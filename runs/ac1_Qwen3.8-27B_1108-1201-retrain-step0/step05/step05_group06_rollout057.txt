"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses adaptive LSE-FISTA with:
- Data-driven beta scheduling based on convolution spread
- Frank-Wolfe escape steps
- Backtracking line search
- Asymmetric perturbation for symmetry breaking
- Multi-n exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_smooth_max(c, beta):
    """Compute LSE approximation of max and its gradient weights."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    lse_val = cm + np.log(tot) / beta
    w = e / tot
    return lse_val, w


def lse_gradient(y, w, c):
    """Gradient of LSE w.r.t. y, given weights w."""
    n = len(y)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_fista_lse(a_init, max_time, rng, beta_init=10.0, beta_max=50000.0,
                   use_fw_every=200, noise_interval=3000):
    """
    LSE-FISTA with adaptive beta and periodic Frank-Wolfe escape steps.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    best_lse = np.inf
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_lse, w = lse_smooth_max(c, beta)
        
        # Track best actual objective
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Adaptive beta: based on spread of top convolution values
        # If top values are spread out, use smaller beta; if concentrated, use larger
        top_k = min(64, len(c))
        top_vals = np.sort(c)[::-1][:top_k]
        spread = top_vals[0] - top_vals[min(top_k, len(c)//4)]
        if spread > 1e-10:
            beta_adaptive = 1.0 / (spread + 1e-10)
        else:
            beta_adaptive = beta_max
        # Blend adaptive with gradual increase
        beta = min(beta_max, max(beta_init, 0.7 * beta + 0.3 * beta_adaptive))
        if stuck < 100:
            beta = beta_init
        elif stuck < 1000:
            beta = min(beta, beta_init * (1.0 + 0.01 * it))
        else:
            beta = min(beta_max, beta * 1.02)
        
        # Recompute LSE with updated beta
        cur_lse, w = lse_smooth_max(c, beta)
        
        # Frank-Wolfe escape step (periodic)
        if use_fw_every > 0 and it % use_fw_every == 0 and stuck > 50:
            # FW step: move towards vertex with minimum gradient component
            grad_fw = lse_gradient(y, w, c)
            j_star = int(np.argmin(grad_fw))
            # Line search: try t in [0.01, 0.05, 0.1, 0.2, 0.5]
            best_t = 0.0
            best_lse_t = cur_lse
            for t_cand in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]:
                a_trial = (1 - t_cand) * y + t_cand * S * np.eye(n)[j_star]
                c_trial = fft_convolve_self(a_trial)
                lse_trial, _ = lse_smooth_max(c_trial, beta)
                if lse_trial < best_lse_t:
                    best_lse_t = lse_trial
                    best_t = t_cand
            if best_t > 0:
                y = (1 - best_t) * y + best_t * S * np.eye(n)[j_star]
                a = y.copy()
                t_mom = 1.0
            continue
        
        # Standard FISTA step
        grad = lse_gradient(y, w, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = 0.8 / (beta * np.max(a) + gnorm + 1.0)
        step = max(step, 1e-10)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Noise when stuck
        if stuck > noise_interval and it % (noise_interval // 10) == 0:
            noise_scale = max(1e-8, 0.01 * S / np.sqrt(n) * np.exp(-stuck / (2 * noise_interval)))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
        
        # Periodic momentum restart
        if it % 5000 == 0:
            t_mom = 1.0
    
    return best_a, best_obj


def run_fw_method(a_init, max_time, rng):
    """
    Pure Frank-Wolfe with backtracking line search.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = 50.0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 100:
            beta = min(1000.0, beta * 1.05)
        
        cur_lse, w = lse_smooth_max(c, beta)
        grad = lse_gradient(a, w, c)
        
        # FW direction: towards vertex with min gradient
        j_star = int(np.argmin(grad))
        
        # Backtracking line search
        t = 2.0 / (it + 2.0)  # Standard FW step size
        best_val = cur_lse
        # Try the standard step and a few alternatives
        for t_try in [t, min(1.0, 2*t), 0.5*t, 0.1, 0.2, 0.3, 0.5]:
            t_try = min(max(t_try, 0.001), 1.0)
            a_trial = (1 - t_try) * a + t_try * S * np.eye(n)[j_star]
            c_trial = fft_convolve_self(a_trial)
            lse_trial, _ = lse_smooth_max(c_trial, beta)
            if lse_trial < best_val:
                best_val = lse_trial
                t = t_try
        
        a = (1 - t) * a + t * S * np.eye(n)[j_star]
        
        # Periodic noise
        if stuck > 2000 and it % 100 == 0:
            idx = rng.integers(0, n, size=min(100, n))
            a[idx] *= (1 + rng.normal(0, 0.1, len(idx)))

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-strategy optimizer combining:
- Adaptive LSE-FISTA with data-driven beta scheduling
- Frank-Wolfe escape steps for local minimum avoidance
- Exact subgradient polishing with adaptive active-set weighting
- Asymmetric perturbation to break mirror symmetry
- Multi-scale (n, 2n) exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights over convolution values."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(len(c))
    return e / tot


def lse_gradient_from_weights(y, w):
    """Gradient of LSE w.r.t. y given precomputed weights w."""
    n = len(y)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgrad_from_lags(y, lags, weights=None):
    """Exact subgradient from specified lags with optional weights."""
    n = len(y)
    if weights is None:
        weights = np.ones(len(lags)) / len(lags)
    w_full = np.zeros(2 * n - 1)
    w_full[lags] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def make_ramp_init(n, rng):
    """Monotonically increasing initialization."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = 0.4 + 0.6 * t
    a += 0.05 * rng.normal(0, 1, n)
    return np.maximum(a, 0.01)


def make_bump_init(n, rng):
    """Bump in the center, low at edges."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    center = n // 2
    dist = np.abs(np.arange(n) - center) / center
    a = 1.0 - 0.7 * dist
    a += 0.03 * rng.normal(0, 1, n)
    return np.maximum(a, 0.01)


def make_two_scale_init(n, rng):
    """Two-scale: coarse structure with fine wiggles."""
    base = 0.5 + 0.3 * np.sin(2 * np.pi * np.arange(n) / n)
    fine = 0.2 * np.sin(2 * np.pi * 7 * np.arange(n) / n)
    a = base + fine
    a += 0.02 * rng.normal(0, 1, n)
    return np.maximum(a, 0.01)


def run_fista_adaptive(a_init, max_time, rng, beta_init=10.0, beta_max=100000.0,
                       fw_interval=150, noise_interval=4000):
    """
    LSE-FISTA with adaptive beta and periodic FW escape steps.
    Key innovation: beta adapts to the spread of top convolution values.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta: respond to the "sharpness" of the max
        top_k = min(128, len(c))
        sorted_c = np.sort(c)[::-1]
        top_vals = sorted_c[:top_k]
        # Spread: difference between max and the (top_k/4)-th value
        spread_idx = max(1, top_k // 4)
        spread = top_vals[0] - top_vals[spread_idx]

        if spread > 1e-12:
            beta_from_spread = 1.0 / spread
        else:
            beta_from_spread = beta_max

        # Blend: gradually increase beta, but also respond to spread
        if stuck < 200:
            target_beta = beta_init * (1.0 + 0.005 * it)
        elif stuck < 2000:
            target_beta = min(beta_max, beta * 1.01)
        else:
            target_beta = min(beta_max, beta * 1.03)

        # Combine both signals
        target_beta = max(target_beta, 0.5 * beta_from_spread)
        beta = 0.8 * beta + 0.2 * min(target_beta, beta_max)
        beta = max(beta, beta_init)
        beta = min(beta, beta_max)

        # Periodic Frank-Wolfe escape
        if fw_interval > 0 and it % fw_interval == 0 and stuck > 30:
            w_lse = lse_weights(c, beta)
            grad = lse_gradient_from_weights(y, w_lse)
            j_star = int(np.argmin(grad))
            # Try several step sizes (FW direction)
            e_j = np.zeros(n)
            e_j[j_star] = 1.0
            best_t = 0.0
            best_cmax = cur_max
            for t_cand in [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03]:
                a_trial = (1.0 - t_cand) * y + t_cand * S * e_j
                # Ensure non-negative
                if a_trial.min() < -1e-10:
                    continue
                c_trial = fft_convolve_self(np.maximum(a_trial, 0))
                if c_trial.max() < best_cmax:
                    best_cmax = c_trial.max()
                    best_t = t_cand
            if best_t > 0:
                y = (1.0 - best_t) * y + best_t * S * e_j
                y = np.maximum(y, 0.0)
                a = y.copy()
                t_mom = 1.0
            continue

        # Standard FISTA step with LSE gradient
        w_lse = lse_weights(c, beta)
        grad = lse_gradient_from_weights(y, w_lse)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: balance between smoothness and progress
        step = 0.9 / (beta * np.max(a) * 2.0 + gnorm + 1e-6)
        step = max(step, 1e-10)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise escape when stuck
        if stuck > noise_interval and it % max(1, noise_interval // 20) == 0:
            scale = max(1e-8, 0.008 * S / np.sqrt(n) * np.exp(-stuck / (3.0 * noise_interval)))
            if scale > 1e-10:
                y = project_simplex(y + rng.normal(0, scale, n), S)
                a = y.copy()
            t_mom = 1.0

        # Periodic momentum reset
        if it % 6000 == 0:
            t_mom = 1.0

    # Final check
    c_final = fft_convolve_self(best_a)
    final_obj = 2.0 * n * c_final.max() / (S * S)
    if final_obj < best_obj:
        best_obj = final_obj
    return best_a, best_obj


def run_exact_polish(a_init, max_time, rng):
    """

    Uses softmax weighting over active lags for smooth gradient.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive active set: include lags within tolerance of max
        if stuck < 100:
            tol_frac = 0.0
            sharpness = 1000.0
        elif stuck < 1000:
            tol_frac = 0.002
            sharpness = 500.0
        elif stuck < 5000:
            tol_frac = 0.01
            sharpness = 200.0
        else:
            tol_frac = min(0.1, 0.01 + 0.02 * (stuck - 5000) / 10000.0)
            sharpness = 100.0

        threshold = cur_max * (1.0 - tol_frac)
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Softmax weighting over active lags
        active_vals = c[active]
        shifted = sharpness * (active_vals - active_vals.max())
        e = np.exp(shifted)
        w_active = e / e.sum()

        grad = exact_subgrad_from_lags(y, active, w_active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size with decrease
        step = 0.8 / gnorm
        step = min(step, S / (gnorm * 2.0))
        step = max(step, 1e-12)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Escape mechanism
        if stuck > 6000 and it % 500 == 0:
            scale = max(1e-9, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 30000.0))

                y = project_simplex(y + rng.normal(0, scale, n), S)
                a = y.copy()
            t_mom = 1.0

            t_mom = 1.0

    final_obj = 2.0 * n * c_final.max() / (S * S)
    if final_obj < best_obj:
        best_obj = final_obj
    return best_a, best_obj

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample using the DFT (spectral method for bandlimited upsample)."""
    n = len(a)
    n_new = n * factor
    # Zero-pad in frequency domain
    size = 1 << (n - 1).bit_length()
    A = fft(a, size)
    # Zero-pad the spectrum
    A_padded = np.zeros(n_new, dtype=np.complex128)
    half = size // 2
    A_padded[:half] = A[:half]
    if size % 2 == 0 and half < n_new:
        A_padded[half] = A[half]
    A_padded[n_new - (size - half - 1):] = A[half + 1:]
    return np.real(ifft(A_padded))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-strategy search for non-negative sequence minimizing
    2*n*max_conv/sum^2.
    Phases:
      1. Adaptive LSE-FISTA (35%)
      2. Frank-Wolfe polish (15%)
      3. Asymmetric perturbation + re-optimize (20%)
      4. Exact subgradient polish (15%)
      5. Multi-scale (2n) exploration (15%)
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    n0 = len(a_start) if a_start is not None else 8192
    S0 = np.sqrt(2.0 * n0)

    print("Base n = %d" % n0)

    # Evaluate starting point
    if a_start is not None:
        obj_start = compute_obj(a_start)
        print("Starting obj: %.6f" % obj_start)
        if obj_start < global_best_obj:
            global_best_obj = obj_start
            global_best_a = a_start.copy()

    if global_best_a is None:
        global_best_a = normalize_seq(make_ramp_init(n0, rng), S0)
        global_best_obj = compute_obj(global_best_a)
        print("Fallback init obj: %.6f" % global_best_obj)

    # === Phase 1: Main LSE-FISTA ===
    remaining = deadline - time.monotonic()
    p1 = remaining * 0.35
    if p1 > 15:
        print("\n=== Phase 1: LSE-FISTA (budget %.0fs) ===" % p1)
        t0 = time.monotonic()
        a1, obj1 = run_fista_adaptive(global_best_a, p1, rng,
                                       beta_init=10.0, beta_max=80000.0,
                                       fw_interval=120, noise_interval=3000)
        print("  obj=%.7f (t=%.1fs)" % (obj1, time.monotonic() - t0))
        if obj1 < global_best_obj:
            global_best_obj = obj1
            global_best_a = a1.copy()

    # === Phase 2: FW Polish ===
    remaining = deadline - time.monotonic()
    p2 = remaining * 0.15
    if p2 > 10:
        print("\n=== Phase 2: FW polish (budget %.0fs) ===" % p2)
        t0 = time.monotonic()
        a2, obj2 = run_fw_polish(global_best_a, p2, rng)
        print("  obj=%.7f (t=%.1fs)" % (obj2, time.monotonic() - t0))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: Asymmetric perturbation + re-optimize ===
    remaining = deadline - time.monotonic()
    p3 = remaining * 0.20
    if p3 > 15:
        print("\n=== Phase 3: Asymmetric + re-opt (budget %.0fs) ===" % p3)
        t0 = time.monotonic()
        n = len(global_best_a)
        S = np.sqrt(2.0 * n)

        # Create asymmetric perturbation: shift the "mass"
        shift = int(n * 0.05)  # 5% shift
        a_shifted = np.zeros(n)
        a_shifted[shift:] = global_best_a[:-shift]
        a_shifted[:shift] = global_best_a[-shift:] * 0.8
        a_shifted = np.maximum(a_shifted, 0)
        a_shifted = normalize_seq(a_shifted, S)

        a3, obj3 = run_fista_adaptive(a_shifted, p3 * 0.6, rng,
                                       beta_init=20.0, beta_max=50000.0,
                                       fw_interval=100, noise_interval=2000)
        print("  Asymmetric: obj=%.7f (t=%.1fs)" % (obj3, time.monotonic() - t0))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

        rem3 = p3 - (time.monotonic() - t0)
        if rem3 > 8:
            a3b, obj3b = run_exact_polish(global_best_a, rem3, rng)
            print("  Polish: obj=%.7f (t=%.1fs)" % (obj3b, time.monotonic() - t0))
            if obj3b < global_best_obj:
                global_best_obj = obj3b
                global_best_a = a3b.copy()

    # === Phase 4: Exact subgradient polish ===
    remaining = deadline - time.monotonic()
    p4 = remaining * 0.15
    if p4 > 10:
        print("\n=== Phase 4: Exact subgrad polish (budget %.0fs) ===" % p4)
        t0 = time.monotonic()
        a4, obj4 = run_exact_polish(global_best_a, p4, rng)
        print("  obj=%.7f (t=%.1fs)" % (obj4, time.monotonic() - t0))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Multi-scale (2n) exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 150 and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        p5 = remaining * 0.80
        print("\n=== Phase 5: Upsample to n=%d (budget %.0fs) ===" % (n2, p5))
        t0 = time.monotonic()
        S2 = np.sqrt(2.0 * n2)
        a_up = upsample(global_best_a, factor=2)
        a_up = np.maximum(a_up, 0)
        a_up = normalize_seq(a_up, S2)
        obj_up_init = compute_obj(a_up)
        print("  Upsampled init: obj=%.7f" % obj_up_init)

        a5, obj5 = run_fista_adaptive(a_up, p5 * 0.5, rng,
                                       beta_init=5.0, beta_max=30000.0,
                                       fw_interval=100, noise_interval=2000)
        print("  Upsampled FISTA: obj=%.7f (t=%.1fs)" % (obj5, time.monotonic() - t0))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        rem5 = p5 - (time.monotonic() - t0)
        if rem5 > 10:
            a5b, obj5b = run_exact_polish(global_best_a, rem5, rng)
            print("  Upsampled polish: obj=%.7f (t=%.1fs)" % (obj5b, time.monotonic() - t0))
            if obj5b < global_best_obj:
                global_best_obj = obj5b
                global_best_a = a5b.copy()

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p6 = remaining * 0.85
        print("\n=== Phase 6: Final polish (budget %.0fs) ===" % p6)
        t0 = time.monotonic()

        # Alternate between exact polish and FW
        half6 = p6 * 0.5
        a6, obj6 = run_exact_polish(global_best_a, half6, rng)
        print("  Final exact: obj=%.7f (t=%.1fs)" % (obj6, time.monotonic() - t0))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

        rem6 = p6 - (time.monotonic() - t0)
        if rem6 > 5:
            a6b, obj6b = run_fw_polish(global_best_a, rem6, rng)
            print("  Final FW: obj=%.7f (t=%.1fs)" % (obj6b, time.monotonic() - t0))
            if obj6b < global_best_obj:
                global_best_obj = obj6b
                global_best_a = a6b.copy()

    # Final verification through evaluate_sequence
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL RESULT ===")
            print("Objective: %.7f (n=%d)" % (ver, len(final_list)))
            return final_list
        else:
            print("WARNING: evaluate_sequence rejected sequence")

    # Fallback
    n_fb = 8192
    fb = np.full(n_fb, np.sqrt(2.0 * n_fb) / n_fb)
    final_list = [float(x) for x in fb]
    ver = evaluate_sequence(final_list)
    print("Fallback: obj=%.7f" % ver)
    return final_list
