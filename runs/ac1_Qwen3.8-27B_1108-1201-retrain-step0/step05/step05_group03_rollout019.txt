6. Deadline-aware time management

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_value_and_weights(c, beta):
    """Compute log-sum-exp value and softmax weights."""
    cm = c.max()
    scaled = beta * (c - cm)
    # Prevent overflow
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c - cm).max())
        beta = scale
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    val = cm + np.log(max(tot, 1e-300)) / beta
    w = e / max(tot, 1e-300)
    return val, w


def lse_gradient_full(a, c, beta):
    """Gradient of full LSE smooth max of self-convolution with respect to a."""
    n = len(a)
    _, w = lse_value_and_weights(c, beta)
    # w has length 2n-1
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def lse_gradient_topk(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_equalize(a, S, alpha):
    """Equalize the power spectrum towards flatness."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a.copy()
    # Move power towards the mean (flatten spectrum)
    new_power = power + alpha * (mean_p - power)
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / power)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def run_fista_lse(a_init, S, max_time, rng,
                  beta_init=10.0, beta_rate=0.004, beta_max=10000.0,
                  use_topk=False, k_frac=0.1):
    """FISTA with full or top-k LSE smoothing, backtracking, momentum."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0
    step = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c_y = fft_convolve_self(y)
        cur_max = c_y.max()
        if cur_max < best - 1e-14:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        if use_topk:
            k = max(4, int(k_frac * (2 * n - 1)))
            grad = lse_gradient_topk(y, c_y, k, beta)
        else:
            grad = lse_gradient_full(y, c_y, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        x_new = project_simplex(y - step * grad, S)
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            # Sufficient decrease condition
            if c_new.max() <= cur_max - 1e-12 * step * gnorm + 1e-14:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new

        # Restart when stuck
        if stuck > 800:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0
            step = 1.0 / (2.0 * beta * 0.1)

        # Spectral equalization periodically
        if it % 1500 == 0 and rng.random() < 0.3:
            y = spectral_equalize(y, S, 0.15)
            t_val = 1.0
            stuck = 0

        # Adaptive step size
        if it % 100 == 0:
            step = min(step * 1.05, 1.0 / (beta * 0.01))

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_mirror_descent(theta_init, max_time, rng,
                       beta_init=5.0, beta_rate=0.005, beta_max=5000.0,
                       lr_init=0.1):
    """
    Mirror descent (exponentiated gradient) in log-space.
    a = exp(theta) / sum(exp(theta)), so a > 0 automatically.
    theta is unconstrained.
    """
    n = len(theta_init)
    theta = theta_init.copy().astype(np.float64)
    start = time.monotonic()
    t = 0
    t_val = 1.0
    y_theta = theta.copy()
    best = np.inf
    best_a = None
    step = lr_init

    # Compute initial a
    def theta_to_a(th):
        th_shifted = th - th.max()
        ea = np.exp(th_shifted)
        return ea / ea.sum()

    a = theta_to_a(theta)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        t += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * t))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-14:
            best = cur_max
            best_a = a.copy()

        # Gradient of LSE w.r.t. a
        grad_a = lse_gradient_full(a, c, beta)

        # Gradient w.r.t. theta: grad_theta = a * grad_a (chain rule through softmax)
        grad_theta = a * grad_a

        # Mirror descent step with momentum
        step_current = step / (1.0 + 0.01 * t)
        y_new = y_theta - step_current * grad_theta

        # FISTA-style update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        theta_new = y_new + ((t_val - 1.0) / t_new) * (y_new - y_theta)

        a_new = theta_to_a(theta_new)
        c_new = fft_convolve_self(a_new)

        # Backtracking if needed
        trials = 0
        while c_new.max() > cur_max + 1e-12 and trials < 8:
            step_current *= 0.5
            y_new = y_theta - step_current * grad_theta
            theta_new = y_new + ((t_val - 1.0) / t_new) * (y_new - y_theta)
            a_new = theta_to_a(theta_new)
            c_new = fft_convolve_self(a_new)
            trials += 1

        a = a_new
        y_theta = theta_new
        theta = a_new_theta = theta_new
        t_val = t_new

        # Periodic perturbation
        if t % 2000 == 0 and rng.random() < 0.15:
            a = theta_to_a(y_theta + rng.normal(0, 0.01, n))
            y_theta = np.log(a + 1e-300)
            t_val = 1.0

    return best_a, best


def local_2exchange(a, S, max_time, rng, n_trials_per_iter=20):
    """
    Local search via 2-element exchanges.
    O(n) per trial using convolution difference formula.
    """
    n = len(a)
    a = normalize(a.copy(), S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    start = time.monotonic()
    total_trials = 0

    while time.monotonic() - start < max_time:
        improved = False
        for _ in range(n_trials_per_iter):
            # Pick random pair
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            # Try a few delta values
            for delta_frac in [0.1, 0.05, -0.1, -0.05, 0.2, -0.2]:
                delta = delta_frac * min(a[i], a[j]) if delta_frac > 0 else \
                        delta_frac * min(a[i], a[j])
                if delta >= 0 and a[i] + delta > 1000:
                    continue
                if delta < 0 and a[j] + delta < 0:
                    continue
                if a[i] + delta < 0:
                    continue

                # Compute change in convolution: dc[k] = 2*delta*(a[k-i] - a[k-j])
                # a[k-i] is valid for k in [i, i+n-1]
                # a[k-j] is valid for k in [j, j+n-1]
                dc = np.zeros(2 * n - 1)
                lo_i = i
                hi_i = min(i + n, 2 * n - 1)
                dc[lo_i:hi_i] += 2.0 * delta * a[:hi_i - lo_i]

                lo_j = j
                hi_j = min(j + n, 2 * n - 1)
                dc[lo_j:hi_j] -= 2.0 * delta * a[:hi_j - lo_j]

                c_new = c + dc
                new_max = c_new.max()

                if new_max < best - 1e-15:
                    # Accept
                    a[i] += delta
                    a[j] -= delta
                    a = np.maximum(a, 0.0)
                    a = normalize(a, S)
                    c = fft_convolve_self(a)  # Recompute to avoid drift
                    best = c.max()
                    best_a = a.copy()
                    improved = True
                    total_trials += 1
                    break
            if improved:
                break
        total_trials += 1

        # If no improvement in this round, try larger perturbations
        if not improved:
            if rng.random() < 0.1:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j:
                    # Swap-style move
                    tmp = a[i] * 0.5
                    a[i] *= 0.5
                    a[j] += tmp
                    a = normalize(a, S)
                    c_new = fft_convolve_self(a)
                    if c_new.max() < best:
                        best = c_new.max()
                        best_a = a.copy()
                        c = c_new
                    else:
                        # Revert
                        a[j] -= tmp
                        a[i] /= 0.5 if a[i] > 0 else 1.0
                        a = normalize(a, S)
                        c = fft_convolve_self(a)

    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length using linear interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src, endpoint=False)
    t_new = np.linspace(0, 1, n_target, endpoint=False)
    return np.interp(t_new, t_old, a)


def geometric_init(n, S, alpha=0.3):
    """Create a geometric decay initialization."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i / n)
    return normalize(a, S)


def triangular_init(n, S):
    """Create a triangular shape initialization."""
    i = np.arange(n, dtype=np.float64)
    mid = n / 2.0
    a = 1.0 - np.abs(i - mid) / mid
    a = np.maximum(a, 0.01)
    return normalize(a, S)


def bimodal_init(n, S, rng):
    """Create a bimodal initialization."""
    i = np.arange(n, dtype=np.float64)
    sigma = n / 8.0
    a = np.exp(-0.5 * ((i - n * 0.3) / sigma) ** 2) + np.exp(-0.5 * ((i - n * 0.7) / sigma) ** 2)
    return normalize(a, S)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: FISTA-LSE -> Spectral Equalization -> Mirror Descent -> Local Search.
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs, deadline in %.0fs" % (budget_s, budget_s - 30))

    # Try to get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.4f" % (len(given), given.sum()))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_global(a, n):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * n * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  *** New global best: %.6f (n=%d) ***" % (obj, n))
        return obj

    # === Phase 1: Main FISTA on given sequence or n=8192 ===
    n_main = 8192
    S_main = np.sqrt(2.0 * n_main)

    if given is not None and len(given) == n_main:
        a_start = normalize(given.copy(), S_main)
        print("Starting from given sequence, n=%d" % n_main)
    elif given is not None:
        a_start = normalize(resample_sequence(given, n_main), S_main)
        print("Resampled given sequence to n=%d" % n_main)
    else:
        a_start = np.ones(n_main) * S_main / n_main
        print("No given sequence, starting uniform, n=%d" % n_main)

    remaining = deadline - time.monotonic()
    if remaining > 60:
        t1 = remaining * 0.35
        print("\n=== Phase 1a: FISTA Full LSE ( %.0fs ) ===" % t1)
        a_opt, val_opt = run_fista_lse(a_start, S_main, t1, rng,
                                        beta_init=8.0, beta_rate=0.003,
                                        beta_max=5000.0, use_topk=False)
        obj = update_global(a_opt, n_main)
        print("  Phase 1a done, obj=%.6f" % (2.0 * n_main * val_opt / (a_opt.sum()**2)))
        a_start = a_opt.copy()

    remaining = deadline - time.monotonic()
    if remaining > 60:
        t1b = remaining * 0.25
        print("\n=== Phase 1b: FISTA Top-k LSE refined ( %.0fs ) ===" % t1b)
        a_opt, val_opt = run_fista_lse(a_start, S_main, t1b, rng,
                                        beta_init=200.0, beta_rate=0.001,
                                        beta_max=20000.0, use_topk=True, k_frac=0.05)
        obj = update_global(a_opt, n_main)
        print("  Phase 1b done, obj=%.6f" % (2.0 * n_main * val_opt / (a_opt.sum()**2)))
        a_start = a_opt.copy()

    # === Phase 2: Spectral shaping + FISTA with higher beta ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        t2 = remaining * 0.20
        print("\n=== Phase 2: High-beta FISTA + spectral shaping ( %.0fs ) ===" % t2)

        # Alternating: FISTA few iterations, then spectral equalize
        a_cur = normalize(global_a.copy(), S_main)
        c_cur = fft_convolve_self(a_cur)
        sub_start = time.monotonic()
        sub_deadline = sub_start + t2
        it = 0
        while time.monotonic() < sub_deadline:
            it += 1
            # FISTA step with high beta
            beta = min(50000.0, 500.0 * (1.0 + it * 0.02))
            grad = lse_gradient_full(a_cur, c_cur, beta)
            step_sz = 0.8 / (beta * 0.1 + 1e-10)
            a_new = project_simplex(a_cur - step_sz * grad, S_main)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < c_cur.max() - 1e-15:
                a_cur = a_new
                c_cur = c_new
            else:
                # Try smaller step
                a_new = project_simplex(a_cur - step_sz * 0.3 * grad, S_main)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < c_cur.max():
                    a_cur = a_new
                    c_cur = c_new

            # Periodic spectral equalization
            if it % 200 == 0:
                a_cur = spectral_equalize(a_cur, S_main, 0.2)
                c_cur = fft_convolve_self(a_cur)

            update_global(a_cur, n_main)

        print("  Phase 2 done, best=%.6f" % global_best)
        a_start = a_cur.copy()

    # === Phase 3: Mirror Descent ===
    remaining = deadline - time.monotonic()
    if remaining > 40 and global_a is not None:
        t3 = remaining * 0.15
        print("\n=== Phase 3: Mirror Descent ( %.0fs ) ===" % t3)
        theta0 = np.log(global_a + 1e-300)
        a_opt, val_opt = run_mirror_descent(theta0, t3, rng,
                                             beta_init=50.0, beta_rate=0.003,
                                             beta_max=5000.0, lr_init=0.05)
        if a_opt is not None:
            a_opt = normalize(a_opt, S_main)
            obj = update_global(a_opt, n_main)
            print("  Phase 3 done, obj=%.6f" % (2.0 * n_main * val_opt / (a_opt.sum()**2)))

    # === Phase 4: Try different initialization ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        t4 = remaining * 0.15
        print("\n=== Phase 4: Different init + FISTA ( %.0fs ) ===" % t4)
        # Try geometric init
        a_geo = geometric_init(n_main, S_main, alpha=0.5)
        a_opt, val_opt = run_fista_lse(a_geo, S_main, t4 * 0.5, rng,
                                        beta_init=10.0, beta_rate=0.004,
                                        beta_max=5000.0, use_topk=False)
        obj = update_global(a_opt, n_main)
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_bim = bimodal_init(n_main, 0.7, rng)
            a_opt2, val_opt2 = run_fista_lse(a_bim, S_main, remaining * 0.6, rng,
                                              beta_init=10.0, beta_rate=0.004,
                                              beta_max=5000.0, use_topk=False)
            obj2 = update_global(a_opt2, n_main)

    # === Phase 5: Local 2-element exchange ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        t5 = remaining * 0.25
        print("\n=== Phase 5: Local 2-exchange ( %.0fs ) ===" % t5)
        S_loc = global_a.sum()
        a_opt, val_opt = local_2exchange(global_a, S_loc, t5, rng,
                                         n_trials_per_iter=30)
        obj = update_global(a_opt, len(a_opt))
        print("  Phase 5 done, best=%.6f" % global_best)

    # === Phase 6: Second length (n=16384) if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n_sec = 16384
        S_sec = np.sqrt(2.0 * n_sec)
        a_sec = normalize(resample_sequence(global_a, n_sec), S_sec)
        t6 = remaining * 0.30
        print("\n=== Phase 6: n=%d FISTA ( %.0fs ) ===" % (n_sec, t6))
        a_opt, val_opt = run_fista_lse(a_sec, S_sec, t6, rng,
                                        beta_init=5.0, beta_rate=0.003,
                                        beta_max=3000.0, use_topk=False)
        obj = update_global(a_opt, n_sec)
        print("  Phase 6 done, best=%.6f" % global_best)

    # === Final verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            print("\nVerification: obj = %.6f" % ver)
            if not np.isfinite(ver) or ver > 100:
                print("  Warning: evaluate_sequence returned unusual value!")
        except Exception as e:
            print("  Verification error: %s" % str(e))
        print("Final sequence length: %d" % len(global_a))
    else:
        print("No valid sequence found, using fallback.")
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Total time used: %.1fs" % (time.monotonic() - start_total))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
