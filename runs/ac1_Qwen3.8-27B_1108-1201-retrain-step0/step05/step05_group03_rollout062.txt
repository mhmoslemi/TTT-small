```
    if n==0: return v

    valid=tc <= w

    if s<=0: res=np.full(n,S/float(n))
```

```python
"""
Search for non-negative sequences minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2.

Approach:
- Normalize each length n to sum S = sqrt(2n), so the objective equals max(convolve(a,a)).
- Build diverse starts (given sequence, power laws, exponentials, random Fourier,
  spectral square-root of a flat polynomial).
- Use first-order optimization with:
    * Adam-style updates on the simplex projected sequence.
    * Log-sum-exp smoothing of the self-convolution, including top-k focus.
    * Projected line-search polishing with active-set / top-k gradients.
- Try a secondary longer length (up to 16384) if it clearly improves.
"""

import time
import numpy as np
from scipy.fft import fft, ifft


def time_remaining(deadline):
    return max(0.0, deadline - time.monotonic())


def normalize_seq(a, S):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return a
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.full(a.size, float(S) / float(a.size), dtype=np.float64)
    a = a * (float(S) / s)
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 <= 1e-300:
        return np.full(a.size, float(S) / float(a.size), dtype=np.float64)
    a = a * (float(S) / s2)
    return a


def project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    n = v.size
    if n == 0:
        return v
    v = np.nan_to_num(v, nan=0.0, posinf=1e6, neginf=-1e6)
    w = np.sort(v)[::-1]
    cum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cum - float(S)) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0])) + 1
    theta = (cum[rho - 1] - float(S)) / float(rho)
    res = np.maximum(v - theta, 0.0)
    s = float(res.sum())
    if s <= 1e-300:
        res = np.full(n, float(S) / float(n), dtype=np.float64)
    else:
        res = res * (float(S) / s)
    return res


def resample_seq(a, n, S):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 0:
        return normalize_seq(np.ones(n, dtype=np.float64), S)
    if a.size == int(n):
        return normalize_seq(a, S)
    ta = np.linspace(0.0, 1.0, int(a.size))
    tb = np.linspace(0.0, 1.0, int(n))
    b = np.interp(tb, ta, a)
    return normalize_seq(b, S)


def get_given_sequence():
    try:
        g = globals().get("height_sequence_1")
        if g is None:
            return None
        a = np.asarray(g, dtype=np.float64).reshape(-1)
        a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.size == 0 or float(a.sum()) <= 1e-12:
            return None
        return a
    except Exception:
        return None


def conv_data(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A, fsize))[:size]
    c = np.where(c > 0.0, c, 0.0)
    c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
    c = np.maximum(c, 0.0)
    maxq = float(c.max()) if c.size else 0.0
    if not np.isfinite(maxq):
        maxq = 0.0
    return c, maxq, A, fsize


def objective_from_array(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    s = float(a.sum())
    if n == 0 or s < 0.01:
        return float("inf")
    _, maxq, _, _ = conv_data(a)
    val = 2.0 * n * maxq / (s * s)
    if not np.isfinite(val):
        return float("inf")
    return float(val)


def lse_weights_full(c, beta):
    c = np.asarray(c, dtype=np.float64)
    m = float(c.max()) if c.size else 0.0
    scaled = beta * (c - m)
    scaled = np.clip(scaled, -500.0, 0.0)
    e = np.exp(scaled)
    Z = float(e.sum())
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.ones_like(c, dtype=np.float64)
        Z = float(c.size) if c.size else 1.0
    e = e / Z
    if beta > 0.0:
        val = m + float(np.log(max(Z, 1e-300))) / beta
    else:
        val = m
    return e, val, m


def lse_weights_topk(c, beta, k):
    c = np.asarray(c, dtype=np.float64).reshape(-1)
    n = c.size
    k = int(k)
    if k >= n:
        return lse_weights_full(c, beta)
    if k <= 0:
        k = 1
    idx = np.argpartition(c, -k)[-k:]
    qtop = c[idx]
    m = float(qtop.max())
    scaled = beta * (qtop - m)
    scaled = np.clip(scaled, -500.0, 0.0)
    e = np.exp(scaled)
    Z = float(e.sum())
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.ones_like(qtop, dtype=np.float64)
        Z = float(qtop.size)
    e = e / Z
    w = np.zeros(n, dtype=np.float64)
    w[idx] = e
    if beta > 0.0:
        val = m + float(np.log(max(Z, 1e-300))) / beta
    else:
        val = m
    return w, val, m


def uniform_top_weights(c, k):
    c = np.asarray(c, dtype=np.float64).reshape(-1)
    n = c.size
    k = int(k)
    if k >= n:
        return np.ones(n, dtype=np.float64) / float(n)
    if k <= 0:
        k = 1
    idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(n, dtype=np.float64)
    w[idx] = 1.0 / float(k)
    return w


def gradient_from_weights(w, a, A, fsize, n):
    w = np.asarray(w, dtype=np.float64).reshape(-1)
    W = fft(w, fsize)
    r = np.real(ifft(W * np.conj(A), fsize))[:n]
    return 2.0 * r


def spectral_perturb(a, amount, rng):
    n = a.size
    S = float(a.sum())
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    phase = rng.normal(0.0, float(amount), fsize)
    phase[0] = 0.0
    an = np.real(ifft(A * np.exp(1j * phase), fsize))[:n]
    return normalize_seq(an, S)


def perturb_sequence(a, rel, rng):
    n = a.size
    S = float(a.sum())
    if rng.random() < 0.65:
        sigma = max(1e-12, float(rel) * S / np.sqrt(float(n)))
        b = a + rng.normal(0.0, sigma, n)
        return normalize_seq(b, S)
    else:
        amount = 0.02 + 2.0 * float(rel)
        return spectral_perturb(a, amount, rng)


def spectral_root_starts(n, S):
    out = []
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    q = np.zeros(fsize, dtype=np.float64)
    q[:size] = (float(S) * float(S)) / float(size)
    Q = fft(q, fsize)
    Q = Q + 1e-300
    A = np.sqrt(Q)
    shifts = (0, -1, 1)
    for sh in shifts:
        phase = np.exp(-1j * 2.0 * np.pi * float(sh) * np.arange(fsize) / float(fsize))
        As = A * phase
        p1 = np.real(ifft(As, fsize))[:n]
        p1 = np.maximum(p1, 0.0)
        if float(p1.sum()) > 1e-300:
            out.append(normalize_seq(p1, S))
        p2 = np.real(ifft(np.conj(As), fsize))[:n]
        p2 = np.maximum(p2, 0.0)
        if float(p2.sum()) > 1e-300:
            out.append(normalize_seq(p2, S))
    return out


def generate_starts(n, S, given, rng):
    starts = []
    if given is not None:
        starts.append(resample_seq(given, n, S))
    starts.append(np.full(n, float(S) / float(n), dtype=np.float64))

    i = np.arange(n, dtype=np.float64)

    for alpha in (0.1, 0.2, 0.35, 0.5):
        vals = (i + 1.0) ** (-alpha)
        starts.append(normalize_seq(vals, S))

    for alpha in (0.3, 0.5):
        for gamma in (0.5, 1.0, 2.0):
            x = i / float(n)
            vals = (1.0 - x) ** (-alpha) * np.exp(-gamma * x)
            starts.append(normalize_seq(vals, S))

    for lam in (1.0, 2.0, 5.0, 10.0):
        vals = np.exp(-lam * i / float(max(1, n - 1)))
        starts.append(normalize_seq(vals, S))

    for gam in (0.5, 1.0, 2.0, 4.0):
        x = i / float(max(1, n - 1))
        vals = 0.05 + x ** gam
        starts.append(normalize_seq(vals, S))

    for _ in range(12):
        vals = np.ones(n, dtype=np.float64)
        for _f in range(5):
            f = int(rng.integers(1, 9))
            amp = float(rng.uniform(0.1, 0.6))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            vals += amp * np.cos(2.0 * np.pi * float(f) * i / float(max(1, n - 1)) + ph)
        starts.append(normalize_seq(vals, S))

    spec = spectral_root_starts(int(n), float(S))
    starts.extend(spec)

    # A few averaged starts to create smoother hybrids.
    if len(starts) >= 2:
        idx = np.argsort([float(np.mean(x)) for x in starts])[-8:]
        for p in range(0, len(idx) - 1):
            a = 0.5 * starts[int(idx[p])] + 0.5 * starts[int(idx[p + 1])]
            starts.append(normalize_seq(a, S))

    return starts


def select_best_starts(starts, S, max_keep=4):
    scored = []
    for j in range(len(starts)):
        a = normalize_seq(starts[j], S)
        val = objective_from_array(a)
        if not np.isfinite(val):
            val = 1e300
        scored.append((val, j, a))
    scored.sort()
    if not scored:
        return [], float("inf")
    best_val = float(scored[0][0])
    selected = []
    for pos in range(min(int(max_keep), len(scored))):
        selected.append(scored[pos][2])
    return selected, best_val


def run_adam_lse(a_init, S, max_time, rng, beta_min=2.0, beta_max=20000.0,
                 lr_init=0.02, topk_frac=1.0, name="adam"):
    a = normalize_seq(a_init, S)
    n = a.size
    S = float(S)
    if max_time <= 0.0:
        return a, objective_from_array(a)

    start = time.monotonic()
    c, maxq, A, fsize = conv_data(a)
    best_F = float(2.0 * n * maxq / (S * S)) if S > 0 else float("inf")
    best_a = a.copy()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta = float(beta_min)

    expected_iters = max(2000.0, float(max_time) * 30.0)
    if beta_max > beta_min:
        growth = (float(beta_max) / float(beta_min)) ** (1.0 / expected_iters)
        growth = min(1.01, max(1.00001, growth))
    else:
        growth = 1.0

    last_improve = start
    last_print = start
    stuck_threshold = 40.0 if max_time > 120.0 else 10.0

    while True:
        elapsed = time.monotonic() - start
        if elapsed >= max_time:
            break

        if maxq < best_F - 1e-12:
            best_F = float(2.0 * n * maxq / (S * S))
            best_a = a.copy()
            last_improve = time.monotonic()

        progress = min(1.0, elapsed / max_time)

        if elapsed - last_improve > stuck_threshold and elapsed < max_time - 20.0:
            rel = 0.02 * (1.0 - progress) + 0.006
            a = perturb_sequence(best_a, rel, rng)
            c, maxq, A, fsize = conv_data(a)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            beta = max(float(beta_min), beta * 0.4)
            last_improve = time.monotonic()

        if float(topk_frac) >= 1.0:
            w, val, curmax = lse_weights_full(c, beta)
        else:
            k = max(16, int(float(topk_frac) * (2 * n - 1)))
            w, val, curmax = lse_weights_topk(c, beta, k)

        g = gradient_from_weights(w, a, A, fsize, n)

        t += 1
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g * g)
        mh = m / (1.0 - 0.9 ** float(t))
        vh = v / (1.0 - 0.999 ** float(t))
        step_vec = mh / (np.sqrt(vh) + 1e-8)
        lr = float(lr_init) * max(0.05, 1.0 - 0.7 * progress)

        a_new = project_simplex(a - lr * step_vec, S)
        if not np.all(np.isfinite(a_new)):
            a_new = a.copy()
        a = a_new

        beta = min(float(beta_max), beta * growth)

        if time.monotonic() - last_print > 60.0:
            print(name, "elapsed=%.0fs best=%.6f beta=%.0f maxq=%.6f"
                  % (elapsed, best_F, beta, maxq))
            last_print = time.monotonic()

    c, maxq, A, fsize = conv_data(a)
    F = float(2.0 * n * maxq / (S * S))
    if F < best_F:
        best_F = F
        best_a = a.copy()
    return best_a, best_F


def run_pg_lse(a_init, S, max_time, rng, beta_start=500.0, beta_max=20000.0,
               step_init=0.5, step_max=2.0, uniform_top=False,
               kfra_start=0.08, kfra_end=0.01, name="pg"):
    a = normalize_seq(a_init, S)
    n = a.size
    S = float(S)
    if max_time <= 0.0:
        return a, objective_from_array(a)

    start = time.monotonic()
    c, maxq, A, fsize = conv_data(a)
    best_F = float(2.0 * n * maxq / (S * S)) if S > 0 else float("inf")
    best_a = a.copy()

    step = float(step_init)
    beta = float(beta_start)
    last_improve = start
    last_print = start
    stuck_threshold = 25.0 if max_time > 120.0 else 5.0
    kmin = 16

    while True:
        elapsed = time.monotonic() - start
        if elapsed >= max_time:
            break

        if maxq < best_F - 1e-12:
            best_F = float(2.0 * n * maxq / (S * S))
            best_a = a.copy()
            last_improve = time.monotonic()

        progress = min(1.0, elapsed / max_time)

        if elapsed - last_improve > stuck_threshold and elapsed < max_time - 10.0:
            rel = 0.006 * (1.0 - progress) + 0.002
            a = perturb_sequence(best_a, rel, rng)
            c, maxq, A, fsize = conv_data(a)
            step = float(step_init)
            if not uniform_top:
                beta = max(float(beta_start), beta * 0.4)
            last_improve = time.monotonic()

        frac = float(kfra_start) + (float(kfra_end) - float(kfra_start)) * progress
        if frac < 0.0:
            frac = 0.0
        k = int(max(kmin, min(2 * n - 1, frac * float(2 * n - 1))))
        if k <= 0:
            k = kmin

        if uniform_top:
            w = uniform_top_weights(c, k)
            val = maxq
        else:
            w, val, curmax = lse_weights_topk(c, beta, k)

        g = gradient_from_weights(w, a, A, fsize, n)

        accepted = False
        old_step = step
        trial = 0
        for trial in range(4):
            a_try = project_simplex(a - step * g, S)
            if not np.all(np.isfinite(a_try)):
                break
            c_try, maxq_try, A_try, fsize_try = conv_data(a_try)
            if uniform_top:
                accept = maxq_try < maxq - 1e-12
            else:
                _, val_try, _ = lse_weights_topk(c_try, beta, k)
                accept = maxq_try < maxq - 1e-12 or val_try < val - 1e-8

            if accept:
                a = a_try
                c = c_try
                maxq = maxq_try
                A = A_try
                fsize = fsize_try
                accepted = True
                break

            step = max(0.01, step * 0.5)

        if accepted:
            if trial == 0:
                step = min(float(step_max), step * 1.05)
            if not uniform_top:
                beta = min(float(beta_max), beta * 1.01)
        else:
            step = max(0.01, old_step * 0.25)
            a = perturb_sequence(a, 0.003, rng)
            c, maxq, A, fsize = conv_data(a)
            if not uniform_top:
                beta = max(float(beta_start), beta * 0.5)

        if time.monotonic() - last_print > 60.0:
            print(name, "elapsed=%.0fs best=%.6f step=%.3f k=%d maxq=%.6f"
                  % (elapsed, best_F, step, k, maxq))
            last_print = time.monotonic()

    c, maxq, A, fsize = conv_data(a)
    F = float(2.0 * n * maxq / (S * S))
    if F < best_F:
        best_F = F
        best_a = a.copy()
    return best_a, best_F


def make_fallback(n=8192):
    S = np.sqrt(2.0 * float(n))
    return np.full(int(n), float(S) / float(n), dtype=np.float64)


def propose_candidate(budget_s=2400.0, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 60.0:
        budget = 2400.0
    budget = min(budget, 2380.0)
    run_budget = max(60.0, budget - 30.0)
    deadline = t0 + run_budget

    rng = np.random.default_rng(123456)
    given = get_given_sequence()

    n_primary = 8192
    if given is not None:
        if 512 <= given.size <= 16384:
            n_primary = int(given.size)
        elif given.size > 16384:
            n_primary = 8192
        else:
            n_primary = 8192

    n_primary = int(min(max(n_primary, 512), 16384))
    S_primary = float(np.sqrt(2.0 * float(n_primary)))

    print("primary n=%d S=%.3f budget=%.0fs" % (n_primary, S_primary, run_budget))

    starts = generate_starts(n_primary, S_primary, given, rng)
    top_starts, best_F = select_best_starts(starts, S_primary, max_keep=6)
    if not top_starts:
        best_a = make_fallback(n_primary)
        best_F = objective_from_array(best_a)
    else:
        best_a = top_starts[0].copy()
    print("initial best F=%.6f len=%d" % (best_F, best_a.size))

    # Multi-start Adam exploration on primary length.
    if n_primary <= 10000:
        num_runs = 4
    else:
        num_runs = 2
    num_runs = min(num_runs, len(top_starts))
    settings = [
        (0.030, 2.0, 1.0),
        (0.020, 5.0, 1.0),
        (0.015, 1.0, 0.5),
        (0.035, 10.0, 0.2),
    ]
    per_run_budget = (0.45 * run_budget) / float(max(1, num_runs))

    for r in range(num_runs):
        rem = time_remaining(deadline)
        if rem < 30.0:
            break
        t_run = min(rem, max(20.0, per_run_budget))
        lr_r = settings[r % len(settings)][0]
        beta_r = settings[r % len(settings)][1]
        tk_r = settings[r % len(settings)][2]
        a_opt, F_opt = run_adam_lse(
            top_starts[r], S_primary, t_run, rng,
            beta_min=beta_r, beta_max=20000.0,
            lr_init=lr_r, topk_frac=tk_r,
            name="adam_run%d" % r
        )
        if F_opt < best_F:
            best_F = F_opt
            best_a = a_opt.copy()
            print("primary adam run %d improved to %.6f" % (r, best_F))

    # Primary projected polishing.
    rem = time_remaining(deadline)
    if rem > 30.0:
        t_pg = min(rem, max(30.0, 0.10 * run_budget))
        a_opt, F_opt = run_pg_lse(
            best_a, S_primary, t_pg, rng,
            beta_start=500.0, beta_max=20000.0,
            step_init=0.5, step_max=2.0,
            uniform_top=False, kfra_start=0.08, kfra_end=0.01,
            name="pg_primary"
        )
        if F_opt < best_F:
            best_F = F_opt
            best_a = a_opt.copy()
            print("primary pg improved to %.6f" % best_F)

    primary_best_a = best_a.copy()
    primary_best_F = best_F

    # Optional secondary longer length.
    if n_primary < 8192:
        n_secondary = min(16384, max(8192, 2 * n_primary))
    elif n_primary <= 8192:
        n_secondary = 16384
    else:
        n_secondary = n_primary

    if n_secondary != n_primary and n_secondary <= 16384:
        rem = time_remaining(deadline)
        if rem > max(90.0, 0.15 * run_budget):
            S_secondary = float(np.sqrt(2.0 * float(n_secondary)))
            a_sec = resample_seq(primary_best_a, n_secondary, S_secondary)
            t_sec = min(time_remaining(deadline), max(30.0, 0.18 * run_budget))
            a_sec_opt, F_sec = run_adam_lse(
                a_sec, S_secondary, t_sec, rng,
                beta_min=3.0, beta_max=20000.0,
                lr_init=0.02, topk_frac=1.0,
                name="adam_secondary"
            )
            print("secondary n=%d F=%.6f" % (n_secondary, F_sec))
            if F_sec < primary_best_F - 1e-5:
                best_A = a_sec_opt.copy()
                best_F = F_sec
                rem2 = time_remaining(deadline)
                if rem2 > 20.0:
                    t_sec_pg = min(rem2, max(20.0, 0.03 * run_budget))
                    a_sec_pg, F_sec_pg = run_pg_lse(
                        best_A, S_secondary, t_sec_pg, rng,
                        beta_start=2000.0, beta_max=30000.0,
                        step_init=0.3, step_max=1.2,
                        uniform_top=False, kfra_start=0.05, kfra_end=0.008,
                        name="pg_secondary"
                    )
                    if F_sec_pg < best_F:
                        best_F = F_sec_pg
                        best_A = a_sec_pg.copy()
                best_a = best_A.copy()
                print("accepted secondary n=%d F=%.6f" % (n_secondary, best_F))

    # Final aggressive active-set / uniform-top polishing.
    rem = time_remaining(deadline)
    if rem > 20.0:
        n_best = int(best_a.size)
        S_best = float(np.sqrt(2.0 * float(n_best)))
        t_final = rem
        a_final, F_final = run_pg_lse(
            best_a, S_best, t_final, rng,
            beta_start=3000.0, beta_max=30000.0,
            step_init=0.2, step_max=1.0,
            uniform_top=True, kfra_start=0.05, kfra_end=0.004,
            name="final_active"
        )
        if F_final < best_F:
            best_F = F_final
            best_a = a_final.copy()
            print("final active polish improved to %.6f" % best_F)

    # Sanitize return value.
    if best_a is None or best_a.size < 2:
        best_a = make_fallback(8192)
    if best_a.size > 16384:
        best_a = resample_seq(best_a, 8192, float(np.sqrt(2.0 * 8192.0)))

    best_a = np.asarray(best_a, dtype=np.float64)
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=1000.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(best_a.sum())
    if s < 0.01:
        best_a = make_fallback(8192)

    final_list = [float(x) for x in best_a]

    # Optional local verification only for moderate lengths and if time is available.
    if best_a.size <= 10000 and time_remaining(deadline) > 10.0:
        try:
            check = evaluate_sequence(final_list)
            print("local evaluate_sequence check=%.6f" % check)
        except Exception:
            pass

    print("returning len=%d internal_best=%.6f" % (len(final_list), best_F))
    return final_list
```
