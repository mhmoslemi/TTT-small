"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses spectral shaping (Gerchberg-Saxton), mirror descent, and active-set refinement.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta, fsize):
    """Compute gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -100, 100)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    # Gradient: 2 * sum_k w[k] * (a shifted for lag k)
    # = 2 * IFFT(FFT(w) * conj(FFT(a)))[:n]
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A), fsize))[:n]
    return grad


def spectral_gs(a, S, max_iters, fsize, alpha=0.5, verbose=False):
    """
    Gerchberg-Saxton style spectral shaping.
    Alternates between time-domain projection and frequency-domain flattening.
    """
    n = len(a)
    c_flat = S * S / (2 * n - 1)
    best_max = np.inf
    best_a = a.copy()
    
    for it in range(max_iters):
        A = fft(a, fsize)
        mag = np.abs(A)
        
        # Compute current autocorrelation
        c = np.real(ifft(A * A, fsize))[:2 * n - 1]
        cur_max = c.max()
        
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
        
        if verbose and it % 200 == 0:
            print(f"    GS it={it}, max={cur_max:.6f}, best={best_max:.6f}")
        
        # Compute target magnitude
        # Method: blend between current and flat
        # Flat target: |A|^2 = S^2/n (by Parseval, sum(|A|^2)/fsize = sum(a^2))
        # But we want the AUTOCORRELATION to be flat, not the spectrum directly.
        # 
        # Alternative: use the "water filling" approach
        # If c[k] > c_flat, we want to reduce |A| at frequencies that contribute to lag k
        # This is complex. Instead, use a simple multiplicative update:
        # new_|A|^2 = |A|^2 * (c_flat / max(c, eps))^alpha
        # This reduces the spectrum where the autocorrelation is high.
        
        eps = 1e-10
        ratio = (c_flat / np.maximum(c, eps)) ** alpha
        # But ratio has length 2n-1 and |A| has length fsize
        # We need to be careful here.
        
        # Actually, let's use a different approach:
        # The autocorrelation c = IFFT(|A|^2). We want c to be flat.
        # If we set |A|^2_new = |A|^2 * f where f is chosen to flatten c,
        # then c_new = IFFT(|A|^2 * f) which is the convolution of c with IFFT(f).
        # This is complex.
        
        # Simpler GS approach: just set |A| to a constant (or blend)
        target_mag = S / np.sqrt(n)  # gives mean |A|^2 = S^2/n
        # Scale: we want the resulting time-domain to have sum = S
        # After IFFT, the DC component A[0] = sum(a) = S
        # So |A[0]| should be S.
        # For flat: |A[k]| = S/sqrt(fsize) for all k? No...
        # By Parseval: sum(k=0 to fsize-1) |A[k]|^2 / fsize = sum(a[j]^2)
        # If we want flat: |A[k]| = c for all k, then c^2 * fsize / fsize = sum(a^2)
        # So c = sqrt(sum(a^2))
        # But also A[0] = sum(a) = S (DC component)
        # So we can't have perfectly flat including DC.
        
        # Better: keep DC fixed, flatten the rest
        A_new = A.copy()
       dc = A_new[0]
        # For the non-DC components, scale toward flat
        non_dc_mag = mag[1:]
        mean_non_dc = non_dc_mag.mean()
        target_non_dc = mean_non_dc ** alpha * mag[1:] ** (1 - alpha)
        # Actually, let's just do: target = geometric mean of current and flat
        flat_val = np.sqrt(mean_non_dc * mean_non_dc)  # = mean_non_dc
        target_non_dc = mag[1:] ** (1-alpha) * flat_val ** alpha
        
        # Reconstruct
        phase = A / np.where(mag > 1e-15, mag, 1.0)
        A_new = np.zeros(fsize, dtype=complex)
        A_new[0] = dc
        A_new[1:] = target_non_dc * phase[1:]
        # Ensure Hermitian symmetry for real output
        # A[fsize-k] = conj(A[k])
        for k in range(1, fsize // 2):
            A_new[fsize - k] = np.conj(A_new[k])
        if fsize % 2 == 0:
            A_new[fsize // 2] = np.real(A_new[fsize // 2])
        
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() > 1e-10:
            a_new = a_new * S / a_new.sum()
        a = a_new
    
    return best_a, best_max

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_grad_fft(a, c, beta, fsize):
    """LSE gradient via FFT."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    w = e / e.sum()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A), fsize))[:n]

    Gerchberg-Saxton style: flatten the power spectrum to flatten autocorrelation.

            best_a = a.copy()
        
        # Compute target: geometric blend of current mag and median

Let me be careful about the step size. The LSE function F_beta(a) = (1/beta) * log(sum exp(beta * c[k])) where c[k] = (a*a)[k].

The gradient is: grad F_beta = (2/beta) * sum_k w[k] * d(a*a)[k]/da where w[k] = exp(beta*c[k]) / sum_j exp(beta*c[j])

The Lipschitz constant of this gradient: each d(a*a)[k]/da is a vector in direction of shifted a. The "Hessian" of (a*a)[k] is 2*C_k (circulant). The spectral norm of the weighted sum is at most 2 * max_k |A[k]| (where A = FFT(a)). For non-negative a, |A[0]| = S is the max, so L <= 2*S.

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
LSE gradient descent with adaptive beta, spectral equalization (Gerchberg-Saxton),
Frank-Wolfe conditional gradient, and random restarts.
Key insight: with normalization sum(a)=sqrt(2n), objective equals max(conv(a,a)),
and the lower bound is 2n/(2n-1)≈1. Target spectrum for flat autocorrelation
is a Dirichlet kernel shape, not flat.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta, fsize):
    """Gradient of LSE-smoothed max of convolution via FFT."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500.0, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A), fsize))[:n]
    return grad


def spectral_target_magnitude(fsize, n, S):
    """
    Compute target |A| for flat autocorrelation.
    Target autocorrelation: c[k] = S^2/(2n-1) for k=0..2n-2
    Target spectrum: P_target = FFT(c_target_padded)
    Take sqrt of max(P_target, 0) as target magnitude.
    """
    c_flat = S * S / (2 * n - 1)
    c_padded = np.zeros(fsize)
    c_padded[:2 * n - 1] = c_flat
    P_target = np.real(fft(c_padded))
    # P_target can be negative; take absolute value for a valid target
    target_mag = np.sqrt(np.abs(P_target))
    # Ensure DC is correct: A[0] should be S
    target_mag[0] = S
    return target_mag


def spectral_equalize_step(a, S, fsize, target_mag, alpha=0.3):
    """
    One step of spectral equalization: blend current |A| toward target.
    """
    A = fft(a, fsize)
    mag = np.abs(A)
    # Geometric blend
    safe_mag = np.maximum(mag, 1e-15)
    new_mag = safe_mag ** (1.0 - alpha) * target_mag ** alpha
    new_mag[0] = mag[0]  # preserve DC
    phase = A / safe_mag
    A_new = new_mag * phase
    a_new = np.real(ifft(A_new, fsize))[:len(a)]
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * S / a_new.sum()
    return a_new


def lse_descent(a_init, S, fsize, max_time, start_beta, end_beta,
                noise_interval=500, noise_scale_base=0.005, verbose=False):
    """
    LSE gradient descent with adaptive beta, step size, and noise.
    """
    n = len(a_init)
    a = project_simplex(a_init, S)
    lower_bound = S * S / (2 * n - 1)

    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Noise injection when stuck
        if stuck > noise_interval:
            scale = noise_scale_base * S / np.sqrt(n) * min(1.0, stuck / (2.0 * noise_interval))
            a = project_simplex(a + rng_global.normal(0, scale, n), S)
            stuck = 0
            continue

        # Adaptive beta
        beta = start_beta * (end_beta / start_beta) ** progress

        grad = lse_gradient(a, c, beta, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Step size: based on Lipschitz constant estimate
        # L ≈ beta * 2 * S (upper bound)
        step = 1.0 / (beta * 2.0 * S)
        # Also limit by gradient norm for stability
        step = min(step, 2.0 / (gnorm + 1e-16))

        # Line search (backtracking)
        accepted = False
        for _ls in range(5):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-9):
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            a = project_simplex(a - step * grad, S)

        if verbose and it % 5000 == 0:
            print(f"      it={it}, t={elapsed:.0f}s, beta={beta:.0f}, max={cur_max:.6f}, best={best_max:.6f}")

    return best_a, best_max


def frank_wolfe_phase(a_init, S, fsize, max_time, start_beta, end_beta):
    """
    Frank-Wolfe (conditional gradient) on the simplex.
    Naturally maintains a >= 0, sum(a) = S without projection.
    """
    n = len(a_init)
    a = project_simplex(a_init, S)

    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()

        beta = start_beta * (end_beta / start_beta) ** progress
        grad = lse_gradient(a, c, beta, fsize)

        # FW: v = S * e_{argmin grad}
        min_idx = np.argmin(grad)
        v = np.zeros(n)
        v[min_idx] = S

        # Step size
        t = 2.0 / (it + 2.0)
        # Blended step: mix FW with gradient direction
        a_fw = (1.0 - t) * a + t * v

        # Also try a projected gradient step and pick the better
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-16:
            step = min(1.0 / (beta * 2.0 * S), 2.0 / gnorm)
            a_pg = project_simplex(a - step * grad, S)
            c_pg = fft_convolve_self(a_pg)
            c_fw = fft_convolve_self(a_fw)
            if c_pg.max() < c_fw.max():
                a = a_pg
            else:
                a = a_fw
        else:
            a = a_fw

    return best_a, best_max


def spectral_gs_phase(a_init, S, fsize, target_mag, max_time, alpha=0.25):
    """
    Iterative spectral equalization (Gerchberg-Saxton style).
    """
    n = len(a_init)
    a = project_simplex(a_init, S)

    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        a = spectral_equalize_step(a, S, fsize, target_mag, alpha)
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()

        # Gradually increase alpha for more aggressive shaping
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        alpha = 0.15 + 0.35 * progress

    return best_a, best_max


def local_search_refine(a_init, S, fsize, max_time, rng):
    """
    Local search: try small redistributions of mass between nearby elements.
    """
    n = len(a_init)
    a = a_init.copy()

    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Try moving mass from a random position to a nearby position
        i = rng.integers(0, n)
        d = rng.integers(-5, 6)
        j = (i + d) % n
        if i == j:
            continue

        # Move a small amount of mass
        amt = 0.01 * S / n * rng.uniform(0.5, 2.0)
        amt = min(amt, a[i])  # can't move more than available

        if amt < 1e-12:
            continue

        a_try = a.copy()
        a_try[i] -= amt
        a_try[j] += amt

        c_try = fft_convolve_self(a_try)
        if c_try.max() < best_max:
            best_max = c_try.max()
            best_a = a_try.copy()
            a = a_try

    return best_a, best_max

    """Compute objective for a normalized sequence (sum = sqrt(2n))."""
    n = len(a)
    S = a.sum()
    if S < 0.01:

    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)
