"""

Strategy: Multi-phase optimization combining:
1. FISTA with LSE smoothing (Euclidean projection) from parent sequence
2. Mirror descent (multiplicative/exponential updates) for non-negativity
3. Spectral equalization periodic reshaping
4. Upsample to n=16384 for more degrees of freedom
5. Active-set subgradient with varying K for final polish

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_grad(a, beta):
    """Compute LSE value and gradient of beta*LSE(c) w.r.t. a. Returns (lse_val, grad, c, w)."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def topk_subgrad(a, c, k):
    """Subgradient from top-k lags of the convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, max_time, rng, beta_init=50.0, beta_rate=0.015,
              beta_max=50000.0, step_scale=1.0, use_backtrack=True):
    """FISTA with adaptive LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_grad(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-16 and bt < 8:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 3:
                step = step_local * 0.7
            else:
                step = min(step * 1.04, 3.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            step = step_scale / max(beta, 1.0)
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)

        t_mom = t_new

def run_mirror_descent(a_init, max_time, rng, beta_init=20.0, beta_rate=0.02,
                        beta_max=50000.0, lr_base=0.5, lr_decay=0.9995):
    """
    Mirror descent with multiplicative (exponential) updates.
    a_new[j] = a[j] * exp(-lr * grad[j]), then normalize to sum=S.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 1e-10, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    lr = lr_base

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_grad(a, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive learning rate
        lr_eff = lr * (S / gnorm) * 0.5
        # Clamp the exponent to avoid overflow
        exponent = -lr_eff * grad
        exponent = np.clip(exponent, -50, 50)
        a_new = a * np.exp(exponent)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new * (S / a_new.sum())

        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-12:
            a = a_new
            lr = min(lr * 1.02, 5.0)
        else:
            # Halve learning rate, try smaller step
            lr *= 0.5
            exponent = -lr * (S / gnorm) * 0.5 * grad
            exponent = np.clip(exponent, -50, 50)
            a_new = a * np.exp(exponent)
            a_new = np.maximum(a_new, 1e-300)
            a_new = a_new * (S / a_new.sum())
            obj_new2 = compute_obj(a_new)
            if obj_new2 <= cur_obj + 1e-12:
                a = a_new

        # Periodic decay of lr
        lr *= lr_decay

        stuck = it - last_improve
        if stuck > 2000 and it % 100 == 0:
            # Random perturbation to escape
            p = 0.01 * np.exp(-stuck / 3000.0)
            mask = rng.random(n) < p
            a[mask] *= rng.uniform(0.5, 2.0, mask.sum()) if mask.sum() > 0 else 1
            a = np.maximum(a, 1e-10)
            a = a * (S / a.sum())

    return best_a, best_obj

    S = a.sum()
    a_work = a.copy()
    fsize = 1 << (2 * n - 1).bit_length()

    for _ in range(max_iterations):
        A = fft(a_work, fsize)
        power = np.abs(A) ** 2
        # Compress dynamic range of power spectrum
        p_min = power.min()
        p_max = power.max()
        if p_max < 1e-30:
            break
        ratio = p_max / (p_min + 1e-30)
        if ratio < 1.1:
            break
        # Apply spectral compression
        target_ratio = min(ratio, ratio ** (1.0 - strength))
        p_new = p_min + (power - p_min) * (target_ratio / ratio)
        # Reconstruct with new amplitude, keep phase
        phase = np.angle(A)
        A_new = np.sqrt(p_new) * np.exp(1j * phase)
        a_candidate = np.real(ifft(A_new, fsize))[:n]
        # Project to non-negative
        a_candidate = np.maximum(a_candidate, 0.0)
        if a_candidate.sum() < 1e-15:
            break
        a_candidate = a_candidate * (S / a_candidate.sum())

        # Check if it improved
        obj_old = compute_obj(a_work)
        obj_new = compute_obj(a_candidate)
        if obj_new < obj_old - 1e-10:
            a_work = a_candidate
        else:
            # Try weaker strength
            strength *= 0.8
            if strength < 0.01:
                break

    return a_work


def run_subgradient_polish(a_init, max_time, rng):
    """Polish using exact top-K subgradient with adaptive K and step size."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    step = S * 0.1

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        # Adaptive K: start small, grow when stuck
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 3
        elif stuck < 2000:
            k = min(10, 3 + stuck // 100)
        else:
            k = min(50, 10 + (stuck - 2000) // 100)

        grad = topk_subgradient(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step with backtracking
        step_local = step
        for _ in range(10):
            a_new = project_simplex(a - step_local * grad, S)
            obj_new = compute_obj(a_new)
            if obj_new <= cur_obj + 1e-13:
                a = a_new
                step = min(step * 1.05, S)
                break
            step_local *= 0.5
        else:
            step = step_local * 0.5

        if stuck > 2000 and it % 200 == 0:
            ns = max(1e-7, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 5000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def block_coordinate_lp(a_fixed, M, n, max_iter=200):
    """
    Approximate LP solve: max sum(b) s.t. b>=0, (a_fixed*b)[k] <= M for all k.
    Uses a projected gradient / active set approximation.
    """
    b = np.ones(n) * M / (np.sum(a_fixed) * 2)  # initial feasible guess (rough)
    S_target = np.sum(a_fixed)  # we'll normalize at the end

    for _ in range(max_iter):
        # Compute convolution c = conv(a_fixed, b)
        fsize = 1 << (2 * n - 1 - 1).bit_length()
        Af = fft(a_fixed, fsize)
        Bf = fft(b, fsize)
        c = np.real(ifft(Af * Bf))[:2 * n - 1]

        # Find violated constraints
        violations = c - M
        if violations.max() < 1e-12:
            break

        # Gradient of objective 1^T b w.r.t. b is just ones
        # Gradient of constraint violation: for each k, dc[k]/db[l] = a_fixed[k-l]
        # We want to increase b where it doesn't violate constraints
        # Use: b_new = b + lr * (ones - penalty * grad_violation)
        
        # Compute subgradient of max constraint violation
        viol_idx = np.argsort(violations)[::-1][:min(10, 2*n-1)]
        w = np.zeros(2 * n - 1)
        w[viol_idx] = 1.0 / len(viol_idx)

        # Gradient of (a_fixed * b) w.r.t. b:
        # dc[k]/db[l] = a_fixed[k-l] if 0<=k-l<n
        # Full gradient: sum_k w[k] * a_fixed[k-l] for each l
        # This is cross-correlation of w and a_fixed
        Wf = fft(w, fsize)
        # grad_b[l] = sum_k w[k] * a_fixed[k-l]
        # = ifft(fft(w) * conj(fft(a_fixed)))[l] ... need to check
        # Actually: cross_corr[w, a_fixed][l] = sum_k w[k+l] * a_fixed[k]
        # We want: sum_k w[k] * a_fixed[k-l]
        # Let m = k-l: sum_m w[m+l] * a_fixed[m] = cross_corr at lag l
        grad_b = np.real(ifft(Wf * np.conj(Af)))[:n]

        # Update: increase b where gradient is low, decrease where high
        lr = M / (np.linalg.norm(grad_b) + 1e-10) * 0.1
        b = b + lr * (1.0 - grad_b / (grad_b.max() + 1e-10))
        b = np.maximum(b, 0.0)

    b_sum = b.sum()
    if b_sum < 1e-15:
        return a_fixed.copy()
    return b * (S_target / b_sum)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA, mirror descent, spectral equalization, upsample, polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point from parent
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        S = np.sqrt(2.0 * n)
        a_start = np.ones(n) * (S / n)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.8f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: FISTA refinement (35% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.35
    if p1_time > 30:
        print("\n=== Phase 1: FISTA refinement (budget %.0fs) ===" % p1_time)

        # Multiple FISTA runs with different parameters
        fista_configs = [
            (20, 0.02, 10000, 1.2, True),
            (50, 0.01, 50000, 0.8, True),
            (200, 0.005, 100000, 0.5, False),
        ]
        for i, (bi, br, bm, ss, bt) in enumerate(fista_configs):
            if time.monotonic() >= deadline:
                break
            rem = min(p1_time / 3.0, deadline - time.monotonic())
            if rem < 5:
                break
            a_opt, obj_opt = run_fista(global_best_a, rem, rng,
                                        beta_init=bi, beta_rate=br,
                                        beta_max=bm, step_scale=ss,
                                        use_backtrack=bt)
            print("  FISTA%d (beta0=%d): obj=%.8f" % (i + 1, bi, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()
                print("  *** NEW BEST ***")

    # === Phase 2: Mirror descent (25% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.25
    if p2_time > 20:
        print("\n=== Phase 2: Mirror descent (budget %.0fs) ===" % p2_time)

        md_configs = [
            (10, 0.03, 50000, 0.8, 0.999),
            (30, 0.015, 20000, 1.5, 0.9995),
            (5, 0.05, 100000, 0.4, 0.998),
        ]
        for i, (bi, br, bm, lb, ld) in enumerate(md_configs):
            if time.monotonic() >= deadline:
                break
            rem = min(p2_time / 3.0, deadline - time.monotonic())
            if rem < 5:
                break
            a_opt, obj_opt = run_mirror_descent(global_best_a, rem, rng,
                                                 beta_init=bi, beta_rate=br,
                                                 beta_max=bm, lr_base=lb,
                                                 lr_decay=ld)
            print("  MirrorDesc%d: obj=%.8f" % (i + 1, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()
                print("  *** NEW BEST ***")

    # === Phase 3: Spectral equalization (10% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.10
    if p3_time > 10 and len(global_best_a) <= 16384:
        print("\n=== Phase 3: Spectral equalization (budget %.0fs) ===" % p3_time)

        for strength in [0.5, 0.3, 0.2, 0.1]:
            if time.monotonic() >= deadline:
                break

                                       strength=strength, rng=rng)

            print("  Spectral (s=%.1f): obj=%.8f" % (strength, obj_spec))
            if obj_spec < global_best_obj:
                global_best_obj = obj_spec
                global_best_a = a_spec.copy()
                print("  *** NEW BEST ***")

        # Follow up with FISTA

                                        beta_init=50, beta_rate=0.01,
                                        beta_max=50000, step_scale=0.8,
                                        use_backtrack=True)
            print("  FISTA after spectral: obj=%.8f" % obj_opt)
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 4: Upsample to larger n (20% time) ===

            print("  *** NEW BEST ***")

        # FISTA on upsampled

                                        beta_init=5, beta_rate=0.01,
                                        beta_max=20000, step_scale=1.0,
                                        use_backtrack=False)
            print("  FISTA upsampled: obj=%.8f" % obj_opt)
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()
                print("  *** NEW BEST ***")

        # Mirror descent on upsampled
        rem4 = deadline - time.monotonic()
        if rem4 > 10:
            a_opt, obj_opt = run_mirror_descent(global_best_a, rem4 * 0.4, rng,
                                                 beta_init=10, beta_rate=0.02,

            print("  MirrorDesc upsampled: obj=%.8f" % obj_opt)

                print("  *** NEW BEST ***")

        # Spectral equalization on upsampled

        # Subgradient polish

            if obj_pol < global_best_obj:
                global_best_obj = obj_pol
                global_best_a = a_pol.copy()

        # High-beta FISTA
        rem5 = deadline - time.monotonic()
        if rem5 > 5:
            a_opt, obj_opt = run_fista(global_best_a, rem5 * 0.5, rng,
                                        beta_init=1000, beta_rate=0.002,
                                        beta_max=500000, step_scale=0.3,

            print("  High-beta FISTA: obj=%.8f" % obj_opt)

        # Mirror descent final

        if rem5 > 5:
