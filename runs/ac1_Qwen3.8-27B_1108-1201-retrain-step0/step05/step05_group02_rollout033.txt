import numpy as np

from scipy.fft import fft, ifft


def fft_convolve(a, b=None):
    """Full convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    if fsize < size:
        fsize = 2 * size
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def lse_and_grad(a, beta):

    n = len(a)

def run_fista(a_init, max_time, rng, beta_init=50.0, beta_rate=0.005,
              beta_max=50000.0, step_scale=1.0, restart_every=2000):
    """FISTA with LSE smoothing and exact-objective backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_and_grad(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking with exact objective
        step_local = step
        x_trial = project_simplex(y - step_local * grad, S)
        obj_trial = compute_obj(x_trial)
        bt = 0
        while obj_trial > cur_obj + 1e-12 and step_local > 1e-14 and bt < 8:
            step_local *= 0.5
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt += 1
        
        if bt > 3:
            step = step_local * 0.7
        else:
            step = min(step * 1.02, 4.0 / max(beta, 1.0))
        
        x_new = x_trial
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        # Restart momentum periodically
        if it % restart_every == 0:
            t_mom = 1.0
            y = a.copy()
        
        # Add noise when stuck
        stuck = it - last_improve
        if stuck > 3000 and it % 100 == 0:
            ns = max(1e-6, 0.01 * S / np.sqrt(n) * np.exp(-stuck / 5000))
            y = project_simplex(y + rng.normal(0, ns, n), S)
            t_mom = 1.0
    
    return best_a, best_obj

        if it - last_improve > 50000:
            break

        k_star = int(np.argmax(c))

                    rough_max = max(new_c_star, c.max() * 0.99)  # rough estimate

def make_init(n, method, rng):
    """Create initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "uniform":
        a = np.ones(n)
    elif method == "triangle":
        a = np.abs(1 - 2 * tt)
    elif method == "bump":
        a = np.exp(-5.0 * (tt - 0.5) ** 2)
    elif method == "chirp_low":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 2 * tt)
    elif method == "chirp_mid":
        a = 0.3 + 0.4 * np.abs(np.sin(2 * np.pi * 5 * tt + 0.5))
    elif method == "chirp_high":
        a = 0.2 + 0.5 * np.abs(np.sin(2 * np.pi * 13 * tt + 1.2))
    elif method == "asym_bump":
        a = np.exp(-5.0 * (tt - 0.3) ** 2) + 0.5 * np.exp(-5.0 * (tt - 0.7) ** 2)
    elif method == "random":
        a = rng.random(n) ** 2 + 0.01
    elif method == "two_peak":
        a = np.exp(-20.0 * (tt - 0.25) ** 2) + np.exp(-20.0 * (tt - 0.75) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def upsample(a, factor=2):
    """Upsample by factor using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-scale FISTA with hybrid LSE/exact-subgradient,
exact backtracking, diverse asymmetric initializations, and
greedy coordinate descent refinement. Uses upscaling to grow
sequence length while preserving structure.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize <<= 1
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_and_grad(a, beta):
    """
    Compute LSE(beta * c) and its gradient w.r.t. a.
    c = a*a (full convolution).
    Returns (lse_value, gradient, c, weights).
    """
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    x = beta * (c - cm)
    x = x - x.max()
    e = np.exp(x)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
    # This equals 2 * (w * a_reversed)[i] convolved appropriately
    # Using FFT: grad = 2 * ifft(conj(FFT(a)) * FFT(w))[:n]
    fsize = 1
    needed = 3 * n - 1
    while fsize < needed:
        fsize <<= 1
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return lse_val, grad, c, w


def exact_subgrad_topk(a, c, k):
    """Exact subgradient using top-k lags of the convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    fsize = 1
    needed = 3 * n - 1
    while fsize < needed:
        fsize <<= 1
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(np.conj(A) * W))[:n]

def coord_descent_fast(a_init, max_time, rng):
    """
    Fast greedy coordinate descent via mass transfer.
    O(n) per iteration by focusing on the peak lag.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    consecutive_fail = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
            consecutive_fail = 0
        
        if it - last_improve > 20000:
            break
        
        k_star = int(np.argmax(c))
        
        # Contribution of coordinate j to c[k_star]: a[k_star - j] (if valid)
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi >= lo:
            j_range = np.arange(lo, hi + 1)
            contrib[j_range] = a[k_star - j_range]
        
        # Score: high contrib + high value = good source
        #         low contrib + low value = good target
        val = a
        src_score = contrib * val
        tgt_score = -contrib * (val + 1e-10)
        
        # Pick top source and target
        src_j = int(np.argmax(src_score))
        tgt_l = int(np.argmin(tgt_score))
        
        if src_j == tgt_l:
            # Find second best target
            tgt_scores_copy = tgt_score.copy()
            tgt_scores_copy[tgt_l] = np.inf
            tgt_l = int(np.argmin(tgt_scores_copy))
        
        if a[src_j] < 1e-12:
            consecutive_fail += 1
            if consecutive_fail > 500:
                idx = rng.integers(0, n)
                delta = 0.01 * S / n
                a[idx] = max(0, a[idx] - delta)
                a[rng.integers(0, n)] += delta
                c = fft_convolve_self(a)
            continue
        
        # Try several transfer fractions
        best_new_obj = cur_obj + 1e-15
        best_frac = 0
        for frac in [0.02, 0.05, 0.1, 0.2]:
            delta = a[src_j] * frac
            if delta < 1e-14:
                continue
            # Quick check: effect on c[k_star]
            # c[k_star] changes by: -delta * a[k_star - src_j] + delta * a[k_star - tgt_l]
            delta_c_star = 0.0
            if 0 <= k_star - src_j < n:
                delta_c_star -= delta * a[k_star - src_j]
            if 0 <= k_star - tgt_l < n:
                delta_c_star += delta * a[k_star - tgt_l]
            new_c_star = c[k_star] + delta_c_star
            if new_c_star >= cur_max and cur_max > 0:
                # Likely won't improve, skip
                continue
            # Verify with exact computation (expensive but needed)
            a[src_j] -= delta
            a[tgt_l] += delta
            c_new = fft_convolve_self(a)
            new_obj = 2.0 * n * c_new.max() / (S * S)
            if new_obj < best_new_obj:
                best_new_obj = new_obj
                best_frac = frac
                best_c_new = c_new.copy()
            # Undo
            a[src_j] += delta
            a[tgt_l] -= delta
        
        if best_frac > 0:
            delta = a[src_j] * best_frac
            a[src_j] -= delta
            a[tgt_l] += delta
            c = best_c_new
        else:
            consecutive_fail += 1
            if consecutive_fail > 100:
                # Random perturbation
                idx = rng.integers(0, n)
                delta = 0.01 * a[idx] if a[idx] > 0 else 0.001 * S / n
                if delta > 1e-14:
                    a[idx] -= delta
                    a[rng.integers(0, n)] += delta

def make_init(n, method, rng):
    """Create diverse initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    j = np.arange(n, dtype=np.float64)
    
    if method == "uniform":
        a = np.ones(n)
    elif method == "triangle":
        a = 1.0 - 2.0 * np.abs(tt - 0.5)
    elif method == "bump_center":
        a = np.exp(-8.0 * (tt - 0.5) ** 2) + 0.01
    elif method == "asym_bump":
        a = np.exp(-8.0 * (tt - 0.35) ** 2) + 0.3 * np.exp(-8.0 * (tt - 0.7) ** 2) + 0.01
    elif method == "two_peak":
        a = np.exp(-15.0 * (tt - 0.2) ** 2) + np.exp(-15.0 * (tt - 0.8) ** 2) + 0.01
    elif method == "chirp_2":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * 2.0 * tt)
    elif method == "chirp_5":
        a = 0.3 + 0.5 * np.abs(np.sin(2 * np.pi * 5.0 * tt + 0.3))
    elif method == "chirp_13":
        a = 0.2 + 0.6 * np.abs(np.sin(2 * np.pi * 13.0 * tt + 1.1)) + 0.05 * np.cos(2 * np.pi * 7.0 * tt)
    elif method == "quadratic_phase":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (j ** 2) / n)
    elif method == "random_pow":
        a = rng.random(n) ** 3 + 0.005
    elif method == "random_bump":
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.3)
        a = np.exp(-(tt - center) ** 2 / (2 * width ** 2)) + 0.01
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def upsample_linear(a, factor=2):
    """Upsample by linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    result = np.interp(t_new, t_old, a)
    return result

            a_opt, obj_opt = run_fista(global_best_a, rem, rng,
                                        beta_init=200, beta_rate=0.002,
                                        beta_max=80000, step_scale=1.0,
                                        restart_every=2000, use_exact_grad=False)
        else:
            a_opt, obj_opt = run_fista(global_best_a, rem, rng,
                                        beta_init=5000, beta_rate=0.0005,
                                        beta_max=200000, step_scale=0.5,
                                        restart_every=3000, use_exact_grad=True)
        print("  FISTA v%d: obj=%.8f" % (variant, obj_opt))
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()
            print("  *** NEW BEST ***")
    
    # === Phase 2: Diverse initializations (15% time) ===
    remaining = deadline - time.monotonic()
    p2 = remaining * 0.15
    if p2 > 30:
        print("\n=== Phase 2: Diverse inits (budget %.0fs) ===" % p2)
        n_try = len(global_best_a)
        methods = ["chirp_5", "quadratic_phase", "two_peak", "random_pow", "asym_bump"]
        per_time = p2 / len(methods)
        for method in methods:
            if time.monotonic() >= deadline:
                break
            a_div = make_init(n_try, method, rng)
            a_opt, obj_opt = run_fista(a_div, per_time * 0.7, rng,
                                        beta_init=100, beta_rate=0.004,
                                        beta_max=50000, step_scale=0.8)
            print("  %s: obj=%.8f" % (method, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()
                print("  *** NEW BEST ***")
    
    # === Phase 3: Coordinate descent (15% time) ===
    remaining = deadline - time.monotonic()
    p3 = remaining * 0.15
    if p3 > 30:
        print("\n=== Phase 3: Coord descent (budget %.0fs) ===" % p3)
        a_cd, obj_cd = coord_descent_fast(global_best_a, p3, rng)
        print("  CD: obj=%.8f" % obj_cd)
        if obj_cd < global_best_obj:
            global_best_obj = obj_cd
            global_best_a = a_cd.copy()
            print("  *** NEW BEST ***")
        
        # Quick FISTA after CD
        remaining3 = deadline - time.monotonic()
        if remaining3 > 15:
            a_cd2, obj_cd2 = run_fista(global_best_a, remaining3 * 0.5, rng,
                                        beta_init=1000, beta_rate=0.001,
                                        beta_max=100000, step_scale=0.5)
            print("  FISTA after CD: obj=%.8f" % obj_cd2)
            if obj_cd2 < global_best_obj:
                global_best_obj = obj_cd2
                global_best_a = a_cd2.copy()
    
    # === Phase 4: Upsample (20% time) ===
    remaining = deadline - time.monotonic()
    p4 = remaining * 0.20
    if p4 > 60 and len(global_best_a) <= 8192:
        factor = 2 if len(global_best_a) <= 4096 else 2
        n2 = len(global_best_a) * factor
        if n2 <= 16384:
            print("\n=== Phase 4: Upsample to n=%d (budget %.0fs) ===" % (n2, p4))
            a_up = upsample_linear(global_best_a, factor=factor)
            S2 = np.sqrt(2.0 * n2)
            a_up = a_up * (S2 / a_up.sum())
            obj_up = compute_obj(a_up)
            print("  Upsampled initial: obj=%.8f" % obj_up)
            if obj_up < global_best_obj:
                global_best_obj = obj_up
                global_best_a = a_up.copy()
            
            a_up_opt, obj_up_opt = run_fista(a_up, p4 * 0.65, rng,
                                              beta_init=50, beta_rate=0.002,
                                              beta_max=50000, step_scale=1.0)
            print("  FISTA upsampled: obj=%.8f" % obj_up_opt)
            if obj_up_opt < global_best_obj:
                global_best_obj = obj_up_opt
                global_best_a = a_up_opt.copy()
                print("  *** NEW BEST ***")
            
            # CD on upsampled
            remaining4 = deadline - time.monotonic()
            if remaining4 > 20:
                a_cd_up, obj_cd_up = coord_descent_fast(global_best_a, remaining4 * 0.6, rng)
                print("  CD upsampled: obj=%.8f" % obj_cd_up)
                if obj_cd_up < global_best_obj:
                    global_best_obj = obj_cd_up
                    global_best_a = a_cd_up.copy()
    
    # === Phase 5: Final polish (15% time) ===
    remaining = deadline - time.monotonic()
    p5 = remaining * 0.90
    if p5 > 10:
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % p5)
        a_pol, obj_pol = run_fista(global_best_a, p5 * 0.6, rng,
                                     beta_init=10000, beta_rate=0.0003,
                                     beta_max=500000, step_scale=0.3,
                                     restart_every=5000, use_exact_grad=True)
        print("  High-beta polish: obj=%.8f" % obj_pol)
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()
        
        remaining5 = deadline - time.monotonic()
        if remaining5 > 5:
            a_pol2, obj_pol2 = run_fista(global_best_a, remaining5, rng,
                                          beta_init=50000, beta_rate=0.0001,
                                          beta_max=1000000, step_scale=0.15,
                                          restart_every=5000, use_exact_grad=True)
            print("  Ultra-high-beta: obj=%.8f" % obj_pol2)
            if obj_pol2 < global_best_obj:
                global_best_obj = obj_pol2
                global_best_a = a_pol2.copy()
    
    # Final verification
    print("\n=== Final Result ===")
    print("Best obj found: %.8f, n=%d" % (global_best_obj, len(global_best_a)))
    
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified: obj=%.8f, n=%d" % (ver, len(final_list)))
            return final_list
        else:
            print("WARNING: invalid result, using fallback")
    
    # Fallback: uniform sequence
    n_fb = 4096
    a_fb = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
    return [float(x) for x in a_fb]
