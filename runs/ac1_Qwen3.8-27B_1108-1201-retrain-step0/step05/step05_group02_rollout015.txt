def fft_convolve_self(a):
    """Full self-convolution via FFT."""

    size = 2 * n - 1

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]

    idx = np.arange(1, n + 1, dtype=np.float64)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""

def lse_value_and_grad(y, c, beta):

        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0

        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()

    else:
        a = a * (S / s)
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 5000:
            break

        # Adaptive top-k
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 2
        elif stuck < 2000:
            k = min(8, 2 + stuck // 200)
        else:
            k = min(32, 8 + stuck // 200)

        # Compute subgradient from top-k peaks
        top_idx = np.argsort(c)[::-1][:k]
        w_full = np.zeros(len(c))
        w_full[top_idx] = 1.0 / k
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = 2.0 * S / (gnorm * (1.0 + 0.005 * it))
        step = min(step, 5.0 * S / gnorm)

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)

        if obj_new <= cur_obj + 1e-13:
            a = a_new
            c = c_new
        else:
            step *= 0.25
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            obj_new = 2.0 * n * c_new.max() / (S * S)
            if obj_new <= cur_obj + 1e-13:
                a = a_new
                c = c_new

        if stuck > 2000 and it % 50 == 0:
            ns = max(1e-8, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 4000))
            a = project_simplex(a + rng.normal(0, ns, n), S)
            c = fft_convolve_self(a)

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via zero-stuffing + simple lowpass (box filter)."""
    n = len(a)
    n_new = n * factor
    a_new = np.zeros(n_new)
    a_new[0::factor] = a
    # Simple box filter to smooth
    if factor >= 2:
        kernel = np.ones(factor) / factor
        a_new = np.convolve(a_new, kernel, mode='same')
        a_new = np.maximum(a_new, 0)
    return a_new


def upsample_interp(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-4.0 * (tt - 0.5) ** 2 / (0.2 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 3 * tt + 0.5)
        a = np.abs(a)
    elif method == "multi_tone":
        a = 0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(
            2 * np.pi * 7 * tt + 1.3) + 0.15 * np.sin(2 * np.pi * 11 * tt + 2.1)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 2.0 * np.abs(tt - 0.5))
    elif method == "edge_weighted":
        a = 0.1 + 0.9 * (np.abs(tt - 0.5) * 2.0) ** 2
    elif method == "two_bursts":
        center = n // 2
        a = np.zeros(n)
        half = n // 4
        a[:half] = 1.0
        a[-half:] = 1.0
        a = a + 0.01 * rng.random(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: multi-scale construction, LSE-FISTA, peak shaving,
    length exploration, and subgradient polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)

    global_best_obj = np.inf
    global_best_a = None

    # Try to get parent sequence
    a_parent = None
    try:
        a_parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_parent = np.clip(a_parent, 0, 1000)
        if a_parent.sum() < 0.01:
            a_parent = None
    except Exception:
        a_parent = None

    if a_parent is not None:
        obj_parent = compute_obj(a_parent)
        print("Parent: n=%d, obj=%.6f" % (len(a_parent), obj_parent))
        global_best_obj = obj_parent
        global_best_a = a_parent.copy()
    else:
        n0 = 8192
        global_best_a = make_init(n0, "cosine", rng)
        global_best_obj = compute_obj(global_best_a)
        print("No parent. Initial: n=%d, obj=%.6f" % (len(global_best_a), global_best_obj))

    # === Phase 1: Multi-scale hierarchical optimization (20% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.20
    if p1_time > 60:
        print("\n=== Phase 1: Multi-scale optimization (budget %.0fs) ===" % p1_time)

        # Start from small n and work up
        sizes = [16, 64, 256, 1024]
        a_scale = make_init(16, "cosine", rng)
        per_time = p1_time / (len(sizes) + 2)

        for i, target_n in enumerate(sizes):
            if time.monotonic() >= deadline:
                break
            if len(a_scale) < target_n:
                # Upsample
                factor = max(1, target_n // len(a_scale))
                a_scale = upsample(a_scale, factor)
                if len(a_scale) != target_n:
                    # Pad or crop
                    if len(a_scale) > target_n:
                        a_scale = a_scale[:target_n]
                    else:
                        pad = target_n - len(a_scale)
                        a_scale = np.concatenate([a_scale, np.zeros(pad)])
                S = np.sqrt(2.0 * target_n)
                a_scale = a_scale * (S / a_scale.sum())

            # Optimize at this scale
            rem = min(per_time, deadline - time.monotonic())
            if rem < 5:
                break
            a_opt, obj_opt = run_lse_fista(a_scale, rem, rng,
                                           beta_init=10, beta_rate=0.01,
                                           beta_max=10000, step_scale=1.0,
                                           use_backtrack=False)
            print("  Scale n=%d: obj=%.6f" % (target_n, obj_opt))
            a_scale = a_opt

        # Now upsample to the target size and compare
        n_target = len(global_best_a)
        if len(a_scale) < n_target:
            factor = max(1, n_target // len(a_scale))
            a_multiscale = upsample(a_scale, factor)
            if len(a_multiscale) != n_target:
                if len(a_multiscale) > n_target:
                    a_multiscale = a_multiscale[:n_target]
                else:
                    pad = n_target - len(a_multiscale)
                    a_multiscale = np.concatenate([a_multiscale, np.zeros(pad)])
            S = np.sqrt(2.0 * n_target)
            a_multiscale = a_multiscale * (S / a_multiscale.sum())
            obj_ms = compute_obj(a_multiscale)
            print("  Multi-scale upsampled: obj=%.6f" % obj_ms)
            if obj_ms < global_best_obj:
                global_best_obj = obj_ms
                global_best_a = a_multiscale.copy()

    # === Phase 2: Diverse initialization screening (15% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.15
    if p2_time > 40:
        print("\n=== Phase 2: Diverse starts (budget %.0fs) ===" % p2_time)
        n_work = len(global_best_a)
        methods = ["cosine", "gaussian", "chirp", "multi_tone", "random",
                   "edge_weighted", "two_bursts"]
        per_time = p2_time / len(methods)

        for method in methods:
            if time.monotonic() >= deadline:
                break
            a_div = make_init(n_work, method, rng)
            a_opt, obj_opt = run_lse_fista(a_div, per_time, rng,
                                           beta_init=15, beta_rate=0.012,
                                           beta_max=5000, step_scale=0.8,
                                           use_backtrack=False)
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()
                print("  %s: obj=%.6f ***NEW BEST***" % (method, obj_opt))
            else:
                print("  %s: obj=%.6f" % (method, obj_opt))

    # === Phase 3: Intensive LSE-FISTA (30% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.30
    if p3_time > 60:
        print("\n=== Phase 3: Intensive FISTA (budget %.0fs) ===" % p3_time)

        variants = [
            (30, 0.008, 20000, 1.0, True),
            (100, 0.005, 50000, 0.6, True),
            (500, 0.003, 100000, 0.4, True),
            (2000, 0.002, 200000, 0.2, False),
        ]
        per_variant = p3_time / len(variants)
        for i, (bi, br, bm, ss, bt) in enumerate(variants):
            if time.monotonic() >= deadline:
                break
            rem = min(per_variant, deadline - time.monotonic())
            if rem < 5:
                break
            a_opt, obj_opt = run_lse_fista(global_best_a, rem, rng,
                                           beta_init=bi, beta_rate=br,
                                           beta_max=bm, step_scale=ss,
                                           use_backtrack=bt)
            print("  FISTA variant %d: obj=%.6f" % (i + 1, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 4: Peak shaving (20% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.20
    if p4_time > 30:
        print("\n=== Phase 4: Peak shaving (budget %.0fs) ===" % p4_time)
        a_ps, obj_ps = peak_shaving(global_best_a, p4_time * 0.6, rng, top_k=5)
        print("  Peak shaving: obj=%.6f" % obj_ps)
        if obj_ps < global_best_obj:
            global_best_obj = obj_ps
            global_best_a = a_ps.copy()

        # Follow up with FISTA
        remaining4 = deadline - time.monotonic()
        if remaining4 > 15:
            a_f2, obj_f2 = run_lse_fista(global_best_a, remaining4 * 0.7, rng,
                                         beta_init=500, beta_rate=0.003,
                                         beta_max=100000, step_scale=0.3,
                                         use_backtrack=True)
            print("  FISTA after PS: obj=%.6f" % obj_f2)
            if obj_f2 < global_best_obj:
                global_best_obj = obj_f2
                global_best_a = a_f2.copy()

    # === Phase 5: Explore different lengths (10% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.10
    if p5_time > 30:
        print("\n=== Phase 5: Length exploration (budget %.0fs) ===" % p5_time)
        n_current = len(global_best_a)
        candidate_ns = [n_current // 2, n_current, n_current * 2]
        # Filter to valid sizes
        candidate_ns = [n for n in candidate_ns if 1024 <= n <= 32768]
        # Remove duplicates while preserving order
        seen = set()
        unique_ns = []
        for n in candidate_ns:
            if n not in seen:
                seen.add(n)
                unique_ns.append(n)

        per_n = p5_time / len(unique_ns)
        for n_try in unique_ns:
            if time.monotonic() >= deadline:
                break
            rem = min(per_n, deadline - time.monotonic())
            if rem < 10:
                break
            if n_try == n_current:
                a_try = global_best_a.copy()
            else:
                # Upsample or downsample
                if n_try > n_current:
                    a_try = upsample_interp(global_best_a, factor=n_try // n_current)
                    if len(a_try) != n_try:
                        if len(a_try) > n_try:
                            a_try = a_try[:n_try]
                        else:
                            a_try = np.concatenate([a_try, np.zeros(n_try - len(a_try))])
                else:
                    # Downsample by averaging
                    n_blocks = n_current // n_try
                    a_try = global_best_a[:n_blocks * n_try].reshape(n_try, n_blocks).mean(axis=1)
                S_try = np.sqrt(2.0 * n_try)
                a_try = a_try * (S_try / a_try.sum())

            a_opt, obj_opt = run_lse_fista(a_try, rem, rng,
                                           beta_init=50, beta_rate=0.006,
                                           beta_max=30000, step_scale=0.8,
                                           use_backtrack=True)
            print("  n=%d: obj=%.6f" % (n_try, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 6: Final polish (5% time) ===
    remaining = deadline - time.monotonic()
    p6_time = remaining * 0.95
    if p6_time > 10:
        print("\n=== Phase 6: Final polish (budget %.0fs) ===" % p6_time)
        a_pol, obj_pol = subgradient_polish(global_best_a, p6_time * 0.5, rng)
        print("  Subgradient polish: obj=%.6f" % obj_pol)
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()

        remaining6 = deadline - time.monotonic()
        if remaining6 > 5:
            a_ps2, obj_ps2 = peak_shaving(global_best_a, remaining6 * 0.8, rng, top_k=8)
            print("  Final peak shaving: obj=%.6f" % obj_ps2)
            if obj_ps2 < global_best_obj:
                global_best_obj = obj_ps2
                global_best_a = a_ps2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver) and ver < 10:
            print("\nFinal verified: obj=%.6f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid result (%.4f), using fallback" % ver)
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
            ver = evaluate_sequence(final_list)
            print("Fallback: obj=%.6f" % ver)
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
