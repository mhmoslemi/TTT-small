```
def random_spectral_candidate(n,N,rng,mode='ones',iters=16):
    c=np.zeros(N)
    if mode=='ones':
       c[:2*n-1]=1.0
    elif mode=='p':
       p=p_init(n)
       c_all=fft_conv(p,N); c[:2*n-1]=np.minimum(1.0,c_all/c_all.max() if c_all.max()>0 else 0)
    elif mode=='random':
       c[:2*n-1]=rng.uniform(0.4,1.0,size=2*n-1)
    else: ...
    a=np.maximum(rng.normal(0,1,n),0)+0.1
    for _ in range(iters):
       # add slack to target to increase sum
       slack=1.0-c
       if np.any(slack>1e-12):
           # add random bump: target = min(1, c + 0.2*rand*slack)
           c=np.minimum(1.0, c + rng.uniform(0,0.3, size=2*n-1)*slack[:2*n-1])
       C=rfft(c,N)
       A=rfft(a,N)
       s=np.sqrt(C)
       scores=np.real(s*np.conj(A))
       signs=np.where(scores>=0,1,-1)
       # random flips 0.3
       flip=rng.random(len(signs))<0.3; signs[flip]*=-1
       signs[0]=1
       B=signs*s
       B[0]=abs(B[0])
       if N%2==0:
           B[-1]=signs[-1]*np.sqrt(abs(np.real(C[-1])))
       b=irfft(B,N)[:n]
       b=np.maximum(b,0)
       if b.sum()<1e-8: b=rng.random(n)+0.1
       # normalize max=1
       cb=fft_conv(b,N); mb=cb.max(); if mb>0: b=b/np.sqrt(mb)
       a=b
       # update target: blend current target with scaled c of a
       # c already zero padded; c_new=fft_conv(a,N)
       c_new=fft_conv(a,N); mn=c_new.max(); c_new=np.minimum(1.0,c_new/mn if mn>0 else 0)
       c=0.5*c + 0.5*c_new
       c=np.clip(c,0,1)
    # normalize a
    cb=fft_conv(a,N); mb=cb.max(); if mb>0: a=a/np.sqrt(mb)
    return a
```

```
def project_simplex(v,S):
    n=len(v); v=np.asarray(v,dtype=np.float64)
    if n==0: return v
    if S<=0: return np.zeros(n)
    w=np.sort(v)[::-1]
    cumsum=np.cumsum(w)
    idx=np.arange(1,n+1,dtype=np.float64)
    theta=(cumsum-S)/idx
    valid=theta <= w
    if np.any(valid):
       rho=int(np.max(np.where(valid)[0])+1)
    else: rho=n
    theta=(cumsum[rho-1]-S)/rho
    return np.maximum(v-theta,0.0)
```

```python
"""
Search for non-negative sequences minimizing
    2*n*max(conv(a,a)) / (sum(a))^2.

The code combines several的策略s:
- Fixed-sum L-BFGS on log-sum-exp smoothing of the peak convolution.
- Nesterov top-k subgradient polishing.
- Spectral square-root projection / random sign restarts to jump basins.
- A structured p-init based on the infinite sequence (1-z)^(-1/2).
- Multi-scale search (8192 baseline, then larger lengths, typically 32768).

All helper functions are top level and do not use closures or lambdas.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def next_pow2(m):
    """Next power of two >= m (m positive integer)."""
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fft_conv(a, N):
    """Linear self-convolution a*a using zero-padded FFT. Returns length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if N < 2 * n - 1:
        N = next_pow2(2 * n - 1)
    Ar = rfft(a, N)
    c = irfft(Ar * Ar, N)
    return c[:2 * n - 1]


def fast_score(a, N):
    """Fast score using FFT. Scale invariant."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    n = a.size
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = fft_conv(a, N)
    cm = float(np.max(c))
    if not np.isfinite(cm) or cm < 0.0:
        cm = 0.0
    return float((2.0 * n * cm) / (s * s))


def consider_candidate(best_a, best_score, cand, N):
    """If cand is better, return copy and score."""
    cand = np.asarray(cand, dtype=np.float64)
    if cand.size == 0:
        return best_a, best_score
    sc = fast_score(cand, N)
    if sc < best_score:
        out = np.where(np.isfinite(cand), cand, 0.0)
        out = np.clip(out, 0.0, 1000.0).copy()
        return out, sc
    return best_a, best_score


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and scale to sum S."""
    a = np.asarray(a, dtype=np.float64)
    if a.size != n:
        a = np.resize(a, n)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(n, dtype=np.float64) * (S / float(n))
    return a * (S / s)


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x)=S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    if S <= 0:
        return np.zeros(n, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0]) + 1)
    else:
        rho = n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(seq, n):
    """Simple linear interpolation of a sequence to length n."""
    seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    if seq.size == 0:
        return np.ones(n, dtype=np.float64)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    if seq.size == n:
        return seq.copy()
    if seq.size == 1:
        return np.full(n, float(seq[0]), dtype=np.float64)
    old = np.linspace(0.0, 1.0, seq.size)
    new = np.linspace(0.0, 1.0, n)
    return np.interp(new, old, seq)


def p_init(n):
    """
    Coefficients of (1-z)^(-1/2) up to length n.
    Its infinite self-convolution is flat, giving a useful structured start.
    """
    p = np.empty(n, dtype=np.float64)
    p[0] = 1.0
    for k in range(1, n):
        p[k] = p[k - 1] * (2 * k - 1) / (2 * k)
    return p


def make_init_rand(n, rng):
    return rng.random(n, dtype=np.float64) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0.0, 1.0, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2.0 * np.pi * (5.0 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    i = np.arange(n)
    c = int(rng.integers(1, max(2, n)))
    phase = -np.pi * i * (2 * i + 1 - c) / max(1, n)
    a = 1.0 + 0.7 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_binary_like(n, rng):
    a = (rng.random(n) > 0.4).astype(np.float64) * (0.5 + 0.5 * rng.random(n))
    return np.clip(a, 0.0, None)


def random_spectral_candidate(n, N, rng, mode="ones", iters=12):
    """
    Generate a candidate by alternating between:
      - clipping a target convolution c to [0,1]
      - taking a square root in frequency, with random/aligned signs
      - clipping inverse values to be non-negative
    This can jump to basins better than local peak minimization from p_init.
    """
    c = np.zeros(N, dtype=np.float64)
    L = 2 * n - 1
    if mode == "ones":
        c[:L] = 1.0
    elif mode == "p":
        p = p_init(n)
        cp = fft_conv(p, N)
        mp = float(cp.max())
        if mp > 1e-14:
            c[:L] = np.minimum(1.0, cp / mp)
        else:
            c[:L] = 1.0
    elif mode == "random":
        c[:L] = rng.uniform(0.45, 1.0, size=L)
    else:
        c[:L] = 1.0

    a = np.maximum(rng.standard_normal(n), 0.0) + 0.1
    for it in range(max(1, int(iters))):
        # Gently fill slack in the target convex coefficients.
        slack = 1.0 - c[:L]
        if np.any(slack > 1e-12):
            add = rng.uniform(0.0, 0.25, size=L) * slack
            c[:L] = np.minimum(1.0, c[:L] + add)

        C = rfft(c, N)
        A = rfft(a, N)
        # Stable square root.
        work = C.copy()
        work = np.where(np.isfinite(work), work, 0.0 + 0.0j)
        svals = np.sqrt(work)
        scores = np.real(svals * np.conj(A))
        scores = np.where(np.isfinite(scores), scores, 0.0)
        signs = np.where(scores >= 0.0, 1.0, -1.0)

        # Random flips are essential for global exploration.
        flip = rng.random(len(signs)) < 0.25
        signs[flip] *= -1.0
        signs[0] = 1.0

        B = signs * svals
        B[0] = abs(B[0])
        if N % 2 == 0:
            B[-1] = complex(signs[-1] * np.sqrt(abs(float(np.real(C[-1])))), 0.0)

        b = irfft(B, N)[:n]
        b = np.where(np.isfinite(b), b, 0.0)
        b = np.maximum(b, 0.0)
        if b.sum() < 1e-10:
            b = np.ones(n, dtype=np.float64)

        cb = fft_conv(b, N)
        mb = float(cb.max())
        if mb > 1e-12:
            b = b / np.sqrt(mb)
        a = b

        # Update target as a blend of an ideal bounded spectrum and current a*a.
        cn = fft_conv(a, N)
        mn = float(cn.max())
        if mn > 1e-14:
            cn = np.minimum(1.0, cn / mn)
        else:
            cn[:L] = 1.0
        c = 0.5 * c + 0.5 * cn
        c = np.clip(c, 0.0, 1.0)

    cb = fft_conv(a, N)
    mb = float(cb.max())
    if mb > 1e-12:
        a = a / np.sqrt(mb)
    return a


def spectral_projection_block(a, N, max_time, rng, random_flip_prob=0.0, blend=1.0):
    """
    Iterate a = Proj_{>=0}( sqrt(Proj_{c<=1}(a*a)) ), with optional frequency-level
    sign randomization. Works in the scale-invariant max=1 normalization.
    """
    n = len(a)
    a = safe_normalize(a, n, 1.0)
    c = fft_conv(a, N)
    cm = float(c.max())
    if cm > 1e-12:
        a = a / np.sqrt(cm)
    best_a = a.copy()
    best_score = fast_score(a, N)

    start = time.monotonic()
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_conv(a, N)
        cm = float(np.max(c))
        s = float(a.sum())
        if cm > 1e-14 and s > 0.0:
            sc = float((2.0 * n * cm) / (s * s))
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if cm > 1e-14:
            target = np.minimum(1.0, c / cm)
        else:
            target = np.zeros(N, dtype=np.float64)

        C = rfft(target, N)
        C = np.where(np.isfinite(C), C, 0.0 + 0.0j)
        A = rfft(a, N)
        svals = np.sqrt(C)
        scores = np.real(svals * np.conj(A))
        scores = np.where(np.isfinite(scores), scores, 0.0)
        signs = np.where(scores >= 0.0, 1.0, -1.0)

        if random_flip_prob > 0.0 and rng is not None:
            flip = rng.random(len(signs)) < random_flip_prob
            signs[flip] *= -1.0
        signs[0] = 1.0

        B = signs * svals
        B[0] = abs(B[0])
        if N % 2 == 0:
            B[-1] = complex(signs[-1] * np.sqrt(abs(float(np.real(C[-1])))), 0.0)

        b = irfft(B, N)[:n]
        b = np.where(np.isfinite(b), b, 0.0)
        b = np.maximum(b, 0.0)
        if blend < 1.0:
            b = (1.0 - blend) * a + blend * b
        if b.sum() < 1e-12:
            b = np.ones(n, dtype=np.float64)

        cb = fft_conv(b, N)
        mb = float(cb.max())
        if mb < 1e-12:
            b = np.ones(n, dtype=np.float64)
            mb = 1.0
        b = b / np.sqrt(mb)
        a = b

        sc = fast_score(a, N)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def spectral_phase_restart(a, N, rng, frac=0.5):
    """Randomize a fraction of FFT phases while preserving magnitudes."""
    n = len(a)
    S = float(a.sum())
    if S <= 1e-15:
        return a.copy()
    A = fft(a, N)
    mag = np.abs(A)
    if frac >= 1.0:
        phase = rng.uniform(0.0, 2.0 * np.pi, size=N)
    else:
        phase = np.angle(A)
        n_rand = max(1, int(N * max(0.01, frac)))
        idx = rng.choice(N, size=n_rand, replace=False)
        phase[idx] = rng.uniform(0.0, 2.0 * np.pi, size=n_rand)
    Anew = mag * np.exp(1j * phase)
    anew = np.real(ifft(Anew))[:n]
    anew = np.where(np.isfinite(anew), anew, 0.0)
    anew = np.maximum(anew, 0.0)
    s = float(anew.sum())
    if s < 1e-15:
        return a.copy()
    return anew * (S / s)


def lse_grad(a, N, beta):
    """
    Gradient of LSE-smoothed peak of conv(a,a).
    Returns (grad, lse_value, conv, conv_max).
    """
    n = len(a)
    c = fft_conv(a, N)
    cmax = float(np.max(c))
    beta = float(beta)
    if beta < 1e-6:
        weights = np.ones(len(c), dtype=np.float64) / float(len(c))
        lse_val = cmax
    else:
        exp_bc = np.exp(beta * (c - cmax))
        total = float(exp_bc.sum())
        if total < 1e-300:
            weights = np.zeros(len(c), dtype=np.float64)
            weights[int(np.argmax(c))] = 1.0
            lse_val = cmax
        else:
            weights = exp_bc / total
            lse_val = cmax + np.log(total) / beta

    fsize = 1 << (3 * n - 2).bit_length()
    Wr = rfft(weights, fsize)
    Ar = rfft(a, fsize)
    grad = 2.0 * irfft(Wr * np.conj(Ar), fsize)[:n]
    return grad, lse_val, c, cmax


def multi_beta_grad(a, N, betas):
    """Weighted combination of LSE gradients at several beta levels."""
    n = len(a)
    total_grad = np.zeros(n, dtype=np.float64)
    nb = len(betas)
    norm_w = float(nb * (nb + 1) / 2.0)
    for i, b in enumerate(betas):
        g, _, _, _ = lse_grad(a, N, b)
        w = (1.0 + i) / norm_w
        total_grad += w * g
    c = fft_conv(a, N)
    return total_grad, c, float(np.max(c))


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n, dtype=np.float64)
        return -g / gn

    q = g.copy()
    alphas = np.zeros(m, dtype=np.float64)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = float(np.dot(y_i, y_i))
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = float(np.dot(s_i, q)) / yy
            q = q - alphas[i] * y_i

    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = float(np.dot(y_last, y_last))
    if yy_last > 1e-30:
        gamma_use = max(1e-8, float(np.dot(s_last, y_last)) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q

    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = float(np.dot(y_i, y_i))
        if yy < 1e-30:
            continue
        beta_i = float(np.dot(s_i, y_i)) / yy
        r = r + (alphas[i] - beta_i) * y_i

    direction = -r
    if float(np.dot(g, direction)) > 0.0:
        direction = -g
    return direction


def lbfgs_lse_block(a_init, n, S, N, max_time, rng,
                    beta_init=5.0, beta_max=200.0, m_lbfgs=10,
                    use_multi_beta=False, deadline=None):
    """
    Fixed-sum L-BFGS on LSE-smoothed peak.
    Returns (best_array, best_peak) where best_peak is with sum normalized to S.
    """
    a = safe_normalize(a_init, n, S)
    beta = float(beta_init)
    best_a = a.copy()
    best_cm = np.inf
    s_hist = []
    y_hist = []

    if use_multi_beta:
        g, c, c_max = multi_beta_grad(a, N, [beta, beta * 3.0, beta * 9.0])
    else:
        g, _, c, c_max = lse_grad(a, N, beta)

    c_max = float(c_max)
    if c_max < best_cm:
        best_cm = c_max
        best_a = a.copy()

    start = time.monotonic()
    it = 0
    no_improve = 0
    step_scale = 2.0 * S / float(n)

    while time.monotonic() - start < max_time and (deadline is None or time.monotonic() < deadline):
        it += 1

        if it % 30 == 0:
            beta = min(float(beta_max), beta * 1.12)
            if use_multi_beta:
                g, c, c_max = multi_beta_grad(a, N, [beta, beta * 3.0, beta * 9.0])
            else:
                g, _, c, c_max = lse_grad(a, N, beta)
            c_max = float(c_max)
            if c_max < best_cm:
                best_cm = c_max
                best_a = a.copy()

        if len(s_hist) > m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_dir(s_hist, y_hist, g)
        dnorm = float(np.linalg.norm(direction))
        if dnorm < 1e-18:
            break

        alpha = max(1e-12, step_scale)
        best_cand = None
        best_c_ls = None
        best_cm_ls = np.inf
        last_ls = 0
        for ls in range(12):
            last_ls = ls
            cand = project_simplex(a + alpha * direction, S)
            c_ls = fft_conv(cand, N)
            cm_ls = float(np.max(c_ls))
            if cm_ls < best_cm_ls:
                best_cm_ls = cm_ls
                best_cand = cand
                best_c_ls = c_ls
            if cm_ls <= c_max * (1.0 + 1e-8) or ls == 11:
                break
            alpha *= 0.5

        accept = False
        if best_cand is not None:
            if best_cm_ls <= c_max * (1.0 + 1e-7):
                accept = True
            elif best_cm_ls < best_cm - 1e-12 and last_ls == 11:
                accept = True

        if not accept:
            no_improve += 1
            if no_improve > 180:
                break
            continue

        a_new = best_cand
        if best_cm_ls < best_cm:
            best_cm = best_cm_ls
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if use_multi_beta:
            g_new, c_new, cm_new = multi_beta_grad(a_new, N, [beta, beta * 3.0, beta * 9.0])
        else:
            g_new, _, c_new, cm_new = lse_grad(a_new, N, beta)

        s_k = a_new - a
        y_k = g_new - g
        sy = float(np.dot(s_k, y_k))
        ss = float(np.dot(s_k, s_k))
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c_max = float(cm_new)

        if c_max < best_cm:
            best_cm = c_max
            best_a = a.copy()

    return best_a, best_cm


def nesterov_toppk_block(a_init, n, S, N, max_time, rng,
                         top_k_init=20, top_k_max=512, deadline=None):
    """Nesterov accelerated top-k subgradient polishing."""
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    c0 = fft_conv(a, N)
    best_score = float((2.0 * n * np.max(c0)) / (S * S))

    start = time.monotonic()
    it = 0
    fsize = 1 << (3 * n - 2).bit_length()

    while time.monotonic() - start < max_time and (deadline is None or time.monotonic() < deadline):
        it += 1

        cy = fft_conv(y, N)
        top_k = min(top_k_max, top_k_init + it // 50)
        if top_k >= len(cy):
            idx = np.arange(len(cy))
        else:
            idx = np.argpartition(cy, -top_k)[-top_k:]
        vals = cy[idx]
        tot = float(np.sum(vals))
        if tot <= 1e-300:
            break

        weights_full = np.zeros(fsize, dtype=np.float64)
        for pos, k in enumerate(idx):
            weights_full[int(k)] = vals[pos]
        weights_full *= (1.0 / tot)

        Wr = rfft(weights_full, fsize)
        Ar = rfft(y, fsize)
        grad = 2.0 * irfft(Wr * np.conj(Ar), fsize)[:n]
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-16:
            break

        step = (2.0 * S / float(n)) / (1.0 + 0.003 * it)
        x_new = project_simplex(y - step * (grad / gnorm), S)

        if it % 5 == 0:
            cx = fft_conv(x_new, N)
            sc = float((2.0 * n * np.max(cx)) / (S * S))
            if sc < best_score:
                best_score = sc
                best_a = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 1200 == 0 and rng.random() < 0.12:
            y = spectral_phase_restart(safe_normalize(a, n, S), N, rng, frac=0.2)
            t = 1.0

    return best_a, best_score


def run_search_for_n(n, budget, rng, given=None, focus_array=None, deadline=None):
    """
    Run a full search for fixed length n.
    Returns (best_array, best_score).
    """
    n = int(n)
    if n <= 0 or budget <= 0.0:
        return None, np.inf

    S = np.sqrt(2.0 * float(n))
    N = next_pow2(2 * n - 1)
    start = time.monotonic()
    end = start + float(budget)

    best_a = None
    best_score = np.inf

    def remaining_limit():
        # This helper would be a closure; instead inline checks below.
        pass

    inits = []

    if focus_array is not None:
        fa = np.asarray(focus_array, dtype=np.float64).reshape(-1)
        if fa.size > 0:
            if fa.size == n:
                inits.append(fa)
            else:
                inits.append(interpolate_sequence(fa, n))

    if given is not None:
        gv = np.asarray(given, dtype=np.float64).reshape(-1)
        if gv.size > 0:
            if gv.size == n:
                inits.append(gv)
            elif gv.size > 100:
                inits.append(interpolate_sequence(gv, n))

    inits.append(p_init(n))
    inits.append(make_init_rand(n, rng))

    if budget > 250.0:
        inits.append(make_init_chirp(n, rng))
    if budget > 400.0:
        inits.append(make_init_zadoff(n, rng))
    if budget > 350.0:
        inits.append(make_init_binary_like(n, rng))
    if budget > 800.0:
        inits.append(random_spectral_candidate(n, N, rng, mode="ones", iters=12))
    if budget > 1500.0:
        inits.append(random_spectral_candidate(n, N, rng, mode="p", iters=10))

    # Phase 1: multi-start local polish.
    if budget < 200.0:
        phase1_budget = max(10.0, budget * 0.55)
    else:
        phase1_budget = min(budget * 0.35, 450.0)
    per_init = max(8.0, min(90.0, phase1_budget / max(1, len(inits))))

    for ai, arr in enumerate(inits):
        if deadline is not None:
            tl = deadline - time.monotonic()
        else:
            tl = end - time.monotonic()
        if tl <= 20.0:
            break

        t0 = time.monotonic()
        rem = max(2.0, min(per_init - (time.monotonic() - t0), tl - 5.0))
        if rem <= 2.0:
            continue

        a = safe_normalize(arr, n, S)

        t_sp = 0.0
        if rem > 30.0:
            t_sp = min(15.0, rem * 0.22)
            a_sp, _ = spectral_projection_block(a, N, t_sp, rng, random_flip_prob=0.05)
            best_a, best_score = consider_candidate(best_a, best_score, a_sp, N)
            a = a_sp

        t_lb = max(5.0, rem - t_sp - 1.0)
        a_lb, _ = lbfgs_lse_block(
            a, n, S, N, t_lb, rng,
            beta_init=5.0, beta_max=200.0, m_lbfgs=8,
            use_multi_beta=False, deadline=deadline
        )
        best_a, best_score = consider_candidate(best_a, best_score, a_lb, N)

        if ai % 2 == 0 or best_score < 1.51:
            print("[n=%d] phase1 init %d: best=%.6f" % (n, ai, best_score))

    if best_a is None:
        best_a = p_init(n)
        best_score = fast_score(best_a, N)

    print("[n=%d] after phase1: best=%.6f" % (n, best_score))

    # Phase 2: focused search with phase restarts.
    final_reserve = max(40.0, min(350.0, budget * 0.2))
    a_curr = safe_normalize(best_a, n, S)
    stall = 0
    use_mb = False

    while True:
        if deadline is not None:
            tl = deadline - time.monotonic()
        else:
            tl = end - time.monotonic()
        if tl <= final_reserve + 30.0:
            break
        block_time = min(120.0, tl - final_reserve - 30.0)
        if block_time < 10.0:
            break

        beta_start = 6.0 + 2.0 * min(stall, 8)
        beta_end = min(500.0, 30.0 * (1.0 + 0.3 * min(stall, 10)))
        prev_best = best_score

        a_new, _ = lbfgs_lse_block(
            a_curr, n, S, N, block_time, rng,
            beta_init=beta_start, beta_max=beta_end, m_lbfgs=12,
            use_multi_beta=use_mb, deadline=deadline
        )
        best_a, best_score = consider_candidate(best_a, best_score, a_new, N)

        if best_score < prev_best - 1e-11:
            stall = 0
            a_curr = safe_normalize(a_new, n, S)
            print("[n=%d] phase2 improved: %.6f" % (n, best_score))
        else:
            stall += 1
            if stall == 2:
                a_curr = spectral_phase_restart(safe_normalize(best_a, n, S), N, rng, 0.4)
                use_mb = not use_mb
                stall = 1
            elif stall >= 4:
                a_curr = spectral_phase_restart(safe_normalize(best_a, n, S), N, rng, 1.0)
                use_mb = not use_mb
                stall = 0
            else:
                a_curr = safe_normalize(a_new, n, S)

    # Phase 3: Nesterov top-k polishing.
    if deadline is not None:
        tl = deadline - time.monotonic()
    else:
        tl = end - time.monotonic()
    tn = min(220.0, final_reserve * 0.55, tl - 20.0)
    if tn > 12.0:
        a_n, _ = nesterov_toppk_block(
            best_a, n, S, N, tn, rng,
            top_k_init=30, top_k_max=500, deadline=deadline
        )
        best_a, best_score = consider_candidate(best_a, best_score, a_n, N)
        print("[n=%d] after nesterov: %.6f" % (n, best_score))

    # Phase 4: final high-beta LBFGS.
    if deadline is not None:
        tl = deadline - time.monotonic()
    else:
        tl = end - time.monotonic()
    tf = min(240.0, tl - 15.0)
    if tf > 15.0:
        a_f, _ = lbfgs_lse_block(
            best_a, n, S, N, tf, rng,
            beta_init=35.0, beta_max=900.0, m_lbfgs=15,
            use_multi_beta=False, deadline=deadline
        )
        best_a, best_score = consider_candidate(best_a, best_score, a_f, N)
        print("[n=%d] final: %.6f" % (n, best_score))

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    """
    t0 = time.monotonic()
    hard_end = t0 + max(0.0, float(budget_s) - 10.0)
    search_deadline = max(t0, hard_end - 130.0)

    rng = np.random.default_rng(seed)

    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == 0:
            given = None
    except Exception:
        given = None

    best_a = None
    best_score = np.inf

    # Baseline on the unavailable/given length, preferably 8192.
    if given is not None:
        base_n = 8192
        if 1024 <= len(given) <= 32768:
            # Use a power of two close to the given length for good FFT behavior.
            base_n = min(16384, max(8192, 1 << (len(given) - 1).bit_length()))
            if base_n > 2 * len(given):
                base_n = 8192
        if search_deadline - time.monotonic() > 300.0:
            base_budget = min(350.0, (search_deadline - time.monotonic()) * 0.22)
            a8, sc8 = run_search_for_n(base_n, base_budget, rng, given=given,
                                       focus_array=given, deadline=search_deadline)
            if a8 is not None:
                N8 = next_pow2(2 * len(a8) - 1)
                best_a, best_score = consider_candidate(best_a, best_score, a8, N8)

    # Main larger-length search. Starting from the best found so far is usually strong.
    remaining = search_deadline - time.monotonic()
    if remaining > 60.0:
        if remaining > 1600.0:
            n_main = 32768
        elif remaining > 900.0:
            n_main = 16384
        else:
            n_main = 8192

        focus = None
        if best_a is not None:
            focus = interpolate_sequence(best_a, n_main)
        elif given is not None:
            focus = interpolate_sequence(given, n_main)

        main_budget = min(remaining * 0.72, 2000.0)
        am, scm = run_search_for_n(n_main, main_budget, rng, given=given,
                                   focus_array=focus, deadline=search_deadline)
        if am is not None:
            Nm = next_pow2(2 * len(am) - 1)
            best_a, best_score = consider_candidate(best_a, best_score, am, Nm)

    # Secondary: if the baseline length remains best, try an intermediate length.
    remaining = search_deadline - time.monotonic()
    if remaining > 320.0 and best_a is not None:
        if len(best_a) < 16384:
            n2 = 16384
        elif len(best_a) < 32768:
            n2 = 32768
        else:
            n2 = len(best_a)

        focus2 = None
        if len(best_a) == n2:
            focus2 = best_a
        else:
            focus2 = interpolate_sequence(best_a, n2)

        sec_budget = min(remaining * 0.65, 900.0)
        a2, sc2 = run_search_for_n(n2, sec_budget, rng, given=given,
                                   focus_array=focus2, deadline=search_deadline)
        if a2 is not None:
            N2 = next_pow2(2 * len(a2) - 1)
            best_a, best_score = consider_candidate(best_a, best_score, a2, N2)

    # Final direct polish of the best length.
    remaining = search_deadline - time.monotonic()
    if best_a is not None and remaining > 80.0:
        nb = len(best_a)
        Sb = np.sqrt(2.0 * float(nb))
        Nb = next_pow2(2 * nb - 1)
        ab, _ = lbfgs_lse_block(
            best_a, nb, Sb, Nb, min(remaining - 15.0, 300.0), rng,
            beta_init=30.0, beta_max=1000.0, m_lbfgs=15,
            use_multi_beta=False, deadline=search_deadline
        )
        best_a, best_score = consider_candidate(best_a, best_score, ab, Nb)

    # Sanitize final output.
    if best_a is None:
        best_a = p_init(2048)
        best_score = fast_score(best_a, next_pow2(2 * len(best_a) - 1))

    best_a = np.asarray(best_a, dtype=np.float64).reshape(-1)
    best_a = np.where(np.isfinite(best_a), best_a, 0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.size == 0 or best_a.sum() < 0.01:
        best_a = np.ones(256, dtype=np.float64)

    # Verify if safe within time.
    Nv = next_pow2(2 * len(best_a) - 1)
    internal_score = fast_score(best_a, Nv)
    print("internal score: %.6f length=%d" % (internal_score, len(best_a)))

    # Only call evaluate_sequence if it is likely to finish safely.
    time_left = hard_end - time.monotonic()
    if time_left > 0.0:
        try:
            if len(best_a) <= 16384 or time_left > 240.0:
                final_list = [float(x) for x in best_a]
                ver = evaluate_sequence(final_list)
                if np.isfinite(ver):
                    best_score = float(ver)
                    print("verified evaluate_sequence: %.6f" % best_score)
                else:
                    print("verify rejected, fallback to safe small sequence")
                    best_a = p_init(2048)
            else:
                print("skipping verify to preserve time")
        except Exception:
            print("verify unavailable/failed")

    return [float(x) for x in best_a]
```
