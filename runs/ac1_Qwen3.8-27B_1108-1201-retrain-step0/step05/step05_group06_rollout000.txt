"""
Minimize 2*n*max(self_convolution)/sum^2 using LSE-FISTA with annealing.
Key improvements over previous:
- Backtracking line search for LSE objective

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value(y, c, beta):
    """Compute LSE value: beta * log(sum(exp(c/beta)))."""
    cm = c.max()
    shifted = c / beta - cm / beta
    return cm + beta * np.log(np.exp(shifted).sum())


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = c / beta - cm / beta
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return np.real(ifft(W * np.conj(A)))[:n] * 2.0


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_lse_fista(a_init, max_time, rng, beta_schedule=None, 
                  use_backtrack=False, perturbation_interval=500,
                  perturbation_scale=0.02):
    """
    LSE-FISTA with adaptive beta and optional perturbations.
    beta_schedule: list of (beta, duration_fraction) pairs
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    if beta_schedule is None:
        beta_schedule = [(10.0, 0.2), (50.0, 0.3), (200.0, 0.3), (1000.0, 0.2)]
    
    total_beta_time = sum(d for _, d in beta_schedule)
    elapsed_frac = 0.0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Determine current beta
        elapsed_frac = (time.monotonic() - start) / max_time
        cum_frac = 0.0
        beta = beta_schedule[-1][0]
        for b, d in beta_schedule:
            cum_frac += d / total_beta_time
            if elapsed_frac < cum_frac:
                beta = b
                break
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Compute gradient
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Step size
        if use_backtrack:
            # Try a large step first
            step = min(2.0 / (beta * np.max(a) + 1.0), S / (gnorm * 2))
            step = max(step, 1e-10)
        else:
            step = 0.5 / (beta * np.max(a) + 2.0 * np.max(a) + 1.0)
            step = max(step, 1e-10)
        
        x_new = project_simplex(y - step * grad, S)
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Perturbation logic
        if stuck > perturbation_interval and it % (perturbation_interval // 10) == 0:
            # Structured perturbation: add a sinusoid
            m = rng.integers(1, n // 4)
            phase = rng.uniform(0, 2 * np.pi)
            j = np.arange(n, dtype=np.float64)
            sin_mode = np.cos(2 * np.pi * m * j / n + phase)
            sin_mode = sin_mode - sin_mode.mean()  # zero mean
            scale = perturbation_scale * S / np.sqrt(n) * np.exp(-stuck / (3 * perturbation_interval))
            if scale > 1e-8:
                y = project_simplex(y + scale * sin_mode, S)
                t_mom = 1.0
                a = y.copy()
                stuck = 0
                last_improve = it
        
        # Periodic momentum restart
        if it % 10000 == 0:
            t_mom = 1.0
        
        # If very stuck, do a bigger jump
        if stuck > 5 * perturbation_interval:
            noise_scale = 0.05 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            a = y.copy()
            stuck = 0
            last_improve = it
    
    return best_a, best_obj


def run_active_set_eq(a_init, max_time, rng):

    """

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        if stuck < 50:
            k = 1
        elif stuck < 200:
            k = 3
        elif stuck < 500:
            k = 10
        elif stuck < 2000:
            k = 30
        elif stuck < 10000:
            k = 100
        else:
            k = min(500, 100 + (stuck - 10000) // 100)

        k_actual = min(k, len(c))
        top_idx = np.argpartition(c, -k_actual)[-k_actual:]
        top_vals = c[top_idx]
        
        # Weight: emphasize the largest
        if k_actual == 1:
            w = np.zeros(len(c))
            w[top_idx[0]] = 1.0
        else:
            # Exponential weighting
            max_top = top_vals.max()
            min_top = top_vals.min()
            if max_top - min_top < 1e-15:
                w = np.zeros(len(c))
                w[top_idx] = 1.0 / k_actual
            else:
                w = np.zeros(len(c))
                w[top_idx] = np.exp(5.0 * (top_vals - max_top) / (min_top - max_top + 1e-30))
                w[top_idx] /= w[top_idx].sum()
        
        # Compute weighted gradient
        fsize = 1 << (len(w) + n - 1).bit_length()

        A = fft(y, fsize)
