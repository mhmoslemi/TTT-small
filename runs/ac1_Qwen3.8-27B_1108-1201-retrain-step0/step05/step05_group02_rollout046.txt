"""

Strategy:

3. LP-guided direction finding (novel: solve LP for optimal mass redistribution)

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (s * s))


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, beta, c_prev=None):
    n = len(y)
    if c_prev is None:
        c = fft_convolve_self(y)
    else:
        c = c_prev
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def run_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.01,
              beta_max=10000.0, step_scale=1.0, use_backtrack=False):
    n = len(a_init)
    S = 1.0  # Use unit sum for numerical stability
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_gradient(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max  # since S=1
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-14 and bt < 5:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            x_new = x_trial
        else:
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        a = x_new
        t_mom = t_new

        if it - last_improve > 3000:
            t_mom = 1.0
            noise_scale = max(1e-7, 0.002 / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    return best_a, best_obj


def solve_lp_direction(a, c, M, S, n_tight=200, time_limit=30.0):
    """
    Solve the LP to find the best direction to move.
    
    LP: minimize t
    subject to: 2*(a*e)_conv[k] - t <= c[k] for k in T
                e >= 0
                sum(e) = S
    
    Returns: e (the new sequence) or None if LP fails
    """
    n = len(a)
    
    # Identify tight lags
    thresholds = np.sort(c)[::-1][:n_tight]
    T = np.where(c >= thresholds[0] - 1e-10)[0]
    T = T[:n_tight]  # Limit to n_tight
    
    # Build constraint matrix
    # For each k in T: 2*sum_i a[k-i]*e[i] - t <= c[k]
    # Variables: e[0],...,e[n-1], t
    # So: [2*A_k, -1] * [e; t] <= c[k]
    
    # Also: e[j] >= 0 => -e[j] <= 0 (n constraints)
    # sum(e) = S (1 equality)
    
    A_rows = []
    b_vals = []
    
    for k in T:
        row = np.zeros(n + 1)
        # 2*(a*e)_conv[k] = 2*sum_{i: 0<=k-i<n} a[k-i]*e[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            row[lo:hi+1] = 2.0 * a[k - hi:k - lo + 1][::-1]
        row[n] = -1.0  # coefficient of t
        A_rows.append(row)
        b_vals.append(c[k])
    
    # Non-negativity: -e[j] <= 0
    for j in range(n):
        row = np.zeros(n + 1)
        row[j] = -1.0
        A_rows.append(row)
        b_vals.append(0.0)
    
    A = np.array(A_rows)
    b = np.array(b_vals)
    
    # Objective: minimize t = e[n]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Equality: sum(e) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: e[j] >= 0, t unbounded below (but will be >= 0 in practice)
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'time_limit': time_limit, 'presolve': True})
        if result.success:
            e = result.x[:n]
            t_val = result.x[n]
            return e, t_val
    except Exception as ex:
        print("  LP error: %s" % str(ex))
    
    return None, None


def lp_refine(a_init, max_time, rng, n_tight=200):
    """LP-guided refinement."""
    n = len(a_init)
    S = 1.0
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = float(2.0 * n * c.max())
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()
        cur_obj = float(2.0 * n * M)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
            stuck_count = 0
        
        stuck_count += 1
        if stuck_count > 30:
            break
        
        # Solve LP
        e, t_pred = solve_lp_direction(a, c, M, S, n_tight=n_tight)
        
        if e is not None:
            # Check actual improvement
            e_sum = e.sum()
            if e_sum > 1e-10:
                e = e / e_sum
            c_e = fft_convolve_self(e)
            obj_e = float(2.0 * n * c_e.max())
            
            if obj_e < cur_obj - 1e-12:
                a = e
                continue
            else:
                # Try scaling: alpha in (0, 1)
                imp = False
                for frac in [0.5, 0.3, 0.1]:
                    a_trial = (1 - frac) * a + frac * e
                    s_t = a_trial.sum()
                    if s_t > 1e-10:
                        a_trial = a_trial / s_t
                    c_t = fft_convolve_self(a_trial)
                    obj_t = float(2.0 * n * c_t.max())
                    if obj_t < cur_obj - 1e-12:
                        a = a_trial
                        imp = True
                        break
                if not imp:
                    stuck_count += 5
        else:
            stuck_count += 5
    
    return best_a, best_obj
