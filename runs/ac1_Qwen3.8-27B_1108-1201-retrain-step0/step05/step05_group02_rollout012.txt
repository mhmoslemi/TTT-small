"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Multi-scale spectral optimization with LSE-FISTA and targeted coordinate descent.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve(a):
    n = len(a)
    size = 2*n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * c.max() / (s*s)

def lse_smooth(a, beta):
    """Compute LSE of convolution and its gradient."""
    n = len(a)
    c = fft_convolve(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = e / tot
    N = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + np.log(tot) / beta
    return lse_val, grad, c, w

def fista_lse(a_init, max_time, rng, beta_init=50.0, beta_max=50000.0, beta_growth=0.005):
    """FISTA with LSE smoothing and adaptive beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(a_init.astype(np.float64), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve_it = 0
    it = 0
    step = 1.0 / max(beta_init, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_growth * it))
        lse_val, grad, c, w = lse_smooth(y, beta)
        cur_obj = 2.0 * n * c.max() / (S*S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve_it = it
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Backtracking line search
        step_l = step
        x_new = project_simplex(y - step_l * grad, S)
        obj_new = compute_obj(x_new)
        bt = 0
        while obj_new > cur_obj + 1e-10 and step_l > 1e-15 and bt < 8:
            step_l *= 0.5
            x_new = project_simplex(y - step_l * grad, S)
            obj_new = compute_obj(x_new)
            bt += 1
        
        if bt > 3:
            step = step_l * 0.7
        else:
            step = min(step * 1.05, 3.0 / beta)
        
        # FISTY momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0*t*t))
        y = x_new + ((t-1.0)/t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t = t_new
        
        # Restart with noise if stuck

        if stuck > 3000:

            noise = rng.normal(0, max(1e-6, 0.002 * S / np.sqrt(n) * np.exp(-stuck/8000)), n)
            y = project_simplex(y + noise, S)
    
    return best_a, best_obj

"""
Multi-scale LSE-FISTA optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy: Polish starting sequence, upsample to larger n, then run progressively
higher-beta FISTA with backtracking line search. Uses targeted subgradient steps
and coordinate descent for final refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(a, beta):
    """Compute LSE smoothing value and gradient of the convolution peak."""
    n = len(a)
    c = fft_convolve(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / tot
    N = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + np.log(tot) / beta
    return lse_val, grad, c, w


def topk_gradient(a, c, k):
    """Subgradient using top-k peak locations of the convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    N = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, N)
    A = fft(a, N)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def fista_lse(a_init, max_time, rng, beta_init=50.0, beta_max=100000.0,
              beta_growth=0.005, step_init=None, use_bt=True):
    """FISTA with LSE smoothing, adaptive beta, and backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    if step_init is not None:
        step = step_init
    else:
        step = 1.0 / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_growth * it))
        lse_val, grad, c, w = lse_gradient(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gn = np.linalg.norm(grad)
        if gn < 1e-16:
            break

        if use_bt:
            step_l = step
            x_new = project_simplex(y - step_l * grad, S)
            obj_try = compute_obj(x_new)
            bt = 0
            while obj_try > cur_obj + 1e-11 and step_l > 1e-16 and bt < 10:
                step_l *= 0.5
                x_new = project_simplex(y - step_l * grad, S)
                obj_try = compute_obj(x_new)
                bt += 1
            if bt > 4:
                step = step_l * 0.6
            else:
                step = min(step * 1.04, 4.0 / beta)
        else:
            step = 2.0 / max(beta, 1.0)
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        stuck = it - last_improve
        if stuck > 4000:
            t_mom = 1.0
            ns = max(1e-7, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 10000))
            y = project_simplex(y + rng.normal(0, ns, n), S)

    return best_a, best_obj


def coord_descent(a_init, max_time, rng, top_k=5):
    """
    Targeted coordinate descent: transfer mass from high-contribution
    to low-contribution coordinates to reduce peak convolution values.
    Uses first-order approximation for speed.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000)
    a = a * (S / a.sum())
    c = fft_convolve(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    recompute_counter = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        if it - last_improve > 20000:
            break

        # Compute multi-peak gradient
        peak_idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros(len(c))
        for k in peak_idx:
            w[int(k)] = 1.0 / top_k
        N = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, N)
        A = fft(a, N)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Score coordinates for source (give mass) and target (receive mass)
        src_score = grad * np.maximum(a, 1e-10)
        tgt_score = -grad * (a.mean() - a + 1e-10)

        n_try = min(30, n)
        src_cand = np.argsort(src_score)[::-1][:n_try]
        tgt_cand = np.argsort(tgt_score)[:n_try]

        best_delta_obj = cur_obj + 1e-15
        best_j, best_l, best_d = -1, -1, 0.0

        # Try candidate pairs with first-order approximation
        for j in src_cand[:12]:
            for l in tgt_cand[:12]:
                if j == l or a[j] < 1e-10:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    d = a[int(j)] * frac
                    if d < 1e-12:
                        continue
                    # First-order change in c: c[k] += -2*d*a[k-j] + 2*d*a[k-l]
                    # (valid where indices are in range)
                    delta_c = np.zeros(len(c))
                    # Contribution from removing d at position j
                    js = max(0, int(j))
                    je = min(2 * n - 2, int(j) + n - 1)
                    for kk in range(js, je + 1):
                        jk = kk - int(j)
                        if 0 <= jk < n:
                            delta_c[kk] -= 2.0 * d * a[jk]
                    # Contribution from adding d at position l
                    ls = max(0, int(l))
                    le = min(2 * n - 2, int(l) + n - 1)
                    for kk in range(ls, le + 1):
                        lk = kk - int(l)
                        if 0 <= lk < n:
                            delta_c[kk] += 2.0 * d * a[lk]

                    new_max = (c + delta_c).max()
                    new_obj = 2.0 * n * new_max / (S * S)
                    if new_obj < best_delta_obj:
                        best_delta_obj = new_obj
                        best_j, best_l, best_d = int(j), int(l), d

        if best_j >= 0:
            j, l, d = best_j, best_l, best_d
            a[j] -= d
            a[l] += d
            recompute_counter += 1
            if recompute_counter % 5 == 0:
                c = fft_convolve(a)
            else:
                # Incremental update (first-order)
                js = max(0, j)
                je = min(2 * n - 2, j + n - 1)
                for kk in range(js, je + 1):
                    jk = kk - j
                    if 0 <= jk < n:
                        c[kk] -= 2.0 * d * a[jk]
                ls = max(0, l)
                le = min(2 * n - 2, l + n - 1)
                for kk in range(ls, le + 1):
                    lk = kk - l
                    if 0 <= lk < n:
                        c[kk] += 2.0 * d * a[lk]
        else:
            # No improvement; perturbation
            if it - last_improve > 500:
                j = int(rng.integers(0, n))
                l = int(rng.integers(0, n))
                if j != l and a[j] > 1e-10:
                    d = a[j] * 0.01
                    a[j] -= d
                    a[l] += d
                    c = fft_convolve(a)

    return best_a, best_obj


def subgradient_refine(a_init, max_time, rng):
    """Refine using top-k subgradient with decreasing step size."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000)
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 2
        elif stuck < 2000:
            k = min(8, 2 + stuck // 200)
        else:
            k = min(32, 8 + stuck // 200)

        grad = topk_gradient(a, c, k)
        gn = np.linalg.norm(grad)
        if gn < 1e-16:
            break

        step = S / (gn * (1.0 + 0.005 * it))
        step = min(step, 3.0 * S / gn)

        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-12:
            a = a_new
        else:
            step *= 0.1
            a_new = project_simplex(a - step * grad, S)
            if compute_obj(a_new) <= cur_obj + 1e-12:
                a = a_new

        if stuck > 2000 and it % 100 == 0:
            ns = max(1e-7, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 5000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: polish, upsample, progressive LSE-FISTA, coordinate descent, refine.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load starting point
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        tt = np.linspace(0, 1, n, endpoint=False)
        a_start = 0.5 + 0.5 * np.cos(2 * np.pi * tt)

    n0 = len(a_start)
    S0 = np.sqrt(2.0 * n0)
    a_start = a_start * (S0 / a_start.sum())
    obj0 = compute_obj(a_start)
    print("Start: n=%d, obj=%.8f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: Quick polish at original n (5% time) ===
    rem = deadline - time.monotonic()
    t1 = rem * 0.05
    if t1 > 8:
        print("\nPhase 1: FISTA polish n=%d (%.0fs)" % (n0, t1))
        a1, obj1 = fista_lse(global_best_a, t1, rng,
                             beta_init=200, beta_max=50000, beta_growth=0.004)
        print("  obj=%.8f" % obj1)
        if obj1 < global_best_obj:
            global_best_obj = obj1
            global_best_a = a1.copy()

    # === Phase 2: Upsample to 2x and initial FISTA (15% time) ===
    rem = deadline - time.monotonic()
    t2 = rem * 0.15
    n2 = n0 * 2
    S2 = np.sqrt(2.0 * n2)
    a2_raw = upsample(global_best_a, 2)
    # Add small noise to help escape local minima at new scale
    noise = rng.normal(0, 0.001 * S2 / np.sqrt(n2), n2)
    a2 = np.maximum(a2_raw + noise, 0)
    a2 = a2 * (S2 / a2.sum())
    obj2_init = compute_obj(a2)
    print("\nPhase 2: Upsample n=%d, obj=%.8f (%.0fs)" % (n2, obj2_init, t2))
    if obj2_init < global_best_obj:
        global_best_obj = obj2_init
        global_best_a = a2.copy()

    a2a, obj2a = fista_lse(a2, t2, rng,
                           beta_init=5, beta_max=5000, beta_growth=0.015)
    print("  FISTA low-beta: obj=%.8f" % obj2a)
    if obj2a < global_best_obj:
        global_best_obj = obj2a
        global_best_a = a2a.copy()

    # === Phase 3: Medium-beta FISTA (25% time) ===
    rem = deadline - time.monotonic()
    t3 = rem * 0.25
    if t3 > 20:
        print("\nPhase 3: FISTA medium-beta (%.0fs)" % t3)
        a3, obj3 = fista_lse(global_best_a, t3, rng,
                             beta_init=100, beta_max=50000, beta_growth=0.006)
        print("  obj=%.8f" % obj3)
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: High-beta FISTA (25% time) ===
    rem = deadline - time.monotonic()
    t4 = rem * 0.25
    if t4 > 20:
        print("\nPhase 4: FISTA high-beta (%.0fs)" % t4)
        a4, obj4 = fista_lse(global_best_a, t4, rng,
                             beta_init=5000, beta_max=500000, beta_growth=0.002)
        print("  obj=%.8f" % obj4)
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Coordinate descent (15% time) ===
    rem = deadline - time.monotonic()
    t5 = rem * 0.15
    if t5 > 15:
        print("\nPhase 5: Coord descent (%.0fs)" % t5)
        a5, obj5 = coord_descent(global_best_a, t5, rng, top_k=5)
        print("  obj=%.8f" % obj5)
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        # Quick FISTA after CD
        rem5 = deadline - time.monotonic()
        if rem5 > 10:
            a5b, obj5b = fista_lse(global_best_a, rem5, rng,
                                   beta_init=10000, beta_max=1000000, beta_growth=0.001)
            print("  FISTA after CD: obj=%.8f" % obj5b)
            if obj5b < global_best_obj:
                global_best_obj = obj5b
                global_best_a = a5b.copy()

    # === Phase 6: Subgradient refinement (10% time) ===
    rem = deadline - time.monotonic()
    t6 = rem * 0.10
    if t6 > 10:
        print("\nPhase 6: Subgradient refine (%.0fs)" % t6)
        a6, obj6 = subgradient_refine(global_best_a, t6, rng)
        print("  obj=%.8f" % obj6)
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

    # === Phase 7: Try upsample to 4x if time permits ===
    rem = deadline - time.monotonic()
    if rem > 180 and len(global_best_a) <= n0 * 2:
        n4 = len(global_best_a) * 2
        S4 = np.sqrt(2.0 * n4)
        a4_raw = upsample(global_best_a, 2)
        noise4 = rng.normal(0, 0.001 * S4 / np.sqrt(n4), n4)
        a4_up = np.maximum(a4_raw + noise4, 0)
        a4_up = a4_up * (S4 / a4_up.sum())
        obj4_up = compute_obj(a4_up)
        print("\nPhase 7: Upsample n=%d, obj=%.8f (%.0fs)" % (n4, obj4_up, rem))

        a4_opt, obj4_opt = fista_lse(a4_up, rem * 0.5, rng,
                                     beta_init=5, beta_max=10000, beta_growth=0.01)
        print("  FISTA upsampled: obj=%.8f" % obj4_opt)
        if obj4_opt < global_best_obj:
            global_best_obj = obj4_opt
            global_best_a = a4_opt.copy()

        rem7 = deadline - time.monotonic()
        if rem7 > 20:
            a4_pol, obj4_pol = fista_lse(global_best_a, rem7, rng,
                                         beta_init=500, beta_max=100000, beta_growth=0.005)
            print("  Polish upsampled: obj=%.8f" % obj4_pol)
            if obj4_pol < global_best_obj:
                global_best_obj = obj4_pol
                global_best_a = a4_pol.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        print("\nFinal: n=%d, obj=%.8f" % (len(final_list), global_best_obj))
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]
        print("\nFallback: n=%d" % n_fb)

    return final_list
