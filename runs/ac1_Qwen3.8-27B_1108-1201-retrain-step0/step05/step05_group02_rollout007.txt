"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-beta LSE FISTA → L-BFGS hybrid → Structured peak-targeting → Polish.
Key innovations: multi-scale gradient (3 betas), superlinear L-BFGS convergence,
and deterministic structured restarts to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-30:
        return np.full(n, S / n)
    if abs(s - S) / S < 1e-12:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc < w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_and_grad(y, beta):
    """LSE value and gradient for single beta."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def multi_lse_grad(y, betas, weights=None):
    """Weighted sum of LSE gradients at multiple beta scales."""
    n = len(y)
    c = fft_convolve_self(y)
    if weights is None:
        weights = np.ones(len(betas)) / len(betas)
    total_grad = np.zeros(n)
    total_val = 0.0
    fsize = 1 << (2 * n - 1 + n - 1).bit_length()
    A = fft(y, fsize)
    for i, beta in enumerate(betas):
        cm = c.max()
        shifted = beta * (c - cm)
        shifted = shifted - shifted.max()
        e = np.exp(shifted)
        tot = e.sum()
        if tot < 1e-300:
            lse_val = cm
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            lse_val = cm + np.log(tot) / beta
            w = e / tot
        total_val += weights[i] * lse_val
        W = fft(w, fsize)
        g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        total_grad += weights[i] * g
    return total_val, total_grad, c


def run_multi_beta_fista(a_init, max_time, rng, betas=(20, 80, 300),
                         weights=(0.3, 0.4, 0.3), beta_growth=0.005):
    """FISTA with multi-beta LSE and adaptive growth."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 5.0 / max(betas)

    while time.monotonic() - start < max_time:
        if time.monotonic() >= start + max_time:
            break
        it += 1
        # Adaptively grow betas
        growth = np.exp(beta_growth * it)
        cur_betas = tuple(min(5000.0, b * growth) for b in betas)

        lse_val, grad, c = multi_lse_grad(y, cur_betas, weights)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Backtracking line search
        step_local = step
        x_trial = project_simplex(y - step_local * grad, S)
        obj_trial = compute_obj(x_trial)
        bt = 0
        while obj_trial > cur_obj + 1e-13 and step_local > 1e-14 and bt < 8:
            step_local *= 0.5
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt += 1
        if bt > 3:
            step = step_local * 0.7
        else:
            step = min(step * 1.02, 10.0 / max(cur_betas[0], 1.0))
        x_new = x_trial

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        # Restart momentum if stuck
        if it - last_improve > 3000:
            t_mom = 1.0
            noise = max(1e-7, 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 3000.0))
            y = project_simplex(y + rng.normal(0, noise, n), S)

    return best_a, best_obj


def lbfgs_hybrid(a_init, max_time, rng, m_store=30, beta=80.0, l2_weight=0.005):
    """
    L-BFGS on hybrid objective: LSE(c; beta) + l2_weight * mean((c - m)^2).
    This gives superlinear convergence for the final refinement.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()

    # L-BFGS storage
    s_store = []
    y_store = []
    rho_store = []

    def objective_and_grad(x):
        """Compute hybrid objective and its gradient."""
        c = fft_convolve_self(x)
        cm = c.max()
        shifted = beta * (c - cm)
        shifted = shifted - shifted.max()
        e = np.exp(shifted)
        tot = e.sum()
        if tot < 1e-300:
            lse_val = cm
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            lse_val = cm + np.log(tot) / beta
            w = e / tot

        # L2 term
        m_val = (x.sum() ** 2) / (2 * n - 1)
        l2_term = l2_weight * np.mean((c - m_val) ** 2)

        # Gradient
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(x, fsize)
        grad_lse = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # L2 gradient: 2 * l2_weight * (2/(2n-1)) * convolve(c - m_val, x)
        e_vec = c - m_val
        fsize2 = 1 << (2 * n - 1 + n).bit_length()
        E = fft(e_vec, fsize2)
        grad_l2 = (4.0 * l2_weight / (2 * n - 1)) * np.real(ifft(E * np.conj(fft(x, fsize2))))[:n]

        total_obj = lse_val + l2_term
        total_grad = grad_lse + grad_l2
        return total_obj, total_grad

    # L-BFGS two-loop recursion
    def lbfgs_direction(x, g, m=m_store):
        """Compute search direction using L-BFGS."""
        k = min(len(s_store), m)
        q = g.copy()
        alphas = np.zeros(k)
        betas = np.zeros(k)
        for i in range(k - 1, -1, -1):
            si = s_store[i]
            yi = y_store[i]
            rhoi = rho_store[i]
            alphas[i] = rhoi * np.dot(si, q)
            q -= alphas[i] * yi
        # Initial Hessian approximation
        yy = np.dot(y_store[-1], y_store[-1]) if k > 0 else 0
        sy = np.dot(s_store[-1], y_store[-1]) if k > 0 else 1
        gamma = sy / yy if yy > 1e-30 else 1.0
        r = gamma * q
        for i in range(k):
            si = s_store[i]
            yi = y_store[i]
            rhoi = rho_store[i]
            betas[i] = rhoi * np.dot(yi, r)
            r += si * (alphas[i] - betas[i])
        return -r

    a_curr = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        f_curr, g_curr = objective_and_grad(a_curr)
        c_curr = fft_convolve_self(a_curr)
        cur_obj = 2.0 * n * c_curr.max() / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a_curr.copy()
            last_improve = it

        gnorm = np.linalg.norm(g_curr)
        if gnorm < 1e-14:
            break

        # Compute L-BFGS direction
        d = lbfgs_direction(a_curr, g_curr)
        # Ensure descent direction
        if np.dot(d, g_curr) > 0:
            d = -g_curr

        # Backtracking line search with Armijo condition
        step = 1.0
        c1 = 1e-4
        dg = np.dot(g_curr, d)
        a_trial = project_simplex(a_curr + step * d, S)
        f_trial, _ = objective_and_grad(a_trial)
        ls_iter = 0
        while f_trial > f_curr + c1 * step * dg and step > 1e-14 and ls_iter < 30:
            step *= 0.5
            a_trial = project_simplex(a_curr + step * d, S)
            f_trial, _ = objective_and_grad(a_trial)
            ls_iter += 1

        a_new = a_trial
        if step < 1e-14:
            # Can't take any step; try steepest descent
            step_sd = S / (gnorm * gnorm) * 0.5
            a_new = project_simplex(a_curr - step_sd * g_curr, S)
            f_new, _ = objective_and_grad(a_new)
        else:
            # Update L-BFGS storage
            s_vec = a_new - a_curr
            g_new, _ = objective_and_grad(a_new)
            y_vec = g_new - g_curr
            sy = np.dot(s_vec, y_vec)
            if sy > 1e-14:
                s_store.append(s_vec)
                y_store.append(y_vec)
                rho_store.append(1.0 / sy)
                if len(s_store) > m_store:
                    s_store.pop(0)
                    y_store.pop(0)
                    rho_store.pop(0)
            f_new = f_trial

        a_curr = a_new

        # Periodic restart
        if it - last_improve > 2000:
            s_store.clear()
            y_store.clear()
            rho_store.clear()
            noise = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 4000))
            a_curr = project_simplex(a_curr + rng.normal(0, noise, n), S)

    return best_a, best_obj


def structured_restart(a, rng, n_peaks=5, frac=0.02):
    """
    Structured perturbation: move mass from peak-contributing to peak-avoiding coordinates.
    """
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    top_idx = np.argsort(c)[::-1][:n_peaks]

    a_new = a.copy()
    for k in top_idx:
        # Coordinates that contribute to c[k]
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        if hi < lo:
            continue
        contrib = a[lo:hi + 1].copy()
        # Find the highest-contributing coordinate and move mass away
        j_rel = int(np.argmax(contrib))
        j = lo + j_rel
        amount = a[j] * frac
        if amount < 1e-12:
            continue
        # Find a low-contributing coordinate to receive mass
        # Score: how much each coordinate contributes to ALL top peaks
        total_contrib = np.zeros(n)
        for kk in top_idx:
            lo2 = max(0, int(kk) - n + 1)
            hi2 = min(n - 1, int(kk))
            if hi2 >= lo2:
                total_contrib[lo2:hi2+1] += a[lo2:hi2+1]
        # Pick from coordinates with low total contribution and not in the top-peak region
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        for kk in top_idx:
            lo3 = max(0, int(kk) - n + 1)
            hi3 = min(n - 1, int(kk))
            mask[lo3:hi3+1] = False
        if mask.sum() < 1:
            mask[j] = True
        candidates = np.where(mask)[0]
        scores = total_contrib[candidates]
        l = candidates[int(np.argmin(scores))]
        a_new[j] -= amount
        a_new[l] += amount

    a_new = np.maximum(a_new, 0)
    a_new = a_new * (S / a_new.sum())
    return a_new


def peak_targeting(a_init, max_time, rng):
    """
    Direct peak-targeting: repeatedly identify the argmax of convolution,
    compute exact subgradient, and take adaptive steps.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    step = 0.1 * S / np.sqrt(n)

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        if it - last_improve > 5000:
            break

        # Top-K subgradient
        stuck = it - last_improve
        if stuck < 200:
            k_top = 1
        elif stuck < 1000:
            k_top = 3
        elif stuck < 3000:
            k_top = 5
        else:
            k_top = min(10, 5 + stuck // 500)

        top_idx = np.argsort(c)[::-1][:k_top]
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / k_top
        fsize = 1 << (2 * n - 1 + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step: larger when making progress
        step = min(2.0 * S / gnorm, max(step * 1.01, 0.01 * S / gnorm))
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)

        if obj_new < cur_obj + 1e-14:
            a = a_new
            c = c_new
        else:
            step *= 0.5
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if 2.0 * n * c_new.max() / (S * S) < cur_obj + 1e-14:
                a = a_new
                c = c_new

        # Structured restart when very stuck
        if stuck > 2000 and it % 200 == 0:
            a_try = structured_restart(a, rng)
            obj_try = compute_obj(a_try)
            if obj_try < best_obj:
                a = a_try
                c = fft_convolve_self(a)
                last_improve = it

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-3.0 * (tt - 0.5) ** 2 / (0.2 ** 2))
    elif method == "multi_tone":
        a = 0.3 + 0.2 * np.sin(2 * np.pi * 3 * tt) + 0.15 * np.sin(2 * np.pi * 7 * tt + 1.3)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "riemann_zeta":
        # Inspired by zeta function structure
        a = 1.0 / (1.0 + tt) ** 0.5
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization:
    1. Multi-beta LSE FISTA (coarse)
    2. L-BFGS hybrid (fine, superlinear)
    3. Peak-targeting with structured restarts
    4. Upsample + re-optimize
    5. Final polish
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        a_start = make_init(n, "cosine", rng)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.8f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: Multi-beta LSE FISTA (25% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.25
    if p1_time > 60:
        print("\n=== Phase 1: Multi-beta FISTA (budget %.0fs) ===" % p1_time)

        # Run with different weight configurations
        configs = [
            ((20, 80, 300), (0.3, 0.4, 0.3)),
            ((10, 50, 200, 800), (0.2, 0.3, 0.3, 0.2)),
        ]
        per_time = p1_time / len(configs)
        for betas, weights in configs:
            if time.monotonic() >= deadline:
                break
            rem = min(per_time, deadline - time.monotonic())
            if rem < 10:
                break
            a_opt, obj_opt = run_multi_beta_fista(global_best_a, rem, rng,
                                                   betas=betas, weights=weights)
            print("  Multi-beta FISTA: obj=%.8f" % obj_opt)
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 2: L-BFGS Hybrid (30% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.30
    if p2_time > 60:
        print("\n=== Phase 2: L-BFGS Hybrid (budget %.0fs) ===" % p2_time)

        # Multiple L-BFGS runs with different beta/l2_weight
        lbfgs_configs = [
            (50.0, 0.003),
            (100.0, 0.005),
            (200.0, 0.002),
        ]
        per_time = p2_time / len(lbfgs_configs)
        for beta, l2w in lbfgs_configs:
            if time.monotonic() >= deadline:
                break
            rem = min(per_time, deadline - time.monotonic())
            if rem < 10:
                break
            a_opt, obj_opt = lbfgs_hybrid(global_best_a, rem, rng,
                                           m_store=40, beta=beta, l2_weight=l2w)
            print("  L-BFGS (beta=%.0f, l2=%.3f): obj=%.8f" % (beta, l2w, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 3: Peak-targeting with structured restarts (25% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.25
    if p3_time > 60:
        print("\n=== Phase 3: Peak-targeting (budget %.0fs) ===" % p3_time)
        a_pt, obj_pt = peak_targeting(global_best_a, p3_time, rng)
        print("  Peak-targeting: obj=%.8f" % obj_pt)
        if obj_pt < global_best_obj:
            global_best_obj = obj_pt
            global_best_a = a_pt.copy()

        # Follow-up L-BFGS on the peak-targeting result
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            a_l, obj_l = lbfgs_hybrid(global_best_a, remaining3 * 0.6, rng,
                                       m_store=30, beta=150.0, l2_weight=0.004)
            print("  L-BFGS after PT: obj=%.8f" % obj_l)
            if obj_l < global_best_obj:
                global_best_obj = obj_l
                global_best_a = a_l.copy()

    # === Phase 4: Upsample + re-optimize (10% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.10
    if p4_time > 45 and len(global_best_a) <= 10000:
        n2 = len(global_best_a) * 2
        if n2 <= 32768:
            print("\n=== Phase 4: Upsample to n=%d (budget %.0fs) ===" % (n2, p4_time))
            a_up = upsample(global_best_a, factor=2)
            S2 = np.sqrt(2.0 * n2)
            a_up = a_up * (S2 / a_up.sum())
            obj_up = compute_obj(a_up)
            print("  Upsampled: obj=%.8f" % obj_up)
            if obj_up < global_best_obj:
                global_best_obj = obj_up
                global_best_a = a_up.copy()

            # Optimize at larger scale
            a_opt, obj_opt = run_multi_beta_fista(a_up, p4_time * 0.5, rng,
                                                   betas=(10, 40, 150),
                                                   weights=(0.4, 0.35, 0.25))
            print("  FISTA upsampled: obj=%.8f" % obj_opt)
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

            remaining4 = deadline - time.monotonic()
            if remaining4 > 15:
                a_l2, obj_l2 = lbfgs_hybrid(global_best_a, remaining4 * 0.7, rng,
                                             m_store=20, beta=60.0, l2_weight=0.005)
                print("  L-BFGS upsampled: obj=%.8f" % obj_l2)
                if obj_l2 < global_best_obj:
                    global_best_obj = obj_l2
                    global_best_a = a_l2.copy()

    # === Phase 5: Final polish (10% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.90
    if p5_time > 10:
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % p5_time)

        # Alternating: L-BFGS → peak targeting → L-BFGS
        half = p5_time * 0.5
        a_p1, obj_p1 = lbfgs_hybrid(global_best_a, half, rng,
                                     m_store=30, beta=200.0, l2_weight=0.001)
        print("  Polish L-BFGS: obj=%.8f" % obj_p1)
        if obj_p1 < global_best_obj:
            global_best_obj = obj_p1
            global_best_a = a_p1.copy()

        remaining5 = deadline - time.monotonic()
        if remaining5 > 10:
            a_p2, obj_p2 = peak_targeting(global_best_a, remaining5, rng)
            print("  Polish PT: obj=%.8f" % obj_p2)
            if obj_p2 < global_best_obj:
                global_best_obj = obj_p2
                global_best_a = a_p2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.8f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
