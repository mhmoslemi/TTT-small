def fft_convolve_self(a):

    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    # Sort in descending order
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    # Find rho = max{i: w[i] > tc[i]}
    valid = (w > tc)
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0])) + 1
    else:
        rho = n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        lse_val, w = lse_value_gradient(c, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        # Gradient of LSE w.r.t. y
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        Y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-14 and bt < 8:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 3:
                step = step_local * 0.7
            else:
                step = min(step * 1.05, 5.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        # Periodic restart with noise
        if it - last_improve > 3000:
            t_mom = 1.0
            y = a.copy()
            noise_scale = max(1e-7, 0.005 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
    
    return best_a, best_obj


def run_threshold_descent(a_init, max_time, rng):
    """
    Gradient descent with threshold penalty: minimize sum_k max(0, c[k]-t)^2.
    The threshold t adaptively tracks just below the current max.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 10000:
            break
        
        # Adaptive threshold: start low, increase over time
        frac = 0.85 + 0.13 * min(1.0, it / 5000.0)
        t = frac * cur_max
        
        grad = threshold_penalty_gradient(a, c, t)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size with decay
        step = S / (gnorm * (1.0 + 0.01 * it))
        step = min(step, 3.0 * S / gnorm)
        
        # Backtracking line search
        for _ in range(5):
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            obj_new = 2.0 * n * c_new.max() / (S * S)
            if obj_new <= cur_obj + 1e-12:
                a = a_new
                break
            step *= 0.5
        else:
            a = project_simplex(a - step * grad, S)
        
        # Escape from local minima
        if stuck > 2000 and it % 100 == 0:
            ns = max(1e-6, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 5000))
            a = project_simplex(a + rng.normal(0, ns, n), S)
    
    return best_a, best_obj


def run_subgradient_line_search(a_init, max_time, rng, top_k=1):
    """
    Exact subgradient descent with top-k lags and backtracking.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 5000:
            break
        
        # Adaptive top-k
        k_eff = min(top_k + stuck // 1000, min(16, 2 * n - 1))
        top_idx = np.argsort(c)[::-1][:k_eff]
        
        # Compute average subgradient over top-k lags
        grad = np.zeros(n)
        for kk in top_idx:
            kk = int(kk)
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            if hi >= lo:
                grad[lo:hi+1] += a[kk - hi:kk - lo + 1][::-1]
        grad /= k_eff
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = 2.0 * S / gnorm
        
        # Backtracking
        for _ in range(10):
            a_new = project_simplex(a - step * grad, S)
            obj_new = compute_obj(a_new)
            if obj_new <= cur_obj + 1e-13:
                a = a_new
                break
            step *= 0.5
        else:
            pass  # No improvement, try next iteration
        
        # Escape
        if stuck > 1000 and it % 50 == 0:
            ns = max(1e-6, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 3000))
            a = project_simplex(a + rng.normal(0, ns, n), S)
    
    return best_a, best_obj


def run_logspace_descent(a_init, max_time, rng):
    """
    Gradient descent in log-space: a[j] = softmax(x[j]).
    Objective: 2n * max_k c_p[k] where p = softmax(x), sum(p) = 1.
    """
    n = len(a_init)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 1e-10, 1000).copy()
    a = a / a.sum()  # normalize to sum = 1
    S = np.sqrt(2.0 * n)
    
    # Initialize log-space
    x = np.log(a)  # a = softmax(x) up to normalization
    x = x - x.max()  # for numerical stability
    
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Compute a from x
        x_shifted = x - x.max()
        a = np.exp(x_shifted)
        a = a / a.sum()
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        # Objective with sum=1: 2n * max(c_p) where c_p[k] = c[k] (since sum=1)
        # But our target objective uses sum = sqrt(2n), so:
        # obj = 2n * max(c_a) / sum_a^2 = 2n * S^2 * max(c_p) / S^2 = 2n * max(c_p)
        cur_obj = 2.0 * n * cur_max
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 10000:
            break
        
        # Find the peak lag
        k_star = int(np.argmax(c))
        
        # Compute gradient in log-space
        # d(c[k*])/dx[l] = 2*a[l]*(a[k*-l] - c[k*]) for valid k*-l
        grad_x = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi >= lo:
            partner = a[k_star - hi:k_star - lo + 1][::-1]
            grad_x[lo:hi+1] = 2.0 * a[lo:hi+1] * (partner - cur_max)
        
        gnorm = np.linalg.norm(grad_x)
        if gnorm < 1e-15:
            break
        
        # Step in log-space
        step = min(1.0 / gnorm, 0.1)
        x = x - step * grad_x
        x = x - x.max()  # renormalize for stability
        
        # Clip to prevent numerical issues
        x = np.clip(x, -50, 50)
        
        # Escape
        if stuck > 2000 and it % 100 == 0:
            ns = max(0.01, 0.1 * np.exp(-stuck / 5000))
            x = x + rng.normal(0, ns, n)
            x = x - x.max()
    
    # Convert back: a = softmax(x) * S
    x_shifted = x - x.max()
    a_final = np.exp(x_shifted)
    a_final = a_final / a_final.sum() * S
    return a_final, best_obj


def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_varied_init(n, rng, seed_offset=0):
    """Create a varied initial sequence."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    method = (seed_offset) % 8
    
    if method == 0:
        # Cosine
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == 1:
        # Gaussian bump
        sigma = 0.1 + 0.1 * rng.random()
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (sigma ** 2))
    elif method == 2:
        # Multi-tone
        freqs = rng.uniform(1, 15, size=5).astype(int)
        phases = rng.uniform(0, 2*np.pi, size=5)
        a = np.ones(n) * 0.3
        for f, p in zip(freqs, phases):
            a += np.abs(np.sin(2 * np.pi * f * tt + p)) * 0.15
    elif method == 3:
        # Random with smoothing
        a = rng.random(n)
        kernel = np.exp(-0.5 * (np.arange(-20, 21) / 5.0) ** 2)
        a = np.convolve(a, kernel, mode='same')
    elif method == 4:
        # Chirp
        a = 0.5 + 0.5 * np.sin(2 * np.pi * (2 * tt + 3 * tt**2))
    elif method == 5:
        # Two bumps
        c1, c2 = 0.3, 0.7
        w1, w2 = 0.1, 0.08
        a = np.exp(-((tt-c1)/w1)**2) + np.exp(-((tt-c2)/w2)**2)
    elif method == 6:
        # Sawtooth-like
        a = np.abs(tt * 2.0 - 1.0) ** 0.5
    else:
        # Random periodic
        period = rng.integers(3, 20)
        a = 0.5 + 0.5 * np.sin(2 * np.pi * ttl if (ttl := tt * period) is not None else 0)
        a = np.abs(a)
    
    a = np.maximum(a, 0.01)
    return a * (S / a.sum())
