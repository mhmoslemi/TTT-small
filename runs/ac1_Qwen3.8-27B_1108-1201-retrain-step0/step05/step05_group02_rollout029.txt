"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Multi-phase optimization:
1. Adaptive LSE-FISTA with backtracking
2. Efficient coordinate descent (mass transport from high to low contribution)
3. Spectral spreading for escaping local minima
4. High-precision polishing
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_val_and_grad(y, beta):
    """Compute beta*LSE(conv(y)) value and gradient."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return val, grad, c, w


def run_adaptive_fista(a_init, max_time, rng, beta_init=20.0, beta_max=50000.0):
    """
    FISTA with adaptive LSE beta and backtracking line search.
    The beta is increased based on the LSE-max gap to tighten the approximation.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / max(beta_init, 1.0)
    beta = beta_init
    c = fft_convolve_self(a)
    cur_max = c.max()
    cur_obj = 2.0 * n * cur_max / (S * S)
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta: increase when LSE is far from max
        lse_val, grad, c_lse, w = lse_val_and_grad(y, beta)
        gap = lse_val - c_lse.max()
        if gap > 0.01 / beta:
            beta = min(beta_max, beta * 1.05)
        elif gap > 0.001 / beta:
            beta = min(beta_max, beta * 1.02)
        
        # Recompute with new beta
        lse_val, grad, c, w = lse_val_and_grad(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        step_local = step
        bt = 0
        improved = False
        while bt < 10:
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = 2.0 * n * fft_convolve_self(x_trial).max() / (S * S)
            if obj_trial <= cur_obj - 1e-14 * max(1, abs(cur_obj)):
                improved = True
                break
            step_local *= 0.5
            bt += 1
        
        if improved:
            step = min(step_local * 1.1, 2.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            # No improvement; try a smaller step or reset
            step_local = step * 0.1
            x_new = project_simplex(y - step_local * grad, S)
            obj_trial = 2.0 * n * fft_convolve_self(x_new).max() / (S * S)
            if obj_trial <= cur_obj + 1e-12:
                x_new = x_trial if obj_trial < cur_obj else a
            else:
                x_new = a
            step = step_local
        
        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        # Restart if stuck
        stuck = it - last_improve
        if stuck > 3000:
            t_mom = 1.0
            noise_scale = max(1e-7, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 5000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
        
        # Early termination if very stuck
        if stuck > 20000:
            break
    
    return best_a, best_obj


def coordinate_descent_v2(a_init, max_time, rng, max_src=8, max_tgt=8):
    """
    Efficient coordinate descent via mass transfer.
    Only considers top-k sources and targets to keep it fast.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 50000:
            break
        
        # Find top few peak lags
        top_k = min(5, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # Compute multi-contribution (weighted sum of contributions to top lags)
        multi_contrib = np.zeros(n)
        for ii, kk in enumerate(top_idx):
            weight = c[kk] - c.min()  # relative weight
            kk = int(kk)
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            if hi >= lo:
                contrib = a[kk - hi:kk - lo + 1][::-1]
                multi_contrib[lo:hi+1] += weight * contrib
        
        # Select sources: high contribution, decent value
        src_scores = multi_contrib * np.maximum(a, 1e-6)
        # Select targets: low contribution, room to grow
        tgt_scores = -multi_contrib * (S / n * 2 - a + 1e-6)
        
        src_candidates = np.argsort(src_scores)[::-1][:max_src]
        tgt_candidates = np.argsort(tgt_scores)[:max_tgt]
        
        best_delta_obj = cur_obj
        best_move = None
        
        for j in src_candidates:
            for l in tgt_candidates:
                if j == l or a[j] < 1e-10:
                    continue
                # Try several transfer amounts
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = a[j] * frac
                    if delta < 1e-12:
                        continue
                    
                    # Compute effect on convolution
                    # Reducing a[j] by delta affects c[k] for k in [j, j+n-1]
                    # c[k] -= delta * a[k-j] (for valid k-j)
                    # Increasing a[l] by delta affects c[k] for k in [l, l+n-1]
                    # c[k] += delta * a[k-l] (for valid k-l)
                    
                    # Only need to check if new max < cur_max
                    # Affected range: union of [j, j+n-1] and [l, l+n-1]
                    # But we need to check the full c + delta_c
                    
                    # Quick check: what's the max change at the current peak?
                    jk = int(top_idx[0])
                    change_at_peak = 0.0
                    if 0 <= jk - j < n:
                        change_at_peak -= delta * a[jk - j]
                    if 0 <= jk - l < n:
                        change_at_peak += delta * a[jk - l]
                    
                    # Estimate new max (conservative)
                    # The peak might shift, so we need to be more careful
                    # For speed, just check if the change at peak is negative
                    # and do full check only for promising candidates
                    
                    if change_at_peak < -1e-10:
                        # Full check
                        delta_c = np.zeros(2 * n - 1)
                        js = j
                        je = min(2 * n - 2, j + n - 1)
                        delta_c[js:je+1] -= delta * a[max(0, js-j):min(n, je-j+1)]
                        ls = l
                        le = min(2 * n - 2, l + n - 1)
                        delta_c[ls:le+1] += delta * a[max(0, ls-l):min(n, le-l+1)]
                        
                        new_c = c + delta_c
                        new_max = new_c.max()
                        new_obj = 2.0 * n * new_max / (S * S)
                        
                        if new_obj < best_delta_obj:
                            best_delta_obj = new_obj
                            best_move = (j, l, delta)
        
        if best_move is not None:
            j, l, delta = best_move
            a[j] -= delta
            a[l] += delta
            # Update c
            js = j
            je = min(2 * n - 2, j + n - 1)
            c[js:je+1] -= delta * a[max(0, js-j):min(n, je-j+1)]
            ls = l
            le = min(2 * n - 2, l + n - 1)
            c[ls:le+1] += delta * a[max(0, ls-l):min(n, le-l+1)]
        else:
            # No improvement; try random perturbation if stuck
            if stuck > 500:
                j = rng.integers(0, n)
                delta = a[j] * 0.01
                if delta > 1e-12:
                    a[j] -= delta
                    l = rng.integers(0, n)
                    a[l] += delta
                    c = fft_convolve_self(a)
        
        # Periodic recompute
        if it % 200 == 0:
            c = fft_convolve_self(a)
    
    return best_a, best_obj


def spectral_spread(a_init, max_time, rng):

        F = fft(a, fsize)

            a = a_new

            step *= 0.1
            a_new = project_simplex(a - step * grad, S)

Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Multi-phase optimization with:
1. Adaptive LSE-FISTA with backtracking and momentum restarts
2. Fast coordinate descent with incremental convolution updates
3. Structured perturbations (subset zeroing) to escape local minima
4. High-precision final polishing

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_val_and_grad(y, beta):
    """Compute LSE value and gradient of LSE(conv(y)) w.r.t. y."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w

    step = 1.0 / max(beta_init, 1.0)
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta based on LSE-max gap
        lse_val, grad, c, w = lse_val_and_grad(y, beta)
        gap = lse_val - c.max()
        if gap > 0.05 / beta:
            beta = min(beta_max, beta * 1.08)
            lse_val, grad, c, w = lse_val_and_grad(y, beta)

        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking
        step_local = step
        bt = 0
        x_new = None
        while bt < 8:
            x_trial = project_simplex(y - step_local * grad, S)
            c_trial = fft_convolve_self(x_trial)
            obj_trial = 2.0 * n * c_trial.max() / (S * S)
            if obj_trial <= cur_obj - 1e-14:
                x_new = x_trial
                break
            step_local *= 0.5
            bt += 1

        if x_new is None:
            # Last resort: very small step
            step_local = step * 0.01
            x_trial = project_simplex(y - step_local * grad, S)
            c_trial = fft_convolve_self(x_trial)
            obj_trial = 2.0 * n * c_trial.max() / (S * S)
            if obj_trial <= cur_obj + 1e-12:
                x_new = x_trial
            else:
                x_new = a.copy()
            step = step_local * 0.5
        else:
            step = min(step_local * 1.05, 3.0 / max(beta, 1.0))

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        # Momentum restart with noise when stuck
        stuck = it - last_improve
        if stuck > 2000:
            t_mom = 1.0
            noise_scale = max(1e-7, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 4000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

        if stuck > 15000:
            break

    return best_a, best_obj


def coord_descent(a_init, max_time, rng, max_src=8, max_tgt=8):
    """
    Fast coordinate descent with incremental O(n) convolution updates.
    Transfers mass from high-contribution to low-contribution coordinates.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    a_max = a.max()
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0
    c_max = c.max()
    c_max_idx = int(np.argmax(c))

    while time.monotonic() - start < max_time:
        it += 1
        cur_obj = 2.0 * n * c_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 200000:
            break

        # Top peak lags
        top_k = min(3, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]

        # Multi-contribution to top lags
        multi_contrib = np.zeros(n)
        for kk in top_idx:
            kk = int(kk)
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            if hi >= lo:
                multi_contrib[lo:hi + 1] += a[kk - hi:kk - lo + 1][::-1]

        # Sources: high contribution × value
        src_scores = multi_contrib * np.maximum(a, 1e-8)
        # Targets: low contribution × room
        tgt_scores = -multi_contrib * (S / n + 1e-8)

        src_cands = np.argsort(src_scores)[::-1][:max_src]
        tgt_cands = np.argsort(tgt_scores)[:max_tgt]

        best_improvement = 0.0
        best_move = None
        a_max = a.max()

        for j in src_cands:
            for l in tgt_cands:
                if j == l or a[j] < 1e-10:
                    continue
                for frac in (0.04, 0.08, 0.15):
                    delta = a[j] * frac
                    if delta < 1e-12:
                        continue

                    # Quick check at current peak
                    ch = 0.0
                    if 0 <= c_max_idx - j < n:
                        ch -= delta * a[c_max_idx - j]
                    if 0 <= c_max_idx - l < n:
                        ch += delta * a[c_max_idx - l]

                    # Conservative bound on increase elsewhere
                    max_possible_increase = 2.0 * delta * a_max
                    if c_max + ch >= c_max - max_possible_increase:
                        # Peak doesn't decrease enough vs worst-case increase
                        # Still might be worth checking, but skip for speed
                        if ch >= -1e-12:
                            continue

                    # Full check
                    new_c = c.copy()
                    new_c[j:j+n] -= delta * a
                    new_c[l:l+n] += delta * a
                    new_max = new_c.max()
                    improvement = c_max - new_max
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_move = (j, l, delta)

        if best_move is not None:
            j, l, delta = best_move
            a[j] -= delta
            a[l] += delta
            c[j:j+n] -= delta * a
            c[l:l+n] += delta * a
            c_max = c.max()
            c_max_idx = int(np.argmax(c))
        else:
            # No improvement; structured perturbation if stuck
            if stuck > 200:
                # Zero a random subset and redistribute
                n_zero = max(10, n // 200)
                zero_mask = rng.random(n) < (n_zero / n)
                mass = a[zero_mask].sum()
                if mass > 1e-10:
                    a[zero_mask] = 0
                    a[~zero_mask] += mass / (~zero_mask).sum()
                    c = fft_convolve_self(a)
                    c_max = c.max()
                    c_max_idx = int(np.argmax(c))

        # Periodic recompute
        if it % 300 == 0:
            c = fft_convolve_self(a)
            c_max = c.max()
            c_max_idx = int(np.argmax(c))

    return best_a, best_obj


def structured_perturb(a_init, max_time, rng):
    """
    Try various structured perturbations to escape local minima.
    Includes: subset zeroing, frequency-domain equalization, random restarts.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()

    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        # Try different perturbation strategies
        strategy = rng.integers(0, 3)

        if strategy == 0:
            # Zero a random contiguous block and redistribute
            block_size = rng.integers(n // 50, n // 10)
            start_idx = rng.integers(0, n - block_size)
            mass = a[start_idx:start_idx + block_size].sum()
            if mass > 1e-10:
                a_trial = a.copy()
                a_trial[start_idx:start_idx + block_size] = 0
                mask = np.ones(n, dtype=bool)
                mask[start_idx:start_idx + block_size] = False
                a_trial[mask] += mass / mask.sum()
                obj_trial = compute_obj(a_trial)
                if obj_trial < best_obj:
                    best_obj = obj_trial
                    best_a = a_trial.copy()
                    a = a_trial

        elif strategy == 1:
            # Frequency-domain equalization: compress dynamic range of |F|^2
            F = fft(a, fsize)
            P = np.abs(F) ** 2
            P_mean = P.mean()
            if P_mean > 0:
                # Soft compression
                alpha = 0.8
                P_new = P ** alpha
                P_new = P_new / P_new.sum() * P.sum()
                # Recover phase from F, use new magnitude
                phase = np.angle(F)
                F_new = np.sqrt(P_new) * np.exp(1j * phase)
                a_trial = np.real(ifft(F_new, fsize)[:n])
                a_trial = np.maximum(a_trial, 0)
                if a_trial.sum() > 1e-10:
                    a_trial = a_trial * (S / a_trial.sum())
                    obj_trial = compute_obj(a_trial)
                    if obj_trial < best_obj:
                        best_obj = obj_trial
                        best_a = a_trial.copy()
                        a = a_trial

        else:
            # Random smooth perturbation
            t = np.linspace(0, 1, n, endpoint=False)
            n_freq = rng.integers(2, 8)
            freqs = rng.integers(1, 20, n_freq)
            phases = rng.uniform(0, 2 * np.pi, n_freq)
            amp = rng.uniform(0.01, 0.05)
            perturbation = np.zeros(n)
            for f, p in zip(freqs, phases):
                perturbation += np.sin(2 * np.pi * f * t + p)
            a_trial = a + amp * perturbation
            a_trial = np.maximum(a_trial, 0)
            if a_trial.sum() > 1e-10:
                a_trial = a_trial * (S / a_trial.sum())
                obj_trial = compute_obj(a_trial)
                if obj_trial < best_obj:
                    best_obj = obj_trial
                    best_a = a_trial.copy()
                    a = a_trial
                # Quick FISTA from the perturbed point
                if obj_trial < best_obj + 0.5:
                    a_f, obj_f = run_fista_adaptive(a_trial, min(5, max(1, (deadline - time.monotonic()) / 3)),
                                                     rng, beta_init=50, beta_max=10000)
                    if obj_f < best_obj:
                        best_obj = obj_f
                        best_a = a_f.copy()
                        a = a_f

        # Brief FISTA from current best
        remaining_inner = max(1, min(3, (deadline - time.monotonic()) / 5))
        if remaining_inner > 0.5:
            a_f, obj_f = run_fista_adaptive(best_a, remaining_inner, rng,
                                            beta_init=100, beta_max=20000)
            if obj_f < best_obj:
                best_obj = obj_f
                best_a = a_f.copy()

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search minimizing 2*n*max_conv/sum^2.
    Combines adaptive FISTA, fast coordinate descent, and structured perturbations.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        tt = np.linspace(0, 1, n, endpoint=False)
        a_start = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
        a_start = np.maximum(a_start, 0.01)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.8f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: FISTA warmup (10% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.10
    if p1_time > 20:
        print("\n=== Phase 1: FISTA warmup (budget %.0fs) ===" % p1_time)
        a_opt, obj_opt = run_fista_adaptive(global_best_a, p1_time, rng,
                                             beta_init=20, beta_max=10000)
        print("  Phase 1: obj=%.8f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

    # === Phase 2: Coordinate descent (25% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.25
    if p2_time > 60:
        print("\n=== Phase 2: Coord descent (budget %.0fs) ===" % p2_time)
        a_cd, obj_cd = coord_descent(global_best_a, p2_time, rng,
                                     max_src=10, max_tgt=10)
        print("  Phase 2: obj=%.8f" % obj_cd)
        if obj_cd < global_best_obj:
            global_best_obj = obj_cd
            global_best_a = a_cd.copy()

    # === Phase 3: FISTA after CD (15% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.15
    if p3_time > 20:
        print("\n=== Phase 3: FISTA polish (budget %.0fs) ===" % p3_time)
        a_opt, obj_opt = run_fista_adaptive(global_best_a, p3_time, rng,
                                             beta_init=100, beta_max=50000)
        print("  Phase 3: obj=%.8f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

    # === Phase 4: Structured perturbations (15% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.15
    if p4_time > 30:
        print("\n=== Phase 4: Structured perturbations (budget %.0fs) ===" % p4_time)
        a_sp, obj_sp = structured_perturb(global_best_a, p4_time, rng)
        print("  Phase 4: obj=%.8f" % obj_sp)
        if obj_sp < global_best_obj:
            global_best_obj = obj_sp
            global_best_a = a_sp.copy()

    # === Phase 5: Coord descent round 2 (15% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.15
    if p5_time > 30:
        print("\n=== Phase 5: CD round 2 (budget %.0fs) ===" % p5_time)
        a_cd2, obj_cd2 = coord_descent(global_best_a, p5_time, rng,
                                       max_src=8, max_tgt=8)
        print("  Phase 5: obj=%.8f" % obj_cd2)
        if obj_cd2 < global_best_obj:
            global_best_obj = obj_cd2
            global_best_a = a_cd2.copy()

    # === Phase 6: High-beta FISTA (10% time) ===
    remaining = deadline - time.monotonic()
    p6_time = remaining * 0.10
    if p6_time > 15:
        print("\n=== Phase 6: High-beta FISTA (budget %.0fs) ===" % p6_time)
        a_opt, obj_opt = run_fista_adaptive(global_best_a, p6_time, rng,
                                             beta_init=500, beta_max=200000)
        print("  Phase 6: obj=%.8f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

    # === Phase 7: Final CD (5% time) ===
    remaining = deadline - time.monotonic()
    p7_time = remaining * 0.90
    if p7_time > 10:
        print("\n=== Phase 7: Final CD (budget %.0fs) ===" % p7_time)
        a_final, obj_final = coord_descent(global_best_a, p7_time, rng,
                                            max_src=6, max_tgt=6)
        print("  Phase 7: obj=%.8f" % obj_final)
        if obj_final < global_best_obj:
            global_best_obj = obj_final
            global_best_a = a_final.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.8f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
