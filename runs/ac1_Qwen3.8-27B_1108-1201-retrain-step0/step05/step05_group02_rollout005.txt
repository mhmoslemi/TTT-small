"""
Minimize 2*n*max_conv/sum^2 using LP-based active set optimization.

Core idea: Iteratively solve an LP to find the best direction g
that maximizes sum(g) subject to (f*g)[k] <= M for active constraints.
This gives a principled descent direction for the autocorrelation peak.

Combined with FISTA for smooth refinement and coordinate descent for polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import scipy.sparse as sp


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-30:
        return np.full(n, S / n)
    return v * (S / s)


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, beta):
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def run_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.01,
              beta_max=10000.0, step_scale=1.0, use_backtrack=True):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()

        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_gradient(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-14 and bt < 5:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1

                step = min(step * 1.03, 2.0 / max(beta, 1.0))
            x_new = x_trial

        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)

def build_active_matrix(f, active_indices, n):
    """Build the constraint matrix A[active_indices, :] where A[k,j] = f[k-j] if valid."""
    K = len(active_indices)
    # A is K x n, sparse
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(active_indices):
        k = int(k)
        # (f*g)[k] = sum_{j} f[k-j] * g[j] where 0 <= k-j <= n-1 and 0 <= j <= n-1
        # So j ranges from max(0, k-n+1) to min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi >= j_lo:
            js = np.arange(j_lo, j_hi + 1)
            fs = f[k - js]  # f[k-j] for j in [j_lo, j_hi]
            rows.extend([i] * len(js))
            cols.extend(js.tolist())
            vals.extend(fs.tolist())
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(K, n))
    return A


def lp_active_set_step(f, M, active_indices, n, method='highs'):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_indices, g >= 0.
    Returns g (unnormalized).
    """
    K = len(active_indices)
    A = build_active_matrix(f, active_indices, n)
    
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(K, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method=method)
    
    if res.success:
        return res.x
    else:
        # Fallback: return f itself (which is feasible)
        return f.copy()


def run_lp_active_set(f_init, max_time, rng, K_start=50, K_max=500, 
                       step_search=True, verbose=True):
    """
    LP-based active set optimization.
    """
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f = np.clip(np.asarray(f_init, dtype=np.float64), 0, 1000).copy()
    f = f * (S / f.sum())
    
    start = time.monotonic()
    best_obj = compute_obj(f)
    best_f = f.copy()
    last_improve = 0
    it = 0
    
    if verbose:
        print(f"  LP Active Set: n={n}, initial obj={best_obj:.8f}")
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(f)
        M = c.max()
        cur_obj = 2.0 * n * M / (S * S)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            last_improve = it
            if verbose and it % 10 == 0:
                print(f"    iter {it}: obj={cur_obj:.8f} (improved)")
        
        # Get active set: top-K indices of c
        K = min(K_start + it * 2, K_max)
        active_indices = np.argsort(c)[::-1][:K]
        
        # Solve LP
        t0 = time.monotonic()
        g = lp_active_set_step(f, M, active_indices, n)
        lp_time = time.monotonic() - t0
        
        if g is None or g.sum() < 1e-15:
            g = f.copy()
        
        # Normalize g
        g = g * (S / g.sum())
        
        # Line search for best alpha in [0, 1]
        if step_search:
            best_alpha = 0.0
            best_alpha_obj = cur_obj
            for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                f_trial = (1 - alpha) * f + alpha * g
                f_trial = np.maximum(f_trial, 0)
                ftrial_s = f_trial.sum()
                if ftrial_s < 1e-15:
                    continue
                f_trial = f_trial * (S / ftrial_s)
                obj_trial = compute_obj(f_trial)
                if obj_trial < best_alpha_obj:
                    best_alpha_obj = obj_trial
                    best_alpha = alpha
            
            if best_alpha > 0:
                f = (1 - best_alpha) * f + best_alpha * g
                f = np.maximum(f, 0)
                f = f * (S / f.sum())
            else:
                # No improvement; try small perturbation
                if it - last_improve > 50:
                    f = project_simplex(f + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
        else:
            f = g
        
        # Check time
        if time.monotonic() - start > max_time:
            break
        
        # Early termination if stuck for too long
        if it - last_improve > 200:
            if verbose:
                print(f"    Stuck after {it} iterations, breaking. Best obj={best_obj:.8f}")
            break
        
        # Periodic verbose
        if verbose and it % 20 == 0:
            c_check = fft_convolve_self(f)
            obj_check = 2.0 * n * c_check.max() / (S * S)
            print(f"    iter {it}: obj={obj_check:.8f}, M={c_check.max():.6f}, "
                  f"K={K}, lp_time={lp_time:.3f}s")
    
    return best_f, best_obj

            js = max(0, j)

def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.15 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 4 * tt + 0.7)
    elif method == "multi_tone":
        a = 0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt + 1.3) + 0.15 * np.sin(2 * np.pi * 11 * tt + 2.1)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 2.0 * np.abs(tt - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())

OK so I had it completely wrong before. The convolution (a*a)(k) = sum_j a[j] * a[k-j] is symmetric: (a*a)(k) = (a*a)(2n-2-k).

The maximum of the convolution can be at any lag, not necessarily at the center.

For a constant sequence a_j = c:
(a*a)(k) = c^2 * min(k+1, n, 2n-1-k)
This is triangular, peaking at k = n-1 with value c^2 * n = S^2/n.
Objective = 2n * S^2/n / S^2 = 2.

For a non-constant sequence, the peak of the convolution can be higher or lower than S^2/n. If it's lower, the objective is < 2.

So the question is: can we find a non-negative sequence where the peak of the full convolution is less than S^2/n?

For a = [1, 2, 3] normalized: S = 6, S^2/n = 36/3 = 12. Peak of convolution = 12. So objective = 2.

So the "bump" shape [3, 1, 3] gives objective 1.59 < 2! The peak is at the center lag (n-1=2) and its value is 13 < 16.33.

The key insight: for a sequence that is large at the ends and small in the middle, the center of the convolution (which is where the "cross terms" contribute most) can be lower than for a constant sequence.

Hmm, so a "bump" shape actually INCREASES the peak at the center. That makes sense because the center of the convolution is where all the "cross terms" add up.

So to DECREASE the peak, we want to REDUCE the center contribution. The center lag (n-1) has the most terms: (a*a)(n-1) = sum_{j=0}^{n-1} a[j]*a[n-1-j]. To minimize this, we want a[j]*a[n-1-j] to be small for most j, which means the sequence should be "anti-palindromic" (large at one end, small at the other).
