"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Algorithm: Multi-phase optimization combining:
1. Adaptive multi-scale FISTA with excess-weighted LSE smoothing
2. Frank-Wolfe (conditional gradient) with subgradient averaging
3. Spectral domain equalization
4. High-beta final polishing
Uses the given sequence as warm start and tries multiple n values.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution at point a."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def excess_weighted_gradient(a, c, threshold_frac, beta):
    """Gradient using peaks above a threshold, weighted by excess."""
    n = len(a)
    target = np.median(c) + threshold_frac * (c.max() - np.median(c))
    mask = c >= target
    if mask.sum() == 0:
        mask = c >= c.max() - 1e-10
    idx = np.where(mask)[0]
    c_sel = c[idx]
    cm = c_sel.max()
    excess = c_sel - target
    excess = np.maximum(excess, 1e-10)
    weights = excess ** beta
    weights = weights / weights.sum()

    w_full = np.zeros(len(c))
    w_full[idx] = weights

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def spectral_flatten(a, alpha, rng):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()

    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a

    # Target: make power more uniform
    # Damp high, boost low
    ratio = power / max(mean_p, 1e-20)
    new_power = power * (1.0 + alpha * (1.0 - np.minimum(ratio, 5.0)) / (1.0 + np.minimum(ratio, 5.0)))
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / np.maximum(power, 1e-30))
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def run_fista_multiscale(a_init, S, max_time, rng,
                         beta_init=20.0, beta_rate=0.002, beta_max=10000.0,
                         k_frac_min=0.01, k_frac_max=0.15):
    """
    FISTA with adaptive multi-scale LSE.
    Adapts k (number of peaks) and beta (sharpening) over time.
    Uses excess-weighted gradient as a blend with top-k LSE.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        # Adaptive parameters
        # Early: broad exploration (large k, small beta)
        # Late: focused refinement (small k, large beta)
        k_frac = k_frac_max - (k_frac_max - k_frac_min) * progress
        k = max(2, int(k_frac * (2 * n - 1)))
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-10:
            best = cur_max
            best_a = a.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        # Compute gradient: blend of top-k LSE and excess-weighted
        grad1 = lse_gradient(a, c, k, beta)
        
        # Excess-weighted (broader)
        k2 = max(k, int(0.3 * (2 * n - 1)))
        grad2 = excess_weighted_gradient(a, c, 0.5, 1.0)
        
        # Blend: more weight on top-k as we progress
        blend = 0.3 * progress + 0.1
        grad = blend * grad1 + (1 - blend) * grad2

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / max(2.0 * beta * 0.05, 1e-10)
        step = min(step, 1.0)  # Cap step size
        x_new = project_simplex(a - step * grad, S)
        accepted = False
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-12:
                accepted = True
                break
            step *= 0.5
            x_new = project_simplex(a - step * grad, S)

        a = x_new

        # Periodic restart when stuck
        if stuck > 800:
            noise_scale = 0.003 * S / np.sqrt(n) * (1 - progress * 0.5)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0

        # Periodic spectral perturbation
        if it % 1500 == 0 and rng.random() < 0.25 and progress < 0.8:
            a = spectral_perturb(a, 0.05 + 0.05 * (1 - progress), rng)
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_frank_wolfe(a_init, S, max_time, rng, k=50):
    """
    Frank-Wolfe (conditional gradient) with subgradient.
    Very cheap per step: O(n log n) for FFT + O(n) for FW step.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0

    # Subgradient memory (exponential averaging)
    grad_ema = np.zeros(n)
    ema_decay = 0.95

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-10:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Compute subgradient: average over top-k peaks
        k_eff = max(2, int(k * (1 + progress)))  # Focus more as we progress
        idx = np.argsort(c)[::-1][:k_eff]
        
        # Gradient at each peak
        # grad_k[i] = a[i+k] + a[i-k] (with bounds)
        grad = np.zeros(n)
        for k_idx in idx:
            kk = int(k_idx)
            g = np.zeros(n)
            if kk == 0:
                g[:] = 2.0 * a
            else:
                # a[i+k] contributes to position i
                if kk < n:
                    g[:n-kk] += a[kk:]  # a[i+kk] for i=0..n-1-kk
                    g[kk:] += a[:n-kk]  # a[i-kk] for i=kk..n-1
            grad += g
        grad /= k_eff  # Average subgradient

        # EMA of gradient
        grad_ema = ema_decay * grad_ema + (1 - ema_decay) * grad

        # Frank-Wolfe: min grad_ema^T x subject to x >= 0, sum(x) = S
        # Solution: put all mass on argmin(grad_ema)
        min_idx = np.argmin(grad_ema)
        s_vertex = np.zeros(n)
        s_vertex[min_idx] = S

        # Step size: t = 2/(it+2)
        t = 2.0 / (it + 2)
        
        # Overshoot: try t*2 as well
        a_new1 = (1 - t) * a + t * s_vertex
        a_new2 = (1 - 2*t) * a + 2*t * s_vertex if 2*t <= 1 else a_new1
        
        c1 = fft_convolve_self(a_new1)
        best_new = a_new1
        best_val = c1.max()
        if 2*t <= 1:
            c2 = fft_convolve_self(a_new2)
            if c2.max() < best_val:
                best_new = a_new2
                best_val = c2.max()
        
        if best_val < cur_max:
            a = best_new
        else:
            a = a_new1  # Still take the step (FW converges without line search)

        # Periodic perturbation
        if stuck > 2000 and rng.random() < 0.1:
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            grad_ema = np.zeros(n)
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_spectral_opt(a_init, S, max_time, rng):
    """
    Spectral domain optimization: iteratively flatten the power spectrum
    while maintaining non-negativity in the time domain.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        # Decrease the flattening strength over time
        alpha = 0.3 * (1 - progress) + 0.01

        a = spectral_flatten(a, alpha, rng)

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Occasionally do a time-domain gradient step
        if it % 10 == 0:
            k = max(4, int(0.05 * (2 * n - 1)))
            beta = 100.0
            grad = lse_gradient(a, c, k, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-14:
                step = 0.1 / (1 + 0.01 * it)
                a_new = project_simplex(a - step * grad, S)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < cur_max:
                    a = a_new
                    c = c_new
                    cur_max = c.max()
                    if cur_max < best:
                        best = cur_max
                        best_a = a.copy()

    return best_a, best


def run_polish_high_beta(a_init, S, max_time, rng):
    """Very high-beta polishing with careful step control."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    k = max(2, int(0.02 * (2 * n - 1)))
    it = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(100000.0, 500.0 * (1.0 + it / 200.0))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()

        grad = lse_gradient(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        base_step = 0.3 / (1.0 + 0.005 * it)
        step = base_step * step_scale

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
            step_scale = min(3.0, step_scale * 1.05)
        else:
            step_scale = max(0.01, step_scale * 0.3)
            # Try even smaller step
            a_new = project_simplex(a - step * 0.1 * grad, S)
            a = a_new

        # Tiny perturbation to escape plateaus
        if it % 1000 == 0 and (time.monotonic() - start) / max_time < 0.7:
            noise_scale = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)
