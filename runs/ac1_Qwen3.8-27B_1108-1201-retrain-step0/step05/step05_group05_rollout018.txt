```python
def compute_subgradient(a, w, active_lags):
    """
    Compute g[j] = 2 * sum_k w[k] * a[k-j] where k ranges over active_lags.
    w is indexed by position in active_lags.
    """
    n = len(a)
    g = np.zeros(n)
    for idx, k in enumerate(active_lags):
        if w[idx] < 1e-15:
            continue
        # a[k-j] for j in [0, n): valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_min = max(0, k - n + 1)

            g[j_min:j_max+1] += 2.0 * w[idx] * a_slice
    return g
```

```python
        # For each s, compute a[k - support[s]]
        idx = k - support  # indices into a
        valid = (idx >= 0) & (idx < n)
        G[valid, ki] = 2.0 * a[idx[valid]]
    
    # Center G: G_c = G - mean(G, axis=0)
    G_mean = G.mean(axis=0, keepdims=True)
    G_c = G - G_mean
    
    # H = G_c^T @ G_c
    H = G_c.T @ G_c
    
    return H
```

```python
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, w, active_lags):
    """
    Compute g[j] = 2 * sum_k w[k_idx] * a[k - j] where k ranges over active_lags.
    w is indexed by position in active_lags.
    """
    n = len(a)
    g = np.zeros(n)
    for idx, k in enumerate(active_lags):
        wi = w[idx]
        if wi < 1e-15:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            a_slice = a[k - j_max: k - j_min + 1][::-1]
            g[j_min: j_max + 1] += 2.0 * wi * a_slice
    return g


def solve_weight_qp(H, w_init=None):
    """Solve min w^T H w s.t. w >= 0, sum(w) = 1 using projected gradient."""
    n = H.shape[0]
    if n == 1:
        return np.array([1.0])
    if w_init is None:
        w = np.ones(n) / n
    else:
        w = np.asarray(w_init, dtype=np.float64).copy()
        w = np.maximum(w, 0)
        s = w.sum()
        if s < 1e-15:
            w = np.ones(n) / n
        else:
            w = w / s

    L = 2.0 * (np.abs(np.linalg.eigvalsh(H)).max() + 1e-10)
    obj_best = w @ H @ w
    for it in range(300):
        grad = 2.0 * (H @ w)
        w_new = w - (1.0 / L) * grad
        w_new = project_simplex(w_new, 1.0)
        obj_new = w_new @ H @ w_new
        if abs(obj_new - obj_best) < 1e-14 * max(1.0, obj_best):
            w = w_new
            break
        w = w_new
        obj_best = obj_new
    return w


def qp_subgradient_method(a, S, max_time, rng, tol_base=1e-4, max_active=40,
                          step_init=0.05, step_decay=0.9999, min_step=1e-8,
                          stall_iters=500, perturb_scale=0.003):
    """
    Main QP-guided subgradient optimization loop.
    Minimize max(conv(a,a)) subject to a >= 0, sum(a) = S.
    """
    n = len(a)
    a = project_simplex(a.copy(), S)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_init
    w_prev = None
    tol_mult = 1.0
    tol_cycle = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive tolerance: cycle through a few values to escape plateaus
        tol_cycle = (it // 2000) % 5
        tol_values = [1e-5, 1e-4, 1e-3, 5e-3, 1e-2]
        tol = tol_values[tol_cycle] * tol_base * (1.0 + 0.5 * tol_cycle)

        cmax = c.max()
        active = np.where(c >= cmax - tol * cmax)[0]
        if len(active) > max_active:
            idx = np.argsort(c[active])[::-1][:max_active]
            active = active[idx]
        na = len(active)

        if na == 1:
            # Single active lag: just use that one
            w = np.array([1.0])
        else:
            # Compute H matrix for the QP
            # H[i,j] = sum_j' G[j',i] * (G[j',j] - mean_j' G[j',j'])
            # where G[j,k] = 2*a[active[k]-j] for j in [0,n)
            # For efficiency, if support is large, use full range
            support = np.where(a > 1e-12 * a.max())[0]
            if len(support) > 0.9 * n:
                support = np.arange(n)
            ns = len(support)

            # Form G (ns x na)
            G = np.zeros((ns, na), dtype=np.float64)
            for ki, k in enumerate(active):
                idx_arr = k - support
                valid = (idx_arr >= 0) & (idx_arr < n)
                G[valid, ki] = 2.0 * a[idx_arr[valid]]

            G_mean = G.mean(axis=0, keepdims=True)
            G_c = G - G_mean
            H = G_c.T @ G_c

            # Solve QP
            w = solve_weight_qp(H, w_init=w_prev)
            w_prev = w

        # Compute subgradient
        g = compute_subgradient(a, w, active)
        gnorm_sq = g @ g
        if gnorm_sq < 1e-30:
            break

        # Step size with geometric decay
        step = max(min_step, step * step_decay)

        # Take step
        a_new = project_simplex(a - step * g, S)

        # Evaluate
        c = fft_convolve_self(a_new)
        a = a_new

        if c.max() < best:
            best = c.max()
            best_a = a.copy()
            last_improve = it
            # Slightly increase step on improvement
            step = min(step_init * 0.5, step * 1.05)

        # Stall detection: perturb to escape local minimum
        if it - last_improve > stall_iters:
            noise_scale = perturb_scale * S / np.sqrt(n) * np.exp(-(it - last_improve) / (2 * stall_iters))
            if noise_scale > 1e-10 * S:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                # Also try a Fourier-domain perturbation
                if it - last_improve > stall_iters * 3:
                    fsize = 1 << (2 * n - 1).bit_length()
                    A = fft(a, fsize)
                    mag = np.abs(A)
                    phase = np.angle(A)
                    freqs = np.arange(fsize, dtype=np.float64)
                    weight = 1.0 + 5.0 * (freqs / fsize) ** 2
                    scale = 0.1 * np.exp(-(it - last_improve) / (5 * stall_iters))
                    phase_noise = rng.normal(0, scale * weight, fsize)
                    new_A = mag * np.exp(1j * (phase + phase_noise))
                    new_A[0] = A[0]
                    a = project_simplex(np.real(ifft(new_A, fsize))[:n], S)
                    c = fft_convolve_self(a)

            t_mom_reset = it
            w_prev = None
            last_improve = it

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def quick_fista(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.005, beta_max=5000.0):
    """Fast FISTA for screening."""
    n = len(a_init)
    a = project_simplex(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        cm = c.max()
        scaled = beta * (c - cm)
        if scaled.max() > 400:
            beta = 400.0 / max(1e-15, (c - cm).max())
            scaled = beta * (c - cm)
        e = np.exp(scaled)
        tot = e.sum()
        if tot < 1e-300:
            break
        w_lse = e / tot
        fsize = 1 << (len(w_lse) + n - 1).bit_length()
        W = fft(w_lse, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (3.0 * beta)
        if it - last_improve > 2000:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def fourier_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + 3.0 * (freqs / fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def basin_hopping(a_init, S, max_time, rng):
    """Basin hopping with quick local optimization."""
    n = len(a_init)
    a = project_simplex(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    opt_time = max(5.0, min(30.0, max_time / 20.0))
    n_iters = max(1, int((max_time - opt_time) / (opt_time + 2)))
    population = [(best, a.copy())]

    for it in range(n_iters):
        if time.monotonic() - start > max_time - opt_time:
            break
        progress = it / max(1, n_iters)
        scale = (0.3 * np.exp(-1.5 * progress) + 0.03) * (1.0 + 0.5 * rng.random())

        r = rng.random()
        if len(population) >= 2 and r < 0.25:
            i1 = rng.integers(len(population))
            i2 = rng.integers(len(population))
            if i1 != i2:
                # Simple interpolation + mutation
                alpha = rng.uniform(0.2, 0.8)
                a = project_simplex(alpha * population[i1][1] + (1 - alpha) * population[i2][1], S)
                a = fourier_mutate(a, rng, scale * 0.5)
            else:
                a = fourier_mutate(a, rng, scale)
        else:
            a = fourier_mutate(a, rng, scale)

        a_opt, val = quick_fista(a, S, opt_time, rng,
                                  beta_init=20 + 30 * it, beta_rate=0.004, beta_max=3000)
        if val < best:
            best = val
            best_a = a_opt.copy()
            print("    BH iter %d: new best max_b=%.8f" % (it, val))

        population.append((val, a_opt.copy()))
        if len(population) > 6:
            population.sort(key=lambda x: x[0])
            population = population[:6]

    return best_a, best


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        inits.append(np.clip(given_seq, 0, 1000).copy())

    # Various window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.5461 + 0.4536 * np.cos(2 * np.pi * tt)
                            + 0.0830 * np.cos(4 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 3 + 0.01)
    inits.append(np.exp(-3.0 * (tt - 0.5) ** 2 / 0.15) + 0.01)
    inits.append(np.maximum(1.0 + 0.2 * np.sin(4 * np.pi * tt) + 0.1 * np.cos(12 * np.pi * tt), 0.01))
    inits.append(rng.random(n) * 0.5 + 0.25)

    # Block structures
    for nb in [8, 16, 32]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.3, 1.2, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n] + 0.01)

    return inits


def multi_resolution_search(n_target, S_target, max_time, rng, given_seq=None):
    """
    Start at low resolution, optimize, then upsample to target resolution.
    """
    resolutions = []
    r = max(64, n_target // 64)
    while r < n_target:
        r = min(r * 2, n_target)
        resolutions.append(r)
    resolutions.append(n_target)

    best_a = None
    best_val = np.inf

    for n_curr in resolutions:
        S_curr = np.sqrt(2.0 * n_curr)
        time_remaining = max_time - (time.monotonic() - multi_resolution_search._start)
        if time_remaining < 60:
            break

        if best_a is not None:
            # Upsample from previous resolution
            a_init = np.interp(np.linspace(0, 1, n_curr, endpoint=False),
                               np.linspace(0, 1, len(best_a), endpoint=False),
                               best_a)
            a_init = np.maximum(a_init, 0.0)
        elif given_seq is not None and n_curr == n_target:
            a_init = np.clip(given_seq[:n_curr], 0, 1000)
            if len(a_init) < n_curr:
                a_init = np.concatenate([a_init, np.zeros(n_curr - len(a_init))])
        else:
            tt = np.linspace(0, 1, n_curr, endpoint=False)
            a_init = np.sin(np.pi * tt) ** 2 + 0.01

        # Optimize at this resolution
        time_alloc = min(60.0, time_remaining * 0.3)
        a_opt, val = qp_subgradient_method(a_init, S_curr, time_alloc, rng,
                                            tol_base=1e-3, max_active=30,
                                            step_init=0.03, step_decay=0.99995)
        if val < best_val:
            best_val = val
            best_a = a_opt.copy()

    return best_a, best_val


def propose_candidate(seed=42, budget_s=1800):

    1. Diverse initialization + quick screening (~60s)
    2. Basin hopping with Fourier mutations (~200s)
    3. Intensive QP-guided subgradient optimization (~1400s)
    4. Final polish and verification (~140s)
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30

    print("Starting optimization, budget=%.0fs" % budget_s)

    n = 8192
    S = np.sqrt(2.0 * n)

    # Load given sequence if available
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            # Pad or trim to length n
            if len(given) < n:
                given = np.concatenate([given, np.zeros(n - len(given))])
            else:
                given = given[:n]
            print("Given sequence: len=%d, sum=%.2f, obj=%.6f" % (
                len(given), given.sum(),
                2.0 * n * fft_convolve_self(given).max() / (given.sum() ** 2) if given.sum() > 0.01 else np.inf))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def check_update(a, val):
        nonlocal global_best, global_a
        obj = 2.0 * n * val / (a.sum() ** 2)
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("    NEW BEST: obj=%.8f (max_b=%.8f, sum=%.4f)" % (obj, val, a.sum()))
            return True
        return False

    # Phase 1: Diverse initialization + quick screening
    print("\n=== Phase 1: Quick screening (~60s) ===")
    p1_start = time.monotonic()
    p1_budget = min(60.0, (deadline - p1_start) * 0.04)
    inits = generate_inits(n, rng, given_seq=given)
    n_screen = min(6, len(inits))
    per_init_time = p1_budget / n_screen

    candidates = []
    for i in range(n_screen):
        if time.monotonic() > deadline - 300:
            break
        t_alloc = min(per_init_time, deadline - time.monotonic() - 300)
        if t_alloc < 5:
            break
        a_opt, val = quick_fista(inits[i], S, t_alloc, rng,
                                  beta_init=10.0, beta_rate=0.006, beta_max=3000)
        obj = 2.0 * n * val / (a_opt.sum() ** 2)
        candidates.append((obj, a_opt.copy()))
        print("  Screen %d: obj=%.6f" % (i, obj))
        check_update(a_opt, val)

    candidates.sort(key=lambda x: x[0])
    print("  Best after screening: obj=%.6f" % global_best)

    # Phase 2: Basin hopping
    print("\n=== Phase 2: Basin hopping (~200s) ===")
    p2_start = time.monotonic()
    p2_budget = min(200.0, (deadline - p2_start) * 0.12)

    bh_starts = min(2, len(candidates))
    for i in range(bh_starts):
        if time.monotonic() > deadline - 500:
            break
        t_alloc = min(p2_budget / (bh_starts - i), deadline - time.monotonic() - 500)
        if t_alloc < 30:
            break
        print("  BH from candidate %d (obj=%.6f), alloc=%.0fs" % (i, candidates[i][0], t_alloc))
        a_bh, val_bh = basin_hopping(candidates[i][1], S, t_alloc, rng)
        check_update(a_bh, val_bh)
        print("  BH %d done: obj=%.6f" % (i, 2.0 * n * val_bh / (a_bh.sum() ** 2)))

    # Phase 3: Intensive QP-guided optimization
    print("\n=== Phase 3: QP-guided optimization (~1400s) ===")
    p3_start = time.monotonic()
    p3_budget = deadline - p3_start - 140

    if global_a is not None and p3_budget > 60:
        # Run multiple rounds with different parameters
        configs = [
            # (tol_base, max_active, step_init, step_decay, stall_iters, perturb_scale)
            (1e-4, 40, 0.04, 0.99995, 400, 0.003),
            (1e-5, 20, 0.02, 0.99997, 600, 0.002),
            (1e-3, 60, 0.06, 0.99993, 300, 0.004),
            (5e-5, 30, 0.03, 0.99996, 500, 0.0025),
        ]

        n_rounds = min(4, max(1, int(p3_budget / 300)))
        for r in range(n_rounds):
            if time.monotonic() > deadline - 140:
                break
            cfg = configs[r % len(configs)]
            t_alloc = min(p3_budget / (n_rounds - r), deadline - time.monotonic() - 140)
            if t_alloc < 30:
                break
            print("  Round %d/%d: tol=%.0e, max_act=%d, step=%.3f, alloc=%.0fs" %
                  (r + 1, n_rounds, cfg[0], cfg[1], cfg[2], t_alloc))
            a_opt, val = qp_subgradient_method(
                global_a, S, t_alloc, rng,
                tol_base=cfg[0], max_active=cfg[1],
                step_init=cfg[2], step_decay=cfg[3],
                stall_iters=cfg[4], perturb_scale=cfg[5])
            check_update(a_opt, val)
            print("  After round %d: obj=%.8f" % (r + 1, global_best))

    # Phase 4: Final polish
    print("\n=== Phase 4: Final polish ===")
    p4_start = time.monotonic()
    p4_budget = deadline - p4_start - 10

    if global_a is not None and p4_budget > 20:
        # Very tight tolerance for final precision
        a_opt, val = qp_subgradient_method(
            global_a, S, p4_budget, rng,
            tol_base=1e-6, max_active=15,
            step_init=0.01, step_decay=0.99999,
            stall_iters=800, perturb_scale=0.001)
        check_update(a_opt, val)
        print("  After final polish: obj=%.8f" % global_best)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.8f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
```
