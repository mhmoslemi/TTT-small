```python
def lp_based_step(a, M, max_constraints=200, max_iter=50):
    """One step of the LP-based alternating method using constraint generation."""
    n = len(a)
    # Start with top constraints
    conv = np.convolve(a, a)
    active = np.argsort(conv)[::-1][:max_constraints]
    
    for iter in range(max_iter):
        # Build LP
        # Variables: g[0], ..., g[n-1]
        # Objective: maximize sum(g) -> minimize -sum(g)
        # Constraints: for k in active: sum_j a[k-j]*g[j] <= M
        #              g[j] >= 0
        
        c = np.zeros(n)
        c[:] = -1  # minimize -sum(g)
        
        A_ub = np.zeros((len(active), n))
        for i, k in enumerate(active):
            for j in range(max(0, k-n+1), min(n, k+1)):
                A_ub[i, j] = a[k-j]
        b_ub = np.full(len(active), M)
        
        # bounds: g >= 0
        bounds = [(0, None)] * n
        
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not result.success:
            break
        
        g = result.x
        
        # Check all constraints
        new_conv = np.convolve(a, g)
        violated = np.where(new_conv > M + 1e-10)[0]
        if len(violated) == 0:
            break
        # Add violated constraints
        active = np.unique(np.concatenate([active, violated]))
        if len(active) > 500:
            # Keep only the most violated
            over = new_conv[active] - M
            top_idx = np.argsort(over)[::-1][:300]
            active = active[top_idx]
    
    return g
```

from scipy.fft import fft, ifft
from scipy.optimize import linprog

def compute_cross(a, b, N):
    """Cross-convolution a*b via FFT."""
    n1 = len(a)
    n2 = len(b)
    if N < n1 + n2 - 1:
        N = 1 << (n1 + n2 - 1).bit_length()
    A = fft(a, N)
    B = fft(b, N)
    c = np.real(ifft(A * B))[:n1 + n2 - 1]

            a_try = (1 - t) * a + t * g_rescaled
            a_try = safe_normalize(a_try, n, S)
            cm = compute_conv(a_try, N).max()
            if cm < best_local_cm:
                best_local_cm = cm
                best_local = a_try

        if best_local is not None:
            a = best_local
            if best_local_cm < best_cmax:
                best_cmax = best_local_cm
                best_a = a.copy()
                stall_count = 0
                if verbose and it % 5 == 0:
                    print("  LP it=%d: peak=%.6f score=%.6f" %
                          (it, best_cmax, 2 * n * best_cmax / (S * S)))
        else:
            stall_count += 1
            if stall_count > 4:
                # Spectral restart
                A = fft(a, N)
                mag = np.abs(A)
                phase = rng.uniform(0, 2 * np.pi, N)
                a_new = np.real(ifft(mag * np.exp(1j * phase)))[:n]
                a_new = np.maximum(a_new, 0.0)
                a = safe_normalize(a_new, n, S)
                stall_count = 0

    return best_a, best_cmax


def tarpit_grad(a, N, top_k):
    """
    Compute gradient targeting top-K peaks of self-convolution.
    Returns (grad, c, c_max).
    """
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n), c, c_max
    weights = top_vals / total

    # Compute gradient: for each peak at lag k, the gradient of (a*a)[k] w.r.t. a
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += w * a[idx]
                if j != idx:
                    grad[j] += w * a[idx]

    return grad, c, c_max


def tarpit_block(a_init, n, S, N, max_time, rng, top_k_init=10, top_k_max=100):
    """
    Tarpit method: iterate top-K peak gradient descent with line search.
    Periodically inject LP-based kicks to escape local minima.
    """
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        rem = max_time - (time.monotonic() - start)
        if rem < 2:
            break

        top_k = min(top_k_max, top_k_init + it // 30)
        grad, c, c_max = tarpit_grad(a, N, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        dir_norm = gnorm
        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        a_new = a
        cm_new = c_max
        for ls in range(15):
            candidate = a - step * grad / dir_norm
            candidate = project_simplex(candidate, S)
            cm_ls = compute_conv(candidate, N).max()
            if cm_ls <= c_max * (1 - 1e-10) or ls == 14:
                a_new = candidate
                cm_new = cm_ls
                break
            step *= 0.6

        if cm_new < best_cmax:
            best_cmax = cm_new
            best_a = a_new.copy()
            stall = 0
        else:
            a_new = project_simplex(a_new, S)
            cm_new = compute_conv(a_new, N).max()
            if cm_new < best_cmax:
                best_cmax = cm_new
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1

        a = a_new

        # Periodic LP kick
        if it % 50 == 0 and rem > 100:
            M_cur = compute_conv(a, N).max()
            g = lp_solve_with_generation(a, n, S, M_cur * 1.003,
                                         max_active=200, max_gen=5)
            if g is not None:
                s_g = g.sum()
                if s_g > 1e-15:
                    g_r = g * (S / s_g)
                    a_kick = 0.7 * a + 0.3 * g_r
                    a_kick = safe_normalize(a_kick, n, S)
                    cm_kick = compute_conv(a_kick, N).max()
                    if cm_kick < best_cmax * 1.01:
                        a = a_kick
                        if cm_kick < best_cmax:
                            best_cmax = cm_kick
                            best_a = a.copy()

        # Spectral restart if stuck
        if stall > 10:
            frac = 0.3 + 0.1 * (stall % 3)
            A = fft(a, N)
            mag = np.abs(A)
            old_phase = np.angle(A)
            n_rand = max(1, int(N * frac))
            rand_idx = rng.choice(N, size=n_rand, replace=False)
            new_phase = old_phase.copy()
            new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
            a_new = np.real(ifft(mag * np.exp(1j * new_phase)))[:n]
            a_new = np.maximum(a_new, 0.0)
            a = safe_normalize(a_new, n, S)
            stall = 0

        if it % 100 == 0:
            score = 2 * n * best_cmax / (S * S)
            print("  Tarpit it=%d: best_peak=%.6f score=%.6f stall=%d" %
                  (it, best_cmax, score, stall))

    return best_a, best_cmax


def lse_block(a_init, n, S, N, max_time, rng, beta_init=20.0, beta_max=500.0):
    """L-BFGS on LSE-smoothed peak (from previous approach, improved)."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    beta = beta_init
    start = time.monotonic()
    it = 0
    s_hist = []
    y_hist = []
    m_lbfgs = 12

    def lse_grad_inner(a_cur, beta_val):
        n_cur = len(a_cur)
        c = compute_conv(a_cur, N)
        c_max = c.max()
        exp_bc = np.exp(beta_val * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
        else:
            weights = exp_bc / total
        fsize = 1 << (3 * n_cur - 2).bit_length()
        W = fft(weights, fsize)
        Af = fft(a_cur, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n_cur]
        return grad, c_max

    g, c_max = lse_grad_inner(a, beta)
    step_scale = 2.0 * S / (n * max(beta, 1.0))

    while time.monotonic() - start < max_time:
        it += 1
        if it % 200 == 0:
            beta = min(beta_max, beta * 1.08)
            step_scale = 2.0 * S / (n * max(beta, 1.0))

        # Simple gradient step with line search (simpler than full L-BFGS)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-16:
            break
        direction = -g / gnorm

        alpha = step_scale
        a_new = a
        cm_new = c_max
        for ls in range(12):
            candidate = project_simplex(a + alpha * direction, S)
            cm_ls = compute_conv(candidate, N).max()
            if cm_ls <= c_max:
                a_new = candidate
                cm_new = cm_ls
                break
            alpha *= 0.5
        else:
            a_new = project_simplex(a + alpha * direction, S)
            cm_new = compute_conv(a_new, N).max()

        if cm_new < best_cmax:
            best_cmax = cm_new
            best_a = a_new.copy()

        g_new, c_new = lse_grad_inner(a_new, beta)
        s_k = a_new - a
        y_k = g_new - g
        a = a_new
        g = g_new
        c_max = c_new

        if it % 300 == 0:
            score = 2 * n * best_cmax / (S * S)
            print("  LSE it=%d: beta=%.0f peak=%.6f score=%.6f" %
                  (it, beta, best_cmax, score))

    return best_a, best_cmax


def coord_descent_block(a_init, n, S, N, max_time, rng):
    """
    Coordinate descent: find the single best coordinate move to reduce the peak.
    Writes mass from high-contributing positions to low-contributing ones.
    """
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()

        k_star = int(np.argmax(c))
        # Contribution to peak from each position
        contrib = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                contrib[j] = a[idx]
                if j != idx:
                    contrib[j] = 2 * a[idx]

        # Find best donor (high contrib) and receiver (low contrib)
        # Try to move mass from top contributor to least contributor
        top_donors = np.argsort(contrib)[::-1][:20]
        bottom_recv = np.argsort(contrib)[:20]

        best_improve = 0
        best_move = None
        for d in top_donors:
            if a[d] < 1e-10:
                continue
            for r in bottom_recv:
                if r == d:
                    continue
                # Try moving a fraction of a[d] to a[r]
                for frac in [0.01, 0.05, 0.1]:
                    amt = frac * a[d]
                    a_try = a.copy()
                    a_try[d] -= amt
                    a_try[r] += amt
                    if a_try[d] < 0 or a_try[r] > 1000:
                        continue
                    cm = compute_conv(a_try, N).max()
                    if c_max - cm > best_improve:
                        best_improve = c_max - cm
                        best_move = (d, r, amt)

        if best_move is not None:
            d, r, amt = best_move
            a[d] -= amt
            a[r] += amt
            a = safe_normalize(a, n, S)
        else:
            # No improvement found; try small random perturbation
            if it % 50 == 0:
                idx = rng.integers(n)
                amt = 0.001 * S / n
                if a[idx] > amt:
                    a[idx] -= amt
                    j2 = rng.integers(n)
                    a[j2] += amt
                    a = safe_normalize(a, n, S)
            else:
                break  # Stuck

        if it % 200 == 0:
            score = 2 * n * best_cmax / (S * S)
            print("  CoordDec it=%d: peak=%.6f score=%.6f" % (it, best_cmax, score))

    return best_a, best_cmax


def spectral_restart(a, N, rng, frac=1.0):
    """Randomize FFT phases (fraction frac) while preserving magnitudes."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    if frac >= 1.0:
        new_phase = rng.uniform(0, 2 * np.pi, N)
    else:
        rand_idx = rng.choice(N, size=n_rand, replace=False)
        new_phase = old_phase.copy()
        new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    a_new = np.real(ifft(mag * np.exp(1j * new_phase)))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    L = n
    i = np.arange(L)
    c = rng.integers(1, L)
    phase = -np.pi * i * (2 * i + 1 - c) / L
    a = 1.0 + 0.7 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_fourier(n, rng, N):
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def deflation_init(n, S, N, n_iters=2000, rng=None):
    """Constructive initialization by iteratively reducing the peak from all-ones."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c = compute_conv(a, N)
        k_star = int(np.argmax(c))
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
        if np.max(contrib) < 1e-15:
            break
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if i_max == i_min:
            break
        transfer = min(0.02 * a[i_max], 0.3 * a[i_max])
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses hybrid LP-alternating + tarpit gradient + LSE polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    # Choose n
    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if 1024 <= given.size <= 32768:
            n = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    def score_from_peak(peak):
        return 2.0 * n * peak / (S * S)

    def update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = safe_normalize(a_candidate, n, S)
        c = compute_conv(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            print("  [%s] peak=%.6f score=%.6f" % (label, sc, score_from_peak(sc)))
        return sc

    # Try to load previous best
    a_start = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            a_start = np.clip(given, 0, 1000).astype(np.float64)
            a_start = safe_normalize(a_start, n, S)
            update_best(a_start, "given")
            print("  Loaded previous best: score=%.6f" % score_from_peak(global_best))
    except Exception:
        pass

    if a_start is None:
        a_start = np.ones(n) * S / n
        update_best(a_start, "ones")

    # === Phase 1: LP-based alternating (most novel) ===
    print("\n=== Phase 1: LP-based alternating ===")
    t1 = time.monotonic()
    phase1_budget = min(700.0, (deadline - t1) * 0.30)
    print("  Budget: %.0fs" % phase1_budget)

    a_l, sc_l = lp_block(a_start, n, S, phase1_budget, rng, verbose=True)
    update_best(a_l, "LP_block")

    # === Phase 2: Tarpit gradient ===
    print("\n=== Phase 2: Tarpit gradient ===")
    t2 = time.monotonic()
    phase2_budget = min(700.0, (deadline - t2) * 0.30)
    print("  Budget: %.0fs" % phase2_budget)

    a_t, sc_t = tarpit_block(global_a, n, S, N, phase2_budget, rng,
                             top_k_init=5, top_k_max=80)
    update_best(a_t, "tarpit")

    # === Phase 3: LSE polishing ===
    print("\n=== Phase 3: LSE polishing ===")
    t3 = time.monotonic()
    phase3_budget = min(400.0, (deadline - t3) * 0.20)
    print("  Budget: %.0fs" % phase3_budget)

    a_lse, sc_lse = lse_block(global_a, n, S, N, phase3_budget, rng,
                              beta_init=30.0, beta_max=600.0)
    update_best(a_lse, "LSE")

    # === Phase 4: Coordinate descent fine-tuning ===
    print("\n=== Phase 4: Coord descent ===")
    t4 = time.monotonic()
    phase4_budget = min(200.0, (deadline - t4) * 0.10)
    print("  Budget: %.0fs" % phase4_budget)

    if phase4_budget > 20:
        a_cd, sc_cd = coord_descent_block(global_a, n, S, N, phase4_budget, rng)
        update_best(a_cd, "coorddec")

    # === Phase 5: Final LP polish ===
    print("\n=== Phase 5: Final LP polish ===")
    t5 = time.monotonic()
    remaining5 = deadline - t5 - 15
    if remaining5 > 30:
        a_lp2, sc_lp2 = lp_block(global_a, n, S, min(remaining5, 200.0), rng, verbose=True)
        update_best(a_lp2, "LP_final")

    # === Phase 6: Tarpit + spectral restarts on remaining time ===
    t6 = time.monotonic()
    remaining6 = deadline - t6 - 10
    if remaining6 > 30:
        for attempt in range(3):
            if remaining6 - (time.monotonic() - t6) < 20:
                break
            a_sr = spectral_restart(global_a, N, rng, frac=0.3 + 0.2 * attempt)
            a_tr, sc_tr = tarpit_block(a_sr, n, S, N, min(remaining6 / 3.0 - (time.monotonic() - t6), 100.0),
                                       rng, top_k_init=5, top_k_max=50)
            update_best(a_tr, "restart_%d" % attempt)

    # === Verification ===
    print("\n=== Final Verification ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            if np.isfinite(ver):
                print("Verification score: %.6f" % ver)
            else:
                print("Verification FAILED (inf), using fallback")
                global_a = np.ones(256, dtype=np.float64)
        except Exception as e:
            print("Verification error: %s, using given" % str(e))
            try:
                given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
                global_a = np.clip(given, 0, 1000)
            except Exception:
                global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    final_score = score_from_peak(global_best) if global_best < np.inf else float('inf')
    print("Returning length %d, peak=%.6f, approx score=%.6f" %
          (len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
