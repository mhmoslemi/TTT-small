def compute_conv_p(p):
    """Compute linear self-convolution of probability distribution p."""
    n = len(p)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    P = fft(p, fsize)
    c = np.real(ifft(P * P))
    return c[:size]

def lse_objective_and_grad(x, beta):
    """Compute LSE objective and gradient w.r.t. x (log-softmax parameterization)."""
    # Softmax
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    p = e / e.sum()
    
    n = len(p)
    # Convolution
    f = compute_conv_p(p)
    
    # LSE
    f_max = f.max()
    shifted = beta * (f - f_max)
    shifted = shifted - shifted.max()
    e_weights = np.exp(shifted)
    w_sum = e_weights.sum()
    w = e_weights / w_sum
    lse_value = f_max + np.log(w_sum) / beta
    
    # Gradient
    # g[m] = sum_k w[k] * p[k-m] (cross-correlation of w and p)
    # = convolve(w, p[::-1]) with appropriate slicing
    # Actually: g[m] = sum_k w[k] * p[k-m]
    # Let's compute via FFT: FFT(g) = conj(FFT(w)) * FFT(p) ... no
    # 
    # g[m] = sum_k w[k] * p[k-m]
    # This is the cross-correlation: (w * p_rev)[m + n - 1] where p_rev = p[::-1]
    # In full convolution: conv(w, p_rev)[m + n - 1] = sum_k w[k] * p_rev[m + n - 1 - k]
    # = sum_k w[k] * p[n-1 - m - n + 1 + k] = sum_k w[k] * p[k-m]
    # Yes! So g = convolve(w, p[::-1])[n-1:n-1+n] (take the middle n elements)
    
    size = w.size + p.size - 1  # = (2n-1) + n - 1 = 3n-2

    return lse_value, grad
