"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Key insight: asymmetric (ramp) initializations give ~1.33 vs ~1.5 for symmetric ones.
Strategy: Start from ramp variants, optimize with FISTA+LSE, refine with coordinate descent.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, beta, n):
    """Compute LSE value, gradient, and convolution."""
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def topk_subgradient(a, c, n, k):
    """Subgradient from top-k lags of the convolution - no FFT needed."""
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for kk in idx:
        kk = int(kk)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if hi >= lo:
            j_indices = np.arange(lo, hi + 1)
            partner = kk - j_indices
            grad[j_indices] += a[partner]
            # Also the "other side" contribution
            lo2 = max(0, kk - (n - 1))
            hi2 = min(n - 1, kk)
            if hi2 >= lo2:
                j2 = np.arange(lo2, hi2 + 1)
                partner2 = kk - j2
                # Avoid double counting when the two ranges overlap
                # Actually for full convolution, c[k] = sum_j a[j]*a[k-j]
                # d(c[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j-k+n-1]... 
                # Let me just use the direct formula
                pass
        # Use direct formula: d(c[k])/d(a[j]) = a[k-j] if 0<=k-j<n, else 0
        # plus a[j-k+n-1] if 0<=j-k+n-1<n... 
        # Actually for np.convolve, c[k] = sum_{j} a[j]*a[k-j] where both indices valid
        # d(c[k])/d(a[j]) = a[k-j] (if 0 <= k-j < n)
        # There's no "other side" - it's a single sum
    # Redo properly
    grad = np.zeros(n)
    for kk in idx:
        kk = int(kk)
        # c[kk] = sum_{j=max(0,kk-n+1)}^{min(n-1,kk)} a[j]*a[kk-j]
        # d(c[kk])/d(a[j]) = a[kk-j] if 0 <= kk-j < n, i.e., kk-n+1 <= j <= kk
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if hi >= lo:
            j_range = np.arange(lo, hi + 1)
            grad[j_range] += a[kk - j_range]
    return grad / k


def run_fista(a_init, max_time, rng, beta_init=30.0, beta_rate=0.008,
              beta_max=50000.0, step_scale=1.0, use_backtrack=True):
    """FISTA with adaptive LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_gradient(y, beta, n)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-14 and bt < 5:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 2:
                step = step_local * 0.8
            else:
                step = min(step * 1.02, 2.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            step = step_scale / max(beta, 1.0)
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        if it - last_improve > 3000:
            t_mom = 1.0
            noise_scale = max(1e-7, 0.002 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    return best_a, best_obj


def run_topk_fista(a_init, max_time, rng, k_init=1, k_max=32, k_rate=0.001):
    """FISTA using top-k subgradients (cheaper than full LSE)."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0
    t_mom = 1.0
    y = a.copy()
    step = 1.0 / n

    while time.monotonic() - start < max_time:
        it += 1
        k = min(k_max, k_init + int(k_rate * it))
        grad = topk_subgradient(y, c, n, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step_new = step * 0.9995
        x_new = project_simplex(y - step_new * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        step = step_new

        # Periodically recompute c and check objective
        if it % 10 == 0 or it < 5:
            c = fft_convolve_self(a)
            cur_obj = 2.0 * n * c.max() / (S * S)
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve = it

        if it - last_improve > 5000 and it % 200 == 0:
            t_mom = 1.0
            y = a.copy()
            ns = max(1e-7, 0.001 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, ns, n), S)

    return best_a, best_obj


def coordinate_descent_fast(a_init, max_time, rng, n_top=5):
    """Fast coordinate descent: transfer mass to reduce peak convolution."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        batch_size = 20
        for _ in range(batch_size):
            if time.monotonic() - start >= max_time:
                break
            it += 1
            cur_max = c.max()
            cur_obj = 2.0 * n * cur_max / (S * S)
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve = it

            if it - last_improve > 5000:
                break

            # Get top few peak lags
            top_idx = np.argsort(c)[::-1][:n_top]
            
            # For each peak lag, find the source coordinate that contributes most
            # and transfer to a coordinate that reduces the peak
            improved = False
            for kk in top_idx[:2]:
                kk = int(kk)
                lo = max(0, kk - n + 1)
                hi = min(n - 1, kk)
                if hi < lo:
                    continue
                j_range = np.arange(lo, hi + 1)
                contrib = a[j_range] * a[kk - j_range]
                total_contrib = contrib.sum()
                if total_contrib < 1e-15:
                    continue
                
                # Source: coordinate with highest contribution
                src_local = int(j_range[np.argmax(contrib)])
                
                # Target: coordinate with lowest contribution to ALL top lags
                min_contrib = np.inf
                tgt_local = -1
                for ll in top_idx[:3]:
                    ll = int(ll)
                    lo2 = max(0, ll - n + 1)
                    hi2 = min(n - 1, ll)
                    if hi2 >= lo2:
                        if src_local >= lo2 and src_local <= hi2:
                            c2 = a[src_local] * a[ll - src_local]
                            min_contrib = min(min_contrib, c2)
                
                # Find a good target: low value, low contribution to peaks
                # Try a few candidates from the low end of the sequence
                candidates = np.random.choice(n, size=min(10, n), replace=False)
                best_reduction = 0
                best_tgt = -1
                delta = a[src_local] * 0.03
                
                for tgt in candidates:
                    if tgt == src_local:
                        continue
                    # Estimate reduction in peak
                    reduction = 0
                    for ll in top_idx[:3]:
                        ll = int(ll)
                        # Current contribution of src to lag ll
                        if 0 <= ll - src_local < n:
                            old_c = a[src_local] * a[ll - src_local]
                        else:
                            old_c = 0
                        # New contribution of tgt to lag ll
                        if 0 <= ll - tgt < n:
                            new_c = delta * a[ll - tgt]
                        else:
                            new_c = 0
                        reduction += old_c - new_c
                    if reduction > best_reduction:
                        best_reduction = reduction
                        best_tgt = tgt

                if best_tgt >= 0 and best_reduction > 0 and delta > 1e-10:
                    a[src_local] -= delta
                    a[best_tgt] += delta
                    # Update c
                    for jj in [src_local, best_tgt]:
                        sign = -1 if jj == src_local else 1
                        sign *= (1 if jj == src_local else 1)
                        # Actually let's just track the changes
                    # Recompute affected parts of c
                    # Change at src_local: -delta
                    # Change at best_tgt: +delta
                    # c[k] changes by: -delta*a[k-src] (if valid) + delta*a[k-best_tgt] (if valid)
                    for jj, sign in [(src_local, -delta), (best_tgt, delta)]:
                        ks = max(0, jj)
                        ke = min(2 * n - 2, jj + n - 1)
                        if ke >= ks:
                            partners = np.arange(ks, ke + 1) - jj
                            valid_p = (partners >= 0) & (partners < n)
                            c[ks:ke + 1][valid_p] += sign * a[partners[valid_p]]
                    improved = True

            if not improved:
                # Small random perturbation
                idx1 = rng.integers(0, n)
                idx2 = rng.integers(0, n)
                if idx1 != idx2 and a[idx1] > 1e-10:
                    delta = a[idx1] * 0.01
                    a[idx1] -= delta
                    a[idx2] += delta
                    c = fft_convolve_self(a)

        # Full recompute periodically
        if it % 100 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def make_ramp(n, p=1.0, rng=None, noise=0.0):
    """Create a power ramp initialization."""
    j = np.arange(1, n + 1, dtype=np.float64)
    a = j ** p
    if noise > 0 and rng is not None:
        a = a + noise * rng.standard_normal(n)
        a = np.maximum(a, 0.0)
    S = np.sqrt(2.0 * n)
    return a * (S / a.sum())


def make_asymmetric_init(n, method, rng):
    """Create various asymmetric initializations."""
    j = np.arange(n, dtype=np.float64)
    tt = j / n
    S = np.sqrt(2.0 * n)

    if method == "ramp":
        a = j + 1
    elif method == "ramp15":
        a = (j + 1) ** 1.5
    elif method == "ramp07":
        a = (j + 1) ** 0.7
    elif method == "ramp12":
        a = (j + 1) ** 1.2
    elif method == "exp_ramp":
        a = np.exp(0.5 * tt)  # mild exponential to avoid end concentration
    elif method == "sine_ramp":
        a = (j + 1) * (1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt))
        a = np.maximum(a, 0.1)
    elif method == "quad_ramp":
        a = (j + 1) + 0.5 * (j + 1) ** 2 / n
    elif method == "ramp_perturb":
        a = (j + 1) * (1.0 + 0.1 * rng.standard_normal(n))
        a = np.maximum(a, 0.1)
    else:
        a = j + 1

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Key: use asymmetric (ramp) initializations for much better starting point.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # === Phase 0: Try ramp initializations with different n ===
    print("=== Phase 0: Ramp initialization ===")
    for n_try in [8192, 16384]:
        a_ramp = make_ramp(n_try, p=1.0)
        obj_ramp = compute_obj(a_ramp)
        print("  n=%d, ramp: obj=%.6f" % (n_try, obj_ramp))
        if obj_ramp < global_best_obj:
            global_best_obj = obj_ramp
            global_best_a = a_ramp.copy()

    n_best = len(global_best_a)
    print("  Best initial: n=%d, obj=%.6f" % (n_best, global_best_obj))

    # === Phase 1: Screen diverse asymmetric inits (15% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.15
    if p1_time > 60:
        print("\n=== Phase 1: Diverse asymmetric starts (budget %.0fs) ===" % p1_time)
        methods = ["ramp", "ramp15", "ramp07", "ramp12", "exp_ramp", "sine_ramp", "quad_ramp", "ramp_perturb"]
        per_time = p1_time / len(methods)

        for method in methods:
            if time.monotonic() >= deadline:
                break
            a_div = make_asymmetric_init(n_best, method, rng)
            a_opt, obj_opt = run_fista(a_div, per_time * 0.7, rng,
                                        beta_init=20, beta_rate=0.005,
                                        beta_max=20000, step_scale=1.0,
                                        use_backtrack=False)
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()
                print("  %s: obj=%.6f ***NEW BEST***" % (method, obj_opt))
            else:
                print("  %s: obj=%.6f" % (method, obj_opt))

    # === Phase 2: Intensive FISTA on best (30% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.30
    if p2_time > 60:
        print("\n=== Phase 2: Intensive FISTA (budget %.0fs) ===" % p2_time)

        variants = [
            (20, 0.003, 10000, 0.8, True),
            (50, 0.002, 30000, 0.5, True),
            (100, 0.001, 100000, 0.3, True),
        ]
        for i, (bi, br, bm, ss, bt) in enumerate(variants):
            if time.monotonic() >= deadline:
                break
            rem = min(p2_time / 3, deadline - time.monotonic() - 10)

                break
            a_opt, obj_opt = run_fista(global_best_a, rem, rng,
                                        beta_init=bi, beta_rate=br,
                                        beta_max=bm, step_scale=ss,
                                        use_backtrack=bt)
            print("  FISTA variant %d: obj=%.6f" % (i + 1, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 3: Top-k subgradient FISTA (20% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.20
    if p3_time > 30:
        print("\n=== Phase 3: Top-k subgradient (budget %.0fs) ===" % p3_time)
        a_opt, obj_opt = run_topk_fista(global_best_a, p3_time, rng,
                                         k_init=1, k_max=16, k_rate=0.005)

        n_fb = 8192
