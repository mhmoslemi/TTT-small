"""
Minimize 2*n*max(a*a)/sum(a)^2 using hybrid LSE/top-K L-BFGS with spectral equalization.

Key innovations:
1. Hybrid gradient: combines LSE smooth max with explicit top-K peak tracking
2. Spectral equalization: periodic reshaping of the power spectrum to promote flatness
3. Peak deflation: targeted mass transfer to reduce the dominant convolution peak
4. Multi-n exploration: try different sequence lengths and keep the best
5. Informed restarts: use spectral analysis of the current solution to guide restarts
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT."""
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * len(a) - 1]
    return c


def compute_conv_fft(a, N):
    """Self-convolution via FFT, returning both c and A."""
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * len(a) - 1]
    return c, A


def lse_gradient(a, N, beta):
    """Gradient of LSE-smoothed peak of conv(a,a)."""
    n = len(a)
    c, A = compute_conv_fft(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
    else:
        weights = exp_bc / total
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    lse_val = c_max + np.log(total) / beta if total > 1e-300 else c_max
    return grad, lse_val, c, c_max


def toppk_gradient(a, N, top_k=10):
    """Gradient of sum of top-K convolution values (normalized)."""
    n = len(a)
    c, A = compute_conv_fft(a, N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n), c, c.max()
    weights = np.zeros(len(c))
    weights[top_idx] = top_vals / total
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, c, c.max()


def hybrid_gradient(a, N, beta, top_k, lse_weight=0.5):
    """Combine LSE and top-K gradients."""
    n = len(a)
    g_lse, _, c, c_max = lse_gradient(a, N, beta)
    g_top, _, _ = toppk_gradient(a, N, top_k)
    grad = lse_weight * g_lse + (1 - lse_weight) * g_top
    return grad, c, c_max


def spectral_equalize(a, N, strength=0.5, dc_floor=0.01, max_boost=5.0):
    """Apply spectral equalization to flatten the power spectrum."""
    A = fft(a, N)
    P = np.abs(A) ** 2
    # Target: make P as flat as possible (excluding DC)
    # Use a geometric mean approach
    P_non_dc = P[1:]
    P_geomean = np.exp(np.mean(np.log(np.maximum(P_non_dc, P_non_dc.min() * 1e-3))))
    # Compute target power
    P_target = P_geomean
    # Scale factor
    scale = np.ones(N)
    for k in range(1, N):
        if P[k] > P_target * dc_floor:
            target_ratio = P_target / P[k]
            scale[k] = 1.0 + strength * (np.sqrt(target_ratio) - 1.0)
            scale[k] = min(scale[k], max_boost)
            scale[k] = max(scale[k], 1.0 / max_boost)
    A_new = A * scale
    # Ensure real result by enforcing Hermitian symmetry (should be automatic)
    a_new = np.real(ifft(A_new))[:len(a)]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    S = a.sum()
    if s < 1e-15:
        return a
    return a_new * (S / s)


def spectral_equalize_fast(a, N, strength=0.5):
    """Vectorized spectral equalization."""
    A = fft(a, N)
    P = np.abs(A) ** 2
    P_non_dc = P[1:]
    # Use median as a robust target
    P_target = np.median(P_non_dc)
    scale = np.ones(N)
    valid = P > P_target * 0.001
    scale[valid] = 1.0 + strength * (np.sqrt(P_target / np.maximum(P[valid], P_target * 0.001)) - 1.0)
    scale = np.clip(scale, 0.5, 2.0)
    scale[0] = 1.0
    A_new = A * scale
    a_new = np.real(ifft(A_new))[:len(a)]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    S_orig = a.sum()
    if s < 1e-15:
        return a
    return a_new * (S_orig / s)


def peak_deflation(a, n, S, N, n_steps=200, step_frac=0.005):
    """Reduce the convolution peak by targeted mass transfer."""
    a = a.copy()
    for step in range(n_steps):
        c = np.real(ifft(np.abs(fft(a, N)) ** 2))[:2 * n - 1]
        m_star = int(c.argmax())
        # Compute gradient of c[m_star] w.r.t. a
        idx1 = m_star - np.arange(n)
        valid1 = (idx1 >= 0) & (idx1 < n)
        contrib1 = np.zeros(n)
        contrib1[valid1] = a[idx1[valid1]]
        
        idx2 = np.arange(n) - m_star
        valid2 = (idx2 >= 0) & (idx2 < n)
        contrib2 = np.zeros(n)
        contrib2[valid2] = a[idx2[valid2]]
        
        grad = contrib1 + contrib2
        
        j_max = int(np.argmax(grad))
        # Find the minimum grad among elements with non-zero a
        masked_grad = grad.copy()
        masked_grad[a < 1e-10 * S / n] = np.inf
        j_min = int(np.argmin(masked_grad))
        
        if j_max == j_min:
            break
        
        # Transfer mass
        max_transfer = a[j_max] * step_frac
        a[j_max] -= max_transfer
        a[j_min] += max_transfer
        a = np.maximum(a, 0)
        a = a * (S / a.sum())
    
    return a


def lbfgs_two_loop(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    if m == 0:
        return -g
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def safe_normalize(a, n, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def optimize_lse(a_init, n, S, N, max_time, rng, 
                 beta_init=5.0, beta_max=500.0, m_lbfgs=10,
                 use_topk=False, top_k_init=10, top_k_max=200,
                 use_spectral_eq=False, spectral_eq_interval=500,
                 spectral_eq_strength=0.3):
    """Main L-BFGS optimization loop with hybrid gradient and spectral equalization."""
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    top_k = top_k_init
    
    # Compute initial gradient
    if use_topk:
        g, c, c_max = hybrid_gradient(a, N, beta, top_k)
    else:
        g, _, c, c_max = lse_gradient(a, N, beta)
    
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()
    
    step_scale = 2.0 * S / (n * max(1.0, beta ** 0.5))
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        # Adaptive parameters
        if it % 200 == 0:
            beta = min(beta_max, beta * 1.08)
            top_k = min(top_k_max, top_k_init + it // 100)
            step_scale = 2.0 * S / (n * max(1.0, beta ** 0.5))
        
        if it % 80 == 0 and len(s_hist) >= m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]
        
        # Spectral equalization
        if use_spectral_eq and it % spectral_eq_interval == 0:
            a = spectral_equalize_fast(a, N, strength=spectral_eq_strength)
            s_hist = []
            y_hist = []
            if use_topk:
                g, c, c_max = hybrid_gradient(a, N, beta, top_k)
            else:
                g, _, c, c_max = lse_gradient(a, N, beta)
        
        # Compute L-BFGS direction
        direction = lbfgs_two_loop(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        # Line search
        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        for ls in range(10):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = np.real(ifft(np.abs(fft(candidate, N)) ** 2))[:2 * n - 1]
            cm_ls = c_ls.max()
            if cm_ls <= c_max * 1.001 + 1e-10 or ls == 9:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_max * 1.001 + 1e-10:
                    break
            alpha_ls *= 0.5
        
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
        
        # Compute new gradient
        if use_topk:
            g_new, c_nw, cmax_nw = hybrid_gradient(a_new, N, beta, top_k)
        else:
            g_new, _, c_nw, cmax_nw = lse_gradient(a_new, N, beta)
        
        # Update L-BFGS history
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        
        a = a_new
        g = g_new
        c_max = cmax_nw
        
        # Random restart
        if it % 1000 == 0 and rng.random() < 0.2:
            # Spectral phase perturbation
            A = fft(a, N)
            mag = np.abs(A)
            frac = 0.3
            n_rand = max(1, int(N * frac))
            rand_idx = rng.choice(N, size=n_rand, replace=False)
            # Maintain Hermitian symmetry
            pairs = []
            for idx in rand_idx:
                if idx == 0 or idx == N // 2:
                    continue
                pairs.append(idx)
            # Only modify positive frequency indices to maintain symmetry
            pos_freqs = [k for k in pairs if 1 <= k < N // 2]
            if pos_freqs:
                for k in pos_freqs:
                    A[k] = mag[k] * np.exp(1j * rng.uniform(0, 2 * np.pi))
                    A[N - k] = np.conj(A[k])
            a_new_spec = np.real(ifft(A))[:n]
            a_new_spec = np.maximum(a_new_spec, 0)
            s_new = a_new_spec.sum()
            if s_new > 1e-15:
                a = a_new_spec * (S / s_new)
                s_hist = []
                y_hist = []
                if use_topk:
                    g, c, c_max = hybrid_gradient(a, N, beta, top_k)
                else:
                    g, _, c, c_max = lse_gradient(a, N, beta)
    
    return best_a, best_cmax


def make_init_random(n, rng):
    return rng.random(n) + 0.01


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (3 * tt + 0.5 * tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    L = n
    i = np.arange(L)
    c = int(rng.integers(1, L))
    phase = -np.pi * i * (2 * i + 1 - c) / L
    a = 1.0 + 0.5 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_fourier(n, rng, N):
    """Random flat-spectrum initialization."""
    P = np.ones(N)
    P[0] = float(n)  # DC component
    # Add some random variation
    P += 0.3 * rng.random(N)
    P[0] = float(n)
    phase = rng.uniform(0, 2 * np.pi, N)
    # Enforce Hermitian symmetry
    for k in range(1, N // 2 + 1):
        phase[N - k] = -phase[k]
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_multi_freq(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for freq in [2, 5, 11, 17, 31, 43]:
        a += 0.1 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return np.clip(a, 0.01, None)


def make_init_concentrated(n, rng):
    """Concentrated in middle, zero at edges (to reduce edge effects)."""
    tt = np.linspace(0, 1, n)
    envelope = np.sin(np.pi * tt) ** 0.5  # Bump function
    a = envelope * (1.0 + 0.2 * rng.standard_normal(n))
    return np.clip(a, 0.0, None)
