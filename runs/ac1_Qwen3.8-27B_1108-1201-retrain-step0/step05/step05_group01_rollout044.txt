"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses adaptive-beta FISTA with reheat, exact active-set polishing,
L-BFGS squeezing, and directional local search to escape local minima.
Improvements over previous: reheat schedule, directional perturbations,
active-set exact subgradient, and multi-length exploration.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if v.sum() == S:
        return np.maximum(v, 0.0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 2).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.0):
    """Top-k weighted gradient for active set method."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    vmax = vals.max()
    # Use linear weighting: weight proportional to (val/vmax)
    w = vals / vmax
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if max_val < 1e-30:
        return np.zeros(n)
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

                       beta_growth=1.002, mode="lse", step_scale=1.0,

    FISTA with monotonically increasing beta and periodic shake.
    The beta grows geometrically, making the LSE approximation tighter over time.
    Periodic shakes help escape local minima.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Monotonically increasing beta
        beta = min(beta_min * (beta_growth ** it), beta_max)
        step = step_scale / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Gradient based on mode
        if mode == "lse":
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
        elif mode == "active":
            if stuck_count < 200:
                k = 1
            elif stuck_count < 1000:
                k = 3
            elif stuck_count < 5000:
                k = min(32, 3 + stuck_count // 200)
            else:
                k = min(128, 32 + stuck_count // 200)
            grad = topk_gradient(y, c, k, power=1.0)
        else:  # hybrid: LSE normally, active when stuck

                k = min(64, 4 + stuck_count // 200)
                grad = topk_gradient(y, c, k, power=1.0)

                w = lse_weights(c, beta)
                grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)

                stuck_count = max(0, stuck_count - shake_interval // 2)

        # FISTA update with backtracking-inspired step adjustment
        # Clip step to ensure we don't overshoot
        max_step = S / (gnorm + 1e-15) * 0.9
        step_used = min(step, max_step)
        x_new = project_simplex(y - step_used * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum reset to avoid overshoot
        if it % 3000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_reheat_fista(a_init, max_time, rng):
    """
    FISTA with a reheat schedule: beta ramps up, then suddenly drops, then ramps up again.
    This helps escape local minima by temporarily smoothing the landscape.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0

    # Reheat schedule: beta goes 5 -> 500, drops to 5, goes 5 -> 2000
    phase_length = max(1000, int(max_time * 60.0 / 2.0))  # approximate iterations per phase

    phases = [
        (5.0, 500.0, 1.003),
        (5.0, 1000.0, 1.004),
        (10.0, 5000.0, 1.005),
    ]

    for phase_idx, (b_min, b_max, growth) in enumerate(phases):
        if time.monotonic() - start > max_time:
            break
        if phase_idx > 0:
            # On reheat, add a perturbation
            perturb = 0.03 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, perturb, n), S)
            t_mom = 1.0
            a = y.copy()

        phase_start = time.monotonic()
        phase_time = min(phase_length / 60.0, max_time - (time.monotonic() - start))

        local_it = 0
        while time.monotonic() - phase_start < phase_time and time.monotonic() - start < max_time:
            local_it += 1
            it += 1
            beta = min(b_min * (growth ** local_it), b_max)
            step = 1.0 / max(beta, 1.0)

            c = fft_convolve_self(y)
            cur_max = c.max()
            if cur_max < best_conv - 1e-14:
                best_conv = cur_max
                best_a = y.copy()

            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            step_used = min(step, 0.9 * S / (gnorm + 1e-15))
            x_new = project_simplex(y - step_used * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

            if local_it % 2000 == 0:
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_lbfgs_polish(a_init, max_time, rng, n_levels=4):
    """
    L-BFGS-B polishing with multiple beta levels.
    Uses the LSE-smoothed objective with penalty for sum constraint.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    beta_schedule = [50.0, 200.0, 1000.0, 5000.0][:n_levels]

    for beta in beta_schedule:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm_local + np.log(e.sum() + 1e-300) / beta
            # Small penalty for sum deviation
            sum_x = x.sum()
            obj += 0.1 * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 0.2 * (sum_x - S)
            return obj, grad

        t0 = time.monotonic()
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 80, 'maxls': 25, 'ftol': 1e-15, 'gtol': 1e-12,
                     'iprint': -1}
        )

        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)

        print("    L-BFGS beta=%.0f: obj=%.8f (t=%.1fs, iters=%d)" %
              (beta, obj_new, elapsed, result.nit))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_local_search(a_init, max_time, rng, n_tries=500):
    """
    Local search: random perturbations of varying scales, keep improvements.
    Uses "snake" perturbations (smooth random directions) for better exploration.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    c = fft_convolve_self(a)
    max_c = c.max()

    for trial in range(n_tries):
        if time.monotonic() - start > max_time:
            break

        # Scale decreases over time (annealing)
        progress = trial / max(n_tries, 1)
        base_scale = 0.05 * (1.0 - 0.7 * progress) * S / np.sqrt(n)

        # Random magnitude
        scale = base_scale * (0.5 + rng.random())

        # Generate smooth random perturbation
        if rng.random() < 0.3:
            # Low-frequency perturbation
            F = fft(rng.normal(0, 1.0, n), n)
            cutoff = max(2, n // 200)
            F[cutoff:-cutoff] *= 0.01
            pert = np.real(ifft(F))[:n]
        else:
            # Mix of low and high frequency
            F = fft(rng.normal(0, 1.0, n), n)
            cutoff = max(3, n // 50)
            F[cutoff:-cutoff] *= 0.3
            pert = np.real(ifft(F))[:n]

        pert_norm = np.linalg.norm(pert)
        if pert_norm < 1e-15:
            continue
        pert = pert / pert_norm * scale

        candidate = project_simplex(a + pert, S)
        obj = compute_obj(candidate)

        if obj < best_obj:
            best_obj = obj
            best_a = candidate.copy()
            a = candidate.copy()
            if trial % 50 == 0:
                print("    Local search trial %d: obj=%.8f (improved)" % (trial, obj))
        elif obj < best_obj + 0.001 and rng.random() < 0.1:
            # Occasional "simulated annealing" accept
            a = candidate.copy()

    return best_a, best_obj


def run_active_polish(a_init, max_time, rng):
    """
    Intensive polishing using exact subgradient with adaptive active set.
    Uses very small steps for the final refinement.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive active set size
        if stuck < 50:
            tol_frac = 0.0
        elif stuck < 200:
            tol_frac = 0.001
        elif stuck < 1000:
            tol_frac = 0.003
        elif stuck < 5000:
            tol_frac = 0.008
        else:
            tol_frac = 0.02

        grad = exact_subgradient(y, c, tol_frac=tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Decreasing step size
        step = S / (gnorm * (1.0 + 0.01 * it))
        step = min(step, 0.5 * S / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 2000 == 0:
            t_mom = 1.0

        # Perturb when very stuck
        if stuck > 5000 and it % 500 == 0:
            noise_scale = max(1e-7, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 10000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: adaptive FISTA, reheat, active-set polish, L-BFGS, local search.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None and len(a_start) > 10:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Loaded height_sequence_1: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    def update_best(a, label=""):
        nonlocal global_best_obj, global_best_a
        obj = compute_obj(a)
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a.copy()
            print("  [%s] NEW BEST: obj=%.8f (n=%d)" % (label, obj, len(a)))
        else:
            print("  [%s] obj=%.8f (no improvement)" % (label, obj))
        return obj

    # === Phase 1: Adaptive FISTA (30% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.30
    print("\n=== Phase 1: Adaptive FISTA (n=%d, budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    a1, _ = run_fista_adaptive(global_best_a, p1_budget * 0.7, rng,
                                beta_min=5.0, beta_max=2000.0,
                                beta_growth=1.003, mode="lse",
                                step_scale=1.0, shake_interval=3000,
                                shake_strength=0.03)
    update_best(a1, "FISTA-adapt")

    rem1 = p1_budget - (time.monotonic() - t1)
    if rem1 > 10:
        a1b, _ = run_fista_adaptive(global_best_a, rem1, rng,
                                     beta_min=20.0, beta_max=5000.0,
                                     beta_growth=1.005, mode="hybrid",
                                     step_scale=0.8, shake_interval=4000,
                                     shake_strength=0.02)
        update_best(a1b, "FISTA-hybrid")

    # === Phase 2: Reheat FISTA (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = remaining * 0.20
        print("\n=== Phase 2: Reheat FISTA (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        a2, _ = run_reheat_fista(global_best_a, p2_budget, rng)
        update_best(a2, "Reheat")

    # === Phase 3: L-BFGS polishing (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = remaining * 0.20
        print("\n=== Phase 3: L-BFGS polishing (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        a3, obj3 = run_lbfgs_polish(global_best_a, p3_budget, rng, n_levels=4)
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()
            print("  [L-BFGS] NEW BEST: obj=%.8f" % obj3)
        else:
            print("  [L-BFGS] obj=%.8f (no improvement)" % obj3)

    # === Phase 4: Active-set exact polish (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = remaining * 0.15
        print("\n=== Phase 4: Active-set polish (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        a4, _ = run_active_polish(global_best_a, p4_budget, rng)
        update_best(a4, "Active")

    # === Phase 5: Local search (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = remaining * 0.10
        print("\n=== Phase 5: Local search (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        n_tries = int(200 * p5_budget / 60.0)  # scale tries with time
        n_tries = max(50, min(1000, n_tries))
        a5, obj5 = run_local_search(global_best_a, p5_budget, rng, n_tries=n_tries)
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()
            print("  [LocalSearch] NEW BEST: obj=%.8f" % obj5)
        else:
            print("  [LocalSearch] obj=%.8f (no improvement)" % obj5)

    # === Phase 6: Final combined polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 6: Final polish (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()

        # Short active-set pass
        a6, _ = run_active_polish(global_best_a, remaining * 0.4, rng)
        update_best(a6, "FinalActive")

        rem6 = remaining - (time.monotonic() - t6)
        if rem6 > 10:
            # Short L-BFGS squeeze
            a7, obj7 = run_lbfgs_polish(global_best_a, rem6 * 0.8, rng, n_levels=2)
            if obj7 < global_best_obj:
                global_best_obj = obj7
                global_best_a = a7.copy()
                print("  [FinalL-BFGS] NEW BEST: obj=%.8f" % obj7)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
