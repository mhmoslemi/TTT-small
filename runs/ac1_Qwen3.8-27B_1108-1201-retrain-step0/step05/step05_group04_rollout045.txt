import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    if rho == 0:
        return np.full(n, S / n)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix numerical issues
    diff = S - result.sum()
    if abs(diff) > 1e-12:
        pos = result > 0
        if pos.any():
            result[pos] += diff / pos.sum()
            result = np.maximum(result, 0.0)
    return result


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def active_gradient(a, c, active_thresh=None, min_active=1, max_active=None):
    """
    Compute the (smoothed) subgradient of max(c) using active set.
    active_thresh: include lags within this distance of max. If None, use adaptive.
    Returns gradient vector of length n.
    """
    n = len(a)
    cm = c.max()
    
    if active_thresh is None:
        # Adaptive: include lags within a small fraction of the range
        c_range = c.max() - c.min()
        active_thresh = max(0.01 * c_range, 1e-10)
    
    mask = c >= (cm - active_thresh)
    if mask.sum() < min_active:
        mask = np.zeros_like(c)
        mask[np.argmax(c)] = 1
    if max_active is not None and mask.sum() > max_active:
        # Keep only top max_active lags
        c_sorted = np.argsort(c)[::-1][:max_active]
        mask = np.zeros_like(c)
        mask[c_sorted] = 1
    
    w = mask.astype(np.float64)
    w /= w.sum()
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return grad


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return grad, cm + np.log(tot) / beta


def fista_active(a_init, n, S, max_time, rng, 
                 beta_start=100.0, beta_end=10000.0, beta_steps=200,
                 use_lse_phase=True, lse_fraction=0.5,
                 step_init=0.01, shrink=0.5, expand=1.2,
                 momentum_decay=1.0, restart_gap=300):
    """
    FISTA with hybrid LSE + active-set gradient.
    Phase 1 (first lse_fraction of time): use LSE gradient for smooth optimization.
    Phase 2 (rest): use active-set gradient for sharp optimization.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    step = step_init
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        frac_time = elapsed / max_time
        in_lse_phase = use_lse_phase and frac_time < lse_fraction
        
        # Compute convolution
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve_it = it
        
        # Compute gradient
        if in_lse_phase:
            # LSE phase: smooth gradient
            frac_beta = min(1.0, it / max(1, int(beta_steps * lse_fraction)))
            beta = beta_start * (beta_end / beta_start) ** frac_beta
            grad, _ = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            # Step size for LSE: roughly 1/beta
            step = 1.0 / np.sqrt(beta)
        else:
            # Active-set phase: sharp gradient
            # Adaptive threshold: start tight, expand if stuck
            stagnant = it - last_improve_it
            base_thresh = max(0.005 * (c.max() - c.min()), 1e-10)
            if stagnant > 100:
                thresh = base_thresh * (1 + stagnant / 50.0)
            else:
                thresh = base_thresh
            grad = active_gradient(y, c, active_thresh=thresh)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            # Step size: adaptive
            step = step_init / (1.0 + it * 0.0001)
        
        # Backtracking line search
        x_trial = project_simplex(y - step * grad, S)
        c_trial = fft_convolve_self(x_trial)
        trial_max = c_trial.max()
        
        accepted = False
        backtracks = 0
        while trial_max > cur_max + 1e-12 and backtracks < 10:
            step *= 0.5
            x_trial = project_simplex(y - step * grad, S)
            c_trial = fft_convolve_self(x_trial)
            trial_max = c_trial.max()
            backtracks += 1
        
        if trial_max <= cur_max + 1e-12:
            x_new = x_trial
            accepted = True
            step = min(step * expand, 0.1)  # Try to expand step next time
        else:
            # Use a very small step
            step *= 0.1
            x_new = project_simplex(y - step * grad, S)
            c_trial = fft_convolve_self(x_new)
            if c_trial.max() < cur_max:
                accepted = True
        
        # Momentum update
        if accepted:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            alpha = (t_mom - 1.0) / t_new
            y = x_new + momentum_decay * alpha * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new
        else:
            # Restart momentum
            t_mom = 1.0
            a = x_new
            y = a.copy()
        
        # Periodic noise injection to escape local minima
        stagnant = it - last_improve_it
        if stagnant > restart_gap and stagnant % 50 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 2000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                a = y.copy()
            if stagnant > restart_gap * 3:
                # Aggressive restart
                noise_scale = 0.01 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = y.copy()
                t_mom = 1.0
                last_improve_it = it
        
        if it % 500 == 0:
            print("    it=%d, best=%.10f, cur=%.10f, step=%.6f, stag=%d" % 
                  (it, best, cur_max, step, stagnant))
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def fista_lse_only(a_init, n, S, max_time, rng,
                   beta_start=50.0, beta_end=50000.0, beta_steps=300):
    """Pure LSE-FISTA for broad optimization."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 100000.0)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve_it = it
        
        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / np.sqrt(beta)
        
        # Backtracking
        x_trial = project_simplex(y - step * grad, S)
        c_trial = fft_convolve_self(x_trial)
        backtracks = 0
        while c_trial.max() > cur_max + 1e-12 and backtracks < 8:
            step *= 0.5
            x_trial = project_simplex(y - step * grad, S)
            c_trial = fft_convolve_self(x_trial)
            backtracks += 1
        
        x_new = x_trial
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        alpha = (t_mom - 1.0) / t_new
        y = x_new + alpha * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Noise for stagnation
        stagnant = it - last_improve_it
        if stagnant > 500:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stagnant / 5000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                a = y.copy()
            if stagnant > 2000:
                last_improve_it = it
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def surgical_refinement(a_init, n, S, max_time, rng, top_k=20):
    """
    Surgical peak shaving: directly target the top peaks.
    For each peak lag, reduce the contributions.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Get top peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        cm = c.max()
        
        # Compute responsibility
        resp = np.zeros(n)
        for k in top_idx:
            weight = max(0.0, c[k] - cm * 0.995)
            if weight < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            np.add.at(resp, js, contrib * weight)
        
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        
        # Multiplicative reduction
        lr = 0.005 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)
        
        # Small LSE polish step
        if it % 10 == 0:
            c = fft_convolve_self(a)
            grad = active_gradient(a, c, active_thresh=0.01 * (c.max() - c.min()))
            step = 0.005
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < c.max():
                a = a_new
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def raised_cosine_init(n, center_frac=0.3, width_frac=0.5):
    """Generate a raised-cosine initialization."""
    t = np.linspace(0, 1, n)
    center = center_frac
    width = width_frac / 2.0
    x = (t - center) / width
    a = np.ones(n)
    mask = np.abs(x) < 1
    a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
    a[~mask] = 0.0
    return np.maximum(a, 1e-15)


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n)
    
    for cf in [0.25, 0.3, 0.35, 0.4, 0.5]:
        for wf in [0.3, 0.4, 0.5, 0.6]:
            inits.append(raised_cosine_init(n, cf, wf))
    
    for sigma in [0.1, 0.15, 0.2, 0.25]:
        for mu in [0.3, 0.4, 0.5, 0.6]:
            a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
            inits.append(np.maximum(a, 1e-15))
    
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        inits.append(np.maximum(t ** p, 1e-15))
        inits.append(np.maximum((1 - t) ** p, 1e-15))
    
    for k in [1, 2, 3]:
        inits.append(np.maximum(0.5 + 0.5 * np.sin(np.pi * k * t), 0.01))
    
    # Random smooth
    for _ in range(5):
        coeffs = rng.normal(0, 1, 8)
        a = np.zeros(n)
        for i, c in enumerate(coeffs):
            a += c * np.sin(np.pi * (i + 1) * t + rng.uniform(0, np.pi))
        inits.append(np.maximum(a, 1e-15))
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """

    Uses hybrid active-set/LSE FISTA with multi-scale approach and surgical refinement.
    """

    budget_s = float(budget_s)

    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))

    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)

            given = np.clip(given, 0, 1000)

    except Exception:
        pass
    
    global_best = np.inf
    global_a = None
    
    def try_sequence(a):
        nonlocal global_best, global_a
        if a is None or len(a) == 0:
            return
        n = len(a)
        S = np.sqrt(2.0 * n)
        a = normalize_sum(a, n)

        val = c.max()

            global_best = val

    if given is not None:

        S_g = np.sqrt(2.0 * n_g)
        given_norm = normalize_sum(given, n_g)
        c_g = fft_convolve_self(given_norm)

        try_sequence(given_norm)

    if global_a is not None:

    if global_a is not None:

                               beta_start=100.0, beta_end=20000.0, beta_steps=300,
                               use_lse_phase=True, lse_fraction=0.4,

        try_sequence(a2)
    
    # === Phase 3: Try different initialization at same n ===
    if global_a is not None:
        remaining = deadline - time.monotonic()

                                       beta_start=50.0, beta_end=15000.0, beta_steps=100,

                                     beta_start=5000.0, beta_end=50000.0, beta_steps=50,
                                     use_lse_phase=False, lse_fraction=0.0,
                                     step_init=0.002, restart_gap=500)
            try_sequence(a_f)
    
    # === Phase 7: One more surgical pass on absolute best ===
    if global_a is not None:
        remaining = deadline - time.monotonic()

            S_s = np.sqrt(2.0 * n_s)

                                              use_lse_phase=False, step_init=0.001,

                    try_sequence(a_s2)
    
    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)

    print("RETURNING: length=%d, best_score=%.10f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
