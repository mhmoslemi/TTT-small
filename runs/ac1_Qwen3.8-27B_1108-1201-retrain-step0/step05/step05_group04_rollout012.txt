Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Phase 1: LSE-FISTA with staircase beta schedule.
Phase 2: Mass-transfer coordinate descent for the last mile.
Phase 3: Perturbation-based restarts.
Phase 4: Final LSE polish.

import numpy as np
import time
from scipy.fft import fft, ifft

N = 8192  # Target length

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta

def run_lse_fista_staircase(a_init, n, S, max_time, rng, 
                             betas=(50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000),
                             iters_per_beta=200, noise_period=500):
    """FISTA with staircase beta schedule."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    for beta in betas:
        if time.monotonic() - start >= max_time:
            break
        t_mom = 1.0  # Restart momentum at each beta level
        for it in range(iters_per_beta):
            if time.monotonic() - start >= max_time:
                break
            
            c = fft_convolve_self(y)
            cur_max = c.max()
            
            if cur_max < best:
                best = cur_max
                best_a = y.copy()
            
            grad, lse_val = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            
            step = 1.0 / beta
            
            # Periodic noise
            if it % noise_period == 0 and it > 0:
                noise_scale = 0.001 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
            
            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences of length 8192.
Strategy: Multi-phase optimization combining LSE-FISTA with staircase beta schedule,
mass-transfer coordinate descent for the last mile, frequency-domain perturbations,
and perturbation-based restarts to escape local minima.
Key insight: The objective equals max(f*f) when sum(f)=sqrt(2n). We minimize the peak
of the self-convolution using smoothed (LSE) gradients with increasing sharpness,
then refine with direct peak manipulation and mass-transfer operations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c). Returns (grad, lse_value)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def active_gradient(a, c, top_k=10):
    """Gradient using hard active set of top-k lags."""
    n = len(a)
    cm = c.max()
    threshold = cm - 2.0 / top_k
    w = np.where(c >= threshold, 1.0, 0.0)
    tot = w.sum()
    if tot < 1:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        tot = 1.0
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_lse_fista(a_init, n, S, max_time, rng,
                  betas=(50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000),
                  iters_per_beta=300, noise_period=500, use_active=False, active_k=10):
    """FISTA with staircase beta schedule and periodic noise."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    stagnant = 0

    for beta in betas:
        if time.monotonic() - start >= max_time:
            break
        t_mom = 1.0
        for it in range(iters_per_beta):
            if time.monotonic() - start >= max_time:
                return best_a, best

            c = fft_convolve_self(y)
            cur_max = c.max()

            if cur_max < best - 1e-12:
                best = cur_max
                best_a = y.copy()
                stagnant = 0
            else:
                stagnant += 1

            if use_active and it % 3 == 0:
                grad = active_gradient(y, c, top_k=active_k)
            else:
                grad, _ = lse_gradient(y, c, beta)

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break

            step = 1.0 / max(beta, 1.0)
            if stagnant > 500:
                step *= 0.3
                t_mom = 1.0

            if it % noise_period == 0 and it > 0:
                noise_scale = 0.002 * S / np.sqrt(n) * max(0.01, 1.0 - it / (iters_per_beta * len(betas)))
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def mass_transfer_cd(a_init, n, S, max_time, rng, max_iters=20000):
    """Mass transfer coordinate descent: move mass from peak contributors to quiet positions."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() - start >= max_time:
            break

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        js = np.arange(lo, hi + 1)
        partners = k_star - js
        resp = a[js] * a[partners]

        # Top contributors to the peak
        top_idx = np.argsort(resp)[::-1][:20]
        # Quietest positions
        quiet_idx = np.argsort(resp)[:20]

        improved = False
        for ti in top_idx[:10]:
            if improved:
                break
            j_main = js[ti]
            for qi in quiet_idx[:10]:
                m_third = js[qi]
                if m_third == j_main:
                    continue
                # Try transfer
                for scale in [1.0, 0.5, 0.1, 0.01]:
                    delta = scale * 0.001 * S / n
                    a_new = a.copy()
                    a_new[j_main] -= delta
                    a_new[m_third] += delta
                    if a_new[j_main] < 0:
                        delta = a[j_main] * 0.5
                        a_new = a.copy()
                        a_new[j_main] -= delta
                        a_new[m_third] += delta
                    a_new = project_simplex(a_new, S)
                    c_new = fft_convolve_self(a_new)
                    if c_new.max() < cur_max - 1e-15:
                        a = a_new
                        improved = True
                        break
                if improved:
                    break

        if not improved and it % 200 == 0:
            # Random perturbation to escape
            noise_scale = 0.001 * S / np.sqrt(n) * max(0.01, 1.0 - it / max_iters)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def freq_perturb_restart(a, n, S, max_time, rng, noise_frac=0.05):
    """Perturb in frequency domain and re-optimize briefly."""
    fsize = 1 << (2 * n - 1 - 1).bit_length()
    A = fft(a, fsize)
    phase = np.angle(A)
    mag = np.abs(A)
    # Perturb magnitudes
    mag_noise = rng.normal(0, noise_frac * mag.mean(), mag.shape)
    mag_new = np.maximum(mag + mag_noise, 0)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(np.maximum(a_new, 0), S)
    return a_new


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n)

    for cf in [0.25, 0.3, 0.35, 0.4, 0.5]:
        for wf in [0.3, 0.4, 0.5, 0.6]:
            x = (t - cf) / (wf / 2.0)
            a = np.ones(n)
            mask = np.abs(x) < 1
            a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
            a[~mask] = 0.0
            inits.append(np.maximum(a, 1e-10))

    for sigma in [0.1, 0.15, 0.2, 0.25, 0.3]:
        for mu in [0.3, 0.4, 0.5, 0.6]:
            a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
            inits.append(np.maximum(a, 1e-10))

    # Two-bump structures (anti-symmetric around center)
    for spread in [0.1, 0.15, 0.2]:
        a = np.exp(-0.5 * ((t - 0.2) / spread) ** 2) + 0.3 * np.exp(-0.5 * ((t - 0.8) / spread) ** 2)
        inits.append(np.maximum(a, 1e-10))

    # Power law (concentrated at one end)
    for p in [0.5, 1.0, 2.0, 3.0]:
        inits.append(np.maximum(t ** p, 1e-10))
        inits.append(np.maximum((1 - t) ** p, 1e-10))

    return inits
