def coordinate_descent_step(a, j, b):
    """Update coordinate j to minimize the objective."""
    n = len(a)
    S = a.sum()
    s = S - a[j]  # sum of all except j
    
    # Range of affected k
    k_lo = j
    k_hi = min(j + n - 1, 2*n - 2)
    
    # Compute M_unaff (max of unaffected b_k)
    # Unaffected k are [0, k_lo-1] and [k_hi+1, 2*n-2]
    mask = np.ones(2*n-1, dtype=bool)
    mask[k_lo:k_hi+1] = False
    M_unaff = b[mask].max() if mask.any() else 0.0
    
    # For affected k, compute b_k(x) = base_k + coeff_k * x (or x^2 for k=2j)
    affected_k = np.arange(k_lo, k_hi + 1)
    # base_k = b_k with a[j] contribution removed
    # coeff_k = coefficient of x (linear part)
    
    # b_k(x) = b_k - (contribution of a[j]) + (new contribution)
    # For k != 2j: b_k(x) = b_k - 2*a[j]*a[k-j] + 2*x*a[k-j]
    # For k = 2j: b_k(x) = b_k - a[j]^2 + x^2
    
    bases = np.zeros(len(affected_k))
    coeffs = np.zeros(len(affected_k))
    quad_mask = np.zeros(len(affected_k), dtype=bool)
    
    for idx, k in enumerate(affected_k):
        kj = k - j  # = k-j, this is the other index
        if k == 2*j:
            bases[idx] = b[k] - a[j]**2
            coeffs[idx] = 0
            quad_mask[idx] = True
        else:
            bases[idx] = b[k] - 2*a[j]*a[kj]
            coeffs[idx] = 2*a[kj]
            quad_mask[idx] = False
    
    # Now find x >= 0 that minimizes max(max(bases + coeffs*x (+ x^2 for quad)), M_unaff) / (s+x)^2
    # Use grid search

    grid = np.linspace(0, x_max, 50)
    
    best_x = x[0]
    best_val = np.inf

        m = max(vals.max(), M_unaff)
        val = m / (s + x)**2

Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: 
1. Start from given sequence or random init
2. FISTA with top-k LSE for initial optimization
3. Exact coordinate descent for precise peak shaving
4. Multi-restart with spectral perturbation
5. Try multiple sequence lengths

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_gradient(a, c, k, beta):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_value(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def resample_sequence(a, n_target):
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_gradient(a, c, k, beta):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)

    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)

            best = cur_max
            best_a = y.copy()

        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new

        if it % 500 == 0:
            c_chk = fft_convolve_self(a)
            if c_chk.max() < best:
                best = c_chk.max()
                best_a = a.copy()

        if it % 3000 == 0 and rng.random() < 0.1:
            noise_scale = 0.004 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def coordinate_descent(a_init, max_time, rng, grid_size=128,
                       max_passes=5000, verbose=True):
    """
    Exact coordinate descent minimizing max(conv)/(sum)^2.
    Scale-invariant, so no renormalization needed.
    Each coordinate update solves a 1D subproblem via grid search.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    b = fft_convolve_self(a)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n)
        b = fft_convolve_self(a)
        S = a.sum()

    obj = b.max() / (S * S)
    best_obj = obj
    best_a = a.copy()
    start = time.monotonic()
    pass_count = 0
    stuck = 0
    last_improve_obj = obj

    while time.monotonic() - start < max_time and pass_count < max_passes:
        pass_count += 1
        obj_before = obj
        S_cur = a.sum()

        order = rng.permutation(n)

        for j in order:
            aj = a[j]
            s = S_cur - aj
            if s < 1e-15:
                continue

            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)

            # Compute M_unaff
            if k_lo == 0 and k_hi == 2 * n - 2:
                M_unaff = 0.0
            else:
                if k_lo == 0:
                    M_unaff = b[k_hi + 1:].max()
                elif k_hi == 2 * n - 2:
                    M_unaff = b[:k_lo].max()
                else:
                    M_unaff = max(b[:k_lo].max(), b[k_hi + 1:].max())

            # Build linear/quadratic coefficients for affected k
            affected_k = np.arange(k_lo, k_hi + 1)
            n_aff = len(affected_k)
            bases = np.empty(n_aff)
            coeffs = np.empty(n_aff)
            quad_pos = -1

            for idx in range(n_aff):
                k = affected_k[idx]
                kj = k - j
                if k == 2 * j:
                    bases[idx] = b[k] - aj * aj
                    coeffs[idx] = 0.0
                    quad_pos = idx
                else:
                    bases[idx] = b[k] - 2.0 * aj * a[kj]
                    coeffs[idx] = 2.0 * a[kj]

            # Grid search for optimal x >= 0
            x_max = 3.0 * max(aj, S_cur / n)
            if x_max < 1e-10:
                x_max = 1e-10
            grid = np.linspace(0.0, x_max, grid_size)

            # Separate linear and quadratic parts
            if quad_pos >= 0:
                lin_bases = np.concatenate([bases[:quad_pos], bases[quad_pos + 1:]])
                lin_coeffs = np.concatenate([coeffs[:quad_pos], coeffs[quad_pos + 1:]])
                quad_base = bases[quad_pos]
            else:
                lin_bases = bases
                lin_coeffs = coeffs
                quad_base = None

            # Vectorized evaluation
            if len(lin_bases) > 0:
                vals_lin = lin_bases[None, :] + lin_coeffs[None, :] * grid[:, None]
                max_lin = vals_lin.max(axis=1)
            else:
                max_lin = np.zeros(grid_size)

            if quad_base is not None:
                max_peak = np.maximum(max_lin, quad_base + grid * grid)
            else:
                max_peak = max_lin.copy()

            max_peak = np.maximum(max_peak, M_unaff)
            obj_grid = max_peak / (s + grid) ** 2
            best_idx = obj_grid.argmin()
            best_x = grid[best_idx]

            # Refine with bisection around best grid point
            if 0 < best_idx < grid_size - 1:
                lo = grid[best_idx - 1]
                hi = grid[best_idx + 1]
                for _ in range(10):
                    mid = 0.5 * (lo + hi)
                    # Evaluate at lo, mid, hi
                    def eval_at(x):
                        v_lin = lin_bases + lin_coeffs * x
                        mp = v_lin.max()
                        if quad_base is not None:
                            mp = max(mp, quad_base + x * x)
                        mp = max(mp, M_unaff)
                        return mp / (s + x) ** 2
                    f_lo = eval_at(lo)
                    f_mid = eval_at(mid)
                    f_hi = eval_at(hi)
                    if f_mid < f_lo and f_mid < f_hi:
                        pass
                    elif f_lo <= f_mid and f_lo <= f_hi:
                        hi = mid
                    else:
                        lo = mid
                best_x = 0.5 * (lo + hi)

            if best_x != aj and best_x >= 0:
                # Update convolution incrementally
                for idx in range(n_aff):
                    k = affected_k[idx]
                    kj = k - j
                    if k == 2 * j:
                        b[k] = bases[idx] + best_x * best_x
                    else:
                        b[k] = bases[idx] + 2.0 * best_x * a[kj]
                S_cur += best_x - aj
                a[j] = best_x

        # Check improvement
        obj = b.max() / (S_cur * S_cur)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if verbose and pass_count % 50 == 0:
            elapsed = time.monotonic() - start
            improve = obj_before - obj
            print(f"  CD pass {pass_count}: obj={obj:.8f}, best={best_obj:.8f}, "
                  f"pass_improve={improve:.2e}, stuck={stuck}, t={elapsed:.0f}s")

        # Periodic renormalization to prevent numerical drift
        if pass_count % 20 == 0:
            S_now = a.sum()
            if S_now > 0:
                scale = np.sqrt(2.0 * n) / S_now
                a = a * scale
                b = fft_convolve_self(a)

        # If stuck, try perturbation
        if stuck > 20:
            noise_scale = 0.002 * a.sum() / np.sqrt(n)
            a = np.maximum(a + noise_scale * rng.standard_normal(n), 0.0)
            b = fft_convolve_self(a)
            stuck = 0
            obj = b.max() / (a.sum() ** 2)

        # Early stop if barely improving
        if pass_count > 100 and obj_before - obj < 1e-14:
            if verbose:
                print(f"  CD converged at pass {pass_count}, obj={obj:.8f}")
            break

    return best_a, best_obj


def spectral_perturb(a, amount, rng):
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[np.abs(A) < 1e-10] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s > 0:
        a_new = a_new * (a.sum() / s)
    return a_new


def propose_candidate(seed=42, budget_s=2300):
    """
    Main search function. Combines FISTA and exact coordinate descent.
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + budget_s - 40
    print(f"Starting optimization, budget={budget_s}s, deadline in {deadline - start_total:.0f}s")

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print(f"Using given sequence of length {len(given)}")
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_length(n, a_start, time_limit, label):
        nonlocal global_best, global_a
        if a_start is not None and len(a_start) != n:
            a_start = resample_sequence(a_start, n)
        if a_start is None:
            a_start = np.ones(n) * np.sqrt(2.0 * n) / n

        S = np.sqrt(2.0 * n)
        a_start = normalize_seq(a_start, S)

        print(f"\n=== {label}: n={n}, time={time_limit:.0f}s ===")
        obj_init = objective_value(a_start)
        print(f"  Initial obj: {obj_init:.6f}")

        best_local_a = a_start.copy()
        best_local_obj = obj_init

        # Phase 1: FISTA (35% of time)
        t_fista = time_limit * 0.35
        if t_fista > 30 and time.monotonic() < deadline - 60:
            a_f, obj_f = run_fista(a_start, S, t_fista, rng,
                                   beta_init=10.0, beta_rate=0.005,
                                   beta_max=2000.0, k_frac=0.03)
            obj_f = objective_value(a_f)
            print(f"  After FISTA: obj={obj_f:.6f}")
            if obj_f < best_local_obj:
                best_local_obj = obj_f
                best_local_a = a_f.copy()

        # Phase 2: Coordinate Descent (55% of time)
        remaining = deadline - time.monotonic() - 60
        t_cd = min(time_limit * 0.55, remaining)
        if t_cd > 30:
            a_cd, cd_obj = coordinate_descent(best_local_a, t_cd, rng,
                                              grid_size=100, max_passes=10000,
                                              verbose=True)
            obj_cd = objective_value(a_cd)
            print(f"  After CD: obj={obj_cd:.6f}")
            if obj_cd < best_local_obj:
                best_local_obj = obj_cd
                best_local_a = a_cd.copy()

        # Phase 3: Another FISTA pass on CD result (10% of time)
        remaining = deadline - time.monotonic() - 30
        t_fista2 = min(time_limit * 0.10, remaining)
        if t_fista2 > 20:
            a_f2, obj_f2 = run_fista(best_local_a, S, t_fista2, rng,
                                     beta_init=200.0, beta_rate=0.001,
                                     beta_max=10000.0, k_frac=0.01)
            obj_f2 = objective_value(a_f2)
            print(f"  After FISTA2: obj={obj_f2:.6f}")
            if obj_f2 < best_local_obj:
                best_local_obj = obj_f2
                best_local_a = a_f2.copy()

        # Phase 4: Final CD polish
        remaining = deadline - time.monotonic() - 20
        if remaining > 20:
            a_cd2, cd_obj2 = coordinate_descent(best_local_a, remaining, rng,
                                                grid_size=100, max_passes=5000,
                                                verbose=False)
            obj_cd2 = objective_value(a_cd2)
            print(f"  After CD2: obj={obj_cd2:.6f}")
            if obj_cd2 < best_local_obj:
                best_local_obj = obj_cd2
                best_local_a = a_cd2.copy()

        print(f"  Best for {label} n={n}: obj={best_local_obj:.6f}")
        if best_local_obj < global_best:
            global_best = best_local_obj
            global_a = best_local_a.copy()
            print(f"  *** New global best: {global_best:.6f} ***")

    # Main run at n=8192
    n_main = 8192
    remaining = deadline - time.monotonic() - 30
    if remaining > 120:
        try_length(n_main, given, remaining * 0.65, "Main")

    # Try a different initialization
    remaining = deadline - time.monotonic() - 30
    if remaining > 60:
        # Snapshot-like initialization
        a_snap = np.zeros(n_main)
        a_snap[:n_main // 2] = 1.0
        a_snap = normalize_seq(a_snap, np.sqrt(2.0 * n_main))
        # Mix with global best
        if global_a is not None and len(global_a) == n_main:
            a_mix = 0.5 * global_a + 0.5 * a_snap
            a_mix = normalize_seq(np.maximum(a_mix, 0), np.sqrt(2.0 * n_main))
        else:
            a_mix = a_snap
        try_length(n_main, a_mix, remaining * 0.5, "Snapshot")

    # Try n=4096 (faster per pass, more CD iterations)
    remaining = deadline - time.monotonic() - 30
    if remaining > 60:
        a_4096 = resample_sequence(global_a, 4096) if global_a is not None else None
        try_length(4096, a_4096, remaining * 0.6, "n4096")

    # Final intensive CD on the global best
    remaining = deadline - time.monotonic() - 20
    if remaining > 30 and global_a is not None:
        print(f"\n=== Final CD polish ({remaining:.0f}s) ===")
        a_fin, obj_fin = coordinate_descent(global_a, remaining, rng,
                                            grid_size=128, max_passes=10000,
                                            verbose=True)
        obj_fin = objective_value(a_fin)
        print(f"  Final obj: {obj_fin:.6f}")
        if obj_fin < global_best:
            global_best = obj_fin
            global_a = a_fin.copy()
            print(f"  *** Final global best: {global_best:.6f} ***")

    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print(f"\nVerification via evaluate_sequence: {ver:.6f}")
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - start_total
    print(f"Total time: {elapsed:.0f}s, returning sequence of length {len(final_list)}")
    return final_list
