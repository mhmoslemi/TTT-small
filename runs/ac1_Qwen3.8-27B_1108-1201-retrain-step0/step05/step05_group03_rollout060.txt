from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 using LSE smoothing, 
symmetry constraint (a[j] = a[n-1-j]), and FFT-based gradient computation.
Multi-length search with multi-start and adaptive beta.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def sym_to_full(p):
    return np.concatenate([p, p[::-1]])


def compute_lse_grad_full(w_full, a, n):
    """Compute gradient of LSE w.r.t. full sequence a using FFT."""
    size2 = (2 * n - 1) + n - 1
    fsize2 = 1 << (size2 - 1).bit_length()
    a_flip = a[::-1]
    W = fft(w_full, fsize2)
    AF = fft(a_flip, fsize2)
    conv_result = np.real(ifft(W * AF))
    return 2.0 * conv_result[n - 1 : 2 * n - 1]


def lse_weights(c, beta, k=None):
    """Compute LSE softmax weights."""
    size = len(c)
    if k is not None and k < size:
        idx = np.argsort(c)[::-1][:k]
        c_top = c[idx]
    else:
        idx = np.arange(size)
        c_top = c
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    max_scaled = scaled.max()
    if max_scaled > 500:
        beta = 500.0 / max(1e-15, max_scaled)
        scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w_full = np.zeros(size)
    w_full[idx] = e / max(tot, 1e-300)
    return w_full


def objective_from_conv(n, c_max, S):
    return 2.0 * n * c_max / (S * S)

                 beta_start=10.0, beta_end=5000.0,

    stuck = 0

            stuck = 0
            step = 0.1

    a = sym_to_full(best_p)
    c = fft_convolve_self(a)
    return best_p, c.max()

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 using:
- Symmetry constraint (a[j] = a[n-1-j]) to halve search space
- Full LSE smoothing with adaptive beta (log-spaced from low to high)
- FFT-based gradient computation (cross-correlation trick)
- FISTA momentum with backtracking line search
- Multi-length, multi-start exploration
- Final high-beta polish phase
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def sym_to_full(p):
    """Convert half-sequence to full symmetric sequence (even length)."""
    return np.concatenate([p, p[::-1]])


def lse_weights(c, beta, k=None):
    """Compute top-k LSE softmax weights."""
    size = len(c)
    if k is not None and k < size and k > 0:
        idx = np.argsort(c)[::-1][:k]
        c_top = c[idx]
    else:
        idx = np.arange(size)
        c_top = c
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    max_s = scaled.max()
    if max_s > 500:
        beta = 500.0 / max(1e-15, max_s)
        scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w_full = np.zeros(size)
    w_full[idx] = e / max(tot, 1e-300)
    return w_full


def compute_grad_full(w_full, a, n, L):
    """Compute LSE gradient w.r.t. full sequence using FFT cross-correlation.
    g[i] = 2 * sum_m w[m+i] * a[m]
    """
    W = fft(w_full, L)
    A = fft(a, L)
    g_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return g_a


def grad_to_sym(g_a, n):
    """Project full gradient to symmetric half-space."""
    half_n = n // 2
    g_p = g_a[:half_n] + g_a[n - 1: half_n - 1: -1]
    return g_p


def objective_val(n, a):
    """Compute the objective value directly."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def optimize_sym(n, p_init, S_full, time_limit, rng,
                 beta_start=5.0, beta_end=5000.0,
                 k_frac=0.08, use_momentum=True,
                 polish_frac=0.15):
    """
    Optimize symmetric sequence of length n (even).
    p_init: initial half-sequence, length n/2
    S_full: target sum for full sequence
    Returns (best_p, best_max_c)
    """
    half_n = n // 2
    S_half = S_full / 2.0

    p = project_simplex(p_init.copy(), S_half)
    y = p.copy()
    t_mom = 1.0
    start = time.monotonic()

    a = sym_to_full(p)
    c = fft_convolve_self(a)
    best_max = c.max()
    best_p = p.copy()

    it = 0
    stuck = 0
    step = 0.3 * S_half

    conv_size = 2 * n - 1
    k = max(8, int(k_frac * conv_size)) if k_frac else None
    L = 1 << (3 * n - 2).bit_length()

    polish_start = start + time_limit * (1.0 - polish_frac)

    while time.monotonic() - start < time_limit:
        it += 1
        elapsed = time.monotonic() - start

        a = sym_to_full(y)
        c = fft_convolve_self(a)

            x_new = y.copy()

        if accepted:

            step = max(step * 0.7, 1e-6 * S_half)

        # Momentum update

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - p)
            y_new = np.maximum(y_new, 0.0)
            y = y_new
            t_mom = t_new
        else:
            y = x_new.copy()

        p = x_new

        if stuck > 400 and frac < 0.7:

            y = project_simplex(y + rng.normal(0, noise_scale, half_n), S_half)
            t_mom = 1.0
            stuck = 0
            step = 0.05 * S_half

    # Final evaluation
    a = sym_to_full(best_p)
    c = fft_convolve_self(a)
    return best_p, c.max()


def resample_to_sym(given_seq, n):
    """Resample a given sequence to a symmetric sequence of length n."""
    half_n = n // 2
    t_old = np.linspace(0, 1, len(given_seq))
    # Sample first half and mirror
    t_new = np.linspace(0, 1, n)
    a_full = np.interp(t_new, t_old, given_seq)
    # Symmetrize
    a_sym = (a_full + a_full[::-1]) / 2.0
    return a_sym[:half_n].copy()
