Uses active-set L-BFGS, time-averaged Nesterov, and approximate LP directions.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 2).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def active_set_gradient(y, c, k_active, power=1.0):
    """Gradient using only the top-k active lags."""
    n = len(y)
    k_active = min(k_active, len(c))
    idx = np.argsort(c)[::-1][:k_active]
    vals = c[idx]
    # Weights: proportional to how close to max they are
    max_val = c.max()
    margin = 0.05 * max_val
    active_mask = c >= max_val - margin
    if active_mask.sum() > k_active:
        active_mask = np.zeros(len(c))
        active_mask[idx] = 1
    active_idx = np.where(active_mask)[0]
    if len(active_idx) == 0:
        active_idx = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active_idx] = 1.0 / len(active_idx)
    return conv_gradient(y, w_full), active_idx


def lse_objective_and_grad(a, beta, n_beta=1):
    """LSE-smoothed objective and gradient for max(conv(a,a))."""
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    w = e / tot
    obj = cm + np.log(tot) / beta
    grad = conv_gradient(a, w)
    return obj, grad, c


def compute_obj(a):

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_time_avg(a_init, max_time, rng, beta_start=20.0, use_time_avg=True):

            stuck_count = 0

        # Run L-BFGS for a limited number of iterations

                     'ftol': 1e-12, 'gtol': 1e-10, 'iprint': -1}

        elapsed = time.monotonic() - t0

        a_new = a_new * (S / max(a_new.sum(), 1e-15))

        else:

    return best_a, best_obj


def run_lp_direction(a_init, max_time, rng):
    """
    Approximate LP direction: find b >= 0 that maximizes sum(b)
    subject to convolve(a, b) <= M.
    Uses a simple iterative method instead of full LP solve.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    M = fft_convolve_self(a).max()
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()

    # Iterative water-filling to find b
    b = np.ones(n) * 0.1
    for it in range(500):
        if time.monotonic() - start > max_time * 0.5:
            break
        # Compute convolve(a, b)
        ab = fft_convolve_self_with(a, b)
        # Find the most violated constraint
        violations = ab - M
        violated = violations > 0
        if not np.any(violated):
            break
        # Reduce b where a is large (to reduce convolution)
        # Simple approach: b_j *= M / max(ab) if max(ab) > M
        max_viol = max_violations = ab.max()
        if max_viol > M:
            scale = M / max_viol
            b = b * (0.5 + 0.5 * scale)
        # Increase b where it can help
        # b_j can be increased by M / max_i a[i] / (something)
        # Simple: b_j += M / (n * max(a)) * (1 - a_j / max(a))
        max_a = a.max()
        if max_a > 0:
            increase = M / (n * max_a) * (1 - a / max_a)
            b = np.maximum(b + increase * 0.01, 0)
        # Project b to sum = S
        b = b * (S / max(b.sum(), 1e-15))

        # Mix with current a
        t = 0.1
        a_candidate = (1 - t) * a + t * b
        a_candidate = np.maximum(a_candidate, 0)
        a_candidate = a_candidate * (S / max(a_candidate.sum(), 1e-15))
        obj_cand = compute_obj(a_candidate)
        if obj_cand < best_obj:
            best_obj = obj_cand
            best_a = a_candidate.copy()
            a = a_candidate.copy()

    return best_a, best_obj

        if len(active_idx) > k_max_active:
            sorted_idx = active_idx[np.argsort(c[active_idx])[::-1]][:k_max_active]
            active_idx = sorted_idx
        elif len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c))])

        # L-BFGS on smooth subproblem: sum of active convolution values
        def smooth_obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            obj = b[active_idx].sum()
            w = np.zeros(len(b))
            w[active_idx] = 1.0
            grad = conv_gradient(x, w)
            return obj, grad

        remaining = max_time - (time.monotonic() - start)
        time_budget = min(20.0, remaining * 0.1)
        maxiter = max(10, int(time_budget * 8))

        t0 = time.monotonic()
        try:
            result = scipy_minimize(
                smooth_obj_and_grad, a.copy(), method='L-BFGS-B',
                jac=True, bounds=[(0, 1000)] * n,
                options={'maxiter': maxiter, 'ftol': 1e-13, 'gtol': 1e-11, 'iprint': -1}
            )
        except Exception:
            break

        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        sn = a_new.sum()
        if sn < 1e-10:
            break
        a_new = a_new * (S / sn)
        obj_new = compute_obj(a_new)

        if obj_new < compute_obj(a) - 1e-12:
            a = a_new.copy()
        else:
            a = a.copy()

    return best_a, best_obj


def run_lp_guided(a_init, max_time, rng):
    """
    LP-guided direction finding.
    Approximates the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0
    using iterative water-filling, then mixes with current solution.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    M = fft_convolve_self(a).max()

    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()

    # Initialize b as anti-correlated with a
    b = np.maximum(1.0 - a / a.max(), 0.01)
    b = b * (S / b.sum())

    for outer in range(30):
        if time.monotonic() - start > max_time:
            break

        # Inner iterations to approximate LP solution
        for inner in range(100):
            ab = fft_convolve_with(a, b)
            max_ab = ab.max()
            if max_ab <= M * (1 + 1e-6):
                break
            # Scale down b where it causes the most violation
            # Find b coordinates that contribute most to the peak
            peak_idx = int(np.argmax(ab))
            # Reduce b at positions that contribute to the peak
            # a[k-j] * b[j] contributes to ab[k]
            contrib = np.zeros(n)
            for j in range(n):
                k_j = peak_idx - j
                if 0 <= k_j < n:
                    contrib[j] = a[k_j] * b[j]
            total_contrib = contrib.sum()
            if total_contrib > 0:
                # Reduce the most contributing b[j]
                # Simple: scale b by M/max_ab with smoothing
                scale = 0.8 * (M / max_ab) + 0.2
                b = b * scale
                # Also push b toward where a is small
                a_norm = a / (a.max() + 1e-15)
                b = b + 0.01 * S * (1.0 - a_norm) / n * (1.0 - scale)
                b = np.maximum(b, 0.0)
                b = b * (S / max(b.sum(), 1e-15))

        # Try mixing
        for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
            candidate = (1.0 - t) * a + t * b
            candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-10:
                continue
            candidate = candidate * (S / cs)
            obj_c = compute_obj(candidate)
            if obj_c < best_obj:
                best_obj = obj_c
                best_a = candidate.copy()
                a = candidate.copy()
                # Update M
                M = fft_convolve_self(a).max()

    return best_a, best_obj


def run_hybrid_polish(a_init, max_time, rng):
    """
    Combined LSE-FISTA + active-set subgradient with adaptive switching.
    Switches between LSE smoothing and exact subgradient based on progress.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    stuck = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        # Adaptive gradient mode
        if stuck < 500:
            # LSE with moderate beta
            beta = 50.0
            cm = c.max()
            shifted = beta * (c - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            grad = conv_gradient(y, w)
        elif stuck < 2000:
            # Top-k gradient
            k = min(16, 4 + stuck // 200)
            idx = np.argsort(c)[::-1][:k]
            w = np.zeros(len(c))
            w[idx] = 1.0 / k
            grad = conv_gradient(y, w)
        else:
            # Exact subgradient with tolerance
            tol = min(0.1, 0.005 * stuck / 1000.0)
            max_val = c.max()
            active = np.where(c >= max_val * (1.0 - tol))[0]
            w = np.zeros(len(c))
            w[active] = 1.0 / len(active)
            grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Adaptive step size
        step = S / (gnorm * (1.0 + 0.01 * it))
        step = min(step, 2.0 * S / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 5000 and it % 500 == 0:
            # Block perturbation
            nblocks = rng.integers(2, 6)
            block_size = n // nblocks
            perm = rng.permutation(nblocks)
            y_new = np.zeros(n)
            for i in range(nblocks):
                start_i = i * block_size
                end_i = start_i + block_size if i < nblocks - 1 else n
                pi = perm[i] * block_size
                pei = pi + block_size if pi < nblocks - 1 else n
                y_new[start_i:end_i] = y[pi:pei]
            y = project_simplex(y_new, S)
            t_mom = 1.0
            stuck = 0

    return best_a, best_conv


def try_different_length(a_ref, n_new, budget, rng):
    """Resample reference to new length and optimize briefly."""
    S_new = np.sqrt(2.0 * n_new)
    # Interpolate from a_ref to length n_new
    n_ref = len(a_ref)
    x_old = np.linspace(0, 1, n_ref, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    a_new = np.interp(x_new, x_old, a_ref)
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new * (S_new / max(a_new.sum(), 1e-15))

    # Quick optimization
    a_opt, _ = run_fista_avg(a_new, budget, rng, beta_start=30.0, step_scale=0.8)
    return a_opt, compute_obj(a_opt)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA avg, active-set L-BFGS, LP-guided, hybrid polish,
    multi-length exploration, and final squeeze.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    n0 = len(a_start) if a_start is not None else 8192

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Init: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a_start = np.maximum(a_start, 0.01)
        obj0 = compute_obj(a_start)
        global_best_obj = obj0
        global_best_a = a_start.copy()
        print("Fallback init: n=%d, obj=%.8f" % (n0, obj0))

    # === Phase 1: FISTA with time averaging (25% time) ===
    remaining = deadline - time.monotonic()
    p1 = remaining * 0.25
    print("\n=== Phase 1: FISTA-avg (n=%d, %.0fs) ===" % (n0, p1))
    t1 = time.monotonic()

    a1, _ = run_fista_avg(a_start, p1 * 0.6, rng, beta_start=15.0, step_scale=1.0)
    obj1 = compute_obj(a1)
    print("  FISTA-avg: obj=%.8f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    rem1 = p1 - (time.monotonic() - t1)
    if rem1 > 5:
        a1b, _ = run_fista_avg(global_best_a, rem1, rng, beta_start=50.0, step_scale=0.5)
        obj1b = compute_obj(a1b)
        print("  FISTA-avg (high beta): obj=%.8f" % obj1b)
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: Active-set L-BFGS (30% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2 = remaining * 0.30
        print("\n=== Phase 2: Active-set L-BFGS (budget %.0fs) ===" % p2)
        t2 = time.monotonic()
        a2, obj2 = run_active_lbfgs(global_best_a, p2, rng, k_max_active=8)
        print("  Active L-BFGS: obj=%.8f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: Hybrid polish (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3 = remaining * 0.20
        print("\n=== Phase 3: Hybrid polish (budget %.0fs) ===" % p3)
        t3 = time.monotonic()
        a3, _ = run_hybrid_polish(global_best_a, p3, rng)
        obj3 = compute_obj(a3)
        print("  Hybrid: obj=%.8f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: LP-guided direction (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4 = remaining * 0.15
        print("\n=== Phase 4: LP-guided (budget %.0fs) ===" % p4)
        t4 = time.monotonic()
        a4, obj4 = run_lp_guided(global_best_a, p4, rng)
        print("  LP-guided: obj=%.8f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Diverse restarts + multi-length (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5 = remaining * 0.10
        print("\n=== Phase 5: Diverse restarts (budget %.0fs) ===" % p5)
        t5 = time.monotonic()

        # Try flipped
        a_flip = global_best_a[::-1].copy()
        S_flip = np.sqrt(2.0 * n0)
        a_flip = a_flip * (S_flip / a_flip.sum())
        a5, _ = run_fista_avg(a_flip, p5 * 0.25, rng, beta_start=30.0, step_scale=0.7)
        obj5 = compute_obj(a5)
        print("  Flipped: obj=%.8f" % obj5)
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        # Try slightly different cosinusoid init
        t = np.linspace(0, 1, n0, endpoint=False)
        for phase in [0.1, 0.3]:
            a_cos = 0.48 + 0.52 * np.cos(2 * np.pi * t + phase)
            a_cos = np.maximum(a_cos, 0.01)
            S_c = np.sqrt(2.0 * n0)
            a_cos = a_cos * (S_c / a_cos.sum())
            a6, _ = run_fista_avg(a_cos, p5 * 0.15, rng, beta_start=20.0, step_scale=0.9)
            obj6 = compute_obj(a6)
            print("  Cos phase %.1f: obj=%.8f" % (phase, obj6))
            if obj6 < global_best_obj:
                global_best_obj = obj6
                global_best_a = a6.copy()

        # Try different length
        remaining5 = p5 - (time.monotonic() - t5)
        if remaining5 > 10:
            n_alt = 16384
            a7, obj7 = try_different_length(global_best_a, n_alt, remaining5 * 0.8, rng)
            print("  n=%d: obj=%.8f" % (n_alt, obj7))
            if obj7 < global_best_obj:
                global_best_obj = obj7
                global_best_a = a7.copy()

    # === Phase 6: Final active-set + FISTA squeeze (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n=== Phase 6: Final squeeze (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()

        # Active set polish
        p6a = remaining * 0.5
        a8, obj8 = run_active_lbfgs(global_best_a, p6a, rng, k_max_active=5)
        print("  Final active L-BFGS: obj=%.8f (t=%.0fs)" % (obj8, time.monotonic() - t6))
        if obj8 < global_best_obj:
            global_best_obj = obj8
            global_best_a = a8.copy()

        # Quick FISTA
        rem6 = remaining - (time.monotonic() - t6)
        if rem6 > 5:
            a9, _ = run_fista_avg(global_best_a, rem6 * 0.7, rng, beta_start=100.0, step_scale=0.3)
            obj9 = compute_obj(a9)
            print("  Final FISTA: obj=%.8f" % obj9)
            if obj9 < global_best_obj:
                global_best_obj = obj9
                global_best_a = a9.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective: %.8f" % global_best_obj)
    return final_list
