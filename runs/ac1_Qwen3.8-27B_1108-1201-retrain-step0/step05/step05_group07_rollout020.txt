"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses LSE-smoothed gradient descent with Nesterov momentum, adaptive beta,
top-k subgradient hybrid, coordinate descent polish, and multi-length exploration.

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    max_sh = shifted.max()
    if max_sh > 400:
        shifted = shifted - max_sh
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(a, c, k, temperature=0.0):
    """Weighted subgradient from top-k lags with optional temperature smoothing."""
    n = len(a)
    k = min(k, 2 * n - 1)
    w_full = np.zeros(2 * n - 1)
    if temperature > 1e-10:
        scaled = c / temperature
        max_s = scaled.max()
        e = np.exp(scaled - max_s)
        w_full = e / e.sum()
        # Boost top-k
        top_idx = np.argsort(c)[::-1][:k]
        boost = 1.0 + 1.0 * (k / (2 * n - 1))
        w_full = w_full * boost
        w_full[top_idx] = w_full[top_idx] * (1.0 + k / (2.0 * n))
        w_full = w_full / w_full.sum()
    else:
        top_idx = np.argsort(c)[::-1][:k]
        median_c = np.median(c)
        weights = np.maximum(c[top_idx] - median_c, 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(k) / k
        else:
            weights = weights / wsum
        w_full[top_idx] = weights

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def optimize_lse_momentum(a0, S, max_time, rng, beta_init=30.0, beta_rate=0.004,

    """
    LSE gradient descent with Nesterov momentum and adaptive beta.
    """
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    beta = beta_init
    momentum = momentum_init

    it = 0
    stuck = 0

        c = fft_convolve_self(a)

            stuck = 0

            beta *= 1.05
            patience = int(patience * 1.1)

            step_look = step

        a_try = project_simplex(a_look + v, S)

            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, mode={mode}, stuck={stuck}")

    return best_a, best


def coordinate_descent(a0, S, max_time, rng, verbose=False):
    """
    Coordinate descent: optimize one coordinate at a time.
    For coordinate i, conv_k is quadratic in a_i for the few k where a_i appears.
    We minimize the max of these quadratics subject to non-negativity.
    """
    n = len(a0)
    a = project_simplex(a0, S).copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    full_pass = 0

    while time.monotonic() - start < max_time:
        full_pass += 1
        improved = 0

        # Random shuffle order
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > max_time:
                break

            # Save current value
            a_save = a[i]

            # The lags where a[i] appears: k in [i, i+n-1] intersect [0, 2n-2]
            k_min = max(0, i)
            k_max = min(2 * n - 2, i + n - 1)
            active_k = np.arange(k_min, k_max + 1)

            # For each active k: conv_k = a[i]^2 * (1 if k==2i else 0)
            #   + a[i] * (sum of a[j]*a[k-j] for j != i, k-j != i) + const
            # Simplification: convex in a[i] for a[i] >= 0 (quadratic with positive leading coeff)

            # Compute coefficients: conv_k = alpha_k * a[i]^2 + beta_k * a[i] + gamma_k
            alphas = np.zeros(len(active_k))
            betas = np.zeros(len(active_k))
            gammas = np.zeros(len(active_k))

            for idx, kk in enumerate(active_k):
                kk = int(kk)
                # Terms: sum_{j} a[j] * a[kk-j]
                # Split into terms involving a[i] and not
                j_vals = np.arange(max(0, kk - n + 1), min(n, kk + 1))
                # For each j in j_vals, the term is a[j] * a[kk-j]
                # a[i] appears when j == i or kk-j == i

                count_ai = 0
                linear_coeff = 0.0
                const = 0.0

                for j in j_vals:
                    j = int(j)
                    partner = kk - j
                    if 0 <= j < n and 0 <= partner < n:
                        if j == i and partner == i:
                            count_ai += 1
                        elif j == i:
                            linear_coeff += a[partner]
                        elif partner == i:
                            linear_coeff += a[j]
                        else:
                            const += a[j] * a[partner]

                alphas[idx] = count_ai * 0.5  # a[i]^2 coefficient
                betas[idx] = linear_coeff
                gammas[idx] = const

            # Now minimize max_k (alpha_k * x^2 + beta_k * x + gamma_k) for x >= 0
            # Also maintain sum(a) = S: if we change a[i] from a_save to x,
            # we need to adjust another coordinate. For simplicity, we adjust
            # the "least important" coordinate (smallest contribution to max conv).

            # Bisection on t: find smallest t such that exists x >= 0 with
            # alpha_k * x^2 + beta_k * x + gamma_k <= t for all k,
            # and we can fix the sum constraint.

            # Simpler approach: try a few candidate values of x and pick best
            # that doesn't increase the max too much.

            # First, what's the gradient at current point?
            # d(conv_k)/d(a_i) at a_i = a_save:
            grads_at_point = 2 * alphas * a_save + betas

            # If all grads are <= 0, a[i] should increase; if all >= 0, decrease.
            # But we need to maintain the sum, so we just do a small line search.

            # Try perturbing a[i] by delta and compensating elsewhere
            delta_options = [0.01 * S / n, -0.01 * S / n,
                             0.005 * S / n, -0.005 * S / n]

            best_obj = None
            best_new_a = None

            for delta in delta_options:
                x = a_save + delta
                if x < 0:
                    continue
                # Compensation: distribute -delta to other coordinates proportionally
                a_new = a.copy()
                a_new[i] = x
                deficit = S - a_new.sum()
                if abs(deficit) < 1e-15:
                    continue
                # Distribute to coordinates with smallest gradient
                other_idx = [j for j in range(n) if j != i]
                other_grads = np.zeros(n)
                # Approximate: just distribute to the coordinate with smallest value
                comp_idx = np.argmin(a[other_idx])
                if a_new[comp_idx] + deficit <= 0:
                    # Can't compensate here, skip
                    continue
                a_new[comp_idx] += deficit
                if a_new[comp_idx] < 0:
                    continue

                c_new = fft_convolve_self(a_new)
                obj_new = c_new.max()
                if best_obj is None or obj_new < best_obj:
                    best_obj = obj_new
                    best_new_a = a_new.copy()

            if best_new_a is not None and best_obj < best * (1 - 1e-10):
                a = best_new_a
                improved += 1
                if a.sum() > 0:
                    c_check = fft_convolve_self(a)
                    if c_check.max() < c.max() * (1 - 1e-8):
                        c = c_check

        # Recompute and check
        c = fft_convolve_self(a)
        if c.max() < best - 1e-12:
            best = c.max()
            best_a = a.copy()

        if improved == 0:
            # Try random restart of coordinate order with slight noise
            break

        if verbose and full_pass % 5 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    CD pass={full_pass}, t={elapsed:.0f}s, obj={cur_obj:.6f}, improved={improved}")

    return best_a, best


def interpolate_sequence(base, n_new):
    """Interpolate a base sequence to a new length."""
    if len(base) < 2:
        return np.ones(n_new) * (np.sqrt(2.0 * n_new) / n_new)
    t_old = np.linspace(0, 1, len(base))
    t_new = np.linspace(0, 1, n_new)
    result = np.interp(t_new, t_old, base)
    return np.maximum(result, 0.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Multi-phase: LSE+momentum, hybrid subgradient, coordinate descent,
    multi-restart, multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    print(f"Starting optimization, budget={budget_s}s")

    # Get provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 128:
            print(f"Using provided sequence: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_global(a, msg=""):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            if msg:
                print(f"  ** NEW GLOBAL BEST: obj={obj:.6f} ({msg})")

    # === Phase 1: Primary LSE+momentum on provided sequence ===
    if a_provided is not None:
        n1 = len(a_provided)
        S1 = np.sqrt(2.0 * n1)
        a1 = a_provided * (S1 / a_provided.sum())

        t1 = min(600, (deadline - time.monotonic()) * 0.30)
        print(f"\nPhase 1a: LSE+momentum, n={n1}, time={t1:.0f}s")
        a_opt, _ = optimize_lse_momentum(a1, S1, t1, rng,
                                         beta_init=20.0, beta_rate=0.003,
                                         momentum_init=0.92, momentum_decay=0.0003,
                                         verbose=True)
        update_global(a_opt, "Phase 1a LSE+momentum")

        # Hybrid refinement
        remaining = deadline - time.monotonic()
        t1b = min(400, remaining * 0.25)
        if t1b > 30:
            print(f"Phase 1b: Hybrid, n={n1}, time={t1b:.0f}s")
            a_hyb, _ = optimize_hybrid(a_opt, S1, t1b, rng, verbose=True)
            update_global(a_hyb, "Phase 1b Hybrid")

    # === Phase 2: Multi-length exploration ===
    remaining = deadline - time.monotonic()
    lengths_to_try = [4096, 8192, 12288, 16384]

    # Remove the length we already tried
    if a_provided is not None:
        lengths_to_try = [l for l in lengths_to_try if l != len(a_provided)]

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 120:
            break
        if global_a is not None and len(global_a) == n_try:
            continue

        S_try = np.sqrt(2.0 * n_try)
        t_alloc = min(300, (deadline - time.monotonic()) * 0.20)
        if t_alloc < 40:
            continue

        print(f"\nPhase 2: n={n_try}, time={t_alloc:.0f}s")
        if global_a is not None:
            init = interpolate_sequence(global_a, n_try)
        else:
            tt = np.linspace(0, 1, n_try, endpoint=False)
            init = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.05
        init = np.maximum(init, 0.0)
        init = project_simplex(init, S_try)

        a_opt, _ = optimize_lse_momentum(init, S_try, t_alloc * 0.55, rng,
                                         beta_init=30.0, beta_rate=0.004,
                                         momentum_init=0.9, momentum_decay=0.0004)
        update_global(a_opt, f"Phase 2 n={n_try} LSE")

        rem = deadline - time.monotonic()
        if rem > 30:
            a_hyb, _ = optimize_hybrid(a_opt, S_try, min(120, rem * 0.35), rng)
            update_global(a_hyb, f"Phase 2 n={n_try} Hybrid")

    # === Phase 3: Coordinate descent polish on global best ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        t3 = min(300, remaining * 0.4)
        print(f"\nPhase 3: Coordinate descent polish, n={n}, time={t3:.0f}s")
        a_cd, _ = coordinate_descent(global_a, S, t3, rng, verbose=True)
        update_global(a_cd, "Phase 3 CD")

    # === Phase 4: Multi-restart from best ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 40:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        n_restarts = 4
        t_per = (remaining - 20) / n_restarts

        print(f"\nPhase 4: {n_restarts} perturbations+refines, {t_per:.0f}s each")
        for ri in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            # Different noise scales for diversity
            noise_scale = (0.01 + 0.015 * ri) * S / np.sqrt(n)
            a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n), S)

            # Quick LSE optimization
            t_quick = max(10, t_per * 0.6)
            a_res, _ = optimize_lse_momentum(a_pert, S, t_quick, rng,
                                             beta_init=50.0, beta_rate=0.005,
                                             momentum_init=0.85, momentum_decay=0.001)
            update_global(a_res, f"Phase 4 restart {ri}")

            # Quick hybrid
            rem = deadline - time.monotonic()
            if rem > 20:
                a_res2, _ = optimize_hybrid(a_res, S, min(60, rem * 0.4), rng)
                update_global(a_res2, f"Phase 4 restart {ri} hybrid")

    # === Phase 5: Final intensive LSE on best ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        t5 = remaining - 10
        print(f"\nPhase 5: Final LSE squeeze, n={n}, time={t5:.0f}s")
        a_final, _ = optimize_lse_momentum(global_a, S, t5, rng,
                                           beta_init=100.0, beta_rate=0.002,
                                           momentum_init=0.7, momentum_decay=0.001)
        update_global(a_final, "Phase 5 Final")

    # === Phase 6: Try a few more lengths if time remains ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        # Try extending the best sequence
        for n_ext in [len(global_a) * 2]:
            if n_ext > 32768 or time.monotonic() > deadline - 80:
                break
            S_ext = np.sqrt(2.0 * n_ext)
            init_ext = interpolate_sequence(global_a, n_ext)
            init_ext = project_simplex(np.maximum(init_ext, 0), S_ext)
            t_ext = min(100, remaining * 0.3)
            print(f"\nPhase 6: Extension to n={n_ext}, time={t_ext:.0f}s")
            a_ext, _ = optimize_lse_momentum(init_ext, S_ext, t_ext, rng,
                                             beta_init=30.0, beta_rate=0.003)
            update_global(a_ext, f"Phase 6 ext n={n_ext}")

    # === Final verification ===
    if global_a is None:
        print("WARNING: No valid solution, using fallback")
        global_a = np.ones(256) * 1.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: len={n_final}, internal_obj={global_best:.6f}")
    print(f"evaluate_sequence verdict: {ver:.6f}")
    print(f"{'='*60}")

    return final_list
