import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_gradient(a, c, k, beta):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_value(a):

    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def resample_sequence(a, n_target):
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def lse_refine(a_init, S, max_time, rng):

    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    k = max(4, int(0.08 * (2 * n - 1)))
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(3000.0, 15.0 * np.exp(0.005 * it))
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
        grad = topk_lse_gradient(y, c, k, beta)
        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(5):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-12:
                break
            step *= 0.5
            if step < 1e-12:
                break
            x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_val = t_new
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
1. Start from the given sequence (previous best at n=8192).
2. Brief LSE-smoothed gradient refinement for basin identification.
3. Extensive pairwise coordinate descent on the simplex (exact 1D solves via bisection).
4. Try multiple sequence lengths (4096, 8192, 16384) and pick the best.
5. Final high-precision polishing.

The key innovation over the parent is the pairwise coordinate descent phase,
which performs exact 1D minimization of the max autocorrelation for each pair,
guaranteeing monotonic progress and avoiding smoothing approximation errors.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_value(a):

    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)

        if j * 2 >= k_lo and j * 2 <= k_hi:
            diag_j_mask[j * 2 - k_lo] = True

        # Bisection to find optimal t in [0, c_val]
        t_lo = 0.0
        t_hi = c_val

        # Warm start: try current value first
        t_mid = ai_old
        for _ in range(bisection_iters):
            t_mid = 0.5 * (t_lo + t_hi)

            # Compute b_k(t_mid) for affected k
            # Delta from i: change when a[i] goes from ai_old to t_mid
            # Delta from j: change when a[j] goes from aj_old to (c_val - t_mid)
            new_b_aff = b_aff.copy()

            if np.any(i_valid):
                idx_i = np.where(i_valid)[0]
                if diag_i_mask[idx_i].any():
                    new_b_aff[idx_i[diag_i_mask[idx_i]]] += t_mid * t_mid - ai_old * ai_old
                off_i = ~diag_i_mask[idx_i]
                if off_i.any():
                    new_b_aff[idx_i[off_i]] += 2.0 * (t_mid - ai_old) * a_ki[off_i]

            if np.any(j_valid):
                idx_j = np.where(j_valid)[0]
                c_minus_t = c_val - t_mid
                if diag_j_mask[idx_j].any():
                    new_b_aff[idx_j[diag_j_mask[idx_j]]] += c_minus_t * c_minus_t - aj_old * aj_old
                off_j = ~diag_j_mask[idx_j]
                if off_j.any():
                    new_b_aff[idx_j[off_j]] += 2.0 * (c_minus_t - aj_old) * a_kj[off_j]

            new_max_aff = new_b_aff.max()
            g_val = max(new_max_aff, M_unaff)

            # Check if we can do no better than M_unaff
            if g_val <= M_unaff + 1e-14:
                # Any t that keeps affected max below M_unaff is optimal
                break

            # Compute subgradient at t_mid
            active = new_b_aff >= (g_val - 1e-12)
            if not np.any(active):
                active = np.zeros(len(new_b_aff), dtype=bool)
                active[np.argmax(new_b_aff)] = True

            d_min = np.inf
            d_max = -np.inf
            for ai_idx in np.where(active)[0]:
                k = k_aff[ai_idx]
                d = 0.0
                # Derivative of i-contribution w.r.t. t:
                if k >= i and k - i < n:
                    if k == 2 * i:
                        d += 2.0 * t_mid
                    else:
                        d += 2.0 * a[k - i]
                # Derivative of j-contribution w.r.t. t (a_j = c_val - t, so da_j/dt = -1):
                if k >= j and k - j < n:
                    if k == 2 * j:
                        d += -2.0 * (c_val - t_mid)
                    else:
                        d += -2.0 * a[k - j]
                if d < d_min:
                    d_min = d
                if d > d_max:
                    d_max = d

            if d_min < -1e-12 or d_max > 1e-12:
                # 0 is in subdifferential (or mist), we're at optimum
                break
            elif d_max < 0:
                # All derivatives negative: increase t
                t_lo = t_mid
            else:
                # All derivatives positive: decrease t
                t_hi = t_mid

        t_opt = 0.5 * (t_lo + t_hi)
        t_opt = np.clip(t_opt, 0.0, c_val)

        # Apply update
        a[i] = t_opt
        a[j] = c_val - t_opt

        # Incremental update of b
        if np.any(i_valid):
            idx_i = np.where(i_valid)[0]
            if diag_i_mask[idx_i].any():
                b[k_lo + idx_i[diag_i_mask[idx_i]]] += t_opt * t_opt - ai_old * ai_old
            off_i = ~diag_i_mask[idx_i]
            if off_i.any():
                b[k_lo + idx_i[off_i]] += 2.0 * (t_opt - ai_old) * a_ki[off_i]

        if np.any(j_valid):
            idx_j = np.where(j_valid)[0]
            c_minus_t_opt = c_val - t_opt
            if diag_j_mask[idx_j].any():
                b[k_lo + idx_j[diag_j_mask[idx_j]]] += c_minus_t_opt * c_minus_t_opt - aj_old * aj_old
            off_j = ~diag_j_mask[idx_j]
            if off_j.any():
                b[k_lo + idx_j[off_j]] += 2.0 * (c_minus_t_opt - aj_old) * a_kj[off_j]

        # Update tracking
        cur_b_max = b.max()
        if cur_b_max < best_b_max * (1.0 - 1e-10):
            best_b_max = cur_b_max
            best_a = a.copy()
            last_improve = it
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Periodic FFT recompute to correct floating point drift
        if it % fft_recompute_interval == 0:
            b = fft_convolve_self(a)
            cur_b_max = b.max()
            if cur_b_max < best_b_max:
                best_b_max = cur_b_max
                best_a = a.copy()

        # Restart if stuck
        if no_improve_count > 200000 and time.monotonic() - start < max_time * 0.8:
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            b = fft_convolve_self(a)
            no_improve_count = 0

        # Progress reporting
        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - start
            obj = 2.0 * n * best_b_max / (S * S)
            print(f"    CD iter {it:>7d}: obj={obj:.7f}, elapsed={elapsed:.0f}s")

    # Final FFT verify
    b_final = fft_convolve_self(a)
    if b_final.max() < best_b_max:
        best_b_max = b_final.max()
        best_a = a.copy()

    return best_a, best_b_max


def high_beta_polish(a_init, S, max_time, rng):
    """Very high beta LSE polishing for final refinement."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    k = max(4, int(0.02 * (2 * n - 1)))
    it = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(50000.0, 200.0 * (1.0 + it / 300.0))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best * (1.0 - 1e-10):
            best = cur_max
            best_a = a.copy()

        grad = topk_lse_gradient(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        base_step = 0.3 / (1.0 + 0.0005 * it)
        step = base_step * step_scale
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
            step_scale = min(3.0, step_scale * 1.05)
        else:
            step_scale = max(0.01, step_scale * 0.5)
            a = a_new

        if it % 1000 == 0 and (time.monotonic() - start) / max_time < 0.6:
            noise_scale = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def propose_candidate():
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses LSE gradient refinement followed by pairwise coordinate descent.
    """
    rng = np.random.default_rng(12345)
    start_total = time.monotonic()
    total_budget = 2380.0
    deadline = start_total + total_budget

    print("Starting optimization, budget=%.0fs" % total_budget)

    # Get starting sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_length(n, time_limit, a_start):
        nonlocal global_best, global_a
        S = np.sqrt(2.0 * n)

        if a_start is not None:
            if len(a_start) != n:
                a_start = resample_sequence(a_start, n)
            a_start = normalize(a_start.copy(), S)
        else:
            a_start = np.ones(n) * S / n

        print("  === n=%d, time_limit=%.0fs ===" % (n, time_limit))
        t0 = time.monotonic()

        # Phase 1: LSE refine (20% of time)
        t1 = time_limit * 0.20
        if t1 > 5:
            a_opt, val_opt = lse_refine(a_start, S, min(t1, deadline - t0 - 10), rng)
        else:
            a_opt = a_start.copy()
            val_opt = fft_convolve_self(a_opt).max()

        obj1 = 2.0 * n * val_opt / (S * S)
        print("    After LSE: obj=%.7f" % obj1)

        # Phase 2: Pairwise CD (65% of time)
        t2 = time_limit * 0.65
        remaining = deadline - time.monotonic() - 10
        t2 = min(t2, remaining)
        if t2 > 5:
            a_opt2, val_opt2 = pairwise_cd(a_opt, S, t2, rng, bisection_iters=25, verbose=True)
            if val_opt2 < val_opt:
                a_opt, val_opt = a_opt2, val_opt2

        obj2 = 2.0 * n * val_opt / (S * S)
        print("    After CD: obj=%.7f" % obj2)

        # Phase 3: Final polish (15% of time)
        remaining = deadline - time.monotonic() - 10
        t3 = min(time_limit * 0.15, remaining)
        if t3 > 3:
            a_opt3, val_opt3 = high_beta_polish(a_opt, S, t3, rng)
            if val_opt3 < val_opt:
                a_opt, val_opt = a_opt3, val_opt3

        obj3 = 2.0 * n * val_opt / (S * S)
        print("    Final: obj=%.7f" % obj3)

        if obj3 < global_best:
            global_best = obj3
            global_a = a_opt.copy()
            print("    *** New global best: %.7f (n=%d) ***" % (global_best, n))

    # Allocation: focus on n=8192, try others
    # n=8192: 50% of time
    t_8192 = total_budget * 0.48
    remaining = deadline - time.monotonic() - 5
    t_8192 = min(t_8192, remaining)
    if t_8192 > 30:
        try_length(8192, t_8192, given)

    # n=4096: 20% of time
    remaining = deadline - time.monotonic() - 5
    t_4096 = min(total_budget * 0.20, remaining)
    if t_4096 > 20 and global_a is not None:
        a_4096 = resample_sequence(global_a, 4096)
        try_length(4096, t_4096, a_4096)

    # n=16384: 15% of time
    remaining = deadline - time.monotonic() - 5
    t_16384 = min(total_budget * 0.15, remaining)
    if t_16384 > 20 and global_a is not None:
        a_16384 = resample_sequence(global_a, 16384)
        try_length(16384, t_16384, a_16384)

    # n=12288: remaining time
    remaining = deadline - time.monotonic() - 10
    if remaining > 30 and global_a is not None:
        t_12288 = remaining
        a_12288 = resample_sequence(global_a, 12288)
        try_length(12288, t_12288, a_12288)

    # Final extensive CD polish on the global best
    remaining = deadline - time.monotonic() - 20
    if remaining > 10 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final CD Polish (%.0fs, n=%d) ===" % (remaining, n_best))
        a_fin, val_fin = pairwise_cd(global_a, S_best, min(remaining * 0.7, deadline - time.monotonic() - 15),
                                     rng, bisection_iters=30, verbose=True)
        obj_fin = 2.0 * n_best * val_fin / (S_best * S_best)
        if obj_fin < global_best:
            global_best = obj_fin
            global_a = a_fin.copy()
            print("  After final CD: obj=%.7f" % global_best)

        # Tiny LSE polish
        remaining2 = deadline - time.monotonic() - 15
        if remaining2 > 5:
            a_pol, val_pol = high_beta_polish(global_a, S_best, remaining2, rng)
            obj_pol = 2.0 * n_best * val_pol / (S_best * S_best)
            if obj_pol < global_best:
                global_best = obj_pol
                global_a = a_pol.copy()
                print("  After final polish: obj=%.7f" % global_best)

    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.7f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, best obj=%.7f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
