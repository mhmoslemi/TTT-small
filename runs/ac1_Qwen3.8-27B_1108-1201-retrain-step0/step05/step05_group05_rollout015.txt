import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_grad(c, beta, a):
    """Compute LSE gradient of max self-convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def fixed_point_diversify(a, S, rng, n_iters=50):
    """
    Fixed-point iteration in frequency domain to generate diverse candidates.
    Philosophy: at optimum, (lambda * p) ≈ const on support, so p ∝ 1/lambda in freq domain.
    """
    n = len(a)
    p = normalize_to_sum(a, 1.0)  # Work with sum=1 for clarity
    best_p = p.copy()
    best_val = fft_convolve_self(p).max()
    
    for it in range(n_iters):
        c = fft_convolve_self(p)
        cmax = c.max()
        if cmax < best_val:
            best_val = cmax
            best_p = p.copy()
        
        # Active set: top 5% of lags
        k_active = max(5, len(c) // 20)
        top_idx = np.argsort(c)[::-1][:k_active]
        
        # Build lambda (uniform over active lags)
        fsize = 1 << (2 * n - 1).bit_length()
        lam = np.zeros(fsize)
        lam[top_idx] = 1.0 / k_active
        
        # Lambda spectrum
        Lam = fft(lam, fsize)
        
        # P = 1/Lam (in frequency domain)
        # Avoid division by zero
        abs_Lam = np.abs(Lam)
        mask = abs_Lam > 1e-10
        P = np.zeros(fsize, dtype=np.complex128)
        P[mask] = 1.0 / Lam[mask]
        
        # Inverse DFT
        p_new = np.real(ifft(P, fsize))[:n]
        
        # Project to simplex
        p_new = project_simplex(p_new, 1.0)
        
        # Check if non-trivial
        if p_new.sum() < 1e-10:
            break
        
        p = p_new
    
    # Scale back to sum S
    best_p = best_p * S
    return best_p


def basin_hopping(a_init, S, max_time, rng, opt_time=20.0):
    """Basin hopping with spectral mutations."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    
    n_rounds = max(1, int((max_time - opt_time) / (opt_time + 2.0)))
    
    for r in range(n_rounds):
        if time.monotonic() - start > max_time - opt_time:
            break
        progress = r / max(1, n_rounds)
        
        # Mutation: spectral
        fsize = 1 << (2 * n - 1).bit_length()
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)
        
        scale = 0.15 * np.exp(-3.0 * progress) + 0.01
        phase_noise = rng.normal(0, scale, fsize)
        mag_noise = 1.0 + rng.normal(0, 0.05 * scale, fsize)
        
        new_A = mag * mag_noise * np.exp(1j * (phase + phase_noise))
        new_A[0] = A[0]  # Preserve DC
        
        a_mut = np.real(ifft(new_A, fsize))[:n]
        a_mut = project_simplex(a_mut, S)

        if val < best:
            best = val
            best_a = a_opt.copy()
            print("    BH round %d: new best max_b=%.10f" % (r, best))

def generate_init(n, rng, given=None):

            print("  Screen %d: max_b=%.8f, obj=%.6f" % (i, maxb_screen, 2.0*n*maxb_screen/S**2))
        
        candidates.sort(key=lambda x: x[0])
        
        # Phase 2: Main FISTA optimization (40% of budget)
        p2_time = t_budget * 0.4
        if p2_time > 10:
            a_main, maxb_main = fista_beta(candidates[0][1], S, p2_time, rng,
                                            beta_init=100.0, beta_max=80000.0,
                                            beta_rate=0.002, max_no_improve=5000)
            try_update(a_main, n)
            print("  After main FISTA: obj=%.6f" % global_best_obj)
        
        # Phase 3: Basin hopping (25% of budget)
        p3_time = t_budget * 0.25
        if p3_time > 30:
            opt_time = max(10.0, p3_time / 15.0)
            a_bh, maxb_bh = basin_hopping(a_main if 'a_main' in dir() else candidates[0][1],
                                           S, p3_time, rng, opt_time=opt_time)
            try_update(a_bh, n
