def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 1e-30, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def soft_max_gradient(a, c, beta):
    """
    Compute gradient of log-sum-exp(beta * c) with respect to a.
    Returns (gradient, lse_value).
    """
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    # Gradient of (2/beta) * LSE(beta*c) w.r.t. a
    # d/d_a [sum_k w_k * c[k]] where c[k] = sum_j a[j]*a[k-j]
    # = 2 * sum_k w_k * a[k-j] (approximately, ignoring diagonal)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + np.log(tot) / beta
    return grad, lse_val

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def cosine_init(n, center=0.35, width=0.4):
    """Raised cosine initialization."""
    t = np.linspace(0, 1, n)
    x = (t - center) / width
    a = np.ones(n)
    mask = np.abs(x) < 1
    a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
    a[~mask] = 0.0
    return np.maximum(a, 1e-10)

def gaussian_init(n, mu=0.4, sigma=0.2):
    """Gaussian bump."""
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    return np.maximum(a, 1e-10)

def asymmetric_init(n, alpha=0.3):
    """Asymmetric shape: rises on left, falls on right with different rates."""
    t = np.linspace(0, 1, n)
    a = np.exp(-alpha * t) * (1 - np.exp(-5 * t))
    return np.maximum(a, 1e-10)

"""
Minimize 2*n*max(f*f)/sum(f)^2 using a multi-stage approach combining
multiplicative (mirror) descent, LSE-FISTA, active-set refinement, and
directed peak shaving. Key innovations over previous: (1) multiplicative
updates that naturally respect non-negativity, (2) adaptive target-tracking
with Polyak-like step sizes, (3) multi-resolution upsampling, (4) improved
asymmetric initializations.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 1e-30, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_grad(a, c, beta):
    """Gradient of LSE(beta*c) w.r.t. a. Returns (grad, lse_value)."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = np.clip(shifted, -100, 100)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + np.log(tot) / beta
    return grad, lse_val


def active_set_grad(a, c, threshold_frac=0.999):
    """Hard active-set gradient using lags above threshold_frac * max."""
    n = len(a)
    cm = c.max()
    thresh = cm * threshold_frac
    w = (c >= thresh).astype(np.float64)
    tot = w.sum()
    if tot < 1:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        tot = 1.0
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

                  beta_start=20.0, beta_end=50000.0, beta_steps=200,
                  momentum_period=300):
    """FISTA with LSE smoothing and periodic momentum restart."""
    a = normalize_sum(a_init.copy(), n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 200000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        grad, _ = lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        step = 1.0 / beta
        # Adaptive step reduction on stagnation
        stagnant = it - last_improve
        if stagnant > 200:
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum restart
        if it % momentum_period == 0:
            t_mom = 1.0
            y = a.copy()

        # Noise injection on long stagnation
        if stagnant > 1000 and it % 100 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 20000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                last_improve = it

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_multiplicative(a_init, n, S, max_time, rng,
                       beta_start=20.0, beta_end=80000.0,
                       beta_steps=150, eta_base=0.4):
    """
    Multiplicative (mirror) descent with LSE smoothing.
    Update: a[j] <- a[j] * exp(-eta * grad[j]), then normalize.
    """
    a = normalize_sum(a_init.copy(), n).clip(min=1e-30)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 200000.0)

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            last_improve = it

        grad, _ = lse_grad(a, c, beta)
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break

        # Adaptive eta: smaller as beta increases
        eta = eta_base / (beta * (gmax / S + 1e-10))
        eta = min(eta, 0.5 / np.log(n + 1))

        # Multiplicative update with numerical stability
        log_update = -eta * grad
        log_update -= log_update.max()
        a = a * np.exp(log_update)
        a = normalize_sum(a, n).clip(min=1e-30)

        # Stagnation handling
        stagnant = it - last_improve
        if stagnant > 500:
            noise_scale = 0.005 * np.exp(-stagnant / 10000.0)
            if noise_scale > 1e-8:
                noise = rng.normal(1.0, noise_scale, n)
                noise = np.clip(noise, 0.5, 2.0)
                a = a * noise
                a = normalize_sum(a, n).clip(min=1e-30)
            last_improve = it

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_active_refinement(a_init, n, S, max_time, rng, 
                          thresh_frac=0.9995):
    """
    Active-set gradient descent with backtracking line search.
    Targets the specific lags that are at or near the maximum.
    """
    a = normalize_sum(a_init.copy(), n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Use progressively tighter active set
        frac = min(0.9999, thresh_frac + it * 0.00001)
        grad = active_set_grad(a, c, threshold_frac=frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / (beta_est(a, c) + 1.0)
        base = cur_max
        improved = False
        for _ in range(10):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            test_val = c_test.max()
            if test_val < base * (1.0 - 1e-7):
                a = a_test
                improved = True
                break
            step *= 0.5

        if not improved and step < 1e-12:
            # Try a small fixed step
            a = project_simplex(a - 1e-4 * grad / (gnorm + 1e-10), S)

        # Periodic noise
        if it % 500 == 0:
            noise_scale = 0.0005 * S / np.sqrt(n) * np.exp(-it / 20000.0)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def beta_est(a, c):
    """Estimate Lipschitz constant from current state."""
    n = len(a)
    amax = a.max()
    # Lipschitz constant of max(conv) is bounded by ~2*max(c)
    return 2.0 * c.max() / (amax + 1e-10)


def run_peak_shave(a_init, n, S, max_time, rng, n_peaks=10):
    """
    Directed peak shaving: identify top peaks, compute responsibility,
    and multiplicatively reduce responsible elements.
    """
    a = normalize_sum(a_init.copy(), n).clip(min=1e-30)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Top peaks
        sorted_lags = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[sorted_lags]
        # Weight by how much each peak exceeds the median
        median_c = np.median(c)
        weights = np.maximum(0, peak_vals - median_c)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights /= wsum

        # Compute responsibility
        resp = np.zeros(n)
        for idx, k in enumerate(sorted_lags):
            if weights[idx] < 1e-15:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            resp[js] += weights[idx] * contrib

        # Multiplicative reduction
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        lr = 0.01 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = normalize_sum(a, n).clip(min=1e-30)

    return best_a, best


def upsample(a, new_n):
    """Upsample sequence to new length using linear interpolation."""
    old_n = len(a)
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def downsample(a, new_n):
    """Downsample sequence to new length."""
    old_n = len(a)
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def sigmoid_init(n, steepness=5.0, center=0.5):
    """Sigmoid-shaped initialization (asymmetric)."""
    t = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + np.exp(-steepness * (t - center)))
    return np.maximum(a, 1e-10)


def exp_ramp_init(n, rate=3.0):
    """Exponential ramp (monotonically increasing)."""
    t = np.linspace(0, 1, n)
    a = np.exp(rate * (t - 1.0))
    return np.maximum(a, 1e-10)


def bimodal_init(n, mu1=0.25, mu2=0.75, sigma=0.1):
    """Bimodal: two bumps."""
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - mu1) / sigma) ** 2) + np.exp(-0.5 * ((t - mu2) / sigma) ** 2)
    return np.maximum(a, 1e-10)


def cosine_asym_init(n, center=0.35, width=0.5):
    """Asymmetric raised cosine."""
    t = np.linspace(0, 1, n)
    x = (t - center) / width
    a = np.ones(n)
    mask = np.abs(x) < 1
    a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
    a[~mask] = 0.0
    return np.maximum(a, 1e-10)


def generate_inits(n, rng):
    """Generate diverse asymmetric initializations."""
    inits = []
    inits.append(sigmoid_init(n, 3.0, 0.5))
    inits.append(sigmoid_init(n, 5.0, 0.4))
    inits.append(sigmoid_init(n, 8.0, 0.5))
    inits.append(exp_ramp_init(n, 2.0))
    inits.append(exp_ramp_init(n, 4.0))
    inits.append(exp_ramp_init(n, 6.0))
    inits.append(bimodal_init(n, 0.2, 0.8, 0.15))
    inits.append(bimodal_init(n, 0.3, 0.7, 0.1))
    inits.append(cosine_asym_init(n, 0.3, 0.4))
    inits.append(cosine_asym_init(n, 0.4, 0.5))
    # Power law
    t = np.linspace(0, 1, n)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(np.maximum(t ** p, 1e-10))
        inits.append(np.maximum((1 - t) ** p, 1e-10))
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Multi-stage: multiplicative descent + LSE-FISTA + active set + peak shaving,
    with multi-resolution and diverse asymmetric initializations.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 90.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)
    t_start = time.monotonic()

    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))

    global_best = np.inf
    global_a = None

    def try_seq(a):
        nonlocal global_best, global_a
        n = len(a)
        S = np.sqrt(2.0 * n)
        a = normalize_sum(a, n)
        c = fft_convolve_self(a)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a.copy()
            elapsed = time.monotonic() - t_start
            print("  *** New best: %.8f (n=%d) at t=%ds" % (val, n, int(elapsed)))

    # Load previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception:
        pass

    # === Phase 0: Quick optimization of given ===
    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        t0 = min(45.0, (deadline - time.monotonic()) * 0.08)
        print("=== Phase 0: Quick LSE on given (n=%d, %.0fs) ===" % (n_g, t0))
        a0, v0 = run_lse_fista(given, n_g, S_g, t0, rng,
                                beta_start=50.0, beta_end=5000.0,
                                beta_steps=40, momentum_period=200)
        try_seq(a0)

    # === Phase 1: Multiplicative descent on best ===
    if global_a is not None:
        n_m = len(global_a)
        S_m = np.sqrt(2.0 * n_m)
        remaining = deadline - time.monotonic()
        t1 = remaining * 0.25
        print("=== Phase 1: Multiplicative descent (n=%d, %.0fs) ===" % (n_m, t1))
        a1, v1 = run_multiplicative(global_a, n_m, S_m, t1, rng,
                                     beta_start=20.0, beta_end=30000.0,
                                     beta_steps=100, eta_base=0.5)
        try_seq(a1)

    # === Phase 2: LSE-FISTA on best ===
    if global_a is not None:
        n_f = len(global_a)
        S_f = np.sqrt(2.0 * n_f)
        remaining = deadline - time.monotonic()
        t2 = remaining * 0.25
        print("=== Phase 2: LSE-FISTA (n=%d, %.0fs) ===" % (n_f, t2))
        a2, v2 = run_lse_fista(global_a, n_f, S_f, t2, rng,
                                beta_start=100.0, beta_end=80000.0,
                                beta_steps=150, momentum_period=400)
        try_seq(a2)

    # === Phase 3: Active set refinement ===
    if global_a is not None:
        n_a = len(global_a)
        S_a = np.sqrt(2.0 * n_a)
        remaining = deadline - time.monotonic()
        t3 = remaining * 0.2
        print("=== Phase 3: Active set (n=%d, %.0fs) ===" % (n_a, t3))
        a3, v3 = run_active_refinement(global_a, n_a, S_a, t3, rng,
                                        thresh_frac=0.9999)
        try_seq(a3)

    # === Phase 4: Peak shaving ===
    if global_a is not None:
        n_p = len(global_a)
        S_p = np.sqrt(2.0 * n_p)
        remaining = deadline - time.monotonic()
        t4 = remaining * 0.15
        print("=== Phase 4: Peak shaving (n=%d, %.0fs) ===" % (n_p, t4))
        a4, v4 = run_peak_shave(global_a, n_p, S_p, t4, rng, n_peaks=15)
        try_seq(a4)
        # Quick LSE polish after shaving
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a4b, v4b = run_lse_fista(a4, n_p, S_p, remaining2 * 0.5, rng,
                                      beta_start=5000.0, beta_end=50000.0,
                                      beta_steps=40, momentum_period=200)
            try_seq(a4b)

    # === Phase 5: Multi-resolution exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print("=== Phase 5: Multi-resolution (remaining %.0fs) ===" % remaining)
        for n_try in [4096, 8192]:
            if time.monotonic() > deadline - 60:
                break
            S_try = np.sqrt(2.0 * n_try)
            t5 = min(60.0, remaining * 0.3)
            print("  Trying n=%d (%.0fs)" % (n_try, t5))

            # Use diverse inits
            inits = generate_inits(n_try, rng)
            # Quick score
            scores = []
            for i, init in enumerate(inits):
                init = normalize_sum(init, n_try)
                c = fft_convolve_self(init)
                scores.append((c.max(), i))
            scores.sort()

            per_init = max(3.0, t5 / 5)
            for score, idx in scores[:3]:
                if time.monotonic() > deadline - 30:
                    break
                init = inits[idx]
                ai = run_multiplicative(init, n_try, S_try, per_init, rng,
                                         beta_start=10.0, beta_end=10000.0,
                                         beta_steps=60, eta_base=0.6)
                try_seq(ai[0])

    # === Phase 6: Try larger n (upsample) ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            n_large = len(global_a) * 2
            if n_large <= 32768:
                S_large = np.sqrt(2.0 * n_large)
                init_large = upsample(global_a, n_large)
                init_large = normalize_sum(init_large, n_large)
                t6 = min(90.0, remaining * 0.4)
                print("=== Phase 6: Upsample to n=%d (%.0fs) ===" % (n_large, t6))
                a6, v6 = run_lse_fista(init_large, n_large, S_large, t6, rng,
                                        beta_start=50.0, beta_end=20000.0,
                                        beta_steps=80, momentum_period=300)
                try_seq(a6)

    # === Phase 7: Final fine-tuning ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            n_final = len(global_a)
            S_final = np.sqrt(2.0 * n_final)
            print("=== Phase 7: Final polish (%.0fs) ===" % remaining)

            # Very high beta LSE
            t7a = remaining * 0.5
            a7, v7 = run_lse_fista(global_a, n_final, S_final, t7a, rng,
                                    beta_start=20000.0, beta_end=200000.0,
                                    beta_steps=30, momentum_period=100)
            try_seq(a7)

            # Active set on the result
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a8, v8 = run_active_refinement(a7, n_final, S_final,
                                               remaining2 * 0.6, rng,
                                               thresh_frac=0.99999)
                try_seq(a8)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.8f, internal best = %.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  Invalid! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No result, using uniform fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
