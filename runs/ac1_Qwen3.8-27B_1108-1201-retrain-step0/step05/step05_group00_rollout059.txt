import time
from scipy.fft import fft, ifft
import scipy.sparse as sp
import cvxpy as cp


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    Nf = 1 << max(1, (2 * n - 1).bit_length())
    if Nf < N:
        Nf = N
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    # Clip tiny negative values from FFT roundoff
    c = np.maximum(c, 0.0)
    return c


def subgrad_peak_vec(a, k_star):
    """Vectorized subgradient of (a*a)(k_star) w.r.t. a."""
    n = len(a)
    j = np.arange(n)
    idx1 = k_star - j
    sg = np.zeros(n)
    m1 = (idx1 >= 0) & (idx1 < n)
    sg[m1] += a[idx1[m1]]
    idx2 = j - k_star
    m2 = (idx2 >= 0) & (idx2 < n)
    sg[m2] += a[idx2[m2]]
    return sg


def spectral_partial_restart(a, N, rng, frac=0.5):
    """Randomize a fraction of FFT phases, preserve magnitudes."""
    n = len(a)
    S = a.sum()
    Nf = max(N, 1 << max(1, n.bit_length()))
    A = fft(a, Nf)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(Nf * frac))
    rand_idx = rng.choice(Nf, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def spectral_full_restart(a, N, rng):
    """Full spectral phase randomization."""
    n = len(a)
    S = a.sum()
    Nf = max(N, 1 << max(1, n.bit_length()))
    A = fft(a, Nf)
    mag = np.abs(A)
    phase = rng.uniform(0, 2 * np.pi, Nf)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    a = a[:n] if len(a) > n else np.pad(a, (0, n - len(a)))
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    a = safe_normalize(a_init, n, S)
    x_prev = a.copy()
    t_val = 1.0
    best_a = a.copy()
    best_peak = np.inf
    start = time.monotonic()
    stall = 0
    it = 0

    c = compute_conv(a, N)
    cur_peak = c.max()
    best_peak = cur_peak

    step_rate = 2.0 * S / np.sqrt(n)

    while time.monotonic() - start < time_budget:
        it += 1
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        momentum = (t_val - 1.0) / t_new

        # Nesterov lookahead
        y = a + momentum * (a - x_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y * (S / sy)

        # Compute top-k peak contributions
        c_y = compute_conv(y, N)
        cur_peak_y = c_y.max()

        if top_k == 1:
            k_star = int(np.argmax(c_y))
            sg = subgrad_peak_vec(y, k_star)
        else:
            top_idx = np.argsort(c_y)[::-1][:top_k]
            top_vals = c_y[top_idx]
            vals_max = top_vals.max()
            weights = np.exp(20.0 * (top_vals - vals_max))
            weights /= weights.sum()
            sg = np.zeros(n)
            for kk, ww in zip(top_idx, weights):
                sg += ww * subgrad_peak_vec(y, int(kk))

        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-18:
            break

        # Adaptive step size
        step_t = step_rate / np.sqrt(it + 1)
        # Backtracking: ensure we don't overshoot
        x_new = project_simplex(y - step_t * sg, S)

        c_new = compute_conv(x_new, N)
        c_new_max = c_new.max()

        if c_new_max < best_peak:
            best_peak = c_new_max
            best_a = x_new.copy()
            stall = 0
        else:
            stall += 1

        x_prev = a
        a = x_new
        t_val = t_new

        # Spectral restart when stuck
        if stall > 300 and (stall % 200 == 0) and rng.random() < 0.3:
            frac = 0.2 + 0.3 * min(stall / 2000.0, 1.0)
            a = spectral_partial_restart(a, N, rng, frac)
            x_prev = a.copy()
            t_val = 1.0
            stall = 0

        if it % 5000 == 0:
            elapsed = time.monotonic() - start
            print("    NV it=%d peak=%.8f stall=%d time=%.0fs" % (it, best_peak, stall, elapsed))

    return best_a, best_peak


def build_conv_matrix(n):
    """Build sparse matrix A where A @ g = f * g (full convolution)."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
    return rows, cols


def solve_lp_direction(f, M, n, rows_pre, cols_pre):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    Returns g (the LP solution) or None if infeasible/failed."""
    nnz = len(rows_pre)
    # Fill in values: A[k, j] = f[k-j]
    vals = np.zeros(nnz, dtype=np.float64)
    rows_arr = np.asarray(rows_pre, dtype=np.int32)
    cols_arr = np.asarray(cols_pre, dtype=np.int32)
    k_vals = rows_arr
    j_vals = cols_arr
    idx = k_vals - j_vals
    valid = (idx >= 0) & (idx < n)
    vals[valid] = f[idx[valid]]

    A = sp.csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n))

    g = cp.Variable(n)
    constraints = [A @ g <= M, g >= 0]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CBC, threads=1, time_limit=120)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return None

    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)


def lp_frank_wolfe(a_init, n, S, time_budget, rng, rows_pre, cols_pre):
    """LP-based Frank-Wolfe iterations."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_peak = np.inf
    start = time.monotonic()
    it = 0
    stall = 0

    c = compute_conv(a, 1)
    best_peak = c.max()

    while time.monotonic() - start < time_budget:
        it += 1
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed

        if remaining < 15:
            break

        c = compute_conv(a, 1)
        M = c.max()
        if M < best_peak:
            best_peak = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Solve LP
        g = solve_lp_direction(a, M, n, rows_pre, cols_pre)
        if g is None:
            # Fallback: subgradient step
            k_star = int(np.argmax(c))
            sg = subgrad_peak_vec(a, k_star)
            sg_norm = np.linalg.norm(sg)
            if sg_norm < 1e-15:
                break
            step = S / (np.sqrt(n) * np.sqrt(it + 1))
            a = project_simplex(a - step * sg, S)
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break

        # Rescale g to sum S
        g = g * (S / g_sum)

        # Backtracking step size
        t = 1.0
        for _ in range(10):
            a_cand = (1 - t) * a + t * g
            a_cand = np.maximum(a_cand, 0)
            a_cand = a_cand * (S / max(a_cand.sum(), 1e-15))
            c_cand = compute_conv(a_cand, 1)
            if c_cand.max() <= M + 1e-10:
                a = a_cand
                break
            t *= 0.5
        else:
            a = a_cand

        if it % 5 == 0:
            c_check = compute_conv(a, 1)
            print("    LP it=%d peak=%.8f stall=%d time=%.0fs" % (it, c_check.max(), stall, time.monotonic() - start))

        # Spectral restart
        if stall > 10 and rng.random() < 0.2:
            a = spectral_partial_restart(a, 1, rng, 0.4)
            stall = 0

    return best_a, best_peak


def deflation_init(n, S, N, n_iters=3000, rng=None):
    """Constructive init: iteratively reduce peak from all-ones."""
    if rng is None:
        rng = np.random.default_rng(0)
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c = compute_conv(a, N)
        k_star = int(np.argmax(c))
        # Compute contribution of each a[j] to c[k_star]
        j = np.arange(n)
        idx1 = k_star - j
        contrib = np.zeros(n)
        m1 = (idx1 >= 0) & (idx1 < n)
        contrib[m1] += a[idx1[m1]]
        idx2 = j - k_star
        m2 = (idx2 >= 0) & (idx2 < n)
        contrib[m2] += a[idx2[m2]]

        i_max = int(np.argmax(contrib))
        # Find a low-contribution, low-value index to add to
        low_vals = np.argsort(a)[:max(1, n // 10)]
        i_min = int(low_vals[np.argmin(contrib[low_vals])])
        if i_max == i_min:
            break
        transfer = min(0.05 * a[i_max], 0.5 * (a[i_max] - a[i_min] + S / n))
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_smooth(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt + rng.random() * 6)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff_hybrid(n, rng):
    """Hybrid: use cos of quadratic phase (real part of Zadoff-Chu) + smoothing."""
    i = np.arange(n)
    c = rng.integers(1, n)
    phase = np.pi * i * (2 * i + 1 - c) / n
    a = 1.0 + 0.6 * np.cos(phase)
    # Add smoothing
    kernel = np.ones(5) / 5
    a = np.convolve(a, kernel, mode='same')
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def lse_refine(a_init, n, S, N, time_budget, rng, beta_init=10.0, beta_max=500.0):
    """L-BFGS on LSE-smoothed peak for polishing."""
    a = safe_normalize(a_init, n, S)
    Nf = 1 << max(1, (2 * n - 1).bit_length())
    best_a = a.copy()
    best_peak = np.inf
    beta = beta_init
    s_hist = []
    y_hist = []
    m_lbfgs = 10
    start = time.monotonic()
    it = 0

    def lse_grad_fn(beta_val):
        c = compute_conv(a, Nf)
        c_max = c.max()
        if beta_val < 0.01:
            weights = np.ones(len(c)) / len(c)
            lse_val = c_max
        else:
            exp_bc = np.exp(beta_val * (c - c_max))
            total = exp_bc.sum()
            if total < 1e-300:
                weights = np.zeros(len(c))
                weights[int(np.argmax(c))] = 1.0
            else:
                weights = exp_bc / total
            lse_val = c_max + np.log(max(total, 1e-300)) / beta_val
        fsize = 1 << max(1, (3 * n).bit_length())
        W = fft(weights, fsize)
        Af = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        return grad, lse_val, c, c_max

    def lbfgs_direction(g):
        m = len(s_hist)
        gn = np.linalg.norm(g)
        if m == 0 or gn < 1e-18:
            return -g / gn if gn > 1e-18 else np.zeros(n)
        q = g.copy()
        alphas = np.zeros(m)
        for i in range(m - 1, -1, -1):
            s_i = s_hist[i]
            y_i = y_hist[i]
            yy = np.dot(y_i, y_i)
            if yy < 1e-30:
                alphas[i] = 0.0
            else:
                alphas[i] = np.dot(s_i, q) / yy
                q = q - alphas[i] * y_i
        s_last = s_hist[-1]
        y_last = y_hist[-1]
        yy_last = np.dot(y_last, y_last)
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last) if yy_last > 1e-30 else 1.0
        r = gamma_use * q
        for i in range(m):
            s_i = s_hist[i]
            y_i = y_hist[i]
            yy = np.dot(y_i, y_i)
            if yy < 1e-30:
                continue
            beta_i = np.dot(s_i, y_i) / yy
            r = r + (alphas[i] - beta_i) * y_i
        direction = -r
        if np.dot(g, direction) > 0:
            direction = -g
        return direction

    g, f_val, c, c_max = lse_grad_fn(beta)
    if c_max < best_peak:
        best_peak = c_max
        best_a = a.copy()

    step_scale = 2.0 * S / (n * max(beta, 1.0))

    while time.monotonic() - start < time_budget:
        it += 1

        if it % 100 == 0:
            beta = min(beta_max, beta * 1.15)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            g, f_val, c, c_max = lse_grad_fn(beta)
            if c_max < best_peak:
                best_peak = c_max
                best_a = a.copy()

        if it % 30 == 0 and len(s_hist) >= m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_direction(g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        alpha = step_scale
        a_old = a
        c_old_max = c_max
        a_new = a
        for ls in range(10):
            cand = project_simplex(a + alpha * direction, S)
            c_ls = compute_conv(cand, Nf)
            cm_ls = c_ls.max()
            if cm_ls <= c_old_max * 1.001 or ls == 9:
                a_new = cand
                break
            alpha *= 0.5

        c_new = compute_conv(a_new, Nf)
        c_new_max = c_new.max()
        if c_new_max < best_peak:
            best_peak = c_new_max
            best_a = a_new.copy()

        g_new, _, _, _ = lse_grad_fn(beta)
        s_k = a_new - a_old
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c_max = c_new_max

        if it % 500 == 0 and rng.random() < 0.2:
            a = spectral_partial_restart(a, Nf, rng, 0.3)
            s_hist = []
            y_hist = []
            g, f_val, c, c_max = lse_grad_fn(beta)

    return best_a, best_peak


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for best non-negative sequence minimizing evaluate_sequence.
    Uses Nesterov subgradient + LP Frank-Wolfe + LSE polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None
    global_n = 0

    def update_best(a_cand, n_cand):
        nonlocal global_best, global_a, global_n
        a_c = safe_normalize(a_cand, n_cand, np.sqrt(2.0 * n_cand))
        Nf = 1 << max(1, (2 * n_cand - 1).bit_length())
        c = compute_conv(a_c, Nf)
        peak = c.max()
        S = np.sqrt(2.0 * n_cand)
        score = 2.0 * n_cand * peak / (S * S)
        if score < global_best:
            global_best = score
            global_a = a_c.copy()
            global_n = n_cand
            print("  [NEW BEST] n=%d score=%.8f peak=%.8f" % (n_cand, score, peak))
        return score

    # Load initial sequence if available
    init_seq = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and given.size <= 32768:
            init_seq = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Budget: %.0fs" % budget_s)

    # === Phase 1: Nesterov subgradient on n=8192 (or given n) ===
    print("=== Phase 1: Nesterov subgradient ===")
    n1 = 8192
    if init_seq is not None and 1024 <= len(init_seq) <= 16384:
        n1 = len(init_seq)
    S1 = np.sqrt(2.0 * n1)
    N1 = 1 << max(1, (2 * n1 - 1).bit_length())

    if init_seq is not None and len(init_seq) == n1:
        a_start = safe_normalize(init_seq, n1, S1)
    else:
        a_start = deflation_init(n1, S1, N1, 2000, rng)

    t1 = time.monotonic()
    p1_budget = min(500.0, (deadline - t1) * 0.22)
    a1, peak1 = nesterov_subgrad(a_start, n1, S1, N1, p1_budget, rng, top_k=1)
    update_best(a1, n1)

    # === Phase 2: LSE polishing on best so far ===
    print("=== Phase 2: LSE polishing ===")
    t2 = time.monotonic()
    p2_budget = min(400.0, (deadline - t2) * 0.18)
    if p2_budget > 30:
        a2, peak2 = lse_refine(a1, n1, S1, N1, p2_budget, rng, beta_init=15.0, beta_max=400.0)
        update_best(a2, n1)

    # === Phase 3: LP Frank-Wolfe for n=2048 ===
    print("=== Phase 3: LP Frank-Wolfe n=2048 ===")
    t3 = time.monotonic()
    p3_budget = min(600.0, (deadline - t3) * 0.25)
    if p3_budget > 60:
        n3 = 2048
        S3 = np.sqrt(2.0 * n3)
        N3 = 1 << max(1, (2 * n3 - 1).bit_length())
        rows_pre, cols_pre = build_conv_matrix(n3)
        print("  LP matrix: %d nonzeros" % len(rows_pre))

            ("defl", deflation_init(n3, S3, N3, 1000, rng)),
            ("rand", make_init_rand(n3, rng)),
            ("smooth", make_init_smooth(n3, rng)),
            ("zadoff", make_init_zadoff_hybrid(n3, rng)),

        if global_a is not None and global_n == n3:
            inits_3.insert(0, ("prev", global_a.copy()))

            if time.monotonic() > deadline - 100:
                break
            a3, peak3 = lp_frank_wolfe(arr3, n3, S3, min(per_init3, p3_budget), rng, rows_pre, cols_pre)
