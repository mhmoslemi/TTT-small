import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy import sparse
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def build_conv_matrix(f, n):
    """Build the sparse convolution matrix."""
    # For efficiency, build in COO format
    row_idx = []
    col_idx = []
    data = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            row_idx.extend([k] * len(js))
            col_idx.extend(js.tolist())
            data.extend((f[k - js]).tolist())
    return sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))

def project_simplex(v, S):
    # Project onto {x >= 0, sum(x) = S}
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def obj_value(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def lp_optimize_step(f, n, S, M, timeout_s=30):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0
    Returns g normalized to sum S, or None if no improvement.
    """
    A = build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, 
                  method='highs', options={'time_limit': timeout_s})
    
    if res.status != 0 or res.x is None:
        return None
    
    g = res.x
    S_g = g.sum()
    if S_g <= S * (1 + 1e-10):
        return None  # No improvement
    
    g_norm = g * (S / S_g)
    return g_norm

    f = f0.copy()
    f = project_simplex(f, S)
    
    c = fft_convolve_self(f)
    best = c.max()
    best_f = f.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        remaining = max_time - elapsed
        if remaining < 5:
            break
        
        c = fft_convolve_self(f)
        cur_M = c.max()
        
        if cur_M < best - 1e-12:
            best = cur_M
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
        
        if verbose and it % 10 == 0:
            obj = n * best / (S * S)  # Since S = sqrt(2n), obj = 2n*best/(2n) = best
            # Actually obj = 2*n*best/(S^2) = 2*n*best/(2*n) = best when S=sqrt(2n)
            print(f"  LP it={it}, t={elapsed:.0f}s, peak={best:.6f}, obj={2*n*best/(S*S):.6f}")
        
        # Solve LP
        g = lp_optimize_step(f, n, S, cur_M, timeout_s=min(20, remaining/3))
        
        if g is None:

                # Add noise to escape
                noise = 0.01 * S / np.sqrt(n)
                f = project_simplex(f + rng.normal(0, noise, n), S)
                stuck = 0
            continue
        
        # Line search on step size
        best_t = None
        for t in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            f_try = (1 - t) * f + t * g
            f_try = np.maximum(f_try, 0)
            # Normalize
            if f_try.sum() < 1e-10:

Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement.
The LP at each step finds the direction that maximizes the sum of the new

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def build_conv_matrix(f, n):
    """Build sparse matrix A where (A@g)(k) = (f*g)(k)."""
    row_idx = np.zeros(n * n, dtype=np.int32)
    col_idx = np.zeros(n * n, dtype=np.int32)
    data = np.zeros(n * n, dtype=np.float64)
    count = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        length = j_max - j_min + 1
        row_idx[count:count+length] = k
        col_idx[count:count+length] = np.arange(j_min, j_max + 1)
        data[count:count+length] = f[k - np.arange(j_min, j_max + 1)]
        count += length
    return sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))

def project_simplex(v, S):
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    A = build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': timeout_s,
                      'presolve': True})
    except Exception:
        return None
    
    if res.status != 0 or res.x is None:
        return None
    
    g = res.x
    if g.sum() <= 0:
        return None
    return g

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid LSE-smoothed gradient
descent with periodic LP-based resets. The LSE approach uses FFT for fast
gradient computation, while the LP provides principled direction changes
to escape local minima. Multi-length exploration is included in later phases.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy import sparse
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def lse_gradient(f, c, beta):
    """Compute gradient of LSE-smoothed max of self-convolution."""
    n = len(f)
    cm = c.max()
    scaled = beta * (c - cm)
    max_scaled = scaled.max()
    if max_scaled > 400:
        scale_factor = 400.0 / max_scaled
        scaled = scaled * scale_factor
        beta_eff = beta * scale_factor
    else:
        beta_eff = beta
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Gradient: grad_j = 2 * sum_k w_k * f(k-j)
    # This is 2 * cross-correlation of w and f
    fsize = 1 << (2 * n - 1).bit_length()
    F = fft(f, fsize)
    W = fft(w, fsize)
    cc = np.real(ifft(W * np.conj(F)))[:n]
    return 2.0 * cc


def build_conv_matrix(f, n):
    """Build sparse matrix A where (A@g)(k) = (f*g)(k) for full convolution."""
    total_nnz = n * n
    row_idx = np.zeros(total_nnz, dtype=np.int64)
    col_idx = np.zeros(total_nnz, dtype=np.int64)
    data = np.zeros(total_nnz, dtype=np.float64)
    count = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        length = j_max - j_min + 1
        row_idx[count:count + length] = k
        col_idx[count:count + length] = np.arange(j_min, j_max + 1)
        data[count:count + length] = f[k - np.arange(j_min, j_max + 1)]
        count += length
    return sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))


def lp_solve(f, n, M, timeout_s=20):
    """Solve LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0."""
    A = build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': timeout_s, 'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None:
        return None
    g = res.x
    if g.sum() <= 0:
        return None
    return g


def optimize_lse(f0, S, max_time, rng, use_lp=True, lp_interval=80, verbose=True):
    """
    LSE-smoothed gradient descent with periodic LP resets.
    Minimizes max(f*f) subject to sum(f)=S, f>=0.
    """
    n = len(f0)
    f = project_simplex(f0.copy(), S)
    c = fft_convolve_self(f)
    best = c.max()
    best_f = f.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    beta = 10.0
    lp_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        remaining = max_time - (time.monotonic() - start)
        if remaining < 3:
            break

        c = fft_convolve_self(f)
        cur_M = c.max()

        if cur_M < best - 1e-14:
            best = cur_M
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if verbose and it % 200 == 0:
            obj = 2.0 * n * best / (S * S)
            print(f"    LSE it={it}, t={it_start:.0f}s, obj={obj:.6f}, beta={beta:.1f}, lp_calls={lp_count}")

        # Increase beta over time
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = 10.0 * np.exp(3.0 * progress)
        beta = min(beta, 2000.0)

        # Compute gradient
        grad = lse_gradient(f, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size with line search
        step = 1.0 / (1.5 * beta * gnorm + 1e-30)
        step = min(step, 5.0 / (gnorm + 1e-30))

        accepted = False
        for _ in range(12):
            f_try = project_simplex(f - step * grad, S)
            c_try = fft_convolve_self(f_try)
            if c_try.max() <= cur_M * (1.0 + 1e-9):
                f = f_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            f = project_simplex(f - 0.01 * step * grad, S)

        # Periodic LP reset
        if use_lp and stuck > lp_interval and remaining > 30 and n <= 16384:
            g = lp_solve(f, n, cur_M, timeout_s=min(15, remaining / 5))
            lp_count += 1
            if g is not None and g.sum() > S * (1 + 1e-6):
                g_norm = g * (S / g.sum())
                improved = False
                for t in [0.5, 0.3, 0.2, 0.1]:
                    f_try = (1 - t) * f + t * g_norm
                    f_try = np.maximum(f_try, 0)
                    f_s = f_try.sum()
                    if f_s < 1e-10:
                        continue
                    f_try = f_try * (S / f_s)
                    c_try = fft_convolve_self(f_try)
                    if c_try.max() < cur_M * (1 - 1e-10):
                        f = f_try
                        improved = True
                        break
                if improved:
                    stuck = 0
                    lp_interval = int(lp_interval * 1.3)

        # Noise injection when very stuck
        if stuck > 500:
            noise_scale = 0.005 * S / np.sqrt(n) * (1.0 - 0.5 * progress)
            if noise_scale > 1e-12:
                f = project_simplex(f + rng.normal(0, noise_scale, n), S)
            stuck = 0
            lp_interval = max(20, int(lp_interval * 0.8))

    return best_f, best

def optimize_subgradient(f0, S, max_time, rng, verbose=True):
    """Top-k weighted subgradient descent for polishing."""
    n = len(f0)
    f = project_simplex(f0.copy(), S)
    c = fft_convolve_self(f)
    best = c.max()
    best_f = f.copy()

    start = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        remaining = max_time - (time.monotonic() - start)
        if remaining < 3:
            break

        c = fft_convolve_self(f)
        cur_M = c.max()

        if cur_M < best - 1e-14:
            best = cur_M
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if verbose and it % 500 == 0:
            obj = 2.0 * n * best / (S * S)
            print(f"    Subgrad it={it}, t={time.monotonic()-start:.0f}s, obj={obj:.6f}")

        # Top-k subgradient
        n_k = min(60, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:n_k]
        ref = np.median(c)
        weights = np.maximum(c[top_idx] - ref, 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(n_k) / n_k
        else:
            weights = weights / wsum

        grad = np.zeros(n)
        for ii in range(n_k):
            ki = int(top_idx[ii])
            wi = weights[ii]
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start <= j_end:
                indices = ki - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * wi * f[indices]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        gap = cur_M - c.mean()
        if gap < 1e-13:
            break

        step = 2.0 * gap / (gnorm * gnorm + 1e-30)
        step = min(step, 4.0 / (gnorm + 1e-30))

        accepted = False
        for _ in range(10):
            f_try = project_simplex(f - step * grad, S)
            c_try = fft_convolve_self(f_try)
            if c_try.max() <= cur_M * (1.0 + 1e-9):
                f = f_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            f = project_simplex(f - 0.005 * step * grad, S)

        if stuck > 300:
            noise_scale = 0.003 * S / np.sqrt(n)
            f = project_simplex(f + rng.normal(0, noise_scale, n), S)
            stuck = 0

    return best_f, best
