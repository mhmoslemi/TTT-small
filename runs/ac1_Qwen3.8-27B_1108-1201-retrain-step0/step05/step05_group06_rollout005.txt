"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: LSE-FISTA with adaptive beta -> Active-set water-filling -> Multi-scale.
Focus on precise optimization with backtracking and diverse restarts.

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_smoothing(c, beta):
    """Computes LSE of c and its gradient w.r.t. c."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        weights = np.zeros_like(c)
        weights[np.argmax(c)] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        weights = e / tot
    return lse_val, weights

def lse_subgradient(a, c, beta):
    """Subgradient of LSE-smoothed max of convolve(a,a) w.r.t. a."""
    n = len(a)
    _, w = lse_smoothing(c, beta)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        obj = 2.0 * n * cur_max / (S * S)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive: start with top-1, expand as we get stuck
        if stuck < 10:
            tol = 0.0
            k = 1
        elif stuck < 100:
            tol = 0.001
            k = 3
        elif stuck < 500:
            tol = 0.005
            k = 10
        elif stuck < 2000:
            tol = 0.01
            k = 50
        else:
            tol = 0.02
            k = min(200, stuck // 10)

        # Find active lags
        threshold = cur_max * (1.0 - tol) - tol * cur_max
        active = np.where(c >= cur_max - max(tol * cur_max, 1e-10))[0]
        if len(active) > k:
            # Keep the top k
            top_k = np.argsort(c[active])[::-1][:k]
            active = active[top_k]

        # Compute subgradient
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size with backtracking
        step = min(S / gnorm, 1.0 / (gnorm + 1e-10))
        for _ in range(15):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-12:
                break
            step *= 0.5

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 2000 == 0:
            t_mom = 1.0
        if stuck > 3000 and it % 100 == 0:
            noise_scale = max(1e-8, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 5000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    return best_a, best_obj

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization combining LSE-FISTA with adaptive beta,
active-set subgradient polishing with backtracking, and multi-scale exploration.
Key improvements: tighter Lipschitz bounds, smarter active-set selection,
and dedicated final polishing with very high smoothing parameter.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value_and_weights(c, beta):
    """Compute LSE value and softmax weights."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        idx = np.argmax(c)
        w = np.zeros_like(c)
        w[idx] = 1.0
        return cm, w
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    return lse_val, w


def conv_subgradient(a, w):
    """Convolve weight vector w with a (truncated to len(a))."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_lse_fista(a_init, max_time, rng, beta_start=10.0, beta_max=50000.0):
    """FISTA with LSE smoothing, adaptive beta, and backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_start

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        obj = 2.0 * n * cur_max / (S * S)
        if obj < best_obj - 1e-11:
            best_obj = obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta: grow when stuck
        if stuck < 100:
            beta = beta_start * (1.0 + 0.005 * it)
        elif stuck < 1000:
            beta = min(beta_max, beta * 1.005)
        elif stuck < 5000:
            beta = min(beta_max, beta * 1.02)
        else:
            beta = min(beta_max, beta * 1.05)
        beta = min(beta, beta_max)

        # Get LSE gradient
        _, w = lse_value_and_weights(c, beta)
        grad = conv_subgradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: use tighter bound
        # L <= 2 * beta * max(w) * ||K||^2, ||K|| <= 2*max(c) roughly
        # Practical: L ~ 4 * beta * c.max() / n * n = 4 * beta * c.max()
        # But this is too conservative; use 2 * beta * c.max()
        L = 2.5 * beta * c.max()
        step = 0.8 / max(L, 1e-10)
        step = max(step, 1e-14)

        # Backtracking (limited, only every 10th iteration to save time)
        if it % 10 == 0:
            lse_old, _ = lse_value_and_weights(c, beta)
            for _ in range(8):
                x_try = project_simplex(y - step * grad, S)
                c_try = fft_convolve_self(x_try)
                lse_try, _ = lse_value_and_weights(c_try, beta)
                # Descent condition for smooth function
                if lse_try <= lse_old + 1e-10:
                    break
                step *= 0.5
            x_new = x_try
        else:
            x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic restart
        if it % 3000 == 0:
            t_mom = 1.0

        # Noise escape when very stuck
        if stuck > 4000 and it % 200 == 0:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                last_improve = it

    return best_a, best_obj


def run_active_subgrad(a_init, max_time, rng):
    """Active-set subgradient method with FISTA momentum and backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step_base = S / (10.0 * np.sqrt(n))

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        obj = 2.0 * n * cur_max / (S * S)
        if obj < best_obj - 1e-11:
            best_obj = obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive active set: start tight, widen when stuck
        if stuck < 50:
            tol_frac = 0.0
            max_k = 1
        elif stuck < 200:
            tol_frac = 0.001
            max_k = 5
        elif stuck < 1000:
            tol_frac = 0.005
            max_k = 20
        elif stuck < 5000:
            tol_frac = 0.01
            max_k = 50
        else:
            tol_frac = 0.02
            max_k = min(200, 50 + stuck // 50)

        # Find active lags
        threshold = cur_max * (1.0 - tol_frac)
        active_indices = np.where(c >= threshold)[0]
        if len(active_indices) > max_k:
            # Keep top max_k by value
            order = np.argsort(c[active_indices])[::-1][:max_k]
            active_indices = active_indices[order]
        if len(active_indices) == 0:
            active_indices = np.array([int(np.argmax(c))])

        # Compute subgradient from active set
        w = np.zeros(len(c))
        w[active_indices] = 1.0 / len(active_indices)
        grad = conv_subgradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size with polynomial decay + backtracking
        step = step_base / (1.0 + 0.01 * it)
        step = min(step, S / gnorm * 0.5)

        # Backtracking (every iteration for precision)
        improved = False
        for _ in range(12):
            x_try = project_simplex(y - step * grad, S)
            c_try = fft_convolve_self(x_try)
            if c_try.max() <= cur_max + 1e-13:
                improved = True
                x_new = x_try
                break
            step *= 0.5
        if not improved:
            x_new = project_simplex(y - step * grad, S)

        # Momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 2000 == 0:
            t_mom = 1.0

        # Noise escape
        if stuck > 5000 and it % 500 == 0:
            noise_scale = max(1e-10, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 20000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it

    return best_a, best_obj


def run_precise_polish(a_init, max_time, rng):
    """
    Very precise polishing: alternating between LSE (high beta) and exact subgradient,
    with aggressive backtracking and hysteresis in active set selection.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    active_set = None

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        obj = 2.0 * n * cur_max / (S * S)
        if obj < best_obj - 1e-11:
            best_obj = obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Alternate between LSE and exact subgradient
        use_lse = (it % 200 < 100)  # First 100 iters of each 200: LSE

        if use_lse:
            beta = min(50000.0, 100.0 * (1.0 + 0.01 * it))
            _, w = lse_value_and_weights(c, beta)
            grad = conv_subgradient(y, w)
            L = 2.5 * beta * c.max()
            step = 0.8 / max(L, 1e-10)
        else:
            # Exact subgradient with hysteresis
            threshold = cur_max * 0.995
            candidate_active = np.where(c >= threshold)[0]
            if len(candidate_active) > 100:
                order = np.argsort(c[candidate_active])[::-1][:100]
                candidate_active = candidate_active[order]
            if len(candidate_active) == 0:
                candidate_active = np.array([int(np.argmax(c))])

            # Hysteresis: keep old active set if it's still valid
            if active_set is not None:
                old_c = c[active_set]
                if np.all(old_c >= threshold * 0.95):
                    w = np.zeros(len(c))
                    w[active_set] = 1.0 / len(active_set)
                    grad = conv_subgradient(y, w)
                else:
                    active_set = candidate_active
                    w = np.zeros(len(c))
                    w[active_set] = 1.0 / len(active_set)
                    grad = conv_subgradient(y, w)
            else:
                active_set = candidate_active
                w = np.zeros(len(c))
                w[active_set] = 1.0 / len(active_set)
                grad = conv_subgradient(y, w)

            gnorm = np.linalg.norm(grad)
            step = S / (gnorm * (1.0 + 0.005 * it) + 1e-10)
            step = min(step, S / gnorm * 0.3)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking
        x_new = y.copy()
        for _ in range(15):
            x_try = project_simplex(y - step * grad, S)
            c_try = fft_convolve_self(x_try)
            if use_lse:
                lse_old, _ = lse_value_and_weights(c, 100.0 * (1.0 + 0.01 * it))
                lse_new, _ = lse_value_and_weights(c_try, 100.0 * (1.0 + 0.01 * it))
                if lse_new <= lse_old + 1e-10:
                    x_new = x_try
                    break
            else:
                if c_try.max() <= cur_max + 1e-12:
                    x_new = x_try
                    break
            step *= 0.4

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 1000 == 0:
            t_mom = 1.0

        if stuck > 3000 and it % 300 == 0:
            noise_scale = max(1e-11, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 15000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            active_set = None
            last_improve = it

    return best_a, best_obj


def upsample_linear(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def upsample_equispaced(a):
    """Upsample by inserting the mean of neighbors."""
    n = len(a)
    n_new = 2 * n
    result = np.zeros(n_new)
    result[0::2] = a
    result[1::2] = 0.5 * (a + np.roll(a, -1))
    # Fix wrap-around
    result[1] = 0.5 * (a[0] + a[1])
    result[n_new - 1] = 0.5 * (a[-1] + a[0])
    return result


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.54 + 0.46 * np.cos(2 * np.pi * tt)


def make_multi_sin_init(n, rng):
    """Multi-frequency sinusoidal initialization."""
    tt = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(3):
        freq = rng.integers(1, n // 4)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.3
        a += amp * np.cos(2 * np.pi * freq * tt / n + phase)
    return np.maximum(a, 0.01)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization: LSE-FISTA -> Active-set subgradient ->
    Precise polish -> Multi-scale upsample -> Final refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
    else:
        n0 = 8192
        a_start = make_cosine_init(n0)

    print("Starting n = %d" % n0)
    S = np.sqrt(2.0 * n0)
    a0 = normalize_seq(a_start, S)

    obj0 = compute_obj(a0)
    print("Initial objective: %.8f" % obj0)
    global_best_obj = obj0
    global_best_a = a0.copy()

    # Also try a multi-sin init and pick the better one
    a_ms = normalize_seq(make_multi_sin_init(n0, rng), S)
    obj_ms = compute_obj(a_ms)
    print("Multi-sin init: %.8f" % obj_ms)
    if obj_ms < global_best_obj:
        global_best_obj = obj_ms
        global_best_a = a_ms.copy()
        a0 = a_ms

    # === Phase 1: LSE-FISTA (warmup + main) ===
    remaining = deadline - time.monotonic()
    p1 = remaining * 0.30
    if p1 > 10:
        print("\n=== Phase 1: LSE-FISTA (budget %.0fs) ===" % p1)
        t1 = time.monotonic()
        a1, _ = run_lse_fista(a0, p1, rng, beta_start=5.0, beta_max=20000.0)
        obj1 = compute_obj(a1)
        print("  Phase 1: obj=%.8f (t=%.1fs)" % (obj1, time.monotonic() - t1))
        if obj1 < global_best_obj:
            global_best_obj = obj1
            global_best_a = a1.copy()

    # === Phase 2: Active-set subgradient ===
    remaining = deadline - time.monotonic()
    p2 = remaining * 0.25
    if p2 > 10:
        print("\n=== Phase 2: Active-set subgradient (budget %.0fs) ===" % p2)
        t2 = time.monotonic()
        a2, _ = run_active_subgrad(global_best_a, p2, rng)
        obj2 = compute_obj(a2)
        print("  Phase 2: obj=%.8f (t=%.1fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: Precise polish (alternating LSE + subgradient) ===
    remaining = deadline - time.monotonic()
    p3 = remaining * 0.20
    if p3 > 10:
        print("\n=== Phase 3: Precise polish (budget %.0fs) ===" % p3)
        t3 = time.monotonic()
        a3, _ = run_precise_polish(global_best_a, p3, rng)
        obj3 = compute_obj(a3)
        print("  Phase 3: obj=%.8f (t=%.1fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Multi-scale (upsample to 2n) ===
    remaining = deadline - time.monotonic()
    if remaining > 200 and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        p4 = remaining * 0.55
        print("\n=== Phase 4: Upsample to n=%d (budget %.0fs) ===" % (n2, p4))
        t4 = time.monotonic()

        # Try both upsample methods
        a_up1 = upsample_linear(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up1 = normalize_seq(a_up1, S2)
        obj_up1 = compute_obj(a_up1)

        a_up2 = upsample_equispaced(global_best_a)
        a_up2 = normalize_seq(a_up2, S2)
        obj_up2 = compute_obj(a_up2)

        print("  Upsample linear: %.8f, equispaced: %.8f" % (obj_up1, obj_up2))
        a_up = a_up1 if obj_up1 < obj_up2 else a_up2

        # Optimize the upsampled sequence
        rem4 = p4 * 0.5
        a4, _ = run_lse_fista(a_up, rem4, rng, beta_start=5.0, beta_max=10000.0)
        obj4 = compute_obj(a4)
        print("  Phase 4a (LSE): obj=%.8f (t=%.1fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

        rem4b = p4 - (time.monotonic() - t4)
        if rem4b > 10:
            a4b, _ = run_active_subgrad(global_best_a, rem4b, rng)
            obj4b = compute_obj(a4b)
            print("  Phase 4b (active): obj=%.8f" % obj4b)
            if obj4b < global_best_obj:
                global_best_obj = obj4b
                global_best_a = a4b.copy()

    # === Phase 5: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % min(remaining, 300))
        t5 = time.monotonic()
        final_budget = min(remaining, 300.0)
        a5, _ = run_precise_polish(global_best_a, final_budget, rng)
        obj5 = compute_obj(a5)
        print("  Final: obj=%.8f (t=%.1fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Final verification and return ===
    print("\n=== Summary ===")
    print("Best objective: %.8f (n=%d)" % (global_best_obj, len(global_best_a)))

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified objective: %.8f" % ver)
            return final_list
        else:
            print("WARNING: evaluate_sequence rejected the result!")

    # Fallback: uniform sequence
    n_fb = 8192
    S_fb = np.sqrt(2.0 * n_fb)
    fallback = np.full(n_fb, S_fb / n_fb)
    final_list = [float(x) for x in fallback]
    ver_fb = evaluate_sequence(final_list)
    print("Fallback objective: %.8f" % ver_fb)
    return final_list
