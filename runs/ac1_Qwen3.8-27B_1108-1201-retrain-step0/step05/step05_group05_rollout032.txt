"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: Multi-strategy optimization combining:
1. LSE-FISTA with aggressive beta schedule
2. Peak-directed subgradient with Nesterov averaging  
3. Spectral shaping initialization
4. Diverse perturbation-based restarts
5. Final exact-subgradient polish
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -500, 500)
    e = np.exp(scaled)
    tot = e.sum()

        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def topk_subgradient(a, c, k=32):
    """Subgradient from top-k lags."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def weighted_topk_subgradient(a, c, k=32):
    """Weighted subgradient from top-k lags, weights proportional to excess."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    cmax = c.max()
    cmean = c.mean()
    weights = np.maximum(c[idx] - cmean, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def run_fista_v2(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.003,
                 beta_max=10000.0, use_acceleration=True):

    - Momentum reset on beta increase
    - Backtracking-inspired step size

    y = a.copy()

            continue

        x_new = project_simplex(y - step * grad, S)
        
        if use_acceleration:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))

            y = x_new

def run_subgradient_averaging(a_init, S, max_time, rng, k=16):
    """
    Subgradient method with Nesterov averaging for the exact max.
    Step size: eta_t = C / sqrt(t)
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    
    # Nesterov averaging
    a_avg = a.copy()
    it = 0
    C = S / np.sqrt(n)  # Initial step scale
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()
        
        grad = topk_subgradient(a, c, k=k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Diminishing step size
        eta = C / np.sqrt(it + 1)
        
        a_new = project_simplex(a - eta * grad / gnorm, S)
        
        # Nesterov averaging
        alpha_t = 2.0 / (it + 2.0)
        a_avg = (1 - alpha_t) * a_avg + alpha_t * a_new
        
        # Track the averaged solution too
        if it % 100 == 0:
            c_avg = fft_convolve_self(a_avg)
            if c_avg.max() < best:
                best = c_avg.max()
                best_a = a_avg.copy()
        
        a = a_new
    
    # Check averaged solution
    c_avg = fft_convolve_self(a_avg)
    if c_avg.max() < best:
        best = c_avg.max()
        best_a = a_avg.copy()
    
    return best_a, best

def spectral_init(n, rng):
    """
    Create a sequence by shaping its spectrum to be as flat as possible
    while keeping the time-domain signal non-negative.
    """
    fsize = 1 << (2 * n - 1).bit_length()
    # Start with a random phase at each frequency
    # This gives a "random" signal with flat spectrum
    A = np.zeros(fsize, dtype=complex)
    A[0] = np.sqrt(2.0 * n)  # DC component
    for k in range(1, fsize // 2 + 1):
        mag = 1.0  # Flat magnitude
        phase = 2 * np.pi * rng.random()
        A[k] = mag * np.exp(1j * phase)
        if k < fsize - k:
            A[fsize - k] = np.conj(A[k])
    
    a = np.real(ifft(A, fsize))[:n]
    return project_simplex(a, np.sqrt(2.0 * n))

def fourier_perturb(a, rng, scale=0.1):
    """Perturb in Fourier domain."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    # More perturbation at mid frequencies
    weight = np.exp(-((freqs - fsize//4) ** 2) / (fsize ** 2 / 16)) + \
             np.exp(-((freqs - 3*fsize//4) ** 2) / (fsize ** 2 / 16))
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    idx = np.argsort(c)[::-1][:k]
    cmean = c.mean()
    weights = np.maximum(c[idx] - cmean, 1e-10)
    wsum = weights.sum()
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, S, max_time, rng, beta_start=10.0, beta_step=1.05,
              beta_max=50000.0, accel=True, noise_period=3000):
    """
    FISTA with stepwise beta increase.
    Returns (best_a, best_max_b).
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_start
    last_beta_change = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Stepwise beta increase
        if it % 50 == 0 and beta < beta_max:
            new_beta = min(beta_max, beta * beta_step)
            if new_beta > beta * 1.01:
                beta = new_beta
                # Reset momentum on significant beta change
                if it - last_beta_change > 20:
                    t_mom = 1.0
                    y = a.copy()
                last_beta_change = it

        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: 1/(beta * Lipschitz_estimate)
        # For LSE with parameter beta, Lipschitz ~ 4*beta*max|a|
        step = 1.0 / (4.0 * beta)
        step = min(step, 2.0 * S / (n * 10.0))

        # Stuck detection
        if it - last_improve > noise_period:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            if beta > beta_start * 100:
                # Reduce beta when stuck at high values
                beta = max(beta_start, beta * 0.5)
                t_mom = 1.0
                y = a.copy()
                last_beta_change = it
            continue

        x_new = project_simplex(y - step * grad, S)

        if accel:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            t_mom = t_new
        else:
            y = x_new

        a = x_new

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_subgrad_avg(a_init, S, max_time, rng, k=8):
    """Subgradient method with Nesterov averaging for exact max."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    a_avg = a.copy()
    it = 0
    C = 0.5 * S / np.sqrt(n)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = a.copy()

        grad = topk_subgradient(a, c, k=k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        eta = C / np.sqrt(it + 1.0)
        a_new = project_simplex(a - eta * grad / gnorm, S)

        alpha_t = 2.0 / (it + 2.0)
        a_avg = (1.0 - alpha_t) * a_avg + alpha_t * a_new

        if it % 200 == 0:
            c_avg = fft_convolve_self(a_avg)
            if c_avg.max() < best:
                best = c_avg.max()
                best_a = a_avg.copy()

        a = a_new

    c_avg = fft_convolve_self(a_avg)
    if c_avg.max() < best:
        best = c_avg.max()
        best_a = a_avg.copy()

    return best_a, best


def fourier_perturb(a, rng, scale=0.1):
    """Perturb the phase of the DFT."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    # Emphasize mid-frequencies
    center = fsize / 4.0
    bandwidth = fsize / 8.0
    weight = np.exp(-((freqs - center) ** 2) / (2 * bandwidth ** 2))
    weight += np.exp(-((freqs - 3 * center) ** 2) / (2 * bandwidth ** 2))
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def peak_targeted_perturb(a, c, rng, S, strength=0.1):
    """
    Perturb coordinates that contribute most to the current peak.
    The peak is at lag k* = argmax c. The coordinates a[j] and a[k*-j]
    contribute to c[k*].
    """
    n = len(a)
    k_star = int(np.argmax(c))
    # Find coordinates that contribute to the peak lag
    contrib = np.zeros(n)
    for j in range(n):
        other = k_star - j
        if 0 <= other < n:
            contrib[j] = a[other]
    # Perturb high-contribution coordinates more
    contrib_norm = contrib / (contrib.max() + 1e-15)
    noise = rng.normal(0, strength * S / np.sqrt(n), n)
    noise = noise * (0.5 + contrib_norm)
    a_new = project_simplex(a + noise, S)
    return a_new


def generate_inits(n, rng, S, given=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given is not None:
        inits.append(normalize_to_sum(given, S))

    # Spectral: flat magnitude random phase
    fsize = 1 << (2 * n - 1).bit_length()
    for trial in range(3):
        A = np.zeros(fsize, dtype=complex)
        A[0] = S
        for k in range(1, fsize // 2 + 1):
            phase = 2.0 * np.pi * rng.random()
            A[k] = np.exp(1j * phase)
            if k < fsize - k:
                A[fsize - k] = np.conj(A[k])
        a = np.real(ifft(A, fsize))[:n]
        inits.append(project_simplex(a, S))

    # Windows
    inits.append(project_simplex(np.sin(np.pi * tt) ** 2 + 0.01, S))
    inits.append(project_simplex(np.maximum(0.5 * (1.0 + np.cos(2.0 * np.pi * tt)), 0.01), S))
    inits.append(project_simplex(np.maximum(0.54 + 0.46 * np.cos(2.0 * np.pi * tt), 0.01), S))

    # Random
    inits.append(project_simplex(rng.random(n) + 0.1, S))
    inits.append(project_simplex(np.abs(rng.normal(0.8, 0.4, n)) + 0.05, S))

    # Multi-frequency
    inits.append(project_simplex(
        np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * tt + 0.7) + 0.2 * np.cos(10 * np.pi * tt), 0.01), S))

    return inits


def compute_obj(a, n):
    """Compute the objective value."""
    s = a.sum()
    if s < 0.01:
        return 1e10
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 25

    n = 8192
    S = np.sqrt(2.0 * n)
    global_best = 1e10
    global_a = None

    print("Starting optimization, n=%d, budget=%.0fs" % (n, budget_s))

    # Load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and given.sum() > 0:
            given = np.clip(given, 0, 1000)
            given = normalize_to_sum(given, S)
            g_obj = compute_obj(given, n)
            print("Given: length=%d, obj=%.6f" % (len(given), g_obj))
    except Exception:
        pass

    # ---- Phase 1: Quick screening ----
    print("\n=== Phase 1: Quick Screening ===")
    p1_start = time.monotonic()
    p1_total = min(120.0, (deadline - p1_start) * 0.10)

    inits = generate_inits(n, rng, S, given)
    per_init = p1_total / len(inits)

    candidates = []
    for i in range(len(inits)):
        if time.monotonic() > deadline - (deadline - total_start) * 0.85:
            break
        t_alloc = min(per_init, deadline - time.monotonic() - 60.0)
        if t_alloc < 8:
            break
        a_opt, val = run_fista(inits[i], S, t_alloc, rng,
                               beta_start=5.0, beta_step=1.08, beta_max=3000)
        obj = compute_obj(a_opt, n)
        candidates.append((obj, a_opt.copy()))
        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
        print("  Screen %d: obj=%.6f" % (i, obj))

    candidates.sort(key=lambda x: x[0])
    print("  Best after Phase 1: obj=%.6f" % global_best)

    # ---- Phase 2: Long FISTA from best candidates ----
    print("\n=== Phase 2: Long FISTA ===")
    p2_start = time.monotonic()
    p2_total = min(500.0, (deadline - p2_start) * 0.30)

    # Run extended FISTA from top 2 candidates
    n_long = min(2, len(candidates))
    per_long = p2_total / n_long

    for i in range(n_long):
        if time.monotonic() > deadline - (deadline - total_start) * 0.45:
            break
        t_alloc = min(per_long, deadline - time.monotonic() - 60.0)
        if t_alloc < 30:
            break
        print("  Long FISTA from candidate %d (obj=%.6f), t=%.0fs" % (i, candidates[i][0], t_alloc))
        a_opt, val = run_fista(candidates[i][1], S, t_alloc, rng,
                               beta_start=20.0, beta_step=1.06, beta_max=20000)
        obj = compute_obj(a_opt, n)
        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
            print("    New best: obj=%.6f" % global_best)
        else:
            print("    obj=%.6f (no improvement)" % obj)

    # ---- Phase 3: Iterated perturbation + re-optimization ----
    print("\n=== Phase 3: Iterated Local Search ===")
    p3_start = time.monotonic()
    p3_total = min(600.0, (deadline - p3_start) * 0.35)

    current = global_a.copy() if global_a is not None else candidates[0][1].copy()
    c_cur = fft_convolve_self(current)
    best_phase3 = c_cur.max()
    best_a_phase3 = current.copy()

    n_rounds = max(8, int(p3_total / 50))
    per_round = p3_total / n_rounds

    for r in range(n_rounds):
        if time.monotonic() > deadline - (deadline - total_start) * 0.25:
            break
        t_alloc = min(per_round, deadline - time.monotonic() - 40.0)
        if t_alloc < 15:
            break

        progress = r / max(1, n_rounds)

        # Choose perturbation strategy
        strat = rng.random()
        if strat < 0.35:
            # Fourier phase perturbation
            scale = 0.25 * (1.0 - 0.7 * progress) + 0.02
            a_pert = fourier_perturb(current, rng, scale=scale)
        elif strat < 0.60:
            # Peak-targeted perturbation
            a_pert = peak_targeted_perturb(current, c_cur, rng, S,
                                           strength=0.08 * (1.0 - 0.5 * progress) + 0.01)
        elif strat < 0.80:
            # Gaussian noise
            scale = 0.04 * (1.0 - 0.7 * progress) + 0.005
            a_pert = project_simplex(current + rng.normal(0, scale * S / np.sqrt(n), n), S)
        else:
            # Blend with random
            a_rand = project_simplex(rng.random(n) + 0.1, S)
            alpha = 0.15 + 0.25 * rng.random()
            a_pert = project_simplex((1.0 - alpha) * current + alpha * a_rand, S)

        # Optimize perturbed point
        opt_time = t_alloc * 0.75
        a_opt, val = run_fista(a_pert, S, opt_time, rng,
                               beta_start=30.0 + 200.0 * progress,
                               beta_step=1.07, beta_max=30000)
        obj = compute_obj(a_opt, n)

        # Subgradient polish if time permits
        if time.monotonic() < deadline - 30 and t_alloc * 0.2 > 10:
            a_sg, val_sg = run_subgrad_avg(a_opt, S, t_alloc * 0.2, rng, k=6)
            if val_sg < val:
                a_opt = a_sg
                val = val_sg
                obj = compute_obj(a_opt, n)

        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
            current = a_opt.copy()
            c_cur = fft_convolve_self(current)
            best_phase3 = c_cur.max()
            print("  Round %d: NEW BEST obj=%.6f" % (r, global_best))
        elif val < best_phase3:
            # Improved locally but not globally
            current = a_opt.copy()
            c_cur = fft_convolve_self(current)
            best_phase3 = c_cur.max()
            print("  Round %d: local improve to obj=%.6f" % (r, compute_obj(current, n)))
        else:
            print("  Round %d: no improve (obj=%.6f)" % (r, obj))

    # ---- Phase 4: Ultra-high-beta FISTA squeeze ----
    print("\n=== Phase 4: Final Squeeze ===")
    p4_start = time.monotonic()
    p4_total = deadline - p4_start - 10

    if global_a is not None and p4_total > 30:
        # Very high beta FISTA
        seg1 = p4_total * 0.5
        a_hb, val_hb = run_fista(global_a, S, seg1, rng,
                                 beta_start=1000.0, beta_step=1.03,
                                 beta_max=100000.0, noise_period=10000)
        obj_hb = compute_obj(a_hb, n)
        if obj_hb < global_best:
            global_best = obj_hb
            global_a = a_hb.copy()
        print("  After high-beta FISTA: obj=%.6f" % global_best)

        # Subgradient polish
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            a_sg, val_sg = run_subgrad_avg(global_a, S, remaining, rng, k=4)
            obj_sg = compute_obj(a_sg, n)
            if obj_sg < global_best:
                global_best = obj_sg
                global_a = a_sg.copy()
            print("  After subgradient: obj=%.6f" % global_best)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nFinal verification: obj=%.6f" % ver)
        if not np.isfinite(ver):
            print("  REJECTED! Using fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.6f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
