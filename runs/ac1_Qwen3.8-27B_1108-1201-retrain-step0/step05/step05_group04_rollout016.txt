"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Multi-phase optimization combining FISTA with soft active-set gradients,
mirror descent (multiplicative weights), LP-based direction finding, and peak shaving.
Key innovations over previous approach:
1. Soft active set with exponential weighting for smoother gradients
2. Mirror descent updates for better handling of non-negativity
3. LP-based direction (simplified cutting plane)
4. Adaptive switching between methods based on progress

import numpy as np
import time
from scipy.fft import fft, ifft

global_best = np.inf
global_a = None

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def soft_active_gradient(a, c, beta, top_k_frac=0.05):
    """
    Gradient using soft active set with exponential weighting.
    Weights lags proportional to exp(beta * (c[k] - c_max)).
    """
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        tot = 1.0
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta

def hard_active_gradient(a, c, n_active=20):
    """Gradient using hard active set (top n_active lags uniformly weighted)."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:n_active]
    w = np.zeros_like(c)
    w[top_idx] = 1.0 / n_active
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def mirror_descent_step(a, grad, eta, S):
    """
    Multiplicative weight update: a_new = a * exp(-eta * grad), then project.
    This is the mirror descent step with KL divergence.
    """
    # Scale gradient to maintain numerical stability
    gmax = np.abs(grad).max()
    if gmax < 1e-30:
        return a.copy()
    grad_scaled = grad / gmax
    log_a = np.log(np.maximum(a, 1e-300)) - eta * grad_scaled
    # Clip to prevent overflow
    log_a = np.clip(log_a, -50, 50)
    a_new = np.exp(log_a)
    a_new = project_simplex(a_new, S)
    return a_new

def fista_with_adaptive_gradient(a_init, n, S, max_time, rng,
                                   beta_start=50.0, beta_end=20000.0,
                                   beta_steps=200, use_mirror=False,
                                   mirror_freq=5):
    """
    FISTA with adaptive gradient (switching between soft and hard active set).
    Optionally interleave mirror descent steps.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        # Adaptive beta
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)

        # Choose gradient type
        if beta > 1000:
            grad = hard_active_gradient(y, c, n_active=30)
        else:
            grad, _ = soft_active_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        step = 1.0 / (beta if beta > 100 else 100)

        # Stagnation handling
        stagnant = it - last_improve
        if stagnant > 500:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            step *= 0.5

        # Mirror descent step (every mirror_freq iterations)
        if use_mirror and it % mirror_freq == 0:
            eta_mirror = 0.1 * step * gnorm
            x_new = mirror_descent_step(y, grad, eta_mirror, S)
        else:
            x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best

def peak_shaving_multiplicative(a, n, S, max_time, rng, n_peaks=10):
    """
    Multiplicative peak shaving: for each top peak lag k, reduce a[j] proportionally
    to a[k-j] (the partner value).
    """
    a = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cm = c.max()
        
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Compute responsibility
        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            weight = max(0, c[k] - cm * 0.95)
            np.add.at(resp, js, contrib * weight)
        
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        
        # Multiplicative update
        lr = 0.01 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)
    
    return a, fft_convolve_self(a).max()

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Multi-phase optimization combining FISTA with soft/hard active-set gradients,
mirror descent (multiplicative weights), multiplicative peak shaving, and diverse
initialization exploration. Key innovations over previous approach:
1. Soft active set with smooth exponential weighting for stable gradients
2. Mirror descent (multiplicative weights) steps interleaved with FISTA
3. Structure-aware peak shaving using responsibility weights
4. Adaptive switching between gradient types based on convergence progress
"""
import numpy as np
import time
from scipy.fft import fft, ifft

# Global state for tracking best solution
fbest_val = np.inf
fbest_a = None


def fft_conv(self_a):
    """Full self-convolution via FFT."""
    n = len(self_a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(self_a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def soft_active_gradient(a, c, beta):
    """
    Gradient using soft active set with exponential weighting.
    Weights lags proportional to exp(beta * (c[k] - c_max)).
    Returns (gradient, lse_value).
    """
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        tot = 1.0
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def hard_active_gradient(a, c, n_active=20):
    """Gradient using hard active set (top n_active lags uniformly weighted)."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:n_active]
    w = np.zeros_like(c)
    w[top_idx] = 1.0 / n_active
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def mirror_descent_step(a, grad, eta, S):
    """
    Multiplicative weight update: a_new = a * exp(-eta * grad), then project.
    This is the mirror descent step with KL divergence.
    """
    gmax = np.abs(grad).max()
    if gmax < 1e-30:
        return a.copy()
    grad_scaled = grad / gmax
    log_a = np.log(np.maximum(a, 1e-300)) - eta * grad_scaled
    log_a = np.clip(log_a, -50, 50)
    a_new = np.exp(log_a)
    a_new = project_simplex(a_new, S)
    return a_new


def run_fista(a_init, n, S, max_time, rng,
              beta_start=50.0, beta_end=20000.0, beta_steps=200,
              use_mirror=False, mirror_freq=5, hard_active_threshold=1000):
    """
    FISTA with adaptive gradient (switching between soft and hard active set).
    Optionally interleave mirror descent steps.
    Returns (best_a, best_val).
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_conv(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        # Adaptive beta (exponential schedule)
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 60000.0)

        # Choose gradient type based on beta
        if beta > hard_active_threshold:
            n_act = min(50, max(10, int(50 * np.exp(-(beta - hard_active_threshold) / 5000))))
            grad = hard_active_gradient(y, c, n_active=n_act)
        else:
            grad, _ = soft_active_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        step = 1.0 / max(beta, 50.0)

        # Stagnation handling
        stagnant = it - last_improve
        if stagnant > 800:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            step *= 0.4

        # Mirror descent step (every mirror_freq iterations if enabled)
        if use_mirror and it % mirror_freq == 0:
            eta_mirror = 0.15 * step * gnorm
            x_new = mirror_descent_step(y, grad, eta_mirror, S)
        else:
            x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    # Final check
    c_final = fft_conv(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_peak_shaving(a_init, n, S, max_time, n_peaks=10, lr_base=0.01):
    """
    Multiplicative peak shaving: for each top peak lag k, reduce a[j] proportionally
    to the partner value a[k-j]. Uses temporal decay on learning rate.
    Returns (best_a, best_val).
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_conv(a)
        cm = c.max()
        if cm < best:
            best = cm
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:n_peaks]

        # Compute responsibility for each element
        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            weight = max(0.0, c[k] - cm * 0.90)
            np.add.at(resp, js, contrib * weight)

        resp_max = resp.max()
        if resp_max < 1e-30:
            break

        # Adaptive learning rate with decay
        elapsed = time.monotonic() - start
        lr = lr_base * S / (n * resp_max) * np.exp(-elapsed / (max_time * 2))
        if lr < 1e-10:
            break

        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)

    # Final check
    c_final = fft_conv(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_combined_gradient(a_init, n, S, max_time, rng):
    """
    Alternate between multiple gradient strategies:
    1. Top-1 lag gradient (most aggressive)
    2. Top-k uniform gradient
    3. Soft exponential gradient

    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    strategies = [
        ("top1", 1),
        ("topk", 15),
        ("soft", 200),

        it += 1

            best_a = a.copy()

            grad = hard_active_gradient(a, c, n_active=1)

            grad = hard_active_gradient(a, c, n_active=sparam)

            if noise_scale > 1e-10:

    return best_a, best


def generate_inits(n, rng):
    """Generate diverse initializations with varying asymmetry."""
    inits = []
    t = np.linspace(0, 1, n)

        for wf in [0.3, 0.4, 0.5, 0.6]:
            center = cf
            width = wf / 2.0
            x = (t - center) / width
            a = np.zeros(n)
            mask = np.abs(x) < 1
            a[mask] = 0.5 * (1.0 + np.cos(np.pi * x[mask]))
            inits.append(np.maximum(a, 1e-10))

    # Gaussian bumps
    for sigma in [0.08, 0.12, 0.15, 0.2]:
        for mu in [0.25, 0.35, 0.45, 0.55]:
            a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
            inits.append(np.maximum(a, 1e-10))

    # Two-bump structures (asymmetric)
    for mu1 in [0.2, 0.3]:
        for mu2 in [0.6, 0.7, 0.8]:
            for s in [0.1, 0.15]:
                a = np.exp(-0.5 * ((t - mu1) / s) ** 2) + \
                    0.5 * np.exp(-0.5 * ((t - mu2) / s) ** 2)
                inits.append(np.maximum(a, 1e-10))

    # Power law
    for p in [0.5, 1.0, 2.0, 3.0, 5.0]:
        inits.append(np.maximum(t ** p, 1e-10))
        inits.append(np.maximum((1.0 - t) ** p, 1e-10))

    # Linear + noise
    a = t + 0.1
    inits.append(np.maximum(a, 0.01))
    a = 1.0 - t + 0.1
    inits.append(np.maximum(a, 0.01))

    # Logarithmic
    inits.append(np.maximum(np.log(t + 0.01), 0.01))
    inits.append(np.maximum(np.log(1.0 - t + 0.01), 0.01))

    return inits
