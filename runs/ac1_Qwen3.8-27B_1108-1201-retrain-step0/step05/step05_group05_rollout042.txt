Or even simpler: just use the projection. After each L-BFGS-B step, project to {b >= 0, sum(b) = 1}. This is the simplex projection which is O(n log n).

Actually, the cleanest approach: since the objective is homogeneous (scale-invariant in the way I showed), we can just minimize 2n * max_k (a*a)[k] / n subject to a >= 0, sum(a) = 1. The factor of 2n is constant.

So minimize: max_k (a*a)[k] subject to a >= 0, sum(a) = 1.

With LSE smoothing: minimize (1/beta) * log(sum_k exp(beta * (a*a)[k])) subject to a >= 0, sum(a) = 1.

For L-BFGS-B, I'll:
1. Use the simplex projection after each step (or better, use the bounds + equality constraint)
2. Actually, let me just work in the "log-barrier" space: parameterize a = exp(x) / sum(exp(x_i)) (softmax). Then any x gives a valid a on the simplex interior. This avoids the constraint entirely!

The softmax parameterization:
a_i = exp(x_i) / sum_j exp(x_j)

This ensures a > 0 and sum(a) = 1 automatically. The gradient w.r.t. x is:
da_i/dx_j = a_i * (delta_ij - a_j)

So d(obj)/dx_j = sum_i d(obj)/da_i * a_i * (delta_ij - a_j)
= d(obj)/da_j - a_j * sum_i d(obj)/da_i

This is a standard softmax gradient transformation.

The advantage: no constraints, no projection needed. L-BFGS-B can work directly on x.

The disadvantage: if any a_i is very close to 0, the corresponding x_i is very negative, which can cause numerical issues. But in practice, this is usually fine.

def softmax_objective(x, beta, n):
    """Compute LSE-smoothed max of auto-convolution for softmax parameterization."""
    a = softmax(x)
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    obj = (1.0/beta) * np.log(np.sum(np.exp(scaled)))
    return obj

def softmax_objective_grad(x, beta, n):
    """Gradient of LSE-smoothed max w.r.t. x (softmax params)."""
    a = softmax(x)
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    w = e / e.sum()
    # grad w.r.t. a: 2*(w * a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    # Transform to x-space: grad_x_j = grad_a_j - a_j * sum_i grad_a_i
    grad_x = grad_a - a * grad_a.sum()
    return grad_x

Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: Multi-phase optimization using:
1. Multi-scale initialization (small-to-large n)

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def lse_obj_and_grad(x, beta, n):
    """
    LSE-smoothed max of auto-convolution for softmax-parameterized a.
    Returns (objective, gradient) w.r.t. x.
    """
    a = softmax(x)
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return 0.0, np.zeros(n)
    w = e / tot
    obj = (1.0 / beta) * np.log(tot)
    
    # Gradient w.r.t. a
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    # Transform to x-space (softmax chain rule)
    grad_x = grad_a - a * grad_a.sum()
    return obj, grad_x


def lse_obj_only(x, beta, n):
    """Objective only (for cases where we don't need gradient)."""
    a = softmax(x)
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    return (1.0 / beta) * np.log(tot)


def exact_objective(a):
    """Compute the exact objective value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def optimize_lbfgsb(x_init, n, betas, max_iter=200, time_limit=60.0):
    """
    Run L-BFGS-B with continuation on beta.
    Returns best x found.
    """
    x = x_init.copy()
    start = time.monotonic()
    
    for beta in betas:
        if time.monotonic() - start > time_limit:
            break
        
        # Run L-BFGS-B for this beta
        result = minimize(
            lse_obj_only, x, args=(beta, n),
            method='L-BFGS-B',
            jac=lambda xx: lse_obj_and_grad(xx, beta, n)[1],
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        if result.x is not None:
            x = result.x
    
    return x


def a_to_x(a):
    """Convert a non-negative vector (sum=1) to softmax parameters."""
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return np.log(a) - np.log(a).mean()  # Center for numerical stability


def upsample(a, factor=2):
    """Upsample a sequence by interpolation, preserving sum."""
    n = len(a)
    n_new = n * factor
    # Use sinc interpolation
    from scipy.interpolate import interp1d
    x_old = np.linspace(0, 1, n, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    f = interp1d(x_old, a, kind='cubic', fill_value=0, assume_sorted=True)
    a_new = f(x_new)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s  # Normalize to sum 1
    return a_new


def fourier_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.1 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]  # Preserve DC
    a_new = np.real(ifft(new_A, fsize))[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses L-BFGS-B with softmax parameterization and LSE smoothing,
combined with multi-scale init, diverse restarts, and basin hopping.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from scipy.interpolate import interp1d


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def a_to_x(a):
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return np.log(a) - np.log(a).mean()


def lse_obj_and_grad(x, beta, n):
    a = softmax(x)
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return 0.0, np.zeros(n)
    w = e / tot
    obj = (1.0 / beta) * np.log(tot)
    
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    grad_x = grad_a - a * np.dot(grad_a, a)
    return obj, grad_x


def lse_fun(x, beta, n):
    return lse_obj_and_grad(x, beta, n)[0]


def lse_jac(x, beta, n):
    return lse_obj_and_grad(x, beta, n)[1]


def exact_objective(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_lbfgsb(x_init, n, betas, max_iter=200, time_limit=30.0):
    x = x_init.copy()
    start = time.monotonic()
    for beta in betas:
        if time.monotonic() - start > time_limit:
            break
        result = minimize(
            lse_fun, x, args=(beta, n),
            method='L-BFGS-B',
            jac=lse_jac,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': max_iter * 5}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            x = result.x
    return x


def upsample_interp(a, n_new):
    n = len(a)
    if n == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    f = interp1d(x_old, a, kind='cubic', fill_value=0, assume_sorted=True)
    a_new = np.maximum(f(x_new), 0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    else:
        a_new = np.ones(n_new) / n_new
    return a_new


def fourier_mutate(a, rng, scale=0.1):
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.1 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new


def generate_starts(n, rng, given_a=None):
    """Generate diverse starting points (normalized to sum=1)."""
    starts = []

    starts.append(rng.random(n) + 0.1)
    starts.append(np.abs(rng.normal(0.8, 0.4, n)) + 0.05)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: L-BFGS-B with softmax parameterization and LSE smoothing,
combined with multi-scale initialization, diverse restarts, and basin hopping.
Key differences from prior approaches: second-order optimization (L-BFGS-B),
softmax reparameterization to eliminate projection overhead, and many fast
restarts enabled by the speed of second-order methods.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from scipy.interpolate import interp1d


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def a_to_x(a):
    """Convert non-negative vector (sum=1) to centered log-space (softmax params)."""
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    return np.log(a) - np.log(a).mean()


def lse_obj_and_grad(x, beta, n):
    """
    LSE-smoothed max of auto-convolution with softmax parameterization.
    Returns (objective, gradient) w.r.t. x.
    """
    a = softmax(x)
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return 0.0, np.zeros(n)
    w = e / tot
    obj = (1.0 / beta) * np.log(tot)

    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    # Softmax chain rule: grad_x_j = a_j * (grad_a_j - dot(grad_a, a))
    grad_x = a * (grad_a - np.dot(grad_a, a))
    return obj, grad_x


def lse_fun(x, beta, n):
    """Objective function wrapper for scipy.optimize."""
    return lse_obj_and_grad(x, beta, n)[0]


def lse_jac(x, beta, n):
    """Gradient function wrapper for scipy.optimize."""
    return lse_obj_and_grad(x, beta, n)[1]


def exact_objective(a):
    """Compute the exact objective: 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_lbfgsb(x_init, n, betas, max_iter=200, time_limit=30.0):
    """
    Run L-BFGS-B with continuation on the LSE smoothing parameter beta.
    Returns the best x found.
    """
    x = x_init.copy()
    start = time.monotonic()
    for beta in betas:
        if time.monotonic() - start > time_limit:
            break
        result = minimize(
            lse_fun, x, args=(beta, n),
            method='L-BFGS-B',
            jac=lse_jac,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': max_iter * 5}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            x = result.x
    return x


def upsample_interp(a, n_new):
    """Upsample a sequence (sum=1) to new length via cubic interpolation."""
    n = len(a)
    if n == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    f = interp1d(x_old, a, kind='cubic', fill_value=0, assume_sorted=True)
    a_new = np.maximum(f(x_new), 0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    else:
        a_new = np.ones(n_new) / n_new
    return a_new


def fourier_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation preserving DC component."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.08 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new


def fourier_mutate_small(a, rng, nbands=8):
    """
    Sparse Fourier mutation: perturb only a random subset of frequency bins.
    More targeted than full mutation.
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase = np.angle(A)
    # Pick random frequency bins to perturb
    n_perturb = max(4, fsize // nbands)
    indices = rng.choice(fsize, size=n_perturb, replace=False)
    scale = 0.1 * rng.random()
    phase[indices] += rng.normal(0, scale, n_perturb)
    mag = np.abs(A)
    new_A = mag * np.exp(1j * phase)
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new


def generate_starts(n, rng, given_a=None):
    """Generate diverse starting points, each normalized to sum=1."""
    starts = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_a is not None and len(given_a) == n:
        starts.append(given_a.copy())

    starts.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    starts.append(0.5 + 0.5 * np.cos(2 * np.pi * tt))
    starts.append(np.sin(np.pi * tt) ** 2 + 0.01)
    starts.append(np.sin(np.pi * tt) ** 4 + 0.01)
    starts.append(np.exp(-8.0 * (tt - 0.5) ** 2) + 0.01)
    starts.append(np.exp(-3.0 * (tt - 0.5) ** 2) + 0.01)
    starts.append(np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * tt + 0.7)
                             + 0.2 * np.cos(10 * np.pi * tt), 0.01))
    starts.append(np.maximum(1.0 + 0.4 * np.sin(6 * np.pi * tt)
                             + 0.15 * np.cos(14 * np.pi * tt + 0.3), 0.01))
    starts.append(np.maximum(1.0 + 0.25 * np.sin(2 * np.pi * tt)
                             + 0.25 * np.cos(3 * np.pi * tt + 1.0), 0.01))
    starts.append(rng.random(n) + 0.1)
    starts.append(np.abs(rng.normal(0.8, 0.4, n)) + 0.05)

    # Block structures
    for nb in [4, 8, 16, 32, 64, 128]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.2, 1.5, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        starts.append(block[:n] + 0.01)

    # Triangular / tent
    starts.append(np.abs(1.0 - 2.0 * tt) + 0.01)
    starts.append(np.maximum(1.0 - 4.0 * np.abs(tt - 0.5), 0.01))

    normalized = []
    for a in starts:
        a = np.maximum(a, 0.001)
        a = a / a.sum()
        normalized.append(a)

    return normalized


def sort_by_first(item):
    """Key function for sorting tuples by first element."""
    return item[0]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def polish_subgradient(a, n, time_limit=30.0):
    """
    Subgradient polish: iteratively take small steps in the negative
    LSE-gradient direction, projected onto the simplex.
    """
    S = a.sum()
    a = a.copy()
    start = time.monotonic()
    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < time_limit:
        c = fft_convolve_self(a)
        cmax = c.max()

        # Adaptive beta: increase as we progress
        elapsed_frac = (time.monotonic() - start) / time_limit
        beta = 500.0 * (1.0 + 20.0 * elapsed_frac)

        scaled = beta * (c - cmax)
        scaled = np.clip(scaled, -500, 500)
        e = np.exp(scaled)
        w = e / e.sum()

        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Adaptive step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = 0.3 * S / (n * gnorm / (S / n) + 1e-10)
        step = min(step, 0.5 * S / n)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 0:
            a_new = a_new * (S / s)
        else:
            break
        a = a_new

    return a
