import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv_full(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_objective_grad(a, N, beta):
    """Compute LSE-smoothed peak value and gradient of conv(a,a)."""
    n = len(a)
    c = compute_conv_full(a, N)
    c_max = c.max()
    if beta < 1e-6:
        weights = np.ones(len(c)) / len(c)
        lse_val = c_max
    else:
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:
            weights = exp_bc / total
            lse_val = c_max + np.log(total) / beta
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def adaptive_beta_from_peaks(c, top_k=10):
    """Choose beta based on the spread of top peaks."""
    top = np.sort(c)[::-1][:min(top_k, len(c))]
    max_val = top[0]
    mean_top = top.mean()
    spread = max_val - mean_top
    if spread < 1e-6:
        return 50.0
    beta = 1.0 / spread
    return np.clip(beta, 5.0, 500.0)


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def eg_update(a, g, eta):
    """Exponentiated gradient (multiplicative weight) update on simplex."""
    n = len(a)
    S = a.sum()
    # a_new proportional to a * exp(-eta * g), normalized to sum = S, non-negative
    log_a = np.full(n, -700.0)
    mask = a > 1e-30
    log_a[mask] = np.log(a[mask])
    log_a -= eta * g
    # softmax to ensure non-negativity and proper scaling
    log_a_max = log_a.max()
    exp_a = np.exp(log_a - log_a_max)
    total = exp_a.sum()
    if total < 1e-300:
        return a.copy()
    a_new = exp_a / total * S
    return a_new


def subgradient_topk_step(a, c, top_k, eta):
    """Subgradient step focusing on top-k peaks simultaneously."""
    n = len(a)
    S = a.sum()
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a.copy()
    weights = top_vals / total
    fsize = 1 << (3 * n - 2).bit_length()
    w_full = np.zeros(fsize)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]
    W = fft(w_full, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    candidate = project_simplex(a - eta * grad, S)
    return candidate


def peak_targeted_perturbation(a, c, S, rng, strength=0.1):
    """Perturb a by adding noise weighted by peak contributions."""
    n = len(a)
    c_max = c.max()
    # Find which indices contribute most to the peak
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for i in range(lo, hi + 1):
        j = k_star - i
        if 0 <= j < n:
            contrib[i] += a[j]
            if i != j:
                contrib[j] += a[i]
    # Add noise: reduce high contributors, increase low ones
    noise = rng.standard_normal(n)
    perturbation = strength * (noise - contrib / max(c_max, 1e-10))
    a_new = a + perturbation
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def spectral_partial_restart(a, N, rng, frac=0.3):
    """Randomize a fraction of FFT phases."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

            elif best_ls is None or cm_ls < best_ls:
                best_ls = cm_ls
                a_new = candidate
                cmax_new = cm_ls
            alpha_ls *= 0.6

        if cmax_new < best_cmax - 1e-12:
            best_cmax = cmax_new
            best_a = a_new.copy()

        g_new, f_new, c_nw, cmax_nw = lse_objective_grad(a_new, N, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c = c_nw
        c_max = cmax_nw

    return best_a, best_cmax


def eg_block(a_init, n, S, N, max_time, rng):
    """Exponentiated gradient optimization block."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0

    c = compute_conv_full(a, N)
    c_max = c.max()
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    base_eta = 4.0 * S / (n * c_max + 1e-10)

    while time.monotonic() - start_time < max_time:
        it += 1
        eta = base_eta / (1.0 + 0.01 * it)

        # Use LSE gradient with moderate beta for smoothness
        beta = 20.0
        g, f_val, c, c_max = lse_objective_grad(a, N, beta)

        a_new = eg_update(a, g, eta)
        a_new = safe_normalize(a_new, n, S)

        c_new = compute_conv_full(a_new, N)
        cm_new = c_new.max()

        if cm_new < best_cmax - 1e-12:
            best_cmax = cm_new
            best_a = a_new.copy()
            a = a_new
        elif cm_new < c_max:
            a = a_new
        else:
            # Revert and try smaller step
            a_new2 = eg_update(a, g, eta * 0.3)
            a_new2 = safe_normalize(a_new2, n, S)
            c_new2 = compute_conv_full(a_new2, N)
            if c_new2.max() < c_max:
                a = a_new2
            else:
                it += 5  # Effective slowdown

        if it % 200 == 0:
            c = compute_conv_full(a, N)
            c_max = c.max()

    return best_a, best_cmax


def subgradient_polish_block(a_init, n, S, N, max_time, rng, top_k_init=5):
    """Subgradient polishing with peak cycling and diminishing steps."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    stall = 0

    c = compute_conv_full(a, N)
    c_max = c.max()
    best_cmax = c_max

    base_step = 0.05 * S / (n * c_max + 1e-10)

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(30, top_k_init + it // 100)
        eta = base_step / (1.0 + 0.002 * it)

        c = compute_conv_full(a, N)
        c_max = c.max()
        if c_max < best_cmax - 1e-12:
            best_cmax = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 100:
            # Peak cycling: focus on a different peak
            c_sorted = np.sort(c)[::-1]
            if len(c_sorted) > top_k and c_sorted[top_k] > c_sorted[0] * 0.9:
                # Many peaks at similar level - use more of them
                top_k = min(50, top_k * 2)
            stall = 0

        a_new = subgradient_topk_step(a, c, top_k, eta)
        c_new = compute_conv_full(a_new, N)
        if c_new.max() <= c_max + 1e-10:
            a = a_new
        else:
            # Try half step
            a_half = subgradient_topk_step(a, c, top_k, eta * 0.5)
            c_half = compute_conv_full(a_half, N)
            if c_half.max() <= c_max:
                a = a_half

    return best_a, best_cmax


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing evaluate_sequence.
    Uses adaptive-beta L-BFGS, exponentiated gradient, and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    # Determine n: use the given sequence length if reasonable, else 8192
    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and given.size <= 32768:
            n = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = safe_normalize(a_candidate, n, S)
        c = compute_conv_full(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            score = float(2 * n * sc / (S * S))
            if label:
                print("  [%s] peak=%.6f score=%.6f" % (label, sc, score))
        return sc

    # Start from given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            a_start = np.clip(given, 0.0, 1000.0).astype(np.float64)
            s = a_start.sum()
            if s > 0.01:
                a_start = a_start * (S / s)
    except Exception:
        a_start = np.ones(n) * S / n

    update_best(a_start, "initial")
    print("Starting score: %.6f" % (2 * n * global_best / (S * S) if global_best < np.inf else np.inf))

    a_curr = global_a.copy() if global_a is not None else a_start.copy()

    # === Phase 1: Adaptive L-BFGS (moderate beta) ===
    print("=== Phase 1: Adaptive L-BFGS (beta 10-100) ===")
    t1 = time.monotonic()
    p1_budget = min(300.0, (deadline - t1) * 0.15)
    a_new, sc_new = lbfgs_adaptive_block(a_curr, n, S, N, p1_budget, rng,
                                          beta_init=10.0, beta_max=150.0, m_lbfgs=10)
    update_best(a_new, "P1-LBFGS")
    if global_a is not None:
        a_curr = global_a.copy()

    # === Phase 2: High-beta L-BFGS ===
    print("=== Phase 2: High-beta L-BFGS (beta 50-500) ===")
    t2 = time.monotonic()
    p2_budget = min(500.0, (deadline - t2) * 0.25)
    # Run in sub-blocks with increasing beta
    n_subblocks = 3
    per_block = p2_budget / n_subblocks
    for sb in range(n_subblocks):
        if time.monotonic() > deadline - 300:
            break
        b_start = 50.0 * (sb + 1)
        b_end = min(500.0, 200.0 * (sb + 1))
        rem = max(1.0, (t2 + p2_budget) - time.monotonic() - 2.0)
        a_new, sc_new = lbfgs_adaptive_block(a_curr, n, S, N, min(per_block, rem), rng,
                                              beta_init=b_start, beta_max=b_end, m_lbfgs=12)
        update_best(a_new, "P2-LBFGS-%d" % sb)
        if global_a is not None and sc_new <= global_best + 1e-9:
            a_curr = global_a.copy()
        else:
            a_curr = a_new

    # === Phase 3: Subgradient polishing ===
    print("=== Phase 3: Subgradient polishing ===")
    t3 = time.monotonic()
    p3_budget = min(400.0, (deadline - t3) * 0.20)
    a_new, sc_new = subgradient_polish_block(global_a, n, S, N, p3_budget, rng, top_k_init=5)
    update_best(a_new, "P3-subgrad")
    if global_a is not None and sc_new <= global_best:
        a_curr = global_a.copy()

    # === Phase 4: EG-based optimization ===
    print("=== Phase 4: Exponentiated gradient ===")
    t4 = time.monotonic()
    p4_budget = min(300.0, (deadline - t4) * 0.15)
    a_new, sc_new = eg_block(global_a, n, S, N, p4_budget, rng)
    update_best(a_new, "P4-EG")
    if global_a is not None and sc_new <= global_best:
        a_curr = global_a.copy()

    # === Phase 5: Final high-beta L-BFGS ===
    print("=== Phase 5: Final high-beta L-BFGS ===")
    t5 = time.monotonic()
    remaining5 = deadline - t5 - 20.0
    if remaining5 > 30:
        a_new, sc_new = lbfgs_adaptive_block(global_a, n, S, N, min(remaining5, 400.0), rng,
                                              beta_init=100.0, beta_max=2000.0, m_lbfgs=15)
        update_best(a_new, "P5-final")

    # === Phase 6: Quick subgradient + L-BFGS alternation ===
    remaining6 = deadline - time.monotonic() - 15.0
    if remaining6 > 40:
        half = remaining6 / 2.0
        a_new, sc_new = subgradient_polish_block(global_a, n, S, N, min(half, 200.0), rng, top_k_init=10)
        update_best(a_new, "P6a-subgrad")
        if deadline - time.monotonic() > 20:
            a_new, sc_new = lbfgs_adaptive_block(global_a, n, S, N,
                                                  min(deadline - time.monotonic() - 5, 200.0), rng,
                                                  beta_init=200.0, beta_max=3000.0, m_lbfgs=15)
            update_best(a_new, "P6b-LBFGS")

    # === Final verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification score: %.6f" % ver)
        else:
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    final_score = 2.0 * len(global_a) * global_best / (S * S) if global_best < np.inf else float('inf')
    print("Returning length %d, internal peak=%.6f, approx score=%.6f" % (len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
