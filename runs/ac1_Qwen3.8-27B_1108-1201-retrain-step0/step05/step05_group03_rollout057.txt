import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def pnorm_gradient(a, p):
    """Compute gradient of (sum c^p)^(1/p) where c = convolve(a,a)."""
    n = len(a)
    c = fft_convolve_self(a)
    n_conv = len(c)  # 2n-1
    
    # w[k] = c[k]^(p-1)
    # Handle potential underflow: use log-space or clip
    c_clipped = np.maximum(c, 1e-30)
    log_w = (p - 1) * np.log(c_clipped)
    # Normalize to avoid overflow
    log_w_max = log_w.max()
    w = np.exp(log_w - log_w_max)  # scaled version

    # J = (sum c^p)^(1/p)
    # dJ/da[i] = (1/p) * (sum c^p)^(1/p - 1) * p * sum_j a[j] * c[j+i]^(p-1)
    #          = (sum c^p)^(1/p - 1) * sum_j a[j] * c[j+i]^(p-1)
    
    # For optimization purposes, we can use the unnormalized gradient:
    # grad[i] = sum_j a[j] * c[j+i]^(p-1)
    # and divide by (sum c^p)^(1-1/p) if needed (but for gradient descent, the scale matters less)
    
    # Actually, let's use the "power" objective: J = sum c^p (simpler gradient)
    # dJ/da[i] = 2p * sum_j a[j] * c[j+i]^(p-1)
    # (factor of 2 from the symmetry of convolution, as derived above)
    
    # Compute using FFT:
    # grad[i] = 2p * (a * w_rev)[2n-2-i]
    w_rev = w[::-1]
    
    # Convolution of a (length n) with w_rev (length 2n-1)
    conv_size = n + (2*n - 1) - 1  # = 3n - 2
    fsize = 1 << (conv_size - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w_rev, fsize)
    conv = np.real(ifft(A * W))[:conv_size]

    grad = 2.0 * p * conv[2*n-2 : 2*n-2-n : -1]

    return grad, (c_clipped ** p).sum() ** (1.0/p), c  # gradient, p-norm value, convolution
