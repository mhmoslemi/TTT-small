def freq_domain_step(a, target_flat, n_steps):
    """Adjust a in frequency domain to make A^2 more flat."""
    n = len(a)
    fsize = 1 << (2*n-1).bit_length()
    A = fft(a, fsize)
    
    for _ in range(n_steps):
        B = A * A  # This is FFT of the convolution
        b = np.real(ifft(B))[:2*n-1]
        
        # Compute the "flatness" target
        # We want B to be close to a constant
        # But B must be the square of A, and A must correspond to a real, non-negative sequence
        
        # Simple approach: scale B towards uniform
        B_mean = B.mean()
        B_new = B + 0.1 * (B_mean - B)
        
        # Take square root (this is the tricky part)
        # A_new = sqrt(B_new) - but this changes the phase
        # Better: adjust only the magnitude of A
        A_mag = np.abs(A)
        A_phase = np.angle(A)
        
        # Target magnitude: sqrt(B_mean) for all frequencies
        target_mag = np.sqrt(np.abs(B_mean))
        A_mag_new = A_mag + 0.05 * (target_mag - A_mag)
        
        A = A_mag_new * np.exp(1j * A_phase)
    
    a_new = np.real(ifft(A))[:n]
    return a_new

`a_proj[j] + a_proj[n-1-j] = (c + a[j] - a[n-1-j])/2 + (c + a[n-1-j] - a[j])/2 = c`. ✓

But this projection changes the sum: `sum(a_proj) = n*c/2`. We need to adjust `c` to maintain the sum.

Actually, `sum(a_proj) = sum_j (c/2 + (a[j]-a[n-1-j])/2) = n*c/2 + 0 = n*c/2`.
And `sum(a) = sum_j (a[j] + a[n-1-j])/2... no, `sum(a) = sum_j a[j]`.
`sum(a_proj) = n*c/2` where `c = (2/n) * sum_j a[j]` (since `sum_j (a[j]+a[n-1-j]) = 2*sum(a)`).

So `sum(a_proj) = n * (2/n * sum(a)) / 2 = sum(a)`. 

So the anti-palindromic projection preserves the sum! And it's a linear projection. We can also project onto the non-negative orthant simultaneously (by taking max with 0), though that breaks the anti-palindromic property.

This is a useful tool. We can periodically project onto the anti-palindromic subspace to guide the optimization towards the expected structure.

    shifted = shifted - shifted.max()

    n = len(y)

    s = a.sum()
    if s < 1e-15:

    """Compute LSE approximation of max."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    return cm + np.log(np.exp(shifted).sum()) / beta


def spectral_flatten(a, S, rng, n_iter=50, strength=0.05):
    """Iteratively flatten the spectrum of A^2 to reduce convolution peaks."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    
    for _ in range(n_iter):
        B = A * A
        b = np.real(ifft(B))[:2 * n - 1]
        b_max = b.max()
        b_mean = b.mean()
        
        # Target: make |B| more uniform
        B_mag = np.abs(B)
        B_mag_target = np.full_like(B_mag, B_mag.mean())
        B_mag = B_mag + strength * (B_mag_target - B_mag)
        
        # Also try to align phases of B to be constructive at DC
        B_phase = np.angle(B)
        # Gently pull B towards its mean (which is real positive for valid a)
        B_new = B + strength * (B.mean() - B)
        
        # Recover A from B_new: A_new = sqrt(B_new)
        # Use magnitude adjustment only to preserve realness
        A_mag = np.abs(A)
        B_new_mag = np.abs(B_new)
        A_mag_target = np.sqrt(np.maximum(B_new_mag, 0))
        A_mag = A_mag + 0.1 * (A_mag_target - A_mag)
        
        A = A_mag * np.exp(1j * np.angle(A))
    
    a_new = np.real(ifft(A))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:

    return a_new

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, compute_obj(best_a)


def generate_anti_pal_init(n, rng, S, freq=1):
    """Generate anti-palindromic initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    # Base: raised cosine (naturally anti-palindromic up to constant)
    base = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
    # Add smooth random perturbation that preserves anti-palindromic structure
    d = rng.normal(0, 0.1, n)
    d = d - d[::-1]  # make anti-symmetric
    F = fft(d)
    cutoff = max(3, n // 50)
    F[cutoff:-cutoff] *= 0.1
    d = np.real(ifft(F))[:n]
    base = base + d
    base = np.maximum(base, 0.01)
    base = base * (S / base.sum())
    return base


def generate_spectral_init(n, rng, S):
    """Generate initialization with flat A^2 spectrum."""
    fsize = 1 << (2 * n - 1).bit_length()
    # Start with random phases
    A = np.zeros(fsize, dtype=np.complex128)
    A[0] = S  # DC component
    # Add small random components
    for k in range(1, fsize // 2):
        mag = rng.uniform(0, 0.1 * S / np.sqrt(n))
        phase = rng.uniform(0, 2 * np.pi)
        A[k] = mag * np.exp(1j * phase)
        A[fsize - k] = mag * np.exp(-1j * phase)  # conjugate for realness
    
    a = np.real(ifft(A))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a * (S / s)
    else:
        a = np.ones(n) * S / n
    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses FISTA with backtracking, anti-palindromic projection, spectral shaping,
    and multi-method polishing with diverse initializations.

    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        S_init = np.sqrt(2.0 * n0)
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.54 + 0.46 * np.cos(2 * np.pi * t)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    print("Starting optimization, target: 1.5027, current: %.8f" % global_best_obj)

    # === Phase 1: FISTA with backtracking + anti-pal (40% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.40
    print("\n=== Phase 1: FISTA-BT with anti-pal (n=%d, budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    a1, obj1 = run_fista_bt(global_best_a, p1_budget * 0.5, rng,
                            beta_start=50.0, beta_rate=0.1,
                            max_beta=10000.0, use_anti_pal=True,
                            anti_pal_every=300, anti_pal_blend=0.08,
                            spectral_every=3000)
    print("  FISTA-BT part 1: obj=%.8f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    rem1 = p1_budget - (time.monotonic() - t1)
    if rem1 > 5:
        a1b, obj1b = run_fista_bt(global_best_a, rem1, rng,
                                  beta_start=200.0, beta_rate=0.05,
                                  max_beta=30000.0, use_anti_pal=True,
                                  anti_pal_every=500, anti_pal_blend=0.03,
                                  spectral_every=5000)
        print("  FISTA-BT part 2: obj=%.8f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: Top-k active set polish (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = remaining * 0.15
        print("\n=== Phase 2: Top-k polish (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()
        a2, obj2 = run_topk_polish(global_best_a, p2_budget, rng)
        print("  Top-k: obj=%.8f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: L-BFGS-B polishing (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = remaining * 0.20
        print("\n=== Phase 3: L-BFGS-B polishing (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        print("  L-BFGS: obj=%.8f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Multi-start with diverse inits (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:

        t4 = time.monotonic()

            else:
                print("  Multi-start %d: obj=%.8f" % (idx, obj_m))

    # === Phase 5: Final intensive polish (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        p5_budget = remaining * 0.10
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        # FISTA with very high beta
        a5, obj5 = run_fista_bt(global_best_a, p5_budget * 0.4, rng,
                                beta_start=1000.0, beta_rate=0.5,
                                max_beta=50000.0, use_anti_pal=False,
                                anti_pal_every=10000, anti_pal_blend=0.0,
                                spectral_every=100000)
        print("  High-beta FISTA: obj=%.8f" % obj5)
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        rem5 = p5_budget - (time.monotonic() - t5)
        if rem5 > 5:
            a5b, obj5b = run_lbfgs_polish(global_best_a, rem5, rng)
            print("  Final L-BFGS: obj=%.8f" % obj5b)
            if obj5b < global_best_obj:
                global_best_obj = obj5b
                global_best_a = a5b.copy()

    # === Final verification ===
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):

            if ver < global_best_obj:

        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
