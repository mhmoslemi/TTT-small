def solve_lp_step(f, M):
    """
    Solve the LP: max sum(b) subject to b >= 0, (f*b)[k] <= M for all k.
    Returns the optimal b.
    """
    n = len(f)
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: (f*b)[k] <= M for k = 0, ..., 2n-2
    # (f*b)[k] = sum_j f[j] * b[k-j] where j ranges over max(0, k-(n-1)) to min(k, n-1)
    
    # Build constraint matrix A (2n-1 rows, n columns)
    # A[k, j] = f[j] if 0 <= k-j <= n-1 else 0
    # i.e., k-(n-1) <= j <= k and 0 <= j <= n-1
    # i.e., max(0, k-(n-1)) <= j <= min(k, n-1)
    
    n_cons = 2 * n - 1
    # For efficiency, use sparse matrix
    from scipy.sparse import cs_matrix
    rows = []
    cols = []
    vals = []
    for k in range(n_cons):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A = cs_matrix((vals, (rows, cols)), shape=(n_cons, n))
    b_ub = np.full(n_cons, M)
    
    from scipy.optimize import linprog
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if result.success:
        return result.x
    else:
        return np.zeros(n)

    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def compute_max_conv(a):
    """Compute just the max of self-convolution."""
    return fft_convolve_self(a).max()


def solve_lp_active(f, M, active_indices):
    """
    Solve LP: max sum(b) s.t. b >= 0, (f*b)[k] <= M for k in active_indices.
    Returns optimal b or None.
    """
    n = len(f)
    n_act = len(active_indices)
    if n_act == 0:
        return None

    # Build sparse constraint matrix
    rows = np.empty(n_act * n, dtype=np.int32)
    cols = np.empty(n_act * n, dtype=np.int32)
    vals = np.empty(n_act * n, dtype=np.float64)
    total_nz = 0

    for idx, k in enumerate(active_indices):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        count = j_max - j_min + 1
        if count > 0:
            rows[total_nz:total_nz + count] = idx
            cols[total_nz:total_nz + count] = np.arange(j_min, j_max + 1)
            vals[total_nz:total_nz + count] = f[j_min:j_max + 1]
            total_nz += count

    if total_nz == 0:
        return None

    A = csr_array((vals[:total_nz], (rows[:total_nz], cols[:total_nz])),
                  shape=(n_act, n))
    b_ub = np.full(n_act, M)
    c_obj = -np.ones(n)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0, 1e30)] * n, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if result.success and result.x is not None:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None

def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross

        beta = beta_start * (1.0 + 0.005 * it)
        beta = min(beta, 20000.0)
        step = step_scale / beta

        c = fft_convolve_self(y)
        cur_max = c.max()
        obj = 2.0 * n * cur_max / S**2

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = y.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Gradient
        w = lse_weights(c, beta)
        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Adaptive step: reduce if stagnant
        step_eff = step * max(0.1, 1.0 / (1.0 + stagnant * 0.01))

        # FISTA update
        x_new = project_simplex(y - step_eff * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Reset momentum periodically
        if it % 3000 == 0 or stagnant > 5000:
            t_mom = 1.0

        # Noise injection when very stagnant
        if stagnant > 10000 and it % 1000 == 0:
            noise = rng.normal(0, 1e-4 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0

    return best_a, best_obj


def run_active_polish(a_init, max_time, rng):
    """
    Polishing with exact subgradient (top-1 and top-k).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        obj = 2.0 * n * cur_max / S**2
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k for top-k
        if stuck < 50:
            k = 1
        elif stuck < 200:
            k = 3
        elif stuck < 1000:
            k = min(16, 2 + stuck // 50)
        else:
            k = min(64, 16 + stuck // 200)

        if k == 1:
            idx = np.argmax(c)
            w_full = np.zeros(len(c))
            w_full[idx] = 1.0
        else:
            idx = np.argsort(c)[::-1][:k]
            w = np.ones(k) / k
            w_full = np.zeros(len(c))
            w_full[idx] = w

        grad = conv_gradient(y, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.001 * it))
        step = min(step, S / (0.5 * gnorm + 1e-15))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 5000 and it % 500 == 0:
            noise_scale = max(1e-7, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 10000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    return best_a, best_obj


def run_lbfgs_polish(a_init, max_time, rng, n_levels=4):
    """L-BFGS-B polishing with increasing beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    betas = [30.0, 100.0, 300.0, 800.0, 2000.0][:n_levels]

    for beta in betas:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            e_sum = e.sum()
            w = e / e_sum
            obj = cm + np.log(e_sum) / beta
            grad = conv_gradient(x, w)
            return obj, grad

        t0 = time.monotonic()
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 50, 'maxls': 20, 'ftol': 1e-14,
                     'gtol': 1e-10, 'iprint': -1}
        )
        elapsed = time.monotonic() - t0

        a_new = np.clip(result.x, 0, 1000)
        a_new_sum = a_new.sum()
        if a_new_sum > 1e-15:
            a_new = a_new * (S / a_new_sum)
            obj_new = compute_obj(a_new)
            print("    L-BFGS beta=%.0f: obj=%.8f (t=%.1fs)" % (beta, obj_new, elapsed))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Primary method: LP-based active-set iterations.
    Secondary: FISTA + active-set polish + L-BFGS.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Init: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.54 + 0.46 * np.cos(2 * np.pi * t)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: LP-based iterations (45% of time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.45
    print("\n=== Phase 1: LP iterations (n=%d, budget %.0fs) ===" % (n0, p1_budget))
    t_phase = time.monotonic()

    a_lp, obj_lp = run_lp_iterations(global_best_a, p1_budget, rng,
                                     n_min_active=30, n_max_active=400)
    print("  LP result: obj=%.8f (t=%.0fs)" % (obj_lp, time.monotonic() - t_phase))
    if obj_lp < global_best_obj:
        global_best_obj = obj_lp
        global_best_a = a_lp.copy()

    # === Phase 2: FISTA polishing (25% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = remaining * 0.25
        print("\n=== Phase 2: FISTA polish (budget %.0fs) ===" % p2_budget)
        t_phase = time.monotonic()

        a_fista, obj_fista = run_fista(global_best_a, p2_budget * 0.6, rng,
                                       beta_start=20.0, step_scale=1.0)
        print("  FISTA: obj=%.8f (t=%.0fs)" % (obj_fista, time.monotonic() - t_phase))
        if obj_fista < global_best_obj:
            global_best_obj = obj_fista
            global_best_a = a_fista.copy()

        rem = p2_budget - (time.monotonic() - t_phase)
        if rem > 20:
            a_f2, obj_f2 = run_fista(global_best_a, rem, rng,
                                     beta_start=100.0, step_scale=0.5)
            print("  FISTA high-beta: obj=%.8f" % obj_f2)
            if obj_f2 < global_best_obj:
                global_best_obj = obj_f2
                global_best_a = a_f2.copy()

    # === Phase 3: Active-set polish (15% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = remaining * 0.15
        print("\n=== Phase 3: Active-set polish (budget %.0fs) ===" % p3_budget)
        t_phase = time.monotonic()

        a_ap, obj_ap = run_active_polish(global_best_a, p3_budget, rng)
        print("  Active polish: obj=%.8f (t=%.0fs)" % (obj_ap, time.monotonic() - t_phase))
        if obj_ap < global_best_obj:
            global_best_obj = obj_ap
            global_best_a = a_ap.copy()

    # === Phase 4: Second LP round on improved solution (10% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = remaining * 0.10
        print("\n=== Phase 4: LP round 2 (budget %.0fs) ===" % p4_budget)
        t_phase = time.monotonic()

        a_lp2, obj_lp2 = run_lp_iterations(global_best_a, p4_budget, rng,
                                           n_min_active=50, n_max_active=600)
        print("  LP round 2: obj=%.8f (t=%.0fs)" % (obj_lp2, time.monotonic() - t_phase))
        if obj_lp2 < global_best_obj:
            global_best_obj = obj_lp2
            global_best_a = a_lp2.copy()

    # === Phase 5: Final L-BFGS squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 5: Final L-BFGS (budget %.0fs) ===" % remaining)
        t_phase = time.monotonic()
        a_lb, obj_lb = run_lbfgs_polish(global_best_a, min(remaining * 0.8, 120), rng, n_levels=4)
        print("  L-BFGS: obj=%.8f (t=%.0fs)" % (obj_lb, time.monotonic() - t_phase))
        if obj_lb < global_best_obj:
            global_best_obj = obj_lb
            global_best_a = a_lb.copy()

    # === Phase 6: Try alternate n ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 6: Alternate n (budget %.0fs) ===" % remaining)
        # Try n = 4096 with quick FISTA
        for n_alt in [4096, 16384]:
            if remaining < 20:
                break
            t_phase = time.monotonic()
            t_arr = np.linspace(0, 1, n_alt, endpoint=False)
            a_alt = 0.54 + 0.46 * np.cos(2 * np.pi * t_arr)
            # Quick FISTA
            a_alt, obj_alt = run_fista(a_alt, min(remaining * 0.4, 120), rng,
                                       beta_start=30.0, step_scale=1.0)
            print("  n=%d: obj=%.8f (t=%.0fs)" % (n_alt, obj_alt, time.monotonic() - t_phase))
            if obj_alt < global_best_obj:
                global_best_obj = obj_alt
                global_best_a = a_alt.copy()
            remaining = deadline - time.monotonic()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using constant fallback")
            n_fb = 4096
            final_list = [float(np.sqrt(2 * n_fb) / n_fb)] * n_fb
            global_best_obj = 2.0
    else:
        n_fb = 4096
        final_list = [float(np.sqrt(2 * n_fb) / n_fb)] * n_fb

    print("\nBest objective: %.8f" % global_best_obj)
    return final_list
