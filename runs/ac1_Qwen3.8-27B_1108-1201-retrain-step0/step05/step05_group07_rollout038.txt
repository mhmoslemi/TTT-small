"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using adaptive LSE-smoothed gradient descent
with Nesterov momentum, active-set tracking, multi-phase beta annealing,
structured perturbations, and length exploration.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection-free method."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = w > theta_candidates
    if not np.any(valid):
        theta = theta_candidates[-1]
    else:
        rho = int(np.max(np.where(valid)[0])) + 1
        theta = theta_candidates[rho - 1]
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500.0, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(np.conj(W) * A))[:n]
    return 2.0 * grad


def topk_subgradient(a, c, k):
    """Weighted subgradient from top-k convolution lags."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    c_vals = c[top_idx]
    c_min = c_vals.min()
    if c_vals.max() - c_min < 1e-15:
        weights = np.ones(k) / k
    else:
        weights = (c_vals - c_min) / (c_vals - c_min).sum()
    grad = np.zeros(n)
    for ii in range(k):
        ki = int(top_idx[ii])
        wi = weights[ii]
        j_start = max(0, ki - (n - 1))
        j_end = min(n - 1, ki)
        for j in range(j_start, j_end + 1):
            grad[j] += 2.0 * wi * a[ki - j]
    return grad


def topk_subgradient_fast(a, c, k):
    """Vectorized top-k subgradient using FFT cross-correlation."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    c_vals = c[top_idx]
    c_min = c_vals.min()
    if c_vals.max() - c_min < 1e-15:
        weights = np.ones(k) / k
    else:
        weights = (c_vals - c_min) / (c_vals - c_min).sum()
    w = np.zeros(2 * n - 1)
    w[top_idx] = weights
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(np.conj(W) * A))[:n]
    return 2.0 * grad


def compute_objective(a):
    """Compute 2n*max(conv)/(sum)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def optimize_core(a0, S, max_time, rng, beta_init=30.0, beta_max=3000.0,
                  momentum_coef=0.85, verbose=True, use_lse=True,

    """
    Core optimization loop with LSE smoothing, momentum, and adaptive step size.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=S.
    """
    n = len(a0)
    a = project_simplex(a0.astype(np.float64), S)
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()
    momentum = np.zeros(n)

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 200

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Adaptive beta schedule
        if use_lse:
            if progress < 0.3:
                beta = beta_init + 20.0 * progress
            elif progress < 0.6:
                beta = beta_init + 6.0 + 50.0 * (progress - 0.3) / 0.3
            elif progress < 0.85:
                beta = beta_init + 6.0 + 15.0 + 200.0 * (progress - 0.6) / 0.25
            else:
                beta = min(beta_max, beta_init + 21.0 + 200.0 + 1000.0 * (progress - 0.85) / 0.15)
        else:
            k = int(topk_base + topk_grow * progress)

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Compute gradient
        if use_lse:
            grad = lse_gradient(a, c, beta)
        else:
            grad = topk_subgradient_fast(a, c, k)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Gap-based step size with momentum
        gap = cur_max - c.mean()
        if gap < 1e-14:
            break

        step = 2.0 * gap / (gnorm * gnorm + 1e-30)
        step = min(step, 3.0 / (gnorm + 1e-30))

        # Momentum update
        momentum = momentum_coef * momentum + (1.0 - momentum_coef) * grad
        mnorm = np.linalg.norm(momentum)

        # Line search
        accepted = False
        for ls_try in range(8):
            a_try = project_simplex(a - step * momentum, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-12) or ls_try >= 5:
                if c_try.max() < cur_max * (1.0 + 1e-8):
                    a = a_try
                    accepted = True
                    break
            step *= 0.5

        if not accepted:
            # Very small step
            a = project_simplex(a - step * momentum, S)

        # Stuck detection and structured perturbation

            # Structured perturbation: target coordinates that contribute to peak lags
            peak_idx = np.argmax(c)

            # Boost noise in active region

            momentum *= 0.5
            stuck = 0
            patience = int(patience * 1.2)

        if verbose and it % 500 == 0:
            cur_obj = 2.0 * n * best_max / (S * S)
            print(f"      it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, "
                  f"beta={beta:.0f}, stuck={stuck}")

def optimize_multi_topk(a0, S, max_time, rng):

    """

            best_a = a.copy()

        else:

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        gap = cur_max - c.mean()
        step = 2.0 * gap / (gnorm * gnorm + 1e-30)

def generate_init(n, rng, base=None):
    """Generate initial sequence with various strategies."""
    if base is not None and len(base) > 1:
        t_old = np.linspace(0, 1, len(base))
        t_new = np.linspace(0, 1, n)
        a = np.interp(t_new, t_old, base)
        a = np.maximum(a, 0.0)
        if a.sum() > 0.01:
            return a
    # Raised cosine with slight asymmetry
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * (tt - 0.1 * rng.standard_normal()))
    a = np.maximum(a, 0.01)
    return a


def resample_sequence(a, new_n):
    """Resample sequence to new length using interpolation."""
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, new_n)
    a_new = np.interp(t_new, t_old, a)
    return np.maximum(a_new, 0.0)
