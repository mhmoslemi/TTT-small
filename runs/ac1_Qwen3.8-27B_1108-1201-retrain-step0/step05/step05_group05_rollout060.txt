from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_lse_grad(y, b, beta, L):
    """Compute LSE-smoothed subgradient using precomputed FFT size L."""
    n = len(y)
    bmax = b.max()
    scaled = beta * (b - bmax)
    if scaled.min() < -500:
        scaled = scaled + scaled.min()  # shift to prevent underflow (doesn't change softmax)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    W = fft(w, L)
    Y = fft(y, L)
    grad = 2.0 * np.real(ifft(W * np.conj(Y), L)[:n])
    return grad


def compute_topk_grad(y, b, k, L):
    """Compute subgradient weighted on top-k lags."""
    n = len(y)
    k = min(k, len(b))
    top_idx = np.argsort(b)[::-1][:k]
    w = np.zeros(len(b))
    w[top_idx] = 1.0 / k
    W = fft(w, L)
    Y = fft(y, L)
    grad = 2.0 * np.real(ifft(W * np.conj(Y), L)[:n])
    return grad


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = (tc <= w)
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    return result


def normalize_to_sum(a, S):
    """Clip to [0, 1000] and rescale to sum S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Strategy: Adaptive LSE subgradient with Nesterov acceleration,
peak-equalization phases, asymmetry-preserving Fourier mutations,
and multi-scale exploration to exploit sequence asymmetry.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_weighted(y, w, L):
    """Compute 2*(w cross-correlate y) using FFT, first len(y) elements."""
    n = len(y)
    W = fft(w, L)
    Y = fft(y, L)
    return 2.0 * np.real(ifft(W * np.conj(Y), L)[:n])


def lse_weights(b, beta):
    """Compute LSE softmax weights."""
    bmax = b.max()
    scaled = beta * (b - bmax)
    if scaled.min() < -500:
        scaled = scaled - scaled.min()
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(b)) / len(b)
    return e / tot


def topk_weights(b, k):
    """Uniform weights on top-k lags."""
    k = min(k, len(b))
    w = np.zeros(len(b))
    top_idx = np.argsort(b)[::-1][:k]
    w[top_idx] = 1.0 / k
    return w


def excess_weights(b, T, p):
    """Weights proportional to (b_k - T)^p_+."""
    excess = np.maximum(b - T, 0.0)
    if excess.max() < 1e-15:
        return np.ones(len(b)) / len(b), 0.0
    if p == 1.0:
        w = excess
    else:
        w = excess ** p
    tot = w.sum()
    if tot < 1e-300:
        return np.ones(len(b)) / len(b), 0.0
    return w / tot, np.sum(excess ** (p + 1)) / (p + 1)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = (tc <= w)
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip to [0, 1000] and rescale to sum S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

    """
    FISTA with adaptive LSE-smoothed subgradient.
    Returns (best_a, best_max_b).
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()

    b0 = fft_convolve_self(a)
    best = b0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    consecutive_worse = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        b = fft_convolve_self(y)
        if b.max() < best:
            best = b.max()
            best_a = y.copy()
            last_improve = it
            consecutive_worse = 0
        else:
            consecutive_worse += 1

        if it - last_improve > stuck_rounds:
            kick = 0.005 * S / np.sqrt(n)
            noise = rng.normal(0, kick, n)
            y = project_simplex(y + noise, S)
            a = y.copy()
            t_k = 1.0
            last_improve = it
            beta = max(beta_init, beta * 0.7)
            step_scale = max(0.3, step_scale * 0.7)
            continue

        w = lse_weights(b, beta)
        grad = compute_grad_weighted(y, w, L)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = step_scale / (4.0 * beta + 1.0)
        x_new = project_simplex(y - step * grad, S)

        b_check = fft_convolve_self(x_new)
        if b_check.max() > best * 1.5:
            step_scale = max(0.1, step_scale * 0.5)
        elif consecutive_worse > 100:
            step_scale = min(3.0, step_scale * 1.1)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

        if verbose and it % 5000 == 0:
            obj = 2.0 * n * b.max() / (y.sum() ** 2)
            print("      FISTA it=%d, max_b=%.10f, obj=%.6f, beta=%.0f" % (it, b.max(), obj, beta))

    b_final = fft_convolve_self(best_a)
    return best_a, b_final.max()


def peak_equalize_phase(a_init, S, max_time, rng, L, verbose=False):
    """
    Targeted peak equalization: cycle through top-K focused subgradients
    with very small steps for precision.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()

    b0 = fft_convolve_self(a)
    best = b0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    mode = 0
    mode_len = 500

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        if it % mode_len == 0:
            mode = (mode + 1) % 3

        b = fft_convolve_self(y)
        if b.max() < best:
            best = b.max()
            best_a = y.copy()
            last_improve = it

        if it - last_improve > 4000:
            kick = 0.001 * S / np.sqrt(n)
            noise = rng.normal(0, kick, n)
            y = project_simplex(y + noise, S)
            a = y.copy()
            t_k = 1.0
            last_improve = it
            continue

        if mode == 0:
            k = min(3 + int(10 * progress), n // 8)
            w = topk_weights(b, k)
        elif mode == 1:
            T = b.mean() + 0.3 * (b.max() - b.mean())
            w, _ = excess_weights(b, T, 2.0)
        else:
            beta = 50.0 + 200.0 * progress
            w = lse_weights(b, beta)

        grad = compute_grad_weighted(y, w, L)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = S / (n * gnorm + 1e-10) * 0.5
        step = min(step, 2.0)

        x_new = project_simplex(y - step * grad, S)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

        if verbose and it % 5000 == 0:
            obj = 2.0 * n * b.max() / (y.sum() ** 2)
            print("      PEQ it=%d, max_b=%.10f, obj=%.6f, mode=%d" % (it, b.max(), obj, mode))

    b_final = fft_convolve_self(best_a)
    return best_a, b_final.max()


def high_beta_fista(a_init, S, max_time, rng, L, beta_init=500.0,
                    beta_rate=0.0005, beta_max=80000.0, verbose=False):
    """Very high beta FISTA for final precision."""
    return fista_main(a_init, S, max_time, rng, L,
                      beta_init=beta_init, beta_rate=beta_rate,
                      beta_max=beta_max, stuck_rounds=5000, verbose=verbose)


def fourier_mutate(a, rng, scale=0.1):
    """Fourier-domain phase perturbation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.1 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def asymmetric_mutate(a, rng):
    """Create asymmetric mutations."""
    n = len(a)
    S = a.sum()
    tt = np.arange(n, dtype=np.float64) / n
    r = rng.random()

    if r < 0.25:
        # Linear ramp perturbation
        c = rng.uniform(-0.3, 0.3)
        perturbed = a * (1.0 + c * (2.0 * tt - 1.0))
    elif r < 0.5:
        # Sinusoidal asymmetry
        c = rng.uniform(-0.4, 0.4)
        freq = rng.uniform(1.0, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        perturbed = a * (1.0 + c * np.sin(2 * np.pi * freq * tt + phase))
    elif r < 0.75:
        # Mirror one half
        half = n // 2
        perturbed = a.copy()
        scale = rng.uniform(0.3, 0.8)
        perturbed[half:] = a[:half][::-1] * scale
    else:
        # Exponential tilt
        c = rng.uniform(-1.0, 1.0)
        perturbed = a * np.exp(c * (tt - 0.5))

    return project_simplex(np.maximum(perturbed, 0.0), S)


def basin_hop(a_init, S, max_time, rng, L, verbose=False):
    """
    Basin hopping: alternate local FISTA with asymmetric mutations.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()

    b0 = fft_convolve_self(a)
    best = b0.max()
    best_a = a.copy()

    opt_time = max(15.0, max_time / 20.0)
    n_iters = max(1, int((max_time - 30) / (opt_time + 10)))

    population = [(best, a.copy())]

    for it in range(n_iters):
        if time.monotonic() - start > max_time - opt_time - 10:
            break
        progress = it / max(1, n_iters)

        base_scale = 0.3 * np.exp(-1.5 * progress) + 0.03
        scale = base_scale * (1.0 + 0.5 * rng.random())

        r = rng.random()
        if len(population) >= 2 and r < 0.4:
            idx = rng.integers(len(population))
            a = asymmetric_mutate(population[idx][1], rng)
        elif r < 0.7:
            a = fourier_mutate(a, rng, scale)
        else:
            a = asymmetric_mutate(a, rng)

        a_opt, val = fista_main(a, S, opt_time, rng, L,
                                beta_init=20 + 30 * it,
                                beta_rate=0.003,
                                beta_max=5000 + 1000 * min(it, 20))
        if val < best:
            best = val
            best_a = a_opt.copy()
            if verbose:
                obj = 2.0 * n * val / (a_opt.sum() ** 2)
                print("    BH iter %d: new best obj=%.6f" % (it, obj))

        population.append((val, a_opt.copy()))
        if len(population) > 6:
            population.sort(key=lambda x: x[0])
            population = population[:6]

    b_final = fft_convolve_self(best_a)
    return best_a, b_final.max()


def generate_inits(n, rng, base=None):
    """Generate diverse asymmetric initializations."""
    inits = []
    tt = np.arange(n, dtype=np.float64) / n

    if base is not None:
        inits.append(np.clip(base, 0, 1000).astype(np.float64))
        # Asymmetric variants of base
        inits.append(base * (1.0 + 0.3 * (2 * tt - 1)))
        inits.append(base * np.exp(0.5 * (tt - 0.5)))
        half = n // 2
        mirror = base.copy()
        mirror[half:] = base[:half][::-1] * 0.6
        inits.append(mirror)

    # Asymmetric windows
    inits.append(np.maximum(0.5 + 0.3 * np.sin(2 * np.pi * tt + 0.5) + 0.2 * (2 * tt - 1), 0.01))
    inits.append(np.maximum(0.6 + 0.3 * np.cos(4 * np.pi * tt) + 0.2 * (1 - 2 * tt), 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.sin(2 * np.pi * tt) * np.cos(4 * np.pi * tt), 0.01))
    inits.append(np.maximum(0.7 + 0.3 * np.exp(-2.0 * (1.5 * tt - 0.5) ** 2), 0.01))
    inits.append(np.maximum(0.4 + 0.6 * tt + 0.1 * np.sin(6 * np.pi * tt), 0.01))
    inits.append(np.maximum(0.8 + 0.2 * np.cos(2 * np.pi * tt - np.pi / 3) - 0.3 * tt, 0.01))

    # Random asymmetric
    inits.append(np.abs(rng.normal(0.8, 0.4, n)) * (1 + 0.3 * tt) + 0.05)
    inits.append(rng.random(n) * (0.5 + 0.5 * tt) + 0.1)

    return [np.clip(x, 0.01, 1000) for x in inits]


def objective_value(a, n=None):
    """Compute the objective directly."""
    if n is None:
        n = len(a)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Diverse asymmetric init + quick FISTA screening (~200s)
    2. Basin hopping with asymmetric/Fourier mutations (~700s)
    3. Extended FISTA with adaptive beta (~600s)
    4. Peak equalization + high-beta polish (~remaining)
    Also evaluates n=4096 track if time permits.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 15

    print("=== Starting optimization, budget=%.0fs ===" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    # Primary: n=8192
    n = 8192
    S = np.sqrt(2.0 * n)
    L_conv = 1 << (2 * n - 1 - 1).bit_length()  # 16384
    L_grad = 1 << (3 * n - 2 - 1).bit_length()  # 32768

    global_best = np.inf
    global_a = None

    def try_update(a, n_val):
        nonlocal global_best, global_a
        a_c = np.clip(a, 0.0, 1000.0)
        s = a_c.sum()
        if s < 0.01:
            return
        b = fft_convolve_self(a_c)
        obj = 2.0 * n_val * b.max() / (s * s)
        if obj < global_best:
            global_best = obj
            global_a = a_c.copy()
            print("  *** New global best: obj=%.6f (n=%d) ***" % (obj, n_val))

    # Phase 1: Quick screening with diverse asymmetric inits
    print("\n=== Phase 1: Diverse screening (~180s) ===")
    p1_start = time.monotonic()
    p1_budget = min(180.0, (deadline - p1_start) * 0.10)

    inits = generate_inits(n, rng, base=given)
    n_screen = min(10, len(inits))
    per_init_time = p1_budget / n_screen

    candidates = []
    for i in range(n_screen):
        remaining = deadline - time.monotonic() - (budget_s * 0.85)
        if time.monotonic() > deadline - remaining:
            break
        t_alloc = min(per_init_time, deadline - time.monotonic() - budget_s * 0.82)
        if t_alloc < 8:
            break
        a_opt, val = fista_main(inits[i], S, t_alloc, rng, L_grad,
                                beta_init=5.0, beta_rate=0.006, beta_max=2000)
        obj = 2.0 * n * val / (a_opt.sum() ** 2)
        candidates.append((obj, a_opt.copy()))
        print("  Screen %d/%d: obj=%.6f" % (i + 1, n_screen, obj))
        try_update(a_opt, n)

    candidates.sort(key=lambda x: x[0])
    print("  Best after screening: obj=%.6f" % global_best)

    # Phase 2: Basin hopping (exploration)
    print("\n=== Phase 2: Basin hopping (~700s) ===")
    p2_start = time.monotonic()
    p2_budget = min(700.0, (deadline - p2_start) * 0.30)

    bh_starts = min(3, len(candidates))
    for i in range(bh_starts):
        remaining = deadline - time.monotonic() - (budget_s * 0.55)
        if time.monotonic() > deadline - remaining:
            break
        t_alloc = min(p2_budget / (bh_starts - i), deadline - time.monotonic() - budget_s * 0.50)
        if t_alloc < 60:
            break
        print("  BH from candidate %d (obj=%.6f), alloc=%.0fs" % (i, candidates[i][0], t_alloc))
        a_bh, val_bh = basin_hop(candidates[i][1], S, t_alloc, rng, L_grad, verbose=True)
        try_update(a_bh, n)
        print("  BH %d done: obj=%.6f" % (i, 2.0 * n * val_bh / (a_bh.sum() ** 2)))

    # Phase 3: Extended FISTA with adaptive beta (exploitation)
    print("\n=== Phase 3: Extended FISTA (~600s) ===")
    p3_start = time.monotonic()
    p3_budget = min(600.0, (deadline - p3_start) * 0.30)

    if global_a is not None and p3_budget > 60:
        t_alloc = min(p3_budget, deadline - time.monotonic() - 180)
        if t_alloc > 60:
            print("  Running extended FISTA, alloc=%.0fs" % t_alloc)
            a_f, val_f = fista_main(global_a, S, t_alloc, rng, L_grad,
                                     beta_init=50.0, beta_rate=0.002,
                                     beta_max=15000.0, stuck_rounds=3000, verbose=True)
            try_update(a_f, n)
        else:
            print("  Insufficient time, skipping.")

    # Phase 4: Peak equalization + high-beta polish
    print("\n=== Phase 4: Peak equalization + high-beta polish ===")
    p4_start = time.monotonic()
    p4_budget = deadline - p4_start - 10

    if global_a is not None and p4_budget > 40:
        half_budget = p4_budget * 0.5

        # Peak equalization
        t_peq = min(half_budget, p4_budget - 30)
        if t_peq > 30:
            print("  Peak equalization, alloc=%.0fs" % t_peq)
            a_peq, val_peq = peak_equalize_phase(global_a, S, t_peq, rng, L_grad, verbose=True)
            try_update(a_peq, n)

        # High-beta FISTA
        remaining = deadline - time.monotonic() - 5
        if remaining > 20:
            print("  High-beta FISTA, alloc=%.0fs" % remaining)
            a_hb, val_hb = high_beta_fista(global_a, S, remaining, rng, L_grad,
                                            beta_init=1000.0, beta_rate=0.0003,
                                            beta_max=100000.0, verbose=True)
            try_update(a_hb, n)

    # Try n=4096 if time remains
    remaining = deadline - time.monotonic() - 5
    if remaining > 120:
        print("\n=== Bonus: n=4096 track, alloc=%.0fs ===" % remaining)
        n2 = 4096
        S2 = np.sqrt(2.0 * n2)
        L2 = 1 << (3 * n2 - 2 - 1).bit_length()  # 16384

        # Resize or create new
        if given is not None and len(given) >= n2:
            base2 = given[:n2]
        elif global_a is not None:
            # Downsample
            base2 = np.interp(np.linspace(0, len(global_a) - 1, n2),
                               np.arange(len(global_a)), global_a)
        else:
            base2 = np.ones(n2) * 0.5

        inits2 = generate_inits(n2, rng, base=base2)
        for i in range(min(5, len(inits2))):
            t_alloc = (deadline - time.monotonic() - 5) / (5 - i) * 0.8
            if t_alloc < 15:
                break
            a_opt2, val2 = fista_main(inits2[i], S2, t_alloc, rng, L2,
                                       beta_init=10.0, beta_rate=0.004, beta_max=5000)
            try_update(a_opt2, n2)

    # Final verification
    print("\n=== Final verification ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(len(global_a)) * np.sqrt(2.0 * len(global_a)) / len(global_a)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification: obj=%.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback obj:", ver)

    elapsed = time.monotonic() - total_start
    print("\n=== Done in %.0fs. Final obj=%.6f, n=%d ===" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
