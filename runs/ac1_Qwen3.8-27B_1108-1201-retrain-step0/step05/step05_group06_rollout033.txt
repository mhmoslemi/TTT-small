def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def topk_gradient(y, c, k, power=0.0):
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    if power == 0.0:
        w = np.ones(k) / k
    else:
        vals = c[idx]
        w = np.power(np.maximum(vals, 1e-30), power)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def run_lse_fista(a_init, max_time, rng, beta_init=5.0, beta_max=5000.0, beta_growth=1.003):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Adaptive beta: increase when stuck
        if stuck < 200:
            beta = beta_init * (1.0 + 0.01 * it)
        elif stuck < 2000:
            beta = min(beta_max, beta * beta_growth)
        else:
            beta = min(beta_max, beta * 1.01)
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking step size
        step = 1.0 / max(beta * np.max(y) + 1.0, 1.0)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        if stuck > 5000:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it
            
        if it % 10000 == 0:
            t_mom = 1.0
    
    return best_a, best_obj

def run_accel_topk(a_init, max_time, rng, k_start=1, k_max=64, k_growth=1.001):

    n = len(a_init)

    start = time.monotonic()

    best_a = a.copy()
    it = 0
    last_improve = 0
    k = k_start
    eta = 0.1  # step size
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Adaptive k
        target_k = min(k_max, int(k_start * (k_growth ** stuck)))
        if target_k != k:
            k = target_k
        
        grad = topk_gradient(y, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Step size with decrease
        eta_new = eta * 0.9999
        if stuck > 1000:
            eta_new = max(eta_new, 0.01)
        
        x_new = project_simplex(y - eta_new * grad, S)
        
        # Check improvement
        c_check = fft_convolve_self(x_new)
        obj_check = 2.0 * n * c_check.max() / (S * S)
        
        if obj_check < cur_obj:
            y = x_new
            eta = max(eta * 1.05, 0.5)  # increase step if improving
        else:
            eta = eta * 0.7  # decrease step

            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

def run_multiplicative(a_init, max_time, rng, eta_init=0.01):

        grad = topk_gradient(a, c, k)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = normalize_seq(a_new, S)

            eta = 0.01
            k = 1

def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)

def make_smooth_bump(n):
    """A smooth bump centered in the middle."""
    tt = np.arange(n, dtype=np.float64)
    center = n / 2.0
    sigma = n / 4.0
    return np.exp(-0.5 * ((tt - center) / sigma) ** 2)

def make_two_bumps(n):
    """Two bumps at 1/3 and 2/3."""
    tt = np.arange(n, dtype=np.float64)
    s1 = n / 6.0
    s2 = n / 6.0
    return np.exp(-0.5 * ((tt - n/3.0) / s1) ** 2) + np.exp(-0.5 * ((tt - 2*n/3.0) / s2) ** 2)

def make_random_smooth(n, rng):
    """Random smooth sequence via random Fourier coefficients."""
    freqs = np.arange(n)
    coeffs = rng.normal(0, 1, n) + 5.0  # Ensure positive mean
    a = np.zeros(n)
    for f in range(min(50, n//2)):
        a += (coeffs[f] * 0.1 / (f+1)) * np.cos(2 * np.pi * f * freqs / n)
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a

"""

Strategy: Multi-phase optimization combining:

- Accelerated top-k subgradient with adaptive step size

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def l2_gradient(a):
    """Gradient of sum(c^2) where c=conv(a,a). Returns 4*Re[ifft(|A|^2*A)]."""
    n = len(a)
    fsize = 1 << (2 * n - 1 - 1).bit_length()
    A = fft(a, fsize)
    absA2 = np.abs(A) ** 2
    grad = 4.0 * np.real(ifft(absA2 * A))[:n]
    return grad


def topk_gradient(y, c, k, power=0.0):
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    if power == 0.0:
        w = np.ones(k) / k
    else:
        vals = c[idx]
        vmin = vals.min()
        w = np.exp((vals - vmin) * 2.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c):
    n = len(y)
    max_val = c.max()
    active = np.where(c >= max_val - 1e-12)[0]
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_hybrid_fista(a_init, max_time, rng, beta_init=5.0, beta_max=20000.0,
                     alpha_init=0.1, alpha_decay=0.999):
    """FISTA with LSE + L2 hybrid objective and backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    alpha = alpha_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta and alpha
        if stuck < 300:
            beta = beta_init * (1.0 + 0.02 * it)
            alpha = alpha_init * (alpha_decay ** it)
        elif stuck < 3000:
            beta = min(beta_max, beta * 1.005)
            alpha = max(0.001, alpha * 0.99)
        else:
            beta = min(beta_max, beta * 1.01)
            alpha = max(0.0, alpha * 0.95)

        # Compute hybrid gradient
        grad_lse = lse_gradient(y, c, beta)
        if alpha > 0.001:
            grad_l2 = l2_gradient(y) * alpha / (2 * n - 1)
            grad = grad_lse + grad_l2
        else:
            grad = grad_lse

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Step size with backtracking
        step = 0.5 / max(beta * np.max(y) + alpha * 4 * c.max() / (2 * n - 1) + 1.0, 1.0)

        x_new = project_simplex(y - step * grad, S)

        # Verify improvement (backtracking)
        c_new = fft_convolve_self(x_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)
        obj_cur = cur_obj
        back = 0
        while obj_new > obj_cur + 1e-12 and back < 5 and step > 1e-12:
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            obj_new = 2.0 * n * c_new.max() / (S * S)
            back += 1

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise restart when stuck
        if stuck > 8000:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it
            beta = max(beta_init, beta * 0.5)

        if it % 15000 == 0:
            t_mom = 1.0

    return best_a, best_obj

def run_mirror_descent(a_init, max_time, rng):
    """Multiplicative weights with adaptive top-k."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    eta = 0.05
    k = 1

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 200:
            k = min(64, k + 1)
        if stuck > 1000:
            eta = max(eta * 0.97, 0.005)

        grad = topk_gradient(a, c, k, power=0.5)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Multiplicative update
        scale = eta * gnorm / k
        a_new = a * np.exp(-scale * grad / (gnorm + 1e-15))
        a_new = normalize_seq(a_new, S)

        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)

        if obj_new <= cur_obj * (1 + 1e-9):
            a = a_new
            if obj_new < cur_obj - 1e-13:
                eta = min(eta * 1.02, 0.2)
        else:
            eta = max(eta * 0.8, 0.001)

        if stuck > 8000 and it % 100 == 0:
            noise = rng.normal(0, 0.01, n)
            a = normalize_seq(a * np.exp(noise * 0.01), S)
            eta = 0.05
            k = 1
            stuck = 0
            last_improve = it

    return best_a, best_obj


def pool_and_expand(a, pool_factor, expand_factor):
    """Pool to reduce dimensions, then expand back."""
    n = len(a)
    n_small = n // pool_factor
    # Pool: take every pool_factor-th element (or average)
    a_pooled = np.array([a[i * pool_factor:(i + 1) * pool_factor].mean()
                         for i in range(n_small)])
    # Expand back
    a_expanded = np.repeat(a_pooled, pool_factor)[:n]
    return a_pooled, a_expanded


def upsample_linear(a, factor):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)


def make_gaussian_init(n):
    tt = np.arange(n, dtype=np.float64)
    center = n / 2.0
    sigma = n / 5.0
    return np.exp(-0.5 * ((tt - center) / sigma) ** 2)

    inits.append((normalize_seq(np.maximum(two_lobe, 0), S), "two_lobe"))

    # Quick evaluation
    print("\n=== Initialization ===")
    best_init = inits[0][0]
    best_init_obj = np.inf
    for a_init, name in inits:
        obj = compute_obj(a_init)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_init
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a_init.copy()
        print("  %s: obj=%.8f" % (name, obj))
    print("Best init: obj=%.8f" % best_init_obj)

    # === Phase 1: Hybrid LSE+L2 FISTA (main workhorse) ===
    remaining = deadline - time.monotonic()
    p1 = remaining * 0.30
    print("\n=== Phase 1: Hybrid FISTA (%.0fs) ===" % p1)
    t = time.monotonic()
    a1, obj1 = run_hybrid_fista(best_init, p1, rng, beta_init=5.0, beta_max=10000.0)
    obj1 = compute_obj(a1)
    print("  obj=%.8f (%.1fs)" % (obj1, time.monotonic() - t))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    # === Phase 2: Accelerated top-k ===
    remaining = deadline - time.monotonic()
    p2 = remaining * 0.25
    print("\n=== Phase 2: Accel top-k (%.0fs) ===" % p2)
    t = time.monotonic()
    a2, obj2 = run_accel_topk(global_best_a, p2, rng)
    obj2 = compute_obj(a2)
    print("  obj=%.8f (%.1fs)" % (obj2, time.monotonic() - t))
    if obj2 < global_best_obj:
        global_best_obj = obj2
        global_best_a = a2.copy()

    # === Phase 3: Mirror descent ===
    remaining = deadline - time.monotonic()
    p3 = remaining * 0.15
    print("\n=== Phase 3: Mirror descent (%.0fs) ===" % p3)
    t = time.monotonic()
    a3, obj3 = run_mirror_descent(global_best_a, p3, rng)
    obj3 = compute_obj(a3)
    print("  obj=%.8f (%.1fs)" % (obj3, time.monotonic() - t))
    if obj3 < global_best_obj:
        global_best_obj = obj3
        global_best_a = a3.copy()

    # === Phase 4: High-beta LSE polish ===
    remaining = deadline - time.monotonic()
    p4 = remaining * 0.20
    print("\n=== Phase 4: High-beta LSE polish (%.0fs) ===" % p4)
    t = time.monotonic()
    a4, obj4 = run_hybrid_fista(global_best_a, p4, rng,
                                 beta_init=100.0, beta_max=50000.0,
                                 alpha_init=0.01, alpha_decay=0.99)
    obj4 = compute_obj(a4)
    print("  obj=%.8f (%.1fs)" % (obj4, time.monotonic() - t))
    if obj4 < global_best_obj:
        global_best_obj = obj4
        global_best_a = a4.copy()

    # === Phase 5: Multi-scale refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and len(global_best_a) <= 16384:
        # Try upsampled version
        n2 = len(global_best_a) * 2
        p5 = remaining * 0.50
        print("\n=== Phase 5: Upsample to n=%d (%.0fs) ===" % (n2, p5))
        t = time.monotonic()
        a_up = upsample_linear(global_best_a, 2)
        S2 = np.sqrt(2.0 * n2)
        a_up = normalize_seq(a_up, S2)
        a5, obj5 = run_hybrid_fista(a_up, p5 * 0.6, rng,
                                     beta_init=5.0, beta_max=5000.0)
        obj5 = compute_obj(a5)
        print("  Upsampled: obj=%.8f (%.1fs)" % (obj5, time.monotonic() - t))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        rem5 = p5 - (time.monotonic() - t)
        if rem5 > 10:
            a5b, obj5b = run_accel_topk(global_best_a, rem5, rng)
            obj5b = compute_obj(a5b)
            print("  Upsampled accel: obj=%.8f" % obj5b)
            if obj5b < global_best_obj:
                global_best_obj = obj5b
                global_best_a = a5b.copy()

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p6 = remaining * 0.80
        print("\n=== Phase 6: Final polish (%.0fs) ===" % p6)
        t = time.monotonic()
        # Alternate between methods
        n_iter = 3
        for i in range(n_iter):
            rem6 = p6 - (time.monotonic() - t)
            if rem6 < 5:
                break
            seg = rem6 / (n_iter - i)
            if i % 2 == 0:
                a6, _ = run_hybrid_fista(global_best_a, seg, rng,
                                          beta_init=200.0, beta_max=100000.0,
                                          alpha_init=0.0, alpha_decay=1.0)
            else:
                a6, _ = run_accel_topk(global_best_a, seg, rng)
            obj6 = compute_obj(a6)
            if obj6 < global_best_obj:
                global_best_obj = obj6
                global_best_a = a6.copy()
        print("  Final: obj=%.8f (%.1fs)" % (global_best_obj, time.monotonic() - t))

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL RESULT ===")
            print("Objective: %.8f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
