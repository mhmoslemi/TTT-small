import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_value(c, beta):
    """Compute LSE beta * max smoothed value."""
    cm = c.max()
    scaled = beta * (c - cm)
    m = np.max(scaled)
    if m > 500:
        return cm + m / beta
    log_sum = np.log(np.sum(np.exp(scaled)))
    return cm + log_sum / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    m = np.max(scaled)
    if m > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
        beta = beta_eff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), beta
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, beta


def fista_with_backtracking(a_init, S, max_time, rng, beta_init=10.0,
                             beta_rate=0.003, beta_max=10000.0,
                             momentum_restarts=1000):
    """
    FISTA with backtracking line search on LSE objective.
    Returns (best_a, best_max_c).
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / (2.0 * beta_init * S)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = y.copy()
            last_improve = it

        grad, beta_eff = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = min(step * 1.5, 1.0 / (2.0 * beta_eff * S * 0.3))
        x_new = project_simplex(y - step * grad, S)

        # Check improvement using LSE value
        c_new = fft_convolve_self(x_new)
        lse_new = lse_value(c_new, beta_eff)
        lse_curr = lse_value(c, beta_eff)

        # Backtrack if no sufficient decrease
        backtrack_count = 0
        while lse_new > lse_curr - 1e-4 * step * gnorm**2 and backtrack_count < 10:
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            lse_new = lse_value(c_new, beta_eff)
            backtrack_count += 1

        # Nesterov momentum with periodic restarts
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum restart to handle non-strongly-convex regions
        if it % momentum_restarts == 0:
            t_mom = 1.0

        # Stuck detection: perturb
        if it - last_improve > 2000:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_mom = 1.0
            last_improve = it
            step = 1.0 / (2.0 * beta_eff * S)

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def spectral_mutate(a, rng, scale=0.15):
    """Fourier-domain phase mutation with frequency-dependent noise."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + 2.0 * freqs / fsize
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)

    if mutation_type == 0:
        # Anti-palindromic push: reduce center, boost edges
        tt = np.linspace(0, 1, n)
        anti_pal = np.abs(1.0 - 2.0 * tt)  # V-shape: 1 at edges, 0 at center
        alpha = 0.3 + 0.3 * rng.random()
        a_new = (1 - alpha) * a + alpha * S * anti_pal / anti_pal.sum()
        return project_simplex(a_new, S)
    elif mutation_type == 1:
        # Split: boost one half
        half = n // 2
        a_new = a.copy()
        boost = rng.random()
        if boost < 0.5:
            a_new[:half] *= 1.5 + rng.random()
            a_new[half:] *= 0.5 + 0.5 * rng.random()
        else:
            a_new[half:] *= 1.5 + rng.random()
            a_new[:half] *= 0.5 + 0.5 * rng.random()
        return normalize_to_sum(a_new, S)
    elif mutation_type == 2:
        # Random block redistribute
        n_blocks = rng.integers(4, 16)
        block_size = n // n_blocks
        blocks = np.split(a, n_blocks)
        # Shuffle blocks
        perm = rng.permutation(n_blocks)
        a_new = np.concatenate([blocks[i] for i in perm])
        return normalize_to_sum(a_new[:n], S)
    else:
        # Gaussian smooth + re-randomize part
        a_new = a.copy()
        frac = 0.3 + 0.4 * rng.random()
        n_rand = int(n * frac)
        idx = rng.choice(n, n_rand, replace=False)
        a_new[idx] = rng.exponential(S / n, n_rand)
        return project_simplex(a_new, S)


def basin_hopping_v2(a_init, S, max_time, rng):
    """
    Basin hopping with structured mutations and candidate pool.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    opt_time = max(10.0, max_time / 20.0)
    n_iters = max(1, int(max_time / (opt_time + 3)))

    # Candidate pool: (value, array)
    pool = [(best, a.copy())]

    for it in range(n_iters):
        if time.monotonic() - start > max_time - opt_time:
            break
        progress = it / max(1, n_iters)

        # Choose mutation strategy based on progress
        r = rng.random()
        if progress < 0.3:
            # Early: large structural mutations
            if r < 0.5:
                a = structural_mutate(a, rng)
            else:
                a = spectral_mutate(a, rng, scale=0.3 + 0.3 * rng.random())
        elif progress < 0.7:
            # Mid: moderate mutations
            if r < 0.4:
                a = structural_mutate(a, rng)
            elif r < 0.8:
                a = spectral_mutate(a, rng, scale=0.1 + 0.15 * rng.random())
            else:
                # Perturb from pool member
                idx = rng.integers(len(pool))
                a = normalize_to_sum(pool[idx][1] + rng.normal(0, 0.05 * S / np.sqrt(n), n), S)
        else:
            # Late: small perturbations
            if r < 0.3:
                a = spectral_mutate(a, rng, scale=0.03 + 0.04 * rng.random())
            elif r < 0.6:
                a = structural_mutate(a, rng, mutation_type=3)
            else:
                a = normalize_to_sum(a + rng.normal(0, 0.02 * S / np.sqrt(n), n), S)

        # Local optimization
        beta_init = 20.0 + 30.0 * it
        a_opt, val = fista_with_backtracking(a, S, opt_time, rng,
                                              beta_init=beta_init,
                                              beta_rate=0.002,
                                              beta_max=8000.0 + 2000 * min(it, 10))
        obj = 2.0 * n * val / (a_opt.sum() ** 2)
        if val < best:
            best = val
            best_a = a_opt.copy()
            print("      BH iter %d: max_b=%.8f, obj=%.6f" % (it, val, obj))

        # Update pool
        pool.append((val, a_opt.copy()))
        pool.sort(key=lambda x: x[0])
        pool = pool[:6]

    return best_a, best


def peak_targeted_reduction(a_init, S, max_time, rng):
    """
    Targeted peak reduction: identify the peak lag and use a
    subgradient that specifically targets it, combined with
    LSE for smoothness.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    y = a.copy()
    t_mom = 1.0
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = y.copy()
            last_improve = it

        # Combined gradient: LSE + peak-targeting
        beta = 50.0 * (1.0 + 50.0 * progress)
        grad_lse, _ = lse_gradient(y, c, beta)

        # Peak-targeting: subgradient from the top few lags
        top_k = min(8 + int(16 * progress), n // 8)
        top_idx = np.argsort(c)[::-1][:top_k]
        c_mean = c.mean()
        weights = np.maximum(c[top_idx] - c_mean, 0.0)
        wsum = weights.sum()
        if wsum > 1e-15:
            weights = weights / wsum
            w_full = np.zeros(len(c))
            w_full[top_idx] = weights
            fsize = 1 << (len(w_full) + n - 1).bit_length()
            W = fft(w_full, fsize)
            A = fft(y, fsize)
            grad_peak = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        else:
            grad_peak = np.zeros(n)

        # Blend gradients
        alpha = 0.3 + 0.5 * progress
        grad = (1 - alpha) * grad_lse + alpha * grad_peak

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (3.0 * beta * S * 0.5 + gnorm * 0.1)
        step = min(step, 5.0 * S / (n * gnorm + 1e-10))

        x_new = project_simplex(y - step * grad, S)

        # Backtrack if needed
        c_new = fft_convolve_self(x_new)
        backtrack = 0
        while c_new.max() > cmax + 1e-6 and backtrack < 8:
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            backtrack += 1

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 800 == 0:
            t_mom = 1.0

        if it - last_improve > 3000:
            noise = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            a = y.copy()
            t_mom = 1.0
            last_improve = it

    return best_a, best


def generate_inits_v2(n, rng, given_seq=None):
    """Generate diverse initializations with emphasis on anti-palindromic structures."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        inits.append(np.clip(given_seq, 0, 1000).copy())

    # Anti-palindromic: V-shape (large at edges, small at center)
    inits.append(np.abs(1.0 - 2.0 * tt) + 0.05)
    inits.append(np.abs(1.0 - 2.0 * tt) ** 2 + 0.05)
    inits.append(np.abs(1.0 - 2.0 * tt) ** 0.5 + 0.05)

    # Split structures
    split1 = np.ones(n)
    split1[:n // 2] = 2.0
    inits.append(split1)
    split2 = np.ones(n)
    split2[n // 3:] = 2.0
    inits.append(split2)

    # Windows
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)

    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * tt + 0.7) +
                            0.2 * np.cos(10 * np.pi * tt), 0.01))
    inits.append(np.maximum(1.0 - 0.4 * np.cos(2 * np.pi * tt) +
                            0.1 * np.sin(6 * np.pi * tt), 0.01))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(rng.exponential(1.0, n))

    # "Staircase" patterns
    for n_stairs in [4, 8, 16]:
        stair = np.repeat(np.arange(1, n_stairs + 1, dtype=np.float64),
                          n // n_stairs + 1)[:n]
        # Randomize order
        perm = rng.permutation(n_stairs)
        stair = np.repeat(perm.astype(np.float64) + 1, n // n_stairs + 1)[:n]
        inits.append(stair + 0.01)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Diverse structural init + quick FISTA (~200s)
    2. Basin hopping with structured mutations (~600s)
    3. Peak-targeted reduction (~500s)
    4. Final high-beta polish (~remaining)
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 40

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    # Choose n - larger is generally better for the objective
    n = 8192
    S = np.sqrt(2.0 * n)
    global_best = np.inf
    global_a = None

    # Phase 1: Quick screening of diverse inits
    print("\n=== Phase 1: Quick screening ===")
    p1_start = time.monotonic()
    p1_budget = min(200.0, (deadline - p1_start) * 0.12)

    inits = generate_inits_v2(n, rng, given_seq=given)
    n_screen = min(10, len(inits))
    per_init_time = p1_budget / n_screen

    candidates = []
    for i in range(n_screen):
        remaining = deadline - time.monotonic() - 300
        if remaining < 30:
            break
        t_alloc = min(per_init_time, remaining)
        if t_alloc < 8:
            break
        a_opt, val = fista_with_backtracking(inits[i], S, t_alloc, rng,
                                              beta_init=5.0, beta_rate=0.005,
                                              beta_max=3000.0)
        obj = 2.0 * n * val / (a_opt.sum() ** 2)
        candidates.append((obj, a_opt.copy()))
        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
        print("  Screen %d: obj=%.6f" % (i, obj))

    candidates.sort(key=lambda x: x[0])
    print("  Best after screening: obj=%.6f" % global_best)

    # Phase 2: Basin hopping from top candidates
    print("\n=== Phase 2: Basin hopping ===")
    p2_start = time.monotonic()
    p2_budget = min(600.0, (deadline - p2_start) * 0.35)

    bh_starts = min(4, len(candidates))
    for i in range(bh_starts):
        remaining = deadline - time.monotonic() - 400
        if remaining < 60:
            break
        t_alloc = min(p2_budget / (bh_starts - i), remaining)
        t_alloc = max(t_alloc, 50)
        if t_alloc < 30:
            break
        print("  BH from candidate %d (obj=%.6f), alloc=%.0fs" % (i, candidates[i][0], t_alloc))
        a_bh, val_bh = basin_hopping_v2(candidates[i][1], S, t_alloc, rng)
        obj_bh = 2.0 * n * val_bh / (a_bh.sum() ** 2)
        if obj_bh < global_best:
            global_best = obj_bh
            global_a = a_bh.copy()
        print("  BH %d done: obj=%.6f" % (i, obj_bh))

    # Phase 3: Peak-targeted reduction
    print("\n=== Phase 3: Peak-targeted reduction ===")
    p3_start = time.monotonic()
    p3_budget = min(500.0, (deadline - p3_start) * 0.30)

    if global_a is not None and p3_budget > 30:
        n_sub = 2
        sub_budget = p3_budget / n_sub
        for sub in range(n_sub):
            remaining = deadline - time.monotonic() - 200
            if remaining < 30:
                break
            t_alloc = min(sub_budget, remaining)
            if t_alloc < 20:
                break
            print("  Peak-reduction sub-phase %d/%d, t=%.0fs" % (sub + 1, n_sub, t_alloc))
            a_p, val_p = peak_targeted_reduction(global_a, S, t_alloc, rng)
            obj_p = 2.0 * n * val_p / (a_p.sum() ** 2)
            if obj_p < global_best:
                global_best = obj_p
                global_a = a_p.copy()
            print("  After sub-phase %d: obj=%.6f" % (sub + 1, global_best))

    # Phase 4: Final high-beta polish
    print("\n=== Phase 4: Final polish ===")
    p4_start = time.monotonic()
    p4_budget = deadline - p4_start - 15

    if global_a is not None and p4_budget > 20:
        # High-beta FISTA
        seg1 = p4_budget * 0.6
        a_hb, val_hb = fista_with_backtracking(global_a, S, seg1, rng,
                                                beta_init=200.0, beta_rate=0.0005,
                                                beta_max=50000.0)
        obj_hb = 2.0 * n * val_hb / (a_hb.sum() ** 2)
        if obj_hb < global_best:
            global_best = obj_hb
            global_a = a_hb.copy()
        print("  After high-beta FISTA: obj=%.6f" % global_best)

        # Peak-targeted final
        remaining = deadline - time.monotonic() - 10
        if remaining > 15:
            a_pt, val_pt = peak_targeted_reduction(global_a, S, remaining, rng)
            obj_pt = 2.0 * n * val_pt / (a_pt.sum() ** 2)
            if obj_pt < global_best:
                global_best = obj_pt
                global_a = a_pt.copy()
            print("  After peak-targeted final: obj=%.6f" % global_best)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.6f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
