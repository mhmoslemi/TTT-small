"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses multi-scale continuation, frequency-domain GS initialization,
and adaptive LSE-continuation FISTA with periodic L-BFGS-B polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.5):
    """Top-k weighted gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def gs_flat_spectrum(a_init, n_iters, rng, clip_frac=0.1):
    """
    Gerchberg-Saxton style iteration to flatten the power spectrum.
    Works in frequency domain: set |FFT(a)| to constant, inverse FFT, clip negatives.
    """
    n = len(a_init)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) * (np.sqrt(2*n) / n)
        S = np.sqrt(2*n)
    
    # Target: flat spectrum means |A(k)|^2 = constant for all k
    # Sum of |A(k)|^2 = fsize * sum(a^2) (by Parseval, with the normalization of numpy FFT)
    # Actually, for numpy: sum(a^2) = (1/fsize) * sum(|A|^2)
    # So sum(|A|^2) = fsize * sum(a^2)
    # For flat spectrum: |A(k)|^2 = fsize * sum(a^2) / fsize = sum(a^2) for all k
    # So |A(k)| = sqrt(sum(a^2)) for all k
    
    for it in range(n_iters):
        A = fft(a, fsize)
        target_mag = np.sqrt(np.sum(a**2))
        # Set magnitude to target, keep phase
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        # Clip negatives
        neg_mass = a_new[a_new < 0].sum()
        a_new = np.maximum(a_new, 0.0)
        # Renormalize
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * (S / s)
        
        #_blend with previous to avoid oscillation
        a = (1 - clip_frac) * a + clip_frac * a_new
        a = np.maximum(a, 0.0)
        a = a * (S / a.sum())
    
    return a

def run_active_polish(a_init, max_time, rng):
    """Intensive polishing using exact subgradient with adaptive top-k."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        obj_cur = 2.0 * n * cur_max / S**2
        if obj_cur < best_obj - 1e-14:
            best_obj = obj_cur
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 3
        elif stuck < 2000:
            k = min(16, 4 + stuck // 200)
        else:
            k = min(64, 16 + stuck // 500)

        if k == 1:
            idx = int(np.argmax(c))
            w_full = np.zeros(len(c))
            w_full[idx] = 1.0
        else:
            w_full = np.zeros(len(c))
            idx = np.argsort(c)[::-1][:k]
            vals = c[idx]
            w = vals / vals.sum()
            w_full[idx] = w

        grad = conv_gradient(y, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Optimal step for subgradient method: step = alpha / (1 + t)
        step = S / (gnorm * (1.0 + 0.01 * np.sqrt(it)))
        step = min(step, 2.0 * S / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 5000 and it % 500 == 0:
            noise_scale = max(1e-7, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 10000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    return best_a, best_obj


def run_lbfgs_polish(a_init, max_time, rng, n_levels=4):
    """L-BFGS-B polishing with increasing beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    betas = [50.0, 150.0, 400.0, 1000.0][:n_levels]
    
    for beta in betas:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break
        
        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / beta
            # Penalty for sum constraint
            sum_x = x.sum()
            obj += 10.0 * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 20.0 * (sum_x - S)
            return obj, grad
        
        t0 = time.monotonic()
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 200, 'maxls': 50, 'ftol': 1e-14, 'gtol': 1e-10,
                     'iprint': -1}
        )
        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        
        print("    L-BFGS beta=%.0f: obj=%.8f (t=%.1fs, iters=%d)" %
              (beta, obj_new, elapsed, result.nit))
    
    return best_a, best_obj


def upsolve_sequence(a_small, n_target):
    """
    Upsolve: take a small optimal sequence and create a larger one by tiling + interpolation.
    """
    n_small = len(a_small)
    n_target = int(n_target)
    S = np.sqrt(2.0 * n_target)
    
    if n_target == n_small:
        return a_small * (S / a_small.sum())
    
    # Tile and truncate
    n_tiles = (n_target + n_small - 1) // n_small
    a_tiled = np.tile(a_small, n_tiles)[:n_target]
    
    # Also try interpolation via FFT
    size_small = 2 * n_small - 1
    fsize_small = 1 << (size_small - 1).bit_length()
    A_small = fft(a_small, fsize_small)
    
    size_target = 2 * n_target - 1
    fsize_target = 1 << (size_target - 1).bit_length()
    
    # Zero-pad the spectrum (this corresponds to interpolation in time)
    n_pad_right = fsize_target // 2 - fsize_small // 2
    n_pad_left = fsize_target - fsize_small - n_pad_right
    A_large = np.zeros(fsize_target, dtype=complex)
    A_large[:fsize_small//2] = A_small[:fsize_small//2]
    A_large[fsize_target - fsize_small//2:] = A_small[fsize_small//2:]
    
    a_interp = np.real(ifft(A_large))[:n_target]
    a_interp = np.maximum(a_interp, 0.0)
    
    # Blend tiling and interpolation
    s_tiled = a_tiled.sum()
    s_interp = a_interp.sum()
    if s_tiled > 1e-10 and s_interp > 1e-10:
        a_blend = 0.5 * a_tiled / s_tiled + 0.5 * a_interp / s_interp
    else:
        a_blend = a_tiled / max(s_tiled, 1e-10)
    
    a_blend = np.maximum(a_blend, 0.0)
    a_blend = a_blend * (S / a_blend.sum())
    return a_blend

    shifted = shifted - shifted.max()

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses L-BFGS-B with LSE+L2 continuation as primary optimizer,
FISTA with variance-aware gradient for fine-tuning,
and active-set subgradient for final polish.
Explores both n=8192 and n=16384 sequence lengths.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_grad(y, w):
    """Gradient of <w, conv(y,y)> w.r.t. y. Returns 2*cross_conv."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def run_lbfgs_continuation(a_init, max_time, rng, n_target, beta_levels, l2_weight=0.1):
    """
    L-BFGS-B with continuation over beta levels.
    Objective: LSE(conv(a,a), beta) + l2_weight * sum(conv^2)/(2n-1) + penalty*(sum(a)-S)^2
    """
    n = n_target
    S = np.sqrt(2.0 * n)
    if len(a_init) != n:
        # Resample to target length
        a_temp = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000)
        if a_temp.sum() < 1e-15:
            a = np.ones(n) * (S / n)
        else:
            size_small = 2 * len(a_temp) - 1
            fs_small = 1 << (size_small - 1).bit_length()
            A_small = fft(a_temp, fs_small)
            size_target = 2 * n - 1
            fs_target = 1 << (size_target - 1).bit_length()
            A_large = np.zeros(fs_target, dtype=complex)
            half_s = fs_small // 2
            half_t = fs_target // 2
            A_large[:min(half_s, half_t)] = A_small[:min(half_s, half_t)]
            A_large[fs_target - min(half_s, half_t):] = A_small[fs_small - min(half_s, half_s // 2):] if half_s > 0 else 0
            # Simpler: just zero-pad
            A_large = np.zeros(fs_target, dtype=complex)
            A_large[:half_s] = A_small[:half_s]
            A_large[fs_target - half_s:] = A_small[half_s:]
            a = np.real(ifft(A_large))[:n]
            a = np.maximum(a, 0.0)
            a = a * (S / max(a.sum(), 1e-15))
    else:
        a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
        a = a * (S / max(a.sum(), 1e-15))

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    conv_size = 2 * n - 1

    for beta_idx, beta in enumerate(beta_levels):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 3:
            break

        penalty = max(1.0, 50.0 * beta / 10.0)

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            lse_val = cm + np.log(e.sum()) / beta
            l2_val = np.sum(b * b) / conv_size
            sum_x = x.sum()
            obj = lse_val + l2_weight * l2_val + penalty * (sum_x - S) ** 2
            grad = conv_grad(x, w) + 2.0 * l2_weight * conv_grad(x, b) / conv_size
            grad += 2.0 * penalty * (sum_x - S)
            return obj, grad

        t0 = time.monotonic()
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0.0, 1000.0)] * n,
            options={'maxiter': 300, 'maxls': 50, 'ftol': 1e-16, 'gtol': 1e-12,
                     'iprint': -1}
        )
        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

        print("    L-BFGS beta=%g: obj=%.10f (t=%.1fs, it=%d)" %
              (beta, best_obj, elapsed, result.nit))

    return best_a, best_obj

            best_obj = obj_cur
            best_a = y.copy()
            last_improve = it
        stuck = it - last_improve

        if stuck > 8000:
            break

        # Combined gradient: LSE + L2 variance
        w_lse = lse_weights(c, beta)
        grad_lse = conv_grad(y, w_lse)
        grad_l2 = conv_grad(y, c) / conv_size
        grad = grad_lse + l2_weight * grad_l2

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step: backtracking-ish
        step = 1.0 / beta
        if stuck > 2000:
            step *= 0.7
        if stuck > 5000:
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 5000 == 0:
            t_mom = 1.0

        # Periodic noise when very stuck
        if stuck > 4000 and it % 200 == 0:
            noise_scale = max(1e-7, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 15000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    return best_a, best_obj


def run_active_polish(a_init, max_time, rng, n_target):
    """Active-set subgradient polish with adaptive k and variance awareness."""
    n = n_target
    S = np.sqrt(2.0 * n)
    if len(a_init) != n:
        a = np.ones(n) * (S / n)
    else:
        a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
        a = a * (S / max(a.sum(), 1e-15))

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    conv_size = 2 * n - 1
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        obj_cur = 2.0 * n * cur_max / S**2
        if obj_cur < best_obj - 1e-14:
            best_obj = obj_cur
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 10000:
            break

        # Adaptive top-k with variance component
        if stuck < 50:
            k = 1
        elif stuck < 300:
            k = 3
        elif stuck < 1000:
            k = min(8, 2 + stuck // 100)
        else:
            k = min(32, 8 + stuck // 200)

        idx_top = np.argsort(c)[::-1][:k]
        vals = c[idx_top]
        w = vals / max(vals.sum(), 1e-30)
        w_full = np.zeros(conv_size)
        w_full[idx_top] = w

        grad = conv_grad(y, w_full)
        # Add small variance reduction component
        grad += 0.05 * conv_grad(y, c / conv_size)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.005 * np.sqrt(it)))
        step = min(step, 2.0 * S / max(gnorm, 1e-15))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 3000 and it % 400 == 0:
            noise_scale = max(1e-8, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 12000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    return best_a, best_obj


def gs_flat_init(n, rng, n_gs_iters=300, base_type="cos"):
    """Generate a diverse initialization using GS-style spectrum flattening."""
    S = np.sqrt(2.0 * n)
    if base_type == "cos":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    elif base_type == "flat":
        a = np.ones(n)
    elif base_type == "random":
        a = rng.random(n) + 0.05
    else:
        a = np.ones(n) * (S / n)
    a = a * (S / a.sum())

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

    for it in range(n_gs_iters):
        A = fft(a, fsize)
        target_mag = np.sqrt(np.sum(a**2))
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * (S / s)
        blend = min(1.0, 0.1 + 0.01 * it)
        a = (1 - blend) * a + blend * a_new
        a = np.maximum(a, 0.0)
        a = a * (S / a.sum())

    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-length exploration with L-BFGS-B continuation, FISTA variance-aware,
    and active-set polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)

    global_best_obj = np.inf
    global_best_a = None
    global_best_n = 0

    # Load initial sequence if available
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.10f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
        global_best_n = n0
    else:
        n0 = 8192
        a_start = gs_flat_init(n0, rng, 200, "cos")
        global_best_obj = compute_obj(a_start)
        global_best_a = a_start.copy()
        global_best_n = n0
        print("GS init: n=%d, obj=%.10f" % (n0, global_best_obj))

    # === Phase 1: L-BFGS-B continuation on current best (n0) ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        p1_budget = remaining * 0.35
        print("\n=== Phase 1: L-BFGS-B continuation n=%d (budget %.0fs) ===" % (n0, p1_budget))
        t1 = time.monotonic()
        beta_levels = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0, 1280.0, 2560.0]
        a1, obj1 = run_lbfgs_continuation(global_best_a, p1_budget, rng, n0, beta_levels, l2_weight=0.05)
        print("  Phase 1 done: obj=%.10f (t=%.0fs)" % (obj1, time.monotonic() - t1))
        if obj1 < global_best_obj:
            global_best_obj = obj1
            global_best_a = a1.copy()
            global_best_n = n0

    # === Phase 2: FISTA variance-aware on best ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        p2_budget = remaining * 0.20
        print("\n=== Phase 2: FISTA variance n=%d (budget %.0fs) ===" % (global_best_n, p2_budget))
        t2 = time.monotonic()
        a2, obj2 = run_fista_var(global_best_a, p2_budget, rng, global_best_n,
                                 beta_start=50.0, l2_weight=0.08)
        print("  Phase 2 done: obj=%.10f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()
            global_best_n = global_best_n

    # === Phase 3: Try n=16384 (larger sequence) ===
    remaining = deadline - time.monotonic()
    if remaining > 200:
        n_try = 16384
        p3_budget = remaining * 0.25
        print("\n=== Phase 3: Try n=%d (budget %.0fs) ===" % (n_try, p3_budget))
        t3 = time.monotonic()

        # Initialize by upsampling current best
        S_try = np.sqrt(2.0 * n_try)
        if global_best_n > 0 and global_best_n < n_try:
            a_small = global_best_a
            size_s = 2 * global_best_n - 1
            fs_s = 1 << (size_s - 1).bit_length()
            A_s = fft(a_small, fs_s)
            size_t = 2 * n_try - 1
            fs_t = 1 << (size_t - 1).bit_length()
            A_t = np.zeros(fs_t, dtype=complex)
            half_s = fs_s // 2
            half_t = min(half_s, fs_t // 2)
            A_t[:half_t] = A_s[:half_t]
            A_t[fs_t - half_t:] = A_s[fs_s - half_t:]
            a_try = np.real(ifft(A_t))[:n_try]
            a_try = np.maximum(a_try, 0.0)
            a_try = a_try * (S_try / max(a_try.sum(), 1e-15))
        else:
            a_try = gs_flat_init(n_try, rng, 200, "cos")

        obj_try = compute_obj(a_try)
        print("  Init n=%d: obj=%.10f" % (n_try, obj_try))

        # L-BFGS-B continuation on n=16384
        beta_levels_3 = [5.0, 10.0, 30.0, 80.0, 200.0, 500.0, 1000.0]
        a3, obj3 = run_lbfgs_continuation(a_try, p3_budget * 0.7, rng, n_try, beta_levels_3, l2_weight=0.05)
        if obj3 < obj_try:
            obj_try = obj3
            a_try = a3
        print("  After L-BFGS-B: obj=%.10f (t=%.0fs)" % (obj_try, time.monotonic() - t3))

        rem3 = p3_budget - (time.monotonic() - t3)
        if rem3 > 20 and obj_try < global_best_obj + 0.01:
            a3b, obj3b = run_fista_var(a_try, rem3, rng, n_try, beta_start=100.0, l2_weight=0.05)
            if obj3b < global_best_obj:
                global_best_obj = obj3b
                global_best_a = a3b.copy()
                global_best_n = n_try
                print("  IMPROVED with n=%d: obj=%.10f" % (n_try, obj3b))

    # === Phase 4: L-BFGS-B on best (high beta) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p4_budget = remaining * 0.20
        print("\n=== Phase 4: High-beta L-BFGS-B n=%d (budget %.0fs) ===" % (global_best_n, p4_budget))
        t4 = time.monotonic()
        beta_levels_4 = [200.0, 500.0, 1000.0, 2000.0, 4000.0]
        a4, obj4 = run_lbfgs_continuation(global_best_a, p4_budget, rng, global_best_n, beta_levels_4, l2_weight=0.02)
        print("  Phase 4 done: obj=%.10f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Active-set subgradient polish ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p5_budget = remaining * 0.15
        print("\n=== Phase 5: Active-set polish n=%d (budget %.0fs) ===" % (global_best_n, p5_budget))
        t5 = time.monotonic()
        a5, obj5 = run_active_polish(global_best_a, p5_budget, rng, global_best_n)
        print("  Phase 5 done: obj=%.10f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Final squeeze with L-BFGS-B ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 6: Final L-BFGS squeeze (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()
        beta_levels_6 = [500.0, 1500.0, 3000.0]
        a6, obj6 = run_lbfgs_continuation(global_best_a, remaining * 0.85, rng, global_best_n, beta_levels_6, l2_weight=0.01)
        print("  Phase 6 done: obj=%.10f (t=%.0fs)" % (obj6, time.monotonic() - t6))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

    # === Final verification ===
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.10f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.10f (n=%d)" % (global_best_obj, len(final_list)))
    return final_list
