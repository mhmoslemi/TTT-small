    """Find optimal delta to move from j to l, exactly."""
    n = len(a)
    # d_k = change in c_k per unit delta
    d = np.zeros(2*n-1)
    # Reducing a[j] by delta: c_k decreases by delta * a[k-j] for k in [j, j+n-1]
    for k in range(j, min(j+n, 2*n-1)):
        if 0 <= k-j < n:
            d[k] -= a[k-j]
    # Increasing a[l] by delta: c_k increases by delta * a[k-l] for k in [l, l+n-1]
    for k in range(l, min(l+n, 2*n-1)):
        if 0 <= k-l < n:
            d[k] += a[k-l]
    
    # f(delta) = max_k (c[k] + delta * d[k]) for delta in [0, a[j]]
    # Binary search for the delta that minimizes f(delta)
    lo, hi = 0.0, a[j]
    # f is convex piecewise-linear in delta
    # Use ternary search or binary search on the subgradient
    
    # Actually, let's use a more efficient approach:
    # The minimum of max_i (c_i + delta * d_i) occurs where the "active" constraints balance.
    # We can use binary search: f'(delta) = sum of d_k over active k / count of active k (in subgradient sense)
    # More precisely, the subgradient of f at delta is the convex hull of {d_k : c_k + delta*d_k = f(delta)}
    # f is minimized when 0 is in the subgradient, i.e., when there exist active k with d_k >= 0 and d_k' <= 0
    
    # Simple binary search approach:
    def f_delta(delta):
        return np.max(c + delta * d)
    
    # Ternary search (works for convex functions)
    for _ in range(60):  # ~60 iterations gives ~1e-18 precision
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if f_delta(m1) <= f_delta(m2):
            hi = m2
        else:
            lo = m1
    
    delta = (lo + hi) / 2
    return delta, f_delta(delta)

"""
Minimize 2*n*max_conv/sum^2 using exact pair coordinate descent and FISTA.
Key innovation: exact 1D optimization of mass transfer between pairs,
replacing the fixed-step heuristic with precise ternary search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def lse_gradient(y, beta):
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        grad = 2.0 * y.copy()  # fallback
    else:
        w = e / tot
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return c.max(), grad

def run_fista(a_init, max_time, rng, beta_init=50.0, beta_rate=0.005,
              beta_max=50000.0, step_scale=0.5):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = 2*n*fft_convolve_self(a).max()/(S*S)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        cm, grad = lse_gradient(y, beta)
        cur_obj = 2.0 * n * cm / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step_local = step
        x_trial = project_simplex(y - step_local * grad, S)
        obj_trial = compute_obj(x_trial)
        bt = 0
        while obj_trial > cur_obj + 1e-12 and step_local > 1e-14 and bt < 5:
            step_local *= 0.5
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt += 1
        step = step_local * (0.8 if bt > 2 else 1.05)
        step = min(step, 2.0 / max(beta, 1.0))
        x_new = x_trial
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        if it - last_improve > 3000:
            t_mom = 1.0
            noise_scale = max(1e-6, 0.002 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
    
    return best_a, best_obj

def exact_cd(a_init, max_time, rng, K=3, n_cand=20, ternary_iters=35):
    """Exact pair coordinate descent with ternary search for optimal delta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a *= S / a.sum()
    c = fft_convolve_self(a)
    
    best_obj = 2*n*c.max()/S**2
    best_a = a.copy()
    last_improve = 0
    it = 0
    start = time.monotonic()
    csize = 2*n - 1
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2*n*cur_max/S**2
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 30000:
            # Restart with noise
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            c = fft_convolve_self(a)
            last_improve = it
            stuck = 0
            continue
        
        # Find top K lags
        K_eff = min(K + stuck // 5000, min(10, csize))
        top_idx = np.argsort(c)[::-1][:K_eff]
        
        # Compute combined leverage
        total_lev = np.zeros(n)
        for kk in top_idx:
            kk_int = int(kk)
            lo = max(0, kk_int - n + 1)
            hi = min(n - 1, kk_int)
            if hi >= lo:
                total_lev[lo:hi+1] += a[kk_int - hi:kk_int - lo + 1][::-1]
        
        # Candidates
        src_score = total_lev * np.maximum(a, 1e-10)
        tgt_score = -total_lev * (S/n - a + 1e-10)
        
        n_src = min(n_cand, n)
        n_tgt = min(n_cand, n)
        src_cands = np.argsort(src_score)[::-1][:n_src]
        tgt_cands = np.argsort(tgt_score)[::-1][:n_tgt]
        
        best_new_obj = cur_obj + 1e-15
        best_jl_delta = None
        
        for j in src_cands:
            j = int(j)
            if a[j] < 1e-10:
                continue
            for l in tgt_cands:
                l = int(l)
                if j == l:
                    continue
                # Compute d = change in c per unit delta
                d = np.zeros(csize)
                # Reducing a[j]: c[k] -= delta * a[k-j] for k in [j, j+n-1]
                k_lo = j
                k_hi = min(j + n - 1, csize - 1)
                if k_hi >= k_lo:
                    d[k_lo:k_hi+1] -= a[max(0, k_lo-j):min(n, k_hi-j+1)]
                # Increasing a[l]: c[k] += delta * a[k-l] for k in [l, l+n-1]
                k_lo2 = l
                k_hi2 = min(l + n - 1, csize - 1)
                if k_hi2 >= k_lo2:
                    d[k_lo2:k_hi2+1] += a[max(0, k_lo2-l):min(n, k_hi2-l+1)]
                
                # Ternary search for optimal delta in [0, a[j]]
                lo_d = 0.0
                hi_d = a[j]
                for _ in range(ternary_iters):
                    m1 = lo_d + (hi_d - lo_d) / 3.0
                    m2 = hi_d - (hi_d - lo_d) / 3.0
                    # f(m) = max(c + m*d)
                    f1 = (c + m1 * d).max()
                    f2 = (c + m2 * d).max()
                    if f1 <= f2:
                        hi_d = m2
                    else:
                        lo_d = m1
                
                opt_delta = (lo_d + hi_d) / 2.0
                if opt_delta < 1e-14:
                    continue
                opt_max = (c + opt_delta * d).max()
                opt_obj = 2.0 * n * opt_max / (S * S)
                
                if opt_obj < best_new_obj:
                    best_new_obj = opt_obj
                    best_jl_delta = (j, l, opt_delta)
        
        if best_jl_delta is not None:
            j, l, delta = best_jl_delta
            a[j] -= delta
            a[l] += delta
            c = fft_convolve_self(a)
        else:
            # No improvement from systematic search
            if stuck > 500:
                # Random pair with small transfer
                j = rng.integers(n)
                l = rng.integers(n)
                if j != l and a[j] > 1e-8:
                    delta = a[j] * 0.05
                    a[j] -= delta
                    a[l] += delta
                    c = fft_convolve_self(a)
            elif it % 100 == 0:
                # Periodic larger random perturbation
                idx = rng.choice(n, size=min(10, n), replace=False)
                for j in idx:
                    if a[j] > 1e-8:
                        delta = a[j] * 0.02
                        a[j] -= delta
                        a[rng.integers(n)] += delta
                c = fft_convolve_self(a)
    
    return best_a, best_obj

def make_init(n, method, rng):
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-3.0 * (tt - 0.5) ** 2 / (0.1 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 6 * tt + 0.5)
    elif method == "multi_tone":
        a = (0.3 + 0.25*np.sin(2*np.pi*3*tt) + 0.2*np.sin(2*np.pi*7*tt+1.0)
             + 0.15*np.sin(2*np.pi*11*tt+2.0) + 0.1*np.sin(2*np.pi*17*tt+0.7))
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 3.0*np.abs(tt - 0.5))
    elif method == "tapered":
        a = np.sin(np.pi * tt)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())
