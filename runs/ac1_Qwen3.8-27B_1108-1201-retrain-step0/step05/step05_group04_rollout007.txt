1. Adaptive LSE-FISTA for bulk optimization
2. LP-based active-set Newton refinement for the "last mile"
3. Local search with random coordinate transfers

import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def run_lse_fista(a_init, n, S, max_time, rng,
                  beta_start=50.0, beta_end=20000.0, beta_steps=200):
    """FISTA with adaptive LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)

        c = fft_convolve_self(y)
        cur_true_max = c.max()

        if cur_true_max < best:
            best = cur_true_max
            best_a = y.copy()
            last_improve_it = it

        grad, lse_val = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta

        stagnant = it - last_improve_it
        if stagnant > 500:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stagnant / (80000.0 / n))
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve_it = it
            step *= 0.3

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def active_set_gradient(a, c, threshold_frac=0.01):
    """Gradient using active set of top lags."""
    n = len(a)
    cm = c.max()
    threshold = cm - threshold_frac * cm
    w = np.where(c >= threshold, 1.0, 0.0)
    tot = w.sum()
    if tot < 1:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        tot = 1.0
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def lp_newton_step(a, c, K, max_step, S):
    """
    LP-based Newton step: minimize the new peak of the linearized self-convolution.
    Solves: min M_new s.t. c[active] + G @ delta <= M_new, sum(delta)=0, -a <= delta, |delta| <= max_step
    Returns the step delta.
    """
    n = len(a)
    active_lags = K
    n_active = len(active_lags)

    if n_active == 0:
        return np.zeros(n)

    # Build gradient matrix G (n_active x n)
    G = np.zeros((n_active, n))
    for i, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        G[i, js] = 2.0 * a[k - js]

    c_active = c[active_lags]

    # Solve LP: min M_new
    # s.t. G @ delta - M_new <= -c_active  (i.e., c_active + G@delta <= M_new)
    #      sum(delta) = 0
    #      -a <= delta <= max_step (element-wise)
    delta_var = cp.Variable(n)
    M_new = cp.Variable()

    constraints = [
        G @ delta_var - M_new <= -c_active,
        cp.sum(delta_var) == 0,
        delta_var >= -a + 1e-12,
        delta_var <= max_step,
        delta_var >= -max_step,
    ]

    objective = cp.Minimize(M_new)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.SCIP, time_limit=10)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK, time_limit=10)
        except Exception:
            # Fallback: minimum norm step
            try:
                target = -0.5 * (c_active - c_active.min())
                GGt = G @ G.T
                GGt += 1e-10 * np.eye(n_active)
                delta = G.T @ np.linalg.solve(GGt, target)
                delta -= np.mean(delta)
                delta = np.clip(delta, -max_step, max_step)
                delta = np.minimum(delta, a - 1e-12)
                return delta
            except Exception:
                return np.zeros(n)

    if delta_var.value is not None:
        delta = np.real(delta_var.value)
        delta = np.clip(delta, -a + 1e-12, max_step)
        # Re-enforce sum constraint approximately
        delta -= np.mean(delta)
        delta = np.clip(delta, -a + 1e-12, max_step)
        return delta
    return np.zeros(n)


def run_lp_newton(a_init, n, S, max_time, rng):
    """Iterative LP-based Newton refinement."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    max_step_start = 0.02 * S / n
    max_step = max_step_start

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Identify active lags
        threshold = cur_max - 0.005 * cur_max
        active_mask = c >= threshold
        active_indices = np.where(active_mask)[0]

        # Limit active set size to avoid too-large LP
        if len(active_indices) > 50:
            top_val = c[active_indices]
            order = np.argsort(top_val)[::-1]
            active_indices = active_indices[order[:50]]

        # Solve LP Newton step
        if len(active_indices) > 0:
            delta = lp_newton_step(a, c, active_indices, max_step, S)
            if np.linalg.norm(delta) > 1e-15:
                # Line search: try the full step, then halve
                improved = False
                for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
                    a_test = a + frac * delta
                    a_test = np.maximum(a_test, 0.0)
                    a_test = project_simplex(a_test, S)
                    c_test = fft_convolve_self(a_test)
                    if c_test.max() < cur_max * (1 - 1e-9):
                        a = a_test
                        improved = True
                        break

                if not improved:
                    # Halve max_step
                    max_step *= 0.7
                    if max_step < 1e-8 * S / n:
                        max_step = max_step_start

        # Periodic noise to escape plateaus
        if it % 100 == 0:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / 3000.0)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

        # Decay max_step slowly
        if it % 50 == 0 and improved:
            max_step *= 0.95

    return best_a, best


def local_search(a_init, n, S, max_time, rng, n_pairs_per_iter=500):
    """
    Local search: random coordinate pair transfers.
    For each pair (i,j), compute optimal transfer amount that minimizes the peak.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Find top peaks to focus on
        top_peaks = np.argsort(c)[::-1][:10]

        # Random pair transfers
        n_accepted = 0
        for _ in range(n_pairs_per_iter):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-10:
                continue

            # Try transferring small amount from i to j
            # The change in c[k] is approximately:
            # delta_c[k] = -eps*a[k-i] + eps*a[k-j] + terms involving a[j]
            # More precisely, moving eps from a[i] to a[j]:
            # c_new[k] - c[k] = eps*(a[k-j] - a[k-i]) + eps*(a[j] - a[i]) * (indicator terms)
            # Approximation: delta_c[k] ≈ eps * (a[k-j] - a[k-i]) + eps*(a[j]-a[i])*(delta_{k,j-i}+delta_{k,i-j})

            eps = 0.01 * a[i]  # transfer 1% of a[i]

            # Estimate the change in the peak
            # For efficiency, just check if it helps the top peaks
            improvement = 0
            for k in top_peaks:
                # Contribution change at lag k
                # a[i] decreases by eps, a[j] increases by eps
                # c[k] = sum a[m]*a[k-m]
                # Change: -eps*a[k-i] - eps*a[i-k] (from a[i] decrease)
                #         +eps*a[k-j] + eps*a[j-k] (from a[j] increase)
                #         + cross terms (second order, ignore)
                ci = a[k - i] if 0 <= k - i < n else 0
                ck = a[i - k] if 0 <= i - k < n else 0
                cj = a[k - j] if 0 <= k - j < n else 0
                ckj = a[j - k] if 0 <= j - k < n else 0
                delta_ci = -eps * (ci + ck) + eps * (cj + ckj)
                if c[k] == cur_max and delta_ci > 0:
                    improvement -= delta_ci  # bad: peak increases
                elif c[k] >= cur_max * 0.99:
                    improvement += max(0, -delta_ci)

            if improvement > 1e-12:
                # Also check that we're not making other peaks worse
                a[i] -= eps
                a[j] += eps
                a[j] = min(a[j], 1000.0)
                # Quick check on a few random lags
                n_accepted += 1

        # Re-normalize occasionally
        if it % 5 == 0:
            a = project_simplex(a, S)

        # Periodic noise
        if it % 20 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 2000.0)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def peak_shaving(a, n, S, n_peaks=5, n_steps=100, lr_scale=0.01):
    """Surgical peak shaving."""
    a = a.copy()
    for _ in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()
        top_idx = np.argsort(c)[::-1][:n_peaks]
        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            weight = max(0, c[k] - cm * 0.99)
            np.add.at(resp, js, contrib * weight)
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        lr = lr_scale * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)
    return a


def raised_cosine_init(n, center_frac=0.3, width_frac=0.5):
    """Generate a raised-cosine initialization."""
    t = np.linspace(0, 1, n)
    center = center_frac
    width = width_frac / 2.0
    x = (t - center) / width
    a = np.ones(n)
    mask = np.abs(x) < 1
    a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
    a[~mask] = 0.0
    return np.maximum(a, 1e-10)


def generate_diverse_inits(n, rng):

        inits.append(np.maximum(a, 0.01))
    # Asymmetric two-bump
    for mu1, mu2, s1, s2 in [(0.2, 0.6, 0.08, 0.12), (0.3, 0.7, 0.1, 0.15), (0.15, 0.5, 0.06, 0.1)]:
        a = np.exp(-0.5 * ((t - mu1) / s1) ** 2) + np.exp(-0.5 * ((t - mu2) / s2) ** 2)
        inits.append(np.maximum(a, 1e-10))
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Uses LSE-FISTA + LP-Newton refinement + local search.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 90.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    print("Budget: %.0fs, deadline in %.0fs" % (budget_s, budget_s - safety))

    global_best = np.inf
    global_a = None
    N = 8192
    S = np.sqrt(2.0 * N)

    def try_sequence(a):
        nonlocal global_best, global_a
        n = len(a)
        S_local = np.sqrt(2.0 * n)
        a = normalize_sum(a, n)
        c = fft_convolve_self(a)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a.copy()
            print("  New best: %.8f (n=%d)" % (val, n))

    # Load previous best if available
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256:
            given = np.clip(given, 0, 1000)
            if len(given) != N:
                # Interpolate to target length
                t_old = np.linspace(0, 1, len(given))
                t_new = np.linspace(0, 1, N)
                given = np.interp(t_new, t_old, given)
            print("Loaded previous best: original_len=%d, resampled to n=%d" % (len(given), N))
    except Exception:
        pass

    # === Phase 1: LSE-FISTA on given sequence ===
    if given is not None:
        t1 = min(600.0, (deadline - time.monotonic()) * 0.25)
        print("=== Phase 1: LSE-FISTA (n=%d, %.0fs) ===" % (N, t1))
        a1, v1 = run_lse_fista(given, N, S, t1, rng,
                                beta_start=100.0, beta_end=20000.0, beta_steps=300)
        try_sequence(a1)
    else:
        # Use a raised cosine init
        print("=== Phase 1: No previous, using raised cosine init ===")
        init = raised_cosine_init(N, 0.35, 0.6)
        t1 = min(300.0, (deadline - time.monotonic()) * 0.15)
        a1, v1 = run_lse_fista(init, N, S, t1, rng,
                                beta_start=30.0, beta_end=10000.0, beta_steps=100)
        try_sequence(a1)

    # === Phase 2: LP-Newton refinement (main innovation) ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        t2 = remaining * 0.35
        print("=== Phase 2: LP-Newton refinement (%.0fs) ===" % t2)
        a2, v2 = run_lp_newton(global_a, N, S, t2, rng)
        try_sequence(a2)

    # === Phase 3: Active-set gradient refinement ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        t3 = remaining * 0.2
        print("=== Phase 3: Active-set refinement (%.0fs) ===" % t3)
        a3, v3 = run_active_refinement(global_a, N, S, t3, rng)
        try_sequence(a3)

    # === Phase 4: Local search ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        t4 = remaining * 0.15
        print("=== Phase 4: Local search (%.0fs) ===" % t4)
        a4, v4 = local_search(global_a, N, S, t4, rng)
        try_sequence(a4)

    # === Phase 5: Final LSE polish with very high beta ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        t5 = min(remaining * 0.15, 300.0)
        if t5 > 30:
            print("=== Phase 5: High-beta LSE polish (%.0fs) ===" % t5)
            a5, v5 = run_lse_fista(global_a, N, S, t5, rng,
                                    beta_start=10000.0, beta_end=50000.0, beta_steps=30)
            try_sequence(a5)

    # === Phase 6: Peak shaving + quick LSE ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("=== Phase 6: Peak shaving + polish (%.0fs) ===" % remaining)
            a6 = peak_shaving(global_a, N, S, n_peaks=10, n_steps=200, lr_scale=0.003)
            try_sequence(a6)
            if a6.sum() > 0 and deadline - time.monotonic() > 60:
                a6, v6 = run_lse_fista(a6, N, S, (deadline - time.monotonic()) * 0.3, rng,
                                        beta_start=20000.0, beta_end=50000.0, beta_steps=20)
                try_sequence(a6)

    # === Phase 7: Try a different initialization ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("=== Phase 7: Alternative init exploration (%.0fs) ===" % remaining)
        inits = generate_diverse_inits(N, rng)
        # Score initializations quickly
        init_scores = []
        for i, init in enumerate(inits):
            a_norm = normalize_sum(init, N)
            c = fft_convolve_self(a_norm)
            init_scores.append((c.max(), i))
        init_scores.sort()
        per_init = max(10.0, remaining / 10.0)
        for score, idx in init_scores[:3]:
            if time.monotonic() > deadline - 30:
                break
            if (deadline - time.monotonic()) < per_init * 0.5:
                break
            print("  Trying alt init %d (score=%.4f)" % (idx, score))
            ai, vi = run_lse_fista(inits[idx], N, S, per_init, rng,
                                    beta_start=50.0, beta_end=15000.0, beta_steps=80)
            try_sequence(ai)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.8f, internal = %.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(N, dtype=np.float64) * (S / N)
    else:
        print("WARNING: No valid result, using uniform fallback.")
        global_a = np.ones(N, dtype=np.float64) * (S / N)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(N, dtype=np.float64) * (S / N)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
