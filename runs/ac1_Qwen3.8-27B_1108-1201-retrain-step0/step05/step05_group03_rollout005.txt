```python
"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
1. LP-based directional method: solve restricted minimax LPs to find descent directions
   that reduce peak autocorrelation while maintaining feasibility
2. FISTA with adaptive top-k LSE smoothing for gradient-based polishing
3. Spectral perturbation for escaping local minima
4. Multi-length exploration (n=8192 primary, n=4096 secondary)
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src, endpoint=False)
    t_new = np.linspace(0, 1, n_target, endpoint=False)
    return np.interp(t_new, t_old, a)


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def solve_direction_lp(f, M, top_k_indices, K):
    """
    Solve minimax LP to find a direction g that minimizes max_k (f*g)_k
    over the top-K indices, subject to g >= 0, sum(g) = 1.
    
    Returns (g, t_opt) where g is the optimal direction and t_opt is the
    minimum peak of f*g over the constrained indices.
    """
    n = len(f)
    # Build constraint matrix: K x n
    # A[i, j] = f[top_k_indices[i] - j] if 0 <= top_k_indices[i] - j < n, else 0
    A = np.zeros((K, n))
    for i in range(K):
        k = top_k_indices[i]
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[i, j_min:j_max+1] = f[k - j_max:k - j_min + 1][::-1]
    
    # LP: min t
    # s.t. A @ g <= t  (K constraints)
    #      sum(g) = 1  (1 equality)
    #      g >= 0
    #      t free (but we can bound t >= 0)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality: A @ g - t <= 0
    A_ub = np.hstack([A, -np.ones((K, 1))])
    b_ub = np.zeros(K)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            g = res.x[:n]
            t_opt = res.x[-1]
            g = np.maximum(g, 0)
            g_sum = g.sum()
            if g_sum > 1e-15:
                g /= g_sum
            return g, t_opt
    except Exception:
        pass
    return None, M


def run_lp_direction(f_init, S, max_time, rng, K=150, verbose=True):
    """
    LP-based directional method:
    At each step, solve a restricted LP to find a descent direction,
    then perform line search.
    """
    n = len(f_init)
    a = normalize(f_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                elapsed = time.monotonic() - start
                obj = 2.0 * n * best / (S * S)
                print("    LP iter %d: max=%.8f, obj=%.6f, t=%.0fs" % (it, best, obj, elapsed))
        else:
            no_improve += 1
        
        if no_improve > 500:
            # Perturb to escape
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            no_improve = 0
            if verbose:
                print("    LP: perturbed at iter %d" % it)
        
        # Find top-K indices of current convolution
        k_effective = min(K, 2 * n - 1)
        top_k_idx = np.argsort(c)[::-1][:k_effective]
        
        # Solve LP
        g_dir, t_opt = solve_direction_lp(a, cur_max, top_k_idx, k_effective)
        
        if g_dir is None:
            # Fallback: use gradient direction
            beta = 50.0
            grad = topk_lse_gradient(a, c, k_effective, beta)
            g_dir = project_simplex(-grad, S)
            t_opt = cur_max
        
        # Line search: try several step sizes
        best_t = 0.0
        best_new_max = cur_max
        for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1.0 - t_try) * a + t_try * g_dir
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try * (S / s_try)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < best_new_max:
                best_new_max = c_try.max()
                best_t = t_try
        
        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * g_dir
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a = a_new * (S / s_new)
            else:
                a = normalize(a, S)
        else:
            # No improvement found; try smaller steps or perturb
            if it % 100 == 0:
                noise_scale = 0.001 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_fista(a_init, S, max_time, rng,
              beta_init=20.0, beta_rate=0.004, beta_max=10000.0,
              k_frac=0.06, use_momentum=True):
    """FISTA with top-k LSE, backtracking, and momentum."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0

    k = max(4, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / max(2.0 * beta * 0.05, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(6):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-14:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        # Periodic restart when stuck
        if stuck > 800:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

        # Periodic spectral perturbation
        if it % 1500 == 0 and rng.random() < 0.15:
            y = spectral_perturb(y, 0.05, rng)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best

        k_eff = min(K, 2 * n - 1)
        top_k_idx = np.argsort(c)[::-1][:k_eff]
        
        g_dir, t_opt = solve_direction_lp(a, cur_max, top_k_idx, k_eff)
        
        if g_dir is None:
            beta = 30.0
            grad = topk_lse_gradient(a, c, min(k_eff, 100), beta)
            g_dir = project_simplex(-grad, S)
        
        # Adaptive line search
        best_t = 0.0
        best_new_max = cur_max
        for t_try in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6]:
            a_try = (1.0 - t_try) * a + t_try * g_dir
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try * (S / s_try)
            max_try = fft_convolve_self(a_try).max()
            if max_try < best_new_max:
                best_new_max = max_try
                best_t = t_try
        
        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * g_dir
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            a = a_new * (S / max(s_new, 1e-15))
        else:
            if it % 200 == 0:
                a = spectral_perturb(a, 0.03, rng)
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best
```
