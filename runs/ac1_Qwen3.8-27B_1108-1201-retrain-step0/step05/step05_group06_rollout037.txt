def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def spectral_flatten(a, n_iters=1000, alpha=0.5, rng=None):
    """
    Phase-retrieval style spectral flattening with non-negativity.
    Iteratively:
    1. Compute DFT of zero-padded a
    2. Compress magnitude: |A| -> |A|^alpha (alpha < 1 flattens)
    3. Inverse DFT
    4. Project onto non-negative (zero out negatives)
    5. Normalize
    """
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    a = np.array(a, dtype=np.float64).copy()
    S = a.sum()
    
    for it in range(n_iters):
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress magnitude
        # To avoid log(0), add small epsilon
        mag_compressed = np.power(mag + 1e-10, alpha)
        
        # Reconstruct with compressed magnitude
        A_new = mag_compressed * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        
        # Project onto non-negative
        a_new = np.maximum(a_new, 0)
        
        # Normalize
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * (S / s)
        
        a = a_new
    
    return a

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def spectral_flatten(a_init, max_time, rng, alpha_init=0.7, alpha_min=0.15):
    """

    Iteratively compresses the DFT magnitude to flatten the convolution.
    """

        A = fft(a, fsize)

def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power <= 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista_lse(a_init, max_time, rng, beta_init=5.0, beta_growth=1.008, beta_max=50000.0):
    """
    FISTA with LSE smoothing and adaptive parameters.
    Uses backtracking step size and periodic momentum restart.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = float(beta_init)
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-12:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Adaptive beta
        if stuck < 200:
            beta = min(beta_max, beta * (1.0 + 0.002))
        elif stuck < 2000:
            beta = min(beta_max, beta * beta_growth)
        else:
            beta = min(beta_max, beta * 1.05)
        
        # Choose gradient type based on stuck state
        if stuck > 3000 and (it % 5 == 0):
            k = min(64, 4 + stuck // 500)
            grad = topk_gradient(y, c, k, power=2.0)
        elif stuck > 1000:
            grad = exact_subgradient(y, c, tol_frac=0.02)
        else:
            grad = lse_gradient(y, c, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking step size
        step = 1.0 / max(beta, 1.0)
        step = min(step, 0.9 / gnorm * S)
        step = max(step, 1e-14)
        
        x_new = project_simplex(y - step * grad, S)
        
        # Check if step was too aggressive (simple increase check)
        c_check = fft_convolve_self(x_new)
        if c_check.max() > cur_max * 1.5 and stuck < 500:
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Restart momentum and add noise when stuck
        if stuck > 4000:
            t_mom = 1.0
            noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-stuck / 8000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it
        
        if it % 5000 == 0:
            t_mom = 1.0
    
    return best_a, best_conv
