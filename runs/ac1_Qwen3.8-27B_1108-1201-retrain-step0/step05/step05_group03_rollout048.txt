import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_lse_grad(a, c, k, beta):
    """Compute gradient of top-k LSE of self-convolution."""
    n = len(a)
    conv_len = len(c)
    k = min(k, conv_len)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    max_scaled = scaled.max()
    if max_scaled > 400:
        beta = 400.0 / max(1e-30, max_scaled)
        scaled = beta * (c_top - cm)
    e = np.exp(scaled - scaled.max())
    tot = e.sum()
    w = e / max(tot, 1e-300)
    w_full = np.zeros(conv_len)
    w_full[idx] = w
    fsize = 1 << (conv_len + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_lse_value(c, k, beta):
    """Compute top-k LSE value."""
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    max_scaled = scaled.max()
    if max_scaled > 400:
        beta = 400.0 / max(1e-30, max_scaled)
        scaled = beta * (c_top - cm)
    e = np.exp(scaled - scaled.max())
    return cm + np.log(max(e.sum(), 1e-300)) / beta


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    S = a.sum()
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def spectral_flatten(a, alpha, rng):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    S = a.sum()
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-30:
        return a
    ratio = power / max(mean_p, 1e-30)
    new_power = power * (1.0 + alpha * (1.0 / (1.0 + ratio) - 0.5))
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / power)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)

            best_local = val_opt
            best_a_local = a_opt.copy()
            obj = 2.0 * n * best_local / (best_a_local.sum() ** 2)
            print("    Phase %d: obj=%.6f" % (i + 1, obj))
    
    obj_final = 2.0 * n * best_local / (best_a_local.sum() ** 2)
    return best_a_local, obj_final


def propose_candidate(budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase optimization with adaptive LSE, peak shaving, and polishing.
    """
    rng = np.random.default_rng(12345)
    start_total = time.monotonic()
    deadline = start_total + max(0.0, budget_s - 30.0)
    
    print("Starting optimization, budget=%.0fs" % budget_s)
    
    # Get given sequence if available
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d, initial obj=%.6f" % (
                len(given),
                2.0 * len(given) * fft_convolve_self(given).max() / (given.sum()**2)
            ))
    except Exception as e:
        print("Could not load given sequence: %s" % str(e))
    
    global_best = np.inf
    global_a = None
    
    def update_global(a, n):
        nonlocal global_best, global_a
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * n * c.max() / (s * s)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  *** New global best: %.6f (n=%d) ***" % (obj, n))
    
    # === Phase 1: Main optimization on parent sequence (n=8192) ===
    n_main = 8192
    S_main = np.sqrt(2.0 * n_main)
    
    if given is not None and len(given) == n_main:
        a_main = normalize(given.copy(), S_main)
    elif given is not None:
        a_main = normalize(resample_sequence(given, n_main), S_main)
    else:
        a_main = np.ones(n_main) * S_main / n_main
    
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print("\n=== Main optimization: n=%d, time=%.0fs ===" % (n_main, remaining * 0.6))
        
        # Phase A: FISTA with adaptive LSE (40% of main time)
        t_a = remaining * 0.40
        if t_a > 30:
            a_opt, val_opt = run_phase_a(a_main, S_main, t_a, rng)
            obj = 2.0 * n_main * val_opt / (a_opt.sum() ** 2)
            print("  Phase A (FISTA-LSE): obj=%.6f" % obj)
            a_main = a_opt
            update_global(a_opt, n_main)
        
        # Phase B: Peak shaving (30%)
        remaining = deadline - time.monotonic()
        t_b = remaining * 0.35
        if t_b > 20:
            a_opt, val_opt = run_phase_b(a_main, S_main, t_b, rng)
            obj = 2.0 * n_main * val_opt / (a_opt.sum() ** 2)
            print("  Phase B (Peak Shave): obj=%.6f" % obj)
            a_main = a_opt
            update_global(a_opt, n_main)
        
        # Phase C: ISRA multiplicative (20%)
        remaining = deadline - time.monotonic()
        t_c = remaining * 0.30
        if t_c > 15:
            a_opt, val_opt = run_phase_c(a_main, S_main, t_c, rng)
            obj = 2.0 * n_main * val_opt / (a_opt.sum() ** 2)
            print("  Phase C (ISRA): obj=%.6f" % obj)
            a_main = a_opt
            update_global(a_opt, n_main)
        
        # Phase D: Final polish (10%)
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt, val_opt = run_phase_d(a_main, S_main, remaining * 0.5, rng)
            obj = 2.0 * n_main * val_opt / (a_opt.sum() ** 2)
            print("  Phase D (Polish): obj=%.6f" % obj)
            a_main = a_opt
            update_global(a_opt, n_main)
    
    # === Phase 2: Try n=16384 (upsample the best found) ===
    remaining = deadline - time.monotonic()
    if remaining > 200 and global_a is not None:
        n_sec = 16384
        S_sec = np.sqrt(2.0 * n_sec)
        a_sec = resample_sequence(global_a, n_sec)
        a_sec = normalize(a_sec, S_sec)
        print("\n=== Secondary: n=%d, time=%.0fs ===" % (n_sec, remaining * 0.4))
        
        t_sec = remaining * 0.4
        a_opt, val_opt = run_phase_a(a_sec, S_sec, t_sec * 0.5, rng)
        obj = 2.0 * n_sec * val_opt / (a_opt.sum() ** 2)
        print("  Sec Phase A: obj=%.6f" % obj)
        update_global(a_opt, n_sec)
        
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_opt, val_opt = run_phase_b(a_opt, S_sec, remaining * 0.4, rng)
            obj = 2.0 * n_sec * val_opt / (a_opt.sum() ** 2)
            print("  Sec Phase B: obj=%.6f" % obj)
            update_global(a_opt, n_sec)
    
    # === Phase 3: Try n=4096 (downsample, might find different optimum) ===
    remaining = deadline - time.monotonic()
    if remaining > 150 and global_a is not None:
        n_ter = 4096
        S_ter = np.sqrt(2.0 * n_ter)
        a_ter = resample_sequence(global_a, n_ter)
        a_ter = normalize(a_ter, S_ter)
        print("\n=== Tertiary: n=%d, time=%.0fs ===" % (n_ter, remaining * 0.3))
        
        t_ter = remaining * 0.3
        a_opt, val_opt = run_phase_a(a_ter, S_ter, t_ter * 0.5, rng)
        obj = 2.0 * n_ter * val_opt / (a_opt.sum() ** 2)
        print("  Ter Phase A: obj=%.6f" % obj)
        update_global(a_opt, n_ter)
        
        remaining = deadline - time.monotonic()
        if remaining > 30:
            a_opt, val_opt = run_phase_b(a_opt, S_ter, remaining * 0.4, rng)
            obj = 2.0 * n_ter * val_opt / (a_opt.sum() ** 2)
            print("  Ter Phase B: obj=%.6f" % obj)
            update_global(a_opt, n_ter)
    
    # === Phase 4: Multi-restart exploration from best found ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Multi-restart: n=%d, time=%.0fs ===" % (n_best, remaining))
        
        n_restarts = max(2, int(remaining / 40))
        t_per_restart = remaining / n_restarts - 5
        
        for r in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            # Different perturbation for each restart
            perturb_amount = 0.1 + 0.1 * r
            a_restart = spectral_perturb(global_a, perturb_amount, rng)
            
            a_opt, val_opt = run_phase_b(a_restart, S_best, t_per_restart, rng)
            obj = 2.0 * n_best * val_opt / (a_opt.sum() ** 2)
            print("  Restart %d: obj=%.6f" % (r + 1, obj))
            update_global(a_opt, n_best)
    
    # === Final verification and return ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        # Fallback: uniform sequence
        n_fb = 8192
        S_fb = np.sqrt(2.0 * n_fb)
        global_a = np.ones(n_fb) * S_fb / n_fb
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback sequence, objective: %.6f" % ver)
    
    total_time = time.monotonic() - start_total
    print("\nTotal time: %.1fs" % total_time)
    print("Returning sequence of length %d, best obj=%.6f" % (len(final_list), global_best))
    return final_list
