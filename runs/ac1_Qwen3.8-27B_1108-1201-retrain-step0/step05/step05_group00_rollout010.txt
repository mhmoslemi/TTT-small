import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    # Clamp tiny negative values from numerical error
    c = np.maximum(c, 0.0)
    return c


def grad_excess(a, N, L, c, target, p=1.0):
    """
    Gradient of sum_k max(0, c[k]-target)^p w.r.t. a.
    Returns (grad, obj_value).
    """
    n = len(a)
    excess = np.maximum(c - target, 0.0)
    if p == 1.0:
        weights = excess.copy()
        obj = weights.sum()
    else:
        weights = p * excess ** (p - 1) * np.where(excess > 0, 1, 0)
        obj = np.sum(excess ** p) / p
    
    total = weights.sum()
    if total < 1e-30:
        # All below target, use uniform weights (shouldn't happen in practice)
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    
    # grad[m] = 2 * sum_j weights[m+j] * a[j]
    a_pad = np.zeros(L)
    a_pad[:n] = a
    w_pad = np.zeros(L)
    w_pad[:2 * n - 1] = weights
    Af = fft(a_pad)
    Wf = fft(w_pad)
    grad = 2.0 * np.real(ifft(Wf * np.conj(Af)))[:n]
    return grad, obj

def lse_gradient(a, N, L, c, beta):
    """
    Gradient of LSE(c, beta) = (1/beta)*log(sum(exp(beta*c))).
    Returns (grad, lse_value).
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        weights = np.zeros(2 * n - 1)
        weights[int(np.argmax(c))] = 1.0
        lse_val = c_max
    else:
        weights = exp_bc / total
        lse_val = c_max + np.log(total) / beta
    
    a_pad = np.zeros(L)
    a_pad[:n] = a
    w_pad = np.zeros(L)
    w_pad[:2 * n - 1] = weights
    Af = fft(a_pad)
    Wf = fft(w_pad)
    grad = 2.0 * np.real(ifft(Wf * np.conj(Af)))[:n]
    return grad, lse_val


def lbfgs_two_loop(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    if m == 0:
        return -g
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

    c = compute_conv(a, N)
    c_max = c.max()
    best_a = a.copy()
    best_cmax = c_max
    
    if mode == 'excess':
        target = target_margin * S * S / (2 * n - 1)
        g, _ = grad_excess(a, N, L, c, target, p)
    elif mode == 'variance':
        g, _ = grad_variance(a, N, L, c)
    elif mode == 'lse':
        g, _ = lse_gradient(a, N, L, c, beta)
    
    gnorm = np.linalg.norm(g)
    if gnorm < 1e-18:
        return best_a, best_cmax
    
    if step_scale is None:
        step_scale = S / (n * 0.1)
    
    s_hist = []
    y_hist = []
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        direction = lbfgs_two_loop(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        # Line search
        alpha = step_scale
        a_best = a
        c_best = c_max
        c_best_val = c_max
        for ls in range(15):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max_val or ls == 14:
                a_best = candidate
                c_best = cm_ls
                c_best_val = cm_ls
                if cm_ls <= c_max_val:
                    break
            alpha *= 0.5
        
        if c_best < best_cmax:
            best_cmax = c_best
            best_a = a_best.copy()
        
        # Compute new gradient
        c_new = c_best
        if mode == 'excess':
            target = target_margin * S * S / (2 * n - 1)
            g_new, _ = grad_excess(a_best, N, L, c_new, target, p)
        elif mode == 'variance':
            g_new, _ = grad_variance(a_best, N, L, c_new)
        elif mode == 'lse':
            g_new, _ = lse_gradient(a_best, N, L, c_new, beta)
        
        s_k = a_best - a
        y_k = g_new - g
        ss = np.dot(s_k, s_k)
        sy = np.dot(s_k, y_k)
        if ss > 1e-20 and sy > 1e-14 * ss:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[1:]
                y_hist = y_hist[1:]
        
        a = a_best
        g = g_new
        c_max = c_best_val
        
        # Adapt step size
        if it % 100 == 0:
            step_scale *= 0.8
    
    return best_a, best_cmax


def coordinate_descent_polish(a, n, S, N, max_time, rng, n_sweeps=50):
    """
    Coordinate descent: for each coordinate i, find the best value of a[i]
    (given all other a[j]) that reduces the max convolution.
    """
    a = project_simplex(a.copy(), S)
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    start = time.monotonic()
    
    # Precompute: for each i, the contribution to each c[k] from a[i]
    # c[k] = a[i]*a[k-i] + (other terms)
    # If we change a[i] to a[i]+delta, c[k] changes by delta*(a[k-i] + a[i-k]) for affected k
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        improved = False
        # Random order each sweep
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > max_time:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            # Try reducing a[i] and adding to others
            # The c[k] values that involve a[i]: c[k] for k in [i, i+n-1]
            # c[k] = ... + a[i]*a[k-i] + ...
            # If we reduce a[i] by delta, c[k] decreases by delta*a[k-i]
            # We add delta to some other coordinate j
            
            # Simple approach: try moving a fraction of a[i] to a random other coordinate
            # Or: try the "best" transfer target
            
            # For efficiency, just try a few candidates
            # Best target: the coordinate j that most reduces the max when we add to it
            # Adding delta to a[j] increases c[k] by delta*a[k-j] for k in [j, j+n-1]
            
            # Greedy: move a small amount from i to the j that minimizes the resulting max
            # This is O(n) per coordinate, O(n^2) per sweep - too slow for n=8192
            # Instead, just try a random target with a fixed fraction
            
            frac = 0.01 + 0.04 * rng.random()
            delta = frac * old_val
            remaining = S - old_val  # sum of all other coordinates
            
            if remaining < 1e-15:
                continue
            
            # Choose target: try a few random targets and pick the best
            best_j = -1
            best_new_cmax = np.inf
            for _ in range(5):
                j = rng.integers(n)
                if j == i:
                    continue
                a[i] = old_val - delta
                a[j] += delta
                # Quick check: recompute only affected c values
                # This is expensive; for simplicity, just do full recompute every few steps
                a[i] = old_val  # revert
                a[j] -= delta  # revert
            
            # For simplicity and speed, do a full recompute approach:
            # Try one random transfer
            j = rng.integers(n)
            while j == i:
                j = rng.integers(n)
            
            a[i] = old_val - delta
            a[j] += delta
            c_check = compute_conv(a, N)
            cm_check = c_check.max()
            
            if cm_check < best_cmax:
                best_cmax = cm_check
                best_a = a.copy()
                improved = True
            else:
                a[i] = old_val
                a[j] -= delta
        
        if not improved and it > 10:
            # Try smaller fractions
            pass
    
    return best_a, best_cmax


def spectral_restart(a, N, rng, frac=0.3):
    """Perturb by randomizing a fraction of FFT phases."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, n_rand, replace=False)
    new_phase = phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def make_init_sinusoidal(n, rng):
    """Initialize with a sinusoidal pattern."""
    t = np.arange(n)
    freqs = rng.uniform(0.1, 5.0, 3)
    a = 1.0
    for f in freqs:
        a += 0.3 * np.sin(2 * np.pi * f * t / n + rng.random() * 2 * np.pi)
    return np.maximum(a, 0.0)


def make_init_gaussian(n, rng):
    """Gaussian-shaped initialization (centered)."""
    t = np.arange(n) - n / 2
    sigma = n * rng.uniform(0.1, 0.3)
    a = np.exp(-t ** 2 / (2 * sigma ** 2))
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.01)


def make_init_two_bump(n, rng):
    """Two-bump initialization."""
    t = np.arange(n)
    center1 = n * rng.uniform(0.2, 0.4)
    center2 = n * rng.uniform(0.6, 0.8)
    sigma = n * rng.uniform(0.05, 0.15)
    a = np.exp(-(t - center1) ** 2 / (2 * sigma ** 2)) + np.exp(-(t - center2) ** 2 / (2 * sigma ** 2))
    a += 0.02 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.

    Phase 2: Excess-weighted gradient (focus on peaks)
    Phase 3: LSE with high beta (fine peak reduction)
    Phase 4: Coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()
    L = 1 << (3 * n - 2).bit_length()
    
    print(f"n={n}, S={S:.4f}, N={N}, L={L}")
    
    best_a = None
    best_cmax = np.inf
    
    def update_best(a_cand, label=""):
        nonlocal best_a, best_cmax
        a_cand = project_simplex(a_cand, S)
        c = compute_conv(a_cand, N)
        cm = c.max()
        if cm < best_cmax:
            best_cmax = cm
            best_a = a_cand.copy()
            score = 2.0 * n * cm / (S * S)
            print(f"  [{label}] max_c={cm:.6f} score={score:.6f}")
        return cm

    inits = []
