Algorithm: Multi-phase optimization with:

2. Adaptive-beta LSE projected gradient with backtracking line search

Key innovations: backtracking line search, adaptive active-set gradients,

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient_vec(a, c, beta):
    """
    Compute gradient of (2/beta) * LSE_k(c_k) where c = conv(a,a).
    Uses the formula: grad_j = (2/beta) * sum_k w_k * a[k-j] * 1(0<=k-j<n)
    which is (2/beta) * cross_correlation(w, a) evaluated at lags 0..n-1.
    """
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Compute cross-correlation: c[j] = sum_m w[j+m] * a[m] for j=0..n-1
    # c = ifft(fft(w, N) * conj(fft(a, N)))[:n] with N >= 3n-2
    N = 1 << (3 * n - 2).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    corr = np.real(ifft(W * np.conj(A)))[:n]
    return (2.0 / beta) * corr


def active_set_gradient(a, c, top_k=8):
    """
    Gradient using only the top-k lags (exact subgradient for those lags).
    grad_j = (2/k) * sum_{k in top} a[k-j] * 1(0<=k-j<n)
    """
    n = len(a)
    top_k = min(top_k, len(c))
    idx = np.argsort(c)[::-1][:top_k]
    w = np.zeros(len(c))
    w[idx] = 1.0 / top_k
    N = 1 << (3 * n - 2).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    corr = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * corr


def objective_max(a):
    """Compute max of self-convolution."""
    c = fft_convolve_self(a)
    return c.max(), c


def backtracking_step(a, grad, S, n, max_steps=30, alpha=0.5, rho=0.5):
    """
    Backtracking line search for projected gradient.
    Minimize LSE along the projected direction.
    Returns the best point found.
    """
    c0 = fft_convolve_self(a)
    f0 = c0.max()
    step = 1.0 / max(1.0, np.linalg.norm(grad) + 1e-15)
    # Scale step based on sequence magnitude
    step *= S / n * 10

    for _ in range(max_steps):
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        f_new = c_new.max()
        if f_new < f0 - 1e-1 * step * np.dot(grad, a - a_new) / max(S, 1e-10) + 1e-12:
            return a_new, f_new
        step *= rho
    return a, f0

        progress = (time.monotonic() - start) / max_time
        beta = min(beta_max, beta_init * np.exp(0.02 * it))

            best_a = a.copy()

            stuck_count = 0
            last_improve = it
            continue

        # Choose gradient type
        if beta > 5000:
            top_k = min(4 + int(16 * progress), 32)
            grad = active_set_gradient(a, c, top_k=top_k)
        else:
            grad = lse_gradient_vec(a, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14 * S:
            break

        # Backtracking line search
        a, f_val = backtracking_step(a, grad, S, n)

        if f_val < best:
            best = f_val
            best_a = a.copy()
            last_improve = it

    return best_a, best


def run_basin_hopping_v2(a_init, S, max_time, rng):
    """
    Basin hopping with Fourier perturbations and aggressive local optimization.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy(), S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    # Determine how many outer iterations we can afford
    opt_time = max(15.0, min(60.0, max_time / 20.0))
    n_outer = max(2, int((max_time - 30) / (opt_time + 3)))

    population = [(best, a.copy())]

    for outer in range(n_outer):
        if time.monotonic() - start > max_time - opt_time - 5:
            break
        progress = outer / max(1, n_outer)

        # Adaptive perturbation scale
        base_scale = 0.3 * np.exp(-1.5 * progress) + 0.03
        scale = base_scale * (0.5 + rng.random())

        # Choose perturbation type
        r = rng.random()
        if r < 0.4:
            # Fourier phase perturbation
            fsize = 1 << (2 * n - 1).bit_length()
            A = fft(a, fsize)
            mag = np.abs(A)
            phase = np.angle(A)
            freqs = np.arange(fsize, dtype=np.float64)
            weight = 1.0 + 2.0 * freqs / fsize
            phase_noise = rng.normal(0, scale * weight, fsize)
            new_A = mag * np.exp(1j * (phase + phase_noise))
            new_A[0] = A[0]
            a_candidate = project_simplex(np.real(ifft(new_A, fsize))[:n], S)
        elif r < 0.7:
            # Crossover with population member
            if len(population) >= 2:
                idx = rng.integers(len(population))
                other = population[idx][1]
                fsize = 1 << (2 * n - 1).bit_length()
                A1 = fft(a, fsize)
                A2 = fft(other, fsize)
                r2 = rng.random(fsize)
                alpha = 0.3 + 0.4 * r2
                new_mag = alpha * np.abs(A1) + (1 - alpha) * np.abs(A2)
                new_phase = np.where(r2 < 0.5, np.angle(A1), np.angle(A2))
                new_A = new_mag * np.exp(1j * new_phase)
                new_A[0] = A1[0]
                a_candidate = project_simplex(np.real(ifft(new_A, fsize))[:n], S)
            else:
                a_candidate = project_simplex(a + rng.normal(0, scale * S / n, n), S)
        else:
            # Coordinate burst: randomly change a subset of coordinates
            a_candidate = a.copy()
            n_change = max(10, n // 20)
            idx = rng.choice(n, n_change, replace=False)
            a_candidate[idx] = 0
            # Redistribute
            remaining = S - a_candidate.sum()
            a_candidate[idx] += remaining / n_change
            a_candidate = np.maximum(a_candidate, 0)
            a_candidate = project_simplex(a_candidate, S)

        # Local optimization
        beta_init = 50 + 100 * outer
        a_opt, val = run_aggressive_pg(a_candidate, S, opt_time, rng,
                                         beta_init=beta_init,
                                         beta_max=min(100000, 5000 + 2000 * outer))
        if val < best:
            best = val
            best_a = a_opt.copy()
            print("    BH %d: new best max_b=%.8f" % (outer, val))

        population.append((val, a_opt.copy()))
        if len(population) > 10:
            population.sort(key=lambda x: x[0])
            population = population[:10]

    return best_a, best


def run_coordinate_descent(a_init, S, max_time, rng):
    """
    Cycle through coordinates, trying small perturbations.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy(), S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    pass_count = 0
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Try random subset of coordinates
        n_try = min(n, 200 + int(800 * (1 - (time.monotonic() - start) / max_time)))
        idx = rng.choice(n, n_try, replace=False)

        improved = False
        for j in idx:
            if time.monotonic() - start > max_time:
                break
            # Try moving mass from j to a random other coordinate
            # or try a small random perturbation
            delta = rng.choice([-1, 1]) * rng.uniform(0.001, 0.05) * S / n
            a_trial = a.copy()
            a_trial[j] += delta
            # Find a coordinate to receive the mass
            if delta < 0:
                # Give mass to a random other coordinate
                m = rng.integers(n)
                if m != j:
                    a_trial[m] -= delta
            a_trial = project_simplex(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < best - 1e-15:
                a = a_trial
                best = c_trial.max()
                best_a = a.copy()
                improved = True
                last_improve = it

        if not improved:
            pass_count += 1
            if pass_count > 50:
                break
        else:
            pass_count = 0

    return best_a, best


def run_fourier_refine(a_init, S, max_time, rng):
    """
    Refine in Fourier domain: optimize the phase/magnitude of the DFT
    to minimize peak autocorrelation.
    """
    n = len(a_init)
    a = project_simplex(a_init.copy(), S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)

    lr = 0.1
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        lr_cur = lr * (1.0 - 0.5 * progress)

        # Reconstruct current a from Fourier
        A_cur = mag * np.exp(1j * phase)
        A_cur[0] = fft(a, fsize)[0]  # Keep DC fixed
        a_cur = project_simplex(np.real(ifft(A_cur, fsize))[:n], S)

        c = fft_convolve_self(a_cur)
        if c.max() < best:
            best = c.max()
            best_a = a_cur.copy()
            last_improve = it

        if it - last_improve > 2000:
            break

        # Gradient in Fourier domain (approximate)
        beta = 100.0 * (1 + 10 * progress)
        grad_real = lse_gradient_vec(a_cur, c, beta)
        # Project gradient to Fourier domain
        G = fft(grad_real, fsize)
        # Update phase (gradient descent on phase)
        phase -= lr_cur * np.real(ifft(G * np.exp(-1j * phase), fsize))[:fsize]
        # Actually, for phase optimization: d/dphase = imaginary part
        # Simplified: just use real-space projected gradient
        a_cur = project_simplex(a_cur - lr_cur * grad_real * S / n, S)
        A = fft(a_cur, fsize)
        mag = np.abs(A)
        phase = np.angle(A)

    return best_a, best


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        g = np.clip(given_seq, 0, 1000)
        s = g.sum()
        if s > 0:
            inits.append(g)

    # Various window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.01)
    inits.append(np.exp(-8.0 * (tt - 0.5) ** 2) + 0.01)

    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * tt + 0.5) + 0.2 * np.cos(8 * np.pi * tt), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.sin(6 * np.pi * tt + 0.3) + 0.1 * np.cos(12 * np.pi * tt), 0.01))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.normal(0.5, 0.3, n)) + 0.05)

    # Block structures
    for nb in [4, 8, 16]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.3, 1.5, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(np.maximum(block[:n], 0.01))

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f, obj=%.6f" % (
                len(given), given.sum(),
                2.0 * len(given) * fft_convolve_self(given).max() / given.sum()**2))
    except Exception:
        pass

    n = 8192
    S = np.sqrt(2.0 * n)

    global_best = np.inf
    global_a = None

    def update_global(a, val):
        nonlocal global_best, global_a
        obj = 2.0 * n * val / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = project_simplex(a.copy(), S)
            print("  *** New global best: obj=%.8f" % global_best)
            return True
        return False

    # Phase 1: Quick screening of initializations
    print("\n=== Phase 1: Screening (~100s) ===")
    p1_start = time.monotonic()
    p1_budget = min(100.0, (deadline - p1_start) * 0.08)
    inits = generate_inits(n, rng, given_seq=given)
    n_screen = min(6, len(inits))
    per_init_time = p1_budget / n_screen

    candidates = []
    for i in range(n_screen):
        if time.monotonic() > deadline - (deadline - total_start) * 0.88:
            break
        t_alloc = min(per_init_time, deadline - time.monotonic() - 120)
        if t_alloc < 5:
            break
        a_opt, val = run_aggressive_pg(inits[i], S, t_alloc, rng,
                                         beta_init=10.0, beta_max=5000.0)
        obj = 2.0 * n * val / (a_opt.sum() ** 2)
        candidates.append((obj, a_opt.copy()))
        print("  Screen %d: obj=%.6f" % (i, obj))
        update_global(a_opt, val)

    candidates.sort(key=lambda x: x[0])
    print("  Best after screening: obj=%.6f" % global_best)

    # Phase 2: Intensive local optimization from best candidates
    print("\n=== Phase 2: Intensive optimization (~400s) ===")
    p2_start = time.monotonic()
    p2_budget = min(400.0, (deadline - p2_start) * 0.25)

    n_refine = min(3, len(candidates))
    for i in range(n_refine):
        if time.monotonic() > deadline - 500:
            break
        t_alloc = min(p2_budget / (n_refine - i), deadline - time.monotonic() - 400)
        if t_alloc < 30:
            break
        print("  Refine candidate %d (obj=%.6f), t=%.0fs" % (i, candidates[i][0], t_alloc))
        a_opt, val = run_aggressive_pg(candidates[i][1], S, t_alloc, rng,
                                         beta_init=100.0, beta_max=50000.0)
        update_global(a_opt, val)

    # Phase 3: Basin hopping
    print("\n=== Phase 3: Basin hopping (~700s) ===")
    p3_start = time.monotonic()
    p3_budget = min(700.0, (deadline - p3_start) * 0.45)

    if global_a is not None and p3_budget > 60:
        n_bh = 2
        for bh in range(n_bh):
            if time.monotonic() > deadline - 200:
                break
            t_alloc = min(p3_budget / (n_bh - bh), deadline - time.monotonic() - 150)
            if t_alloc < 60:
                break
            # Perturb the global best before BH
            pert_scale = 0.1 * S / np.sqrt(n)
            a_start = project_simplex(global_a + rng.normal(0, pert_scale, n), S)
            print("  BH %d/%d, t=%.0fs" % (bh + 1, n_bh, t_alloc))
            a_bh, val_bh = run_basin_hopping_v2(a_start, S, t_alloc, rng)
            update_global(a_bh, val_bh)

    # Phase 4: Fourier refinement
    print("\n=== Phase 4: Fourier refinement (~200s) ===")
    p4_start = time.monotonic()
    p4_budget = min(200.0, deadline - p4_start - 100)

    if global_a is not None and p4_budget > 30:
        a_fr, val_fr = run_fourier_refine(global_a, S, p4_budget, rng)
        update_global(a_fr, val_fr)

    # Phase 5: Coordinate descent polish
    print("\n=== Phase 5: Coordinate descent polish ===")
    p5_start = time.monotonic()
    p5_budget = deadline - p5_start - 10

    if global_a is not None and p5_budget > 20:
        # Run multiple short CD rounds with different random seeds
        n_cd = max(1, int(p5_budget / 30))
        for cd in range(min(n_cd, 5)):
            t_alloc = min(30.0, p5_budget / (n_cd - cd))
            if t_alloc < 5:
                break
            a_cd, val_cd = run_coordinate_descent(global_a, S, t_alloc, rng)
            update_global(a_cd, val_cd)

    # Phase 6: Final high-beta FISTA-like polish
    print("\n=== Phase 6: Final polish ===")
    p6_start = time.monotonic()
    p6_budget = deadline - p6_start - 5

    if global_a is not None and p6_budget > 10:
        a_final, val_final = run_aggressive_pg(global_a, S, p6_budget, rng,
                                                 beta_init=5000.0, beta_max=100000.0)
        update_global(a_final, val_final)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.8f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
