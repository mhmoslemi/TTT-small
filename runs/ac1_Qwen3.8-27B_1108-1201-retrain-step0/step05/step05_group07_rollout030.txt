def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()

        return np.zeros(n)

    valid = tc <= w

        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho

    scaled -= scaled.max()
    x_new = x * np.exp(scaled)
    s = x_new.sum()
    if s < 1e-300:
        return x.copy()
    return x_new / s

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses mirror descent (exponential/multiplicative updates) with adaptive top-k
subgradient, anti-concentration pair moves, LSE smoothing, and multi-phase
strategy. Key differences from pure additive subgradient: natural simplex
preservation, faster convergence via entropic regularization, and targeted
pair unbalancing at peak lags.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient_topk(x, c, k, temperature):
    """Weighted subgradient from top-k convolution lags with softmax weighting."""
    n = len(x)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    top_c = c[top_idx]
    scaled = (top_c - top_c.max()) / max(temperature, 1e-15)
    e = np.exp(scaled)
    weights = e / e.sum()
    grad = np.zeros(n)
    for ii in range(k):
        kidx = int(top_idx[ii])
        wi = weights[ii]
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * x[indices]
    return grad


def compute_lse_gradient(x, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(x)
    cm = c.max()
    diff = c - cm
    scaled = beta * diff
    smax = scaled.max()
    if smax > 400:
        beta = 400.0 / max(1e-15, smax)
        scaled = beta * diff
    e = np.exp(scaled - scaled.max())
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    X = fft(x, fsize)
    return 2.0 * np.real(ifft(W * np.conj(X)))[:n]


def mirror_step(x, grad, eta):
    """Exponential mirror descent step: x_i <- x_i * exp(-eta*g_i), renormalize to sum 1."""
    scaled = -eta * grad
    scaled -= scaled.max()
    x_new = x * np.exp(scaled)
    s = x_new.sum()
    if s < 1e-300:
        return x.copy()
    return x_new / s


def anti_concentration_move(x, c, n, rng, strength=0.05, max_lags=5):
    """
    For top peak lags, unbalance pairs to reduce the convolution value.
    Pairs with similar values contribute most to the product; making them
    more unequal reduces the product (quadratic improvement).
    """
    x_new = x.copy()
    top_lags = np.argsort(c)[::-1][:min(max_lags, 2 * n - 1)]
    total_change = 0.0
    for kidx_int in top_lags:
        kidx = int(kidx_int)
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_end - j_start + 1 < 2:
            continue
        indices = np.arange(j_start, j_end + 1)
        partners = kidx - indices
        # Only consider pairs where both indices are valid and distinct
        valid_mask = (indices != partners) & (indices < n) & (partners >= 0) & (partners < n)
        if not np.any(valid_mask):
            continue
        idx_v = indices[valid_mask]
        par_v = partners[valid_mask]
        # Avoid double counting: only process i < partner
        unique_mask = idx_v < par_v
        if not np.any(unique_mask):
            continue
        i_arr = idx_v[unique_mask]
        j_arr = par_v[unique_mask]
        # Compute imbalance
        xi = x_new[i_arr]
        xj = x_new[j_arr]
        # Target pairs that are relatively balanced (high product for given sum)
        min_vals = np.minimum(xi, xj)
        max_vals = np.maximum(xi, xj)
        sums = xi + xj
        # Product/sum^2 is maximized at 0.25 (equal split)
        balance = 4.0 * min_vals * max_vals / np.maximum(sums * sums, 1e-30)
        # Only modify pairs that are fairly balanced (balance > 0.4 means close to equal)
        target = balance > 0.5
        if not np.any(target):
            continue
        ti = i_arr[target]
        tj = j_arr[target]
        # Transfer mass to increase imbalance
        for a_i in ti[:200]:  # limit work per lag
            a_j = tj[np.where(ti == a_i)[0][0]]
            xa = x_new[a_i]
            xb = x_new[a_j]
            if xa < 1e-15 or xb < 1e-15:
                continue
            smaller = min(xa, xb)
            larger = max(xa, xb)
            delta = strength * smaller * 0.5
            if xa <= xb:
                x_new[a_i] = max(0, xa - delta)
                x_new[a_j] = xb + delta
            else:
                x_new[a_j] = max(0, xb - delta)
                x_new[a_i] = xa + delta
            total_change += delta
    if total_change > 1e-15:
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s > 1e-30:
            x_new = x_new / s
    return x_new


def optimize_mirror(x0, max_time, rng, verbose=True):
    """
    Mirror descent with exponential updates for min max(conv(x,x)), sum(x)=1.
    """
    n = len(x0)
    x = x0 / max(x0.sum(), 1e-30)
    c = fft_convolve_self(x)
    best_max = c.max()
    best_x = x.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 600

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(x)
        cur_max = c.max()
        cur_mean = c[1:2*n-2].mean() if n > 1 else c[0]
        gap = cur_max - cur_mean

        if cur_max < best_max - 1e-13:
            best_max = cur_max
            best_x = x.copy()
            stuck = 0
        else:
            stuck += 1

        progress = min(1.0, (time.monotonic() - start) / max_time)
        if gap < 1e-15:
            break

        # Adaptive k: start focused, broaden over time
        k = int(4 + 120 * progress ** 0.4)
        k = min(k, 2 * n - 1)

        # Temperature: controls spread of softmax weights
        # Lower temp = more focus on top; higher = smoother
        temp = gap * n * (0.3 + 0.7 * (1.0 - progress))

        grad = compute_subgradient_topk(x, c, k, temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Step size for mirror descent
        eta = 1.5 * gap / (gnorm * gnorm)
        eta = min(eta, 1.0 / max(gnorm, 1e-16))

        # Line search with backtracking
        accepted = False
        eta_try = eta
        for _ls in range(10):
            x_try = mirror_step(x, grad, eta_try)
            c_try = fft_convolve_self(x_try)
            if c_try.max() <= cur_max * (1.0 + 1e-11):
                x = x_try
                accepted = True
                break
            eta_try *= 0.45

        if not accepted:
            # Fallback: tiny additive step
            step = 0.0005 / gnorm
            x_try = project_simplex(x - step * grad, 1.0)
            x = x_try

        # Periodic anti-concentration
        if it % 300 == 0:
            c = fft_convolve_self(x)
            x = anti_concentration_move(x, c, n, rng, strength=0.03, max_lags=3)

        # Anti-concentration with more lags every 1000 iters

            c = fft_convolve_self(x)
            x = anti_concentration_move(x, c, n, rng, strength=0.02, max_lags=8)

        # Noise when stuck
        if stuck > patience:
            noise_scale = min(0.04, 0.004 * (stuck / patience)) / np.sqrt(n)
            x = x + rng.normal(0, noise_scale, n)
            x = np.maximum(x, 0)
            x = x / x.sum()
            stuck = 0
            patience = int(patience * 1.25)

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - start
            obj = 2.0 * n * best_max
            print(f"    it={it}, t={elapsed:.0f}s, obj={obj:.6f}, max={best_max:.8f}, gap={gap:.8f}")

    return best_x, best_max

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - start
            print(f"    it={it}, t={elapsed:.0f}s, obj={2.0*n*best:.6f}")

    return best_x, best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: mirror descent, additive subgradient, LSE polish,
    anti-concentration moves, and multi-length exploration.
    Works with sum(x)=1 normalization (objective is scale-invariant).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 55.0)

    print(f"Starting optimization, budget={budget_s}s, deadline in {budget_s-55:.0f}s")

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 256:
            print(f"Using provided sequence: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_global(a, msg=""):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  ** NEW BEST: obj={obj:.6f} ({msg})")

    # === Phase 1: Mirror descent on provided sequence ===
    if a_provided is not None:
        n1 = len(a_provided)
        x1 = a_provided / a_provided.sum()  # normalize to sum=1

        t1 = min(700, (deadline - time.monotonic()) * 0.30)
        print(f"\nPhase 1a: Mirror descent, n={n1}, time={t1:.0f}s")
        x_opt, _ = optimize_mirror(x1, t1, rng, verbose=True)
        update_global(x_opt, "Phase 1a mirror")

        # Phase 1b: Additive subgradient continuation
        remaining = deadline - time.monotonic()
        t1b = min(350, remaining * 0.20)
        if t1b > 30:
            print(f"Phase 1b: Additive subgrad, time={t1b:.0f}s")
            x_opt2, _ = optimize_additive(x_opt, t1b, rng)
            update_global(x_opt2, "Phase 1b additive")
        else:
            x_opt2 = x_opt

        # Phase 1c: LSE polish
        remaining = deadline - time.monotonic()
        t1c = min(300, remaining * 0.18)
        if t1c > 25:
            print(f"Phase 1c: LSE polish, time={t1c:.0f}s")
            x_pol, _ = optimize_lse(x_opt2, t1c, rng, beta_init=300.0, beta_max=40000.0)
            update_global(x_pol, "Phase 1c LSE")

    # === Phase 2: Try larger length if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 200 and a_provided is not None and len(a_provided) == 8192:
        n2 = 16384
        S2 = 1.0  # normalized
        t2 = min(400, remaining * 0.25)
        if t2 > 80:
            print(f"\nPhase 2: Length n={n2}, time={t2:.0f}s")
            # Upsample the current best
            if global_a is not None:
                t_old = np.linspace(0, 1, len(global_a))
                t_new = np.linspace(0, 1, n2)
                x2 = np.interp(t_new, t_old, global_a)
                x2 = np.maximum(x2, 0)
                x2 = x2 / x2.sum()
            else:
                x2 = np.ones(n2) / n2

            x_opt2, _ = optimize_mirror(x2, t2 * 0.6, rng, verbose=False)
            update_global(x_opt2, f"Phase 2 n={n2} mirror")

            rem = deadline - time.monotonic()
            if rem > 30:
                x_pol2, _ = optimize_lse(x_opt2, min(150, rem * 0.4), rng,
                                         beta_init=500.0, beta_max=50000.0)
                update_global(x_pol2, f"Phase 2 n={n2} LSE")

    # === Phase 3: Intensive refinement on global best ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 80:
        n = len(global_a)
        t3 = remaining - 30
        print(f"\nPhase 3: Intensive refinement, n={n}, time={t3:.0f}s")

        x_curr = global_a / global_a.sum()

        # Alternating multi-method
        n_rounds = min(6, int(t3 // 40))
        t_per = t3 / max(n_rounds, 1)

        for ri in range(n_rounds):
            if time.monotonic() > deadline - 15:
                break
            t_phase = min(t_per, deadline - time.monotonic() - 10)
            if t_phase < 8:
                break

            if ri % 3 == 0:
                x_curr, _ = optimize_mirror(x_curr, t_phase, rng, verbose=False)
            elif ri % 3 == 1:
                x_curr, _ = optimize_additive(x_curr, t_phase, rng)
            else:
                x_curr, _ = optimize_lse(x_curr, t_phase, rng,
                                        beta_init=1000.0, beta_max=60000.0)

            # Anti-concentration between rounds
            c_check = fft_convolve_self(x_curr)
            x_curr = anti_concentration_move(x_curr, c_check, n, rng,
                                            strength=0.015, max_lags=5)
            update_global(x_curr, f"Phase 3 round {ri}")

    # === Phase 4: Targeted peak reduction ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 50:
        n = len(global_a)
        t4 = remaining - 25
        print(f"\nPhase 4: Targeted peak reduction, time={t4:.0f}s")

        x_curr = global_a / global_a.sum()

        # Aggressive: focus on top 1-3 lags with very small steps
        c = fft_convolve_self(x_curr)
        round_start = time.monotonic()
        while time.monotonic() - round_start < t4:
            c = fft_convolve_self(x_curr)
            cur_max = c.max()
            cur_mean = c[1:2*n-2].mean() if n > 1 else c[0]
            gap = cur_max - cur_mean
            if gap < 1e-15:
                break

            # Very focused: top 2-3 lags
            k = 2
            temp = gap * n * 0.1  # very low temp = extreme focus
            grad = compute_subgradient_topk(x_curr, c, k, temp)
            gnorm = np.linalg.norm(grad)

            if gnorm > 1e-16:
                # Small mirror step
                eta = gap / (gnorm * gnorm * 3.0)
                x_try = mirror_step(x_curr, grad, eta)
                c_try = fft_convolve_self(x_try)
                if c_try.max() < cur_max:
                    x_curr = x_try
                else:
                    # Try additive
                    step = gap / (gnorm * gnorm * 5.0)
                    x_try2 = project_simplex(x_curr - step * grad, 1.0)
                    c_try2 = fft_convolve_self(x_try2)
                    if c_try2.max() < cur_max:
                        x_curr = x_try2

            # Brief LSE micro-polish
            t_micro = min(15.0, deadline - time.monotonic() - 8)
            if t_micro > 3:
                x_curr, _ = optimize_lse(x_curr, t_micro, rng,
                                        beta_init=2000.0, beta_max=80000.0)
            update_global(x_curr, "Phase 4")

    # === Phase 5: Final micro-polish ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 20:
        n = len(global_a)
        x_final = global_a / global_a.sum()
        t5 = remaining - 10
        print(f"\nPhase 5: Final micro-polish, time={t5:.0f}s")
        x_final, _ = optimize_lse(x_final, t5, rng, beta_init=3000.0, beta_max=100000.0)
        update_global(x_final, "Phase 5 final")

    # === Verify and return ===
    if global_a is None:
        print("WARNING: No valid solution found, using uniform fallback")
        global_a = np.ones(1024) / 1024
        global_best = 2.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024) / 1024

    final_list = [float(v) for v in global_a]
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL: len={n_final}, obj={global_best:.6f}")
    print(f"evaluate_sequence: {ver:.6f}")
    print(f"{'='*60}")

    return final_list
