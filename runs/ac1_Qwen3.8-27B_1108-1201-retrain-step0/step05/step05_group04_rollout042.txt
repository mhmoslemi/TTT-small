"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 for non-negative sequences.
Improved approach: aggressive LSE-FISTA continuation with geometric beta schedule,
active-set identification, Frank-Wolfe polishing, and multi-restart strategy.
Key insight: asymmetric sequences break the symmetry barrier (symmetric >= 2),
and the optimization should converge to a shape where the convolution is as flat as possible.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_and_gradient(a, c, beta):
    """Compute LSE value and gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    lse_val = cm + np.log(tot) / beta
    if tot < 1e-300:
        return np.zeros(n), lse_val
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, lse_val


def active_set_gradient(a, c, tol_frac=0.01):
    """Gradient using active set weighted by proximity to max."""
    n = len(a)
    cm = c.max()
    # Weight lags by how close they are to the max
    weights = np.maximum(0, 1.0 - (cm - c) / (tol_frac * cm + 1e-30))
    tot = weights.sum()
    if tot < 1e-15:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = weights / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_lse_fista_continuation(a_init, n, S, max_time, rng,
                                beta_start=50.0, beta_end=200000.0,
                                n_beta_steps=300, iters_per_beta=50):
    """
    LSE-FISTA with geometric beta continuation.
    At each beta level, run a fixed number of FISTA iterations.
    Adaptive: if convergence is slow at a beta, do more iterations.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    last_improve_time = start

    for beta_idx in range(n_beta_steps):
        frac = float(beta_idx) / max(1, n_beta_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 500000.0)

        iters = iters_per_beta
        # If gap is large, do more iterations
        for it in range(iters):
            if time.monotonic() - start > max_time:
                break

            c = fft_convolve_self(y)
            cur_max = c.max()

            if cur_max < best:
                best = cur_max
                best_a = y.copy()
                last_improve_time = time.monotonic()

            grad, lse_val = lse_and_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break

            step = 1.0 / (beta + 1e-10)

            # FISTA update
            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

        # Momentum restart check at end of beta level
        if time.monotonic() - last_improve_time > 5.0:
            t_mom = 1.0
            y = a.copy()

    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_frank_wolfe(a_init, n, S, max_time, rng, beta_start=100.0,
                     beta_end=50000.0, n_beta_steps=100):
    """
    Frank-Wolfe (conditional gradient) method with LSE smoothing.
    Natural for simplex constraints. Uses LSE-smoothed gradient.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    for beta_idx in range(n_beta_steps):
        frac = float(beta_idx) / max(1, n_beta_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 200000.0)

        t_beta0 = time.monotonic()
        # Run FW for ~2 seconds per beta level
        while time.monotonic() - t_beta0 < 2.0:
            if time.monotonic() - start > max_time:
                break

            c = fft_convolve_self(a)
            cur_max = c.max()
            if cur_max < best:
                best = cur_max
                best_a = a.copy()

            grad, lse_val = lse_and_gradient(a, c, beta)

            # FW: find the vertex (standard basis) that minimizes grad^T * e_i
            i = np.argmin(grad)

            # Step size: 2/(t+2) or backtracking
            step = 2.0 / (float(beta_idx) + 2.0)
            # Adaptive: try to improve
            a_trial = (1.0 - step) * a + step * S * np.zeros(n)
            a_trial[i] = step * S

            # Check if trial is better
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() > cur_max * 1.001:
                step = 0.5 * step
                a_trial = (1.0 - step) * a + step * S * np.zeros(n)
                a_trial[i] = step * S

            a = a_trial
            a = np.maximum(a, 0.0)
            # Re-normalize (FW should maintain simplex, but numerical errors)
            s = a.sum()
            if s > 1e-15:
                a = a * (S / s)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_active_refinement(a_init, n, S, max_time, rng):
    """
    Active-set refinement: identify top lags, compute weighted gradient,
    use backtracking line search.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Active set: top lags within 1% of max
        threshold = cur_max * 0.995
        active = c >= threshold
        n_active = active.sum()

        if n_active < 1:
            active[np.argmax(c)] = True
            n_active = 1

        # Weighted gradient (weight by how active each lag is)
        weights = np.where(active, 1.0, 0.0) / n_active
        fsize = 1 << (len(weights) + n - 1).bit_length()
        W = fft(weights, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / (n_active * 2.0)
        improved = False
        for _ in range(10):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur_max * (1 - 1e-7):
                a = a_test
                improved = True
                break
            step *= 0.5

        if not improved and it % 100 == 0:
            # Perturb to escape
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    return best_a, best


def run_hybrid_optimize(a_init, n, S, max_time, rng):
    """
    Hybrid: LSE-FISTA for bulk, then active-set refinement, then FW polish.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    def track(a_new):
        nonlocal best, best_a
        c = fft_convolve_self(a_new)
        if c.max() < best:
            best = c.max()
            best_a = a_new.copy()
        return best

    # Phase A: LSE-FISTA continuation (60% of time)
    time_a = max_time * 0.55
    a, _ = run_lse_fista_continuation(a, n, S, time_a, rng,
                                       beta_start=50.0, beta_end=100000.0,
                                       n_beta_steps=250, iters_per_beta=40)
    track(a)
    print("    After LSE-FISTA: %.8f" % best)

    # Phase B: Active-set refinement (20% of time)
    remaining = max(1.0, max_time - (time.monotonic() - start))
    time_b = remaining * 0.35
    a, _ = run_active_refinement(a, n, S, time_b, rng)
    track(a)
    print("    After active refinement: %.8f" % best)

    # Phase C: High-beta LSE polish (remaining time)
    remaining = max(1.0, max_time - (time.monotonic() - start))
    if remaining > 5:
        a, _ = run_lse_fista_continuation(a, n, S, remaining * 0.7, rng,
                                           beta_start=5000.0, beta_end=500000.0,
                                           n_beta_steps=80, iters_per_beta=30)
        track(a)
        print("    After high-beta polish: %.8f" % best)

    return best_a, best


def generate_exponential_init(n, c):
    """Generate exponential decay initialization with parameter c."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(-c * j / n)
    return a


def generate_raised_cosine_asymmetric(n, center_frac, width_frac):
    """Asymmetric raised cosine."""
    t = np.linspace(0, 1, n)
    center = center_frac
    width = width_frac
    x = (t - center) / width
    a = np.where(np.abs(x) < 1, 0.5 * (1 + np.cos(np.pi * x)), 0.0)
    return np.maximum(a, 1e-15)


def generate_gaussian_asymmetric(n, mu, sigma):
    """Asymmetric Gaussian (not centered)."""
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    return np.maximum(a, 1e-15)


def generate_power_law(n, p, direction='decay'):
    """Power law shape."""
    t = np.linspace(0.001, 1, n)
    if direction == 'decay':
        a = t ** (-p)
    else:
        a = t ** p
    return a / a.max()


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Uses aggressive LSE-FISTA continuation, active-set refinement,
    and multi-restart strategy with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    print("Budget: %.0fs, safety margin: %.0fs" % (budget_s, safety))

    global_best = np.inf
    global_a = None

    def try_sequence(a):
        nonlocal global_best, global_a
        n = len(a)
        S = np.sqrt(2.0 * n)
        a = normalize_sum(a, n)
        c = fft_convolve_self(a)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a.copy()
            print("  New best: %.8f (n=%d)" % (val, n))

    # === Load given sequence ===
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
            # Quick evaluation
            n_g = len(given)
            S_g = np.sqrt(2.0 * n_g)
            a_g = normalize_sum(given, n_g)
            c_g = fft_convolve_self(a_g)
            print("  Initial value: %.8f" % c_g.max())
            try_sequence(a_g)
    except Exception as e:
        print("Could not load previous: %s" % str(e))

    # === Phase 1: Optimize given sequence with aggressive LSE ===
    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        remaining = deadline - time.monotonic()
        t1 = min(remaining * 0.4, 600.0)
        print("=== Phase 1: Aggressive LSE on given (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_lse_fista_continuation(given, n_g, S_g, t1, rng,
                                             beta_start=200.0, beta_end=200000.0,
                                             n_beta_steps=300, iters_per_beta=30)
        try_sequence(a1)

        # Active-set refinement on the result
        remaining = deadline - time.monotonic()
        t1b = min(remaining * 0.2, 200.0)
        if t1b > 10:
            print("=== Phase 1b: Active refinement (%.0fs) ===" % t1b)
            a1b, v1b = run_active_refinement(a1, n_g, S_g, t1b, rng)
            try_sequence(a1b)

    # === Phase 2: Try different initializations ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        n_try = 8192
        S_try = np.sqrt(2.0 * n_try)
        t2 = remaining * 0.3
        print("=== Phase 2: Diverse inits (n=%d, %.0fs) ===" % (n_try, t2))

        # Generate diverse inits
        inits = []
        names = []

        # Exponential with various c
        for c in [0.8, 1.0, 1.2, 1.5, 2.0, 2.5, 3.0]:
            inits.append(generate_exponential_init(n_try, c))
            names.append("exp_c%.1f" % c)

        # Asymmetric raised cosine
        for cf in [0.3, 0.4, 0.5]:
            for wf in [0.3, 0.4, 0.5]:
                inits.append(generate_raised_cosine_asymmetric(n_try, cf, wf))
                names.append("rc_%.1f_%.1f" % (cf, wf))

        # Asymmetric Gaussian
        for mu in [0.3, 0.4, 0.5]:
            for sigma in [0.15, 0.2, 0.25, 0.3]:
                inits.append(generate_gaussian_asymmetric(n_try, mu, sigma))
                names.append("gauss_%.1f_%.2f" % (mu, sigma))

        # Power law
        for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
            inits.append(generate_power_law(n_try, p, 'decay'))
            names.append("pow_%.1f" % p)

        # Mixed: exponential * raised cosine
        t = np.linspace(0, 1, n_try)
        for c in [1.0, 2.0, 3.0]:
            a = np.exp(-c * t) * (1 + np.cos(np.pi * t))
            a = np.maximum(a, 1e-15)
            inits.append(a)
            names.append("mix_%.1f" % c)

        # Score inits and pick best few
        init_scores = []
        for i, init in enumerate(inits):
            a_init = normalize_sum(init, n_try)
            c_conv = fft_convolve_self(a_init)
            init_scores.append((c_conv.max(), i))

        init_scores.sort()
        print("  Top 5 inits:")
        for score, idx in init_scores[:5]:
            print("    %s: %.6f" % (names[idx], score))

        # Optimize top inits
        per_init = max(10.0, t2 / 5)
        for score, idx in init_scores[:5]:
            if time.monotonic() > deadline - 60:
                break
            init = inits[idx]
            print("  Optimizing init '%s' (score=%.6f, %.0fs)" % (names[idx], score, per_init))
            a_i, v_i = run_lse_fista_continuation(init, n_try, S_try, per_init, rng,
                                                   beta_start=50.0, beta_end=100000.0,
                                                   n_beta_steps=200, iters_per_beta=30)
            try_sequence(a_i)

    # === Phase 3: If we have a good result, do intensive refinement ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        n_main = len(global_a)
        S_main = np.sqrt(2.0 * n_main)

        if remaining > 60:
            t3 = remaining * 0.4
            print("=== Phase 3: Intensive hybrid (n=%d, %.0fs) ===" % (n_main, t3))
            a3, v3 = run_hybrid_optimize(global_a, n_main, S_main, t3, rng)
            try_sequence(a3)

    # === Phase 4: Try larger n ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 120:
            n_large = len(global_a) * 2
            if n_large <= 32768:
                S_large = np.sqrt(2.0 * n_large)
                t_old = np.linspace(0, 1, len(global_a))
                t_new = np.linspace(0, 1, n_large)
                init_large = np.interp(t_new, t_old, global_a)
                init_large = normalize_sum(init_large, n_large)

                t4 = remaining * 0.3
                print("=== Phase 4: Larger n=%d (%.0fs) ===" % (n_large, t4))
                a4, v4 = run_hybrid_optimize(init_large, n_large, S_large, t4, rng)
                try_sequence(a4)

    # === Phase 5: Final polish with very high beta ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            n_f = len(global_a)
            S_f = np.sqrt(2.0 * n_f)
            t5 = remaining * 0.5
            print("=== Phase 5: Final high-beta polish (%.0fs) ===" % t5)
            a5, v5 = run_lse_fista_continuation(global_a, n_f, S_f, t5, rng,
                                                 beta_start=20000.0, beta_end=500000.0,
                                                 n_beta_steps=50, iters_per_beta=20)
            try_sequence(a5)

            # Active-set final refinement
            remaining2 = max(1.0, deadline - time.monotonic())
            if remaining2 > 10:
                t5b = remaining2 * 0.7
                a5b, v5b = run_active_refinement(a5, n_f, S_f, t5b, rng)
                try_sequence(a5b)

    # === Final verification ===
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.8f, internal = %.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result found, using uniform fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
