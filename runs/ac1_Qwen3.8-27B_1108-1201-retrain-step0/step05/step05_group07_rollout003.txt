"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using entropic mirror descent,
adaptive LSE smoothing, and multi-phase optimization with length exploration.
Key innovation: entropic (multiplicative) updates that naturally preserve
non-negativity without explicit projection, combined with aggressive LSE polishing.

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    v = np.maximum(v, 0)
    s = v.sum()
    if abs(s - S) < 1e-12:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def entropic_update(a_old, g, eta, S):
    """Mirror descent step with entropy: a_new[j] ∝ a_old[j] * exp(-eta * g[j])."""
    exponent = -eta * g
    # Clip for numerical stability
    exponent = np.clip(exponent, -50, 50)
    weights = a_old * np.exp(exponent)
    wsum = weights.sum()
    if wsum < 1e-300:
        # Fallback: uniform
        return np.full_like(a_old, S / len(a_old))
    return weights * (S / wsum)


def lse_gradient(a, c, beta):

    # Gradient = 2 * (w * a) via cross-correlation
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

            continue

        if progress < 0.25:
            grad = topk_subgradient(a, c, k, use_softmax=True, softmax_temp=3.0)
        elif progress < 0.6:
            grad_lse = lse_gradient(a, c, beta)
            grad_topk = topk_subgradient(a, c, k, use_softmax=True, softmax_temp=5.0)
            blend = (progress - 0.25) / 0.35
            grad = (1.0 - blend) * grad_topk + blend * grad_lse
        else:
            grad = lse_gradient(a, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        eta_eff = eta / max(gnorm, 1e-10)
        a_new = entropic_update(a, grad, eta_eff, S)

        c_new = fft_convolve_self(a_new)
        if c_new.max() <= cur_max * (1.0 + 1e-8) + 1e-14:
            a = a_new
        else:
            a_new = entropic_update(a, grad, eta_eff * 0.5, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() <= cur_max * (1.0 + 1e-8) + 1e-14:
                a = a_new
            else:
                a = entropic_update(a, grad, eta_eff * 0.1, S)

        if it % 1000 == 0:
            a = np.maximum(a, 1e-12)

        if verbose and it % 3000 == 0:
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    MD it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, beta={beta:.0f}, k={k}")

    return best_a, best


def optimize_lse_polish(a0, S, max_time, rng, beta_init=500.0, beta_max=200000.0,
                        verbose=True):
    """
    High-beta LSE polishing using entropic updates for endgame convergence.
    """
    n = len(a0)
    a = project_simplex(a0.copy(), S)
    a = np.maximum(a, 1e-12)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 400

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-16:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > patience:
            noise_scale = 0.003 * S / np.sqrt(n) * max(0.1, 1.0 - progress)
            if noise_scale > 1e-14:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                a = np.maximum(a, 1e-12)
            stuck = 0
            patience = int(patience * 1.15)
            continue

        beta = min(beta_max, beta_init * (1.003 ** it))
        grad = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        eta = 1.5 / max(gnorm * np.sqrt(n), 1e-10)
        eta = min(eta, 2.0 / max(gnorm, 1e-10))

        a_new = entropic_update(a, grad, eta, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() <= cur_max + 1e-15:
            a = a_new
        else:
            a_new = entropic_update(a, grad, eta * 0.5, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() <= cur_max + 1e-15:
                a = a_new
            else:
                a = entropic_update(a, grad, eta * 0.1, S)

        if verbose and it % 5000 == 0:
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    LSE it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, beta={beta:.0f}")

    return best_a, best


def optimize_subgradient_projected(a0, S, max_time, rng,
                                   k_init=4, k_max=80, verbose=True):
    """
    Projected subgradient with top-k weighting and Polyak step size.
    """
    n = len(a0)
    a = project_simplex(a0.copy(), S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 250

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-14:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        k = int(k_init + (k_max - k_init) * (progress ** 0.5))
        grad = topk_subgradient(a, c, k, use_softmax=True, softmax_temp=4.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        f_lb = (S * S) / (2.0 * n - 1.0)
        polyak_step = (cur_max - f_lb) / max(gnorm * gnorm, 1e-30)
        step = min(polyak_step, 4.0 / gnorm)
        step = max(step, 1e-6 / gnorm)

        accepted = False
        for _ in range(8):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-10):
                a = a_try
                accepted = True
                break
            step *= 0.4

        if not accepted:
            a = project_simplex(a - (0.0005 / gnorm) * grad, S)

        if stuck > patience:
            noise_scale = min(0.025, 0.002 * (stuck // patience + 1)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.2)

        if verbose and it % 3000 == 0:
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    SG it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, k={k}")

    return best_a, best


def generate_init(n, rng, base=None):
    """Generate initial sequence with structured + noise."""
    if base is not None and len(base) > 1:
        t_old = np.linspace(0, 1, len(base))
        t_new = np.linspace(0, 1, n)
        a = np.interp(t_new, t_old, base)
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            return a
    tt = np.linspace(0, 1, n, endpoint=False)
    a = (1.0 + 0.25 * np.cos(2 * np.pi * tt) + 0.1 * np.cos(4 * np.pi * tt)
         + 0.05 * np.cos(6 * np.pi * tt) + 0.03 * np.cos(8 * np.pi * tt))
    a = a + 0.02 * rng.normal(size=n)
    a = np.maximum(a, 0.01)
    return a


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Combines entropic mirror descent, projected subgradient, and LSE polishing
    across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    print(f"=== Optimization Start | budget={budget_s}s ===")

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0.01 and len(a_provided) >= 256:
            print(f"Provided sequence: len={len(a_provided)}, sum={a_provided.sum():.2f}")
            cur_obj = obj_value(a_provided)
            print(f"  Initial objective: {cur_obj:.6f}")
        else:
            a_provided = None
    except Exception:
        a_provided = None

    global_best = np.inf
    global_a = None

    def _update_global(a, msg=""):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  *** NEW BEST: obj={obj:.6f} ({msg})")

    # === Phase 1: Mirror descent on provided sequence ===
    if a_provided is not None:
        n1 = len(a_provided)
        S1 = np.sqrt(2.0 * n1)
        a1 = a_provided * (S1 / a_provided.sum())
        a1 = project_simplex(a1, S1)

        rem = deadline - time.monotonic()
        t1 = min(500, rem * 0.25)
        if t1 > 30:
            print(f"\n--- Phase 1: Mirror descent n={n1}, t={t1:.0f}s ---")
            a_md, _ = optimize_mirror_descent(a1, S1, t1, rng,
                                              beta_init=40, beta_max=20000,
                                              k_init=4, k_max=60,
                                              eta_init=0.35, eta_min=0.02)
            _update_global(a_md, "P1 mirror")

    # === Phase 2: Projected subgradient (complementary) ===
    if a_provided is not None:
        n1 = len(a_provided)
        S1 = np.sqrt(2.0 * n1)
        base = global_a if (global_a is not None and len(global_a) == n1) else a1
        rem = deadline - time.monotonic()
        t2 = min(400, rem * 0.2)
        if t2 > 30:
            print(f"\n--- Phase 2: Projected subgradient n={n1}, t={t2:.0f}s ---")
            a_sg, _ = optimize_subgradient_projected(base, S1, t2, rng,
                                                      k_init=8, k_max=80)
            _update_global(a_sg, "P2 subgrad")

    # === Phase 3: LSE polish on global best ===
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        rem = deadline - time.monotonic()
        t3 = min(500, rem * 0.25)
        if t3 > 40:
            print(f"\n--- Phase 3: LSE polish n={n_best}, t={t3:.0f}s ---")
            a_pol, _ = optimize_lse_polish(global_a, S_best, t3, rng,
                                            beta_init=500, beta_max=80000)
            _update_global(a_pol, "P3 LSE")

    # === Phase 4: Explore alternative length ===
    rem = deadline - time.monotonic()
    if rem > 120:
        n4 = 16384
        S4 = np.sqrt(2.0 * n4)
        t4 = min(500, rem * 0.3)
        if t4 > 50:
            print(f"\n--- Phase 4: New length n={n4}, t={t4:.0f}s ---")
            base4 = global_a if global_a is not None else None
            init4 = generate_init(n4, rng, base=base4)
            init4 = project_simplex(init4, S4)

            a4, _ = optimize_mirror_descent(init4, S4, t4 * 0.45, rng,
                                             beta_init=30, beta_max=15000,
                                             k_init=4, k_max=50,
                                             eta_init=0.3, eta_min=0.02,
                                             verbose=False)
            _update_global(a4, f"P4 n={n4} mirror")

            rem = deadline - time.monotonic()
            if rem > 40:
                a4p, _ = optimize_lse_polish(a4, S4, min(250, rem * 0.5), rng,
                                              beta_init=300, beta_max=50000,
                                              verbose=False)
                _update_global(a4p, f"P4 n={n4} polish")

    # === Phase 5: Final intensive polishing ===
    rem = deadline - time.monotonic()
    if global_a is not None and rem > 60:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        t5 = rem - 25
        print(f"\n--- Phase 5: Final polish n={n}, t={t5:.0f}s ---")

        a_curr = global_a.copy()
        n_rounds = min(4, max(2, int(t5 / 80)))
        for ri in range(n_rounds):
            if time.monotonic() > deadline - 15:
                break
            t_r = min(t5 / n_rounds, deadline - time.monotonic() - 12)
            if t_r < 15:
                break
            beta_start = 1000 * (5 ** ri)
            a_curr, _ = optimize_lse_polish(a_curr, S, t_r, rng,
                                             beta_init=min(beta_start, 100000),
                                             beta_max=300000, verbose=False)
            _update_global(a_curr, f"P5 round {ri}")

    # === Phase 6: Quick perturb-and-refine restarts ===
    rem = deadline - time.monotonic()
    if global_a is not None and rem > 50:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        n_restarts = min(4, int(rem / 25))
        t_per = (rem - 15) / max(n_restarts, 1)
        print(f"\n--- Phase 6: {n_restarts} restarts, {t_per:.0f}s each ---")
        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            noise_scale = 0.008 * S / np.sqrt(n)
            a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n), S)
            a_pert, _ = optimize_lse_polish(a_pert, S, max(t_per, 10), rng,
                                             beta_init=2000, beta_max=200000,
                                             verbose=False)
            _update_global(a_pert, f"P6 restart {ri}")

    # === Final output ===
    if global_a is None:
        print("WARNING: fallback to uniform")
        global_a = np.ones(256) * 1.0
        global_best = obj_value(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 1.0

    final_list = [float(x) for x in global_a]
    n_final = len(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL: len={n_final}, internal_obj={global_best:.6f}")
    ver = evaluate_sequence(final_list)
    print(f"evaluate_sequence: {ver:.6f}")
    print(f"{'='*60}")

    return final_list
