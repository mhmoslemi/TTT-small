```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 for nonnegative a.

This version uses:
- a corrected smooth gradient for the square convolution a*a,
- L-BFGS on a log-sum-exp smoothed max objective,
- exact top-k subgradient polishing,
- spectral phase restarts,
- one-dimensional transform searches,
- coarse-to-fine continuation,
- optional extension to a longer sequence when time remains.
"""
import numpy as np
import time


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def safe_normalize(a, n, S):
    a = np.asarray(a, dtype=np.float64).reshape(-1).copy()
    if a.size != n:
        if a.size > n:
            a = a[:n]
        else:
            tmp = np.zeros(n, dtype=np.float64)
            tmp[:a.size] = a
            a = tmp
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.full(n, S / float(n), dtype=np.float64)
    a *= S / s
    a = np.minimum(a, 1000.0)
    s2 = float(a.sum())
    if s2 <= 1e-15:
        return np.full(n, S / float(n), dtype=np.float64)
    a *= S / s2
    return a


def project_fast(v, S):
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.maximum(v, 0.0)
    v = np.minimum(v, 1000.0)
    s = float(v.sum())
    n = len(v)
    if s <= 1e-15:
        return np.full(n, S / float(n), dtype=np.float64)
    v *= S / s
    v = np.minimum(v, 1000.0)
    s2 = float(v.sum())
    if s2 <= 1e-15:
        return np.full(n, S / float(n), dtype=np.float64)
    v *= S / s2
    return v


def compute_conv(a, N):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if N < 2 * n - 1:
        N = next_power_of_two(2 * n - 1)
    A = np.fft.fft(a, N)
    c = np.real(np.fft.ifft(A * A))[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def hankel_conv_grad(a, w, gs, n):
    """
    Gradient of sum_k w_k * (conv(a,a))_k for linear convolution square.
    Returns g_i = 2 * sum_q w_{i+q} a_q for valid indices.
    """
    r = w[::-1].copy()
    R = np.fft.fft(r, gs)
    A = np.fft.fft(a, gs)
    conv_ra = np.real(np.fft.ifft(R * A))[:3 * n - 1]
    return 2.0 * conv_ra[2 * n - 2:n - 1:-1]


def lse_value(c, beta):
    cmax = float(c.max())
    if beta <= 1e-8:
        return cmax
    e = np.exp(beta * (c - cmax))
    total = float(e.sum())
    if total <= 1e-300:
        return cmax
    return cmax + float(np.log(total)) / float(beta)


def compute_lse(a, N, gs, n, beta):
    c = compute_conv(a, N)
    cmax = float(c.max())
    val = lse_value(c, beta)
    if beta <= 1e-8:
        w = np.full(len(c), 1.0 / len(c), dtype=np.float64)
    else:
        w = np.exp(beta * (c - cmax))
        total = float(w.sum())
        if total <= 1e-300:
            w = np.zeros(len(c), dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= total
    grad = hankel_conv_grad(a, w, gs, n)
    return grad, c, cmax, val


def lbfgs_dir(s_hist, y_hist, g):
    m = len(s_hist)
    if m == 0:
        return -g
    n = len(g)
    q = g.copy()
    alphas = np.zeros(m, dtype=np.float64)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = float(np.dot(y_i, y_i))
        if yy > 1e-30:
            alphas[i] = float(np.dot(s_i, q)) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = float(np.dot(y_last, y_last))
    if yy_last > 1e-30:
        gamma = max(1e-8, float(np.dot(s_last, y_last)) / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = float(np.dot(y_i, y_i))
        if yy <= 1e-30:
            continue
        beta_i = float(np.dot(s_i, y_i)) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if float(np.dot(g, direction)) > 0:
        direction = -g
    return direction


def spectral_restart(a, N, rng, frac, S):
    a = safe_normalize(a, len(a), S)
    n = len(a)
    A = np.fft.fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    nrand = max(1, int(round(N * float(frac))))
    if nrand <= 1:
        newphase = rng.uniform(0.0, 2.0 * np.pi, N)
    else:
        idx = rng.choice(N, size=nrand, replace=False)
        newphase = phase.copy()
        newphase[idx] = rng.uniform(0.0, 2.0 * np.pi, nrand)
    A2 = mag * np.exp(1j * newphase)
    a2 = np.real(np.fft.ifft(A2))[:n]
    a2 = safe_normalize(a2, n, S)
    return a2


def central_sequence(n):
    out = np.ones(n, dtype=np.float64)
    if n <= 1:
        return out
    for i in range(1, n):
        out[i] = out[i - 1] * ((2 * i - 1) / float(2 * i))
    return out


def make_fourier_init(n, rng, N):
    A = rng.normal(size=N) + 1j * rng.normal(size=N)
    A[0] = abs(A[0]) + 1.0
    a = np.real(np.fft.ifft(A))[:n]
    return np.abs(a) + 0.01


def make_cos_asym(n, r):
    t = np.linspace(0.0, 1.0, n)
    return 1.0 + float(r) * np.cos(np.pi * t)


def score_candidate(v, n, S, N):
    a = safe_normalize(v, n, S)
    c = compute_conv(a, N)
    return float(c.max()), a


def downsample_seq(a, target):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = len(a)
    if n == target:
        return a.copy()
    if n > target:
        if n % target == 0:
            return a.reshape(target, n // target).mean(axis=1)
        x_old = np.linspace(0.0, 1.0, n)
        x_new = np.linspace(0.0, 1.0, target)
        return np.interp(x_new, x_old, a)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, target)
    return np.interp(x_new, x_old, a)


def resize_sequence(a, target):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if len(a) == target:
        return a.copy()
    if len(a) > target:
        return downsample_seq(a, target)
    x_old = np.linspace(0.0, 1.0, len(a))
    x_new = np.linspace(0.0, 1.0, target)
    return np.interp(x_new, x_old, a)


def upsample_seq(a, target, S_target, N_target):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = len(a)
    if n == target:
        return score_candidate(a, target, S_target, N_target)

    candidates = []

    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, target)
    candidates.append(np.interp(x_new, x_old, a))

    A = np.fft.fft(a)
    A_new = np.zeros(target, dtype=np.complex128)
    pos = min((n - 1) // 2, (target - 1) // 2)
    A_new[0] = A[0]
    if pos > 0:
        A_new[1:pos + 1] = A[1:pos + 1]
        A_new[-pos:] = A[-pos:]
    if (n % 2 == 0) and (target % 2 == 0):
        A_new[target // 2] = A[n // 2]
    band = np.real(np.fft.ifft(A_new))
    candidates.append(band)

    if target == 2 * n:
        out = np.zeros(target, dtype=np.float64)
        out[0::2] = a
        if n > 1:
            out[1:-1:2] = 0.5 * (a[:-1] + a[1:])
        out[-1] = a[-1]
        candidates.append(out)

    best_peak = np.inf
    best_a = safe_normalize(candidates[0], target, S_target)
    for cand in candidates:
        p, an = score_candidate(cand, target, S_target, N_target)
        if p < best_peak:
            best_peak = p
            best_a = an
    return best_a, best_peak


def lbfgs_lse(a_init, n, S, N, gs, budget_s, rng, beta):
    a = safe_normalize(a_init, n, S)
    grad, c, cmax, fval = compute_lse(a, N, gs, n, beta)
    best_a = a.copy()
    best_c = float(cmax)

    s_hist = []
    y_hist = []
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s))
    it = 0
    no_progress = 0

    while time.monotonic() < deadline:
        it += 1
        d = lbfgs_dir(s_hist, y_hist, grad)
        dnorm = float(np.linalg.norm(d))
        if dnorm <= 1e-16:
            break
        d = d / dnorm

        alpha = 50.0 * S / float(n)
        accepted = False
        best_ls_a = a
        best_ls_c = float(cmax)
        best_ls_f = float(fval)

        for _ in range(12):
            cand = project_fast(a + alpha * d, S)
            c_new = compute_conv(cand, N)
            cm = float(c_new.max())
            f_new = lse_value(c_new, beta)

            if cm < best_ls_c - 1e-12 or f_new < best_ls_f - 1e-12:
                best_ls_a = cand
                best_ls_c = cm
                best_ls_f = f_new

            if f_new <= fval + 1e-12 or cm < cmax - 1e-12:
                grad_new, c_new2, cm2, f_new2 = compute_lse(best_ls_a, N, gs, n, beta)
                sk = best_ls_a - a
                yk = grad_new - grad
                sy = float(np.dot(sk, yk))
                ss = float(np.dot(sk, sk))
                if sy > 1e-14 * max(1e-300, ss) and ss > 1e-20:
                    s_hist.append(sk)
                    y_hist.append(yk)
                    if len(s_hist) > 10:
                        s_hist = s_hist[-10:]
                        y_hist = y_hist[-10:]
                a = best_ls_a
                grad = grad_new
                c = c_new2
                cmax = float(cm2)
                fval = float(f_new2)
                if cmax < best_c:
                    best_c = cmax
                    best_a = a.copy()
                accepted = True
                no_progress = 0
                break

            alpha *= 0.5

        if not accepted:
            if best_ls_c < cmax - 1e-12:
                grad_new, c_new2, cm2, f_new2 = compute_lse(best_ls_a, N, gs, n, beta)
                sk = best_ls_a - a
                yk = grad_new - grad
                sy = float(np.dot(sk, yk))
                ss = float(np.dot(sk, sk))
                if sy > 1e-14 * max(1e-300, ss) and ss > 1e-20:
                    s_hist.append(sk)
                    y_hist.append(yk)
                    if len(s_hist) > 10:
                        s_hist = s_hist[-10:]
                        y_hist = y_hist[-10:]
                a = best_ls_a
                grad = grad_new
                c = c_new2
                cmax = float(cm2)
                fval = float(f_new2)
                if cmax < best_c:
                    best_c = cmax
                    best_a = a.copy()
                no_progress = 0
            else:
                no_progress += 1
                if no_progress > 45:
                    break

    # Final recompute for safety.
    c_final = compute_conv(best_a, N)
    best_c = min(best_c, float(c_final.max()))
    return best_a, best_c


def topk_subgrad_block(a_init, n, S, N, gs, budget_s, rng, top_k_init=32, top_k_max=256):
    a = safe_normalize(a_init, n, S)
    c = compute_conv(a, N)
    best_a = a.copy()
    best_c = float(c.max())

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s))
    it = 0
    no_progress = 0
    step = 25.0 * S / float(n)
    M = 2 * n - 1

    while time.monotonic() < deadline:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 20)
        if top_k <= 0:
            top_k = 1
        if top_k >= M:
            idx = np.arange(M)
        else:
            idx = np.argpartition(c, -top_k)[-top_k:]
        vals = c[idx]
        total = float(vals.sum())
        if total <= 1e-300:
            break

        w = np.zeros(M, dtype=np.float64)
        w[idx] = vals / total
        grad = hankel_conv_grad(a, w, gs, n)
        gnorm = float(np.linalg.norm(grad))
        if gnorm <= 1e-16:
            break
        grad = grad / gnorm

        alpha = step
        accepted = False
        for _ in range(10):
            cand = project_fast(a - alpha * grad, S)
            cm = float(compute_conv(cand, N).max())
            if cm < best_c - 1e-12:
                best_c = cm
                best_a = cand.copy()
            if cm < float(c.max()) - 1e-12:
                a = cand
                accepted = True
                no_progress = 0
                break
            alpha *= 0.6

        if not accepted:
            no_progress += 1
            if no_progress > 60:
                break
            if it % 12 == 0:
                a = spectral_restart(a, N, rng, 0.15, S)
                c = compute_conv(a, N)
                if c.max() < best_c:
                    best_c = float(c.max())
                    best_a = a.copy()
            else:
                a = spectral_restart(a, N, rng, 0.07, S)
                c = compute_conv(a, N)
                if c.max() < best_c:
                    best_c = float(c.max())
                    best_a = a.copy()
        else:
            c = compute_conv(a, N)
            if c.max() < best_c:
                best_c = float(c.max())
                best_a = a.copy()

        step = max(25.0 * S / float(n) / (1.0 + 0.012 * it), 1e-7 * S)

    c_final = compute_conv(best_a, N)
    best_c = min(best_c, float(c_final.max()))
    return best_a, best_c


def transform_search(a_best, n, S, N, budget_s, rng):
    a_best = safe_normalize(a_best, n, S)
    c0 = compute_conv(a_best, N)
    best_peak = float(c0.max())
    best_a = a_best.copy()
    if budget_s <= 1.0:
        return best_a, best_peak

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s))
    tt = np.linspace(0.0, 1.0, n)

    # Exponential tilt.
    lams = np.linspace(-2.0, 2.0, 13)
    for lam in lams:
        if time.monotonic() > deadline:
            break
        v = best_a * np.exp(float(lam) * tt)
        p, an = score_candidate(v, n, S, N)
        if p < best_peak:
            best_peak = p
            best_a = an

    # Cosine tilt.
    cos_t = np.cos(np.pi * tt)
    for lam in np.linspace(-0.85, 0.85, 9):
        if time.monotonic() > deadline:
            break
        v = best_a * (1.0 + float(lam) * cos_t)
        p, an = score_candidate(v, n, S, N)
        if p < best_peak:
            best_peak = p
            best_a = an

    # Reverse blend.
    rev = best_a[::-1]
    for al in (0.2, 0.4, 0.6, 0.8):
        if time.monotonic() > deadline:
            break
        v = (1.0 - al) * best_a + al * rev
        p, an = score_candidate(v, n, S, N)
        if p < best_peak:
            best_peak = p
            best_a = an

    # Blend with central binomial one-sided singular sequence.
    if n >= 64:
        cent = central_sequence(n)
        for al in (0.25, 0.5, 0.75):
            if time.monotonic() > deadline:
                break
            v = (1.0 - al) * best_a + al * cent
            p, an = score_candidate(v, n, S, N)
            if p < best_peak:
                best_peak = p
                best_a = an
        for al in (0.25, 0.5, 0.75):
            if time.monotonic() > deadline:
                break
            v = (1.0 - al) * best_a + al * cent[::-1]
            p, an = score_candidate(v, n, S, N)
            if p < best_peak:
                best_peak = p
                best_a = an

    # Spectral perturbations.
    for frac in (0.3, 0.6, 1.0):
        if time.monotonic() > deadline:
            break
        v = spectral_restart(best_a, N, rng, frac, S)
        p, an = score_candidate(v, n, S, N)
        if p < best_peak:
            best_peak = p
            best_a = an

    return best_a, best_peak


def optimize_scale(a_init, n, S, N, gs, budget_s, rng):
    a_best = safe_normalize(a_init, n, S)
    c_best = compute_conv(a_best, N)
    best_peak = float(c_best.max())
    a_cur = a_best.copy()

    print("optimize_scale n=%d current=%.8f budget=%.1f" % (n, best_peak, float(budget_s)))

    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s))
    block_id = 0
    stall = 0

    while time.monotonic() < deadline - 15:
        rem = deadline - time.monotonic()
        if rem < 18:
            break
        block_time = min(35.0, rem * 0.22)
        if block_time < 8:
            break

        beta = 6.0 + 7.0 * min(block_id, 14) + 4.0 * min(stall, 6)
        beta = min(700.0, beta)

        a_new, pe = lbfgs_lse(a_cur, n, S, N, gs, block_time, rng, beta)
        if pe < best_peak - 1e-10:
            best_peak = pe
            a_best = a_new.copy()
            a_cur = a_best
            stall = 0
            block_id += 1
            print("  [%d] LSE improved -> %.8f" % (block_id, best_peak))
        else:
            stall += 1
            if pe < best_peak * 1.003:
                a_cur = a_new.copy()
            else:
                a_cur = a_best.copy()

            if stall == 2:
                a_cur = spectral_restart(a_cur, N, rng, 0.4, S)
                stall = 0
            elif stall >= 4:
                a_cur = spectral_restart(a_cur, N, rng, 1.0, S)
                stall = 0

        if block_id % 3 == 0 and time.monotonic() < deadline - 70:
            tb = min(14.0, deadline - time.monotonic() - 70)
            a_tr, pe_tr = transform_search(a_best, n, S, N, tb, rng)
            if pe_tr < best_peak - 1e-10:
                best_peak = pe_tr
                a_best = a_tr.copy()
                a_cur = a_best
                stall = 0
                block_id += 1
                print("  [transform] -> %.8f" % best_peak)

    if time.monotonic() < deadline - 20:
        rem = deadline - time.monotonic() - 20
        beta_final = min(900.0, 80.0 + 10.0 * block_id)
        a_fin, pe_fin = lbfgs_lse(a_best, n, S, N, gs, min(180.0, rem), rng, beta_final)
        if pe_fin < best_peak - 1e-10:
            best_peak = pe_fin
            a_best = a_fin.copy()
            print("  [final LSE] -> %.8f" % best_peak)

    if time.monotonic() < deadline - 12:
        rem = deadline - time.monotonic() - 12
        a_tk, pe_tk = topk_subgrad_block(a_best, n, S, N, gs, min(90.0, rem), rng)
        if pe_tk < best_peak - 1e-10:
            best_peak = pe_tk
            a_best = a_tk.copy()
            print("  [topk] -> %.8f" % best_peak)

    c_fin = compute_conv(a_best, N)
    best_peak = min(best_peak, float(c_fin.max()))
    return a_best, best_peak


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 35.0)

    # Determine base working length.
    base_n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size > 0:
            # Choose nearest 8192 unless given is clearly larger.
            if given.size >= 4096 and given.size <= 16384:
                base_n = 1 << int(round(np.log2(given.size)))
                base_n = max(4096, min(16384, base_n))
    except Exception:
        given = None

    S_base = float(np.sqrt(2.0 * base_n))
    N_base = next_power_of_two(2 * base_n - 1)
    gs_base = next_power_of_two(3 * base_n - 2)

    global_best = np.inf
    global_a = None

    def _try_update(arr, source_name):
        nonlocal global_best, global_a
        n = len(arr)
        if n <= 0:
            return
        S = float(np.sqrt(2.0 * n))
        N = next_power_of_two(2 * n - 1)
        peak, an = score_candidate(arr, n, S, N)
        if peak < global_best:
            global_best = peak
            global_a = an.copy()
            print("update %-14s n=%d score=%.8f" % (source_name, n, global_best))

    print("base_n=%d S=%.4f N=%d gs=%d" % (base_n, S_base, N_base, gs_base))

    # Initial candidates at base length.
    candidates = []

    if given is not None:
        g = resize_sequence(given, base_n)
        candidates.append(("given", g))

    cent = central_sequence(base_n)
    candidates.append(("central", cent))
    candidates.append(("central_rev", cent[::-1]))
    candidates.append(("central_sym", 0.5 * (cent + cent[::-1])))
    candidates.append(("cos_05", make_cos_asym(base_n, 0.5)))
    candidates.append(("cos_08", make_cos_asym(base_n, 0.8)))
    candidates.append(("fourier1", make_fourier_init(base_n, rng, next_power_of_two(2 * base_n))))
    candidates.append(("fourier2", make_fourier_init(base_n, rng, next_power_of_two(2 * base_n))))
    candidates.append(("uniform", np.ones(base_n, dtype=np.float64)))

    for name, arr in candidates:
        if time.monotonic() > deadline - 40:
            break
        _try_update(arr, name)

    if global_a is None:
        global_a = safe_normalize(np.ones(base_n), base_n, S_base)
        global_best = float(compute_conv(global_a, N_base).max())
        print("fallback global score=%.8f" % global_best)

    # Coarse-to-fine continuation at base_n: 4096 then optionally 2048.
    if time.monotonic() < deadline - 250:
        coarse_n = 4096
        S_c = float(np.sqrt(2.0 * coarse_n))
        N_c = next_power_of_two(2 * coarse_n - 1)
        gs_c = next_power_of_two(3 * coarse_n - 2)
        a_c = downsample_seq(global_a, coarse_n)
        tb = min(90.0, deadline - time.monotonic() - 250)
        if tb > 20:
            a_c, pe_c = optimize_scale(a_c, coarse_n, S_c, N_c, gs_c, tb, rng)
            a_up, pe_up = upsample_seq(a_c, base_n, S_base, N_base)
            if pe_up < global_best:
                global_best = pe_up
                global_a = a_up.copy()
                print("coarse 4096 upsample -> %.8f" % global_best)

    if time.monotonic() < deadline - 350 and base_n >= 4096:
        coarse_n = 2048
        S_c = float(np.sqrt(2.0 * coarse_n))
        N_c = next_power_of_two(2 * coarse_n - 1)
        gs_c = next_power_of_two(3 * coarse_n - 2)
        a_c = downsample_seq(global_a, coarse_n)
        tb = min(50.0, deadline - time.monotonic() - 350)
        if tb > 15:
            a_c, pe_c = optimize_scale(a_c, coarse_n, S_c, N_c, gs_c, tb, rng)
            a_up, pe_up = upsample_seq(a_c, base_n, S_base, N_base)
            if pe_up < global_best:
                global_best = pe_up
                global_a = a_up.copy()
                print("coarse 2048 upsample -> %.8f" % global_best)

    # Main optimization at base length.
    if time.monotonic() < deadline - 120:
        rem = deadline - time.monotonic()
        main_budget = min(1500.0, max(250.0, rem * 0.62))
        a_fin, pe_fin = optimize_scale(global_a, len(global_a),
                                       float(np.sqrt(2.0 * len(global_a))),
                                       next_power_of_two(2 * len(global_a) - 1),
                                       next_power_of_two(3 * len(global_a) - 2),
                                       main_budget, rng)
        if pe_fin < global_best:
            global_best = pe_fin
            global_a = a_fin.copy()
            print("main base -> %.8f" % global_best)

    # Extra transform/refinement at base length.
    if time.monotonic() < deadline - 80 and len(global_a) == base_n:
        tb = min(90.0, deadline - time.monotonic() - 80)
        a_tr, pe_tr = transform_search(global_a, base_n, S_base, N_base, tb, rng)
        if pe_tr < global_best:
            global_best = pe_tr
            global_a = a_tr.copy()
            print("extra transform base -> %.8f" % global_best)

    # Optional longer sequence: 16384 if base is smaller and time remains.
    if len(global_a) < 16384 and time.monotonic() < deadline - 650:
        target_n = 16384
        S_lg = float(np.sqrt(2.0 * target_n))
        N_lg = next_power_of_two(2 * target_n - 1)
        gs_lg = next_power_of_two(3 * target_n - 2)
        a_lg, pe_lg = upsample_seq(global_a, target_n, S_lg, N_lg)
        if pe_lg < global_best * 1.03:
            tb = min(550.0, deadline - time.monotonic() - 300)
            if tb > 60:
                a_lg, pe_lg2 = optimize_scale(a_lg, target_n, S_lg, N_lg, gs_lg, tb, rng)
                if pe_lg2 < global_best:
                    global_best = pe_lg2
                    global_a = a_lg.copy()
                    print("long 16384 -> %.8f" % global_best)
            c_chk = compute_conv(a_lg, N_lg)
            if float(c_chk.max()) < global_best:
                global_best = float(c_chk.max())
                global_a = a_lg.copy()
                print("long 16384 initial -> %.8f" % global_best)

    # Optional very long sequence 32768 only if ample time remains.
    if len(global_a) < 32768 and time.monotonic() < deadline - 1300:
        target_n = 32768
        S_vlg = float(np.sqrt(2.0 * target_n))
        N_vlg = next_power_of_two(2 * target_n - 1)
        gs_vlg = next_power_of_two(3 * target_n - 2)
        a_vlg, pe_vlg = upsample_seq(global_a, target_n, S_vlg, N_vlg)
        if pe_vlg < global_best * 1.03:
            tb = min(800.0, deadline - time.monotonic() - 250)
            if tb > 120:
                a_vlg, pe_vlg2 = optimize_scale(a_vlg, target_n, S_vlg, N_vlg, gs_vlg, tb, rng)
                if pe_vlg2 < global_best:
                    global_best = pe_vlg2
                    global_a = a_vlg.copy()
                    print("very long 32768 -> %.8f" % global_best)

    # Final safe normalization and verification.
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)
    n_final = len(global_a)
    S_final = float(np.sqrt(2.0 * n_final))
    global_a = safe_normalize(global_a, n_final, S_final)

    final_list = [float(x) for x in global_a]
    try:
        ver = evaluate_sequence(final_list)
    except Exception:
        ver = np.inf

    if not np.isfinite(ver):
        # Fallback to a simple valid sequence.
        if given is not None and len(given) > 0:
            fb = safe_normalize(given, len(given), float(np.sqrt(2.0 * len(given))))
            final_list = [float(x) for x in fb]
        else:
            final_list = [1.0] * 256
        try:
            ver2 = evaluate_sequence(final_list)
            if np.isfinite(ver2):
                ver = ver2
        except Exception:
            ver = np.inf

    print("internal best=%.8f length=%d verified=%.8f" % (global_best, n_final, ver))
    return final_list
```
