            else: high=mid

```python
"""
Search for a non-negative sequence minimizing

    2*n*max(conv(a,a)) / (sum(a))^2

with a hybrid continuation / alternating-projection method.

The main new idea is to work in the coefficient space of the first half of a*a.
If a is normalized to sum sqrt(2n), the objective is just max(a*a).  For the
first-half vector h = (a*a)[0:n], the full sum is weighted:
    2*sum(h[0:n-1]) + h[n-1] = (sum a)^2.

Given current a, we project h onto {0 <= h <= T, weighted sum fixed}.  Then we
take one formal polynomial square-root Newton step

    a <- 1/2 (a + h_target * a^{-1})  (mod z^n)

with clipping and normalization.  This is a very cheap way to push the whole
convolution vector toward a lower box bound, much more global than a single
gradient step.  The method is combined with active-set polish and a few
restarts / length extrapolations.
"""

import numpy as np
import math
import time
from scipy.fft import rfft, irfft


def next_power_of_two_at_least(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def uniform_sequence(n):
    S = math.sqrt(2.0 * n)
    return np.ones(n, dtype=np.float64) * (S / float(n))


def normalize_sequence(a, S):
    a = np.asarray(a, dtype=np.float64).reshape(-1).copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(a > 0.0, a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(a.size, dtype=np.float64) * (S / float(a.size))
    a *= S / s
    return a


def orient_for_sqrt(a, S):
    """
    Make the sequence suitable for formal series square root:
    - non-negative,
    - normalized to sum S,
    - endpoint a[0] not too small.
    Reversal preserves the objective.
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1).copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(a > 0.0, a, 0.0)
    if a.size > 1 and a[-1] > a[0]:
        a = a[::-1].copy()
    floor = 1e-5 * S
    if a[0] < floor:
        a[0] = floor
    s = float(a.sum())
    if s < 1e-15:
        return normalize_sequence(a, S)
    a *= S / s
    return a


def resample_sequence(a, n_new):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    old = a.size
    if old == n_new:
        return a.copy()
    if old == 0 or n_new == 0:
        return np.ones(n_new, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n_new, dtype=np.float64)
    return np.interp(x_new, x_old, a, dtype=np.float64)


def series_mul(a, b, trunc):
    """First `trunc` coefficients of the formal product a*b."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    if trunc <= 0:
        return np.zeros(0, dtype=np.float64)

    la = a.size
    lb = b.size
    if la == 0 or lb == 0:
        return np.zeros(trunc, dtype=np.float64)

    if la == 1:
        c = np.zeros(trunc, dtype=np.float64)
        m = min(trunc, lb)
        c[:m] = a[0] * b[:m]
        return c

    if lb == 1:
        c = np.zeros(trunc, dtype=np.float64)
        m = min(trunc, la)
        c[:m] = b[0] * a[:m]
        return c

    full = la + lb - 1
    L = next_power_of_two_at_least(full)
    fa = rfft(a, L)
    fb = rfft(b, L)
    c = irfft(fa * fb, L)
    return c[:trunc].copy()


def series_inverse(f, n):
    """Formal inverse of f modulo z^n, assuming f[0] != 0."""
    f = np.asarray(f, dtype=np.float64)
    if f.size < n:
        f = np.concatenate((f, np.zeros(n - f.size, dtype=np.float64)))
    else:
        f = f[:n].copy()

    if abs(f[0]) < 1e-14:
        f[0] = 1e-14

    g = np.array([1.0 / f[0]], dtype=np.float64)
    m = 1
    while m < n:
        m2 = min(2 * m, n)
        fg = series_mul(f[:m2], g, m2)
        corr = 2.0 - fg
        g = series_mul(g, corr, m2)
        m = m2
    return g[:n].copy()


def self_conv_first_n(a):
    """First n coefficients of a*a (the full convolution is symmetric)."""
    n = a.size
    return series_mul(a, a, n)


def hmax_seq(a):
    """Internal objective assuming a is normalized to sum sqrt(2n)."""
    c = self_conv_first_n(a)
    if c.size == 0:
        return np.inf
    return float(np.max(c))


def project_box_weighted(H, T, total, weights=None):
    """
    L2 projection of H onto {0 <= x <= T, weights dot x = total}.
    For first-half convolution h from a length-n sequence, weights are 2
    everywhere except h[n-1], where the weight is 1.
    """
    H = np.asarray(H, dtype=np.float64).copy()
    H = np.maximum(H, 0.0)
    n = H.size
    if n == 0:
        return H
    if T <= 0.0:
        return np.zeros(n, dtype=np.float64)

    if weights is None:
        weights = np.full(n, 2.0, dtype=np.float64)
        weights[-1] = 1.0
    weights = np.asarray(weights, dtype=np.float64)

    hmax = float(np.max(H))
    if T >= hmax - 1e-12:
        x = H.copy()
    else:
        low = 0.0
        high = T - float(np.min(H))
        if high < 0.0:
            high = 0.0

        # Find lambda >= 0 such that sum(w * min(T, H + lambda)) = total.
        for _ in range(45):
            mid = 0.5 * (low + high)
            x_mid = np.minimum(T, H + mid)
            s = float(np.dot(weights, x_mid))
            if s < total:
                low = mid
            else:
                high = mid

        lam = 0.5 * (low + high)
        x = np.minimum(T, H + lam)

    # Small final correction for roundoff.
    diff = total - float(np.dot(weights, x))
    for _ in range(100):
        if abs(diff) <= 1e-9:
            break
        if diff > 0.0:
            slack = T - x
            idx = int(np.argmax(slack))
            if slack[idx] <= 0.0:
                break
            add = min(diff / weights[idx], slack[idx] * 0.999999)
            x[idx] += add
        else:
            idx = int(np.argmax(x))
            if x[idx] <= 0.0:
                break
            sub = min(-diff / weights[idx], x[idx] * 0.999999)
            x[idx] -= sub
        diff = total - float(np.dot(weights, x))

    x = np.clip(x, 0.0, max(0.0, T + 1e-12))
    return x


def poly_step_multi(a, H, S, T, weights):
    """
    One continuation step from current normalized a.

    H is the current first-half convolution a*a.
    T is a lower target max.
    Several target vectors are formed from H and inverse polynomial Newton
    roots are evaluated.  The best blend with a is returned.
    """
    n = a.size
    current = float(np.max(H))

    try:
        invA = series_inverse(a, n)
    except Exception:
        return a, current, False

    if not np.all(np.isfinite(invA)):
        return a, current, False

    H0 = np.maximum(H, 0.0)
    total = S * S

    B_list = []

    # 1) Exact weighted box + sum projection.
    try:
        B1 = project_box_weighted(H0, T, total, weights)
        B_list.append(B1)
    except Exception:
        pass

    # 2) Endpoint/simple cap, no explicit sum restoration.
    B2 = np.clip(H0, 0.0, T)
    B_list.append(B2)

    # 3) Soft peak reduction: changes only where H is above T.
    excess = np.maximum(H0 - T, 0.0)
    B3 = np.maximum(H0 - 0.5 * excess, 0.0)
    B_list.append(B3)

    best = a
    best_val = current
    etas = (1.0, 0.5, 0.25)

    for B in B_list:
        if B.size != n:
            continue
        if not np.all(np.isfinite(B)):
            continue

        try:
            ba = series_mul(B, invA, n)
        except Exception:
            continue

        if ba.size != n:
            continue
        if not np.all(np.isfinite(ba)):
            continue

        root = 0.5 * (a + ba)
        if not np.all(np.isfinite(root)):
            continue

        for eta in etas:
            cand = (1.0 - eta) * a + eta * root
            if not np.all(np.isfinite(cand)):
                continue
            cand = np.maximum(cand, 0.0)
            ss = float(cand.sum())
            if ss <= 0.0:
                continue
            cand *= S / ss

            H2 = self_conv_first_n(cand)
            val = float(np.max(H2))
            if val < best_val - 1e-12:
                best_val = val
                best = cand.copy()

    if best_val < current - 1e-11:
        return best, best_val, True
    return a, current, False


def run_poly_descent(a_init, max_time, min_T=1.0, initial_step=3e-4, max_step=5e-3, rng=None):
    """
    Continuation descent through lower and lower T targets.
    """
    n = a_init.size
    S = math.sqrt(2.0 * n)
    weights = np.full(n, 2.0, dtype=np.float64)
    weights[-1] = 1.0

    a = orient_for_sqrt(a_init, S)
    H = self_conv_first_n(a)
    best_val = float(np.max(H)) if H.size else np.inf
    best_a = a.copy()

    deadline = time.monotonic() + max(max_time, 0.0)
    step = max(1e-8, initial_step)
    no_improve = 0
    restarts = 0

    while time.monotonic() < deadline:
        a = orient_for_sqrt(a, S)
        H = self_conv_first_n(a)
        current = float(np.max(H)) if H.size else np.inf

        if current < best_val:
            best_val = current
            best_a = a.copy()
            no_improve = 0

        if current <= min_T or not np.isfinite(current):
            break

        T_target = current - step
        if T_target < min_T:
            T_target = min_T
        if T_target >= current:
            T_target = current - max(1e-8, step * 0.1)

        old = current
        a2, val2, improved = poly_step_multi(a, H, S, T_target, weights)

        if improved:
            a = a2
            if val2 < best_val:
                best_val = val2
                best_a = a.copy()

            improvement = old - val2
            if improvement > step * 0.25:
                step = min(max_step, step * 1.25 + 1e-8)
            else:
                step = min(max_step, step * 1.05)
            no_improve = 0
        else:
            step *= 0.5
            no_improve += 1

            if step < 1e-8 or no_improve > 80:
                if rng is not None and restarts < 8:
                    scale = 0.01 * S / math.sqrt(n)
                    nz = unit_lowpass_noise(n, rng, 15.0 + 10.0 * float(restarts))
                    cand = orient_for_sqrt(a + scale * nz, S)
                    vv = hmax_seq(cand)
                    if vv < best_val:
                        best_val = vv
                        best_a = cand.copy()
                    a = cand
                    step = max(0.5 * initial_step, min(max_step, step * 2.0))
                    no_improve = 0
                    restarts += 1
                else:
                    break

    return best_a, best_val


def cross_corr(w, a, n):
    """
    Gradient-like cross-correlation:
      2 * sum_k w_k a_{k-i}, zero-extended, taking n terms.
    """
    full = len(w) + n - 1
    L = next_power_of_two_at_least(full)
    fw = rfft(w, L)
    fa = rfft(a, L)
    r = irfft(fw * np.conj(fa), L)
    return 2.0 * r[:n].copy()


def unit_lowpass_noise(n, rng, alpha=20.0):
    z = rng.normal(0.0, 1.0, n)
    f = np.fft.rfftfreq(n)
    w = 1.0 / (1.0 + (alpha * f) ** 2)
    spec = rfft(z, n) * w
    noise = irfft(spec, n)
    noise -= float(noise.mean())
    std = float(np.std(noise))
    if std > 1e-14:
        noise /= std
    else:
        noise = rng.normal(0.0, 1.0, n)
    return noise


def run_active_polish(a_init, max_time, K=32, rng=None):
    """
    Active-set / top-lag subgradient polish.
    Uses simple nonnegative normalized updates and a short line search.
    """
    n = a_init.size
    S = math.sqrt(2.0 * n)
    if K <= 0 or K > n:
        K = n

    a = normalize_sequence(a_init, S)
    best_a = a.copy()
    best_val = hmax_seq(a)

    deadline = time.monotonic() + max(max_time, 0.0)
    steps = (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01)
    stagnant = 0

    while time.monotonic() < deadline:
        H = self_conv_first_n(a)
        val = float(np.max(H)) if H.size else np.inf
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stagnant = 0

        if K < n:
            top = np.argpartition(H, -K)[-K:]
        else:
            top = np.arange(n)

        w = np.zeros(n, dtype=np.float64)
        w[top] = 1.0 / float(K)

        grad = cross_corr(w, a, n)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            break

        improved = False
        for st in steps:
            cand = np.maximum(a - st * grad, 0.0)
            ss = float(cand.sum())
            if ss <= 0.0:
                continue
            cand *= S / ss
            v = hmax_seq(cand)
            if v < val - 1e-12:
                a = cand
                if v < best_val:
                    best_val = v
                    best_a = a.copy()
                improved = True
                stagnant = 0
                break

        if not improved:
            # Tiny accepted step helps leave plateaus.
            cand = np.maximum(a - 1e-6 * grad, 0.0)
            ss = float(cand.sum())
            if ss > 0.0:
                cand *= S / ss
                v = hmax_seq(cand)
                if v < val + 1e-10:
                    a = cand
                    if v < best_val:
                        best_val = v
                        best_a = a.copy()
                        stagnant = 0

            stagnant += 1

            if rng is not None and stagnant % 40 == 0 and stagnant <= 400:
                scale = 0.004 * S / math.sqrt(n)
                nz = unit_lowpass_noise(n, rng, 25.0)
                cand = normalize_sequence(a + scale * nz, S)
                v = hmax_seq(cand)
                if v < best_val:
                    best_val = v
                    best_a = cand.copy()
                    stagnant = 0

            if stagnant > 400:
                break

    return best_a, best_val


def make_inits(n, rng):
    """Diverse non-starts for length n."""
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    inits = []

    inits.append(np.ones(n, dtype=np.float64))

    # Raised cosines.
    for cf in (0.25, 0.35, 0.45, 0.55):
        for wf in (0.4, 0.6, 0.8):
            x = (t - cf) / (0.5 * wf)
            a = np.zeros(n, dtype=np.float64)
            m = np.abs(x) < 1.0
            a[m] = 0.5 * (1.0 + np.cos(np.pi * x[m]))
            inits.append(a)

    # Gaussians.
    for mu in (0.30, 0.45, 0.60):
        for sig in (0.08, 0.15, 0.25):
            inits.append(np.exp(-0.5 * ((t - mu) / sig) ** 2))

    # Sinusoidal nonnegative shapes.
    for k in (1, 2, 3, 5):
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        inits.append(0.1 + 0.9 * (0.5 + 0.5 * np.sin(np.pi * k * t + phase)))

    # Power laws.
    for p in (0.5, 1.0, 2.0, 3.0):
        inits.append(t ** p)
        inits.append((1.0 - t) ** p)

    # Smooth random shapes.
    for alpha in (5.0, 20.0, 50.0):
        nz = unit_lowpass_noise(n, rng, alpha)
        inits.append(np.maximum(1.0 + 0.5 * nz, 0.05))

    # Smooth two-level windows.
    for lo, hi in ((0.15, 0.45), (0.40, 0.70), (0.10, 0.55)):
        edge = 0.04
        left = 0.5 * (1.0 + np.tanh((t - lo) / edge))
        right = 0.5 * (1.0 + np.tanh((hi - t) / edge))
        inits.append(0.2 + 0.8 * left * right)

    return inits

            vv = hmax_seq(aa)
            if np.isfinite(vv):
                scored.append((vv, idx))
        scored.sort()

        per = max(25.0, min(90.0, rem * 0.04))
        tried = 0
        for vv, idx in scored:
            rem = remaining()
            if tried >= 5 or rem < per + 10:
                break
            if vv > best_val + 0.25:
                # Too far to be worth the budget, unless we have lots of time.
                if not (vv < best_val + 0.45 and rem > 600):
                    continue

            aa = normalize_sequence(inits[idx], S_cur)
            a2, v2 = run_poly_descent(aa, per,
                                      min_T=1.0,
                                      initial_step=1e-4,
                                      max_step=2.5e-3,
                                      rng=rng)
            tried += 1
            if v2 < best_val:
                best_a, best_val = a2, v2
                print("Diverse init %d improved: %.10f n=%d" % (idx, v2, len(best_a)), flush=True)

    # Active-set polish on the best sequence.
    rem = remaining()
    if rem > 90:
        t = min(130.0, rem * 0.25)
        a5, v5 = run_active_polish(best_a, t, K=32, rng=rng)
        if v5 < best_val:
            best_a, best_val = a5, v5
            print("Active polish K=32 improved: %.10f" % best_val, flush=True)

    rem = remaining()
    if rem > 70:
        t = min(70.0, rem * 0.25)
        a5, v5 = run_active_polish(best_a, t, K=128, rng=rng)
        if v5 < best_val:
            best_a, best_val = a5, v5
            print("Active polish K=128 improved: %.10f" % best_val, flush=True)

    # Try a longer length if there is room and potential.
    rem = remaining()
    if rem > 450 and len(best_a) < MAX_N:
        n2 = min(len(best_a) * 2, MAX_N)
        S2 = math.sqrt(2.0 * n2)
        a2 = resample_sequence(best_a, n2)
        a2 = orient_for_sqrt(a2, S2)
        v2_0 = hmax_seq(a2)
        print("Trying longer length n=%d, initial interpolated bound=%.10f" % (n2, v2_0), flush=True)

        t = min(600.0, rem * 0.35)
        a2, v2 = run_poly_descent(a2, t,
                                  min_T=max(1.0, best_val - 0.02),
                                  initial_step=2e-4,
                                  max_step=4e-3,
                                  rng=rng)

        rem = remaining()
        if rem > 40 and v2 < best_val + 0.005:
            a2, v2 = run_active_polish(a2, min(70.0, rem * 0.25), K=64, rng=rng)

        if v2 < best_val:
            best_a, best_val = a2, v2
            print("Longer length improved: %.10f n=%d" % (best_val, len(best_a)), flush=True)

    # Final micro-continuation.
    rem = remaining()
    if rem > 30:
        t = min(140.0, rem * 0.55)
        a6, v6 = run_poly_descent(best_a, t,
                                  min_T=1.0,
                                  initial_step=1e-5,
                                  max_step=8e-4,
                                  rng=rng)
        if v6 < best_val:
            best_a, best_val = a6, v6
            print("Final micro-poly improved: %.10f" % best_val, flush=True)

    rem = remaining()
    if rem > 15:
        a6, v6 = run_active_polish(best_a, min(25.0, rem * 0.6), K=64, rng=rng)
        if v6 < best_val:
            best_a, best_val = a6, v6

    # Final safety / cleanup.
    final_a = np.asarray(best_a, dtype=np.float64).reshape(-1).copy()
    final_a = np.clip(final_a, 0.0, 1000.0)

    if final_a.size < 16 or not np.all(np.isfinite(final_a)) or final_a.sum() < 0.01:
        final_a = uniform_sequence(8192)
    elif final_a.size > MAX_N:
        final_a = resample_sequence(final_a, MAX_N)
        final_a = normalize_sequence(final_a, math.sqrt(2.0 * MAX_N))
        final_a = np.clip(final_a, 0.0, 1000.0)

    print("Returning length=%d internal bound=%.10f runtime=%.0f" %
          (len(final_a), best_val, time.monotonic() - start), flush=True)

    return [float(x) for x in final_a]
```
