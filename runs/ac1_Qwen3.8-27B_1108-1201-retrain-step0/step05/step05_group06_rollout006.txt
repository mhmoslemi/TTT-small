"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
- Scale-invariant optimization (no simplex constraint, just non-negativity)
- LSE smoothing with adaptive beta and backtracking line search
- Multi-start with diverse initializations
- Active-set polishing
- Multi-resolution (upsampling)

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the actual objective value 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_sum(a, S):
    """Normalize sequence to have sum S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_weights(c, beta):
    """Compute softmax weights for LSE approximation of max."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()  # numerical stability
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.full(len(c), 1.0 / len(c))
    return e / tot


def lse_gradient_nosimplex(a, c, beta):
    """
    Gradient of 2*n*M/S^2 where M = LSE(c; beta), S = sum(a).
    Uses the scale-invariant formulation.
    """
    n = len(a)
    S = a.sum()
    M = (1.0 / beta) * np.log(np.sum(np.exp(beta * (c - c.max())))) + c.max()
    
    w = lse_weights(c, beta)
    
    # Compute g = dc/da (the "standard" convolution gradient)
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    # Full gradient of f = 2n*M/S^2:
    # df/da[i] = 2n/S^3 * (S*g[i] - 2*M)
    # We can drop the constant factor 2n/S^3 (doesn't affect direction)
    # and just use: grad[i] = S*g[i] - 2*M
    grad = S * g - 2.0 * M
    return grad, M, S


def run_lse_nosimplex(a_init, max_time, rng, beta_init=5.0, beta_max=500.0,
                      beta_increase=1.05, patience=200):
    """
    LSE optimization without simplex constraint (just non-negativity).
    Uses adaptive beta and backtracking line search.
    """
    n = len(a_init)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    if a.sum() < 1e-15:
        a = np.ones(n) / n
    a = normalize_sum(a, np.sqrt(2.0 * n))
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        S = a.sum()
        M_true = c.max()
        cur_obj = 2.0 * n * M_true / (S * S)
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Adaptive beta
        if stuck < patience:
            pass  # keep current beta
        elif stuck < patience * 3:
            beta = min(beta_max, beta * beta_increase)
        else:
            beta = min(beta_max, beta * 1.02)
        
        # Compute gradient
        grad, M_lse, S = lse_gradient_nosimplex(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        step = 1.0 / (beta * (np.max(a) + 1.0) + 1.0)
        step = max(step, 1e-10)
        
        best_step_obj = cur_obj
        best_step_a = a.copy()
        
        for ls in range(10):
            a_trial = np.maximum(a - step * grad, 0.0)
            if a_trial.sum() < 1e-15:
                step *= 0.5
                continue
            c_trial = fft_convolve_self(a_trial)
            S_trial = a_trial.sum()
            obj_trial = 2.0 * n * c_trial.max() / (S_trial * S_trial)
            
            if obj_trial < cur_obj - 1e-12:
                best_step_obj = obj_trial
                best_step_a = a_trial.copy()
                break
            step *= 0.5
        
        a = best_step_a
        
        # Periodic renormalization
        if it % 500 == 0:
            a = normalize_sum(a, np.sqrt(2.0 * n))
        
        # Noise injection when stuck
        if stuck > patience * 5 and it % 100 == 0:
            noise_scale = 0.01 * np.sqrt(2.0 * n) / np.sqrt(n)
            a = np.maximum(a + rng.normal(0, noise_scale, n), 0)
            a = normalize_sum(a, np.sqrt(2.0 * n))
            last_improve = it  # Reset stuck counter
        
        # Check time
        if time.monotonic() - start > max_time * 0.95:
            break
    
    # Final check
    final_obj = compute_obj(best_a)
    if final_obj < best_obj:
        best_obj = final_obj
    return best_a, best_obj

def simplex_project(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)
