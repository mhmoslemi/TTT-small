def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):

    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using KKT-optimized mirror descent
with optimal active-set weight computation via eigendecomposition.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_pressure(a, lambda_weights, active_indices, n):
    """
    Compute pressure w[p] = sum_k lambda_k * a[k-p] for all p in [0, n-1].
    Uses FFT for efficiency.
    """
    size = 2 * n - 1
    L = np.zeros(size)
    L[active_indices] = lambda_weights
    fsize = 1 << (size + n - 1).bit_length()
    # w[p] = sum_k L[k] * a[k-p] = cross_correlation(L, a)[p]
    # In FFT: ifft(fft(L) * conj(fft(a)))
    Lf = fft(L, fsize)
    Af = fft(a, fsize)
    w = np.real(ifft(Lf * np.conj(Af), fsize))[:n]
    return w


def solve_optimal_lambda(a, active_indices, support_mask, tol=1e-8):
    """
    Solve for optimal lambda on active set that minimizes variance of pressure on support.
    
    Returns: lambda weights (normalized to sum to 1)
    """
    n = len(a)
    m = len(active_indices)
    ns = int(support_mask.sum())
    
    if m == 1:
        return np.ones(1)
    
    if m > 200:
        # Too many active lags, use uniform weights
        return np.ones(m) / m
    
    # Build G matrix: G[s, i] = a[active_indices[i] - support_idx[s]]
    support_idx = np.where(support_mask)[0]
    G = np.zeros((ns, m))
    for i, k in enumerate(active_indices):
        # G[s, i] = a[k - support_idx[s]] (0 if out of bounds)
        idx = k - support_idx
        valid = (idx >= 0) & (idx < n)
        G[valid, i] = a[idx[valid]]
    
    # Center G (remove mean over support)
    P = G - G.mean(axis=0, keepdims=True)
    
    # Solve: minimize lambda^T (P^T P) lambda subject to sum(lambda) = 1, lambda >= 0
    # Unconstrained: eigenvector of P^T P with smallest eigenvalue
    M = P.T @ P  # m x m matrix
    M = (M + M.T) / 2  # Symmetrize
    
    # Find smallest eigenvector
    eigvals, eigvecs = np.linalg.eigh(M)
    lambda_vec = np.abs(eigvecs[:, 0])  # First eigenvector (smallest eigenvalue)
    
    # Project to simplex
    if lambda_vec.sum() < tol:
        return np.ones(m) / m
    lambda_vec = lambda_vec / lambda_vec.sum()
    
    # Check if the solution is good; if not, try uniform
    obj_opt = lambda_vec @ M @ lambda_vec
    obj_unif = (1.0/m) * M @ (1.0/m) * np.ones(m)  # Hmm this isn't right
    unif = np.ones(m) / m
    obj_unif = unif @ M @ unif
    
    if obj_opt < obj_unif:
        return lambda_vec
    else:
        return unif


def kkt_mirror_descent(a, S, max_time, rng, 
                        active_tol=0.01, line_search=True,
                        verbose=True, max_active=100):
    """
    KKT-optimized mirror descent for minimizing max(conv(a,a)).
    """
    n = len(a)
    a = normalize_to_sum(a.copy(), S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    it = 0
    last_improve = 0
    eta = 0.1  # Initial step size
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cmax = c.max()
        
        if cmax < best:
            best = cmax
            best_a = a.copy()
            last_improve = it
            if verbose and it % 1000 == 0:
                print("      iter %d: max_b=%.10f" % (it, best))
        
        # Identify active set
        # Use relative tolerance
        tol = active_tol * max(cmax - c.min(), 1e-10)
        active_mask = c >= (cmax - tol)
        active_indices = np.where(active_mask)[0]
        
        # Limit active set size
        if len(active_indices) > max_active:

            top_vals = c[active_indices]
            top_idx = np.argsort(top_vals)[::-1][:max_active]
            active_indices = active_indices[top_idx]

        w_support = w[support_mask]
        if len(w_support) > 0:
            w_mean = w_support.mean()
        else:
            w_mean = w.mean()
        
        # Scale eta based on the pressure variation
        w_var = np.std(w[support_mask]) if support_mask.sum() > 0 else 1.0
        if w_var < 1e-15:
            break  # KKT conditions are (approximately) satisfied
        
        eta_eff = eta * a.mean() / (w_var * n)  # Scale appropriately
        
        # Clip to avoid overflow
        scaled = -np.clip(eta_eff * (w - w_mean), -20, 20)
        a_new = a * np.exp(scaled)
        a_new = project_simplex(a_new, S)
        
        # Line search (optional)
        if line_search:
            c_new = fft_convolve_self(a_new)
            if c_new.max() > cmax:
                # Try smaller step
                for attempt in range(5):
                    eta_eff = eta_eff * 0.5
                    scaled = -np.clip(eta_eff * (w - w_mean), -20, 20)
                    a_new = a * np.exp(scaled)
                    a_new = project_simplex(a_new, S)
                    c_new = fft_convolve_self(a_new)
                    if c_new.max() <= cmax:

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using KKT-optimized mirror descent.
Key innovation: compute optimal active-set weights via eigendecomposition
to get the exact KKT subgradient direction, combined with careful line search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_pressure_fft(a, lam, active_indices, n):
    """Compute w[p] = sum_k lam[k] * a[k-p] using FFT."""
    size = 2 * n - 1
    L = np.zeros(size)
    L[active_indices] = lam
    fsize = 1 << (size + n - 1).bit_length()
    Lf = fft(L, fsize)
    Af = fft(a, fsize)
    w = np.real(ifft(Lf * np.conj(Af), fsize))[:n]
    return w


def solve_lambda(a, active_indices, support_idx, n):
    """
    Solve for optimal lambda minimizing variance of pressure on support.
    Returns normalized lambda vector.
    """
    m = len(active_indices)
    ns = len(support_idx)
    
    if m == 1:
        return np.ones(1)
    if m > 300:
        return np.ones(m) / m
    
    # Build G: ns x m, G[s, i] = a[active_indices[i] - support_idx[s]]
    G = np.zeros((ns, m))
    for i, k in enumerate(active_indices):
        idx = k - support_idx
        valid = (idx >= 0) & (idx < n)
        G[valid, i] = a[idx[valid]]
    
    # Center
    G_c = G - G.mean(axis=0, keepdims=True)
    
    # Compute M = G_c^T G_c (m x m)
    M = G_c.T @ G_c
    M = (M + M.T) / 2.0
    
    # Find eigenvector of smallest eigenvalue
    eigvals, eigvecs = np.linalg.eigh(M)
    lam = np.abs(eigvecs[:, 0])
    s = lam.sum()
    if s < 1e-15:
        return np.ones(m) / m
    lam = lam / s
    
    # Compare with uniform
    obj_lam = lam @ M @ lam
    unif = np.ones(m) / m
    obj_unif = unif @ M @ unif
    
    if obj_lam <= obj_unif:
        return lam
    return unif

def generate_perturbations(a, rng, n_pert=5):
    """Generate perturbed versions of a for diversification."""
    n = len(a)
    S = a.sum()
    perturbs = [a.copy()]
    
    for _ in range(n_pert):

        a_new = np.real(ifft(new_A, fsize))[:n]
        a_new = project_simplex(a_new, S)
        perturbs.append(a_new)

        n = 8192
        given = np.ones(n)

    S = np.sqrt(2.0 * n)
    given = normalize_to_sum(given, S)
    
    global_best = np.inf
    global_a = given.copy()
    
    def check_and_update(a):
        nonlocal global_best, global_a
        c = fft_convolve_self(a)
        val = c.max()
        obj = 2.0 * n * val / (a.sum() ** 2)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            return True
        return False
    
    # Initial evaluation
    check_and_update(given)
    print("Initial: obj=%.6f" % global_best)
    
    # Phase 1: Screen perturbations
    print("\n=== Phase 1: Screening perturbations ===")
    p1_budget = min(120.0, (deadline - time.monotonic()) * 0.1)
    perturbs = generate_perturbations(given, rng, n_pert=5)
    per_pert_time = p1_budget / (len(perturbs) + 1)
    
    # Also try a chirp initialization
    tt = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 8]:

    candidates = []
    for i, a_pert in enumerate(perturbs):
        if time.monotonic() > deadline - (deadline - total_start) * 0.8:
            break
        t_alloc = min(per_pert_time, deadline - time.monotonic() - 300)
        if t_alloc < 30:

        print("  Pert %d: optimizing..." % i)
        a_opt, val = kkt_optimize(a_pert, S, t_alloc, rng, verbose=False)
        obj = 2.0 * n * val / (a_opt.sum() ** 2)
        candidates.append((obj, a_opt.copy()))
        print("  Pert %d: obj=%.6f" % (i, obj))
        check_and_update(a_opt)

    print("  Best after screening: obj=%.6f" % global_best)

    ver = evaluate_sequence(final_list)

        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
