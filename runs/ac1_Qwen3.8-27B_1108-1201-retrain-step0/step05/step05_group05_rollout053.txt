def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()

        return np.zeros(n)

    A = fft(a, fsize)

    Compute KKT subgradient with optimized weights on active lags.
    The weights minimize the max subgradient on the support (LP relaxation).
    """
    n = len(a)
    cmax = c.max()
    active = np.where(c >= cmax - tol_rel * cmax)[0]
    active = active[:max_active]
    
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    support = np.where(a > 1e-10)[0]
    
    # Compute g^{(k)}_j = 2 * a_{k-j} for j in support, k in active
    # g_matrix[j_idx, k_idx] = 2 * a[active[k_idx] - support[j_idx]] (if valid)
    G = np.zeros((len(support), len(active)))
    for ki, k in enumerate(active):
        for ji, j in enumerate(support):
            if 0 <= k - j < n:
                G[ji, ki] = 2.0 * a[k - j]
    
    # Find weights w on simplex that minimize max_j (G w)_j
    # This is a minimax LP: min t s.t. G w <= t, w >= 0, sum(w) = 1
    # Use a simple approach: iterative projection
    m = len(active)
    w = np.ones(m) / m
    
    for _ in range(50):  # 50 iterations of projected gradient
        s = G @ w  # subgradient values on support
        s_max = s.max()
        # Gradient of max(s) is the one-hot at argmax(s)
        idx_max = np.argmax(s)
        grad_w = G[idx_max, :]
        # Projected gradient step on simplex
        w = w - 0.01 * grad_w
        w = np.maximum(w, 0)
        w = w / w.sum()
    
    s = G @ w
    # Return the full subgradient
    sg = np.zeros(n)
    for ji, j in enumerate(support):
        sg[j] = s[ji]

def spectral_flatten(a, n_iters=10, S=None):
    """Apply GS-type spectral flattening."""
    n = len(a)
    if S is None:
        S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    
    for _ in range(n_iters):
        A = fft(a, fsize)
        # Target: flatten the power spectrum (excluding DC)
        P = np.abs(A) ** 2
        P_mean = P[1:].mean()
        target_mag = np.sqrt(np.maximum(P_mean, P.max() * 0.001))
        target_mag[0] = np.abs(A[0])  # Preserve DC

        a_new = np.real(ifft(A_new, fsize))[:n]
        a = project_simplex(a_new, S)
    
    return a
