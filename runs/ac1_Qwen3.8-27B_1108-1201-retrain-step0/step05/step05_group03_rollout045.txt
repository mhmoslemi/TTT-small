import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if v.sum() <= S:
        return np.maximum(v, 0.0) * (S / max(v.sum(), 1e-300))
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

    if scaled.max() > 500:
        scaled = 500.0
        beta_adj = 500.0 / max(1e-300, (c - mc).max())
        scaled = beta_adj * (c - mc)
        beta = beta_adj
    e = np.exp(scaled - scaled.max())
    tot = e.sum()
    w = e / tot

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def lse_value(c, beta):
    """Compute LSE value."""
    mc = c.max()
    scaled = beta * (c - mc)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-300, (c - mc).max())
        scaled = beta * (c - mc)
    e = np.exp(scaled - scaled.max())
    return mc + np.log(e.sum()) / beta


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_equalize(a, alpha, rng):
    """Spectral equalization: reduce non-DC power to flatten autocorrelation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()

    A = fft(a, fsize)
    power = np.abs(A) ** 2

    # Reduce non-DC components
    mask = np.ones(fsize, dtype=bool)
    mask[0] = False
    # For real signals, also protect the Nyquist if fsize is even
    if fsize % 2 == 0:
        mask[fsize // 2] = False

    power_new = power.copy()
    power_new[mask] = power_new[mask] * (1.0 - alpha)
    power_new = np.maximum(power_new, 0.0)

    # Preserve phase
    phase = np.angle(A)
    A_new = np.sqrt(power_new / np.maximum(power, 1e-300)) * np.exp(1j * phase)
    A_new[0] = A[0]  # Preserve DC
    if fsize % 2 == 0:
        A_new[fsize // 2] = A[fsize // 2]

    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    if fsize % 2 == 0:
        phase_perturb[fsize // 2] = 0
    A_new = A * np.exp(1j * phase_perturb)
    # Ensure conjugate symmetry for real output
    for i in range(1, fsize // 2 + 1):
        A_new[fsize - i] = np.conj(A_new[i])
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def run_high_beta_fista(a_init, S, max_time, rng, beta_init=100.0, beta_rate=0.002,
                        beta_max=100000.0, k_frac=1.0, use_momentum=True,
                        momentum_decay_to=0.5):
    """FISTA with full LSE, adaptive momentum, and backtracking."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        grad = lse_gradient_full(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-13:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        # Adaptive momentum: reduce as we get closer to convergence
        if use_momentum:
            progress = min(1.0, it / 5000.0)
            mom_factor = 1.0 - (1.0 - momentum_decay_to) * progress
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            mu = ((t_val - 1.0) / t_new) * mom_factor
            y_new = x_new + mu * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        # Periodic restart when stuck
        if stuck > 800:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_val = 1.0
            stuck = 0

        # Periodic spectral equalization
        if it % 500 == 0 and stuck > 200:
            y = spectral_equalize(y, 0.15, rng)
            a = y.copy()
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_spectral_iterations(a_init, S, max_time, rng, alpha=0.1):
    """Iterative spectral equalization with refinement."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        a = spectral_equalize(a, alpha, rng)

        # Small gradient step after each spectral step
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = a.copy()

        beta = min(5000.0, 50.0 * (1.0 + it * 0.01))
        grad = lse_gradient_full(a, c, beta)
        step = 0.5 / beta
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max:
            a = a_new
            if a.max() < best - 1e-15:
                best = c_new.max()
                best_a = a.copy()
        else:
            a = a  # keep the spectral step result

        # Occasional perturbation
        if it % 1000 == 0 and (time.monotonic() - start) / max_time < 0.6:
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def run_exact_subgradient_polish(a_init, S, max_time, rng):
    """Polish using exact subgradient (max element) with Polyak averaging."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    # Track running best for Polyak step size
    f0 = best
    t = 0.0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = a.copy()
            f0 = min(f0, cur_max)

        # Find all elements within 1% of max (active set)
        threshold = cur_max * 0.99
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Compute subgradient: average of gradients at active lags
        # grad of b[k] = a[k] + ... is 2 * P_k * a where P_k is the appropriate matrix
        # For full convolution, grad_bk[a] = 2 * (a shifted by k) for appropriate k
        grad_sum = np.zeros(n)
        for k in active:
            if k < n:
                # b[k] = sum_{i=0}^{k} a[i]*a[k-i], grad wrt a[j] = a[k-j] + (a[-j] if j>k else 0)
                # Actually for linear conv: db[k]/da[j] = a[k-j] if 0<=k-j<n else 0, plus a[-j] if 0<=-j<n else 0
                # More carefully: b[k] = sum_i a[i]*a[i+k-n+1+... ] 
                # Let's just use the FFT-based approach for correctness
                pass
        
        # Use FFT approach for the subgradient at the max lag
        # For the true max (single element), grad = 2 * circ_shift(a, k)
        k_max = np.argmax(c)
        if k_max < n:
            grad = np.zeros(n)
            # b[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]
            # db[dk]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            # = 2*a[k-j] if k>=0 and 0<=k-j<n and k-j != j (i.e., k != 2j)
            # Simplified: use the FFT method
            w = np.zeros(2 * n - 1)
            w[k_max] = 1.0
            fsize = 1 << (2 * n - 1).bit_length()
            W = fft(w, fsize)
            A = fft(a, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        else:
            k_adj = 2 * n - 2 - k_max
            w = np.zeros(2 * n - 1)
            w[k_adj] = 1.0
            fsize = 1 << (2 * n - 1).bit_length()
            W = fft(w, fsize)
            A = fft(a, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Polyak step size
        t += 1
        step = (cur_max - f0) / (np.linalg.norm(grad) ** 2 + 1e-30)
        step = min(step, 1.0 / (1.0 + 0.01 * it))

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller step
            a_new = project_simplex(a - step * 0.1 * grad, S)
            c_new2 = fft_convolve_self(a_new)
            if c_new2.max() < cur_max:
                a = a_new

        # Small perturbation every few thousand iterations
        if it % 3000 == 0 and (time.monotonic() - start) / max_time < 0.5:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def run_perturbation_search(a_init, S, max_time, rng):
    """Try many random perturbations and keep improvements (basin hopping)."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Generate perturbation
        pert_type = rng.random()
        if pert_type < 0.3:
            # Small Gaussian noise
            scale = 0.01 * S / np.sqrt(n)
            a_try = project_simplex(a + rng.normal(0, scale, n), S)
        elif pert_type < 0.5:
            # Spectral perturbation
            a_try = spectral_perturb(a, 0.05 + 0.05 * rng.random(), rng)
        elif pert_type < 0.7:
            # Spectral equalization
            a_try = spectral_equalize(a, 0.1 + 0.2 * rng.random(), rng)
        elif pert_type < 0.85:
            # Random coordinate swaps
            a_try = a.copy()
            for _ in range(rng.integers(5, 50)):
                i, j = rng.integers(0, n, 2)
                delta = (a_try[i] - a_try[j]) * 0.1 * rng.random()
                a_try[i] -= delta
                a_try[j] += delta
            a_try = project_simplex(a_try, S)
        else:
            # Block perturbation
            a_try = a.copy()
            start_idx = rng.integers(0, n - 64)
            block = a_try[start_idx:start_idx + 64].copy()
            block = block + rng.normal(0, 0.01 * S / np.sqrt(n), 64)
            a_try[start_idx:start_idx + 64] = np.maximum(block, 0)
            a_try = project_simplex(a_try, S)

        # Evaluate
        c_try = fft_convolve_self(a_try)
        if c_try.max() < best:
            a = a_try
            best = c_try.max()
            best_a = a_try.copy()

    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def propose_candidate():
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: High-beta FISTA -> Spectral -> Polish -> Perturbation.
    """
    start_total = time.monotonic()
    budget = 2400.0
    deadline = start_total + budget - 30.0
    rng = np.random.default_rng(12345)

    print("Starting optimization, budget=%.0fs" % budget)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d, sum=%.4f" % (len(given), given.sum()))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_and_update(a_candidate, label):
        """Evaluate and update global best."""
        nonlocal global_best, global_a
        a_candidate = np.clip(np.asarray(a_candidate, dtype=np.float64), 0, 1000)
        s = a_candidate.sum()
        if s < 0.01:
            return
        b = fft_convolve_self(a_candidate)
        obj = 2.0 * len(a_candidate) * b.max() / (s * s)
        if obj < global_best:
            global_best = obj
            global_a = a_candidate.copy()
            print("  [%s] New best: obj=%.8f, max_conv=%.8f, n=%d" % (label, obj, b.max(), len(a_candidate)))

    # Phase 1: High-beta FISTA on n=8192 from given sequence
    n1 = 8192
    S1 = np.sqrt(2.0 * n1)
    if given is not None and len(given) == n1:
        a1 = normalize(given.copy(), S1)
    elif given is not None:
        a1 = normalize(resample_sequence(given, n1), S1)
    else:
        a1 = np.ones(n1) * S1 / n1

    print("\n=== Phase 1: High-beta FISTA (n=%d) ===" % n1)
    try_and_update(a1, "init")

    t1_limit = min(budget * 0.35, deadline - time.monotonic())
    if t1_limit > 30:
        a_opt, val_opt = run_high_beta_fista(a1, S1, t1_limit, rng,
                                             beta_init=200.0, beta_rate=0.001,
                                             beta_max=80000.0, use_momentum=True,
                                             momentum_decay_to=0.3)
        try_and_update(a_opt, "fista")

    # Phase 2: Spectral iterations
    print("\n=== Phase 2: Spectral iterations (n=%d) ===" % n1)
    t2_limit = min(budget * 0.15, deadline - time.monotonic())
    if t2_limit > 20:
        a_opt, val_opt = run_spectral_iterations(global_a if global_a is not None else a1,
                                                 S1, t2_limit, rng, alpha=0.08)
        try_and_update(a_opt, "spectral")

    # Phase 3: Exact subgradient polishing
    print("\n=== Phase 3: Subgradient polish (n=%d) ===" % n1)
    t3_limit = min(budget * 0.20, deadline - time.monotonic())
    if t3_limit > 20:
        a_opt, val_opt = run_exact_subgradient_polish(global_a if global_a is not None else a1,
                                                      S1, t3_limit, rng)
        try_and_update(a_opt, "polish")

    # Phase 4: Perturbation search (basin hopping)
    print("\n=== Phase 4: Perturbation search (n=%d) ===" % n1)
    t4_limit = min(budget * 0.15, deadline - time.monotonic())
    if t4_limit > 15:
        a_opt, val_opt = run_perturbation_search(global_a if global_a is not None else a1,
                                                 S1, t4_limit, rng)
        try_and_update(a_opt, "perturb")

    # Phase 5: Try n=16384
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n2 = 16384
        S2 = np.sqrt(2.0 * n2)
        a2 = normalize(resample_sequence(global_a, n2), S2)
        print("\n=== Phase 5: Optimization at n=%d ===" % n2)
        try_and_update(a2, "init_n2")

        t5_limit = remaining * 0.5
        if t5_limit > 30:
            a_opt, val_opt = run_high_beta_fista(a2, S2, t5_limit, rng,
                                                 beta_init=200.0, beta_rate=0.001,
                                                 beta_max=80000.0, use_momentum=True,
                                                 momentum_decay_to=0.3)
            try_and_update(a_opt, "fista_n2")

        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            a_opt, val_opt = run_exact_subgradient_polish(global_a if len(global_a) == n2 else a2,
                                                          S2, remaining2 * 0.6, rng)
            try_and_update(a_opt, "polish_n2")

    # Phase 6: Final intensive polish on the best found
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Phase 6: Final polish (n=%d, %.0fs) ===" % (n_best, remaining))

        # Very high beta FISTA
        t_fin = remaining * 0.5
        a_fin, val_fin = run_high_beta_fista(global_a, S_best, t_fin, rng,
                                             beta_init=1000.0, beta_rate=0.0005,
                                             beta_max=200000.0, use_momentum=False)
        try_and_update(a_fin, "final_fista")

        remaining2 = deadline - time.monotonic() - 10
        if remaining2 > 15:
            a_pol, val_pol = run_exact_subgradient_polish(global_a, S_best, remaining2, rng)
            try_and_update(a_pol, "final_polish")

    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - start_total
    print("Total time: %.1fs" % elapsed)
    print("Returning sequence of length %d, best obj=%.8f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
