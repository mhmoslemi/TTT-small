The function f(a) = max_k b[k] is convex in... wait, it's not convex as I discussed. But the subgradient at a point a is: (1/|S|) * sum_{k in S} grad_b[k] where S is the set of k achieving the max (or near-max).

The subgradient method (with appropriate step sizes) should converge to a local minimum for this problem. The issue is that it's slow.

**My refined approach:**

1. Use the subgradient of the max (or top-k max) to get a direction.
2. But instead of projecting onto the simplex (which is what FISTA does), use a smarter update:
   - Compute the subgradient g = (1/k) sum_{top k} grad_b[lag_i]
   - The direction -g tells us which coordinates to decrease
   - Project -g onto the tangent space of the simplex (i.e., subtract its mean)
   - Take a step in this direction with line search

3. Additionally, use "coordinate pairing": if the subgradient says to decrease a[j] and increase a[l], do a pairwise transfer with optimal amount.

def active_set_refine(a, max_time, rng, top_k=10):
    """
    Active-set refinement: iteratively solve a linearized problem
    focused on the top-k lags of the autocorrelation.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    N = 1 << (3*n).bit_length()  # FFT size for gradient computation
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        if it - last_improve > 5000:
            break
        
        # Find active set (top-k lags)
        k_active = min(top_k, 2*n-1)
        active_idx = np.argsort(c)[::-1][:k_active]
        
        # Compute multi-lag gradient
        w = np.zeros(2*n-1)
        w[active_idx] = 1.0 / k_active
        W = fft(w, N)
        A = fft(a, N)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Solve LP: min t s.t. grad^T a <= t + c[active] (linearized), sum(a)=S, a>=0
        # Actually the linearization is:
        # b_lin[k](a) = c[k] + grad_k^T (a - a_current)
        # We want b_lin[k](a) <= t
        # grad_k^T a <= t - c[k] + grad_k^T a_current
        # Since grad_k^T a_current = 2*c[k] (Euler's theorem), this becomes:
        # grad_k^T a <= t + c[k]
        
        # For the combined gradient: sum_k w[k]*grad_k^T a <= t + sum_k w[k]*c[k]
        # sum_k w[k]*c[k] = mean of top-k c values
        rhs_t = np.mean(c[active_idx])
        
        # LP: min t
        # s.t.: grad^T a - t <= rhs_t
        #        sum(a) = S
        #        a >= 0
        # Variables: a[0], ..., a[n-1], t (total n+1 variables)
        # Constraints: 
        #   grad^T a - t <= rhs_t (1 inequality)
        #   sum(a) = S (1 equality)
        #   a >= 0 (n inequalities, handled by bounds)
        
        # Using scipy linprog:
        # c_obj = [0, 0, ..., 0, 1] (minimize t)
        # A_ub = [grad[0], grad[1], ..., grad[n-1], -1], b_ub = [rhs_t]
        # A_eq = [1, 1, ..., 1, 0], b_eq = [S]
        # bounds = [(0, None)]*n + [(None, None)]
        
        c_obj = np.zeros(n+1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((1, n+1))
        A_ub[0, :n] = grad
        A_ub[0, n] = -1.0
        b_ub = np.array([rhs_t])
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])
        bounds = [(0, None)] * n + [(None, None)]
        
        from scipy.optimize import linprog
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        
        if res.success:
            a_new = res.x[:n]
            t_new = res.x[n]
            
            # Verify improvement
            obj_new_test = 2.0 * n * t_new / (S * S)  # approximate
            # Actually let's compute the true objective
            a_trial = np.maximum(a_new, 0)
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_convolve_self(a_trial)
            obj_trial = 2.0 * n * c_trial.max() / (S * S)
            
            if obj_trial < cur_obj:
                a = a_trial
                c = c_trial
            else:
                # Try a fraction of the step
                alpha = 0.5
                a_trial = a + alpha * (a_new - a)
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial * (S / a_trial.sum())
                c_trial = fft_convolve_self(a_trial)
                obj_trial = 2.0 * n * c_trial.max() / (S * S)
                if obj_trial < cur_obj:
                    a = a_trial
                    c = c_trial
                else:
                    # No improvement, try random perturbation
                    idx = rng.integers(0, n)
                    delta = a[idx] * 0.01
                    if delta > 0:
                        a[idx] -= delta
                        a[rng.integers(0, n)] += delta
                        c = fft_convolve_self(a)
        else:
            # LP failed, try subgradient step
            step = S / (np.linalg.norm(grad) * (1.0 + 0.01 * it))
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if 2.0 * n * c_new.max() / (S * S) < cur_obj:
                a = a_new
                c = c_new
    
    return best_a, best_obj

def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

        a = 0.5 + 0.5 * np.sin(2 * np.pi * 16 * tt)
    elif method == "broadband":
        a = 0.4 + 0.3 * np.sin(2 * np.pi * 5 * tt) + 0.2 * np.sin(2 * np.pi * 13 * tt + 0.5) + 0.1 * np.sin(2 * np.pi * 29 * tt + 1.2)
        a = np.abs(a)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA-LSE, active-set subgradient, paired CD, upsample, polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load starting point
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        a_start = make_init(n, "multi_tone", rng)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.8f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: FISTA-LSE variants (25% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.25
    if p1_time > 60:
        print("\n=== Phase 1: FISTA-LSE (budget %.0fs) ===" % p1_time)
        variants = [
            (20, 0.01, 20000, 1.0),
            (50, 0.005, 50000, 0.8),
            (100, 0.003, 100000, 0.5),
            (200, 0.001, 50000, 0.3),
        ]
        per_var = p1_time / len(variants)
        for i, (bi, br, bm, ss) in enumerate(variants):
            if time.monotonic() >= deadline:
                break
            rem = min(per_var, deadline - time.monotonic())
            if rem < 10:
                break
            a_opt, obj_opt = run_fista_lse(global_best_a, rem, rng,
                                            beta_init=bi, beta_rate=br,
                                            beta_max=bm, step_scale=ss)
            print("  FISTA-LSE v%d (beta_init=%d): obj=%.8f" % (i + 1, bi, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 2: Active-set subgradient (20% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.20
    if p2_time > 30:
        print("\n=== Phase 2: Active-set subgradient (budget %.0fs) ===" % p2_time)
        for k_top in [3, 5, 10, 20]:
            if time.monotonic() >= deadline:
                break
            rem = min(p2_time * 0.25, deadline - time.monotonic())
            if rem < 5:
                break
            a_opt, obj_opt = active_set_subgrad(global_best_a, rem, rng, top_k=k_top)
            print("  Subgrad k=%d: obj=%.8f" % (k_top, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 3: Paired coordinate descent (25% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.25
    if p3_time > 30:
        print("\n=== Phase 3: Paired CD (budget %.0fs) ===" % p3_time)
        a_opt, obj_opt = paired_coord_descent_fast(global_best_a, p3_time * 0.6, rng,
                                                    n_candidates=6)
        print("  Paired CD: obj=%.8f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

        # Follow up with FISTA
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            a_opt2, obj_opt2 = run_fista_lse(global_best_a, remaining3 * 0.7, rng,
                                              beta_init=50, beta_rate=0.003,
                                              beta_max=20000, step_scale=0.5)
            print("  FISTA after CD: obj=%.8f" % obj_opt2)
            if obj_opt2 < global_best_obj:
                global_best_obj = obj_opt2
                global_best_a = a_opt2.copy()

    # === Phase 4: Multi-scale upsample (25% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.25
    if p4_time > 60 and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        print("\n=== Phase 4: Upsample to n=%d (budget %.0fs) ===" % (n2, p4_time))
        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = a_up * (S2 / a_up.sum())
        obj_up = compute_obj(a_up)
        print("  Upsampled initial: obj=%.8f" % obj_up)
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up.copy()

        # FISTA on upsampled
        rem4 = min(p4_time * 0.5, deadline - time.monotonic())
        if rem4 > 20:
            a_up_opt, obj_up_opt = run_fista_lse(a_up, rem4, rng,
                                                  beta_init=5, beta_rate=0.005,
                                                  beta_max=10000, step_scale=1.0)
            print("  FISTA upsampled: obj=%.8f" % obj_up_opt)
            if obj_up_opt < global_best_obj:
                global_best_obj = obj_up_opt
                global_best_a = a_up_opt.copy()

        # Subgradient on upsampled
        rem4b = deadline - time.monotonic()
        if rem4b > 20:
            a_sg, obj_sg = active_set_subgrad(global_best_a, rem4b * 0.5, rng, top_k=5)
            print("  Subgrad upsampled: obj=%.8f" % obj_sg)
            if obj_sg < global_best_obj:
                global_best_obj = obj_sg
                global_best_a = a_sg.copy()

        # Paired CD on upsampled
        rem4c = deadline - time.monotonic()
        if rem4c > 20:
            a_cd, obj_cd = paired_coord_descent_fast(global_best_a, rem4c * 0.7,
                                                      rng, n_candidates=4)
            print("  Paired CD upsampled: obj=%.8f" % obj_cd)
            if obj_cd < global_best_obj:
                global_best_obj = obj_cd
                global_best_a = a_cd.copy()

    # === Phase 5: Multi-restart with perturbation (5% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.05
    if p5_time > 30:
        print("\n=== Phase 5: Multi-restart (budget %.0fs) ===" % p5_time)
        n_restarts = 5
        per_restart = p5_time / n_restarts
        for r in range(n_restarts):
            if time.monotonic() >= deadline:
                break
            # Structured perturbation: modulate with a random frequency
            n_cur = len(global_best_a)
            tt = np.linspace(0, 1, n_cur, endpoint=False)
            freq = rng.uniform(2, 20)
            phase = rng.uniform(0, 2 * np.pi)
            modulation = 1.0 + 0.1 * np.sin(2 * np.pi * freq * tt + phase)
            a_pert = global_best_a * modulation
            a_pert = np.maximum(a_pert, 0)
            S_cur = np.sqrt(2.0 * n_cur)
            a_pert = a_pert * (S_cur / a_pert.sum())
            a_r, obj_r = run_fista_lse(a_pert, min(per_restart, deadline - time.monotonic()),
                                        rng, beta_init=50, beta_rate=0.005,
                                        beta_max=10000, step_scale=0.5)
            print("  Restart %d: obj=%.8f" % (r + 1, obj_r))
            if obj_r < global_best_obj:
                global_best_obj = obj_r
                global_best_a = a_r.copy()

    # === Phase 6: Final high-precision polish (5% time) ===
    remaining = deadline - time.monotonic()
    p6_time = remaining * 0.90
    if p6_time > 10:
        print("\n=== Phase 6: Final polish (budget %.0fs) ===" % p6_time)
        a_pol, obj_pol = run_fista_lse(global_best_a, p6_time * 0.4, rng,
                                        beta_init=500, beta_rate=0.001,
                                        beta_max=100000, step_scale=0.2)
        print("  High-beta FISTA: obj=%.8f" % obj_pol)
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()

        rem6 = deadline - time.monotonic()
        if rem6 > 10:
            a_sg2, obj_sg2 = active_set_subgrad(global_best_a, rem6, rng, top_k=3)
            print("  Final subgrad: obj=%.8f" % obj_sg2)
            if obj_sg2 < global_best_obj:
                global_best_obj = obj_sg2
                global_best_a = a_sg2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.8f, n=%d" % (ver, len(final_list)))
            global_best_obj = ver
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
