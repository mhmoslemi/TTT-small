"""

Strategy: Multi-phase optimization combining adaptive LSE-FISTA with active-set

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_full(a):
    """Full self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection for the threshold."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    
    # Sort once for bisection
    order = np.argsort(v)[::-1]
    w = v[order]
    cumsum = np.cumsum(w)
    
    # Find the threshold using the sorted approach first (fast path)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    
    # Fix any numerical drift
    s = result.sum()
    if abs(s - S) > 1e-12 * max(1.0, abs(S)):
        # Try to fix by adjusting the smallest positive elements
        s_diff = S - s
        if s_diff > 0:
            # Need to add mass
            pos = result > 1e-15
            if pos.sum() > 0:
                result[pos] += s_diff / pos.sum()
        else:
            # Need to remove mass
            pos = result > 1e-15
            if pos.sum() > 0:
                frac = 1.0 + s_diff / result[pos].sum()
                frac = max(frac, 0.0)
                result[pos] *= frac
    return result


def normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_grad(a, c, beta):
    """Gradient of LSE approximation. Returns (grad, lse_value)."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    # Numerical stability
    shifted = np.minimum(shifted, 700.0)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    
    # convolve w with a to get gradient
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + np.log(tot) / beta
    return grad, lse_val


def active_set_grad(a, c, top_k=20):
    """Gradient using uniform weights on top_k active lags."""
    n = len(a)
    cm = c.max()
    # Select top_k lags
    top_idx = np.argsort(c)[::-1][:top_k]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / top_k
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

    """
    Active-set refinement with equality enforcement.

    """

    a = normalize(a_init, n)

    best = np.inf
    best_a = a.copy()
    it = 0
    rng_local = np.random.default_rng(99999 + n)

        c = fft_convolve_full(a)
        cur_max = c.max()

            best = cur_max
            best_a = a.copy()

        active_vals = c[top_idx]
        active_mean = active_vals.mean()

            weights = np.maximum(weights, 0)  # Only push down the ones above mean

            if wsum > 1e-15:

            else:

            w = np.zeros(len(c))

                w[k] = weights[i]
        else:
            w = np.zeros(len(c))
            w[top_idx] = 1.0 / len(top_idx)

        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step with backtracking
        step = 0.01 / max(1.0, it * 0.001)
        base_obj = cur_max
        improved = False
        
        for _ in range(10):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_full(a_test)
            test_obj = c_test.max()
            if test_obj < base_obj - 1e-14:
                a = a_test
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # Very small fixed step
            a = project_simplex(a - (step * 0.1) * grad, S)
        
        # Periodic noise
        if it % 100 == 0:
            ns = 0.0005 * S / np.sqrt(n) * np.exp(-it / 2000.0)
            if ns > 1e-14:
                a = project_simplex(a + rng_local.normal(0, ns, n), S)
    
    c_final = fft_convolve_full(best_a)
    if c_final.max() < best:
        best = c_final.max()
    return best_a, best


def peak_equalize(a, n, top_k=10, n_iter=200):
    """
    Iteratively equalize the top peaks by solving small linear systems.
    For each pair of active lags (k1, k2) with c[k1] > c[k2],
    find a direction that reduces c[k1] - c[k2].
    """
    S = np.sqrt(2.0 * n)
    a = normalize(a.copy() if isinstance(a_init, np.ndarray) else a, n) if False else a.copy()
    a = normalize(a, n)
    
    for _ in range(n_iter):
        c = fft_convolve_full(a)
        cm = c.max()
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        
        if top_vals.min() >= cm - 1e-10:
            break
        
        # For each pair (max_lag, min_lag), compute direction to equalize
        max_lag = top_idx[0]
        min_lag = top_idx[-1]
        
        # Gradient of c[max_lag] - c[min_lag]
        grad_diff = np.zeros(n)
        for k in [max_lag]:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            grad_diff[lo:hi+1] += 2.0 * a[k - np.arange(lo, hi+1)]
        for k in [min_lag]:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            grad_diff[lo:hi+1] -= 2.0 * a[k - np.arange(lo, hi+1)]
        
        # Project gradient onto the tangent space (remove component along 1-vector)
        grad_diff -= np.mean(grad_diff)
        
        gnorm = np.linalg.norm(grad_diff)
        if gnorm < 1e-15:
            break
        
        # Take a small step
        step = (top_vals[0] - top_vals[-1]) / (gnorm ** 2) * 0.5
        step = min(step, S / n * 0.01)
        a = project_simplex(a - step * grad_diff, S)
    
    return a


def spectral_init(n, num_bumps=1, asymmetry=0.0):
    """
    Generate initializations based on spectral considerations.
    A sequence whose DFT has certain properties will have controlled autocorrelation peaks.
    """
    t = np.linspace(0, 1, n)
    
    if num_bumps == 1:
        # Raised cosine with adjustable center
        center = 0.5 + asymmetry
        width = 0.4
        x = (t - center) / width
        a = np.where(np.abs(x) < 1, 0.5 * (1 + np.cos(np.pi * x)), 0.0)
    elif num_bumps == 2:
        # Two bumps
        a = 0.0
        for c in [0.3, 0.7]:
            x = (t - c) / 0.15
            a += np.where(np.abs(x) < 1, 0.5 * (1 + np.cos(np.pi * x)), 0.0)
    else:
        # Gaussian
        a = np.exp(-0.5 * ((t - 0.5) / 0.2) ** 2)
    
    return np.maximum(a, 0.0)


def multi_bump_init(n, n_bumps=3, rng=None):
    """Generate a multi-bump initialization."""
    if rng is None:
        rng = np.random.default_rng(42)
    t = np.linspace(0, 1, n)
    a = np.zeros(n)
    positions = np.sort(rng.uniform(0.1, 0.9, n_bumps))
    for pos in positions:
        sigma = rng.uniform(0.05, 0.2)
        a += np.exp(-0.5 * ((t - pos) / sigma) ** 2)
    return np.maximum(a, 0.0)


def wrapper_lse_phase(a, n, max_time, phase_label):
    """Wrapper for LSE phase with progress reporting."""
    start = time.monotonic()
    if phase_label == "coarse":
        a, val = adaptive_lse_fista(a, n, max_time,
                                     beta_start=30.0, beta_end=5000.0,
                                     beta_power=1.5, momentum_restart=300,
                                     noise_period=300, noise_scale_base=0.003)
    elif phase_label == "fine":
        a, val = adaptive_lse_fista(a, n, max_time,
                                     beta_start=2000.0, beta_end=50000.0,
                                     beta_power=2.0, momentum_restart=200,
                                     noise_period=500, noise_scale_base=0.001,
                                     use_backtrack=True)
    elif phase_label == "ultra_fine":
        a, val = adaptive_lse_fista(a, n, max_time,
                                     beta_start=10000.0, beta_end=100000.0,
                                     beta_power=1.0, momentum_restart=150,
                                     noise_period=1000, noise_scale_base=0.0005,
                                     use_backtrack=True)
    elapsed = time.monotonic() - start
    print("  %s: best=%.10f (%.1fs)" % (phase_label, val, elapsed))
    return a, val


def wrapper_active_phase(a, n, max_time, top_k=30):
    """Wrapper for active-set phase."""
    start = time.monotonic()
    a, val = active_set_refine(a, n, max_time, top_k=top_k, use_equality=True)
    elapsed = time.monotonic() - start
    print("  active_set(top_k=%d): best=%.10f (%.1fs)" % (top_k, val, elapsed))
    return a, val


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Multi-phase: multi-scale warm start -> coarse LSE -> fine LSE -> active-set
    refinement -> ultra-fine LSE -> peak equalization -> final polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 30.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)
    
    print("Propose candidate: budget=%.0fs, safety=%.0fs" % (budget_s, safety))
    
    global_best = np.inf
    global_a = None
    
    def update_best(a):
        nonlocal global_best, global_a
        n = len(a)
        S = np.sqrt(2.0 * n)
        a_norm = normalize(a, n)
        c = fft_convolve_full(a_norm)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a_norm.copy()
            print("  ** New best: %.10f (n=%d) ** [time remaining: %.0fs]" %
                  (val, n, deadline - time.monotonic()))
    
    # Load previous best
    a_start = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            a_start = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(a_start))
    except Exception as e:
        print("Could not load previous: %s" % str(e))
    
    if a_start is None:
        a_start = spectral_init(8192)
    
    n = len(a_start)
    S = np.sqrt(2.0 * n)
    a_start = normalize(a_start, n)
    
    # Initial evaluation
    c0 = fft_convolve_full(a_start)
    print("Initial: max_conv=%.10f (n=%d)" % (c0.max(), n))
    update_best(a_start)
    
    # === Phase 0: Quick coarse pass to ensure we're in a good basin ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t0 = min(60.0, remaining * 0.08)
        print("\n=== Phase 0: Coarse LSE warm-up (%.0fs) ===" % t0)
        a, v = wrapper_lse_phase(a, n, t0, "coarse")
        update_best(a)
    
    # === Phase 1: Main fine optimization ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t1 = remaining * 0.30
        print("\n=== Phase 1: Fine LSE (%.0fs) ===" % t1)
        a, v = wrapper_lse_phase(a, n, t1, "fine")
        update_best(a)
    
    # === Phase 2: Active-set refinement ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t2 = remaining * 0.25
        print("\n=== Phase 2: Active-set refinement (%.0fs) ===" % t2)
        a, v = wrapper_active_phase(a, n, t2, top_k=40)
        update_best(a)
    
    # === Phase 3: Ultra-fine LSE ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t3 = remaining * 0.25
        print("\n=== Phase 3: Ultra-fine LSE (%.0fs) ===" % t3)
        a, v = wrapper_lse_phase(a, n, t3, "ultra_fine")
        update_best(a)
    
    # === Phase 4: Peak equalization ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 4: Peak equalization ===")
        a_eq = peak_equalize(global_a if global_a is not None else a, n,
                             top_k=15, n_iter=500)
        update_best(a_eq)
        # Quick LSE polish after equalization
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_p, v_p = adaptive_lse_fista(a_eq, n, remaining2 * 0.5,
                                           beta_start=20000.0, beta_end=80000.0,
                                           beta_power=1.0, momentum_restart=100,
                                           noise_period=1000, noise_scale_base=0.0003)
            update_best(a_p)
    
    # === Phase 5: Final active-set with very small top_k ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 5: Final active-set (top_k=10) ===")
        t5 = min(remaining * 0.4, 120.0)
        a, v = wrapper_active_phase(a, n, t5, top_k=10)
        update_best(a)
    
    # === Phase 6: Alternative initializations (if time remains) ===
    remaining = deadline - time.monotonic()
    if remaining > 90:
        print("\n=== Phase 6: Alternative inits ===")
        # Try a few different initializations with short optimization
        alt_inits = [
            spectral_init(n, num_bumps=1, asymmetry=-0.15),
            spectral_init(n, num_bumps=1, asymmetry=0.15),
            spectral_init(n, num_bumps=2, asymmetry=0.0),
            multi_bump_init(n, n_bumps=4, rng=rng),
        ]
        
        for i, init in enumerate(alt_inits):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            t_alt = min(remaining * 0.15, 60.0)
            if t_alt < 10:
                break
            print("  Alt init %d: optimizing for %.0fs" % (i, t_alt))
            a_alt, v_alt = adaptive_lse_fista(init, n, t_alt,
                                               beta_start=100.0, beta_end=20000.0,
                                               beta_power=2.0, momentum_restart=200,
                                               noise_period=200, noise_scale_base=0.002)
            update_best(a_alt)
    
    # === Phase 7: Final polish on the absolute best ===
    remaining = deadline - time.monotonic()
    if remaining > 15 and global_a is not None:
        print("\n=== Phase 7: Final polish (%.0fs) ===" % remaining)
        t7 = remaining * 0.7
        a_final, v_final = adaptive_lse_fista(global_a, n, t7,
                                               beta_start=30000.0, beta_end=80000.0,
                                               beta_power=1.0, momentum_restart=100,
                                               noise_period=500, noise_scale_base=0.0002,
                                               use_backtrack=True)
        update_best(a_final)
        
        # One more active-set pass
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a_as, v_as = active_set_refine(a_final, n, remaining2 * 0.8,
                                           top_k=20, use_equality=True)
            update_best(a_as)
    
    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        n_final = len(global_a)
        print("\nFinal: evaluate_sequence=%.10f, internal=%.10f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, n_final))
        if not np.isfinite(ver):
            print("  WARNING: evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(8192, dtype=np.float64) * (np.sqrt(2 * 8192) / 8192)
    else:
        print("WARNING: No valid result, using uniform fallback.")
        global_a = np.ones(8192, dtype=np.float64) * (np.sqrt(2 * 8192) / 8192)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64) * (np.sqrt(2 * 8192) / 8192)
    
    print("RETURNING: length=%d, score=%.10f, time_used=%.0fs" %
          (len(global_a), global_best, time.monotonic() - (deadline - (budget_s - safety))))
    return [float(x) for x in global_a]
