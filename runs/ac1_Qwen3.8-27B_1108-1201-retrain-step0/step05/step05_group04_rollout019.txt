Strategy: Multi-phase optimization combining LSE-FISTA (broad search),
active-set Newton refinement (fine-tuning), and surgical peak shaving.

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    n = len(a)
    size = n + len(b) - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c). Returns (gradient, lse_value)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def active_set_gradient(a, c, beta_w=100.0):
    """Weighted active-set gradient using LSE weights on active lags."""
    n = len(a)
    cm = c.max()
    # LSE weights over all lags (effectively focuses on near-max)
    e = np.exp(beta_w * (c - cm))
    tot = e.sum()
    if tot < 1e-300:

        w[np.argmax(c)] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_autocorr(f):
    """Compute aperiodic autocorrelation r(m) = sum_l f[l]*f[l+m] for all m."""
    n = len(f)

    fsize2 = 1 << (2 * n - 1 + n - 1).bit_length()
    U = fft(u, fsize2)
    Fr = fft(f, fsize2)
    # cross-corr: sum_i lam[i]*f[k_i - j] = ifft(fft(u) * conj(fft(f)))[:n]
    cross = np.real(ifft(U * np.conj(Fr)))[:n]
    delta = -0.5 * (2.0 * cross + mu)
    
    # Ensure sum(delta) = 0 (numerical correction)
    delta -= np.mean(delta)
    
    return delta, active_idx


def run_lse_fista_v2(a_init, n, S, max_time, rng,
                     beta_start=20.0, beta_end=30000.0, beta_steps=300,
                     use_bb=True):
    """
    Improved LSE-FISTA with Barzilai-Borwein step and adaptive restart.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    
    prev_grad = None
    prev_y = None
    bb_step = 1.0 / beta_start
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 80000.0)
        
        # Compute convolution
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve_it = it
        
        # Compute LSE gradient
        grad, lse_val = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size
        if use_bb and prev_grad is not None and prev_y is not None:
            s_vec = y - prev_y
            y_vec = grad - prev_grad
            sy = np.dot(s_vec, y_vec)
            if abs(sy) > 1e-14:
                ss = np.dot(s_vec, s_vec)
                yy = np.dot(y_vec, y_vec)
                if ss > 1e-14 and yy > 1e-14:
                    # Alternate between BB1 and BB2
                    if it % 2 == 0:
                        bb_candidate = ss / sy
                    else:
                        bb_candidate = sy / yy
                    if 0.1 * (1.0/beta) < bb_candidate < 10.0 * (1.0/beta):
                        bb_step = bb_candidate
        step = min(bb_step, 2.0 / beta)
        
        # Stagnation handling
        stagnant = it - last_improve_it
        if stagnant > 300:
            t_mom = 1.0
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stagnant / 20000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve_it = it
            step *= 0.5
            bb_step = 1.0 / beta
        
        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-project to ensure feasibility
        y = project_simplex(y, S)
        a = x_new
        t_mom = t_new
        
        # Store for BB
        prev_grad = grad.copy()
        prev_y = y.copy()
        
        # Periodic check
        if it % 200 == 0:
            c_check = fft_convolve_self(y)
            if c_check.max() < best:
                best = c_check.max()
                best_a = y.copy()
                last_improve_it = it
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_newton_active_set(a_init, n, S, max_time, rng, max_active=30, 
                          tol=0.001, line_search_max=10):
    """
    Newton method with active set identification.
    Uses the structured G matrix for efficient direction computation.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # If stuck, add noise and reduce active set
        if no_improve > 50:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-no_improve / 500.0)
            if noise_scale < 1e-12:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            no_improve = 0
            continue
        
        # Newton step
        try:
            delta, active_idx = newton_active_set_step(a, c, max_active, S)
        except Exception:
            # Fallback to LSE gradient
            grad = active_set_gradient(a, c, beta_w=200.0)
            delta = grad - np.mean(grad)
        
        # Line search
        base_obj = cur_max
        accepted = False
        alpha = 1.0
        for ls in range(line_search_max):
            a_trial = a + alpha * delta
            a_trial = project_simplex(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            trial_obj = c_trial.max()
            if trial_obj < base_obj * (1.0 - 1e-7):
                a = a_trial
                accepted = True
                break
            alpha *= 0.5
        
        if not accepted:
            # Steepest descent fallback
            grad = active_set_gradient(a, c, beta_w=500.0)
            direction = -(grad - np.mean(grad))
            alpha = 0.01 * S / (np.linalg.norm(direction) + 1e-10)
            for ls in range(5):
                a_trial = project_simplex(a + alpha * direction, S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < base_obj:
                    a = a_trial
                    accepted = True
                    break
                alpha *= 0.5
            if not accepted:
                # Tiny random perturbation
                a = project_simplex(a + rng.normal(0, 1e-5 * S / n, n), S)
        
        # Convergence check
        if it > 100 and (best - cur_max) < 1e-8 * best:
            # Check if we're at a fixed point
            delta_norm = np.linalg.norm(delta)
            if delta_norm < 1e-10 * S:
                break
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_hybrid_v2(a_init, n, S, max_time, rng):
    """
    Hybrid: LSE-FISTA -> Newton Active Set -> Peak Shaving.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    def try_seq(seq):
        nonlocal best, best_a
        c = fft_convolve_self(seq)
        if c.max() < best:
            best = c.max()
            best_a = seq.copy()
        return best
    
    # Phase A: LSE-FISTA (40% of time)
    t_a = max_time * 0.4
    remaining = deadline_global - time.monotonic()
    t_a = min(t_a, remaining * 0.4)
    if t_a > 5:
        print("  Phase A: LSE-FISTA (%.0fs)" % t_a)
        a, va = run_lse_fista_v2(a, n, S, t_a, rng,
                                  beta_start=20.0, beta_end=50000.0, beta_steps=200)
        try_seq(a)
        print("  After Phase A: %.8f" % best)
    
    # Phase B: Newton Active Set (40% of time)
    remaining = deadline_global - time.monotonic()
    t_b = remaining * 0.45
    if t_b > 10:
        print("  Phase B: Newton Active Set (%.0fs)" % t_b)
        a, vb = run_newton_active_set(a, n, S, t_b, rng,
                                       max_active=25, line_search_max=8)
        try_seq(a)
        print("  After Phase B: %.8f" % best)
    
    # Phase C: Another round of LSE with high beta (20% of time)
    remaining = deadline_global - time.monotonic()
    t_c = remaining * 0.3
    if t_c > 5:
        print("  Phase C: High-beta LSE (%.0fs)" % t_c)
        a, vc = run_lse_fista_v2(a, n, S, t_c, rng,
                                  beta_start=5000.0, beta_end=80000.0, beta_steps=50)
        try_seq(a)
        print("  After Phase C: %.8f" % best)
    
    # Phase D: Short Newton refinement
    remaining = deadline_global - time.monotonic()
    t_d = remaining * 0.4
    if t_d > 5:
        print("  Phase D: Final Newton (%.0fs)" % t_d)
        a, vd = run_newton_active_set(a, n, S, t_d, rng,
                                       max_active=35, line_search_max=6)
        try_seq(a)
        print("  After Phase D: %.8f" % best)
    
    return best_a, best


def peak_shaving_v2(a, n, S, max_steps=200, n_peaks=10):
    """
    Improved peak shaving with responsibility-based updates.
    """
    a = a.copy()
    for step in range(max_steps):
        c = fft_convolve_self(a)
        cm = c.max()
        
        # Get top peaks
        top_m = min(n_peaks, len(c))
        top_idx = np.argsort(c)[::-1][:top_m]
        
        # If peaks are all very close, we're near optimal
        if c[top_idx[-1]] > cm * 0.999:
            break
        
        # Compute responsibility vector
        resp = np.zeros(n)
        for k in top_idx:
            weight = max(0.0, c[k] - cm * 0.995)
            if weight < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            np.add.at(resp, js, contrib * weight)
        
        resp_max = resp.max()
        if resp_max < 1e-20:
            break
        
        # Multiplicative update
        lr = 0.005 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)
    
    return a


def coordinate_descent_polish(a, n, S, max_iter=500):
    """
    Greedy coordinate descent: adjust individual coordinates to reduce max.
    """
    a = a.copy()
    for it in range(max_iter):
        c = fft_convolve_self(a)
        cm = c.max()
        k_star = np.argmax(c)
        
        # Find the coordinate j that, if changed, most reduces c[k_star]
        # c[k_star] = sum_j a[j]*a[k_star-j]
        # d c[k_star]/d a[j] = 2*a[k_star-j] (if valid)
        # We want to decrease a[j] where a[k_star-j] is large
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        js = np.arange(lo, hi + 1)
        partners = k_star - js
        grads = 2.0 * a[partners]
        
        # Find the j with maximum positive gradient (most to reduce)
        j_rel = np.argmax(grads)
        j = js[j_rel]
        
        # Try reducing a[j] and increasing some other a[m]
        # To keep sum constant, we need to increase another coordinate
        # Simplest: reduce a[j] by epsilon and increase a[0] (or another low-impact coord)
        eps = 0.01 * a[j]
        if eps < 1e-10:
            break
        
        a_trial = a.copy()
        a_trial[j] -= eps
        # Distribute eps to the coordinate with smallest impact on c[k_star]
        # Impact of a[m] on c[k_star] is 2*a[k_star-m] (if valid)
        impacts = np.zeros(n)
        for m in range(n):
            if 0 <= k_star - m < n:
                impacts[m] = 2.0 * a[k_star - m]
            else:
                impacts[m] = 0.0
        # Add eps to the coordinate with minimum impact
        m_best = np.argmin(impacts)
        a_trial[m_best] += eps
        a_trial = project_simplex(a_trial, S)
        
        c_trial = fft_convolve_self(a_trial)
        if c_trial.max() < cm:
            a = a_trial
        else:
            break
    
    return a


def generate_smart_init(n, rng):
    """Generate a smart initialization based on known good shapes."""
    t = np.linspace(0, 1, n)
    inits = []
    
    # Asymmetric raised cosine (key: asymmetry breaks the center peak)
    for cf in [0.3, 0.35, 0.4, 0.45]:
        for wf in [0.4, 0.5, 0.6]:
            width = wf / 2.0
            x = (t - cf) / width
            a = np.zeros(n)
            mask = np.abs(x) < 1
            a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
            inits.append(np.maximum(a, 1e-10))
    
    # Skewed Gaussian
    for mu in [0.35, 0.4, 0.45]:
        for sigma in [0.15, 0.2, 0.25]:
            a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
            inits.append(np.maximum(a, 1e-10))
    
    # Product of two bumps (creates a "valley" in the convolution)
    for mu1 in [0.25, 0.3]:
        for mu2 in [0.6, 0.7]:
            a = np.exp(-0.5 * ((t - mu1) / 0.15) ** 2) + np.exp(-0.5 * ((t - mu2) / 0.15) ** 2)
            inits.append(np.maximum(a, 1e-10))
    
    # Power law (compacts energy toward one end)
    for p in [0.5, 1.0, 2.0, 3.0]:
        inits.append(np.maximum(t ** p, 1e-10))
        inits.append(np.maximum((1 - t) ** p, 1e-10))
    
    return inits


deadline_global = 0.0


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Multi-phase: LSE-FISTA -> Newton Active Set -> Peak Shaving -> Polish.
    """
    global deadline_global
    
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 90.0
    deadline_global = time.monotonic() + max(1.0, budget_s - safety)
    
    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))
    
    # Load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded given sequence: length=%d" % len(given))
    except Exception:
        pass
    
    global_best = np.inf
    global_a = None
    
    def evaluate_internal(a):
        nonlocal global_best, global_a
        n = len(a)
        S = np.sqrt(2.0 * n)
        a = normalize_sum(a, n)
        c = fft_convolve_self(a)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a.copy()
            print("  ** New best: %.10f (n=%d, t=%.0fs)" % (val, n, time.monotonic() - (deadline_global - budget_s + safety)))
        return val
    
    # === Phase 1: Quick optimization of given sequence ===
    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        t1 = min(120.0, (deadline_global - time.monotonic()) * 0.15)
        print("=== Phase 1: LSE-FISTA on given (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_lse_fista_v2(given, n_g, S_g, t1, rng,
                                   beta_start=100.0, beta_end=10000.0, beta_steps=100)
        evaluate_internal(a1)
    
    # === Phase 2: Newton Active Set on best ===
    if global_a is not None:
        n_main = len(global_a)
        S_main = np.sqrt(2.0 * n_main)
        remaining = deadline_global - time.monotonic()
        t2 = remaining * 0.4
        print("=== Phase 2: Newton Active Set (n=%d, %.0fs) ===" % (n_main, t2))
        a2, v2 = run_newton_active_set(global_a, n_main, S_main, t2, rng,
                                        max_active=20, line_search_max=10)
        evaluate_internal(a2)
    
    # === Phase 3: Hybrid (LSE + Newton alternating) ===
    if global_a is not None:
        n_h = len(global_a)
        S_h = np.sqrt(2.0 * n_h)
        remaining = deadline_global - time.monotonic()
        t3 = remaining * 0.3
        if t3 > 30:
            print("=== Phase 3: Full Hybrid (n=%d, %.0fs) ===" % (n_h, t3))
            a3, v3 = run_hybrid_v2(global_a, n_h, S_h, t3, rng)
            evaluate_internal(a3)
    
    # === Phase 4: Peak shaving + coordinate descent ===
    if global_a is not None:
        n_p = len(global_a)
        S_p = np.sqrt(2.0 * n_p)
        remaining = deadline_global - time.monotonic()
        if remaining > 20:
            print("=== Phase 4: Peak shaving + CD (remaining %.0fs) ===" % remaining)
            a_p = peak_shaving_v2(global_a, n_p, S_p, max_steps=n_p // 4, n_peaks=15)
            evaluate_internal(a_p)
            
            if global_a is not None:
                a_cd = coordinate_descent_polish(global_a, n_p, S_p, max_iter=200)
                evaluate_internal(a_cd)
    
    # === Phase 5: Try different initialization ===
    remaining = deadline_global - time.monotonic()
    if remaining > 60:
        n_try = 8192
        S_try = np.sqrt(2.0 * n_try)
        t5 = remaining * 0.3
        print("=== Phase 5: Alternative init (n=%d, %.0fs) ===" % (n_try, t5))
        
        inits = generate_smart_init(n_try, rng)
        # Quick score
        init_scores = []
        for i, init in enumerate(inits[:15]):
            init = normalize_sum(init, n_try)
            c = fft_convolve_self(init)
            init_scores.append((c.max(), i))
        init_scores.sort()
        
        per_init = max(10.0, t5 / max(1, len(init_scores)))
        for score, idx in init_scores[:4]:
            if time.monotonic() > deadline_global - 30:
                break
            init = inits[idx]
            ai, vi = run_lse_fista_v2(init, n_try, S_try, per_init, rng,
                                       beta_start=30.0, beta_end=20000.0, beta_steps=150)
            evaluate_internal(ai)
    
    # === Phase 6: Final Newton polish ===
    if global_a is not None:
        n_f = len(global_a)
        S_f = np.sqrt(2.0 * n_f)
        remaining = deadline_global - time.monotonic()
        if remaining > 15:
            print("=== Phase 6: Final Newton polish (%.0fs) ===" % remaining)
            a_f, v_f = run_newton_active_set(global_a, n_f, S_f, remaining * 0.7, rng,
                                              max_active=40, line_search_max=6)
            evaluate_internal(a_f)
    
    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.10f, internal = %.10f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = None
    
    if global_a is None:
        print("WARNING: No valid result. Using uniform fallback.")
        n_fb = 8192
        global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2 * n_fb) / n_fb)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 8192
        global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2 * n_fb) / n_fb)
    
    result = [float(x) for x in global_a]
    print("RETURNING: length=%d, best_score=%.10f" % (len(global_a), global_best))
    return result
