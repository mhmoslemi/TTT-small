    c = np.real(ifft(A * A))
    return c[:size]

    """Compute a * b (full linear convolution) using FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:size]

    v = np.asarray(v, dtype=np.float64)

    return np.maximum(v - theta, 0.0)

    c = fft_convolve_self(a)
