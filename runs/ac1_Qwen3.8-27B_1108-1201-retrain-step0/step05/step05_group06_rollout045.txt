Minimize 2*n*max_conv/sum^2 using:
1. Nesterov-accelerated subgradient method with exact subgradient
2. Coordinate descent polish with exact line search
3. Adaptive smoothness (top-k subgradient averaging)
4. Multi-scale exploration (try different n)
5. LP-based alternating optimization (tug-of-war)

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def normalize_seq(a, S):
    """Normalize to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def subgradient_at_max(a, c, top_k=1):
    """Compute (sub)gradient of max(c) w.r.t. a.
    gradient_k[m] = 2 * a[k-m] for valid m (0 <= k-m < n)
    Average over top_k lags.
    """
    n = len(a)
    k_idx = np.argsort(c)[::-1][:top_k]
    grad = np.zeros(n)
    for k in k_idx:
        # grad[m] += (1/top_k) * 2 * a[k-m] for valid m
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            grad[m_start:m_end+1] += (2.0 / top_k) * a[k - m_end:k - m_start + 1][::-1]
    return grad

def nesterov_subgrad(a_init, max_time, top_k_schedule=None, verbose=True):
    """Nesterov-accelerated subgradient method."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(np.clip(a_init, 0, 1000), S).copy()
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    # Adaptive top_k
    if top_k_schedule is None:
        top_k_schedule = [(0, 1), (2000, 2), (5000, 4), (10000, 8), (20000, 16), (50000, 32)]
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        # Track best (using normalized objective)
        obj = cur_max  # since sum = S = sqrt(2n), obj = 2n*max/S^2 = 2n*max/(2n) = max
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Determine top_k
        tk = top_k_schedule[0][1]
        for threshold, k_val in top_k_schedule:
            if it >= threshold:
                tk = k_val
        # If stuck longer, increase k
        if stuck > 5000:
            tk = min(64, tk * 2)
        if stuck > 20000:
            tk = min(128, tk * 2)
        
        # Compute subgradient
        grad = subgradient_at_max(y, c, top_k=tk)
        
        # Lipschitz constant L = 2 (as derived above, per-active-lag)
        # For averaged top-k, it's still 2 (convex combination)
        L = 2.0
        step = 1.0 / L
        
        # Nesterov update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Ensure sum is maintained (projection already does this, but just in case)
        a = x_new
        t_mom = t_new
        
        # Periodic restart
        if it % 10000 == 0:
            t_mom = 1.0
        
        # Escape from local minima
        if stuck > 30000 and it % 1000 == 0:
            noise = np.random.randn(n) * 0.01 * S / np.sqrt(n)
            y = project_simplex(y + noise, S)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)
