"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses adaptive LSE-continuation FISTA with backtracking,
entropy-based beta scheduling, direct subgradient polishing,
and active-set refinement. Exploits symmetry for efficiency.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """LSE gradient w.r.t. a. d/d(a) [1/beta * log(sum(exp(beta*c_i)))] convolved back."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Gradient of LSE wrt a: 2 * convolve(w, a)[:n]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_at_max(a, c, top_k=1):
    """Subgradient using top-k convolution values."""
    n = len(a)
    k = min(top_k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def make_cosine_init(n):

    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)


def make_symmetric_cosine_init(n):
    """Symmetric cosine window."""
    half = n // 2
    tt = np.linspace(0, 0.5, half, endpoint=False)
    first = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    a = np.zeros(n)
    a[:half] = first
    a[half:] = first[::-1]
    return a


def make_random_symmetric_init(n, rng):

    first = rng.random(half) ** 2
    a = np.zeros(n)

def make_windowed_init(n, alpha=2.0):
    """Raised cosine window to the power alpha."""
    tt = np.linspace(0, 1, n)
    w = 0.5 + 0.5 * np.cos(2 * np.pi * tt)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1

    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        grad = subgrad_topk(y, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking on actual max
        step = step_cache
        accepted = False
        for _ in range(10):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-12 or step < 1e-16:
                accepted = True
                break
            step *= 0.5

        if not accepted:
            x_new = project_simplex(y - step * grad, S)

        if stuck > 500:
            t_mom = 1.0
            stuck = 0
            step_cache = step
        else:
            step_cache = min(step * 1.05, 0.1)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise escape
        if stuck > 2000 and it % 300 == 0:
            ns = 0.005 * S / np.sqrt(n) * np.exp(-stuck / 4000.0)
            if ns > 1e-10:
                y = project_simplex(y + rng.normal(0, ns, n), S)
                t_mom = 1.0

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a, best_conv


def annealed_polish(a_init, max_time, rng):
    """Alternating fine subgradient steps with small random perturbations."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        frac = min(1.0, (time.monotonic() - start) / max_time)

        # Adaptive k: start small, grow
        k = 1 + int(frac * 15)
        k = min(k, 32)

        grad = subgrad_topk(y, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Very small steps for polish
        step = 0.01 / (1.0 + 0.1 * it)
        step = min(step, S / (gnorm * 10.0))
        step = max(step, 1e-14)

        x_new = project_simplex(y - step * grad, S)

        if stuck > 100:
            t_mom = 1.0

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Random perturbation every 200 iters when stuck
        if stuck > 200 and it % 200 == 0:
            scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck / 5000.0)
            if scale > 1e-12:
                perturbed = project_simplex(y + rng.normal(0, scale, n), S)
                c_p = fft_convolve_self(perturbed)
                if c_p.max() < cur_max:
                    y = perturbed
                    t_mom = 1.0

        if it % 5000 == 0:
            t_mom = 1.0

    return best_a, best_conv


def structured_perturb(a_init, max_time, rng):
    """Try structured perturbations (low-frequency modes) of current best."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()

    c_best = fft_convolve_self(a)
    best_conv = c_best.max()

    # Try various frequency perturbations
    for mode in range(1, min(20, n // 2)):
        if time.monotonic() - start > max_time:
            break
        for amplitude in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            if time.monotonic() - start > max_time:
                break
            pert = a + amplitude * np.cos(2 * np.pi * mode * np.arange(n) / n)
            pert = np.maximum(pert, 0.0)
            pert = normalize_seq(pert, S)
            c_p = fft_convolve_self(pert)
            if c_p.max() < best_conv:
                best_conv = c_p.max()
                best_a = pert.copy()
                print("    Mode %d, amp %.4f: conv%.8f" % (mode, amplitude, c_p.max()))

    # Try random structured perturbations
    for _ in range(50):
        if time.monotonic() - start > max_time:
            break
        mode1 = rng.integers(1, n // 2)
        mode2 = rng.integers(1, n // 2)
        amp = rng.uniform(0.001, 0.01)
        pert = a + amp * (np.cos(2 * np.pi * mode1 * np.arange(n) / n) +
                           0.5 * np.sin(2 * np.pi * mode2 * np.arange(n) / n))
        pert = np.maximum(pert, 0.0)
        pert = normalize_seq(pert, S)
        c_p = fft_convolve_self(pert)
        if c_p.max() < best_conv:
            best_conv = c_p.max()
            best_a = pert.copy()

    return best_a, best_conv


def upsample_linear(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: diverse init -> LSE FISTA backtrack -> subgrad polish
    -> annealed polish -> structured perturb -> optional upsample.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 35.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    n = 8192
    S = np.sqrt(2.0 * n)

    # Generate diverse initializations
    inits = []
    if a_start is not None and len(a_start) == n:
        inits.append(normalize_seq(a_start, S))
    inits.append(normalize_seq(make_cosine_init(n), S))
    inits.append(normalize_seq(make_symmetric_cosine_init(n), S))
    inits.append(normalize_seq(make_windowed_init(n, 2.0), S))
    inits.append(normalize_seq(make_windowed_init(n, 1.5), S))
    inits.append(normalize_seq(make_random_symmetric_init(n, rng), S))

    # Quick evaluation
    best_init = None
    best_init_obj = np.inf
    for a_init in inits:
        obj = compute_obj(a_init)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_init
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a_init.copy()

    print("Best init obj: %.8f" % best_init_obj)
    print("Starting optimization, n=%d" % n)

    # Phase 1: LSE FISTA with backtracking (main optimization)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        b1 = remaining * 0.45
        print("\nPhase 1: LSE FISTA backtrack (budget %.0fs)" % b1)
        t1 = time.monotonic()
        a1, _ = fista_lse_backtrack(best_init, b1, rng,
                                     beta_start=3.0, beta_end=30000.0)
        obj1 = compute_obj(a1)
        print("  obj=%.8f (t=%.1fs)" % (obj1, time.monotonic() - t1))
        if obj1 < global_best_obj:
            global_best_obj = obj1
            global_best_a = a1.copy()

    # Phase 2: Subgradient with backtracking
    remaining = deadline - time.monotonic()
    if remaining > 30:
        b2 = remaining * 0.30
        print("\nPhase 2: Subgrad backtrack (budget %.0fs)" % b2)
        t2 = time.monotonic()
        a2, _ = fista_subgrad_backtrack(global_best_a, b2, rng,
                                         k_start=1, k_end=24)
        obj2 = compute_obj(a2)
        print("  obj=%.8f (t=%.1fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # Phase 3: Annealed polish (very fine steps + perturbations)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        b3 = remaining * 0.30
        print("\nPhase 3: Annealed polish (budget %.0fs)" % b3)
        t3 = time.monotonic()
        a3, _ = annealed_polish(global_best_a, b3, rng)
        obj3 = compute_obj(a3)
        print("  obj=%.8f (t=%.1fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # Phase 4: Structured perturbation search
    remaining = deadline - time.monotonic()
    if remaining > 15:
        b4 = remaining * 0.20
        print("\nPhase 4: Structured perturb (budget %.0fs)" % b4)
        t4 = time.monotonic()
        a4, _ = structured_perturb(global_best_a, b4, rng)
        obj4 = compute_obj(a4)
        print("  obj=%.8f (t=%.1fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # Phase 5: One more LSE round at high beta from best found
    remaining = deadline - time.monotonic()
    if remaining > 30:
        b5 = remaining * 0.50
        print("\nPhase 5: High-beta LSE polish (budget %.0fs)" % b5)
        t5 = time.monotonic()
        a5, _ = fista_lse_backtrack(global_best_a, b5, rng,
                                     beta_start=100.0, beta_end=200000.0)
        obj5 = compute_obj(a5)
        print("  obj=%.8f (t=%.1fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # Phase 6: Final annealed polish
    remaining = deadline - time.monotonic()
    if remaining > 15:
        b6 = remaining * 0.70
        print("\nPhase 6: Final anneal (budget %.0fs)" % b6)
        t6 = time.monotonic()
        a6, _ = annealed_polish(global_best_a, b6, rng)
        obj6 = compute_obj(a6)
        print("  obj=%.8f (t=%.1fs)" % (obj6, time.monotonic() - t6))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

    # Phase 7: Try upsample to 2n if time and size allow
    remaining = deadline - time.monotonic()
    if remaining > 120 and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        S2 = np.sqrt(2.0 * n2)
        b7 = remaining * 0.65
        print("\nPhase 7: Upsample to n=%d (budget %.0fs)" % (n2, b7))
        t7 = time.monotonic()
        a_up = upsample_linear(global_best_a, factor=2)
        a_up = normalize_seq(a_up, S2)

        a_up1, _ = fista_lse_backtrack(a_up, b7 * 0.5, rng,
                                         beta_start=5.0, beta_end=20000.0)
        obj_up = compute_obj(a_up1)
        print("  Upsampled LSE: obj=%.8f (t=%.1fs)" % (obj_up, time.monotonic() - t7))
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up1.copy()

        rem7 = b7 - (time.monotonic() - t7)
        if rem7 > 20:
            a_up2, _ = fista_subgrad_backtrack(global_best_a, rem7, rng,
                                                 k_start=1, k_end=16)
            obj_up2 = compute_obj(a_up2)
            print("  Upsampled subgrad: obj=%.8f (t=%.1fs)" % (obj_up2, time.monotonic() - t7))
            if obj_up2 < global_best_obj:
                global_best_obj = obj_up2
                global_best_a = a_up2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver) and ver < 100.0:
            print("\n=== FINAL RESULT ===")
            print("Objective: %.8f (n=%d)" % (ver, len(final_list)))
            return final_list
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")

    # Fallback
    a_fb = np.ones(8192) * (np.sqrt(2 * 8192) / 8192)
    return [float(x) for x in a_fb]
