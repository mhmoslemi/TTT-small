"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: L2 pre-conditioning + adaptive LSE-FISTA + multi-restart.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def next_pow2(m):
    return 1 << (m - 1).bit_length()

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    L = next_pow2(size)
    A = fft(a, L)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w

def next_pow2(m):
    return 1 << (max(m, 1) - 1).bit_length()


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    L = next_pow2(size)
    A = fft(a, L)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

                global_best_obj = obj2b
                global_best_a = a2b.copy()

        rem2b = rem2 - (time.monotonic() - t2 - rem2 * 0.0) if rem2 > 0 else 0
        rem2b = p2_budget - (time.monotonic() - t2)
        if rem2b > 15:
            a2c, obj2c = run_lse_fista(global_best_a, rem2b, rng, beta_start=50.0, adaptive=False)
            print("  LSE-fista (ramp): obj=%.8f" % obj2c)
            if obj2c < global_best_obj:
                global_best_obj = obj2c
                global_best_a = a2c.copy()

    # === Phase 3: Top-k FISTA (20% time) ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.20
    if p3_budget > 30:
        print("\n=== Phase 3: Top-k FISTA (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()
        a3, obj3 = run_topk_fista(global_best_a, p3_budget, rng)
        print("  Top-k FISTA: obj=%.8f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Multi-restart exploration (15% time) ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.15
    if p4_budget > 30:
        print("\n=== Phase 4: Multi-restart (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        # Generate diverse perturbations of current best
        n_restarts = 4
        time_per = p4_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            if time_per < 10:
                break
            # Perturb best with smooth noise
            noise = rng.normal(0, 1.0, n0)
            F = fft(noise, n0)
            cutoff = max(3, n0 // 80)
            F[cutoff:-cutoff] *= 0.03
            noise_s = np.real(ifft(F))[:n0]
            S_curr = np.sqrt(2.0 * n0)
            scale = rng.uniform(0.02, 0.08)
            a_pert = global_best_a + scale * noise_s
            a_pert = np.maximum(a_pert, 0.0)
            if a_pert.sum() > 1e-10:
                a_pert = a_pert * (S_curr / a_pert.sum())
                a4, obj4 = run_lse_fista(a_pert, time_per * 0.9, rng, beta_start=50.0, adaptive=True)
                print("  Restart %d: obj=%.8f" % (r, obj4))
                if obj4 < global_best_obj:
                    global_best_obj = obj4
                    global_best_a = a4.copy()

        # Try a different n value
        rem4 = p4_budget - (time.monotonic() - t4)
        if rem4 > 30:
            n_alt = 16384
            S_alt = np.sqrt(2.0 * n_alt)
            t_alt = np.linspace(0, 1, n_alt, endpoint=False)
            a_alt = 0.5 + 0.5 * np.cos(2 * np.pi * t_alt)
            print("  Trying n=%d, remaining %.0fs" % (n_alt, rem4))
            a_alt_l2, _ = run_l2_phase(a_alt, rem4 * 0.3, rng)
            a_alt_fista, obj_alt = run_lse_fista(a_alt_l2, rem4 * 0.6, rng, beta_start=50.0, adaptive=True)
            print("  n=%d: obj=%.8f" % (n_alt, obj_alt))
            if obj_alt < global_best_obj:
                global_best_obj = obj_alt
                global_best_a = a_alt_fista.copy()
                n0 = n_alt

    # === Phase 5: Final polish (5% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % remaining)
        t5 = time.monotonic()
        a5, obj5 = run_lse_fista(global_best_a, remaining * 0.6, rng, beta_start=200.0, adaptive=True)
        print("  High-beta polish: obj=%.8f" % obj5)
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

        rem5 = remaining - (time.monotonic() - t5)
        if rem5 > 10:
            a5b, obj5b = run_topk_fista(global_best_a, rem5, rng)
            print("  Top-k polish: obj=%.8f" % obj5b)
            if obj5b < global_best_obj:
                global_best_obj = obj5b
                global_best_a = a5b.copy()

    # Final verification and return
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
            global_best_obj = 2.0
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
