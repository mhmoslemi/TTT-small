def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(y, beta):
    """Compute LSE value and gradient of beta*LSE(c) w.r.t. y."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def topk_subgradient(a, c, k):
    """Subgradient from top-k lags of the convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def coordinate_descent(a_init, max_time, rng, max_pairs_per_iter=8):
    """
    Direct coordinate descent via mass transfer.
    Transfers small amounts of mass from high-contribution to low-contribution
    coordinates to reduce the peak of the autocorrelation.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 20000:
            break

        k_star = int(np.argmax(c))
        # Contribution of coordinate j to c[k_star]:
        # a[j] appears in c[k_star] as a[j]*a[k_star-j] if 0<=k_star-j<n
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi >= lo:
            contrib[lo:hi+1] = a[k_star - hi:k_star - lo + 1][::-1]

        # Also consider the top few lags (not just argmax)
        top_k = min(3, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        multi_contrib = np.zeros(n)
        for kk in top_idx:
            lo2 = max(0, int(kk) - n + 1)
            hi2 = min(n - 1, int(kk))
            if hi2 >= lo2:
                multi_contrib[lo2:hi2+1] += a[int(kk) - hi2:int(kk) - lo2 + 1][::-1]

        # Select source (j): high contribution, relatively large value
        # Select target (l): low contribution, relatively small value
        # Score: we want to move from high multi_contrib to low multi_contrib
        src_scores = multi_contrib * np.maximum(a, 0.001)
        tgt_scores = -multi_contrib * (S / n - a + 0.001)

        # Try a few candidates
        n_try = min(max_pairs_per_iter, n)
        src_candidates = np.argsort(src_scores)[::-1][:n_try]
        tgt_candidates = np.argsort(-tgt_scores)[:n_try]

        best_delta_obj = cur_obj + 1e-15
        best_j, best_l, best_delta = -1, -1, 0.0

        for j in src_candidates[:n_try // 2]:
            for l in tgt_candidates[:n_try // 2]:
                if j == l or a[j] < 1e-10:
                    continue
                # Try transferring a small fraction of a[j]
                delta = a[j] * 0.05
                if delta < 1e-12:
                    continue
                # Compute new max approximately
                # c changes by: -delta*a[k-j] + delta*a[k-l] for affected k
                delta_c = np.zeros(2 * n - 1)
                # Effect of reducing a[j] by delta
                js = max(0, j)
                je = min(2 * n - 2, j + n - 1)
                coeff_j = a[max(0, js - j):min(n, je - j + 1)]
                delta_c[js:je+1] -= delta * coeff_j
                # Effect of increasing a[l] by delta
                ls = max(0, l)
                le = min(2 * n - 2, l + n - 1)
                coeff_l = a[max(0, ls - l):min(n, le - l + 1)]
                delta_c[ls:le+1] += delta * coeff_l

                new_max = np.max(c + delta_c)
                new_obj = 2.0 * n * new_max / (S * S)
                if new_obj < best_delta_obj:
                    best_delta_obj = new_obj
                    best_j, best_l, best_delta = j, l, delta

        if best_j >= 0:
            j, l, delta = best_j, best_l, best_delta
            a[j] -= delta
            a[l] += delta
            # Update c directly
            js = max(0, j)
            je = min(2 * n - 2, j + n - 1)
            c[js:je+1] -= delta * a[max(0, js - j):min(n, je - j + 1)]
            ls = max(0, l)
            le = min(2 * n - 2, l + n - 1)
            c[ls:le+1] += delta * a[max(0, ls - l):min(n, le - l + 1)]
        else:
            # No improvement found; try a random small perturbation
            if stuck > 1000:
                idx = rng.integers(0, n)
                delta = a[idx] * 0.01
                if delta > 1e-12:
                    a[idx] -= delta
                    a[rng.integers(0, n)] += delta
                    # Recompute c (expensive but needed after random change)
                    c = fft_convolve_self(a)

        # Periodic full recompute to avoid drift
        if it % 100 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def run_subgradient_polish(a_init, max_time, rng):
    """Polish using exact subgradient with decreasing step size."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck < 200:
            k = 1
        elif stuck < 1000:
            k = 2
        elif stuck < 5000:
            k = min(8, 2 + stuck // 500)
        else:
            k = min(32, 8 + stuck // 1000)

        grad = topk_subgradient(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.002 * it))
        step = min(step, 2.0 * S / gnorm)

        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-13:
            a = a_new
        else:
            step *= 0.1
            a_new = project_simplex(a - step * grad, S)
            if compute_obj(a_new) <= cur_obj + 1e-13:
                a = a_new

        if stuck > 3000 and it % 100 == 0:
            ns = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 8000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.15 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 4 * tt + 0.7)
    elif method == "multi_tone":
        a = 0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt + 1.3) + 0.15 * np.sin(2 * np.pi * 11 * tt + 2.1)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 2.0 * np.abs(tt - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())
