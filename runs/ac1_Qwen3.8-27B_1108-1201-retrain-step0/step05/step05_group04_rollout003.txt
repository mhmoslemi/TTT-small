import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c). Returns (grad, lse_value)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def penalty_gradient(a, c, T, active_top=20):

    Gradient of quadratic penalty: sum_k max(0, c[k]-T)^2 / (2*active_count).
    Only consider top 'active_top' lags above T for efficiency and stability.

    # Find lags above T, take top active_top
    above = c > T
    if not np.any(above):
        return np.zeros(n), 0.0

    above_vals = c[above_idx]
    if len(above_vals) > active_top:

        above_idx = above_idx[top_local]

    excess = c[above_idx] - T  # all positive
    count = float(len(above_idx))
    obj = np.sum(excess ** 2) / (2.0 * count)

    W = fft(w, fsize)

        active_idx = active_idx[np.argsort(c[active_idx])[::-1][:n_active]]
    w = np.zeros(2 * n - 1)
    w[active_idx] = 1.0 / len(active_idx)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm


def fourier_noise(n, rng, scale):
    """Generate smooth noise in the spatial domain via random Fourier phase."""
    fsize = 1 << (n - 1).bit_length()
    A = np.zeros(fsize, dtype=np.complex128)
    k = np.arange(fsize)
    # Low-frequency noise for smoothness
    mask = k < n // 8
    A[mask] = rng.normal(0, scale, mask.sum()) + 1j * rng.normal(0, scale, mask.sum())
    A[fsize - mask[::-1][1:]] = np.conj(A[mask][1:]
)
    noise = np.real(ifft(A))[:n]
    return noise


def run_lse_fista(a_init, n, S, max_time, rng,
                  beta_start=30.0, beta_end=30000.0, beta_steps=200,
                  momentum_restart_gap=300, noise_period=500):
    """
    FISTA with LSE smoothing and adaptive beta.
    Returns (best_a, best_val).
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stall_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        # Plateau schedule: beta stays high for a while before increasing
        if frac < 0.3:
            beta = beta_start * (1 + 2 * frac / 0.3)
        elif frac < 0.6:
            beta = beta_start * 3.0 * (1 + (frac - 0.3) / 0.3)
        elif frac < 0.85:
            beta = beta_start * 9.0 * (1 + 2 * (frac - 0.6) / 0.25)
        else:
            beta = beta_start * 27.0 * (1 + (frac - 0.85) / 0.15)
        beta = min(beta, 50000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stall_count = 0
        else:
            stall_count += 1

        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta

        # Adaptive step reduction on stall
        if stall_count > momentum_restart_gap:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stall_count / 2000.0)
            if noise_scale > 1e-9:
                noise = fourier_noise(n, rng, noise_scale)
                y = project_simplex(y + noise, S)
            last_improve = it
            step *= 0.5
            stall_count = 0

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_penalty_refinement(a_init, n, S, max_time, rng,
                           T_start_delta=0.005, T_min_delta=0.0005,
                           T_decay=0.995, active_top=20):
    """
    Quadratic penalty refinement: minimize sum(max(0, c[k]-T)^2) with decreasing T.
    This provides a smooth gradient for the last mile.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    c0 = fft_convolve_self(a)
    cur_max = c0.max()
    best = cur_max
    best_a = a.copy()

    T = cur_max + T_start_delta
    T_floor = cur_max * (1.0 - T_min_delta / max(cur_max, 1e-10))

    y = a.copy()
    t_mom = 1.0
    it = 0
    last_improve = 0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Decay T toward the floor
        T = max(T * T_decay, T_floor)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stall = 0
        else:
            stall += 1

        grad, pen_obj = penalty_gradient(y, c, T, active_top=active_top)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            # Try direct subgradient step
            grad, _ = active_set_subgradient(y, c, n_active=5)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

        # Step size: adaptive based on penalty curvature
        step = 1.0 / (n * max(pen_obj + 1e-10, 0.01))
        step = min(step, 0.1)
        step = max(step, 1e-6)

        if stall > 200:
            t_mom = 1.0
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-stall / 1000.0)
            if noise_scale > 1e-10:
                noise = fourier_noise(n, rng, noise_scale)
                y = project_simplex(y + noise, S)
            last_improve = it
            step *= 0.5
            stall = 0

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_active_refinement(a_init, n, S, max_time, rng):
    """
    Direct active-set subgradient with backtracking line search.
    For the final "last mile" where we need very precise optimization.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    stall = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            last_improve = it
            stall = 0
        else:
            stall += 1

        # Adaptive active set size: start broad, narrow down
        if stall < 100:
            n_active = 8
        elif stall < 500:
            n_active = 5
        else:
            n_active = 3

        grad, _ = active_set_subgradient(a, c, n_active=n_active, n_fallback=2)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        base_obj = cur_max
        step = 0.005 / (1 + 0.001 * it)
        improved = False
        for _ in range(10):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            test_obj = c_test.max()
            if test_obj < base_obj - 1e-9:
                a = a_test
                improved = True
                break
            step *= 0.5

        if not improved:
            # Tiny step in a random smooth direction
            noise = fourier_noise(n, rng, 0.0005 * S / np.sqrt(n))
            a = project_simplex(a + noise, S)

        if stall > 300:
            # More aggressive escape
            noise = fourier_noise(n, rng, 0.003 * S / np.sqrt(n))
            a = project_simplex(a + noise, S)
            stall = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def peak_shaving(a, n, S, n_peaks=6, n_steps=80, lr_scale=0.008):
    """
    Surgical peak shaving: reduce contributions to top peaks via multiplicative updates.
    """
    a = normalize_sum(a, n).copy()
    for step_i in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()
        top_idx = np.argsort(c)[::-1][:n_peaks]

        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            weight = max(0.0, c[k] - cm * 0.995)
            np.add.at(resp, js, contrib * weight)

        resp_max = resp.max()
        if resp_max < 1e-30:
            break

        lr = lr_scale * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)

    return a


def try_sequence(a, state):
    """Evaluate a sequence and update global best if improved."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a, n)
    c = fft_convolve_self(a)
    val = c.max()
    if val < state['best']:
        state['best'] = val
        state['best_a'] = a.copy()
        print("  New best: %.10f (n=%d, it context: %s)" % (val, n, state.get('context', '')))
    return val


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Multi-phase: LSE-FISTA -> Penalty refinement -> Active-set -> Peak shaving.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    print("Budget: %.0fs" % budget_s)

    # Load previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception as e:
        print("Could not load previous: %s" % str(e))
        pass

    state = {'best': np.inf, 'best_a': None, 'context': ''}

    def _try(a, context=''):
        state['context'] = context
        try_sequence(a, state)

    # === Phase 1: Quick LSE on given sequence ===
    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        t1 = min(40.0, (deadline - time.monotonic()) * 0.05)
        print("=== Phase 1: Quick LSE (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_lse_fista(given, n_g, S_g, t1, rng,
                                beta_start=100.0, beta_end=8000.0, beta_steps=80)
        _try(a1, "phase1")

    # If we don't have a good starting point, generate one
    if state['best_a'] is None:
        n_main = 8192
        S_main = np.sqrt(2.0 * n_main)
        # Raised cosine init centered off-center
        t = np.linspace(0, 1, n_main)
        center = 0.35
        width = 0.3
        x = (t - center) / width
        a_init = np.ones(n_main)
        mask = np.abs(x) < 1
        a_init[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
        a_init = np.maximum(a_init, 1e-10)
        a_init = normalize_sum(a_init, n_main)
        print("Generated initial sequence (n=%d)" % n_main)
        _try(a_init, "init")

    n_main = len(state['best_a'])
    S_main = np.sqrt(2.0 * n_main)

    # === Phase 2: Main LSE-FISTA ===
    remaining = deadline - time.monotonic()
    t2 = remaining * 0.40
    print("=== Phase 2: Main LSE-FISTA (n=%d, %.0fs, best=%.8f) ===" %
          (n_main, t2, state['best']))
    a2, v2 = run_lse_fista(state['best_a'], n_main, S_main, t2, rng,
                            beta_start=50.0, beta_end=25000.0, beta_steps=250)
    _try(a2, "phase2")

    # === Phase 3: Penalty refinement (key for last mile) ===
    remaining = deadline - time.monotonic()
    t3 = remaining * 0.35
    print("=== Phase 3: Penalty refinement (%.0fs, best=%.8f) ===" %
          (t3, state['best']))
    a3, v3 = run_penalty_refinement(state['best_a'], n_main, S_main, t3, rng,
                                     T_start_delta=0.004, T_min_delta=0.0003,
                                     T_decay=0.996, active_top=15)
    _try(a3, "phase3")

    # === Phase 4: Active-set subgradient (last mile) ===
    remaining = deadline - time.monotonic()
    t4 = remaining * 0.30
    print("=== Phase 4: Active-set refinement (%.0fs, best=%.8f) ===" %
          (t4, state['best']))
    a4, v4 = run_active_refinement(state['best_a'], n_main, S_main, t4, rng)
    _try(a4, "phase4")

    # === Phase 5: Peak shaving + polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20 and state['best_a'] is not None:
        n_p = len(state['best_a'])
        S_p = np.sqrt(2.0 * n_p)
        print("=== Phase 5: Peak shaving (%.0fs) ===" % remaining)
        a5 = peak_shaving(state['best_a'], n_p, S_p, n_peaks=6, n_steps=60, lr_scale=0.006)
        _try(a5, "peak_shave")

        # LSE polish after shaving
        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            a5b, _ = run_lse_fista(a5, n_p, S_p, remaining2 * 0.5, rng,
                                    beta_start=5000.0, beta_end=40000.0, beta_steps=50)
            _try(a5b, "post_shave_lse")

    # === Phase 6: Final active-set on best ===
    remaining = deadline - time.monotonic()
    if remaining > 15 and state['best_a'] is not None:
        n_f = len(state['best_a'])
        S_f = np.sqrt(2.0 * n_f)
        print("=== Phase 6: Final active-set (%.0fs) ===" % remaining)
        a6, _ = run_active_refinement(state['best_a'], n_f, S_f, remaining * 0.7, rng)
        _try(a6, "final_active")

    # === Phase 7: Penalty on latest best ===
    remaining = deadline - time.monotonic()
    if remaining > 15 and state['best_a'] is not None:
        n_f = len(state['best_a'])
        S_f = np.sqrt(2.0 * n_f)
        print("=== Phase 7: Final penalty (%.0fs) ===" % remaining)
        a7, _ = run_penalty_refinement(state['best_a'], n_f, S_f, remaining * 0.8, rng,
                                        T_start_delta=0.002, T_min_delta=0.0002,
                                        T_decay=0.997, active_top=10)
        _try(a7, "final_penalty")

    # Final verification
    if state['best_a'] is not None:
        final_list = [float(x) for x in np.clip(state['best_a'], 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.10f, internal = %.10f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, state['best'], len(state['best_a'])))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            state['best_a'] = np.ones(8192, dtype=np.float64) * (np.sqrt(2 * 8192) / 8192)
    else:
        print("WARNING: No valid result, using uniform fallback (n=8192).")
        state['best_a'] = np.ones(8192, dtype=np.float64) * (np.sqrt(2 * 8192) / 8192)

    state['best_a'] = np.clip(state['best_a'], 0.0, 1000.0)
    if state['best_a'].sum() < 0.01:
        state['best_a'] = np.ones(8192, dtype=np.float64) * (np.sqrt(2 * 8192) / 8192)

    print("RETURNING: length=%d, best_score=%.10f" % (len(state['best_a']), state['best']))
    return [float(x) for x in state['best_a']]
