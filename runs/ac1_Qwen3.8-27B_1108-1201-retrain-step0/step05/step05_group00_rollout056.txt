"""
Minimize 2n*max_conv/sum^2 using GS flat-spectrum initialization + adaptive LSE/L-BFGS + direct de-peak.

Key innovations:
1. Gerchberg-Saxton alternating projections for flat-spectrum non-negative initialization
2. Larger sequence length (16384) for lower theoretical bound
3. Direct peak-removal steps interleaved with LSE refinement
4. Adaptive spectrum target accounting for non-negativity
5. Multi-scale initialization (coarse-to-fine)
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    if np.all(v <= 0):
        return np.full(n, S / n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = (theta_candidates <= w) & (idx > 0)
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    return result


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_grad(a, N, beta):
    """Gradient of LSE-smoothed peak of conv(a,a)."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    if beta < 1e-6:
        weights = np.ones(len(c)) / len(c)
        lse_val = c_max
    else:
        shifted = beta * (c - c_max)
        shifted = np.clip(shifted, -700, 700)
        exp_bc = np.exp(shifted)
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:
            weights = exp_bc / total
            lse_val = c_max + np.log(total) / beta
    fsize = N
    if fsize < 2 * n:
        fsize = 1 << (2 * n).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def gs_flat_spectrum(a_init, n, S, N, n_iters, rng, lam=0.3):
    """
    Gerchberg-Saxton iterations to flatten the amplitude spectrum.
    Alternates between:
      - Time domain: clip to non-negative, fix sum
      - Frequency domain: partially flatten |A| toward target
    """
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
        return a
    a *= S / s

    for it in range(n_iters):
        A = fft(a, N)
        mag = np.abs(A)
        
        # Target: partial flattening
        mean_mag = np.mean(mag[1:]) if N > 1 else mag[0]
        if it < 50:
            # Gentle start
            eff_lam = lam * (it / 50.0)
        else:
            eff_lam = lam
        
        # New magnitude: interpolate between current and flat target
        target = np.full(N, mean_mag)
        target[0] = A[0].real  # DC = sum(a), must preserve
        new_mag = (1 - eff_lam) * mag + eff_lam * target
        new_mag[0] = mag[0]
        
        # Update
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new * (S / s_new)
    
    return a


def gs_with_target_shape(a_init, n, S, N, n_iters, rng, lam=0.2, low_freq_boost=1.5):
    """GS with a target that's boosted at low frequencies (to aid non-negativity)."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
        return a
    a *= S / s

    m_idx = np.arange(N)
    # Target shape: boost at low frequencies
    target_shape = np.ones(N)
    target_shape[0] = S / np.mean(np.abs(fft(a, N))[1:]) if N > 1 else S
    bandwidth = max(1, N // 16)
    for m in range(1, min(bandwidth, N)):
        target_shape[m] = low_freq_boost * (1 - m / bandwidth) + (1 - (1 - m / bandwidth) * (low_freq_boost - 1))
    # Symmetric
    target_shape[N // 2:] = target_shape[N // 2:0:-1] if N % 2 == 0 else target_shape[N // 2:0:-1]

    for it in range(n_iters):
        A = fft(a, N)
        mag = np.abs(A)
        mean_mag = np.mean(mag[1:]) if N > 1 else mag[0]
        
        target = mean_mag * target_shape / np.mean(target_shape[1:]) if N > 1 else np.array([S])
        target[0] = S
        
        eff_lam = min(lam, 0.5 * (it + 1) / 100)
        new_mag = (1 - eff_lam) * mag + eff_lam * target
        new_mag[0] = S
        
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new * (S / s_new)
    
    return a


def depeak_iterative(a, n, S, N, n_iters, rng, top_k=50):
    """Direct iterative peak reduction on the self-convolution."""
    a = a.copy()
    for it in range(n_iters):
        c = compute_conv(a, N)
        c_max = c.max()
        k_star = int(np.argmax(c))
        
        # Compute gradient of c[k_star] w.r.t. a
        # d c[k*] / d a[j] = a[k*-j] (if valid) + a[k*-j] (if k*-j != j and valid)
        grad = np.zeros(n)
        for j in range(n):
            other = k_star - j
            if 0 <= other < n:
                grad[j] = a[other]
                if other != j:
                    grad[j] += a[other]
        
        # Top-k positions with highest gradient (contribute most to peak)
        if np.max(grad) < 1e-30:
            break
        top_idx = np.argsort(grad)[::-1][:top_k]
        
        # Create direction: decrease top-k, increase others
        direction = np.zeros(n)
        direction[top_idx] = -1.0
        rest = np.setdiff1d(np.arange(n), top_idx)
        if len(rest) > 0:
            direction[rest] = top_k / len(rest)
        
        # Line search
        best_alpha = 0.0
        best_cmax = c_max
        for alpha_try in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            a_try = project_simplex(a + alpha_try * direction, S)
            c_try = compute_conv(a_try, N)
            if c_try.max() < best_cmax:
                best_cmax = c_try.max()
                best_alpha = alpha_try
        
        if best_alpha == 0.0:
            break
        a = project_simplex(a + best_alpha * direction, S)
    
    return a

def spectral_restart(a, N, rng, frac=0.5):
    """Randomize a fraction of FFT phases."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def nesterov_toppk(a_init, n, S, N, max_time, rng, top_k_init=50, top_k_max=512):
    """Nesterov accelerated top-k subgradient."""
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)
        c = compute_conv(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        fsize = N
        w_full = np.zeros(fsize)
        for i, k in enumerate(top_idx):
            if k < fsize:
                w_full[int(k)] = weights[i]
        W = fft(w_full, fsize)
        Af = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y_s = y.sum()
        if y_s > 1e-15:
            y = y * (S / y_s)
        a = x_new
        t = t_new
        
        if it % 500 == 0 and rng.random() < 0.2:
            y = spectral_restart(y, N, rng, frac=0.2)
            t = 1.0
    
    return best_a, best_cmax


def make_random_init(n, rng):
    return rng.random(n) + 0.01


def make_chirp_init(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.3 * np.sin(2 * np.pi * (3 * tt + 0.5 * tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bump_init(n, rng):
    """Gaussian bump centered."""
    tt = np.linspace(0, 1, n)
    center = 0.5
    sigma = 0.2 + 0.1 * rng.random()
    a = np.exp(-0.5 * ((tt - center) / sigma) ** 2)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_arcosine_init(n, rng):
    """Arcsine distribution (concentrated at edges)."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 / np.sqrt(tt * (1 - tt) + 1e-6)
    a = a / np.max(a)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_multi_bump_init(n, rng):
    """Multiple Gaussian bumps."""
    a = np.zeros(n)
    n_bumps = rng.integers(3, 8)
    for _ in range(n_bumps):
        center = rng.random()
        sigma = rng.uniform(0.05, 0.2)
        tt = np.linspace(0, 1, n)
        a += np.exp(-0.5 * ((tt - center) / sigma) ** 2)
    a += 0.05
    return a

        print("  Phase 3: De-peak + LSE alternation")
        t3 = time.monotonic()
        depeak_budget = min(400.0, (deadline - t3) * 0.2)
        
        a_curr = best_a.copy()
        for round_i in range(5):
            if time.monotonic() > t3 + depeak_budget:
                break
            # De-peak step
            dp_time = min(30.0, (t3 + depeak_budget) - time.monotonic())
            if dp_time > 5:
                a_dp = depeak_iterative(a_curr, n, S, N, int(dp_time * 50), rng, top_k=100)
                c_dp = compute_conv(a_dp, N).max()
                if c_dp < best_cmax:
                    best_cmax = c_dp
                    best_a = a_dp.copy()
                    a_curr = a_dp.copy()
                    print("    De-peak r%d: score=%.6f" % (round_i, 2.0 * n * c_dp / (S * S)))
            
            # LSE polish
            lse_time = min(60.0, (t3 + depeak_budget) - time.monotonic())
            if lse_time > 10:
                a_l, c_l = lbfgs_lse_block(a_curr, n, S, N, lse_time, rng,
                                            beta_init=20.0, beta_max=200.0, m_lbfgs=8)
                if c_l < best_cmax:
                    best_cmax = c_l
                    best_a = a_l.copy()
                    a_curr = a_l.copy()
                    print("    LSE r%d: score=%.6f" % (round_i, 2.0 * n * c_l / (S * S)))
        
        print("  After de-peak: best score=%.6f" % (2.0 * n * best_cmax / (S * S)))
        
        # Phase 4: Nesterov polishing
        print("  Phase 4: Nesterov polishing")
        t4 = time.monotonic()
        nesterov_budget = min(200.0, (deadline - t4) * 0.15)
        if nesterov_budget > 20:
            a_ng, c_ng = nesterov_toppk(best_a, n, S, N, nesterov_budget, rng,
                                         top_k_init=50, top_k_max=300)
            if c_ng < best_cmax:
                best_cmax = c_ng
                best_a = a_ng.copy()
                print("    Nesterov: score=%.6f" % (2.0 * n * c_ng / (S * S)))
        
        # Phase 5: Final high-beta L-BFGS
        t5 = time.monotonic()
        final_budget = deadline - t5 - 10
        if final_budget > 30:
            a_fin, c_fin = lbfgs_lse_block(best_a, n, S, N, min(final_budget, 200.0), rng,
                                            beta_init=30.0, beta_max=600.0, m_lbfgs=12)
            if c_fin < best_cmax:
                best_cmax = c_fin
                best_a = a_fin.copy()
                print("    Final LSE: score=%.6f" % (2.0 * n * c_fin / (S * S)))
        
        # Update global best
        final_score = 2.0 * n * best_cmax / (S * S)
        print("  n=%d final score: %.6f" % (n, final_score))
        
        if final_score < global_best_score:
            global_best_score = final_score
            global_best_seq = best_a.copy()
            print("  *** New global best: %.6f ***" % global_best_score)
    
    # Verify and return
    if global_best_seq is not None:
        global_best_seq = np.clip(global_best_seq, 0.0, 1000.0)
        if global_best_seq.sum() < 0.01:
            global_best_seq = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_best_seq]
        ver = evaluate_sequence(final_list)
        print("Verification score: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_best_seq = np.ones(256, dtype=np.float64)
    else:
        global_best_seq = np.ones(256, dtype=np.float64)
    
    print("Returning length %d" % len(global_best_seq))
    return [float(x) for x in global_best_seq]
