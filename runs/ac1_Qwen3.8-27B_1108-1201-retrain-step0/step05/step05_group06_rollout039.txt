"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses LSE-FISTA with adaptive beta, power-p polishing, peak-shaving local search,
and multi-n exploration. Key improvements over previous:
- Backtracking step size for robustness
- Power-p objective for intermediate polishing
- Direct peak-shaving local search
- Better beta schedule (geometric)
- Symmetric sequence exploitation
- Multi-n exploration (4096, 8192, 16384)

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def power_p_gradient(y, c, p):
    """Gradient of sum(c^p) w.r.t. a."""
    n = len(y)
    w = np.maximum(c, 0.0) ** (p - 1.0)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return p * 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(y, c, k=1, tol_frac=0.0):
    """Subgradient from top-k or tolerance-based active set."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        k = min(k, len(c))
        active = np.argsort(c)[::-1][:k]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_fista_adaptive(a_init, max_time, rng, beta_init=10.0, beta_max=50000.0,
                       beta_multiplier=1.005, use_backtrack=True):
    """LSE-FISTA with adaptive beta and optional backtracking step size."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj_val = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj_val - 1e-12:
            best_obj_val = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Geometric beta increase
        if stuck < 200:
            beta = max(beta, beta_init)
        elif stuck < 2000:
            beta = min(beta_max, beta * beta_multiplier)
        elif stuck < 10000:
            beta = min(beta_max, beta * 1.01)
        else:
            beta = min(beta_max, beta * 1.05)

        # Gradient computation
        if beta > 1000 and stuck > 500:
            # Switch to top-k subgradient for stability
            k = min(10, 1 + stuck // 2000)
            grad = topk_subgradient(y, c, k=k, tol_frac=0.001)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        if use_backtrack:
            # Conservative initial step
            L_est = 2.0 * beta + 2.0 * np.sum(np.maximum(c, 0) ** (1.0)) / n
            step = 1.0 / max(L_est, 1.0)
            step = min(step, 1.0 / gnorm)

            # Backtracking (simplified: just use the conservative step)
            x_new = project_simplex(y - step * grad, S)

            # Check if objective decreased (using LSE as proxy)
            c_new = fft_convolve_self(x_new)
            if c_new.max() > cur_max * 1.01 and stuck < 100:
                # Step too aggressive, halve it
                step *= 0.5
                x_new = project_simplex(y - step * grad, S)
        else:
            step = 1.0 / max(2.0 * beta + 1.0, gnorm)
            x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Renormalize y to maintain feasibility approximately
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        a = x_new
        t_mom = t_new

        # Periodic restart
        if it % 5000 == 0:
            t_mom = 1.0

        # Noise when very stuck
        if stuck > 8000 and it % 300 == 0:
            noise_scale = max(1e-7, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 15000.0))
            noise = rng.normal(0, noise_scale, n)
            y = project_simplex(y + noise, S)
            a = y.copy()
            t_mom = 1.0
            stuck = 0
            last_improve = it

    c_final = fft_convolve_self(best_a)
    final_obj = 2.0 * n * c_final.max() / (best_a.sum() ** 2)
    return best_a, final_obj

    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()

        if cur_obj < best_obj_val - 1e-12:

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)

    return best_a, best_obj_val


def peak_shaving(a_init, max_time, rng, max_rounds=50):
    """Local peak-shaving search: directly target the worst autocorrelation peak."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_obj_val = compute_obj(a)
    best_a = a.copy()
    round_num = 0

    while time.monotonic() - start < max_time and round_num < max_rounds:
        round_num += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj_val - 1e-12:
            best_obj_val = cur_obj
            best_a = a.copy()

        # Find the worst peak
        k_star = int(np.argmax(c))
        # Find contributing pairs
        # c[k_star] = sum_i a[i] * a[k_star - i]
        # The largest contributor is the pair (i, k_star-i) with max product

        # Try to reduce the peak by moving mass from high-contributing positions
        # to low-contributing positions
        improved = False
        for attempt in range(20):
            if time.monotonic() - start > max_time:
                break

            # Find the index i that maximizes a[i] * a[k_star - i]
            valid_i = []
            contrib = []
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    prod = a[i] * a[j]
                    if prod > 0:
                        valid_i.append(i)
                        contrib.append(prod)

            if not valid_i:
                break

            valid_i = np.array(valid_i)
            contrib = np.array(contrib)

            # Find the top contributor
            idx_max = valid_i[np.argmax(contrib)]
            j_max = k_star - idx_max

            # Find a target position to move mass to (where contribution is low)
            # We want to move mass from idx_max to some position p where
            # a[p] is small and a[k_star - p] is also small
            scores = a.copy()
            if 0 <= k_star - idx_max < n:
                scores[idx_max] = -1e9  # Don't move to the peak contributor
            # Prefer positions with low a[p] * a[k_star - p]
            target_scores = np.zeros(n)
            for p in range(n):
                q = k_star - p
                if 0 <= q < n:
                    target_scores[p] = a[p] * a[q]
                else:
                    target_scores[p] = -1  # Moving here reduces the peak
            target_scores[idx_max] = -1e9
            if j_max != idx_max:
                target_scores[j_max] = -1e9

            # Pick the position with lowest target score (least contribution to peak)
            # but also not too large currently
            p_target = int(np.argmin(target_scores))

            if p_target == idx_max:
                continue

            # Move a small amount of mass
            delta = 0.01 * a[idx_max]
            if delta > a[idx_max]:
                delta = 0.005 * a[idx_max]
            if delta < 1e-10:
                continue

            a_trial = a.copy()
            a_trial[idx_max] -= delta
            a_trial[p_target] += delta

            if a_trial[idx_max] < 0:
                a_trial[idx_max] = 0
                a_trial[p_target] += a_trial[idx_max]  # adjust

            # Check if this improves the objective
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_max - 1e-10:
                a = a_trial
                improved = True
                # Continue shaving

        if not improved:
            # Try a random perturbation
            noise_scale = 0.002 * S / np.sqrt(n)
            a = normalize_seq(a + rng.normal(0, noise_scale, n), S)
            c_check = fft_convolve_self(a)
            if c_check.max() > cur_max * 1.05:
                a = best_a.copy()

    return best_a, best_obj_val


def subgradient_polish(a_init, max_time, rng):
    """Polish using exact/top-k subgradient with diminishing step size."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_obj_val = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step_scale = 0.5 * S

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj_val - 1e-12:
            best_obj_val = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive top-k
        if stuck < 100:
            k = 1
            tol = 0.0
        elif stuck < 500:
            k = min(5, 1 + stuck // 100)
            tol = 0.001
        elif stuck < 2000:
            k = min(20, 5 + stuck // 200)
            tol = 0.005
        else:
            k = min(50, 20 + stuck // 100)
            tol = 0.01

        if tol > 0:
            active = np.where(c >= c.max() * (1.0 - tol))[0]
            if len(active) > k:
                active = np.argsort(c)[::-1][:k]
        else:
            active = np.argsort(c)[::-1][:k]

        w_full = np.zeros(len(c))
        w_full[active] = 1.0 / len(active)
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Diminishing step size (Polyak-like)
        step = step_scale / (1.0 + 0.01 * it)
        step = min(step, S / gnorm)

        a_new = project_simplex(a - step * grad, S)

        # Accept if it doesn't worsen too much
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max * 1.001:
            a = a_new
        else:
            step *= 0.5
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max:
                a = a_new

        if stuck > 5000 and it % 200 == 0:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            last_improve = it

    return best_a, best_obj_val


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.54 + 0.46 * np.cos(2 * np.pi * tt)


def make_hann_init(n):
    tt = np.arange(n)
    return 0.5 * (1.0 + np.cos(2 * np.pi * tt / (n - 1))) if n > 1 else np.ones(1)


def make_blackman_init(n):
    tt = np.arange(n)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * tt / (n - 1)) + 0.08 * np.cos(4 * np.pi * tt / (n - 1))
    return np.maximum(a, 0.01) if n > 1 else np.ones(1)


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging."""
    n = len(a)
    n_new = n // factor
    a_padded = a[:n_new * factor]
    return a_padded.reshape(n_new, factor).mean(axis=1)

                global_best_a = a2.copy()

        print("  Phase 2 total time: %.1fs" % (time.monotonic() - t2))

    # === Phase 3: Peak shaving local search ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.20
    if p3_budget > 10:
        print("\n=== Phase 3: Peak shaving (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()
        a3, obj3 = peak_shaving(global_best_a, p3_budget, rng)
        print("  Peak shaving: obj=%.6f (t=%.1fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Subgradient polish ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.20
    if p4_budget > 10:
        print("\n=== Phase 4: Subgradient polish (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        a4, obj4 = subgradient_polish(global_best_a, p4_budget, rng)
        print("  Subgradient: obj=%.6f (t=%.1fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Random restarts ===
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.10
    if p5_budget > 15:
        print("\n=== Phase 5: Random restarts (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()
        n_restarts = min(5, int(p5_budget / 20))
        for r in range(n_restarts):
            rem5 = p5_budget - (time.monotonic() - t5)
            if rem5 < 15:
                break
            # Perturb the best sequence
            noise_scale = 0.01 * S / np.sqrt(n)
            a_pert = normalize_seq(global_best_a + rng.normal(0, noise_scale, n), S)
            a5, obj5 = lse_fista_adaptive(a_pert, min(rem5 * 0.8, 60), rng,
                                           beta_init=10.0, beta_max=5000.0,
                                           beta_multiplier=1.005)
            print("  Restart %d: obj=%.6f" % (r + 1, obj5))
            if obj5 < global_best_obj:
                global_best_obj = obj5
                global_best_a = a5.copy()
        print("  Phase 5 total time: %.1fs" % (time.monotonic() - t5))

    # === Phase 6: Multi-n exploration ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print("\n=== Phase 6: Multi-n exploration (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()

        # Try 2n (upsample)
        if len(global_best_a) <= 8192:
            n2 = len(global_best_a) * 2
            budget_2n = min(remaining * 0.4, 300)
            print("  Trying n=%d (budget %.0fs)..." % (n2, budget_2n))
            a_up = upsample(global_best_a, factor=2)
            S2 = np.sqrt(2.0 * n2)
            a_up = normalize_seq(a_up, S2)
            a_up1, obj_up1 = lse_fista_adaptive(a_up, budget_2n, rng,
                                                  beta_init=5.0, beta_max=5000.0,
                                                  beta_multiplier=1.005)
            print("  n=%d: obj=%.6f (t=%.1fs)" % (n2, obj_up1, time.monotonic() - t6))
            if obj_up1 < global_best_obj:
                global_best_obj = obj_up1
                global_best_a = a_up1.copy()

        # Try n/2 (downsample) if n is large enough
        remaining6 = deadline - time.monotonic()
        if remaining6 > 60 and len(global_best_a) > 4096:
            n_half = len(global_best_a) // 2
            budget_half = min(remaining6 * 0.5, 120)
            print("  Trying n=%d (budget %.0fs)..." % (n_half, budget_half))
            a_ds = downsample(global_best_a, factor=2)
            S_half = np.sqrt(2.0 * n_half)
            a_ds = normalize_seq(a_ds, S_half)
            a_ds1, obj_ds1 = lse_fista_adaptive(a_ds, budget_half, rng,
                                                  beta_init=5.0, beta_max=5000.0,
                                                  beta_multiplier=1.005)
            # Peak shave the downsampled result
            a_ds2, obj_ds2 = peak_shaving(a_ds1, 30, rng)
            print("  n=%d: obj=%.6f (t=%.1fs)" % (n_half, obj_ds2, time.monotonic() - t6))
            if obj_ds2 < global_best_obj:
                global_best_obj = obj_ds2
                global_best_a = a_ds2.copy()

        print("  Phase 6 total time: %.1fs" % (time.monotonic() - t6))

    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 7: Final polish (budget %.0fs) ===" % remaining)
        t7 = time.monotonic()

        # Peak shaving
        a7a, obj7a = peak_shaving(global_best_a, remaining * 0.4, rng)
        print("  Final peak shave: obj=%.6f" % obj7a)
        if obj7a < global_best_obj:
            global_best_obj = obj7a
            global_best_a = a7a.copy()

        # High-beta LSE
        rem7 = remaining - (time.monotonic() - t7) - 5
        if rem7 > 10:
            a7b, obj7b = lse_fista_adaptive(global_best_a, rem7, rng,
                                             beta_init=200.0, beta_max=100000.0,
                                             beta_multiplier=1.01)
            print("  Final LSE: obj=%.6f" % obj7b)
            if obj7b < global_best_obj:
                global_best_obj = obj7b
                global_best_a = a7b.copy()

        # One more peak shave
        rem7 = deadline - time.monotonic()
        if rem7 > 10:
            a7c, obj7c = peak_shaving(global_best_a, rem7, rng)
            print("  Final shave 2: obj=%.6f" % obj7c)
            if obj7c < global_best_obj:
                global_best_obj = obj7c
                global_best_a = a7c.copy()

    # Final verification
    print("\n=== Final result: obj=%.6f, n=%d ===" % (global_best_obj, len(global_best_a)))

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified objective: %.6f" % ver)
            return final_list
        else:
            print("WARNING: evaluate_sequence rejected!")

    # Fallback
    n_fb = 8192
    a_fb = make_cosine_init(n_fb)
    a_fb = normalize_seq(a_fb, np.sqrt(2.0 * n_fb))
    final_list = [float(x) for x in a_fb]
    print("Using fallback, obj=%.6f" % evaluate_sequence(final_list))
    return final_list
