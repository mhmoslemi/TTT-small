import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

"""
Minimize 2*n*max_conv/sum^2 using p-norm FISTA with annealing,
peak-directed coordinate descent, and multi-scale refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def pnorm_and_grad(a, p):
    """Compute p-norm of self-convolution and gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve_self(a)
    c = np.maximum(c, 0.0)  # Clip tiny negatives from FFT
    
    if p >= 1000:
        # Effectively the max
        val = c.max()
        idx = np.argmax(c)
        w = np.zeros(len(c))
        w[idx] = 1.0
    else:
        cp = c ** p
        sum_cp = cp.sum()
        val = sum_cp ** (1.0 / p)
        # grad[i] = 2 * sum_k w[k] * a[k-i] where w[k] = p * c[k]^(p-1) / sum_cp^(1-1/p)
        # Simplified: w[k] = p * c[k]^(p-1) / sum_cp (we'll handle the constant in step size)
        w = p * c ** (p - 1) / sum_cp
    
    # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j] (cross-correlation)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return val, grad, c


def run_pnorm_fista(a_init, max_time, rng, p_start=8.0, p_end=10000.0, 
                    p_rate=0.02, step_scale=1.0):
    """FISTA with p-norm smoothing and p-annealing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        p = min(p_end, p_start * np.exp(p_rate * it))
        
        val, grad, c = pnorm_and_grad(y, p)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        step = step_scale / max(p * 1.0, 10.0)
        step = min(step, 2.0 / max(p, 1.0))
        
        x_new = project_simplex(y - step * grad, S)
        
        # Backtracking
        obj_new = compute_obj(x_new)
        bt = 0
        while obj_new > cur_obj + 1e-12 and step > 1e-14 and bt < 4:
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
            obj_new = compute_obj(x_new)
            bt += 1
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        # Restart momentum if stuck
        if it - last_improve > 3000:
            t_mom = 1.0
            noise = max(1e-6, 0.002 * S / np.sqrt(n))

    return best_a, best_obj

One more idea: **spectral approach**. The self-convolution b = a*a has Fourier transform B(f) = |A(f)|^2. If we want b to be flat (value v = S^2/(2n-1)), we need |A(f)|^2 = v * (2n-1) * delta... no, that's not right for linear convolution.

For linear convolution, the relationship between the Fourier transform of b and a is more complex due to the windowing. But in the "overlap-add" sense, the DFT of the linear convolution is related to the DFT of a.

Actually, for the full linear convolution b = convolve(a, a) (length 2n-1), the DFT is B(f) = A(f)^2 (where A is the DFT of a zero-padded to length 2n-1). So |B(f)| = |A(f)|^2.

If we want b to be flat, we need B(f) to be a delta at f=0, which means A(f) = 0 for all f≠0. This means a is a DC signal (constant), which we already know gives a triangular (not flat) convolution.

The resolution: the DFT of the ZERO-PADDED a is not the same as the DFT of a itself. The zero-padding introduces the "triangular" envelope.

So the problem is fundamentally about designing a non-negative sequence whose zero-padded DFT is as "flat" as possible (in the sense that |A_pad(f)|^2 is as constant as possible away from f=0).

This is related to "flat-top" window design in signal processing! The flat-top window is designed to have a very flat main lobe in the frequency domain.

"""
Minimize 2*n*max_conv/sum^2 using p-norm FISTA with adaptive annealing,
peak-directed coordinate descent with verification, and multi-scale refinement.
Key improvements over previous: better p-norm scheduler, verified CD steps,
diverse structural initializations, and optional larger-n exploration.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0.0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def pnorm_val_and_grad(a, p):
    """
    Compute p-norm of self-convolution and its gradient w.r.t. a.
    Returns (pnorm_value, gradient, convolution).
    """
    n = len(a)
    c = fft_convolve_self(a)
    c = np.maximum(c, 0.0)

    if p >= 500:
        val = c.max()
        idx = int(np.argmax(c))
        w = np.zeros(len(c))
        w[idx] = 1.0
    else:
        cp = c ** p
        sum_cp = cp.sum()
        if sum_cp < 1e-300:
            val = c.max()
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            val = sum_cp ** (1.0 / p)
            w = p * c ** (p - 1) / sum_cp

    # Gradient: grad[i] = 2 * sum_j w[i+j] * a[j] (cross-correlation of w and a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    return val, grad, c


def run_fista_pnorm(a_init, max_time, rng, p_start=4.0, p_end=2000.0,
                    p_rate=0.015, step_base=1.0, use_bt=True):
    """
    FISTA with p-norm smoothing and p-annealing.
    Uses backtracking line search for stability.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_base / max(p_start, 2.0)

    while time.monotonic() - start < max_time:
        it += 1
        p = min(p_end, p_start * np.exp(p_rate * it))

        val, grad, c = pnorm_val_and_grad(y, p)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_bt:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-12 and bt < 5:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 2:
                step = step_local * 0.7
            else:
                step = min(step * 1.02, 4.0 / max(p, 2.0))
            x_new = x_trial
        else:
            step = step_base / max(p, 2.0)
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        if it - last_improve > 4000:
            t_mom = 1.0
            noise_scale = max(1e-6, 0.003 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    return best_a, best_obj


def run_cd_verified(a_init, max_time, rng, top_k=4, n_cands=12, fracs=None):
    """
    Coordinate descent with verified full-convolution check.
    Proposes transfers based on top-lag contributions, verifies with FFT.
    """
    if fracs is None:
        fracs = [0.03, 0.06, 0.10, 0.15]

    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        if it - last_improve > 20000:
            # Escalate: try random restart with noise
            noise_scale = max(1e-5, 0.005 * S / np.sqrt(n))
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if time.monotonic() - start < max_time and it - last_improve > 40000:
                break

        # Compute contribution vector from top lags
        top_lags = np.argsort(c)[::-1][:top_k]
        contrib = np.zeros(n)
        for k in top_lags:
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            partners = a[ki - hi:ki - lo + 1][::-1]
            contrib[lo:hi + 1] += partners

        # Score: source = high contrib * high value; target = low contrib
        src_score = contrib * a
        tgt_score = -contrib - 0.001 * a  # small penalty for already-large targets

        src_cands = np.argsort(src_score)[::-1][:n_cands]
        tgt_cands = np.argsort(tgt_score)[:n_cands]

        best_trial_obj = cur_obj + 1e-15
        best_move = None

        for j in src_cands[:8]:
            if a[j] < 1e-10:
                continue
            j = int(j)
            for l in tgt_cands[:8]:
                l = int(l)
                if j == l:
                    continue
                for frac in fracs:
                    delta = a[j] * frac
                    if delta < 1e-12 or delta > a[j]:
                        continue
                    a[j] -= delta
                    a[l] += delta
                    c_new = fft_convolve_self(a)
                    obj_new = 2.0 * n * c_new.max() / (S * S)
                    a[j] += delta  # revert
                    a[l] -= delta
                    if obj_new < best_trial_obj:
                        best_trial_obj = obj_new
                        best_move = (j, l, delta, c_new)

            a[j] -= delta
            a[l] += delta
        else:
            # No improvement from structured moves; try a small random move
            if it - last_improve > 500:
                j = int(rng.integers(0, n))
                l = int(rng.integers(0, n))
                if j != l and a[j] > 1e-10:
                    delta = a[j] * 0.02
                    a[j] -= delta
                    a[l] += delta
                    c = fft_convolve_self(a)

def run_subgrad_polish(a_init, max_time, rng):
    """Subgradient method with top-k lags and decreasing step size."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 2
        elif stuck < 2000:
            k = min(8, 2 + stuck // 200)
        else:
            k = min(32, 8 + stuck // 500)

        top_idx = np.argsort(c)[::-1][:k]
        w = np.zeros(len(c))
        # Weight by how close each lag is to the max
        weights = np.exp(-0.5 * (c[top_idx] - c[max(top_idx)]) / max(c[max(top_idx)] * 0.01, 1e-10))
        w[top_idx] = weights / weights.sum()

        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * np.sqrt(1.0 + 0.01 * it))
        step = min(step, 3.0 * S / gnorm)

        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-13:
            a = a_new
        else:
            a_new = project_simplex(a - 0.1 * step * grad, S)
            if compute_obj(a_new) <= cur_obj + 1e-13:
                a = a_new

        if stuck > 2000 and it % 200 == 0:
            ns = max(1e-6, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 10000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj


def make_init(n, method, rng):

def upsample_interp(a, n_new):

            global_best_obj = obj_cd
            global_best_a = a_cd.copy()

        # FISTA refinement after CD
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            a_f, obj_f = run_fista_pnorm(global_best_a, remaining3 * 0.4, rng,
                                         p_start=30, p_end=3000, p_rate=0.008,
                                         step_base=0.5, use_bt=True)
            print("  FISTA after CD: obj=%.6f" % obj_f)
            if obj_f < global_best_obj:
                global_best_obj = obj_f
                global_best_a = a_f.copy()

        # Second CD pass
        remaining3 = deadline - time.monotonic()
        if remaining3 > 15:
            a_cd2, obj_cd2 = run_cd_verified(global_best_a, remaining3 * 0.5, rng,
                                             top_k=3, n_cands=8,
                                             fracs=[0.02, 0.05, 0.10])
            print("  CD pass 2: obj=%.6f" % obj_cd2)
            if obj_cd2 < global_best_obj:
                global_best_obj = obj_cd2
                global_best_a = a_cd2.copy()

    # === Phase 4: Multi-scale exploration (18% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.18
    if p4_time > 45:
        # Try upsample to 2x
        n2 = len(global_best_a) * 2
        print("\n=== Phase 4: Multi-scale n=%d (budget %.0fs) ===" % (n2, p4_time))

        a_up = upsample_interp(global_best_a, n2)
        S2 = np.sqrt(2.0 * n2)
        a_up = a_up * (S2 / a_up.sum())
        obj_up = compute_obj(a_up)
        print("  Upsampled: obj=%.6f" % obj_up)

        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up.copy()

        a_up_opt, obj_up_opt = run_fista_pnorm(a_up, p4_time * 0.4, rng,
                                               p_start=4, p_end=300, p_rate=0.02,
                                               step_base=0.8, use_bt=False)
        print("  FISTA upsampled: obj=%.6f" % obj_up_opt)
        if obj_up_opt < global_best_obj:
            global_best_obj = obj_up_opt
            global_best_a = a_up_opt.copy()

        remaining4 = deadline - time.monotonic()
        if remaining4 > 20:
            a_cd_up, obj_cd_up = run_cd_verified(global_best_a, remaining4 * 0.5, rng,
                                                 top_k=3, n_cands=8,
                                                 fracs=[0.03, 0.07])
            print("  CD upsampled: obj=%.6f" % obj_cd_up)
            if obj_cd_up < global_best_obj:
                global_best_obj = obj_cd_up
                global_best_a = a_cd_up.copy()

    # === Phase 5: Final polish (12% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.12
    if p5_time > 10:
        print("\n=== Phase 5: Final Polish (budget %.0fs) ===" % p5_time)

        a_pol, obj_pol = run_subgrad_polish(global_best_a, p5_time * 0.6, rng)
        print("  Subgradient: obj=%.6f" % obj_pol)
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()

        remaining5 = deadline - time.monotonic()
        if remaining5 > 5:
            a_pol2, obj_pol2 = run_fista_pnorm(global_best_a, remaining5, rng,
                                               p_start=500, p_end=100000, p_rate=0.002,
                                               step_base=0.2, use_bt=True)
            print("  High-p final: obj=%.6f" % obj_pol2)
            if obj_pol2 < global_best_obj:
                global_best_obj = obj_pol2
                global_best_a = a_pol2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal: obj=%.6f, n=%d" % (ver, len(final_list)))
            global_best_obj = ver
        else:
            print("WARNING: invalid, using fallback")
            global_best_a = make_init(n0, "flattop", rng)
            final_list = [float(x) for x in global_best_a]
    else:
        global_best_a = make_init(n0, "flattop", rng)
        final_list = [float(x) for x in global_best_a]

    return final_list
