Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses LP-based direction finding (cutting plane) combined with 
gradient polishing and LSE-FISTA for local refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog, minimize as scipy_minimize


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Compute np.convolve(a, b) via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    # = sum_{m} a[k-m] * g[m] (with 0<=k-m<n, 0<=m<n)
    # So the coefficient of g[m] in c[k] is a[k-m] (if 0<=k-m<n)
    
    # For a given constraint row k, the non-zero entries are at columns m
    # where 0 <= k-m < n, i.e., k-n < m <= k, intersected with [0, n-1].
    # So m ranges from max(0, k-n+1) to min(k, n-1).
    
    # Initial active set: just the peak lag
    peak_k = int(np.argmax(fft_convolve_self(a)))
    active_set = [peak_k]
    
    best_g = None
    best_obj = 0.0
    
    for cp_iter in range(max_cp_iter):
        # Build constraint matrix for active constraints
        num_constraints = len(active_set)
        A_ub = np.zeros((num_constraints, n))
        
        for i, k in enumerate(active_set):
            # Coefficient of g[m] in c[k] is a[k-m] if 0 <= k-m < n
            m_start = max(0, k - n + 1)
            m_end = min(k, n - 1)
            if m_start <= m_end:
                # g[m_start..m_end] has coefficients a[k-m_start..k-m_end]
                # a[k-m] for m in [m_start, m_end]
                # = a[k-m_end .. k-m_start] (reversed)
                coeffs = a[k - m_end : k - m_start + 1][::-1]
                A_ub[i, m_start:m_end+1] = coeffs
        
        b_ub = np.full(num_constraints, M)
        
        # Objective: maximize sum(g) => minimize -sum(g)
        c_obj = -np.ones(n)
        
        # Solve LP
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='HiGHS')
        
        if not result.success:
            break
        
        g = result.x
        obj_val = g.sum()
        
        if obj_val < best_obj + 1e-10:
            best_obj = obj_val
            best_g = g.copy()
        
        # Check for violated constraints
        c_full = fft_cross_convolve(a, g)
        violations = c_full - M
        if violations.max() < tol:
            break
        
        # Add the most violated constraint
        violated_k = int(np.argmax(violations))
        if violated_k not in active_set:
            active_set.append(violated_k)
        else:
            # Already in active set but violated - numerical issue
            break
        
        if len(active_set) > 500:
            # Too many constraints, stop
            break
    
    if best_g is None:
        best_g = np.zeros(n)
    
    return best_g, len(active_set)


def line_search_max_conv(a, g, t_max=1.0, n_eval=20):
    """
    Find t in [0, t_max] that minimizes max(((1-t)*a + t*g) * ((1-t)*a + t*g)).
    """
    # Precompute the three convolutions
    c_aa = fft_convolve_self(a)
    c_ag = fft_cross_convolve(a, g)
    c_gg = fft_convolve_self(g)
    
    def f(t):
        return np.max((1-t)**2 * c_aa + 2*t*(1-t) * c_ag + t**2 * c_gg)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    lo = 0.0
    hi = t_max
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    
    for _ in range(n_eval):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    
    t_opt = (lo + hi) / 2
    return t_opt, f(t_opt)


def run_lp_iteration(a, S, max_cp_iter=100):
    """
    One LP-based iteration: find direction g, do line search, update a.
    Returns updated a and the new max convolution.
    """
    n = len(a.normalized() if hasattr(a, 'normalized') else a)
    # Ensure sum(a) = S
    a = a.copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)
    
    c = fft_convolve_self(a)
    M = c.max()
    
    # Solve LP for direction
    g, num_active = lp_solve_direction(a, M, max_cp_iter=max_cp_iter)
    
    g_sum = g.sum()
    if g_sum < 1e-15:
        return a, M
    
    # Rescale g to have sum S
    g = g * (S / g_sum)
    
    # Line search
    t_opt, f_val = line_search_max_conv(a, g, t_max=1.0, n_eval=30)
    
    # Update
    a_new = (1 - t_opt) * a + t_opt * g
    a_new = np.maximum(a_new, 0.0)
    
    # Verify
    c_new = fft_convolve_self(a_new)
    M_new = c_new.max()
    
    return a_new, M_new


def run_fista_lse(a_init, max_time, rng, beta=50.0, step_scale=1.0):
    """FISTA with LSE smoothing for local refinement."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
        
        # LSE weights
        cm = c.max()
        shifted = beta * (c - cm)
        shifted = shifted - shifted.max()
        e = np.exp(shifted)
        w = e / e.sum()
        
        # Gradient: 2 * cross-correlation of w and y
        n_c = len(w)
        L = 1 << (n_c + n - 1 - 1).bit_length()
        y_rev = y[::-1]
        conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
        cross = conv_wy[n - 1: 2 * n - 1]
        grad = 2.0 * cross
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = step_scale / max(beta, 1.0)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        if it % 5000 == 0:
            t_mom = 1.0
    
    return best_a

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Primary method: LP-based direction finding with cutting plane (from the
state-of-the-art description), combined with FISTA-LSE polishing,
adaptive restarts, and multi-scale spectral initialization.
Key innovation over previous code: the LP finds the OPTIMAL direction
at each step (not just a gradient), avoiding poor local minima that
gradient methods get trapped in.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def build_constraint_row(a, k, n):
    """Build the row of the constraint matrix for lag k.
    C[k,m] = a[k-m] if 0 <= k-m < n, else 0.
    Returns (start_col, coeffs) where coeffs fills columns start_col..start_col+len(coeffs)-1.
    """
    m_start = max(0, k - n + 1)
    m_end = min(k, n - 1)
    if m_start > m_end:
        return m_start, np.array([])
    # Coefficient of g[m] in c[k] is a[k-m] for m in [m_start, m_end]
    coeffs = a[k - m_end : k - m_start + 1][::-1].copy()
    return m_start, coeffs


def lp_solve_direction(a, M, max_cp_iter=150, tol=1e-9, prev_active=None):
    """
    Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Uses cutting plane method with active set management.
    Returns (g, active_set, num_lp_solves).
    """
    n = len(a)

    # Initialize active set
    if prev_active is not None and len(prev_active) > 0:
        active_set = list(prev_active[:100])
    else:
        c_a = fft_convolve_self(a)
        peak_k = int(np.argmax(c_a))
        # Include a few lags near the peak
        peaks = np.argsort(c_a)[::-1][:5]
        active_set = list(peaks)

    best_g = None
    best_obj = 0.0
    num_solves = 0

    for cp_iter in range(max_cp_iter):
        num_solves += 1
        num_constraints = len(active_set)

        # Build constraint matrix
        A_ub = np.zeros((num_constraints, n))
        for i, k in enumerate(active_set):
            m_start, coeffs = build_constraint_row(a, k, n)
            if len(coeffs) > 0:
                A_ub[i, m_start:m_start + len(coeffs)] = coeffs

        b_ub = np.full(num_constraints, M)
        c_obj = -np.ones(n)

        # Solve LP with HiGHS
        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                             bounds=[(0, None)] * n, method='HiGHS')
        except Exception:
            break

        if not result.success:
            if result.message == 'Iteration limit reached':
                if result.x is not None:
                    g = np.maximum(result.x, 0)
                else:
                    break
            else:
                break
        else:
            g = result.x

        obj_val = g.sum()
        if obj_val < best_obj + 1e-12:
            best_obj = obj_val
            best_g = g.copy()
        else:
            best_obj = obj_val
            best_g = g.copy()

        # Check for violated constraints via FFT
        c_check = fft_cross_convolve(a, g)
        violations = c_check - M
        max_viol = violations.max()

        if max_viol < max(tol * M, 1e-12):
            break

        # Add most violated constraint(s)
        num_add = min(5, max(1, int(np.sum(violations > 0.5 * max_viol))))
        violated = np.argsort(violations)[::-1][:num_add]
        for vk in violated:
            vk = int(vk)
            if vk not in active_set:
                active_set.append(vk)

        # Limit active set size - drop least violated
        if len(active_set) > 200:
            # Keep the most violated 200
            viol_arr = np.array([c_check[k] - M for k in active_set])
            keep = np.argsort(viol_arr)[::-1][:200]
            active_set = [active_set[i] for i in keep]

    if best_g is None:
        best_g = np.zeros(n)

    return best_g, active_set, num_solves


def line_search_max_conv(a, g, t_max=1.0, n_eval=30):
    """
    Find t in [0, t_max] minimizing max(((1-t)*a + t*g) * ((1-t)*a + t*g)).
    Uses golden section search.
    """
    c_aa = fft_convolve_self(a)
    c_ag = fft_cross_convolve(a, g)
    c_gg = fft_convolve_self(g)

    def f(t):
        return np.max((1 - t) ** 2 * c_aa + 2 * t * (1 - t) * c_ag + t ** 2 * c_gg)

    # Try t=1 first (full replacement)
    f1 = f(1.0)
    f0 = f(0.0)

    if f1 <= f0:
        # t=1 is better, but check if there's something even better
        # Use golden section on [0.3, 1.0]
        lo, hi = 0.3, 1.0
    else:
        lo, hi = 0.0, 1.0

    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)

    for _ in range(n_eval - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)

    t_opt = (lo + hi) / 2
    return t_opt, f(t_opt)


def run_lp_phase(a_init, S, max_time, rng, verbose=True):
    """Run LP-based direction finding iterations."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    active_set = None
    iter_count = 0
    stall_count = 0

    while time.monotonic() - start < max_time:
        iter_count += 1
        it_start = time.monotonic()
        rem = max_time - (time.monotonic() - start)
        if rem < 3:
            break

        c = fft_convolve_self(a)
        M = c.max()

        # Solve LP
        g, active_set, n_solves = lp_solve_direction(a, M, max_cp_iter=200,
                                                     prev_active=active_set)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break

        # Rescale g to have sum S
        g = g * (S / g_sum)

        # Check if g is essentially same as a (stuck)
        diff = np.linalg.norm(g - a) / (np.linalg.norm(a) + 1e-15)
        if diff < 1e-8:
            stall_count += 1
            if stall_count > 20:
                break
            continue

        stall_count = 0

        # Line search
        t_opt, f_val = line_search_max_conv(a, g, t_max=1.0, n_eval=25)

        # Update
        a_new = (1 - t_opt) * a + t_opt * g
        a_new = np.maximum(a_new, 0.0)

        # Verify improvement
        obj_new = compute_obj(a_new)
        obj_old = compute_obj(a)

        if t_opt > 0 and obj_new < obj_old - 1e-10:
            a = a_new
            if verbose and (iter_count <= 10 or iter_count % 20 == 0):
                print("    LP iter %d: obj=%.8f, t=%.4f, active=%d, solves=%d (t=%.1fs)" %
                      (iter_count, obj_new, t_opt, len(active_set) if active_set else 0,
                       n_solves, time.monotonic() - start))
        else:
            # No improvement - try smaller step
            if t_opt > 0.01:
                t_small = t_opt * 0.1
                a_try = (1 - t_small) * a + t_small * g
                a_try = np.maximum(a_try, 0.0)
                obj_try = compute_obj(a_try)
                if obj_try < obj_old - 1e-12:
                    a = a_try
            # Stall
            stall_count += 1
            if stall_count > 30:
                break

    return a, iter_count


def run_fista_lse(a_init, max_time, rng, beta=50.0, step_scale=1.0):
    """FISTA with LSE smoothing for local refinement."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    stuck = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
        stuck = it - last_improve

        # LSE weights
        cm = c.max()
        shifted = beta * (c - cm)
        shifted = shifted - shifted.max()
        e = np.exp(shifted)
        w = e / e.sum()

        # Gradient: 2 * cross-correlation of w and y
        n_c = len(w)
        L = 1 << (n_c + n - 1 - 1).bit_length()
        y_rev = y[::-1]
        conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
        cross = conv_wy[n - 1:2 * n - 1]
        grad = 2.0 * cross

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step
        eff_beta = beta * (1.0 + 0.01 * it)
        step = step_scale / max(eff_beta, 1.0)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart momentum periodically
        if it % 3000 == 0:
            t_mom = 1.0

        # DCT noise when very stuck
        if stuck > 5000 and it % 500 == 0:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
            if noise_scale > 1e-8:
                noise = rng.normal(0, 1.0, n)
                F = fft(noise)
                cutoff = max(5, n // 40)
                F[cutoff:-cutoff] *= 0.05
                noise_smooth = np.real(ifft(F))[:n]
                ns = np.linalg.norm(noise_smooth)
                if ns > 1e-15:
                    noise_smooth = noise_smooth / ns * noise_scale
                    y = project_simplex(y + noise_smooth, S)
                    t_mom = 1.0
                    stuck = 0

    return best_a


def run_lbfgs_refine(a_init, max_time, rng):
    """L-BFGS-B with LSE objective for final polishing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    for beta in [30.0, 80.0, 200.0]:
        if time.monotonic() - start > max_time:
            break
        rem = max_time - (time.monotonic() - start)
        if rem < 5:
            break

        c_cur = fft_convolve_self(a)
        cm_cur = c_cur.max()

        def obj_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / beta
            # Gradient
            n_c = len(w)
            L = 1 << (n_c + n - 1 - 1).bit_length()
            x_rev = x[::-1]
            conv = np.real(ifft(fft(w, L) * fft(x_rev, L)))
            cross = conv[n - 1:2 * n - 1]
            grad = 2.0 * cross
            return obj, grad

        result = scipy_minimize(obj_grad, a.copy(), method='L-BFGS-B',
                                jac=True, bounds=[(0, 1000)] * n,
                                options={'maxiter': 50, 'ftol': 1e-15,
                                         'gtol': 1e-11, 'iprint': -1})
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

        if time.monotonic() - start > max_time:
            break

    return best_a, best_obj


def spectral_init(n, rng):
    """Generate a spectral initialization."""
    # Random phases, flat magnitude
    k = np.fft.fftfreq(n)
    # Cosine-tapered magnitude
    mag = 1.0 / (1.0 + 0.5 * np.abs(k) * n)
    phases = rng.uniform(0, 2 * np.pi, n)
    A = mag * np.exp(1j * phases)
    # Make spectrum Hermitian for real signal
    for i in range(1, (n // 2) + 1):
        A[n - i] = np.conj(A[i])
    A[0] = np.abs(A[0])
    if n % 2 == 0:
        A[n // 2] = np.abs(A[n // 2])
    a = np.real(ifft(A))
    a = np.maximum(a, 0.0)
    a = a + 0.01  # Ensure positivity
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a


def cosine_init(n, freq=1):
    """Cosine-based initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
    a = np.maximum(a, 0.01)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a

    # === Phase 3: LP refinement on polished result (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_time = remaining * 0.20
        print("\n=== Phase 3: LP refinement (budget %.0fs) ===" % p3_time)
        t3 = time.monotonic()

        a_lp2 = run_lp_phase(global_best_a, S, p3_time, rng, verbose=True)
        obj_lp2 = compute_obj(a_lp2)
        print("  LP phase 3: obj=%.8f (t=%.0fs)" % (obj_lp2, time.monotonic() - t3))
        if obj_lp2 < global_best_obj:
            global_best_obj = obj_lp2
            global_best_a = a_lp2.copy()

    # === Phase 4: Multi-start with diverse inits (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_time = remaining * 0.10
        print("\n=== Phase 4: Multi-start (budget %.0fs) ===" % p4_time)
        t4 = time.monotonic()

        inits = [
            cosine_init(n, freq=1),
            cosine_init(n, freq=2),
            cosine_init(n, freq=3),
            spectral_init(n, rng),
        ]

        # Include current best
        inits.append(global_best_a.copy())

        time_per = p4_time / len(inits)
        for idx, init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            if time_per < 5:
                break
            # Quick FISTA
            a_m, _ = run_fista_lse(init, min(time_per, 20.0), rng, beta=50.0, step_scale=0.8)
            obj_m = compute_obj(a_m)
            if obj_m < global_best_obj:
                global_best_obj = obj_m
                global_best_a = a_m.copy()
                print("  Multi-start %d: obj=%.8f (NEW BEST!)" % (idx, obj_m))
            else:
                print("  Multi-start %d: obj=%.8f" % (idx, obj_m))

    # === Phase 5: L-BFGS-B final squeeze (5% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p5_time = remaining * 0.05
        print("\n=== Phase 5: L-BFGS-B squeeze (budget %.0fs) ===" % p5_time)
        t5 = time.monotonic()
        a_b, obj_b = run_lbfgs_refine(global_best_a, p5_time, rng)
        print("  L-BFGS: obj=%.8f (t=%.0fs)" % (obj_b, time.monotonic() - t5))
        if obj_b < global_best_obj:
            global_best_obj = obj_b
            global_best_a = a_b.copy()

    # === Phase 6: Final LP + FISTA alternating ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 6: Final alternating (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()
        half = remaining / 3
        for round_i in range(3):
            if time.monotonic() > deadline:
                break
            # LP step
            a_l, _ = run_lp_phase(global_best_a, S, min(half, remaining - 5), rng, verbose=False)
            obj_l = compute_obj(a_l)
            if obj_l < global_best_obj:
                global_best_obj = obj_l
                global_best_a = a_l.copy()
            # FISTA step
            rem = deadline - time.monotonic()
            if rem > 10:
                a_f, _ = run_fista_lse(global_best_a, min(half, rem - 5), rng, beta=150.0, step_scale=0.3)
                obj_f = compute_obj(a_f)
                if obj_f < global_best_obj:
                    global_best_obj = obj_f
                    global_best_a = a_f.copy()
        print("  Final alternating: obj=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t6))

    # Final verification
    final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
        if ver != global_best_obj:
            print("  Note: verified=%s, computed=%s" % (ver, global_best_obj))
    else:
        print("WARNING: evaluate_sequence rejected result, using fallback")
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback objective: %.8f" % ver)

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
