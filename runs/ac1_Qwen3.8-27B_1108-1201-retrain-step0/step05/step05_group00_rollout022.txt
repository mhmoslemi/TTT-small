```python
"""
Minimize 2*n*max_conv/sum^2 using adaptive LSE gradient descent with
Nesterov acceleration, diverse restarts, and multi-scale optimization.
"""
import numpy as np
import time


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def conv_fft(a, N):
    """Compute linear convolution a*a via FFT. N is the FFT size."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    return c


def lse_gradient(a, c, beta, Nf):
    """
    Compute gradient of LSE(beta * c) w.r.t. a.
    a: input sequence of length n
    c: convolution of length 2n-1
    beta: LSE parameter
    Nf: FFT size for gradient computation (>= 3n-2)
    Returns gradient of length n.
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        lse_val = c_max
    else:
        w = exp_bc / total
        lse_val = c_max + np.log(total) / beta
    
    # Gradient: grad[i] = (2/beta) * sum_{j=0}^{n-1} w[i+j] * a[j]
    # = (2/beta) * (w cross-correlate a)[i]
    w_pad = np.zeros(Nf)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    
    W = np.fft.rfft(w_pad)
    Af = np.fft.rfft(a_pad)
    grad = (2.0 / beta) * np.fft.irfft(W * np.conj(Af), Nf)[:n]
    
    return grad, lse_val


def topk_gradient(a, c, Nf, top_k=50):
    """
    Compute gradient using top-k peak weights.
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / top_k
    
    w_pad = np.zeros(Nf)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    
    W = np.fft.rfft(w_pad)
    Af = np.fft.rfft(a_pad)
    grad = 2.0 * np.fft.irfft(W * np.conj(Af), Nf)[:n]
    
    return grad


def nesterov_optimize(a_init, n, S, N, Nf, max_time, rng,
                       beta_init=20.0, beta_max=2000.0,
                       nesterov_mom_max=0.85):
    """
    Nesterov-accelerated gradient descent on LSE with adaptive beta.
    """
    a = project_simplex(a_init.copy(), S)
    c = conv_fft(a, N)
    c_max = c.max()
    best_a = a.copy()
    best_cmax = c_max
    
    # Nesterov state
    y = a.copy()  # lookahead point
    momentum = np.zeros(n)
    t = 1.0
    
    beta = beta_init
    step = 2.0 * S / (n * beta)
    
    start_time = time.monotonic()
    it = 0
    stall_count = 0
    patience = 500
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        # Adaptive beta
        if it % 200 == 0:
            beta = min(beta_max, beta * 1.15)
            step = 2.0 * S / (n * max(beta, 1.0))
        
        # Compute gradient at lookahead point y
        c_y = conv_fft(y, N)
        c_max_y = c_y.max()
        
        if c_max_y < best_cmax:
            best_cmax = c_max_y
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        grad, lse_val = lse_gradient(y, c_y, beta, Nf)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Nesterov step
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        
        # Gradient step on y
        y_new = y - step * grad / gnorm
        y_new = project_simplex(y_new, S)
        
        # Momentum update
        momentum = ((t - 1.0) / t_new) * momentum + (y_new - a)
        a = y_new
        y = y_new + momentum
        y = np.maximum(y, 0.0)
        # Re-project y to simplex (approximately)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        t = t_new
        
        # Periodic re-anchoring to avoid drift
        if it % 500 == 0:
            y = a.copy()
            momentum = np.zeros(n)
            t = 1.0
        
        # Stall detection
        if stall_count > patience:
            break
    
    return best_a, best_cmax


def perturb_subset(a, rng, frac=0.2, noise_scale=0.3):
    """Perturb a random subset of coordinates."""
    n = len(a)
    S = a.sum()
    a_new = a.copy()
    n_perturb = max(1, int(n * frac))
    indices = rng.choice(n, size=n_perturb, replace=False)
    a_new[indices] += noise_scale * S / n * rng.standard_normal(n_perturb)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new * (S / s)


def perturb_block_reverse(a, rng):
    """Reverse a random contiguous block."""
    n = len(a)
    S = a.sum()
    block_len = rng.integers(n // 10, n // 2)
    start = rng.integers(0, n - block_len)
    a_new = a.copy()
    a_new[start:start + block_len] = a_new[start:start + block_len][::-1]
    return a_new


def spectral_perturb(a, N, rng, frac=0.5):
    """Randomize a fraction of FFT phases."""
    n = len(a)
    S = a.sum()
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
```

```python
Minimize 2*n*max_conv/sum^2 using adaptive LSE Nesterov optimization with
diverse perturbations, multi-n exploration, and exact-max final polish.

Key improvements over parent:

import numpy as np
import time


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def conv_fft(a, N):
    """Compute linear convolution a*a via real FFT. N is FFT size >= 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    return c


def lse_grad_and_val(a, c, beta, Nf):
    """
    Compute gradient of LSE(beta*c) w.r.t. a, and the LSE value.
    a: sequence of length n
    c: convolution of length 2n-1
    beta: LSE parameter
    Nf: FFT size for gradient (>= 3n-2)
    Returns (grad, lse_val).
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
        lse_val = c_max
    else:
        w = exp_bc / total
        lse_val = c_max + np.log(total) / beta

    w_pad = np.zeros(Nf)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(Nf)
    a_pad[:n] = a

    W = np.fft.rfft(w_pad)
    Af = np.fft.rfft(a_pad)
    grad = (2.0 / beta) * np.fft.irfft(W * np.conj(Af), Nf)[:n]
    return grad, lse_val


def exact_max_grad(a, c, Nf, top_k=20):
    """
    Compute weighted subgradient of max(c) focusing on top_k peaks.
    Returns gradient of length n.
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    # Weight by exponential of excess
    top_max = top_vals[0]
    w_exp = np.exp(50.0 * (top_vals - top_max))
    weights = w_exp / w_exp.sum()

    w = np.zeros(2 * n - 1)
    w[top_idx] = weights

    w_pad = np.zeros(Nf)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(Nf)
    a_pad[:n] = a

    W = np.fft.rfft(w_pad)
    Af = np.fft.rfft(a_pad)
    grad = 2.0 * np.fft.irfft(W * np.conj(Af), Nf)[:n]
    return grad


def nesterov_lse_block(a_init, n, S, N, Nf, max_time, rng,
                       beta_init=20.0, beta_max=3000.0,
                       mom_max=0.85, patience=400):
    """
    Nesterov-accelerated gradient descent on LSE with adaptive beta.
    Returns (best_a, best_cmax).
    """
    a = project_simplex(a_init.copy(), S)
    c = conv_fft(a, N)
    c_max = c.max()
    best_a = a.copy()
    best_cmax = c_max

    y = a.copy()
    momentum = np.zeros(n)
    t = 1.0
    beta = beta_init
    step = 4.0 * S / (n * beta)

    start_time = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        if it % 100 == 0:
            beta = min(beta_max, beta * 1.2)
            step = 4.0 * S / (n * max(beta, 1.0))

        # Evaluate at lookahead point
        c_y = conv_fft(y, N)
        c_max_y = c_y.max()

        if c_max_y < best_cmax - 1e-12:
            best_cmax = c_max_y
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        if stall > patience:
            break

        grad, _ = lse_grad_and_val(y, c_y, beta, Nf)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        mom_coeff = (t - 1.0) / t_new
        if mom_coeff > mom_max:
            mom_coeff = mom_max

        x_new = project_simplex(y - step * grad / gnorm, S)
        momentum = mom_coeff * momentum + (x_new - a)
        a = x_new
        y = a + momentum
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        t = t_new

        # Periodic re-anchoring
        if it % 300 == 0:
            y = a.copy()
            momentum = np.zeros(n)
            t = 1.0

    return best_a, best_cmax

    best_a = a.copy()
    best_cmax = c_max

def perturb_subset(a, rng, frac=0.25, noise_scale=0.5):
    """Add noise to a random subset of coordinates, renormalize."""
    n = len(a)
    S = a.sum()
    a_new = a.copy()

    idx = rng.choice(n, size=n_p, replace=False)

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new * (S / s)


def perturb_block_reverse(a, rng):

    n = len(a)
    a_new = a.copy()

    S = a.sum()
    s = a_new.sum()
    if s < 1e-15:

    return a_new * (S / s)

def make_init_deflation(n, S, N, rng, n_iters=3000):
    """Deflation: iteratively reduce peak by moving mass."""
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c = conv_fft(a, N)
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
        if contrib.max() < 1e-15:
            break
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        transfer = min(0.05 * a[i_max], 0.5 * a[i_max])
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

        a_best, peak, S, n_used = run_optimization_for_n(
            n_try, S=None, budget_s=alloc, rng=rng, given_seq=given_for_n)

        if a_best is not None:
            score = 2.0 * n_used * peak / (S * S)
            print("  Best for n=%d: score=%.6f" % (n_used, score))
            if score < best_score:
                best_score = score
                best_overall = a_best

    # Final verification
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0.0, 1000.0)
        if best_overall.sum() < 0.01:
            best_overall = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in best_overall]
        ver = evaluate_sequence(final_list)
        print("Final verification: %s" % (str(ver) if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            best_overall = np.ones(256, dtype=np.float64)
    else:
        best_overall = np.ones(256, dtype=np.float64)

    print("Returning length %d" % len(best_overall))
    return [float(x) for x in best_overall]
```
