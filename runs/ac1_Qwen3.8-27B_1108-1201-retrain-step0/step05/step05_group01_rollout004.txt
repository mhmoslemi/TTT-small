```python
    """
    Compute the LSE-smoothed hinge: sum_k softplus(beta * (b_k - T))
    This is a smooth approximation of sum_k max(b_k - T, 0).
    """
    b = fft_convolve_self(a)
    excess = b - T
    # softplus(x) = log(1 + exp(x))
    # For numerical stability: softplus(x) = max(x,0) + log1p(exp(-|x|))
    sp = np.maximum(excess, 0) + np.log1p(np.exp(-np.abs(excess)))
    obj = sp.sum()
    # Gradient: sum_k sigmoid(beta*(b_k - T)) * 2 * C_k * a
    w = 1.0 / (1.0 + np.exp(-beta * excess))  # sigmoid
    grad = conv_gradient(a, w)
    return obj, grad
```

For the LSE objective f(a) = (1/beta) * log(sum_k exp(beta * b_k)), the gradient Lipschitz constant is bounded by 2*beta*max_i |a_i| * (something related to the convolution operator norm).

The convolution operator norm: ||conv(a, delta)|| <= ||a||_1 * ||delta||_1... this is getting complicated. Let me just use a conservative step size and rely on the projection to maintain feasibility.

Let me simplify and focus on what's most likely to help:

1. **Adaptive beta FISTA with restart**: The key insight is to use a beta that starts moderate (say 50) and increases over time. When the iterate starts oscillating (detected by the iterate not improving), reset the momentum.

2. **Top-K gradient for final polish**: In the last phase, switch to using only the top K convolution values for the gradient. With K=1, this is the exact subgradient of the max.

3. **Structured perturbation**: When stuck, instead of random noise, apply a perturbation that is designed to reduce the max convolution. For example, "push mass away from the location that contributes most to the max convolution coefficient."

Let me think about idea 3 more carefully. If b_k = max, then b_k = sum_i a_i * a_{i+k}. To reduce b_k, we want to reduce the product a_i * a_{i+k}. One way: reduce a at positions i where a_i * a_{i+k} is large. But this might increase other b_j values.

A simpler approach: if the max is at lag k, shift some mass from position i to position i-d where d is chosen to reduce b_k. This is like a "gradient step" in a specific direction.

Actually, the gradient of b_k with respect to a is:
(db_k / da_i) = a_{i+k} + a_{i-k} (with appropriate boundary conditions)

So the gradient of the max (at the unique argmax k*) is:
grad_i = a_{i+k*} + a_{i-k*}

This tells us exactly which positions to reduce. If we decrease a_i by grad_i * step, we reduce b_{k*} (first order).

This is exactly what the FISTA/subgradient method does. The issue is avoiding increasing other b_j values. The LSE smoothing handles this by considering all b_j, not just the max.

OK let me just write the code. I'll make it a clean, well-structured implementation that:
1. Uses FISTA with adaptive LSE beta and Nesterov restart
2. Has a "hinge" mode that focuses on the top values
3. Uses L-BFGS-B for polishing
4. Tries multiple initializations
5. Has a final intensive polish phase

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses adaptive FISA with LSE smoothing, Nesterov restart, hinge-based gradients,
L-BFGS-B polishing, and structured multi-start.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def conv_gradient(a, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. a, where b=conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wa = np.real(ifft(fft(w, L) * fft(a_rev, L)))
    cross = conv_wa[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_adaptive(a_init, max_time, rng, beta_start=50.0, beta_end=5000.0,
                       mode='lse', restart_threshold=500):
    """
    FISTA with adaptive beta (exponential increase) and Nesterov restart.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    f_prev = None

    # Precompute for step size
    max_a_init = a.max()
    L_est = 4.0 * beta_start * max_a_init  # Lipschitz constant estimate

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: exponential ramp
        frac = min(1.0, it / (max_time * 100))  # rough fraction of time elapsed
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = max(10.0, min(beta, 100000.0))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-12:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Nesterov restart: if stuck, reset momentum
        if stuck > restart_threshold:
            t_mom = 1.0
            y = a.copy()
            if stuck > restart_threshold * 3:
                # Structured perturbation: redistribute mass from high-prod positions
                # Find the lag with max conv
                k_max = int(np.argmax(c))
                # Gradient direction for this lag
                a_rev = a[::-1]
                e = np.zeros(2*n-1)
                e[k_max] = 1.0
                grad = conv_gradient(a, e)
                # Perturb in negative gradient direction
                perturb = -grad / (np.linalg.norm(grad) + 1e-15)
                scale = 0.01 * S / np.sqrt(n) * min(1.0, stuck / (restart_threshold * 5))
                y = project_simplex(a + scale * perturb, S)
                t_mom = 1.0

        # Compute gradient
        cm = c.max()
        shifted = beta * (c - cm)
        shifted = shifted - shifted.max()
        e = np.exp(shifted)
        tot = e.sum()
        if tot < 1e-300:
            w = np.ones(len(c)) / len(c)
        else:
            w = e / tot
        grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Step size: use 2/L where L is Lipschitz constant estimate
        # L ≈ 2 * beta * ||C|| where ||C|| is the convolution operator norm
        # Conservative: L = 4 * beta * max(a)
        L_step = 4.0 * beta * max_a_init
        step = 2.0 / max(L_step, 1e-10)
        # Cap step to not be too large
        step = min(step, S / (gnorm + 1e-15) * 0.5)

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Update max_a estimate
        max_a_init = max(max_a_init * 0.9999, a.max())

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

def run_lbfgs(a_init, max_time, rng, beta_levels=None):
    """L-BFGS-B polishing with LSE objective."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    if beta_levels is None:
        beta_levels = [50.0, 200.0, 1000.0, 5000.0]

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    for beta in beta_levels:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        c = fft_convolve_self(a)
        cm = c.max()

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm_local + np.log(e.sum()) / beta
            # Gradient
            grad = conv_gradient(x, w)
            return obj, grad

        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 200, 'maxls': 50, 'ftol': 1e-15, 'gtol': 1e-12,
                     'iprint': -1}
        )

        a_new = np.clip(result.x, 0, 1000)
        if a_new.sum() > 1e-15:
            a_new = a_new * (S / a_new.sum())
        obj_new = compute_obj(a_new)
        print("    L-BFGS beta=%.0f: obj=%.8f (nit=%d)" % (beta, obj_new, result.nit))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_exact_subgradient_polish(a_init, max_time, rng):
    """
    Polish using exact subgradient of the max with diminishing step size.
    This is slow but can make very precise final improvements.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0

    # Initial step size
    c0 = fft_convolve_self(a)
    grad0 = conv_gradient(a, np.zeros(2*n-1))  # dummy
    k_max = int(np.argmax(c0))
    e = np.zeros(2*n-1)
    e[k_max] = 1.0
    grad0 = conv_gradient(a, e)
    step0 = S / (np.linalg.norm(grad0) + 1e-15)

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:

            best_a = y.copy()

        # Find all near-max lags
        max_val = c.max()
        active = np.where(c >= max_val * 0.9999)[0]
        e = np.zeros(2*n-1)
        e[active] = 1.0 / len(active)
        grad = conv_gradient(y, e)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:

        step = step0 / np.sqrt(1.0 + 0.01 * it)

    c_final = fft_convolve_self(best_a)

    return best_a, best_conv


def generate_inits(n, rng, S):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Cosine
    for freq in [1, 2]:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        a = np.maximum(a, 0.01)
        a = a * (S / a.sum())
        inits.append(a.copy())

    # Linear ramp
    a = np.linspace(0.1, 1.0, n)
    a = a * (S / a.sum())
    inits.append(a.copy())

    # Two peaks
    p1, p2 = n//4, 3*n//4
    sig = n//16
    a = np.exp(-0.5*((np.arange(n)-p1)/sig)**2) + np.exp(-0.5*((np.arange(n)-p2)/sig)**2)
    a = a * (S / a.sum())
    inits.append(a.copy())

    # Smoothed random
    raw = rng.random(n)
    F = fft(raw)
    cut = max(3, n//200)
    F[cut:-cut] *= 0.01
    smooth = np.real(ifft(F))[:n]
    smooth = np.maximum(smooth, 0.0)
    if smooth.sum() > 1e-10:
        smooth = smooth * (S / smooth.sum())
        inits.append(smooth.copy())

    # Near-uniform with small perturbation
    a = np.ones(n) + 0.1 * np.cos(2*np.pi*t) + 0.05*np.cos(4*np.pi*t)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    inits.append(a.copy())

    return inits
```

Also, let me reconsider the approach. The key issue is that we're at 1.506810 and need to get to 1.5027. That's a 0.27% improvement. Let me think about what could achieve this.

One thing I notice: the previous code's FISTA uses `step = step_scale / max(beta, 1.0)`. With beta_start=10 and step_scale=1.0, the step is 0.1 initially. But the Lipschitz constant of the LSE gradient is much larger than beta (it's proportional to beta * max(a) * some operator norm). So the step might be too large.

Let me use a more principled step size. For the LSE objective f(a) = (1/beta) * log(sum exp(beta * b_k)), the gradient is:
grad_i = (1/beta) * sum_k exp(beta*b_k) / sum exp(beta*b_j) * 2 * (a_{i+k} + a_{i-k})

The Lipschitz constant of this gradient is bounded by:
