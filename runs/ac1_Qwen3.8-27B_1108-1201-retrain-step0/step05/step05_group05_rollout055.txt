def fft_conv(a):

    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size], A, fsize

def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Standard algorithm
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.where(valid)[0].max() + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def smart_subgradient_step(a, c, S, fsize, rng, k_active=5, n_trials=3):
    """
    Try multiple subgradient directions and pick the best via backtracking.
    Returns (new_a, improved).
    """
    n = len(a)
    cmax = c.max()
    # Find top-k active lags (within 1% of max)
    tol = 0.01 * cmax
    candidates = np.where(c >= cmax - tol)[0]
    if len(candidates) == 0:
        candidates = np.array([np.argmax(c)])
    # Sort by value, take top k
    if len(candidates) > k_active:
        sorted_idx = np.argsort(c[candidates])[::-1][:k_active]
        candidates = candidates[sorted_idx]

        candidates = candidates[np.argsort(c[candidates])[::-1]]
    
    n_cand = len(candidates)
    
    # Generate different weightings
    weight_options = []
    # Option 1: uniform

    # Option 2: top-1 only
    weight_options.append(np.array([1.0] + [0.0] * (n_cand - 1)))
    # Option 3: proportional to excess above mean of candidates
    c_act = c[candidates]
    excess = np.maximum(c_act - c_act[:-1].min() if len(c_act) > 1 else c_act - 0, 0)
    if excess.sum() > 1e-30:
        weight_options.append(excess / excess.sum())
    # Option 4: inverse (spread more to less-active)
    inv = 1.0 / (c_act + 1e-10)
    weight_options.append(inv / inv.sum())
    
    # Deduplicate (keep at most n_trials)
    unique_weights = []
    for w in weight_options:
        if w not in [allclose(u, w) for u in unique_weights]:
            unique_weights.append(w)
        if len(unique_weights) >= n_trials:
            break
    
    best_a = a.copy()
    best_val = cmax
    base_step = 0.5 * S / n / (np.linalg.norm(a) / S + 1e-10)
    
    for w in unique_weights[:n_trials]:
        g = compute_subgrad_fft(a, candidates, w, fsize)

def fourier_phase_mutate(a, rng, scale=0.15):
    """Perturb Fourier phases and project."""
    n = len(a)
    S = a.sum()
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + 2.0 * freqs / fsize
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    # Ensure Hermitian symmetry
    new_A = 0.5 * (new_A + np.conj(new_A[np.mod(-np.arange(fsize), fsize)]))
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def block_mutate(a, rng, n_blocks=None):
    """Replace random blocks with new random values."""
    n = len(a)
    S = a.sum()
    if n_blocks is None:
        n_blocks = rng.integers(2, 8)
    a_new = a.copy()
    for _ in range(n_blocks):
        block_len = max(1, int(n * rng.uniform(0.01, 0.1)))
        start_idx = rng.integers(0, n - block_len)
        a_new[start_idx:start_idx + block_len] = rng.uniform(0, 2 * S / n, block_len)
    return project_simplex(a_new, S)


def targeted_mutate(a, c, rng, S, fsize):
    """Target the worst peak: reduce elements that contribute to it."""
    n = len(a)
    k_peak = np.argmax(c)
    # Elements that contribute to c[k_peak]: a_j * a[k_peak - j]
    contrib = np.zeros(n)
    for j in range(n):
        kpj = k_peak - j
        if 0 <= kpj < n:
            contrib[j] = a[j] * a[kpj]
    # Reduce the top contributors
    top_contrib = np.argsort(contrib)[::-1][:max(1, n // 20)]
    a_new = a.copy()
    reduction = rng.uniform(0.1, 0.4)
    a_new[top_contrib] *= (1.0 - reduction)
    # Add small noise to restore sum
    a_new = project_simplex(a_new + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
    return a_new


def spectral_flatten(a, S, fsize, rng, gamma=0.3):

        c, _, _ = fft_conv(a)
        cmax = c.max()

            best = cmax
            best_a = a.copy()

            g_min_idx = np.argmin(g_supp)

            # Just nudge weights
            # Compute sensitivity: for each active lag i, how much does it affect g at the two positions?
            # Approximate: shift the weight from the lag that's "worst" at max position
            # Simpler: just try swapping some weight between first and last active lag
            if n_act >= 2:
                delta = 0.1 / n_act
                w[0] = max(0, w[0] - delta)
                w[-1] = min(1, w[-1] + delta)
                w = np.maximum(w, 0)
                wsum = w.sum()
                if wsum > 1e-10:
                    w /= wsum

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        step = 0.1 * S / (n * (1.0 + 0.005 * it))
        a_new = project_simplex(a - step * g / gnorm * S / np.sqrt(n), S)
        
        c_new, _, _ = fft_conv(a_new)

    return best_a, c_final.max()
