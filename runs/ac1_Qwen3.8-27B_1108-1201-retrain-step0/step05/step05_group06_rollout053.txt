def coord_line_search(a, b, S, i, tol=1e-10):
    """
    Find optimal a_i' by 1D search.
    a: current sequence
    b: current full convolution (a*a)
    S: current sum
    i: coordinate to update
    """
    n = len(a)
    
    # Affected lags: j in [i, min(i+n-1, 2n-2)]
    j_start = i
    j_end = min(i + n, 2*n - 1)  # exclusive
    affected_b = b[j_start:j_end]  # b[i], b[i+1], ..., b[j_end-1]
    affected_a = a[0:j_end - j_start]  # a[0], a[1], ..., a[j_end-1-j_start] = a_{j-i} for j in range
    
    # The diagonal: j = 2i
    diag_idx = 2*i - j_start  # index within affected arrays
    diag_valid = (0 <= 2*i < j_end) and (0 <= diag_idx < len(affected_b))
    
    # Unaffected max
    # All lags except [j_start, j_end)
    if j_start > 0 or j_end < 2*n - 1:
        mask = np.ones(2*n-1, dtype=bool)
        mask[j_start:j_end] = False
        M_unaffected = b[mask].max() if mask.any() else 0.0
    else:
        M_unaffected = 0.0
    
    def objective(t):
        """Compute objective value for change t in coordinate i."""
        new_S = S + t
        if new_S <= 0:
            return np.inf
        # Affected lags: b[j] + 2*t*a[j-i]
        new_affected = affected_b + 2.0 * t * affected_a
        if diag_valid:
            # Override diagonal: b[2i] + 2*a_i*t + t^2
            new_affected[diag_idx] = b[2*i] + 2.0 * a[i] * t + t * t
        M_new = max(M_unaffected, new_affected.max())
        return 2.0 * n * M_new / (new_S * new_S)
    
    # Golden section search on t in [-a[i], a[i]*10] (or some upper bound)
    t_lower = -a[i]
    t_upper = a[i] + S / n  # reasonable upper bound
    
    # Golden section
    gr = (np.sqrt(5) - 1) / 2
    a_gs = t_lower
    b_gs = t_upper
    c_gs = b_gs - gr * (b_gs - a_gs)
    d_gs = a_gs + gr * (b_gs - a_gs)
    fc = objective(c_gs)
    fd = objective(d_gs)
    
    for _ in range(50):
        if fc < fd:
            b_gs = d_gs
            d_gs = c_gs
            fd = fc
            c_gs = b_gs - gr * (b_gs - a_gs)
            fc = objective(c_gs)
        else:
            a_gs = c_gs
            c_gs = d_gs
            fc = fd
            d_gs = a_gs + gr * (b_gs - a_gs)
            fd = objective(d_gs)
    
    t_opt = (a_gs + b_gs) / 2
    return max(0.0, a[i] + t_opt)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
- FISTA with exact/hybrid subgradient (top-k active lags)
- Coordinate descent with golden section 1D search
- LP-based direction finding (active set method)
- Multi-scale exploration (n and 2n)
- Adaptive restarting and diverse initializations
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_corr(a, w):
    """
    Compute cross-correlation: result[i] = sum_j w[j] * a[j-i]
    i.e., (a ⊛ w)_i = sum_m a[m] * w[m+i]
    Using: ifft(conj(fft(a)) * fft(w))
    """
    n = len(a)
    nw = len(w)
    fsize = 1 << (n + nw - 1 - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    return np.real(ifft(np.conj(A) * W))[:n]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def get_subgradient(a, b, k=1, tol_frac=0.0):
    """
    Compute subgradient of max(b) w.r.t. a using top-k active lags.
    Returns gradient vector of length n.
    """
    n = len(a)
    max_val = b.max()
    if tol_frac > 0 and max_val > 0:
        active = np.where(b >= max_val * (1.0 - tol_frac))[0]
    else:
        # Top-k
        if k >= len(b):
            active = np.arange(len(b))
        else:
            active = np.argsort(b)[::-1][:k]
    
    w = np.zeros(len(b))
    w[active] = 1.0 / len(active)
    
    # subgrad = 2 * cross_corr(a, w)
    # But we need to be careful: the cross-correlation gives sum_j w[j]*a[j-i]
    # which is what we want (up to the diagonal correction, which is a second-order effect)
    sg = 2.0 * fft_cross_corr(a, w)
    return sg


def get_subgradient_hybrid(a, b, k_low, k_high, tol_frac):
    """Hybrid subgradient: uniform over active lags within tol, capped at k_high."""
    n = len(a)
    max_val = b.max()
    if max_val <= 0:
        return np.zeros(n)
    # Lags within tolerance
    thresh = max_val * (1.0 - tol_frac)
    active_all = np.where(b >= thresh)[0]
    if len(active_all) > k_high:
        # Take top k_high
        active = active_all[np.argsort(b[active_all])[::-1][:k_high]]
    else:
        active = active_all
    if len(active) < k_low:
        active = np.argsort(b)[::-1][:k_low]
    
    w = np.zeros(len(b))
    w[active] = 1.0 / len(active)
    sg = 2.0 * fft_cross_corr(a, w)
    return sg

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000.0).copy()
    a = project_simplex(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_max = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    b0 = fft_convolve_self(a)
    L_est = 4.0 * np.max(a) + 1.0 / max(n, 1)
    step = 1.0 / max(L_est, 1e-6)
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_max - 1e-15 * max(1.0, best_max):
            best_max = cur_max
            best_a = y.copy()
            last_improve = it

        # Get subgradient
        if hybrid:
            if stuck < 200:
                k = k_init
                tol = 0.0
            elif stuck < 2000:
                k = max(k_init, min(64, k_init + stuck // 50))
                tol = tol_frac * min(1.0, stuck / 2000.0)
            else:
                k = max(32, min(256, k_init + stuck // 20))
                tol = tol_frac
            grad = get_subgradient_hybrid(y, c, 1, k, tol)
        else:
            grad = get_subgradient(y, c, k=k_init, tol_frac=tol_frac)
        
        # Also consider the sum constraint gradient
        # We're projecting to simplex, so the gradient for the constrained problem
        # is just the unconstrained gradient (the projection handles the constraint)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size (backtracking-lite)
        step = min(step * 1.05, 2.0 / (gnorm / S + 1e-10))
        step = max(step, 1e-10)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Momentum restart
        if stuck > 3000 or it % 5000 == 0:
            t_mom = 1.0
        
        # Noise injection when very stuck
        if stuck > 10000 and it % 500 == 0:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck / 20000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck = 0
                last_improve = it
    
    return best_a, best_max


def coord_descent_pass(a, b, S, n_iter=None, rng=None, focus_active=True):
    """
    One pass of coordinate descent with golden section search.
    Updates each coordinate to its optimal value (given others fixed).
    """
    n = len(a)
    a = a.copy()
    b = b.copy()
    
    # Determine order: focus on coordinates with large gradient
    if focus_active and rng is not None:
        # Get a rough gradient to prioritize
        max_val = b.max()
        active = np.where(b >= max_val * 0.95)[0]
        # Prioritize coordinates that affect active lags
        grad_approx = np.zeros(n)
        for j in active[:min(32, len(active))]:
            for i in range(max(0, j - n + 1), min(n, j + 1)):
                if 0 <= j - i < n:
                    grad_approx[i] += 2.0 * a[j - i]
        # Order: high gradient first, then random
        order = np.argsort(-grad_approx)
        if rng is not None:
            # Randomize within blocks to avoid bias
            block = 128
            for start_idx in range(0, n, block):
                end_idx = min(start_idx + block, n)
                order[start_idx:end_idx] = rng.permutation(order[start_idx:end_idx])
    else:
        order = np.arange(n)
    
    if n_iter is not None:
        order = order[:min(n_iter, n)]
    
    count = 0
    for i in order:
        count += 1
        # Golden section search for optimal a[i]
        # Affected lags: j in [i, min(i+n-1, 2n-2)]
        j_start = i
        j_end = min(i + n, 2 * n - 1)
        
        if j_start >= j_end:
            continue
        
        affected_b = b[j_start:j_end]
        affected_a = a[0:j_end - j_start]  # a[j-i] for j in [j_start, j_end)
        
        # Diagonal index: j = 2i
        diag_pos = 2 * i - j_start
        diag_valid = (0 <= 2 * i < j_end) and (0 <= diag_pos < len(affected_b))
        
        # Unaffected max
        total_len = 2 * n - 1
        if j_start > 0 or j_end < total_len:
            if j_start > 0 and j_end < total_len:
                M_unaff = max(b[:j_start].max(), b[j_end:].max())
            elif j_start > 0:
                M_unaff = b[:j_start].max()
            else:
                M_unaff = b[j_end:].max()
        else:
            M_unaff = 0.0
        
        a_i = a[i]
        
        def obj_at_t(t):
            new_S = S + t
            if new_S <= 1e-15:
                return np.inf
            new_affected = affected_b + 2.0 * t * affected_a
            if diag_valid:
                new_affected[diag_pos] = b[2 * i] + 2.0 * a_i * t + t * t
            M_new = max(M_unaff, new_affected.max())
            return M_new / (new_S * new_S)
        
        # Search range: t in [-a_i, a_i + S/n * 2]
        t_lo = -a_i
        t_hi = a_i + 2.0 * S / n
        
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = t_hi - gr * (t_hi - t_lo)
        c2 = t_lo + gr * (t_hi - t_lo)
        f1 = obj_at_t(c1)
        f2 = obj_at_t(c2)
        
        for _ in range(40):
            if f1 < f2:
                t_hi = c2
                c2 = c1
                f2 = f1
                c1 = t_hi - gr * (t_hi - t_lo)
                f1 = obj_at_t(c1)
            else:
                t_lo = c1
                c1 = c2
                f1 = f2
                c2 = t_lo + gr * (t_hi - t_lo)
                f2 = obj_at_t(c2)
        
        t_opt = (t_lo + t_hi) / 2.0
        new_val = max(0.0, a_i + t_opt)
        
        if new_val != a_i:
            a[i] = new_val
            # Update b
            t = new_val - a_i  # already applied, t is the actual change
            # Wait, we already set a[i] = new_val, so t = new_val - a_i (old)
            # But we need to recompute t relative to the old value
            # Actually, let's recompute: the change is new_val - a_i (where a_i was the old value)
            pass
        
        # Recompute b for affected lags only
        # Actually, for correctness, let's just recompute the full b every K updates
        # For now, update incrementally
        # The change in b[j] for j in [i, i+n-1]:
        # b[j] += 2 * (new_val - old_val) * a[j-i] (for j != 2i)
        # b[2i] += 2 * old_val * (new_val - old_val) + (new_val - old_val)^2
        # But we already modified a[i], so we need the OLD a_i
        # Let me redo this more carefully
    
    # For simplicity and correctness, recompute b at the end
    b_new = fft_convolve_self(a)
    return a, b_new


def coord_descent_pass_fast(a, max_coords=2000, rng=None):
    """
    Faster coordinate descent: update a subset of coordinates, recompute b.
    """
    n = len(a)
    S = a.sum()
    b = fft_convolve_self(a)
    
    # Compute approximate gradient for prioritization
    max_val = b.max()
    if max_val > 0:
        active = np.where(b >= max_val * 0.9)[0][:64]
        grad = np.zeros(n)
        w = np.zeros(len(b))
        w[active] = 1.0 / len(active)
        grad = 2.0 * fft_cross_corr(a, w)
    else:
        grad = np.zeros(n)
        active = np.array([0])
    
    # Select coordinates to update: those with largest |gradient|
    if rng is not None and max_coords < n:
        # Bias towards high-gradient coordinates
        probs = np.abs(grad) + 1e-10
        probs /= probs.sum()
        idx = rng.choice(n, size=min(max_coords, n), replace=False, p=probs)
    else:
        idx = np.argsort(-np.abs(grad))[:max_coords]
    
    # Update selected coordinates
    for i in idx:
        a_i = a[i]
        j_start = i
        j_end = min(i + n, 2 * n - 1)
        
        if j_start >= j_end:
            continue
        
        affected_b = b[j_start:j_end].copy()
        affected_a = a[0:j_end - j_start].copy()
        # Fix the diagonal in affected_a
        diag_pos = 2 * i - j_start
        diag_valid = (0 <= 2 * i < j_end) and (0 <= diag_pos < len(affected_b))
        if diag_valid:
            affected_a[diag_pos] = 0.0  # Handle diagonal separately
        
        # Unaffected max (precompute once)
        total_len = 2 * n - 1
        if j_start > 0 and j_end < total_len:
            M_unaff = max(b[:j_start].max(), b[j_end:].max())
        elif j_start > 0:
            M_unaff = b[:j_start].max()
        elif j_end < total_len:
            M_unaff = b[j_end:].max()
        else:
            M_unaff = 0.0
        
        def obj_at_t(t):
            new_S = S + t
            if new_S <= 1e-15:
                return np.inf
            new_affected = affected_b + 2.0 * t * affected_a
            if diag_valid:
                new_affected[diag_pos] = b[2 * i] + 2.0 * a_i * t + t * t
            M_new = max(M_unaff, new_affected.max())
            return M_new / (new_S * new_S)
        
        t_lo = -a_i
        t_hi = a_i + 2.0 * S / n
        
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = t_hi - gr * (t_hi - t_lo)
        c2 = t_lo + gr * (t_hi - t_lo)
        f1 = obj_at_t(c1)
        f2 = obj_at_t(c2)
        
        for _ in range(30):
            if f1 < f2:
                t_hi = c2
                c2 = c1
                f2 = f1
                c1 = t_hi - gr * (t_hi - t_lo)
                f1 = obj_at_t(c1)
            else:
                t_lo = c1
                c1 = c2
                f1 = f2
                c2 = t_lo + gr * (t_hi - t_lo)
                f2 = obj_at_t(c2)
        
        t_opt = (t_lo + t_hi) / 2.0
        new_val = max(0.0, a_i + t_opt)
        
        if abs(new_val - a_i) > 1e-12:
            a[i] = new_val
            S += new_val - a_i
            # Update b incrementally
            t = new_val - a_i  # This is wrong since a_i was already updated
            # Actually a[i] = new_val now, so the change was (new_val - a_i_old)
            # But we can't easily recover a_i_old. Let me just not do incremental update.
    
    # Recompute b
    b = fft_convolve_self(a)
    # Renormalize
    S_actual = a.sum()
    if S_actual > 0:
        target_S = np.sqrt(2.0 * n)
        a = a * (target_S / S_actual)
        b = fft_convolve_self(a)
    
    return a, b


def run_cd_phase(a_init, max_time, rng, batch_size=1000):
    """
    Coordinate descent phase: repeatedly do CD passes.
    """
    n = len(a_init)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000.0).copy()
    S = np.sqrt(2.0 * n)
    a = project_simplex(a, S)
    
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        a, b = coord_descent_pass_fast(a, max_coords=batch_size, rng=rng)
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 20:
            # Try larger batch or different strategy
            batch_size = min(n, batch_size * 2)
            if stuck > 50:
                # Add noise
                noise = rng.normal(0, 0.01 / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
                batch_size = 1000
                stuck = 0
                last_improve = it
    
    return best_a, best_obj


def solve_lp_direction(f, M, n, max_active=50, max_iters=20):
    """
    Solve restricted LP: max sum(g) s.t. (f*g)_j <= M for active j, g >= 0.
    Uses active set method.
    Returns the direction g (or None if failed).
    """
    from scipy.optimize import linprog
    
    # Get initial active set: top lags of f*f
    ff = fft_convolve_self(f)
    active = np.argsort(ff)[::-1][:max_active]
    
    for it in range(max_iters):
        # Build constraint matrix for active lags
        # Constraint j: sum_i f[j-i] * g[i] <= M (for valid j-i in [0, n-1])
        k = len(active)
        A_ub = np.zeros((k, n))
        for row, j in enumerate(active):
            # g[i] appears with coefficient f[j-i] when 0 <= j-i <= n-1
            i_start = max(0, j - n + 1)
            i_end = min(n, j + 1)
            for i in range(i_start, i_end):
                if 0 <= j - i < n:
                    A_ub[row, i] = f[j - i]
        
        # Solve LP
        c = -np.ones(n)  # maximize sum(g) = minimize -sum(g)
        try:
            result = linprog(c, A_ub=A_ub, b_ub=np.full(k, M),
                           bounds=[(0, None)] * n, method='highs',
                           options={'maxiter': 5000})
        except Exception:
            return None
        
        if not result.success:
            return None
        
        g = result.x
        
        # Check all constraints
        fg = np.convolve(f, g, mode='full')
        violations = np.where(fg > M + 1e-8)[0]
        
        if len(violations) == 0:
            return g
        
        # Add violated constraints
        active = np.unique(np.concatenate([active, violations[:20]]))[:2*max_active]
    
    return None


def run_lp_phase(a_init, max_time, rng):
    """
    Phase using LP-based direction finding.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000.0).copy()
    a = project_simplex(a, S)
    
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        if time.monotonic() - start < 2.0 and it > 0:
            # LP is expensive, limit frequency
            pass
        
        b = fft_convolve_self(a)
        M = b.max()
        obj = compute_obj(a)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Solve LP direction
        g = solve_lp_direction(a, M, n, max_active=30, max_iters=10)
        
        if g is not None and g.sum() > 0:
            # Scale g to match normalization
            g_scaled = g * (S / g.sum())
            
            # Line search for step size
            best_t = 0.0
            best_obj_step = obj
            for t_candidate in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_trial = (1 - t_candidate) * a + t_candidate * g_scaled
                a_trial = np.maximum(a_trial, 0)
                obj_trial = compute_obj(a_trial)
                if obj_trial < best_obj_step:
                    best_obj_step = obj_trial
                    best_t = t_candidate
            
            if best_t > 0:
                a = (1 - best_t) * a + best_t * g_scaled
                a = np.maximum(a, 0)
                a = project_simplex(a, S)
        
        it += 1
    
    return best_a, best_obj


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.54 + 0.46 * np.cos(2 * np.pi * tt)


def make_random_structured(n, rng):
    """Random initialization with some structure."""
    a = rng.random(n) ** 3
    # Add some low-frequency structure
    freq = rng.integers(1, 5)
    phase = rng.uniform(0, 2 * np.pi)
    a += 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    return a


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

            obj_blk = compute_obj(a_fin)
            if obj_blk < global_best_obj:
                global_best_obj = obj_blk
                global_best_a = a_fin.copy()
            
            # CD block
            rem_block = block_time
            if time.monotonic() + rem_block > deadline:
                break
            a_fin, obj_cd = run_cd_phase(global_best_a, rem_block, rng,
                                          batch_size=5000)
            if obj_cd < global_best_obj:
                global_best_obj = obj_cd
                global_best_a = a_fin.copy()
        
        print("  Final: obj=%.8f (time=%.1fs)" % (global_best_obj, time.monotonic() - t0))
    
    # Final between n and 2n: try both and pick best
    print("\n=== Checking both lengths ===")
    # Evaluate current best
    final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
    ver = evaluate_sequence(final_list)
    print("Final objective (evaluate_sequence): %.8f (n=%d)" % (ver, len(final_list)))
    
    if not np.isfinite(ver):
        # Fallback
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]
    
    print("\n=== SUMMARY ===")
    print("Best objective: %.8f" % global_best_obj)
    print("Sequence length: %d" % len(global_best_a))
    print("Total time: %.0fs" % (time.monotonic() - total_start))
    
    return final_list
