def fft_convolve_self(a):

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

"""
Minimize 2*n*max_conv/sum^2 using coordinate descent with exact 1D subproblem
solving, adaptive FISTA smoothing, and multi-scale initialization.
Key innovation: coordinate descent exploits the structure that changing one
coordinate only affects O(n) convolution lags, enabling efficient 1D
minimization via golden section search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def coord_descent_1d(c, a, j, n, x_old, S_rest, base, lin, quad, max_unaffected):
    """Solve 1D subproblem for coordinate j via golden section search."""
    # f(x) = max(max_unaffected, (base + lin*x + quad*x^2).max()) / (S_rest + x)^2
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    def f_of_x(x):
        vals = base + lin * x + quad * x * x
        max_aff = vals.max()
        max_tot = max(max_unaffected, max_aff)
        denom = S_rest + x
        if denom < 1e-15:
            return np.inf
        return max_tot / (denom * denom)

    # Bracket
    x_hi = max(x_old * 3.0, S_rest * 0.1, 1.0)
    # Expand bracket if needed
    f_lo = f_of_x(0.0)
    f_hi = f_of_x(x_hi)
    while f_hi > f_lo and x_hi < S_rest * 10.0:
        x_hi *= 2.0
        f_hi = f_of_x(x_hi)

    a_lo, a_hi = 0.0, x_hi
    c_pt = a_hi - gr * (a_hi - a_lo)
    d_pt = a_lo + gr * (a_hi - a_lo)
    f_c = f_of_x(c_pt)
    f_d = f_of_x(d_pt)

    for _ in range(50):
        if f_c < f_d:
            a_hi = d_pt
            d_pt = c_pt
            f_d = f_c
            c_pt = a_hi - gr * (a_hi - a_lo)
            f_c = f_of_x(c_pt)
        else:
            a_lo = c_pt
            c_pt = d_pt
            f_c = f_d
            d_pt = a_lo + gr * (a_hi - a_lo)
            f_d = f_of_x(d_pt)

    x_opt = (a_lo + a_hi) / 2.0
    f_opt = f_of_x(x_opt)

    # Check boundaries
    if f_lo < f_opt:
        x_opt = 0.0
    elif f_hi < f_opt:
        x_opt = x_hi
    # Also check x_old (no change)
    f_xold = f_of_x(x_old)
    if f_xold < f_opt:
        x_opt = x_old

    return x_opt


def run_coordinate_descent(a_init, max_time, rng, verbose=True):
    """Run coordinate descent with incremental convolution updates."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = compute_obj(a)
    best_a = a.copy()

    start = time.monotonic()
    pid = 0
    obj_history = []

    while time.monotonic() - start < max_time:
        pid += 1
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > max_time:
                break

            x_old = a[j]
            S_rest = S - x_old  # Since a is normalized to sum S

            # Affected lags: [j, j+n-1]
            affected_k = np.arange(j, j + n)
            k_offset = np.arange(n)  # k - j
            is_2j_pos = j  # position in affected_k where k=2j (since affected_k[j] = j+j = 2j)
            is_2j_mask = (k_offset == j)  # k-j == j means k == 2j

            # Precompute base, lin, quad for the 1D subproblem
            c_aff = c[affected_k]
            base = c_aff.copy()
            base[~is_2j_mask] -= 2.0 * x_old * a[k_offset[~is_2j_mask]]
            base[is_2j_mask] -= x_old * x_old

            lin = np.zeros(n)
            lin[~is_2j_mask] = 2.0 * a[k_offset[~is_2j_mask]]

            quad = np.zeros(n)
            quad[is_2j_mask] = 1.0

            # Max of unaffected lags
            mask_aff = np.zeros(2 * n - 1, dtype=bool)
            mask_aff[j:j + n] = True
            if np.any(~mask_aff):
                max_unaff = c[~mask_aff].max()
            else:
                max_unaff = 0.0

            # Solve 1D subproblem
            x_new = coord_descent_1d(c, a, j, n, x_old, S_rest, base, lin, quad, max_unaff)

            # Apply update
            if abs(x_new - x_old) > 1e-15:
                delta = x_new - x_old
                # Update c incrementally
                c[j:j + n] += 2.0 * delta * a[:n]
                # Fix k=2j
                c[2 * j] -= 2.0 * delta * a[j]
                c[2 * j] += x_new * x_new - x_old * x_old
                a[j] = x_new

        # Renormalize (small drift from floating point)
        a = a * (S / a.sum())
        c = fft_convolve_self(a)  # Refresh to avoid drift accumulation
        obj = compute_obj(a)
        obj_history.append(obj)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose and pid % 5 == 0:
                elapsed = time.monotonic() - start
                print("  CD pass %d: obj=%.8f (t=%.0fs)" % (pid, best_obj, elapsed))
        elif verbose and pid % 10 == 0:
            elapsed = time.monotonic() - start
            print("  CD pass %d: obj=%.8f, best=%.8f (t=%.0fs)" % (pid, obj, best_obj, elapsed))

        # Convergence check
        if len(obj_history) > 20:
            recent = obj_history[-20:]
            if max(recent) - min(recent) < 1e-9:
                if verbose:
                    print("  CD converged at pass %d: obj=%.8f" % (pid, best_obj))
                break

    return best_a, best_obj


def run_fista(a_init, max_time, rng, beta_start=20.0, mode="lse"):
    """FISTA with LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    stuck = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        obj_cur = 2.0 * n * cur_max / S**2
        if obj_cur < best_obj - 1e-12:
            best_obj = obj_cur
            best_a = y.copy()
            last_improve = it
        stuck = it - last_improve

        # Adaptive beta
        beta = beta_start * (1.0 + 0.01 * it)
        if stuck > 500:
            beta = max(10.0, beta * 0.5)  # Reduce beta when stuck to explore more
        beta = min(beta, 50000.0)
        step = 1.0 / max(beta, 1.0)

        if mode == "lse":
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
        else:
            # Top-k gradient
            k = max(4, min(128, n // 16))
            if stuck > 500:
                k = max(2, k // 4)
            idx = np.argsort(c)[::-1][:k]
            w = np.zeros(2 * n - 1)
            w[idx] = 1.0 / k
            grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # DCT noise when stuck
        if stuck > 2000 and it % 100 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
            if noise_scale > 1e-8:
                noise = rng.normal(0, 1.0, n)
                F = fft(noise)
                cutoff = max(3, n // 100)
                F[cutoff:-cutoff] *= 0.05
                noise = np.real(ifft(F))[:n]
                noise = noise / (np.linalg.norm(noise) + 1e-15) * noise_scale
                y = project_simplex(y + noise, S)
                t_mom = 1.0
                stuck = 0

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 5000 == 0:
            t_mom = 1.0

    return best_a, best_obj


def upsample_banded(a_small, n_large):
    """Band-limited upsample via frequency domain zero-padding."""
    n_small = len(a_small)
    # Compute DFT of small sequence
    A_small = fft(a_small, n_small)
    # Zero-pad in frequency domain
    # For real signal, A is conjugate symmetric
    A_large = np.zeros(n_large, dtype=complex)
    # Place the first half of A_small into A_large
    half_small = n_small // 2
    half_large = n_large // 2
    A_large[:half_small + 1] = A_small[:half_small + 1]
    A_large[n_large - half_small:] = A_small[n_small - half_small:]
    a_large = np.real(ifft(A_large))
    return a_large

    # === Phase 2: Main coordinate descent (35% time) ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.35
    print("\n=== Phase 2: Coordinate descent (budget %.0fs) ===" % p2_budget)
    t2 = time.monotonic()

    a_cd, obj_cd = run_coordinate_descent(global_best_a, p2_budget, rng, verbose=True)
    obj_cd = compute_obj(a_cd)
    print("  CD done: obj=%.8f (t=%.0fs)" % (obj_cd, time.monotonic() - t2))
    if obj_cd < global_best_obj:
        global_best_obj = obj_cd
        global_best_a = a_cd.copy()

    # === Phase 3: Interleaved CD + FISTA refinement (25% time) ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.25
    print("\n=== Phase 3: Interleaved CD+FISTA (budget %.0fs) ===" % p3_budget)
    t3 = time.monotonic()

    a_if, obj_if = run_interleaved_cd_fista(global_best_a, p3_budget, rng)
    obj_if = compute_obj(a_if)
    print("  Interleaved done: obj=%.8f (t=%.0fs)" % (obj_if, time.monotonic() - t3))
    if obj_if < global_best_obj:
        global_best_obj = obj_if
        global_best_a = a_if.copy()

    # === Phase 4: Perturbation + restart (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = remaining * 0.10
        print("\n=== Phase 4: Perturbation restarts (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        n = len(global_best_a)
        S = np.sqrt(2.0 * n)

        # Try a few different perturbations
        n_pert = 4
        pert_time = p4_budget / n_pert
        for p in range(n_pert):
            if time.monotonic() > deadline:
                break
            # Different perturbation strategies
            if p == 0:
                # Add smooth noise
                noise = rng.normal(0, 0.05 / np.sqrt(n), n)
                F = fft(noise)
                F[n // 20:-n // 20] *= 0.01
                noise = np.real(ifft(F))[:n]
                a_pert = global_best_a + noise
            elif p == 1:
                # Modulate amplitude
                t = np.linspace(0, 1, n, endpoint=False)
                mod = 1.0 + 0.1 * np.cos(2 * np.pi * rng.integers(1, 5) * t)
                a_pert = global_best_a * mod
            elif p == 2:
                # Random permutation of blocks
                block_size = n // 32
                n_blocks = n // block_size
                perm = rng.permutation(n_blocks)
                a_pert = np.zeros(n)
                for i in range(n_blocks):
                    a_pert[i * block_size:(i + 1) * block_size] = \
                        global_best_a[perm[i] * block_size:(perm[i] + 1) * block_size]
            else:
                # Gaussian smoothing with random sigma
                sigma = rng.uniform(n // 100, n // 20)
                from scipy.ndimage import gaussian_filter1d
                a_pert = gaussian_filter1d(global_best_a, sigma, mode='wrap')

            a_pert = np.maximum(a_pert, 0.0)
            if a_pert.sum() > 1e-10:
                a_pert = a_pert * (S / a_pert.sum())
                a_p, obj_p = run_coordinate_descent(a_pert, pert_time, rng, verbose=False)
                obj_p = compute_obj(a_p)
                if obj_p < global_best_obj:
                    global_best_obj = obj_p
                    global_best_a = a_p.copy()
                    print("  Pert %d: obj=%.8f (improved!)" % (p, obj_p))
                else:
                    print("  Pert %d: obj=%.8f" % (p, obj_p))

    # === Phase 5: Final polish (5% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % remaining)
        t5 = time.monotonic()
        a_final, obj_final = run_coordinate_descent(global_best_a, remaining * 0.7, rng, verbose=False)
        obj_final = compute_obj(a_final)
        if obj_final < global_best_obj:
            global_best_obj = obj_final
            global_best_a = a_final.copy()

        rem = deadline - time.monotonic()
        if rem > 3:
            a_f2, obj_f2 = run_fista(global_best_a, rem, rng, beta_start=100.0, mode="lse")
            obj_f2 = compute_obj(a_f2)
            if obj_f2 < global_best_obj:
                global_best_obj = obj_f2
                global_best_a = a_f2.copy()

        print("  Final: obj=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t5))

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
