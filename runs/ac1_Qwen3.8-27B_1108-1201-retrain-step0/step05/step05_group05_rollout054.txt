This requires choosing `mu` too. We can eliminate `mu` by projecting: minimize `||P(sum_k lambda_k * g_k)||^2` where `P` projects onto the subspace orthogonal to `1`.

Option B: Maximize the "descent" in the projected gradient direction. The projected gradient at `a` in the direction `d` is `d - (sum(d)/n) * 1` (for the sum=1 constraint). The objective decrease is approximately `a^T * (projected d) = a^T d - (sum(d)/n) * sum(a) = a^T d - sum(d)/n`.

`a^T g_k = c_k` (autocorrelation at lag k). And `sum(g_k) = sum_j a_{k-j} = sum(a) = 1` (for the linear convolution, this is approximately 1 for lags in the middle).

So the projected descent is approximately `c_k - 1/n ≈ c_k` (since `c_k >> 1/n` for peak lags).

This means all active lags give approximately the same descent. The choice of `lambda` doesn't matter much for the immediate descent, but it matters for the trajectory.

I'll go with Option A: find `lambda` that makes the subgradient as flat as possible. This should lead to a more "uniform" update that doesn't create sharp features.

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses KKT-optimal subgradient selection with adaptive active set,
backtracking line search, and multi-scale initialization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_subgradient(a, k_lag):
    """Compute subgradient vector for a specific lag k_lag.
    (g_k)_j = a_{k-j} if 0 <= k-j < n, else 0.
    """
    n = len(a)
    g = np.zeros(n)
    # For lag k, g[j] = a[k-j] for valid j
    # j ranges from max(0, k-(n-1)) to min(k, n-1)
    j_start = max(0, k_lag - (n - 1))
    j_end = min(k_lag, n - 1)
    if j_start <= j_end:
        # g[j] = a[k_lag - j] for j in [j_start, j_end]
        # a[k_lag - j] for j = j_start, ..., j_end
        # = a[k_lag - j_start], a[k_lag - j_start - 1], ..., a[k_lag - j_end]
        # = a[j_end], a[j_end-1], ..., a[j_start]  (reversed)
        g[j_start:j_end+1] = a[j_end:j_start-1:-1]
    return g


def compute_subgradient_fast(a, lags, fsize):
    """Compute subgradients for multiple lags using FFT.
    For lag k: g_k = ifft(fft(a, fsize) * exp(-2j*pi*k/fsize))[:n]
    Actually, the circular correlation of a with a shifted by k.
    For linear convolution c_k = sum_j a_j a_{k-j}, the gradient w.r.t. a is:
    (grad)_j = a_{k-j} for valid j.
    This is the same as the cross-correlation of a with a delta at position k,
    which in Fourier domain is: conj(FFT(a)) * FFT(delta_k)
    FFT(delta_k) = exp(-2j*pi*k/fsize)
    So grad = real(ifft(conj(A) * exp(-2j*pi*k/fsize)))[:n]
    """
    n = len(a)
    A = fft(a, fsize)
    grads = np.zeros((len(lags), n))
    for i, k in enumerate(lags):
        ph = np.exp(-2j * np.pi * k / fsize)
        grads[i] = np.real(ifft(np.conj(A) * ph))[:n]
    return grads


def kkt_optimal_subgradient(a, c, k_active, fsize):
    """
    Find the optimal convex combination of subgradients for the active lags
    that is as 'flat' as possible (minimum variance after removing mean).
    
    Returns the combined subgradient direction.
    """
    n = len(a)
    m = len(k_active)
    
    if m == 0:
        return np.zeros(n)
    if m == 1:
        return compute_subgradient(a, k_active[0])
    
    # Compute subgradients for active lags
    grads = compute_subgradient_fast(a, k_active, fsize)  # shape (m, n)
    
    # Center them (remove mean)
    grads_centered = grads - grads.mean(axis=1, keepdims=True)
    
    # We want lambda that minimizes ||sum_k lambda_k * grads_centered[k]||^2
    # = lambda^T G^T G lambda where G = grads_centered (m x n)
    # subject to lambda >= 0, sum(lambda) = 1
    
    # G^T G is m x m
    # But computing G^T G directly is O(m^2 * n) which for m=50, n=8192 is ~20M - fine
    # Actually G is m x n, so G^T G is m x m, computed in O(m^2 * n)
    # But we can do it via (G.T @ G) which is efficient
    
    # Actually, let's use a simpler approach: just minimize the norm
    # QP: min lambda^T (G^T G) lambda s.t. lambda >= 0, 1^T lambda = 1
    # This is a standard QP. For m <= 50, we can solve it with simple methods.
    
    # Simpler approach: use iterative water-filling
    # Start with uniform lambda, iterate:
    # lambda_k proportional to exp(-gamma * (G^T G @ lambda)_k)
    
    # Even simpler: just use uniform weights (average of active subgradients)
    # This is already better than single-lag subgradient
    lambda_w = np.ones(m) / m
    combined = (lambda_w[:, None] * grads_centered).sum(axis=0)
    # Add back the mean (so the subgradient is correct)
    # Actually for the gradient step, we want the actual subgradient, not centered
    combined_actual = (lambda_w[:, None] * grads).sum(axis=0)
    
    return combined_actual


def kkt_optimal_subgradient_v2(a, c, k_active, fsize):
    """
    Improved KKT-optimal subgradient: solve the QP properly.
    """
    n = len(a)
    m = len(k_active)
    
    if m == 0:
        return np.zeros(n)
    if m == 1:
        return compute_subgradient(a, k_active[0])
    
    # Compute subgradients
    A = fft(a, fsize)
    grads = np.zeros((m, n))
    for i, k in enumerate(k_active):
        ph = np.exp(-2j * np.pi * k / fsize)
        grads[i] = np.real(ifft(np.conj(A) * ph))[:n]
    
    # Center for the QP
    means = grads.mean(axis=1, keepdims=True)
    G = grads - means  # m x n, centered
    
    # QP: min lambda^T (G^T G) lambda s.t. lambda >= 0, sum(lambda) = 1
    # G^T G = (G.T @ G)  -- but G is m x n, so G.T is n x m, G.T @ G is m x m
    # This is O(m^2 * n) 
    
    # For efficiency with m up to ~30:
    # G^T G = sum of outer products, or just matrix multiply
    Q = G.T @ G  # m x m
    
    # Solve QP: min 0.5 lambda^T Q lambda s.t. 1^T lambda = 1, lambda >= 0
    # Use projected gradient on the probability simplex
    if m <= 50:
        lam = np.ones(m) / m
        # Gradient of 0.5 lambda^T Q lambda is Q @ lambda
        for it in range(200):
            grad_q = Q @ lam
            # Projected gradient step
            lr = 1.0 / max(np.abs(Q).max(), 1e-10)
            lam = project_simplex(lam - lr * grad_q, 1.0)
            lam = np.maximum(lam, 0)
            s = lam.sum()
            if s > 0:
                lam = lam / s
        # Ensure feasibility
        lam = np.maximum(lam, 0)
        lam = lam / lam.sum()
    else:
        lam = np.ones(m) / m
    
    combined = (lam[:, None] * grads).sum(axis=0)
    return combined, lam


def lse_subgradient(a, c, beta, fsize):
    """LSE-smoothed subgradient."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    # Combined subgradient
    A = fft(a, fsize)
    # w is defined on the full convolution (length 2n-1)
    W = fft(w, fsize)
    grad = np.real(ifft(W * np.conj(A)))[:n]
    return grad

def fourier_phase_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * fsize)

    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses sum-of-top-k objective with adaptive k, FISTA momentum,
backtracking line search, and basin hopping with Fourier mutations.
Key innovations over parent:
  - Sum-of-top-k smooth surrogate (avoids subgradient selection issues)
  - Adaptive k schedule (broad focus -> narrow focus)
  - Backtracking line search for robust step sizes
  - KKT-support identification for zero coordinates
  - Multi-scale spectral re-flattening

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_gradient(a, c, k, fsize):
    """
    Gradient of sum of top-k autocorrelation values.
    Uses FFT for efficiency.
    """
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    # Weight by excess above median for smoothness
    median_c = np.median(c[idx])
    weights = np.maximum(c[idx] - median_c * 0.5, 0.1)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum
    w[idx] = weights
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_gradient_hard(a, c, k, fsize):
    """
    Hard top-k gradient: uniform weight on top-k lags.
    """
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def lse_gradient(a, c, beta, fsize):
    """LSE-smoothed max gradient."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    mx = scaled.max()
    if mx > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
    e = np.exp(scaled - scaled.max())
    tot = e.sum()
    w = e / tot
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def kkt_combined_subgradient(a, c, active_lags, fsize):
    """
    KKT-optimal convex combination of active lag subgradients.
    Minimizes variance of the combined subgradient (makes it flat).
    """
    n = len(a)
    m = len(active_lags)
    if m == 0:
        return np.zeros(n)
    if m == 1:
        w = np.zeros(len(c))
        w[active_lags[0]] = 1.0
        W = fft(w, fsize)
        A = fft(a, fsize)
        return 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    A = fft(a, fsize)
    # Compute all subgradient directions
    grads = np.zeros((m, n))
    for i, k in enumerate(active_lags):
        ph = np.exp(-2j * np.pi * k / fsize)
        grads[i] = np.real(ifft(np.conj(A) * ph))[:n]
    
    # Center
    means = grads.mean(axis=1, keepdims=True)
    G = grads - means  # m x n
    
    # QP: min lambda^T (G^T G) lambda s.t. sum(lambda)=1, lambda>=0
    # G^T G is m x m
    Q = G.T @ G  # m x m
    
    # Solve with active set method
    active_mask = np.ones(m, dtype=bool)
    lam = np.ones(m) / m
    for outer in range(m):
        # Solve equality-constrained QP on active set
        idx_active = np.where(active_mask)[0]
        Qaa = Q[np.ix_(idx_active, idx_active)]
        ma = len(idx_active)
        if ma == 1:
            lam_sub = np.array([1.0])
        else:
            try:
                Qinv = np.linalg.inv(Qaa + 1e-12 * np.eye(ma))
                lam_sub = Qinv @ np.ones(ma)
                lam_sub = lam_sub / lam_sub.sum()
            except np.linalg.LinAlgError:
                lam_sub = np.ones(ma) / ma
        
        # Check if all positive
        if np.all(lam_sub > -1e-10):
            lam = np.zeros(m)
            lam[idx_active] = np.maximum(lam_sub, 0)
            lam = lam / lam.sum()
            break
        else:
            # Deactivate the most negative component
            worst_local = np.argmin(lam_sub)
            global_idx = idx_active[worst_local]
            active_mask[global_idx] = False
            if not active_mask.any():
                lam = np.ones(m) / m
                break
    
    combined = (lam[:, None] * grads).sum(axis=0)
    return 2.0 * combined


def fourier_phase_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def spectral_reflat(a, S, rng, strength=0.1):
    """
    Attempt to flatten the power spectrum |A(f)|^2 while maintaining non-negativity.
    This reduces the peak-to-average ratio of the autocorrelation.
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    P = np.abs(A) ** 2
    
    # Compute target: geometric mean of P (makes P as flat as possible)
    log_P = np.log(np.maximum(P, 1e-30))
    target_log_P = np.ones(fsize) * np.mean(log_P)
    
    # Partially move toward flat spectrum
    new_log_P = (1 - strength) * log_P + strength * target_log_P
    new_P = np.exp(new_log_P)
    
    # Need to find phase such that IFFT(sqrt(new_P) * exp(i*phase)) >= 0
    # Use current phase as starting point
    new_mag = np.sqrt(new_P)
    new_phase = np.angle(A)
    new_A = new_mag * np.exp(1j * new_phase)
    new_A[0] = A[0]  # Preserve DC
    
    a_new = np.real(ifft(new_A, fsize))[:n]
    
    # If non-negativity is badly violated, reduce strength
    violations = np.sum(np.maximum(-a_new, 0))
    if violations > 0.1 * a.sum():
        return None
    
    return project_simplex(a_new, S)

def generate_diverse_inits(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Windows

    inits.append(np.sin(np.pi * tt) ** 2 + 0.001)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.001)
    inits.append(np.exp(-8.0 * (tt - 0.5) ** 2) + 0.001)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * 3 * tt + 0.5), 0.001))
    
    # Random
    inits.append(rng.random(n) + 0.01)
    inits.append(np.abs(rng.normal(1.0, 0.3, n)) + 0.01)
    inits.append(np.abs(rng.normal(0, 1, n)) + 0.1)
    
    # Block structures
    for nb in [8, 16, 32]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.3, 2.0, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n] + 0.01)
    
    return inits

            print("  Opt sub-phase %d/%d (t=%.0fs):" % (sub + 1, n_sub, t_alloc))
            # Add small perturbation for diversity
            a_pert = global_a + rng.normal(0, 0.0005 * S / np.sqrt(n), n)
            a_pert = normalize_to_sum(a_pert, S)
            a_p, val_p = optimize_sequence(a_pert, S, t_alloc, rng)
            obj_p = 2.0 * n * val_p / (a_p.sum() ** 2)
            if obj_p < global_best:
                global_best = obj_p
                global_a = a_p.copy()
                print("    ** NEW BEST: %.6f" % obj_p)
            else:
                print("    obj=%.6f (no improvement)" % obj_p)
    
    # Phase 4: High-beta LSE polish
    print("\n=== Phase 4: High-Beta Polish ===")
    p4_start = time.monotonic()
    p4_budget = deadline - p4_start - 10
    
    if global_a is not None and p4_budget > 30:
        # Split into two segments with different beta ranges
        seg1 = p4_budget * 0.6
        seg2 = p4_budget * 0.4 - 5
        
        a_hb, val_hb = high_beta_fista(global_a, S, seg1, rng,
                                         beta_start=50.0, beta_end=5000.0)
        obj_hb = 2.0 * n * val_hb / (a_hb.sum() ** 2)
        if obj_hb < global_best:
            global_best = obj_hb
            global_a = a_hb.copy()
        print("  After HB FISTA: obj=%.6f" % global_best)
        
        if seg2 > 15:
            a_hb2, val_hb2 = high_beta_fista(global_a, S, seg2, rng,
                                               beta_start=200.0, beta_end=20000.0)
            obj_hb2 = 2.0 * n * val_hb2 / (a_hb2.sum() ** 2)
            if obj_hb2 < global_best:
                global_best = obj_hb2
                global_a = a_hb2.copy()
            print("  After HB FISTA 2: obj=%.6f" % global_best)
    
    # Final: try one more round of adaptive from best
    remaining = deadline - time.monotonic() - 5
    if global_a is not None and remaining > 20:
        a_fin, val_fin = optimize_sequence(global_a, S, remaining, rng)
        obj_fin = 2.0 * n * val_fin / (a_fin.sum() ** 2)
        if obj_fin < global_best:
            global_best = obj_fin
            global_a = a_fin.copy()
        print("  Final adaptive: obj=%.6f" % global_best)
    
    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n)
            final_list = [float(x) for x in global_a]
            ver = evaluate_sequence(final_list)
            print("  Fallback obj=%.6f" % ver)
            global_best = ver
    else:
        global_a = np.ones(n)
        final_list = [float(x) for x in global_a]
    
    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.8f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
