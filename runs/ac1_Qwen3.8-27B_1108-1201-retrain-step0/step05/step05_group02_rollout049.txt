"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Combines mirror descent, FISTA with LSE smoothing, simulated annealing,
and multi-scale upscaling to improve upon previous best of 1.5068.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_smooth_max(c, beta):
    """Compute LSE approximation of max and weights."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return cm, w
    w = e / tot
    return cm, w


def lse_gradient(a, beta):
    """LSE gradient of beta * LSE(conv(a,a))."""
    n = len(a)
    c = fft_convolve_self(a)
    cm, w = lse_smooth_max(c, beta)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return cm, grad, c


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def mirror_descent(a_init, max_time, rng, eta_init=0.1, eta_decay=0.999,
                   beta_init=5.0, beta_rate=0.005, beta_max=5000.0):
    """Mirror descent with LSE smoothing on the simplex."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 1e-30, 1000).copy()
    a = a / a.sum() * S
    
    start = time.monotonic()
    best_obj = 2.0 * n * fft_convolve_self(a).max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0
    eta = eta_init
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        cm, grad, c = lse_gradient(a, beta)
        cur_obj = 2.0 * n * c.max() / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 5000:
            break
        
        # Mirror descent step
        log_a = np.log(a)
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()  # numerical stability
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum() * S
        
        # Ensure no underflow issues
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum() * S
        
        a = a_new
        eta = eta * eta_decay
        
        # Adaptive eta adjustment
        if cur_obj > best_obj * 1.01:
            eta *= 0.5
        
    return best_a, best_obj


def fista_lse(a_init, max_time, rng, beta_init=10.0, beta_rate=0.01,
              beta_max=10000.0, step_scale=1.0):
    """FISTA with LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = 2.0 * n * fft_convolve_self(a).max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        cm, grad, c = lse_gradient(y, beta)
        cur_obj = 2.0 * n * c.max() / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        step_local = step
        for _ in range(5):
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = 2.0 * n * fft_convolve_self(x_trial).max() / (S * S)
            if obj_trial <= cur_obj + 1e-10:
                break
            step_local *= 0.5
        
        x_new = project_simplex(y - step_local * grad, S)
        step = min(step * 1.02, 2.0 / max(beta, 1.0))
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        if it - last_improve > 3000:
            t_mom = 1.0

def simulated_annealing(a_init, max_time, rng, T_init=0.01, T_min=1e-6, cooldown=0.9999):
    """Simulated annealing with efficient c-update."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    cur_max = c.max()
    cur_obj = 2.0 * n * cur_max / (S * S)
    
    start = time.monotonic()
    best_obj = cur_obj
    best_a = a.copy()
    T = T_init
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        if it % 1000 == 0:
            T = max(T_min, T_init * cooldown ** it)
        
        # Pick two random indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try transferring delta from i to j
        delta = a[i] * 0.01 * rng.uniform(0.5, 1.5)
        if delta < 1e-12 or delta >= a[i]:
            delta = a[i] * 0.005
        
        # Compute change in c
        # c[k] changes by: -delta*a[k-i] + delta*a[k-j] for k in affected range
        # (ignoring O(delta^2) terms)
        # Affected k for index i: k in [i, i+n-1]
        # Affected k for index j: k in [j, j+n-1]
        
        # For efficiency, only check if the new max might decrease
        # The max can only decrease if we're reducing a peak
        k_star = int(np.argmax(c))
        
        # Check contribution of i and j to c[k_star]
        contrib_i = 0.0
        if 0 <= k_star - i < n:
            contrib_i = a[k_star - i]
        contrib_j = 0.0
        if 0 <= k_star - j < n:
            contrib_j = a[k_star - j]
        
        # New c[k_star] ≈ c[k_star] - 2*delta*contrib_i + 2*delta*contrib_j
        # (factor of 2 because of the subgradient formula)
        new_peak_est = cur_max - 2.0 * delta * (contrib_i - contrib_j)
        
        if new_peak_est >= cur_max:
            # This move doesn't help the peak, skip
            continue
        
        # Actually compute the new c (only affected parts)
        c_new = c.copy()
        
        # Update for reducing a[i] by delta
        lo_i = max(0, i)
        hi_i = min(2 * n - 2, i + n - 1)
        for k in range(lo_i, hi_i + 1):
            m = k - i
            if 0 <= m < n:
                c_new[k] -= 2.0 * delta * a[m] - delta * delta * (1.0 if k == 2 * i else 0.0)
        
        # Update for increasing a[j] by delta
        lo_j = max(0, j)
        hi_j = min(2 * n - 2, j + n - 1)
        for k in range(lo_j, hi_j + 1):
            m = k - j
            if 0 <= m < n:
                c_new[k] += 2.0 * (a[m] + (delta if m == 0 else 0.0)) * delta - delta * delta * (1.0 if k == 2 * j else 0.0)
        
        new_max = c_new.max()
        new_obj = 2.0 * n * new_max / (S * S)
        delta_obj = new_obj - cur_obj
        
        # Accept or reject
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, T_min)):
            a[i] -= delta
            a[j] += delta
            c = c_new
            cur_max = new_max
            cur_obj = new_obj
            
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve = it
        
        if it - last_improve > 50000:
            # Restart with perturbation
            a = project_simplex(best_a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            c = fft_convolve_self(a)
            cur_max = c.max()
            cur_obj = 2.0 * n * cur_max / (S * S)
            last_improve = it
            T = T_init * 0.1
    
    return best_a, best_obj

def spectral_equalization(a_init, max_time, rng, target_flatness=1.0):
    """Iteratively flatten the power spectrum of a."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    # Target: flat spectrum (white noise)
    # But DC component must be S (the sum)
    
    while time.monotonic() - start < max_time:
        it += 1
        A = fft(a, n)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target magnitude: DC = S, others = S/sqrt(n-1) (to match total energy)
        # Actually, for flat spectrum: |A[k]|^2 sum = sum(a^2) by Parseval
        # We want |A[k]|^2 to be as flat as possible
        # Target: |A[0]| = S, |A[k]| = sqrt(sum(a^2)/n) for k != 0
        
        sum_a2 = np.sum(a**2)
        target_mag = np.full(n, np.sqrt(sum_a2 / n))
        target_mag[0] = S
        
        # Blend: A_new = A * (target_mag / mag)^alpha
        alpha = 0.1  # blending parameter
        ratio = target_mag / np.maximum(mag, 1e-10)
        ratio = np.clip(ratio, 0.5, 2.0)
        mag_new = mag * ratio ** alpha
        mag_new[0] = S  # preserve DC
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))
        
        # Project to simplex
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * (S / s_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            last_improve = it
            a = a_new
        else:
            # Reduce alpha
            alpha *= 0.9
            if alpha < 0.001:
                break
        
        a = a_new
    
    return best_a, best_obj

def topk_lse_gradient(a, beta, K=10):
    """Weighted gradient using top-K lags with smooth weighting."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    
    # Use LSE weighting but with a cap on K
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    w = e / e.sum()
    
    # Optional: zero out weights below a threshold (keep top K effectively)
    threshold = np.sort(w)[::-1][min(K, len(w)-1)]
    w[w < threshold * 0.01] = 0
    w /= w.sum()
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return cm, grad, c

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-algorithm search combining FISTA with LSE smoothing,
mirror descent (exponentiated gradient), simulated annealing with efficient
convolution updates, and multi-scale upscaling. Key improvements over previous:
- Mirror descent with natural simplex geometry
- Simulated annealing for escaping local minima
- Better beta scheduling to maintain informative gradient signal
- Upscaling to n=16384 for more degrees of freedom
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def compute_obj_from_c(c, n, S):
    return 2.0 * n * c.max() / (S * S)


def lse_gradient(a, beta):
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return cm, grad, c


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fista_optimize(a_init, max_time, rng, beta_init=10.0, beta_rate=0.008,
                   beta_max=5000.0, step_scale=1.0, use_backtrack=True):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best_obj = compute_obj_from_c(c0, n, S)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        cm, grad, c = lse_gradient(y, beta)
        cur_obj = compute_obj_from_c(c, n, S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            step_local = step
            for _ in range(4):
                x_trial = project_simplex(y - step_local * grad, S)
                c_trial = fft_convolve_self(x_trial)
                obj_trial = compute_obj_from_c(c_trial, n, S)
                if obj_trial <= cur_obj + 1e-10:
                    break
                step_local *= 0.5
            x_new = project_simplex(y - step_local * grad, S)
            step = min(step * 1.02, 2.0 / max(beta, 1e-6))
        else:
            step = step_scale / max(beta, 1.0)
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        stuck = it - last_improve
        if stuck > 4000:
            t_mom = 1.0
            noise_scale = max(1e-6, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 8000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    return best_a, best_obj


def mirror_descent_optimize(a_init, max_time, rng, eta_init=0.5, eta_decay=0.9995,
                            beta_init=5.0, beta_rate=0.005, beta_max=3000.0):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 1e-15, 1000).copy()
    a = a / a.sum() * S

    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best_obj = compute_obj_from_c(c0, n, S)
    best_a = a.copy()
    it = 0
    last_improve = 0
    eta = eta_init
    log_a = np.log(a)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        cm, grad, c = lse_gradient(a, beta)
        cur_obj = compute_obj_from_c(c, n, S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 8000:
            break

        # Mirror descent: a_new = a * exp(-eta * grad) / Z
        log_a -= eta * grad
        # Numerical stability
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            # Reset
            a_new = np.ones(n) / n
            log_a = np.log(a_new)
            a_new_sum = 1.0
        a_new = a_new / a_new_sum * S
        a = a_new
        log_a = np.log(a)

        eta *= eta_decay
        # Adaptive: if stuck, try larger eta
        if stuck > 2000 and it % 500 == 0:
            eta = eta * 1.5

    return best_a, best_obj


def simulated_annealing_optimize(a_init, max_time, rng, T_init=0.02, T_min=1e-8,
                                 cooldown=0.99999):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a).copy()
    cur_max = c.max()
    cur_obj = compute_obj_from_c(c, n, S)

    start = time.monotonic()
    best_obj = cur_obj
    best_a = a.copy()
    T = T_init
    it = 0
    last_improve = 0
    n_accept = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Update temperature
        T = max(T_min, T_init * cooldown ** (it % 500000))

        # Pick source and target
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Delta: fraction of a[i] to transfer
        frac = rng.uniform(0.002, 0.015)
        delta = a[i] * frac
        if delta < 1e-14 or delta >= a[i] * 0.99:
            continue

        # Efficient c update: only modify affected positions
        # c[k] = sum_j a[j]*a[k-j]
        # Changing a[i] by -delta and a[j] by +delta:
        # c[k] -= 2*delta*a[k-i] (for valid k-i) + 2*delta*a[k-j] (for valid k-j)
        # (the cross term delta^2 is negligible)

        lo_i = max(0, i)
        hi_i = min(2 * n - 2, i + n - 1)
        lo_j = max(0, j)
        hi_j = min(2 * n - 2, j + n - 1)

        # Work on a temporary copy of c
        c_new = c.copy()
        idx_i = np.arange(lo_i, hi_i + 1)
        m_i = idx_i - i
        c_new[idx_i] -= 2.0 * delta * a[m_i]

        idx_j = np.arange(lo_j, hi_j + 1)
        m_j = idx_j - j
        c_new[idx_j] += 2.0 * delta * a[m_j]

        new_max = c_new.max()
        new_obj = compute_obj_from_c(c_new, n, S)
        delta_obj = new_obj - cur_obj

        # Metropolis acceptance
        if delta_obj < 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta_obj / max(T, 1e-15))

        if accept:
            a[i] -= delta
            a[j] += delta
            c = c_new
            cur_max = new_max
            cur_obj = new_obj
            n_accept += 1
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve = it

        # Periodic restart if stuck
        if it - last_improve > 30000 and it % 5000 == 0:
            #Perturb from best
            n_perturb = rng.integers(5, 50)
            for _ in range(n_perturb):
                src = rng.integers(0, n)
                tgt = rng.integers(0, n)
                if src != tgt and a[src] > 1e-10:
                    d = a[src] * 0.01
                    a[src] -= d
                    a[tgt] += d
            a = np.maximum(a, 0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            cur_max = c.max()
            cur_obj = compute_obj_from_c(c, n, S)
            last_improve = it
            T = max(T, 0.01)

    return best_a, best_obj


def upsample_smooth(a, factor=2):
    n = len(a)
    n_new = n * factor
    # Use FFT-based interpolation for smooth upscaling
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    # Zero-pad in frequency to upsample
    A_new = np.zeros(n_new, dtype=complex)
    half = fsize // 2
    A_new[:n] = A[:n]
    A_new[-(fsize - n):] = A[half+n - fsize // 2:] if fsize > n else 0
    # Simpler: just use interpolation
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    a_new = np.maximum(0, np.interp(t_new, t_old, a))
    if a_new.sum() < 1e-15:
        a_new = np.ones(n_new)
    return a_new


def make_spectral_init(n, rng):
    """Create initial sequence with near-flat power spectrum."""
    S = np.sqrt(2.0 * n)
    # Create a phase-randomized flat-spectrum signal
    mag = np.ones(n)
    mag[0] = n  # DC component will be the sum
    phase = rng.uniform(0, 2 * np.pi, n)
    phase[0] = 0
    # Add some spectral shaping to ensure non-negativity is achievable
    freq_idx = np.arange(n)
    # Low-pass + some high freq content
    shaping = 1.0 + 0.5 * np.exp(-freq_idx.astype(float) / (n * 0.05))
    shaping[0] = 1.0
    A = mag * shaping * np.exp(1j * phase)
    a = np.real(ifft(A))
    a = np.maximum(a, 0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    a = a * (S / a.sum())
    return a


def make_cosine_init(n, rng):
    S = np.sqrt(2.0 * n)
    t = np.linspace(0, 1, n, endpoint=False)
    # Multi-frequency cosine for good spectral properties
    a = 1.0
    for f, amp, ph in [(2, 0.3, 0.0), (3, 0.2, 1.0), (5, 0.15, 2.0),
                       (7, 0.1, 0.5), (11, 0.08, 3.0), (13, 0.06, 1.5)]:
        a += amp * np.sin(2 * np.pi * f * t + ph)
    a = np.abs(a)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Try to load previous best
    n_start = 8192
    S_start = np.sqrt(2.0 * n_start)
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        a_start = make_cosine_init(n_start, rng)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Start: n=%d, obj=%.6f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 0: Quick FISTA at original size (5% time) ===
    remaining = deadline - time.monotonic()
    p0_time = remaining * 0.05
    if p0_time > 20:
        print("\n--- Phase 0: Quick FISTA n=%d (%.0fs) ---" % (n0, p0_time))
        a_opt, obj_opt = fista_optimize(global_best_a, p0_time, rng,
                                         beta_init=20, beta_rate=0.01,
                                         beta_max=3000, step_scale=0.8,
                                         use_backtrack=False)
        print("  obj=%.6f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

    # === Phase 1: Upsample to 2x (10% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.10
    n_target = min(16384, n0 * 2)
    if n_target > n0 and p1_time > 30:
        print("\n--- Phase 1: Upsample to n=%d (%.0fs) ---" % (n_target, p1_time))
        a_up = upsample_smooth(global_best_a, factor=n_target // n0)
        S_up = np.sqrt(2.0 * n_target)
        a_up = a_up * (S_up / a_up.sum())
        obj_up = compute_obj(a_up)
        print("  Upsampled: obj=%.6f" % obj_up)
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up.copy()

        a_up_opt, obj_up_opt = fista_optimize(a_up, p1_time * 0.7, rng,
                                               beta_init=5, beta_rate=0.008,
                                               beta_max=2000, step_scale=1.0,
                                               use_backtrack=False)
        print("  FISTA upsampled: obj=%.6f" % obj_up_opt)
        if obj_up_opt < global_best_obj:
            global_best_obj = obj_up_opt
            global_best_a = a_up_opt.copy()
    else:
        n_target = n0

    # === Phase 2: Intensive FISTA at target size (25% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.25
    print("\n--- Phase 2: FISTA n=%d (%.0fs) ---" % (len(global_best_a), p2_time))

    # Run multiple FISTA variants
    variants = [
        (15, 0.008, 5000, 1.0, False),
        (50, 0.004, 10000, 0.7, True),
        (200, 0.002, 30000, 0.4, False),
    ]
    for vi, (bi, br, bm, ss, bt) in enumerate(variants):
        if time.monotonic() >= deadline:
            break
        rem = min(p2_time * 0.35, deadline - time.monotonic())
        if rem < 10:
            break
        a_opt, obj_opt = fista_optimize(global_best_a, rem, rng,
                                         beta_init=bi, beta_rate=br,
                                         beta_max=bm, step_scale=ss,
                                         use_backtrack=bt)
        print("  FISTA variant %d: obj=%.6f" % (vi, obj_opt))
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()
            print("    *** NEW BEST ***")

    # === Phase 3: Mirror Descent (20% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.20
    print("\n--- Phase 3: Mirror Descent n=%d (%.0fs) ---" % (len(global_best_a), p3_time))

    md_variants = [
        (0.5, 0.9995, 5, 0.005, 3000),
        (1.0, 0.999, 10, 0.003, 2000),
        (2.0, 0.998, 3, 0.008, 5000),
    ]
    for vi, (ei, ed, bi, br, bm) in enumerate(md_variants):
        if time.monotonic() >= deadline:
            break
        rem = min(p3_time * 0.35, deadline - time.monotonic())
        if rem < 10:
            break
        a_opt, obj_opt = mirror_descent_optimize(global_best_a, rem, rng,
                                                  eta_init=ei, eta_decay=ed,
                                                  beta_init=bi, beta_rate=br,
                                                  beta_max=bm)
        print("  MD variant %d: obj=%.6f" % (vi, obj_opt))
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()
            print("    *** NEW BEST ***")

    # === Phase 4: Simulated Annealing (25% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.25
    n_sa = len(global_best_a)
    # SA is expensive for very large n; cap at 16384
    if n_sa > 16384:
        # Downsample for SA
        n_sa = 16384
        t_old = np.linspace(0, 1, len(global_best_a), endpoint=False)
        t_new = np.linspace(0, 1, n_sa, endpoint=False)
        a_sa = np.interp(t_new, t_old, global_best_a)
        a_sa = np.maximum(a_sa, 0)
        S_sa = np.sqrt(2.0 * n_sa)
        a_sa = a_sa * (S_sa / a_sa.sum())
    else:
        a_sa = global_best_a.copy()
        S_sa = np.sqrt(2.0 * n_sa)

    print("\n--- Phase 4: Simulated Annealing n=%d (%.0fs) ---" % (n_sa, p4_time))
    a_sa_opt, obj_sa_opt = simulated_annealing_optimize(a_sa, p4_time * 0.7, rng,
                                                        T_init=0.015, T_min=1e-9,
                                                        cooldown=0.99998)
    print("  SA: obj=%.6f" % obj_sa_opt)
    if obj_sa_opt < global_best_obj:
        global_best_obj = obj_sa_opt
        global_best_a = a_sa_opt.copy()
        print("    *** NEW BEST ***")

    # FISTA polish after SA
    remaining4 = deadline - time.monotonic()
    if remaining4 > 15:
        a_pol, obj_pol = fista_optimize(global_best_a, remaining4 * 0.6, rng,
                                         beta_init=100, beta_rate=0.005,
                                         beta_max=20000, step_scale=0.5,
                                         use_backtrack=True)
        print("  FISA after SA: obj=%.6f" % obj_pol)
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()

    # === Phase 5: Try larger n if time permits (15% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.15
    n_current = len(global_best_a)
    if n_current <= 16384 and p5_time > 60:
        n_big = n_current * 2
        print("\n--- Phase 5: Upscale to n=%d (%.0fs) ---" % (n_big, p5_time))
        a_big = upsample_smooth(global_best_a, factor=2)
        S_big = np.sqrt(2.0 * n_big)
        a_big = a_big * (S_big / a_big.sum())
        obj_big = compute_obj(a_big)
        print("  Upscaled: obj=%.6f" % obj_big)
        if obj_big < global_best_obj:
            global_best_obj = obj_big
            global_best_a = a_big.copy()

        a_big_opt, obj_big_opt = fista_optimize(a_big, p5_time * 0.8, rng,
                                                 beta_init=3, beta_rate=0.006,
                                                 beta_max=2000, step_scale=1.0,
                                                 use_backtrack=False)
        print("  FISTA big: obj=%.6f" % obj_big_opt)
        if obj_big_opt < global_best_obj:
            global_best_obj = obj_big_opt
            global_best_a = a_big_opt.copy()

    # === Phase 6: Final high-precision polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n--- Phase 6: Final polish (%.0fs) ---" % remaining)
        # Very high beta FISTA
        a_final, obj_final = fista_optimize(global_best_a, remaining * 0.5, rng,
                                             beta_init=500, beta_rate=0.002,
                                             beta_max=100000, step_scale=0.2,
                                             use_backtrack=True)
        print("  High-beta FISTA: obj=%.6f" % obj_final)
        if obj_final < global_best_obj:
            global_best_obj = obj_final
            global_best_a = a_final.copy()

        remaining6 = deadline - time.monotonic()
        if remaining6 > 10:
            # Mirror descent final
            a_md_final, obj_md_final = mirror_descent_optimize(global_best_a,
                                                                remaining6, rng,
                                                                eta_init=0.1,
                                                                eta_decay=0.9999,
                                                                beta_init=100,
                                                                beta_rate=0.002,
                                                                beta_max=50000)
            print("  MD final: obj=%.6f" % obj_md_final)
            if obj_md_final < global_best_obj:
                global_best_obj = obj_md_final
                global_best_a = a_md_final.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL: obj=%.6f, n=%d ===" % (ver, len(final_list)))
        else:
            print("WARNING: invalid, using fallback")
            n_fb = 8192
            fb = make_cosine_init(n_fb, rng)
            final_list = [float(x) for x in fb]
    else:
        n_fb = 8192
        fb = make_cosine_init(n_fb, rng)
        final_list = [float(x) for x in fb]

    return final_list
