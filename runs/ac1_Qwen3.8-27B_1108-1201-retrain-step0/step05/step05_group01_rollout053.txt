"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses adaptive LSE-FISTA with backtracking, multi-lag objective,
KKT-based refinement, and multi-scale search (n=8192, 16384).
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def cross_corr_weighted(a, w):
    """Compute 2 * cross-correlation of w and reversed a (gradient of weighted sum of conv)."""
    n = len(a)
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    a_rev = a[::-1]
    result = np.real(ifft(fft(w, L) * fft(a_rev, L)))
    return 2.0 * result[n-1:2*n-1]


def kkt_residual(a, c):
    """Compute KKT residual: how far the gradient is from being constant on the support."""
    n = len(a)
    max_val = c.max()
    active = np.where(c >= max_val * 0.999)[0]
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    grad = cross_corr_weighted(a, w)
    support = a > 1e-10 * a.max()
    if support.sum() < 2:
        return 0.0
    grad_support = grad[support]
    return np.std(grad_support) / (np.abs(grad_support).mean() + 1e-15)


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

                        beta_start=20.0, beta_rate=1.02, beta_max=20000.0,
                        anneal_period=3000, backtrack_max=10):
    """
    FISTA with adaptive LSE/multi-lag objective, backtracking line search,
    and periodic annealing.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    anneal_count = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive number of lags
        n_lags = min(n_lags_max, n_lags_start + it // 500)

        # LSE beta with periodic annealing
        if it % anneal_period == 0:
            anneal_count += 1
            beta = beta_start * (1.0 + 0.5 * anneal_count)
            t_mom = 1.0  # Reset momentum on anneal
        else:
            beta = min(beta_max, beta_start * (beta_rate ** (it % anneal_period)))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()

        # Compute gradient using top-k or LSE
        if n_lags <= 8:
            # Use LSE
            cm = c.max()
            shifted = beta * (c - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
        else:
            # Use top-k
            k = min(n_lags, len(c))
            idx = np.argsort(c)[::-1][:k]
            w = np.zeros(len(c))
            w[idx] = 1.0 / k

        grad = cross_corr_weighted(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step = S / (beta * gnorm + 1.0)
        obj_target = cur_max  # We want to decrease the max

        for bt in range(backtrack_max):
            x_new = project_simplex(y - step * grad, S)
            # Check if this improves the weighted objective
            c_new = fft_convolve_self(x_new)
            if (c_new).max() < cur_max + 0.01 * cur_max:  # Allow small increase
                break
            step *= 0.5

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 5000 == 0:
            t_mom = 1.0

    return best_a, best_conv


def run_kkt_refine(a_init, max_time, rng):
    """Refine using KKT conditions: drive gradient to be constant on support."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()

        # Find active lags (within tolerance of max)
        max_val = c.max()
        tol = max(1e-6, 0.001 * max_val * (1 + 0.1 * np.log1p(it / 100.0)))
        active = np.where(c >= max_val - tol)[0]

        if len(active) == 0:
            active = np.array([np.argmax(c)])

        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        grad = cross_corr_weighted(a, w)

        # Project gradient: make it constant on the support
        support = a > 1e-8 * a.max()
        if support.sum() > 0:
            grad_const = grad[support].mean()
            grad_proj = grad.copy()
            grad_proj[support] = grad_const
            # For non-support, keep the gradient as is (it should push toward 0)
        else:
            grad_proj = grad

        gnorm = np.linalg.norm(grad_proj)
        if gnorm < 1e-12:
            break

        # Adaptive step: aim to reduce the KKT residual
        step = 0.5 * S / (gnorm * np.sqrt(n))
        a_new = project_simplex(a - step * grad_proj, S)

        # Check if improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller step
            a_new = project_simplex(a - 0.1 * step * grad_proj, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max:
                a = a_new
            else:
                # Accept anyway with very small step
                a = project_simplex(a - 0.01 * step * grad_proj, S)

    return best_a, best_conv

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization combining adaptive LSE-FISTA with backtracking,
KKT-driven exact subgradient refinement, L-BFGS polishing, and multi-scale search
(n=8192 and n=16384). Key innovations: (1) oscillating-then-growing beta schedule
with full momentum resets, (2) KKT residual tracking to switch between smooth and
exact modes, (3) projected-gradient equalization on the support set.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def cross_corr_grad(a, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. a, where b=conv(a,a)."""
    n = len(a)
    L = 1 << (len(w) + n - 2).bit_length()
    a_rev = a[::-1]
    result = np.real(ifft(fft(w, L) * fft(a_rev, L)))
    return 2.0 * result[n - 1:2 * n - 1]


def lse_weights(c, beta):
    """Softmax weights for LSE approximation of max."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def topk_weights(c, k, power=1.0):
    """Top-k weights with power weighting."""
    n = len(c)
    k = min(k, n)
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(n)
    w_full[idx] = w
    return w_full


def compute_obj(a):
    """Compute the actual objective 2*n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def kkt_residue(a, c):
    """Compute KKT residual: std of gradient on support / mean magnitude."""
    n = len(a)
    max_val = c.max()
    active = np.where(c >= max_val * 0.9995)[0]
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    grad = cross_corr_grad(a, w)
    support = a > 1e-10 * a.max()
    if support.sum() < 2:
        return 0.0
    g_s = grad[support]
    return np.std(g_s) / (np.abs(g_s).mean() + 1e-15)


def run_fista_v2(a_init, max_time, rng, beta_start=20.0, beta_growth=1.005,
                 beta_max=50000.0, reset_period=4000, mode="adaptive"):
    """
    FISTA with adaptive mode switching:
    - Starts with LSE (smooth), transitions to top-k, then to exact subgradient.
    - Periodic momentum resets and beta resets for escape.
    - Adaptive step size based on gradient norm.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    best_iter = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: grows, then resets periodically
        phase_len = it % reset_period
        if phase_len == 0 and it > 0:
            # Reset: lower beta, reset momentum
            beta = beta_start * (1.0 + 0.3 * (it // reset_period))
            t_mom = 1.0
            y = a.copy()
        else:
            beta = min(beta_max, beta_start * (beta_growth ** phase_len))

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_conv - 1e-13:
            best_conv = cur_max
            best_a = y.copy()
            best_iter = it
            no_improve = 0
        else:
            no_improve += 1

        # Determine mode based on progress
        if no_improve < 200:
            # Fresh: use LSE with current beta
            w = lse_weights(c, min(beta, 500.0))
            grad = cross_corr_grad(y, w)
            step = S / (min(beta, 500.0) * np.linalg.norm(grad) / np.sqrt(n) + 1.0)
        elif no_improve < 1500:
            # Stuck: use top-k with growing k
            k = min(32, 4 + (no_improve - 200) // 50)
            w = topk_weights(c, k, power=1.0)
            grad = cross_corr_grad(y, w)
            step = S / (np.linalg.norm(grad) / np.sqrt(n) * 2.0 + 1.0)
        else:
            # Very stuck: exact subgradient with wider active set
            max_val = c.max()
            tol_frac = min(0.05, 0.002 * (no_improve - 1500) / 500.0)
            active = np.where(c >= max_val * (1.0 - tol_frac))[0]
            w = np.zeros(len(c))
            w[active] = 1.0 / len(active)
            grad = cross_corr_grad(y, w)
            gnorm = np.linalg.norm(grad)
            step = S / (gnorm * (1.0 + 0.001 * it))
            step = min(step, 0.5 * S / (gnorm + 1e-15))

        # DCT-domain perturbation when very stuck
        if no_improve > 3000 and it % 200 == 0:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-no_improve / 10000.0)
            if noise_scale > 1e-8:
                noise = rng.normal(0, 1.0, n)
                F = fft(noise)
                cutoff = max(3, n // 100)
                F[cutoff:-cutoff] *= 0.05
                noise_s = np.real(ifft(F))[:n]
                noise_s = noise_s / (np.linalg.norm(noise_s) + 1e-15) * noise_scale
                y = project_simplex(y + noise_s, S)
                t_mom = 1.0
                no_improve = 0
                continue

        # FISTA update
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 3000 == 0:
            t_mom = 1.0
            y = a.copy()

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_kkt_refine_v2(a_init, max_time, rng):
    """
    KKT-driven refinement: directly minimize the KKT residual.
    At optimum, gradient must be constant on support. We project the
    gradient to be constant on support and take steps in that direction.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    y = a.copy()
    t_mom = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_conv - 1e-13:
            best_conv = cur_max
            best_a = y.copy()

        # Active set with adaptive tolerance
        max_val = c.max()
        # Tolerance grows slowly to include more lags
        tol = max_val * (1.0 - min(0.01, 0.0005 * it / 500.0))
        active = np.where(c >= tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        grad = cross_corr_grad(y, w)

        # KKT projection: equalize gradient on support
        support = a > 1e-9 * a.max()
        if support.sum() > 1:
            grad_eq = grad.copy()
            grad_eq[support] = grad[support].mean()
            # For non-support, use a damping (push toward boundary)
            grad_eq[~support] = grad[~support] * 0.5
        else:
            grad_eq = grad

        gnorm = np.linalg.norm(grad_eq)
        if gnorm < 1e-14:
            break

        # Adaptive step: small for stability
        step = 0.3 * S / (gnorm * np.sqrt(n))
        step = min(step, S / (0.5 * gnorm + 1e-15))

        x_new = project_simplex(y - step * grad_eq, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic check: try exact max reduction
        if it % 100 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() < best_conv - 1e-13:
                best_conv = c_check.max()
                best_a = a.copy()
                y = a.copy()
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_lbfgs_v2(a_init, max_time, rng, n_levels=4):
    """L-BFGS-B polish with increasing LSE smoothness."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    configs = [
        (30.0, 5.0, 150),
        (100.0, 10.0, 150),
        (400.0, 20.0, 200),
        (1000.0, 50.0, 200),
    ][:n_levels]

    for beta, pen, maxiter in configs:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 3:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / beta
            sum_x = x.sum()
            obj += pen * (sum_x - S) ** 2
            grad = cross_corr_grad(x, w)
            grad += 2.0 * pen * (sum_x - S)
            return obj, grad

        t0 = time.monotonic()
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': maxiter, 'maxls': 40,
                     'ftol': 1e-15, 'gtol': 1e-12, 'iprint': -1}
        )
        elapsed = time.monotonic() - t0

        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            print("    L-BFGS beta=%.0f: obj=%.8f (t=%.1fs, nit=%d)" %
                  (beta, best_obj, elapsed, result.nit))
        else:
            print("    L-BFGS beta=%.0f: no improvement (t=%.1fs, nit=%d)" %
                  (beta, elapsed, result.nit))

    return best_a, best_obj

    # === Phase 7: Final squeeze with remaining time ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n=== Phase 7: Final squeeze (%.0fs) ===" % remaining)
        t7 = time.monotonic()
        # Quick LSE FISTA with high beta
        a7, _ = run_fista_v2(global_best_a, remaining * 0.5, rng,
                             beta_start=50.0, beta_growth=1.01,
                             beta_max=50000.0, reset_period=2000, mode="adaptive")
        update_best(a7)
        rem7 = remaining - (time.monotonic() - t7)
        if rem7 > 5:
            a7b, _ = run_kkt_refine_v2(global_best_a, rem7, rng)
            update_best(a7b)
        print("  Final: obj=%.8f (t=%.0fs)" % (global_best_obj, time.monotonic() - t7))

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nVerified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: verification failed, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective: %.8f" % global_best_obj)
    return final_list
