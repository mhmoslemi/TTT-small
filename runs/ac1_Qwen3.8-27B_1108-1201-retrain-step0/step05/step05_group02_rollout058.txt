import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def solve_lp_direction(f, M, active_indices):
    """
    Solve: max sum(g) s.t. g >= 0, (f*g)(k) <= M for k in active_indices
    
    Returns the optimal g.
    """
    n = len(f)
    K = len(active_indices)
    
    # Build constraint matrix A such that A @ g <= M * ones(K)
    # (f*g)(k) = sum_j f[j] * g[k-j] for valid j
    # For each active index k, the constraint is:
    # sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j] <= M
    
    # A is K x n
    A = np.zeros((K, n))
    for i, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[i, j] = f[k - j]
    
    # Objective: max sum(g) = -min(-sum(g))
    c_obj = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    # Constraints: A @ g <= M
    b = np.full(K, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        return None

def lp_active_set_method(a_init, max_time, K=30, mixing_rate=0.1):
    """
    Active set LP method.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()
        cur_obj = 2.0 * n * M / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        # Get top-K active indices
        top_idx = np.argsort(c)[::-1][:K]
        # Only use those that are close to M
        threshold = M * 0.95
        active = top_idx[c[top_idx] >= threshold]
        if len(active) == 0:
            active = top_idx[:min(5, K)]
        
        # Solve LP
        g = solve_lp_direction(a, M, active)
        if g is None:
            # LP failed, try with fewer constraints
            active_small = top_idx[:min(5, K)]
            g = solve_lp_direction(a, M, active_small)
            if g is None:
                break
        
        sum_g = g.sum()
        if sum_g < S * 0.99:
            # g is not significantly better, break
            if it - last_improve > 100:
                break
            # Try mixing with a smaller rate
            t = 0.01
        else:
            t = mixing_rate
        
        # Rescale and mix
        g_rescaled = g * (S / sum_g)
        a_new = (1 - t) * a + t * g_rescaled
        a_new = np.maximum(a_new, 0)
        a_new = a_new * (S / a_new.sum())
        
        a = a_new
        
        if it % 50 == 0:
            print("  LP iter %d: obj=%.6f, M=%.4f, sum_g=%.2f" % (it, cur_obj, M, sum_g))
    
    return best_a, best_obj

Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses pair-exchange optimization with exact line search, LP-based
direction finding, and subgradient polish.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def find_best_delta(v, s, max_delta):

def lse_gradient(y, beta):
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def run_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.01,
              beta_max=10000.0, step_scale=1.0):

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_gradient(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = step_scale / max(beta, 1.0)
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        if it - last_improve > 3000:
            t_mom = 1.0
            noise_scale = max(1e-6, 0.002 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
    
    return best_a, best_obj


def subgradient_polish(a_init, max_time, rng):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        k = min(16, 1 + stuck // 200)
        
        # Top-k subgradient
        idx = np.argsort(c)[::-1][:k]
        w_full = np.zeros(len(c))
        w_full[idx] = 1.0 / k
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = S / (gnorm * (1.0 + 0.001 * it))
        step = min(step, 3.0 * S / gnorm)
        
        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-13:
            a = a_new
        else:
            step *= 0.1
            a_new = project_simplex(a - step * grad, S)
            if compute_obj(a_new) <= cur_obj + 1e-13:
                a = a_new
        
        if stuck > 2000 and it % 200 == 0:
            ns = max(1e-7, 0.0005 * S / np.sqrt(n) * np.exp(-stuck / 5000))
            a = project_simplex(a + rng.normal(0, ns, n), S)
    
    return best_a, best_obj

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses pair-exchange optimization with exact bisection line search,
FISTA with LSE smoothing, and subgradient polish.
The key innovation is the pair-exchange method which directly finds
optimal mass transfers that reduce the autocorrelation peak.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def find_best_delta(c, s, max_delta):
    """Find delta in [0, max_delta] minimizing max(c + delta*s) via bisection."""
    if max_delta <= 0:
        return 0.0
    lo = 0.0
    hi = max_delta
    for _ in range(50):
        mid = (lo + hi) * 0.5
        c_mid = c + mid * s
        k_star = np.argmax(c_mid)
        if s[k_star] > 1e-18:
            hi = mid
        elif s[k_star] < -1e-18:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5


def compute_slope(c_len, n, j, l, a):
    """Compute s[k] = a[k-l] - a[k-j] for k in [0, 2n-2]."""
    s = np.zeros(c_len)
    js = max(0, j)
    je = min(c_len - 1, j + n - 1)
    if je >= js:
        s[js:je + 1] -= a[0:je - js + 1]
    ls = max(0, l)
    le = min(c_len - 1, l + n - 1)
    if le >= ls:
        s[ls:le + 1] += a[0:le - ls + 1]
    return s


def pair_exchange_optimize(a_init, max_time, rng, n_pairs=16, top_peaks=5):
    """
    Pair exchange: transfer mass from high-contributing to low-contributing
    positions using exact bisection line search for optimal transfer amount.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    c_len = len(c)
    cur_max = c.max()

    start = time.monotonic()
    best_obj = cur_max
    best_a = a.copy()
    it = 0
    last_improve = 0
    stagnant = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()

        if cur_max < best_obj - 1e-13:
            best_obj = cur_max
            best_a = a.copy()
            last_improve = it
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 8000:
            # Random restart within neighborhood
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            c = fft_convolve_self(a)
            stagnant = 0
            continue

        if it % 2000 == 0:
            print("    PE iter %d: obj=%.8f, stagnant=%d" % (it, cur_max, stagnant))

        # Identify top peak lags
        top_idx = np.argsort(c)[::-1][:top_peaks]

        # Compute contribution of each position to the top peaks
        contrib = np.zeros(n)
        for kk in top_idx:
            kk = int(kk)
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            if hi >= lo:
                contrib[lo:hi + 1] += a[kk - hi:kk - lo + 1][::-1]

        # Source: high contribution * high value (big impact on peak)
        src_score = contrib * np.maximum(a, 1e-15)
        # Target: low contribution (adding mass here won't hurt peaks much)
        tgt_score = contrib + 0.001 * a  # slight penalty for large values

        n_src = min(n_pairs // 2, n)
        n_tgt = min(n_pairs // 2, n)
        src_candidates = np.argsort(src_score)[::-1][:n_src]
        tgt_candidates = np.argsort(tgt_score)[:n_tgt]

        best_delta_obj = cur_max + 1e-15
        best_j = -1
        best_l = -1
        best_delta = 0.0
        best_s = None

        for j in src_candidates:
            j = int(j)
            if a[j] < 1e-14:
                continue
            for l in tgt_candidates:
                l = int(l)
                if j == l:
                    continue
                s = compute_slope(c_len, n, j, l, a)
                max_d = a[j] * 0.4
                delta = find_best_delta(c, s, max_d)
                if delta > 1e-14:
                    new_max = np.max(c + delta * s)
                    if new_max < best_delta_obj:
                        best_delta_obj = new_max
                        best_j = j
                        best_l = l
                        best_delta = delta
                        best_s = s

        # Also try a few random pairs if stuck
        if best_j < 0 and stagnant > 500:
            for _ in range(20):
                j = int(rng.integers(0, n))
                l = int(rng.integers(0, n))
                if j == l or a[j] < 1e-14:
                    continue
                s = compute_slope(c_len, n, j, l, a)
                delta = find_best_delta(c, s, a[j] * 0.3)
                if delta > 1e-14:
                    new_max = np.max(c + delta * s)
                    if new_max < cur_max - 1e-14:
                        best_j = j
                        best_l = l
                        best_delta = delta
                        best_s = s
                        best_delta_obj = new_max
                        break

        if best_j >= 0:
            a[best_j] -= best_delta
            a[best_l] += best_delta
            c += best_delta * best_s
            # Ensure non-negativity
            if a[best_j] < 0:
                a[best_j] = 0.0
        else:
            # Try reverse: maybe we need to move mass the other way
            if stagnant > 1000:
                # Random small perturbation
                idx1 = int(rng.integers(0, n))
                idx2 = int(rng.integers(0, n))
                if idx1 != idx2 and a[idx1] > 1e-14:
                    d = a[idx1] * 0.01
                    s = compute_slope(c_len, n, idx1, idx2, a)
                    delta = find_best_delta(c, s, d)
                    if delta > 1e-14:
                        a[idx1] -= delta
                        a[idx2] += delta
                        c += delta * s

        # Periodic full recompute to avoid numerical drift
        if it % 1000 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def lse_gradient(y, beta):
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return c, w, grad


def run_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.008,
              beta_max=5000.0, step_scale=0.8):
    """FISTA with adaptive LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c, w, grad = lse_gradient(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = step_scale / max(beta, 1.0)
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        if it - last_improve > 3000:
            t_mom = 1.0
            noise_scale = max(1e-6, 0.002 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    return best_a, best_obj


def subgradient_polish(a_init, max_time, rng):
    """Polish using exact subgradient with decreasing step size."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        k = min(32, 1 + stuck // 100)

        idx = np.argsort(c)[::-1][:k]
        w_full = np.zeros(len(c))
        w_full[idx] = 1.0 / k
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.001 * it))
        step = min(step, 3.0 * S / gnorm)

        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-13:
            a = a_new
        else:
            step *= 0.1
            a_new = project_simplex(a - step * grad, S)
            if compute_obj(a_new) <= cur_obj + 1e-13:
                a = a_new

        if stuck > 3000 and it % 500 == 0:
            ns = max(1e-7, 0.0005 * S / np.sqrt(n) * np.exp(-stuck / 6000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj


def multi_pair_exchange(a_init, max_time, rng, n_rounds_per_iter=4):
    """
    Enhanced pair exchange that does multiple non-overlapping transfers
    per iteration for faster convergence.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    c_len = len(c)
    cur_max = c.max()

    start = time.monotonic()
    best_obj = cur_max
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        if cur_max < best_obj - 1e-13:
            best_obj = cur_max
            best_a = a.copy()
            last_improve = it

        if it - last_improve > 10000:
            break

        top_idx = np.argsort(c)[::-1][:5]
        contrib = np.zeros(n)
        for kk in top_idx:
            kk = int(kk)
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            if hi >= lo:
                contrib[lo:hi + 1] += a[kk - hi:kk - lo + 1][::-1]

        src_score = contrib * np.maximum(a, 1e-15)
        tgt_score = contrib

        n_cand = min(20, n)
        src_candidates = np.argsort(src_score)[::-1][:n_cand]
        tgt_candidates = np.argsort(tgt_score)[:n_cand]

        improved = False
        for _ in range(n_rounds_per_iter):
            best_delta_obj = c.max() + 1e-15
            best_j = -1
            best_l = -1
            best_delta = 0.0
            best_s = None

            for j in src_candidates[:8]:
                j = int(j)
                if a[j] < 1e-14:
                    continue
                for l in tgt_candidates[:8]:
                    l = int(l)
                    if j == l:
                        continue
                    s = compute_slope(c_len, n, j, l, a)
                    delta = find_best_delta(c, s, a[j] * 0.3)
                    if delta > 1e-14:
                        new_max = np.max(c + delta * s)
                        if new_max < best_delta_obj:
                            best_delta_obj = new_max
                            best_j = j
                            best_l = l
                            best_delta = delta
                            best_s = s

            if best_j >= 0 and best_delta_obj < c.max() - 1e-14:
                a[best_j] -= best_delta
                a[best_l] += best_delta
                c += best_delta * best_s
                improved = True
            else:
                break

        if not improved:
            # Try random exploration
            for _ in range(30):
                j = int(rng.integers(0, n))
                l = int(rng.integers(0, n))
                if j == l or a[j] < 1e-14:
                    continue
                s = compute_slope(c_len, n, j, l, a)
                delta = find_best_delta(c, s, a[j] * 0.2)
                if delta > 1e-14:
                    new_max = np.max(c + delta * s)
                    if new_max < c.max() - 1e-14:
                        a[j] -= delta
                        a[l] += delta
                        c += delta * s
                        break

        if it % 2000 == 0:
            c = fft_convolve_self(a)
            print("    MPE iter %d: obj=%.8f" % (it, c.max()))

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA warm-up, pair-exchange optimization (primary),
    multi-round pair exchange, subgradient polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        tt = np.linspace(0, 1, n, endpoint=False)
        a_start = 0.5 + 0.5 * np.cos(2 * np.pi * tt)

    n0 = len(a_start)
    S0 = np.sqrt(2.0 * n0)
    a_start = a_start * (S0 / a_start.sum())
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.6f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: FISTA warm-up (8% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.08
    if p1_time > 30:
        print("\n=== Phase 1: FISTA warm-up (%.0fs) ===" % p1_time)
        a_opt, obj_opt = run_fista(global_best_a, p1_time, rng,
                                    beta_init=50, beta_rate=0.006,
                                    beta_max=5000, step_scale=0.7)
        print("  FISTA: obj=%.6f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

    # === Phase 2: Primary pair exchange (40% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.40
    if p2_time > 60:
        print("\n=== Phase 2: Pair exchange (%.0fs) ===" % p2_time)
        a_pe, obj_pe = pair_exchange_optimize(global_best_a, p2_time, rng,
                                               n_pairs=16, top_peaks=5)
        print("  Pair exchange: obj=%.6f" % obj_pe)
        if obj_pe < global_best_obj:
            global_best_obj = obj_pe
            global_best_a = a_pe.copy()

    # === Phase 3: FISTA on PE result (10% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.10
    if p3_time > 20:
        print("\n=== Phase 3: FISTA after PE (%.0fs) ===" % p3_time)
        a_f, obj_f = run_fista(global_best_a, p3_time, rng,
                                beta_init=100, beta_rate=0.004,
                                beta_max=10000, step_scale=0.5)
        print("  FISTA: obj=%.6f" % obj_f)
        if obj_f < global_best_obj:
            global_best_obj = obj_f
            global_best_a = a_f.copy()

    # === Phase 4: Multi-round pair exchange (25% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.25
    if p4_time > 60:
        print("\n=== Phase 4: Multi-round PE (%.0fs) ===" % p4_time)
        a_mpe, obj_mpe = multi_pair_exchange(global_best_a, p4_time, rng,
                                              n_rounds_per_iter=3)
        print("  Multi-round PE: obj=%.6f" % obj_mpe)
        if obj_mpe < global_best_obj:
            global_best_obj = obj_mpe
            global_best_a = a_mpe.copy()

    # === Phase 5: Subgradient polish (10% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.10
    if p5_time > 20:
        print("\n=== Phase 5: Subgradient polish (%.0fs) ===" % p5_time)
        a_sp, obj_sp = subgradient_polish(global_best_a, p5_time, rng)
        print("  Subgradient: obj=%.6f" % obj_sp)
        if obj_sp < global_best_obj:
            global_best_obj = obj_sp
            global_best_a = a_sp.copy()

    # === Phase 6: Final pair exchange (7% time) ===
    remaining = deadline - time.monotonic()
    p6_time = remaining * 0.70
    if p6_time > 20:
        print("\n=== Phase 6: Final PE (%.0fs) ===" % p6_time)
        a_final_pe, obj_final_pe = pair_exchange_optimize(global_best_a, p6_time, rng,
                                                           n_pairs=12, top_peaks=3)
        print("  Final PE: obj=%.6f" % obj_final_pe)
        if obj_final_pe < global_best_obj:
            global_best_obj = obj_final_pe
            global_best_a = a_final_pe.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.6f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
