import numpy as np

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot

    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross

        if gnorm < 1e-15:
            break

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 3000 == 0:
            t_mom = 1.0

        # DCT noise when stuck
        if stuck_count > 3000 and it % 100 == 0:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck_count / 10000.0)
            if noise_scale > 1e-8:
                noise = rng.normal(0, 1.0, n)
                F = fft(noise)
                cutoff = max(3, n // 100)
                F[cutoff:-cutoff] *= 0.1
                noise_smooth = np.real(ifft(F))[:n]
                noise_smooth = noise_smooth / (np.linalg.norm(noise_smooth) + 1e-15) * noise_scale
                y = project_simplex(y + noise_smooth, S)
                t_mom = 1.0
                stuck_count = 0

    return best_a, best_conv


def run_cutting_plane(a_init, max_time, active_set_size=100, damp=0.5):
    """
    Cutting-plane LP refinement.
    At each step, linearize the max constraints and solve an LP.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    best_max_b = fft_convolve_self(a).max()
    
    it = 0
    no_improve = 0
    # Track active set as the top lags
    b = fft_convolve_self(a)
    active_set = list(np.argsort(b)[::-1][:active_set_size])
    
    alpha = damp  # damping factor
    
    while time.monotonic() - start < max_time:
        it += 1
        b = fft_convolve_self(a)
        cur_max = b.max()
        
        if cur_max < best_max_b - 1e-10:
            best_max_b = cur_max
            best_a = a.copy()
            best_obj = 2.0 * n * cur_max / (S * S)
            no_improve = 0
            print("  CP iter %d: max_b=%.8f, obj=%.8f" % (it, cur_max, best_obj))
        else:
            no_improve += 1
            
        if no_improve > 500:
            # Reset: re-derive active set and reduce damping
            active_set = list(np.argsort(b)[::-1][:active_set_size])
            alpha = max(0.1, alpha * 0.5)
            no_improve = 0
        
        if alpha < 0.05:
            break
            
        # Build LP
        # Variables: a[0..n-1], M  (n+1 vars)
        # Minimize M
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # M is last variable
        
        A_rows = []
        b_rows = []
        
        for k in active_set:
            # Compute gradient of b[k] w.r.t. a at current a
            g = np.zeros(n)
            if k == 0:
                g = 2.0 * a
            elif k < n:
                g[:n-k] += a[k:]
                g[k:] += a[:n-k]
            else:
                # k >= n: b[k] = sum_{i=k-n+1}^{n-1} a[i] * a[i+k-n+1]... 
                # Actually for k >= n, b[k] = sum_{i=0}^{2n-2-k} a[n-1-i] * a[n-1-(i+k-n+1)]
                # Let me think... b[k] = sum_j a[j] * a[j+k-(n-1)] for j in range where both are valid
                # Actually, for the full convolution: b[k] = sum_{j} a[j] * a[k-j] where 0<=j<n and 0<=k-j<n
                # So j ranges from max(0, k-n+1) to min(n-1, k)
                # db[k]/da[j] = a[k-j] + a[j-(k-n+1)]... this is getting complicated
                # Let me just compute it directly
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for j in range(lo, hi + 1):
                    partner = k - j
                    # db[k]/da[j] gets a[partner]
                    g[j] += a[partner]
                    # db[k]/da[partner] gets a[j] (if partner != j)
                    if partner != j:
                        g[partner] += a[j]
            
            # Constraint: g^T a_new - M <= b[k](a)
            # i.e., g^T a_new - M <= b[k]
            row = np.concatenate([g, [-1.0]])
            A_rows.append(row)
            b_rows.append(float(b[k]))
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])
        
        bounds = [(0, None)] * (n + 1)
        
        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                            bounds=bounds, method='highs',
                            options={'maxiter': 5000, 'presolve': True})
        except Exception as e:
            print("  LP error: %s" % str(e))
            break
        
        if not result.success:
            if no_improve > 100:
                break
            continue
        
        a_lp = result.x[:n]
        M_lp = result.x[-1]
        
        # Damped step
        a_new = (1.0 - alpha) * a + alpha * a_lp
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * (S / s_new)
        
        b_new = fft_convolve_self(a_new)
        max_new = b_new.max()
        
        if max_new <= cur_max + 1e-10:
            a = a_new
        else:
            # Step was too aggressive, reduce alpha
            alpha = max(0.05, alpha * 0.7)
            a = a_new  # still take the step but with reduced future damping
        
        # Update active set
        b = fft_convolve_self(a)
        # Keep top lags, adding any newly violating ones
        top_lags = np.argsort(b)[::-1]
        active_set = list(top_lags[:active_set_size])
        
        # Adaptive: increase active set if we keep improving
        if no_improve == 0 and it % 50 == 0:
            active_set_size = min(200, active_set_size + 5)
            active_set = list(top_lags[:active_set_size])
    
    return best_a, best_obj


def run_active_polish(a_init, max_time, rng):
    """
    Intensive polishing using exact subgradient with adaptive top-k.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        if stuck < 50:
            k = 1
        elif stuck < 200:
            k = 2
        elif stuck < 500:
            k = 3
        elif stuck < 1000:
            k = 5
        elif stuck < 3000:
            k = min(20, 5 + stuck // 200)
        else:
            k = min(100, 20 + stuck // 300)

        tol_frac = min(0.05, 0.002 * stuck) if stuck > 100 else 0.0

        if k == 1:
            grad = exact_subgradient(y, c, tol_frac=0.0)
        else:
            grad = topk_gradient(y, c, k, power=1.0)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = S / (gnorm * (1.0 + 0.01 * it))
        step = min(step, 2.0 * S / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv
