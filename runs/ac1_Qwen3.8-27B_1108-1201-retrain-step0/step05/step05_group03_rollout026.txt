If `a` is non-negative and has length n, zero-padded to L, then:

For a zero-padded signal (L >= 2n-1), the circular autocorrelation equals the linear autocorrelation (with the appropriate indexing). Specifically:
- `IFFT(|A|^2)[0] = sum(a^2)` (lag 0)
- `IFFT(|A|^2)[k] = sum_{i} a[i]*a[i+k]` (lag k, for 0 <= k < n)
- `IFFT(|A|^2)[L-k] = sum_{i} a[i]*a[i+k]` (negative lag k)

So the linear autocorrelation `c = convolve(a, a)` of length 2n-1 is:
`c[j] = IFFT(|A|^2)[(j + n - 1) mod L]` ... hmm, the indexing is a bit tricky.

Actually, in numpy:
- `np.convolve(a, a)` gives c[0..2n-2] where c[j] = sum_i a[i]*a[j-i]
- The FFT approach: `A = fft(append(a, zeros(L-n)))`, `B = |A|^2`, `r = real(ifft(B))`
- Then r[0] = sum(a^2) (lag 0), r[1] = sum a[i]*a[i-1] (lag -1 or +1 depending on convention)

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovation: Spectral flattening iteration that directly targets
flattening the autocorrelation in the frequency domain, combined with
adaptive gradient-based refinement using top-k LSE smoothing.

The spectral iteration preserves the phase of the FFT while modifying
the power spectrum to reduce autocorrelation peaks, then projects back
to the non-negative simplex.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_flatten_iteration(a, alpha, L):
    """
    One step of spectral flattening.
    Preserves phase of FFT, modifies power spectrum to flatten autocorrelation.
    """
    n = len(a)
    A = fft(a, L)
    P = np.abs(A) ** 2
    
    # Compute autocorrelation
    r = np.real(ifft(P))
    
    # Ensure non-negative (should be for non-negative a, but numerical issues)
    r = np.maximum(r, 1e-30)
    
    mean_r = r.mean()
    
    # Geometric interpolation toward flat
    # r_new = r * (mean_r / r)^alpha = r^(1-alpha) * mean_r^alpha
    r_new = r * (mean_r / r) ** alpha
    
    # Get new power spectrum
    P_new = np.real(fft(r_new))
    P_new = np.maximum(P_new, 1e-30)
    
    # Update A: preserve phase, adjust magnitude
    ratio = np.sqrt(P_new / np.maximum(P, 1e-30))
    A_new = A * ratio
    
    # Recover time domain
    a_new = np.real(ifft(A_new, L))[:n]
    a_new = np.clip(a_new, 0.0, None)
    
    return a_new


def run_spectral_optimization(a_init, S, max_time, rng, 
                               alpha_init=0.3, alpha_min=0.01,
                               L=None):
    """
    Spectral flattening optimization with cooling schedule.
    """
    n = len(a_init)
    if L is None:
        L = 1 << (2 * n - 1).bit_length()
    
    a = normalize(a_init.copy(), S)
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    no_improve = 0
    
    total_iters = 500  # target number of iterations
    
    while time.monotonic() - start < max_time:
        it += 1
        # Cool down alpha over time
        progress = min(1.0, it / total_iters)
        alpha = alpha_init * (alpha_min / alpha_init) ** progress
        
        # Multiple sub-steps with decreasing alpha
        for sub in range(3):
            alpha_sub = alpha * (0.5 ** sub)
            a_new = spectral_flatten_iteration(a, alpha_sub, L)
            a = normalize(a_new, S)
        
        # Check improvement
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # If stuck, try a small perturbation
        if no_improve > 20:
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            no_improve = 0
        
        if it % 100 == 0:
            obj = 2.0 * n * best / (best_a.sum() ** 2)
            print("    Spectral iter %d: obj=%.6f, alpha=%.4f" % (it, obj, alpha))
    
    # Final check
    c = fft_convolve_self(a)
    if c.max() < best:
        best = c.max()
        best_a = a.copy()
    
    return best_a, best


def run_gradient_fista(a_init, S, max_time, rng,
                       beta_init=50.0, beta_rate=0.002, beta_max=20000.0,
                       k_frac=0.05, use_momentum=True):
    """FISTA with top-k LSE, adaptive backtracking."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0

    k = max(4, int(k_frac * (2 * n - 1)))
    L = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1

        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 1.0 / (2.0 * beta * 0.1 + 1e-10)
        x_new = project_simplex(y - step * grad, S)
        
        # Backtracking
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        # Periodic restart
        if stuck > 800:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

        # Spectal micro-perturbation
        if it % 500 == 0:
            a_temp = spectral_flatten_iteration(y, 0.1, L)
            y = normalize(a_temp, S)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_combined_polish(a_init, S, max_time, rng):
    """
    Combined polishing: alternate between spectral flattening and gradient steps.
    Uses very high beta for the gradient to target the exact max.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    L = 1 << (2 * n - 1).bit_length()
    
    k = max(4, int(0.02 * (2 * n - 1)))
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        # Spectral step (gentler as we progress)
        alpha = 0.1 * (1.0 - 0.8 * progress)
        a_spec = spectral_flatten_iteration(a, alpha, L)
        a = normalize(a_spec, S)
        
        # Gradient step
        beta = min(50000.0, 100.0 * (1.0 + it / 200.0))
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        grad = topk_lse_gradient(a, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-14:
            step = 0.5 / (1.0 + 0.0005 * it)
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max:
                a = a_new
        
        if cur_max < best or c_new.max() < best:
            final_c = fft_convolve_self(a)
            if final_c.max() < best:
                best = final_c.max()
                best_a = a.copy()

def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src, endpoint=False)
    t_new = np.linspace(0, 1, n_target, endpoint=False)
    return np.interp(t_new, t_old, a)


def random_nonneg_init(n, S, rng):
    """Generate a random non-negative initialization."""
    # Mix of uniform and random
    r = rng.random()
    if r < 0.3:
        a = np.ones(n)
    elif r < 0.6:
        a = rng.dirichlet(np.ones(n) * 0.5) * n
    else:
        a = np.abs(rng.normal(1.0, 0.5, n))
    return normalize(a, S)

Actually, one more thought. The Barzilai-Borwein step size uses:
`alpha_k = (s_k^T y_k) / (y_k^T y_k)` or `(s_k^T s_k) / (s_k^T y_k)`
