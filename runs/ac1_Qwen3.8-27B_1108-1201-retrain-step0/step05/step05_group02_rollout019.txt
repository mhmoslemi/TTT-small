"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase: FISTA (coarse) -> L-BFGS (fine) -> Active-set search (polish) -> Multi-length.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_smoothed_obj_and_grad(a, beta):
    """Compute LSE-smoothed max of convolution and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()  # numerical stability
    e = np.exp(shifted)
    tot = e.sum()
    lse_val = cm + np.log(tot) / beta
    w = e / tot
    # Gradient: d/d a_j of lse = (2/beta) * sum_k w_k * (d c_k / d a_j)
    # c_k = sum_j a[j] * a[k-j], so d c_k / d a_j = a[k-j] + a[k-j'] (symmetry)
    # Actually: c_k = sum_i a[i] * a[k-i], so d c_k / d a_j = a[k-j] (if 0<=k-j<n)
    # But also a[j] appears as a[k-i] when k-i=j, i.e., i=k-j, giving a[k-j].
    # Wait, c_k = sum_i a[i]*a[k-i]. If j is in [0,n-1], then a[j] appears when i=j (giving a[k-j])
    # and when k-i=j, i.e., i=k-j (giving a[k-j]). These are the same term!
    # Actually no: c_k = sum_i a[i]*a[k-i]. The term involving a[j] is:
    # - i=j: a[j]*a[k-j] (requires 0<=k-j<n)
    # - k-i=j, i=k-j: a[k-j]*a[j] (requires 0<=k-j<n) -- same term!
    # So d c_k / d a_j = 2*a[k-j] if 0<=k-j<n, else a[k-j] if j=k (i.e., k-j=0... wait)
    
    # Let me be more careful. c_k = sum_i a[i]*a[k-i] where a is zero-padded.
    # d c_k / d a_j: a[j] appears in the sum when i=j (term: a[j]*a[k-j]) 
    # and when k-i=j (term: a[k-j]*a[j]).
    # These are the same term (a[j]*a[k-j] = a[k-j]*a[j]), so it's counted once in the sum.
    # Wait no, in the sum sum_i a[i]*a[k-i], the term a[j]*a[k-j] appears when i=j.
    # The term a[k-j]*a[j] appears when i=k-j. If j != k-j (i.e., j != k/2), these are two different terms.
    # So d c_k / d a_j = a[k-j] (from i=j) + a[k-j] (from i=k-j, if k-j != j)
    # = 2*a[k-j] if j != k/2, and a[k-j] = a[j] if j = k/2.
    
    # More simply: d c_k / d a_j = a[k-j] + a[j-(k-j)] = a[k-j] + a[2j-k]
    # No wait, that's wrong. Let me just use the FFT approach.
    
    # The gradient of c w.r.t. a is: grad_a = 2 * ifft(fft(w) * conj(fft(a)))[:n]
    # (with proper zero-padding)
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def run_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.005,
              beta_max=50000.0, step_scale=1.0):
    """FISTA with adaptive LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_smoothed_obj_and_grad(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = step_scale / max(beta, 1.0)
        x_new = project_simplex(y - step * grad, S)
        
        # Check if step is too large (objective increased)
        obj_new = 2.0 * n * fft_convolve_self(x_new).max() / (S * S)
        if obj_new > cur_obj * 1.01 + 1e-10:
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        # Periodic restart with noise
        if it - last_improve > 2000 and it % 500 == 0:
            t_mom = 1.0
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            a = y.copy()
    
    return best_a, best_obj


def l_bfgs(a_init, max_time, rng, beta=5000.0, m_history=20):
    """L-BFGS optimization on the LSE-smoothed objective."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    # L-BFGS history
    S_list = []  # step directions
    Y_list = []  # gradient differences
    g_prev = None
    
    while time.monotonic() - start < max_time:
        it += 1
        lse_val, grad, c, w = lse_smoothed_obj_and_grad(a, beta)
        cur_obj = 2.0 * n * c.max() / (S * S)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # L-BFGS two-loop recursion to compute search direction
        if g_prev is not None and len(S_list) > 0:
            m = len(S_list)
            q = grad.copy()
            rho = [1.0 / np.dot(S_list[i], Y_list[i]) for i in range(m)]
            alpha = np.zeros(m)
            for i in range(m - 1, -1, -1):
                alpha[i] = rho[i] * np.dot(S_list[i], q)
                q -= alpha[i] * Y_list[i]
            # Initial Hessian approximation
            if m > 0:
                gamma = np.dot(S_list[-1], Y_list[-1]) / np.dot(Y_list[-1], Y_list[-1])
            else:
                gamma = 1.0 / gnorm
            r = gamma * q
            for i in range(m):
                beta_i = rho[i] * np.dot(Y_list[i], r)
                r += (alpha[i] - beta_i) * S_list[i]
            direction = -r
        else:
            direction = -grad / gnorm
        
        # Project direction to simplex tangent
        # We need direction to be in the tangent space of the simplex
        # Simple approach: project (a + direction) onto simplex, direction = projected - a
        trial = project_simplex(a + 0.1 * direction, S)
        direction = trial - a
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-15:
            break
        
        # Line search (backtracking)
        step = 1.0 / max(beta, 1.0)
        for ls in range(20):
            a_trial = a + step * direction
            a_trial = project_simplex(a_trial, S)
            obj_trial = 2.0 * n * fft_convolve_self(a_trial).max() / (S * S)
            if obj_trial < cur_obj - 1e-12 * gnorm * step * dnorm:
                break
            step *= 0.5
        else:
            step = 0.0
        
        if step < 1e-15:
            break
        
        # Update history
        a_new = project_simplex(a + step * direction, S)
        lse_new, grad_new, _, _ = lse_smoothed_obj_and_grad(a_new, beta)
        
        s_vec = a_new - a
        y_vec = grad_new - grad
        sy = np.dot(s_vec, y_vec)
        if sy > 1e-12:
            S_list.append(s_vec)
            Y_list.append(y_vec)
            if len(S_list) > m_history:
                S_list.pop(0)
                Y_list.pop(0)
        
        a = a_new
        g_prev = grad
        
        if it - last_improve > 500:
            # Restart L-BFGS history
            S_list = []
            Y_list = []
            g_prev = None
            # Add small noise
            noise = rng.normal(0, 0.0005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
    
    return best_a, best_obj


def active_set_search(a_init, max_time, rng, max_k=50):
    """Subgradient method with active set and exact line search."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    step = 0.01
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        # Determine active set size based on stagnation
        stuck = it - last_improve
        if stuck < 50:
            k = 1
        elif stuck < 200:
            k = 2
        elif stuck < 500:
            k = 5
        elif stuck < 2000:
            k = min(20, 5 + stuck // 100)
        else:
            k = min(max_k, 20 + stuck // 200)
        
        # Compute subgradient from top-k lags
        top_idx = np.argsort(c)[::-1][:k]
        fsize = 1 << (len(c) + n - 1).bit_length()
        w_full = np.zeros(len(c))
        w_full[top_idx] = 1.0 / k
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step_local = step
        for ls in range(15):
            a_trial = project_simplex(a - step_local * grad, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = 2.0 * n * c_trial.max() / (S * S)
            if obj_trial < cur_obj - 1e-14:
                break
            step_local *= 0.6
        else:
            step_local = 0.0
        
        if step_local > 1e-15:
            a = project_simplex(a - step_local * grad, S)
            step = min(step_local * 1.1, 0.1)
        else:
            step = max(0.001, step * 0.5)
            # Add noise to escape
            if stuck > 100:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n) * min(1.0, stuck/500), n)
                a = project_simplex(a + noise, S)
                if compute_obj(a) < best_obj:
                    best_obj = compute_obj(a)
                    best_a = a.copy()
                    last_improve = it
    
    return best_a, best_obj


def make_init(n, method, rng):
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5)**2 / (0.15**2))
    elif method == "u_shape":
        a = 0.3 + 0.7 * (2 * tt - 1)**2
    elif method == "bimodal":
        a = np.exp(-10*(tt-0.15)**2) + np.exp(-10*(tt-0.85)**2)
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "chirp":
        a = 0.5 + 0.5 * np.abs(np.sin(2 * np.pi * 3 * tt))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())
