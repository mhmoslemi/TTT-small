```python
        A = fft(a, fsize)
        power = np.abs(A)**2
        
        # Target: make power as flat as possible
        # Current autocorrelation
        c = np.real(ifft(power, fsize))[:2*n-1]
        cur_max = c.max()

        mean_power = power.mean()

        power_new = power * np.exp(-alpha * (power - mean_power) / mean_power)

        a = normalize(a_new, S)
    
    return a
```

```python
def multiplicative_weights(a, S, max_time, rng, beta_init=10, beta_rate=0.01):
    n = len(a)
    a = normalize(a.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    fsize_cc = 1 << (3*n - 2 - 1).bit_length()  # for cross-correlation
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(10000, beta_init * np.exp(beta_rate * it))
        
        b = fft_convolve_self(a)
        cur_max = b.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Compute λ = softmax(β * b)
        b_scaled = beta * (b - b.max())
        lam = np.exp(b_scaled)
        lam /= lam.sum()
        
        # Compute g_i = sum_k λ_k * a_{k-i} + a_i * λ_{2i}
        # Cross-correlation part
        L = fft(lam, fsize_cc)
        A = fft(a, fsize_cc)
        cross = np.real(ifft(L * np.conj(A), fsize_cc))[:n]
        
        # Diagonal correction: a_i * λ_{2i}
        diag = np.zeros(n)
        valid_2i = 2 * np.arange(n) < len(lam)
        diag[valid_2i] = a[valid_2i] * lam[2 * np.arange(n)[valid_2i]]
        
        g = cross + diag
        g = np.maximum(g, 1e-15)  # avoid division by zero
        
        # Multiplicative update
        C = S * np.mean(a) / np.mean(g)  # or just use a constant
        # Actually, for the multiplicative update to preserve the sum approximately:
        # sum(a * C/g) = C * sum(a/g) should ≈ S
        # So C = S / sum(a/g)
        ratio = np.sum(a / g)
        if ratio > 0:
            C = S / ratio
        else:
            C = S
        a_new = a * C / g
        a_new = np.clip(a_new, 0, 1000)
        
        # Renormalize
        a = normalize(a_new, S)
        
        # Damping for stability (optional)
        # a = 0.5 * best_a + 0.5 * a  (hybrid with best)
    
    return best_a, best
```

Actually, one more thought: I should also try a "pairwise" coordinate descent approach. Pick two coordinates (i, j) and optimize the redistribution of mass between them. This is exact (1D convex minimization) and can make progress even when the gradient-based methods are stuck.

For a pair (i, j), fix all other coordinates. Let x = a_i and y = a_j with x + y = c (constant). We want to minimize:
f(x) = max_k b_k(x, c-x)

where b_k depends on x and c-x linearly (for k where k-i and k-j are both in range and k-i ≠ k-j) or as x*(c-x) terms.

Actually, b_k = (base terms) + x * a_{k-i} + (c-x) * a_{k-j} + x * a_{i-k} + (c-x) * a_{j-k} (with appropriate terms)

Wait, I already established that ∂b_k/∂a_i = 2*a_{k-i} (if k-i in range). So:
b_k(x, c-x) = b_k(0, c) + x * (2*a_{k-i} - 2*a_{k-j}) (for the linear part)

Hmm, this isn't quite right because changing a_i also affects b_k through the a_{k-i} term when the "other" index is i.

Let me just do it directly. If we change a_i from x to x+δ and a_j from y to y-δ (maintaining the sum):
Δb_k = δ * (∂b_k/∂a_i - ∂b_k/∂a_j) = δ * 2 * (a_{k-i} - a_{k-j}) (assuming k-i and k-j are in range)

So b_k changes linearly in δ. The max of b_k is a piecewise-linear function of δ, and we can find the optimal δ by checking the breakpoints.

This is a 1D piecewise-linear minimization, which can be solved in O(n) time (or O(n log n) with sorting).

But doing this for all pairs is O(n^2) per sweep, which is too slow for n = 8192. We can do a "Gauss-Seidel" style: sweep through pairs in a fixed order, updating each pair exactly.

Actually, for a single pair update, we just need to find the δ that minimizes max_k (b_k + δ * d_k) where d_k = 2*(a_{k-i} - a_{k-j}). This is equivalent to finding the δ that minimizes the max of (b_k + δ*d_k) over k.

The optimal δ* is where the "active" constraints (those achieving the max) balance. This can be found by:
1. Computing the "slope" of the max function at δ=0: s = max_k {d_k : b_k + δ*d_k is the max at δ=0}... actually this is the subgradient.
2. If s < 0, increase δ; if s > 0, decrease δ.
3. Use binary search on δ.

Or more practically: the function f(δ) = max_k (b_k + δ*d_k) is convex in δ. We can find its minimum by:
- Evaluating f at δ = 0
- Computing the subgradient at δ = 0 (which is the set of d_k for k that achieve the max)
- If 0 is in the subgradient, we're at the optimum
- Otherwise, move in the direction -sign(subgradient) and use line search

This is essentially subgradient descent in 1D, which is simple to implement.

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using multiplicative weights (MW)
and FISTA with LSE smoothing, combined in a basin-hopping framework.
Uses cross-correlation-based repulsion for peak reduction.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def cross_correlate(lam, a):
    """Compute cross[i] = sum_k lam[k] * a[k-i] for i=0..n-1."""
    n = len(a)
    N = len(lam)
    fsize = 1 << (N + n - 1 - 1).bit_length()
    L = fft(lam, fsize)
    A = fft(a, fsize)
    result = np.real(ifft(L * np.conj(A), fsize))
    return result[:n]


def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_value(c, beta):
    """Log-sum-exp of beta * c."""
    cm = c.max()
    return cm + np.log(np.sum(np.exp(beta * (c - cm)))) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE(beta*c) with respect to a."""
    n = len(a)
    size = 2 * n - 1
    w = np.exp(beta * (c - c.max()))
    w /= w.sum()
    fsize = 1 << (size + n - 1 - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A), fsize))[:n]
    return grad


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def run_mw(a_init, S, max_time, rng, beta_init=5.0, beta_rate=0.005, beta_max=5000.0,
           damping=0.5, noise_interval=500):
    """Multiplicative weights algorithm for autocorrelation minimization."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        b = fft_convolve_self(a)
        cur_max = b.max()
        if cur_max < best * (1 - 1e-10):
            best = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute lambda = softmax(beta * b)
        b_shifted = beta * (b - b.max())
        # Clip to avoid underflow
        b_shifted = np.maximum(b_shifted, -500)
        lam = np.exp(b_shifted)
        lam_sum = lam.sum()
        if lam_sum < 1e-300:
            lam = np.ones(len(b)) / len(b)
        else:
            lam /= lam_sum
        
        # Cross-correlation: cross[i] = sum_k lam[k] * a[k-i]
        cross = cross_correlate(lam, a)
        cross = np.maximum(cross, 1e-30)
        
        # Multiplicative update
        # a_i * C / cross[i], choose C to preserve sum
        ratio = np.sum(a / cross)
        if ratio > 1e-30:
            C = S / ratio
        else:
            C = S
        a_new = a * C / cross
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        # Damping
        a_damped = damping * a + (1 - damping) * a_new
        a = normalize(a_damped, S)
        
        # Noise injection for exploration
        if it % noise_interval == 0 and stall > 100:
            noise_scale = 0.01 * S / np.sqrt(n)
            a = normalize(a + rng.normal(0, noise_scale, n), S)
            stall = 0
        
        # If completely stalled, do a bigger perturbation
        if stall > 2000:
            # Spectral perturbation
            fsize = 1 << (2 * n - 1 - 1).bit_length()
            A = fft(a, fsize)
            phase_noise = rng.normal(0, 0.1, fsize)
            A = A * np.exp(1j * phase_noise)
            a = np.real(ifft(A, fsize))[:n]
            a = normalize(a, S)
            stall = 0
    
    # Final check
    b_final = fft_convolve_self(a)
    if b_final.max() < best:
        best = b_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_fista_lse(a_init, S, max_time, rng, beta_init=20.0, beta_rate=0.003,
                  beta_max=5000.0, use_momentum=True):
    """FISTA with full LSE smoothing."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stall = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best * (1 - 1e-10):
            best = cur_max
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking
        step = 1.0 / max(0.1 * beta, 1e-10)
        for _ in range(10):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-12:
                break
            step *= 0.5
        
        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
            y_new = np.maximum(y_new, 0.0)
            a = x_new
            y = y_new
            t_val = t_new
        else:
            a = x_new
            y = a.copy()
        
        # Restart if stuck
        if stall > 1000:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stall = 0
    
    b_final = fft_convolve_self(a)
    if b_final.max() < best:
        best = b_final.max()
        best_a = a.copy()
    
    return best_a, best


def spectral_perturb(a, amount, rng):
    """Perturb in frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1 - 1).bit_length()
    A = fft(a, fsize)
    phase = rng.normal(0, amount, fsize)
    phase[0] = 0
    A_new = A * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() < 1e-15:
        return a.copy()
    return a_new * (a.sum() / a_new.sum())


def spectral_flatten(a, alpha, rng):
    """Flatten the power spectrum."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1 - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a
    # Soft flattening: new_power = power * (1 + alpha*(1 - power/mean_p))
    ratio = power / mean_p
    new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + ratio))
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / power)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() < 1e-15:
        return a
    return a_new * (S / a_new.sum())


def make_cosine_start(n, alpha, phase, freq=1):
    """Create a cosine-modulated starting sequence."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2 * np.pi * freq * i / n + phase)
    a = np.clip(a, 0, None)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a


def make_random_start(n, rng):
    """Create a random starting sequence."""
    strategy = rng.integers(0, 4)
    if strategy == 0:
        a = rng.uniform(0, 1, n)
    elif strategy == 1:
        # Random spectral
        fsize = 1 << (2 * n - 1 - 1).bit_length()
        A = rng.normal(0, 1, fsize) + 1j * rng.normal(0, 1, fsize)
        A[0] = abs(A[0])
        # Make it symmetric for real output
        for k in range(1, fsize // 2 + 1):
            if k < fsize - k:
                A[fsize - k] = np.conj(A[k])
        a = np.real(ifft(A, fsize))[:n]
        a = np.clip(a, 0, None)
        if a.sum() < 1e-15:
            a = rng.uniform(0, 1, n)
    elif strategy == 2:
        # Cosine
        alpha = rng.uniform(0.3, 0.9)
        phase = rng.uniform(0, 2 * np.pi)
        a = make_cosine_start(n, alpha, phase)
    else:
        # Exponential with noise
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-i / n) + 0.1 * rng.standard_normal(n)
        a = np.clip(a, 0, None)
    
    return a


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)
