    else:
        w = vals / total
    total_grad = np.zeros(n)
    for i, k in enumerate(k_list):
        total_grad += w[i] * exact_peak_grad(a, N, k)
    return total_grad, c, c.max()


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def spectral_partial_restart(a, N, rng, frac=0.3):
    """Randomize a fraction of FFT phases, preserve magnitudes."""
    n = len(a)
    S = a.sum()
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    A = fft(a, Nf)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(Nf * frac))
    rand_idx = rng.choice(Nf, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2.0 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def spectral_full_restart(a, N, rng):
    """Full spectral phase randomization."""
    n = len(a)
    S = a.sum()
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    A = fft(a, Nf)
    mag = np.abs(A)
    phase = rng.uniform(0, 2.0 * np.pi, Nf)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_init_tapered(n, S, rng, taper_power=1.0):
    """Raised-cosine or other tapered initialization."""
    t = np.linspace(0, 1, n)
    if taper_power == 1.0:
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5)))
    elif taper_power == 2.0:
        a = np.sin(np.pi * t) ** 2
    else:
        a = np.sin(np.pi * t) ** taper_power
    a += 0.02 * rng.standard_normal(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    return a


def make_init_smooth(n, S, rng):
    """Smooth multi-frequency initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for freq in [2, 5, 13, 37]:
        a += 0.1 * np.sin(2.0 * np.pi * freq * t + rng.random() * 2.0 * np.pi)
    a = np.clip(a, 0.01, None)
    s = a.sum()
    return a * (S / s)


def make_init_random(n, S, rng):
    a = rng.random(n) + 0.05
    return a * (S / a.sum())


def make_init_antisymmetric(n, S, rng):
    """Front half different from back half."""
    half = n // 2
    front = 1.0 + 0.3 * rng.random(half)
    back = 1.0 - 0.3 * rng.random(n - half)
    a = np.concatenate([front, back])
    a = np.clip(a, 0.05, None)
    return a * (S / a.sum())


def lbfgs_block(a_init, n, S, N, max_time, rng, beta_init=10.0, beta_max=300.0,
                 m_lbfgs=10, use_multiscale=False, top_k=100):
    """L-BFGS on LSE-smoothed objective."""
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    start_time = time.monotonic()
    it = 0

    if use_multiscale:
        betas = [beta, beta * 5, beta * 20]
        g, c, c_max = multi_scale_grad(a, N, betas)
    else:
        g, _, c, c_max = lse_grad_topk(a, N, beta, top_k)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    step_scale = 2.0 * S / (n * max(beta, 1.0))

    while time.monotonic() - start_time < max_time:
        if time.monotonic() - start_time + 0.5 > max_time:
            break
        it += 1

        if it % 100 == 0:
            beta = min(beta_max, beta * 1.08)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            if use_multiscale:
                betas = [beta, beta * 5, beta * 20]
                g, c, c_max = multi_scale_grad(a, N, betas)
            else:
                g, _, c, c_max = lse_grad_topk(a, N, beta, top_k)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

        if it % 30 == 0 and len(s_hist) >= m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-20:
            break

        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        for ls in range(10):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls, _ = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max * 1.001 or ls == 9:
                a_new = candidate
                cmax_new = cm_ls
                break
            alpha_ls *= 0.5

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        if use_multiscale:
            betas = [beta, beta * 5, beta * 20]
            g_new, c_nw, cmax_nw = multi_scale_grad(a_new, N, betas)
        else:
            g_new, _, c_nw, cmax_nw = lse_grad_topk(a_new, N, beta, top_k)

        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * max(ss, 1e-30) and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c_max = cmax_nw

        if it % 300 == 0 and rng.random() < 0.2:
            a = spectral_partial_restart(a, N, rng, frac=0.15)
            s_hist = []
            y_hist = []
            if use_multiscale:
                betas = [beta, beta * 5, beta * 20]
                g, c, c_max = multi_scale_grad(a, N, betas)
            else:
                g, _, c, c_max = lse_grad_topk(a, N, beta, top_k)

    return best_a, best_cmax


def peak_chase_block(a_init, n, S, N, max_time, rng, mode="topk"):
    """Exact peak-directed projected gradient descent."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    it = 0
    step_base = 0.1 * S / n

    while time.monotonic() - start_time < max_time:
        if time.monotonic() - start_time + 0.3 > max_time:
            break
        it += 1

        c, Nf = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()

        if mode == "top1":
            k_star = int(np.argmax(c))
            g = exact_peak_grad(a, N, k_star)
        elif mode == "topk":
            top_k = min(10, len(c))
            k_list = np.argsort(c)[::-1][:top_k]
            g, _, _ = topk_combined_grad(a, N, k_list)
        else:
            top_k = min(50, len(c))
            k_list = np.argsort(c)[::-1][:top_k]
            g, _, _ = topk_combined_grad(a, N, k_list)

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-18:
            break

        g_tangent = g - g.mean()
        g_t_norm = np.linalg.norm(g_tangent)
        if g_t_norm < 1e-18:
            break

        step = step_base * (1.0 / np.sqrt(1.0 + it * 0.01))
        decay = 1.0 - 0.3 * min(1.0, it / max(1, int(max_time * 30)))
        step *= decay

        a_new = project_simplex(a - step * g_tangent / g_t_norm, S)
        c_new, _ = compute_conv(a_new, N)
        cm_new = c_new.max()

        if cm_new > c_max * 1.01:
            step *= 0.5
            a_new = project_simplex(a - step * g_tangent / g_t_norm, S)
            c_new, _ = compute_conv(a_new, N)
            cm_new = c_new.max()
            if cm_new > c_max * 1.005:
                continue

        a = a_new

        if it % 500 == 0 and rng.random() < 0.15:
            a = spectral_partial_restart(a, N, rng, frac=0.1)

    return best_a, best_cmax


def annealing_block(a_init, n, S, N, max_time, rng):
    """Simulated annealing with spectral perturbations."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    c, _ = compute_conv(a, N)
    best_cmax = c.max()
    start_time = time.monotonic()
    it = 0
    T_initial = 0.1
    T_final = 0.001

    while time.monotonic() - start_time < max_time:
        if time.monotonic() - start_time + 0.5 > max_time:
            break
        it += 1
        progress = min(1.0, it / max(1, int(max_time * 10)))
        T = T_initial * (T_final / T_initial) ** progress

        frac = 0.1 + 0.2 * rng.random()
        a_prop = spectral_partial_restart(a, N, rng, frac)
        c_prop, _ = compute_conv(a_prop, N)
        cmax_prop = c_prop.max()

        delta = cmax_prop - best_cmax
        if delta < 0 or rng.random() < np.exp(-max(0, delta) / max(T, 1e-10)):
            a = a_prop
            if cmax_prop < best_cmax:
                best_cmax = cmax_prop
                best_a = a_prop.copy()

    return best_a, best_cmax


def deflation_init(n, S, N, n_iters=1000, rng=None):
    """Iteratively reduce peak from all-ones."""
    if rng is None:
        rng = np.random.default_rng(99)
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c, _ = compute_conv(a, N)
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                if i == j:
                    contrib[i] += a[j]
                else:
                    contrib[i] += 2.0 * a[j]
        if contrib.max() < 1e-15:
            break
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if i_max == i_min:
            break
        transfer = 0.01 * a[i_max]
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv)/sum^2.
    Multi-phase: LSE-LBFGS exploration, multi-scale exploitation,
    peak-chase polishing, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    n = 8192
    try:
        given_arr = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if 2048 <= given_arr.size <= 32768:
            n = len(given_arr)
            given_arr = np.clip(given_arr, 0, 1000)
    except Exception:
        given_arr = None

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()
    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    def _update_best(a_cand, label=""):
        nonlocal global_best, global_a
        a_c = safe_normalize(a_cand, n, S)
        c, _ = compute_conv(a_c, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_c.copy()
            score = 2.0 * n * sc / (S * S)
            if label:
                print("  [%s] peak=%.6f score=%.6f" % (label, sc, score))
        return sc

    # === Phase 1: Multi-start LSE-LBFGS exploration (30% time) ===
    print("=== Phase 1: Multi-start exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(720.0, (deadline - t1) * 0.30)

    inits = []
    if given_arr is not None and given_arr.size == n:
        inits.append(("given", safe_normalize(given_arr, n, S)))
    inits.append(("taper1", make_init_tapered(n, S, rng, 1.0)))
    inits.append(("taper2", make_init_tapered(n, S, rng, 2.0)))
    inits.append(("smooth", make_init_smooth(n, S, rng)))
    inits.append(("random", make_init_random(n, S, rng)))
    inits.append(("antisym", make_init_antisymmetric(n, S, rng)))
    inits.append(("deflation", deflation_init(n, S, N, 2000, rng)))

    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 400:
            break
        rem = max(2.0, (t1 + phase1_budget) - time.monotonic())
        alloc = min(per_init, rem)
        a_l, sc_l = lbfgs_block(arr, n, S, N, alloc, rng,
                                beta_init=5.0, beta_max=80.0,
                                m_lbfgs=8, use_multiscale=False, top_k=50)
        _update_best(a_l, name + "/P1")

    print("Phase 1 done. Best score: %.6f" % (2.0 * n * global_best / (S * S) if global_best < np.inf else 2.0))

    # === Phase 2: Focused multi-scale LSE-LBFGS (35% time) ===
    print("=== Phase 2: Multi-scale exploitation ===")
    t2 = time.monotonic()
    phase2_budget = min(840.0, (deadline - t2) * 0.35)

    a_curr = global_a.copy() if global_a is not None else np.ones(n) * S / n
    stall = 0
    use_ms = True

    while time.monotonic() < t2 + phase2_budget:
        block_time = min(120.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 10:
            break
        beta_start = 10.0 + 5.0 * min(stall, 8)
        a_new, sc_new = lbfgs_block(a_curr, n, S, N, block_time, rng,
                                     beta_init=beta_start, beta_max=500.0,
                                     m_lbfgs=12, use_multiscale=use_ms, top_k=200)
        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            score = 2.0 * n * sc_new / (S * S)
            print("  [P2] score=%.6f (beta_start=%.0f, ms=%s)" % (score, beta_start, use_ms))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall == 2:
                a_curr = spectral_partial_restart(a_curr, N, rng, frac=0.3)
                use_ms = not use_ms
                stall = 0
            elif stall >= 4:
                a_curr = spectral_full_restart(a_curr, N, rng)
                stall = 0

    # === Phase 3: Peak-chase polishing (20% time) ===
    print("=== Phase 3: Peak-chase polishing ===")
    t3 = time.monotonic()
    remaining3 = min(480.0, (deadline - t3) * 0.20)
    if remaining3 > 20:
        a_pc, sc_pc = peak_chase_block(global_a, n, S, N, remaining3, rng, mode="topk")
        _update_best(a_pc, "peakchase_topk")
        if deadline - time.monotonic() > 20:
            a_pc2, sc_pc2 = peak_chase_block(global_a, n, S, N,
                                            min(120.0, deadline - time.monotonic() - 10),
                                            rng, mode="top50")
            _update_best(a_pc2, "peakchase_top50")

    # === Phase 4: Simulated annealing (10% time) ===
    print("=== Phase 4: Annealing ===")
    t4 = time.monotonic()
    remaining4 = min(240.0, (deadline - t4) * 0.10)
    if remaining4 > 20:
        a_an, sc_an = annealing_block(global_a, n, S, N, remaining4, rng)
        _update_best(a_an, "annealing")

    # === Phase 5: Final high-beta L-BFGS ===
    print("=== Phase 5: Final L-BFGS ===")
    remaining5 = deadline - time.monotonic() - 15
    if remaining5 > 20:
        a_fin, sc_fin = lbfgs_block(global_a, n, S, N, min(remaining5, 300.0), rng,
                                     beta_init=50.0, beta_max=1000.0,
                                     m_lbfgs=15, use_multiscale=True, top_k=300)
        _update_best(a_fin, "final")

    # === Verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.8f" % ver)
        else:
            print("Verification FAILED, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    final_score = 2.0 * len(global_a) * global_best / (S * S) if global_best < np.inf else float('inf')
    print("Returning length %d, internal peak=%.6f, approx score=%.6f" % (len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
