"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.
Strategy: Multi-start optimization combining FISTA with adaptive top-k LSE,
exact max-subgradient polishing, spectral flattening, and diverse initializations.
Improves upon parent by using exact subgradient in final phase, wider spectral
perturbation, and adaptive beta scheduling with periodic restarts.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_value_and_gradient(a, c, k, beta):
    """Compute top-k LSE value and gradient."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    max_scaled = scaled.max()
    if max_scaled > 500:
        scale = 500.0 / max(1e-15, max_scaled)
        beta = beta * scale
        scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    val = cm + np.log(max(tot, 1e-300)) / beta
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return val, grad, idx, w


def exact_max_subgradient(a, c):
    """Compute subgradient of max(convolve(a,a)) using uniform weights over argmax."""
    n = len(a)
    max_val = c.max()
    tol = max_val * 1e-10 + 1e-15
    mask = c >= max_val - tol
    if not np.any(mask):
        mask = c >= max_val * (1 - 1e-12)
    if not np.any(mask):
        mask = np.zeros(len(c), dtype=bool)
        mask[np.argmax(c)] = True
    w = np.zeros(len(c))
    w[mask] = 1.0 / mask.sum()

    # Gradient = 2 * cross-correlation of w with a
    # g[i] = 2 * sum_m w[i+m] * a[m]
    # = 2 * (convolve(w, a_flipped))[n-1 : 2n-1]
    a_flip = a[::-1]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    Af = fft(a_flip, fsize)
    conv = np.real(ifft(W * Af))
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad


def spectral_flatten_step(a, alpha):
    """Spectral flattening to reduce autocorrelation peaks."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()

    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a

    ratio = power / max(mean_p, 1e-20)
    # Damp high-power, boost low-power frequencies (bounded)
    new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + alpha * ratio))
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / power)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain (phase perturbation)."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def spectral_mirror_step(a, rng, strength=0.3):
    """Apply a random spectral mask to encourage diversity."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    # Random gain in [1-strength, 1+strength]
    gain = 1.0 + strength * (rng.random(fsize) * 2 - 1)
    gain[0] = 1.0
    A_new = A * gain
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() < 1e-10:
        return a
    return normalize(a_new, a.sum())


def gaussian_window(n, sigma_frac=0.4):
    """Create a Gaussian window of length n."""
    t = np.arange(n) - n / 2.0
    sigma = sigma_frac * n / 2.0
    w = np.exp(-t**2 / (2 * sigma**2))
    return w / w.sum()


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def run_fista(a_init, S, max_time, rng,
              beta_init=20.0, beta_rate=0.004, beta_max=8000.0,
              k_frac=0.08, use_momentum=True, use_exact_grad=False):
    """FISTA with top-k LSE or exact max subgradient."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0

    k = max(4, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        if it % 500 == 1:
            elapsed = time.monotonic() - start
            frac = min(1.0, elapsed / max_time)
            # Adaptive beta: start moderate, increase over time
            beta = min(beta_max, beta_init * np.exp(beta_rate * it))

            if use_exact_grad and it > max_time * 0.5 / max(0.01, (time.monotonic() - start) / max_time if max_time > 0 else 1):
                pass

        beta = min(beta_max, beta_init * np.exp(beta_rate * it * 0.5))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        # Choose gradient type
        if use_exact_grad and stuck > 200:
            grad = exact_max_subgradient(y, c)
        else:
            _, grad, _, _ = topk_lse_value_and_gradient(y, c, k, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / max(2.0 * beta * 0.1, 1e-8)
        if use_exact_grad and stuck > 200:
            step = min(step, 0.5 / max(gnorm, 1e-8))
        x_new = project_simplex(y - step * grad, S)
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        # Periodic restart when stuck
        if stuck > 800:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

        # Periodic spectral perturbation
        if it % 1500 == 0 and rng.random() < 0.25:
            y = spectral_perturb(y, 0.1, rng)
            t_val = 1.0
            stuck = 0

        # Sparse spectral mirror
        if it % 3000 == 0 and rng.random() < 0.1:
            y = spectral_mirror_step(y, rng, strength=0.15)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_exact_polish(a_init, S, max_time, rng):
    """Polish using exact max subgradient with careful step control."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    it = 0
    step_scale = 1.0
    k_lse = max(4, int(0.02 * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Mix: use exact subgradient with LSE smoothing for direction
        beta = min(20000.0, 100.0 * (1.0 + it / 200.0))
        _, grad_lse, idx, w = topk_lse_value_and_gradient(a, c, k_lse, beta)
        grad_exact = exact_max_subgradient(a, c)

        # Blend the two gradients
        alpha_blend = min(1.0, it / 500.0)  # gradually shift to exact
        grad = (1 - alpha_blend) * grad_lse + alpha_blend * grad_exact

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 0.5 * step_scale / max(1.0, gnorm / (S * 0.01))
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
            step_scale = min(3.0, step_scale * 1.05)
        else:
            step_scale = max(0.005, step_scale * 0.4)
            a = project_simplex(a - step * 0.5 * grad, S)

        # Small perturbation to escape plateaus
        if it % 500 == 0 and (time.monotonic() - start) / max_time < 0.8:
            noise_scale = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def run_spectral_iteration(a_init, S, max_time, rng, flatness=0.1):
    """Iterative spectral flattening with time-domain projection."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        # Spectral flattening
        a = spectral_flatten_step(a, flatness)

        # Small gradient step to refine
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        k = max(4, int(0.05 * (2 * n - 1)))
        beta = 50.0
        _, grad, _, _ = topk_lse_value_and_gradient(a, c, k, beta)
        step = 0.1 / max(np.linalg.norm(grad), 1e-10)
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max:
            a = a_new

        # Occasional perturbation
        if rng.random() < 0.02:
            a = spectral_perturb(a, 0.05, rng)

    return best_a, best


def run_multi_start(n, given_seq, time_budget, rng):
    """Run multi-start optimization for a given sequence length."""
    S = np.sqrt(2.0 * n)
    global_best = np.inf
    global_a = None

    def update_best(a, desc):
        nonlocal global_best, global_a
        obj = objective_value(a)
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("    [%s] obj=%.8f" % (desc, obj))
        return obj

    print("  Starting multi-start for n=%d, budget=%.0fs" % (n, time_budget))
    t_start = time.monotonic()
    deadline = t_start + time_budget - 5.0

    # Generate diverse starting points
    starts = []
    names = []

    # 1. Given sequence (if valid)
    if given_seq is not None and len(given_seq) >= 64:
        g = np.clip(np.asarray(given_seq, dtype=np.float64), 0, 1000)
        if len(g) != n:
            g = resample_sequence(g, n)
        g = normalize(g, S)
        starts.append(g)
        names.append("given")

    # 2. Uniform
    starts.append(np.ones(n) * S / n)
    names.append("uniform")

    # 3. Gaussian window
    starts.append(normalize(gaussian_window(n, 0.35) * S, S))
    names.append("gaussian0.35")

    # 4. Gaussian window wider
    starts.append(normalize(gaussian_window(n, 0.5) * S, S))
    names.append("gaussian0.5")

    # 5. Random
    r = rng.random(n)
    starts.append(normalize(r, S))
    names.append("random")

    # 6. Triangular
    t = np.linspace(-1, 1, n)
    tri = np.clip(1 - np.abs(t), 0, 1)
    starts.append(normalize(tri, S))
    names.append("triangular")

    # Evaluate all starts
    for i, (s, name) in enumerate(zip(starts, names)):
        obj = update_best(s, "start:" + name)

    # Allocate time to top starts
    # Sort by objective
    start_objs = []
    for s in starts:
        c = fft_convolve_self(s)
        start_objs.append(c.max())
    order = np.argsort(start_objs)

    # Run FISTA on top 3 starting points
    n_starts_to_optimize = 3
    time_per_start = (deadline - time.monotonic()) / n_starts_to_optimize

    for rank in range(min(n_starts_to_optimize, len(starts))):
        si = order[rank]
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        t_alloc = min(time_per_start, remaining * 0.7)
        print("  Optimizing start '%s' (rank %d) for %.0fs..." % (names[si], rank, t_alloc))

        a_opt, val_opt = run_fista(starts[si], S, t_alloc, rng,
                                   beta_init=15.0, beta_rate=0.005,
                                   beta_max=5000.0, k_frac=0.08,
                                   use_momentum=True, use_exact_grad=True)
        obj = update_best(a_opt, "fista:" + names[si])

        # Polish with exact max
        remaining = deadline - time.monotonic()
        if remaining > 20 and obj < global_best * 1.01:
            t_polish = min(remaining * 0.3, 120)
            a_pol, val_pol = run_exact_polish(a_opt, S, t_polish, rng)
            update_best(a_pol, "polish:" + names[si])

    # Final: spectral iteration on best found
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        a_sp, val_sp = run_spectral_iteration(global_a, S, min(remaining * 0.5, 100), rng,
                                              flatness=0.05)
        update_best(a_sp, "spectral")

        # Final exact polish
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            a_fin, val_fin = run_exact_polish(global_a, S, remaining, rng)
            update_best(a_fin, "final_polish")

    print("  Best for n=%d: obj=%.8f" % (n, global_best))
    return global_a, global_best

            a4, val4 = run_exact_polish(global_a, S_best, remaining, rng)
            obj4 = objective_value(a4)
            if obj4 < global_best:
                global_best = obj4
                global_a = a4.copy()
                print("  Exact polish: obj=%.8f" % global_best)

        # One more spectral pass
        remaining = budget - (time.monotonic() - start_total) - 5.0
        if remaining > 10:
            a5, val5 = run_spectral_iteration(global_a, S_best, remaining, rng,
                                              flatness=0.03)
            obj5 = objective_value(a5)
            if obj5 < global_best:
                global_best = obj5
                global_a = a5.copy()
                print("  Spectral: obj=%.8f" % global_best)

    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        print("No valid sequence found! Using fallback.")
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    # Final evaluation
    final_obj = evaluate_sequence([float(x) for x in global_a])
    print("\nFinal answer: length=%d, objective=%.8f" % (len(global_a), final_obj))
    print("Total time: %.1fs" % (time.monotonic() - start_total))

    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
