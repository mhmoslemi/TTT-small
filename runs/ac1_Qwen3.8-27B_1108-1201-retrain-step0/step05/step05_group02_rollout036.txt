"""
Minimize 2*n*max_conv/sum^2 using LP-guided direction finding combined with 
FISTA and coordinate descent for refinement.

Key innovation: Use a reduced-constraint LP to find descent directions that 
simultaneously address the top-K autocorrelation peaks, providing a more 
global guidance than pure subgradient methods.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_subgradient(a, beta):
    """Compute LSE of autocorrelation and its subgradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    s = a.sum()
    obj = 2.0 * n * (cm + np.log(tot) / beta) / (s * s)
    return obj, grad, c


def solve_reduced_lp(a, M, K=50):
    """
    Solve reduced LP: max sum(b) s.t. b >= 0, (a*b)[k_i] <= M for top K peaks.
    Returns the optimal b, or None if infeasible.
    """
    n = len(a)
    c = fft_convolve_self(a)
    # Get top K peak indices
    K = min(K, 2 * n - 1)
    top_k = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix: K x n
    # Row i: (a*b)[top_k[i]] = sum_j a[j]*b[top_k[i]-j] (valid j)
    A_mat = np.zeros((K, n), dtype=np.float64)
    for i, k in enumerate(top_k):
        k = int(k)
        # (a*b)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j] * b[k-j]
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        # A_mat[i, j] = a[j] contributes to b[k-j]
        # So A_mat[i, k-j] += a[j] for j in [j_lo, j_hi]
        # Equivalently: for j in [j_lo, j_hi], A_mat[i, k-j] = a[j]
        # Let's index differently: b[m] where m = k-j, so j = k-m
        # A_mat[i, m] = a[k-m] if 0 <= k-m < n
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        A_mat[i, m_lo:m_hi+1] = a[k - m_hi:k - m_lo + 1][::-1]
    
    b_var = cp.Variable(n)
    constraints = [A_mat @ b_var <= M * np.ones(K), b_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return None
    
    if b_var.value is None:
        return None
    return np.maximum(b_var.value, 0.0)


def lp_guided_optimize(a_init, max_time, rng, K=50, t_init=0.3):
    """
    LP-guided optimization: iteratively solve reduced LP to find directions,
    then update the sequence.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    t = t_init
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        M = cur_max
        # Solve reduced LP
        b_star = solve_reduced_lp(a, M, K=K)
        if b_star is None:
            # Fallback: use subgradient
            _, grad, _ = lse_subgradient(a, 100.0)
            step = S / max(np.linalg.norm(grad), 1e-10) * 0.1
            a = project_simplex(a - step * grad, S)
            c = fft_convolve_self(a)
            continue
        
        b_sum = b_star.sum()
        if b_sum < S * 1.001:
            # LP doesn't improve much; reduce step and try
            t = max(0.05, t * 0.9)
            b_rescaled = b_star * (S / max(b_sum, 1e-15))
        else:
            b_rescaled = b_star * (S / b_sum)
            # The ratio S/sum(b*) tells us how much room we have
            ratio = b_sum / S
            t = min(0.5, max(0.1, 0.3 * (ratio - 1.0) + 0.2))
        
        # Update
        a_new = (1 - t) * a + t * b_rescaled
        a_new = project_simplex(a_new, S)
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S * S)
        
        if obj_new < cur_obj * (1 + 1e-10):
            a = a_new
            c = c_new
            t = min(0.5, t * 1.05)  # Increase step slightly
        else:
            # Try smaller step
            for t_try in [t * 0.5, t * 0.25, t * 0.1]:
                a_try = (1 - t_try) * a + t_try * b_rescaled
                a_try = project_simplex(a_try, S)
                c_try = fft_convolve_self(a_try)
                obj_try = 2.0 * n * c_try.max() / (S * S)
                if obj_try < cur_obj * (1 + 1e-10):
                    a = a_try
                    c = c_try
                    t = t_try
                    break
            else:
                t = max(0.05, t * 0.5)
        
        # Periodic noise injection
        if it - last_improve > 50 and it % 20 == 0:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            c = fft_convolve_self(a)
    
    return best_a, best_obj


def run_fista(a_init, max_time, rng, beta_init=50.0, beta_rate=0.008,
              beta_max=50000.0, step_scale=1.0):
    """FISTA with adaptive LSE."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        obj_lse, grad, c = lse_subgradient(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = step_scale / max(beta, 1.0)
        # Backtracking
        x_trial = project_simplex(y - step * grad, S)
        obj_trial = compute_obj(x_trial)
        bt = 0
        while obj_trial > cur_obj + 1e-12 and step > 1e-14 and bt < 5:
            step *= 0.5
            x_trial = project_simplex(y - step * grad, S)
            obj_trial = compute_obj(x_trial)
            bt += 1
        if bt > 2:
            step = step * 0.8
        else:
            step = min(step * 1.05, 2.0 / max(beta, 1.0))
        
        x_new = x_trial
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        if it - last_improve > 3000:
            t_mom = 1.0
            noise_scale = max(1e-6, 0.002 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
    
    return best_a, best_obj

Actually, let me reconsider the LP formulation. The constraint `(a*b)[k] <= M` means the cross-correlation between `a` and `b` at lag k is bounded by M. This is different from the self-correlation `a*a`. The LP is finding a `b` that "fits" under the peak of `a`'s autocorrelation spectrum.

The intuition: if `a*a` has a peak at lag k, then `(a*b)[k]` being small means `b` is "anti-aligned" with `a` at that lag. By maximizing `sum(b)`, we find the `b` that has the most "energy" while being anti-aligned with the peaks of `a*a`.

When we update `a_new = (1-t)*a + t*b`, the new autocorrelation is:
`(a_new * a_new)[k] = (1-t)^2 (a*a)[k] + t(1-t) * 2(a*b)[k] + t^2 (b*b)[k]`

The key term is `t(1-t) * 2(a*b)[k] <= t(1-t) * 2M` (from the LP constraint). And `(a*a)[k] <= M`. So:
`(a_new * a_new)[k] <= (1-t)^2 M + 2t(1-t)M + t^2 (b*b)[k]`
`= M(1-t)^2 + 2M*t(1-t) + t^2 (b*b)[k]`
`= M(1 - 2t + t^2 + 2t - 2t^2) + t^2 (b*b)[k]`
`= M(1 - t^2) + t^2 (b*b)[k]`
`= M - t^2 M + t^2 (b*b)[k]`
`= M + t^2 ((b*b)[k] - M)`

So the new peak is bounded by `M + t^2 * max(b*b) - t^2 * M = M + t^2 (max(b*b) - M)`.

For this to be less than M, we need `max(b*b) < M`. But the LP doesn't guarantee this! The LP bounds `(a*b)[k]` but not `(b*b)[k]`.

So the improvement is:
- The cross term `2t(1-t)M` is at most the same as the original peak `M`.
- The quadratic term `t^2 (b*b)[k]` could be larger or smaller than `t^2 M`.

In the best case (if `b*b` is also flat with max <= M), the new peak is `M(1-t^2) + t^2 M = M`. No improvement!

In practice, `b*b` will have a lower peak than `M` because `b` is "designed" to be anti-correlated with `a` at the peak lags. So `max(b*b) < M` typically, and we get improvement.

The amount of improvement depends on how much `max(b*b) < M`. If `b` is very different from `a` (e.g., `b` is more "flat"), then `max(b*b)` could be much less than `M`, giving significant improvement.

"""
Minimize 2*n*max_conv/sum^2 using LP-guided direction finding (top-K peak constraints),
combined with adaptive FISTA (LSE-smoothed) and peak-targeted coordinate descent.

Key innovation: A reduced-constraint LP identifies descent directions that
simultaneously address multiple autocorrelation peaks, providing more global
guidance than pure subgradient methods. Combined with adaptive step sizing
and multi-phase refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_subgradient(a, beta):
    """Compute LSE of autocorrelation and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A_fft = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A_fft)))[:n]
    s = a.sum()
    obj = 2.0 * n * (cm + np.log(tot) / beta) / (s * s)
    return obj, grad, c


def solve_reduced_lp(a, M, K=50):
    """
    Solve reduced LP: max sum(b) s.t. b >= 0, (a*b)[k_i] <= M for top K peaks.
    Returns optimal b or None.
    """
    n = len(a)
    c = fft_convolve_self(a)
    K = min(K, 2 * n - 1)
    top_k = np.argsort(c)[::-1][:K]

    # Build constraint matrix: K x n
    # (a*b)[k] = sum_j a[j]*b[k-j] for valid j
    # A_mat[i, m] = a[k_i - m] if 0 <= k_i - m < n
    A_mat = np.zeros((K, n), dtype=np.float64)
    for i in range(K):
        k = int(top_k[i])
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_hi >= m_lo:
            A_mat[i, m_lo:m_hi + 1] = a[k - m_hi:k - m_lo + 1][::-1]

    b_var = cp.Variable(n)
    try:
        constraints = [A_mat @ b_var <= M * np.ones(K), b_var >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        prob.solve(solver=cp.GLOP)
        if b_var.value is None:
            prob.solve(solver=cp.CBC)
        if b_var.value is None:
            prob.solve(solver=cp.ECOS)
    except Exception:
        try:
            constraints = [A_mat @ b_var <= M * np.ones(K), b_var >= 0]
            prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
            prob.solve(solver=cp.CBC)
        except Exception:
            return None

    if b_var.value is None:
        return None
    return np.maximum(b_var.value, 0.0)


def lp_guided_optimize(a_init, max_time, rng, K=50):
    """
    LP-guided optimization: iteratively solve reduced LP for descent directions.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0
    t = 0.3
    consecutive_fails = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
            consecutive_fails = 0
        else:
            consecutive_fails += 1

        # Adaptive K: increase when stuck
        K_eff = K
        if consecutive_fails > 30:
            K_eff = min(2 * n - 1, K + consecutive_fails // 5)

        # Solve reduced LP
        M = cur_max
        b_star = solve_reduced_lp(a, M, K=K_eff)

        if b_star is None:
            # Fallback: FISTA step
            _, grad, _ = lse_subgradient(a, 200.0)
            step = 0.1 * S / max(np.linalg.norm(grad), 1e-10)
            a = project_simplex(a - step * grad, S)
            c = fft_convolve_self(a)
            continue

        b_sum = b_star.sum()
        if b_sum < S * 1.0001:
            # LP barely improves; use subgradient
            _, grad, _ = lse_subgradient(a, 100.0)
            step = 0.05 * S / max(np.linalg.norm(grad), 1e-10)
            a = project_simplex(a - step * grad, S)
            c = fft_convolve_self(a)
            continue

        b_rescaled = b_star * (S / b_sum)

        # Adaptive step size based on LP ratio
        ratio = b_sum / S
        t = min(0.6, max(0.1, 0.2 + 0.3 * min(ratio - 1.0, 1.0)))

        # Try the update with line search
        improved = False
        for t_try in [t, t * 0.6, t * 0.3, t * 0.1]:
            a_new = (1 - t_try) * a + t_try * b_rescaled
            a_new = project_simplex(a_new, S)
            c_new = fft_convolve_self(a_new)
            obj_new = 2.0 * n * c_new.max() / (S * S)
            if obj_new <= cur_obj * (1 + 1e-9):
                a = a_new
                c = c_new
                t = t_try
                improved = True
                break

        if not improved:
            t = max(0.05, t * 0.7)
            # Add small noise to escape
            if consecutive_fails > 20:
                noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
                c = fft_convolve_self(a)

        if it % 20 == 0:
            elapsed = time.monotonic() - start
            print("  LP it %d: obj=%.6f, t=%.3f, K=%d, ratio=%.3f (%.0fs)" %
                  (it, cur_obj, t, K_eff, ratio, elapsed))

    return best_a, best_obj


def run_fista(a_init, max_time, rng, beta_init=50.0, beta_rate=0.008,
              beta_max=50000.0, step_scale=1.0):
    """FISTA with adaptive LSE and backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        _, grad, c = lse_subgradient(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step_local = step
        x_trial = project_simplex(y - step_local * grad, S)
        obj_trial = compute_obj(x_trial)
        bt = 0
        while obj_trial > cur_obj + 1e-12 and step_local > 1e-16 and bt < 6:
            step_local *= 0.5
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt += 1

        if bt > 3:
            step = step_local * 0.7
        else:
            step = min(step * 1.04, 3.0 / max(beta, 1.0))

        x_new = x_trial
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        if it - last_improve > 5000:
            t_mom = 1.0
            noise_scale = max(1e-7, 0.003 * S / np.sqrt(n) * np.exp((it - last_improve) / -3000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    return best_a, best_obj


def coordinate_descent_peak(a_init, max_time, rng):
    """Peak-targeted coordinate descent with exact max computation."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        if it - last_improve > 30000:
            break

        # Multi-peak contribution
        top_k = min(4, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        contrib = np.zeros(n)
        for kk in top_idx:
            kk = int(kk)
            j_lo = max(0, kk - n + 1)
            j_hi = min(n - 1, kk)
            if j_hi >= j_lo:
                contrib[j_lo:j_hi + 1] += a[kk - j_hi:kk - j_lo + 1][::-1]

        avg_val = S / n
        src_score = contrib * a
        tgt_score = -contrib * (avg_val - a + 1e-12)

        n_cand = min(15, n)
        src_cand = np.argsort(src_score)[::-1][:n_cand]
        tgt_cand = np.argsort(tgt_score)[:n_cand]

        best_obj_trial = cur_obj
        best_move = None

        for j in src_cand[:8]:
            if a[j] < 1e-12:
                continue
            for l in tgt_cand[:8]:
                if j == l:
                    continue
                for frac in [0.08, 0.04, 0.015]:
                    delta = a[j] * frac
                    if delta < 1e-13:
                        continue
                    # Compute delta_c
                    delta_c = np.zeros(2 * n - 1)
                    js, je = j, min(2 * n - 2, j + n - 1)
                    delta_c[js:je + 1] -= delta * a[max(0, js - j):min(n, je - j + 1)]
                    ls, le = l, min(2 * n - 2, l + n - 1)
                    delta_c[ls:le + 1] += delta * a[max(0, ls - l):min(n, le - l + 1)]

                    new_max = np.max(c + delta_c)
                    new_obj = 2.0 * n * new_max / (S * S)
                    if new_obj < best_obj_trial - 1e-14:
                        best_obj_trial = new_obj
                        best_move = (j, l, delta, delta_c)

        if best_move is not None:
            j, l, delta, delta_c = best_move
            a[j] -= delta
            a[l] += delta
            c = c + delta_c
        elif it - last_improve > 500:
            j = rng.integers(0, n)
            l = rng.integers(0, n)
            if j != l and a[j] > 1e-10:
                delta = a[j] * 0.01
                a[j] -= delta
                a[l] += delta
                c = fft_convolve_self(a)

        if it % 100 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample using linear interpolation, then renormalize."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-3.0 * (tt - 0.5) ** 2 / (0.12 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 5 * tt + 1.1)
    elif method == "multi_tone":
        a = (0.25 + 0.2 * np.sin(2 * np.pi * 3 * tt) +
             0.2 * np.sin(2 * np.pi * 7 * tt + 1.3) +
             0.15 * np.sin(2 * np.pi * 13 * tt + 2.1))
        a = np.abs(a) + 0.05
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "bump":
        a = np.maximum(0.02, 1.0 - 1.5 * np.abs(tt - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())
