**Another idea: Use the "infimal convolution" or "dual" perspective.**

The problem `minimize max_k c[k] s.t. sum(a) = S, a >= 0` can be rewritten as:
`minimize M s.t. c[k] <= M for all k, sum(a) = S, a >= 0`

The Lagrangian is:
`L = M + sum_k lambda_k * (c[k] - M) + mu * (sum(a) - S) - sum_j nu_j * a[j]`

where `lambda_k >= 0`, `nu_j >= 0`.

At optimality:
- `1 - sum_k lambda_k = 0` (derivative w.r.t. M) => `sum_k lambda_k = 1`
- `sum_k lambda_k * dc[k]/da[i] + mu - nu[i] = 0` for each `i`
- Complementary slackness: `lambda_k * (c[k] - M) = 0`, `nu[i] * a[i] = 0`

So at the optimum, the weights `lambda_k` are concentrated on the active peaks (where `c[k] = M`), and the gradient condition says:
`sum_k lambda_k * 2*a[k-i] + mu = nu[i]`

For `i` where `a[i] > 0` (interior), `nu[i] = 0`, so:
`sum_k lambda_k * 2*a[k-i] = -mu` (constant for all interior `i`)

This means: at the optimum, the "weighted sum of shifted versions of a" is constant at all interior points. This is a nice characterization!

We could use this to design an iterative algorithm:
1. Start with some `a` and some weights `lambda` on the peaks.
2. Update `a` to try to make `sum_k lambda_k * 2*a[k-i]` as constant as possible (at interior points).
3. Update `lambda` to concentrate on the current peaks.
4. Repeat.

This is essentially a **mirror descent / multiplicative weights** approach!

More concretely:
1. Compute `c = convolve(a, a)`.
2. Set `lambda[k] = exp(gamma * c[k]) / sum(exp(gamma * c))` (softmax over the convolution values). As `gamma -> infinity`, this concentrates on the maximum.
3. Compute `g[i] = sum_k lambda[k] * 2 * a[k-i]` (the "weighted autocorrelation").
4. Update `a[i] <- a[i] * exp(-eta * (g[i] - mean(g)))` (multiplicative update to make `g` more uniform).
5. Normalize `a` to have sum `S`.
6. Repeat.

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. Softmax-weighted multiplicative iteration (mirror descent with entropy) - novel approach
2. FISTA with top-k LSE smoothing for fine-tuning
3. Active-set subgradient method for focused peak reduction
4. Spectral perturbation for exploration
5. Multi-length search (n=8192, 16384, 32768)
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_corr(w, a):
    """
    Compute cross-correlation: result[i] = sum_k w[k] * a[k-i] for i=0..len(a)-1
    where a[j]=0 for j outside [0, len(a)-1].
    w has length M, a has length n. Need FFT size >= M+n-1.
    """
    M = len(w)
    n = len(a)
    fsize = 1 << (M + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return corr[:n]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max() + 1e-15)
        beta_eff = beta * scale
        scaled = beta_eff * (c_top - cm)
    else:
        beta_eff = beta
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    M = len(w_full)
    fsize = 1 << (M + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def run_softmax_iter(a_init, S, max_time, rng,
                     gamma_init=10.0, gamma_end=500.0,
                     eta_init=1.0, eta_decay=0.9997,
                     k_init=50, k_end=5):
    """
    Softmax-weighted multiplicative iteration (mirror descent with entropy).
    
    This uses the dual variable insight: at optimum, weighted sum of shifted
    a should be constant at interior points. We iteratively:
    1. Compute softmax weights on convolution peaks
    2. Compute the "pressure" gradient
    3. Apply multiplicative update to equalize pressure
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    eta = eta_init
    last_improve_time = start
    fsize_conv = 1 << (2 * n - 1).bit_length()
    fsize_corr = 1 << (3 * n - 2).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Anneal parameters
        gamma = gamma_init + (gamma_end - gamma_init) * (progress ** 0.4)
        k_active = max(3, int(k_init - (k_init - k_end) * progress))

        c = np.real(ifft(fft(a, fsize_conv) * fft(a, fsize_conv)))[:2*n-1]
        cur_max = c.max()
        if cur_max < best - 1e-10:
            best = cur_max
            best_a = a.copy()
            last_improve_time = time.monotonic()

        # Compute weights: use top-k with softmax for stability
        idx_top = np.argsort(c)[::-1][:k_active]
        c_top = c[idx_top]
        c_min = c_top.min()
        w_top = np.exp(gamma * (c_top - c_min))
        w_top /= w_top.sum()

        # Build full weight vector
        w_full = np.zeros(2 * n - 1)
        w_full[idx_top] = w_top

        # Compute gradient: g[i] = 2 * sum_k w_full[k] * a[k-i]
        g = 2.0 * fft_cross_corr(w_full, a)

        # Multiplicative update with centering
        g_mean = g.mean()
        g_centered = g - g_mean
        g_std = np.std(g)
        if g_std < 1e-15:
            break

        # Scale step size appropriately
        scale = S / (n * g_std + 1e-15)
        effective_eta = eta * min(scale, 10.0)

        exponent = np.clip(-effective_eta * g_centered / (g_std + 1e-15), -3.0, 3.0)
        a = a * np.exp(exponent)
        a = np.maximum(a, 1e-30)
        a = normalize(a, S)

        # Decay step size
        eta *= eta_decay

        # Periodic restart when stuck (no improvement for 30s)
        if time.monotonic() - last_improve_time > 30 and rng.random() < 0.3:
            noise_scale = 0.008 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            last_improve_time = time.monotonic()
            eta = max(eta, 0.3)

        # Spectral perturbation for exploration (early)
        if progress < 0.3 and it % 2000 == 0 and rng.random() < 0.2:
            a = spectral_perturb(a, 0.05, rng)
            last_improve_time = time.monotonic()

    return best_a, best


def run_fista_topk(a_init, S, max_time, rng,
                   beta_init=50.0, beta_rate=0.005, beta_max=10000.0,
                   k_frac=0.02, use_momentum=True):
    """FISTA with top-k LSE, backtracking, and adaptive momentum."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0
    fsize = 1 << (2 * n - 1).bit_length()

    k = max(5, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = np.real(ifft(fft(y, fsize) * fft(y, fsize)))[:2*n-1]
        cur_max = c.max()
        if cur_max < best - 1e-10:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        step = min(step, S / (gnorm + 1e-15) * 0.5)
        x_new = project_simplex(y - step * grad, S)

        improved = False
        for _ in range(6):
            c_new = np.real(ifft(fft(x_new, fsize) * fft(x_new, fsize)))[:2*n-1]
            if c_new.max() < cur_max + 1e-10:
                improved = True
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if use_momentum and improved:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        else:
            y_new = x_new.copy()
            t_val = 1.0
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        if improved and use_momentum:
            t_val = t_new

        # Periodic restart when stuck
        if stuck > 500:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

    return best_a, best


def run_active_subgrad(a_init, S, max_time, rng,
                       K=20, step_init=0.1, step_decay=0.999):
    """
    Active-set subgradient method.
    Uses equal weights on top K peaks as a subgradient of the max function.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    step = step_init
    fsize = 1 << (2 * n - 1).bit_length()
    fsize_corr = 1 << (3 * n - 2).bit_length()

    # SMAVE (smoothed average of values) for adaptive step
    avg_obj = best
    time_avg = 0.0

    while time.monotonic() - start < max_time:
        it += 1
        c = np.real(ifft(fft(a, fsize) * fft(a, fsize)))[:2*n-1]
        cur_max = c.max()

        if cur_max < best - 1e-10:
            best = cur_max
            best_a = a.copy()

        # Update time-averaged objective for adaptive step
        time_avg = 0.99 * time_avg + 0.01 * cur_max
        obj_diff = avg_obj - cur_max
        if obj_diff > 1e-12:
            step = min(step * 1.01, 2.0 * obj_diff / (cur_max + 1e-15))
        else:
            step *= 0.95
        step = max(step, step_init * 0.01)

        # Get top K peaks
        K_eff = min(K, 2 * n - 1)
        idx_top = np.argsort(c)[::-1][:K_eff]
        # Equal weights on top K (subgradient of max)
        w = np.zeros(2 * n - 1)
        w[idx_top] = 1.0 / K_eff

        # Compute subgradient: g[i] = 2 * sum_k w[k] * a[k-i]
        g = 2.0 * fft_cross_corr(w, a)

        # Step with projection
        a_new = project_simplex(a - step * g / (np.linalg.norm(g) + 1e-15), S)

        # Check if improvement; if not, reduce step
        c_new = np.real(ifft(fft(a_new, fsize) * fft(a_new, fsize)))[:2*n-1]
        if c_new.max() < cur_max + 1e-10:
            a = a_new
        else:
            step *= 0.5
            a = a_new  # Still update, just accept the smaller step

        # Periodic perturbation
        if it % 1000 == 0 and rng.random() < 0.15:
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def run_polish_coordinate(a_init, S, max_time, rng):
    """
    Coordinate-descent-like polishing.
    For each element, try to reduce the max convolution value.
    Uses a greedy approach: identify the current peak, then try to reduce it
    by adjusting elements that contribute most to it.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    fsize = 1 << (2 * n - 1).bit_length()
    fsize_corr = 1 << (3 * n - 2).bit_length()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = np.real(ifft(fft(a, fsize) * fft(a, fsize)))[:2*n-1]
        cur_max = c.max()
        if cur_max < best - 1e-10:
            best = cur_max
            best_a = a.copy()

        # Find the peak position
        peak_idx = np.argmax(c)

        # The peak is c[peak_idx] = sum_j a[j] * a[peak_idx - j]
        # Gradient of c[peak_idx] w.r.t. a[i] is 2 * a[peak_idx - i] (if valid)
        # To reduce c[peak_idx], decrease a[i] where a[peak_idx-i] is large.
        
        # Compute the gradient at the peak
        # grad_peak[i] = 2 * a[peak_idx - i] for valid i
        valid = (peak_idx >= 0) & (peak_idx < 2 * n - 1)
        grad_peak = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                grad_peak[i] = 2.0 * a[j]
        
        # Also consider the second peak
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) > 1 and c_sorted[1] > 0.95 * c_sorted[0]:
            # Two similar peaks - use average gradient
            peak_idx2 = np.argsort(c)[::-1][1]
            grad_peak2 = np.zeros(n)
            for i in range(n):
                j = peak_idx2 - i
                if 0 <= j < n:
                    grad_peak2[i] = 2.0 * a[j]
            grad_peak = 0.5 * (grad_peak + grad_peak2)

        gnorm = np.linalg.norm(grad_peak)
        if gnorm < 1e-14:
            break

        # Take a small step against the peak gradient
        # But also add a small component to reduce other large values
        top_10 = np.argsort(c)[::-1][:10]
        w_avg = np.zeros(2 * n - 1)
        w_avg[top_10] = 1.0 / 10.0
        grad_avg = 2.0 * fft_cross_corr(w_avg, a)
        
        # Combined gradient
        combined = 0.5 * grad_peak + 0.5 * grad_avg
        combined_norm = np.linalg.norm(combined)
        if combined_norm < 1e-14:
            break

        # Adaptive step size
        step = 0.1 * S / (combined_norm * np.sqrt(n) + 1e-15)
        step = min(step, 0.5)

        a_new = project_simplex(a - step * combined, S)
        
        # Verify and accept
        c_new = np.real(ifft(fft(a_new, fsize) * fft(a_new, fsize)))[:2*n-1]
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller step
            a_new = project_simplex(a - step * 0.1 * combined, S)
            c_new2 = np.real(ifft(fft(a_new, fsize) * fft(a_new, fsize)))[:2*n-1]
            if c_new2.max() < cur_max:
                a = a_new

        # Periodic perturbation to escape local minima
        if it % 500 == 0 and rng.random() < 0.2:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def run_adam_lse(a_init, S, max_time, rng,
                 lr_init=0.01, beta_adam=50.0, beta_rate=0.005,
                 k_frac=0.03):
    """Adam optimizer with LSE gradient and adaptive learning rate."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    fsize = 1 << (2 * n - 1).bit_length()
    fsize_corr = 1 << (3 * n - 2).bit_length()

    beta1 = 0.85
    beta2 = 0.999
    eps_adam = 1e-8

    k = max(5, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        t += 1
        c = np.real(ifft(fft(a, fsize) * fft(a, fsize)))[:2*n-1]
        cur_max = c.max()
        if cur_max < best - 1e-10:
            best = cur_max
            best_a = a.copy()

        beta = min(5000.0, beta_adam * np.exp(beta_rate * t))
        grad = topk_lse_gradient(a, c, k, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        progress = min(1.0, (time.monotonic() - start) / max_time)
        lr = lr_init * (1.0 - 0.5 * progress) ** 2

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = project_simplex(a_new, S)

        # Periodic restart
        if t % 3000 == 0 and rng.random() < 0.1:
            noise_scale = 0.002 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                m = np.zeros(n)
                v = np.zeros(n)

    return best_a, best


def make_gaussian_init(n, S):
    """Create a Gaussian-shaped initial sequence."""
    x = np.linspace(-1, 1, n)
    a = np.exp(-2.0 * x ** 2)
    return normalize(a, S)


def make_flat_init(n, S):
    """Create a uniform initial sequence."""
    return np.full(n, S / n)


def make_window_init(n, S, width=0.6):
    """Create a raised-cosine window initial sequence."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    # Focus on the center
    center = np.linspace(-1, 1, n)
    a = a * np.exp(-center ** 2 / (2 * width ** 2))
    return normalize(a, S)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: softmax iter -> FISTA -> Adam -> subgradient -> polish.
    Tries multiple n values and initializations.
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Using given sequence of length %d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_config(n, a_start, time_limit, phase_desc):
        """Run optimization pipeline for a given length and start."""
        nonlocal global_best, global_a
        S = np.sqrt(2.0 * n)

        if a_start is not None and len(a_start) != n:
            a_start = resample_sequence(a_start, n)
            a_start = normalize(a_start, S)

        if a_start is None:
            a_start = make_gaussian_init(n, S)

        obj_start = objective_value(a_start)
        print("\n=== %s: n=%d, init obj=%.6f ===" % (phase_desc, n, obj_start))

        best_local = fft_convolve_self(a_start).max()
        best_a_local = a_start.copy()

        # Phase A: Softmax multiplicative iteration (45% of time)
        t_a = time_limit * 0.45
        if t_a > 30:
            a_opt, val_opt = run_softmax_iter(a_start, S, t_a, rng,
                                              gamma_init=15.0, gamma_end=300.0,
                                              eta_init=1.5, eta_decay=0.9996,
                                              k_init=30, k_end=5)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
                print("  Softmax: obj=%.6f (max_b=%.4f)" % (obj, val_opt))

        # Phase B: FISTA (25% of time)
        remaining = deadline - time.monotonic()
        t_b = min(time_limit * 0.25, remaining * 0.5)
        if t_b > 30 and time.monotonic() < deadline - 60:
            a_opt, val_opt = run_fista_topk(best_a_local, S, t_b, rng,
                                            beta_init=100.0, beta_rate=0.003,
                                            beta_max=5000.0, k_frac=0.015)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
                print("  FISTA: obj=%.6f (max_b=%.4f)" % (obj, val_opt))

        # Phase C: Active subgradient (15% of time)
        remaining = deadline - time.monotonic()
        t_c = min(time_limit * 0.15, remaining * 0.4)
        if t_c > 20 and time.monotonic() < deadline - 40:
            a_opt, val_opt = run_active_subgrad(best_a_local, S, t_c, rng,
                                                K=15, step_init=0.05)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
                print("  Subgrad: obj=%.6f (max_b=%.4f)" % (obj, val_opt))

        # Phase D: Coordinate polish (15% of time)
        remaining = deadline - time.monotonic()
        t_d = min(time_limit * 0.15, remaining * 0.5)
        if t_d > 15 and time.monotonic() < deadline - 20:
            a_opt, val_opt = run_polish_coordinate(best_a_local, S, t_d, rng)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * val_opt / (best_a_local.sum() ** 2)
                print("  Polish: obj=%.6f (max_b=%.4f)" % (obj, val_opt))

        obj_final = 2.0 * n * best_local / (best_a_local.sum() ** 2)
        print("  Best for n=%d: obj=%.6f" % (n, obj_final))

        if obj_final < global_best:
            global_best = obj_final
            global_a = best_a_local.copy()
            print("  *** New global best: %.6f ***" % global_best)

        return best_a_local, best_local

    # Main run: n = 16384 with given sequence
    remaining = deadline - time.monotonic()
    n_primary = 16384
    if remaining > 180:
        if given is not None:
            a_start = resample_sequence(given, n_primary)
        else:
            a_start = make_gaussian_init(n_primary, np.sqrt(2 * n_primary))
        try_config(n_primary, a_start, remaining * 0.50, "Main n=16384")

    # Second run: n = 8192 with given sequence (different length may find different basin)
    remaining = deadline - time.monotonic()
    if remaining > 120:
        n_sec = 8192
        if given is not None:
            a_start2 = resample_sequence(given, n_sec)
        else:
            a_start2 = make_gaussian_init(n_sec, np.sqrt(2 * n_sec))
        try_config(n_sec, a_start2, remaining * 0.25, "Secondary n=8192")

    # Third run: n = 32768 if time permits
    remaining = deadline - time.monotonic()
    if remaining > 200 and global_a is not None:
        n_ter = 32768
        a_start3 = resample_sequence(global_a, n_ter)
        try_config(n_ter, a_start3, remaining * 0.30, "Tertiary n=32768")

    # Final intensive polishing on the best found
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final Polish (remaining %.0fs) ===" % remaining)

        # FISTA with high beta
        t_fin1 = remaining * 0.35
        if t_fin1 > 15:
            a_fin, val_fin = run_fista_topk(global_a, S_best, t_fin1, rng,
                                            beta_init=200.0, beta_rate=0.001,
                                            beta_max=10000.0, k_frac=0.008,
                                            use_momentum=False)
            obj_fin = 2.0 * n_best * val_fin / (a_fin.sum() ** 2)
            if obj_fin < global_best:
                global_best = obj_fin
                global_a = a_fin.copy()
                print("  Final FISTA: obj=%.6f" % global_best)

        # Coordinate polish
        remaining2 = deadline - time.monotonic() - 5
        if remaining2 > 15:
            a_pol, val_pol = run_polish_coordinate(global_a, S_best, remaining2, rng)
            obj_pol = 2.0 * n_best * val_pol / (a_pol.sum() ** 2)
            if obj_pol < global_best:
                global_best = obj_pol
                global_a = a_pol.copy()
                print("  Final Polish: obj=%.6f" % global_best)

    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, best obj=%.6f" % (len(global_a), global_best))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
