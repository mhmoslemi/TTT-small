import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def grad_conv_at_lag(a, k):
    """
    Gradient of b_k = sum_i a_i * a_{k-i} with respect to a.
    Result: grad[j] = 2*a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    j = np.arange(n)
    idx = k - j
    valid = (idx >= 0) & (idx < n)
    grad = np.zeros(n)
    grad[valid] = 2.0 * a[idx[valid]]
    return grad


def lse_subgradient(a, c, beta):
    """LSE approximation gradient for smooth optimization."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), 0.0
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, w


def active_set_subgradient(a, c, tol_frac=0.0, max_active=64):
    """
    Subgradient using convex combination of active lags.
    Active lags are those within tol_frac of the max.
    """
    n = len(a)
    cmax = c.max()
    if tol_frac <= 0:
        active_idx = np.array([int(np.argmax(c))])
        w = np.ones(1)
    else:
        thresh = cmax * (1.0 - tol_frac)
        active_idx = np.where(c >= thresh)[0]
        if len(active_idx) > max_active:
            # Keep only the top max_active
            top_vals = np.sort(c)[::-1][:max_active]
            thresh = top_vals[-1]
            active_idx = np.where(c >= thresh)[0]
        # Weight by proximity to max
        w = np.maximum(c[active_idx] / cmax - (1.0 - tol_frac), 1e-10)
        w = w / w.sum()
    
    grad = np.zeros(n)
    for k in active_idx:
        k = int(k)
        grad += w[list(active_idx).index(k)] * grad_conv_at_lag(a, k)
    return grad


def lse_subgradient_fast(a, c, beta):
    """Fast LSE subgradient using FFT."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)

    if tot < 1e-300:

    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

                            else:
                                new_b[k] = b[k] + 2.0 * delta * a[k - j]
                    new_max_b = new_b.max()
                
                obj_candidate = 2.0 * n * new_max_b / (new_s * new_s)
                if obj_candidate < best_obj_candidate:
                    best_obj_candidate = obj_candidate
                    best_x = x
            
            if best_x != old_val:
                a[j] = best_x
                # Update b and s incrementally
                delta = best_x - old_val
                lo = j
                hi = min(j + n - 1, 2 * n - 2)
                if lo <= hi:
                    for k in range(lo, hi + 1):
                        if k == 2 * j:
                            b[k] = b[k] - old_val * old_val + best_x * best_x
                        else:
                            b[k] = b[k] + 2.0 * delta * a[k - j]
                s = s + delta
                improved = True
        
        if not improved:
            break
    
    return a


def coordinate_descent_vectorized(a, n_iterations=3):
    """
    Vectorized coordinate descent pass. More efficient than element-by-element.
    Uses a greedy approach: sort by "importance" and update in that order.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    b = fft_convolve_self(a)
    s = a.sum()
    current_obj = 2.0 * n * b.max() / (s * s)
    
    # Process coordinates in order of their contribution to the peak
    # Coordinates that appear in the worst convolution lags are most important
    bmax_idx = np.argmax(b)
    
    for it in range(n_iterations):
        # Find the worst lag
        worst_k = int(np.argmax(b))
        M = b.max()
        
        # The gradient at the worst lag tells us which coordinates to adjust
        # grad[j] = 2*a[worst_k - j] for 0 <= worst_k - j < n
        j_vals = np.arange(n)
        idx = worst_k - j_vals
        valid = (idx >= 0) & (idx < n)
        
        # Try reducing coordinates with high a[worst_k - j]
        # These are the coordinates that most contribute to the peak
        if not np.any(valid):
            break
        
        # For each valid j, the "contribution" to b[worst_k] is 2*a[j]*a[worst_k-j]
        # (or a[j]^2 if j == worst_k-j)
        # We want to reduce the coordinates that most reduce the peak
        
        # Simple approach: try a small step in the negative gradient direction
        grad = np.zeros(n)
        grad[valid] = 2.0 * a[idx[valid]]
        
        # Try reducing: a_new = a - step * grad, projected
        # Use a small step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Try several step sizes
        best_step = 0.0
        best_obj = current_obj
        for step_frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
            step = step_frac * S / gnorm if (S := a.sum()) > 0 else 0.01
            a_try = np.maximum(a - step * grad, 0.0)
            a_try_sum = a_try.sum()
            if a_try_sum < 1e-15:
                continue
            # Renormalize to keep sum the same (for fair comparison)
            # Actually, don't renormalize - changing the sum changes the objective
            b_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * b_try.max() / (a_try_sum * a_try_sum)
            if obj_try < best_obj:
                best_obj = obj_try
                best_step = step
        
        if best_step == 0.0:
            # Try increasing some coordinates (to increase sum without increasing peak)
            # Increase coordinates that don't contribute much to the peak
            low_contrib = np.zeros(n)
            low_contrib[valid] = grad[valid]
            # Normalize to find "safe" directions
            safe_dir = 1.0 - low_contrib / max(low_contrib.max(), 1e-15)
            safe_dir = np.maximum(safe_dir, 0.0)
            safe_dir_sum = safe_dir.sum()
            if safe_dir_sum > 1e-15:
                safe_dir = safe_dir / safe_dir_sum * a.sum()
                for step_frac in [0.05, 0.1, 0.2]:
                    a_try = a + step_frac * safe_dir * 0.1
                    a_try = np.maximum(a_try, 0.0)
                    b_try = fft_convolve_self(a_try)
                    obj_try = 2.0 * n * b_try.max() / (a_try.sum() ** 2)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        a = a_try
                        s = a_try.sum()
                        b = b_try
        
        if best_step > 0:
            a = np.maximum(a - best_step * grad, 0.0)
            s = a.sum()
            b = fft_convolve_self(a)
            current_obj = 2.0 * n * b.max() / (s * s)
    
    return a


def upsample_smooth(a, factor=2):
    """Upsample using cubic-like smoothing."""
    from scipy.ndimage import zoom
    n = len(a)
    n_new = n * factor
    # Simple linear interpolation
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    a_new = np.interp(t_new, t_old, a)
    return a_new


def make_cosine_init(n, phase=0.0):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * (tt + phase))


def make_dual_bump_init(n):
    """Two bumps: high at start and end, low in middle."""
    tt = np.linspace(0, 1, n, endpoint=False)
    # Create a W-shaped profile
    return 0.3 + 0.7 * (0.5 + 0.5 * np.cos(4 * np.pi * tt)) ** 2


def run_refined_optimization(a_init, max_time, rng):
    """
    Refined optimization: alternate between FISTA and coordinate descent.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    # Phase A: FISTA with moderate beta
    t_A = min(max_time * 0.4, time.monotonic() + max_time * 0.4 - start)
    a_fista, _ = run_fista_backtrack(a, max(1.0, max_time * 0.4), rng, beta_init=5.0)
    obj_fista = compute_obj(a_fista)
    if obj_fista < best_obj:
        best_obj = obj_fista
        best_a = a_fista.copy()
    print("    FISTA: obj=%.8f" % obj_fista)
    
    # Phase B: Coordinate descent polishing
    remaining = max_time - (time.monotonic() - start)
    if remaining > 10:
        t_cd = min(remaining * 0.5, 300.0)
        a_cd = coordinate_descent_vectorized(best_a, n_iterations=5)
        obj_cd = compute_obj(a_cd)
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
        print("    CoordDesc: obj=%.8f" % obj_cd)
    
    # Phase C: More FISTA with higher beta
    remaining = max_time - (time.monotonic() - start)
    if remaining > 10:
        a_f2, _ = run_fista_backtrack(best_a, remaining * 0.6, rng, beta_init=50.0)
        obj_f2 = compute_obj(a_f2)
        if obj_f2 < best_obj:
            best_obj = obj_f2
            best_a = a_f2.copy()
        print("    FISTA-high: obj=%.8f" % obj_f2)
    
    # Phase D: Final coordinate descent
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        a_cd2 = coordinate_descent_vectorized(best_a, n_iterations=3)
        obj_cd2 = compute_obj(a_cd2)
        if obj_cd2 < best_obj:
            best_obj = obj_cd2
            best_a = a_cd2.copy()
        print("    Final CD: obj=%.8f" % obj_cd2)
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA -> coord descent -> upsample -> refine.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    global_best_obj = np.inf
    global_best_a = None
    
    # Try to get the provided sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None
    
    # Determine base n
    if a_start is not None:
        n0 = len(a_start)
    else:
        n0 = 4096
        a_start = make_cosine_init(n0)
    
    print("Base n = %d" % n0)
    print("Total budget: %ds" % budget_s)
    
    S = np.sqrt(2.0 * n0)
    
    # Evaluate starting point
    if a_start is not None and a_start.sum() > 0.01:
        a_norm = a_start * (S / a_start.sum())
        obj_start = compute_obj(a_norm)
        print("Starting obj: %.8f" % obj_start)
        if obj_start < global_best_obj:
            global_best_obj = obj_start
            global_best_a = a_norm.copy()
    else:
        a_norm = make_cosine_init(n0)
        a_norm = a_norm * (S / a_norm.sum())
        obj_start = compute_obj(a_norm)
        print("Cosine init obj: %.8f" % obj_start)
        if obj_start < global_best_obj:

    # === Main optimization loop ===
    # Strategy: run refined optimization, then try upsample
