import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_cross_convolve(a, b):
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]

def normalize(a, S):
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def solve_lp_for_g(f, M, n_constraints=200, max_rounds=20):
    """Solve max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Use cutting-plane with limited constraints."""
    n = len(f)
    
    # Start with the most peaked constraints
    ff = fft_convolve_self(f)
    top_k = np.argsort(ff)[::-1][:n_constraints]
    active_constraints = set(top_k.tolist())
    
    g = None
    for round in range(max_rounds):
        constraints = sorted(active_constraints)
        nc = len(constraints)
        
        # Build sparse matrix
        rows = []
        cols = []
        vals = []
        for i, k in enumerate(constraints):
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                rows.append(i)
                cols.append(j)
                vals.append(f[k - j])
        
        A = csr_matrix((vals, (rows, cols)), shape=(nc, n))
        b = np.full(nc, M)
        c = -np.ones(n)
        
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            break
        
        g = res.x
        
        # Check for violated constraints

        violated = []
        for k in range(len(fg)):

                violated.append((k, fg[k]))
        
        if not violated:
            break
        
        violated.sort(key=lambda x: -x[1])
        for k, _ in violated[:20]:
            active_constraints.add(k)
    
    return g

This is asking: what's the minimum-weight non-negative signal $\lambda$ such that its "cross-correlation" with $f$ is everywhere at least 1?

A simple approach: set $\lambda[k] = 1 / \sum_j f[j]^2$ for all $k$ (uniform). Then $(f \star \lambda)[j] = \sum_k f[k-j] / \sum f^2 = S / \sum f^2$ (constant). If this is $\geq 1$, we're done. Otherwise, we need to increase $\lambda$ where the cross-correlation is low.

2. Compute $c = f \star \lambda$ (cross-correlation)
3. Where $c[j] < 1$, increase $\lambda$ at nearby positions
4. Where $c[j] \gg 1$, decrease $\lambda$ at nearby positions
5. Repeat

At convergence, we get the optimal $\lambda$, and the primal solution $g$ can be recovered from complementary slackness.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full linear cross-convolution via FFT."""
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def normalize(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(a, 0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, constraint_indices):
    """Build the sparse constraint matrix for the LP.
    Row k: A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(f)
    nc = len(constraint_indices)
    rows = []
    cols = []
    vals = []
    
    for i, k in enumerate(constraint_indices):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        js = np.arange(j_min, j_max + 1)
        ks = k - js
        rows.extend([i] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[ks].tolist())
    
    return csr_matrix((vals, (rows, cols)), shape=(nc, n))


def solve_lp_fixed_point(f, M, max_constraints=500, max_rounds=30, verbose=False):
    """Solve max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses cutting-plane with limited constraints."""
    n = len(f)
    
    # Start with the most peaked constraints
    ff = fft_convolve_self(f)
    top_k = np.argsort(ff)[::-1][:max_constraints]
    active_set = set(top_k.tolist())
    
    g = None
    for rnd in range(max_rounds):
        constraints = sorted(active_set)
        nc = len(constraints)
        
        if verbose:
            print(f"    LP round {rnd}: {nc} constraints")
        
        A = build_constraint_matrix(f, constraints)
        b = np.full(nc, M)
        c_obj = -np.ones(n)
        
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            if res.x is not None:
                g = np.maximum(res.x, 0)
            break
        
        g = res.x
        
        # Check for violated constraints
        if g is not None and g.sum() > 0:
            fg = fft_cross_convolve(f, g)
            violated = []
            for k in range(len(fg)):
                if k not in active_set and fg[k] > M * 1.001:
                    violated.append((k, fg[k]))
            
            if violated:
                violated.sort(key=lambda x: -x[1])
                for k, _ in violated[:50]:
                    active_set.add(k)
            else:
                break
        else:
            break
    
    return g if g is not None else np.zeros(n)


def lp_fixed_point_iteration(a, S, max_iters=50, step_size=0.3, max_constraints=300, verbose=True):
    """LP-based fixed-point iteration."""
    n = len(a)
    a = normalize(a.copy(), S)
    
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    for it in range(max_iters):
        c = fft_convolve_self(a)
        M = c.max()
        
        if M < best:
            best = M
            best_a = a.copy()
        
        if verbose:
            obj = 2.0 * n * M / (S * S)
            print(f"  LP-FP iter {it}: M={M:.6f}, obj={obj:.6f}")
        
        # Solve LP
        g = solve_lp_fixed_point(a, M, max_constraints=max_constraints)
        
        if g is None or g.sum() < 1e-10:
            if verbose:
                print("    LP failed, stopping")
            break
        
        # Normalize g to have sum S
        g_norm = g * (S / g.sum())
        
        # Update
        t = step_size
        a_new = (1 - t) * a + t * g_norm
        a = project_simplex(a_new, S)
        
        # Check convergence
        if np.max(np.abs(a - best_a)) / S < 1e-6:
            if verbose:
                print(f"    Converged at iter {it}")
            break
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def lse_smoothed_gradient(a, c, k, beta):
    """Compute gradient of top-k LSE of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    
    w_full = np.zeros(len(c))
    w_full[idx] = w
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista(a_init, S, max_time, rng, beta_init=20.0, beta_rate=0.002, beta_max=3000.0, k_frac=0.05):
    """FISTA with top-k LSE smoothing."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0
    
    k = max(4, int(k_frac * (2 * n - 1)))
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1
        
        grad = lse_smoothed_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / max(beta * 0.5, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        
        for _ in range(5):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-10:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new
        
        if stuck > 2000:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0
    
    return best_a, best


def spectral_flatten(a, alpha, rng):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a
    
    # Damp high-power frequencies
    target_power = power * (1.0 - alpha * (power / mean_p - 1.0) / (1.0 + power / mean_p))
    target_power = np.maximum(target_power, 1e-30)
    
    A_new = A * np.sqrt(target_power / np.maximum(power, 1e-30))
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)

def lp_fixed_point_iteration(a, S, max_time, rng, step_size=0.3, max_constraints=300, verbose=True):
    """LP-based fixed-point iteration with time limit."""
    n = len(a)
    a = normalize(a.copy(), S)
    start = time.monotonic()
    
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()
        
        if M < best:
            best = M
            best_a = a.copy()
        
        if verbose:
            obj = 2.0 * n * M / (S * S)
            print(f"  LP-FP iter {it}: M={M:.8f}, obj={obj:.6f}, time={time.monotonic()-start:.1f}s")
        
        # Solve LP
        g = solve_lp_fixed_point(a, M, max_constraints=max_constraints)
        
        if g is None or g.sum() < 1e-10:
            if verbose:
                print("    LP failed, stopping")
            break
        
        # Normalize g to have sum S
        g_sum = g.sum()
        g_norm = g * (S / g_sum)
        
        # Line search for step size
        best_t = step_size
        best_obj = M
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1 - t) * a + t * g_norm
            a_trial = project_simplex(a_trial, S)
            M_trial = fft_convolve_self(a_trial).max()
            if M_trial < best_obj:
                best_obj = M_trial
                best_t = t
        
        a = (1 - best_t) * a + best_t * g_norm
        a = project_simplex(a, S)
        
        # Check for convergence
        if abs(M - best) / best < 1e-8:
            if verbose:
                print(f"    Converged")
            break
        
        # Perturbation
        if it % 10 == 0:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

def build_constraint_matrix_fast(f, constraint_indices):
    """Build the sparse constraint matrix for the LP using vectorized operations."""
    n = len(f)
    nc = len(constraint_indices)
    
    # For each constraint k, the non-zero positions are j in [max(0, k-n+1), min(k, n-1)]
    # with value f[k-j]
    
    all_rows = []
    all_cols = []
    all_vals = []
    
    for i, k in enumerate(constraint_indices):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min > j_max:
            continue
        js = np.arange(j_min, j_max + 1)
        all_rows.extend([i] * len(js))
        all_cols.extend(js.tolist())
        all_vals.extend(f[k - js].tolist())
    
    return csr_matrix((all_vals, (all_rows, all_cols)), shape=(nc, n))

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def normalize(a, S):
    a = np.clip(a, 0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, constraint_indices):
    """Build sparse constraint matrix for LP. Row k: A[k,j] = f[k-j] if valid."""
    n = len(f)
    nc = len(constraint_indices)
    rows = np.empty(nc * n, dtype=np.int32)  # worst case
    cols = np.empty(nc * n, dtype=np.int32)
    vals = np.empty(nc * n, dtype=np.float64)
    
    total = 0
    for i, k in enumerate(constraint_indices):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        js = np.arange(j_min, j_max + 1)
        count = len(js)
        rows[total:total+count] = i
        cols[total:total+count] = js
        vals[total:total+count] = f[k - js]
        total += count
    
    return csr_matrix((vals[:total], (rows[:total], cols[:total])), shape=(nc, n))

One more thought: I should also try a "dual ascent" approach where I iteratively improve the dual solution. The dual problem is:
$$\min M \cdot \sum_k \lambda_k \quad \text{s.t.} \quad \sum_m f[m] \lambda[m+j] \geq 1 \; \forall j, \; \lambda \geq 0$$

This is a linear program with $2n-1$ variables and $n$ constraints. The constraint matrix is again a convolution matrix, so I can use FFT for matrix-vector products.

A projected gradient method for the dual:
1. Initialize $\lambda > 0$
2. Compute the constraint violation: $v[j] = \max(0, 1 - \sum_m f[m] \lambda[m+j])$
3. Update: $\lambda \leftarrow \text{proj}_{\lambda \geq 0}(\lambda - \eta \cdot (M \cdot \mathbf{1} - A^T v))$

where $A$ is the constraint matrix and $A^T v$ is the gradient of the penalty term.

The matrix-vector product $A^T v$ can be computed via FFT:
$(A^T v)[k] = \sum_j A[k,j] v[j] = \sum_j f[k-j] v[j]$ (cross-correlation)

This is efficient! Let me implement this as an alternative to the primal LP:

def solve_lp_dual_pgm(f, M, max_iters=10000, tol=1e-8):
    """Solve the dual LP using projected gradient method with FFT-accelerated matrix-vector products."""
    n = len(f)
    lam_size = 2 * n - 1
    
    # FFT size for cross-correlation
    fsize = 1 << (n + lam_size - 1).bit_length()
    F = fft(f, fsize)
    
    # Initialize lambda
    lam = np.full(lam_size, 1.0 / (n * M))
    
    # Learning rate (determined by Lipschitz constant of the gradient)
    # The gradient of the Lagrangian w.r.t. lambda involves A^T * v, which has norm bounded by
    # ||f||_1 * ||v||_2 <= S * ||v||_2
    # So the Lipschitz constant is M + S^2 (approximately)
    # For safety, use a small step size
    eta = 1.0 / (M + 1.0)
    
    best_obj = np.inf
    best_lam = lam.copy()
    
    for it in range(max_iters):
        # Compute constraint violation: v[j] = max(0, 1 - sum_m f[m] * lam[m+j])
        L = fft(lam, fsize)
        # cross-correlation: corr[k] = sum_j f[k-j] * lam[j] ... 
        # Actually, I need: c[j] = sum_m f[m] * lam[m+j] for j = 0, ..., n-1
        # This is: c = ifft(conj(FFT(f)) * FFT(lam_padded)) ... 
        
        # Let me think about this more carefully.
        # I want c[j] = sum_{m=0}^{n-1} f[m] * lam[m+j] for j = 0, ..., n-1
        # 
        # Define the linear cross-correlation: (f ⋆ lam)[j] = sum_m f[m] * lam[m+j]
        # This is equivalent to: (f ⋆ lam)[j] = sum_m f[m] * conj(conj(lam))[m - (-j)]
        # = cross_correlation(f, lam)[j]
        #
        # Via FFT: cross_correlation(f, g)[j] = ifft(FFT(f) * conj(FFT(g)))[j]
        # (with appropriate zero-padding for linear correlation)
        
        corr = np.real(ifft(F * np.conj(L), fsize))
        # corr[j] = sum_m f[m] * lam[m+j] (circular, but with sufficient padding it's linear)
        c = corr[:n]  # the part we need
        
        # Deficit
        deficit = np.maximum(0, 1.0 - c)
        
        # Check feasibility
        if deficit.max() < tol:
            obj = M * lam.sum()
            if obj < best_obj:
                best_obj = obj
                best_lam = lam.copy()
            break
        
        # Gradient step
        # The dual objective is M * sum(lam) + penalty for constraint violation
        # L = M * sum(lam) + sum_j v[j] * (1 - c[j])  (Lagrangian with v >= 0 as multipliers)
        # dL/dlam[k] = M - sum_j v[j] * f[k-j]  (where the sum is over valid j)
        # = M - (v ⋆ f)[k]  (cross-correlation of v and f at position k)
        
        # Extend deficit to full length for FFT
        deficit_full = np.zeros(fsize)
        deficit_full[:n] = deficit
        
        D = fft(deficit_full, fsize)
        # (deficit ⋆ f)[k] = sum_j deficit[j] * f[k-j]
        # = ifft(conj(FFT(deficit)) * FFT(f))[k] ... 
        # Wait, let me be careful.
        # cross_correlation(a, b)[k] = sum_j a[j] * b[j+k] = ifft(FFT(a) * conj(FFT(b)))[k]
        # So (deficit ⋆ f)[k] = sum_j deficit[j] * f[j+k] = ifft(FFT(deficit) * conj(FFT(f)))[k]
        
        # But I want: sum_j deficit[j] * f[k-j] = (f ⋆ deficit)[k] = ifft(FFT(f) * conj(FFT(deficit)))[k]
        
        grad_lam = M - np.real(ifft(F * np.conj(D), fsize))[:lam_size]
        
        # Wait, that's not right. Let me re-derive.
        # c[j] = sum_m f[m] * lam[m+j]
        # dc[j]/dlam[k] = f[k-j] if 0 <= k-j < n, else 0
        # 
        # The Lagrangian (for the dual) is:
        # L = M * sum_k lam[k] + sum_j v[j] * (1 - c[j])
        # dL/dlam[k] = M - sum_j v[j] * f[k-j]
        # 
        # Now, sum_j v[j] * f[k-j] = (f * v)[k] where * is convolution... 
        # No: sum_j v[j] * f[k-j] IS the convolution (f * v)[k]!
        # So dL/dlam[k] = M - (f * v)[k]
        
        # Compute (f * deficit)[k] for k = 0, ..., 2n-2
        # This is the linear convolution of f and deficit_padded
        # deficit has length n (positions 0 to n-1), f has length n
        # The convolution has length 2n-1
        
        conv = fft_cross_convolve(f, deficit)  # length 2n-1
        # conv[k] = sum_j f[k-j] * deficit[j]
        # But I want sum_j deficit[j] * f[k-j] which is the same thing!
        
        grad_lam = M - conv  # length 2n-1 = lam_size
        
        # Projected gradient step
        lam_new = np.maximum(lam - eta * grad_lam, 0)
        
        # Adaptive step size
        obj_new = M * lam_new.sum()
        obj_old = M * lam.sum()
        
        # We want to decrease the dual objective while maintaining feasibility
        # Actually, for the dual, we're minimizing, so we want obj_new < obj_old
        # But we also need to maintain the constraint c[j] >= 1
        # 
        # Let's use a simple backtracking:
        if obj_new > obj_old and it > 0:
            eta *= 0.5
            lam_new = np.maximum(lam - eta * grad_lam, 0)
        
        lam = lam_new
        lam = np.maximum(lam, 1e-15)  # keep positive for next iteration
        
        if it % 1000 == 0:
            print(f"    Dual PGM iter {it}: obj={M*lam.sum():.6f}, max_deficit={deficit.max():.8f}")
    
    return best_lam
