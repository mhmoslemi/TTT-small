```python
"""
Search algorithm for non-negative step sequences minimizing
    2*n*max(conv(a,a)) / sum(a)^2.

Approach:
- Homogeneous objective: normalize sum(a) = sqrt(2n), so score = max conv.
- Adaptive log-sum-exp (LSE) smoothing of the convolution peak.
- Proximal L-BFGS directions on the simplex {a >= 0, sum(a)=S}.
- FFT-based convolution and gradients.
- Spectral phase/noise restarts to escape local minima.
- Multiscale run: improve a length-8192 candidate, then upscale to 16384.
- Optional Nesterov top-k polishing for exact peak.
- Final direct calls to evaluate_sequence are used only to validate/select.
"""

import numpy as np
import time

try:
    from scipy.fft import fft, ifft
except Exception:
    from numpy.fft import fft, ifft


def next_pow2_ge(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_sequence(v, n, S):
    """Resize, clean, clip, and rescale to sum S."""
    try:
        arr = np.asarray(v, dtype=np.float64).ravel()
    except Exception:
        return np.ones(n, dtype=np.float64) * (S / max(1, n))

    if arr.size == 0:
        return np.ones(n, dtype=np.float64) * (S / max(1, n))

    if arr.size != n:
        if arr.size == 1:
            arr = np.full(n, float(arr[0]), dtype=np.float64)
        else:
            x_old = np.linspace(0.0, 1.0, arr.size)
            x_new = np.linspace(0.0, 1.0, n)
            arr = np.interp(x_new, x_old, arr)

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)

    s = float(np.sum(arr))
    if s <= 1e-15:
        return np.ones(n, dtype=np.float64) * (S / max(1, n))

    arr = arr * (S / s)

    # Extremely unlikely for our scales, but keep it valid for the evaluator.
    if arr.size and float(np.max(arr)) > 1000.0:
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(np.sum(arr))
        if s <= 1e-15:
            return np.ones(n, dtype=np.float64) * (S / max(1, n))
        arr = arr * (S / s)

    return arr


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x)=S}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if S <= 0.0:
        return np.zeros(n, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.ones(n, dtype=np.float64) * (S / n)

    # Fast path: if a uniform shift works, avoid sorting.
    s = float(np.sum(v))
    delta = (S - s) / float(n)
    if float(np.min(v)) + delta >= -1e-12:
        out = np.maximum(v + delta, 0.0)
        ss = float(np.sum(out))
        if ss > 0.0:
            out = out * (S / ss)
            if float(np.min(out)) >= -1e-12 and abs(float(np.sum(out)) - S) <= 1e-7 * max(1.0, S):
                return out

    # Full projection by sorting.
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0])) + 1
    theta = (cumsum[rho - 1] - S) / rho
    out = np.maximum(v - theta, 0.0)

    ss = float(np.sum(out))
    if ss <= 0.0:
        return np.ones(n, dtype=np.float64) * (S / n)

    out = out * (S / ss)
    diff = S - float(np.sum(out))
    if diff != 0.0 and float(np.max(out)) > 0.0:
        j = int(np.argmax(out))
        out[j] += diff
        if out[j] < 0.0:
            out = np.maximum(out, 0.0)
            ss = float(np.sum(out))
            if ss > 0.0:
                out = out * (S / ss)
            else:
                out = np.ones(n, dtype=np.float64) * (S / n)
    return out


def compute_conv(a, N):
    """Linear convolution a*a via FFT; returns first 2*len(a)-1 coefficients."""
    n = a.size
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))
    c = c[:2 * n - 1]
    if c.size and not np.all(np.isfinite(c)):
        c = np.where(np.isfinite(c), c, 0.0)
    return c


def lse_value(c, beta):
    """Stable log-sum-exp of c with smoothing beta."""
    cmax = float(np.max(c))
    if beta < 1e-8:
        return cmax
    expb = np.exp(beta * (c - cmax))
    tot = float(np.sum(expb))
    if not np.isfinite(tot) or tot <= 0.0:
        return cmax
    return cmax + float(np.log(tot)) / beta


def lse_weights(c, beta):
    """Weights for LSE-smoothed max and the corresponding LSE value."""
    cmax = float(np.max(c))
    if beta < 1e-8:
        w = np.full(c.shape, 1.0 / max(1, c.size), dtype=np.float64)
        return w, cmax, cmax

    expb = np.exp(beta * (c - cmax))
    tot = float(np.sum(expb))
    if not np.isfinite(tot) or tot <= 0.0:
        w = np.zeros(c.shape, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w, cmax, cmax

    w = expb / tot
    lse = cmax + float(np.log(tot)) / beta
    return w, lse, cmax


def gradient_from_conv(a, c, beta, F):
    """
    Gradient of LSE-smoothed peak of a*a.

    For b_k = sum_i a_i a_{k-i},
    grad_l = 2 * sum_k w_k a_{k-l}.
    """
    n = a.size
    w, lse, cmax = lse_weights(c, beta)

    W = fft(w, F)
    Af = fft(a, F)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]

    if not np.all(np.isfinite(grad)):
        grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad, lse, cmax


def lbfgs_dir(s_hist, y_hist, g):
    """Two-loop L-BFGS direction, normalized to unit length."""
    m = len(s_hist)
    nvec = g.size

    if m == 0:
        dn = float(np.linalg.norm(g))
        if dn < 1e-300:
            return np.zeros(nvec, dtype=np.float64)
        return -g / dn

    q = g.copy()
    alphas = np.zeros(m, dtype=np.float64)

    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = float(np.dot(y_i, y_i))
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = float(np.dot(s_i, q)) / yy
            q = q - alphas[i] * y_i

    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = float(np.dot(y_last, y_last))
    if yy_last > 1e-30:
        gamma = max(1e-8, float(np.dot(s_last, y_last)) / yy_last)
    else:
        gamma = 1.0

    r = gamma * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = float(np.dot(y_i, y_i))
        if yy < 1e-30:
            continue
        beta_i = float(np.dot(s_i, y_i)) / yy
        r = r + (alphas[i] - beta_i) * y_i

    direction = -r
    if float(np.dot(g, direction)) > 0.0:
        direction = -g

    dn = float(np.linalg.norm(direction))
    if dn < 1e-300:
        direction = -g
        dn = float(np.linalg.norm(direction))
    if dn > 1e-300:
        direction = direction / dn
    return direction


def spectral_restart(a, N, rng, frac=1.0):
    """Randomize FFT phases of a zero-padded sequence, then project to nonnegative."""
    n = a.size
    S = float(np.sum(a))
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = fft(a_pad)
    mag = np.abs(A)

    if frac >= 0.999:
        phase = rng.uniform(0.0, 2.0 * np.pi, N)
    else:
        phase = np.angle(A)
        n_rand = max(1, int(N * frac))
        if n_rand >= N:
            phase = rng.uniform(0.0, 2.0 * np.pi, N)
        else:
            idx = rng.choice(N, size=n_rand, replace=False)
            phase[idx] = rng.uniform(0.0, 2.0 * np.pi, n_rand)

    A_new = mag * np.exp(1j * phase)
    anew = np.real(ifft(A_new))[:n]
    anew = np.maximum(anew, 0.0)
    s = float(np.sum(anew))
    if s <= 1e-15:
        return a.copy()
    return anew * (S / s)


def noisy_restart(a, rng, frac=0.1):
    """Add relative Gaussian noise and renormalize."""
    n = a.size
    S = float(np.sum(a))
    sigma = frac * S / max(1, n)
    anew = a + rng.normal(0.0, sigma, n)
    return normalize_sequence(anew, n, S)


def uniform_start(n, S):
    return np.ones(n, dtype=np.float64) * (S / n)


def random_smooth_start(n, S, rng):
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    base = np.ones(n, dtype=np.float64)
    for k in range(1, 17):
        amp = rng.uniform(0.05, 0.6) / np.sqrt(float(k))
        base += amp * np.sin(2.0 * np.pi * k * t + rng.uniform(0.0, 2.0 * np.pi))
    base = np.maximum(base, 1e-4)
    return normalize_sequence(base, n, S)


def beta_profile_start(n, S, rng):
    t = np.linspace(1e-6, 1.0 - 1e-6, n)
    alpha = rng.uniform(0.3, 2.5)
    beta = rng.uniform(0.3, 2.5)
    p = np.power(t, alpha) * np.power(1.0 - t, beta) + 0.25
    p = np.maximum(p, 1e-4)
    return normalize_sequence(p, n, S)


def two_block_start(n, S, rng):
    a = np.zeros(n, dtype=np.float64)
    l1 = int(n * rng.uniform(0.25, 0.45))
    l2 = int(n * rng.uniform(0.25, 0.45))
    if l1 + l2 > n:
        l2 = max(1, n - l1)
    a[:l1] = 1.0
    a[n - l2:] = 1.0
    a += 0.05
    return normalize_sequence(a, n, S)


def make_initial_sequence(n, S, rng, heights=None, use_given=True):
    if use_given and heights is not None and heights.size > 0:
        return normalize_sequence(heights, n, S)

    choice = int(rng.integers(0, 4))
    if choice == 0:
        return random_smooth_start(n, S, rng)
    if choice == 1:
        return beta_profile_start(n, S, rng)
    if choice == 2:
        return two_block_start(n, S, rng)
    return uniform_start(n, S)


def run_optimizer(a_init, n, S, N, F, budget, rng,
                  beta_start=3.0, beta_max=800.0, m_lbfgs=12,
                  max_line=10, step_mult=1.0):
    """Adaptive projected L-BFGS on LSE-smoothed convolution peak."""
    a = normalize_sequence(a_init, n, S)
    c = compute_conv(a, N)
    cmax = float(np.max(c))
    best_a = a.copy()
    best_c = cmax

    if budget <= 0.1:
        return best_a, best_c

    beta = float(beta_start)
    grad, f, _ = gradient_from_conv(a, c, beta, F)
    s_hist = []
    y_hist = []

    start = time.monotonic()
    last_best_time = start
    it = 0
    step_factor = float(step_mult)
    stall = 0
    avg = S / float(n)

    while time.monotonic() - start < budget:
        it += 1
        now = time.monotonic()

        # Adapt smoothing level.
        if it % 80 == 0:
            slow = (now - last_best_time) > 60.0
            if slow:
                beta = min(beta_max, beta * (1.05 + 0.02 * min(stall, 20)))
            else:
                beta = min(beta_max, beta * 1.015)
            step_factor = min(step_mult * 1.3, max(0.25 * step_mult, step_factor * 1.08))

        # Restart on long stalls or occasionally.
        if stall > 80 or (it % 220 == 0 and rng.random() < 0.22):
            if stall > 180:
                a = spectral_restart(a, N, rng, 1.0)
            elif rng.random() < 0.55:
                a = spectral_restart(a, N, rng, 0.25 + 0.45 * rng.random())
            else:
                a = noisy_restart(a, rng, 0.05 + 0.15 * rng.random())
            s_hist = []
            y_hist = []
            beta = max(beta_start, beta * 0.4)
            stall = 0
            c = compute_conv(a, N)
            cmax = float(np.max(c))
            grad, f, _ = gradient_from_conv(a, c, beta, F)
            if cmax < best_c - 1e-12:
                best_c = cmax
                best_a = a.copy()
                last_best_time = time.monotonic()
            continue

        direction = lbfgs_dir(s_hist, y_hist, grad)
        denom = max(1.0, beta ** 0.35)
        alpha = 0.75 * avg * step_factor / denom

        accepted = False
        cand = a
        c_ls = c
        cmax_ls = cmax

        for _ in range(max_line):
            candidate = project_simplex(a + alpha * direction, S)
            c_candidate = compute_conv(candidate, N)
            cm_candidate = float(np.max(c_candidate))

            if cm_candidate <= cmax - 1e-12 or cm_candidate < best_c - 1e-12:
                accepted = True
                cand = candidate
                c_ls = c_candidate
                cmax_ls = cm_candidate
                break

            lse_candidate = lse_value(c_candidate, beta)
            if lse_candidate <= f + 1e-12:
                accepted = True
                cand = candidate
                c_ls = c_candidate
                cmax_ls = cm_candidate
                break

            alpha *= 0.5
            if alpha <= avg * 1e-9:
                break

        # Fallback: pure projected gradient step.
        if not accepted:
            gdir = -grad
            gnorm = float(np.linalg.norm(gdir))
            alpha = avg * step_factor / denom
            if gnorm > 1e-300:
                gdir = gdir / gnorm
                for _ in range(5):
                    candidate = project_simplex(a + alpha * gdir, S)
                    c_candidate = compute_conv(candidate, N)
                    cm_candidate = float(np.max(c_candidate))
                    if cm_candidate <= cmax - 1e-12 or cm_candidate < best_c - 1e-12:
                        accepted = True
                        cand = candidate
                        c_ls = c_candidate
                        cmax_ls = cm_candidate
                        break
                    alpha *= 0.5

        if not accepted:
            # Force a local escape instead of getting stuck.
            if rng.random() < 0.6:
                a = spectral_restart(a, N, rng, 0.3 + 0.5 * rng.random())
            else:
                a = noisy_restart(a, rng, 0.1)
            s_hist = []
            y_hist = []
            beta = max(beta_start, beta * 0.5)
            c = compute_conv(a, N)
            cmax = float(np.max(c))
            grad, f, _ = gradient_from_conv(a, c, beta, F)
            stall += 8
            step_factor = max(0.25 * step_mult, step_factor * 0.6)
            if stall > 400:
                break
            continue

        grad_new, f_new, cmax_new = gradient_from_conv(cand, c_ls, beta, F)
        sk = cand - a
        yk = grad_new - grad
        ss = float(np.dot(sk, sk))
        sy = float(np.dot(sk, yk))
        if ss > 1e-20 and sy > 1e-12 * max(1.0, ss):
            s_hist.append(sk)
            y_hist.append(yk)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = cand
        grad = grad_new
        f = f_new
        c = c_ls
        cmax = cmax_new

        if cmax < best_c - 1e-11:
            best_c = cmax
            best_a = a.copy()
            stall = 0
            last_best_time = time.monotonic()
            step_factor = min(step_mult * 1.5, step_factor * 1.12)
        else:
            stall += 1
            if stall > 50:
                step_factor = max(0.25 * step_mult, step_factor * 0.9)

    return best_a, best_c


def nesterov_topk(a_init, n, S, N, F, budget, rng, top_k_init=16, top_k_max=256):
    """Momentum top-k subgradient polishing for the exact maximum coefficient."""
    a = normalize_sequence(a_init, n, S)
    c0 = compute_conv(a, N)
    best_a = a.copy()
    best_c = float(np.max(c0))

    if budget <= 0.1:
        return best_a, best_c

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    it = 0
    avg = S / float(n)

    while time.monotonic() - start < budget:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 25)

        y = np.maximum(y, 0.0)
        sy_sum = float(np.sum(y))
        if sy_sum > 0.0:
            y = y * (S / sy_sum)

        c = compute_conv(y, N)
        cur = float(np.max(c))
        if cur < best_c - 1e-12:
            best_c = cur
            best_a = y.copy()

        order = np.argsort(c)[::-1][:top_k]
        vals = c[order]
        total = float(np.sum(vals))
        if total <= 0.0:
            break

        weights = vals / total
        weights = 0.75 * weights + 0.25 * (1.0 / float(len(order)))
        weights = weights / float(np.sum(weights))

        w_full = np.zeros(len(c), dtype=np.float64)
        for i, idx in enumerate(order):
            w_full[int(idx)] = weights[i]

        W = fft(w_full, F)
        Yf = fft(y, F)
        grad = 2.0 * np.real(ifft(W * np.conj(Yf)))[:n]
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-18:
            break

        step = 0.18 * avg / (1.0 + 0.001 * it)
        x = project_simplex(y - step * grad / gnorm, S)
        x = np.maximum(x, 0.0)

        if it % 4 == 0:
            cx = compute_conv(x, N)
            cxmax = float(np.max(cx))
            if cxmax < best_c - 1e-12:
                best_c = cxmax
                best_a = x.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        yy_sum = float(np.sum(y))
        if yy_sum > 0.0:
            y = y * (S / yy_sum)
        else:
            y = x.copy()

        a = x
        t = t_new

        if it % 350 == 0 and rng.random() < 0.12:
            y = spectral_restart(y, N, rng, 0.25)

    return best_a, best_c


def to_safe_list(a):
    out = []
    for x in a:
        xf = float(x)
        if not np.isfinite(xf):
            xf = 0.0
        if xf < 0.0:
            xf = 0.0
        if xf > 1000.0:
            xf = 1000.0
        out.append(xf)
    return out


def get_previous_heights():
    try:
        hs = height_sequence_1
        arr = np.asarray(hs, dtype=np.float64).ravel()
        if arr.size == 0:
            return None
        arr = np.where(np.isfinite(arr), arr, 0.0)
        return arr
    except Exception:
        return None


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    """
    Return a list of non-negative heights minimizing the evaluator score.
    """
    start_global = time.monotonic()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0
    if not np.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 2400.0

    # Leave time for final direct evaluation and fallback.
    deadline = start_global + max(60.0, budget_s) - 160.0
    rng = np.random.default_rng(int(seed) if np.isfinite(getattr(seed, "item", float("nan"))) else 42)

    heights = get_previous_heights()

    def remaining_time():
        rem = deadline - time.monotonic()
        if not np.isfinite(rem):
            rem = 0.0
        return float(max(0.0, rem))

    # ------------------- Phase 1: main length ~8192 -------------------
    n1 = 8192
    S1 = float(np.sqrt(2.0 * float(n1)))
    N1 = next_pow2_ge(2 * n1 - 1)
    F1 = next_pow2_ge(3 * n1 - 2)

    a1 = make_initial_sequence(n1, S1, rng, heights=heights, use_given=True)
    c1 = compute_conv(a1, N1)
    best_score = float(np.max(c1))
    best_seq = a1.copy()
    initial_candidates = [(best_score, a1.copy())]

    print("start n1=%d internal_score=%.6f" % (n1, best_score))

    budget1 = min(750.0, remaining_time() * 0.40)
    if budget1 > 30.0:
        a1, best1 = run_optimizer(a1, n1, S1, N1, F1, budget1, rng,
                                  beta_start=3.0, beta_max=1000.0,
                                  m_lbfgs=12, max_line=10, step_mult=1.0)
        if best1 < best_score - 1e-12:
            best_score = best1
            best_seq = a1.copy()
            print("after n1 optimizer score=%.6f time=%.1f" % (best_score, time.monotonic() - start_global))
    else:
        print("skipping n1 optimizer, budget too small")

    initial_candidates.append((best_score if best_seq is not None else np.inf, best_seq.copy() if best_seq is not None else None))

    # ------------------- Phase 2: upscale to 16384 -------------------
    rem = remaining_time()
    use_large = rem > 550.0
    if use_large:
        n2 = 16384
        S2 = float(np.sqrt(2.0 * float(n2)))
        N2 = next_pow2_ge(2 * n2 - 1)
        F2 = next_pow2_ge(3 * n2 - 2)

        # Primary: continue from the best current sequence.
        a2 = normalize_sequence(best_seq, n2, S2)
        budget2_primary = min(remaining_time() - 130.0, 1550.0)
        if budget2_primary > 60.0:
            a2, best2 = run_optimizer(a2, n2, S2, N2, F2, budget2_primary, rng,
                                      beta_start=2.5, beta_max=1400.0,
                                      m_lbfgs=14, max_line=10, step_mult=1.0)
            if best2 < best_score - 1e-12:
                best_score = best2
                best_seq = a2.copy()
                print("after n2 primary score=%.6f time=%.1f" % (best_score, time.monotonic() - start_global))
            initial_candidates.append((best2, a2.copy()))

        # Secondary: a fresh random-ish 16k start if time remains.
        rem = remaining_time()
        if rem > 260.0 and (heights is None or best_score > 1.52):
            a2b = make_initial_sequence(n2, S2, rng, heights=heights, use_given=False)
            budget2_second = min(rem - 120.0, 500.0)
            if budget2_second > 40.0:
                a2b, best2b = run_optimizer(a2b, n2, S2, N2, F2, budget2_second, rng,
                                            beta_start=3.0, beta_max=1400.0,
                                            m_lbfgs=14, max_line=10, step_mult=1.0)
                if best2b < best_score - 1e-12:
                    best_score = best2b
                    best_seq = a2b.copy()
                    print("after n2 secondary score=%.6f time=%.1f" % (best_score, time.monotonic() - start_global))
                initial_candidates.append((best2b, a2b.copy()))

    # ------------------- Phase 3: Nesterov top-k polish -------------------
    rem = remaining_time()
    if rem > 100.0 and best_seq is not None:
        np_len = int(best_seq.size)
        Sfin = float(np.sqrt(2.0 * float(np_len)))
        Nfin = next_pow2_ge(2 * np_len - 1)
        Ffin = next_pow2_ge(3 * np_len - 2)
        budget_polish = min(rem - 50.0, 350.0)
        if budget_polish > 30.0:
            a_pol, best_pol = nesterov_topk(best_seq, np_len, Sfin, Nfin, Ffin,
                                            budget_polish, rng,
                                            top_k_init=24, top_k_max=320)
            if best_pol < best_score - 1e-12:
                best_score = best_pol
                best_seq = a_pol.copy()
                print("after nesterov polish score=%.6f time=%.1f" % (best_score, time.monotonic() - start_global))
            initial_candidates.append((best_pol, a_pol.copy()))

    # Ensure a safe best sequence exists.
    if best_seq is None:
        best_seq = uniform_start(256, float(np.sqrt(512.0)))
        best_score = compute_conv(best_seq, next_pow2_ge(2 * 256 - 1)).max()

    # ------------------- Final direct evaluation / selection -------------------
    final_list = to_safe_list(best_seq)

    # Sanitize final list.
    if len(final_list) == 0:
        final_list = [1.0]

    if sum(final_list) < 0.01:
        final_list = [1.0] * 256

    # Evaluate final and one earlier candidate if length differs and time allows.
    candidates = []

    # Add final candidate.
    candidates.append((len(final_list), final_list))

    # Add best earlier candidate with different length, if any.
    for sc, seq in initial_candidates:
        if seq is None:
            continue
        lst = to_safe_list(seq)
        if len(lst) != len(final_list):
            # only keep earlier candidates that are not obviously bad
            if np.isfinite(sc) and sc < max(2.0, best_score + 0.05):
                candidates.append((len(lst), lst))
            break

    best_actual = None
    best_actual_list = final_list

    for _len, lst in candidates:
        lst = to_safe_list(lst)
        if len(lst) == 0:
            continue
        if sum(lst) < 0.01:
            continue
        rem = remaining_time()
        if rem > 40.0 and _len <= 20000:
            try:
                val = evaluate_sequence(lst)
                if not np.isfinite(val):
                    val = np.inf
            except Exception:
                val = np.inf
        else:
            val = np.inf
            # If no evaluation budget, use internal score as proxy.
            if _len == len(final_list):
                val = best_score

        if np.isfinite(val):
            if best_actual is None or val < best_actual:
                best_actual = val
                best_actual_list = lst

    # If direct evaluation did not produce anything finite, return internal best.
    if best_actual is None:
        best_actual_list = final_list

    # Defensive fallback if final sequence somehow invalid.
    try:
        if len(best_actual_list) == 0 or not all(np.isfinite(x) for x in best_actual_list):
            best_actual_list = [1.0] * 256
    except Exception:
        best_actual_list = [1.0] * 256

    print("final length=%d internal_score=%.6f" % (len(best_actual_list), best_score))
    return [float(x) for x in best_actual_list]
```
