def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Ensure exact sum
    diff = S - result.sum()
    if abs(diff) > 1e-12:
        # Distribute the difference among positive elements
        pos = result > 0
        if pos.any():
            result[pos] += diff / pos.sum()
            result = np.maximum(result, 0)
    return result

    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip to non-negative and rescale to sum S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_lse_gradient(a, c, L, beta):
    """
    Compute gradient of LSE_beta(a*a) with respect to a.
    Uses the identity: grad = 2 * IFFT(FFT(w) * conj(FFT(a)))[:n]
    where w = softmax(beta * c).
    """
    n = len(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -600.0, 0.0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        w = np.full(len(c), 1.0 / len(c))
    else:
        w = e / tot
    W = fft(w, L)
    A = fft(a, L)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista_lse(a_init, S, max_time, rng,
                  beta_start=None, beta_end=None,
                  momentum=True, print_interval=5000):
    """
    FISTA (Nesterov accelerated gradient) with full LSE smoothing.
    Minimizes max(a*a) approximately by minimizing LSE_beta(a*a).
    """
    n = len(a_init)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()

    a = normalize_to_sum(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()

    c0, _ = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    no_improve = 0

    if beta_start is None:
        beta_start = 2.0 / max(best, 0.1)
    if beta_end is None:
        beta_end = 500.0 / max(best, 0.1)

    total_time = max_time
    while time.monotonic() - start < total_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / total_time)
        # Geometric interpolation of beta
        log_beta = np.log(beta_start) + progress * (np.log(beta_end) - np.log(beta_start))
        beta = np.exp(log_beta)
        beta = min(beta, 100000.0)

        c_y, _ = fft_convolve_self(y)
        cur_max = c_y.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute LSE gradient
        grad = compute_lse_gradient(y, c_y, L, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: heuristic based on beta and signal scale
        # The gradient magnitude scales with ~ beta * max(a), so step ~ 1/(beta * max(a))
        step = 1.0 / (2.0 * beta * max(y.max(), 1e-10) + 1e-10)
        # Additional damping for stability
        step *= 0.8

        x_new = project_simplex(y - step * grad, S)

        if momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
            y_new = np.clip(y_new, 0.0, None)
            s_new = y_new.sum()
            if s_new > 1e-15:
                y_new *= S / s_new
            else:
                y_new = a.copy()
            t_val = t_new
        else:
            y_new = x_new.copy()

        a = x_new
        y = y_new

        # Periodic restart when stuck
        if no_improve > 800:
            noise_scale = 0.002 * S / np.sqrt(n) * (1.0 - progress) + 1e-6
            y = project_simplex(a + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            no_improve = 0

        # Spectral perturbation for exploration
        if it % 10000 == 0 and progress < 0.7 and rng.random() < 0.3:
            fsize = L
            A_y = fft(y, fsize)
            phase = rng.normal(0, 0.05, fsize)
            phase[0] = 0
            A_new = A_y * np.exp(1j * phase)
            y_new = np.real(ifft(A_new, fsize))[:n]
            y = normalize_to_sum(y_new, S)
            t_val = 1.0
            no_improve = 0

        if it % print_interval == 0:
            obj = 2.0 * n * best / (S * S)
            print("    FISTA it=%d, beta=%.1f, best_max=%.8f, obj=%.6f" %
                  (it, beta, best, obj))

    # Final check
    c_final, _ = fft_convolve_self(y)
    if c_final.max() < best:
        best = c_final.max()
        best_a = y.copy()
    c_a, _ = fft_convolve_self(a)
    if c_a.max() < best:
        best = c_a.max()
        best_a = a.copy()

    return best_a, best


def run_mirror_descent(a_init, S, max_time, rng,
                       beta_start=5.0, beta_end=200.0,
                       print_interval=5000):
    """
    Multiplicative (mirror) descent with LSE gradient.
    Naturally maintains positivity: a_j *= exp(-eta * grad_j).
    """
    n = len(a_init)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()

    a = normalize_to_sum(a_init.copy(), S)
    start = time.monotonic()

    c0, _ = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress
        beta = min(beta, 50000.0)

        c_a, _ = fft_convolve_self(a)
        cur_max = c_a.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        grad = compute_lse_gradient(a, c_a, L, beta)

        # Multiplicative update with small step
        eta = 1.0 / (beta * max(a.max(), 1e-10) + 1e-10) * 0.5
        # Clip the exponent to avoid underflow/overflow
        exponent = np.clip(-eta * grad, -10.0, 10.0)
        multiplier = np.exp(exponent)
        a_new = a * multiplier
        a = normalize_to_sum(a_new, S)

        if no_improve > 1000:
            noise_scale = 0.002 * S / np.sqrt(n) * (1.0 - progress) + 1e-6
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            no_improve = 0

        if it % print_interval == 0:
            obj = 2.0 * n * best / (S * S)
            print("    MD it=%d, beta=%.1f, best_max=%.8f, obj=%.6f" %
                  (it, beta, best, obj))

    c_final, _ = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_polish(a_init, S, max_time, rng,
               beta_start=200.0, beta_end=5000.0,
               print_interval=3000):
    """
    High-beta polishing: very small steps with high beta to squeeze
    out remaining improvement in the max convolution value.
    """
    n = len(a_init)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()

    a = normalize_to_sum(a_init.copy(), S)
    start = time.monotonic()

    c0, _ = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress
        beta = min(beta, 1000000.0)

        c_a, _ = fft_convolve_self(a)
        cur_max = c_a.max()

        if cur_max < best - 1e-13:
            best = cur_max
            best_a = a.copy()

        grad = compute_lse_gradient(a, c_a, L, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Very small step for polishing
        base_step = 0.3 / (beta * max(a.max(), 1e-10) + 1e-10)
        step = base_step * step_scale

        a_new = project_simplex(a - step * grad, S)
        c_new, _ = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
            step_scale = min(3.0, step_scale * 1.01)
        else:
            step_scale = max(0.01, step_scale * 0.5)
            a = a_new  # Still move, but smaller step next time

        # Occasional tiny perturbation
        if it % 3000 == 0 and progress < 0.8:
            noise_scale = 1e-4 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

        if it % print_interval == 0:
            obj = 2.0 * n * best / (S * S)
            print("    Polish it=%d, beta=%.0f, best_max=%.8f, obj=%.6f" %
                  (it, beta, best, obj))

    c_final, _ = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length using linear interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain for diversity."""
    n = len(a)
    S = a.sum()
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = fft(a, L)
    phase = rng.normal(0, amount, L)
    phase[0] = 0
    A_new = A * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, L))[:n]
    return normalize_to_sum(a_new, S)


def spectral_flatten(a, alpha, rng):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    S = a.sum()
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()

    A = fft(a, L)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a

    ratio = power / max(mean_p, 1e-20)
    new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + ratio + 1e-10))
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / (power + 1e-30))
    a_new = np.real(ifft(A_new, L))[:n]
    return normalize_to_sum(a_new, S)


def run_spectral_iteration(a_init, S, max_time, rng,
                           flatten_alpha=0.1, perturb_amount=0.03,
                           print_interval=200):
    """
    Iterate: spectral flatten -> perturbation -> evaluate.
    Exploits the connection between flat spectrum and flat autocorrelation.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init.copy(), S)
    start = time.monotonic()

    c0, _ = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Flatten spectrum
        a = spectral_flatten(a, flatten_alpha, rng)

        # Small perturbation
        if it % 10 == 0 and rng.random() < 0.3:
            a = spectral_perturb(a, perturb_amount, rng)

        c_a, _ = fft_convolve_self(a)
        cur = c_a.max()
        if cur < best - 1e-12:
            best = cur
            best_a = a.copy()

        if it % print_interval == 0:
            obj = 2.0 * n * best / (S * S)
            print("    Spectral it=%d, best_max=%.8f, obj=%.6f" % (it, best, obj))

    return best_a, best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: FISTA+LSE -> Mirror Descent -> Polish -> Spectral.
    Tries n=8192 primarily, n=16384 if time permits.
    """
    rng = np.random.default_rng(seed)
    start_total = time.monotonic()
    deadline = start_total + max(0.0, budget_s - 30.0)

    print("Starting optimization, budget=%.0fs, deadline in %.0fs" %
          (budget_s, deadline - start_total))

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given = np.clip(given, 0, 1000.0)
            if given.sum() > 0.01:
                print("Using given sequence of length %d, sum=%.2f" % (len(given), given.sum()))
            else:
                given = None
    except Exception:
        given = None

    global_best = np.inf
    global_a = None

    def try_optimization(n, a_start, time_limit, label):
        """Run pipeline for given (n, a_start)."""
        nonlocal global_best, global_a
        S = np.sqrt(2.0 * n)

        if a_start is not None and len(a_start) != n:
            a_start = resample_sequence(a_start, n)
        if a_start is not None:
            a_start = normalize_to_sum(a_start, S)
        else:
            # Random initialization with some structure
            a_base = np.ones(n)
            # Add a cosine window to break uniformity
            t = np.linspace(0, 1, n)
            a_base *= (0.5 + 0.5 * np.cos(2 * np.pi * t))
            a_base += 0.1
            a_start = normalize_to_sum(a_base, S)

        print("\n=== %s: n=%d, time=%.0fs ===" % (label, n, time_limit))
        c0, _ = fft_convolve_self(a_start)
        obj_start = 2.0 * n * c0.max() / (S * S)
        print("  Initial: max_conv=%.8f, obj=%.6f" % (c0.max(), obj_start))

        best_local = c0.max()
        best_a_local = a_start.copy()

        # Phase 1: FISTA with full LSE (45% of time)
        t1 = time_limit * 0.45
        if t1 > 30:
            print("  Phase 1: FISTA+LSE (%.0fs)" % t1)
            a_opt, val_opt = run_fista_lse(
                best_a_local, S, t1, rng,
                beta_start=2.0 / max(best_local, 0.1),
                beta_end=300.0 / max(best_local, 0.1))
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * best_local / (S * S)
                print("  After FISTA: max_conv=%.8f, obj=%.6f" % (best_local, obj))
                if obj < global_best:
                    global_best = obj
                    global_a = best_a_local.copy()
                    print("  *** New global best: %.6f ***" % global_best)

        # Phase 2: Mirror descent (20% of time)
        t2 = time_limit * 0.20
        remaining = deadline - time.monotonic()
        if t2 > 20 and remaining > 60:
            t2 = min(t2, remaining - 60)
            print("  Phase 2: Mirror Descent (%.0fs)" % t2)
            a_opt, val_opt = run_mirror_descent(
                best_a_local, S, t2, rng,
                beta_start=5.0, beta_end=200.0)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * best_local / (S * S)
                print("  After MD: max_conv=%.8f, obj=%.6f" % (best_local, obj))
                if obj < global_best:
                    global_best = obj
                    global_a = best_a_local.copy()
                    print("  *** New global best: %.6f ***" % global_best)

        # Phase 3: High-beta polish (20% of time)
        t3 = time_limit * 0.20
        remaining = deadline - time.monotonic()
        if t3 > 15 and remaining > 40:
            t3 = min(t3, remaining - 40)
            print("  Phase 3: Polish (%.0fs)" % t3)
            a_opt, val_opt = run_polish(
                best_a_local, S, t3, rng,
                beta_start=200.0, beta_end=5000.0)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * best_local / (S * S)
                print("  After Polish: max_conv=%.8f, obj=%.6f" % (best_local, obj))
                if obj < global_best:
                    global_best = obj
                    global_a = best_a_local.copy()
                    print("  *** New global best: %.6f ***" % global_best)

        # Phase 4: Spectral iteration (15% of time)
        t4 = time_limit * 0.15
        remaining = deadline - time.monotonic()
        if t4 > 10 and remaining > 30:
            t4 = min(t4, remaining - 30)
            print("  Phase 4: Spectral (%.0fs)" % t4)
            a_opt, val_opt = run_spectral_iteration(
                best_a_local, S, t4, rng,
                flatten_alpha=0.05, perturb_amount=0.02)
            if val_opt < best_local:
                best_local = val_opt
                best_a_local = a_opt.copy()
                obj = 2.0 * n * best_local / (S * S)
                print("  After Spectral: max_conv=%.8f, obj=%.6f" % (best_local, obj))
                if obj < global_best:
                    global_best = obj
                    global_a = best_a_local.copy()
                    print("  *** New global best: %.6f ***" % global_best)

        obj_final = 2.0 * n * best_local / (best_a_local.sum() ** 2)
        print("  Best for %s (n=%d): obj=%.6f" % (label, n, obj_final))

        if obj_final < global_best:
            global_best = obj_final
            global_a = best_a_local.copy()
            print("  *** Updated global best: %.6f ***" % global_best)

        return best_a_local, best_local

    # Main optimization: n = 8192
    n_primary = 8192
    remaining = deadline - time.monotonic()
    if remaining > 120:
        try_optimization(n_primary, given, remaining * 0.60, "Primary")

    # Second try: n = 16384 (resample best to longer)
    remaining = deadline - time.monotonic()
    if remaining > 200 and global_a is not None:
        n_secondary = 16384
        a_sec = resample_sequence(global_a, n_secondary) if global_a is not None else None
        try_optimization(n_secondary, a_sec, remaining * 0.35, "Secondary")

    # Final intensive polish on the best found
    remaining = deadline - time.monotonic() - 10
    if remaining > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final Intensive Polish (%.0fs) ===" % remaining)

        # Very high beta FISTA without momentum
        t_fin1 = remaining * 0.4
        a_fin, val_fin = run_fista_lse(
            global_a, S_best, t_fin1, rng,
            beta_start=500.0, beta_end=10000.0,
            momentum=False)
        obj_fin = 2.0 * n_best * val_fin / (a_fin.sum() ** 2)
        if obj_fin < global_best:
            global_best = obj_fin
            global_a = a_fin.copy()
            print("  Final FISTA: obj=%.6f" % global_best)

        # Polish
        remaining2 = deadline - time.monotonic() - 10
        if remaining2 > 20:
            a_pol, val_pol = run_polish(
                global_a, S_best, remaining2, rng,
                beta_start=1000.0, beta_end=50000.0)
            obj_pol = 2.0 * n_best * val_pol / (a_pol.sum() ** 2)
            if obj_pol < global_best:
                global_best = obj_pol
                global_a = a_pol.copy()
                print("  Final Polish: obj=%.6f" % global_best)

    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
            final_list = [float(x) for x in global_a]
            ver = evaluate_sequence(final_list)
            print("  Fallback verified: %.6f" % ver)
    else:
        print("WARNING: No valid solution found, using uniform fallback.")
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback verified: %.6f" % ver)

    print("\nFinal: length=%d, objective=%.6f, time=%.0fs" %
          (len(global_a), global_best, time.monotonic() - start_total))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
