import numpy as np
from scipy.fft import fft, ifft, next_fast_len
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = next_fast_len(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection."""
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64)
    lo = -1e6
    hi = v.max() + 1e6
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > S:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    result = np.maximum(v - theta, 0)
    # Ensure exact sum
    diff = S - result.sum()
    if diff != 0:
        result += diff / n
        result = np.maximum(result, 0)
    return result


def compute_topk_gradient(a, c, K):
    """Compute weighted subgradient from top-K convolution lags."""
    n = len(a)
    K = min(K, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:K]
    c_top = c[top_idx]
    c_med = np.median(c)
    weights = np.maximum(c_top - c_med, 1e-10)
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for ii in range(K):
        kidx = int(top_idx[ii])
        wi = 2.0 * weights[ii]
        j_lo = max(0, kidx - n + 1)
        j_hi = min(n - 1, kidx)
        if j_lo <= j_hi:
            idx = kidx - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi+1] += wi * a[idx]
    return grad


def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    c_shifted = c - c.max()
    scaled = beta * c_shifted
    max_scaled = scaled.max()
    if max_scaled > 350:
        scaled = scaled * (350.0 / max_scaled)
        beta_eff = beta * (350.0 / max_scaled)
    else:
        beta_eff = beta
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = next_fast_len(len(w) + n - 1)
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * A))[:n]
    return grad

def try_lp_direction(a, c, M, n):
    """
    Try to find a good direction using LP.
    Returns the direction vector g or None if LP fails.
    """
    from scipy.optimize import linprog
    
    # LP: maximize sum(g) subject to conv(a, g)[k] <= M, g >= 0
    # conv(a, g)[k] = sum_i a[i] * g[k-i] for valid indices
    
    # Build constraint matrix (sparse)
    # For each lag k (0 to 2n-2):
    #   sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * g[k-i] <= M
    
    # This is a large LP. For n > 4096, skip it.
    if n > 4096:
        return None
    
    # Build A matrix (2n-1 rows, n cols)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A_ub[k, j] = a[k - j]
    
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.ones(2*n-1) * M
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                        method='highs', options={'time_limit': 30})
        if result.success:
            return -result.x  # We minimized -sum(g), so direction is -x... wait
            # Actually result.x is g that maximizes sum(g). We want to move toward g.
            return result.x
    except Exception:
        pass
    return None

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = next_fast_len(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection."""
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.maximum(v, 0)
    lo = 0.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    result = np.maximum(v - theta, 0.0)
    diff = S - result.sum()
    if abs(diff) > 1e-12 * S:
        mask = result > 0
        if mask.any():
            result[mask] += diff / mask.sum()
            result = np.maximum(result, 0)
    return result


def obj_internal(a):

def compute_topk_gradient(a, c, K):

def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)

    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = next_fast_len(len(w) + n - 1)
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * A))[:n]

def try_lp_direction(a, c, M, max_time_lp=15.0):
    """
    Try LP to find direction g maximizing sum(g) s.t. conv(a,g)[k] <= M, g >= 0.
    Returns g or None.
    """
    from scipy.optimize import linprog
    n = len(a)
    if n > 3000:
        return None

    # Build constraint matrix
    n_rows = 2 * n - 1
    A_ub = np.zeros((n_rows, n))
    for k in range(n_rows):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A_ub[k, j] = a[k - j]

    c_obj = -np.ones(n)
    b_ub = np.ones(n_rows) * M

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': max_time_lp})
        if result.success:
            return result.x
    except Exception:
        pass
    return None

            a_curr = peak_shave(a_curr, S, 200, rng)
            update_global(a_curr, f"Phase 4 shave {ri}")

    # === Phase 5: Diverse perturbations ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 50:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        n_pert = max(2, min(5, int((remaining - 20) / 60)))
        t_pert = (remaining - 20) / n_pert
        print(f"\nPhase 5: {n_pert} perturbations, ~{t_pert:.0f}s each")

        for pi in range(n_pert):
            if time.monotonic() > deadline - 15:
                break
            if pi % 3 == 0:
                noise = 0.02 * S / np.sqrt(n)
                a_pert = project_simplex(global_a + rng.normal(0, noise, n), S)
            elif pi % 3 == 1:
                a_pert = global_a.copy()
                n_ch = max(10, n // 50)
                idx = rng.choice(n, n_ch, replace=False)
                a_pert[idx] *= rng.uniform(0.3, 3.0, n_ch)
                a_pert = project_simplex(a_pert, S)
            else:
                # Frequency perturbation
                A = fft(global_a, n)
                A *= (1.0 + 0.15 * rng.standard_normal(n) + 0.1j * rng.standard_normal(n))
                a_pert = np.real(ifft(A))
                a_pert = np.maximum(a_pert, 0)
                a_pert = project_simplex(a_pert, S)

            a_pert, _ = optimize_adaptive(a_pert, S, min(t_pert, deadline - time.monotonic() - 10),
                                          rng, use_annealing=True)
            update_global(a_pert, f"Phase 5 pert {pi}")

    # === Final: try a few more lengths ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        # Try length 4096 and 16384
        for n_try in [4096, 16384]:
            if time.monotonic() > deadline - 20:
                break
            if global_a is not None and len(global_a) == n_try:
                continue
            S_try = np.sqrt(2.0 * n_try)
            if global_a is not None:
                t_old = np.linspace(0, 1, len(global_a))
                t_new = np.linspace(0, 1, n_try)
                a_try = np.interp(t_new, t_old, global_a)
            else:
                t_new = np.linspace(0, 1, n_try)
                a_try = 0.5 * (1.0 + np.cos(2.0 * np.pi * t_new)) + 0.05
            a_try = project_simplex(a_try, S_try)
            t_alloc = min(80, deadline - time.monotonic() - 15)
            if t_alloc > 20:
                print(f"\nFinal: Try n={n_try}, {t_alloc:.0f}s")
                a_try, _ = optimize_adaptive(a_try, S_try, t_alloc, rng)
                update_global(a_try, f"Final n={n_try}")

    # === Return best ===
    if global_a is None:
        print("WARNING: No valid solution, using fallback")
        global_a = np.ones(256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print(f"\n{'=' * 60}")
    print(f"FINAL: len={len(global_a)}, internal_obj={global_best:.6f}, eval={ver:.6f}")
    print(f"{'=' * 60}")

    return final_list
