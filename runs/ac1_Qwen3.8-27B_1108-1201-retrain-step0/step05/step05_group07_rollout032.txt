def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    if v.sum() <= S:
        # Already below target, scale up
        # Actually we need sum = S exactly
        pass
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_topk_grad(a, c, k):
    """Compute subgradient from top-k convolution lags."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    # Weight by how much above median
    c_median = np.median(c)
    raw_w = np.maximum(c[top_idx] - c_median, 1e-10)
    w = raw_w / raw_w.sum()
    grad = np.zeros(n)
    for ii in range(k):
        kidx = int(top_idx[ii])
        wi = w[ii]
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
    return grad

def compute_lse_grad(a, c, beta):
    """Gradient of LSE_smoothed_max."""
    n = len(a)
    cmax = c.max()
    scaled = beta * (c - cmax)
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def objective(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def adam_subgradient(a0, S, max_time, rng, base_lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam-style subgradient descent with simplex projection."""
    n = len(a0)
    a = a0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_c = fft_convolve_self(a)
    best_val = best_c.max()
    start = time.monotonic()
    stuck = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_val - 1e-12:
            best_val = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Adaptive k
        k = int(4 + 60 * progress**0.4)
        
        grad = compute_topk_grad(a, c, k)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Scale learning rate
        lr = base_lr / (1.0 + 0.1 * progress)  # decay lr over time
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a_try = a - step
        a_try = project_simplex(a_try, S)
        
        c_try = fft_convolve_self(a_try)
        
        if c_try.max() >= cur_max * (1 + 1e-10) and stuck < 100:
            # Reject, try smaller step
            a_try = a - 0.5 * step
            a_try = project_simplex(a_try, S)
            c_try = fft_convolve_self(a_try)
        
        a = a_try
        if c_try.max() < cur_max:
            pass  # improvement
        
        # Noise injection when stuck
        if stuck > 200:
            noise_scale = min(0.05, 0.005 * (stuck / 200)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
    
    return best_a, best_val

def coordinate_refine(a0, S, max_time, rng, m=200):
    """Coordinate descent refinement."""
    n = len(a0)
    a = a0.copy()
    c = fft_convolve_self(a)
    best_val = c.max()
    best_a = a.copy()
    start = time.monotonic()
    iterations = 0
    
    while time.monotonic() - start < max_time:
        iterations += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_val - 1e-14:
            best_val = cur_max
            best_a = a.copy()
        
        # Find top-k lags
        k = min(16, 2*n-1)
        top_idx = np.argsort(c)[::-1][:k]
        
        # Compute gradient to identify important coordinates
        grad = compute_topk_grad(a, c, k)
        
        # Pick top-m coordinates by |grad|
        m_actual = min(m, n)
        coord_idx = np.argsort(np.abs(grad))[::-1][:m_actual]
        
        improved = False
        for j in coord_idx:
            if time.monotonic() - start > max_time - 5:
                break
            aj = a[j]
            if aj < 1e-10:
                # Try making it positive
                delta = S / n * 0.01
            else:
                delta = aj * 0.1
            
            # Try positive and negative delta
            for d in [delta, -delta, 2*delta, -2*delta, delta*0.5, -delta*0.5]:
                new_val = aj + d
                if new_val < 0:
                    continue
                a[j] = new_val
                # Adjust to maintain sum (simple: just check objective)
                # Actually, we should project to simplex
                c_try = fft_convolve_self(a)
                a[j] = aj  # restore
                break  # Just one trial per direction for now
        
        # Simpler approach: for each important coordinate, try a few values
        for j in coord_idx[:50]:
            if time.monotonic() - start > max_time - 5:
                break
            aj = a[j]
            deltas = [0.05, -0.05, 0.1, -0.1, 0.02, -0.02] * max(aj, S/n * 0.1)
            best_delta = 0
            best_obj = cur_max
            for d in deltas:
                new_val = aj + d
                if new_val < 0:
                    continue
                a[j] = new_val
                # Quick check: recompute only the affected conv values
                # For simplicity, just check the top-k lags
                # c_k change = 2*d*a[k-j] + d^2 (if k=2j)
                max_change = 0
                for kidx in top_idx:
                    kidx = int(kidx)
                    if 0 <= kidx - j < n:
                        change = 2 * d * a[kidx - j] + d * d if kidx == 2 * j else 2 * d * a[kidx - j]
                        # Approximate new value
                        new_c = c[kidx] + change
                        if new_c > max_change:
                            max_change = new_c
                if max_change < best_obj:
                    best_obj = max_change
                    best_delta = d
            a[j] = aj + best_delta
        
        if not improved and iterations % 50 == 0:
            # Small noise
            noise_scale = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best_val

def lse_polish(a0, S, max_time, rng, beta_start=50, beta_end=5000):
    """LSE smoothing polishing with adaptive beta."""
    n = len(a0)
    a = a0.copy()
    best_val = fft_convolve_self(a).max()
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_val - 1e-14:
            best_val = cur_max
            best_a = a.copy()
        
        grad = compute_lse_grad(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (1.5 * beta)
        step = min(step, 2.0 / gnorm)
        step *= (1.0 - 0.5 * progress)  # decay step size
        
        a_try = a - step * grad
        a_try = project_simplex(a_try, S)
        c_try = fft_convolve_self(a_try)
        
        if c_try.max() <= cur_max * (1 + 1e-10):
            a = a_try
        else:
            a_try = a - 0.5 * step * grad
            a_try = project_simplex(a_try, S)
            a = a_try
    
    return best_a, best_val

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimize 2n*max(conv(a,a))/sum(a)^2 using adaptive subgradient (Adam),
    coordinate refinement, and LSE polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    print(f"Starting optimization, budget={budget_s}s")
    
    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 128:
            print(f"Using provided sequence: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        pass
    
    if a_provided is None:
        n = 8192
        a_provided = np.ones(n)
        print(f"Using default uniform sequence: len={n}")
    
    n = len(a_provided)
    S = np.sqrt(2.0 * n)
    a_init = a_provided * (S / a_provided.sum())
    a_init = project_simplex(a_init, S)
    
    global_best = np.inf
    global_a = None
    
    def update_global(a, msg=""):
        nonlocal global_best, global_a
        S_a = a.sum()
        if S_a < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S_a * S_a)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  NEW BEST: obj={obj:.8f} ({msg})")
    
    update_global(a_init, "init")
    
    remaining = deadline - time.monotonic()
    
    # Phase 1: Adam subgradient (45% of time)
    t1 = remaining * 0.45
    print(f"\nPhase 1: Adam subgradient, n={n}, time={t1:.0f}s")
    a_opt, _ = adam_subgradient(a_init, S, t1, rng, base_lr=0.05)
    update_global(a_opt, "Phase 1 Adam")
    
    # Phase 2: LSE polish (25% of time)
    remaining = deadline - time.monotonic()
    t2 = remaining * 0.35
    print(f"\nPhase 2: LSE polishing, time={t2:.0f}s")
    a_opt, _ = lse_polish(a_opt, S, t2, rng, beta_start=50, beta_end=5000)
    update_global(a_opt, "Phase 2 LSE")
    
    # Phase 3: More Adam (15% of time)
