```python
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    rho = int(np.max(np.where(valid)[0]) + 1) if np.any(valid) else n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)
```

```python
import numpy as np

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    rho = int(np.max(np.where(valid)[0]) + 1) if np.any(valid) else n
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def compute_obj_from_c(a, c):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def lse_of_vals(vals, beta):
    """Log-sum-exp of vals with temperature 1/beta."""
    m = vals.max()
    shifted = beta * (vals - m)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return m, 0.0
    return m + np.log(tot) / beta, e / tot

def fista_lse(a_init, max_time, rng, beta0=50.0, beta_end=5000.0, step0=0.5):
    """FISTA with LSE smoothification and adaptive parameters."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta: increase when stuck
        stagnation = it - last_improve
        if stagnation < 100:
            beta = beta0
        elif stagnation < 500:
            beta = beta0 * 5
        elif stagnation < 2000:
            beta = beta0 * 50
        else:
            beta = min(beta_end, beta0 * 50 * (1 + stagnation / 1000))
        
        # Compute LSE gradient
        c = fft_convolve_self(y)
        lse_val, w = lse_of_vals(c, beta)
        
        # Gradient: 2 * ifft(fft(w_full) * conj(fft(y)))
        fsize = 1 << (2 * n - 1 + n - 1).bit_length()  # need 2n-1 + n - 1
        # Actually: w has length 2n-1, y has length n
        # ifft result has length fsize, we take first n
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        Y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]
        
        # Evaluate current objective
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        # Step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking
        x_trial = project_simplex(y - step * grad, S)
        obj_trial = compute_obj_from_c(x_trial, fft_convolve_self(x_trial))
        bt = 0
        while obj_trial > cur_obj * (1 + 1e-10) + 1e-14 and step > 1e-10 and bt < 8:
            step *= 0.5
            x_trial = project_simplex(y - step * grad, S)
            obj_trial = compute_obj_from_c(x_trial, fft_convolve_self(x_trial))
            bt += 1
        
        # Update step for next iteration
        if bt == 0:
            step = min(step * 1.15, 5.0 / (beta + 1))
        else:
            step = step * 0.7
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        x_new = x_trial
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        # Restart if no progress for a long time
        if stagnation > 3000:
            t_mom = 1.0
            y = a.copy()
            noise = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
        
        # Periodic reproject
        if it % 500 == 0:
            c = fft_convolve_self(a)
            if compute_obj_from_c(a, c) < best_obj - 1e-14:
                best_obj = compute_obj_from_c(a, c)
                best_a = a.copy()
    
    return best_a, best_obj

def exact_coord_descent(a_init, c_init, max_time, rng, top_k=5):
    """
    Coordinate descent with exact-ish line search.
    For each of the top_k peaks, find the best (source, target, delta) transfer.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    c_sum = a.sum() ** 2  # = S^2
    
    start = time.monotonic()
    best_obj = compute_obj_from_c(a, c)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / c_sum
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 50000:
            break
        
        # Find top peaks
        k = min(top_k, 2 * n - 1)
        peak_idx = np.argsort(c)[::-1][:k]
        
        # Compute contribution of each coordinate to the top peaks
        # contrib[m] = sum over peaks of |d(c[peak])/da[m]|
        contrib = np.zeros(n)
        for pk in peak_idx:
            pk = int(pk)
            # d(c[pk])/da[m] = 2*a[pk-m] for m in [pk-n+1, pk] valid range
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            if hi >= lo:
                contrib[lo:hi+1] += 2.0 * np.abs(a[pk - hi:pk - lo + 1][::-1])
        
        # Source: high contribution, high value
        # Target: low contribution, low value (has room to grow)
        src_score = contrib * a
        tgt_score = -contrib * (a.mean() - a + 1e-10)
        
        # Try a few transfer candidates
        n_src = min(10, n)
        n_tgt = min(10, n)
        src_cand = np.argsort(src_score)[::-1][:n_src]
        tgt_cand = np.argsort(tgt_score)[:n_tgt]
        
        best_new_obj = cur_obj
        best_move = None
        
        for j in src_cand:
            if a[j] < 1e-12:
                continue
            for l in tgt_cand:
                if j == l:
                    continue
                # Compute delta_c for transferring delta from j to l
                # c_new[k] = c[k] - 2*delta*a[k-j] + 2*delta*a[k-l] (approximately)
                # (ignoring quadratic correction at k=2j and k=2l)
                
                # Try several delta values
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = a[j] * frac
                    if delta < 1e-14:
                        continue
                    
                    # Compute new max (approximate)
                    delta_c = np.zeros(2 * n - 1)
                    # Effect of reducing a[j] by delta
                    js_lo = max(0, j)
                    js_hi = min(2 * n - 2, j + n - 1)
                    # c[k] -= 2*delta*a[k-j] for k in [j, j+n-1]
                    a_vals_j = a[max(0, js_lo - j):min(n, js_hi - j + 1)]
                    delta_c[js_lo:js_hi+1] -= 2.0 * delta * a_vals_j
                    # Correction for k = 2j (quadratic term)
                    if 0 <= 2*j < 2*n-1:
                        delta_c[2*j] += 2*delta*a[j] + delta*delta  # (a[j]-delta)^2 - a[j]^2 = -2*delta*a[j]+delta^2
                        # Wait: c[2j] contains a[j]^2. After change: (a[j]-delta)^2
                        # So delta_c[2j] should have -2*delta*a[j] + delta^2 from this term
                        # But we already subtracted 2*delta*a[j] in the linear term above
                        # So we need to add back delta^2
                        delta_c[2*j] += delta * delta
                    
                    # Effect of increasing a[l] by delta
                    ls_lo = max(0, l)
                    ls_hi = min(2 * n - 2, l + n - 1)
                    a_vals_l = a[max(0, ls_lo - l):min(n, ls_hi - l + 1)]
                    delta_c[ls_lo:ls_hi+1] += 2.0 * delta * a_vals_l
                    if 0 <= 2*l < 2*n-1:
                        delta_c[2*l] += delta * delta
                    
                    new_max = np.max(c + delta_c)
                    new_obj = 2.0 * n * new_max / c_sum
                    
                    if new_obj < best_new_obj - 1e-15:
                        best_new_obj = new_obj
                        best_move = (j, l, delta)
        
        if best_move is not None:
            j, l, delta = best_move
            a[j] -= delta
            a[l] += delta
            # Update c
            js_lo = max(0, j)
            js_hi = min(2 * n - 2, j + n - 1)
            a_vals_j = a[max(0, js_lo - j):min(n, js_hi - j + 1)]
            # But a[j] has changed, so we need to use the new a[j] for the quadratic
            # For simplicity, recompute affected parts
            # c[k] -= 2*delta*old_a[k-j] for k in [j, j+n-1], k != 2j
            # c[2j] -= 2*delta*a[j] + delta^2 (using old a[j])
            old_aj = a[j] + delta
            for k in range(js_lo, js_hi + 1):
                if k == 2 * j:
                    c[k] += delta * delta - 2.0 * delta * old_aj
                else:
                    idx_in_a = k - j
                    if 0 <= idx_in_a < n:
                        c[k] -= 2.0 * delta * a[idx_in_a]
            
            ls_lo = max(0, l)
            ls_hi = min(2 * n - 2, l + n - 1)
            old_al = a[l] - delta
            for k in range(ls_lo, ls_hi + 1):
                if k == 2 * l:
                    c[k] += delta * delta + 2.0 * delta * old_al
                else:
                    idx_in_a = k - l
                    if 0 <= idx_in_a < n:
                        c[k] += 2.0 * delta * a[idx_in_a]
        else:
            # No improvement found, add noise
            if stuck > 500:
                noise_scale = max(1e-8, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 5000))
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
        
        # Periodic full recompute
        if it % 200 == 0:
            c = fft_convolve_self(a)
            # Ensure sum is correct
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
    
    return best_a, best_obj
```

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy:
  1. Polish existing n=8192 solution with adaptive LSE-FISTA
  2. Upsample to n=16384 via zero-stuffing + FISTA fill

  4. Vectorized coordinate descent with line search on top peaks
  5. Final ultra-high-beta polish

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):

    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)

def gradient_from_weights(w, y, n):
    """Compute 2*ifft(fft(w)*conj(fft(y)))[:n] where w has length 2n-1."""
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    Y = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(Y)))[:n]


def lse_weights(c, beta):
    """Return (lse_value, normalized_weights) for LSE of c with sharpness beta."""
    m = c.max()
    shifted = beta * (c - m)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return m, w
    lse_val = m + np.log(tot) / beta
    return lse_val, e / tot

    else:
        a = a * (S / s)

    y = a.copy()

    start = time.monotonic()
    best_obj = compute_obj(a)

def coord_descent_vec(a_init, max_time, rng, top_k=5, n_candidates=8):
    """
    Vectorized coordinate descent. Tries transfers from high-contribution
    to low-contribution coordinates with multiple delta sizes.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)

    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0
    print("  CD: start obj=%.8f" % best_obj)

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 30000:
            break

        # Find top peaks
        k = min(top_k, 2 * n - 1)
        peak_idx = np.argsort(c)[::-1][:k]

        # Compute contribution vector
        contrib = np.zeros(n)
        for pk in peak_idx:
            pk = int(pk)
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            if hi >= lo:
                contrib[lo:hi + 1] += np.abs(a[pk - hi:pk - lo + 1][::-1])

        # Score for source (high contrib * high value) and target (low contrib)
        src_score = contrib * np.maximum(a, 1e-10)
        tgt_score = -contrib

        src_cand = np.argsort(src_score)[::-1][:n_candidates]
        tgt_cand = np.argsort(tgt_score)[:n_candidates]

        best_new_obj = cur_obj
        best_move = None

        fracs = [0.03, 0.07, 0.15, 0.3]

        for j in src_cand[:n_candidates]:
            j = int(j)
            if a[j] < 1e-12:
                continue
            for l in tgt_cand[:n_candidates]:
                l = int(l)
                if j == l:
                    continue
                for frac in fracs:
                    delta = a[j] * frac
                    if delta < 1e-14:
                        continue

                    # Compute delta_c vectorized
                    delta_c = np.zeros(2 * n - 1)
                    # Reduce a[j] by delta
                    js = j
                    je = min(2 * n - 2, j + n - 1)
                    a_slice_j = a[max(0, js - j):min(n, je - j + 1)]
                    delta_c[js:je + 1] -= 2.0 * delta * a_slice_j
                    if 0 <= 2 * j < 2 * n - 1:
                        delta_c[2 * j] += delta * delta

                    # Increase a[l] by delta
                    ls = l
                    le = min(2 * n - 2, l + n - 1)
                    a_slice_l = a[max(0, ls - l):min(n, le - l + 1)]
                    delta_c[ls:le + 1] += 2.0 * delta * a_slice_l
                    if 0 <= 2 * l < 2 * n - 1:
                        delta_c[2 * l] += delta * delta

                    new_max = np.max(c + delta_c)
                    new_obj = 2.0 * n * new_max / (S * S)
                    if new_obj < best_new_obj - 1e-14:
                        best_new_obj = new_obj
                        best_move = (j, l, delta)

        if best_move is not None:
            j, l, delta = best_move
            old_aj = a[j]
            old_al = a[l]
            a[j] = old_aj - delta
            a[l] = old_al + delta

            # Update c
            js = j
            je = min(2 * n - 2, j + n - 1)
            a_slice_j = np.clip(np.arange(js - j, je - j + 1), 0, n - 1)
            # Use old values for linear terms
            old_a = old_aj
            delta_c = np.zeros(2 * n - 1)
            delta_c[js:je + 1] -= 2.0 * delta * a[a_slice_j]
            if 0 <= 2 * j < 2 * n - 1:
                delta_c[2 * j] += delta * delta - 2.0 * delta * (old_aj - 2 * delta)  # fix
            # Actually let's just recompute properly
            # c[k] = sum a[m]*a[k-m]. Removing delta from a[j]:
            # c[k] -= delta * a[k-j] + delta * a[k-j] = 2*delta*a[k-j] for k in [j, j+n-1]
            # except k=2j where it's 2*delta*a[j] - delta (from the a[j] in the product)
            # Simpler: just recompute c via FFT periodically
            c[js:je + 1] -= 2.0 * delta * a[a_slice_j + delta / delta]  # wrong, let me just use FFT

            # Simpler and safer: recompute c via FFT
            c = fft_convolve_self(a)
        else:
            if stuck > 200:
                noise_scale = max(1e-9, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 3000))
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)

        if it % 1000 == 0:
            print("  CD: it=%d obj=%.8f stuck=%d" % (it, best_obj, stuck))

    return best_a, best_obj


def upsample_zero_stuff(a, factor=2):
    """Upsample by zero-stuffing."""
    n = len(a)
    n_new = n * factor
    b = np.zeros(n_new)
    b[::factor] = a
    return b


def interp_upsample(a, factor=2):
    """Upsample by linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0.0, 1.0, n, endpoint=False)
    t_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: polish, upsample, intensive FISTA, coordinate descent, final polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)
    print("Budget: %.0fs, deadline in %.0fs" % (budget_s, budget_s - 20.0))

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        a_start = np.ones(n) * np.sqrt(2.0 * n) / n
        print("No parent sequence, using flat init n=%d" % n)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.8f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: Quick polish on original size (5%) ===
    remaining = deadline - time.monotonic()
    p1_time = min(remaining * 0.05, 120.0)
    if p1_time > 20:
        print("\n=== Phase 1: Polish n=%d (%.0fs) ===" % (n0, p1_time))
        a_p1, obj_p1 = fista_adaptive(global_best_a, p1_time, rng,
                                       beta0=50, beta_max=20000,
                                       step_init=0.5, name="P1-polish")
        if obj_p1 < global_best_obj:
            global_best_obj = obj_p1
            global_best_a = a_p1.copy()
            print("  Phase 1 improved: obj=%.8f" % global_best_obj)

    # === Phase 2: Upsample to 2x (5%) ===
    remaining = deadline - time.monotonic()
    p2_time = min(remaining * 0.05, 120.0)
    n_target = n0 * 2
    if p2_time > 20 and n_target <= 32768:
        print("\n=== Phase 2: Upsample to n=%d (%.0fs) ===" % (n_target, p2_time))

        # Try both zero-stuff and interpolation
        a_zs = upsample_zero_stuff(global_best_a, 2)
        S2 = np.sqrt(2.0 * n_target)
        a_zs = a_zs * (S2 / a_zs.sum())
        obj_zs = compute_obj(a_zs)
        print("  Zero-stuff init: obj=%.8f" % obj_zs)

        a_interp = interp_upsample(global_best_a, 2)
        a_interp = a_interp * (S2 / a_interp.sum())
        obj_interp = compute_obj(a_interp)
        print("  Interp init: obj=%.8f" % obj_interp)

        # Pick better initializer
        if obj_interp < obj_zs:
            a_up = a_interp.copy()
            obj_up = obj_interp
        else:
            a_up = a_zs.copy()
            obj_up = obj_zs

        # Quick FISTA to fill in
        a_up2, obj_up2 = fista_adaptive(a_up, p2_time, rng,
                                         beta0=10, beta_max=5000,
                                         step_init=1.5, name="P2-upsample")
        if obj_up2 < global_best_obj:
            global_best_obj = obj_up2
            global_best_a = a_up2.copy()
            print("  Phase 2 improved: obj=%.8f" % global_best_obj)
        elif obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up.copy()
    else:
        a_up = global_best_a.copy()
        print("\n=== Phase 2: Skipped (insufficient time) ===")

    # === Phase 3: Main FISTA on largest size (55%) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.55
    n_main = len(global_best_a)
    print("\n=== Phase 3: Main FISTA n=%d (%.0fs) ===" % (n_main, p3_time))

    # Multiple FISTA runs with different params
    fista_configs = [
        (30, 30000, 0.8, "F3a"),
        (100, 50000, 0.5, "F3b"),
        (500, 100000, 0.3, "F3c"),
    ]
    time_per_fista = p3_time / len(fista_configs)

    for bi, bm, si, nm in fista_configs:
        if time.monotonic() >= deadline:
            break
        t_avail = min(time_per_fista, deadline - time.monotonic())
        if t_avail < 10:
            break
        a_opt, obj_opt = fista_adaptive(global_best_a, t_avail, rng,
                                         beta0=bi, beta_max=bm,
                                         step_init=si, name=nm)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()
            print("  %s improved: obj=%.8f ***" % (nm, global_best_obj))

    # === Phase 4: Coordinate descent (25%) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.25
    if p4_time > 30:
        print("\n=== Phase 4: Coordinate descent (%.0fs) ===" % p4_time)
        # Adjust top_k and candidates based on n
        if len(global_best_a) > 10000:
            tk = 4
            nc = 6
        else:
            tk = 5
            nc = 8
        a_cd, obj_cd = coord_descent_vec(global_best_a, p4_time, rng,
                                          top_k=tk, n_candidates=nc)
        if obj_cd < global_best_obj:
            global_best_obj = obj_cd
            global_best_a = a_cd.copy()
            print("  CD improved: obj=%.8f ***" % global_best_obj)

        # FISTA after CD
        remaining4 = deadline - time.monotonic()
        if remaining4 > 30:
            a_cd2, obj_cd2 = fista_adaptive(global_best_a, remaining4 * 0.6, rng,
                                             beta0=200, beta_max=80000,
                                             step_init=0.3, name="P4b-FISTA")
            if obj_cd2 < global_best_obj:
                global_best_obj = obj_cd2
                global_best_a = a_cd2.copy()
                print("  Post-CD FISTA improved: obj=%.8f ***" % global_best_obj)

    # === Phase 5: Final polish (15%) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.8
    if p5_time > 15:
        print("\n=== Phase 5: Final polish (%.0fs) ===" % p5_time)
        a_pol, obj_pol = fista_adaptive(global_best_a, p5_time, rng,
                                         beta0=1000, beta_max=200000,
                                         step_init=0.15, name="P5-final")
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()
            print("  Final polish improved: obj=%.8f ***" % global_best_obj)

    # === Final verification ===
    print("\n=== Final ===")
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        # Ensure sum is sufficient
        if sum(final_list) < 0.01:
            n_fb = 8192
            final_list = [float(np.sqrt(2.0 * n_fb) / n_fb)] * n_fb
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified: obj=%.8f, n=%d" % (ver, len(final_list)))
            global_best_obj = ver
        else:
            print("WARNING: evaluation failed, using fallback")
            n_fb = 8192
            final_list = [float(np.sqrt(2.0 * n_fb) / n_fb)] * n_fb
    else:
        n_fb = 8192
        final_list = [float(np.sqrt(2.0 * n_fb) / n_fb)] * n_fb

    print("Best found: obj=%.8f" % global_best_obj)
    return final_list
```
