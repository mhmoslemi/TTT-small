"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Improved multi-method: FISTA with adaptive step size and annealing beta,
frequency-domain structured perturbations, diverse multi-start,
and L-BFGS-B penalty polishing.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_adaptive(a_init, max_time, rng, beta_start=20.0, beta_end=2000.0,

    for idx, (beta, pen) in enumerate(zip(betas, penalties)):

            shifted = shifted - shifted.max()

            sum_x = x.sum()

            obj_and_grad, a.copy(), method='L-BFGS-B',

                     'iprint': -1}

    step_bb = 1.0 / (2.0 * n)  # initial step
    
    while time.monotonic() - start < max_time:
        it += 1

            k = 1
        elif stuck < 1000:
            k = min(4, 1 + stuck // 300)

        else:

        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum

            break

            dg = g - g_prev
            dx = a - x_prev
            denom = np.dot(dg, dg)
            if denom > 1e-30:
                # BB2
                num = np.dot(dx, dg)
                if num > 1e-30:
                    step_bb = np.dot(dx, dx) / num
                else:
                    step_bb = 2.0 * step_bb
            step_bb = max(1e-10, min(step_bb, 10.0 / gnorm))

        g_prev = g.copy()

        c_new = fft_convolve_self(a_new)
        if c_new.max() > cur_max * 1.01:

            a_new = project_simplex(a - step_bb * g, S)

def generate_inits(n, rng, S, n_inits=12):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Cosine with different frequencies and phases
    for freq in [1, 2, 3, 5]:
        for phase in [0, 0.5]:
            a = 0.5 + 0.5 * np.cos(2 * np.pi * (freq * t + phase))
            a = np.maximum(a, 0.001)
            a = a * (S / a.sum())
            inits.append(a.copy())
    
    # Smooth random
    for _ in range(3):
        raw = rng.random(n)
        F = fft(raw)
        cutoff = max(3, n // 50)
        F[cutoff:-cutoff] *= 0.01
        smooth = np.real(ifft(F))[:n]
        smooth = np.maximum(smooth, 0.0)
        if smooth.sum() > 1e-10:
            smooth = smooth * (S / smooth.sum())
            inits.append(smooth.copy())
    
    # Triangular
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    tri = tri * (S / tri.sum())
    inits.append(tri.copy())
    
    # Two-peak
    peak1 = n // 3
    peak2 = 2 * n // 3
    sigma = n // 15
    a = np.exp(-0.5 * ((np.arange(n) - peak1) / sigma) ** 2) + \
        np.exp(-0.5 * ((np.arange(n) - peak2) / sigma) ** 2)
    a = a * (S / a.sum())
    inits.append(a.copy())
    
    return inits[:n_inits]

import numpy as np

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.0):
    """Top-k weighted gradient for the max function."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    vmin = vals.min()
    w = np.power(np.maximum(vals - vmin, 1e-10), power)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def freq_perturb(a, rng, scale, band_boost=None):
    """Add frequency-domain structured noise to a sequence."""
    n = len(a)
    F = fft(a)

        for i in range(n):

            if dist > width:
                noise_F[i] *= 0.05
                if i > 0 and i < n:
                    noise_F[n - i] *= 0.05
    a_perturbed = a + scale * np.real(ifft(noise_F))[:n]
    return a_perturbed

            step = step_start / max(1.0, beta / beta_start)
        elif stuck < 300:
            step = step_start * 0.5 / max(1.0, beta / beta_start)
        elif stuck < 1000:
            step = step_start * 0.25 / max(1.0, beta / beta_start)
        else:
            step = max(min_step, step_start * 0.1 / max(1.0, beta / beta_start))

        # Compute gradient
        if mode == "lse":
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
        elif mode == "topk":
            k = max(1, min(64, stuck // 50))
            grad = topk_gradient(y, c, k, power=1.0)
        else:  # hybrid: alternate between lse and topk
            if (it // 100) % 3 == 0:
                k = max(1, min(32, stuck // 100))
                grad = topk_gradient(y, c, k, power=1.0)
            else:
                w = lse_weights(c, beta)
                grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum reset to avoid oscillation
        if it % 2000 == 0:
            t_mom = 1.0

        # Aggressive structured perturbation when stuck
        if stuck > 2000 and it % 300 == 0:
            noise_scale = 0.008 * S / np.sqrt(n) * np.exp(-stuck / 20000.0)
            if noise_scale > 1e-9:
                # Rotate the frequency band we perturb
                band_center = (it // 300) % (n // 4)
                band_width = max(3, n // 64)
                y_pert = freq_perturb(y, rng, noise_scale, (band_center, band_width))
                y_pert = project_simplex(y_pert, S)
                y = y_pert
                t_mom = 1.0
                stuck = 0  # reset stuck counter after perturbation

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_bb_descent(a_init, max_time, rng):
    """
    Gradient descent with Barzili-Borwein step size.
    Uses adaptive top-k exact subgradient.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    x_prev = a.copy()
    g_prev = None
    step_bb = 1.0 / (2.0 * n * 0.5)  # conservative initial step
    step_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        stuck = it - last_improve

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        # Adaptive top-k: start sharp, widen when stuck
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = min(3, 1 + stuck // 200)
        elif stuck < 2000:
            k = min(12, 3 + stuck // 250)
        elif stuck < 8000:
            k = min(48, 12 + stuck // 300)
        else:
            k = min(128, 48 + stuck // 500)

        grad = topk_gradient(a, c, k, power=1.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Barzili-Borwein step size
        if g_prev is not None and step_count > 0:
            dg = grad - g_prev
            dx = a - x_prev
            denom_bb = np.dot(dg, dg)
            if denom_bb > 1e-30:
                num_bb = np.dot(dx, dg)
                if num_bb > 1e-30:
                    # BB2 formula
                    step_candidate = np.dot(dx, dx) / num_bb
                    # Damp to avoid too-large steps
                    step_candidate = min(step_candidate, 5.0 * step_bb)
                    step_bb = 0.5 * step_bb + 0.5 * step_candidate
                else:
                    step_bb *= 0.7  # reduce if num is bad
            step_bb = max(1e-12, min(step_bb, 20.0 / max(gnorm, 1e-15)))

        x_prev = a.copy()
        g_prev = grad.copy()
        step_count += 1

        a_new = project_simplex(a - step_bb * grad, S)

        # Backtracking: if new point is much worse, reduce step
        c_new = fft_convolve_self(a_new)
        if c_new.max() > cur_max * 1.05 + 1e-10:
            step_bb *= 0.4
            a_new = project_simplex(a - step_bb * grad, S)

        a = a_new

        # Perturbation when very stuck
        if stuck > 6000 and it % 2000 == 0:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck / 30000.0)
            if noise_scale > 1e-10:
                band_center = (it // 2000) % (n // 4)
                a_pert = freq_perturb(a, rng, noise_scale, (band_center, max(3, n // 32)))
                a = project_simplex(a_pert, S)
                step_bb = 1.0 / (2.0 * n * 0.5)
                g_prev = None

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_lbfgs_polish(a_init, max_time, rng, n_beta_levels=6):
    """L-BFGS-B polishing with penalty method for sum constraint."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    betas = [30.0, 80.0, 200.0, 500.0, 1200.0, 3000.0][:n_beta_levels]
    penalties = [10.0, 30.0, 80.0, 200.0, 500.0, 1000.0][:n_beta_levels]

    for idx, (beta, pen) in enumerate(zip(betas, penalties)):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        def obj_and_grad(x, _beta=beta, _pen=pen):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm_local = b.max()
            shifted = _beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm_local + np.log(e.sum()) / _beta
            sum_x = x.sum()
            obj += _pen * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 2.0 * _pen * (sum_x - S)
            return obj, grad

        t0 = time.monotonic()
        # Scale iterations based on remaining time
        max_iter = min(300, max(30, int((remaining * 0.8) / 0.3)))

        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': max_iter, 'maxls': 50, 'ftol': 1e-15, 'gtol': 1e-12,
                     'iprint': -1}
        )

        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)

        print("    L-BFGS beta=%.0f pen=%.0f: obj=%.8f (t=%.1fs, iters=%d)" %
              (beta, pen, obj_new, elapsed, result.nit))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def generate_inits(n, rng, S, n_inits=10):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Cosine with different frequencies
    for freq in [1, 2, 3, 4, 6]:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        a = np.maximum(a, 0.001)
        a = a * (S / a.sum())
        inits.append(a.copy())

    # Smooth random (very low frequency)
    for _ in range(3):
        raw = rng.random(n)
        F = fft(raw)
        cutoff = max(2, n // 200)
        F[cutoff:-cutoff] *= 0.005
        smooth = np.real(ifft(F))[:n]
        smooth = np.maximum(smooth, 0.0)
        if smooth.sum() > 1e-10:
            smooth = smooth * (S / smooth.sum())
            inits.append(smooth.copy())

    # Triangular
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    tri = tri * (S / tri.sum())
    inits.append(tri.copy())

    # Two-peak
    peak1 = n // 3
    peak2 = 2 * n // 3
    sigma = n // 12
    a = (np.exp(-0.5 * ((np.arange(n) - peak1) / sigma) ** 2) +
         np.exp(-0.5 * ((np.arange(n) - peak2) / sigma) ** 2))
    a = a * (S / a.sum())
    inits.append(a.copy())

    return inits[:n_inits]


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-method: adaptive FISTA with annealing beta, BB gradient descent,
    frequency-domain structured perturbations, diverse multi-start,
    and L-BFGS-B penalty polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    n0 = len(a_start) if a_start is not None else 8192

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.6f" % (n0, obj0))
        if obj0 < global_best_obj:
            global_best_obj = obj0
            global_best_a = a_start.copy()
    else:
        n0 = 8192
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.54 + 0.46 * np.cos(2 * np.pi * t)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    S = np.sqrt(2.0 * n0)

    # === Phase 1: Adaptive FISTA with annealing (25% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.25
    print("\n=== Phase 1: FISTA anneal (n=%d, budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    a1, _ = run_fista_anneal(a_start, p1_budget * 0.55, rng,
                              beta_start=10.0, beta_end=1500.0,
                              step_start=1.0, min_step=0.02, mode="lse")
    obj1 = compute_obj(a1)
    print("  FISTA-lse: obj=%.8f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    rem1 = p1_budget - (time.monotonic() - t1)
    if rem1 > 10:
        a1b, _ = run_fista_anneal(global_best_a, rem1, rng,
                                   beta_start=50.0, beta_end=3000.0,
                                   step_start=0.5, min_step=0.01, mode="hybrid")
        obj1b = compute_obj(a1b)
        print("  FISTA-hybrid: obj=%.8f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: BB Gradient Descent (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = remaining * 0.20
        print("\n=== Phase 2: BB Gradient Descent (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        a2, _ = run_bb_descent(global_best_a, p2_budget, rng)
        obj2 = compute_obj(a2)
        print("  BB-GD: obj=%.8f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: L-BFGS-B polishing (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = remaining * 0.20
        print("\n=== Phase 3: L-BFGS-B polishing (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        a3, obj3 = run_lbfgs_polish(global_best_a, p3_budget, rng, n_beta_levels=6)
        print("  L-BFGS done: obj=%.8f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Multi-start with diverse inits (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p4_budget = remaining * 0.15
        print("\n=== Phase 4: Multi-start (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        S4 = np.sqrt(2.0 * n0)
        inits = generate_inits(n0, rng, S4, n_inits=8)
        # Include current best
        inits.append(global_best_a.copy())

        time_per_init = p4_budget / len(inits)
        for idx, init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            if time_per_init < 5:
                break
            a_m, _ = run_fista_anneal(init, min(time_per_init, 25.0), rng,
                                       beta_start=10.0, beta_end=500.0,
                                       step_start=1.0, min_step=0.05, mode="lse")
            obj_m = compute_obj(a_m)
            if obj_m < global_best_obj:
                global_best_obj = obj_m
                global_best_a = a_m.copy()
                print("  Multi-start %d: obj=%.8f (improved!)" % (idx, obj_m))
            else:
                print("  Multi-start %d: obj=%.8f" % (idx, obj_m))

    # === Phase 5: Final BB polish + L-BFGS squeeze (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % remaining)
        t5 = time.monotonic()

        # BB gradient descent final
        a5, _ = run_bb_descent(global_best_a, remaining * 0.45, rng)
        obj5 = compute_obj(a5)
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()
        print("  Final BB: obj=%.8f (t=%.0fs)" % (obj5, time.monotonic() - t5))

        rem5 = remaining - (time.monotonic() - t5)
        if rem5 > 10:
            a6, obj6 = run_lbfgs_polish(global_best_a, rem5 * 0.85, rng, n_beta_levels=4)
            print("  Final L-BFGS: obj=%.8f (t=%.0fs)" % (obj6, time.monotonic() - t5))
            if obj6 < global_best_obj:
                global_best_obj = obj6
                global_best_a = a6.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj + 1e-10:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
