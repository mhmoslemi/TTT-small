import numpy as np
import time

from scipy.sparse import lil_matrix, csc_matrix

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, beta):
    """Compute LSE value and gradient of beta*LSE(c) w.r.t. y."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def topk_subgradient(a, c, k):
    """Subgradient from top-k lags of the convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def build_conv_csc(n):
    """Build the sparse CSC matrix A where (A @ b)[k] = (a*b)[k] for full convolution.
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    Column j has entries a[0],...,a[n-1] at rows j, j+1, ..., j+n-1.
    """
    A_lil = lil_matrix((2 * n - 1, n))
    for j in range(n):
        A_lil[j:j + n, j] = 1
    return A_lil.tocsc()


def update_conv_csc(A_csc, a):
    """Update data values of the CSC convolution matrix in-place."""
    n = len(a)
    for j in range(n):
        A_csc.data[A_csc.indptr[j]:A_csc.indptr[j + 1]] = a


def solve_lp_direction(a, M, A_csc=None):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns the optimal b or None on failure.
    """
    n = len(a)
    if A_csc is None:
        A_csc = build_conv_csc(n)
    update_conv_csc(A_csc, a)

    b = cp.Variable(n, nonneg=True)
    constraints = [A_csc @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)

    solvers_to_try = [cp.MOSEK, cp.ECOS, cp.SCS]
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver)
            if b.value is not None:
                return np.maximum(b.value, 0)
        except Exception:
            continue
    return None


def run_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.01,
              beta_max=10000.0, step_scale=1.0, use_backtrack=True):
    """FISTA with adaptive LSE and optional backtracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_gradient(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-14 and bt < 5:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 2:
                step = step_local * 0.8
            else:
                step = min(step * 1.03, 2.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            step = step_scale / max(beta, 1.0)
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        if it - last_improve > 5000:
            t_mom = 1.0
            noise_scale = max(1e-6, 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 5000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    return best_a, best_obj


def run_lp_iteration(a_init, max_time, rng, A_csc=None):
    """LP-based iteration: at each step, solve the linearized LP and use line search."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    if A_csc is None:
        A_csc = build_conv_csc(n)

    c = fft_convolve_self(a)
    M = c.max()
    cur_obj = 2.0 * n * M / (S * S)
    start = time.monotonic()
    best_obj = cur_obj
    best_a = a.copy()
    it = 0
    last_improve = 0
    no_lp_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        if time.monotonic() - start > max_time:
            break

        # Solve LP for direction
        b_opt = solve_lp_direction(a, M, A_csc)

        if b_opt is None or np.sum(b_opt) < 1e-10:
            no_lp_count += 1
            if no_lp_count > 10:
                break
            # Fallback: use gradient step
            grad = topk_subgradient(a, c, min(5, 2 * n - 1))
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = S / (gnorm * (1.0 + 0.01 * it))
            a = project_simplex(a - step * grad, S)
            c = fft_convolve_self(a)
            M = c.max()
            cur_obj = 2.0 * n * M / (S * S)
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve = it
            continue
        else:
            no_lp_count = 0

        # Normalize b_opt to have same sum as a
        b_sum = b_opt.sum()
        if b_sum < 1e-10:
            continue
        b_norm = b_opt * (S / b_sum)

        # Direction: d = b_norm - a (move from a toward b_norm)
        d = b_norm - a

        # Line search: find best t in [0, 1]
        best_t = 0.0
        best_val = cur_obj
        for t_candidate in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9, 1.0]:
            a_trial = a + t_candidate * d
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial * (S / max(a_trial.sum(), 1e-15))
            obj_trial = compute_obj(a_trial)
            if obj_trial < best_val:
                best_val = obj_trial
                best_t = t_candidate

        if best_t > 0:
            a = a + best_t * d
            a = np.maximum(a, 0)
            a = a * (S / max(a.sum(), 1e-15))
            c = fft_convolve_self(a)
            M = c.max()
            cur_obj = 2.0 * n * M / (S * S)
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve = it
        else:
            # No improvement from LP direction; try a small gradient step
            grad = topk_subgradient(a, c, min(3, 2 * n - 1))
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = S / (gnorm * (1.0 + 0.02 * it))
            a = project_simplex(a - step * grad, S)
            c = fft_convolve_self(a)
            M = c.max()
            cur_obj = 2.0 * n * M / (S * S)
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve = it

        # Check for stagnation
        if it - last_improve > 100:
            # Perturb to escape
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            c = fft_convolve_self(a)
            M = c.max()
            last_improve = it - 50  # Give it some grace

    return best_a, best_obj


def coordinate_descent(a_init, max_time, rng, max_pairs_per_iter=8):
    """Direct coordinate descent via mass transfer."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 20000:
            break

        k_star = int(np.argmax(c))
        top_k = min(3, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        multi_contrib = np.zeros(n)
        for kk in top_idx:
            lo2 = max(0, int(kk) - n + 1)
            hi2 = min(n - 1, int(kk))
            if hi2 >= lo2:
                multi_contrib[lo2:hi2 + 1] += a[int(kk) - hi2:int(kk) - lo2 + 1][::-1]

        src_scores = multi_contrib * np.maximum(a, 0.001)
        tgt_scores = -multi_contrib * (S / n - a + 0.001)

        n_try = min(max_pairs_per_iter, n)
        src_candidates = np.argsort(src_scores)[::-1][:n_try]
        tgt_candidates = np.argsort(-tgt_scores)[:n_try]

        best_delta_obj = cur_obj + 1e-15
        best_j, best_l, best_delta = -1, -1, 0.0

        half_try = max(2, n_try // 2)
        for j in src_candidates[:half_try]:
            for l in tgt_candidates[:half_try]:
                if j == l or a[j] < 1e-10:
                    continue
                delta = a[j] * 0.05
                if delta < 1e-12:
                    continue
                delta_c = np.zeros(2 * n - 1)
                js = max(0, j)
                je = min(2 * n - 2, j + n - 1)
                coeff_j = a[max(0, js - j):min(n, je - j + 1)]
                delta_c[js:je + 1] -= delta * coeff_j
                ls = max(0, l)
                le = min(2 * n - 2, l + n - 1)
                coeff_l = a[max(0, ls - l):min(n, le - l + 1)]
                delta_c[ls:le + 1] += delta * coeff_l

                new_max = np.max(c + delta_c)
                new_obj = 2.0 * n * new_max / (S * S)
                if new_obj < best_delta_obj:
                    best_delta_obj = new_obj
                    best_j, best_l, best_delta = j, l, delta

        if best_j >= 0:
            j, l, delta = best_j, best_l, best_delta
            a[j] -= delta
            a[l] += delta
            js = max(0, j)
            je = min(2 * n - 2, j + n - 1)
            c[js:je + 1] -= delta * a[max(0, js - j):min(n, je - j + 1)]
            ls = max(0, l)
            le = min(2 * n - 2, l + n - 1)
            c[ls:le + 1] += delta * a[max(0, ls - l):min(n, le - l + 1)]
        else:
            if stuck > 1000:
                idx = rng.integers(0, n)
                delta = a[idx] * 0.01
                if delta > 1e-12:
                    a[idx] -= delta
                    a[rng.integers(0, n)] += delta
                    c = fft_convolve_self(a)

        if it % 100 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def run_subgradient_polish(a_init, max_time, rng):
    """Polish using exact subgradient with decreasing step size."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck < 200:
            k = 1
        elif stuck < 1000:
            k = 2
        elif stuck < 5000:
            k = min(8, 2 + stuck // 500)
        else:
            k = min(32, 8 + stuck // 1000)

        grad = topk_subgradient(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.002 * it))
        step = min(step, 2.0 * S / gnorm)

        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-13:
            a = a_new
        else:
            step *= 0.1
            a_new = project_simplex(a - step * grad, S)
            if compute_obj(a_new) <= cur_obj + 1e-13:
                a = a_new

        if stuck > 3000 and it % 100 == 0:
            ns = max(1e-6, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 8000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.15 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 4 * tt + 0.7)
    elif method == "multi_tone":
        a = 0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt + 1.3) + 0.15 * np.sin(2 * np.pi * 11 * tt + 2.1)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 2.0 * np.abs(tt - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA warm-up, LP-based iteration, coordinate descent, polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        a_start = make_init(n, "cosine", rng)

    n0 = len(a_start)
    S0 = np.sqrt(2.0 * n0)
    a_start = a_start * (S0 / a_start.sum())
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.6f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: FISTA warm-up (20% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.20
    if p1_time > 60:
        print("\n=== Phase 1: FISTA warm-up (budget %.0fs) ===" % p1_time)
        a_opt, obj_opt = run_fista(global_best_a, p1_time, rng,
                                    beta_init=15, beta_rate=0.012,
                                    beta_max=5000, step_scale=0.8,
                                    use_backtrack=True)
        print("  FISTA warm-up: obj=%.6f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

        # Try a second FISTA with different params
        remaining1 = deadline - time.monotonic()
        if remaining1 > 30:
            a_opt2, obj_opt2 = run_fista(global_best_a, remaining1 * 0.5, rng,
                                          beta_init=100, beta_rate=0.005,
                                          beta_max=20000, step_scale=0.5,
                                          use_backtrack=True)
            print("  FISTA v2: obj=%.6f" % obj_opt2)
            if obj_opt2 < global_best_obj:
                global_best_obj = obj_opt2
                global_best_a = a_opt2.copy()

    # === Phase 2: LP-based iteration (35% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.35
    if p2_time > 60:
        print("\n=== Phase 2: LP-based iteration (budget %.0fs) ===" % p2_time)
        n_lp = len(global_best_a)
        print("  Building convolution matrix for n=%d..." % n_lp)
        t_build = time.monotonic()
        A_csc = build_conv_csc(n_lp)
        print("  Matrix built in %.1fs" % (time.monotonic() - t_build))

        a_lp, obj_lp = run_lp_iteration(global_best_a, p2_time, rng, A_csc=A_csc)
        print("  LP iteration: obj=%.6f" % obj_lp)
        if obj_lp < global_best_obj:
            global_best_obj = obj_lp
            global_best_a = a_lp.copy()

        # Follow up with FISTA
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_lp2, obj_lp2 = run_fista(global_best_a, remaining2 * 0.5, rng,
                                        beta_init=50, beta_rate=0.008,
                                        beta_max=20000, step_scale=0.7,
                                        use_backtrack=True)
            print("  FISTA after LP: obj=%.6f" % obj_lp2)
            if obj_lp2 < global_best_obj:
                global_best_obj = obj_lp2
                global_best_a = a_lp2.copy()

    # === Phase 3: Coordinate descent (25% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.25
    if p3_time > 60:
        print("\n=== Phase 3: Coordinate descent (budget %.0fs) ===" % p3_time)
        a_cd, obj_cd = coordinate_descent(global_best_a, p3_time, rng,
                                           max_pairs_per_iter=6)
        print("  Coord descent: obj=%.6f" % obj_cd)
        if obj_cd < global_best_obj:
            global_best_obj = obj_cd
            global_best_a = a_cd.copy()

        # Second CD round with more patience
        remaining3 = deadline - time.monotonic()
        if remaining3 > 30:
            a_cd2, obj_cd2 = coordinate_descent(global_best_a, remaining3 * 0.6, rng,
                                                 max_pairs_per_iter=4)
            print("  CD round 2: obj=%.6f" % obj_cd2)
            if obj_cd2 < global_best_obj:
                global_best_obj = obj_cd2
                global_best_a = a_cd2.copy()

    # === Phase 4: Multi-scale upsample (10% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.10
    if p4_time > 45 and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        print("\n=== Phase 4: Upsample to n=%d (budget %.0fs) ===" % (n2, p4_time))
        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = a_up * (S2 / a_up.sum())
        obj_up = compute_obj(a_up)
        print("  Upsampled initial: obj=%.6f" % obj_up)
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up.copy()

        a_up_opt, obj_up_opt = run_fista(a_up, p4_time * 0.6, rng,
                                          beta_init=5, beta_rate=0.008,
                                          beta_max=5000, step_scale=1.0,
                                          use_backtrack=False)
        print("  FISTA upsampled: obj=%.6f" % obj_up_opt)
        if obj_up_opt < global_best_obj:
            global_best_obj = obj_up_opt
            global_best_a = a_up_opt.copy()

        remaining4 = deadline - time.monotonic()
        if remaining4 > 15:
            a_cd_up, obj_cd_up = coordinate_descent(global_best_a, remaining4 * 0.7,
                                                     rng, max_pairs_per_iter=4)
            print("  CD upsampled: obj=%.6f" % obj_cd_up)
            if obj_cd_up < global_best_obj:
                global_best_obj = obj_cd_up
                global_best_a = a_cd_up.copy()

    # === Phase 5: Final polish (10% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.85
    if p5_time > 10:
        print("\n=== Phase 5: Final polish (budget %.0fs) ===" % p5_time)
        a_pol, obj_pol = run_subgradient_polish(global_best_a, p5_time * 0.5, rng)
        print("  Subgradient polish: obj=%.6f" % obj_pol)
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()

        remaining5 = deadline - time.monotonic()
        if remaining5 > 5:
            a_pol2, obj_pol2 = run_fista(global_best_a, remaining5, rng,
                                          beta_init=1000, beta_rate=0.001,
                                          beta_max=200000, step_scale=0.2,
                                          use_backtrack=True)
            print("  High-beta final: obj=%.6f" % obj_pol2)
            if obj_pol2 < global_best_obj:
                global_best_obj = obj_pol2
                global_best_a = a_pol2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.6f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
