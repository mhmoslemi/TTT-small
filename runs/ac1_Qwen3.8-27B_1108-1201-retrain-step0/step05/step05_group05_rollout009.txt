"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: Multi-strategy optimization combining:
1. Multi-scale initialization (solve small-n problems, upscale)
2. Phase-domain alternating projection (POCS-style)
3. Adaptive active-set subgradient with variance reduction
4. High-beta LSE FISTA for final polish
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
        beta = beta_eff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_active(a, c, K=5):
    """Subgradient from top-K lags of the convolution."""
    n = len(a)
    K = min(K, len(c))
    idx = np.argsort(c)[::-1][:K]
    w = np.zeros(len(c))
    w[idx] = 1.0 / K
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def phase_domain_step(a, S, fsize, alpha_mag=0.1, alpha_phase=0.5):
    """
    Phase-domain alternating step:
    1. Adjust magnitude to flatten autocorrelation
    2. Enforce non-negativity via phase adjustment
    """
    n = len(a)
    # Pad and FFT
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Compute current autocorrelation
    r = np.real(ifft(mag * mag))[:2*n-1]
    r_mean = r.mean()
    
    # Gradient of max(r) with respect to mag^2:
    # r_k = (1/fsize) * IFFT(mag^2)_k
    # dr_k / d(mag^2)_omega = (1/fsize) * exp(2*pi*i*omega*k/fsize)
    # We want to reduce the peak, so decrease mag^2 at frequencies that
    # contribute to the peak lag
    peak_idx = np.argmax(r)
    # The contribution to r[peak_idx] from frequency omega is:
    # (1/fsize) * mag^2[omega] * exp(2*pi*i*omega*peak_idx/fsize)
    # Real part: (1/fsize) * mag^2[omega] * cos(2*pi*omega*peak_idx/fsize)
    # To reduce r[peak_idx], decrease mag^2 where cos is positive
    cos_term = np.cos(2 * np.pi * np.arange(fsize) * peak_idx / fsize)
    # Only decrease where it helps
    delta_mag_sq = -alpha_mag * r[peak_idx] * np.maximum(cos_term, 0) / fsize
    # Also add a small regularization to keep mag from becoming too uneven
    delta_mag_sq += 0.01 * alpha_mag * (mag * mag).mean() * np.ones(fsize) / fsize
    
    new_mag_sq = np.maximum(mag * mag + delta_mag_sq, 1e-10)
    new_mag = np.sqrt(new_mag_sq)
    
    # Reconstruct with new magnitude, same phase
    A_new = new_mag * np.exp(1j * phase)
    A_new[0] = A[0]  # Preserve DC (sum)
    
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.maximum(a_new, 0.0)
    
    # Normalize
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new * (S / s)
    
    return a_new


def run_fista(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.005,
              beta_max=5000.0, K_active=5, use_phase=False, phase_alpha=0.05):
    """
    FISTA with adaptive LSE smoothing and optional phase-domain steps.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    fsize = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it
        
        # Choose gradient type
        r = rng.random()
        if use_phase and r < 0.1:
            # Phase domain step
            y = phase_domain_step(y, S, fsize, alpha_mag=phase_alpha)
            t_mom = 1.0
            continue
        elif r < 0.3:
            # Active set subgradient
            grad = subgradient_active(y, c, K=K_active)
        else:
            # LSE gradient
            grad = lse_gradient(y, c, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        if r < 0.3:
            step = 0.5 * S / (n * max(gnorm / (S/n), 1e-10))
        else:
            step = 1.0 / (3.0 * beta)
        
        # Stuck detection
        if it - last_improve > 3000:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 20000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()

For this to equal $\mu$ (a constant) for all $j$:
$\sum_k 2\lambda_k (a_{j-k} + a_{j+k}) = \mu$

This is a linear system in $a$ (given $\lambda$) or in $\lambda$ (given $a$). At the optimum, both are determined simultaneously.

**Algorithm based on KKT**:
1. Compute the autocorrelation $r$
2. Find the active lags $\tau = \{k : r_k \geq r_{\max} - \epsilon\}$
3. Solve for $\lambda_k$ (weights on active lags) such that $\sum_k \lambda_k \nabla r_k$ is as close to constant as possible (on the support of $a$)

4. Update: $a \leftarrow a - \eta \cdot (\sum_k \lambda_k g_k - \mu \mathbf{1})$ projected to the simplex
