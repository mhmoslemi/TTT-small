"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses a multi-phase approach:
1. Fourier-parameterized low-dimensional optimization for global shape
2. Full-dimensional mirror descent (exponential weights) with LSE-smoothed objective
3. Subgradient polish on peak lags
4. Adaptive beta scheduling and momentum
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def fourier_sequence(params, n):
    """
    Generate a non-negative sequence from Fourier parameters.
    params: [A1, B1, A2, B2, ..., AK, BK]
    a[j] = exp(sum_k (Ak*cos(2*pi*k*j/n) + Bk*sin(2*pi*k*j/n))) normalized to sum S
    """
    K = len(params) // 2
    t = np.arange(n) / n
    v = np.zeros(n)
    for k in range(K):
        Ak = params[2*k]
        Bk = params[2*k+1]
        v += Ak * np.cos(2*np.pi*(k+1)*t) + Bk * np.sin(2*np.pi*(k+1)*t)
    # Softmax normalization to ensure non-negativity
    v_max = np.max(v)
    a = np.exp(v - v_max)
    a = a / a.sum()  # normalize to sum 1
    return a

def fourier_objective(params, n, S_target):
    """Objective for Fourier parameter optimization."""
    a = fourier_sequence(params, n)
    a = a * S_target
    c = fft_convolve_self(a)
    return c.max()  # minimize this (equivalent to minimizing the full objective)

def mirror_descent_step(a, grad, eta):
    """
    Mirror descent (exponential weights) step on the simplex.
    a_new[j] = a[j] * exp(-eta * grad[j]) / Z
    """
    log_a = np.where(a > 0, np.log(a), -100.0)
    log_new = log_a - eta * grad
    # Normalize to sum S
    log_new_max = np.max(log_new)
    new_exp = np.exp(log_new - log_new_max)
    S = new_exp.sum()
    if S < 1e-300:
        return a  # fallback
    a_new = new_exp / S * a.sum()
    return np.maximum(a_new, 0.0)

def lse_smooth_max(c, beta):
    """Log-sum-exp smooth approximation of max(c)."""
    c_max = np.max(c)
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 0)
    return c_max + np.log(np.sum(np.exp(scaled))) / beta

def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution w.r.t. a."""
    n = len(a)
    c_max = np.max(c)
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def topk_subgradient(a, c, k):

import numpy as np
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)

    return 2.0 * n * c.max() / (S * S)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution w.r.t. a."""
    n = len(a)
    c_max = np.max(c)
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)

    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

    Mirror descent (exponential weights) step on the simplex.

    total = new_exp.sum()
    if total < 1e-300:
        return a.copy()
    a_new = new_exp / total * S
    return np.maximum(a_new, 0.0)

    a_cos = (0.5 * (1 + np.cos(2 * np.pi * t)) + 0.1)
    a_cos = a_cos / a_cos.sum() * S_primary
    inits.append(a_cos)

    # Slightly asymmetric init
    a_asym = np.exp(-0.3 * (t - 0.5) ** 2 * 4) * (0.7 + 0.3 * np.cos(2 * np.pi * 2 * t))
    a_asym = np.maximum(a_asym, 0.01)
    a_asym = a_asym / a_asym.sum() * S_primary
    inits.append(a_asym)

    # === Phase 1: Primary optimization (Mirror Descent) ===
    remaining = deadline - time.monotonic()
    # Allocate time to each init based on quality
    t_md = min(800, remaining * 0.35)
    print(f"\nPhase 1: Mirror descent, total time={t_md:.0f}s")

    # Sort inits by quality (waste less time on bad ones)
    init_scores = []
    for i, a_init in enumerate(inits):
        c_init = fft_convolve_self(a_init)
        init_scores.append((c_init.max(), i))
    init_scores.sort()

    # Use best init for most time
    best_init_idx = init_scores[0][1]
    a_curr = inits[best_init_idx].copy()
    t_main = t_md * 0.6
    print(f"  Main MD: using init {best_init_idx}, time={t_main:.0f}s")

    a_md, _ = optimize_mirror_descent(a_curr, S_primary, t_main, rng,
                                      beta_init=30.0, beta_max=80000.0)
    update_global(a_md, "Phase 1 Mirror Descent")

    # Also run on second-best init with remaining time
    remaining = deadline - time.monotonic()
    if remaining > 60:
        second_idx = init_scores[1][1] if len(init_scores) > 1 else best_init_idx
        if second_idx != best_init_idx:
            a_curr2 = inits[second_idx].copy()
            t2 = min(300, remaining * 0.2)
            print(f"  Secondary MD: init {second_idx}, time={t2:.0f}s")
            a_md2, _ = optimize_mirror_descent(a_curr2, S_primary, t2, rng,
                                               beta_init=50.0, beta_max=60000.0)
            update_global(a_md2, "Phase 1 Secondary MD")

    # === Phase 2: Nesterov LSE acceleration ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 100:
        n2 = len(global_a)
        S2 = np.sqrt(2.0 * n2)
        t_nv = min(500, remaining * 0.25)
        print(f"\nPhase 2: Nesterov LSE, time={t_nv:.0f}s")

        a_nv, _ = optimize_nesterov_lse(global_a, S2, t_nv, rng,
                                        beta_init=100.0, beta_max=100000.0)
        update_global(a_nv, "Phase 2 Nesterov LSE")

    # === Phase 3: Subgradient polish ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 80:
        n3 = len(global_a)
        S3 = np.sqrt(2.0 * n3)
        t_sg = min(400, remaining * 0.2)
        print(f"\nPhase 3: Subgradient polish, time={t_sg:.0f}s")

        a_sg, _ = optimize_subgradient(global_a, S3, t_sg, rng, verbose=False)
        update_global(a_sg, "Phase 3 Subgradient")

    # === Phase 4: Alternating refinement ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        n4 = len(global_a)
        S4 = np.sqrt(2.0 * n4)
        print(f"\nPhase 4: Alternating refinement, time={remaining:.0f}s")

        a_curr = global_a.copy()
        for phase_i in range(5):
            if time.monotonic() > deadline - 30:
                break
            t_phase = min(150, (deadline - time.monotonic() - 30) / (5 - phase_i))
            if t_phase < 10:
                break

            if phase_i % 3 == 0:
                a_curr, _ = optimize_mirror_descent(a_curr, S4, t_phase, rng,
                                                    beta_init=200.0, beta_max=150000.0)
            elif phase_i % 3 == 1:
                a_curr, _ = optimize_nesterov_lse(a_curr, S4, t_phase, rng,
                                                  beta_init=300.0, beta_max=200000.0)
            else:
                a_curr, _ = optimize_subgradient(a_curr, S4, t_phase, rng, verbose=False)
            update_global(a_curr, f"Phase 4 iter {phase_i}")

    # === Phase 5: Multi-restart perturbation ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 40:
        n5 = len(global_a)
        S5 = np.sqrt(2.0 * n5)
        n_restarts = 4
        t_per = (remaining - 30) / n_restarts
        print(f"\nPhase 5: {n_restarts} perturb+refine, {t_per:.0f}s each")

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            # Different noise scales for diversity
            scales = [0.005, 0.01, 0.02, 0.003]
            noise_scale = scales[ri % len(scales)] * S5 / np.sqrt(n5)
            a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n5), S5)
            a_pert, _ = optimize_mirror_descent(a_pert, S5, t_per, rng,
                                                beta_init=300.0, beta_max=200000.0)
            update_global(a_pert, f"Phase 5 restart {ri} (scale={noise_scale:.4f})")

    # === Phase 6: Try different length ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 100:
        n_alt = 16384
        S_alt = np.sqrt(2.0 * n_alt)
        t_alt = min(300, remaining * 0.3)
        if t_alt > 50:
            print(f"\nPhase 6: Alternative length n={n_alt}, time={t_alt:.0f}s")
            # Upsample global best
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n_alt)
            a_up = np.interp(t_new, t_old, global_a)
            a_up = np.maximum(a_up, 0.0)
            a_up = a_up / a_up.sum() * S_alt
            a_alt, _ = optimize_mirror_descent(a_up, S_alt, t_alt, rng,
                                               beta_init=30.0, beta_max=50000.0)
            update_global(a_alt, f"Phase 6 n={n_alt}")

    # === Final verification ===
    if global_a is None:
        print("WARNING: No valid solution found, using fallback")
        global_a = np.ones(256) * 1.0
        global_best = obj_value(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: len={n_final}, internal_obj={global_best:.6f}")
    print(f"evaluate_sequence verdict: {ver:.6f}")
    print(f"{'='*60}")

    return final_list
