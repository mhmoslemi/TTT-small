"""
Minimize 2*n*max_conv/sum^2 using proximal QP with adaptive dual lag-weights.
Key idea: maintain a probability vector p over convolution lags (dual variable).
At each step, solve the convex proximal QP: min a^T Q(p)a + (1/(2*eta))||a-a_prev||^2
s.t. a>=0, sum(a)=s. The QP has Hankel structure (Q_{ik}=p_{i+k}) and is solved
via FISTA in O(n log n) per iteration. The dual p is updated to concentrate on
active (high) lags, driving the max down.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def hankel_matvec(p, x):
    """Compute Q @ x where Q_{ik} = p_{i+k}. p has length 2n-1, x has length n."""
    n = len(x)
    fsize = 1 << (3 * n - 2).bit_length()
    P = fft(p, fsize)
    X = fft(x, fsize)
    result = np.real(ifft(P * np.conj(X)))
    return result[:n]


def solve_prox_qp(p, a_prev, s, eta, max_fista=60, tol=1e-11):
    """Solve convex QP: min a^T Q(p) a + (1/(2*eta))||a - a_prev||^2 s.t. a>=0, 1^T a = s.
    Uses accelerated projected gradient (FISTA)."""
    n = len(a_prev)
    L = 2.0 + 1.0 / eta

    x = a_prev.copy()
    x_old = a_prev.copy()
    t_mom = 1.0

    for it in range(max_fista):
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        if it == 0:
            y = x
        else:
            y = x + ((t_mom - 1.0) / t_new) * (x - x_old)
            y = np.maximum(y, 0.0)

        Qy = hankel_matvec(p, y)
        grad = 2.0 * Qy + (y - a_prev) / eta

        x_new = project_simplex(y - grad / L, s)

        diff = np.linalg.norm(x_new - x)
        x_old = x
        x = x_new
        t_mom = t_new

        if diff < tol * max(1.0, np.linalg.norm(x)):
            break

    return x


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def update_dual(p, c, beta, temperature_schedule=None):
    """Update dual variable p based on convolution values c."""
    c_max = c.max()
    if c_max < 1e-30:
        return p.copy()
    log_p = np.log(p + 1e-300) + beta * c / c_max
    log_p -= log_p.max()
    p_new = np.exp(log_p)
    p_new /= p_new.sum()
    return p_new


def update_dual_power(p, c, alpha):
    """Update dual using power rule: p_j proportional to (c_j/c_max)^alpha."""
    c_max = c.max()
    if c_max < 1e-30:
        return p.copy()
    ratios = np.maximum(c / c_max, 0.0)
    p_new = p * np.power(ratios, alpha)
    p_sum = p_new.sum()
    if p_sum < 1e-30:
        return np.ones_like(p) / len(p)
    p_new /= p_sum
    return p_new


def active_set_dual(c, n_active, margin=0.0):
    """Create dual concentrated on top n_active lags within margin of max."""
    c_max = c.max()
    threshold = c_max * (1.0 - margin)
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    active = active[:n_active] if len(active) > n_active else active
    p = np.zeros(len(c))
    p[active] = np.ones(len(active)) / len(active)
    return p

    tt = np.linspace(0, 1, n, endpoint=False)
