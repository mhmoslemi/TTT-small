def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

        lse_val, grad, c, w = lse_gradient(y, beta, mu)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # BB step size update
        if use_bb and s_prev is not None and y_prev is not None:
            dy = y - y_prev
            ds = grad - s_prev
            dyds = np.dot(dy, ds)
            if dyds > 1e-15:
                dwn = np.dot(dy, dy)
                dsn = np.dot(ds, ds)
                if dsn > 1e-15:
                    step_bb = max(1e-10, min(10.0 / max(beta, 1.0), dwn / dsn))
            s_prev = grad.copy()
            y_prev = y.copy()

        if use_backtrack:
            step_local = step_bb if use_bb else step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-14 and bt < 6:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 2:
                step_bb = step_local * 0.7
            else:
                if not use_bb:
                    step = min(step * 1.05, 5.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            x_new = project_simplex(y - step_bb * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 1e-10)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        # Restart momentum if stuck
        if it - last_improve > 3000:
            t_mom = 1.0
            noise_scale = max(1e-7, 0.005 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 5000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            s_prev = None
            y_prev = None

        it2 = it
        if it2 % 2000 == 0:
            elapsed = time.monotonic() - start
            print("    FISTA-B: it=%d, obj=%.6f, best=%.6f, t=%.0fs" % (it, cur_obj, best_obj, elapsed))

    return best_a, best_obj


def run_subgradient_frank_wolfe(a_init, max_time, rng, k_top=5, k_grow_rate=500):
    """
    Subgradient method with adaptive top-k and line search.
    Uses exact subgradient at top-k lags with Barzilai-Borwein-like step.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 1e-10, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    step_prev = S / np.sqrt(n)

    # Track previous for BB
    a_prev = a.copy()
    g_prev = None

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        # Adaptive k: start small, grow when stuck
        k = min(k_top + stuck // k_grow_rate, 64)

        # Compute subgradient from top-k lags
        top_idx = np.argsort(c)[::-1][:k]
        w_full = np.zeros(len(c))
        w_full[top_idx] = 1.0 / k
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Center the gradient (for the sum constraint)
        grad_centered = grad - grad.mean()
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-15:
            break

        # BB step size
        if g_prev is not None:
            dy = a - a_prev
            ds = grad_centered - g_prev
            dyds = np.dot(dy, ds)
            if dyds > 1e-15:
                dwn = np.dot(dy, dy)
                dsn = np.dot(ds, ds)
                if dsn > 1e-15:
                    step = max(1e-6, min(10 * S / gnorm, dwn / dsn))
                else:
                    step = step_prev
            else:
                step = step_prev
        else:
            step = S / gnorm

        step_prev = step
        g_prev = grad_centered.copy()
        a_prev = a.copy()

        # Try the step with line search (a few candidates)
        candidates = [step, step * 0.5, step * 0.25, step * 0.1]
        best_x = a
        best_c = cur_obj
        for t in candidates:
            a_trial = project_simplex(a - t * grad_centered, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = 2.0 * n * c_trial.max() / (S * S)
            if obj_trial < best_c:
                best_c = obj_trial
                best_x = a_trial

        a = best_x

        if stuck > 200 and it % 50 == 0:
            ns = max(1e-7, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 8000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj


def mass_transfer_cd(a_init, max_time, rng, max_pairs=8):
    """
    Coordinate descent via mass transfer.
    Transfers mass from coordinates with high contribution to the peak
    lag(s) to coordinates with low contribution.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 50000:
            break

        # Find top-k peak lags
        k_top = min(3 + stuck // 10000, 8)
        top_idx = np.argsort(c)[::-1][:k_top]

        # Compute total contribution of each coordinate to the top-k lags
        contrib = np.zeros(n)
        for kk in top_idx:
            kk = int(kk)
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            if hi >= lo:
                # contrib[j] += a[kk - j] for j in [lo, hi]
                for j in range(lo, hi + 1):
                    contrib[j] += a[kk - j]

        # Source: high contribution AND high value (more to give)
        # Target: low contribution AND low value (room to receive)
        src_score = contrib * (a + 1e-10)
        tgt_score = -contrib * (a.max() - a + 1e-10)

        n_try = min(max_pairs, n)
        src_cands = np.argsort(src_score)[::-1][:n_try]
        tgt_cands = np.argsort(tgt_score)[:n_try]

        best_delta_obj = cur_obj
        best_move = None

        for j in src_cands[:n_try // 2 + 1]:
            for l in tgt_cands[:n_try // 2 + 1]:
                if j == l or a[j] < 1e-8:
                    continue
                # Try a few transfer fractions
                for frac in [0.03, 0.01, 0.005]:
                    delta = a[j] * frac
                    if delta < 1e-12:
                        continue
                    # Compute change in convolution
                    # a[j] decreases by delta, a[l] increases by delta
                    # c_k changes by: -delta * a[k-j] + delta * a[k-l] (approximately)
                    dc = np.zeros(2 * n - 1)
                    # Effect of -delta at position j
                    js = max(0, j)
                    je = min(2 * n - 2, j + n - 1)
                    range_j = range(js, je + 1)
                    coeff_j = a[max(0, js - j):min(n, je - j + 1)]
                    dc[js:je + 1] -= delta * coeff_j
                    # Effect of +delta at position l
                    ls = max(0, l)
                    le = min(2 * n - 2, l + n - 1)
                    coeff_l = a[max(0, ls - l):min(n, le - l + 1)]
                    dc[ls:le + 1] += delta * coeff_l

                    new_max = np.max(c + dc)
                    new_obj = 2.0 * n * new_max / (S * S)
                    if new_obj < best_delta_obj:
                        best_delta_obj = new_obj
                        best_move = (j, l, delta)

        if best_move is not None:
            j, l, delta = best_move
            a[j] -= delta
            a[l] += delta
            # Update c
            js = max(0, j)
            je = min(2 * n - 2, j + n - 1)
            coeff_j = a[max(0, js - j):min(n, je - j + 1)]
            c[js:je + 1] -= delta * coeff_j
            ls = max(0, l)
            le = min(2 * n - 2, l + n - 1)
            coeff_l = a[max(0, ls - l):min(n, le - l + 1)]
            c[ls:le + 1] += delta * coeff_l
        else:
            # Random perturbation when stuck
            if stuck > 500:
                n_perturb = min(10, n)
                for _ in range(n_perturb):
                    j = rng.integers(0, n)
                    l = rng.integers(0, n)
                    if j != l and a[j] > 1e-8:
                        delta = a[j] * 0.01
                        a[j] -= delta
                        a[l] += delta
                c = fft_convolve_self(a)

        if it % 200 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via monotone cubic interpolation (preserves non-negativity better)."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    result = np.interp(t_new, t_old, a)
    return np.maximum(result, 0)


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "ramp":
        a = np.linspace(2.0, 0.2, n)
    elif method == "ramp2":
        a = 1.0 + 1.5 * (1.0 - tt)
    elif method == "gaussian":
        a = np.exp(-3.0 * (tt - 0.3) ** 2 / (0.2 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 6 * tt + 1.0)
    elif method == "power_law":
        a = (1.0 - tt + 0.1) ** 3
    elif method == "two_bump":
        a = np.exp(-30 * (tt - 0.25) ** 2) + 0.5 * np.exp(-30 * (tt - 0.75) ** 2)
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "flat":
        a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase with barrier-augmented FISTA, subgradient Frank-Wolfe,
    mass-transfer CD, upsample, and final polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)

    global_best_obj = np.inf
    global_best_a = None

    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01 or len(a_start) < 100:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        a_start = make_init(n, "ramp", rng)
        print("Using default ramp init, n=%d" % n)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Starting: n=%d, obj=%.6f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: Diverse initializations + quick FISTA (12% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.12
    if p1_time > 60:
        print("\n=== Phase 1: Diverse starts (budget %.0fs) ===" % p1_time)
        methods = ["ramp", "ramp2", "power_law", "two_bump", "cosine", "chirp"]
        per_time = p1_time / len(methods)
        for method in methods:

            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()
                print("  %s: obj=%.6f ***NEW BEST***" % (method, obj_opt))
            else:
                print("  %s: obj=%.6f" % (method, obj_opt))

    # === Phase 2: Main FISTA with barrier (25% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.25

        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

        # Variant B: no barrier, aggressive LSE
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_opt2, obj_opt2 = run_fista_barrier(global_best_a, remaining2 * 0.55, rng,
                                                  beta_init=100, beta_rate=0.008,
                                                  beta_max=80000, mu=0.0, use_bb=True)
            print("  FISTA-B(mu=0, high beta): obj=%.6f" % obj_opt2)
            if obj_opt2 < global_best_obj:
                global_best_obj = obj_opt2
                global_best_a = a_opt2.copy()

    # === Phase 3: Subgradient Frank-Wolfe (20% time) ===
    remaining = deadline - time.monotonic()

                                            beta_init=10, beta_rate=0.015,
                                            beta_max=20000, mu=0.0, use_bb=True)
        print("  FISTA upsampled: obj=%.6f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

        # CD on upsampled
        remaining5 = deadline - time.monotonic()
        if remaining5 > 20:
            a_opt2, obj_opt2 = mass_transfer_cd(global_best_a, remaining5 * 0.6, rng, max_pairs=6)
            print("  CD upsampled: obj=%.6f" % obj_opt2)
            if obj_opt2 < global_best_obj:

    # === Phase 6: Final polish (8% time) ===

                                            beta_init=500, beta_rate=0.003,
                                            beta_max=200000, mu=0.0,
                                            use_bb=False, use_backtrack=True)
        print("  High-beta polish: obj=%.6f" % obj_pol)
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()

        # Subgradient polish
        remaining6 = deadline - time.monotonic()
        if remaining6 > 10:
            a_pol2, obj_pol2 = run_subgradient_frank_wolfe(global_best_a, remaining6, rng,
                                                            k_top=10, k_grow_rate=200)
            print("  SubGrad polish: obj=%.6f" % obj_pol2)
            if obj_pol2 < global_best_obj:
                global_best_obj = obj_pol2
                global_best_a = a_pol2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]

    return final_list
