"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences of length 8192.
Strategy: Multi-phase optimization combining LSE-FISTA, active-set subgradient
with Polyak step size, and targeted peak shaving. Key innovations over previous:
(1) Polyak step size for self-adaptive convergence, (2) direct active-set
optimization via subgradient, (3) pair-level peak shaving that identifies and
reduces specific contributing pairs.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c). Returns (grad, lse_value)."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    # grad[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def active_set_gradient(a, c, n_active=20):
    """
    Subgradient using top n_active lags.
    Returns gradient vector.
    """
    n = len(a)
    # Get indices of top lags
    idx = np.argsort(c)[::-1][:n_active]
    w = np.zeros(len(c))
    w[idx] = 1.0 / n_active
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, idx


def run_lse_fista(a_init, n, S, max_time, rng,
                  beta_start=50.0, beta_end=30000.0, beta_steps=200,
                  momentum_gap=300):
    """FISTA with LSE smoothing and adaptive beta."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 100000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / beta

        # Stagnation handling
        if it - last_improve > momentum_gap:
            t_mom = 1.0
            noise = 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 2000.0)
            if noise > 1e-9:
                y = project_simplex(y + rng.normal(0, noise, n), S)
            step *= 0.5
            last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_active_subgradient(a_init, n, S, max_time, rng, n_active=15):
    """
    Subgradient method with Polyak step size on active set.
    This is the key novel component: uses (f(a) - f_lower) / ||g||^2 as step size.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    # Lower bound: average of c[k] = sum(a)^2 / (2n-1) = 2n/(2n-1)
    f_lower = 2.0 * n / (2.0 * n - 1)  # ≈ 1.0
    
    # Polyak step Parameters
    damp = 0.9  # Damping factor for Polyak step
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Active set gradient
        grad, active_idx = active_set_gradient(a, c, n_active)
        gnorm2 = np.dot(grad, grad)
        if gnorm2 < 1e-30:
            break
        
        # Polyak step size
        polyak_step = damp * (cur_max - f_lower) / gnorm2
        
        # Cap the step to avoid overshooting
        step = min(polyak_step, 0.1 * S / np.sqrt(n) / np.max(np.abs(grad)) if np.max(np.abs(grad)) > 0 else 0.01)
        
        # Also use a minimum step to avoid stalling
        min_step = 1e-6 / (1 + it * 1e-5)
        step = max(step, min_step)
        
        a_new = project_simplex(a - step * grad, S)
        
        # Check if this is an improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max * (1 + 1e-8):
            a = a_new
        else:
            # Try smaller step
            a_new = project_simplex(a - step * 0.5 * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max * (1 + 1e-8):
                a = a_new
            else:
                # Keep old a, add small noise
                noise = 0.001 * S / np.sqrt(n) * np.exp(-it / 5000.0)
                if noise > 1e-10:
                    a = project_simplex(a + rng.normal(0, noise, n), S)
        
        # Periodically increase active set size
        if it % 500 == 0 and n_active < 50:
            n_active += 5

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_mixed_method(a_init, n, S, max_time, rng):
    """
    Alternates between LSE-FISTA (smooth) and active-set subgradient (sharp).
    Uses a smart switching criterion: switch to active-set when LSE plateaus.

    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

def targeted_peak_shaving(a, n, S, n_peaks=5, n_steps=200, rng=None):
    """
    Targeted peak shaving: For each peak lag k, identify the top-contributing
    pairs (j, k-j) and multiplicatively reduce them.
    More surgical than uniform reduction.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    a = a.copy()
    
    for step_i in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()
        
        # Get top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # For each peak, compute responsibility
        resp = np.zeros(n)
        for k in top_idx:
            excess = c[k] - cm * 0.995
            if excess <= 0:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            np.add.at(resp, js, contrib * excess)
        
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        
        # Multiplicative update with smaller rate
        lr = 0.01 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)
        
        # Small noise to avoid getting stuck
        if step_i % 50 == 0:
            noise = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)
    
    return a

Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Multi-phase optimization combining LSE-FISTA, active-set subgradient
with Polyak step size, KKT-based refinement (new: targets the optimality condition

peak shaving. Key innovations: (1) KKT refinement via multiplicative updates,
(2) adaptive Lipschitz-based step sizing, (3) simulated annealing for escape.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c). Returns (grad, lse_value)."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def active_set_gradient(a, c, n_active=20):
    """Subgradient using top n_active lags with uniform weights."""
    n = len(a)
    idx = np.argsort(c)[::-1][:n_active]
    w = np.zeros(len(c))
    w[idx] = 1.0 / n_active
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, idx


def active_set_gradient_weighted(a, c, n_active=20):
    """Subgradient using top n_active lags with LSE weights among them."""
    n = len(a)
    idx = np.argsort(c)[::-1][:n_active]
    c_top = c[idx]
    cm = c_top.max()
    shifted = 200.0 * (c_top - cm)
    e = np.exp(np.clip(shifted, -500, 500))
    w_top = e / e.sum()
    w = np.zeros(len(c))
    w[idx] = w_top
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, idx

                  momentum_gap=250):

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    lipchitz_est = 1.0 / beta_start

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 100000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best:

        step = 1.0 / beta
        # Estimate local Lipschitz: ||grad(a) - grad(a')|| / ||a - a'|| ~ 2*beta*max|a|
        lipchitz_est = max(lipchitz_est * 0.99, 2.0 * beta * np.max(np.abs(y)) / S)
        step = min(step, 1.0 / max(lipchitz_est, 1e-10) * 0.5)

        # Stagnation handling with momentum restart
        stagnant = it - last_improve
        if stagnant > momentum_gap:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 1500.0)
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            step *= 0.4
            last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_active_subgradient(a_init, n, S, max_time, rng, n_active=15):
    """
    Subgradient method with Polyak step size on active set.
    Uses (f(a) - f_lower) / ||g||^2 as step size with damping.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    f_lower = 2.0 * n / (2.0 * n - 1)
    damp = 0.8
    best_f = np.inf
    it_since_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            it_since_improve = 0
        else:
            it_since_improve += 1

        grad, _ = active_set_gradient_weighted(a, c, n_active)
        gnorm2 = np.dot(grad, grad)
        if gnorm2 < 1e-30:
            break

        # Polyak step with damping
        # Use best_f as a tighter lower bound approximation
        f_target = max(f_lower, best_f * 0.999)
        polyak_step = damp * (cur_max - f_target) / gnorm2

        # Also compute gradient-based step as upper bound
        max_grad = np.max(np.abs(grad))
        grad_step = 0.5 * S / (n * max_grad) if max_grad > 0 else 0.01

        step = min(polyak_step, grad_step)
        step = max(step, 1e-8 / (1 + it * 1e-6))

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        new_max = c_new.max()

        # Simulated annealing: occasionally accept worse solutions
        temp = 0.001 * (cur_max - f_lower) * np.exp(-it / 5000.0)
        if new_max < cur_max or (temp > 1e-10 and rng.random() < np.exp(-(new_max - cur_max) / temp)):
            a = a_new
            if new_max < best_f:
                best_f = new_max
        else:
            # Try half step
            a_new2 = project_simplex(a - 0.5 * step * grad, S)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < cur_max:
                a = a_new2
            elif it_since_improve > 50 and rng.random() < 0.3:
                # Escape: add noise
                noise = 0.002 * S / np.sqrt(n) * np.exp(-it_since_improve / 1000.0)
                if noise > 1e-10:
                    a = project_simplex(a + rng.normal(0, noise, n), S)

        # Periodically increase active set
        if it % 800 == 0 and n_active < 50:
            n_active = min(50, n_active + 5)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_kkt_refinement(a_init, n, S, max_time, rng, n_active=20):
    """
    KKT-based refinement: At the optimum, the active-set gradient should be
    constant on the support of a. Use multiplicative updates to drive the
    gradient toward uniformity on the support.
    
    This is the key novel method: it directly targets the first-order
    optimality condition rather than just descending the objective.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    eta = 0.05  # Learning rate for multiplicative update

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad, _ = active_set_gradient_weighted(a, c, n_active)

        # Get support (elements above threshold)
        threshold = 1e-8
        support = a > threshold
        n_support = support.sum()
        if n_support < 2:
            break

        # Compute gradient on support
        g_sup = grad[support]
        g_mean = np.mean(g_sup)
        g_std = np.std(g_sup)
        g_gap = np.max(g_sup) - np.min(g_sup)

        if g_gap < 1e-10:
            break

        # Adaptive learning rate: proportional to the gap
        eta_adapt = eta * min(1.0, g_gap / (0.1 * S / n))
        eta_adapt = max(eta_adapt, 1e-6)

        # Multiplicative update: reduce high-grad, increase low-grad on support
        # a[j] *= exp(-eta * (grad[j] - g_mean))
        log_update = np.zeros(n)
        log_update[support] = -eta_adapt * (g_sup - g_mean)
        # Clip to avoid overflow
        log_update = np.clip(log_update, -5, 5)
        a_new = a * np.exp(log_update)
        a_new = project_simplex(a_new, S)

        # Check improvement
        c_new = fft_convolve_self(a_new)
        new_max = c_new.max()

        if new_max < cur_max * (1 + 1e-9):
            a = a_new
        else:
            # Try smaller step
            a_half = a * np.exp(0.5 * log_update)
            a_half = project_simplex(a_half, S)
            c_half = fft_convolve_self(a_half)
            if c_half.max() < cur_max * (1 + 1e-9):
                a = a_half
            else:
                # Add small noise to escape
                noise = 0.001 * S / np.sqrt(n) * np.exp(-it / 3000.0)
                if noise > 1e-10:
                    a = project_simplex(a + rng.normal(0, noise, n), S)

        # Decay learning rate slowly
        if it % 500 == 0:
            eta *= 0.95

        # Periodically increase active set
        if it % 1000 == 0 and n_active < 40:
            n_active += 5

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_mixed_optimization(a_init, n, S, max_time, rng):
    """
    Alternates between LSE-FISTA, active-set subgradient, and KKT refinement.
    Switches based on which method is making the most progress.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    # Track progress of each method
    method_best = {'lse': np.inf, 'active': np.inf, 'kkt': np.inf}
    method_last = {'lse': 0, 'active': 0, 'kkt': 0}

    # Phase A: LSE-FISTA (40% of time)
    tA = max_time * 0.40
    a, vA = run_lse_fista(a, n, S, tA, rng, beta_start=30, beta_end=15000, beta_steps=120)
    c = fft_convolve_self(a)
    method_best['lse'] = c.max()
    if c.max() < best:
        best = c.max()
        best_a = a.copy()
    print("  Mixed LSE: %.8f" % best)

    # Phase B: Active-set subgradient (25% of time)
    remaining = max(1.0, max_time - (time.monotonic() - start))
    tB = remaining * 0.45
    a, vB = run_active_subgradient(a, n, S, tB, rng, n_active=12)
    c = fft_convolve_self(a)
    method_best['active'] = c.max()
    if c.max() < best:
        best = c.max()
        best_a = a.copy()
    print("  Mixed Active: %.8f" % best)

    # Phase C: KKT refinement (20% of time)
    remaining = max(1.0, max_time - (time.monotonic() - start))
    tC = remaining * 0.5
    a, vC = run_kkt_refinement(a, n, S, tC, rng, n_active=15)
    c = fft_convolve_self(a)
    method_best['kkt'] = c.max()
    if c.max() < best:
        best = c.max()
        best_a = a.copy()
    print("  Mixed KKT: %.8f" % best)

    # Phase D: Final high-beta LSE polish
    remaining = max(1.0, max_time - (time.monotonic() - start))
    if remaining > 5:
        a, vD = run_lse_fista(a, n, S, remaining * 0.7, rng,
                               beta_start=5000, beta_end=80000, beta_steps=30)
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()
        print("  Mixed Final LSE: %.8f" % best)

    return best_a, best


def targeted_peak_shaving(a_init, n, S, n_peaks=5, n_steps=300, rng=None):
    """
    Surgical peak shaving: identify top-contributing pairs for each peak lag
    and multiplicatively reduce them. More targeted than uniform reduction.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    a = normalize_sum(a_init, n)

    for step_i in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()

        top_idx = np.argsort(c)[::-1][:n_peaks]

        resp = np.zeros(n)
        for k in top_idx:
            excess = c[k] - cm * 0.998
            if excess <= 0:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            np.add.at(resp, js, contrib * excess)

        resp_max = resp.max()
        if resp_max < 1e-30:
            break

        lr = 0.005 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)

        if step_i % 100 == 0:
            noise = 0.0003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)

    return a


def run_gradient_descent_ls(a_init, n, S, max_time, rng, n_active=15):
    """
    Pure gradient descent with backtracking line search on active-set subgradient.
    More stable than FISTA for the final refinement.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    stagnation_count = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1

        grad, _ = active_set_gradient_weighted(a, c, n_active)
        gnorm2 = np.dot(grad, grad)
        if gnorm2 < 1e-30:
            break

        # Backtracking line search
        step = 0.1 * S / (n * np.max(np.abs(grad))) if np.max(np.abs(grad)) > 0 else 0.01
        improved = False
        for _ in range(12):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur_max - 1e-12:
                a = a_test
                improved = True
                break
            step *= 0.5

        if not improved and stagnation_count > 20:
            # Escape with noise
            noise = 0.001 * S / np.sqrt(n) * np.exp(-stagnation_count / 500.0)
            if noise > 1e-10:
                a = project_simplex(a + rng.normal(0, noise, n), S)

        if it % 1500 == 0 and n_active < 40:
            n_active += 5

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_creep_mode(a_init, n, S, max_time, rng):
    """
    Final creep mode: very small steps with very high beta LSE.
    Designed for the absolute last mile of optimization.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    # Start with very high beta, very small steps
    beta = 50000.0
    step_base = 0.01

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad, lse_val = lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Very conservative step
        step = step_base / beta * S
        step = min(step, 0.001 * S / n)

        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try even smaller
            a_new2 = project_simplex(a - 0.1 * step * grad, S)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < cur_max:
                a = a_new2

        # Slowly increase beta
        if it % 100 == 0:
            beta = min(beta * 1.5, 200000.0)
            step_base *= 0.9

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def generate_asymmetric_init(n, rng):
    """
    Generate an asymmetric initialization designed to reduce center peak.
    Uses a skewed distribution that puts more mass on one side.
    """
    t = np.linspace(0, 1, n)
    # Skewed raised cosine
    center = 0.4
    width = 0.3
    x = (t - center) / width
    a = np.where(np.abs(x) < 1, 0.5 * (1 + np.cos(np.pi * x)), 0.0)
    # Add asymmetry: multiply by a linear ramp
    ramp = 0.5 + 0.5 * t
    a = a * ramp
    return np.maximum(a, 1e-10)


def generate_bump_init(n, n_bumps=3, rng=None):
    """Generate initialization with multiple bumps to reduce concentration."""
    if rng is None:
        rng = np.random.default_rng(0)
    t = np.linspace(0, 1, n)
    a = np.zeros(n)
    for i in range(n_bumps):
        mu = 0.2 + 0.6 * i / max(1, n_bumps - 1)
        sigma = 0.08 + 0.04 * rng.random()
        amp = 0.5 + 0.5 * rng.random()
        a += amp * np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    return np.maximum(a, 1e-10)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Phases: LSE-FISTA -> Active-set subgradient -> KKT refinement -> Peak shaving
    -> Gradient descent with line search -> Creep mode.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 30.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)
    t0 = time.monotonic()

    print("Budget: %.0fs, n target: 8192" % budget_s)

    # Load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded given: n=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_seq(a, label=""):
        nonlocal global_best, global_a
        n = len(a)
        S = np.sqrt(2.0 * n)
        a = normalize_sum(a, n)
        c = fft_convolve_self(a)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a.copy()
            elapsed = time.monotonic() - t0
            print("  *** NEW BEST: %.8f (n=%d, t=%.0fs) %s" % (val, n, elapsed, label))

    # === Phase 1: Quick LSE-FISTA on given (20% time) ===
    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        t1 = min(150.0, (deadline - time.monotonic()) * 0.20)
        print("\n=== Phase 1: LSE-FISTA (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_lse_fista(given, n_g, S_g, t1, rng,
                                beta_start=50, beta_end=10000, beta_steps=120)
        try_seq(a1, "LSE")
    else:
        # Generate initial sequence
        n_g = 8192
        S_g = np.sqrt(2.0 * n_g)
        a1 = generate_asymmetric_init(n_g, rng)
        a1 = normalize_sum(a1, n_g)
        print("  Generated asymmetric init")
        try_seq(a1, "init")

    # === Phase 2: Active-set subgradient (25% time) ===
    if global_a is not None:
        n_a = len(global_a)
        S_a = np.sqrt(2.0 * n_a)
        remaining = deadline - time.monotonic()
        t2 = remaining * 0.25
        print("\n=== Phase 2: Active-set subgradient (%.0fs) ===" % t2)
        a2, v2 = run_active_subgradient(global_a, n_a, S_a, t2, rng, n_active=12)
        try_seq(a2, "Active")

    # === Phase 3: KKT refinement (20% time) ===
    if global_a is not None:
        n_k = len(global_a)
        S_k = np.sqrt(2.0 * n_k)
        remaining = deadline - time.monotonic()
        t3 = remaining * 0.25
        print("\n=== Phase 3: KKT refinement (%.0fs) ===" % t3)
        a3, v3 = run_kkt_refinement(global_a, n_k, S_k, t3, rng, n_active=15)
        try_seq(a3, "KKT")

    # === Phase 4: Peak shaving + LSE polish (15% time) ===
    if global_a is not None:
        n_p = len(global_a)
        S_p = np.sqrt(2.0 * n_p)
        remaining = deadline - time.monotonic()
        t4 = remaining * 0.20
        if t4 > 20:
            print("\n=== Phase 4: Peak shaving + polish (%.0fs) ===" % t4)
            a4 = targeted_peak_shaving(global_a, n_p, S_p, n_peaks=8, n_steps=200, rng=rng)
            try_seq(a4, "Shave")
            # LSE polish after shaving
            remaining2 = max(1.0, deadline - time.monotonic())
            if remaining2 > 15:
                a4b, _ = run_lse_fista(a4, n_p, S_p, remaining2 * 0.6, rng,
                                        beta_start=3000, beta_end=30000, beta_steps=40)
                try_seq(a4b, "Shave+LSE")

    # === Phase 5: Gradient descent with line search (15% time) ===
    if global_a is not None:
        n_gd = len(global_a)
        S_gd = np.sqrt(2.0 * n_gd)
        remaining = deadline - time.monotonic()
        t5 = remaining * 0.25
        if t5 > 15:
            print("\n=== Phase 5: Gradient descent + LS (%.0fs) ===" % t5)
            a5, v5 = run_gradient_descent_ls(global_a, n_gd, S_gd, t5, rng, n_active=20)
            try_seq(a5, "GD")

    # === Phase 6: KKT refinement again (on best found) (10% time) ===
    if global_a is not None:
        n_k2 = len(global_a)
        S_k2 = np.sqrt(2.0 * n_k2)
        remaining = deadline - time.monotonic()
        t6 = remaining * 0.3
        if t6 > 10:
            print("\n=== Phase 6: KKT refinement round 2 (%.0fs) ===" % t6)
            a6, v6 = run_kkt_refinement(global_a, n_k2, S_k2, t6, rng, n_active=25)
            try_seq(a6, "KKT2")

    # === Phase 7: Creep mode (final 5-10%) ===
    if global_a is not None:
        n_c = len(global_a)
        S_c = np.sqrt(2.0 * n_c)
        remaining = deadline - time.monotonic()
        t7 = remaining * 0.5
        if t7 > 5:
            print("\n=== Phase 7: Creep mode (%.0fs) ===" % t7)
            a7, v7 = run_creep_mode(global_a, n_c, S_c, t7, rng)
            try_seq(a7, "Creep")

    # === Final: Quick active-set polish on very best ===
    if global_a is not None:
        n_f = len(global_a)
        S_f = np.sqrt(2.0 * n_f)
        remaining = max(1.0, deadline - time.monotonic())
        if remaining > 5:
            print("\n=== Final: Quick polish (%.0fs) ===" % remaining)
            a_f, _ = run_active_subgradient(global_a, n_f, S_f, remaining * 0.6, rng, n_active=30)
            try_seq(a_f, "Final")

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nFinal: evaluate_sequence=%.8f, internal=%.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  WARNING: evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result, using uniform fallback")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
