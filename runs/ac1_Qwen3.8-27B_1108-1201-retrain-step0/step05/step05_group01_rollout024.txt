Minimize 2*n*max_conv/sum^2 over non-negative sequences using multi-scale LSE continuation,
adaptive FISTA, and L-BFGS-B polishing.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Softmax weights for LSE gradient computation."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*conv(y,y)[k] w.r.t. y."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross

    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def run_lbfgs_polish(a_init, beta, S, max_time, penalty=50.0):

    L-BFGS-B polishing with penalty method for sum constraint.
    """

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    def obj_and_grad(x):
        x = np.maximum(x, 0.0)
        b = fft_convolve_self(x)
        cm = b.max()
        shifted = beta * (b - cm)
        shifted = shifted - shifted.max()
        e = np.exp(shifted)
        w = e / e.sum()
        lse_val = cm + np.log(e.sum()) / beta
        sum_x = x.sum()
        obj = lse_val + penalty * (sum_x - S) ** 2
        grad = conv_gradient(x, w) + 2.0 * penalty * (sum_x - S)
        return obj, grad

        obj_and_grad, a.copy(), method='L-BFGS-B',
        jac=True, bounds=[(0, 1000)] * n,
        options={'maxiter': 200, 'maxls': 40, 'ftol': 1e-15,
                 'gtol': 1e-10, 'iprint': -1}
    )

    a_new = a_new * (S / max(a_new.sum(), 1e-15))
    obj_new = compute_obj(a_new)
    
    if obj_new < best_obj:
        best_obj = obj_new
        best_a = a_new.copy()
    
    return best_a, best_obj, result.nit


def run_active_subgrad(a_init, S, max_time, rng):

        it += 1

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
- Multi-scale LSE continuation FISTA with backtracking
- L-BFGS-B polishing at high beta levels
- Active-set subgradient refinement
- Structured perturbation for escaping local minima
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*conv(y,y)[k] w.r.t. y. Returns 2*cross_corr(w, y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def fista_lse(a_init, beta, S, max_time, rng):
    """
    FISTA for LSE-smoothed max of self-convolution.
    Uses backtracking line search and adaptive momentum.
    Returns (best_a, best_lse_value, num_iterations).
    """
    n = len(a_init)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_lse = np.inf
    it = 0
    no_improve = 0

    # Initial step estimate
    c = fft_convolve_self(y)
    w = lse_weights(c, beta)
    grad = conv_gradient(y, w)
    gnorm = np.linalg.norm(grad)
    step = 1.0 / (beta * 2.0 * np.linalg.norm(y) + 1e-10)
    step = min(step, 2.0 * S / max(gnorm, 1e-15))

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cm = c.max()
        shifted = beta * (c - cm)
        shifted = shifted - shifted.max()
        e = np.exp(shifted)
        lse_val = cm + np.log(max(e.sum(), 1e-300)) / beta

        if lse_val < best_lse - 1e-13 * max(1.0, abs(lse_val)):
            best_lse = lse_val
            best_a = y.copy()
            no_improve = 0

def lbfgs_polish(a_init, beta, S, max_time, penalty=100.0):
    """
    L-BFGS-B with penalty method for sum constraint.
    Minimizes LSE(conv(a,a), beta) + penalty*(sum(a)-S)^2 subject to a>=0.
    """
    n = len(a_init)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    def obj_and_grad(x):
        x = np.maximum(x, 0.0)
        b = fft_convolve_self(x)
        cm = b.max()
        shifted = beta * (b - cm)
        shifted = shifted - shifted.max()
        e = np.exp(shifted)
        w = e / e.sum()
        lse_val = cm + np.log(max(e.sum(), 1e-300)) / beta
        sum_x = x.sum()
        obj = lse_val + penalty * (sum_x - S) ** 2
        grad = conv_gradient(x, w) + 2.0 * penalty * (sum_x - S)
        return obj, grad

    time_budget = max_time - (time.monotonic() - start)
    if time_budget < 5:
        return best_a, best_obj, 0

    maxiter = min(300, max(50, int(time_budget * 20)))

    result = scipy_minimize(
        obj_and_grad, a.copy(), method='L-BFGS-B',
        jac=True, bounds=[(0.0, 1000.0)] * n,
        options={'maxiter': maxiter, 'maxls': 30, 'ftol': 1e-15,
                 'gtol': 1e-12, 'factr': 1e10, 'iprint': -1}
    )

    a_new = np.clip(result.x, 0, 1000)
    a_new = a_new * (S / max(a_new.sum(), 1e-15))
    obj_new = compute_obj(a_new)

    if obj_new < best_obj:
        best_obj = obj_new
        best_a = a_new.copy()

    return best_a, best_obj, result.nit
