"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses refined FISTA with backtracking, adaptive active-set,
and local perturbation search to improve upon previous results.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def power_p_gradient(y, c, p):
    """Gradient of sum(c^p) w.r.t. a."""
    n = len(y)
    c_pos = np.maximum(c, 0)
    w = c_pos ** (p - 1)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w = w / wsum * p
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return np.real(ifft(W * np.conj(A)))[:n] * 2.0


def exact_subgradient(y, c, active_indices):
    """Subgradient using given active indices."""
    n = len(y)
    w_full = np.zeros(len(c))
    k = len(active_indices)
    w_full[active_indices] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def fista_lse(a_init, max_time, rng, beta_init=10.0, beta_max=50000.0,
              beta_growth=1.002, use_backtrack=True):
    """FISTA with LSE smooth approximation and backtracking line search."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    
    # Compute initial
    c_init = fft_convolve_self(a)
    best_conv = c_init.max()
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Adaptive beta
        if stuck > 200:
            beta = min(beta_max, beta * beta_growth)
        elif it < 100:
            beta = max(1.0, beta_init * 0.9)
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size with backtracking
        if use_backtrack:
            # Lipschitz estimate: approximately 2*beta*max(a) + 2*mean(a)
            L = 2.0 * beta * a.max() + 4.0 * c.max() / n
            step = min(1.0 / max(L, 1e-10), 1.0 / gnorm * 0.5)
        else:
            step = 0.5 / max(beta, 1.0)
        step = max(step, 1e-12)
        
        x_new = project_simplex(y - step * grad, S)
        
        # Backtracking: check if objective decreased
        if use_backtrack and it % 5 == 0:
            c_new = fft_convolve_self(x_new)
            if c_new.max() > cur_max * 1.01:
                step *= 0.5
                x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Restart with noise when stuck
        if stuck > 3000:
            t_mom = 1.0
            noise_scale = max(1e-8, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 5000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it
            beta = max(beta_init, beta * 0.5)
        
        # Periodic momentum restart
        if it % 5000 == 0:
            t_mom = 1.0
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

def local_perturbation_search(a_init, max_time, rng):

                a = a_new.copy()
            # Decrease scale over time
            elapsed = time.monotonic() - start

def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample_avg(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    if n < 2:
        return a.copy()
    a_padded = np.resize(a, n_new * factor)
    return a_padded.reshape(n_new, factor).mean(axis=1)

    """Find optimal b >= 0 such that ||a*b||_inf <= M, maximizing sum(b)."""
    n = len(a)
    conv_len = 2 * n - 1
    
    # This is a linear program:
    # max sum(b_j)
    # s.t. (a*b)_k <= M for k = 0, ..., 2n-2
    #      (a*b)_k >= -M for k = 0, ..., 2n-2 (usually not binding if a,b >= 0)
    #      b_j >= 0
    
    # For non-negative a and b, (a*b)_k >= 0 always, so we only need (a*b)_k <= M
    
    # This is: max 1^T b s.t. C b <= M, b >= 0
    # where C is a (2n-1) x n matrix where C[k,j] = FK(a, k-j) (the "unfolding" of convolution)

    # The optimal b is an extreme point, which means at most (2n-1) constraints are active.
    # But solving the full LP is expensive for large n.

    # Then sum(b) = M * sum(y) (by strong duality).

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses refined FISTA with backtracking line search, adaptive active-set management,
KKT-based polishing, and multi-scale search. Key improvements over previous:
- Proper backtracking to avoid overshooting near optimum
- KKT-condition checking for convergence certification
- Adaptive active set with running average for stability
- Multi-scale exploration (n=8192 and n=16384)
- Local perturbation with simulated annealing acceptance
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_xy(x, y):
    """Full linear convolution of x and y via FFT."""
    nx = len(x)
    ny = len(y)
    size = nx + ny - 1
    fsize = 1 << (size - 1).bit_length()
    X = fft(x, fsize)
    Y = fft(y, fsize)
    return np.real(ifft(X * Y))[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def power_p_weight(c, p):
    """Compute normalized power-p weights for gradient."""
    c_pos = np.maximum(c, 1e-30)
    w = c_pos ** (p - 1)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.ones(len(c)) / len(c)
    return (p / wsum) * w


def subgradient_from_weights(y, w):
    """Compute subgradient given weight distribution w on convolution lags."""
    n = len(y)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def fista_backtrack(a_init, max_time, rng, beta_init=10.0, beta_max=50000.0,
                    beta_growth=1.003, use_lse=True):
    """FISTA with backtracking line search and adaptive restart."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    y_prev = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    c_init = fft_convolve_self(a)
    best_conv = c_init.max()

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta
        if stuck > 200:
            beta = min(beta_max, beta * beta_growth)
        elif it < 50:
            beta = max(2.0, beta_init * 0.8)

        # Gradient
        if use_lse:
            grad = lse_gradient(y, c, beta)
        else:
            active = np.where(c >= cur_max * (1 - 0.01))[0]
            if len(active) > 10:
                active = np.argsort(c)[::-1][:10]
            elif len(active) == 0:
                active = np.array([int(np.argmax(c))])
            w = np.zeros(len(c))
            w[active] = 1.0 / len(active)
            grad = subgradient_from_weights(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size with backtracking
        L_est = 2.0 * beta * a.max() + 4.0 * c.max() / (n * 0.1)
        step = min(1.0 / max(L_est, 1e-10), 0.5 / max(gnorm, 1e-10))
        step = max(step, 1e-12)

        # FISTA update
        x_new = project_simplex(y - step * grad, S)

        # Restart condition: if x_k is close to x_{k-1}
        dist = np.linalg.norm(x_new - y_prev)
        if dist < 1e-10 * S:
            t_mom = 1.0
            y = x_new
            a = x_new
            y_prev = x_new
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a_prev = a.copy()
        a = x_new
        y_prev = x_new
        t_mom = t_new

        # Noise restart when stuck
        if stuck > 3000:
            t_mom = 1.0
            noise_scale = max(1e-8, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 8000.0))
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = project_simplex(a + rng.normal(0, noise_scale * 0.5, n), S)
            beta = max(beta_init * 0.5, beta * 0.5)
            stuck = 0
            last_improve = it

        # Periodic momentum restart
        if it % 4000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def fista_adaptive_active(a_init, max_time, rng):
    """FISTA with adaptive active-set and running average for stability."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    # Running average of active set (smoothed probability distribution)
    avg_active = np.full(2 * n - 1, 1.0 / (2 * n - 1))

    c_curr = fft_convolve_self(a)
    best_conv = c_curr.max()

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive active set with tolerance
        if stuck < 100:
            tol_frac = 0.0
            k_max = 1
        elif stuck < 500:
            tol_frac = 0.001
            k_max = 3
        elif stuck < 2000:
            tol_frac = 0.005
            k_max = 8
        else:
            tol_frac = 0.01
            k_max = 32

        threshold = cur_max * (1.0 - tol_frac)
        active = np.where(c >= threshold)[0]
        if len(active) > k_max:
            active = np.argsort(c)[::-1][:k_max]
        elif len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Update running average (exponential moving average)
        w_inst = np.zeros(len(c))
        w_inst[active] = 1.0 / len(active)
        alpha_smooth = 0.01 if stuck < 1000 else 0.05
        avg_active = (1 - alpha_smooth) * avg_active + alpha_smooth * w_inst
        avg_active = np.maximum(avg_active, 0)
        avg_active = avg_active / avg_active.sum()

        # Use the averaged active set for gradient (more stable)
        if it % 3 == 0 and stuck > 200:
            grad = subgradient_from_weights(y, avg_active)
        else:
            grad = subgradient_from_weights(y, w_inst)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size
        step = min(0.8 / max(gnorm, 1e-10), S * 0.02 / max(gnorm, 1e-10))

        a = x_new
        t_mom = t_new

        # Reset smoothed active set on restart

            t_mom = 1.0
            avg_active = np.full(2 * n - 1, 1.0 / (2 * n - 1))
            noise_scale = max(1e-8, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 15000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it

        if it % 4000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def kkt_polish(a_init, max_time, rng):
    """Polish using KKT conditions: on support, subgradient should be constant."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)
    best_conv = c.max()

    # Initial active set
    active = np.array([int(np.argmax(c))])

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = a.copy()

        # Update active set
        tol = max(cur_max * 0.0005, 1e-15)
        active = np.where(c >= cur_max - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        # Cap active set size
        if len(active) > 20:
            active = np.argsort(c)[::-1][:20]

        # Compute subgradient
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        grad = subgradient_from_weights(a, w)

        # KKT check: on support, grad should be constant
        support = np.where(a > 1e-8 * S)[0]
        if len(support) > 1:
            g_supp = grad[support]
            g_mean = g_supp.mean()
            g_var = np.var(g_supp)

            # Direction: equalize grad on support
            # We want to move a so that grad becomes more uniform on support
            # grad = 2 * (w * a), so d(grad)/da = 2 * (w * da)
            # To reduce variance of grad on support, move in direction:
            # da_i proportional to -(grad_i - g_mean) for i in support
            direction = np.zeros(n)
            direction[support] = -(g_supp - g_mean)
            # Zero-sum direction
            direction = direction - direction.mean()

            dnorm = np.linalg.norm(direction)
            if dnorm < 1e-15:
                # Try removing a small element from support
                idx_min = support[np.argmin(a[support])]
                if a[idx_min] < 0.01 * S / n:
                    a[idx_min] = 0
                    a = normalize_seq(a, S)
                    continue
                break

            # Adaptive step size
            step = 0.005 * S / dnorm
            # Try the step
            a_new = a + step * direction
            a_new = np.maximum(a_new, 0)
            a_new = normalize_seq(a_new, S)
            c_new = fft_convolve_self(a_new)

            if c_new.max() < cur_max:
                a = a_new
            else:
                # Halve step
                step *= 0.5
                a_new = a + step * direction
                a_new = np.maximum(a_new, 0)
                a_new = normalize_seq(a_new, S)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < cur_max:
                    a = a_new
                else:
                    # Try perturbation to escape
                    if rng.random() < 0.3:
                        noise_scale = 0.005 * S / np.sqrt(n)
                        idx = rng.choice(n, size=max(5, n // 200), replace=False)
                        a[idx] += rng.normal(0, noise_scale, len(idx))
                        a = np.maximum(a, 0)
                        a = normalize_seq(a, S)
                    break

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def local_annealing(a_init, max_time, rng):
    """Local search with simulated annealing acceptance."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_conv = fft_convolve_self(a).max()
    best_a = a.copy()
    cur_conv = best_conv

    # Temperature schedule
    T0 = 0.01 * S / np.sqrt(n)
    T_min = 1e-12

    n_attempt = 0
    while time.monotonic() - start < max_time:
        n_attempt += 1
        # Annealing schedule
        frac = min(1.0, (time.monotonic() - start) / max_time)
        T = T0 * np.exp(-5 * frac)

        if T < T_min:
            T = T_min

        # Generate perturbation
        # Mix of small random and structured perturbations
        if rng.random() < 0.7:
            # Small random noise
            noise = rng.normal(0, T, n)
        elif rng.random() < 0.5:
            # Structured: modify a random subset
            k = rng.integers(max(3, n // 100), max(10, n // 20))
            idx = rng.choice(n, size=k, replace=False)
            noise = np.zeros(n)
            noise[idx] = rng.normal(0, T * 3, k)
        else:
            # Frequency-domain perturbation
            A = fft(a, n)
            A[:n] += rng.normal(0, T * 0.5, n)  # Small real freq perturbation
            a_try = np.real(ifft(A))[:n]
            a_try = np.maximum(a_try, 0)
            a_try = normalize_seq(a_try, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < cur_conv:
                cur_conv = c_try.max()
                a = a_try
                if cur_conv < best_conv:
                    best_conv = cur_conv
                    best_a = a.copy()
            continue

        a_new = a + noise
        a_new = np.maximum(a_new, 0)
        a_new = normalize_seq(a_new, S)
        c_new = fft_convolve_self(a_new)
        new_conv = c_new.max()

        # Acceptance criterion
        if new_conv < cur_conv:
            accept = True
        else:
            delta = new_conv - cur_conv
            accept = rng.random() < np.exp(-delta / max(T, 1e-15))

        if accept:
            a = a_new
            cur_conv = new_conv
            if cur_conv < best_conv:
                best_conv = cur_conv
                best_a = a.copy()

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

        a_up = normalize_seq(a_up, S2)

        # FISTA on upsampled
        b1 = max(15, s5_budget * 0.4)
        a_out, _ = fista_backtrack(a_up, b1, rng,
                                    beta_init=5.0, beta_max=10000.0)
        obj_out = compute_obj(a_out)
        print("  Upscale FISTA: %.6f (t=%.1fs)" % (obj_out, time.monotonic() - t), flush=True)
        if obj_out < global_best_obj:
            global_best_obj = obj_out
            global_best_a = a_out.copy()
            global_best_n = n2

        # Active set on upsampled
        remaining = deadline - time.monotonic()
        if remaining > 20:
            b2 = remaining * 0.5
            a_out, _ = fista_adaptive_active(global_best_a, b2, rng)
            obj_out = compute_obj(a_out)
            print("  Upscale active: %.6f (t=%.1fs)" % (obj_out, time.monotonic() - t), flush=True)
            if obj_out < global_best_obj:
                global_best_obj = obj_out
                global_best_a = a_out.copy()
                global_best_n = len(a_out)

    # === Stage 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        s6_budget = remaining * 0.85
        print("--- Stage 6: Final polish (%.0fs) ---" % s6_budget, flush=True)
        t = time.monotonic()

        # KKT polish
        b1 = s6_budget * 0.4
        a_out, _ = kkt_polish(global_best_a, b1, rng)
        obj_out = compute_obj(a_out)
        print("  KKT: %.6f (t=%.1fs)" % (obj_out, time.monotonic() - t), flush=True)
        if obj_out < global_best_obj:
            global_best_obj = obj_out
            global_best_a = a_out.copy()
            global_best_n = len(a_out)

        # Final FISTA
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_out, _ = fista_backtrack(global_best_a, remaining * 0.7, rng,
                                        beta_init=50.0, beta_max=50000.0,
                                        beta_growth=1.001)
            obj_out = compute_obj(a_out)
            print("  FISTA: %.6f (t=%.1fs)" % (obj_out, time.monotonic() - t), flush=True)
            if obj_out < global_best_obj:
                global_best_obj = obj_out
                global_best_a = a_out.copy()
                global_best_n = len(a_out)

        # Final annealing
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_out, _ = local_annealing(global_best_a, remaining * 0.7, rng)
            obj_out = compute_obj(a_out)
            print("  Anneal: %.6f (t=%.1fs)" % (obj_out, time.monotonic() - t), flush=True)
            if obj_out < global_best_obj:
                global_best_obj = obj_out
                global_best_a = a_out.copy()
                global_best_n = len(a_out)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL: obj=%.6f, n=%d ===" % (ver, len(final_list)), flush=True)
            print("Best internal obj: %.6f" % global_best_obj, flush=True)
        else:
            print("WARNING: rejected by evaluator, using fallback", flush=True)
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
