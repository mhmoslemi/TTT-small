import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates < w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def conv_full(a, N):
    """Self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    if N < 2 * n - 1:
        N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def peak_shaving_step(a, S, N, top_k=10, max_ls=10):
    """
    One peak-shaving iteration: compute top-K peaks, form gradient,
    line search, project. Returns (new_a, new_max_b).

    n = len(a)

        top_idx = np.argpartition(c, -top_k)[-top_k:]

        top_vals = c[top_idx]

    total = top_vals.sum()
    if total < 1e-30:
        return a, cur_max
    weights = top_vals / total
    
    # Compute gradient: g[i] = 2 * sum_m w[m] * a[k_m - i]
    # This is a weighted sum of shifted/reversed a
    g = np.zeros(n)
    for m in range(len(top_idx)):
        k = int(top_idx[m])
        w = weights[m]
        # a[k - i] for i = 0..n-1, valid when 0 <= k-i < n
        # i ranges from max(0, k-n+1) to min(k, n-1)
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:

def lse_objective_grad(a, N, beta, top_k=None):
    """Compute LSE-smoothed peak value and gradient."""
    n = len(a)
    c = conv_full(a, N)
    c_max = c.max()
    
    if beta < 1e-6:
        weights = np.ones(len(c)) / len(c)
        lse_val = c_max
    else:
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:
            weights = exp_bc / total
            lse_val = c_max + np.log(total) / beta
    
    # Gradient via FFT
    fsize = N  # use same N for efficiency
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def lbfgs_lse_block(a_init, n, S, N, max_time, rng, beta_init=10.0, beta_max=500.0,
                    m_lbfgs=12):
    """L-BFGS on LSE-smoothed objective."""
    a = a_init.copy()
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    g, f_val, c, c_max = lse_objective_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        if it % 200 == 0:
            beta = min(beta_max, beta * 1.15)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            g, f_val, c, c_max = lse_objective_grad(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

        direction = lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        for ls in range(15):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = conv_full(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max * (1 + 1e-10) or ls == 14:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_max:
                    break
            alpha_ls *= 0.5

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        g_new, f_new, c_nw, cmax_nw = lse_objective_grad(a_new, N, beta)

        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c_max = cmax_nw

        if it % 500 == 0 and rng.random() < 0.3:
            # Spectral perturbation
            A = fft(a, N)
            mag = np.abs(A)
            phase = np.angle(A)
            n_rand = max(1, int(N * 0.2))
            rand_idx = rng.choice(N, size=n_rand, replace=False)
            phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
            A_new = mag * np.exp(1j * phase)
            a_new = np.real(ifft(A_new))[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a = a_new * (S / s)
                s_hist = []
                y_hist = []
                g, f_val, c, c_max = lse_objective_grad(a, N, beta)

    return best_a, best_cmax


def make_ushape_init(n, S, p=2.0, alpha=1.0):
    """U-shaped (bowl) initialization."""
    t = np.linspace(-1, 1, n)
    a = 1.0 + alpha * np.abs(t) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_fourier_flat(n, S, N, rng):
    """Target flat |A(f)| spectrum."""
    A = np.zeros(N, dtype=complex)
    A[0] = S  # DC
    target_mag = S / np.sqrt(2 * n - 1)
    for f in range(1, N):
        # Conjugate symmetry for real sequence
        A[f] = target_mag
    # Ensure conjugate symmetry
    for f in range(1, N // 2 + 1):
        A[N - f] = np.conj(A[f])
    # Random phases
    for f in range(1, N // 2 + 1):
        A[f] = target_mag * np.exp(1j * rng.uniform(0, 2 * np.pi))
        A[N - f] = np.conj(A[f])
    A[0] = S
    a = np.real(ifft(A))[:n]
    a = np.abs(a) + 0.01
    s = a.sum()
    return a * (S / s)


def make_init_multi_freq(n, S, rng):
    t = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n)
    n_freqs = 8
    for _ in range(n_freqs):
        freq = rng.integers(2, 50)
        phase = rng.uniform(0, 2 * np.pi)
        amp = 0.15 + 0.3 * rng.random()
        a += amp * np.sin(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.01)
    s = a.sum()
    return a * (S / s)


def safe_normalize(a, n, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
