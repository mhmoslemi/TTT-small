Hybrid optimization for minimizing the convolution peak ratio:
  objective = 2 * n * max(conv(a,a)) / (sum(a))^2
Uses three complementary methods:
  1. Subgradient descent with momentum and adaptive active-set size
  2. Exact coordinate descent (1D optimization per coordinate)
  3. Log-sum-exp (LSE) smooth polishing
Explores multiple sequence lengths and random restarts.
"""

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def subgradient_optimize(a0, S, max_time, rng, verbose=True):

    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=S.
    """

    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    it = 0
    stuck = 0

        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        progress = min(1.0, (time.monotonic() - start) / max_time)
        k = int(8 + 128 * progress ** 0.5)
        k = min(k, 2 * n - 1)

        top_idx = np.argsort(c)[::-1][:k]
        median_c = np.median(c)
        weights = np.maximum(c[top_idx] - median_c, 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(k) / k
        else:
            weights = weights / wsum

        for ii in range(k):
            kidx = int(top_idx[ii])
            wi = weights[ii]
            j_start = max(0, kidx - n + 1)

            if j_start <= j_end:
                indices = kidx - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * wi * a[indices]

        if gnorm < 1e-14:
            break

        if gap < 1e-12:
            break

        step = 2.0 * gap / (gnorm * gnorm)

def coordinate_descent(a0, max_time, rng, verbose=True):
    """
    Exact coordinate descent: for each coordinate, find the optimal
    1D step that minimizes the max convolution value.
    Works with sum(a) = 1 normalization (scale-invariant objective).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        return a
    a = a / S  # Normalize to sum = 1
    # Add small floor to avoid division by zero
    a = np.maximum(a, 1e-12)
    a = a / a.sum()  # Renormalize
    
    best_max = np.inf
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        b = fft_convolve_self(a)
        cur_max = b.max()
        
        if cur_max < best_max * (1 - 1e-10):
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Compute prefix and suffix max of b
        # pmax[i] = max(b[0..i])
        pmax = np.maximum.accumulate(b)
        # smax[i] = max(b[i..end])
        smax = np.maximum.accumulate(b[::-1])[::-1]
        
        # For each coordinate i: mr[i] = max(b outside [i, i+n-1])
        mr = np.full(n, -np.inf)
        # Left part: b[0..i-1], max = pmax[i-1] for i >= 1
        if n > 1:
            mr[1:] = np.maximum(mr[1:], pmax[:n-1])
        # Right part: b[i+n..2n-2], max = smax[i+n] for i+n <= 2n-2, i.e., i <= n-2
        if n > 1:
            mr[:n-1] = np.maximum(mr[:n-1], smax[n:2*n-1])
        # For i=0: only right part. For i=n-1: only left part.
        # The above handles it: mr[0] gets smax[n], mr[n-1] gets pmax[n-2]
        
        # f0[i] = max(b[i:i+n]) - sliding window max
        f0 = np.empty(n)
        for i in range(n):
            f0[i] = b[i:i+n].max()
        
        # Compute optimal delta for each coordinate
        improved = False
        for i in range(n):
            if f0[i] > mr[i] + 1e-14:
                w = b[i:i+n]
                # d_opt = min_j (mr[i] - w[j]) / (2 * a[j])
                ratios = (mr[i] - w) / (2.0 * a)
                d = ratios.min()
                # Ensure a[i] + d >= 0
                d = max(d, -a[i] + 1e-15)
                if d < -1e-15:
                    a[i] += d
                    improved = True
        
        if improved:
            # Rescale to sum = 1
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
        
        # Noise injection when stuck
        if stuck > 100:
            noise_scale = 0.01 / np.sqrt(n)
            a = a + rng.normal(0, noise_scale, n)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            stuck = 0
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            obj = 2.0 * n * best_max
            print(f"    CD it={it}, t={elapsed:.0f}s, obj={obj:.8f}, max={best_max:.6f}")
        
        # Convergence check
        if cur_max - b.min() < 1e-10:
            break
    
    return best_a


def lse_optimize(a0, S, max_time, rng, beta_init=50.0, verbose=True):
    """LSE-smoothed gradient descent for final polishing."""
    n = len(a0)
    a = project_simplex(a0, S).copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_init * np.exp(0.01 * it)
        beta = min(beta, 50000.0)
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 300:
            noise_scale = 0.003 * S / np.sqrt(n) * max(0.01, 1.0 - progress)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            continue
        
        # LSE gradient
        cm = c.max()
        scaled = beta * (c - cm)
        e = np.exp(scaled)
        tot = e.sum()
        w = e / tot
        
        # grad = 2 * ifft(FFT(w) * conj(FFT(a)))[:n]
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = min(1.0 / (1.5 * beta), 1.0 / gnorm)
        a_try = project_simplex(a - step * grad, S)
        c_try = fft_convolve_self(a_try)
        if c_try.max() <= cur_max * (1.0 + 1e-8):
            a = a_try
        else:
            a = project_simplex(a - 0.5 * step * grad, S)
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    LSE it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}")
    
    return best_a, best


def generate_ideal_init(n):
    """
    Generate the 'flat self-convolution' ideal sequence.
    This is the unique (up to scale) non-negative sequence whose
    self-convolution is constant (for infinite support).
    Truncated to length n as an initialization.
    """
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        # b[k] = sum_{j=0}^{k} a[j]*a[k-j] = a[0]^2 = 1
        # 2*sum_{j=0}^{k-1} a[j]*a[k-j] + a[k]^2 = ... wait
        # b[k] = 2*sum_{j=0}^{floor((k-1)/2)} a[j]*a[k-j] + (a[k/2]^2 if k even) = a[0]^2 = 1
        # Actually: b[k] = sum_{j=0}^{k} a[j]*a[k-j] = 1
        # a[k] appears as: a[0]*a[k] + a[k]*a[0] = 2*a[k] (when k > 0, a[0]=1)
        # Plus the "cross terms" that don't involve a[k]
        # So: 2*a[k] + (b[k] - 2*a[k]) = 1 → 2*a[k] = 1 - (cross terms)
        # Cross terms = sum_{j=1}^{k-1} a[j]*a[k-j] (j from 1 to k-1, each counted twice except middle)
        # Wait: b[k] = a[0]*a[k] + a[k]*a[0] + sum_{j=1}^{k-1} a[j]*a[k-j]
        #        = 2*a[k] + sum_{j=1}^{k-1} a[j]*a[k-j]
        # So: 2*a[k] = 1 - sum_{j=1}^{k-1} a[j]*a[k-j]
        cross = 0.0
        for j in range(1, k):
            cross += a[j] * a[k - j]
        a[k] = max(0.0, (1.0 - cross) / 2.0)
    
    if a.sum() > 0:
        a = a / a.sum()
    return a

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

        t1b = min(200, rem * 0.12)
        if t1b > 20:
            print(f"Phase 1b: LSE polish, t={t1b:.0f}s")
            a_pol, _ = lse_optimize(a_opt, S1, t1b, rng, beta_init=100.0, verbose=True)
            update_global(a_pol, "P1b LSE")

    # --- Phase 2: Coordinate descent on best so far ---
    if global_a is not None:
        rem = deadline - time.monotonic()
        t2 = min(700, rem * 0.35)
        if t2 > 30:
            n2 = len(global_a)
            S2 = np.sqrt(2.0 * n2)
            a2 = global_a * (S2 / global_a.sum())
            print(f"\nPhase 2: Coord descent n={n2}, t={t2:.0f}s")
            a_cd = coordinate_descent(a2, t2, rng, verbose=True)
            update_global(a_cd, "P2 CD")

            # LSE after CD
            rem = deadline - time.monotonic()
            t2b = min(150, rem * 0.10)
            if t2b > 15:
                a_pol2, _ = lse_optimize(a_cd, S2, t2b, rng, beta_init=200.0, verbose=True)
                update_global(a_pol2, "P2b LSE post-CD")

    # --- Phase 3: Try larger length n=16384 ---
    rem = deadline - time.monotonic()
    if rem > 120:
        n3 = 16384
        S3 = np.sqrt(2.0 * n3)
        t3 = min(500, rem * 0.25)
        if t3 > 30:
            print(f"\nPhase 3: New length n={n3}, t={t3:.0f}s")
            if global_a is not None:
                base = global_a
            else:
                base = np.ones(256)
            t_old = np.linspace(0, 1, len(base))
            t_new = np.linspace(0, 1, n3)
            init3 = np.interp(t_new, t_old, base)
            init3 = np.maximum(init3, 0)
            init3 = init3 * (S3 / init3.sum()) if init3.sum() > 0 else np.ones(n3) * S3 / n3

            a3, _ = subgradient_optimize(init3, S3, t3 * 0.5, rng, verbose=False)
            update_global(a3, "P3 subgrad n=16384")

            rem = deadline - time.monotonic()
            t3b = min(200, rem * 0.30)
            if t3b > 20:
                a3c = coordinate_descent(a3, t3b, rng, verbose=False)
                update_global(a3c, "P3 CD n=16384")

    # --- Phase 4: Ideal init + optimization on n=8192 ---
    rem = deadline - time.monotonic()
    if rem > 100:
        n4 = 8192
        S4 = np.sqrt(2.0 * n4)
        a_ideal = generate_ideal_init(n4)
        a_ideal = a_ideal * (S4 / a_ideal.sum())
        t4 = min(300, rem * 0.20)
        if t4 > 20:
            print(f"\nPhase 4: Ideal init n={n4}, t={t4:.0f}s")
            a4, _ = subgradient_optimize(a_ideal, S4, t4 * 0.5, rng, verbose=False)
            update_global(a4, "P4 ideal subgrad")
            rem = deadline - time.monotonic()
            if rem > 20:
                a4c = coordinate_descent(a4, min(100, rem * 0.5), rng, verbose=False)
                update_global(a4c, "P4 ideal CD")

    # --- Phase 5: Intensive refinement on global best ---
    rem = deadline - time.monotonic()
    if global_a is not None and rem > 60:
        n5 = len(global_a)
        S5 = np.sqrt(2.0 * n5)
        t5 = rem - 30
        print(f"\nPhase 5: Final refinement n={n5}, t={t5:.0f}s")

        a_curr = global_a.copy() * (S5 / global_a.sum())
        t_sub = t5 * 0.35
        t_cd = t5 * 0.40
        t_lse = t5 * 0.20

        a_curr, _ = subgradient_optimize(a_curr, S5, t_sub, rng, verbose=False)
        update_global(a_curr, "P5 subgrad")

        a_curr = coordinate_descent(a_curr, t_cd, rng, verbose=False)
        update_global(a_curr, "P5 CD")

        if t_lse > 10:
            S5b = global_a.sum() if global_a.sum() > 0.01 else S5
            a_curr, _ = lse_optimize(a_curr, S5b, t_lse, rng, beta_init=500.0, verbose=False)
            update_global(a_curr, "P5 LSE")

    # --- Phase 6: Perturbation restarts ---
    rem = deadline - time.monotonic()
    if global_a is not None and rem > 40:
        n6 = len(global_a)
        S6 = np.sqrt(2.0 * n6)
        n_restarts = max(1, int(rem / 80))
        t_per = (rem - 20) / n_restarts
        print(f"\nPhase 6: {n_restarts} restarts, {t_per:.0f}s each")

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            noise_scale = 0.015 * S6 / np.sqrt(n6)
            a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n6), S6)
            a_pert, _ = subgradient_optimize(a_pert, S6, t_per * 0.5, rng, verbose=False)
            update_global(a_pert, f"P6 restart {ri}")
            rem2 = deadline - time.monotonic()
            if rem2 > 20 and global_a is not None:
                a_pert2 = coordinate_descent(a_pert, min(60, rem2 * 0.4), rng, verbose=False)
                update_global(a_pert2, f"P6 restart {ri} CD")

    # --- Final: one more LSE polish ---
    rem = deadline - time.monotonic()
    if global_a is not None and rem > 15:
        nf = len(global_a)
        Sf = np.sqrt(2.0 * nf)
        af = global_a * (Sf / global_a.sum())
        af, _ = lse_optimize(af, Sf, min(rem - 5, 60), rng, beta_init=1000.0, verbose=False)
        update_global(af, "Final LSE")

    # --- Return best ---
    if global_a is None:
        print("WARNING: fallback")
        global_a = np.ones(256)
        global_best = compute_obj(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print(f"\n{'=' * 60}")
    print(f"FINAL: len={len(global_a)}, internal={global_best:.8f}, eval={ver:.8f}")
    print(f"{'=' * 60}")
    return final_list
