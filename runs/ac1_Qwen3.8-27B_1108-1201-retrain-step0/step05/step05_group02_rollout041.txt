Actually, there's another idea. The objective is:
f(a) = 2n * max_k c[k] / (sum a)^2

By the theory of sequence design (Sidon sets, Golay complements, etc.), sequences with low autocorrelation peaks exist. For non-negative sequences, the problem is related to "flat" power spectra.

A key insight: if a has a flat power spectrum (i.e., |FFT(a)|^2 is constant), then the autocorrelation is a delta function, meaning c[0] = sum(a^2) and c[k] = 0 for k != 0. But for non-negative a, we can't have c[k] = 0 for k != 0 (since all terms in the sum are non-negative). So the best we can do is make c[k] as uniform as possible.

In the limit, the optimal solution would have all c[k] equal (except c[0] which must be larger since it's sum of a[i]^2). The objective would then be approximately 2n * (S^2/(2n-1)) / S^2 = 2n/(2n-1) ≈ 1.

But in practice, for non-negative sequences, we can't achieve perfectly flat autocorrelation. The best we can do is related to the "Barker sequence" type problems but for non-negative reals.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

"""
Minimize 2*n*max_conv/sum^2 using ACL (Alternating Convex Linearization) with LP sub-solves,
combined with LSE-FISTA for polishing and multi-start diversity.
The ACL step linearizes the quadratic convolution at the current point and solves an LP
to find the best direction in the linearized problem, then verifies actual improvement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def compute_conv_and_obj(a):
    """Compute convolution and objective together."""
    n = len(a)
    s = a.sum()
    c = fft_convolve_self(a)
    obj = 2.0 * n * c.max() / (s * s) if s > 1e-15 else np.inf
    return c, obj


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def build_acl_rows(a, active_indices):
    """Build constraint rows for ACL LP. Returns (A_ub, b_ub) arrays."""
    n = len(a)
    m = len(active_indices)
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    c = fft_convolve_self(a)
    
    for i, kk in enumerate(active_indices):
        kk = int(kk)
        # Row: 2*a[kk-j] for j where 0<=kk-j<n, i.e., kk-n+1 <= j <= kk
        m_start = max(0, kk - n + 1)
        m_end = min(n - 1, kk)
        if m_start <= m_end:
            a_idx = kk - np.arange(m_start, m_end + 1)
            A_ub[i, m_start:m_end + 1] = 2.0 * a[a_idx]
        A_ub[i, n] = -1.0
        b_ub[i] = c[kk]
    
    return A_ub, b_ub, c


def acl_lp_step(a, S, K=100, tol=1e-8):
    """
    One ACL step: solve LP to minimize linearized peak.
    Returns (a_new, objective_of_a_new, c).
    """
    n = len(a)
    c, cur_obj = compute_conv_and_obj(a)
    
    # Select top-K active constraints
    K = min(K, 2 * n - 1)
    active_idx = np.argsort(c)[::-1][:K]
    
    # Build LP
    A_ub, b_ub, c = build_acl_rows(a, active_idx)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None, cur_obj, c
    
    if result.status != 0:
        return None, cur_obj, c
    
    a_new = result.x[:n].copy()
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return None, cur_obj, c
    a_new = a_new * (S / s_new)
    
    obj_new = compute_obj(a_new)
    return a_new, obj_new, c


def run_acl(a_init, S, max_time, rng, K_start=50, K_end=200, verbose=True):
    """Run ACL iterations."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    c, _ = compute_conv_and_obj(a)
    
    start = time.monotonic()
    it = 0
    last_improve = 0
    failed_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptively increase K
        progress = min(1.0, it / 100.0)
        K = int(K_start + (K_end - K_start) * progress)
        
        a_new, obj_new, c = acl_lp_step(a, S, K=K)
        
        if a_new is not None and obj_new < best_obj - 1e-12:
            a = a_new
            best_obj = obj_new
            best_a = a.copy()
            last_improve = it
            failed_count = 0
            if verbose and it % 20 == 0:
                print("    ACL it %d: obj=%.6f (K=%d)" % (it, best_obj, K))
        elif a_new is not None:
            # LP solution didn't improve; try damped step
            for damp in [0.5, 0.2, 0.1, 0.05]:
                a_trial = (1 - damp) * a + damp * a_new
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial * (S / a_trial.sum())
                obj_trial = compute_obj(a_trial)
                if obj_trial < best_obj - 1e-12:
                    a = a_trial
                    best_obj = obj_trial
                    best_a = a.copy()
                    last_improve = it
                    failed_count = 0
                    break

                failed_count += 1

def lse_fista(a_init, S, max_time, rng, beta_init=50.0, beta_rate=0.005,
              beta_max=50000.0, step_scale=1.0):
    """FISTA with LSE smoothing."""
    n = len(a_init)
    a = np.clip(a_init, 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cm = c.max()
        shifted = beta * (c - cm)
        shifted -= shifted.max()
        e = np.exp(shifted)
        tot = e.sum()
        if tot < 1e-300:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / tot
        
        # Gradient: 2 * ifft(fft(w) * conj(fft(y)))
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        Y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]
        
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search (limited)
        step_local = step
        x_trial = project_simplex(y - step_local * grad, S)
        obj_trial = compute_obj(x_trial)
        bt = 0
        while obj_trial > cur_obj + 1e-10 and step_local > 1e-14 and bt < 4:
            step_local *= 0.5
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt += 1
        if bt > 1:
            step = step_local * 0.7
        else:
            step = min(step * 1.05, 2.0 / max(beta, 1.0))
        
        x_new = x_trial
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        if it - last_improve > 3000:
            t_mom = 1.0
            ns = max(1e-7, 0.002 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, ns, n), S)
    
    return best_a, best_obj

def spectral_init(n, rng, method='flat'):
    """Create initial sequence with approximately flat power spectrum."""
    S = np.sqrt(2.0 * n)
    if method == 'flat':
        # Random phases -> flat power spectrum
        phases = rng.uniform(0, 2 * np.pi, n)
        amp = np.ones(n)
        a = np.real(ifft(amp * np.exp(1j * phases)))
    elif method == 'cosine':
        t = np.linspace(0, 1, n, endpoint=False)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    elif method == 'random':
        a = rng.random(n) + 0.1
    elif method == 'chirp':
        t = np.linspace(0, 1, n, endpoint=False)
        a = 0.5 + 0.5 * np.sin(2 * np.pi * (3 * t + t**2))
    elif method == 'multi_tone':
        t = np.linspace(0, 1, n, endpoint=False)
        a = (0.25 + 0.2 * np.sin(2*np.pi*3*t) + 0.15*np.sin(2*np.pi*7*t+1)
             + 0.1*np.sin(2*np.pi*11*t+2) + 0.1*np.sin(2*np.pi*13*t+3))
        a = np.abs(a) + 0.05
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())

            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 3
        elif stuck < 2000:
            k = min(16, 3 + stuck // 200)
        else:
            k = min(64, 16 + stuck // 500)
        
        idx = np.argsort(c)[::-1][:k]
        w_full = np.zeros(len(c))
        w_full[idx] = 1.0 / k
        
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = S / (gnorm * (1.0 + 0.003 * it))
        step = min(step, 2.0 * S / gnorm)
        
        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-12:
            a = a_new
        else:
            step *= 0.2
            a_new = project_simplex(a - step * grad, S)
            if compute_obj(a_new) <= cur_obj + 1e-12:
                a = a_new
        
        if stuck > 2000 and it % 50 == 0:
            ns = max(1e-7, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 5000))
            a = project_simplex(a + rng.normal(0, ns, n), S)
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses ACL+LP as primary method with LSE-FISTA and subgradient for polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)
    
    # Get starting point
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            raise ValueError
    except Exception:
        n = 8192
        a_start = spectral_init(n, rng, 'cosine')
    
    n = len(a_start)
    S = np.sqrt(2.0 * n)
    a_start = a_start * (S / a_start.sum())
    
    best_obj = compute_obj(a_start)
    best_a = a_start.copy()
    print("Initial: n=%d, obj=%.6f" % (n, best_obj))
    
    # === Phase 1: Quick FISTA to get a good starting point (10% time) ===
    remaining = deadline - time.monotonic()
    p1_time = min(remaining * 0.10, 120)
    print("\n=== Phase 1: FISTA warmup (%.0fs) ===" % p1_time)
    if p1_time > 10:
        a_f, obj_f = lse_fista(a_start, S, p1_time, rng,
                                beta_init=30, beta_rate=0.008, beta_max=20000,
                                step_scale=1.0)
        print("  FISTA warmup: obj=%.6f" % obj_f)
        if obj_f < best_obj:
            best_obj = obj_f
            best_a = a_f.copy()
    
    # === Phase 2: Multi-K ACL (40% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.40
    print("\n=== Phase 2: Multi-K ACL (%.0fs) ===" % p2_time)
    if p2_time > 30:
        a_acl, obj_acl = multi_k_acl(best_a, S, p2_time, rng)
        print("  Multi-K ACL: obj=%.6f" % obj_acl)
        if obj_acl < best_obj:
            best_obj = obj_acl
            best_a = a_acl.copy()
    
    # === Phase 3: FISTA refinement (20% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.20
    print("\n=== Phase 3: FISTA refinement (%.0fs) ===" % p3_time)
    if p3_time > 15:
        # Multiple FISTA variants
        variants = [
            (100, 0.008, 50000, 0.8),
            (500, 0.003, 100000, 0.4),
            (2000, 0.001, 200000, 0.2),
        ]
        for i, (bi, br, bm, ss) in enumerate(variants):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            seg = min(p3_time / len(variants), rem - 5)
            if seg < 5:
                break
            a_f2, obj_f2 = lse_fista(best_a, S, seg, rng,
                                      beta_init=bi, beta_rate=br,
                                      beta_max=bm, step_scale=ss)
            print("  FISTA v%d: obj=%.6f" % (i+1, obj_f2))
            if obj_f2 < best_obj:
                best_obj = obj_f2
                best_a = a_f2.copy()
    
    # === Phase 4: More ACL with higher K (15% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.15
    print("\n=== Phase 4: ACL high-K (%.0fs) ===" % p4_time)
    if p4_time > 20:
        a_acl2, obj_acl2 = run_acl(best_a, S, p4_time, rng,
                                    K_start=200, K_end=500, verbose=True)
        print("  ACL high-K: obj=%.6f" % obj_acl2)
        if obj_acl2 < best_obj:
            best_obj = obj_acl2
            best_a = a_acl2.copy()
    
    # === Phase 5: Final polish (10% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.12
    print("\n=== Phase 5: Subgradient polish (%.0fs) ===" % p5_time)
    if p5_time > 10:
        a_pol, obj_pol = subgradient_polish(best_a, S, p5_time, rng)
        print("  Subgradient: obj=%.6f" % obj_pol)
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()
    
    # === Phase 6: Final ACL squeeze ===
    remaining = deadline - time.monotonic()
    p6_time = remaining * 0.5
    if p6_time > 10:
        print("\n=== Phase 6: Final ACL (%.0fs) ===" % p6_time)
        a_fin, obj_fin = run_acl(best_a, S, p6_time, rng,
                                  K_start=100, K_end=300, verbose=True)
        print("  Final ACL: obj=%.6f" % obj_fin)
        if obj_fin < best_obj:
            best_obj = obj_fin
            best_a = a_fin.copy()
    
    # === Phase 7: Try different n (if time remains) ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and n == 8192:
        print("\n=== Phase 7: Try n=4096 (%.0fs) ===" % remaining * 0.5)
        n2 = 4096
        S2 = np.sqrt(2.0 * n2)
        # Downsample
        idx = np.linspace(0, n - 1, n2).astype(int)
        a2 = a_start[idx]
        a2 = a2 * (S2 / a2.sum())
        obj2 = compute_obj(a2)
        if obj2 < best_obj:
            print("  n=4096 initial: obj=%.6f" % obj2)
            # Optimize
            a2_f, obj2_f = lse_fista(a2, S2, remaining * 0.3, rng,
                                       beta_init=50, beta_rate=0.005,
                                       beta_max=20000, step_scale=1.0)
            print("  n=4096 FISTA: obj=%.6f" % obj2_f)
            if obj2_f < best_obj:
                best_obj = obj2_f
                best_a = a2_f.copy()
    
    # Final verification
    print("\n=== Final: obj=%.6f, n=%d ===" % (best_obj, len(best_a)))
    final_list = [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: %.6f" % ver)
    else:
        print("WARNING: invalid, using fallback")
        n_fb = 8192
        final_list = [float(np.sqrt(2 * n_fb) / n_fb)] * n_fb
    
    return final_list
