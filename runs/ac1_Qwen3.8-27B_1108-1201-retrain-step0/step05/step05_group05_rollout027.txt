    """Find the KKT-optimal subgradient direction."""
    n = len(a)
    cmax = c.max()
    # Active lags
    active = np.where(c >= cmax - 1e-4 * cmax)[0]
    active = active[:50]  # Limit for efficiency
    
    # Support of a
    support = np.where(a > support_tol * a.max())[0]
    
    # For each active lag k, the subgradient contribution at position i is:
    # 2 * a[k-i] if 0 <= k-i < n
    # We want to find weights w such that sum_k w_k * grad_k is "most constant" on support
    
    # Compute all subgradients
    m = len(active)
    grads = np.zeros((m, len(support)))
    for idx, k in enumerate(active):
        for j, i in enumerate(support):
            jk = k - i
            if 0 <= jk < n:
                grads[idx, j] = 2 * a[jk]
    
    # Find w >= 0, sum w = 1, that minimizes ||grads.T @ w - mean(grads.T @ w)||^2
    # This is equivalent to minimizing the variance of the weighted average
    # = minimizing w^T (grads @ grads.T) w - (w^T grads @ 1)(1^T grads @ w) / n_su
    
    # Simpler: use the weights proportional to c[k] for active k
    # Or: solve the QP (but that might be slow)
    
    # For efficiency, use the simple heuristic: weights proportional to c[k] - mean(c)
    w = np.ones(m) / m
    
    # Combine
    grad = np.zeros(n)
    for idx, k in enumerate(active):
        for i in range(n):
            jk = k - i
            if 0 <= jk < n:
                grad[i] += w[idx] * 2 * a[jk]
    
    return grad

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using subgradient descent with
Nesterov momentum, backtracking line search, multi-peak soft weighting,
and periodic Fourier perturbation. Different from previous approach by
using direct subgradients (no LSE smoothing) with adaptive step sizes.

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Use the sorted approach (exact)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, c, soft_width=0.005):
    """
    Compute soft subgradient of max(c) at a.
    Uses exponential soft weighting centered at the max.
    """
    n = len(a)
    cmax = c.max()
    scale = max(soft_width * cmax, 1e-10)
    # Soft weights: peaks closer to max get more weight
    weights = np.exp(-np.maximum(cmax - c, 0.0) / scale)
    wsum = weights.sum()
    if wsum < 1e-30:
        k_star = np.argmax(c)
        weights = np.zeros(len(c))
        weights[k_star] = 1.0
    else:
        weights /= wsum
    
    # Compute gradient via FFT
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def objective_nonscaled(a):
    """Return max(conv(a,a)) for a with fixed sum."""
    c = fft_convolve_self(a)
    return c.max()

def run_subgradient_descent(a_init, S, max_time, rng, soft_width=0.005):
    """
    Nesterov accelerated subgradient descent with backtracking.
    """
    n = len(a_init)
    a = project_simplex(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c = fft_convolve_self(y)
    best = c.max()
    best_a = y.copy()
    it = 0
    last_improve = 0
    step_size = 0.01
    patience = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cmax = c.max()
        
        if cmax < best - 1e-12:
            best = cmax
            best_a = y.copy()
            last_improve = it
            patience = 0
        else:
            patience += 1
        
        # Adaptive soft width: tighten over time
        effective_sw = soft_width * np.exp(-it / 5000.0) + 0.001
        grad = compute_subgradient(y, c, soft_width=effective_sw)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = step_size
        improved = False
        for _ in range(10):
            x_trial = project_simplex(y - step * grad, S)
            c_trial = fft_convolve_self(x_trial)
            if c_trial.max() < cmax - 1e-10:
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # If backtracking fails, try a smaller fixed step
            step = step_size * 0.1
            x_trial = project_simplex(y - step * grad, S)
        
        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        momentum_coeff = (t_mom - 1.0) / t_new
        a_new = x_trial
        y_new = a_new + momentum_coeff * (a_new - a)
        y_new = np.maximum(y_new, 0.0)
        # Re-project to maintain sum
        if y_new.sum() > 0:
            y_new = y_new * (S / y_new.sum())
        a = a_new
        y = y_new
        t_mom = t_new
        
        # Periodic perturbation
        if patience > 3000:
            noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-patience / 10000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                patience = 0
                last_improve = it
        
        # Every 5000 iters, try a Fourier perturbation
        if it % 5000 == 0 and patience > 1000:
            fsize = 1 << (2 * n - 1).bit_length()
            A = fft(y, fsize)
            mag = np.abs(A)
            phase = np.angle(A)
            scale = 0.05
            phase_noise = rng.normal(0, scale, fsize)
            new_A = mag * np.exp(1j * (phase + phase_noise))
            new_A[0] = A[0]
            y_cand = np.real(ifft(new_A, fsize))[:n]
            y_cand = project_simplex(y_cand, S)
            c_cand = fft_convolve_self(y_cand)
            if c_cand.max() < cmax * 1.01:  # Accept if not much worse
                y = y_cand
                t_mom = 1.0
                patience = 0
    
    return best_a, best

def run_lse_fista(a_init, S, max_time, rng, beta_start=100, beta_end=100000):
    """LSE-FISTA for final polishing."""
    n = len(a_init)
    a = project_simplex(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c = fft_convolve_self(y)
    best = c.max()
    best_a = y.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best - 1e-12:
            best = cmax
            best_a = y.copy()
            last_improve = it
        
        # Adaptive beta
        progress = min(1.0, it / max(1, (max_time / 0.05)))  # roughly 20 iters per 0.05s
        beta = beta_start * (beta_end / beta_start) ** progress
        beta = min(beta, beta_end)
        
        # LSE gradient
        scaled = beta * (c - cmax)
        e = np.exp(np.maximum(scaled, -500))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (2.0 * beta + 1e-10)
        
        if it - last_improve > 3000:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    return best_a, best

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} exactly."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient_fft(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""

        lr = 1.0 / (2.0 * np.linalg.norm(G.T @ G, ord=np.inf) / ns + 1e-10)
        w_new = w_qp - lr * grad_qp
        w_new = project_simplex(w_new, 1.0)
        
        if np.linalg.norm(w_new - w_qp) < 1e-10:
            break
        w_qp = w_new
    
    # Compute final subgradient
    fsize = 1 << (len(c) + n - 1).bit_length()
    final_w = np.zeros(len(c))
    final_w[active] = w_qp
    W = fft(final_w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def lse_fista_sharp(a_init, S, max_time, rng, beta_start=100.0, beta_end=500000.0,
                    perturb_every=3000, perturb_scale=0.003):
    """
    LSE-FISTA with sharp initial beta for fast convergence to the max structure.
    """
    n = len(a_init)
    a = project_simplex(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c = fft_convolve_self(y)
    best = c.max()
    best_a = y.copy()
    it = 0
    last_improve = 0
    stuck_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cmax = c.max()
        
        if cmax < best - 1e-15:
            best = cmax
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        
        # Adaptive beta: logarithmic ramp
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)
        beta = min(beta, beta_end)
        
        grad, beta_eff = lse_gradient_fft(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = 1.0 / (2.5 * beta_eff + 1e-10)
        
        # Stuck detection
        if it - last_improve > perturb_every:
            stuck_count += 1
            if stuck_count > 5:
                # Fourier perturbation
                scale = perturb_scale * S / np.sqrt(n) * np.exp(-stuck_count / 10.0)
                if scale > 1e-12:
                    fsize = 1 << (2 * n - 1).bit_length()
                    A = fft(y, fsize)
                    mag = np.abs(A)
                    phase = np.angle(A)
                    phase_noise = rng.normal(0, scale * 10.0, fsize)
                    # More noise at high frequencies
                    freqs = np.arange(fsize)
                    freq_weight = 1.0 + 3.0 * freqs / fsize
                    phase_noise *= freq_weight
                    new_A = mag * np.exp(1j * (phase + phase_noise))
                    new_A[0] = A[0]
                    y_cand = np.real(ifft(new_A, fsize))[:n]
                    y_cand = project_simplex(y_cand, S)
                    c_cand = fft_convolve_self(y_cand)
                    if c_cand.max() < cmax * 1.05:
                        y = y_cand
                        t_mom = 1.0
                        last_improve = it
                        stuck_count = 0
                        continue
                # Fallback: small Gaussian noise
                noise_scale = perturb_scale * S / np.sqrt(n) * 0.1
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                last_improve = it
                stuck_count = 0
                continue
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    return best_a, best


def subgradient_polish(a_init, S, max_time, rng, top_k=16, smooth_frac=0.003):
    """
    Direct subgradient method with KKT-optimal weighting and backtracking.
    Uses Nesterov momentum.
    """
    n = len(a_init)
    a = project_simplex(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c = fft_convolve_self(y)
    best = c.max()
    best_a = y.copy()
    it = 0
    last_improve = 0
    step_size = 0.5 * S / n
    prev_grad = None
    prev_step = None
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cmax = c.max()
        
        if cmax < best - 1e-14:
            best = cmax
            best_a = y.copy()
            last_improve = it
        
        # Compute KKT-weighted subgradient
        grad = kkt_weighted_subgradient(y, c, top_k=top_k, smooth_frac=smooth_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Barzilai-Borwein step size (if we have previous iterate)
        if prev_grad is not None and prev_step is not None:
            s_vec = y - prev_step
            y_vec = grad - prev_grad
            sTs = s_vec @ s_vec
            sTy = s_vec @ y_vec
            if sTs > 1e-30:
                if abs(sTy) > 1e-30:
                    step_bb = sTs / abs(sTy)
                    step_bb = np.clip(step_bb, 0.1 * S / n, 10.0 * S / n)
                else:
                    step_bb = step_size
                # Blend BB with previous step
                step_size = 0.5 * step_size + 0.5 * step_bb
        
        # Backtracking line search
        step = step_size
        found = False
        for _ in range(8):
            x_trial = project_simplex(y - step * grad, S)
            c_trial = fft_convolve_self(x_trial)
            if c_trial.max() < cmax * (1.0 - 1e-8):
                found = True
                break
            step *= 0.6
        
        if not found:
            x_trial = project_simplex(y - step * grad, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        momentum_coeff = (t_mom - 1.0) / t_new
        a_new = x_trial
        y_new = a_new + momentum_coeff * (a_new - a)
        y_new = np.maximum(y_new, 0.0)
        if y_new.sum() > 0:
            y_new = y_new * (S / y_new.sum())
        a = a_new
        y = y_new
        t_mom = t_new
        prev_grad = grad.copy()
        prev_step = a_new.copy()
        
        # Periodic perturbation if stuck
        if it - last_improve > 4000:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 20000.0)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                last_improve = it
                prev_grad = None
                prev_step = None
    
    return best_a, best


def basin_hopping_fourier(a_init, S, max_time, rng):
    """
    Basin hopping: Fourier-domain mutations followed by short LSE-FISTA.
    """
    n = len(a_init)
    a = project_simplex(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    
    # Time per iteration
    opt_time = 15.0
    it = 0
    
    # Population for diversity
    population = [(best, a.copy())]
    
    while time.monotonic() - start < max_time - opt_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        
        # Adaptive mutation scale
        base_scale = 0.3 * np.exp(-3.0 * progress) + 0.03
        
        # Choose mutation source
        if len(population) >= 2 and rng.random() < 0.3:
            idx = rng.integers(len(population))
            source = population[idx][1]
        else:
            source = a
        
        # Fourier phase mutation
        fsize = 1 << (2 * n - 1).bit_length()
        A = fft(source, fsize)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Frequency-dependent noise
        scale = base_scale * (1.0 + 0.5 * rng.random())
        freqs = np.arange(fsize, dtype=np.float64)
        freq_weight = 1.0 + 2.0 * freqs / fsize
        phase_noise = rng.normal(0, scale * freq_weight, fsize)
        
        new_A = mag * np.exp(1j * (phase + phase_noise))
        new_A[0] = A[0]  # Preserve DC
        a_cand = np.real(ifft(new_A, fsize))[:n]
        a_cand = project_simplex(a_cand, S)
        
        # Short local optimization
        a_opt, val = lse_fista_sharp(a_cand, S, opt_time, rng,
                                       beta_start=200.0, beta_end=100000.0,
                                       perturb_every=2000, perturb_scale=0.002)
        
        if val < best:
            best = val
            best_a = a_opt.copy()
            print("    BH iter %d: max_b=%.10f (obj=%.6f)" % (it, val, 2.0*n*val/S**2))
        
        # Update population
        population.append((val, a_opt.copy()))
        if len(population) > 6:
            population.sort(key=lambda x: x[0])
            population = population[:6]
    
    return best_a, best
