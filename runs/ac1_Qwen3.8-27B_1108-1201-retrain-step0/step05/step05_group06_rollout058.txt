"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses subgradient optimization with adaptive active set,
KKT-based refinement, and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0)
    # Standard simplex projection
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_from_active(a, c, active_indices):
    """Compute subgradient from active set of convolution lags."""
    n = len(a)
    size = 2 * n - 1
    w_full = np.zeros(size)
    w_full[active_indices] = 1.0 / len(active_indices)
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    # g_k = sum_j w_j * (a_{j-k} + a_{k-j})
    # First term: convolution of w (reversed) with a
    # Second term: convolution of w with a
    # In FFT: ifft(W * A) gives circular conv of w and a (padded)
    # We need linear conv, handled by sufficient padding
    result = 2.0 * np.real(ifft(W * A))[:n]
    return result


def lse_subgradient(a, c, beta):
    """LSE-based smooth approximation of max subgradient."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        active = np.array([int(np.argmax(c))])
        return subgradient_from_active(a, c, active)
    w = e / tot
    size = len(c)
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * A))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_subgradient_opt(a_init, max_time, rng, beta_start=5.0, beta_end=5000.0, 
                        active_tol_start=0.05, active_tol_end=0.001):
    """
    Main optimization using subgradient with adaptive active set and LSE.
    Switches from LSE (smooth) to active-set (sharp) as optimization progresses.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()  # FISTA momentum variable
    t_mom = 1.0
    
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / S**2  # since sum = S
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        progress = min(1.0, it / max(1, it + stuck))  # 0 if completely stuck
        
        # Adaptive parameters
        if stuck < 500:
            mode = 'lse'
            beta = beta_start * (1.0 + 0.01 * it)
            tol = active_tol_start
        elif stuck < 3000:
            mode = 'lse'
            beta = min(beta_end, beta_start * np.exp(0.005 * it))
            tol = max(active_tol_end, active_tol_start * np.exp(-0.002 * stuck))
        else:
            mode = 'active'
            tol = active_tol_end * (1.0 + 0.01 * (stuck - 3000) / 3000)
            tol = min(tol, 0.05)
        
        # Compute subgradient
        if mode == 'lse':
            grad = lse_subgradient(y, c, beta)
        else:
            max_val = c.max()
            active = np.where(c >= max_val * (1.0 - tol))[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(c))])
            grad = subgradient_from_active(y, c, active)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # KKT-based step size
        # We want to make the subgradient more uniform
        # lambda = mean of grad on support
        support = y > 1e-10
        if np.sum(support) > 0:
            lam = np.mean(grad[support])
        else:
            lam = np.mean(grad)
        
        # Direction: reduce variance of subgradient
        direction = grad - lam
        # Project direction to maintain sum
        direction = direction - np.mean(direction)
        
        # Step size via simple heuristics
        step = S / (gnorm * np.sqrt(n))
        step = min(step, 0.5 / max(beta if mode == 'lse' else 10.0, 1.0))
        
        # Try the step
        x_new = project_simplex(y - step * direction, S)
        
        # Backtracking if needed
        for _ in range(5):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max * (1 + 1e-10):
                break
            step *= 0.5
            x_new = project_simplex(y - step * direction, S)
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Renormalize
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        a = x_new
        t_mom = t_new
        
        # Escape mechanism
        if stuck > 5000 and it % 100 == 0:
            noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                a = y.copy()
            
        # Periodic restart
        if it % 10000 == 0:
            t_mom = 1.0
    
    return best_a, best_obj


def run_coordinate_descent(a_init, max_time, rng, n_sweeps=None):
    """
    Coordinate descent: optimize pairs of coordinates to reduce max convolution.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    sweep = 0
    
    while time.monotonic() - start < max_time:
        sweep += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / S**2
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Find the lag with maximum convolution
        j_max = int(np.argmax(c))
        
        # Try to reduce c[j_max] by adjusting coordinates
        # c[j_max] = sum_k a_k * a_{j_max - k}
        # To reduce it, decrease a_k or a_{j_max-k} for large terms
        
        # Get the top contributors to c[j_max]
        k_idx = np.arange(n)
        valid_k = (k_idx >= 0) & (j_max - k_idx >= 0) & (j_max - k_idx < n)
        k_valid = k_idx[valid_k]

                if target == top_k:

                if c_new.max() < cur_max:

def run_kkt_polish(a_init, max_time, rng):
    """
    KKT-based polishing: iteratively adjust a to make the subgradient
    from the active set as uniform as possible.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    tol = 0.01  # Active set tolerance
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / S**2
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Identify active set
        active = np.where(c >= cur_max * (1.0 - tol))[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # Compute subgradient
        grad = subgradient_from_active(a, c, active)
        
        # Target: make grad uniform on support
        support = a > 1e-10
        if np.sum(support) < 2:
            break
        lam = np.mean(grad[support])
        
        # Newton-like step: we want to find delta such that
        # grad + J * delta is more uniform
        # Approximate J as diagonal: J_kk ~ 2*a_k (self-contribution)
        # This gives: new_grad_k ~ grad_k + 2*a_k*delta_k
        # We want new_grad_k = lam for all k in support
        # So delta_k = (lam - grad_k) / (2*a_k)
        
        delta = np.zeros(n)
        mask = support & (a > 1e-10)
        delta[mask] = (lam - grad[mask]) / (2.0 * a[mask])
        
        # Limit the step
        max_delta = np.max(np.abs(delta))
        if max_delta > 0:
            scale = min(1.0, (0.1 * S) / (n * max_delta))
            delta = delta * scale
        
        # Apply and project
        a_new = a + delta
        a_new = project_simplex(a_new, S)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max * (1.0 + 1e-10):
            a = a_new
        else:
            # Try smaller step
            a_new = a + delta * 0.5
            a_new = project_simplex(a_new, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max * (1.0 + 1e-10):
                a = a_new
            else:
                a = a_new  # Accept anyway to make progress
        
        # Adapt tolerance
        if stuck > 2000:
            tol = max(0.001, tol * 0.95)
        elif stuck < 100:
            tol = min(0.05, tol * 1.05)
        
        # Escape
        if stuck > 5000 and it % 50 == 0:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
    
    return best_a, best_obj


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)


def make_hann_init(n):
    tt = np.arange(n)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt / n)


def make_gaussian_init(n, sigma_frac=0.3):
    tt = np.arange(n) - n / 2
    sigma = sigma_frac * n
    return np.exp(-tt**2 / (2 * sigma**2))


def make_bump_init(n):
    """Raised cosine bump."""
    tt = np.linspace(-1, 1, n)
    return 0.5 + 0.5 * np.cos(np.pi * tt)

    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0.0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_active(a, c, active_idx):
    """Compute subgradient from a set of active convolution lags."""
    n = len(a)
    size = 2 * n - 1
    w_full = np.zeros(size)
    w_full[active_idx] = 1.0 / len(active_idx)
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * A))[:n]


def lse_subgradient(a, c, beta):
    """LSE-smoothed subgradient of max(c)."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return subgradient_active(a, c, np.array([int(np.argmax(c))]))
    w = e / tot
    size = len(c)
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * A))[:n]


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_subgradient_opt(a_init, max_time, rng, beta_start=5.0, beta_end=5000.0,
                        active_tol_start=0.05, active_tol_end=0.002):
    """
    Main subgradient optimization with adaptive active set, FISTA,
    and backtracking line search.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0

    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_start

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive mode and parameters
        if stuck < 200:
            mode = 'lse'
            beta = beta_start * (1.0 + 0.02 * it)
            tol = active_tol_start
        elif stuck < 2000:
            mode = 'lse'
            beta = min(beta_end, beta * 1.01)
            tol = max(active_tol_end, active_tol_start * np.exp(-0.003 * stuck))
        elif stuck < 10000:
            mode = 'active'
            tol = max(active_tol_end, active_tol_start * np.exp(-0.005 * stuck))
            tol = min(tol, 0.05)
        else:
            mode = 'active'
            tol = active_tol_end * (1.0 + 0.05 * (stuck - 10000) / 10000.0)
            tol = min(tol, 0.1)

        # Compute subgradient
        if mode == 'lse':
            beta = min(beta, beta_end)
            grad = lse_subgradient(y, c, beta)
        else:
            max_val = c.max()
            active = np.where(c >= max_val * (1.0 - tol))[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(c))])
            grad = subgradient_active(y, c, active)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Direction: make subgradient uniform on support
        support = y > 1e-12
        n_support = np.sum(support)
        if n_support < 2:
            break
        lam = np.mean(grad[support])
        direction = grad.copy()
        direction[support] -= lam
        direction = direction - np.mean(direction)

        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-15:
            break

        # Step size
        step = S / (dnorm * np.sqrt(n))
        if mode == 'lse':
            step = min(step, 0.8 / max(beta, 1.0))
        else:
            step = min(step, 0.5)

        # Backtracking line search
        x_new = project_simplex(y - step * direction, S)
        accepted = False
        for bt in range(8):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max * (1.0 + 1e-10):
                accepted = True
                break
            step *= 0.5
            x_new = project_simplex(y - step * direction, S)
        if not accepted:
            x_new = project_simplex(y - step * direction, S)

        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new = y_new * (S / s_y)
        y = y_new
        a = x_new
        t_mom = t_new

        # Escape mechanism
        if stuck > 3000 and it % 200 == 0:
            noise_scale = 0.02 * S / np.sqrt(n) * np.exp(-stuck / 8000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = y.copy()
                t_mom = 1.0

        # Periodic momentum restart
        if it % 15000 == 0:
            t_mom = 1.0

    return best_a, best_obj


def run_kkt_polish(a_init, max_time, rng):
    """
    KKT-based polishing: iteratively minimize the variance of the active-set
    subgradient on the support (which is the KKT residual).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)

    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    tol = 0.02

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve

        # Active set
        active = np.where(c >= cur_max * (1.0 - tol))[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Subgradient
        grad = subgradient_active(a, c, active)

        # KKT target: make grad uniform on support
        support = a > 1e-10
        n_sup = np.sum(support)
        if n_sup < 2:
            break
        lam = np.mean(grad[support])

        # Newton-like step: delta_k = (lam - grad_k) / (2*a_k)
        delta = np.zeros(n)
        mask = support & (a > 1e-10)
        delta[mask] = (lam - grad[mask]) / (2.0 * a[mask])

        # Limit step magnitude
        max_d = np.max(np.abs(delta))
        if max_d > 1e-30:
            scale = min(1.0, 0.5 * S / (n * max_d))
            delta = delta * scale

        # Apply and verify
        a_new = a + delta
        a_new = project_simplex(a_new, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max * (1.0 + 1e-10):
            a = a_new
        else:
            # Try half step
            a_new = a + delta * 0.5
            a_new = project_simplex(a_new, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max * (1.0 + 1e-10):
                a = a_new
            else:
                # Accept anyway (non-monotone)
                a = project_simplex(a + delta * 0.25, S)

        # Adapt tolerance
        if stuck > 3000:
            tol = max(0.002, tol * 0.98)
        elif stuck < 200:
            tol = min(0.1, tol * 1.02)

        # Escape
        if stuck > 8000 and it % 100 == 0:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n) * np.exp(-stuck / 20000.0), n)
            a = project_simplex(a + noise, S)

    return best_a, best_obj


def run_coord_descent(a_init, max_time, rng):
    """
    Coordinate descent: try pairwise transfers to reduce max convolution.
    Focuses on the lag achieving the maximum.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)

    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    sweep = 0

    while time.monotonic() - start < max_time:
        sweep += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()

        j_max = int(np.argmax(c))

        # Find top contributors to c[j_max]
        k_idx = np.arange(n)
        valid = (k_idx >= 0) & (j_max - k_idx >= 0) & (j_max - k_idx < n)
        k_v = k_idx[valid]
        if len(k_v) == 0:
            break
        contrib = a[k_v] * a[j_max - k_v]

        # Try a few random pair transfers
        n_tries = min(5, len(k_v))
        top_tracks = k_v[np.argsort(contrib)[::-1][:min(10, len(k_v))]]

        for _ in range(n_tries):
            src = top_tracks[rng.integers(0, len(top_tracks))]
            dst = rng.integers(0, n)
            if src == dst:
                continue

            # Try moving a small fraction from src to dst
            for frac in [0.05, 0.1, 0.02]:
                delta = a[src] * frac
                if delta < 1e-15:
                    continue
                a_new = a.copy()
                a_new[src] -= delta
                a_new[dst] += delta
                if a_new[src] < 0:
                    continue
                a_new = normalize_seq(a_new, S)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < cur_max:
                    a = a_new
                    cur_max = c_new.max()
                    break

    return best_a, best_obj


def run_power2_warmup(a_init, max_time, rng):
    """Warmup: minimize sum(c^2) to spread out the convolution."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0

    start = time.monotonic()
    best_a = a.copy()
    best_obj2 = np.sum(fft_convolve_self(y) ** 2)
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        obj2 = np.sum(c * c)
        if obj2 < best_obj2:
            best_obj2 = obj2
            best_a = y.copy()

        # Gradient of sum(c^2) w.r.t. a: 4 * (c * a)[0:n] (circular approx)
        fsize = 1 << (len(c) + n - 1).bit_length()
        C = fft(c, fsize)
        A = fft(y, fsize)
        grad = 4.0 * np.real(ifft(C * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 0.5 / (4.0 * c.max() + 1.0)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        if y.sum() > 1e-15:
            y = y * (S / y.sum())
        a = x_new
        t_mom = t_new

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)


def make_hann_init(n):
    tt = np.arange(n)
    return 0.5 + 0.5 * np.cos(2.0 * np.pi * tt / (n - 1))


def make_gaussian_init(n, sigma_frac=0.3):
    tt = (np.arange(n) - n / 2.0) / (n / 2.0)
    sigma = sigma_frac
    return np.exp(-tt * tt / (2.0 * sigma * sigma))


def make_barker_mod_init(n, rng):
    """Gaussian envelope modulated by a pseudo-random binary sequence."""
    env = make_gaussian_init(n, 0.35)
    # Generate a random +/- 1 sequence
    b = rng.choice([-1.0, 1.0], size=n)
    # Smooth the binary sequence slightly
    # Map to [0, 1]: b_mapped = (b+1)/2 gives {0, 1}
    # But we want a smoother effect: use a raised version
    b_smooth = 0.3 + 0.7 * (b + 1.0) / 2.0
    a = env * b_smooth
    return np.maximum(a, 0.0)


def make_dual_bump_init(n):
    """Two-bump: large at edges, small in middle."""
    tt = np.linspace(0, 1, n)
    a = np.exp(-20.0 * (tt - 0.0) ** 2) + np.exp(-20.0 * (tt - 1.0) ** 2)
    return a + 0.1


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: diverse init -> power2 warmup -> subgradient -> KKT -> coord descent.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)

    global_best_obj = np.inf
    global_best_a = None

    # Try to get the provided sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    # Determine base n
    if a_start is not None:
        n0 = len(a_start)
    else:
        n0 = 8192

    print("Base n = %d" % n0)
    n = n0
    S = np.sqrt(2.0 * n)

    # Generate diverse initializations
    inits = []
    names = []

    if a_start is not None:
        inits.append(normalize_seq(a_start, S))
        names.append("provided")

    inits.append(normalize_seq(make_cosine_init(n), S))
    names.append("cosine")

    inits.append(normalize_seq(make_hann_init(n), S))
    names.append("hann")

    inits.append(normalize_seq(make_gaussian_init(n, 0.25), S))
    names.append("gauss25")

    inits.append(normalize_seq(make_gaussian_init(n, 0.35), S))
    names.append("gauss35")

    inits.append(normalize_seq(make_gaussian_init(n, 0.45), S))
    names.append("gauss45")

    inits.append(normalize_seq(make_barker_mod_init(n, rng), S))
    names.append("barker_mod")

    inits.append(normalize_seq(make_dual_bump_init(n), S))
    names.append("dual_bump")

    # Quick evaluation
    print("\n=== Initialization ===")
    best_init = None
    best_init_obj = np.inf
    for a_init, name in zip(inits, names):
        obj = compute_obj(a_init)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_init.copy()
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a_init.copy()
        print("  %s: obj=%.6f" % (name, obj))

    # Also try a random init
    a_rand = normalize_seq(rng.random(n) ** 3, S)
    obj_rand = compute_obj(a_rand)
    print("  random: obj=%.6f" % obj_rand)
    if obj_rand < best_init_obj:
        best_init = a_rand.copy()
        best_init_obj = obj_rand
    if obj_rand < global_best_obj:
        global_best_obj = obj_rand
        global_best_a = a_rand.copy()

    print("Best init: obj=%.6f" % best_init_obj)

    # === Phase 1: Power-2 warmup ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.08
    if p1_budget > 5:
        print("\n=== Phase 1: Power-2 warmup (%.0fs) ===" % p1_budget)
        t1 = time.monotonic()
        a_p2 = run_power2_warmup(best_init, p1_budget, rng)
        obj_p2 = compute_obj(a_p2)
        print("  obj=%.6f (t=%.1fs)" % (obj_p2, time.monotonic() - t1))
        if obj_p2 < global_best_obj:
            global_best_obj = obj_p2
            global_best_a = a_p2.copy()
        best_init = a_p2

    # === Phase 2: Main subgradient optimization ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.35
    if p2_budget > 20:

    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.25

            global_best_a = a_kkt.copy()

    remaining = deadline - time.monotonic()

        a_cd, _ = run_coord_descent(global_best_a, p4_budget, rng)
        obj_cd = compute_obj(a_cd)
        print("  obj=%.6f (t=%.1fs)" % (obj_cd, time.monotonic() - t4))

            global_best_obj = obj_cd

            global_best_obj = obj_fin
            global_best_a = a_fin.copy()

    # === Phase 6: Final KKT ===
    remaining = deadline - time.monotonic()
    if remaining > 15:

        obj_fin2 = compute_obj(a_fin2)
        print("  Final KKT: obj=%.6f" % obj_fin2)
        if obj_fin2 < global_best_obj:
            global_best_obj = obj_fin2
            global_best_a = a_fin2.copy()

        obj_up = compute_obj(a_up_sg)
        print("  Upsampled: obj=%.6f (t=%.1fs)" % (obj_up, time.monotonic() - t7))
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up_sg.copy()

        rem7 = p7_budget - (time.monotonic() - t7)
        if rem7 > 20:
            a_up2, _ = run_kkt_polish(global_best_a, rem7, rng)
