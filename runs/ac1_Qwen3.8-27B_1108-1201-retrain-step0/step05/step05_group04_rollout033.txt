import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta

def active_set_gradient(a, c, tol=0.01):
    """
    Compute gradient using active set: lags within tol of max.
    Weights proportional to (c[k] - (max - tol))^2 for emphasis on highest.
    """
    n = len(a)
    cm = c.max()
    # Active set: lags within tol of max
    active = c >= (cm - tol)
    if not np.any(active):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        # Weight by how close to max (quadratic emphasis)
        w = np.where(active, (c - (cm - tol)) ** 2, 0.0)
        tot = w.sum()
        if tot < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
            tot = 1.0
        w /= tot
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def dual_multiplicative_update(a, c, n, S, eta=0.1, top_k=None):
    """
    Multiplicative weight update based on dual variables.
    The idea: if the gradient (from active set) is not constant on the support,
    we adjust a multiplicatively to make it more constant.
    """
    cm = c.max()
    # Get active set gradient
    if top_k is not None:
        # Use top_k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / top_k
    else:
        # Use lags within small tol of max
        tol = max(0.001 * cm, cm * 0.005)
        w = np.where(c >= (cm - tol), 1.0, 0.0)
        tot = w.sum()
        if tot < 1:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
            tot = 1.0
        w /= tot
    
    # Compute grad = 2 * sum_k w_k * a[k-j]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    # lambda = mean of grad on support (where a > 0)
    support = a > 1e-10
    if support.sum() == 0:
        return a
    lam = grad[support].mean()
    
    # Multiplicative update: a_new = a * exp(-eta * (grad - lam))
    # But only on the support; off-support stays 0 (or small)
    delta = grad - lam
    a_new = a * np.exp(-eta * delta)
    
    # Project onto simplex
    a_new = project_simplex(a_new, S)
    return a_new

def fista_lse_refined(a_init, n, S, max_time, rng, 
                       beta_start=100.0, beta_end=100000.0, 
                       beta_steps=100, noise_scale=0.002):
    """
    Refined LSE-FISTA with:
    - Faster beta ramp
    - Adaptive noise
    - Periodic restart
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        # Use exponential ramp for faster convergence
        beta = beta_start * (beta_end / beta_start) ** frac
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
        
        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / beta
        
        # Stagnation handling
        stagnant = it - last_improve
        if stagnant > 200:
            t_mom = 1.0
            # Structured noise: shift or flip a component
            if stagnant > 500:
                # Try reversing a random fraction
                frac_rev = 0.1 + 0.2 * rng.random()
                start_idx = rng.integers(0, n)
                end_idx = min(n, start_idx + int(frac_rev * n))
                a_rev = y.copy()
                a_rev[start_idx:end_idx] = y[start_idx:end_idx][::-1]
                a_rev = normalize_sum(a_rev, n)
                y = a_rev
            else:
                noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
                y = project_simplex(y + noise, S)
            last_improve = it
            step *= 0.5
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

    """

    best = np.inf
    best_a = a.copy()
    it = 0

        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Active set gradient
        tol = tol_frac * cur_max
        w = np.where(c >= (cur_max - tol), 1.0, 0.0)
        tot = w.sum()
        if tot < 1:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
            tot = 1.0
        w /= tot
        
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = 0.01
        improved = False
        for _ in range(6):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur_max * (1 - 1e-7):
                a = a_test
                improved = True
                break
            step *= 0.5
        
        if not improved:

            a = project_simplex(a - step * grad, S)
        
        # Periodic perturbation

                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    c_final = fft_convolve_self(a)

        best_a = a.copy()

def multiplicative_refine(a_init, n, S, max_time, rng, eta=0.05, top_k=20):

   b. Transfer a small amount from i to j.

        g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Lambda = mean of g on support
        support = a > 1e-10
        if support.sum() < 2:
            break
        lam = g[support].mean()
        
        # Find indices to transfer from/to
        above = np.where(support & (g > lam + 1e-8))[0]
        below = np.where(support & (g < lam - 1e-8))[0]
        
        if len(above) == 0 or len(below) == 0:
            # KKT approximately satisfied
            break
        
        # Pick the most extreme pair
        i = above[np.argmax(g[above])]
        j = below[np.argmin(g[below])]
        
        # Transfer amount: proportional to the gap and current values
        gap = g[i] - g[j]
        delta = 0.1 * min(a[i], a[j]) * min(1.0, gap / (abs(lam) + 1e-10))
        delta = min(delta, a[i] * 0.5)  # Don't transfer more than half
        
        # Apply transfer
        a[i] -= delta
        a[j] += delta
        a = project_simplex(a, S)
        
        iteration += 1
        if iteration % 10000 == 0:
            print("    KKT iter %d, obj=%.10f" % (iteration, cur_max))
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Multi-phase optimization with zoom-in active-set refinement,
KKT-based mass transfer, and adaptive LSE-FISTA with BB stepping.
Key improvements over previous: tighter active sets, zoom approach,
KKT balance phase, and trying nearby lengths.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def get_topk_weights(c, k):
    """Return uniform weights on top-k lags."""
    size = len(c)
    top_idx = np.argsort(c)[::-1][:k]
    w = np.zeros(size)
    w[top_idx] = 1.0 / k
    return w


def get_tight_weights(c, tol_frac):
    """Return uniform weights on lags within tol_frac of max."""
    size = len(c)
    cm = c.max()
    threshold = cm * (1.0 - tol_frac)
    w = np.where(c >= threshold, 1.0, 0.0)
    tot = w.sum()
    if tot < 1:
        w = np.zeros(size)
        w[np.argmax(c)] = 1.0
        tot = 1.0
    return w / tot


def lse_weights(c, beta):
    """Return LSE soft-max weights."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return e / tot


def compute_gradient(a, w):
    """Compute 2*IFFT(W*conj(A))[:n] where w are lag weights."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista_zoom(a_init, n, S, max_time, rng,
                   beta_schedule=None,
                   k_schedule=None,
                   noise_scale=0.001,
                   stagnation_gap=200):
    """
    FISTA with zoom: beta and active set size increase over time.
    beta_schedule: list of (beta, duration_frac) pairs
    k_schedule: list of (n_lags, duration_frac) pairs
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    last_improve = 0
    it = 0

    # Default schedules
    if beta_schedule is None:
        beta_schedule = [(100, 0.2), (500, 0.2), (2000, 0.2), (10000, 0.2), (50000, 0.2)]
    if k_schedule is None:
        k_schedule = [(10, 0.3), (7, 0.2), (5, 0.2), (3, 0.15), (2, 0.15)]

    # Build phase boundaries
    total_frac = sum(d for _, d in beta_schedule)
    phase_beta = []
    phase_k = []
    for (b, d) in beta_schedule:
        phase_beta.append((b, d / total_frac))
    total_k_frac = sum(d for _, d in k_schedule)
    for (k, d) in k_schedule:
        phase_k.append((k, d / total_k_frac))

    prev_y = None
    prev_grad = None
    bb_step = None

    while time.monotonic() - start < max_time:
        it += 1
        elapsed_frac = min(1.0, (time.monotonic() - start) / max_time)

        # Determine current beta and k
        cum_b = 0
        cur_beta = phase_beta[-1][0]
        for b, d in phase_beta:
            cum_b += d
            if elapsed_frac <= cum_b:
                cur_beta = b
                break

        cum_k = 0
        cur_k = phase_k[-1][0]
        for k, d in phase_k:
            cum_k += d
            if elapsed_frac <= cum_k:
                cur_k = k
                break

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        # Choose weights: switch between LSE and top-k
        if cur_beta < 1000:
            w = lse_weights(c, cur_beta)
        else:
            w = get_topk_weights(c, cur_k)

        grad = compute_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Step size
        stagnant = it - last_improve
        if cur_beta < 1000:
            step = 1.0 / cur_beta
        else:
            # For top-k gradient, use adaptive step
            if prev_grad is not None and prev_y is not None:
                s = y - prev_y
                gz = grad - prev_grad
                sTs = np.dot(s, s)
                sTz = np.dot(s, gz)
                if sTs > 1e-30 and abs(sTz) > 1e-30:
                    step = min(0.01, max(0.0001, sTs / (abs(sTz) + 1e-30)))
                else:
                    step = 0.001
            else:
                step = 0.005

        # Backtracking for top-k phase
        if cur_beta >= 1000:
            improved = False
            for _ in range(5):
                y_test = project_simplex(y - step * grad, S)
                c_test = fft_convolve_self(y_test)
                if c_test.max() < cur_max * (1 - 1e-9):
                    y = y_test
                    improved = True
                    break
                step *= 0.5
            if not improved:
                step = max(1e-6, step * 0.1)
                y = project_simplex(y - step * grad, S)
        else:
            # FISTA with momentum
            x_new = project_simplex(y - step * grad, S)
            if stagnant < stagnation_gap:
                t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
                y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
                y = np.maximum(y, 0.0)
                t_mom = t_new
            else:
                t_mom = 1.0
                y = x_new
            a = x_new

        # Stagnation handling
        stagnant = it - last_improve
        if stagnant > stagnation_gap * 2:
            t_mom = 1.0
            noise = rng.normal(0, noise_scale * S / np.sqrt(n) * np.exp(-stagnant / 1000.0), n)
            if np.abs(noise).max() > 1e-12:
                y = project_simplex(y + noise, S)
            a = y.copy()
            last_improve = it

        prev_y = y.copy()
        prev_grad = grad.copy()

    c_final = fft_convolve_self(y)
    if c_final.max() < best:
        best = c_final.max()
        best_a = y.copy()

    return best_a, best


def run_active_refine_tight(a_init, n, S, max_time, rng,
                            n_lags=3, tol_frac=0.0005):
    """
    Very tight active-set refinement.
    Uses only the top n_lags or lags within tol_frac of max.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Use very tight active set
        if it % 500 < 250:
            w = get_topk_weights(c, n_lags)
        else:
            w = get_tight_weights(c, tol_frac)

        grad = compute_gradient(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Backtracking line search
        step = 0.005
        improved = False
        for _ in range(8):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur_max * (1 - 1e-10):
                a = a_test
                improved = True
                break
            step *= 0.5

        if not improved:
            a = project_simplex(a - 1e-5 * grad / (gnorm + 1e-30), S)

        # Very periodic tiny noise
        if it % 1000 == 0:
            noise_scale = 0.0003 * S / np.sqrt(n) * np.exp(-it / 20000.0)
            if noise_scale > 1e-14:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_kkt_balance(a_init, n, S, max_time, rng, n_lags=5, transfer_frac=0.05):
    """
    KKT-based mass transfer: equalize the active-set gradient on the support.
    Cheap per iteration, can run many times.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    count = 0

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Compute active gradient
        w = get_topk_weights(c, n_lags)
        g = compute_gradient(a, w)

        # Lambda = mean of g on support
        support = a > 1e-10
        n_sup = support.sum()
        if n_sup < 2:
            break
        lam = g[support].mean()

        # Find most imbalanced pair
        g_sup = g[support]
        max_dev = np.max(np.abs(g_sup - lam))
        if max_dev < 1e-10:
            break

        # Get index arrays for support
        sup_idx = np.where(support)[0]
        # Find index with max positive deviation
        pos_dev = g_sup - lam
        i_local = np.argmax(pos_dev)
        j_local = np.argmin(pos_dev)
        i = sup_idx[i_local]
        j = sup_idx[j_local]

        # Transfer amount
        gap = g[i] - g[j]
        if gap <= 0:
            break
        delta = transfer_frac * min(a[i], a[j])
        delta = min(delta, gap * 0.01 / (abs(g[i]) + abs(g[j]) + 1e-10) * max(a[i], a[j]))
        delta = max(delta, 1e-12 * S)

        # Apply
        a[i] -= delta
        a[j] += delta
        a = project_simplex(a, S)

        count += 1
        if count % 50000 == 0:
            print("      KKT iter %d, obj=%.10f" % (count, c.max()))

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_lse_polish(a_init, n, S, max_time, rng, beta_start=100000, beta_end=1000000):
    """Final LSE polish with very high beta, no momentum (to avoid oscillation)."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Slightly increase beta over time
        frac = min(1.0, it / 200)
        beta = beta_start * (beta_end / beta_start) ** frac

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        w = lse_weights(c, beta)
        grad = compute_gradient(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Small fixed step (no momentum to avoid oscillation)
        step = 0.5 / beta

        # Backtracking
        for _ in range(5):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur_max:
                a = a_test
                break
            step *= 0.5

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def try_different_length(a_base, n_target, max_time, rng):
    """Try optimizing at a different length by interpolating."""
    n_orig = len(a_base)
    t_old = np.linspace(0, 1, n_orig)
    t_new = np.linspace(0, 1, n_target)
    a_new = np.interp(t_new, t_old, a_base)
    S = np.sqrt(2.0 * n_target)
    a_new = normalize_sum(a_new, n_target)

    # Quick zoom optimization

                                           k_schedule=[(8, 0.3), (5, 0.3), (3, 0.4)],
                                           noise_scale=0.002)
    return result_a, result_val

        t2 = min(remaining * 0.2, 300.0)
        print("=== Phase 2: Tight active-set (n=%d, %.0fs) ===" % (n, t2))
        a2, v2 = run_active_refine_tight(global_a, n, S, t2, rng,
                                          n_lags=3, tol_frac=0.0003)
        record(a2)

    # Phase 3: KKT balance (15%)
    if global_a is not None and time.monotonic() < deadline:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        t3 = min(remaining * 0.15, 200.0)
        if t3 > 20:
            print("=== Phase 3: KKT balance (n=%d, %.0fs) ===" % (n, t3))
            a3, v3 = run_kkt_balance(global_a, n, S, t3, rng, n_lags=5, transfer_frac=0.03)
            record(a3)
            # Quick FISTA after KKT
            remaining2 = deadline - time.monotonic()
            if remaining2 > 30:
                t3b = min(remaining2 * 0.3, 100.0)
                a3b, v3b = run_fista_zoom(a3, n, S, t3b, rng,
                                           beta_schedule=[(5000, 0.3), (20000, 0.4), (100000, 0.3)],
                                           k_schedule=[(5, 0.4), (3, 0.3), (2, 0.3)],
                                           noise_scale=0.0005)
                record(a3b)

    # Phase 4: LSE polish (15%)
    if global_a is not None and time.monotonic() < deadline:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        t4 = min(remaining * 0.15, 250.0)
        if t4 > 20:
            print("=== Phase 4: LSE polish (n=%d, %.0fs) ===" % (n, t4))
            a4, v4 = run_lse_polish(global_a, n, S, t4, rng,
                                     beta_start=50000, beta_end=500000)
            record(a4)

    # Phase 5: Try a slightly different length (remaining time)
    if global_a is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 180:
            n_try = len(global_a) * 2
            if n_try > 16384:
                n_try = 16384
            S_try = np.sqrt(2.0 * n_try)
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n_try)
            a_try = np.interp(t_new, t_old, global_a)
            a_try = normalize_sum(a_try, n_try)
            t5 = min(remaining * 0.4, 300.0)
            print("=== Phase 5: Try n=%d (%.0fs) ===" % (n_try, t5))
            a5, v5 = try_different_length(global_a, n_try, t5, rng)
            record(a5)

            # Also try n/2
            remaining2 = deadline - time.monotonic()
            if remaining2 > 120:
                n_try2 = max(2048, len(global_a) // 2)
                if n_try2 != n_try and n_try2 != len(global_a):
                    S_try2 = np.sqrt(2.0 * n_try2)
                    t_old2 = np.linspace(0, 1, len(global_a))
                    t_new2 = np.linspace(0, 1, n_try2)
                    a_try2 = np.interp(t_new2, t_old2, global_a)
                    a_try2 = normalize_sum(a_try2, n_try2)
                    t5b = min(remaining2 * 0.3, 150.0)
                    if t5b > 30:
                        print("=== Phase 5b: Try n=%d (%.0fs) ===" % (n_try2, t5b))
                        a5b, v5b = try_different_length(global_a, n_try2, t5b, rng)
                        record(a5b)

    # Final: one more tight refinement on the best
    if global_a is not None and time.monotonic() < deadline - 10:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            print("=== Final polish (%.0fs) ===" % remaining)
            af, vf = run_active_refine_tight(global_a, n, S, remaining * 0.6, rng,
                                              n_lags=2, tol_frac=0.0002)
            record(af)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.10f, internal = %.10f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("WARNING: evaluate_sequence rejected! Fallback.")
            global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result, using fallback.")
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.10f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
