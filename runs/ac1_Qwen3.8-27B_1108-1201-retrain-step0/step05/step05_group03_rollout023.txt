    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def lse_value_and_grad(a, c, k, beta):
    """Top-k LSE: value and gradient."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    # Numerical safety
    if np.any(scaled > 500):
        beta = 500.0 / (c_top - cm).max() if (c_top - cm).max() > 0 else beta
        scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    val = cm + np.log(max(tot, 1e-300)) / beta
    w = e / max(tot, 1e-300)
    
    w_full = np.zeros(len(c))
    w_full[idx] = w
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return val, grad

        else:

        gnorm = np.linalg.norm(grad)

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()

def coordinate_descent_polish(a, S, time_limit, rng):
    """Polish using coordinate descent on top peaks."""
    n = len(a)
    a = normalize(a.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < time_limit:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Find the peak location
        peak_idx = np.argmax(c)
        
        # Gradient of the peak value conv(peak_idx)
        # conv(peak_idx) = sum_i a_i * a_{peak_idx - i}
        # d(conv(peak_idx))/d(a_j) = a_{peak_idx - j} + (j != peak_idx - j) * a_j * 0
        # Actually: conv(k) = sum_i a_i * a_{k-i}, so d(conv(k))/d(a_j) = a_{k-j} + a_j * (contribution from the other side)
        # For self-convolution: conv(k) = sum_{i: max(0,k-n+1)<=i<=min(k,n-1)} a_i * a_{k-i}
        # d(conv(k))/d(a_j) = a_{k-j} + a_{k-j} (if j != k-j, i.e., j != k/2)
        #                    = 2*a_{k-j} (if j != k/2)
        #                    = a_j (if j == k/2, i.e., k is even and j = k/2)
        
        k = peak_idx
        grad = np.zeros(n)
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                if j == kj:
                    grad[j] += a[j]
                else:
                    grad[j] += 2 * a[kj]
        
        # Try to reduce the peak by decreasing a at positions that contribute most
        # But we need to maintain sum = S
        # Simple approach: find j that maximizes grad[j] and try to decrease it
        # while increasing some other element
        
        # Actually, let's just do a small gradient step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Try a few step sizes
        for step in [0.01, 0.05, 0.1, 0.5, 1.0, 2.0]:
            a_new = project_simplex(a - step * grad / gnorm, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max:
                a = a_new
                break
        else:
            # No improvement, try a random perturbation
            noise_scale = 0.002 * S / np.sqrt(n)
            a_try = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < cur_max:
                a = a_try
    
    return best_a, best

Hmm, the coordinate descent is going to be very slow because each iteration requires a full FFT. Let me think of a better approach.

Actually, for the coordinate descent, I can compute the effect of changing one element a_j on the entire convolution analytically:
- If I change a_j by δ, then conv(k) changes by δ * a_{k-j} + (if the same element appears twice) δ^2
- For small δ: Δconv(k) ≈ 2δ * a_{k-j} (for k-j ≠ j) or δ * (2*a_j + δ) (for k-j = j)

So I can compute the new max by checking which conv values are affected.

But this is getting complex. Let me simplify my approach:

**Final plan:**
1. Load parent sequence
2. Run main FISTA optimization (60% of time)
3. Run high-beta polishing (25% of time)  
4. Try n=4096 with a fresh start (15% of time)
5. Return the best

Let me also add a different approach: **use the "flattening" idea from the frequency domain**.

If the power spectrum |A(ω)|^2 is flat, then the circular autocorrelation is a delta function (only at lag 0). For the linear autocorrelation, a flat power spectrum still helps reduce peaks.

So I can add a "spectral flattening" step: modify the phase of A(ω) to make |A(ω)| more uniform, while keeping a >= 0.

Actually, that's what the previous program does with `spectral_flatten_step`. Let me include a version of that.

Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.
Strategy: Start from parent sequence, run multi-phase FISTA with adaptive
LSE smoothing, add spectral flattening, and try multiple sequence lengths.

def fft_convolve_self(a):

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize(a, S):

    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def lse_grad(a, c, k, beta):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if np.any(scaled > 500):
        max_diff = (c_top - cm).max()
        if max_diff > 0:
            beta = 500.0 / max_diff
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)

def spectral_flatten(a, alpha, rng):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a
    ratio = power / max(mean_p, 1e-20)
    # Damp high-power frequencies, boost low-power
    new_power = power * np.exp(-alpha * (ratio - 1.0))
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / np.maximum(power, 1e-30))
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)

            best_a = y.copy()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

def run_random_restart(a_init, S, time_limit, rng, n_restarts=3):

def resample_sequence(a, n_target):

import numpy as np

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def lse_grad(a, c, k, beta):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()

    max_scaled = scaled.max()

            scaled = beta * (c_top - cm)

    n = len(a)

            if 0 <= kj < n:
                if j == kj:
                    grad[j] += a[j]
                else:
                    grad[j] += 2.0 * a[kj]

        return a

    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / np.maximum(power, 1e-30))
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
