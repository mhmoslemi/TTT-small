import numpy as np

from scipy.fft import fft, ifft


def fft_convolve(a, b=None):
    """Linear convolution via FFT. If b is None, computes a*a."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        theta = (cumsum[-1] - S) / n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, active_lags):
    """
    Compute the combined subgradient for the given active lags with uniform weights.
    The subgradient of max_k conv(a,a)[k] is a convex combination of gradients at active lags.
    
    For lag k (in 0..2n-2), the gradient w.r.t. a[i] is:
        a[k-i] (if 0 <= k-i < n) + a[i-k] (if 0 <= i-k < n)
    """
    n = len(a)
    m = len(active_lags)
    if m == 0:
        return np.zeros(n)
    
    # For each active lag k, the gradient vector v_k has:
    # v_k[i] = a[k-i] (if valid) + a[i-k] (if valid)
    
    # We can compute this efficiently:
    # v_k[i] for i=0..n-1:
    #   Part 1: a[k-i] valid when 0 <= k-i < n => max(0,k-n+1) <= i <= min(k,n-1)
    #   Part 2: a[i-k] valid when 0 <= i-k < n => max(0,k) <= i <= min(n-1,k+n-1)
    
    grad = np.zeros(n)
    for k in active_lags:
        # Part 1: a[k-i] for valid i
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            # a[k-i] for i = i_start..i_end => a[k-i_start]..a[k-i_end] = a[i_end:(k-i_start)+1][::-1]
            # k-i goes from k-i_start to k-i_end
            grad[i_start:i_end+1] += a[k - np.arange(i_start, i_end + 1)]
        
        # Part 2: a[i-k] for valid i
        i_start2 = max(0, k)
        i_end2 = min(n - 1, k + n - 1)
        if i_start2 <= i_end2:
            # a[i-k] for i = i_start2..i_end2 => a[0..(i_end2-k)]
            grad[i_start2:i_end2+1] += a[np.arange(i_start2, i_end2 + 1) - k]
    
    return grad / m  # Uniform weights


def compute_subgradient_weighted(a, active_lags, weights):
    """Compute weighted subgradient for active lags."""
    n = len(a)
    m = len(active_lags)
    if m == 0:
        return np.zeros(n)
    
    grad = np.zeros(n)
    for idx, k in enumerate(active_lags):
        w = weights[idx]
        if w < 1e-15:

        i_start = max(0, k - n + 1)

        i_end2 = min(n - 1, k + n - 1)
        if i_start2 <= i_end2:
            grad[i_start2:i_end2+1] += w * a[np.arange(i_start2, i_end2 + 1) - k]
    
    return grad

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key innovations:
1. Exact active-set subgradient with QP-optimized weights (Frank-Wolfe)
2. Backtracking line search for adaptive step sizing
3. Zadoff-Chu and structured initialization for better starting points
4. Symmetric variant exploration
5. Multi-phase schedule: screen -> KKT-optimize -> polish

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_full(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        theta = (cumsum[-1] - S) / n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip to non-negative and rescale to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_subgradient_fft(a, active_lags, weights, fsize):
    """
    Compute sum_k weights[k] * 2*a[k-i] (if valid) using FFT.
    This is the subgradient of max conv at the active lags.
    """
    n = len(a)
    w = np.zeros(fsize)
    for i, k in enumerate(active_lags):
        if weights[i] > 1e-15:
            w[min(k, fsize - 1)] = weights[i]
    A = fft(a, fsize)
    W = fft(w, fsize)
    result = np.real(ifft(np.conj(A) * W))[:n]
    return 2.0 * result


def solve_lambda_qp(V, m):
    """
    Frank-Wolfe for: min lambda' Q lambda s.t. lambda >= 0, sum lambda = 1
    where Q = V'V (centered).
    V is (n x m) matrix of centered subgradients on the support.
    Returns optimal lambda of length m.
    """
    if m == 1:
        return np.array([1.0])
    # Compute Q = V^T V
    Q = V.T @ V
    # Frank-Wolfe
    lam = np.ones(m) / m
    for fw_it in range(100):
        grad = 2.0 * Q @ lam
        # Find vertex: minimizer of grad over simplex is e_{argmin grad}
        v_idx = np.argmin(grad)
        e_v = np.zeros(m)
        e_v[v_idx] = 1.0
        # Line search for step size
        best_t = 0.0
        best_val = lam @ Q @ lam
        for t in np.linspace(0, 1, 20):
            cand = (1 - t) * lam + t * e_v
            val = cand @ Q @ cand
            if val < best_val:
                best_val = val
                best_t = t
        lam_new = (1 - best_t) * lam + best_t * e_v
        if np.max(np.abs(lam_new - lam)) < 1e-10:
            lam = lam_new
            break
        lam = lam_new
    lam = np.maximum(lam, 0)
    lam_sum = lam.sum()
    if lam_sum < 1e-15:
        lam = np.ones(m) / m
    else:
        lam = lam / lam_sum
    return lam


def get_active_set(c, tol_frac=0.05, max_lags=60):
    """Get active lags: those within tol_frac of the max."""
    cmax = c.max()
    threshold = cmax * (1.0 - tol_frac)
    active = np.where(c >= threshold)[0]
    if len(active) > max_lags:
        # Keep the top max_lags
        active = active[np.argsort(c[active])[::-1][:max_lags]]
    return active


def backtracking_line_search(a, S, direction, c_current, max_conv, n, fsize,
                              max_steps=30, rho=0.5, c_alpha=0.01):
    """
    Backtracking line search to find step size.
    direction: the direction to step (negative gradient direction)
    Returns optimal step size.
    """
    step = 1.0
    base_val = max_conv
    for _ in range(max_steps):
        a_new = project_simplex(a + step * direction, S)
        c_new = fft_convolve_full(a_new)
        new_max = c_new.max()
        if new_max < base_val - c_alpha * step * np.linalg.norm(direction):
            return step, new_max, a_new
        step *= rho
    return step, base_val, project_simplex(a + step * direction, S)


def objective_from_conv(max_conv, n, S):
    """Compute objective from max of convolution."""
    return 2.0 * n * max_conv / (S * S)


def generate_zadoff_chu_init(n, rng, S):
    """Generate initialization based on Zadoff-Chu-like structure."""
    # Use chirp: a[i] = |cos(pi * i * (i+1) / P)| for prime P near n
    P = 8191  # prime
    if n <= P:
        i = np.arange(n)
        a = np.abs(np.cos(np.pi * i * (i + 1) / P))
    else:
        i = np.arange(n)
        a = np.abs(np.cos(np.pi * (i % P) * ((i % P) + 1) / P))
    a = a + 0.01
    return normalize_to_sum(a, S)


def generate_inits(n, S, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        gs = np.clip(given_seq, 0, 1000)
        if gs.sum() > 0.01:
            inits.append(normalize_to_sum(gs, S))

    # Zadoff-Chu based
    inits.append(generate_zadoff_chu_init(n, rng, S))

    # Standard windows
    inits.append(normalize_to_sum(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01), S))
    inits.append(normalize_to_sum(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01), S))
    inits.append(normalize_to_sum(np.sin(np.pi * tt) ** 2 + 0.01, S))
    inits.append(normalize_to_sum(np.sin(np.pi * tt) ** 4 + 0.01, S))
    inits.append(normalize_to_sum(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.25) + 0.01, S))

    # Multi-frequency
    inits.append(normalize_to_sum(
        np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * tt + 0.7) + 0.2 * np.cos(10 * np.pi * tt), 0.01), S))
    inits.append(normalize_to_sum(
        np.maximum(1.0 + 0.4 * np.sin(6 * np.pi * tt) + 0.15 * np.cos(14 * np.pi * tt + 0.3), 0.01), S))

    # Random
    inits.append(normalize_to_sum(rng.random(n) + 0.1, S))
    inits.append(normalize_to_sum(np.abs(rng.normal(0.8, 0.4, n)) + 0.05, S))

    # Block structures
    for nb in [4, 8, 16]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.2, 1.5, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(normalize_to_sum(block[:n] + 0.01, S))

    # Symmetric uniform with perturbation
    base = np.ones(n)
    base += 0.3 * np.sin(2 * np.pi * 3 * tt)
    inits.append(normalize_to_sum(np.abs(base) + 0.01, S))

    return inits


def quick_fista(a_init, S, max_time, rng, beta=50.0):
    """Quick FISTA with LSE for initial screening."""
    n = len(a_init)
    fsize = 1 << (2 * n - 1).bit_length()
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c = fft_convolve_full(a)
    best = c.max()
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_full(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        # LSE gradient
        cm = c.max()
        scaled = beta * (c - cm)
        if scaled.max() > 400:
            beta_eff = 400.0 / max(1e-15, (c - cm).max())
            scaled = beta_eff * (c - cm)
            beta = beta_eff
        e = np.exp(scaled)
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        Wf = fft(w, fsize)
        Af = fft(a, fsize)
        grad = 2.0 * np.real(ifft(Wf * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it - last_improve > 3000:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it

    return best_a, best


def kkt_optimize(a_init, S, max_time, rng):
    """
    Main KKT-guided optimization with exact subgradient and QP-optimized weights.
    Uses backtracking line search and periodic perturbations.
    """
    n = len(a_init)
    fsize = 1 << (2 * n - 1).bit_length()
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()

    c = fft_convolve_full(a)
    best = c.max()
    best_a = a.copy()
    last_improve = 0
    it = 0

    # Precompute FFT of a (will update as a changes)
    A_fft = fft(a, fsize)

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        c = fft_convolve_full(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()
            last_improve = it
            if it % 5000 == 0:
                obj = objective_from_conv(best, n, S)
                print("  KKT it %d: max_b=%.8f obj=%.6f" % (it, best, obj))

        # Stuck handling
        if it - last_improve > 5000:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 10000.0)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            last_improve = it
            continue

        # Get active set (adaptively sized)
        tol_frac = 0.01 + 0.04 * (1.0 - progress)
        max_lags = min(40 + int(40 * (1 - progress)), 80)
        active = get_active_set(c, tol_frac=tol_frac, max_lags=max_lags)
        m = len(active)

        if m == 0:
            active = np.array([np.argmax(c)])
            m = 1

        # Solve QP for optimal weights (every 50 iterations or first time)
        if it % 50 == 1 or not hasattr(kkt_optimize, '_last_it'):

            # Compute V (support x m) for QP
            # V[j, i] = 2*a[active[i] - j] (if valid)

            Q = V_centered.T @ V_centered
            # Add small regularization for numerical stability

            lam = solve_lambda_qp(V_centered, m)
            kkt_optimize._last_it = it
            kkt_optimize._lam = lam
            kkt_optimize._active = active.copy()
        else:
            lam = kkt_optimize._lam
            active = kkt_optimize._active

        # Compute subgradient
        A_fft = fft(a, fsize)
        grad = compute_subgradient_fft(a, active, lam, fsize)

        # Compute mu = mean of gradient on support
        support = np.where(a > 1e-6)[0]
        if len(support) == 0:
            mu = grad.mean()
        else:
            mu = grad[support].mean()

        # Direction: negative of (grad - mu*1) on the simplex

        if gnorm < 1e-14:
            # Try uniform perturbation
            direction = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S) - a
            gnorm = np.linalg.norm(direction)
            if gnorm < 1e-14:

        # Adaptive step size
        base_step = S / (n * (1.0 + 0.1 * progress))
        step_scale = base_step / gnorm
        step = step_scale

            c_new = fft_convolve_full(a_new)
            if c_new.max() < c.max() * (1.0 + 1e-10):
                a = a_new
                found = True
                break
            step *= 0.5

        if not found:
            # Accept anyway if it's much better
            a_new = project_simplex(a + step * direction, S)
            c_new = fft_convolve_full(a_new)
            if c_new.max() < c.max():
                a = a_new

        # Periodic Fourier mutation to explore

            Af = fft(a, fsize)
            mag = np.abs(Af)
            phase = np.angle(Af) + rng.normal(0, scale, fsize)
            new_Af = mag * np.exp(1j * phase)
            new_Af[0] = Af[0]
            a_mut = np.real(ifft(new_Af, fsize))[:n]
            a_mut = project_simplex(a_mut, S)
            c_mut = fft_convolve_full(a_mut)

                break
            step *= 0.5

    return best_a, best


def try_symmetric(a_init, S, max_time, rng):
    """
    Try optimizing in the symmetric subspace: a[i] = a[n-1-i].
    This halves the effective dimension.
    """
    n = len(a_init)
    fsize = 1 << (2 * n - 1).bit_length()

    # Make symmetric
    half_n = n // 2
    a = normalize_to_sum(a_init, S)
    a_sym = np.zeros(n)
    for i in range(n):
        a_sym[i] = 0.5 * (a[i] + a[n - 1 - i])
    a_sym = normalize_to_sum(a_sym, S)

    start = time.monotonic()
    c = fft_convolve_full(a_sym)
    best = c.max()
    best_a = a_sym.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        c = fft_convolve_full(a_sym)
        if c.max() < best:
            best = c.max()
            best_a = a_sym.copy()
            last_improve = it

        if it - last_improve > 3000:
            noise_scale = 0.001 * S / np.sqrt(n)
            a_sym = project_simplex(a_sym + rng.normal(0, noise_scale, n), S)
            # Re-symmetrize
            for i in range(n):
                a_sym[i] = 0.5 * (a_sym[i] + a_sym[n - 1 - i])
            a_sym = normalize_to_sum(a_sym, S)
            last_improve = it
            continue

        active = get_active_set(c, tol_frac=0.02, max_lags=30)
        m = len(active)
        lam = np.ones(m) / m

        grad = compute_subgradient_fft(a_sym, active, lam, fsize)
        # Symmetrize gradient
        grad_sym = 0.5 * (grad + grad[::-1])

        support = np.where(a_sym > 1e-6)[0]
        if len(support) > 0:
            mu = grad_sym[support].mean()
        else:
            mu = grad_sym.mean()

        direction = -(grad_sym - mu)
        gnorm = np.linalg.norm(direction)
        if gnorm < 1e-14:
            break

        step = S / (n * gnorm * (2.0 + progress))

        for _ in range(10):
            a_new = project_simplex(a_sym + step * direction, S)
            for i in range(n):
                a_new[i] = 0.5 * (a_new[i] + a_new[n - 1 - i])
            a_new = normalize_to_sum(a_new, S)
            c_new = fft_convolve_full(a_new)
            if c_new.max() <= c.max() + 1e-12:
                a_sym = a_new
                break
            step *= 0.5

    return best_a, best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Diverse initialization + quick FISTA screening (~150s)
    2. KKT-guided optimization with QP weights (~1500s)
    3. Symmetric variant exploration (~300s)
    4. Final high-precision polish (~350s)
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30

    print("Starting optimization, budget=%.0fs, deadline in %.0fs" % (budget_s, deadline - total_start))

    # Get given sequence if available
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    n = 8192
    S = np.sqrt(2.0 * n)
    global_best = np.inf
    global_a = None

    def update_global(a, val):
        nonlocal global_best, global_a
        obj = 2.0 * n * val / (a.sum() ** 2)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            return True
        return False

    # Phase 1: Quick screening
    print("\n=== Phase 1: Quick screening ===")
    p1_start = time.monotonic()
    p1_budget = min(150.0, (deadline - p1_start) * 0.10)
    inits = generate_inits(n, S, rng, given_seq=given)

    # Sort inits by quick evaluation
    init_scores = []
    for i, a_init in enumerate(inits):
        c = fft_convolve_full(a_init)
        obj = 2.0 * n * c.max() / (a_init.sum() ** 2)
        init_scores.append((obj, i))
    init_scores.sort()
    print("  Initial scores (top 5):")
    for obj, idx in init_scores[:5]:
        print("    init %d: obj=%.6f" % (idx, obj))

    # Quick FISTA on top candidates
    n_screen = min(6, len(init_scores))
    per_time = p1_budget / n_screen

    candidates = []
    for rank in range(n_screen):
        if time.monotonic() > deadline - (deadline - total_start) * 0.85:
            break
        idx = init_scores[rank][1]
        t_alloc = min(per_time, deadline - time.monotonic() - 60)
        if t_alloc < 10:
            break
        a_opt, val = quick_fista(inits[idx], S, t_alloc, rng, beta=30.0)
        obj = 2.0 * n * val / (a_opt.sum() ** 2)
        candidates.append((obj, a_opt.copy()))
        print("  Screen %d (init %d): obj=%.6f" % (rank, idx, obj))
        update_global(a_opt, val)

    candidates.sort(key=lambda x: x[0])
    print("  Best after screening: obj=%.6f" % global_best)

    # Phase 2: KKT-guided main optimization
    print("\n=== Phase 2: KKT-guided optimization ===")
    p2_start = time.monotonic()
    p2_budget = min(1500.0, (deadline - p2_start) * 0.55)

    # Run from best candidate and second best
    n_runs = min(3, len(candidates))
    for run_i in range(n_runs):
        if time.monotonic() > deadline - 400:
            break
        t_alloc = (deadline - time.monotonic() - 400) / (n_runs - run_i)
        t_alloc = min(t_alloc, p2_budget / (n_runs - run_i) + 30)
        if t_alloc < 60:
            break
        start_a = candidates[run_i][1]
        print("  KKT run %d (start obj=%.6f), alloc=%.0fs" % (run_i, candidates[run_i][0], t_alloc))
        a_opt, val = kkt_optimize(start_a, S, t_alloc, rng)
        obj = 2.0 * n * val / (a_opt.sum() ** 2)
        print("  KKT run %d done: obj=%.6f" % (run_i, obj))
        update_global(a_opt, val)

    print("  Best after KKT: obj=%.6f" % global_best)

    # Phase 3: Symmetric variant
    print("\n=== Phase 3: Symmetric exploration ===")
    p3_start = time.monotonic()
    p3_budget = min(300.0, (deadline - p3_start) * 0.15)

    if global_a is not None and p3_budget > 30:
        a_sym, val_sym = try_symmetric(global_a, S, p3_budget, rng)
        obj_sym = 2.0 * n * val_sym / (a_sym.sum() ** 2)
        print("  Symmetric result: obj=%.6f" % obj_sym)
        update_global(a_sym, val_sym)

    # Also try symmetric from a different starting point
    if candidates[0][1] is not None and p3_budget > 60:
        t_rem = deadline - time.monotonic() - 200
        if t_rem > 30:
            a_sym2, val_sym2 = try_symmetric(candidates[0][1], S, min(t_rem, 100), rng)
            obj_sym2 = 2.0 * n * val_sym2 / (a_sym2.sum() ** 2)
            print("  Symmetric (alt) result: obj=%.6f" % obj_sym2)
            update_global(a_sym2, val_sym2)

    print("  Best after symmetric: obj=%.6f" % global_best)

    # Phase 4: Final polish
    print("\n=== Phase 4: Final polish ===")
    p4_start = time.monotonic()
    p4_budget = deadline - p4_start - 10

    if global_a is not None and p4_budget > 30:
        # Split polish time
        seg1 = p4_budget * 0.6
        seg2 = p4_budget * 0.4

        a_p1, val_p1 = final_polish(global_a, S, seg1, rng)
        obj_p1 = 2.0 * n * val_p1 / (a_p1.sum() ** 2)
        print("  After polish 1: obj=%.8f" % obj_p1)
        update_global(a_p1, val_p1)

        # Another polish from updated best
        remaining = deadline - time.monotonic() - 5
        if remaining > 20:
            a_p2, val_p2 = final_polish(global_a, S, min(seg2, remaining), rng)
            obj_p2 = 2.0 * n * val_p2 / (a_p2.sum() ** 2)
            print("  After polish 2: obj=%.8f" % obj_p2)
            update_global(a_p2, val_p2)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
            ver = evaluate_sequence(final_list)
            print("  Fallback obj=%.6f" % ver)
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.8f, n=%d" % (elapsed, ver if np.isfinite(ver) else global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
