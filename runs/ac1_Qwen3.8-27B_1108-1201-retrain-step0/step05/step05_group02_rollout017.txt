"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses L-BFGS with adaptive LSE smoothing and penalty method for constraints.
Multi-phase: smooth-to-sharp beta schedule, upsample, final subgradient polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_obj_grad(a, beta, S, mu):
    """
    Compute LSE objective + penalty and its gradient.
    """
    n = len(a)
    size = 2 * n - 1
    c = fft_convolve_self(a)
    
    # LSE
    bc = beta * c
    bc_max = bc.max()
    shifted = bc - bc_max
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    e_sum = e.sum()
    lse_val = (bc_max + np.log(e_sum)) / beta if e_sum > 0 else bc_max / beta
    w = e / e_sum if e_sum > 0 else np.zeros(size)
    w[int(np.argmax(c))] = 1.0 if e_sum < 1e-300 else w[int(np.argmax(c))]
    
    # Gradient w.r.t. a: g_j = 2 * sum_k w_k * a_{k-j}
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_lse = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    # Penalty: mu * (sum(a) - S)^2 + mu * sum(max(0, -a)^2)
    sum_a = a.sum()
    diff = sum_a - S
    obj_pen = mu * diff * diff
    grad_pen = 2.0 * mu * diff * np.ones(n)
    
    neg = np.maximum(0, -a)
    obj_pen += mu * (neg * neg).sum()
    grad_pen += -2.0 * mu * neg
    
    total = lse_val + obj_pen
    grad = grad_lse + grad_pen
    return total, grad


def run_lbfgs(a_init, max_time, beta, mu, maxiter=500):
    """Run L-BFGS-B optimization."""
    S = np.sqrt(2.0 * len(a_init))
    a0 = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a0 = a0 * (S / a0.sum())
    
    start = time.monotonic()
    best_a = a0.copy()
    best_obj = compute_obj(a0)
    
    def fun(x):
        val, _ = lse_obj_grad(x, beta, S, mu)
        return val
    
    def jac(x):
        _, g = lse_obj_grad(x, beta, S, mu)
        return g
    
    result = minimize(fun, a0, jac=jac, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8,
                               'maxfun': maxiter * 3, 'iprint': 0})
    
    a_opt = result.x
    a_opt = np.clip(a_opt, 0, 1000)
    a_opt = a_opt * (S / a_opt.sum())
    
    obj_opt = compute_obj(a_opt)
    if obj_opt < best_obj:
        best_obj = obj_opt
        best_a = a_opt.copy()
    
    return best_a, best_obj, result

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        if it - last_improve > 3000:
            t_mom = 1.0
            noise_scale = max(1e-7, 0.002 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    return best_a, best_obj


def topk_subgradient(a, c, k):
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_polish(a_init, max_time, rng):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = project_simplex(np.clip(a_init, 0, 1000), S)
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck < 100:
            k = 1
        elif stuck < 500:
            k = 2
        elif stuck < 2000:
            k = min(8, 2 + stuck // 200)
        else:
            k = min(32, 8 + stuck // 500)

        grad = topk_subgradient(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (gnorm * (1.0 + 0.003 * it))
        step = min(step, 3.0 * S / gnorm)

        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        if obj_new <= cur_obj + 1e-13:
            a = a_new
        else:
            step *= 0.1
            a_new = project_simplex(a - step * grad, S)
            if compute_obj(a_new) <= cur_obj + 1e-13:
                a = a_new

        if stuck > 2000 and it % 200 == 0:
            ns = max(1e-7, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 10000))
            a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best_obj


def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)
