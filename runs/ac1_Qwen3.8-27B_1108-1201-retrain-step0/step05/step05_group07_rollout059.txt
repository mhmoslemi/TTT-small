```python
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csc_matrix
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(n, b):
    """Build the sparse matrix M where (M @ a)[k] = (a * b)[k]."""
    # M is (2n-1) x n
    # M[k, i] = b[k-i] if 0 <= k-i <= n-1, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(b[k - i])
    M = csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return M

def solve_lp_for_a(b, S, n):
    """Solve: minimize t s.t. (a*b)[k] <= t, sum(a)=S, a>=0."""
    M = build_conv_matrix(n, b)
    
    # Construct A_ub: [M, -1] <= 0
    # A_eq: [1^T, 0] = S
    # Variables: [a_0, ..., a_{n-1}, t]
    
    # A_ub shape: (2n-1) x (n+1)
    # We need to add a column of -1's for t
    from scipy.sparse import identity as speye
    from scipy.sparse import vstack, hstack, csr_matrix
    
    # t_col is a column vector of -1's
    t_col = csr_matrix(np.ones((2*n-1, 1)) * (-1))
    A_ub = hstack([M, t_col], format='csc')
    b_ub = np.zeros(2*n-1)
    
    # A_eq
    ones_row = csr_matrix(np.ones((1, n)))
    zero_row = csr_matrix(np.zeros((1, 1)))
    A_eq = hstack([ones_row, zero_row], format='csc')
    b_eq = np.array([S])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Bounds: a[i] >= 0, t free
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if not result.success:
        return None, np.inf
    
    a_opt = result.x[:n]
    t_opt = result.x[n]
    return a_opt, t_opt
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using alternating LP + LSE polishing.

The key insight is that fixing one copy of a makes the problem a linear program:
minimize t s.t. (a*b)[k] <= t for all k, sum(a)=S, a>=0.
We alternate between optimizing a (given b) and b (given a), solving an LP at each step.
This avoids non-convexity issues that plague pure subgradient methods.
"""

import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csc_matrix, csr_matrix, hstack
import time

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(n, b):
    """Build sparse matrix M where (M @ a)[k] = (a*b)[k] for self-conv style."""
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        cnt = i_end - i_start + 1
        rows_list.append(np.full(cnt, k, dtype=np.int32))
        cols_list.append(np.arange(i_start, i_end + 1, dtype=np.int32))
        vals_list.append(b[k - np.arange(i_start, i_end + 1)])
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def solve_conv_lp(fixed, S, n):
    """
    Solve: minimize t s.t. (x * fixed)[k] <= t for all k, sum(x)=S, x>=0.
    Returns (x_opt, t_opt) or (None, inf) on failure.
    """
    M = build_conv_matrix(n, fixed)
    
    # A_ub: [M | -1] @ [x; t] <= 0
    # A_eq: [1^T | 0] @ [x; t] = S
    
    ones_col = csr_matrix(np.ones((2*n-1, 1)) * (-1.0))
    A_ub = hstack([M, ones_col], format='csc')
    b_ub = np.zeros(2*n - 1)
    
    ones_row = csr_matrix(np.ones((1, n)))
    zero_row = csr_matrix(np.zeros((1, 1)))
    A_eq = hstack([ones_row, zero_row], format='csc')
    b_eq = np.array([S])
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 120})
    
    if not result.success:
        return None, np.inf
    
    x_opt = np.maximum(result.x[:n], 0.0)
    t_opt = result.x[n]
    return x_opt, t_opt

def alternating_lp(a_init, S, n, max_iter=20, time_limit=600):
    """
    Alternating LP: minimize max_k (a*a)[k] by alternating LP solves.
    """
    a = project_simplex(a_init.copy(), S)
    b = a.copy()
    
    best_obj = np.inf
    best_a = a.copy()
    
    start = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break
        
        t0 = time.monotonic()
        
        # Solve for a given b
        a_new, t_a = solve_conv_lp(b, S, n)
        t1 = time.monotonic()
        
        if a_new is None:
            print(f"  LP solve for a failed at iter {it}")
            break
        
        a_new = project_simplex(a_new, S)
        
        # Solve for b given a_new
        b_new, t_b = solve_conv_lp(a_new, S, n)
        t2 = time.monotonic()
        
        if b_new is None:
            print(f"  LP solve for b failed at iter {it}")
            break
        
        b_new = project_simplex(b_new, S)
        
        # Evaluate
        a_curr = a_new
        c = fft_convolve_self(a_curr)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_curr.copy()
        
        # Check convergence
        diff_a = np.abs(a_new - a).max()
        diff_b = np.abs(b_new - b).max()
        
        print(f"  LP iter {it}: obj={cur_obj:.6f}, max={cur_max:.6f}, "
              f"t_a={t_a:.6f}, t_b={t_b:.6f}, "
              f"diff_a={diff_a:.6f}, diff_b={diff_b:.6f}, "
              f"time={t2-t0:.1f}s")
        
        if max(diff_a, diff_b) < 1e-8:
            print(f"  Converged at iter {it}")
            break
        
        a = a_new
        b = b_new
    
    return best_a, best_obj
```

```python
def make_conv_matrix(n, b, rows, cols):
    """Create sparse convolution matrix for fixed b."""
    vals = b[rows - cols].astype(np.float64)
    return csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_conv_lp(fixed, S, n, rows, cols, time_per_solve=120):
    """
    Solve: minimize t s.t. (x * fixed)[k] <= t for all k, sum(x)=S, x>=0.
    Returns (x_opt, t_opt).
    """
    M = make_conv_matrix(n, fixed, rows, cols)
    
    num_constraints = 2 * n - 1
    ones_col = csr_matrix(-np.ones((num_constraints, 1)))
    A_ub = hstack([M, ones_col], format='csc')
    b_ub = np.zeros(num_constraints)
    
    ones_row = csr_matrix(np.ones((1, n)))
    zero_row = csr_matrix(np.zeros((1, 1)))
    A_eq = hstack([ones_row, zero_row], format='csc')
    b_eq = np.array([S])
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_per_solve})
    except Exception as e:
        print(f"    LP error: {e}")
        return None, np.inf
    
    if not result.success:
        return None, np.inf
    
    x_opt = np.maximum(result.x[:n], 0.0)
    t_opt = float(result.x[n])
    return x_opt, t_opt


def alternating_lp(a_init, S, n, rows, cols, max_iter=30, time_limit=500):
    """
    Alternating LP optimization.
    """
    a = project_simplex(a_init.copy(), S)
    b = a.copy()
    
    best_obj = np.inf
    best_a = a.copy()
    
    start = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break
        
        t0 = time.monotonic()
        
        # Solve for a given b
        a_new, t_a = solve_conv_lp(b, S, n, rows, cols)
        
        if a_new is None:
            print(f"    LP(a|b) failed at iter {it}, stopping")
            break
        
        a_new = project_simplex(a_new, S)
        
        # Solve for b given a_new
        b_new, t_b = solve_conv_lp(a_new, S, n, rows, cols)
        
        if b_new is None:
            # Just use a_new
            b_new = a_new
            b_new = project_simplex(b_new, S)
        
        b_new = project_simplex(b_new, S)
        
        # Evaluate self-convolution of a_new (and also of the average)
        c_a = fft_convolve_self(a_new)
        obj_a = 2.0 * n * c_a.max() / (S * S)
        
        avg = project_simplex(0.5 * (a_new + b_new), S)
        c_avg = fft_convolve_self(avg)
        obj_avg = 2.0 * n * c_avg.max() / (S * S)
        
        # Pick the best
        if obj_a <= obj_avg:
            a_curr = a_new
            cur_obj = obj_a
        else:
            a_curr = avg
            cur_obj = obj_avg
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_curr.copy()
        
        elapsed = time.monotonic() - t0
        diff = np.abs(a_new - a).max()
        
        print(f"    LP iter {it}: obj={cur_obj:.6f} (a:{obj_a:.6f}, avg:{obj_avg:.6f}), "
              f"diff={diff:.2e}, time={elapsed:.1f}s")
        
        if diff < 1e-10:
            print(f"    Converged at iter {it}")
            break
        
        a = a_new
        b = b_new
    
    return best_a, best_obj


def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution: (1/beta)log(sum exp(beta*c))."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Gradient of (a*a)[k] w.r.t. a[i] is a[k-i] (for valid k-i)
    # So grad = 2 * (w * a) (cross-correlation)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def lse_polish(a0, S, max_time, rng, beta_init=50.0, beta_max=50000.0, verbose=True):
    """LSE-smoothed gradient polishing."""
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 200
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = min(beta_max, beta_init * np.exp(0.01 * it))
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best - 1e-14:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Noise injection when stuck
        if stuck > patience:
            noise_scale = min(0.05, 0.005 * (stuck / patience)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.3)
            continue
        
        grad = compute_lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: for LSE, the optimal step is ~1/beta
        step = 1.0 / (2.0 * beta)
        step = min(step, 1.0 / gnorm)
        
        # Backtracking line search
        accepted = False
        for ls in range(15):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-12):
                a = a_try
                accepted = True
                break
            step *= 0.5
        
        if not accepted:
            step = 0.001 / max(gnorm, 1e-15)
            a = project_simplex(a - step * grad, S)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    LSE iter {it}: obj={cur_obj:.6f}, beta={beta:.0f}, t={elapsed:.0f}s")
    
    return best_a, best


def subgradient_optimize(a0, S, max_time, rng, verbose=False):
    """Top-k weighted subgradient method (improved from previous)."""
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 200
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best - 1e-14:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        progress = min(1.0, (time.monotonic() - start) / max_time)
        k = int(4 + 64 * progress ** 0.7)
        k = min(k, 2 * n - 1)
        
        top_idx = np.argsort(c)[::-1][:k]
        weights = np.maximum(c[top_idx] - c[top_idx].min(), 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(k) / k
        else:
            weights = weights / wsum
        
        grad = np.zeros(n)
        for ii in range(k):
            kidx = int(top_idx[ii])
            wi = weights[ii]
            j_start = max(0, kidx - n + 1)
            j_end = min(n - 1, kidx)
            if j_start <= j_end:
                indices = kidx - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        if stuck > patience:
            noise_scale = min(0.03, 0.003 * (stuck / patience)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.2)
            continue
        
        gap = cur_max - c.mean()
        step = 2.0 * gap / max(gnorm * gnorm, 1e-30)
        step = min(step, 5.0 / gnorm)
        
        accepted = False
        for _ in range(10):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-10):
                a = a_try
                accepted = True
                break
            step *= 0.5
        
        if not accepted:
            a = project_simplex(a - (0.001 / gnorm) * grad, S)
    
    return best_a, best
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach combining:
- Alternating LP (exploits linearity when one copy is fixed, solving global subproblems)
- LSE-smoothed gradient descent with adaptive beta for fine convergence
- Top-k subgradient method as complement
- Multiple restarts with noise injection to escape local minima

Key innovation: The alternating LP directly solves a global optimization subproblem
at each iteration (minimize max_k (a*b)[k] with b fixed is a LINEAR program),
avoiding the slow convergence and local minimum trapping of pure subgradient methods.
"""

import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csc_matrix, csr_matrix, hstack
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    v = np.maximum(v, 0.0)
    if v.sum() < 1e-15:
        return np.full(n, S / n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def precompute_conv_indices(n):
    """Precompute row/col indices for the convolution matrix structure."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        cnt = i_end - i_start + 1
        rows_list.append(np.full(cnt, k, dtype=np.int32))
        cols_list.append(np.arange(i_start, i_end + 1, dtype=np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    return rows, cols


def make_conv_matrix(n, b, rows, cols):
    """Create sparse convolution matrix for fixed b."""
    vals = b[rows - cols].astype(np.float64)
    return csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_conv_lp(fixed, S, n, rows, cols, time_per_solve=90):
    """
    Solve: minimize t s.t. (x * fixed)[k] <= t for all k, sum(x)=S, x>=0.
    Returns (x_opt, t_opt) or (None, inf) on failure.
    """
    M = make_conv_matrix(n, fixed, rows, cols)
    num_constraints = 2 * n - 1
    ones_col = csr_matrix(-np.ones((num_constraints, 1)))
    A_ub = hstack([M, ones_col], format='csc')
    b_ub = np.zeros(num_constraints)
    ones_row = csr_matrix(np.ones((1, n)))
    zero_row = csr_matrix(np.zeros((1, 1)))
    A_eq = hstack([ones_row, zero_row], format='csc')
    b_eq = np.array([S])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = [(0, None)] * n + [(-np.inf, None)]
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_per_solve})
    except Exception as e:
        print(f"    LP error: {e}")
        return None, np.inf
    if not result.success:
        return None, np.inf
    x_opt = np.maximum(result.x[:n], 0.0)
    t_opt = float(result.x[n])
    return x_opt, t_opt


def alternating_lp(a_init, S, n, rows, cols, max_iter=30, time_limit=500):
    """Alternating LP: minimize max_k (a*a)[k] by alternating LP solves."""
    a = project_simplex(a_init.copy(), S)
    b = a.copy()
    best_obj = np.inf
    best_a = a.copy()
    start = time.monotonic()
    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break
        t0 = time.monotonic()
        a_new, t_a = solve_conv_lp(b, S, n, rows, cols)
        if a_new is None:
            print(f"    LP(a|b) failed at iter {it}, stopping")
            break
        a_new = project_simplex(a_new, S)
        b_new, t_b = solve_conv_lp(a_new, S, n, rows, cols)
        if b_new is None:
            b_new = a_new.copy()
        b_new = project_simplex(b_new, S)
        c_a = fft_convolve_self(a_new)
        obj_a = 2.0 * n * c_a.max() / (S * S)
        avg = project_simplex(0.5 * (a_new + b_new), S)
        c_avg = fft_convolve_self(avg)
        obj_avg = 2.0 * n * c_avg.max() / (S * S)
        if obj_a <= obj_avg:
            a_curr = a_new
            cur_obj = obj_a
        else:
            a_curr = avg
            cur_obj = obj_avg
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_curr.copy()
        elapsed = time.monotonic() - t0
        diff = np.abs(a_new - a).max()
        print(f"    LP iter {it}: obj={cur_obj:.6f} (a:{obj_a:.6f}, avg:{obj_avg:.6f}), "
              f"diff={diff:.2e}, time={elapsed:.1f}s")
        if diff < 1e-10:
            print(f"    LP converged at iter {it}")
            break
        a = a_new
        b = b_new
    return best_a, best_obj


def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max: grad of (1/beta)*log(sum_k exp(beta*c_k))."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def lse_polish(a0, S, max_time, rng, beta_init=50.0, beta_max=50000.0, verbose=False):
    """LSE-smoothed gradient polishing with adaptive beta and noise."""
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 150
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = min(beta_max, beta_init * np.exp(0.008 * it))
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-14:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        if stuck > patience:
            noise_scale = min(0.04, 0.004 * (stuck / patience)) * S / np.sqrt(n) * (1.0 - 0.5 * progress)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.25)
            continue
        grad = compute_lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)
        step = min(step, 0.5 / gnorm)
        accepted = False
        for _ in range(20):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-12):
                a = a_try
                accepted = True
                break
            step *= 0.4
        if not accepted:
            step = 0.0005 / max(gnorm, 1e-15)
            a = project_simplex(a - step * grad, S)
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    LSE iter {it}: obj={cur_obj:.6f}, beta={beta:.0f}, t={elapsed:.0f}s")
    return best_a, best


def subgradient_optimize(a0, S, max_time, rng, verbose=False):
    """Top-k weighted subgradient with line search and adaptive noise."""
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 250
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-14:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        k = int(4 + 64 * progress ** 0.6)
        k = min(k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        weights = np.maximum(c[top_idx] - c[top_idx].min(), 0.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(k) / k
        else:
            weights = weights / wsum
        grad = np.zeros(n)
        for ii in range(k):
            kidx = int(top_idx[ii])
            wi = weights[ii]
            j_start = max(0, kidx - n + 1)
            j_end = min(n - 1, kidx)
            if j_start <= j_end:
                indices = kidx - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        if stuck > patience:
            noise_scale = min(0.025, 0.003 * (stuck / patience)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.2)
            continue
        gap = cur_max - c.mean()
        step = 2.0 * gap / max(gnorm * gnorm, 1e-30)
        step = min(step, 4.0 / gnorm)
        accepted = False
        for _ in range(12):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-10):
                a = a_try
                accepted = True
                break
            step *= 0.5
        if not accepted:
            a = project_simplex(a - (0.001 / gnorm) * grad, S)
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    SG iter {it}: obj={cur_obj:.6f}, t={elapsed:.0f}s")
    return best_a, best


def try_update_best(best_state, a, label=""):
    """Evaluate objective and update best_state if improved."""
    S = a.sum()
    if S < 0.01 or len(a) < 2:
        return
    c = fft_convolve_self(a)
    obj = 2.0 * len(a) * c.max() / (S * S)
    if obj < best_state['obj']:
        best_state['obj'] = obj
        best_state['a'] = a.copy()
        print(f"  ** NEW BEST: obj={obj:.6f} ({label})")


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    
    Multi-phase strategy:
    Phase 1: Alternating LP (solves global subproblems, avoids local minima)
    Phase 2: LSE polishing for fine convergence
    Phase 3: Subgradient complement
    Phase 4: Perturbation restarts
    Phase 5: Final intensive LSE
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    print(f"Starting optimization, budget={budget_s}s, deadline in {budget_s-30:.0f}s")

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 256:
            print(f"Provided sequence: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        pass

    best_state = {'obj': np.inf, 'a': None}

    # Determine primary length
    if a_provided is not None:
        n_primary = len(a_provided)
    else:
        n_primary = 8192
    S_primary = np.sqrt(2.0 * n_primary)

    # Initial sequence
    if a_provided is not None and len(a_provided) == n_primary:
        a_init = a_provided * (S_primary / a_provided.sum())
    else:
        tt = np.linspace(0, 1, n_primary, endpoint=False)
        a_init = 0.5 * (1.0 + np.cos(2.0 * np.pi * tt)) + 0.05
        a_init = project_simplex(a_init, S_primary)

    # Evaluate initial
    try_update_best(best_state, a_init, "initial")

    # === Phase 1: Alternating LP ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Alternating LP, n={n_primary}")
    print(f"{'='*60}")

    if n_primary <= 8192:
        print(f"  Precomputing convolution matrix structure...")
        t0 = time.monotonic()
        conv_rows, conv_cols = precompute_conv_indices(n_primary)
        print(f"  Structure: nnz={len(conv_rows)}, time={time.monotonic()-t0:.1f}s")

        # Check if we have enough time
        remaining_for_lp = deadline - time.monotonic()
        t_alloc = min(500.0, remaining_for_lp * 0.3)
        if t_alloc > 60:
            print(f"  LP allocation: {t_alloc:.0f}s")
            t1 = time.monotonic()
            a_lp, obj_lp = alternating_lp(a_init, S_primary, n_primary, conv_rows, conv_cols,
                                           max_iter=25, time_limit=t_alloc)
            print(f"  LP done: {time.monotonic()-t1:.1f}s, best obj={obj_lp:.6f}")
            try_update_best(best_state, a_lp, "Phase 1 LP")
        else:
            print(f"  Not enough time for LP ({t_alloc:.0f}s < 60s), skipping")
    else:
        print(f"  n={n_primary} too large for LP, skipping")

    # === Phase 2: LSE Polish ===
    remaining = deadline - time.monotonic()
    if best_state['a'] is not None and remaining > 60:
        n = len(best_state['a'])
        S = np.sqrt(2.0 * n)
        a_work = best_state['a'] * (S / best_state['a'].sum())

        t_alloc = min(350.0, remaining * 0.3)
        print(f"\n{'='*60}")
        print(f"Phase 2: LSE polish, n={n}, time={t_alloc:.0f}s")
        print(f"{'='*60}")

        a_pol, _ = lse_polish(a_work, S, t_alloc, rng,
                              beta_init=30.0, beta_max=30000.0, verbose=True)
        try_update_best(best_state, a_pol, "Phase 2 LSE")

    # === Phase 3: Subgradient ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n = n_primary
        S = np.sqrt(2.0 * n)

        if best_state['a'] is not None and len(best_state['a']) != n:
            t_old = np.linspace(0, 1, len(best_state['a']))
            t_new = np.linspace(0, 1, n)
            a_start = np.interp(t_new, t_old, best_state['a'])
        elif best_state['a'] is not None:
            a_start = best_state['a'].copy()
        else:
            a_start = a_init.copy()
        a_start = project_simplex(a_start, S)

        t_alloc = min(300.0, remaining * 0.25)
        print(f"\n{'='*60}")
        print(f"Phase 3: Subgradient, n={n}, time={t_alloc:.0f}s")
        print(f"{'='*60}")

        a_sg, _ = subgradient_optimize(a_start, S, t_alloc, rng, verbose=True)
        try_update_best(best_state, a_sg, "Phase 3 subgradient")

    # === Phase 4: Perturbation + Refinement ===
    remaining = deadline - time.monotonic()
    if best_state['a'] is not None and remaining > 50:
        n = len(best_state['a'])
        S = np.sqrt(2.0 * n)
        n_restarts = max(2, int(remaining / 180))
        t_per = (remaining - 20) / n_restarts

        print(f"\n{'='*60}")
        print(f"Phase 4: {n_restarts} perturbations, {t_per:.0f}s each")
        print(f"{'='*60}")

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            noise_scales = [0.005, 0.01, 0.02, 0.015, 0.008]
            ns = noise_scales[ri % len(noise_scales)] * S / np.sqrt(n)
            a_pert = project_simplex(best_state['a'] + rng.normal(0, ns, n), S)

            t_half = t_per * 0.5
            a_pert, _ = subgradient_optimize(a_pert, S, t_half, rng, verbose=False)
            try_update_best(best_state, a_pert, f"Phase 4 r{ri} subgrad")

            if time.monotonic() > deadline - 15:
                break
            rem = min(t_half, deadline - time.monotonic() - 10)
            if rem > 5:
                a_pert, _ = lse_polish(a_pert, S, rem, rng,
                                       beta_init=200.0, beta_max=50000.0, verbose=False)
                try_update_best(best_state, a_pert, f"Phase 4 r{ri} LSE")

    # === Phase 5: Final intensive LSE ===
    remaining = deadline - time.monotonic()
    if best_state['a'] is not None and remaining > 30:
        n = len(best_state['a'])
        S = np.sqrt(2.0 * n)
        a_work = best_state['a'] * (S / best_state['a'].sum())

        print(f"\n{'='*60}")
        print(f"Phase 5: Final LSE, n={n}, time={remaining-10:.0f}s")
        print(f"{'='*60}")

        a_final, _ = lse_polish(a_work, S, remaining - 10, rng,
                                beta_init=500.0, beta_max=100000.0, verbose=True)
        try_update_best(best_state, a_final, "Phase 5 final")

    # === Final verification ===
    if best_state['a'] is None:
        print("WARNING: No valid solution, using fallback")
        best_state['a'] = np.ones(n_primary) * (S_primary / n_primary)
        best_state['obj'] = 2.0

    final_a = np.clip(best_state['a'], 0.0, 1000.0)
    if final_a.sum() < 0.01:
        final_a = np.ones(n_primary) * 1.0

    final_list = [float(x) for x in final_a]
    ver = evaluate_sequence(final_list)
    n_final = len(final_a)

    print(f"\n{'='*60}")
    print(f"FINAL: len={n_final}, internal={best_state['obj']:.6f}, evaluate={ver:.6f}")
    print(f"{'='*60}")

    return final_list
```
