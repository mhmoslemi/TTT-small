- For each i in {1,...,m}: the subgradient from lag ki should be a convex combination that, together with the constraint gradients (from a >= 0 and sum(a) = S), spans the normal cone at the optimum.

The subgradient from lag k is 2 * J_k a (where J_k is the matrix for lag k).

So at the optimum, there exist non-negative multipliers λ_1, ..., λ_m (summing to 1) and multipliers μ_i >= 0 for the non-negativity constraints such that:
2 * sum_i λ_i * J_{k_i} a = ν * 1 - sum_j μ_j * e_j

where ν is the multiplier for the sum constraint and e_j are standard basis vectors.

This means: for all j where a_j > 0 (μ_j = 0):
2 * sum_i λ_i * (J_{k_i} a)_j = ν

And for j where a_j = 0 (μ_j >= 0):
2 * sum_i λ_i * (J_{k_i} a)_j <= ν

So the "active" subgradient should be constant (= ν) on the support of a, and <= ν outside.

This gives us a way to check if we're at the optimum: compute the active subgradient, check if it's constant on the support.

Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences of length 8192.

Algorithm:

2. Multi-resolution LSE-FISTA optimization (low beta to high beta)

4. Basin hopping with structural mutations
5. Final KKT-guided polishing

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, c, beta):
    """Gradient of log-sum-exp smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip to prevent overflow
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def compute_subgradient_for_lag(a, k, fsize=None):
    """Compute subgradient 2*J_k*a for a specific lag k."""
    n = len(a)
    if fsize is None:
        fsize = 1 << (2 * n - 1).bit_length()
    # J_k * a: this is the cross-correlation of a with a shifted by k
    # (J_k a)_i = a[i] * a[i+k] + a[i+k] * a[i] ... 
    # Actually, the gradient of b_k = sum_i a_i * a_{i+k} (for k >= 0)
    # db_k/da_j = a_{j+k} + a_{j-k} (with appropriate boundaries)
    # In FFT terms: grad_k = 2 * ifft(fft(e_k) * conj(fft(a)))[:n]
    w = np.zeros(fsize)
    w[k] = 1.0
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    that is most constant on the support of a.

    cmax = c.max()
    active = np.where(c >= cmax - tol * cmax)[0]
    active = active[:max_active]
    
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    if len(active) == 1:
        fsize = 1 << (2 * n - 1).bit_length()
        return compute_subgradient_for_lag(a, active[0], fsize)
    
    support = a > 1e-10
    nsup = support.sum()
    if nsup < 2:
        fsize = 1 << (2 * n - 1).bit_length()
        return compute_subgradient_for_lag(a, active[0], fsize)

def fourier_crossover(a1, a2, n, S, rng):

    new_A[0] = A1[0]
    a_new = np.real(ifft(new_A, fsize))[:n]

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lag_subgrad(a, k):
    """Compute 2*B_k*a (subgradient of conv[a,k]) efficiently without FFT."""
    n = len(a)
    result = np.zeros(n)
    i_start = max(0, k - n + 1)
    i_end = min(k, n - 1)
    if i_start <= i_end:
        indices = np.arange(i_start, i_end + 1)
        result[indices] = 2.0 * a[k - indices]
    return result


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def optimal_active_subgradient(a, c, tol=1e-4, max_active=40):
    """
    Find optimal convex combination of active-lag subgradients
    that is most constant on the support of a (KKT-aligned direction).
    """
    n = len(a)
    cmax = c.max()
    if cmax < 1e-30:
        return np.zeros(n)
    active = np.where(c >= cmax - tol * cmax)[0]
    active = active[:max_active]
    if len(active) == 0:
        active = np.array([np.argmax(c)])

    if len(active) == 1:
        return lag_subgrad(a, int(active[0]))

    support = a > 1e-10 * a.max()
    sup_idx = np.where(support)[0]
    nsup = len(sup_idx)
    if nsup < 2:
        return lag_subgrad(a, int(active[0]))

    # Compute subgradients on support for each active lag
    m = len(active)
    grads_sup = np.zeros((m, nsup))
    for i, k in enumerate(active):
        g_full = lag_subgrad(a, int(k))
        grads_sup[i] = g_full[sup_idx]

    # Find lambda >= 0, sum=1 minimizing variance of (grads_sup @ lambda)
    lam = np.ones(m) / m
    # Projected gradient descent on simplex
    # Objective: ||M@lam||^2 - (1/nsup)*(sum(M@lam))^2 = lam^T M^T M lam - (1/nsup)(1^T M lam)^2
    # Gradient: 2*M^T(M@lam) - (2/nsup)*M^T@(1^T M lam)*1
    M = grads_sup
    MtM = M.T @ M
    M1 = M.T @ np.ones(nsup)
    for _ in range(100):
        g_lam = 2.0 * (MtM @ lam) - (2.0 / nsup) * M1 * (M1 @ lam)
        # Lipschitz constant of g_lam
        L = 4.0 * np.linalg.norm(MtM, 2) / nsup + 1e-10
        step = 1.0 / L
        lam_new = project_simplex(lam - step * g_lam, 1.0)
        if np.linalg.norm(lam_new - lam) < 1e-12:
            lam = lam_new
            break
        lam = lam_new

    # Compute combined subgradient
    combined = np.zeros(n)
    for i in range(m):
        if lam[i] > 1e-10:
            combined += lam[i] * lag_subgrad(a, int(active[i]))
    return combined


def run_lse_fista(a_init, S, max_time, rng, beta_start=10.0, beta_end=50000.0):
    """FISTA with progressively increasing LSE beta."""
    n = len(a_init)
    a = project_simplex(np.maximum(a_init, 0.0), S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        beta = beta_start * (beta_end / beta_start) ** progress
        if beta > 1e6:
            beta = 1e6

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta + 10.0)

        if it - last_improve > 4000 and progress > 0.2:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 8000.0)
            if noise_scale > 1e-14:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 10000 == 0:
            print("      LSE iter %d: beta=%.0f, max_b=%.10f, obj=%.6f" %
                  (it, beta, best, 2.0 * n * best / S**2))

    return best_a, best


def run_active_set_polish(a_init, S, max_time, rng):
    """Polish using optimal active-set subgradients with FISTA momentum."""
    n = len(a_init)
    a = project_simplex(np.maximum(a_init, 0.0), S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    y = a.copy()
    t_mom = 1.0
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        tol = max(1e-5, 5e-4 * (1.0 - progress))
        grad = optimal_active_subgradient(y, c, tol=tol, max_active=30)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 0.3 * S / (n * (1.0 + 0.0005 * it))
        if it - last_improve > 3000:
            step = max(step * 0.3, 1e-6 * S / n)
            noise = rng.normal(0, 1e-5 * S / n, n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
            last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 5000 == 0:
            print("      AS iter %d: max_b=%.10f" % (it, best))

    return best_a, best


def block_perturb(a, S, rng, n):
    """Perturb random blocks of the sequence."""
    a_new = a.copy()
    n_blocks = rng.integers(4, 15)
    for _ in range(n_blocks):
        st = rng.integers(0, n)
        ln = rng.integers(5, max(10, n // 40))
        en = min(st + ln, n)
        if en > st:
            a_new[st:en] *= rng.uniform(0.4, 2.5)
    return project_simplex(a_new, S)


def sinusoidal_perturb(a, S, rng, n):
    """Add random sinusoidal perturbations."""
    a_new = a.copy()
    for _ in range(rng.integers(1, 4)):
        freq = rng.uniform(1, n / 15)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.15) * S / n
        x = np.arange(n)
        a_new += amp * np.sin(2 * np.pi * freq * x / n + phase)
    return project_simplex(a_new, S)


def fourier_phase_mutate(a, S, rng, n):
    """Fourier phase perturbation with frequency-dependent scaling."""
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.08 * fsize)
    scale = 0.03 + 0.1 * rng.random()
    noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def fourier_crossover(a1, a2, n, S, rng):
    """Fourier-domain crossover."""
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    r = rng.random(fsize)
    alpha = 0.3 + 0.4 * r
    new_mag = alpha * np.abs(A1) + (1 - alpha) * np.abs(A2)
    new_phase = np.where(r < 0.5, np.angle(A1), np.angle(A2))
    new_A = new_mag * np.exp(1j * new_phase)
    new_A[0] = A1[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def peak_rebalance(a, S, rng, n, n_peaks=5):
    """
    Direct peak reduction: identify top peaks in convolution,
    reduce contributors proportionally, redistribute to flat areas.
    """
    c = fft_convolve_self(a)
    cmax = c.max()
    # Find lags near the peak
    threshold = cmax - 0.1 * cmax
    hot_lags = np.where(c >= threshold)[0]
    if len(hot_lags) == 0:
        hot_lags = np.array([np.argmax(c)])
    hot_lags = hot_lags[:n_peaks]

    # Compute "heat" at each index: sum of contributions from hot lags
    heat = np.zeros(n)
    for k in hot_lags:
        i_start = max(0, int(k) - n + 1)
        i_end = min(int(k), n - 1)
        if i_start <= i_end:
            indices = np.arange(i_start, i_end + 1)
            heat[indices] += a[k - indices]

    # Reduce hot indices, increase cool indices
    total_heat = heat.sum()
    if total_heat < 1e-20:
        return a
    a_new = a.copy()
    # Reduce where heat is high
    reduction = np.minimum(a_new, 0.1 * heat)
    a_new -= reduction
    # Redistribute to low-heat areas
    low_mask = heat < 0.3 * heat.max()
    if low_mask.any():
        a_new[low_mask] += reduction.sum() / low_mask.sum()
    return project_simplex(a_new, S)


def run_basin_hopping(a_init, S, max_time, rng):
    """Basin hopping with diverse structural mutations."""
    n = len(a_init)
    a = project_simplex(np.maximum(a_init, 0.0), S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    opt_time = max(15.0, min(60.0, max_time / 15.0))
    population = [(best, a.copy())]
    it = 0

    while time.monotonic() - start < max_time - opt_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        r = rng.random()

        if r < 0.2 and len(population) >= 2:
            idx1 = rng.integers(len(population))
            idx2 = rng.integers(len(population))
            if idx1 != idx2:
                a = fourier_crossover(population[idx1][1], population[idx2][1], n, S, rng)
            else:
                a = block_perturb(a, S, rng, n)
        elif r < 0.4:
            a = block_perturb(a, S, rng, n)
        elif r < 0.55:
            a = sinusoidal_perturb(a, S, rng, n)
        elif r < 0.7:
            a = fourier_phase_mutate(a, S, rng, n)
        elif r < 0.85:
            a = peak_rebalance(a, S, rng, n)
            # Small random perturbation on top
            a = project_simplex(a + rng.normal(0, 0.01 * S / n, n), S)
        else:
            # Larger structural change
            a = block_perturb(a, S, rng, n)
            a = fourier_phase_mutate(a, S, rng, n)

        # Local optimization with moderate beta range
        beta_start = 30.0 + 50.0 * min(it, 10)
        beta_end = 5000.0 + 5000.0 * min(it, 20)
        a_opt, val = run_lse_fista(a, S, opt_time, rng,
                                    beta_start=beta_start, beta_end=beta_end)
        if val < best:
            best = val
            best_a = a_opt.copy()
            print("      BH iter %d: new best max_b=%.10f (obj=%.6f)" %
                  (it, val, 2.0 * n * val / S**2))

        population.append((val, a_opt.copy()))
        if len(population) > 8:
            population.sort(key=lambda x: x[0])
            population = population[:8]

    return best_a, best


def run_mirror_descent_polish(a_init, S, max_time, rng):
    """Mirror descent (KL-proximal) polish for non-Euclidean exploration."""
    n = len(a_init)
    a = project_simplex(np.maximum(a_init, 0.0), S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()

        # Use active-set subgradient
        tol = max(1e-5, 1e-3 * (1.0 - progress))
        grad = optimal_active_subgradient(a, c, tol=tol, max_active=25)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Mirror descent step: a_j <- a_j * exp(-eta * grad_j / gmax)
        eta = 0.5 * (1.0 - 0.5 * progress)
        gmax = np.max(np.abs(grad))
        if gmax < 1e-20:
            break
        scaled = -np.clip(eta * grad / gmax, -20, 20)
        a_new = a * np.exp(scaled)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new * (S / s)
        a = np.maximum(a_new, 0.0)

        # Occasionally add small noise to escape flat regions
        if it % 100 == 0:
            noise = rng.normal(0, 1e-5 * S / n, n)
            a = project_simplex(a + noise, S)

    return best_a, best


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. LSE-FISTA on given/best starting point (multi-resolution beta)
    2. Active-set polish with optimal multiplier combination
    3. Basin hopping with diverse mutations (block, sinusoidal, Fourier, peak-rebalance)
    4. Mirror descent polish
    5. Final high-beta LSE + active-set refinement
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30

    n = 8192
    S = np.sqrt(2.0 * n)
    global_best = np.inf
    global_a = None

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    # Initialize
    if given is not None and len(given) == n:
        a0 = project_simplex(given, S)
    else:
        a0 = np.ones(n) * S / n

    obj0 = objective_value(a0)
    global_best = obj0
    global_a = a0.copy()
    print("Initial: obj=%.6f" % obj0)

    # Phase 1: Multi-resolution LSE on best starting point
    print("\n=== Phase 1: LSE-FISTA (400s) ===")
    p1_budget = min(400.0, (deadline - time.monotonic()) * 0.22)
    a1, val1 = run_lse_fista(global_a, S, p1_budget, rng,
                              beta_start=10.0, beta_end=50000.0)
    obj1 = 2.0 * n * val1 / (a1.sum() ** 2)
    if obj1 < global_best:
        global_best = obj1
        global_a = a1.copy()
        print("  After LSE: obj=%.6f" % obj1)

    # Phase 2: Active-set polish
    print("\n=== Phase 2: Active-set polish (250s) ===")
    p2_budget = min(250.0, (deadline - time.monotonic()) * 0.15)
    a2, val2 = run_active_set_polish(global_a, S, p2_budget, rng)
    obj2 = 2.0 * n * val2 / (a2.sum() ** 2)
    if obj2 < global_best:
        global_best = obj2
        global_a = a2.copy()
        print("  After active-set: obj=%.6f" % obj2)

    # Phase 3: Basin hopping
    print("\n=== Phase 3: Basin hopping (1000s) ===")
    p3_budget = min(1000.0, (deadline - time.monotonic()) * 0.50)
    a3, val3 = run_basin_hopping(global_a, S, p3_budget, rng)
    obj3 = 2.0 * n * val3 / (a3.sum() ** 2)
    if obj3 < global_best:
        global_best = obj3
        global_a = a3.copy()
        print("  After BH: obj=%.6f" % obj3)

    # Phase 4: Mirror descent polish
    print("\n=== Phase 4: Mirror descent polish (150s) ===")
    p4_budget = min(150.0, (deadline - time.monotonic()) * 0.12)
    a4, val4 = run_mirror_descent_polish(global_a, S, p4_budget, rng)
    obj4 = 2.0 * n * val4 / (a4.sum() ** 2)
    if obj4 < global_best:
        global_best = obj4
        global_a = a4.copy()
        print("  After mirror: obj=%.6f" % obj4)

    # Phase 5: Final refinement
    print("\n=== Phase 5: Final refinement ===")
    remaining = deadline - time.monotonic() - 10
    if remaining > 60 and global_a is not None:
        # High-beta LSE
        seg1 = remaining * 0.5
        a5, val5 = run_lse_fista(global_a, S, seg1, rng,
                                  beta_start=500.0, beta_end=100000.0)
        obj5 = 2.0 * n * val5 / (a5.sum() ** 2)
        if obj5 < global_best:
            global_best = obj5
            global_a = a5.copy()
            print("  After high-beta LSE: obj=%.6f" % obj5)

        # Final active-set
        remaining2 = deadline - time.monotonic() - 5
        if remaining2 > 20:
            a6, val6 = run_active_set_polish(global_a, S, remaining2, rng)
            obj6 = 2.0 * n * val6 / (a6.sum() ** 2)
            if obj6 < global_best:
                global_best = obj6
                global_a = a6.copy()
                print("  After final active-set: obj=%.6f" % obj6)

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(n) * S / n
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\nVerification: obj=%.6f" % ver)

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Using uniform fallback.")
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback obj: %.6f" % ver)

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.6f, n=%d" % (elapsed, ver, len(global_a)))
    return final_list
