"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy:
1. Spectral-shaping initialization (Gerchberg-Saxton inspired)
2. Improved FISTA with adaptive LSE and gradient-norm-based step size
3. Peak-directed coordinate descent (mass transfer from peak-contributing pairs)
4. Multi-scale upsample for more degrees of freedom
5. Diverse restarts to escape local minima
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def spectral_shaping(a_init, n_iters=200, rng=None):
    """
    Gerchberg-Saxton style spectral shaping to flatten the autocorrelation.
    Target: make |A(f)| as flat as possible (which approximately flattens b).
    """
    n = len(a_init)
    S = a_init.sum()
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    a = a * (S / a.sum())
    
    fsize = 1 << (2 * n - 2).bit_length()
    # Target magnitude: S / sqrt(n) for all frequencies
    target_mag = S / np.sqrt(n)
    
    for it in range(n_iters):
        A = fft(a, fsize)
        # Set magnitude to target (keep phase)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        # For f=0, keep the original magnitude (which should be ~S)
        new_mag = np.full_like(mag, target_mag)
        new_mag[0] = mag[0]  # Keep DC
        A_new = phase * new_mag
        # Ensure Hermitian symmetry for real output
        a_new = np.real(ifft(A_new, fsize))[:n]
        a = np.maximum(a_new, 0.0)
        if a.sum() < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / a.sum())
    
    return a


def lse_gradient(y, beta, c=None):
    """Compute LSE value and gradient of the max function w.r.t. y."""
    n = len(y)
    if c is None:
        c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c, w


def run_fista_v2(a_init, max_time, rng, beta_init=10.0, beta_rate=0.02,
                 beta_max=50000.0, use_backtrack=True, max_restarts=5):
    """
    Improved FISTA with:
    - Gradient-norm-based step size
    - Aggressive beta growth
    - Restart from best point
    - L2 regularization to avoid degeneracy
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / max(beta_init, 1.0)
    no_improve_count = 0
    restart_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_gradient(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Adaptive step size based on gradient norm
        # The Lipschitz constant is roughly 2 * beta * max(w) * ||a||^2
        # For safety, use a smaller step
        step_base = 1.0 / (beta * gnorm) if beta > 1 else 1.0 / gnorm
        step_local = min(step_base * 0.5, 2.0 * S / gnorm)
        step_local = max(step_local, 1e-15)
        
        if use_backtrack:
            obj_trial = compute_obj(project_simplex(y - step_local * grad, S))
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-16 and bt < 8:
                step_local *= 0.5
                obj_trial = compute_obj(project_simplex(y - step_local * grad, S))
                bt += 1
            if bt > 3:
                step = step_local * 0.7
            elif bt < 2:
                step = min(step * 1.05, 2.0 / max(beta, 1.0))
            else:
                step = step_local
            x_new = project_simplex(y - step_local * grad, S)
        else:
            step = step_local
            x_new = project_simplex(y - step * grad, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        # Restart strategy
        if no_improve_count > 2000 and restart_count < max_restarts:
            restart_count += 1
            t_mom = 1.0
            # Inject noise
            noise_scale = 0.01 * S / np.sqrt(n)
            y = project_simplex(best_a + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            step = 1.0 / max(beta, 1.0)
            no_improve_count = 0
        
        if no_improve_count > 5000:
            break
    
    return best_a, best_obj


def peak_directed_cd(a_init, max_time, rng, top_k_lags=3, transfer_frac=0.05):
    """
    Peak-directed coordinate descent:
    - Find the top-k peak lags
    - For each peak, identify the pair (i, k-i) with maximum product
    - Transfer mass from that pair to a low-contribution coordinate
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        if it - last_improve > 10000:
            break
        
        # Find top-k peak lags
        top_idx = np.argsort(c)[::-1][:top_k_lags]
        
        # For each peak, find the max-contributing pair
        best_improvement = 0
        best_action = None
        
        for k_star in top_idx:
            k_star = int(k_star)
            # contrib[i] = a[i] * a[k_star - i] for valid i
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            if hi < lo:
                continue
            # Pairs: i from lo to hi, partner is k_star - i
            i_range = np.arange(lo, hi + 1)
            partners = k_star - i_range
            products = a[i_range] * a[partners]
            
            # Find the pair with max product
            max_prod_idx = int(np.argmax(products))
            i_max = int(i_range[max_prod_idx])
            j_max = int(partners[max_prod_idx])
            
            if i_max == j_max:
                continue
            
            # Determine source (larger value) and target
            if a[i_max] >= a[j_max]:
                src = i_max
                other = j_max
            else:
                src = j_max
                other = i_max
            
            delta = a[src] * transfer_frac
            if delta < 1e-12:
                continue
            
            # Find a good target: coordinate with low contribution to all top peaks
            # Compute contribution of each coordinate to the top peaks
            contrib_total = np.zeros(n)
            for kk in top_idx:
                kk = int(kk)
                lo2 = max(0, kk - n + 1)
                hi2 = min(n - 1, kk)
                if hi2 >= lo2:
                    i_range2 = np.arange(lo2, hi2 + 1)
                    partners2 = kk - i_range2
                    contrib_total[i_range2] += a[partners2]
                    contrib_total[partners2] += a[i_range2]
            
            # Target: low contribution, not the source
            target_candidates = np.where(contrib_total < contrib_total[src] * 0.5)[0]
            target_candidates = target_candidates[target_candidates != src]
            
            if len(target_candidates) == 0:
                target = rng.integers(0, n)
                if target == src:
                    target = (target + 1) % n
            else:
                # Pick the one with lowest contribution
                target = int(target_candidates[np.argmin(contrib_total[target_candidates])])
            
            # Try the transfer
            a_trial = a.copy()
            a_trial[src] -= delta
            a_trial[target] += delta
            
            # Quick check: does this reduce the peak?
            # Only need to check the affected lags
            # Lags affected by changing a[src]: k in [src, src+n-1] (and [src-n+1, src] if applicable)
            # For simplicity, recompute c (expensive but correct)
            if it % 5 == 0 or cur_obj - best_obj > 0.001:
                c_trial = fft_convolve_self(a_trial)
                obj_trial = 2.0 * n * c_trial.max() / (S * S)
                if obj_trial < cur_obj - 1e-14:
                    best_improvement = cur_obj - obj_trial
                    best_action = (src, target, delta, c_trial)
            else:
                # Approximate check: just check the top peaks
                affected = True
                for kk in top_idx:
                    kk = int(kk)
                    # New c[kk] = c[kk] - delta * a[kk-src] + delta * a[kk-target] (if valid)
                    old_val = c[kk]
                    delta_c = 0
                    if 0 <= kk - src < n:
                        delta_c -= delta * a[kk - src]
                    if 0 <= kk - target < n:
                        delta_c += delta * a[kk - target]
                    # Also the "other" side
                    if 0 <= kk - src < n:
                        # a[src] also appears as the "second" factor when partner is src
                        pass  # Already accounted for
                    new_val = old_val + delta_c
                    if new_val > cur_max + 1e-12:
                        affected = False
                        break
                if affected:
                    best_improvement = 1.0  # placeholder
                    best_action = (src, target, delta, None)
        
        if best_action is not None:
            src, target, delta, c_new = best_action
            a[src] -= delta
            a[target] += delta
            if c_new is not None:
                c = c_new
            else:
                if it % 10 == 0:
                    c = fft_convolve_self(a)
        else:
            # No improvement: try random perturbation
            if it - last_improve > 1000:
                src = int(rng.integers(0, n))
                if a[src] > 1e-10:
                    tgt = int(rng.integers(0, n))
                    delta = a[src] * 0.01
                    a[src] -= delta
                    a[tgt] += delta
                    c = fft_convolve_self(a)
        
        if it % 200 == 0:
            c = fft_convolve_self(a)

def exponential_gradient(a_init, max_time, rng, eta0=0.1, decay=0.001):
    """
    Exponential gradient (multiplicative weights) for minimizing max_k b[k].
    Naturally maintains non-negativity and sum constraint.
    """

    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = np.maximum(a, 1e-10)  # Avoid zeros
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        if it - last_improve > 5000:
            break
        
        k_star = int(np.argmax(c))
        
        # Subgradient: g[i] = 2 * a[k_star - i] for valid i
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi >= lo:
            g[lo:hi+1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        eta = eta0 / (1.0 + decay * it)
        eta = min(eta, 0.5 * S / gnorm)
        
        # Multiplicative update
        a = a * np.exp(-eta * g)
        a = np.maximum(a, 1e-15)
        a = a * (S / a.sum())
    
    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_diverse_inits(n, rng, n_inits=8):
    """Create diverse initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    inits = []
    
    # Cosine
    a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    inits.append(np.maximum(a, 0.01) * (S / np.maximum(a, 0.01).sum()))
    
    # Sine
    a = 0.5 + 0.5 * np.sin(2 * np.pi * 3 * tt)
    inits.append(np.maximum(a, 0.01) * (S / np.maximum(a, 0.01).sum()))
    
    # Random
    a = rng.random(n) + 0.1
    inits.append(a * (S / a.sum()))
    
    # Two-hump
    a = np.exp(-8 * (tt - 0.25)**2) + np.exp(-8 * (tt - 0.75)**2)
    inits.append(a * (S / a.sum()))
    
    # Multi-frequency
    a = 1.0
    for freq, amp, phase in [(2, 0.3, 0), (5, 0.2, 1.0), (11, 0.15, 2.0)]:
        a += amp * np.sin(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    inits.append(a * (S / a.sum()))
    
    # Tent (high at center)
    a = 1.0 - 2.0 * np.abs(tt - 0.5)
    inits.append(a * (S / a.sum()))
    
    # Gaussian
    a = np.exp(-4 * (tt - 0.5)**2 / (0.2**2))
    inits.append(a * (S / a.sum()))
    
    # Exponential decay
    a = np.exp(-3 * tt)
    inits.append(a * (S / a.sum()))
    
    return inits
