```python
Minimize 2*n*max(f*f)/sum(f)^2 using p-norm gradient descent + active-set QP.

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def cross_corr_weights(w, f):
    """Compute (w ⊛ f)(i) = sum_k w[k] f[k-i] for i = 0, ..., n-1."""
    n = len(f)
    L = 1 << (3 * n - 3).bit_length()
    W = fft(w, L)
    F = fft(f, L)
    corr = np.real(ifft(W * np.conj(F)))
    return corr[:n]


def pnorm_grad(f, c, p):
    """Gradient of (sum |c_k|^p)^(1/p) w.r.t. f."""
    n = len(f)
    c_p = np.abs(c) ** p
    C = c_p.sum()
    J = C ** (1.0 / p)
    if J < 1e-30:
        return np.zeros(n), 0.0
    # weights: w_k = |c_k|^(p-1) * sign(c_k) / J^(p-1)
    weights = (np.abs(c) ** (p - 1)) * np.sign(c)
    weights = weights / (J ** (p - 1))
    # grad ≈ 2 * (weights ⊛ f)
    grad = 2.0 * cross_corr_weights(weights, f)
    return grad, J


def matvec_Mp(p, f):
    """Compute M(p) @ f where M(p)_{i,j} = p_{i+j}."""
    n = len(f)
    g = f[::-1]
    size = 3 * n - 2
    fsize = 1 << (size - 1).bit_length()
    G = fft(g, fsize)
    P = fft(p, fsize)
    conv = np.real(ifft(G * P))[:size]
    return conv[n-1:n-1+n]


def solve_qp_active(f_init, p, S, n, max_iter=200, tol=1e-10):
    """
    Solve min f^T M(p) f s.t. f >= 0, sum(f) = S using projected gradient.
    p is a probability vector of length 2n-1.
    """
    f = f_init.copy()
    # Estimate Lipschitz constant
    # lambda_max of M(p) is at most sum(p) * max |p_k| * n (rough bound)
    # Better: use power iteration
    v = f / np.linalg.norm(f)
    for _ in range(20):
        Mv = matvec_Mp(p, v)
        nv = np.linalg.norm(Mv)
        if nv < 1e-30:
            break
        v = Mv / nv
    lambda_max = v @ matvec_Mp(p, v)
    eta = 1.0 / (2.0 * max(lambda_max, 1e-10))
    
    for it in range(max_iter):
        grad = 2.0 * matvec_Mp(p, f)
        f_new = project_simplex(f - eta * grad, S)
        if np.linalg.norm(f_new - f) < tol * S:
            f = f_new
            break
        f = f_new
    
    return f


def active_set_qp_refine(f, S, n, max_outer=50, n_active_threshold=1e-6):
    """
    Alternating: update active set p, solve QP for f.
    """
    f = f.copy()
    for outer in range(max_outer):
        c = fft_convolve_self(f)
        c_max = c.max()
        
        # Determine active set: lags within threshold of max
        threshold = c_max * (1 - n_active_threshold)
        active = c >= threshold
        if active.sum() == 0:
            active[np.argmax(c)] = True
        
        # p is uniform over active set
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / active.sum()
        
        # Solve QP
        f_qp = solve_qp_active(f, p, S, n, max_iter=100)
        
        # Check convergence
        f_new = f_qp
        if np.linalg.norm(f_new - f) < 1e-10 * S:
            f = f_new
            break
        
        f = f_new
    
    return f
```

```python
"""
Minimize 2*n*max(f*f)/sum(f)^2 using p-norm gradient descent, active-set QP
alternation, and subgradient polishing. Key innovation over previous approach:
(1) Use exact p-norm gradients (no LSE approximation error), 
(2) Solve the active-set QP via projected gradient with FFT mat-vec,
(3) Growing active set that adapts to the problem structure,
(4) Adaptive step sizes with backtracking for the final "last mile."
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def cross_corr_weights(w, f):
    """Compute (w cross-corr f)(i) = sum_k w[k] * f[k-i] for i=0..n-1.
    w has length 2n-1, f has length n."""
    n = len(f)
    L = 1 << (3 * n - 3).bit_length()
    W = fft(w, L)
    F = fft(f, L)
    corr = np.real(ifft(W * np.conj(F)))
    return corr[:n]


def pnorm_grad(f, c, p):
    """Gradient of (sum|c_k|^p)^(1/p) w.r.t. f. Returns (grad, J)."""
    n = len(f)
    c_abs = np.abs(c)
    # Compute J = (sum |c_k|^p)^(1/p) in a numerically stable way
    max_c = c_abs.max()
    if max_c < 1e-300:
        return np.zeros(n), 0.0
    scaled = c_abs / max_c
    J = max_c * (scaled ** p).sum() ** (1.0 / p)
    if J < 1e-30:
        return np.zeros(n), 0.0
    # Weights: w_k = (|c_k|/J)^(p-1) * sign(c_k)
    ratio = c_abs / J
    ratio = np.minimum(ratio, 1.0)
    weights = ratio ** (p - 1) * np.sign(c)
    grad = 2.0 * cross_corr_weights(weights, f)
    return grad, J


def matvec_Mp(p, f):
    """Compute M(p) @ f where M(p)_{i,j} = p_{i+j}.
    p has length 2n-1, f has length n."""
    n = len(f)
    g = f[::-1]
    size = 3 * n - 2
    fsize = 1 << (size - 1).bit_length()
    G = fft(g, fsize)
    P = fft(p, fsize)
    conv = np.real(ifft(G * P))[:size]
    return conv[n - 1:n - 1 + n]


def estimate_lipschitz(p, f, n, iters=15):
    """Estimate lambda_max of M(p) via power iteration."""
    v = f.copy()
    nv = np.linalg.norm(v)
    if nv < 1e-30:
        return 1.0
    v = v / nv
    lam = 0.0
    for _ in range(iters):
        Mv = matvec_Mp(p, v)
        lam = v @ Mv
        nv = np.linalg.norm(Mv)
        if nv < 1e-30:
            break
        v = Mv / nv
    return max(abs(lam), 1e-10)


def solve_qp_projected_grad(f_init, p, S, n, max_iter=300, tol=1e-12):
    """Solve min f^T M(p) f s.t. f>=0, sum(f)=S using projected gradient."""
    f = f_init.copy()
    lam_max = estimate_lipschitz(p, f, n, iters=10)
    eta = 1.0 / (2.0 * lam_max)
    
    prev_obj = np.inf
    for it in range(max_iter):
        grad = 2.0 * matvec_Mp(p, f)
        f_new = project_simplex(f - eta * grad, S)
        
        if np.linalg.norm(f_new - f) < tol * S:
            f = f_new
            break
        
        f = f_new
        
        if it % 50 == 49:
            obj = f @ matvec_Mp(p, f)
            if prev_obj - obj < 1e-15:
                break
            prev_obj = obj
    
    return f


def active_set_qp_refine(f, S, n, max_outer=40, grow_factor=1.5):

    2. Solve QP for that active set
    3. Grow active set if non-active lags exceed threshold

    best_f = f.copy()
    best_obj = fft_convolve_self(f).max()
    
    for outer in range(max_outer):
        c = fft_convolve_self(f)
        c_max = c.max()
        if c_max < best_obj:
            best_obj = c_max
            best_f = f.copy()
        
        # Determine active set: lags within relative tolerance of max
        rel_tol = max(1e-8, (c_max - best_obj * 0.999) / c_max)
        threshold = c_max * (1 - rel_tol)

        if active.sum() == 0:

    Uses adaptive step size and periodic noise for escape.

    best_f = f.copy()

    stagnation = 0

        c_max = c.max()

            best_f = f.copy()

        else:
            stagnation += 1

            active_mask[np.argmax(c)] = True
        
        # Subgradient: uniform over active lags
        w = np.zeros(2 * n - 1)
        w[active_mask] = 1.0 / active_mask.sum()
        grad = 2.0 * cross_corr_weights(w, f)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size: start moderate, reduce if stuck
        base_eta = 0.1 / (gnorm + 1e-30)
        if stagnation > 50:
            base_eta *= 0.3
        
        # Line search
        eta = base_eta
        improved = False
        for _ in range(6):
            f_new = project_simplex(f - eta * grad, S)
            c_new = fft_convolve_self(f_new)
            if c_new.max() < c_max * (1 - 1e-10):
                f = f_new
                improved = True
                break
            eta *= 0.4
        
        if not improved:
            # Try noise escape
            if stagnation > 20:
                noise_scale = 1e-5 * S / np.sqrt(n) * np.exp(-stagnation / 200.0)
                if noise_scale > 1e-12:
                    f_trial = project_simplex(f + rng.normal(0, noise_scale, n), S)
                    c_trial = fft_convolve_self(f_trial)
                    if c_trial.max() < c_max:
                        f = f_trial
                        stagnation = 0
                    else:
                        if stagnation > 500:
                            break
                else:
                    break
    
    return best_f


def pnorm_descent(f, S, n, max_time, rng, p_start=4, p_end=512, n_phases=8):
    """
    p-norm gradient descent with increasing p.
    """
    f = f.copy()
    start = time.monotonic()
    
    # Create p schedule (logarithmic)
    p_schedule = np.geomspace(p_start, p_end, n_phases)
    time_per_phase = max_time / n_phases
    
    for p_val in p_schedule:
        phase_end = time.monotonic() + time_per_phase
        it = 0
        
        while time.monotonic() < phase_end:
            c = fft_convolve_self(f)
            grad, J = pnorm_grad(f, c, float(p_val))
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            
            # Step size: 1/L where L is Lipschitz constant estimate
            # For p-norm, L ~ 2 * max|w_k| * 2 (rough)
            max_w = np.max(np.abs(c)) / J if J > 0 else 1.0
            max_w = min(max_w, 1.0)
            L_est = 4.0 * max_w ** max(1, int(p_val) - 1) + 0.01
            eta = 1.0 / L_est
            
            # Cap step size
            eta = min(eta, 0.5 / gnorm)
            
            # Try step with backtracking
            improved = False
            for _ in range(8):
                f_new = project_simplex(f - eta * grad, S)
                c_new = fft_convolve_self(f_new)
                new_max = c_new.max()
                old_max = c.max()
                if new_max < old_max:
                    f = f_new
                    improved = True
                    break
                eta *= 0.5
            
            if not improved:
                # Very small step
                eta = 1e-8 / max(gnorm, 1e-30)
                f = project_simplex(f - eta * grad, S)
            
            it += 1
            if it % 2000 == 0:
                c_check = fft_convolve_self(f)
                print("    p=%d, iter=%d, max=%.10f" % (int(p_val), it, c_check.max()))
    
    return f


def try_new_length(a_best, n_new, S_new, time_budget, rng):
    """Try optimizing at a different sequence length via interpolation."""
    n_old = len(a_best)
    t_old = np.linspace(0, 1, n_old)
    t_new = np.linspace(0, 1, n_new)
    init = np.interp(t_new, t_old, a_best)
    init = normalize_sum(init, n_new)
    
    # Quick p-norm optimization
    f = pnorm_descent(init, S_new, n_new, time_budget * 0.6, rng,
                      p_start=8, p_end=128, n_phases=4)
    # Active set refinement
    f = active_set_qp_refine(f, S_new, n_new, max_outer=10)
    
    c = fft_convolve_self(f)
    return f, c.max()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimize 2*n*max(f*f)/sum(f)^2 using a multi-phase strategy:
    Phase 1: p-norm gradient descent (smooth to sharp)
    Phase 2: Active-set QP alternation
    Phase 3: Subgradient polishing with exact max
    Phase 4: Try alternative sequence lengths
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 90.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)
    
    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))
    
    # Load initial sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 128 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception:
        pass
    
    if given is None:
        n = 8192
        given = np.ones(n)
        print("Using uniform fallback: n=%d" % n)
    
    n = len(given)
    S = np.sqrt(2.0 * n)
    global_best = np.inf
    global_a = None
    
    def try_sequence(a, n_local=None):
        nonlocal global_best, global_a
        if n_local is None:
            n_local = len(a)
        S_local = np.sqrt(2.0 * n_local)
        a = normalize_sum(a, n_local)
        c = fft_convolve_self(a)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a.copy()
            print("  ** New best: %.10f (n=%d) **" % (val, n_local))
    
    # Initial evaluation
    a = normalize_sum(given, n)
    c_init = fft_convolve_self(a)
    print("Initial: max=%.10f, n=%d" % (c_init.max(), n))
    try_sequence(a)
    
    # === Phase 1: p-norm gradient descent ===
    remaining = deadline - time.monotonic()
    t1 = remaining * 0.35
    print("=== Phase 1: p-norm descent (%.0fs) ===" % t1)
    
    f = pnorm_descent(a, S, n, t1, rng, p_start=4, p_end=512, n_phases=8)
    try_sequence(f)
    
    # === Phase 2: Active-set QP refinement ===
    remaining = deadline - time.monotonic()
    t2 = min(remaining * 0.30, 300.0)
    print("=== Phase 2: Active-set QP (%.0fs) ===" % t2)
    
    # Run active set QP with time limit
    f_start = f.copy()
    best_phase2 = f_start
    best_phase2_obj = fft_convolve_self(f_start).max()
    start_2 = time.monotonic()
    
    for outer_iter in range(30):
        if time.monotonic() - start_2 > t2:
            break
        
        c = fft_convolve_self(f)
        c_max = c.max()
        
        # Multiple active set sizes to try
        for top_k in [1, 3, 5, 10, 20, 50]:
            if time.monotonic() - start_2 > t2:
                break
            
            sorted_idx = np.argsort(c)[::-1]
            active = np.zeros(2 * n - 1, dtype=bool)
            active[sorted_idx[:top_k]] = True
            
            # Also add lags close to the max
            threshold = c_max * 0.999
            active[c >= threshold] = True
            
            p = np.zeros(2 * n - 1)
            p[active] = 1.0 / active.sum()
            
            # Solve QP
            f_qp = solve_qp_projected_grad(f, p, S, n, max_iter=100)
            c_qp = fft_convolve_self(f_qp)
            c_qp_max = c_qp.max()
            
            if c_qp_max < best_phase2_obj:
                best_phase2_obj = c_qp_max
                best_phase2 = f_qp.copy()
                f = f_qp.copy()
            
            # Also try averaging
            f_avg = 0.5 * (f + f_qp)
            f_avg = normalize_sum(f_avg, n)
            c_avg = fft_convolve_self(f_avg)
            if c_avg.max() < best_phase2_obj:
                best_phase2_obj = c_avg.max()
                best_phase2 = f_avg.copy()
                f = f_avg.copy()
    
    f = best_phase2
    try_sequence(f)
    
    # === Phase 3: Subgradient polishing ===
    remaining = deadline - time.monotonic()
    t3 = min(remaining * 0.25, 200.0)
    print("=== Phase 3: Subgradient polish (%.0fs) ===" % t3)
    
    f_polished = subgradient_polish(f, S, n, t3, rng)
    try_sequence(f_polished)
    f = f_polished
    
    # === Phase 4: Try different active set configurations ===
    remaining = deadline - time.monotonic()
    print("=== Phase 4: Active set exploration (%.0fs) ===" % remaining)
    
    if remaining > 20:
        # Try expanding the active set
        c = fft_convolve_self(f)
        c_max = c.max()
        
        for width in [0.001, 0.005, 0.01, 0.02, 0.05]:
            if remaining - (deadline - time.monotonic()) > remaining * 0.8:
                break
            
            threshold = c_max * (1 - width)
            active = c >= threshold
            if active.sum() < 1:
                active[np.argmax(c)] = True
            
            p = np.zeros(2 * n - 1)
            p[active] = 1.0 / active.sum()
            
            f_qp = solve_qp_projected_grad(f, p, S, n, max_iter=50)
            try_sequence(f_qp)
            
            if f_qp is not None:
                c_qp = fft_convolve_self(f_qp)
                if c_qp.max() < c_max:
                    f = f_qp
    
    # === Phase 5: Try different sequence lengths ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        print("=== Phase 5: Alternative lengths (%.0fs) ===" % remaining)
        
        for n_try in [4096, 16384]:
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            
            S_try = np.sqrt(2.0 * n_try)
            t_try = remaining * 0.35
            print("  Trying n=%d (%.0fs)" % (n_try, t_try))
            
            f_try, val_try = try_new_length(global_a, n_try, S_try, t_try, rng)
            try_sequence(f_try, n_try)
    
    # === Phase 6: Final round of subgradient polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 10 and global_a is not None:
        print("=== Phase 6: Final polish (%.0fs) ===" % remaining)
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * n_final)
        f_final = subgradient_polish(global_a, S_final, n_final, remaining * 0.8, rng)
        try_sequence(f_final, n_final)
    
    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.10f, internal = %.10f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Fallback.")
            n_fb = 8192
            global_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
    else:
        print("WARNING: No valid result, using uniform fallback.")
        n_fb = 8192
        global_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 8192
        global_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
    
    print("RETURNING: length=%d, best_internal=%.10f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
```
