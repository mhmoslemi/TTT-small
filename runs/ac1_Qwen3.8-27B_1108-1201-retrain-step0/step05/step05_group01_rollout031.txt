1. Compute the exact max location k* (or the active set)
2. Compute the exact subgradient at the active set
3. Blend with the LSE gradient for smoothing

This is a hybrid approach that should be more effective than pure LSE or pure subgradient.

**Let me also think about the "LLR" (Log-Likelihood Ratio) approach from information theory:**

The problem is similar to finding a probability distribution p on {0,...,n-1} that minimizes the peak of the "convolution" p*p. In information theory, this is related to the "collision probability" of a distribution.

The collision probability is Pr[X = Y] where X and Y are independent copies, which is sum_i p_i^2 = (p*p)_0. But we want to minimize the max over ALL lags, not just lag 0.

This is related to the "maximum correlation" or "autocorrelation peak" problem in signal processing, which is well-studied for complex sequences (Zipf-Mandelbrot, Barker codes, etc.). For real non-negative sequences, it's more constrained.

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Improved multi-method: Hybrid gradient (subgradient + LSE blend), 
Barzilai-Borwein step size, multi-n search, KKT flatness refinement,
and local perturbation search.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute LSE weights."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_hybrid_fista(a_init, max_time, rng, 
                     beta_start=10.0, beta_end=5000.0,
                     blend_schedule="linear", 
                     bb_momentum=0.7):
    """
    FISTA with hybrid gradient: blend of exact subgradient and LSE.
    Uses Barzilai-Borwein step size adaptation.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0
    
    # BB step size tracking
    prev_y = None
    prev_grad = None
    step_bb = 1.0 / beta_start

    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta: ramp up over time
        progress = min(1.0, it / max(1, int(max_time * 10)))  # ~10 iterations per second
        if blend_schedule == "linear":
            beta = beta_start * (beta_end / beta_start) ** progress
        else:  # exponential
            beta = beta_start * np.exp(progress * np.log(beta_end / beta_start))
        beta = max(5.0, min(beta, 100000.0))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Hybrid gradient: blend of LSE and exact subgradient
        # The blend weight increases as we get closer to convergence
        c_max = c.max()
        active_mask = c >= c_max * (1.0 - 0.01)  # 1% tolerance
        n_active = active_mask.sum()
        
        # LSE weights
        w_lse = lse_weights(c, beta)
        
        # Exact subgradient weights (uniform over active set)
        w_sub = np.zeros_like(c)
        w_sub[active_mask] = 1.0 / n_active
        
        # Blend: more subgradient as beta increases
        alpha = min(1.0, beta / 100.0)  # transition at beta=100
        w = alpha * w_sub + (1.0 - alpha) * w_lse
        w = w / w.sum()
        
        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Barzilai-Borwein step size
        if prev_y is not None and prev_grad is not None:
            s_vec = y - prev_y
            y_vec = grad - prev_grad
            s_norm = np.dot(s_vec, s_vec)
            y_norm = np.dot(y_vec, y_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            if abs(s_dot_y) > 1e-30:
                step_bb_alt = s_norm / abs(s_dot_y)
                # Damped BB step
                step_bb = bb_momentum * step_bb + (1 - bb_momentum) * step_bb_alt
            step_bb = max(1e-10, min(step_bb, S / gnorm))
        
        # Fallback step if BB fails
        step = min(step_bb, S / (0.5 * gnorm))
        
        prev_y = y.copy()
        prev_grad = grad.copy()

        # When stuck, add structured noise
        if stuck_count > 500 and it % 100 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck_count / 5000.0)
            if noise_scale > 1e-8:
                noise = rng.normal(0, 1.0, n)
                # Apply low-pass filter for smooth noise
                F = fft(noise)
                cutoff = max(3, n // 100)
                F[cutoff:-cutoff] *= 0.1
                noise = np.real(ifft(F))[:n]
                noise = noise / (np.linalg.norm(noise) + 1e-15) * noise_scale
                y = project_simplex(y + noise, S)
                t_mom = 1.0
                stuck_count = stuck_count // 2  # Partial reset

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum reset when stuck
        if stuck_count > 2000:
            t_mom = 1.0

        # Print progress

            print("    Hybrid FISTA: it=%d, obj=%.8f, beta=%.0f, stuck=%d" %

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_flatness_refine(a_init, max_time, rng):
    """
    KKT-based refinement: enforce the "flatness" condition on the active set.
    At optimum, the subgradient should be uniform on the support.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))
    
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    y = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
        
        # Active set with tolerance
        tol = 0.005
        active = np.where(c >= cur_max * (1.0 - tol))[0]
        
        # Compute the "pressure" (subgradient) at active set
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        grad = conv_gradient(y, w)
        
        # The "flatness residual" is the deviation of grad from its mean on support
        support = np.where(y > 1e-10)[0]
        if len(support) > 10:
            grad_on_support = grad[support]
            mean_grad = grad_on_support.mean()
            residual = grad_on_support - mean_grad
            rnorm = np.linalg.norm(residual)
            
            if rnorm > 1e-10:
                # Correction direction: reduce high-pressure, increase low-pressure
                correction = np.zeros(n)
                correction[support] = -residual / rnorm
                # Normalize the correction
                cstep = 0.01 * S / n  # Small step
                y_new = project_simplex(y + cstep * correction, S)
                
                # Check if improvement
                c_new = fft_convolve_self(y_new)
                if c_new.max() < cur_max:
                    y = y_new
        
        # Also try the standard gradient step
        w_full = lse_weights(c, 50.0)
        grad_full = conv_gradient(y, w_full)
        gnorm = np.linalg.norm(grad_full)
        if gnorm > 1e-15:
            step = 0.5 * S / gnorm
            y2 = project_simplex(y - step * grad_full, S)
            c2 = fft_convolve_self(y2)
            if c2.max() < c.max():
                y = y2
        else:
            break

        if (time.monotonic() - start) % 10 < 0.1:
            pass  # Rate limit print
    
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv

def run_lbfgs_polish(a_init, max_time, rng, n_beta_levels=4):

    for idx, beta in enumerate(betas):
        if time.monotonic() - start > max_time:
            break

        if remaining < 5:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)

            obj = cm + np.log(e.sum()) / beta
            sum_x = x.sum()
            obj += 10.0 * (sum_x - S) ** 2

def generate_cosine_init(n, freq=1.0, phase=0.0, amplitude=0.5):
    """Generate a cosine-based initial sequence."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 + amplitude * np.cos(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.01)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a


def upsample_init(a_small, n_target):
    """Upsample a smaller sequence to a larger size using band-limited interpolation."""
    n_small = len(a_small)
    if n_small >= n_target:
        return a_small[:n_target].copy()
    
    # Pad to power of 2 for FFT
    fsize = 1 << (n_target - 1).bit_length()
    F = fft(a_small, fsize)
    # Low-pass filter to avoid aliasing
    cutoff = n_small // 2
    if fsize > n_target:
        F[cutoff:-cutoff] = 0
    a_large = np.real(ifft(F))[:n_target]
    a_large = np.maximum(a_large, 0.0)
    S = np.sqrt(2.0 * n_target)
    if a_large.sum() < 1e-10:
        a_large = np.ones(n_target) * S / n_target
    else:
        a_large = a_large * (S / a_large.sum())
    return a_large

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Improved multi-method: Hybrid subgradient+LSE FISTA with BB step size,
KKT flatness refinement, multi-n exploration, spectral initialization,
and local block perturbation search.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if v.sum() <= S:
        return np.maximum(v, 0.0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_hybrid_fista(a_init, max_time, rng,
                     beta_start=10.0, beta_end=5000.0,
                     schedule="sqrt", bb_alpha=0.6,
                     active_tol=0.005, noise_period=800):
    """
    FISTA with hybrid gradient (subgradient + LSE blend) and
    Barzilai-Borwein adaptive step size.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    # BB tracking
    prev_y = None
    prev_grad = None
    step_bb = 1.0 / beta_start

    total_iters_est = max(1, int(max_time * 8))  # estimate total iterations

    while time.monotonic() - start < max_time:
        it += 1

        # Beta schedule
        progress = min(1.0, it / total_iters_est)
        if schedule == "sqrt":
            beta = beta_start * (beta_end / beta_start) ** np.sqrt(progress)
        elif schedule == "log":
            beta = beta_start * np.exp(progress * np.log(beta_end / beta_start))
        else:  # linear
            beta = beta_start + (beta_end - beta_start) * progress
        beta = max(5.0, min(beta, 100000.0))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Active set
        active = np.where(c >= cur_max * (1.0 - active_tol))[0]
        n_active = max(1, len(active))

        # Hybrid weights
        w_sub = np.zeros(len(c))
        w_sub[active] = 1.0 / n_active
        w_lse = lse_weights(c, beta)
        # Blend factor: more subgradient at high beta
        alpha_blend = min(1.0, 0.3 + 0.7 * (beta / 500.0))
        w = alpha_blend * w_sub + (1.0 - alpha_blend) * w_lse
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = w_sub
        else:
            w = w / w_sum

        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Barzilai-Borwein step size
        if prev_y is not None and prev_grad is not None:
            s_vec = y - prev_y
            y_vec = grad - prev_grad
            s_norm_sq = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            if abs(s_dot_y) > 1e-30:
                step_bb_new = s_norm_sq / abs(s_dot_y)
                step_bb = bb_alpha * step_bb + (1.0 - bb_alpha) * step_bb_new
        step_bb = max(1e-12, min(step_bb, 2.0 * S / gnorm))

        step = step_bb
        if step * gnorm > S:
            step = S / gnorm

        prev_y = y.copy()
        prev_grad = grad.copy()

        # Structured noise when stuck
        if stuck_count > 300 and stuck_count % noise_period == 0:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck_count / 10000.0)
            if noise_scale > 1e-9:
                noise = rng.normal(0, 1.0, n)
                F_noise = fft(noise)
                cutoff = max(3, n // 200)
                F_noise[cutoff:-cutoff] *= 0.05
                noise = np.real(ifft(F_noise))[:n]
                nrm = np.linalg.norm(noise)
                if nrm > 1e-15:
                    y = project_simplex(y + (noise / nrm) * noise_scale, S)
                    t_mom = 1.0
                    stuck_count = stuck_count // 2
                    prev_y = None
                    prev_grad = None

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck_count > 3000:
            t_mom = 1.0

        if it % 8000 == 0:
            cur_obj = 2.0 * n * cur_max / (S * S)
            print("    HybridF it=%d obj=%.8f beta=%.0f stuck=%d" %
                  (it, cur_obj, beta, stuck_count))

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_kkt_flatness(a_init, max_time, rng):
    """
    KKT-based refinement: enforce that the subgradient is uniform on support.
    Also performs line search on the pure active-set direction.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))
    y = a.copy()
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()

        # Active set (tight constraints)
        tol = 0.002
        active = np.where(c >= cur_max * (1.0 - tol))[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Subgradient on active set
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        grad = conv_gradient(y, w)

        # KKT residual: on support, grad should be constant (= lambda)
        support = np.where(y > 1e-8)[0]
        if len(support) >= 20:
            g_s = grad[support]
            lam = g_s.mean()
            residual = g_s - lam
            rnorm = np.linalg.norm(residual)

            if rnorm > 1e-12:
                # Direction to flatten: move mass from high-pressure to low-pressure
                direction = np.zeros(n)
                direction[support] = -residual
                direction = direction / (np.linalg.norm(direction) + 1e-30)

                # Try a few step sizes
                best_y_trial = y.copy()
                best_c_trial = cur_max
                for scale in [0.5, 1.0, 2.0, 4.0, 8.0]:
                    trial = project_simplex(y + scale * 0.005 * S / n * direction, S)
                    c_trial = fft_convolve_self(trial)
                    if c_trial.max() < best_c_trial:
                        best_c_trial = c_trial.max()
                        best_y_trial = trial.copy()
                y = best_y_trial

        # Also try LSE gradient step as backup
        beta = 100.0
        w_lse = lse_weights(c, beta)
        grad_lse = conv_gradient(y, w_lse)
        gn = np.linalg.norm(grad_lse)
        if gn > 1e-15:
            for scale in [0.3, 0.5, 1.0]:
                step = scale * S / gn
                trial = project_simplex(y - step * grad_lse, S)
                c_trial = fft_convolve_self(trial)
                if c_trial.max() < c.max() * (1 - 1e-10):
                    y = trial
                    c = c_trial
                    break

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_local_perturb(a_init, max_time, rng):
    """

    """
    n = len(a_init)

    no_improve = 0
    base_scale = 0.005

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive block size
        block_size = int(rng.integers(max(4, n // 1000), min(512, n // 8)))
        start_idx = int(rng.integers(0, n - block_size))
        end_idx = start_idx + block_size

        # Adaptive noise scale
        local_mean = a[start_idx:end_idx].mean()
        scale = base_scale * local_mean * (1.0 + 0.05 * no_improve)

        c_new = fft_convolve_self(a_new)
        if c_new.max() < best_conv - 1e-12:
            a = a_new.copy()
            best_a = a_new.copy()

def run_lbfgs_polish(a_init, max_time, rng, n_beta_levels=4):
    """L-BFGS-B polishing with LSE objective."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)

    for idx, beta in enumerate(betas):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / beta
            sum_x = x.sum()
            obj += 20.0 * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 2.0 * 20.0 * (sum_x - S)
            return obj, grad

        t0 = time.monotonic()
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 60, 'maxls': 50, 'ftol': 1e-16, 'gtol': 1e-12,
                     'iprint': -1}
        )
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

        print("    L-BFGS beta=%.0f: obj=%.8f (%.1fs)" %
              (beta, obj_new, time.monotonic() - t0))

    return best_a, best_obj


def generate_cosine_init(n, freq=1.0, phase=0.0, amp=0.5):
    """Cosine-based initial sequence."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.01)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a


def upsample_sequence(a_small, n_target):
    """Band-limited upsample a sequence to a larger size."""
    n_small = len(a_small)
    if n_small >= n_target:
        return a_small[:n_target].copy()
    fsize = 1 << (n_target - 1).bit_length()
    F = fft(a_small, fsize)
    # Zero out high frequencies to avoid aliasing
    cutoff = n_small
    if fsize > n_target:
        keep = min(cutoff, n_target // 2)
        F[keep:n_target - keep] = 0.0
        F[n_target:-keep] = 0.0
    a_large = np.real(ifft(F))[:n_target]
    a_large = np.maximum(a_large, 0.0)
    S = np.sqrt(2.0 * n_target)
    if a_large.sum() < 1e-10:
        a_large = np.ones(n_target) * S / n_target
    else:
        a_large = a_large * (S / a_large.sum())
    return a_large
