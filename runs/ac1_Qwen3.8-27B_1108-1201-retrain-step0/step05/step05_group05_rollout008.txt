from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_to_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
        beta = beta_eff
    e = np.exp(scaled)

    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def peak_active_optimize(a_init, S, max_time, rng, k_active=3):
    """
    Optimize using an explicit active-set of top peaks.
    At each step, identify the top-k lags and use their combined
    subgradient for the descent direction.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        if c.max() < best - 1e-12:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        if it - last_improve > 5000:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        # Identify active peaks
        top_k = min(k_active, len(c))
        sorted_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[sorted_idx]
        cmax = c.max()
        # Weight by how close to max
        tol = 0.01 * cmax
        weights = np.maximum(top_vals - (cmax - tol), 0.0)
        if weights.sum() < 1e-15:

        else:
            weights = weights / weights.sum()

        w = np.zeros(len(c))
        for i, idx in enumerate(sorted_idx):
            w[idx] = weights[i]

        # Gradient via FFT
        grad = weighted_conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 0.5 * S / (n * gnorm / S + 1e-15)
        step = min(step, 1.0)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def fourier_perturb(a, S, rng, scale=0.1):
    """Fourier-domain phase perturbation."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.1 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)

            # Spectral flattening (gentle)
            alpha = 0.2 + 0.3 * (1.0 - progress)
            a = spectral_flatten_step(a, S, n_iter=20, alpha=alpha)
        elif r < 0.7:
            # Fourier phase perturbation
            scale = 0.3 * (1.0 - progress) + 0.05
            a = fourier_perturb(a, S, rng, scale=scale)
        else:
            # Small random perturbation
            a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)

        # Local optimization: FISTA with moderate beta
        t_alloc = min(opt_time, max_time - (time.monotonic() - start) - 10)
        if t_alloc < 5:
            break
        a_opt, val = fista_lse(a, S, t_alloc, rng,
                               beta_init=50.0, beta_rate=0.003,
                               beta_max=20000.0)
        if val < best - 1e-10:
            best = val
            best_a = a_opt.copy()
            print("      BH-spectral iter %d: max_b=%.10f" % (it, val))

    return best_a, best


def ultra_high_beta_polish(a_init, S, max_time, rng):
    """
    Final polish with extremely high LSE beta to resolve the
    exact active set and optimize precisely.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    # Phase A: Very high beta FISTA
    seg_time = max_time * 0.6
    a_opt, val = fista_lse(a, S, seg_time, rng,
                           beta_init=5000.0, beta_rate=0.001,
                           beta_max=500000.0, use_backtrack=True)
    if val < best:
        best = val
        best_a = a_opt.copy()

    # Phase B: Peak-active-set optimization
    remaining = max_time - (time.monotonic() - start) - 5
    if remaining > 10:
        a_opt2, val2 = peak_active_optimize(best_a, S, remaining, rng, k_active=5)
        if val2 < best:
            best = val2
            best_a = a_opt2.copy()

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def try_length(n, a_given, S, budget, rng):
    """
    Run the full optimization pipeline for a given sequence length n.
    Returns (best_a, best_obj) where best_obj = 2n*max_b/S^2.
    """
    start = time.monotonic()
    deadline = start + budget

    # Initialize
    if a_given is not None and len(a_given) == n:
        a = normalize_to_sum(a_given, S)
    else:
        # Generate from given (interpolate or truncate)
        if a_given is not None and len(a_given) > 0:
            a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(a_given)), a_given)
            a = normalize_to_sum(a, S)
        else:
            a = np.ones(n) * S / n

    # Quick spectral flatten to get a better starting point
    a = spectral_flatten_step(a, S, n_iter=30, alpha=0.3)

    # Quick FISTA to establish baseline
    t1 = min(30.0, budget * 0.1)
    a_opt, val = fista_lse(a, S, t1, rng, beta_init=20.0, beta_rate=0.005, beta_max=5000.0)
    obj = 2.0 * n * val / (a_opt.sum() ** 2)
    print("    n=%d: after quick FISTA, obj=%.8f" % (n, obj))
    best_a = a_opt.copy()
    best_obj = obj

    # Basin hopping with spectral mutations
    t2_end = start + budget * 0.55
    if time.monotonic() < t2_end:
        t2 = t2_end - time.monotonic()
        a_bh, val_bh = basin_hopping_spectral(best_a, S, t2, rng)
        obj_bh = 2.0 * n * val_bh / (a_bh.sum() ** 2)
        if obj_bh < best_obj:
            best_obj = obj_bh
            best_a = a_bh.copy()
            print("    n=%d: after basin hopping, obj=%.8f" % (n, best_obj))

    # Ultra-high-beta polish
    t3_end = deadline - 5
    if time.monotonic() < t3_end:
        t3 = t3_end - time.monotonic()
        a_pol, val_pol = ultra_high_beta_polish(best_a, S, t3, rng)
        obj_pol = 2.0 * n * val_pol / (a_pol.sum() ** 2)
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()
            print("    n=%d: after polish, obj=%.8f" % (n, best_obj))

    elapsed = time.monotonic() - start
    print("    n=%d: final obj=%.8f (took %.0fs)" % (n, best_obj, elapsed))
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30

    print("Starting optimization, budget=%.0fs, deadline in %.0fs" % (budget_s, budget_s - 30))

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
            # Compute its objective
            n_g = len(given)
            S_g = given.sum()
            c_g = fft_convolve_self(given)
            obj_g = 2.0 * n_g * c_g.max() / (S_g * S_g)
            print("Given sequence obj: %.8f" % obj_g)
    except Exception:
        pass

    # Try multiple lengths
    n_values = [4096, 8192, 16384]

    # Allocate time: give more to larger n (more DOF)
    # But also try smaller n (faster per-iteration, might find different minima)
    # Weight: 4096 gets 20%, 8192 gets 40%, 16384 gets 40%
    total_budget = deadline - total_start - 10
    time_alloc = {}
    time_alloc[4096] = total_budget * 0.15
    time_alloc[8192] = total_budget * 0.45
    time_alloc[16384] = total_budget * 0.40

    global_best_obj = np.inf
    global_best_a = None

    # Try n=8192 first (same as previous, likely good)
    for n in [8192, 4096, 16384]:
        if time.monotonic() > deadline - 60:
            break
        S = np.sqrt(2.0 * n)
        t_alloc = min(time_alloc[n], deadline - time.monotonic() - 30)
        if t_alloc < 30:
            continue
        print("\n  Trying n=%d (budget=%.0fs)..." % (n, t_alloc))
        a_res, obj_res = try_length(n, given, S, t_alloc, rng)
        if obj_res < global_best_obj:
            global_best_obj = obj_res
            global_best_a = a_res.copy()
            print("  NEW GLOBAL BEST: obj=%.8f at n=%d" % (global_best_obj, n))

    # If we have remaining time, do additional refinement on the best
    remaining = deadline - time.monotonic() - 10
    if global_best_a is not None and remaining > 30:
        n_best = len(global_best_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n  Final refinement (budget=%.0fs)..." % remaining)

        # Multiple short rounds of polish
        n_rounds = min(3, int(remaining / 30))
        for r in range(n_rounds):
            t_round = (remaining - 5) / (n_rounds - r)
            if t_round < 10:
                break
            # Slight perturbation
            a_pert = fourier_perturb(global_best_a, S_best, rng, scale=0.02)
            a_res, val_res = ultra_high_beta_polish(a_pert, t_round, rng)
            obj_res = 2.0 * n_best * val_res / (a_res.sum() ** 2)
            if obj_res < global_best_obj:
                global_best_obj = obj_res
                global_best_a = a_res.copy()
                print("    Refinement round %d: obj=%.8f" % (r, global_best_obj))

    # Final verification
    if global_best_a is not None:
        global_best_a = np.clip(global_best_a, 0.0, 1000.0)
        if global_best_a.sum() < 0.01:
            n_fallback = len(global_best_a)
            global_best_a = np.ones(n_fallback) * np.sqrt(2.0 * n_fallback) / n_fallback
        final_list = [float(x) for x in global_best_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            n_fallback = 8192
            global_best_a = np.ones(n_fallback) * np.sqrt(2.0 * n_fallback) / n_fallback
            final_list = [float(x) for x in global_best_a]
    else:
        n_fallback = 8192
        global_best_a = np.ones(n_fallback) * np.sqrt(2.0 * n_fallback) / n_fallback
        final_list = [float(x) for x in global_best_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.8f, n=%d" % (elapsed, global_best_obj, len(global_best_a)))
    return [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
