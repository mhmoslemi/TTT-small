"""
Minimize 2*n*max(f*f)/sum(f)^2 for non-negative sequences of length 8192.
Strategy: Hybrid LP-direction + LSE-FISTA + active-set refinement.
Key improvement over previous: LP-based direction finding using the SOTA approach,
combined with FFT-based gradient methods for fine-tuning.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta

def run_lse_fista(a_init, n, S, max_time, rng, beta_start=50.0, beta_end=20000.0, beta_steps=200):
    """FISTA with LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)

        c = fft_convolve_self(y)
        cur_true_max = c.max()

        if cur_true_max < best:
            best = cur_true_max
            best_a = y.copy()
            last_improve_it = it

        grad, _ = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta

        stagnant = it - last_improve_it
        if stagnant > 500:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stagnant / (80000.0 / max(n,1)))
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve_it = it
            step *= 0.3

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best

def run_lp_iterations(a_init, n, S, max_time, rng, K=50, blend_rate=0.3):
    """
    Iterate the LP-based direction finding.
    At each step:
    1. Compute M = max(f*f)
    2. Solve LP to get direction b (sum(b)=1)
    3. Update: f = (1-t)*f/S + t*b  (then renormalize to sum=S)
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()
        
        if M < best:
            best = M
            best_a = a.copy()
            print("  LP iter %d: M = %.10f" % (it, M))
        
        # Solve LP for direction
        b_dir = lp_direction(a, M, K=K, max_time_lp=20.0)
        
        if b_dir is None:
            print("  LP failed at iter %d, stopping." % it)
            break
        
        # Normalize b_dir to sum = S
        b_dir = b_dir * S
        
        # Blend
        t = blend_rate
        a_new = (1 - t) * a + t * b_dir
        a = normalize_sum(a_new, n)
        
        # Quick check
        c_new = fft_convolve_self(a)
        if c_new.max() < best:
            best = c_new.max()
            best_a = a.copy()
            print("  LP iter %d (after blend): M = %.10f" % (it, best))
    
    return best_a, best

def active_set_gradient(a, c, beta_at=300.0):
    """Gradient using hard active set."""
    n = len(a)
    cm = c.max()
    threshold = cm - 5.0 / beta_at
    w = np.where(c >= threshold, 1.0, 0.0)
    tot = w.sum()
    if tot < 1:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        tot = 1.0
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def run_active_refinement(a_init, n, S, max_time, rng):
    """Refine using active-set gradient with backtracking."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad = active_set_gradient(a, c, beta_at=200.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 0.01
        improved = False
        for _ in range(8):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur_max * (1 - 1e-6):
                a = a_test
                improved = True
                break
            step *= 0.5

        if not improved:
            step = 0.001 / (1 + it * 0.001)
            a = project_simplex(a - step * grad, S)

        if it % 200 == 0:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / 5000.0)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best

def peak_shaving(a, n, S, n_peaks=8, n_steps=100, lr_scale=0.01):
    """Surgical peak shaving."""
    a = a.copy()
    for _ in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()
        top_idx = np.argsort(c)[::-1][:n_peaks]
        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            weight = max(0, c[k] - cm * 0.95)
            np.add.at(resp, js, contrib * weight)
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        lr = lr_scale * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)
    return a

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):

    cm = c.max()

        return np.zeros(n), cm

    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    b_ub = np.full(K, M)

            if s > 1e-10:

            b = np.asarray(res.x, dtype=np.float64)
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception as e:
        print("  LP error: %s" % str(e))

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        grad = active_set_gradient(a, c, beta_at=200.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 0.01
        improved = False
        for _ in range(8):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur_max * (1.0 - 1e-6):
                a = a_test
                improved = True
                break
            step *= 0.5

        if not improved:
            step = 0.001 / (1.0 + it * 0.001)
            a = project_simplex(a - step * grad, S)

        if it % 200 == 0:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / 5000.0)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def peak_shaving(a, n, S, n_peaks=8, n_steps=100, lr_scale=0.01):
    """Surgical peak shaving: reduce contributions to top convolution peaks."""
    a = a.copy()
    for _ in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()
        top_idx = np.argsort(c)[::-1][:n_peaks]
        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            weight = max(0.0, c[k] - cm * 0.95)
            np.add.at(resp, js, contrib * weight)
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        lr = lr_scale * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)
    return a


def spectral_shaping(a, n, S, max_time, rng):
    """
    Alternative approach: work in frequency domain to flatten the power spectrum.
    In freq domain, self-convolution peak relates to the time-domain IFFT of |A|^2.
    We want to minimize max(IFFT(|A|^2)), which is equivalent to minimizing the
    peak of the aperiodic autocorrelation.
    Use iterative spectral leveling.
    """
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        # Target: flat power spectrum (water-filling)
        # But we need a >= 0, so this is tricky
        # Instead, use amplitude compression
        # Compress dynamic range of |A|
        amp = np.sqrt(power)
        amp_min = amp.min()
        amp_max = amp.max()
        if amp_max - amp_min < 1e-10:
            break
        # Gentle compression
        t_comp = 0.3
        amp_new = amp_min + (amp_max - amp_min) * ((amp - amp_min) / (amp_max - amp_min)) ** (1.0 + t_comp)
        # Reconstruct with original phase
        phase = np.angle(A)
        A_new = amp_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        a = project_simplex(a_new, S)

        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()

        if it % 50 == 0:
            print("    Spectral iter %d: max = %.8f" % (it, best))

    return best_a, best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence of length 8192 minimizing
    2*n*max(f*f)/sum(f)^2. Uses LP-direction + LSE-FISTA + active-set + peak shaving.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    n = 8192
    S = np.sqrt(2.0 * n)

    print("Budget: %.0fs, n=%d, S=%.4f" % (budget_s, n, S))

    # Load previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    # If given has different length, interpolate to n
    if given is not None and len(given) != n:
        t_old = np.linspace(0, 1, len(given))
        t_new = np.linspace(0, 1, n)
        given = np.interp(t_new, t_old, given)
        print("Interpolated to n=%d" % n)

    if given is not None:
        given = normalize_sum(given, n)
        c_g = fft_convolve_self(given)
        print("Given sequence: max_conv = %.8f, obj = %.8f" % (c_g.max(), 2.0 * n * c_g.max() / (given.sum() ** 2)))
        global_best = c_g.max()
        global_a = given.copy()

    # === Phase 1: Quick LSE-FISTA on given ===
    if given is not None:
        t1 = min(100.0, (deadline - time.monotonic()) * 0.12)
        print("=== Phase 1: LSE-FISTA on given (%.0fs) ===" % t1)
        a1, v1 = run_lse_fista(given, n, S, t1, rng,
                                beta_start=100.0, beta_end=10000.0, beta_steps=120)
        c1 = fft_convolve_self(a1)
        if c1.max() < global_best:
            global_best = c1.max()
            global_a = a1.copy()
            print("  Phase 1 best: %.8f" % global_best)

    # === Phase 2: LP-based iterations (key SOTA improvement) ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        t2 = min(500.0, remaining * 0.30)
        print("=== Phase 2: LP-based iterations (%.0fs) ===" % t2)
        a2, v2 = run_lp_iterations(global_a, n, S, t2, rng, K=60, blend_rate=0.3)
        if v2 < global_best:
            global_best = v2
            global_a = a2.copy()
            print("  Phase 2 best: %.8f" % global_best)

    # === Phase 3: LSE-FISTA refinement on LP result ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        t3 = min(350.0, remaining * 0.28)
        print("=== Phase 3: LSE-FISTA refinement (%.0fs) ===" % t3)
        a3, v3 = run_lse_fista(global_a, n, S, t3, rng,
                                beta_start=500.0, beta_end=50000.0, beta_steps=200)
        if v3 < global_best:
            global_best = v3
            global_a = a3.copy()
            print("  Phase 3 best: %.8f" % global_best)

    # === Phase 4: Active-set refinement ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        t4 = min(250.0, remaining * 0.22)
        print("=== Phase 4: Active-set refinement (%.0fs) ===" % t4)
        a4, v4 = run_active_refinement(global_a, n, S, t4, rng)
        if v4 < global_best:
            global_best = v4
            global_a = a4.copy()
            print("  Phase 4 best: %.8f" % global_best)

    # === Phase 5: Peak shaving + polish ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("=== Phase 5: Peak shaving (%.0fs) ===" % remaining)
            a5 = peak_shaving(global_a, n, S, n_peaks=10, n_steps=200, lr_scale=0.005)
            c5 = fft_convolve_self(a5)
            if c5.max() < global_best:
                global_best = c5.max()
                global_a = a5.copy()
                print("  Phase 5 (shaved): %.8f" % global_best)

            remaining2 = max(5.0, deadline - time.monotonic())
            if remaining2 > 20:
                a5b, v5b = run_lse_fista(global_a, n, S, remaining2 * 0.5, rng,
                                          beta_start=5000.0, beta_end=30000.0, beta_steps=50)
                if v5b < global_best:
                    global_best = v5b
                    global_a = a5b.copy()
                    print("  Phase 5 (polished): %.8f" % global_best)

    # === Phase 6: Second LP round if time permits ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 120:
            print("=== Phase 6: Second LP round (%.0fs) ===" % (remaining * 0.6))
            a6, v6 = run_lp_iterations(global_a, n, S, remaining * 0.6, rng, K=80, blend_rate=0.2)
            if v6 < global_best:
                global_best = v6
                global_a = a6.copy()
                print("  Phase 6 best: %.8f" % global_best)

    # === Phase 7: Final active-set polish ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print("=== Phase 7: Final active refinement (%.0fs) ===" % remaining)
            a7, v7 = run_active_refinement(global_a, n, S, remaining * 0.8, rng)
            if v7 < global_best:
                global_best = v7
                global_a = a7.copy()
                print("  Phase 7 best: %.8f" % global_best)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.8f, internal = %.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n, dtype=np.float64) * (S / n)
    else:
        print("WARNING: No valid result found, using uniform fallback.")
        global_a = np.ones(n, dtype=np.float64) * (S / n)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(n, dtype=np.float64) * (S / n)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
