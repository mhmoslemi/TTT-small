"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Multi-stage optimization combining coarse block-based search,
DIRECT-COM subgradient averaging, active-set refinement, and peak shaving.
Key insight: The subgradient of max(conv) is a shifted copy of the sequence itself,
enabling efficient computation. DIRECT-COM averaging provides convergence guarantees.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def compute_subgradient(a, k):
    """Compute subgradient of c[k] = (a*a)[k] with respect to a.
    dc[k]/da[i] = a[k-i] + a[i-k] (when indices valid)"""
    n = len(a)
    g = np.zeros(n)
    # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
    lo1 = max(0, k - n + 1)
    hi1 = min(k, n - 1)
    # a[i-k] is valid when 0 <= i-k < n, i.e., k <= i <= k+n-1
    lo2 = max(0, k)
    hi2 = min(k + n - 1, n - 1)
    
    for i in range(lo1, hi1 + 1):
        g[i] += a[k - i]
    for i in range(lo2, hi2 + 1):
        g[i] += a[i - k]
    return g

def compute_subgradient_fft(a, k):
    """Compute subgradient of c[k] using FFT (faster for large n).
    dc[k]/da[i] = a[k-i] + a[i-k]
    This is (reverse(a) convolved with delta_k)[i] + (a convolved with delta_k)[i]
    = (shifted reverse a)[i] + (shifted a)[i]
    """
    n = len(a)
    # a[k-i] for valid i: this is a reversed and shifted copy
    # Let's think: b[i] = a[k-i] for 0 <= k-i < n
    # So b[i] = a[k-i], which means b = reverse(a) shifted by k
    # reverse(a)[j] = a[n-1-j]
    # a[k-i] = reverse(a)[n-1-k+i] ... hmm this is getting confusing
    
    # Simpler: just do it directly
    g = np.zeros(n)
    lo1 = max(0, k - n + 1)
    hi1 = min(k, n - 1)
    g[lo1:hi1+1] = a[k-hi1:k-lo1+1][::-1]
    
    lo2 = max(0, k)
    hi2 = min(k + n - 1, n - 1)
    g[lo2:hi2+1] += a[k:(hi2-k+1)]
    
    return g

def compute_subgradient_multi(a, k_list):
    """Compute average subgradient over multiple lags."""
    n = len(a)
    g = np.zeros(n)
    for k in k_list:
        lo1 = max(0, k - n + 1)
        hi1 = min(k, n - 1)
        g[lo1:hi1+1] += a[k-hi1:k-lo1+1][::-1]
        
        lo2 = max(0, k)
        hi2 = min(k + n - 1, n - 1)
        g[lo2:hi2+1] += a[k:(hi2-k+1)]
    return g / len(k_list)

def direct_com(a_init, n, max_time, rng, n_top=1, alpha_init=1.0, momentum=True):
    """
    DIRECT-COM style subgradient method with averaging.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = S.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    x_avg = a.copy()
    
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cm = c.max()
        
        if cm < best:
            best = cm
            best_a = a.copy()
        
        # Find top lags
        if n_top == 1:
            k_max = np.argmax(c)
            k_list = [k_max]
        else:
            threshold = cm - 0.1 / n  # relative threshold
            top_idx = np.where(c >= threshold)[0]
            if len(top_idx) == 0:
                top_idx = [np.argmax(c)]
            elif len(top_idx) > n_top:
                # Take the top n_top
                top_vals = np.argsort(c)[::-1][:n_top]
                top_idx = top_vals
            k_list = list(top_idx)
        
        # Compute subgradient
        g = compute_subgradient_multi(a, k_list)
        
        # DIRECT-COM step size: 2 / (it + 1)
        step = 2.0 / (it + 1)
        
        # Update
        a_new = a - step * g
        a_new = project_simplex(a_new, S)
        
        # Update average
        x_avg = x_avg + (a_new - x_avg) / (it + 1)
        
        a = a_new
        
        # Also check the average periodically
        if it % 100 == 0:
            x_avg_check = normalize_sum(x_avg, n)
            c_avg = fft_convolve_self(x_avg_check)
            if c_avg.max() < best:
                best = c_avg.max()
                best_a = x_avg_check.copy()
    
    # Final check on average
    x_avg_final = normalize_sum(x_avg, n)
    c_final = fft_convolve_self(x_avg_final)
    if c_final.max() < best:
        best = c_final.max()
        best_a = x_avg_final.copy()
    
    return best_a, best

def peak_shaving(a_init, n, max_iters=200, lr=0.01):
    """
    Direct peak shaving: reduce a[j] where it contributes most to the peak.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    best = np.inf
    best_a = a.copy()
    
    for it in range(max_iters):
        c = fft_convolve_self(a)
        cm = c.max()
        
        if cm < best:
            best = cm
            best_a = a.copy()
        
        # Top 3 peaks
        top = np.argsort(c)[::-1][:3]
        
        # Responsibility
        resp = np.zeros(n)
        for k in top:
            weight = c[k] - cm * 0.999
            if weight <= 0:
                weight = 0.01
            lo1 = max(0, k - n + 1)
            hi1 = min(k, n - 1)
            js = np.arange(lo1, hi1 + 1)
            resp[js] += a[k - js] * weight
            lo2 = max(0, k)
            hi2 = min(k + n - 1, n - 1)
            js2 = np.arange(lo2, hi2 + 1)
            resp[js2] += a[js2 - k] * weight
        
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        
        # Multiplicative reduction
        a = a * np.exp(-lr * resp / resp_max)
        a = project_simplex(a, S)
    
    return best_a, best

def block_coarse_search(n, max_time, rng, n_blocks=128):
    """
    Coarse optimization using block representation.
    Each block has a single value. Search over block values.
    """
    S = np.sqrt(2.0 * n)
    block_size = n // n_blocks
    n_blocks_actual = n // block_size  # Adjust for exact fit
    
    # Initial: random block values
    blocks = rng.uniform(0.5, 1.5, n_blocks_actual)
    
    def blocks_to_seq(b):
        a = np.repeat(b, block_size)
        if len(a) < n:
            a = np.concatenate([a, np.zeros(n - len(a))])
        return a[:n]
    
    best_val = np.inf
    best_blocks = blocks.copy()
    
    start = time.monotonic()
    it = 0
    temperature = 1.0
    
    while time.monotonic() - start < max_time:
        it += 1
        a = blocks_to_seq(blocks)
        a = normalize_sum(a, n)
        c = fft_convolve_self(a)
        val = c.max()
        
        if val < best_val:
            best_val = val
            best_blocks = blocks.copy()
        
        # Temperature decay
        temperature *= 0.999
        
        # Propose: perturb one or a few random blocks
        n_perturb = rng.integers(1, min(5, n_blocks_actual // 4))
        idx = rng.choice(n_blocks_actual, size=n_perturb, replace=False)
        
        new_blocks = blocks.copy()
        for i in idx:
            new_blocks[i] = max(0.01, blocks[i] + rng.normal(0, 0.1))
        
        new_a = normalize_sum(blocks_to_seq(new_blocks), n)
        new_c = fft_convolve_self(new_a)
        new_val = new_c.max()
        
        # Metropolis criterion
        if new_val < val or rng.random() < np.exp(-(new_val - val) / (temperature * 0.01)):
            blocks = new_blocks
    
    return blocks_to_seq(best_blocks), best_val

def gradient_descent_fine(a_init, n, max_time, rng, lr=0.005, decay=0.9999):

    start = time.monotonic()

        cm = c.max()
        
        if cm < best:
            best = cm
            best_a = a.copy()

                noise = 0.0001 * S / np.sqrt(n)

    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_subgradient_single(a, k):
    """Subgradient of c[k] = (a*a)[k]: g[i] = 2*a[k-i] if 0<=k-i<n."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def compute_subgradient_multi(a, k_list):
    """Average subgradient over multiple lags."""
    n = len(a)
    g = np.zeros(n)
    for k in k_list:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(k_list)
    return g


def kkt_residual(a, c, active_lags):
    """Check KKT: how uniform is the subgradient on the support?"""
    n = len(a)
    support = a > 1e-10
    if support.sum() < 2:
        return np.inf
    g = compute_subgradient_multi(a, active_lags)
    g_sup = g[support]
    return g_sup.max() - g_sup.min()


def direct_com(a_init, n, max_time, rng, n_top=3, step_mode="standard"):
    """
    DIRECT-COM subgradient method with averaging.
    Minimizes max(conv(a,a)) s.t. a>=0, sum(a)=S.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    x_avg = a.copy()

    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    best_avg_val = np.inf

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cm = c.max()

        if cm < best:
            best = cm
            best_a = a.copy()

        # Find active lags
        if n_top <= 1:
            k_list = [int(np.argmax(c))]
        else:
            # Use lags within a relative tolerance of max
            threshold = cm - 1.0 / n
            active = np.where(c >= threshold)[0]
            if len(active) == 0:
                active = np.array([np.argmax(c)])
            elif len(active) > n_top:
                active = np.argsort(c)[::-1][:n_top]
            k_list = [int(k) for k in active]

        # Subgradient
        g = compute_subgradient_multi(a, k_list)

        # Step size
        if step_mode == "standard":
            step = 1.0 / (it ** 0.5)
        elif step_mode == "slow":
            step = 0.5 / (it ** 0.5)
        else:
            step = 2.0 / (it + 1)

        # Update
        a_new = a - step * g
        a_new = project_simplex(a_new, S)

        # Update running average
        x_avg = x_avg + (a_new - x_avg) / (it + 1)

        a = a_new

        # Periodically check the average
        if it % 200 == 0:
            x_avg_norm = normalize_sum(x_avg, n)
            c_avg = fft_convolve_self(x_avg_norm)
            if c_avg.max() < best_avg_val:
                best_avg_val = c_avg.max()
            if c_avg.max() < best:
                best = c_avg.max()
                best_a = x_avg_norm.copy()

    # Final check
    x_avg_final = normalize_sum(x_avg, n)
    c_final = fft_convolve_self(x_avg_final)
    if c_final.max() < best:
        best = c_final.max()
        best_a = x_avg_final.copy()

    return best_a, best


def subgradient_with_backtrack(a_init, n, max_time, rng, n_top=3, lr_init=0.5):
    """
    Subgradient method with backtracking line search.
    More aggressive than DIRECT-COM but can get stuck.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cm = c.max()

        if cm < best:
            best = cm
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000:
            # Restart with perturbation
            noise = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)
            no_improve = 0
            lr = lr_init

        # Active lags
        threshold = cm - 1.0 / n
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        elif len(active) > n_top:
            active = np.argsort(c)[::-1][:n_top]
        k_list = [int(k) for k in active]

        # Subgradient
        g = compute_subgradient_multi(a, k_list)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g_normalized = g / gnorm

        # Backtracking
        step = lr
        improved = False
        for _ in range(8):
            a_test = project_simplex(a - step * g, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cm * (1.0 - 1e-9):
                a = a_test
                improved = True
                break
            step *= 0.5

        if not improved:
            # Tiny step or noise
            if it % 100 == 0:
                noise = 0.005 * S / np.sqrt(n) * np.exp(-it / 20000.0)
                a = project_simplex(a + rng.normal(0, noise, n), S)
            lr *= 0.99
            if lr < 1e-6:
                lr = lr_init * 0.1
                # Also try larger step in random direction
                a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)

    return best_a, best


def peak_shaving_v2(a_init, n, max_iters=500, lr=0.01):
    """
    Refined peak shaving: identify top lags, compute responsibility,
    apply multiplicative updates, then LSE polish.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    best = np.inf
    best_a = a.copy()

    for it in range(max_iters):
        c = fft_convolve_self(a)
        cm = c.max()

        if cm < best:
            best = cm
            best_a = a.copy()

        # Top peaks with weights
        top_k = min(5, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]

        # Responsibility: how much does each a[j] contribute to the peaks?
        resp = np.zeros(n)
        for k in top_idx:
            weight = max(0.0, c[k] - cm * 0.995)
            if weight == 0:
                weight = 0.01
            # g[i] = 2*a[k-i] contribute to peak at lag k
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                resp[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1] * weight

        resp_max = resp.max()
        if resp_max < 1e-30:
            break

        # Multiplicative update
        a = a * np.exp(-lr * resp / resp_max)
        a = project_simplex(a, S)

    return best_a, best


def lse_fista_refined(a_init, n, max_time, rng, beta_start=100.0, beta_end=50000.0,
                      beta_steps=200, restart_gap=1000):
    """
    Refined LSE-FISTA with better beta schedule and restart logic.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 100000.0)

        c = fft_convolve_self(y)
        cm = c.max()

        if cm < best:
            best = cm
            best_a = y.copy()
            last_improve = it

        # LSE gradient
        cm_val = c.max()
        e = np.exp(beta * (c - cm_val))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot

        # Gradient via FFT: grad = 2 * ifft(FFT(w) * conj(FFT(a)))[:n]
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta

        # Stagnation handling
        stagnant = it - last_improve
        if stagnant > restart_gap:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / (10000.0 / max(1, n // 1000)))
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            step *= 0.5

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    # Final
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def active_set_kkt_refine(a_init, n, max_time, rng, n_active_max=50):
    """
    KKT-based refinement: identify active set, check KKT residual,
    and take steps to reduce the residual.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cm = c.max()

        if cm < best:
            best = cm
            best_a = a.copy()

        # Active set: lags very close to max
        threshold = cm - 0.01 / n
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        if len(active) > n_active_max:
            active = np.argsort(c)[::-1][:n_active_max]
        k_list = [int(k) for k in active]

        # Compute subgradient
        g = compute_subgradient_multi(a, k_list)

        # KKT residual: want g to be constant on support
        support = a > 1e-8
        if support.sum() < 2:
            break
        g_sup = g[support]
        g_mean = g_sup.mean()
        residual = g_sup - g_mean

        # Step: reduce the residual
        g_step = np.zeros(n)
        g_step[support] = residual
        gnorm = np.linalg.norm(g_step)
        if gnorm < 1e-14:
            break

        # Try several step sizes
        improved = False
        for scale in [0.01, 0.005, 0.001, 0.0005]:
            a_test = project_simplex(a - scale * g_step, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cm * (1 - 1e-10):
                a = a_test
                improved = True
                break

        if not improved:
            # Very small step
            a = project_simplex(a - 1e-5 * g_step, S)
            # Or noise
            if best * 1000 % 100 < 1:
                noise = 0.001 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)

    return best_a, best


def generate_asymmetric_init(n, rng):
    """Generate diverse asymmetric initializations."""
    inits = []
    t = np.linspace(0, 1, n)

    # Asymmetric raised cosines
    for cf in [0.25, 0.3, 0.35, 0.4]:
        for wf in [0.3, 0.4, 0.5]:
            x = (t - cf) / (wf / 2.0)
            a = np.zeros(n)
            mask = np.abs(x) < 1
            a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
            a = np.maximum(a, 1e-10)
            inits.append(a)

    # Skewed Gaussians
    for mu in [0.3, 0.35, 0.4]:
        for sigma in [0.15, 0.2, 0.25]:
            a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
            a = np.maximum(a, 1e-10)
            inits.append(a)

    # Power ramps (asymmetric)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = t ** p
        a = np.maximum(a, 0.01)
        inits.append(a)
        a = (1 - t) ** p
        a = np.maximum(a, 0.01)
        inits.append(a)

    # Sinusoidal (asymmetric)
    for k in [1, 2, 3]:
        a = 0.5 + 0.5 * np.sin(np.pi * k * t - np.pi / 4)
        a = np.maximum(a, 0.01)
        inits.append(a)

    # Multi-bump
    for n_bumps in [2, 3]:
        centers = np.linspace(0.2, 0.8, n_bumps)
        a = np.zeros(n)
        for mu in centers:
            a += np.exp(-0.5 * ((t - mu) / 0.15) ** 2)
        a = np.maximum(a, 1e-10)
        inits.append(a)

    return inits
