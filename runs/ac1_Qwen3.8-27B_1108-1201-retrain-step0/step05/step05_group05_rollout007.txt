import numpy as np

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_value(c, beta):
    cm = c.max()
    scaled = beta * (c - cm)
    # Clamp for numerical stability
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    return cm + np.log(e.sum()) / beta

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def exact_subgradient(a, c, top_k=5):
    """Exact subgradient from top-k lags."""
    n = len(a)
    k = min(top_k, len(c))
    idx = np.argsort(c)[::-1][:k]
    # Weights proportional to excess above second-highest
    c_sorted = np.sort(c)[::-1]
    if k > 1 and c_sorted[1] > 0:
        weights = np.maximum(c[idx] - c_sorted[min(k, len(c_sorted))-1], 0)
        if weights.sum() < 1e-30:
            weights = np.ones(k)
    else:
        weights = np.ones(k)
    weights = weights / weights.sum()
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def run_lse_fista_v2(a_init, S, max_time, rng, 
                      beta_start=20.0, beta_max=50000.0,
                      beta_rate=0.002, use_polyak=True):
    """
    Improved LSE-FISTA with Polyak step size and adaptive beta.
    """
    n = len(a_init)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    
    # Lower bound on max conv: average value
    T_lower = S * S / (2 * n - 1)
    
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_start * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best - 1e-15:
            best = cmax
            best_a = y.copy()
            last_improve = it
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14 * S:
            break
        
        # Step size
        if use_polyak:
            # Polyak step: (f(y) - f*) / ||g||^2
            # Use T_lower as estimate of f*
            gap = max(cmax - T_lower, 1e-10)
            step = gap / (gnorm * gnorm + 1e-30)
            # Cap the step to not be too aggressive
            step = min(step, 3.0 / (2.0 * beta))
        else:
            step = 1.0 / (2.5 * beta)
        
        # Stuck detection
        if it - last_improve > 5000:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            continue
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()

def generate_symmetric_inits(n, rng):
    """Generate symmetric initializations (a_j = a_{n-1-j})."""
    m = n // 2
    inits = []
    tt = np.linspace(0, 1, m, endpoint=False)
    
    # Raised cosine
    half = np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01)
    inits.append(_make_symmetric(half, n))
    
    # Triangle (tent)
    half = np.maximum(1.0 - 2.0 * tt, 0.01)
    inits.append(_make_symmetric(half, n))
    
    # Gaussian
    half = np.exp(-4.0 * tt ** 2) + 0.01
    inits.append(_make_symmetric(half, n))
    
    # Sinusoidal bump
    half = np.sin(np.pi * tt) ** 2 + 0.01
    inits.append(_make_symmetric(half, n))
    
    # Sharper bump
    half = np.sin(np.pi * tt) ** 4 + 0.01
    inits.append(_make_symmetric(half, n))
    
    # Bohman-like
    half = (1 - tt) * (1 + np.cos(2 * np.pi * tt)) + 0.01
    half = np.maximum(half, 0.01)
    inits.append(_make_symmetric(half, n))
    
    # Random symmetric
    half = rng.random(m) * 0.5 + 0.5
    inits.append(_make_symmetric(half, n))
    
    return inits

def _make_symmetric(half, n):
    """Create a full symmetric sequence from its first half."""
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([half[:m], half[:m][::-1]])
    else:
        return np.concatenate([half[:m], [half[m-1]], half[:m-1][::-1]])

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Approach:
1. Start from given sequence + diverse symmetric initializations
2. LSE-FISTA with Polyak step size and adaptive beta schedule
3. Active-set exact subgradient polishing with LP-guided directions
4. Basin hopping with Fourier mutations for escaping local minima
5. Ultra-fine final polish with very high LSE beta
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c, top_k=8):
    """Exact subgradient from top-k lags with excess weights."""
    n = len(a)
    k = min(top_k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_vals = c[idx]
    if k > 1:
        ref = c_vals[-1]
        weights = np.maximum(c_vals - ref, 0.0)
        if weights.sum() < 1e-30:
            weights = np.ones(k)
    else:
        weights = np.ones(1)
    weights = weights / weights.sum()
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def fourier_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    phase_noise = rng.normal(0, scale, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def fourier_crossover(a1, a2, rng):
    """Fourier-domain crossover."""
    n = len(a1)
    S = a1.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    r = rng.random(fsize)
    alpha = 0.3 + 0.4 * r
    new_mag = alpha * np.abs(A1) + (1 - alpha) * np.abs(A2)
    new_phase = np.where(r < 0.5, np.angle(A1), np.angle(A2))
    new_A = new_mag * np.exp(1j * new_phase)
    new_A[0] = A1[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def run_lse_fista(a_init, S, max_time, rng,
                  beta_start=20.0, beta_max=50000.0,
                  beta_rate=0.002, use_polyak=True,
                  switch_to_exact_at=100.0):
    """
    LSE-FISTA with adaptive beta, Polyak step, and switch to exact subgradient.
    """
    n = len(a_init)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()

    T_lower = S * S / (2 * n - 1)

    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_start * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best - 1e-14:
            best = cmax
            best_a = y.copy()
            last_improve = it

        # Choose gradient method
        if beta > switch_to_exact_at:
            active = np.where(c >= cmax - 1e-6 * max(cmax, 1e-10))[0]
            n_active = len(active)
            if n_active <= 12:
                grad = exact_subgradient(y, c, top_k=n_active)
            else:
                grad = lse_gradient(y, c, beta)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14 * S:
            break

        # Step size
        if use_polyak:
            gap = max(cmax - T_lower, 1e-12)
            step = gap / (gnorm * gnorm + 1e-30)
            step_cap = 3.0 / (max(beta, 1.0))
            step = min(step, step_cap)
        else:
            step = 1.0 / (2.5 * max(beta, 1.0))

        # Stuck detection
        if it - last_improve > 4000:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(best_a)
    best_a = project_simplex(best_a, best_a.sum())
    return best_a, c_final.max()


def run_precise_polish(a_init, S, max_time, rng):
    """
    Ultra-precise polishing using exact subgradient with Polyak steps.
    Switches between LSE and exact modes based on active set size.
    """
    n = len(a_init)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)

    start = time.monotonic()
    T_lower = S * S / (2 * n - 1)

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    t_mom = 1.0
    y = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time

        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best - 1e-14:
            best = cmax
            best_a = y.copy()
            last_improve = it

        # Active set identification with tightening tolerance
        tol = max(1e-10, 1e-4 * (1 - progress))
        active = np.where(c >= cmax - tol * max(cmax, 1e-10))[0]
        n_active = len(active)

        # Gradient selection
        if n_active <= 6:
            grad = exact_subgradient(y, c, top_k=n_active)
        elif n_active <= 20:
            # Weighted subgradient emphasizing the very top
            top_n = min(n_active, 8)
            top_idx = np.argsort(c)[::-1][:top_n]
            w_full = np.zeros(len(c))
            w_full[top_idx] = 1.0 / top_n
            fsize = 1 << (len(c) + n - 1).bit_length()
            W = fft(w_full, fsize)
            A = fft(y, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        else:
            beta = 500.0 + 5000.0 * progress
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14 * S:
            break

        # Polyak step with safety
        gap = max(cmax - T_lower, 1e-14)
        step = gap / (gnorm * gnorm + 1e-30)
        step = min(step, 0.5 * S / (n * gnorm / S + 1e-10))

        # Backtracking if step seems too large
        x_trial = project_simplex(y - step * grad, S)
        c_trial = fft_convolve_self(x_trial)
        if c_trial.max() > cmax * 1.01:
            step *= 0.1
            x_trial = project_simplex(y - step * grad, S)
            c_trial = fft_convolve_self(x_trial)

        if it - last_improve > 3000:
            noise_scale = max(1e-8, 0.0005 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 5000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_trial + ((t_mom - 1.0) / t_new) * (x_trial - a)
        y = np.maximum(y, 0.0)
        a = x_trial
        t_mom = t_new

    c_final = fft_convolve_self(best_a)
    best_a = project_simplex(best_a, best_a.sum())
    return best_a, c_final.max()


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    m = n // 2
    tt_full = np.linspace(0, 1, n, endpoint=False)
    tt_half = np.linspace(0, 1, m, endpoint=False)

    if given_seq is not None:
        g = np.clip(given_seq, 0, 1000).copy()
        if len(g) == n:
            inits.append(g)
            inits.append(0.5 * (g + g[::-1]))  # Symmetric version

    def make_sym(half):
        if n % 2 == 0:
            return np.concatenate([half[:m], half[:m][::-1]])
        else:
            return np.concatenate([half[:m], [half[m - 1]], half[:m - 1][::-1]])

    # Raised cosine
    inits.append(make_sym(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt_half)), 0.01)))
    # Triangle
    inits.append(make_sym(np.maximum(1.0 - 2.0 * tt_half, 0.01)))
    # Gaussian
    inits.append(make_sym(np.exp(-3.0 * tt_half ** 2) + 0.01))
    # Sin^2 bump
    inits.append(make_sym(np.sin(np.pi * tt_half) ** 2 + 0.01))
    # Sin^4 bump (sharper)
    inits.append(make_sym(np.sin(np.pi * tt_half) ** 4 + 0.01))
    # Bohman-like
    bohman = (1 - tt_half) * (1 + np.cos(2 * np.pi * tt_half))
    inits.append(make_sym(np.maximum(bohman, 0.01)))
    # Flat top with tapered edges
    flat = np.ones(m) * 0.8
    taper = int(0.1 * m)
    for i in range(taper):
        t = i / taper
        flat[i] *= 0.8 * t + 0.2
        flat[m - 1 - i] *= 0.8 * t + 0.2
    inits.append(make_sym(np.maximum(flat, 0.01)))
    # Random
    inits.append(rng.random(n) * 0.5 + 0.5)
    inits.append(make_sym(rng.random(m) * 0.5 + 0.5))

    return inits


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.

    Phases:
    1. Diverse initialization + quick LSE-FISTA screening
    2. Basin hopping with Fourier mutations
    3. Precise active-set polishing
    4. Ultra-fine final convergence
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 25

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f, obj=%.8f" % (
                len(given), given.sum(), objective_value(given)))
    except Exception:
        pass

    n = 8192
    S = np.sqrt(2.0 * n)
    global_best = np.inf
    global_a = None

    def update_global(a, val):
        nonlocal global_best, global_a
        obj = 2.0 * n * val / (a.sum() ** 2)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  >>> NEW BEST: obj=%.8f" % obj)
            return True
        return False

    # ===== Phase 1: Quick screening =====
    print("\n=== Phase 1: Quick screening (~200s) ===")
    p1_start = time.monotonic()
    p1_budget = min(200.0, (deadline - p1_start) * 0.12)

    inits = generate_inits(n, rng, given_seq=given)
    n_screen = min(10, len(inits))
    per_init = p1_budget / n_screen
    candidates = []

    for i in range(n_screen):
        if time.monotonic() > deadline - (deadline - total_start) * 0.82:
            break
        t_alloc = min(per_init, deadline - time.monotonic() - 300)
        if t_alloc < 8:
            break
        a_opt, val = run_lse_fista(inits[i], S, t_alloc, rng,
                                    beta_start=10.0, beta_max=3000.0,
                                    beta_rate=0.008, use_polyak=True,
                                    switch_to_exact_at=200.0)
        obj = 2.0 * n * val / (a_opt.sum() ** 2)
        candidates.append((obj, a_opt.copy()))
        print("  Screen %d: obj=%.8f" % (i, obj))
        update_global(a_opt, val)

    candidates.sort(key=lambda x: x[0])
    print("  Best after screening: obj=%.8f" % global_best)

    # ===== Phase 2: Basin hopping =====
    print("\n=== Phase 2: Basin hopping (~600s) ===")
    p2_start = time.monotonic()
    p2_budget = min(600.0, (deadline - p2_start) * 0.33)

    bh_starts = min(3, len(candidates))
    population = [(candidates[0][0], candidates[0][1].copy())]

    for bh_i in range(bh_starts):
        if time.monotonic() > deadline - 400:
            break
        remaining = deadline - time.monotonic() - 400
        t_alloc = remaining * (0.5 / max(1, bh_starts - bh_i))
        t_alloc = min(t_alloc, p2_budget / max(1, bh_starts - bh_i) + 20)
        if t_alloc < 40:
            break

        print("  BH round %d (start obj=%.8f), t=%.0fs" % (bh_i, candidates[bh_i][0], t_alloc))

        a = candidates[bh_i][1].copy()
        n_iters = max(4, int(t_alloc / 80))
        opt_time = max(20.0, t_alloc / n_iters - 3)

        for it in range(n_iters):
            if time.monotonic() > deadline - 400:
                break
            progress = it / max(1, n_iters)
            scale = 0.25 * np.exp(-2.5 * progress) + 0.02

            r = rng.random()
            if len(population) >= 2 and r < 0.35:
                idx2 = int(rng.integers(len(population)))
                a = fourier_crossover(a, population[idx2][1], rng)
            else:
                a = fourier_mutate(a, rng, scale)

            a_opt, val = run_lse_fista(a, S, opt_time, rng,
                                        beta_start=30 + 80 * it,
                                        beta_max=30000.0,
                                        beta_rate=0.003, use_polyak=True,
                                        switch_to_exact_at=200.0)
            obj = 2.0 * n * val / (a_opt.sum() ** 2)
            update_global(a_opt, val)

            population.append((obj, a_opt.copy()))
            population.sort(key=lambda x: x[0])
            population = population[:8]

    # ===== Phase 3: Precise polishing =====
    print("\n=== Phase 3: Precise polishing (~500s) ===")
    p3_start = time.monotonic()
    p3_budget = min(500.0, (deadline - p3_start) * 0.30)

    if global_a is not None and p3_budget > 30:
        n_sub = 4
        for sub in range(n_sub):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(p3_budget / (n_sub - sub), deadline - time.monotonic() - 150)
            if t_alloc < 20:
                break
            print("  Polish sub-phase %d/%d, t=%.0fs" % (sub + 1, n_sub, t_alloc))
            a_p, val_p = run_precise_polish(global_a, S, t_alloc, rng)
            update_global(a_p, val_p)

    # ===== Phase 4: Final ultra-fine convergence =====
    print("\n=== Phase 4: Final ultra-fine polish ===")
    p4_start = time.monotonic()
    p4_budget = deadline - p4_start - 10

    if global_a is not None and p4_budget > 20:
        # High-beta LSE FISTA
        seg1 = p4_budget * 0.5
        a_hb, val_hb = run_lse_fista(global_a, S, seg1, rng,
                                      beta_start=200.0, beta_max=200000.0,
                                      beta_rate=0.0003, use_polyak=True,
                                      switch_to_exact_at=10.0)
        update_global(a_hb, val_hb)

        # Final precise polish
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            a_fp, val_fp = run_precise_polish(global_a, S, remaining, rng)
            update_global(a_fp, val_fp)

    # ===== Final verification =====
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.8f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
