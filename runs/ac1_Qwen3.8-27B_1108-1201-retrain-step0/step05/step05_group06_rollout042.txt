def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_smooth_max(c, beta):
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    return cm + np.log(e.sum()) / beta

def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Gradient of LSE w.r.t. y: 2 * convolve(w_reversed, y) [0:n]
    # Actually, d/dy_j [LSE(c)] = 2 * sum_k w_k * dc_k/dy_j
    # dc_k/dy_j involves the convolution structure
    # More precisely: c = y*y (full conv), so dc_k = dy_k * y + y * dy_k
    # d(LSE)/dy_j = 2 * sum_k w_k * (dy_k/dy_j convolved with y + y convolved with dy_k/dy_j)
    # = 2 * (w * y) [first n elements] (by symmetry of convolution)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def make_fbm_init(n, rng, self_sim=0.7):
    """Generate fractional Brownian motion-like initialization."""
    freqs = np.fft.rfftfreq(n)
    freqs[0] = 1.0
    power = 1.0 / (freqs ** (2 * self_sim + 0.1))
    phase = rng.uniform(0, 2 * np.pi, len(freqs))
    spectrum = np.sqrt(power) * (np.cos(phase) + 1j * np.sin(phase))
    a = np.fft.irfft(spectrum, n)
    a = a - a.min() + 0.1
    return a

def make_log_uniform_init(n, rng):
    """Log-uniform distribution initialization."""
    a = rng.uniform(0.01, 1.0, n)
    return a

def make_multi_scale_cosine(n, rng):

    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    n_harmonics = 5

        amp = 1.0 / k
        a += amp * np.cos(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))

                if np.all(x_trial >= 0):

import numpy as np

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

"""
Minimize 2*n*max_conv/sum^2 using multi-strategy optimization:
- Hybrid FISTA with adaptive LSE/exact-subgradient switching and backtracking
- Log-parameterization for different optimization dynamics  
- Multi-peak weighted gradient for simultaneous peak reduction
- Pairwise local search for fine-grained improvements
- Multi-scale exploration (n -> 2n upsampling)
- Stochastic perturbation for local minimum escape

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def weighted_peak_gradient(y, c, n_peaks, tol_frac):
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
        if len(active) > n_peaks:
            idx = np.argsort(c)[::-1][:n_peaks]
        else:
            idx = active
    else:
        idx = np.argsort(c)[::-1][:min(n_peaks, len(c))]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / len(idx)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exp_weighted_peak_gradient(y, c, n_peaks, scale_frac):
    n = len(y)
    idx = np.argsort(c)[::-1][:min(n_peaks, len(c))]
    diffs = c.max() - c[idx]
    scale = max(c.max() * scale_frac, 1e-10)
    weights = np.exp(-diffs / scale)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(len(idx)) / len(idx)
    else:
        weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def upsample_linear(a, factor):

    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)

    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve

        if stuck < 500:
            grad = weighted_peak_gradient(a, c, 1, 0.0)
        else:
            grad = exp_weighted_peak_gradient(a, c, min(16, 4 + stuck // 100), 0.01)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Find best donor (high positive grad) and receiver (high negative grad)
        pos_idx = np.argmax(grad)
        neg_idx = np.argmin(grad)

        if grad[pos_idx] <= 0 or a[pos_idx] <= 0:
            # Fallback to random
            pos_idx = rng.integers(0, n)
            neg_idx = rng.integers(0, n)
            if pos_idx == neg_idx:
                neg_idx = (neg_idx + 1) % n

        # Also try random pairs in addition to gradient-guided
        n_pairs = max(5, min(50, stuck // 50))
        improved = False

        for pair_i in range(n_pairs):
            if pair_i == 0:
                src = pos_idx
                dst = neg_idx
            else:
                # Gradient-weighted random selection
                probs = np.maximum(grad - grad.min(), 1e-15)
                probs = probs / probs.sum()
                src = rng.choice(n, p=probs)
                probs2 = np.maximum(grad.min() - grad + 1e-15, 1e-15)
                probs2 = probs2 / probs2.sum()
                dst = rng.choice(n, p=probs2)
                if src == dst:
                    continue

            transfer_frac = rng.uniform(0.001, 0.01)
            transfer = a[src] * transfer_frac
            if transfer <= 0 or a[src] < transfer:
                continue

            a[src] -= transfer
            a[dst] += transfer
            a = np.maximum(a, 0)
            a = normalize_seq(a, S)

            c_new = fft_convolve_self(a)
            new_max = c_new.max()

            if new_max < cur_max - 1e-13:
                cur_max = new_max
                c = c_new
                improved = True
                if cur_max < best_conv:
                    best_conv = cur_max
                    best_a = a.copy()
            else:
                # Revert
                a[dst] -= transfer
                a[src] += transfer
                a = normalize_seq(a, S)

        if not improved and stuck > 5000:
            # Random perturbation
            noise_scale = max(1e-8, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 30000.0))
            a_pert = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c_pert = fft_convolve_self(a_pert)
            if c_pert.max() < best_conv or rng.random() < 0.05:
                a = a_pert

    return best_a, best_conv


def run_focused_polish(a_init, max_time, rng):
    """
    Very focused polishing: uses exact subgradient with careful step sizes.
    Tries multiple step size candidates and picks the best.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve

        # Use exact peak gradient (or small top-k when stuck)
        if stuck < 200:
            grad = weighted_peak_gradient(a, c, 1, 0.0)
        elif stuck < 1000:
            grad = weighted_peak_gradient(a, c, min(4, 1 + stuck // 200), 0.0)
        else:
            grad = exp_weighted_peak_gradient(a, c, min(16, 4 + (stuck - 1000) // 200), 0.01)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Try multiple step sizes, pick the best
        base_step = min(1.0 / gnorm, S / (gnorm + 1e-15))
        candidates = [base_step * f for f in [1.0, 0.5, 0.25, 2.0, 0.125]]
        candidates = [max(s, 1e-14) for s in candidates]

        best_trial = None
        best_trial_max = cur_max

        for s in candidates:
            x_trial = project_simplex(a - s * grad, S)
            c_trial = fft_convolve_self(x_trial)
            trial_max = c_trial.max()
            if trial_max < best_trial_max:
                best_trial_max = trial_max
                best_trial = x_trial

        if best_trial is not None:
            a = best_trial
        else:
            # No improvement from gradient; try small random perturbation
            if stuck > 3000:
                noise_scale = max(1e-9, 0.001 * S / np.sqrt(n))
                a_pert = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c_pert = fft_convolve_self(a_pert)
                if c_pert.max() < cur_max:
                    a = a_pert
                else:
                    break  # Truly stuck

    return best_a, best_conv


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-strategy optimization for minimizing 2*n*max_conv/sum^2.
    Combines hybrid FISTA, log-space optimization, gradient-guided local search,
    focused polishing, and multi-scale exploration.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get provided sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj_start = compute_obj(a_start)
        print("Provided sequence: n=%d, obj=%.6f" % (n0, obj_start))
        global_best_obj = obj_start
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        a_start = np.ones(n0)
        obj_start = compute_obj(a_start)
        print("Using uniform init: n=%d, obj=%.6f" % (n0, obj_start))
        global_best_obj = obj_start
        global_best_a = a_start.copy()

    # Helper to track best (using local variables returned)
    def _try_update(a, name):
        nonlocal global_best_obj, global_best_a
        obj = compute_obj(a)
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a.copy()
            print("  *** NEW BEST [%s]: obj=%.6f (n=%d)" % (name, obj, len(a)))
        else:
            print("  [%s]: obj=%.6f" % (name, obj))
        return obj

    # === Phase 1: Hybrid FISTA main (n=original) ===
    remaining = deadline - time.monotonic()
    p1 = remaining * 0.25
    print("\n=== Phase 1: Hybrid FISTA (%.0fs, n=%d) ===" % (p1, len(global_best_a)))
    t = time.monotonic()
    a1, _ = run_hybrid_fista(global_best_a, p1, rng)
    _try_update(a1, "HybridF1")
    print("  t=%.1fs" % (time.monotonic() - t))

    # === Phase 2: Log-space ===
    remaining = deadline - time.monotonic()
    p2 = remaining * 0.15
    if p2 > 15:
        print("\n=== Phase 2: Log-space (%.0fs) ===" % p2)
        t = time.monotonic()
        a2, _ = run_logspace_fista(global_best_a, p2, rng)
        _try_update(a2, "LogSpace")
        print("  t=%.1fs" % (time.monotonic() - t))

    # === Phase 3: Gradient-guided local search ===
    remaining = deadline - time.monotonic()
    p3 = remaining * 0.15
    if p3 > 15:
        print("\n=== Phase 3: Grad-LSearch (%.0fs) ===" % p3)
        t = time.monotonic()
        a3, _ = run_gradient_ascent_descent(global_best_a, p3, rng)
        _try_update(a3, "GradLS")
        print("  t=%.1fs" % (time.monotonic() - t))

    # === Phase 4: Focused polish ===
    remaining = deadline - time.monotonic()
    p4 = remaining * 0.15
    if p4 > 10:
        print("\n=== Phase 4: Focused Polish (%.0fs) ===" % p4)
        t = time.monotonic()
        a4, _ = run_focused_polish(global_best_a, p4, rng)
        _try_update(a4, "Polish")
        print("  t=%.1fs" % (time.monotonic() - t))

    # === Phase 5: Upsample to 2n and optimize ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and len(global_best_a) <= 16384:
        n2 = len(global_best_a) * 2
        p5 = remaining * 0.50
        print("\n=== Phase 5: Upsample n=%d (%.0fs) ===" % (n2, p5))
        t = time.monotonic()

        a_up = upsample_linear(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = normalize_seq(a_up, S2)
        obj_up = compute_obj(a_up)
        print("  Upsampled obj: %.6f" % obj_up)

        a5, _ = run_hybrid_fista(a_up, p5 * 0.45, rng)
        _try_update(a5, "Up-Hybrid")

        rem5 = p5 - (time.monotonic() - t)
        if rem5 > 10:
            a5b, _ = run_focused_polish(global_best_a, rem5, rng)
            _try_update(a5b, "Up-Polish")

        print("  t=%.1fs" % (time.monotonic() - t))

    # === Phase 6: Second round on best ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p6 = remaining * 0.85
        print("\n=== Phase 6: Final round (%.0fs, n=%d) ===" % (p6, len(global_best_a)))
        t = time.monotonic()

        a6, _ = run_hybrid_fista(global_best_a, p6 * 0.4, rng)
        _try_update(a6, "Final1")

        rem6 = p6 - (time.monotonic() - t)
        if rem6 > 5:
            a6b, _ = run_focused_polish(global_best_a, rem6, rng)
            _try_update(a6b, "Final2")

        print("  t=%.1fs" % (time.monotonic() - t))

    # Final verification
    print("\n=== SUMMARY ===")
    print("Best objective: %.6f (n=%d)" % (global_best_obj, len(global_best_a)))

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified objective: %.6f" % ver)
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            fallback = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in fallback]
    else:
        n_fb = 8192
        fallback = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in fallback]

    return final_list
