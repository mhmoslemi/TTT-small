import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_objective_and_grad(a, beta):
    """Compute LSE-smoothed max of convolution and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(n)
    w = e / tot
    # Gradient of LSE w.r.t. a: 2 * (w * a) convolved, truncated to n
    # d(c_k)/da_i = a_{k-i} (for valid k-i) + a_{k-i} ... 
    # Actually d(c_k)/da_i = a_{k-i} if 0 <= k-i < n (first term) + a_{k-i} if... 
    # Wait, c_k = sum_j a_j * a_{k-j}
    # d(c_k)/da_i = a_{k-i} (from the first factor) + a_{k-i} (from the second factor, when j = k-i gives a_i * a_{k-j} = a_i * a_i... no)
    # d(c_k)/da_i: c_k = sum_j a_j * a_{k-j}
    # = ... + a_i * a_{k-i} + ... + a_{k-i} * a_i + ...
    # If i != k-i (i.e., 2i != k): d(c_k)/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i}
    # If i == k-i (i.e., 2i == k): d(c_k)/da_i = a_i (only one term)
    # So d(c_k)/da_i = 2*a_{k-i} - a_i * [2i == k]
    
    # The gradient of LSE = sum_k w_k * d(c_k)/da = sum_k w_k * 2*a_{k-i} (approximately)
    # = 2 * (w * a)_i where (w * a) is the cross-correlation
    # More precisely: sum_k w_k * 2 * a_{k-i} = 2 * sum_k w_k * a_{k-i}
    # Let m = k-i, then k = m+i, sum_m w_{m+i} * a_m
    # This is the cross-correlation of w and a at lag i, times 2.
    
    # Using FFT: cross-correlation can be computed as ifft(conj(FFT(w)) * FFT(a))
    # But we need to be careful about the convolution vs correlation convention.
    
    # Actually, let's think of it differently.
    # sum_k w_k * 2 * a_{k-i} (where k-i is in range [0, n-1])
    # = 2 * sum_{j=0}^{n-1} w_{j+i} * a_j (where j = k-i)
    # This is 2 times the cross-correlation of w and a at lag i.
    
    # The full convolution w * a has length len(w) + n - 1 = (2n-1) + n - 1 = 3n-2
    # (w*a)_m = sum_j w_j * a_{m-j}
    # We want sum_j w_{j+i} * a_j = sum_j a_j * w_{j+i}
    # Let p = j+i: sum_p a_{p-i} * w_p = (a * w)_{2i} ... hmm this is getting confusing.
    
    # Let me use a different approach. Define the "autocorrelation-like" quantity:
    # g_i = 2 * sum_k w_k * a_{k-i} (with a_j = 0 for j outside [0, n-1])
    # = 2 * (cross-correlation of w and a at lag i)
    
    # In terms of convolution: if we define the cross-correlation as
    # (w ⋆ a)_i = sum_k w_k * a_{k+i}
    # Then g = 2 * (w ⋆ a)
    
    # The cross-correlation can be computed as:
    # cross_corr(w, a) = ifft(conj(fft(w)) * fft(a))
    # But we need to be careful about zero-padding.
    
    # Actually, the simplest way: 
    # g_i = 2 * sum_{k=0}^{2n-2} w_k * a_{k-i} (with a_j=0 for j notin [0,n-1])
    # = 2 * (w convolved with reverse(a)) evaluated at position... 
    
    # Let me just use the direct formula with FFT:
    # Define b as the reversal of a: b_j = a_{n-1-j}
    # Then (w * b)_m = sum_j w_j * b_{m-j} = sum_j w_j * a_{n-1-m+j}
    # Hmm, this is also confusing.
    
    # Let me just use the property that:
    # gradient_i = 2 * sum_k w_k * (d c_k / d a_i)
    # d c_k / d a_i = a_{k-i} if k-i in [0,n-1] else 0, PLUS a_{k-i} if k-i in [0,n-1] (from the other factor)
    # Wait no. c_k = sum_j a_j * a_{k-j}
    # d c_k / d a_i: 
    #   - When j = i: derivative of a_i * a_{k-i} w.r.t. a_i is a_{k-i} (if k-i in [0,n-1])
    #   - When k-j = i, i.e., j = k-i: derivative of a_{k-i} * a_i w.r.t. a_i is a_{k-i} (if k-i in [0,n-1])
    #   - If i = k-i (i.e., k = 2i): both terms give a_i, so total is 2*a_i... wait no.
    #     If k = 2i, then j = i gives term a_i * a_i = a_i^2, derivative is 2*a_i.
    #     But we also have j = k-i = i, which is the same term. So we only count it once.
    #     So d c_k / d a_i = a_{k-i} + a_{k-i} = 2*a_{k-i} if 2i != k
    #                     = a_i if 2i == k (which is the same as a_{k-i} = a_i)
    #     Wait, if 2i == k, then a_{k-i} = a_i, and the derivative is 2*a_i (from a_i^2).
    #     So d c_k / d a_i = 2*a_{k-i} in all cases (when k-i in [0,n-1]).
    
    # So: d c_k / d a_i = 2 * a_{k-i} * [0 <= k-i < n]
    
    # Therefore: gradient_i = sum_k w_k * 2 * a_{k-i} * [0 <= k-i < n]
    # = 2 * sum_{k: 0<=k-i<n} w_k * a_{k-i}
    # = 2 * sum_{j=0}^{n-1} w_{j+i} * a_j (substituting j = k-i)
    # = 2 * (cross-correlation of w and a at lag i) [defined as sum_j w_{j+i} * a_j]
    
    # This is 2 * flip_convolve: 2 * (w * flip(a))_{n-1-i} ... let me just use FFT directly.
    
    # Using the fact that sum_j w_{j+i} * a_j is the cross-correlation at lag i,
    # we can compute it as: ifft(fft(w, N) * conj(fft(a, N)))[:n]
    # where N is large enough.
    
    fsize = 1 << (2 * n + 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # Cross-correlation: ifft(W * conj(A)) gives sum_j w_j * conj(a_{j-i}) for real a = sum_j w_j * a_{j-i}
    # We want sum_j w_{j+i} * a_j = sum_j a_j * w_{j+i}
    # = sum_j a_j * w_{j-(-i)} ... 
    
    # Actually: cross_corr[i] = sum_j w[j+i] * a[j] 
    # = ifft(conj(FFT(w)) * FFT(a))[i] ... let me verify.
    # FFT: W[k] = sum_m w[m] exp(-2πimk/N), A[k] = sum_m a[m] exp(-2πimk/N)
    # ifft(conj(W) * A)[i] = (1/N) sum_k conj(W[k]) * A[k] * exp(2πik i/N)
    # = (1/N) sum_k (sum_m w[m] exp(2πimk/N)) * (sum_l a[l] exp(-2πilk/N)) * exp(2πik i/N)
    # = sum_m sum_l w[m] a[l] (1/N) sum_k exp(2πik(m-l+i)/N)
    # = sum_m sum_l w[m] a[l] * [m - l + i = 0 mod N]
    # = sum_m w[m] a[m+i] (for m+i < N, assuming no wraparound)
    # = sum_m w[m] * a[m+i]
    
    # So ifft(conj(W) * A)[i] = sum_m w[m] * a[m+i]
    # We want sum_j w[j+i] * a[j]. Let m = j+i: sum_m w[m] * a[m-i] = ifft(conj(W) * A)[-i] = ifft(conj(W) * A)[N-i]
    
    # Hmm, this is getting confusing with the indexing. Let me just use:
    # g[i] = 2 * sum_j w[j+i] * a[j]
    
    # Using the above: ifft(conj(W)*A)[p] = sum_m w[m]*a[m+p]
    # So sum_j w[j+i]*a[j] = ifft(conj(W)*A)[-i] ... but negative indices wrap around.
    
    # Alternative: sum_j w[j+i]*a[j] = sum_j a[j]*w[j+i]
    # Let's define h[j] = w[j+i]. Then this is sum_j a[j]*h[j] = <a, h> (inner product)
    # h[j] = w[j+i] is w shifted left by i.
    
    # In FFT terms: shift left by i corresponds to multiplying FFT by exp(2πik*i/N)
    # FFT(h)[k] = exp(2πik*i/N) * W[k]
    # <a, h> = ifft(conj(FFT(a)) * FFT(h))[0] = ifft(conj(A) * exp(2πik*i/N) * W)[0]
    
    # This is a circular correlation, which is what we get with FFT.
    
    # For our purposes, since w has length 2n-1 and a has length n, and we zero-pad to fsize >= 3n,
    # the circular and linear versions agree for the range we care about.
    
    # Let me just compute it directly:
    # g = 2 * correlate(w, a) where correlate[w,a][i] = sum_j w[j+i]*a[j]
    # = 2 * ifft(conj(fft(w)) * fft(a)) but shifted appropriately.
    
    # Actually the simplest correct formula:
    # The cross-correlation (w ⋆ a)[i] = sum_n w[n+i] * conj(a[n]) = sum_n w[n+i] * a[n] (for real)
    # In numpy/scipy: scipy.signal.correlate(w, a, mode='full')[offset]
    # But to use FFT: correlate(w, a) = ifft(conj(fft(w)) * fft(a)) with proper padding.
    
    # Let me verify: 
    # scipy.signal.correlate(w, a)[i] = sum_n w[n+i] * a[n] (for i >= 0, 'full' mode)
    # FFT formula: ifft(conj(W) * A) gives the circular cross-correlation.
    # With zero-padding to size >= len(w)+len(a)-1, circular = linear.
    
    # So: cross_corr = np.real(ifft(np.conj(W) * A))
    # cross_corr[i] = sum_n w[n+i]*a[n] for i in valid range
    
    # But wait, the "full" cross-correlation has length len(w)+len(a)-1.
    # The centered part (i=0) is at index len(a)-1 in the full output.
    # With FFT, ifft(conj(W)*A) gives a circular result of length fsize.
    # The linear cross-correlation is embedded in this:
    # For i = 0, ..., len(w)-1: cross_corr[i] = result[i + len(a) - 1] ... 
    # Actually I think the convention is:
    # result[m] = sum_n w[(m - n) mod fsize] * a[n] ... no.
    
    # Let me just use a simple approach: compute the gradient directly using the convolution.
    # gradient_i = 2 * sum_k w_k * a_{k-i}
    # = 2 * (w * a_reversed)[2n-2-i] ... 
    
    # You know what, let me just use the most straightforward method:
    # The gradient of sum_k w_k * c_k w.r.t. a_i is:
    # sum_k w_k * d(c_k)/d(a_i) = sum_k w_k * 2 * a_{k-i} (for valid k-i)
    # = 2 * sum_{j=0}^{n-1} w_{j+i} * a_j
    
    # This is exactly 2 times the cross-correlation of w and a at lag i.
    # I'll compute it as: 2 * ifft(conj(fft(w, fsize)) * fft(a, fsize)) and take the right portion.
    
    # With zero-padding to fsize:
    # ifft(conj(W)*A)[p] for p=0,...,fsize-1 gives circular cross-correlation.
    # For linear cross-correlation of w (length Lw=2n-1) and a (length La=n):
    # The result has length Lw+La-1 = 3n-2.
    # In the circular result (length fsize >= 3n-2), the linear result is:
    #   for p = 0, ..., Lw+La-2: linear_corr[Lw-1+p] = circular[p]  (for p < La)
    #   Hmm, I'm getting confused. Let me just use a different formula.
    
    # Actually, the cleanest way:
    # Define g as the vector where g_i = 2 * sum_k w_k * a_{k-i} for i=0,...,n-1
    # (with a_j = 0 for j < 0 or j >= n)
    
    # This is 2 * (the cross-correlation of w and a, evaluated at lags 0 to n-1)
    
    # Using FFT with zero padding to size fsize >= len(w) + len(a) - 1:
    # cross_corr_full = ifft(conj(fft(w, fsize)) * fft(a, fsize))
    # The full cross-correlation (length len(w)+len(a)-1) is:
    # cross_corr_full[0 : len(a)-1] gives lags -(len(a)-1) to -1 (or +len(w)-1 to +len(w)-len(a)+1, depending on convention)
    # cross_corr_full[len(a)-1 : len(w)+len(a)-1] gives lags 0 to len(w)-1
    
    # Wait, I need to be more careful. Let me use the definition:
    # (w ⋆ a)[i] = sum_n w[n+i] * a[n] for i = -(len(a)-1), ..., len(w)-1
    # So lag 0 is at index len(a)-1 in the zero-based representation of the full correlation.
    
    # In the FFT computation with zero-padding to fsize:
    # result = ifft(conj(fft(w, fsize)) * fft(a, fsize))
    # result is circular, but with enough padding it equals linear for the valid range.
    # The full linear cross-correlation (w ⋆ a) has indices from -(len(a)-1) to len(w)-1.
    # In the circular result of length fsize:
    #   index 0 corresponds to lag 0
    #   index 1 corresponds to lag 1
    #   ...
    #   index len(a)-2 corresponds to lag len(a)-2
    #   index len(a)-1 corresponds to lag -(len(a)-1) (wraparound) -- NO, this is wrong.
    
    # I think the correct mapping is:
    # result[k] = sum_n w[(n+k) mod fsize] * a[n] for k = 0, ..., fsize-1
    # With zero-padding, for k = 0, ..., len(w)-1: result[k] = sum_n w[n+k]*a[n] (the positive lags)
    # For k = fsize - (len(a)-1), ..., fsize-1: result[k] = sum_n w[n+k]*a[n] (the negative lags, wrapped)
    
    # So for our purpose (lags 0 to n-1):
    # g_i = 2 * result[i] for i = 0, ..., n-1
    
    # Let me verify with a small example.
    # w = [w0, w1, w2] (length 3), a = [a0, a1] (length 2)
    # (w ⋆ a)[0] = w0*a0 + w1*a1
    # (w ⋆ a)[1] = w1*a0 + w2*a1
    # (w ⋆ a)[2] = w2*a0
    # (w ⋆ a)[-1] = w0*a1
    
    # FFT with fsize=4:
    # W = fft([w0,w1,w2,0], 4), A = fft([a0,a1,0,0], 4)
    # result = ifft(conj(W)*A)
    # result[0] = w0*a0 + w1*a1 + w2*a2(=0) + w3(=0)*a... 
    # Actually with circular convolution:
    # result[k] = sum_n w[(n+k) mod 4] * a[n]
    # result[0] = w[0]*a[0] + w[1]*a[1] + w[2]*a[2] + w[3]*a[3] = w0*a0 + w1*a1
    # result[1] = w[1]*a[0] + w[2]*a[1] + w[3]*a[2] + w[0]*a[3] = w1*a0 + w2*a1
    # result[2] = w[2]*a[0] + w[3]*a[1] + w[0]*a[2] + w[1]*a[3] = w2*a0
    # result[3] = w[3]*a[0] + w[0]*a[1] + w[1]*a[2] + w[2]*a[3] = w0*a1
    
    # So result[0] = (w⋆a)[0], result[1] = (w⋆a)[1], result[2] = (w⋆a)[2], result[3] = (w⋆a)[-1]
    
    # Great! So for our case with fsize >= len(w) + len(a) - 1:
    # result[i] = (w ⋆ a)[i] for i = 0, ..., len(w)-1
    # result[fsize - (len(a)-1 - 1 + 1)] ... the negative lags are at the end.
    
    # For our gradient: g_i = 2 * (w ⋆ a)[i] for i = 0, ..., n-1
    # = 2 * result[i] for i = 0, ..., n-1
    
    # But we need fsize >= len(w) + n - 1 = (2n-1) + n - 1 = 3n-2.
    # So fsize = next_power_of_2(3*n-2) or similar.
    
    return cm + np.log(tot) / beta, 2.0 * np.real(np.ifft(np.conj(W) * A))[:n]


def compute_obj(a):
    """Compute the actual objective."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def adam_optimize(a0, S, max_time, rng, beta_start=10.0, beta_end=5000.0, 
                  lr_start=0.01, lr_end=0.0001, verbose=True):
    """
    Adam optimizer for minimizing LSE-smoothed max of convolution.
    Subject to sum(a) = S, a >= 0.
    """
    n = len(a0)
    a = project_simplex(a0, S).copy()
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # We also need to handle the simplex constraint. 
    # Strategy: do Adam update, then project onto simplex.
    # The projection might undo some of the gradient direction, but it's a common approach.
    
    it = 0
    stuck = 0
    patience = 2000
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        # Schedule beta
        beta = beta_start * (beta_end / beta_start) ** progress
        # Schedule learning rate
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Compute LSE objective and gradient
        c = fft_convolve_self(a)
        cm = c.max()
        scaled = beta * (c - cm)
        scaled = np.clip(scaled, -500, 500)
        e = np.exp(scaled)
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        
        # Gradient: 2 * cross-correlation of w and a
        fsize = 1 << (3 * n).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
        
        # Adam update
        beta1_t = beta1 ** (it + 1)
        beta2_t = beta2 ** (it + 1)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1_t)
        v_hat = v / (1 - beta2_t)
        
        a_try = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_try = project_simplex(a_try, S)
        
        # Check improvement
        c_try = fft_convolve_self(a_try)
        cur_obj = 2.0 * n * c_try.max() / (S * S)
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a_try.copy()
            stuck = 0
        else:
            stuck += 1
        
        a = a_try
        
        # Noise injection when stuck
        if stuck > patience:
            noise_scale = 0.01 * S / np.sqrt(n) * (1.0 + progress)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.3)
            # Reset Adam state to avoid bias from old gradients
            m = np.zeros(n)
            v = np.zeros(n)
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    it={it}, t={elapsed:.0f}s, obj={best_obj:.6f}, beta={beta:.1f}")
    
    return best_a, best_obj


def generate_ramp_init(n, rng, asymmetry=1.0):
    """Generate a ramp-like initialization."""
    t = np.arange(n, dtype=np.float64) / n
    # Asymmetric ramp: more weight on one side
    a = (1 - t) ** asymmetry + 0.01
    return a


def generate_gaussian_init(n, rng, center=0.3, width=0.2):
    """Generate a Gaussian-like initialization."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((t - center) / width) ** 2) + 0.01
    return a


def generate cosine_init(n):
    """Generate a cosine-bell initialization."""
    t = np.arange(n, dtype=np.float64) / n
    a = 0.5 * (1 + np.cos(2 * np.pi * t)) + 0.01
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Minimize 2n*max(conv(a,a))/(sum(a))^2 using Adam optimizer with LSE smoothing.
    Multi-phase: adaptive beta schedule, multiple initializations, periodic restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    print(f"Starting Adam-LSE optimization, budget={budget_s}s")
    
    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 256:
            print(f"Using provided sequence: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        pass
    
    global_best = np.inf
    global_a = None
    
    def update_global(a, msg=""):
        nonlocal global_best, global_a
        obj = compute_obj(a)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            if msg:
                print(f"  NEW BEST: obj={obj:.6f} ({msg})")
        return obj
    
    # Determine primary length
    if a_provided is not None:
        n_primary = len(a_provided)
    else:
        n_primary = 8192
    
    S = np.sqrt(2.0 * n_primary)
    
    # === Phase 1: Optimize the provided sequence ===
    if a_provided is not None:
        t1 = min(600, (deadline - time.monotonic()) * 0.3)
        print(f"\nPhase 1: Adam-LSE on provided sequence, n={n_primary}, time={t1:.0f}s")
        a1 = a_provided * (S / a_provided.sum())
        a_opt, _ = adam_optimize(a1, S, t1, rng, 
                                  beta_start=50.0, beta_end=3000.0,
                                  lr_start=0.005, lr_end=0.0001)
        update_global(a_opt, "Phase 1 provided")
    
    # === Phase 2: Try different initializations ===
    remaining = deadline - time.monotonic()
    
    inits = [
        ("ramp", generate_ramp_init(n_primary, rng, asymmetry=1.5)),
        ("ramp2", generate_ramp_init(n_primary, rng, asymmetry=2.5)),
        ("gauss", generate_gaussian_init(n_primary, rng, center=0.35, width=0.15)),
        ("cosine", generate_cosine_init(n_primary)),
    ]
    
    if global_a is not None:
        inits.insert(0, ("perturbed_best", global_a + rng.normal(0, 0.02 * S / np.sqrt(n_primary), n_primary)))
    
    t_per_init = min(300, remaining * 0.5 / len(inits))
    for name, init in inits:
        if time.monotonic() > deadline - 60:
            break
        if t_per_init < 10:
            break
        print(f"\nPhase 2: Init '{name}', n={n_primary}, time={t_per_init:.0f}s")

"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses Adam optimizer with LSE-smoothed max of self-convolution,
multiple initialization strategies, adaptive beta/lr schedules,
and periodic noise injection to escape local minima.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    if n == 1:
        return np.array([S])
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(a, c, beta):
    """Compute gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    # grad_i = 2 * sum_j w[j+i] * a[j]
    # = 2 * ifft(conj(FFT(a)) * FFT(w))[i] for i=0,...,n-1
    fsize = 1 << (3 * n).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    lse_val = cm + np.log(tot) / beta
    return grad, lse_val


def compute_obj(a):
    """Compute the actual objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def adam_optimize(a0, S, max_time, rng, beta_start=50.0, beta_end=3000.0,
                  lr_start=0.005, lr_end=0.0002, verbose=True):
    """
    Adam optimizer minimizing LSE-smoothed max of self-convolution.
    Constraint: sum(a) = S, a >= 0.
    """
    n = len(a0)
    a = project_simplex(a0.copy(), S)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8

    it = 0
    stuck = 0
    patience = 1500

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Adaptive schedules
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)
        lr = lr_start * (lr_end / lr_start) ** (progress ** 0.5)

        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Compute gradient
        grad, _ = compute_lse_grad(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_try = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_try = project_simplex(a_try, S)

        # Verify improvement (accept if not worse by too much)
        c_try = fft_convolve_self(a_try)
        obj_try = 2.0 * n * c_try.max() / (S * S)

        if obj_try <= cur_obj * (1.0 + 1e-9) + 1e-12:
            a = a_try
        else:
            # Try smaller step
            a_half = a - 0.25 * lr * m_hat / (np.sqrt(v_hat) + eps)
            a_half = project_simplex(a_half, S)
            c_half = fft_convolve_self(a_half)
            obj_half = 2.0 * n * c_half.max() / (S * S)

            else:

            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            # Reset momentum to avoid stale gradients
            m *= 0.1
            v *= 0.1
            stuck = 0
            patience = int(patience * 1.4)

        if verbose and it % 1000 == 0:

    return best_a, best_obj


def subgradient_optimize(a0, S, max_time, rng, verbose=True):
    """
    Subgradient method with top-k lag weighting as a complementary approach.
    """
    n = len(a0)
    a = project_simplex(a0.copy(), S)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()

    it = 0
    stuck = 0
    patience = 2000

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Top-k subgradient
        k = int(8 + 40 * np.sqrt(progress))
        k = min(k, 2 * n - 1)
        top_idx = np.argpartition(c, -k)[-k:]
        top_c = c[top_idx]
        weights = np.maximum(top_c - top_c.min(), 1e-10)
        weights /= weights.sum()

        # Compute gradient
        grad = np.zeros(n)
        for ii in range(k):
            kk = int(top_idx[ii])
            wi = 2.0 * weights[ii]
            j_start = max(0, kk - n + 1)
            j_end = min(n - 1, kk)
            if j_start <= j_end:
                indices = kk - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += wi * a[indices]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak step size
        gap = cur_max - c.mean()
        step = 2.0 * max(gap, 1e-10) / (gnorm * gnorm)
        step = min(step, 10.0 / gnorm)

        a_try = project_simplex(a - step * grad, S)
        c_try = fft_convolve_self(a_try)
        if c_try.max() <= cur_max * (1.0 + 1e-9):
            a = a_try
        else:
            a = project_simplex(a - 0.3 * step * grad, S)

        if stuck > patience:
            noise_scale = 0.015 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.3)

        if verbose and it % 2000 == 0:
            print(f"    [subgrad] it={it}, t={elapsed:.0f}s, obj={best_obj:.6f}", flush=True)

    return best_a, best_obj


def generate_ramp_init(n, asymmetry=1.5):
    """Asymmetric ramp initialization."""
    t = np.arange(n, dtype=np.float64) / n
    a = (1.0 - t) ** asymmetry + 0.005
    return a


def generate_gauss_init(n, center=0.3, width=0.2):
    """Gaussian bump initialization."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((t - center) / width) ** 2) + 0.005
    return a


def generate_bimodal_init(n, c1=0.25, c2=0.65, w=0.12):
    """Bimodal initialization."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((t - c1) / w) ** 2) + np.exp(-((t - c2) / w) ** 2) + 0.005
    return a


def generate_cosine_init(n):
    """Cosine bell initialization."""
    t = np.arange(n, dtype=np.float64) / n
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) + 0.005
    return a


def generate_power_init(n, p=2.0):
    """Power law ramp."""
    t = np.arange(n, dtype=np.float64) / n
    a = (1.0 - t) ** p + 0.005
    return a
