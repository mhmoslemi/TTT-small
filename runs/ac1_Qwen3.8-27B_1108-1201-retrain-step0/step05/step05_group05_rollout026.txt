def fft_convolve_self(a):

    fsize = 1 << (size - 1).bit_length()

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]

    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def kkt_refine(a, S, n_active_lags=10, n_iter=100, rng=None):
    """
    KKT-based refinement: find the best convex combination of active lag
    subgradients and take a step in that direction.
    """
    n = len(a)
    a_work = normalize_sum(a, S)
    c = fft_convolve_self(a_work)
    cmax = c.max()
    
    # Find active lags
    tol = 0.01 * cmax  # 1% of max
    active = np.where(c >= cmax - tol)[0]
    active = active[:n_active_lags]
    
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    # Compute subgradients for each active lag
    fsize = 1 << (2 * n).bit_length()
    A_fft = fft(a_work, fsize)
    
    grads = np.zeros((len(active), n))
    for i, k in enumerate(active):
        w = np.zeros(2 * n - 1)
        w[k] = 1.0
        W = fft(w, fsize)
        grads[i] = 2.0 * np.real(ifft(W * np.conj(A_fft)))[:n]
    
    # Find weights that minimize variance of combined grad on support
    support = a_work > 1e-10 * a_work.max()
    if support.sum() < 2:
        support = np.ones(n, dtype=bool)
    
    best_weights = np.ones(len(active)) / len(active)
    best_var = np.inf
    
    # Try a few weight combinations
    for _ in range(20):
        if len(active) == 1:
            w_try = np.ones(1)
        else:
            w_try = rng.dirichlet(np.ones(len(active)) * 0.5)
        combined = w_try @ grads
        vals = combined[support]
        var = vals.var()
        if var < best_var:
            best_var = var
            best_weights = w_try
    
    combined_grad = best_weights @ grads
    gnorm = np.linalg.norm(combined_grad)
    if gnorm < 1e-15:
        return a_work, cmax
    
    # Take a small step
    step = S / (n * gnorm) * 0.5
    a_new = project_simplex(a_work - step * combined_grad, S)
    c_new = fft_convolve_self(a_new)
    
    if c_new.max() < cmax:
        return a_new, c_new.max()
    return a_work, cmax


def spectral_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    # More perturbation at mid frequencies
    weight = 1.0 + 2.0 * np.sin(np.pi * freqs / fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]  # Preserve sum
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: Multi-phase optimization combining:
1. Diverse initialization with FISTA screening
2. Long FISTA with adaptive LSE smoothing and backtracking line search
3. KKT active-set refinement for subgradient selection
4. Basin hopping with frequency-dependent spectral mutations
5. Ultra-high-beta final polish with tiny steps
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, S):
    """Clip to [0,1000] and rescale to sum S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_compute(a, beta):
    """Compute LSE value, gradient, convolution, and weights."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500.0, 0.0)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    fsize = 1 << (2 * n).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n] / beta
    return lse_val, grad, c, w


def fista_optimize(a_init, S, max_time, rng, beta_init=10.0,
                   beta_growth=1.02, beta_max=50000.0,
                   use_ls=True, ls_max_iter=8):
    """
    FISTA with adaptive LSE smoothing and optional backtracking line search.
    Returns (best_a, best_max_b).
    """
    n = len(a_init)
    a = normalize_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()

    c0 = fft_convolve_self(a)
    best_max = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: grow when making progress
        if it % 30 == 0:
            if it - last_improve < 300:
                beta = min(beta_max, beta * beta_growth)
            elif it - last_improve > 2000:
                beta = max(beta_init, beta * 0.95)

        c = fft_convolve_self(y)
        curr_max = c.max()
        if curr_max < best_max - 1e-13 * max(1.0, best_max):
            best_max = curr_max
            best_a = y.copy()
            last_improve = it

        # Compute LSE gradient
        lse_val, grad, c, w = lse_compute(y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Step size with optional line search
        if use_ls and beta < 10000:
            step = 1.0 / (2.0 * beta)
            c1 = 1e-4
            accepted = False
            for ls in range(ls_max_iter):
                y_trial = y - step * grad
                x_trial = project_simplex(y_trial, S)
                c_trial = fft_convolve_self(x_trial)
                f_trial = c_trial.max()
                f_curr = c.max()
                if f_trial <= f_curr - c1 * step * (grad @ (y - x_trial)):
                    accepted = True
                    break
                step *= 0.5
            if not accepted:
                step = max(step, 1e-10 / max(gnorm, 1e-10))
        else:
            step = 1.0 / (2.5 * beta)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Stuck handling: inject perturbation
        if it - last_improve > 3000:
            elapsed_frac = (time.monotonic() - start) / max_time
            noise_scale = 0.005 * S / np.sqrt(n) * (1.0 - elapsed_frac) * 0.5
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it

    return best_a, best_max


def kkt_refine_step(a, S, n_lags=12, rng=None):
    """
    KKT-inspired refinement: find optimal weights on active lags,
    take a projected step in the combined subgradient direction.
    Returns (new_a, new_max_b).
    """
    n = len(a)
    a_work = normalize_sum(a, S)
    c = fft_convolve_self(a_work)
    cmax = c.max()

    # Active lags: within 2% of max
    tol = 0.02 * cmax
    active = np.where(c >= cmax - tol)[0]
    active = active[:n_lags]
    if len(active) == 0:
        active = np.array([np.argmax(c)])

    # Compute subgradients for active lags
    fsize = 1 << (2 * n).bit_length()
    A_fft = fft(a_work, fsize)
    grads = np.zeros((len(active), n))
    for i in range(len(active)):
        w_emp = np.zeros(fsize)
        w_emp[active[i]] = 1.0
        grads[i] = 2.0 * np.real(ifft(w_emp * np.conj(A_fft)))[:n]

    # Find weights minimizing gradient variance on support
    support = a_work > 1e-8 * a_work.max()
    n_sup = support.sum()
    if n_sup < 2:
        support = np.ones(n, dtype=bool)

    best_var = np.inf
    best_weights = np.ones(len(active)) / len(active)

    if rng is not None:
        n_trials = min(30, max(10, len(active) * 3))
        for _ in range(n_trials):
            if len(active) == 1:
                w_try = np.ones(1)
            else:
                alpha = rng.uniform(0.3, 2.0, len(active))
                w_try = alpha / alpha.sum()
            combined = w_try @ grads
            vals = combined[support]
            var = np.var(vals)
            if var < best_var:
                best_var = var
                best_weights = w_try.copy()
    else:
        # Try uniform and a few heuristics
        candidates_w = [np.ones(len(active)) / len(active)]
        if len(active) > 1:
            c_vals = c[active]
            w_prop = (c_vals ** 4) / (c_vals ** 4).sum()
            candidates_w.append(w_prop)
        for w_try in candidates_w:
            combined = w_try @ grads
            vals = combined[support]
            var = np.var(vals)
            if var < best_var:
                best_var = var
                best_weights = w_try.copy()

    combined_grad = best_weights @ grads
    gnorm = np.linalg.norm(combined_grad)
    if gnorm < 1e-15:
        return a_work, cmax

    # Compute optimal step via minimization along the direction
    # We want to minimize (a - step*g)^T H (a - step*g) approximately
    # Use a small fixed step scaled by the geometry
    step = 0.3 * S / (n * gnorm + 1e-10)

    a_new = project_simplex(a_work - step * combined_grad, S)
    c_new = fft_convolve_self(a_new)
    new_max = c_new.max()

    if new_max < cmax - 1e-14:
        return a_new, new_max
    return a_work, cmax


def spectral_mutate(a, rng, scale=0.1):
    """
    Fourier-domain mutation: perturb phases with frequency-dependent noise.
    Mid-frequencies get more perturbation (they control the "shape").
    """
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A).copy()
    phase = np.angle(A)

    # Frequency-dependent noise: emphasizing mid-frequencies
    freqs = np.arange(fsize, dtype=np.float64)
    # Smooth bump centered at fsize/4 (quarter frequency)
    weight = 1.0 + 2.0 * np.exp(-0.5 * ((freqs - fsize / 4.0) / (fsize / 8.0)) ** 2)
    weight += 1.0 * np.exp(-0.5 * ((freqs - 3.0 * fsize / 8.0) / (fsize / 10.0)) ** 2)

    phase_noise = rng.normal(0, 1.0, fsize) * scale * weight
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]  # Preserve DC (sum)
    # Enforce conjugate symmetry for real output
    for k in range(1, fsize // 2 + 1):
        new_A[fsize - k] = np.conj(new_A[k])

    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def spectral_crossover(a1, a2, rng):
    """Fourier-domain crossover between two sequences."""
    n = len(a1)
    S = a1.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    r = rng.random(fsize)
    alpha = 0.3 + 0.4 * r
    new_mag = alpha * np.abs(A1) + (1.0 - alpha) * np.abs(A2)
    new_phase = np.where(r < 0.5, np.angle(A1), np.angle(A2))
    new_A = new_mag * np.exp(1j * new_phase)
    new_A[0] = A1[0]
    for k in range(1, fsize // 2 + 1):
        new_A[fsize - k] = np.conj(new_A[k])
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def peak_shave_step(a, S, k_top=8, rng=None):
    """
    Targeted peak shaving: reduce the top-k peaks using a weighted
    subgradient from those specific lags.
    """
    n = len(a)
    a_work = normalize_sum(a, S)
    c = fft_convolve_self(a_work)
    cmax = c.max()

    # Top k lags weighted by excess above median
    c_sorted_idx = np.argsort(c)[::-1]
    k_top = min(k_top, len(c))
    top_idx = c_sorted_idx[:k_top]
    median_c = np.median(c)
    weights = np.maximum(c[top_idx] - median_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-15:
        weights = np.ones(k_top) / k_top
    else:
        weights = weights / wsum

    # Combined subgradient
    fsize = 1 << (2 * n).bit_length()
    A_fft = fft(a_work, fsize)
    combined = np.zeros(n)
    for i in range(k_top):
        w_emp = np.zeros(fsize)
        w_emp[top_idx[i]] = weights[i]
        combined += np.real(ifft(w_emp * np.conj(A_fft)))[:n]
    combined *= 2.0

    gnorm = np.linalg.norm(combined)
    if gnorm < 1e-15:
        return a_work, cmax

    step = 0.5 * S / (n * gnorm + 1e-10)
    a_new = project_simplex(a_work - step * combined, S)
    c_new = fft_convolve_self(a_new)
    new_max = c_new.max()

    if new_max < cmax - 1e-14:
        return a_new, new_max
    return a_work, cmax


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        g = np.clip(given_seq, 0, 1000)
        if g.size == n and g.sum() > 0:
            inits.append(g.copy())

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * tt)))
    # Sine power
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.01)
    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4 * np.pi * tt + 0.5)
                            + 0.15 * np.cos(10 * np.pi * tt + 1.2), 0.01))
    inits.append(np.maximum(1.0 + 0.25 * np.sin(6 * np.pi * tt)
                            + 0.2 * np.cos(14 * np.pi * tt + 0.3), 0.01))
    # Skewed (non-symmetric) - important for breaking the symmetric optimum
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * tt)
                            + 0.2 * np.cos(4 * np.pi * tt + np.pi / 3), 0.01))
    inits.append(np.maximum(1.0 - 0.3 * np.cos(2 * np.pi * tt)
                            + 0.1 * np.sin(6 * np.pi * tt), 0.01))
    # Random
    inits.append(rng.random(n) + 0.1)
    # Gaussian-like
    center = np.linspace(-1, 1, n)
    inits.append(np.exp(-2.0 * center ** 2) + 0.05)
    # Double-hump (non-symmetric)
    inits.append(np.maximum(0.5 + 0.4 * np.exp(-8 * (tt - 0.3) ** 2)
                            + 0.3 * np.exp(-8 * (tt - 0.7) ** 2), 0.01))

    return inits


def run_basin_hopping(a_init, S, max_time, rng):
    """
    Basin hopping: mutate in spectral domain, then local FISTA optimization.
    Maintains a small population for diversity.
    """
    n = len(a_init)
    a = normalize_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best_max = c0.max()
    best_a = a.copy()

    opt_time = max(10.0, max_time / 25.0)
    n_pop = 6
    population = [(best_max, a.copy())]

    it = 0
    while time.monotonic() - start < max_time - opt_time * 0.5:
        it += 1
        progress = it / 30.0  # Normalize over ~30 iterations

        # Adaptive mutation scale
        base_scale = 0.35 * np.exp(-1.5 * progress) + 0.02
        scale = base_scale * (0.7 + 0.6 * rng.random())

        # Choose operation
        r = rng.random()
        if len(population) >= 2 and r < 0.25:
            i1 = rng.integers(len(population))
            i2 = rng.integers(len(population))
            if i1 != i2:
                a_mut = spectral_crossover(population[i1][1], population[i2][1], rng)
            else:
                a_mut = spectral_mutate(a, rng, scale)
        else:
            a_mut = spectral_mutate(best_a, rng, scale)

        # Local optimization
        t_alloc = min(opt_time, max_time - (time.monotonic() - start) - 30)
        if t_alloc < 5:
            break
        a_opt, max_b = fista_optimize(a_mut, S, t_alloc, rng,
                                       beta_init=30.0 + 20.0 * min(it, 10),
                                       beta_growth=1.012,
                                       beta_max=15000.0,
                                       use_ls=True)

        if max_b < best_max:
            best_max = max_b
            best_a = a_opt.copy()

        # Update population
        population.append((max_b, a_opt.copy()))
        if len(population) > n_pop:
            population.sort(key=lambda x: x[0])
            population = population[:n_pop]

    return best_a, best_max


def mirror_step(a, grad, eta, S):
    """KL mirror descent step: a_new ~ a * exp(-eta * grad), normalized."""
    g_max = np.max(np.abs(grad))
    if g_max < 1e-30:
        return a.copy()
    scaled = -np.clip(eta * grad / g_max, -25, 25)
    a_new = a * np.exp(scaled)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new = a_new * (S / s)
    return np.maximum(a_new, 0.0)

        nonlocal global_best, global_a
        obj = 2.0 * n * max_b / (a.sum() ** 2)
        if obj < global_best and np.isfinite(obj):
            global_best = obj
            global_a = a.copy()
            print("  [*] NEW BEST: obj=%.8f (max_b=%.8f) at t=%.0fs"
                  % (obj, max_b, time.monotonic() - total_start))

    # ==================== Phase 1: Screening ====================
    print("\n=== Phase 1: Diverse screening ===")
    p1_budget = min(150.0, (deadline - total_start) * 0.10)
    inits = generate_inits(n, rng, given_seq=given)
    n_screen = min(8, len(inits))
    per_time = p1_budget / n_screen

    candidates = []
    for i in range(n_screen):
        remaining = deadline - time.monotonic() - (deadline - total_start) * 0.85
        if remaining < 20:
            break
        t_alloc = min(per_time, remaining)
        if t_alloc < 10:
            break
        a_opt, max_b = fista_optimize(inits[i], S, t_alloc, rng,
                                       beta_init=5.0, beta_growth=1.03,
                                       beta_max=5000.0)
        obj = 2.0 * n * max_b / (a_opt.sum() ** 2)
        candidates.append((obj, a_opt.copy(), max_b))
        print("  Screen %d: obj=%.6f" % (i, obj))
        _try_update(a_opt, max_b)

    candidates.sort(key=lambda x: x[0])
    print("  Phase 1 best: obj=%.8f" % global_best)

    # ==================== Phase 2: Main FISTA ====================
    print("\n=== Phase 2: Long FISTA ===")
    p2_budget = min(700.0, (deadline - total_start) * 0.35)
    if global_a is not None:
        remaining = deadline - time.monotonic() - (deadline - total_start) * 0.50
        t_alloc = min(p2_budget, remaining)
        if t_alloc > 50:
            a_opt, max_b = fista_optimize(global_a, S, t_alloc, rng,
                                           beta_init=15.0, beta_growth=1.012,
                                           beta_max=30000.0, use_ls=True)
            _try_update(a_opt, max_b)
            print("  Phase 2 best: obj=%.8f" % global_best)

    # ==================== Phase 3: Basin Hopping ====================
    print("\n=== Phase 3: Basin hopping ===")
    p3_budget = min(600.0, (deadline - total_start) * 0.30)
    if global_a is not None:
        remaining = deadline - time.monotonic() - (deadline - total_start) * 0.30
        t_alloc = min(p3_budget, remaining)
        if t_alloc > 60:
            a_bh, max_bh = run_basin_hopping(global_a, S, t_alloc, rng)
            _try_update(a_bh, max_bh)
            print("  Phase 3 best: obj=%.8f" % global_best)

    # ==================== Phase 4: Final Polish ====================
    print("\n=== Phase 4: Exhaustive polish ===")
    if global_a is not None:
        remaining = deadline - time.monotonic() - 20
        if remaining > 30:
            a_pol, max_pol = exhaustive_kkt_polish(global_a, S, remaining, rng)
            _try_update(a_pol, max_pol)
            print("  Phase 4 best: obj=%.8f" % global_best)

    # ==================== Final Verification ====================
    print("\n=== Final Result ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verified objective: %.8f" % ver)
        if not np.isfinite(ver):
            print("  WARNING: evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
            ver = evaluate_sequence(final_list)
            print("  Fallback objective: %.8f" % ver)
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Used uniform fallback: obj=%.8f" % ver)

    elapsed = time.monotonic() - total_start
    print("\nComplete in %.1fs. Final obj=%.8f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
