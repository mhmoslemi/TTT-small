"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses a hybrid approach: subgradient descent with top-k lags, mirror descent
mixture, ultra-high-beta LSE polishing, and coordinate-wise refinement.

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient_topk(a, c, k):
    """Weighted subgradient from top-k convolution lags."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    median_c = np.median(c)
    weights = np.maximum(c[top_idx] - median_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum

    grad = np.zeros(n)
    for ii in range(k):
        kidx = int(top_idx[ii])
        wi = weights[ii]
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
    return grad


def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    result = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return result


def mirror_descent_step(a, c, eta, k):
    """Mirror descent (multiplicative weights) step on the simplex."""
    n = len(a)
    k_min = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k_min]
    
    # Subgradient at top-k lags
    grad = np.zeros(n)
    for kidx in top_idx:
        kidx = int(kidx)
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += a[indices]
    
    # Mirror descent update: a_i <- a_i * exp(-eta * grad_i) / Z
    # Use log-space for numerical stability
    log_a = np.where(a > 1e-300, np.log(a), -300)
    new_log_a = log_a - eta * grad
    # Center for numerical stability
    new_log_a -= new_log_a.max()
    new_a = np.exp(new_log_a)
    new_sum = new_a.sum()
    if new_sum < 1e-300:
        return a
    new_a = new_a / new_sum * a.sum()
    return new_a


def optimize_subgradient(a0, S, max_time, rng, verbose=False):
    """Subgradient optimization with adaptive k and line search."""
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 500

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-14:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        progress = min(1.0, (time.monotonic() - start) / max_time)
        # Start with small k, grow to large k
        k = int(max(4, min(2*n-1, 8 + 120 * progress ** 0.7)))

        grad = compute_subgradient_topk(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        gap = cur_max - c.mean()
        if gap < 1e-14:
            break

        # Step size: use polyak-type with cap
        step = 2.0 * gap / (gnorm * gnorm + 1e-30)
        step = min(step, 3.0 / (gnorm + 1e-30))

        # Line search
        accepted = False
        for _ in range(12):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max + 1e-14:
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            step = 0.001 / (gnorm + 1e-30)
            a = project_simplex(a - step * grad, S)

        # Noise when stuck
        if stuck > patience:
            noise_scale = min(0.05, 0.003 * (stuck / patience)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.2)

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, max={best:.8f}")

    return best_a, best

def optimize_lse_polish(a0, S, max_time, rng, beta_init=100.0, beta_max=100000.0):
    """Ultra-high-beta LSE polishing."""
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Beta grows exponentially
        beta = min(beta_max, beta_init * np.exp(0.01 * it))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 200:
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            continue

        grad = compute_lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size for LSE: 1/(beta * L) where L is smoothness constant
        step = 1.0 / (1.0 * beta * 0.1)  # conservative
        step = min(step, 5.0 / (gnorm + 1e-30))

        # Line search
        accepted = False
        for trial in range(8):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max + 1e-15:
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            step = 0.01 / (gnorm + 1e-30)
            a = project_simplex(a - step * grad, S)

    return best_a, best


def coordinate_refine(a0, S, max_time, rng, verbose=False):
    """
    Coordinate-wise refinement: for each coordinate, find the optimal value
    given all others fixed (approximately, using the peak lags).
    """
    n = len(a0)
    a = project_simplex(a0, S).copy()
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = a.copy()

        # Find top-k lags that are "active"
        k = min(32, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]

        # For each coordinate, try to decrease the peak
        # Compute which coordinates contribute to each peak lag
        # A coordinate a_i contributes to lag k if we look at c[k] which involves a[i]*a[k-i]
        # To decrease c[k], we want to decrease a[i] for i that make a[i]*a[k-i] large

        # Simple approach: for each top lag, find the coordinate that contributes most
        # and try to decrease it (redistributing to other coordinates)
        for kidx in top_idx[:16]:
            kidx = int(kidx)
            j_start = max(0, kidx - n + 1)
            j_end = min(n - 1, kidx)
            if j_start > j_end:
                continue
            indices = kidx - np.arange(j_start, j_end + 1)
            contributions = a[j_start:j_end + 1] * a[indices]
            
            # Find the coordinate with the largest contribution
            max_c_idx = np.argmax(contributions)
            coord = j_start + max_c_idx
            
            # Try to reduce this coordinate and redistribute
            reduction = 0.1 * a[coord]
            if reduction < 1e-15:
                continue
            
            # Redistribute to coordinates with smallest values
            a_try = a.copy()
            a_try[coord] -= reduction
            # Add to the 3 smallest coordinates
            smallest_idx = np.argsort(a_try)[:3]
            a_try[smallest_idx] += reduction / 3.0
            a_try = np.maximum(a_try, 0.0)
            a_try = project_simplex(a_try, S)

            c_try = fft_convolve_self(a_try)
            if c_try.max() < cur_max:
                a = a_try
                cur_max = c_try.max()

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    coord it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}")

    return best_a, best


def generate_init(n, rng, base=None):
    """Generate initial sequence."""
    if base is not None and len(base) > 1:
        t_old = np.linspace(0, 1, len(base))
        t_new = np.linspace(0, 1, n)
        a = np.interp(t_new, t_old, base)
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            return a
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.05
    return a


def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Hybrid approach: subgradient descent, mirror descent, ultra-high-beta LSE
polishing, and coordinate-wise refinement for final convergence.

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient_topk(a, c, k):
    """Weighted subgradient from top-k convolution lags."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    median_c = np.median(c[top_idx])
    weights = np.maximum(c[top_idx] - median_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum

    grad = np.zeros(n)
    for ii in range(k):
        kidx = int(top_idx[ii])
        wi = weights[ii]
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * wi * a[indices]
    return grad


def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution with numerical safeguards."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -600, 0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    result = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return result

def mirror_descent_step(a, c, eta, k):
    """Mirror descent (multiplicative weights) step."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]

    grad = np.zeros(n)
    for kidx in top_idx:
        kidx = int(kidx)
        j_start = max(0, kidx - n + 1)
        j_end = min(n - 1, kidx)
        if j_start <= j_end:
            indices = kidx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += a[indices]

    log_a = np.where(a > 1e-300, np.log(a), -300.0)
    new_log_a = log_a - eta * grad
    new_log_a -= new_log_a.max()
    new_a = np.exp(new_log_a)
    new_sum = new_a.sum()
    if new_sum < 1e-300:
        return a.copy()
    new_a = new_a / new_sum * a.sum()
    return new_a
