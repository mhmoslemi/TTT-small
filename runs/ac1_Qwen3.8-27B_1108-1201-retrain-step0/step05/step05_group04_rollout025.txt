import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    # Clip to avoid overflow
    shifted = beta * (c - cm)
    shifted = np.clip(shifted, -50, 50)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta

def active_set_gradient_weighted(a, c, top_k=20, beta_weight=50.0):
    """Compute gradient using weighted active set (weights based on how close to max)."""
    n = len(a)
    cm = c.max()
    # Weight: exp(-beta_weight * (cm - c_k) / cm) for top_k lags
    top_idx = np.argsort(c)[::-1][:top_k]
    weights = np.exp(-beta_weight * (cm - c[top_idx]) / max(cm, 1e-10))
    weights /= weights.sum()
    
    # Gradient is sum of weights * 2 * shifted(a)
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        # c_idx = sum_j a_j * a[idx-j]
        # d(c_idx)/d(a_i) = 2 * a[idx-i] (with boundary handling)
        lo = max(0, idx - n + 1)
        hi = min(idx, n - 1)
        # a[idx - j] for j in [lo, hi] means a values at positions [idx-hi, idx-lo] = [lo, hi] reversed... 
        # Actually: for j in [lo, hi], the partner is idx-j which ranges from idx-hi to idx-lo
        # So grad[lo:hi+1] += 2 * w * a[idx-hi : idx-lo+1][::-1]
        partners = idx - np.arange(lo, hi + 1)
        grad[lo:hi+1] += 2.0 * w * a[partners]
    
    return grad

def newton_step_active(a, c, top_k=10, S=None, max_cg_iter=50, damping=0.5):
    """
    Compute a damped Newton step for the weighted active-set objective.
    H = sum_k w_k * H_k where H_k has 2 on the anti-diagonal for sum=k.
    We solve H*d = -g approximately using CG.
    """
    if S is None:
        S = np.sqrt(2.0 * len(a))
    n = len(a)
    cm = c.max()
    top_idx = np.argsort(c)[::-1][:top_k]
    weights = np.exp(-30.0 * (cm - c[top_idx]) / max(cm, 1e-10))
    weights /= weights.sum()
    
    # Compute gradient g = sum_k w_k * grad_k
    g = np.zeros(n)
    for k, w in zip(top_idx, weights):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        partners = k - np.arange(lo, hi + 1)
        g[lo:hi+1] += 2.0 * w * a[partners]
    
    gnorm = np.linalg.norm(g)
    if gnorm < 1e-15:
        return a.copy()
    
    # H is defined by: (H*d)_i = sum_k w_k * 2 * sum_{j: i+j=k, valid} d_j
    # = sum_k w_k * 2 * d_{k-i} (if k-i is valid)
    # So H*d can be computed as: for each i, (Hd)_i = 2 * sum_k w_k * d_{k-i} (if k-i valid)
    
    def matvec(d):
        Hd = np.zeros(n)
        for k, w in zip(top_idx, weights):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            partners = k - np.arange(lo, hi + 1)
            Hd[lo:hi+1] += 2.0 * w * d[partners]
        return Hd
    
    # Regularize: H + lambda*I to ensure positive definiteness
    lambda_reg = 1e-6 * np.trace(np.diag([matvec(np.eye(n)[:, i])[i] for i in range(min(10, n))])) if n > 10 else 1e-6
    
    # Actually, let's just add a small diagonal
    lambda_reg = 0.001 * S / n
    
    def matvec_reg(d):
        return matvec(d) + lambda_reg * d
    
    # CG to solve (H + lambda*I) * d = -g
    # But CG requires SPD. Let's use a simple iteration instead.
    # Or just use the gradient direction scaled appropriately.
    
    # Simple approach: use preconditioned gradient = H^{-1} g approximation
    # Since H is banded, a few Jacobi iterations might help.
    
    # For now, use a scaled gradient step (acts like a preconditioned step)
    # The "natural" step size for the quadratic a^T H a / 2 is 1/L where L = max eigenvalue of H
    # L ≈ 2 * sum_k w_k * (max number of valid pairs) ≈ 2 * max(w_k) * n
    # But more conservatively: L ≈ 2 * sum(w_k) * min(k+1, 2n-1-k, n)
    
    # Let's estimate L using power iteration (2 iterations)
    v = np.ones(n) / np.sqrt(n)
    for _ in range(3):
        v_new = matvec(v)
        norm = np.linalg.norm(v_new)
        if norm < 1e-30:
            break
        v_new /= norm
        v = v_new
    L_est = float(v @ matvec(v))
    L_est = max(L_est, 1e-10)
    
    step_size = damping / L_est
    d = -step_size * g
    
    # Project
    a_new = project_simplex(a + d, S)
    
    # Check if this is an improvement
    c_new = fft_convolve_self(a_new)
    if c_new.max() > c.max() * 1.001:
        # Backtrack
        for factor in [0.5, 0.25, 0.1]:
            a_test = project_simplex(a + factor * d, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < c.max():
                return a_test
        return a.copy()
    
    return a_new

def pair_surgical(a, n, S, max_time, rng):
    """
    Directly manipulate pairs of coordinates to reduce the peak lag.
    For the max lag k, find pairs (j, k-j) with high product and
    redistribute mass to reduce c_k.
    """
    a = a.copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cm = c.max()
        if cm < best:
            best = cm
            best_a = a.copy()
        
        # Find top lag
        k_star = int(np.argmax(c))
        
        # Find the top contributing pairs for lag k_star
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        js = np.arange(lo, hi + 1)
        partners = k_star - js
        products = a[js] * a[partners]
        
        # Get top pairs
        top_pairs_idx = np.argsort(products)[::-1][:50]
        
        improved = False
        for pi in top_pairs_idx:
            j = js[pi]
            p = partners[pi]
            if j == p:
                continue
            
            # Try transferring mass from a[j] to a[p] (or vice versa)
            # c_k changes by: -a[j]*da + a[p]*da + ... (first order)
            # Actually, if we reduce a[j] by delta and increase a[p] by delta:
            # Delta c_k = -delta*a[p] + delta*a[j] = delta*(a[j]-a[p])
            # We want Delta c_k < 0, so if a[j] > a[p], increase a[p] and decrease a[j]
            # If a[j] < a[p], increase a[j] and decrease a[p]
            
            if a[j] > a[p] + 1e-10:
                # Move mass from j to p
                delta = min(a[j] * 0.01, 0.001 * S / n)
            elif a[p] > a[j] + 1e-10:
                # Move mass from p to j
                delta = min(a[p] * 0.01, 0.001 * S / n)
            else:
                continue
            
            a[j] -= delta
            a[p] += delta
            
            # Check if improvement
            c_test = fft_convolve_self(a)
            if c_test.max() > cm * 1.0001:
                # Revert
                a[j] += delta
                a[p] -= delta
            else:
                improved = True
                break
        
        if not improved:
            # Add small noise
            noise_scale = 0.001 * S / n
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best

    if gnorm_sq < 1e-30:

    # Hg (one matvec)

    d = -alpha * g
    return d, alpha

Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Multi-phase optimization combining LSE-FISTA, Newton-direction active-set
refinement with backtracking, and pair-based surgical optimization.
Key innovation: Uses Hessian-vector products for optimal step sizing along gradient,

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):

    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    cm = c.max()
    top_idx = np.argsort(c)[::-1][:top_k]
    weights = np.exp(-conc * (cm - c[top_idx]) / max(cm, 1e-10))
    weights /= weights.sum()
    Hv = np.zeros(len(v))
    for k, w in zip(top_idx, weights):
        lo = max(0, k - len(v) + 1)
        hi = min(k, len(v) - 1)
        partners = k - np.arange(lo, hi + 1)
        Hv[lo:hi + 1] += 2.0 * w * v[partners]
    return Hv


def compute_newton_step(a, c, S, top_k=15, conc=20.0, max_alpha=0.5):
    """
    Compute optimal step along gradient using quadratic (Newton) approximation.
    Step size = ||g||^2 / (g^T H g), giving exact minimizer of quadratic along grad dir.
    """
    n = len(a)
    cm = c.max()
    top_idx = np.argsort(c)[::-1][:top_k]
    weights = np.exp(-conc * (cm - c[top_idx]) / max(cm, 1e-10))
    weights /= weights.sum()

    # Gradient
    g = np.zeros(n)
    for k, w in zip(top_idx, weights):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        partners = k - np.arange(lo, hi + 1)
        g[lo:hi + 1] += 2.0 * w * a[partners]

    gnorm_sq = np.sum(g * g)
    if gnorm_sq < 1e-30:
        return a.copy()

    # Hg (one mat-vec)
    Hg = np.zeros(n)
    for k, w in zip(top_idx, weights):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        partners = k - np.arange(lo, hi + 1)
        Hg[lo:hi + 1] += 2.0 * w * g[partners]

    gHg = np.sum(g * Hg)
    if gHg < 1e-20:
        alpha = 0.1 / np.linalg.norm(g)
    else:
        alpha = gnorm_sq / gHg
        alpha = min(alpha, max_alpha)

    a_new = a - alpha * g
    return a_new


def run_fista(a_init, n, max_time, rng, beta_start=50, beta_end=2000, beta_steps=80):
    """LSE-FISTA with adaptive beta and stagnation handling."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / beta

        stagnant = it - last_improve
        if stagnant > 300:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stagnant / 1000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            step *= 0.4

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_newton_active(a_init, n, max_time, rng, top_k=15, conc=20.0):
    """
    Newton-direction active-set optimization with backtracking line search.
    Key: uses Hessian-vector product to determine optimal step size.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            last_improve = it

        # Compute Newton step
        a_candidate = compute_newton_step(a, c, S, top_k=top_k, conc=conc)
        a_candidate = project_simplex(a_candidate, S)

        # Backtracking line search
        c_test = fft_convolve_self(a_candidate)
        test_max = c_test.max()

        if test_max >= cur_max:
            # Try smaller steps
            found = False
            for factor in [0.5, 0.25, 0.1, 0.05]:
                a_try = a + factor * (a_candidate - a)
                a_try = project_simplex(a_try, S)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < cur_max * (1 - 1e-7):
                    a = a_try
                    found = True
                    break
            if not found:
                # Stagnation: inject noise
                stagnant = it - last_improve
                noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 500.0)
                if noise_scale > 1e-10:
                    a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                    last_improve = it
                # Try asymmetric perturbation
                if stagnant > 200:
                    mask = rng.random(n) < 0.1
                    a[~mask] *= (1 + rng.normal(0, 0.01, n)[~mask])
                    a = project_simplex(a, S)
        else:
            a = a_candidate

        # Adaptive: increase top_k over time
        eff_top_k = min(30, top_k + it // 50)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_surgical(a_init, n, max_time, rng):
    """
    Pair-based surgical optimization: directly manipulate coordinate pairs
    to reduce the peak convolution lag.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()
        if cm < best:
            best = cm
            best_a = a.copy()

        # Find top 3 lags
        top_lags = np.argsort(c)[::-1][:3]

        improved = False
        for k_star in top_lags:
            k_star = int(k_star)
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            j_range = np.arange(lo, hi + 1)
            partners = k_star - j_range
            products = a[j_range] * a[partners]

            # Get top contributing pairs
            top_pairs = np.argsort(products)[::-1][:30]

            for pi in top_pairs:
                j = int(j_range[pi])
                p = int(partners[pi])
                if j == p:
                    continue
                if a[j] <= 0 or a[p] <= 0:
                    continue

                # Determine direction: move mass from higher to lower to reduce product
                if a[j] > a[p]:
                    src, dst = j, p
                else:
                    src, dst = p, j

                delta = min(a[src] * 0.005, S / (2 * n))

                a[src] -= delta
                a[dst] += delta

                # Quick check (only check the target lag and a few neighbors)
                c_new = fft_convolve_self(a)
                if c_new.max() < cm:
                    improved = True
                    break
                else:
                    # Revert
                    a[src] += delta
                    a[dst] -= delta

            if improved:
                break

        if not improved:
            # Periodic noise with decreasing scale
            noise_scale = max(1e-6, 0.001 * S / n * np.exp(-it / 3000.0))
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def run_asymmetric_refine(a_init, n, max_time, rng):
    """
    Exploit asymmetry: try shifting mass to one side to reduce center peak.
    Uses a smooth tilt factor that gradually increases.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    t = np.linspace(0, 1, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    # Try various tilt levels
    for tilt in [0.1, 0.2, 0.3, 0.15, 0.25, 0.05, -0.1, -0.2]:
        if time.monotonic() - start > max_time:
            break
        # Apply smooth tilt: multiply by exp(tilt * (2*t - 1))
        tilt_factor = np.exp(tilt * (2 * t - 1))
        a_tilt = a * tilt_factor
        a_tilt = normalize_sum(a_tilt, n)

        # Quick LSE refinement
        c = fft_convolve_self(a_tilt)
        if c.max() < best:
            # Do a few LSE iterations
            sub_time = min(10.0, (max_time - (time.monotonic() - start)) * 0.3)
            a_ref, v_ref = run_fista(a_tilt, n, sub_time, rng,
                                      beta_start=200, beta_end=5000, beta_steps=20)
            c_ref = fft_convolve_self(a_ref)
            if c_ref.max() < best:
                best = c_ref.max()
                best_a = a_ref.copy()

    # Also try the original
    c_orig = fft_convolve_self(a)
    if c_orig.max() < best:
        best = c_orig.max()
        best_a = a.copy()

    return best_a, best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Multi-phase: FISTA -> Newton active-set -> Surgical -> Asymmetric refine -> Final polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)
    t0 = time.monotonic()

    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))

    # Load previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 512 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_seq(a):
        nonlocal global_best, global_a
        if a is None or len(a) < 2:
            return
        n = len(a)
        S = np.sqrt(2.0 * n)
        a_norm = normalize_sum(a, n)
        c = fft_convolve_self(a_norm)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a_norm.copy()
            elapsed = time.monotonic() - t0
            print("  New best: %.8f (n=%d) at %.0fs" % (val, n, elapsed))

    # === Phase 1: LSE-FISTA bulk optimization (25% time) ===
    if given is not None:
        n_g = len(given)
        t1 = min(300.0, (deadline - time.monotonic()) * 0.25)
        print("=== Phase 1: LSE-FISTA (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_fista(given, n_g, t1, rng, beta_start=50, beta_end=2000, beta_steps=80)
        try_seq(a1)
        print("  Phase 1 done: best=%.8f" % global_best)

    # === Phase 2: Newton active-set (30% time) ===
    if global_a is not None:
        n_m = len(global_a)
        remaining = deadline - time.monotonic()
        t2 = remaining * 0.30
        print("=== Phase 2: Newton active-set (n=%d, %.0fs) ===" % (n_m, t2))
        a2, v2 = run_newton_active(global_a, n_m, t2, rng, top_k=15, conc=20.0)
        try_seq(a2)

        # Second run with different parameters
        if time.monotonic() < deadline - 60:
            remaining = deadline - time.monotonic()
            t2b = remaining * 0.3
            print("=== Phase 2b: Newton active-set (top_k=10, conc=50) ===")
            a2b, v2b = run_newton_active(global_a, n_m, t2b, rng, top_k=10, conc=50.0)
            try_seq(a2b)

    # === Phase 3: Surgical pair optimization (20% time) ===
    if global_a is not None:
        n_s = len(global_a)
        remaining = deadline - time.monotonic()
        t3 = remaining * 0.20
        print("=== Phase 3: Surgical (n=%d, %.0fs) ===" % (n_s, t3))
        a3, v3 = run_surgical(global_a, n_s, t3, rng)
        try_seq(a3)

    # === Phase 4: Asymmetric refinement (10% time) ===
    if global_a is not None:
        n_as = len(global_a)
        remaining = deadline - time.monotonic()
        t4 = remaining * 0.10
        print("=== Phase 4: Asymmetric refine (n=%d, %.0fs) ===" % (n_as, t4))
        a4, v4 = run_asymmetric_refine(global_a, n_as, t4, rng)
        try_seq(a4)

    # === Phase 5: Final high-beta LSE polish (15% time) ===
    if global_a is not None:
        n_f = len(global_a)
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t5 = remaining * 0.5
            print("=== Phase 5: Final LSE polish (n=%d, %.0fs) ===" % (n_f, t5))
            a5, v5 = run_fista(global_a, n_f, t5, rng,
                                beta_start=2000, beta_end=30000, beta_steps=20)
            try_seq(a5)

    # === Phase 6: One more surgical pass on best ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_p = len(global_a)
            t6 = remaining * 0.6
            print("=== Phase 6: Final surgical (%.0fs) ===" % t6)
            a6, v6 = run_surgical(global_a, n_p, t6, rng)
            try_seq(a6)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence=%.8f, internal=%.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  WARNING: evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result, using uniform fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
