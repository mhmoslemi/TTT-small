Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses L-BFGS in log-space with multi-start strategy.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_obj_grad(x, beta, n):
    """
    Objective and gradient for f(x) = 2n * LSE_beta(conv(exp(x), exp(x))) / (sum exp(x))^2
    with gauge fixing x[0] = 0 (we optimize x[1:]).
    """
    a = np.exp(x)
    S = a.sum()
    c = fft_convolve_self(a)
    
    # LSE computation
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    e_sum = e.sum()
    LSE = cm + np.log(e_sum) / beta
    w = e / e_sum
    
    # Objective (using LSE as proxy for max)
    obj = 2.0 * n * LSE / (S * S)
    
    # Gradient computation
    # dLSE/dx[j] = a[j] * h[j] where h[j] = sum_k w[k]*(a[k-j] + a[j-k])
    fsize = 1 << (2 * n).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    A_flip = fft(a[::-1], fsize)
    
    wa = np.real(ifft(W * A))  # (w * a)[m]
    wa_flip = np.real(ifft(W * A_flip))  # (w * a_flip)[m]
    
    # h[j] = wa[j] + wa_flip[j + n - 1]
    h = np.zeros(n)
    h[:] = wa[:n]
    h[:] += wa_flip[n-1:n-1+n]
    
    dLSE_dx = a * h
    
    # Full gradient of f = 2n * LSE / S^2
    # df/dx[j] = 2n/S^2 * (dLSE/dx[j] - 2*LSE/S * S_part[j])
    # where S_part[j] = dS/dx[j] = a[j]
    grad_full = (2.0 * n / (S * S)) * (dLSE_dx - 2.0 * LSE / S * a)
    
    # Remove the gauge-fixed component (x[0])
    grad = grad_full[1:].copy()
    
    return obj, grad


def lse_obj_grad_with_max(x, beta, n):
    """
    Alternative: use actual max for objective, LSE for gradient.
    """
    a = np.exp(x)
    S = a.sum()
    c = fft_convolve_self(a)
    c_max = c.max()
    obj = 2.0 * n * c_max / (S * S)
    
    # LSE for gradient
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    e_sum = e.sum()
    w = e / e_sum
    
    fsize = 1 << (2 * n).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    A_flip = fft(a[::-1], fsize)
    
    wa = np.real(ifft(W * A))
    wa_flip = np.real(ifft(W * A_flip))
    
    h = np.zeros(n)
    h[:] = wa[:n]
    h[:] += wa_flip[n-1:n-1+n]
    
    dLSE_dx = a * h
    LSE = cm + np.log(e_sum) / beta
    
    grad_full = (2.0 * n / (S * S)) * (dLSE_dx - 2.0 * LSE / S * a)
    grad = grad_full[1:].copy()
    
    return obj, grad


def run_lbfgs(a_init, max_time, rng, beta_start=50.0, beta_end=500.0, n_restarts=5):
    """
    Run L-BFGS in log-space with multiple restarts.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 1e-10, 1000).copy()
    a = normalize_seq(a, S)
    
    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    
    for restart in range(n_restarts):
        if time.monotonic() - start > max_time:
            break
        
        # Initialize x in log-space
        if restart == 0:
            x_log = np.log(a)
        else:
            # Perturb the best found
            noise_scale = 0.05 / (1 + restart * 0.1)
            x_new = np.log(np.maximum(best_a, 1e-10)) + rng.normal(0, noise_scale, n)
            x_log = x_new
        
        # Fix gauge: x[0] = 0
        x_log = x_log - x_log[0]
        x0 = x_log[1:].copy()
        
        # Define the objective function for scipy
        beta = beta_start * (beta_end / beta_start) ** (restart / max(n_restarts - 1, 1))
        
        def fun_and_grad(x_opt):
            x_full = np.concatenate([[0.0], x_opt])
            return lse_obj_grad_with_max(x_full, beta, n)
        
        # Run L-BFGS-B
        t0 = time.monotonic()
        try:
            result = minimize(fun_and_grad, x0, method='L-BFGS-B',
                             jac=True, options={'maxiter': 500, 'gtol': 1e-8,
                                                'maxls': 20, 'disp': False})
        except Exception as e:
            print("  L-BFGS error: %s" % str(e))
            continue
        
        x_opt = result.x
        x_full = np.concatenate([[0.0], x_opt])
        a_new = np.exp(x_full)
        a_new = np.clip(a_new, 0, 1000)
        a_new = normalize_seq(a_new, S)
        
        obj_new = compute_obj(a_new)
        elapsed = time.monotonic() - t0
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            print("  Restart %d: obj=%.8f (improved, %.1fs)" % (restart, obj_new, elapsed))
        else:
            print("  Restart %d: obj=%.8f (no improvement, %.1fs)" % (restart, obj_new, elapsed))
    
    return best_a, best_obj


def run_fista_polish(a_init, max_time, rng, beta=200.0):
    """Quick FISTA polish with high-beta LSE."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 10000:
            break
        
        # LSE gradient
        cm = c.max()
        shifted = beta * (c - cm)
        shifted = shifted - shifted.max()
        e = np.exp(shifted)
        w = e / e.sum()
        
        fsize = 1 << (2 * n).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        
        # Gradient of LSE w.r.t. a: 2 * (w_conv_a) where w_conv_a[k] = sum_j w[j]*a[k-j]
        # dLSE/da[j] = 2 * sum_k w[k] * a[k-j] (approximately, using the symmetric form)
        # Actually: dLSE/da[j] = 2 * (w * a)[j] + 2 * (w * a_flip)[n-1+j] ... 
        # Let me use the same formula as before but for gradient w.r.t. a directly.
        # dLSE/da[j] = 2 * sum_k w[k] * (a[k-j] + a[j-k]) / 2 ... 
        
        # Simpler: dLSE/da[j] = 2 * (w_conv_a)[j] where w_conv_a = ifft(W*conj(A))
        # Wait, that's for the case where LSE = (1/beta)*log(sum exp(beta*c)) and c[k] = sum_j a[j]*a[k-j]
        # dLSE/da[j] = sum_k w[k] * dc[k]/da[j] = sum_k w[k] * 2*a[k-j] (for valid k-j)
        # = 2 * sum_k w[k]*a[k-j] (only one term, since dc[k]/da[j] = a[k-j] + a[j-k] = 2*a[k-j] when symmetric)
        # Wait, dc[k]/da[j] = a[k-j] (from i=j) + a[j-k] (from i=k-j)
        # For real a, a[j-k] = a[k-j] only if a is symmetric. In general they're different.
        # So dc[k]/da[j] = a[k-j] + a[j-k]
        
        # Using FFT: sum_k w[k]*a[k-j] = ifft(conj(W)*A)[j+n-1] (cross-correlation)
        # And sum_k w[k]*a[j-k] = ifft(W*A)[j]
        
        A_rev = fft(a[::-1], fsize)  # Hmm, this uses 'a' not 'y'
        # Let me use y:
        Y = fft(y, fsize)
        Y_flip = fft(y[::-1], fsize)
        
        # sum_k w[k]*a[j-k] = ifft(W*Y)[j]
        term1 = np.real(ifft(W * Y))[:n]
        # sum_k w[k]*a[k-j] = ifft(W*Y_flip)[j+n-1]
        term2 = np.real(ifft(W * Y_flip))[n-1:n-1+n]
        
        grad_a = 2.0 * (term1 + term2)
        
        # Step size
        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-15:
            break
        step = 1.0 / (2.0 * beta * y.max() + gnorm + 1e-10)
        step = max(step, 1e-12)
        step = min(step, 0.5 / (beta * y.max() + 1e-10))
        
        x_new = project_simplex(y - step * grad_a, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        if it % 10000 == 0:
            t_mom = 1.0
    
    return best_a, best_obj


def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def run_active_polish(a_init, max_time, rng):
    """Adaptive top-k subgradient polish."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 50000:
            break
        
        # Adaptive k
        if stuck < 100:
            k = 1
        elif stuck < 1000:
            k = 4
        elif stuck < 5000:
            k = 16
        else:
            k = min(64, 16 + stuck // 200)
        
        # Top-k gradient
        k = min(k, len(c))
        idx = np.argsort(c)[::-1][:k]
        w_full = np.zeros(len(c))
        w_full[idx] = 1.0 / k
        
        fsize = 1 << (2 * n).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        A_flip = fft(y[::-1], fsize)
        
        term1 = np.real(ifft(W * A))[:n]
        term2 = np.real(ifft(W * A_flip))[n-1:n-1+n]
        grad = 2.0 * (term1 + term2)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size with backtracking
        step = min(1.0 / gnorm, 0.1 * S / (gnorm + 1e-15))
        step = max(step, 1e-14)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        if stuck > 5000 and it % 1000 == 0:
            noise_scale = max(1e-6, 0.001 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
    
    return best_a, best_obj
