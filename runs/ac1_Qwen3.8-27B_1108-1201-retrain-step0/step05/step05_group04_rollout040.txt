"""
Minimize 2*n*max(f*f)/sum(f)^2 using a multi-strategy approach:
1. LSE-FISTA with adaptive beta (log-spaced schedule)
2. Multiplicative weights (mirror descent with entropy)

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_full(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Normalize a to have sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_values(c, beta):
    """Compute LSE of c and the softmax weights."""
    cm = c.max()
    shifted = beta * (c - cm)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        e = np.zeros_like(c)
        e[np.argmax(c)] = 1.0
        tot = 1.0
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    return w, lse_val


def gradient_from_weights(w, a):
    """Compute gradient: 2 * ifft(FFT(w) * conj(FFT(a)))[:n]"""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_lse_fista(a_init, n, max_time, rng,
                  beta_start=50.0, beta_end=50000.0, n_beta_steps=100):
    """
    FISTA with log-spaced beta schedule.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    # Pre-compute log-spaced beta values
    betas = np.geomspace(beta_start, beta_end, n_beta_steps)

    while time.monotonic() - start < max_time:
        it += 1
        beta = betas[min(it - 1, len(betas) - 1)]

        c = fft_convolve_full(y)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        w, lse_val = lse_values(c, beta)
        grad = gradient_from_weights(w, a)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta

        # Momentum restart
        if it - last_improve > 200:
            t_mom = 1.0
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            last_improve = it
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_full(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_mirror_descent(a_init, n, max_time, rng,
                       eta_start=0.1, eta_end=0.001, n_steps=5000):
    """
    Multiplicative weights / mirror descent with entropy kernel.
    a_new = a * exp(-eta * grad) / sum(...)
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Log-spaced eta
        frac = min(1.0, it / n_steps)
        eta = eta_start * (eta_end / eta_start) ** frac

        c = fft_convolve_full(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Use LSE with moderate beta for gradient
        beta = max(50.0, 2.0 / eta)
        w, _ = lse_values(c, beta)
        grad = gradient_from_weights(w, a)

        # Multiplicative update
        log_update = -eta * grad
        log_update = np.clip(log_update, -50, 50)  # Numerical stability
        a_new = a * np.exp(log_update)
        s = a_new.sum()
        if s < 1e-300:
            a_new = a
        else:
            a_new = a_new * (S / s)
        a_new = np.clip(a_new, 0.0, 1000.0)
        a = a_new

        # Periodic restart with noise
        if it % 500 == 0:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    c_final = fft_convolve_full(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_adaptive_subgrad(a_init, n, max_time, rng,
                         window_size=50, beta=100.0):
    """
    Subgradient method with exponential moving average of the active set.
    Uses a running window of top lags for a smoothed subgradient.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    # Exponential average of weights
    w_avg = None
    decay = 0.95

    step = S / (np.linalg.norm(a) * 10)

    while time.monotonic() - start < max_time:
        it += 1
        # Decrease step size
        step = S / (np.linalg.norm(a) * (10 + it * 0.01))

        c = fft_convolve_full(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Compute current weights
        w, _ = lse_values(c, beta)

        # Exponential average
        if w_avg is None:
            w_avg = w.copy()
        else:
            w_avg = decay * w_avg + (1 - decay) * w

        grad = gradient_from_weights(w_avg, a)
        a = project_simplex(a - step * grad, S)

        # Periodic reset of the average
        if it % 200 == 0:
            w_avg = None

        # Noise injection
        if it % 100 == 0:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / 10000.0)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_full(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best

def proposal_shortest_grad(a, n, S, n_iter=200):
    """
    Shortest vector gradient: directly minimize the peak lag.
    For the current max lag k*, the gradient is 2*a shifted by k*.
    We take a step in the direction that most reduces c[k*].
    """
    a = a.copy()
    for i in range(n_iter):
        c = fft_convolve_full(a)
        k_max = np.argmax(c)
        # Subgradient: 2 * a[k_max - j] for valid j
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        grad = np.zeros(n)
        # d c[k_max] / d a[j] = a[k_max - j]
        for j in range(lo, hi + 1):
            grad[j] = a[k_max - j]
        grad *= 2.0

        # Step size via backtracking
        step = 0.1 * S / np.linalg.norm(grad) if np.linalg.norm(grad) > 0 else 0
        for _ in range(10):
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_full(a_new)
            if c_new.max() < c.max():
                a = a_new
                break
            step *= 0.5
        else:
            # If no improvement, try smaller fixed step
            a = project_simplex(a - 0.001 * step * grad, S)

    c_final = fft_convolve_full(a)
    return a, c_final.max()

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()

"""
Minimize 2*n*max(f*f)/sum(f)^2 using multi-strategy optimization:
- Adaptive LSE-FISTA with backtracking step size
- Mirror descent (multiplicative weights) for diverse trajectories
- Targeted peak-lag gradient descent with line search
- Convex combination exploration between diverse solutions
- Homotopy-style beta annealing
- Structured perturbation targeting the dominant lag
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_full(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Normalize a to have sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_weights(c, beta):
    """Compute softmax weights from LSE."""
    cm = c.max()
    shifted = np.clip(beta * (c - cm), -500, 500)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        e = np.zeros_like(c)
        e[np.argmax(c)] = 1.0
        tot = 1.0
    return e / tot


def gradient_from_weights(w, a):
    """Compute gradient 2*ifft(FFT(w)*conj(FFT(a)))[:n]."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_lse_fista(a_init, n, max_time, rng,
                  beta_start=50.0, beta_end=50000.0, n_beta=100,
                  use_backtrack=False, momentum_decay=0.999):
    """FISTA with log-spaced beta and optional backtracking step size."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    betas = np.geomspace(beta_start, beta_end, n_beta)

    while time.monotonic() - start < max_time:
        it += 1
        beta = betas[min(it - 1, len(betas) - 1)]
        c = fft_convolve_full(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        w = lse_weights(c, beta)
        grad = gradient_from_weights(w, a)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        if use_backtrack:
            step = 2.0 / beta
            improved = False
            for _ in range(6):
                x_try = project_simplex(y - step * grad, S)
                c_try = fft_convolve_full(x_try)
                if c_try.max() < cur_max:
                    a = x_try
                    improved = True
                    break
                step *= 0.5
            if not improved:
                a = project_simplex(y - 0.1 / beta * grad, S)
            # Update y with momentum (simplified)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
            y = a + ((t_mom - 1.0) / t_new) * (a - y)
            y = np.maximum(y, 0.0)
            t_mom = t_new * momentum_decay
        else:
            step = 1.0 / beta
            # Stagnation handling
            if it - last_improve > 300:
                t_mom = 1.0
                noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 800.0)
                if noise_scale > 1e-10:
                    y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                last_improve = it
                step *= 0.5

            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

    c_final = fft_convolve_full(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_mirror_descent(a_init, n, max_time, rng, eta_start=0.1, eta_end=0.001):
    """Multiplicative weights / mirror descent with entropy kernel."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / 5000.0)
        eta = eta_start * (eta_end / eta_start) ** frac
        c = fft_convolve_full(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        beta = max(50.0, 2.0 / eta)
        w = lse_weights(c, beta)
        grad = gradient_from_weights(w, a)
        log_update = np.clip(-eta * grad, -50, 50)
        a_new = a * np.exp(log_update)
        s = a_new.sum()
        if s < 1e-300:
            a_new = a
        else:
            a_new = a_new * (S / s)
        a = np.clip(a_new, 0.0, 1000.0)

        if it % 400 == 0:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    c_final = fft_convolve_full(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_adaptive_subgrad(a_init, n, max_time, rng, beta=200.0, decay=0.92):
    """Subgradient with exponential moving average of active set."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    w_avg = None

    while time.monotonic() - start < max_time:
        it += 1
        step = S / (np.linalg.norm(a) * (10.0 + 0.003 * it))
        c = fft_convolve_full(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        w = lse_weights(c, beta)
        if w_avg is None:
            w_avg = w.copy()
        else:
            w_avg = decay * w_avg + (1.0 - decay) * w

        grad = gradient_from_weights(w_avg, a)
        a = project_simplex(a - step * grad, S)

        if it % 250 == 0:
            w_avg = None
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 15000.0)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_full(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def targetable_gradient_step(a, n, S, max_time, rng):
    """Directly target the peak lag with backtracking line search."""
    a = a.copy()
    start = time.monotonic()
    best = fft_convolve_full(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_full(a)
        k_max = int(np.argmax(c))
        cur_max = c[k_max]

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Compute subgradient for the dominant lag
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        grad = np.zeros(n)
        js = np.arange(lo, hi + 1)
        grad[js] = 2.0 * a[k_max - js]

        # Also try top-2 lags with equal weight for more robustness
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) >= 2 and c_sorted[1] > c_sorted[0] * 0.95:
            top2 = np.argsort(c)[::-1][:2]
            for k2 in top2[1:]:
                lo2 = max(0, k2 - n + 1)
                hi2 = min(k2, n - 1)
                js2 = np.arange(lo2, hi2 + 1)
                grad[js2] += 2.0 * a[k2 - js2] * 0.5

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 0.5 * S / gnorm
        improved = False
        for _ in range(10):
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_full(a_new)
            if c_new.max() < cur_max * (1 - 1e-9):
                a = a_new
                improved = True
                break
            step *= 0.5
        if not improved:
            # Small fixed step
            a = project_simplex(a - 0.01 * step * grad, S)

        # Occasional noise
        if it % 100 == 0:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / 3000.0)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_full(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def convex_combination_explore(a1, a2, n, S, rng, n_trials=20):
    """Try convex combinations of two solutions and optimize the best."""
    best_val = np.inf
    best_a = a1.copy()
    c1 = fft_convolve_full(a1).max()
    c2 = fft_convolve_full(a2).max()
    if c1 < best_val:
        best_val = c1
    if c2 < best_val:
        best_val = c2
        best_a = a2.copy()

    ts = np.linspace(0.1, 0.9, n_trials)
    for t in ts:
        a_mix = (1 - t) * a1 + t * a2
        a_mix = normalize_sum(a_mix, n)
        c_mix = fft_convolve_full(a_mix).max()
        if c_mix < best_val:
            best_val = c_mix
            best_a = a_mix.copy()

    return best_a, best_val


def run_multi_trajectory(a_init, n, max_time, rng):
    """Run multiple optimization trajectories and keep the best."""
    S = np.sqrt(2.0 * n)
    a0 = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a0.copy()
    solutions = [a0]

    # Run 3 different trajectories
    n_trajs = 3
    t_per = max_time / n_trajs

    # Trajectory 1: Standard LSE-FISTA
    t1 = min(t_per, max_time - (time.monotonic() - start))
    a1, v1 = run_lse_fista(a0, n, t1, rng, beta_start=50, beta_end=30000, n_beta=60)
    if v1 < best:
        best = v1
        best_a = a1.copy()
    solutions.append(a1.copy())
    print("    Traj 1 (LSE): %.8f" % v1)

    # Trajectory 2: Mirror descent
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        t2 = min(t_per, remaining)
        a2, v2 = run_mirror_descent(a0, n, t2, rng, eta_start=0.08, eta_end=0.001)
        if v2 < best:
            best = v2
            best_a = a2.copy()
        solutions.append(a2.copy())
        print("    Traj 2 (Mirror): %.8f" % v2)

    # Trajectory 3: Adaptive subgradient
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        t3 = min(t_per, remaining)
        a3, v3 = run_adaptive_subgrad(a0, n, t3, rng, beta=150.0, decay=0.93)
        if v3 < best:
            best = v3
            best_a = a3.copy()
        solutions.append(a3.copy())
        print("    Traj 3 (Subgrad): %.8f" % v3)

    # Try convex combinations between the best solutions
    if len(solutions) >= 2:
        remaining = max_time - (time.monotonic() - start)
        if remaining > 10:
            for i in range(len(solutions)):
                for j in range(i + 1, len(solutions)):
                    if remaining < 5:
                        break
                    a_mix, v_mix = convex_combination_explore(
                        solutions[i], solutions[j], n, S, rng, n_trials=10)
                    if v_mix < best:
                        best = v_mix
                        best_a = a_mix.copy()
                        print("    Convex combo (%d,%d): %.8f" % (i, j, v_mix))
                    remaining = max_time - (time.monotonic() - start)

    return best_a, best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy search for non-negative sequence minimizing
    2*n*max(convolve(a,a))/sum(a)^2.
    Uses LSE-FISTA, mirror descent, adaptive subgradient, targeted
    gradient steps, and convex combination exploration.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 30.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))

    # Load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded sequence: length=%d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    # === Phase 1: Quick LSE on given ===
    if given is not None:
        n_g = len(given)
        t1 = min(45.0, (deadline - time.monotonic()) * 0.06)
        print("Phase 1: Quick LSE (n=%d, %.0fs)" % (n_g, t1))
        a1, v1 = run_lse_fista(given, n_g, t1, rng,
                                beta_start=100, beta_end=10000, n_beta=40,
                                use_backtrack=True)
        global_best = v1
        global_a = a1.copy()
        print("  Phase 1 result: %.8f" % global_best)

    # === Phase 2: Multi-trajectory exploration ===
    if global_a is not None:
        n_m = len(global_a)
        remaining = deadline - time.monotonic()
        t2 = remaining * 0.35
        print("Phase 2: Multi-trajectory (n=%d, %.0fs)" % (n_m, t2))
        a2, v2 = run_multi_trajectory(global_a, n_m, t2, rng)
        if v2 < global_best:
            global_best = v2
            global_a = a2.copy()
        print("  Phase 2 result: %.8f" % global_best)

    # === Phase 3: High-beta LSE-FISTA with backtracking ===
    if global_a is not None:
        n_h = len(global_a)
        remaining = deadline - time.monotonic()
        t3 = remaining * 0.25
        print("Phase 3: High-beta LSE + backtrack (n=%d, %.0fs)" % (n_h, t3))
        a3, v3 = run_lse_fista(global_a, n_h, t3, rng,
                                beta_start=5000, beta_end=200000, n_beta=50,
                                use_backtrack=True, momentum_decay=0.999)
        if v3 < global_best:
            global_best = v3
            global_a = a3.copy()
        print("  Phase 3 result: %.8f" % global_best)

    # === Phase 4: Targeted gradient steps ===
    if global_a is not None:
        n_t = len(global_a)
        S_t = np.sqrt(2.0 * n_t)
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t4 = remaining * 0.2
            print("Phase 4: Targeted gradient (%.0fs)" % t4)
            a4, v4 = targetable_gradient_step(global_a, n_t, S_t, t4, rng)
            if v4 < global_best:
                global_best = v4
                global_a = a4.copy()
            print("  Phase 4 result: %.8f" % global_best)

    # === Phase 5: Final mirror descent push ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t5 = remaining * 0.6
            print("Phase 5: Mirror descent final (%.0fs)" % t5)
            a5, v5 = run_mirror_descent(global_a, len(global_a), t5, rng,
                                         eta_start=0.03, eta_end=0.0003)
            if v5 < global_best:
                global_best = v5
                global_a = a5.copy()
            print("  Phase 5 result: %.8f" % global_best)

    # === Phase 6: Final ultra-high-beta LSE polish ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            t6 = remaining * 0.9
            print("Phase 6: Final polish (%.0fs)" % t6)
            a6, v6 = run_lse_fista(global_a, len(global_a), t6, rng,
                                     beta_start=50000, beta_end=500000, n_beta=20,
                                     use_backtrack=True)
            if v6 < global_best:
                global_best = v6
                global_a = a6.copy()
            print("  Phase 6 result: %.8f" % global_best)

    # === Final verification ===
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence=%.8f, internal=%.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            final_list = [float(x) for x in global_a]
    else:
        print("WARNING: No valid result, using uniform fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
