from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import numpy as np

def solve_lp_refinement(f, S, N, K=200, t=0.5):
    n = len(f)
    c = np.real(ifft(fft(f, N)**2))[:2*n-1]
    M = c.max()
    
    # Select top-K constraint indices
    top_k_idx = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix (sparse)
    rows = []
    cols = []
    vals = []
    for k in top_k_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n:
                rows.append(k)  # constraint index
                cols.append(j)  # variable index
                vals.append(f[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(K, n))
    b_ub = np.full(K, M)
    
    # Objective: max sum(b) => min -sum(b)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if not res.success:
        return f, M
    
    b = res.x
    b_sum = b.sum()
    if b_sum < 1e-10:
        return f, M
    
    # Rescale to match sum S
    b = b * (S / b_sum)
    
    # Blend
    f_new = (1 - t) * f + t * b
    f_new = np.maximum(f_new, 0)
    f_new = f_new * (S / f_new.sum())
    
    return f_new, M

"""
Minimize 2*n*max_conv/sum^2 using LP-based refinement combined with L-BFGS.

Key innovation: Use the linear program formulation to find optimal
refinement directions. Given current f with peak M, solve:
  max sum(b) s.t. b >= 0, (f*b)_k <= M for top-K constraint indices
Then blend f and rescaled b. This exploits the full convolution structure
and should escape local minima more effectively than subgradient methods.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

def lse_objective_and_grad(a, N, beta):
    """LSE-smoothed peak of conv(a,a) and its gradient."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
    else:
        weights = exp_bc / total
    lse_val = c_max + np.log(total) / beta if total > 0 else c_max
    
    fsize = N
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return lse_val, grad, c, c_max

def safe_normalize(a, n, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def spectral_partial_restart(a, N, rng, frac=0.3):
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)

def lbfgs_lse_block(a_init, n, S, N, max_time, rng, beta_init=5.0, beta_max=200.0, m_lbfgs=10):
    """L-BFGS on LSE-smoothed objective."""
    from scipy.fft import fft, ifft
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    
    f_val, g, c, c_max = lse_objective_and_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()
    
    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.15)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            f_val, g, c, c_max = lse_objective_and_grad(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
        
        if it % 30 == 0 and len(s_hist) >= m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]
        
        # L-BFGS direction
        direction = _lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        alpha_ls = step_scale
        a_new = a
        for ls in range(10):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max or ls == 9:
                a_new = candidate
                c_max_new = cm_ls
                if cm_ls <= c_max:
                    break
            alpha_ls *= 0.5
        
        if c_max_new < best_cmax:
            best_cmax = c_max_new
            best_a = a_new.copy()
        
        f_new, g_new, c_nw, cmax_nw = lse_objective_and_grad(a_new, N, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        
        a = a_new
        g = g_new
        c_max = cmax_nw
        
        if it % 300 == 0 and rng.random() < 0.2:
            a = spectral_partial_restart(a, N, rng, frac=0.3)
            s_hist = []
            y_hist = []
            f_val, g, c, c_max = lse_objective_and_grad(a, N, beta)
    
    return best_a, best_cmax

def _lbfgs_dir(s_hist, y_hist, g):
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction

def nesterov_toppk_block(a_init, n, S, N, max_time, rng, top_k_init=20, top_k_max=512):
    """Nesterov top-K subgradient."""
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0
    
    while time.monotonic() - start_time < max_time:

def lp_refinement_step(f, S, N, K, t_blend, max_lp_time=30):
    """
    One LP refinement step.
    Given f with sum S, solve LP to find b that maximizes sum(b)
    subject to (f*b)_k <= M for top-K constraint indices and b >= 0.
    Returns (f_new, success).
    """
    n = len(f)
    c = compute_conv(f, N)
    M = c.max()
    
    # Select top-K constraint indices
    top_k_idx = np.argsort(c)[::-1][:K]
    
    # Build sparse constraint matrix
    # Constraint k: sum_i f_i * b_{k-i} <= M (valid indices)
    # Variable j (b_j): appears in constraint k with coefficient f_{k-j}
    
    nnz_estimate = K * min(n, K)  # rough estimate
    rows_list = []
    cols_list = []
    vals_list = []
    
    for ci, k in enumerate(top_k_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n and f[i] > 1e-15:
                rows_list.append(ci)
                cols_list.append(j)
                vals_list.append(f[i])
    
    if not rows_list:
        return f, M, False
    
    A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(len(top_k_idx), n))
    b_ub = np.full(len(top_k_idx), M)
    
    # Objective: max sum(b) => min -sum(b)
    c_obj = -np.ones(n)
    
    start_lp = time.monotonic()
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                  bounds=[(0, 1000.0)] * n,
                  method='highs',
                  options={'time_limit': max_lp_time, 'presolve': True})
    
    if time.monotonic() - start_lp > max_lp_time:
        return f, M, False
    
    if not res.success or res.x is None:
        return f, M, False
    
    b = res.x
    b_sum = b.sum()
    if b_sum < 1e-10:
        return f, M, False
    
    # Rescale to match sum S
    b = b * (S / b_sum)
    
    # Blend
    f_new = (1 - t_blend) * f + t_blend * b
    f_new = np.maximum(f_new, 0.0)
    s_new = f_new.sum()
    if s_new < 1e-15:
        return f, M, False
    f_new = f_new * (S / s_new)
    
    return f_new, M, True

def deflation_init(n, S, N, n_iters=1500, rng=None):
    if rng is None:
        rng = np.random.default_rng(123)
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c = compute_conv(a, N)
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
        if np.max(contrib) < 1e-15:
            break
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if i_max == i_min:
            break
        transfer = min(0.02 * a[i_max], 0.3 * a[i_max])
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def make_init_rand(n, rng):
    return rng.random(n) + 0.05

def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_init_zadoff(n, rng):
    L = n
    i = np.arange(L)
    c = rng.integers(1, L)
    phase = -np.pi * i * (2 * i + 1 - c) / L
    a = 1.0 + 0.7 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)

def make_init_fourier(n, rng, N):
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01

"""
Minimize 2*n*max_conv/sum^2 using LP-guided refinement combined with L-BFGS.

Key innovation: Solve the linear program
  max sum(b) s.t. b >= 0, (f*b)_k <= M for top-K constraint indices
to find a "complementary" refinement direction. This directly exploits the
convolution structure, providing a principled direction that subgradient
methods approximate only crudely. Combined with L-BFGS on LSE for smooth
optimization, adaptive spectral restarts, and Nesterov polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_objective_and_grad(a, N, beta):
    """LSE-smoothed peak of conv(a,a) and its gradient."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    if beta < 1e-6:
        weights = np.ones(len(c)) / len(c)
        lse_val = c_max
    else:
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:
            weights = exp_bc / total
            lse_val = c_max + np.log(total) / beta
    W = fft(weights, N)
    Af = fft(a, N)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return lse_val, grad, c, c_max


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def spectral_partial_restart(a, N, rng, frac=0.3):
    """Randomize a fraction of FFT phases, project to non-negative."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def spectral_full_restart(a, N, rng):
    """Fully randomize FFT phases, project to non-negative."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    phase = rng.uniform(0, 2 * np.pi, N)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def lbfgs_solve_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def lbfgs_lse_block(a_init, n, S, N, max_time, rng, beta_init=5.0, beta_max=200.0, m_lbfgs=10):
    """L-BFGS on LSE-smoothed objective with adaptive beta."""
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    f_val, g, c, c_max = lse_objective_and_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        if it % 100 == 0:
            beta = min(beta_max, beta * 1.12)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            f_val, g, c, c_max = lse_objective_and_grad(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

        if it % 30 == 0 and len(s_hist) >= m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_solve_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        alpha_ls = step_scale
        a_new = a
        c_max_new = c_max
        for ls in range(10):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max or ls == 9:
                a_new = candidate
                c_max_new = cm_ls
                if cm_ls <= c_max:
                    break
            alpha_ls *= 0.5

        if c_max_new < best_cmax:
            best_cmax = c_max_new
            best_a = a_new.copy()

        f_new, g_new, c_nw, cmax_nw = lse_objective_and_grad(a_new, N, beta)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c_max = cmax_nw

        if it % 250 == 0 and rng.random() < 0.2:
            a = spectral_partial_restart(a, N, rng, frac=0.25)
            s_hist = []
            y_hist = []
            f_val, g, c, c_max = lse_objective_and_grad(a, N, beta)

    return best_a, best_cmax


def nesterov_toppk_block(a_init, n, S, N, max_time, rng, top_k_init=20, top_k_max=512):
    """Nesterov top-K subgradient for polishing."""
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 60)
        c = compute_conv(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        w_full = np.zeros(N)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        W = fft(w_full, N)
        Af = fft(y, N)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        if it % 15 == 0:
            c_x = compute_conv(x_new, N)
            if c_x.max() < best_cmax:
                best_cmax = c_x.max()
                best_a = x_new.copy()
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        if it % 1500 == 0 and rng.random() < 0.12:
            y = spectral_partial_restart(y, N, rng, frac=0.2)
            t = 1.0
    return best_a, best_cmax


def solve_lp_complement(f, S, N, K, max_time_lp=15.0):
    """
    Solve LP: max sum(b) s.t. b >= 0, (f*b)_k <= M for top-K indices.
    Returns (b_opt, M, success).
    """
    n = len(f)
    c = compute_conv(f, N)
    M = c.max()

    top_k_idx = np.argsort(c)[::-1][:K]

    rows_list = []
    cols_list = []
    vals_list = []

    for ci, k in enumerate(top_k_idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n:
                fv = f[i]
                if fv > 1e-12:
                    rows_list.append(ci)
                    cols_list.append(j)
                    vals_list.append(fv)

    if not rows_list:
        return None, M, False

    A = csr_matrix((vals_list, (rows_list, cols_list)), shape=(K, n))
    b_ub = np.full(K, M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, 1000.0)] * n

    t0 = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': max_time_lp, 'presolve': True})
    except Exception:
        return None, M, False

    if time.monotonic() - t0 > max_time_lp + 5:
        return None, M, False

    if not res.success or res.x is None:
        return None, M, False

    return res.x, M, True


def lp_refinement_step(f, S, N, K, t_blend, max_time_lp=15.0):
    """
    One LP refinement step: find complement b, blend with f.
    Returns (f_new, M, success, improved).
    """
    n = len(f)
    b_opt, M, success = solve_lp_complement(f, S, N, K, max_time_lp)
    if not success:
        return f, M, False, False

    b_sum = b_opt.sum()
    if b_sum < 1e-10:
        return f, M, False, False

    b_scaled = b_opt * (S / b_sum)

    # Try multiple blend parameters, pick best
    best_f = f.copy()
    best_peak = M
    for t in [t_blend, t_blend * 0.5, t_blend * 1.5, 0.1, 0.3]:
        t = min(0.8, t)
        f_try = (1.0 - t) * f + t * b_scaled
        f_try = np.maximum(f_try, 0.0)
        s_try = f_try.sum()
        if s_try < 1e-15:
            continue
        f_try = f_try * (S / s_try)
        c_try = compute_conv(f_try, N)
        peak_try = c_try.max()
        if peak_try < best_peak:
            best_peak = peak_try
            best_f = f_try.copy()

    improved = best_peak < M - 1e-10
    return best_f, M, True, improved


def lp_refinement_block(f_init, n, S, N, max_time, rng, K_init=60, K_max=400,

        else:
            stall += 1

        K = min(K_max, K_init + it * 3)
        t_blend = max(t_min, t_init * (0.98 ** it))

        f_new, M, success, improved = lp_refinement_step(f, S, N, K, t_blend, max_lp_time)

        if not success:

            if K > 30:

            if not success:

            else:
                lp_fail_count += 1

        if success:
            lp_success_count += 1
            if improved:
                f = f_new
            else:
                # LP didn't help; try smaller blend

def deflation_init(n, S, N, n_iters=1200, rng=None):
    """Constructive initialization by iteratively reducing the peak from all-ones."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c = compute_conv(a, N)
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
        if np.max(contrib) < 1e-15:
            break
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if i_max == i_min:
            break
        transfer = min(0.02 * a[i_max], 0.3 * a[i_max])
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    L = n
    i = np.arange(L)
    c = rng.integers(1, L)
    phase = -np.pi * i * (2 * i + 1 - c) / L
    a = 1.0 + 0.7 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_fourier(n, rng, N):
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_windowed(n, rng, N):
    """Windowed initialization with various window shapes."""
    tt = np.linspace(0, 1, n, endpoint=False)
    shape = rng.integers(0, 3)
    if shape == 0:
        a = np.cos(np.pi * tt) ** 2
    elif shape == 1:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    else:
        a = np.exp(-4 * (tt - 0.5) ** 2)
    a += 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses LP-guided refinement (novel) combined with L-BFGS on LSE,
    Nesterov acceleration, and spectral phase restarts.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 35.0)

    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and given.size <= 32768:
            n = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    def update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = safe_normalize(a_candidate, n, S)
        c = compute_conv(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            score = float(2 * n * sc / (S * S))
            if label:
                print("  [%s] peak=%.6f score=%.6f" % (label, sc, score))
        return sc

    # --- Phase 1: Multi-start exploration ---
    print("=== Phase 1: Multi-start L-BFGS exploration ===")
    t1 = time.monotonic()
    phase1_budget = min(450.0, (deadline - t1) * 0.18)

    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
    except Exception:
        pass
    inits.append(("rand", make_init_rand(n, rng)))
    inits.append(("chirp", make_init_chirp(n, rng)))
    inits.append(("zadoff", make_init_zadoff(n, rng)))
    inits.append(("fourier", make_init_fourier(n, rng, N)))
    inits.append(("window", make_init_windowed(n, rng, N)))
    inits.append(("deflation", deflation_init(n, S, N, 1000, rng)))

    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 500:
            break
        a_norm = safe_normalize(arr, n, S)
        t0 = time.monotonic()
        rem = max(1.0, per_init - (time.monotonic() - t0) - 0.5)
        a_l, sc_l = lbfgs_lse_block(a_norm, n, S, N, min(rem, 70.0), rng,
                                     beta_init=5.0, beta_max=100.0, m_lbfgs=10)
        update_best(a_l, name + "/LBFGS")

    print("Phase 1 done. Best score: %.6f" % (2 * n * global_best / (S * S) if global_best < np.inf else np.inf))

    # --- Phase 2: LP-based refinement (KEY NOVELTY) ---
    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = 2.0

    print("=== Phase 2: LP-based refinement ===")
    t2 = time.monotonic()
    phase2_budget = min(1100.0, (deadline - t2) * 0.48)

    a_curr = global_a.copy()
    f_lp, sc_lp = lp_refinement_block(a_curr, n, S, N, phase2_budget, rng,
                                       K_init=50, K_max=350,
                                       t_init=0.3, t_min=0.04, max_lp_time=12.0)
    update_best(f_lp, "LP_block")

    if global_a is None:
        global_a = a_curr.copy()
        global_best = 2.0

    # --- Phase 3: Focused L-BFGS with spectral restarts ---
    print("=== Phase 3: Focused L-BFGS refinement ===")
    t3 = time.monotonic()
    phase3_budget = min(350.0, (deadline - t3) * 0.18)

    a_curr = global_a.copy()
    stall = 0
    while time.monotonic() < t3 + phase3_budget:
        block_time = min(100.0, (t3 + phase3_budget) - time.monotonic())
        if block_time < 10:
            break
        beta_start = 10.0 + 4.0 * min(stall, 8)
        beta_end = min(500.0, 50.0 * (1 + 0.5 * stall))
        a_new, sc_new = lbfgs_lse_block(a_curr, n, S, N, block_time, rng,
                                         beta_init=beta_start, beta_max=beta_end, m_lbfgs=12)
        if sc_new < global_best - 1e-9:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            print("  [P3] score=%.6f" % (2 * n * sc_new / (S * S)))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall == 2:
                a_curr = spectral_partial_restart(a_curr, N, rng, frac=0.35)
                stall = 0
            elif stall >= 4:
                a_curr = spectral_full_restart(a_curr, N, rng)
                a_curr = safe_normalize(a_curr, n, S)
                # Re-optimize from scratch-ish
                a_reshot, sc_reshot = lbfgs_lse_block(a_curr, n, S, N, min(30, block_time * 0.5),
                                                       rng, beta_init=5.0, beta_max=50.0, m_lbfgs=8)
                a_curr = a_reshot
                if sc_reshot < global_best:
                    global_best = sc_reshot
                    global_a = a_curr.copy()
                stall = 0

    # --- Phase 4: Nesterov + final LP polish ---
    print("=== Phase 4: Final polishing ===")
    t4 = time.monotonic()
    remaining4 = deadline - t4 - 25
    if remaining4 > 40:
        # Nesterov
        nest_time = min(remaining4 * 0.4, 150.0)
        if nest_time > 20:
            a_ng, sc_ng = nesterov_toppk_block(global_a, n, S, N, nest_time, rng,
                                                top_k_init=30, top_k_max=400)
            update_best(a_ng, "nesterov")

        # Short LP block
        remaining4b = deadline - time.monotonic() - 20
        if remaining4b > 60:
            f_lp2, sc_lp2 = lp_refinement_block(global_a, n, S, N, min(remaining4b * 0.5, 200.0),
                                                 rng, K_init=100, K_max=400,
                                                 t_init=0.2, t_min=0.03, max_lp_time=10.0)
            update_best(f_lp2, "LP_block2")

        # Final high-beta L-BFGS
        remaining4c = deadline - time.monotonic() - 15
        if remaining4c > 20:
            a_fin, sc_fin = lbfgs_lse_block(global_a, n, S, N, min(remaining4c, 200.0),
                                             rng, beta_init=40.0, beta_max=800.0, m_lbfgs=15)
            update_best(a_fin, "final_LBFGS")

    # --- Final Nesterov squeeze ---
    remaining_fin = deadline - time.monotonic() - 10
    if remaining_fin > 20:
        a_s, sc_s = nesterov_toppk_block(global_a, n, S, N, min(remaining_fin, 100.0),
                                          rng, top_k_init=50, top_k_max=500)
        update_best(a_s, "final_nest")

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("Verification: inf - using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    final_score = 2.0 * len(global_a) * global_best / (S * S) if global_best < np.inf else float('inf')
    print("Returning length %d, internal peak=%.6f, approx score=%.6f" % (len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
