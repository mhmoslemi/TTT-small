from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def conv_peak_grad(a, N, top_k):
    """Compute convolution, find top-k peaks, and return weighted subgradient."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2*n-1]
    
    # Find top-k peaks
    k = min(top_k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx]
    
    # Weight by excess (how much above the median)
    med = np.median(c)
    excess = np.maximum(0, top_vals - med)
    total_excess = excess.sum()
    if total_excess < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = excess / total_excess
    
    # Compute gradient: d(c[k])/d(a[j]) = a[k-j] + a[j-k] (for valid indices)
    # Using FFT: grad = 2 * real(ifft(C * conj(A))) where C is the weighted convolution
    fsize = N  # Use same FFT size
    C = np.zeros(N, dtype=np.complex128)
    for i, idx in enumerate(top_idx):
        C[int(idx)] = weights[i]
    C_f = fft(C)
    grad = 2.0 * np.real(ifft(C_f * np.conj(A)))[:n]
    
    c_max = c.max()
    return grad, c_max, c, top_idx, weights

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_conv(a, N):
    """Compute linear self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def topk_subgradient(a, N, top_k):
    """
    Compute weighted subgradient from top-K convolution peaks.
    Returns (grad, c_max, c).
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()

    k = min(top_k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx]

    # Weight by excess above median
    med = np.median(c)
    excess = np.maximum(0.0, top_vals - med)
    total_excess = excess.sum()
    if total_excess < 1e-30:
        excess = np.ones(k)
        total_excess = k
    weights = excess / total_excess

    # Build weighted impulse train and FFT
    W = np.zeros(N)
    for i, idx in enumerate(top_idx):
        W[int(idx)] = weights[i]
    W_f = fft(W)
    A_conj = np.conj(A)
    grad = 2.0 * np.real(ifft(W_f * A_conj))[:n]

    return grad, c_max, c


def lse_subgradient(a, N, beta):
    """LSE-smoothed subgradient. Returns (grad, lse_val, c_max)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()

    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
    else:
        weights = exp_bc / total

    W_f = fft(weights, N)
    A_conj = np.conj(A)
    grad = 2.0 * np.real(ifft(W_f * A_conj))[:n]

    lse_val = c_max + np.log(total) / beta
    return grad, lse_val, c_max


def pair_swap_local_search(a, n, S, N, max_iters=500):
    """
    Local search: for the peak lag, find the worst contributing pair and try to
    reduce it by transferring mass to a less contributing position.
    """
    best_a = a.copy()
    c = compute_conv(a, N)
    best_peak = c.max()

    for it in range(max_iters):
        k_star = int(np.argmax(c))
        # Find contributing pairs
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[i] * a[j]

        i_max = int(np.argmax(contrib))
        if contrib[i_max] < 1e-15:
            break

        # Try transferring small amount from i_max to various other positions
        j_partner = k_star - i_max
        a_i = a[i_max]
        a_j = a[j_partner] if 0 <= j_partner < n else 0.0

        # Find a "safe" destination: a position that contributes little to the peak
        # and where adding mass won't create a new peak
        best_dest = -1
        best_gain = 0.0
        # Try a few random destinations
        for _ in range(20):
            d = int(np.random.randint(n))
            if d == i_max or d == j_partner:
                continue
            # Estimate gain: reducing a[i_max] by delta reduces c[k_star] by delta*a[j_partner]
            # (and also affects c at other lags involving position i_max)
            # Adding to position d affects lags d+i for all i
            # Simple heuristic: prefer positions with small current value
            if a[d] < a[i_max] * 0.5:
                gain = a[j_partner]  # per unit transferred
                if gain > best_gain:
                    best_gain = gain
                    best_dest = d

        if best_dest < 0:
            break

        # Try a few step sizes
        improved = False
        for frac in [0.05, 0.02, 0.01, 0.005]:
            delta = frac * a[i_max]
            if delta < 1e-15:
                break
            a_trial = a.copy()
            a_trial[i_max] -= delta
            a_trial[best_dest] += delta
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = safe_normalize(a_trial, n, S)
            c_trial = compute_conv(a_trial, N)
            if c_trial.max() < best_peak - 1e-12:
                a = a_trial
                best_a = a.copy()
                best_peak = c_trial.max()
                c = c_trial
                improved = True
                break

        if not improved:
            break

    return best_a, best_peak


def jolt_perturb(a, n, S, rng, strength=0.3):
    """Random time-domain perturbation to escape local minima."""
    a_new = a.copy()
    n_perturb = max(1, int(n * 0.05))
    idx = rng.choice(n, size=n_perturb, replace=False)
    noise = rng.standard_normal(n_perturb) * strength * (S / n)
    a_new[idx] += noise
    a_new = np.maximum(a_new, 0.0)
    a_new = safe_normalize(a_new, n, S)
    return a_new


def spectral_jolt(a, n, S, N, rng, frac=0.3):
    """Spectral phase randomization on a fraction of bins."""
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    a_new = safe_normalize(a_new, n, S)
    return a_new

    """
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    c = compute_conv(a, N)
    best_peak = c.max()
    momentum_vec = np.zeros(n)
    stall = 0
    t = 0
    start_time = time.monotonic()

    while time.monotonic() - start_time < max_time:
        # Adaptive top-K
        top_k = min(topk_max, topk_init + t // 200)

        # Compute subgradient
        if t % 1000 == 0:
            # Occasionally use LSE for smoother gradient

        # Momentum (heavy ball)
        if use_momentum:
            mom = mom_init * (1.0 - 0.01 * min(stall, 20))
            step_dir = grad + mom * momentum_vec
            momentum_vec = step_dir
        else:
            step_dir = grad

        # Step size with decay
        step = step_scale * 2.0 * S / (n * (1.0 + 0.005 * t))

        # Backtracking line search
        a_new = a
        cmax_new = c.max()
        for ls in range(10):
            candidate = project_simplex(a - step * step_dir, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= cmax_new * (1.0 - 1e-8) or ls == 9:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= cmax_new:
                    break
            step *= 0.5

        a = a_new
        if cmax_new < best_peak:
            best_peak = cmax_new
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Diversity restarts

            else:

            stall = 0

        if t % 5000 == 0:

    tt = np.linspace(0, 1, n, endpoint=False)
    a = np.ones(n)
    n_freqs = int(rng.integers(3, 8))
    for _ in range(n_freqs):
        freq = float(rng.integers(2, 50))
        amp = float(rng.uniform(0.05, 0.2))
        phase = float(rng.uniform(0, 2 * np.pi))
        a += amp * np.sin(2 * np.pi * freq * tt + phase)
    return np.clip(a, 0.001, None)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses adaptive top-K projected gradient with momentum, diverse restarts,
    pair-swap local search, and multi-scale refinement.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    # Try to use the provided sequence as warm start
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_best(a_cand, n_cand, label=""):
        nonlocal global_best, global_a
        S_cand = np.sqrt(2.0 * float(n_cand))
        a_norm = safe_normalize(a_cand, n_cand, S_cand)
        N_cand = 1 << (2 * n_cand - 1).bit_length()
        c = compute_conv(a_norm, N_cand)
        peak = c.max()
        score = 2.0 * n_cand * peak / (S_cand * S_cand)
        # Actually with S = sqrt(2n), score = peak
        # But let's compute it properly
        sum_a = a_norm.sum()
        if sum_a < 0.01:
            return np.inf
        score = 2.0 * n_cand * peak / (sum_a * sum_a)
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
            print("  [%s] n=%d score=%.8f" % (label, n_cand, score))
        return score

    # ============================================================
    # Phase 1: Warm start from given sequence (if available)
    # ============================================================
    if given is not None and len(given) >= 1024:
        print("=== Phase 1: Warm start from given sequence (n=%d) ===" % len(given))
        n1 = len(given)
        S1 = np.sqrt(2.0 * n1)
        N1 = 1 << (2 * n1 - 1).bit_length()
        a1 = safe_normalize(given, n1, S1)

        # Quick polish with high top-K
        t0 = time.monotonic()
        polish_time = min(300.0, (deadline - t0) * 0.15)
        a1_best, peak1 = optimize_sequence(a1, n1, S1, N1, polish_time, rng,
                                            use_momentum=False,
                                            topk_init=50, topk_max=512,
                                            step_scale=0.3, jolt_interval=500)
        update_best(a1_best, n1, "warmstart_polish")

        # Pair-swap local search
        t0 = time.monotonic()
        if time.monotonic() < deadline - 300:
            a1_ps, peak_ps = pair_swap_local_search(a1_best, n1, S1, N1, max_iters=2000)
            update_best(a1_ps, n1, "warmstart_pairswap")

    # ============================================================
    # Phase 2: Fresh optimization at n=8192
    # ============================================================
    n2 = 8192
    S2 = np.sqrt(2.0 * n2)
    N2 = 1 << (2 * n2 - 1).bit_length()

    print("=== Phase 2: Fresh optimization at n=%d ===" % n2)
    t2 = time.monotonic()
    phase2_budget = min(800.0, (deadline - t2) * 0.35)

    # Try multiple initializations
    inits = [
        ("random", make_init_random(n2, rng)),
        ("chirp", make_init_chirp(n2, rng)),
        ("zadoff", make_init_zadoff(n2, rng)),
        ("multi_sine", make_init_multi_sine(n2, rng)),
    ]
    if given is not None and len(given) == n2:
        inits.insert(0, ("given_n8192", given.copy()))

    per_init = phase2_budget / len(inits)
    print("  %d inits, ~%.1f sec each" % (len(inits), per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break
        a_norm = safe_normalize(arr, n2, S2)
        t0 = time.monotonic()
        rem = max(5.0, per_init - (time.monotonic() - t0) - 2.0)
        a_best, peak = optimize_sequence(a_norm, n2, S2, N2, min(rem, 200.0), rng,
                                          use_momentum=True, mom_init=0.7,
                                          topk_init=5, topk_max=256,
                                          step_scale=1.0, jolt_interval=200)
        update_best(a_best, n2, name)

    # ============================================================
    # Phase 3: Focused refinement of the best solution
    # ============================================================
    print("=== Phase 3: Focused refinement ===")
    if global_a is not None:
        n3 = len(global_a)
        S3 = np.sqrt(2.0 * n3)
        N3 = 1 << (2 * n3 - 1).bit_length()
        t3 = time.monotonic()
        phase3_budget = min(600.0, (deadline - t3) * 0.3)

        # Multiple refinement passes with different parameters
        a_curr = global_a.copy()
        for pass_i in range(3):
            if time.monotonic() > deadline - 100:
                break
            block_time = min(200.0, (deadline - 100.0) - time.monotonic())
            if block_time < 20:
                break

            if pass_i == 0:
                a_curr, pk = optimize_sequence(a_curr, n3, S3, N3, block_time, rng,
                                               use_momentum=True, mom_init=0.8,
                                               topk_init=20, topk_max=512,
                                               step_scale=0.5, jolt_interval=400)
            elif pass_i == 1:
                # No momentum, very small steps for fine tuning
                a_curr, pk = optimize_sequence(a_curr, n3, S3, N3, block_time, rng,
                                               use_momentum=False,
                                               topk_init=100, topk_max=1024,
                                               step_scale=0.2, jolt_interval=800)
            else:
                # LSE-based polish
                a_curr, pk = optimize_sequence(a_curr, n3, S3, N3, block_time, rng,
                                               use_momentum=True, mom_init=0.5,
                                               topk_init=50, topk_max=512,
                                               step_scale=0.3, jolt_interval=600)
            update_best(a_curr, n3, "refine_pass_%d" % pass_i)

        # Pair-swap on the best
        if time.monotonic() < deadline - 60:
            a_ps, pk = pair_swap_local_search(global_a, n3, S3, N3, max_iters=5000)
            update_best(a_ps, n3, "pairswap_final")

    # ============================================================
    # Phase 4: Try a different n (coarser or finer)
    # ============================================================
    print("=== Phase 4: Alternative n exploration ===")
    t4 = time.monotonic()
    if time.monotonic() < deadline - 120:
        for n_alt in [4096, 16384]:
            if time.monotonic() > deadline - 100:
                break
            S_alt = np.sqrt(2.0 * n_alt)
            N_alt = 1 << (2 * n_alt - 1).bit_length()
            a_alt = make_init_multi_sine(n_alt, rng)
            a_alt = safe_normalize(a_alt, n_alt, S_alt)
            block_time = min(150.0, (deadline - 100.0) - time.monotonic())
            if block_time < 30:
                break
            a_alt_best, pk = optimize_sequence(a_alt, n_alt, S_alt, N_alt, block_time, rng,
                                               topk_init=10, topk_max=256,
                                               step_scale=1.0, jolt_interval=200)
            update_best(a_alt_best, n_alt, "n%d" % n_alt)

    # ============================================================
    # Phase 5: Final intensive polishing
    # ============================================================
    print("=== Phase 5: Final polishing ===")
    if global_a is not None:
        n5 = len(global_a)
        S5 = np.sqrt(2.0 * n5)
        N5 = 1 << (2 * n5 - 1).bit_length()
        remaining = deadline - time.monotonic() - 15
        if remaining > 30:
            a_final, pk = optimize_sequence(global_a, n5, S5, N5, min(remaining, 200.0), rng,
                                            use_momentum=False,
                                            topk_init=200, topk_max=2048,
                                            step_scale=0.1, jolt_interval=1000)
            update_best(a_final, n5, "final_polish")

    # ============================================================
    # Verification
    # ============================================================
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: score = %.8f (length=%d)" % (ver, len(global_a)))
            global_best = ver
        else:
            print("Verification failed, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        print("No valid solution found, using uniform fallback")
        global_a = np.ones(256, dtype=np.float64)
        global_best = 2.0

    final_list = [float(x) for x in global_a]
    print("Returning length %d, best score=%.8f" % (len(global_a), global_best))
    return final_list
