"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Strategy: LP-based constraint generation (active set) + FISTA polishing + L-BFGS.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize
from scipy.sparse import csr_matrix
import cvxpy as cp

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def convolve_f_g(f, g):
    """Compute conv(f, g) via FFT."""
    n = len(f)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    F = fft(f, fsize)
    G = fft(g, fsize)
    c = np.real(ifft(F * G))
    return c[:size]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_weights(c, beta):
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot

def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross

def run_lp_active_set(f_init, max_time, rng, K=200, alpha=1.0):
    """
    LP-based active set method.
    Iteratively solves: max sum(g) s.t. g>=0, conv(f,g)_k <= M for active k.
    """
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f = np.clip(np.asarray(f_init, dtype=np.float64), 0, 1000).copy()
    f = f * (S / f.sum())
    
    start = time.monotonic()
    c = fft_convolve_self(f)
    M = c.max()
    best_M = M
    best_f = f.copy()
    it = 0
    
    # Active set: top K indices
    active_set = np.argsort(c)[::-1][:K]
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        
        # Solve LP: max sum(g) s.t. g >= 0, conv(f,g)[k] <= M for k in active_set
        # conv(f,g)[k] = sum_j f[k-j]*g[j] (with valid indices)
        # Build constraint matrix B (K x n)
        B_rows = []
        for k in active_set:
            row = np.zeros(n)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                row[j] = f[k - j]
            B_rows.append(row)
        B = np.array(B_rows)
        
        # LP: max sum(g) s.t. B@g <= M, g >= 0
        g = cp.Variable(n)
        obj = cp.Maximize(cp.sum(g))
        cons = [B @ g <= M, g >= 0]
        prob = cp.Problem(obj, cons)
        
        t0 = time.monotonic()
        try:
            prob.solve(solver=cp.ECOS)  # or CBC, GLOP
        except Exception as e:
            print("  LP failed: %s" % str(e))
            break
        
        lp_time = time.monotonic() - t0
        sum_g = g.value.sum() if g.value is not None else 0
        
        if g.value is None or sum_g <= S * (1 + 1e-10):
            # Can't improve, try different active set or break
            if it > 5:
                # Try expanding active set
                c_new = fft_convolve_self(f)
                M_new = c_new.max()
                active_set = np.argsort(c_new)[::-1][:min(K * 2, 2 * n - 1)]
                M = M_new
                continue
            else:
                break
        
        # Rescale g to sum S
        g_new = g.value * (S / sum_g)
        g_new = np.maximum(g_new, 0)
        
        # Mix
        f_new = (1 - alpha) * f + alpha * g_new
        f_new = np.maximum(f_new, 0)
        f_new = f_new * (S / f_new.sum())
        
        # Check new objective
        c_new = fft_convolve_self(f_new)
        M_new = c_new.max()
        
        if M_new < best_M:
            best_M = M_new
            best_f = f_new.copy()
        
        # Also check: does f_new * f_new violate any inactive constraints?
        # (This is just the new max, which we already computed)
        
        f = f_new
        M = M_new
        
        # Update active set periodically
        if it % 10 == 0:
            c_cur = fft_convolve_self(f)
            active_set = np.argsort(c_cur)[::-1][:K]
            M = c_cur.max()
        
        if it % 100 == 0:

                  (it, M, best_M, sum_g/S, lp_time))
    
    return best_f, best_M

def fft_convolve_self(a):

    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    n = len(v)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y_mom if use_momentum else f)
        M_cur = c.max()

        if M_cur < best_M - 1e-12:
            best_M = M_cur
            best_f = (y_mom if use_momentum else f).copy()
            stuck = 0
        else:
            stuck += 1

        # Active set: constraints near the max
        frac = min(0.10, 0.01 + 0.005 * (stuck / 500.0))
        threshold = M_cur * (1.0 - frac)
        active_mask = c >= threshold
        active_indices = np.where(active_mask)[0]
        if len(active_indices) > K:
            active_indices = active_indices[np.argsort(c[active_indices])[::-1][:K]]
        if len(active_indices) == 0:
            active_indices = np.array([int(np.argmax(c))])

        # Build B and solve dual LP
        f_work = np.asarray(f, dtype=np.float64)
        B = build_B(f_work, active_indices, n)
        BT = B.T

        y_vals = solve_dual_lp(BT, len(active_indices))

        if y_vals is not None and np.isfinite(y_vals).all() and y_vals.sum() > 1e-10:
            grad = compute_lp_subgrad(f_work, active_indices, y_vals, n)
            lp_fail_count = 0
        else:
            # Fallback: top-1 subgradient
            lp_fail_count += 1
            k_max = int(np.argmax(c))
            j_start = max(0, k_max - n + 1)
            j_end = min(k_max, n - 1)
            js = np.arange(j_start, j_end + 1)
            grad = 2.0 * np.zeros(n)
            grad[js] = 2.0 * f_work[k_max - js]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking step size
        step = min(step * 1.05, S / gnorm)
        for _ in range(20):
            f_trial = project_simplex(f - step * grad, S)
            c_trial = fft_convolve_self(f_trial)
            if c_trial.max() < M_cur + 1e-10:
                break
            step *= 0.5

        f_new = project_simplex(f - step * grad, S)
        c_new = fft_convolve_self(f_new)
        M_new = c_new.max()

        if M_new < M_cur:
            # FISTA momentum update
            if use_momentum:
                t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
                y_mom = f_new + ((t_mom - 1.0) / t_new) * (f_new - prev_f)
                y_mom = np.maximum(y_mom, 0.0)
                y_mom = y_mom * (S / max(y_mom.sum(), 1e-15))
                t_mom = t_new
            prev_f = f
            f = f_new
            M = M_new

            if it % 500 == 0:
                if use_momentum:
                    t_mom = 1.0
        else:
            # Reduce step, stay in place
            step *= 0.7
            if stuck > 5000:
                # Random restart perturbation
                noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
                f = project_simplex(f + noise, S)
                if use_momentum:
                    y_mom = f.copy()
                    t_mom = 1.0
                stuck = 0

        if it % 2000 == 0:
            print("    LP-SG it=%d, M=%.8f, best=%.8f, step=%.4e, stuck=%d, frac=%.3f" %
                  (it, M, best_M, step, stuck, frac))

        # Periodic momentum reset
        if use_momentum and it % 3000 == 0:
            t_mom = 1.0
            y_mom = f.copy()

    return best_f, best_M


def run_fista(f_init, max_time, rng, beta_start=20.0, mode="lse"):
    """Standard FISTA with LSE or top-k gradient."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f = np.clip(np.asarray(f_init, dtype=np.float64), 0, 1000).copy()
    f = f * (S / f.sum())
    y = f.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_M = np.inf
    best_f = f.copy()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = beta_start * (1.0 + 0.002 * it)
        beta = min(beta, 100000.0)
        step = 1.0 / max(beta, 1.0)

        c = fft_convolve_self(y)
        M_cur = c.max()
        if M_cur < best_M - 1e-12:
            best_M = M_cur
            best_f = y.copy()
            stuck = 0
        else:
            stuck += 1

        if mode == "lse":
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
        elif mode == "topk":
            k = max(4, min(128, n // 16))
            idx = np.argsort(c)[::-1][:k]
            vals = c[idx]
            w = np.zeros(len(c))
            w[idx] = (vals ** 1.5)
            wsum = w.sum()
            if wsum > 0:
                w /= wsum
            else:
                w[int(np.argmax(c))] = 1.0
            grad = conv_gradient(y, w)
        else:
            # Hybrid: use exact when stuck
            if stuck > 1000:
                k_max = int(np.argmax(c))
                j_start = max(0, k_max - n + 1)
                j_end = min(k_max, n - 1)
                js = np.arange(j_start, j_end + 1)
                grad = 2.0 * np.zeros(n)
                grad[js] = 2.0 * y[k_max - js]
            else:
                w = lse_weights(c, beta)
                grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking
        for _ in range(15):
            x_new = project_simplex(y - step * grad, S)
            if x_new.sum() > 1e-10:
                break
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0.0)
        f = x_new
        t_mom = t_new

        if it % 5000 == 0:
            t_mom = 1.0

        # Restart if stuck
        if stuck > 3000 and it % 200 == 0:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            stuck = 0

    return best_f, best_M


def run_lbfgs(f_init, max_time, rng, n_levels=4):
    """L-BFGS-B with LSE objective and penalty for sum constraint."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(f_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    betas = [30.0, 80.0, 200.0, 500.0][:n_levels]

    for beta in betas:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        def obj_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / beta
            penalty = 10.0 * (x.sum() - S) ** 2
            obj += penalty
            grad = conv_gradient(x, w)
            grad += 20.0 * (x.sum() - S)
            return obj, grad

        result = scipy_minimize(
            obj_grad, a.copy(), method='L-BFGS-B', jac=True,
            bounds=[(0, 1000)] * n,
            options={'maxiter': 80, 'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1}
        )
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_coordinate_descent(f_init, max_time, rng):
    """Coordinate descent: optimize one element at a time."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(f_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    c = fft_convolve_self(a)
    M = c.max()
    best_M = M
    best_a = a.copy()
    order = rng.permutation(n)
    pos = 0

    while time.monotonic() - start < max_time:
        i = order[pos % n]
        pos += 1

        # Optimize a[i] by golden section search
        # The objective as a function of a[i] involves c[k] for k where i participates
        # c[k] = a[i]*a[k-i] + (rest) for k where both i and k-i are valid

        # Quick perturbation approach
        old_val = a[i]
        # Try a few candidate values
        candidates = [
            old_val * 1.1, old_val * 0.9,
            old_val + 0.01 * S / n, old_val - 0.01 * S / n,
            0.0, old_val * 1.5, old_val * 0.5
        ]
        candidates = [max(0, min(1000, c)) for c in candidates]

        best_val = old_val
        best_obj_i = M
        for cv in candidates:
            if abs(cv - old_val) < 1e-12:
                continue
            a[i] = cv
            # Recompute affected convolution values
            # c[k] for k where a[i] appears: k = i + j for j in [0, n-1]
            # c[k] = a[i]*a[k-i] + (sum of other terms)
            # This is expensive to do exactly, so just recompute full conv
            # (for n=8192, one FFT conv is ~1ms, so this is fine)
            c_check = fft_convolve_self(a)
            M_check = c_check.max()
            if M_check < best_obj_i:
                best_obj_i = M_check
                best_val = cv
            a[i] = old_val

        if best_val != old_val:
            a[i] = best_val
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            M = c.max()
            if M < best_M - 1e-12:
                best_M = M
                best_a = a.copy()

        if pos % 50000 == 0:
            print("    CoordDesc pos=%d, M=%.8f" % (pos, best_M))

    return best_a, best_M


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max(conv)/sum^2.
    Multi-phase: LP-optimal subgradient -> FISTA -> L-BFGS -> coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Init: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    n = n0
    S = np.sqrt(2.0 * n)

    # === Phase 1: LP-optimal subgradient (45% time) ===
    remaining = deadline - time.monotonic()
    p1 = remaining * 0.45
    if p1 > 30:
        print("\n=== Phase 1: LP-optimal subgradient (budget=%.0fs) ===" % p1)
        t1 = time.monotonic()
        a1, _ = run_lp_subgrad(global_best_a, p1 * 0.7, rng, K=300, use_momentum=True)
        obj1 = compute_obj(a1)
        print("  LP-SG (momentum): obj=%.8f (t=%.0fs)" % (obj1, time.monotonic() - t1))
        if obj1 < global_best_obj:
            global_best_obj = obj1
            global_best_a = a1.copy()

        rem1 = p1 - (time.monotonic() - t1)
        if rem1 > 30:
            a1b, _ = run_lp_subgrad(global_best_a, rem1, rng, K=500, use_momentum=False)
            obj1b = compute_obj(a1b)
            print("  LP-SG (no momentum, K=500): obj=%.8f (t=%.0fs)" %
                  (obj1b, time.monotonic() - t1))
            if obj1b < global_best_obj:
                global_best_obj = obj1b
                global_best_a = a1b.copy()

    # === Phase 2: FISTA polishing (25% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2 = remaining * 0.25
        print("\n=== Phase 2: FISTA polishing (budget=%.0fs) ===" % p2)
        t2 = time.monotonic()

        a2, _ = run_fista(global_best_a, p2 * 0.5, rng, beta_start=30.0, mode="lse")
        obj2 = compute_obj(a2)
        print("  FISTA-lse: obj=%.8f" % obj2)
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

        rem2 = p2 - (time.monotonic() - t2)
        if rem2 > 20:
            a2b, _ = run_fista(global_best_a, rem2, rng, beta_start=50.0, mode="hybrid")
            obj2b = compute_obj(a2b)
            print("  FISTA-hybrid: obj=%.8f" % obj2b)
            if obj2b < global_best_obj:
                global_best_obj = obj2b
                global_best_a = a2b.copy()

    # === Phase 3: L-BFGS-B (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p3 = remaining * 0.20
        print("\n=== Phase 3: L-BFGS-B (budget=%.0fs) ===" % p3)
        t3 = time.monotonic()
        a3, obj3 = run_lbfgs(global_best_a, p3, rng, n_levels=4)
        print("  L-BFGS: obj=%.8f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Coordinate descent (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4 = remaining * 0.10
        print("\n=== Phase 4: Coordinate descent (budget=%.0fs) ===" % p4)
        t4 = time.monotonic()
        a4, _ = run_coordinate_descent(global_best_a, p4, rng)
        obj4 = compute_obj(a4)
        print("  CoordDesc: obj=%.8f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Final LP squeeze (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 5: Final LP squeeze (budget=%.0fs) ===" % remaining)
        t5 = time.monotonic()
        a5, _ = run_lp_subgrad(global_best_a, remaining * 0.8, rng, K=400, use_momentum=True)
        obj5 = compute_obj(a5)
        print("  Final LP: obj=%.8f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence failed, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            global_best_obj = compute_obj(global_best_a)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective: %.8f" % global_best_obj)
    return final_list
