import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_objective_and_grad(c, beta):
    """Compute LSE smooth approximation of max(c) and its gradient weights."""
    cm = c.max()
    shifted = beta * (c - cm)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    e = np.exp(shifted)
    tot = e.sum()
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    return lse_val, w

def active_set_weights(c, tol_frac=0.001):
    """Get weights for active set (lags within tol_frac of max)."""
    cm = c.max()
    threshold = cm * (1 - tol_frac)
    w = np.zeros_like(c)
    active = c >= threshold
    if active.sum() == 0:
        w[np.argmax(c)] = 1.0
    else:
        w[active] = 1.0 / active.sum()
    return w

def run_active_gradient(a, n, S, max_time, rng, tol_frac=0.002, max_backtrack=15):
    """
    Active-set gradient descent with backtracking line search.
    Focus on the active lags (within tol_frac of max).
    """
    a = normalize_sum(a, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Get active set weights
        w = active_set_weights(c, tol_frac)
        
        # Compute gradient
        grad = compute_gradient_from_weights(w, a, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Backtracking line search
        step = 0.1 / (1 + 0.01 * gnorm)
        improved = False
        for bt in range(max_backtrack):
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max * (1 - 1e-7):
                a = a_new
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # Very small step
            a = project_simplex(a - 0.01 * step * grad, S)
        
        # Periodic noise
        if it % 100 == 0:
            noise_scale = 0.0005 * S / np.sqrt(n) * np.exp(-it / 5000.0)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best

def run_peak_surgery(a, n, S, max_time, n_peaks=10, lr=0.01):
    """
    Direct peak surgery: identify elements contributing to peaks and reduce them.
    """
    a = a.copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Get top n_peaks lags
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # For each j, compute total "blame"
        blame = np.zeros(n)
        for k in top_idx:
            weight = max(0.0, c[k] - cur_max * 0.995)
            if weight == 0:
                weight = 1.0  # At least the top one
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            np.add.at(blame, js, contrib * weight)
        
        blame_max = blame.max()
        if blame_max < 1e-30:
            break
        
        # Reduce high-blame elements
        effective_lr = lr * S / (n * blame_max + 1e-30)
        a = a * np.exp(-effective_lr * blame)
        a = project_simplex(a, S)
    
    return best_a, best

from scipy.fft import fft, ifft

def fft_conv_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def proj_simplex(v, S):
    """Project v onto {x: x>=0, sum(x)=S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cs - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.where(valid)[0].max() + 1)
    theta = (cs[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_grad(a, c, beta):
    """LSE-smoothed gradient of max(c)."""
    cm = c.max()
    e = np.exp(np.clip(beta * (c - cm), -50, 50))
    w = e / e.sum()
    lse_val = cm + np.log(e.sum()) / beta
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, lse_val

    if s < 1e-15:
