```python
"""
Search for non-negative coefficients minimizing
    2*n*max(np.convolve(a,a)) / (sum(a)^2)
by optimizing the probability vector p = a / sum(a), where the objective is
    2*n*max(p*p)
and a = sqrt(2*n) * p.

The optimizer combines:
- L-BFGS on a softmax parameterization with a log-sum-exp modular approximation
  of the maximum convolution coefficient,
- exact Frank-Wolfe / pair-move polishing with line search on the true max,
- multi-scale (coarse-to-fine) restarts,
- Fourier-smoothed random restarts.
"""

import numpy as np
import time
from scipy.optimize import minimize as _scipy_minimize

# Globals for the L-BFGS callable.  Top-level helper functions are required,
# so small problem state is stored here during each L-BFGS run.
_OPT_N = 0
_OPT_SCALE = 0.0
_OPT_FSIZE = 0
_OPT_BETA = 1.0
_MIN_CB_DEADLINE = 0.0


def _minimize_callback(xk):
    if time.monotonic() > _MIN_CB_DEADLINE:
        return True
    return False


def _time_left(deadline):
    return max(0.0, deadline - time.monotonic())


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _fft_convolve_self(x, fsize=None):
    n = int(x.size)
    size = 2 * n - 1
    if fsize is None:
        fsize = _next_pow2(size)
    X = np.fft.rfft(np.asarray(x, dtype=np.float64), fsize)
    out = np.fft.irfft(X * X, fsize)
    return out[:size]


def _obj_p_fixed(p, scale, fsize):
    return float(scale * np.max(_fft_convolve_self(np.asarray(p, dtype=np.float64), fsize)))


def _obj_p(p):
    n = int(np.asarray(p).size)
    fsize = _next_pow2(2 * n - 1)
    scale = 2.0 * n
    return _obj_p_fixed(p, scale, fsize)


def _lse_weights(r, beta):
    r = np.asarray(r, dtype=np.float64)
    beta = float(beta)
    if beta <= 0:
        return np.full(r.size, 1.0 / r.size)
    rmax = float(np.max(r))
    z = beta * (r - rmax)
    z -= float(np.max(z))
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0:
        return np.full(r.size, 1.0 / r.size)
    return ez / s


def _topk_weights(r, k, power=1.0):
    r = np.asarray(r, dtype=np.float64)
    k = int(k)
    if k <= 0:
        return np.full(r.size, 1.0 / r.size)
    if k >= r.size:
        return np.full(r.size, 1.0 / r.size)
    idx = np.argpartition(r, -k)[-k:]
    w = np.zeros(r.size, dtype=np.float64)
    if power == 1:
        w[idx] = 1.0
    else:
        w[idx] = np.power(np.maximum(r[idx], 0.0), float(power))
    s = float(w.sum())
    if s <= 0 or not np.isfinite(s):
        return np.full(r.size, 1.0 / r.size)
    return w / s


def _conv_grad_weights(p, w, fsize=None):
    """
    Gradient of sum_k w[k] * (p*p)[k] with respect to p.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(p.size)
    size = 2 * n - 1
    if fsize is None:
        fsize = _next_pow2(size)
    W = np.fft.rfft(w, fsize)
    Pr = np.fft.rfft(p[::-1], fsize)
    cross = np.fft.irfft(W * Pr, fsize)[:size]
    return 2.0 * cross[n - 1:2 * n - 1]


def _set_opt_globals(n, beta):
    global _OPT_N, _OPT_SCALE, _OPT_FSIZE, _OPT_BETA
    _OPT_N = int(n)
    _OPT_SCALE = float(2 * _OPT_N)
    _OPT_FSIZE = _next_pow2(2 * _OPT_N - 1)
    _OPT_BETA = float(beta)


def _softmax(theta):
    theta = np.asarray(theta, dtype=np.float64)
    if theta.size == 0:
        return np.asarray([], dtype=np.float64)
    if not np.all(np.isfinite(theta)):
        theta = np.where(np.isfinite(theta), theta, 0.0)
    theta = theta - float(np.max(theta))
    e = np.exp(theta)
    z = float(e.sum())
    if not np.isfinite(z) or z <= 0:
        return np.ones(theta.size, dtype=np.float64) / theta.size
    return e / z


def _lbfgs_objgrad(x):
    n = _OPT_N
    scale = _OPT_SCALE
    fsize = _OPT_FSIZE
    beta = _OPT_BETA
    theta = np.asarray(x, dtype=np.float64).reshape(-1)
    if theta.size != n or not np.all(np.isfinite(theta)):
        return 1.0e100, np.zeros(n, dtype=np.float64)

    # Remove the mean-null direction of the softmax.
    theta = theta - float(np.mean(theta))
    p = _softmax(theta)

    q = _fft_convolve_self(p, fsize)
    r = scale * q
    rmax = float(np.max(r))
    if not np.isfinite(rmax):
        return 1.0e100, np.zeros(n, dtype=np.float64)

    z = beta * (r - rmax)
    z -= float(np.max(z))
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0:
        return float(n * rmax), np.zeros(n, dtype=np.float64)

    val = rmax + float(np.log(s)) / beta
    w = ez / s

    gp = scale * _conv_grad_weights(p, w, fsize)
    dot = float(np.dot(p, gp))
    gt = p * gp - p * dot
    gt -= float(np.mean(gt))

    if not np.all(np.isfinite(gt)):
        gt = np.where(np.isfinite(gt), gt, 0.0)

    gmax = float(np.max(np.abs(gt)))
    if gmax > 1.0e9:
        gt = gt * (1.0e9 / gmax)

    # Scale the objective and gradient by n to improve L-BFGS conditioning,
    # since p entries are O(1/n).
    obj_scale = float(n)
    return float(val) * obj_scale, (gt * obj_scale).astype(np.float64)


def _theta_from_p(p):
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, None)
    s = float(p.sum())
    n = p.size
    if s <= 0 or not np.isfinite(s):
        return np.zeros(n, dtype=np.float64)
    p = p / s
    logp = np.log(np.maximum(p, 1.0e-300))
    logp -= float(np.mean(logp))
    return logp


def run_lbfgs_stage(theta0, beta, max_time, maxiter=200, gtol=1.0e-8):
    global _MIN_CB_DEADLINE
    theta0 = np.asarray(theta0, dtype=np.float64).reshape(-1)
    if theta0.size == 0:
        return theta0
    n = int(theta0.size)
    _set_opt_globals(n, beta)
    theta = theta0 - float(np.mean(theta0))
    start = time.monotonic()
    max_time = float(max_time)

    while True:
        remain = max_time - (time.monotonic() - start)
        if remain <= 0:
            break
        chunk = min(40.0, remain)
        if chunk < 5.0:
            break
        _MIN_CB_DEADLINE = time.monotonic() + chunk
        try:
            res = _scipy_minimize(
                _lbfgs_objgrad,
                theta,
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": int(maxiter),
                    "maxls": 20,
                    "ftol": 1.0e-10,
                    "gtol": float(gtol),
                    "iprint": 0,
                },
                callback=_minimize_callback,
            )
        except Exception:
            break

        if hasattr(res, "x") and res.x is not None:
            x = np.asarray(res.x, dtype=np.float64).reshape(-1)
            if x.size == n and np.all(np.isfinite(x)):
                theta = x - float(np.mean(x))

        if hasattr(res, "status") and res.status == 0:
            break
    return theta


def _lowpass_noise(n, rng, kc=None):
    n = int(n)
    if kc is None:
        kc = max(5.0, n / 30.0)
    F = np.fft.rfft(rng.normal(size=n))
    k = np.arange(F.size, dtype=np.float64)
    mask = np.exp(-((k / kc) ** 2))
    mask[0] = 0.0
    noise = np.fft.irfft(F * mask, n)
    norm = float(np.max(np.abs(noise)))
    if norm > 1.0e-15:
        noise = noise / norm
    return noise


def _smooth_p(p, kc=None):
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    n = int(p.size)
    p = np.clip(p, 0.0, None)
    s = float(p.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / n
    p = p / s
    if kc is None:
        kc = max(10.0, n / 30.0)
    F = np.fft.rfft(p)
    k = np.arange(F.size, dtype=np.float64)
    mask = np.exp(-((k / kc) ** 2))
    q = np.fft.irfft(F * mask, n)
    q = np.clip(q, 0.0, None)
    qs = float(q.sum())
    if qs <= 0 or not np.isfinite(qs):
        return p
    return q / qs


def _resize_p(p, n):
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, None)
    n = int(n)
    m = p.size
    s = float(p.sum())
    if m == n:
        if s > 0 and np.isfinite(s):
            return p / s
        return np.ones(n, dtype=np.float64) / n
    if m == 1:
        return np.ones(n, dtype=np.float64) / n
    old_t = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
    new_t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    q = np.interp(new_t, old_t, p, left=p[0], right=p[-1])
    q = np.clip(q, 0.0, None)
    qs = float(q.sum())
    if qs <= 0 or not np.isfinite(qs):
        return np.ones(n, dtype=np.float64) / n
    return q / qs


def _downsample_p(p, m):
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    p = np.clip(p, 0.0, None)
    n = int(p.size)
    m = int(m)
    if m <= 1:
        return np.ones(m, dtype=np.float64) / m
    if n == m:
        s = float(p.sum())
        if s > 0:
            return p / s
        return np.ones(m, dtype=np.float64) / m
    if n % m == 0:
        q = p.reshape(m, n // m).mean(axis=1)
    else:
        old_t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        new_t = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
        q = np.interp(new_t, old_t, p, left=p[0], right=p[-1])
    q = np.clip(q, 0.0, None)
    s = float(q.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones(m, dtype=np.float64) / m
    return q / s


def _upsample_p(q, n):
    q = np.asarray(q, dtype=np.float64).reshape(-1)
    q = np.clip(q, 0.0, None)
    s = float(q.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / n
    q = q / s
    m = q.size
    n = int(n)
    if m == n:
        return q
    if n % m == 0:
        full = np.repeat(q, n // m)
    else:
        old_t = (np.arange(m, dtype=np.float64) + 0.5) / float(m)
        new_t = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
        full = np.interp(new_t, old_t, q, left=q[0], right=q[-1])
    full = np.clip(full, 0.0, None)
    fs = float(full.sum())
    if fs <= 0 or not np.isfinite(fs):
        return np.ones(n, dtype=np.float64) / n
    return full / fs


def _random_positive_p(n, rng, base=None, mix=0.5, kc=None):
    n = int(n)
    if base is None:
        base = np.ones(n, dtype=np.float64)
    base = np.asarray(base, dtype=np.float64).reshape(-1)
    base = np.clip(base, 0.0, None)
    bs = float(base.sum())
    if bs <= 0 or not np.isfinite(bs):
        base = np.ones(n, dtype=np.float64) / n
    else:
        base = base / bs

    noise = _lowpass_noise(n, rng, kc)
    rand = np.clip(1.0 + 0.9 * noise, 1.0e-8, None)
    rs = float(rand.sum())
    if rs <= 0 or not np.isfinite(rs):
        rand = np.ones(n, dtype=np.float64) / n
    else:
        rand = rand / rs

    mix = float(np.clip(mix, 0.0, 1.0))
    p = (1.0 - mix) * base + mix * rand
    p = np.clip(p, 0.0, None)
    ps = float(p.sum())
    if ps <= 0 or not np.isfinite(ps):
        return base.copy()
    return p / ps


def _perturb_p(base, rng, amp=None, kc=None):
    base = np.asarray(base, dtype=np.float64).reshape(-1)
    n = int(base.size)
    base = np.clip(base, 0.0, None)
    bs = float(base.sum())
    if bs <= 0 or not np.isfinite(bs):
        base = np.ones(n, dtype=np.float64) / n
    else:
        base = base / bs
    if amp is None:
        amp = float(0.03 + 0.10 * rng.random())
    mean = float(np.mean(base))
    noise = _lowpass_noise(n, rng, kc)
    p = base + 5.0 * float(amp) * mean * noise
    p = np.clip(p, 0.0, None)
    ps = float(p.sum())
    if ps <= 0 or not np.isfinite(ps):
        return base.copy()
    return p / ps


def run_pair_polish(p0, max_time=60.0, beta=50.0, K_init=12, rng=None):
    p = np.asarray(p0, dtype=np.float64).reshape(-1)
    p = np.clip(p, 0.0, None)
    s = float(p.sum())
    if s <= 0 or not np.isfinite(s):
        return p0.copy(), np.inf
    p = p / s
    n = int(p.size)
    fsize = _next_pow2(2 * n - 1)
    scale = 2.0 * n

    best_p = p.copy()
    best_obj = _obj_p_fixed(p, scale, fsize)
    start = time.monotonic()
    max_time = float(max_time)

    beta_cur = float(beta)
    K = int(max(2, K_init))
    step_frac = 0.05
    stuck = 0
    it = 0

    while True:
        if time.monotonic() - start >= max_time:
            break
        it += 1

        q = _fft_convolve_self(p, fsize)
        r = scale * q
        cur_obj = float(np.max(r))
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_p = p.copy()

        if (it % 3 == 0):
            k = max(1, K // 4)
            w = _topk_weights(r, k, 1.0)
        else:
            w = _lse_weights(r, beta_cur)

        g = scale * _conv_grad_weights(p, w, fsize)
        g -= float(np.mean(g))
        if not np.all(np.isfinite(g)):
            break

        g_inf = float(np.max(np.abs(g)))
        if g_inf <= 1.0e-15:
            break

        min_p = max(1.0e-15, 0.01 * float(np.mean(p)))
        g_src = g.copy()
        g_src[p < min_p] = -np.inf

        ks = int(min(K, max(2, n // 16)))
        if int(np.sum(np.isfinite(g_src))) >= ks:
            src = np.argpartition(g_src, -ks)[-ks:]
        else:
            src = np.argsort(g)[::-1][:ks]
        tgt = np.argpartition(g, ks - 1)[:ks]

        mask = np.zeros(n, dtype=np.float64)
        mask[tgt] = 1.0
        mask[src] = -1.0

        src_mask = mask < 0
        if not np.any(src_mask):
            stuck += 50
            continue

        min_src = float(np.min(p[src_mask]))
        s_max = step_frac * min_src
        if s_max <= 1.0e-20:
            stuck += 50
            continue

        best_s = 0.0
        best_local = cur_obj
        if stuck > 300:
            fracs = (0.001, 0.003, 0.01, 0.03, 0.10, 0.30, 0.60)
        else:
            fracs = (0.01, 0.03, 0.10, 0.30, 0.60, 1.00)

        for fr in fracs:
            sv = float(fr) * s_max
            p_try = p + sv * mask
            if p_try.min() < 0.0:
                p_try = np.clip(p_try, 0.0, None)
            psum = float(p_try.sum())
            if psum <= 0.0 or not np.isfinite(psum):
                continue
            if abs(psum - 1.0) > 1.0e-12:
                p_try = p_try / psum
            ot = _obj_p_fixed(p_try, scale, fsize)
            if ot < best_local:
                best_local = ot
                best_s = sv

        if best_s > 0.0:
            p = p + best_s * mask
            p = np.clip(p, 0.0, None)
            ps = float(p.sum())
            if ps <= 0.0 or not np.isfinite(ps):
                p = best_p.copy()
                stuck += 100
                continue
            p = p / ps
            cur_obj = _obj_p_fixed(p, scale, fsize)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_p = p.copy()
                stuck = 0
                if step_frac < 0.35:
                    step_frac = min(0.35, step_frac * 1.12)
            else:
                stuck += 1
                if step_frac > 0.004:
                    step_frac *= 0.75
        else:
            stuck += 1
            if step_frac > 0.002:
                step_frac *= 0.5

        # Adapt the active set / smoothing.
        if stuck > 0 and (it % 75 == 0):
            if stuck < 300:
                beta_cur = max(1.0, beta_cur * 0.6)
                K = max(2, K // 2)
                step_frac = max(0.002, step_frac * 0.5)
            elif stuck < 1200:
                beta_cur = min(1000.0, beta_cur * 1.3)
                K = min(64, K * 2)
            else:
                if rng is not None:
                    p = _perturb_p(p, rng, amp=0.05 + 0.12 * rng.random())
                else:
                    p = _perturb_p(p, np.random.default_rng(it + 17), amp=0.07)
                stuck = 0
                step_frac = 0.05

        # Avoid drifting too far from the best exact point.
        if best_obj < cur_obj and cur_obj > best_obj * (1.0 + 1.0e-6):
            if stuck > 500 and (it % 200 == 0):
                p = best_p.copy()
                stuck = 250

    return best_p, best_obj

    # ------------------------------------------------------------------
    res = _time_left(deadline)
    phase1 = min(1000.0, max(30.0, 0.40 * res))
    phase1 = min(phase1, max(0.0, res - 550.0))
    if phase1 > 30.0:
        betas = (5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0)
        fracs = (0.04, 0.08, 0.18, 0.24, 0.22, 0.12, 0.12)
        sfrac = float(sum(fracs))
        for beta, frac in zip(betas, fracs):
            t = phase1 * float(frac) / sfrac
            t = min(t, _time_left(deadline) - 550.0)
            if t < 12.0:
                break
            theta_cur = run_lbfgs_stage(theta_cur, beta, t, maxiter=180, gtol=1.0e-8)
            p_cur = _softmax(theta_cur)
            obj_cur = _obj_p(p_cur)
            if obj_cur < best_obj:
                best_obj = obj_cur
                best_p = p_cur.copy()
                theta_best = theta_cur.copy()
                print(f"  L-BFGS beta={beta:g}: {best_obj:.10f}", flush=True)
            else:
                # If the smooth stage drifted too far from the best exact point,
                # revert for the next stage.
                if obj_cur > best_obj + 0.005:
                    theta_cur = theta_best.copy()

    # ------------------------------------------------------------------
    # Phase 2: Exact pair-move polishing on the current best.
    # ------------------------------------------------------------------
    res = _time_left(deadline)
    t = min(300.0, max(30.0, 0.13 * res))
    t = min(t, max(0.0, res - 500.0))
    if t > 20.0:
        p_try, obj_try = run_pair_polish(best_p, t, beta=20.0, K_init=10, rng=rng)
        if obj_try < best_obj:
            best_obj = obj_try
            best_p = p_try.copy()
            theta_best = _theta_from_p(best_p)
            print(f"  Pair polish 1: {best_obj:.10f}", flush=True)

    res = _time_left(deadline)
    t = min(220.0, max(20.0, 0.09 * res))
    t = min(t, max(0.0, res - 450.0))
    if t > 15.0:
        p_try, obj_try = run_pair_polish(best_p, t, beta=120.0, K_init=24, rng=rng)
        if obj_try < best_obj:
            best_obj = obj_try
            best_p = p_try.copy()
            theta_best = _theta_from_p(best_p)
            print(f"  Pair polish 2: {best_obj:.10f}", flush=True)

    # ------------------------------------------------------------------
    # Phase 3: Coarse-to-fine restarts.
    # ------------------------------------------------------------------
    res = _time_left(deadline)
    if res > 450.0:
        coarse_budget = min(350.0, 0.18 * res)
        for m in (64, 128, 256):
            if _time_left(deadline) < 380.0:
                break
            t_coarse = max(30.0, coarse_budget / 3.0)
            t_coarse = min(t_coarse, _time_left(deadline) - 380.0)
            if t_coarse < 25.0:
                break

            pm_base = _downsample_p(best_p, m)
            variants = [
                pm_base,
                _random_positive_p(m, rng, base=pm_base, mix=0.5, kc=float(m // 10)),
            ]

            for vm, pm_start in enumerate(variants):
                if _time_left(deadline) < 350.0:
                    break
                theta_m = _theta_from_p(pm_start)
                t1 = min(35.0, max(10.0, 0.40 * t_coarse))
                t2 = min(45.0, max(10.0, 0.60 * t_coarse))

                theta_m = run_lbfgs_stage(theta_m, 20.0, t1, maxiter=220, gtol=1.0e-7)
                theta_m = run_lbfgs_stage(theta_m, 200.0, t2, maxiter=220, gtol=1.0e-7)
                p_m = _softmax(theta_m)
                p_full = _upsample_p(p_m, target_n)
                p_full = _smooth_p(p_full, kc=max(20.0, target_n / 50.0))

                if _time_left(deadline) > 260.0:
                    p_full, obj_full = run_pair_polish(
                        p_full,
                        min(35.0, _time_left(deadline) - 220.0),
                        beta=60.0,
                        K_init=16,
                        rng=rng
                    )
                else:
                    obj_full = _obj_p(p_full)

                if obj_full < best_obj:
                    best_obj = obj_full
                    best_p = p_full.copy()
                    theta_best = _theta_from_p(best_p)
                    print(f"  Coarse m={m} variant={vm}: {best_obj:.10f}", flush=True)

    # ------------------------------------------------------------------
    # Phase 4: Basin hopping via smoothed random restarts.
    # ------------------------------------------------------------------
    attempts = 0
    while True:
        if _time_left(deadline) <= 320.0 or attempts >= 220:
            break
        attempts += 1

        u = rng.random()
        if u < 0.40:
            p_cand = _perturb_p(best_p, rng, amp=0.02 + 0.08 * rng.random())
        elif u < 0.80:
            p_cand = _random_positive_p(target_n, rng, base=best_p,
                                        mix=0.30 + 0.45 * rng.random(),
                                        kc=float(target_n / 20.0))
        else:
            p_cand = _random_positive_p(target_n, rng, base=best_p,
                                        mix=0.60 + 0.35 * rng.random(),
                                        kc=float(target_n / 8.0))

        if _time_left(deadline) > 280.0:
            theta_c = _theta_from_p(p_cand)
            theta_c = run_lbfgs_stage(theta_c, 40.0, 22.0, maxiter=100, gtol=1.0e-7)
            p_cand = _softmax(theta_c)

        if _time_left(deadline) > 200.0:
            p_cand, obj_cand = run_pair_polish(
                p_cand,
                min(55.0, _time_left(deadline) - 170.0),
                beta=60.0,
                K_init=16,
                rng=rng
            )
        else:
            obj_cand = _obj_p(p_cand)

        if obj_cand < best_obj:
            best_obj = obj_cand
            best_p = p_cand.copy()
            theta_best = _theta_from_p(best_p)
            print(f"  Basin attempt {attempts}: {best_obj:.10f}", flush=True)

    # ------------------------------------------------------------------
    # Phase 5: Final exact polishing.
    # ------------------------------------------------------------------
    if _time_left(deadline) > 130.0:
        t = min(240.0, _time_left(deadline) - 70.0)
        p_try, obj_try = run_pair_polish(best_p, t, beta=80.0, K_init=18, rng=rng)
        if obj_try < best_obj:
            best_obj = obj_try
            best_p = p_try.copy()
            theta_best = _theta_from_p(best_p)
            print(f"  Final pair polish: {best_obj:.10f}", flush=True)

    if _time_left(deadline) > 90.0:
        theta_f = _theta_from_p(best_p)
        theta_f = run_lbfgs_stage(theta_f, 500.0, min(50.0, _time_left(deadline) - 60.0),
                                  maxiter=120, gtol=1.0e-8)
        p_try = _softmax(theta_f)
        obj_try = _obj_p(p_try)
        if obj_try < best_obj:
            best_obj = obj_try
            best_p = p_try.copy()
            theta_best = _theta_from_p(best_p)
            print(f"  Final L-BFGS high-beta: {best_obj:.10f}", flush=True)

        if _time_left(deadline) > 60.0:
            p_try, obj_try = run_pair_polish(best_p, min(40.0, _time_left(deadline) - 40.0),
                                             beta=250.0, K_init=32, rng=rng)
            if obj_try < best_obj:
                best_obj = obj_try
                best_p = p_try.copy()
                print(f"  Final sharp pair polish: {best_obj:.10f}", flush=True)

    best_p = np.asarray(best_p, dtype=np.float64).reshape(-1)
    best_p = np.clip(best_p, 0.0, None)
    ps = float(best_p.sum())
    if ps <= 0.0 or not np.isfinite(ps):
        best_p = np.ones(target_n, dtype=np.float64) / target_n
        ps = float(best_p.sum())
    best_p = best_p / ps

    # Convert to sequence a = sqrt(2*n) * p.
    total_sequence_sum = float(np.sqrt(2.0 * best_p.size))
    a_out = total_sequence_sum * best_p
    a_out = np.where(np.isfinite(a_out), a_out, 0.0)
    a_out = np.clip(a_out, 0.0, 1000.0)

    final_list = [float(x) for x in a_out]

    # One final verification, only if the evaluator is available.
    try:
        ver = evaluate_sequence(final_list)
        print(f"Verified objective: {ver:.10f} (n={len(final_list)})", flush=True)
        if not np.isfinite(ver):
            fb = np.ones(target_n, dtype=np.float64)
            fb_sum = float(np.sqrt(2.0 * target_n) / target_n)
            final_list = [fb_sum] * target_n
            ver = evaluate_sequence(final_list)
            print(f"Fallback objective: {ver:.10f}", flush=True)
    except Exception as e:
        print("final evaluate_sequence unavailable/error:", e, flush=True)

    return final_list
```
