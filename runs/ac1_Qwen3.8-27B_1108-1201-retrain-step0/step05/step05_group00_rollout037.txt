    idx = np.arange(1, n + 1, dtype=np.float64)

"""
Minimize 2*n*max_conv/sum^2 using adaptive LSE + peak deflation + top-K refinement.

Key improvements over previous approach:
1. Peak deflation: structured perturbation that directly targets and reduces the max peak
2. Adaptive top-K gradient: smoothly transitions between smooth LSE and sharp top-K focus
3. Multi-n search: tries multiple sequence lengths to find the best
4. Armijo backtracking line search for robust convergence
5. Warm start from given sequence with spectral reshaping
6. Entropic (multiplicative) updates for better handling of non-negativity
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= v_sorted
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Compute full self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_conv_fft(a, N):
    """Compute full self-convolution and FFT of padded sequence."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_grad(a, N, beta):
    """Compute LSE-smoothed max of self-convolution and its gradient.
    Returns (grad, lse_val, c, c_max, k_star)."""
    n = len(a)
    c, A = compute_conv_fft(a, N)
    c_max = c.max()
    k_star = int(np.argmax(c))

    if beta < 1e-4:
        # Use hard max gradient
        a_pad = np.zeros(N)
        a_pad[:n] = a
        Af = fft(a_pad)
        w = np.zeros(N)
        w[k_star] = 1.0
        W = fft(w)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        return grad, c_max, c, c_max, k_star

    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -50, 0.0)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        exp_vals = np.zeros(len(c))
        exp_vals[k_star] = 1.0
        total = 1.0
    lse_val = c_max + np.log(total) / beta
    weights = exp_vals / total

    # Gradient via FFT
    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = weights
    W = fft(w_pad)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, lse_val, c, c_max, k_star


def toppk_grad(a, N, K, temp):
    """Gradient from top-K peaks weighted by softmax temperature.
    Returns (grad, c, c_max, k_star)."""
    n = len(a)
    c, A = compute_conv_fft(a, N)
    c_max = c.max()
    k_star = int(np.argmax(c))

    if K >= len(c):
        return lse_grad(a, N, 1.0 / max(temp, 1e-6))[0:2] + (c, c_max, k_star)

    # Select top K
    top_idx = np.argsort(c)[::-1][:K]
    top_vals = c[top_idx]
    # Softmax weights over top K
    shifted = (top_vals - top_vals[0]) / max(temp, 1e-10)
    shifted = np.clip(shifted, -50, 0.0)
    exp_w = np.exp(shifted)
    total_w = exp_w.sum()
    w_top = exp_w / total_w

    w_full = np.zeros(len(c))
    for i, idx in enumerate(top_idx):
        w_full[int(idx)] = w_top[i]

    w_pad = np.zeros(N)
    w_pad[:2 * n - 1] = w_full
    W = fft(w_pad)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, c, c_max, k_star


def max_loc_grad(a, N, k_star):
    """Gradient at the single maximum location."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    Af = fft(a_pad)
    w = np.zeros(N)
    w[k_star] = 1.0
    W = fft(w)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad


def peak_deflation(a, c, N, n, k_star, strength=0.05):
    """Structured perturbation: reduce elements contributing to the peak.
    Returns modified a."""
    S = a.sum()
    # Compute contribution of each a[i] to c[k_star]
    # c[k_star] = sum_i a[i]*a[k_star-i]
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    contrib = np.zeros(n)
    for i in range(lo, hi + 1):
        j = k_star - i
        if 0 <= j < n:
            contrib[i] += a[j]
    # Reduce high contributors, add to low contributors
    max_contrib = contrib.max()
    min_contrib = contrib.min()
    if max_contrib - min_contrib < 1e-12:
        return a.copy()
    # Transfer mass from high to low contributors
    excess = np.maximum(contrib - np.median(contrib), 0)
    deficit = np.maximum(np.median(contrib) - contrib, 0)
    excess_sum = excess.sum()
    deficit_sum = deficit.sum()
    if excess_sum < 1e-15 or deficit_sum < 1e-15:
        return a.copy()
    scale = min(1.0, strength * S / excess_sum)
    a_new = a.copy()
    a_new -= scale * excess * (a / max(a.sum(), 1e-15))
    a_new += scale * deficit * (a / max(a.sum(), 1e-15))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def spectral_reshaping(a, N, rng, frac=0.3):
    """Randomize a fraction of FFT phases, project to non-negative."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def armijo_ls(a, grad, S, N, n, f_curr, c_max, c_sort, alpha_init, c1=0.01, max_ls=15):
    """Armijo backtracking line search.
    Returns (a_new, cmax_new, alpha_used)."""
    alpha = alpha_init
    for ls in range(max_ls):
        a_try = project_simplex(a - alpha * grad, S)
        c_try = compute_conv(a_try, N)
        cm_try = c_try.max()
        # Armijo condition: new peak should decrease
        if cm_try <= c_max * (1.0 - c1 * alpha * np.linalg.norm(grad) / max(c_max, 1e-10)) or ls == max_ls - 1:
            return a_try, cm_try, alpha
        alpha *= 0.5
    return a_try, cm_try, alpha


def lbfgs_two_loop(s_hist, y_hist, g, m=10):
    """L-BFGS two-loop recursion for search direction."""
    m_use = min(len(s_hist), m)
    if m_use == 0:
        gn = np.linalg.norm(g)
        if gn < 1e-18:
            return np.zeros(len(g))
        return -g / gn
    s_h = s_hist[-m_use:]
    y_h = y_hist[-m_use:]
    q = g.copy()
    alphas = np.zeros(m_use)
    for i in range(m_use - 1, -1, -1):
        yy = np.dot(y_h[i], y_h[i])
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_h[i], q) / yy
            q = q - alphas[i] * y_h[i]
    s_last = s_h[-1]
    y_last = y_h[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma = max(1e-10, np.dot(s_last, y_last) / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m_use):
        yy = np.dot(y_h[i], y_h[i])
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_h[i], y_h[i]) / yy
        r = r + (alphas[i] - beta_i) * y_h[i]
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def optimize_lse(a_init, n, S, N, max_time, rng, beta_start=5.0, beta_end=300.0, m_lbfgs=10, use_toppk=False, topk_start=50, topk_end=1):
    """Main LSE-based optimization loop with adaptive beta and optional top-K switching."""
    a = safe_normalize(a_init, n, S)
    beta = beta_start
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    start_time = time.monotonic()
    it = 0
    stall_count = 0
    best_cmax_for_stall = best_cmax

    # Initial gradient
    if use_toppk:
        g, c, c_max, k_star = toppk_grad(a, N, topk_start, 1.0 / beta)
    else:
        g, lse_val, c, c_max, k_star = lse_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    gn = np.linalg.norm(g)
    if gn < 1e-18:
        return best_a, best_cmax

    alpha_init = 2.0 * S / (n * max(beta, 1.0) * max(gn, 1e-10))

    while time.monotonic() - start_time < max_time:
        it += 1

        # Adaptive beta increase
        if it % 100 == 0:
            progress = min(1.0, it / max(1, int(max_time * 2)))
            beta = beta_start * (beta_end / beta_start) ** progress
            alpha_init = 2.0 * S / (n * max(beta, 1.0) * max(gn, 1e-10))

        # Switch to top-K as we progress
        if it % 50 == 0:
            if use_toppk:
                k_progress = min(1.0, it / 300.0)
                K_cur = max(topk_end, int(topk_start - k_progress * (topk_start - topk_end)))
                g, c, c_max, k_star = toppk_grad(a, N, K_cur, 1.0 / beta)
            else:
                g, lse_val, c, c_max, k_star = lse_grad(a, N, beta)

        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
            stall_count = 0
            best_cmax_for_stall = c_max
        else:
            stall_count += 1

        # Peak deflation when stalled
        if stall_count % 50 == 40 and stall_count > 0:
            a_deflated = peak_deflation(a, c, N, n, k_star, strength=0.03 + 0.01 * min(stall_count / 100.0, 1.0))
            c_defl = compute_conv(a_deflated, N)
            if c_defl.max() < best_cmax:
                a = project_simplex(a_deflated, S)
                best_cmax = c_defl.max()
                best_a = a.copy()
                if use_toppk:
                    g, c, c_max, k_star = toppk_grad(a, N, topk_start, 1.0 / beta)
                else:
                    g, lse_val, c, c_max, k_star = lse_grad(a, N, beta)
                s_hist = []
                y_hist = []
                stall_count = 0
                continue

        # Get L-BFGS direction
        direction = lbfgs_two_loop(s_hist, y_hist, g, m_lbfgs)
        dir_n = np.linalg.norm(direction)
        if dir_n < 1e-18:
            break

        # Line search
        a_new, cmax_new, alpha_used = armijo_ls(a, direction, S, N, n, lse_val if not use_toppk else c_max, c_max, None, alpha_init)

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        # Update L-BFGS history
        if use_toppk:
            k_progress = min(1.0, it / 300.0)
            K_cur = max(topk_end, int(topk_start - k_progress * (topk_start - topk_end)))
            g_new, c_new, cmax_new, k_star_new = toppk_grad(a_new, N, K_cur, 1.0 / beta)
        else:
            g_new, lse_new, c_new, cmax_new, k_star_new = lse_grad(a_new, N, beta)

        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c = c_new
        c_max = cmax_new
        k_star = k_star_new
        gn = np.linalg.norm(g)
        alpha_init = 2.0 * S / (n * max(beta, 1.0) * max(gn, 1e-10))

        # Spectral restart when very stalled
        if stall_count > 200 and it % 200 == 0 and rng.random() < 0.3:
            a = spectral_reshaping(a, N, rng, frac=0.3)
            s_hist = []
            y_hist = []
            stall_count = 0
            if use_toppk:
                g, c, c_max, k_star = toppk_grad(a, N, topk_start, 1.0 / beta)
            else:
                g, lse_val, c, c_max, k_star = lse_grad(a, N, beta)

    return best_a, best_cmax


def multi_start_optimize(inits, n, S, N, per_init_time, rng, deadline):
    """Run short optimizations from multiple initializations."""
    best_a = None
    best_score = np.inf
    for name, arr in inits:
        if time.monotonic() > deadline - 100:
            break
        a_norm = safe_normalize(arr, n, S)
        t0 = time.monotonic()
        rem = max(1.0, per_init_time - (time.monotonic() - t0))
        a_opt, cmax = optimize_lse(a_norm, n, S, N, min(rem, per_init_time), rng,
                                    beta_start=3.0, beta_end=100.0, m_lbfgs=8,
                                    use_toppk=False)
        score = 2.0 * n * cmax / (S * S)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print("  [%s] n=%d score=%.6f" % (name, n, score))
    return best_a, best_score


def make_init_random(n, rng):
    return rng.random(n) + 0.01


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (3 * tt + 0.5 * tt * tt)) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    L = max(n, 2)
    i = np.arange(L)
    c = int(rng.integers(1, L))
    phase = -np.pi * i * (2 * i + 1 - c) / L
    a = 1.0 + 0.5 * np.cos(phase) + 0.05 * rng.standard_normal(L)
    return np.clip(a[:n], 0.0, None)


def make_init_binary(n, rng):
    a = (rng.random(n) > 0.35).astype(np.float64) * (0.3 + 0.7 * rng.random(n))
    return a


def make_init_multi_freq(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for freq in [2, 5, 11, 23, 47, 89]:
        a += 0.1 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return np.clip(a, 0.01, None)


def make_init_from_spectral(n, rng, N):
    """Generate from random spectral magnitudes with structured phases."""
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    # Use conjugate symmetric phases for real signal
    phase = np.zeros(N)
    phase[1:N//2] = rng.uniform(0, 2 * np.pi, N//2 - 1)
    for k in range(1, N//2):
        phase[N-k] = -phase[k]
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses multi-n search with adaptive LSE + peak deflation + top-K refinement.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_score = np.inf
    global_best_a = None
    global_best_n = 0

    def try_update(a_cand, n_val):
        nonlocal global_best_score, global_best_a, global_best_n
        S_v = np.sqrt(2.0 * float(n_val))
        a_norm = safe_normalize(a_cand, n_val, S_v)
        N_v = 1 << (2 * n_val - 1).bit_length()
        c_v = compute_conv(a_norm, N_v)
        score = 2.0 * n_val * c_v.max() / (S_v * S_v)
        if score < global_best_score:
            global_best_score = score
            global_best_a = a_norm.copy()
            global_best_n = n_val
            print("  NEW BEST: n=%d score=%.6f (peak=%.4f)" % (n_val, score, c_v.max()))
        return score

    # Try to use given sequence
    given_seq = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and given.size <= 32768:
            given_seq = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Decide which n values to try
    n_candidates = [8192]
    if given_seq is not None and 1024 <= len(given_seq) <= 32768:
        n_candidates.insert(0, len(given_seq))
    # Add alternates
    if 4096 not in n_candidates:
        n_candidates.append(4096)
    if 16384 not in n_candidates:
        n_candidates.append(16384)

    # Sort by priority: given first, then 8192, then others
    n_candidates.sort(key=lambda x: (0 if given_seq is not None and x == len(given_seq) else 1,
                                      0 if x == 8192 else 1, x))

    print("n_candidates: %s" % str(n_candidates[:5]))
    print("Budget: %.0fs, deadline in %.0fs" % (budget_s, deadline - time.monotonic()))

    # Phase 1: Quick multi-start exploration across n values (300s)
    print("=== Phase 1: Multi-start exploration ===")
    t1 = time.monotonic()
    phase1_time = min(300.0, (deadline - t1) * 0.15)

    for n_val in n_candidates[:3]:
        if time.monotonic() > deadline - 200:
            break
        S_v = np.sqrt(2.0 * float(n_val))
        N_v = 1 << (2 * n_val - 1).bit_length()
        print("  Exploring n=%d (N=%d)..." % (n_val, N_v))

        inits = []
        if given_seq is not None and len(given_seq) == n_val:
            inits.append(("given", given_seq.copy()))
        inits.append(("rand", make_init_random(n_val, rng)))
        inits.append(("chirp", make_init_chirp(n_val, rng)))
        inits.append(("zadoff", make_init_zadoff(n_val, rng)))
        inits.append(("binary", make_init_binary(n_val, rng)))
        inits.append(("spectral", make_init_from_spectral(n_val, rng, N_v)))

        per_init = min(15.0, phase1_time / (len(n_candidates[:3]) * len(inits) + 1))
        for name, arr in inits:
            if time.monotonic() > deadline - 200:
                break
            a_norm = safe_normalize(arr, n_val, S_v)
            t0 = time.monotonic()
            a_opt, cmax = optimize_lse(a_norm, n_val, S_v, N_v, per_init, rng,
                                        beta_start=3.0, beta_end=50.0, m_lbfgs=6,
                                        use_toppk=False)
            try_update(a_opt, n_val)

    print("Phase 1 done. Best: %.6f" % global_best_score)

    # Phase 2: Focused optimization on best n (1200s)
    print("=== Phase 2: Focused optimization ===")
    t2 = time.monotonic()
    phase2_time = min(1200.0, (deadline - t2) * 0.55)

    if global_best_a is None:
        n_best = 8192
        global_best_a = np.ones(n_best) * np.sqrt(2.0 * n_best) / n_best
        global_best_n = n_best
        global_best_score = 2.0
    else:
        n_best = global_best_n

    S_best = np.sqrt(2.0 * float(n_best))
    N_best = 1 << (2 * n_best - 1).bit_length()

    a_curr = global_best_a.copy()
    # Multiple rounds with different settings
    rounds = [
        {"beta_s": 5.0, "beta_e": 80.0, "topk": False, "time_frac": 0.35},
        {"beta_s": 10.0, "beta_e": 200.0, "topk": True, "time_frac": 0.30, "tk_s": 30, "tk_e": 1},
        {"beta_s": 20.0, "beta_e": 500.0, "topk": False, "time_frac": 0.20},
        {"beta_s": 50.0, "beta_e": 800.0, "topk": True, "time_frac": 0.15, "tk_s": 15, "tk_e": 1},
    ]

    for ri, rd in enumerate(rounds):
        if time.monotonic() > deadline - 60:
            break
        block_time = phase2_time * rd["time_frac"]
        use_tk = rd["topk"]
        tk_s = rd.get("tk_s", 50)
        tk_e = rd.get("tk_e", 1)

        a_new, cmax_new = optimize_lse(a_curr, n_best, S_best, N_best, block_time, rng,
                                         beta_start=rd["beta_s"], beta_end=rd["beta_e"],
                                         m_lbfgs=12, use_toppk=use_tk,
                                         topk_start=tk_s, topk_end=tk_e)
        score_new = 2.0 * n_best * cmax_new / (S_best * S_best)
        if score_new < global_best_score:
            global_best_score = score_new
            global_best_a = a_new.copy()
            a_curr = a_new.copy()
            print("  Round %d: score=%.6f improved" % (ri, score_new))
        else:
            a_curr = a_new.copy()
            # Try spectral restart before next round
            if ri < len(rounds) - 1:
                a_curr = spectral_reshaping(a_curr, N_best, rng, frac=0.2)
                print("  Round %d: no improvement (score=%.6f), restarting" % (ri, score_new))

    print("Phase 2 done. Best: %.6f" % global_best_score)

    # Phase 3: Try other n values with focused time (400s)
    print("=== Phase 3: Alternative n values ===")
    t3 = time.monotonic()
    phase3_time = min(400.0, (deadline - t3) * 0.20)
    other_ns = [x for x in n_candidates if x != n_best][:2]

    for n_alt in other_ns:
        if time.monotonic() > deadline - 40:
            break
        if phase3_time < 30:
            break
        S_alt = np.sqrt(2.0 * float(n_alt))
        N_alt = 1 << (2 * n_alt - 1).bit_length()
        block_time = min(150.0, phase3_time / len(other_ns))

        # Initialize from best scaled
        a_init_alt = np.interp(np.linspace(0, n_alt - 1, n_alt),
                                np.linspace(0, len(global_best_a) - 1, len(global_best_a)),
                                global_best_a)
        a_init_alt = safe_normalize(a_init_alt, n_alt, S_alt)

        inits_alt = [("rescaled", a_init_alt),
                     ("rand", make_init_random(n_alt, rng)),
                     ("chirp", make_init_chirp(n_alt, rng))]

        per = block_time / len(inits_alt)
        for nm, ar in inits_alt:
            if time.monotonic() > deadline - 40:
                break
            a_opt, cmax = optimize_lse(ar, n_alt, S_alt, N_alt, per, rng,
                                        beta_start=5.0, beta_end=200.0, m_lbfgs=10,
                                        use_toppk=False)
            try_update(a_opt, n_alt)

    print("Phase 3 done. Best: %.6f (n=%d)" % (global_best_score, global_best_n))

    # Phase 4: Final polishing with very high beta (remaining time)
    print("=== Phase 4: Final polishing ===")
    t4 = time.monotonic()
    remaining = deadline - t4 - 15
    if remaining > 30:
        n_fin = global_best_n
        S_fin = np.sqrt(2.0 * float(n_fin))
        N_fin = 1 << (2 * n_fin - 1).bit_length()

        a_fin, cmax_fin = optimize_lse(global_best_a, n_fin, S_fin, N_fin, remaining, rng,
                                         beta_start=30.0, beta_end=1000.0, m_lbfgs=15,
                                         use_toppk=True, topk_start=10, topk_end=1)
        score_fin = 2.0 * n_fin * cmax_fin / (S_fin * S_fin)
        if score_fin < global_best_score:
            global_best_score = score_fin
            global_best_a = a_fin.copy()
            print("  Final polish: score=%.6f" % score_fin)
        else:
            print("  Final polish: no improvement (score=%.6f)" % score_fin)

    # Verification
    if global_best_a is not None:
        final_list = [float(x) for x in global_best_a]
        try:
            ver = evaluate_sequence(final_list)
            print("Verification score: %.6f" % ver)
            if not np.isfinite(ver):
                print("  Verification failed, using fallback")
                global_best_a = np.ones(256, dtype=np.float64)
                global_best_n = 256
        except Exception as e:
            print("  Verification error: %s" % str(e))

    # Final output
    final_list = [float(max(0.0, min(1000.0, x))) for x in global_best_a]
    print("Returning: length=%d, internal_score=%.6f" % (len(final_list), global_best_score))
    return final_list
