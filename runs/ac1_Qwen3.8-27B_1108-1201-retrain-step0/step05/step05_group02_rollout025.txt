"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Algorithm: Multi-phase optimization combining:
1. LSE-FISTA with adaptive beta and noise injection
2. Mirror descent (multiplicative weight updates) 
3. Direct peak sculpting with lookahead

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(w_sorted)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w_sorted
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_smooth_max_and_grad(c, beta):
    """Compute LSE and its gradient weights."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        lse_val = cm
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    return lse_val, w


def grad_from_weights(a, w):
    """Compute gradient of sum_k w[k]*b[k] w.r.t. a.
    grad[j] = 2 * sum_m a[m] * w[j+m] (cross-correlation)
    """
    n = len(a)
    # w has length 2n-1
    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[j+m] for j+m < 2n-1
    # This is the cross-correlation of a and w (with w indexed from 0)
    N = 1 << (2 * n - 1 - 1 + n - 1).bit_length()  # enough for the correlation
    N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    # cross-correlation: ifft(conj(A) * W) gives sum_m a[m]*w[m+j]... 
    # Actually: ifft(conj(fft(a)) * fft(w))[k] = sum_j a[j] * w[j+k]
    # That's what we want: grad[j] = 2 * sum_m a[m] * w[m+j]
    corr = np.real(ifft(np.conj(A) * W))[:n]
    return 2.0 * corr


def run_lse_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.008,
                  beta_max=50000.0, use_backtrack=True, noise_start=3000):
    """LSE-FISTA with adaptive beta, backtracking, and noise injection."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        lse_val, w = lse_smooth_max_and_grad(fft_convolve_self(y), beta)
        grad = grad_from_weights(y, w)
        
        cur_c = fft_convolve_self(y)
        cur_obj = 2.0 * n * cur_c.max() / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-14 and bt < 5:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 2:
                step = max(step_local * 0.7, 1e-10)
            else:
                step = min(step * 1.02, 3.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        # Noise injection to escape local minima
        if it - last_improve > noise_start:
            noise_scale = max(1e-7, 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 5000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    return best_a, best_obj

                            best_delta_obj = new_obj
                            best_move = (j, l, delta)

            a[l] += delta

            for k in range(2 * n - 1):
                k_int = k
                a_kj = a[j + delta] if 0 <= k_int - j < n else 0.0  # wrong, need old a
                # This is getting complicated; just recompute periodically

                idx2 = rng.integers(0, n)
                if idx1 != idx2 and a[idx1] > 1e-10:

                    a[idx1] -= delta
                    a[idx2] += delta

def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "asymmetric_linear":
        a = np.linspace(0.1, 1.0, n)
    elif method == "exponential_decay":
        a = np.exp(-1.5 * tt)
    elif method == "asymmetric_bump":
        center = 0.3
        a = np.exp(-5.0 * (tt - center) ** 2)
    elif method == "two_peaks":
        a = np.exp(-20.0 * (tt - 0.2) ** 2) + 0.5 * np.exp(-20.0 * (tt - 0.8) ** 2)
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "sawtooth":
        a = np.abs(tt * 2.0 - 1.0) + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())
