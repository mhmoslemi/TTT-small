    a = a_init.copy()
    a = np.clip(a, 0, None)
    a *= S / a.sum()
    
    best_a = a.copy()
    best_score = np.inf
    lam = 0.5  # softness parameter
    
    while time budget:
        A = fft(a, N)
        power = np.abs(A)**2
        avg_power = power.sum() / N
        
        # Soft flatten: new_mag = power^((1-lam)/2) * avg_power^(lam/2)
        # This pushes power toward the average
        new_power = power ** (1-lam) * avg_power ** lam
        new_mag = np.sqrt(new_power)
        new_phase = np.angle(A)
        A_new = new_mag * np.exp(1j * new_phase)
        
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new *= S / s
        
        # Update
        c = conv(a_new)
        max_c = c.max()
        if max_c < best_score:
            best_score = max_c
            best_a = a_new.copy()
        a = a_new
        
        # Adapt lam
        lam = min(0.99, lam * 1.01)
    
    return best_a

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):

    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2*n-1]
    return c, A

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns (c[2n-1], A)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_gradient(a, N, beta, top_k=None):
    """Gradient of LSE-smoothed peak (or top-k weighted) of conv(a,a)."""
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[idx]
        total = top_vals.sum()
        if total < 1e-30:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
        else:
            weights = np.zeros(len(c))
            for i, k in enumerate(idx):
                weights[k] = top_vals[i] / total
        f_val = (weights * c).sum()
    elif beta < 1e-6:
        weights = np.ones(len(c)) / len(c)
        f_val = c_max
    else:
        shifted = c - c_max
        max_exp = np.max(shifted * beta)
        if max_exp > 50:
            shifted = shifted - max_exp / beta
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            f_val = c_max
        else:
            weights = exp_bc / total
            f_val = c_max + np.log(total) / beta

    fsize = 1 << (3 * n - 2).bit_length()
    w_full = np.zeros(fsize)
    w_full[:len(c)] = weights
    W = fft(w_full)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, f_val, c, c_max


def lbfgs_two_loop(s_hist, y_hist, g):
    """L-BFGS two-loop recursion for search direction."""
    m = len(s_hist)
    n = len(g)
    if m == 0:
        gn = np.linalg.norm(g)
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def spectral_equalization(a_init, n, S, N, max_time, rng, lam_init=0.3, lam_max=0.95):
    """GS-like algorithm: alternate spectrum flattening and time-domain projection."""
    a = np.clip(a_init.copy(), 0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a *= S / s

    best_a = a.copy()
    best_cmax = np.inf
    lam = lam_init
    start = time.monotonic()
    it = 0

    c, _ = compute_conv(a, N)
    best_cmax = c.max()

    while time.monotonic() - start < max_time:
        it += 1
        t_since = time.monotonic() - start
        lam = min(lam_max, lam_init + (lam_max - lam_init) * (t_since / max_time) ** 0.5)

        A = fft(a, N)
        power = np.abs(A) ** 2
        avg_power = power[1:].mean()
        if avg_power < 1e-30:
            break

        ratio = np.ones(N)
        mask = power[1:] > 1e-30
        ratio[1:] = (avg_power / (power[1:] + 1e-30))
        ratio[1:] = np.clip(ratio[1:], 0.01, 100.0)
        new_power = power * (ratio ** lam)
        new_power[0] = power[0]
        new_mag = np.sqrt(new_power)
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new *= S / s_new

        c_new, _ = compute_conv(a_new, N)
        cmax_new = c_new.max()
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
        a = a_new

        if it % 500 == 0:
            score = 2.0 * n * best_cmax / (S * S)
            print("    [SE] it=%d peak=%.6f score=%.6f" % (it, best_cmax, score))

    return best_a, best_cmax


def peak_deflation(a_init, n, S, N, max_time, rng):
    """Directly reduce the dominant peak by adjusting a."""
    a = np.clip(a_init.copy(), 0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a *= S / s

    best_a = a.copy()
    best_cmax = np.inf
    start = time.monotonic()
    it = 0

    c, _ = compute_conv(a, N)
    best_cmax = c.max()

    while time.monotonic() - start < max_time:
        it += 1
        c, _ = compute_conv(a, N)
        cmax = c.max()
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()

        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]

        if contrib.max() - contrib.min() < 1e-15:
            break

        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if i_max == i_min:
            break

        transfer = 0.01 * a[i_max]
        a[i_max] -= transfer
        a[i_min] += transfer
        a = np.maximum(a, 0)
        s_curr = a.sum()
        if s_curr > 1e-15:
            a *= S / s_curr

        if it % 2000 == 0:
            score = 2.0 * n * best_cmax / (S * S)
            print("    [PD] it=%d peak=%.6f score=%.6f" % (it, best_cmax, score))

    return best_a, best_cmax


def lbfgs_lse_optimize(a_init, n, S, N, max_time, rng, beta_init=5.0, beta_max=300.0,
                       m_lbfgs=12, top_k=None, top_k_max=None):
    """L-BFGS on LSE-smoothed or top-k objective."""
    a = np.clip(a_init.copy(), 0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a *= S / s

    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    g, f_val, c, c_max = lse_gradient(a, N, beta, top_k)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    step_scale = 2.0 * S / (n * max(beta, 1.0))
    if top_k is not None:
        step_scale = 2.0 * S / (n * 10.0)
    start_time = time.monotonic()
    it = 0
    stall_count = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        if it % 200 == 0:
            beta = min(beta_max, beta * 1.08)
            if top_k is not None and top_k_max is not None:
                top_k = min(top_k_max, top_k + top_k_max // 100)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            if top_k is not None:
                step_scale = 2.0 * S / (n * 10.0)

        if it % 60 == 0 and len(s_hist) >= m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_two_loop(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        for ls in range(15):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls, _ = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max + 1e-12 or ls == 14:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_max + 1e-12:
                    break
            alpha_ls *= 0.4

        if cmax_new < best_cmax - 1e-10:
            best_cmax = cmax_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        g_new, f_new, c_nw, cmax_nw = lse_gradient(a_new, N, beta, top_k)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * max(ss, 1e-30) and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c_max = cmax_nw

        if stall_count > 200 and it % 100 == 0:
            s_hist = []
            y_hist = []
            stall_count = 0
            if top_k is not None:
                top_k = max(5, top_k // 2)
            beta = max(5.0, beta * 0.5)

    return best_a, best_cmax


def spectral_phase_restart(a, N, rng, frac=1.0):
    """Randomize FFT phases (partially or fully) while preserving magnitudes."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def make_init_spectral(n, N, S, rng):
    """Initialize via random phases in frequency domain."""
    A = np.zeros(N, dtype=np.complex128)
    A[0] = S
    for k in range(1, N):
        A[k] = rng.standard_normal() + 1j * rng.standard_normal()
    A[N - k] = np.conj(A[k]) if k < N // 2 else A[k]
    a = np.real(ifft(A))[:n]
    a = np.abs(a) + 0.01
    a = np.clip(a, 0, None)
    s = a.sum()
    if s > 1e-15:
        a *= S / s
    return a


def make_init_rand(n, S, rng):
    a = rng.random(n) + 0.01
    a *= S / a.sum()
    return a


def make_init_chirp(n, S, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (7 * tt + 3 * tt * tt)) + 0.1 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)

def make_init_asymmetric(n, S, rng):
    """Initialize with asymmetric mass distribution to reduce center peak."""
    decay = rng.uniform(0.5, 2.0)
    weights = np.exp(-decay * np.abs(np.arange(n) - n * 0.3) / n)
    a = weights + 0.05 * rng.random(n)
    a = np.clip(a, 0, None)
    a *= S / a.sum()
    return a


def make_init_two_sided(n, S, rng):
    """Two gaussian bumps away from center."""
    center1 = n * rng.uniform(0.2, 0.4)
    center2 = n * rng.uniform(0.6, 0.8)
    width = n * rng.uniform(0.05, 0.15)
    a = np.exp(-0.5 * ((np.arange(n) - center1) / width) ** 2) + \
        np.exp(-0.5 * ((np.arange(n) - center2) / width) ** 2)
    a += 0.01 * rng.random(n)
    a = np.clip(a, 0, None)
    a *= S / a.sum()
    return a


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses spectral equalization, L-BFGS on LSE/top-k, peak deflation,
    and stochastic spectral restarts.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 1024 and given.size <= 32768:
            n = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    global_best = np.inf
    global_a = None

    def _update_best(a_candidate, label=""):
        nonlocal global_best, global_a
        a_cand = safe_normalize(a_candidate, n, S)
        c, _ = compute_conv(a_cand, N)
        sc = c.max()
        if sc < global_best:
            global_best = sc
            global_a = a_cand.copy()
            score = float(2 * n * sc / (S * S))
            if label:
                print("  [%s] peak=%.6f score=%.6f" % (label, sc, score))
        return sc

    # --- Phase 1: Spectral Equalization (fast convergence) ---
    print("=== Phase 1: Spectral Equalization ===")
    t1 = time.monotonic()
    phase1_budget = min(600.0, (deadline - t1) * 0.20)

    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
    except Exception:
        pass
    inits.append(("rand", make_init_rand(n, S, rng)))
    inits.append(("chirp", make_init_chirp(n, S, rng)))
    inits.append(("asym", make_init_asymmetric(n, S, rng)))
    inits.append(("twoside", make_init_two_sided(n, S, rng)))
    inits.append(("spectral", make_init_spectral(n, N, S, rng)))
    inits.append(("spectral2", make_init_spectral(n, N, S, rng)))

    n_inits = len(inits)
    per_init_time = phase1_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init_time))

    for name, arr in inits:
        if time.monotonic() > deadline - 400:
            break
        t0 = time.monotonic()
        rem = max(5.0, per_init_time - (time.monotonic() - t0) - 0.5)
        a_se, sc_se = spectral_equalization(arr, n, S, N, min(rem, 120.0), rng,
                                            lam_init=0.3, lam_max=0.9)
        _update_best(a_se, name + "/SE")

    print("Phase 1 done. Best score: %.6f" % (2 * n * global_best / (S * S) if global_best < np.inf else np.inf))

    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = 2.0

    # --- Phase 2: L-BFGS with top-K gradient ---
    print("=== Phase 2: L-BFGS top-K optimization ===")
    t2 = time.monotonic()
    phase2_budget = min(800.0, (deadline - t2) * 0.35)

    a_curr = global_a.copy()
    stall = 0

    while time.monotonic() < t2 + phase2_budget:
        block_time = min(200.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 15:
            break

        top_k_start = 10 + 5 * min(stall, 8)
        top_k_end = min(200, 50 + 20 * stall)
        a_new, sc_new = lbfgs_lse_optimize(a_curr, n, S, N, block_time, rng,
                                            beta_init=10.0, beta_max=200.0,
                                            m_lbfgs=12, top_k=top_k_start, top_k_max=top_k_end)
        if sc_new < global_best - 1e-9:
            _update_best(a_new, "P2/toppk")
            a_curr = a_new.copy()
            stall = 0
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall == 3:
                a_curr = spectral_phase_restart(a_curr, N, rng, frac=0.3)
                stall = 0
            elif stall >= 6:
                a_curr = spectral_phase_restart(a_curr, N, rng, frac=0.7)
                a_curr = safe_normalize(a_curr, n, S)
                stall = 0

    # --- Phase 3: Peak deflation refinement ---
    print("=== Phase 3: Peak deflation ===")
    t3 = time.monotonic()
    remaining3 = min(300.0, (deadline - t3) * 0.15)
    if remaining3 > 20:
        a_pd, sc_pd = peak_deflation(global_a, n, S, N, remaining3, rng)
        _update_best(a_pd, "peak_deflation")

    # --- Phase 4: High-beta L-BFGS polish ---
    print("=== Phase 4: High-beta L-BFGS polish ===")
    t4 = time.monotonic()
    remaining4 = deadline - t4 - 20
    if remaining4 > 30:
        a_fin, sc_fin = lbfgs_lse_optimize(global_a, n, S, N, min(remaining4 * 0.6, 400.0),
                                            rng, beta_init=30.0, beta_max=500.0,
                                            m_lbfgs=15, top_k=None, top_k_max=None)
        _update_best(a_fin, "final_LBFGS")

    # --- Phase 5: Alternating short blocks ---
    remaining5 = deadline - time.monotonic() - 10
    if remaining5 > 30:
        third = remaining5 / 3.0
        a_tk, sc_tk = lbfgs_lse_optimize(global_a, n, S, N, min(third, 150.0),
                                          rng, beta_init=50.0, beta_max=600.0,
                                          m_lbfgs=10, top_k=5, top_k_max=30)
        _update_best(a_tk, "top5_polish")

        if deadline - time.monotonic() > 20:
            a_se2, sc_se2 = spectral_equalization(global_a, n, S, N,
                                                   min(third, 150.0), rng,
                                                   lam_init=0.7, lam_max=0.95)
            _update_best(a_se2, "SE_polish")

        if deadline - time.monotonic() > 20:
            a_fin2, sc_fin2 = lbfgs_lse_optimize(global_a, n, S, N,
                                                  min(deadline - time.monotonic() - 5, 200.0),
                                                  rng, beta_init=80.0, beta_max=800.0,
                                                  m_lbfgs=15, top_k=None, top_k_max=None)
            _update_best(a_fin2, "final2")

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("Verification: inf - using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    final_score = 2.0 * len(global_a) * global_best / (S * S) if global_best < np.inf else float('inf')
    print("Returning length %d, internal peak=%.6f, approx score=%.6f" % (len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
