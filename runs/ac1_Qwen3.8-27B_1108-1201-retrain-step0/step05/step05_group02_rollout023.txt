"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: 
1. Asymmetric + diverse initialization with quick FISTA screening
2. Adaptive LSE-FISTA with backtracking
3. Frank-Wolfe refinement with active-set subgradients
4. Multi-scale upsample (zero-stuffing + interpolation)
5. High-precision final polish
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_grad(y, beta):
    """Compute LSE value, gradient, and convolution."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        lse_val = cm
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c

        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_grad(y, beta):
    """Compute LSE value, gradient, and convolution of y with itself."""
    n = len(y)
    c = fft_convolve_self(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        lse_val = cm
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A2 = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A2)))[:n]
    return lse_val, grad, c

            last_improve = it
            consecutive_bad = 0

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            c_trial = fft_convolve_self(x_trial)
            obj_trial = 2.0 * n * c_trial.max() / (S * S)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-14 and bt < 8:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                c_trial = fft_convolve_self(x_trial)
                obj_trial = 2.0 * n * c_trial.max() / (S * S)
                bt += 1
            if bt > 4:
                step = step_local * 0.7
                consecutive_bad += 1
            else:
                step = min(step * 1.04, 5.0 / max(beta, 1.0))
                if bt <= 1:
                    consecutive_bad = max(0, consecutive_bad - 1)
            x_new = x_trial
        else:
            step = step_scale / max(beta, 1.0)
            x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum with restart on bad steps
        if consecutive_bad > 6:
            t_mom = 1.0
            consecutive_bad = 0
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        stuck = it - last_improve
        if stuck > 5000:
            break
        elif stuck > 2000:
            t_mom = 1.0
            noise_scale = max(1e-7, 0.004 * S / np.sqrt(n) * np.exp(-stuck / 4000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    return best_a, best_obj


def peel_step(a, max_time, rng, n_peels=500):
    """
    Targeted peeling: transfer mass from high-contribution pairs to reduce peak lags.
    Operates directly on the convolution.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time and it < n_peels:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 200:
            break

        # Find top lags
        top_k = min(3, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]

        # Compute "danger score" for each position: how much it contributes to top lags
        danger = np.zeros(n)
        for kk in top_idx:
            kk = int(kk)
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            danger[lo:hi + 1] += a[kk - hi:kk - lo + 1][::-1]

        # Source: positions with high danger AND high value
        # Target: positions with low danger AND low value
        src_score = danger * np.maximum(a, 1e-10)
        tgt_score = danger * (S / n - a + 1e-10)

        # Try several candidate transfers
        n_src = min(n, max(5, n // 500))
        n_tgt = min(n, max(5, n // 500))
        src_cands = np.argsort(src_score)[::-1][:n_src]
        tgt_cands = np.argsort(tgt_score)[:n_tgt]

        best_gain = 0.0
        best_j, best_l, best_delta = -1, -1, 0.0

        for j in src_cands[:min(n_src, 10)]:
            for l in tgt_cands[:min(n_tgt, 10)]:
                j, l = int(j), int(l)
                if j == l or a[j] < 1e-10:
                    continue
                # Try transferring fraction of a[j]
                for frac in [0.03, 0.08, 0.15]:
                    delta = a[j] * frac
                    if delta < 1e-12:
                        continue
                    # Compute change in top lags
                    # Δc[k] = 2*delta*(a[k-j] - a[k-l]) for the transfer j->l (reducing j, increasing l)
                    # Actually: reducing a[j] by delta, increasing a[l] by delta
                    # Δc[k] = -2*delta*a[k-j] + 2*delta*a[k-l] = 2*delta*(a[k-l] - a[k-j])
                    max_new = cur_max
                    improved = True
                    for kk in top_idx[:2]:  # only check top 2 for speed
                        kk = int(kk)
                        val_j = a[kk - j] if 0 <= kk - j < n else 0.0
                        val_l = a[kk - l] if 0 <= kk - l < n else 0.0
                        new_ck = c[kk] + 2.0 * delta * (val_l - val_j)
                        if new_ck > max_new + 1e-12:
                            # Check if overall max increases
                            if new_ck > max_new:
                                max_new = new_ck
                    # Also check the overall max approximately
                    if max_new < cur_max - 1e-14:
                        gain = cur_max - max_new
                        if gain > best_gain:
                            best_gain = gain
                            best_j, best_l, best_delta = j, l, delta

        if best_j >= 0:
            j, l, delta = best_j, best_l, best_delta
            a[j] -= delta
            a[l] += delta
            # Update c incrementally
            # Reducing a[j] by delta: c[k] -= 2*delta*a[k-j] for valid k
            for k in range(2 * n - 1):
                idx = k - j
                if 0 <= idx < n:
                    c[k] -= 2.0 * delta * a[idx]
            # Increasing a[l] by delta: c[k] += 2*delta*a[k-l] for valid k
            for k in range(2 * n - 1):
                idx = k - l
                if 0 <= idx < n:
                    c[k] += 2.0 * delta * a[idx]
            # Clip small negatives from numerical error
            c = np.maximum(c, 0.0)
        else:
            if stuck > 50:
                # Random perturbation to escape
                idx = rng.integers(0, n)
                delta = a[idx] * 0.02
                if delta > 1e-12:
                    a[idx] -= delta
                    a[rng.integers(0, n)] += delta
                    c = fft_convolve_self(a)
                else:
                    break

        if it % 50 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def frank_wolfe_refine(a_init, max_time, rng, n_active=8):
    """Frank-Wolfe with line search and active-set subgradients."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 3000:
            break

        # Active lags
        threshold = cur_max * (0.95 + 0.04 * min(1.0, stuck / 1000.0))
        active_mask = c >= threshold
        active_idx = np.where(active_mask)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c))])
        if len(active_idx) > n_active:
            active_idx = active_idx[np.argsort(c[active_idx])[::-1][:n_active]]

        # Combined subgradient
        grad = np.zeros(n)
        for k in active_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]

        # FW direction: vertex S*e_i where i minimizes grad[i]
        i_min = int(np.argmin(grad))

        # Line search for step size
        best_gamma = 0.0
        best_obj_ls = cur_obj
        for gamma in [0.5, 0.3, 0.2, 0.1, 0.05, 0.01]:
            v = np.zeros(n)
            v[i_min] = S
            a_try = (1 - gamma) * a + gamma * v
            a_try = np.maximum(a_try, 0.0)
            a_try = a_try * (S / a_try.sum())
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / (S * S)
            if obj_try < best_obj_ls:
                best_obj_ls = obj_try
                best_gamma = gamma

        if best_gamma > 0:
            v = np.zeros(n)
            v[i_min] = S
            a = (1 - best_gamma) * a + best_gamma * v
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
        else:
            # No improvement from single vertex; try small random moves
            if stuck > 500:
                break

    return best_a, best_obj


def upsample_interp(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def upsample_zero(a, factor=2):
    """Upsample via zero-stuffing."""
    n = len(a)
    n_new = n * factor
    b = np.zeros(n_new)
    b[::factor] = a
    return b


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian_center":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.1 ** 2))
    elif method == "gaussian_left":
        a = np.exp(-2.0 * (tt - 0.3) ** 2 / (0.12 ** 2))
    elif method == "gaussian_right":
        a = np.exp(-2.0 * (tt - 0.7) ** 2 / (0.12 ** 2))
    elif method == "ramp_up":
        a = tt + 0.1
    elif method == "ramp_down":
        a = (1.0 - tt) + 0.1
    elif method == "exp_decay":
        a = np.exp(-3.0 * tt)
    elif method == "exp_rise":
        a = np.exp(-3.0 * (1.0 - tt))
    elif method == "chirp":
        a = 0.3 + 0.7 * np.abs(np.sin(2 * np.pi * 5 * tt + 1.0))
    elif method == "multi_tone":
        a = (0.2 + 0.3 * np.sin(2 * np.pi * 3 * tt) +
             0.25 * np.sin(2 * np.pi * 7 * tt + 1.3) +
             0.25 * np.sin(2 * np.pi * 13 * tt + 2.1))
        a = np.abs(a) + 0.05
    elif method == "two_bump":
        a = (np.exp(-2.0 * (tt - 0.25) ** 2 / (0.08 ** 2)) +
             np.exp(-2.0 * (tt - 0.75) ** 2 / (0.08 ** 2)))
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "bump_offset":
        a = np.maximum(0.05, 1.0 - 2.5 * np.abs(tt - 0.35))
    elif method == "sawtooth":
        a = 0.3 + 0.7 * (tt % 0.5) * 2.0
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: asymmetric diverse init -> FISTA -> peeling -> upsample -> polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point from provided sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Starting from provided: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        a_start = make_init(n0, "gaussian_left", rng)
        obj0 = compute_obj(a_start)
        print("No provided sequence. Starting with n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: Diverse asymmetric screening (20% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.20
    if p1_time > 60:
        print("\n=== Phase 1: Diverse starts (budget %.0fs) ===" % p1_time)
        methods = ["gaussian_left", "gaussian_right", "ramp_up", "exp_decay",
                   "two_bump", "bump_offset", "chirp", "sawtooth"]
        per_time = p1_time / len(methods)

        for method in methods:
            if time.monotonic() >= deadline:
                break
            a_div = make_init(n0, method, rng)
            a_opt, obj_opt = run_fista(a_div, per_time * 0.65, rng,
                                        beta_init=10, beta_rate=0.02,
                                        beta_max=20000, step_scale=0.8,
                                        use_backtrack=False)
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()
                print("  %s: obj=%.6f ***NEW BEST***" % (method, obj_opt))
            else:
                print("  %s: obj=%.6f" % (method, obj_opt))

    # === Phase 2: Intensive FISTA on best (25% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.25
    if p2_time > 90:
        print("\n=== Phase 2: Intensive FISTA (budget %.0fs) ===" % p2_time)

        variants = [
            (15, 0.02, 20000, 1.0, True),
            (50, 0.01, 50000, 0.7, True),
            (200, 0.005, 200000, 0.4, True),
            (10, 0.025, 30000, 1.2, False),
        ]
        per_variant = p2_time / len(variants)
        for i, (bi, br, bm, ss, bt) in enumerate(variants):
            if time.monotonic() >= deadline:
                break
            rem = min(per_variant, deadline - time.monotonic())
            if rem < 10:
                break
            a_opt, obj_opt = run_fista(global_best_a, rem, rng,
                                        beta_init=bi, beta_rate=br,
                                        beta_max=bm, step_scale=ss,
                                        use_backtrack=bt)
            print("  FISTA%d (beta_init=%d): obj=%.6f" % (i + 1, bi, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 3: Peeling refinement (25% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.25
    if p3_time > 60:
        print("\n=== Phase 3: Peeling (budget %.0fs) ===" % p3_time)
        # Split time for multiple rounds of peel+FISTA
        n_rounds = 3
        per_round = p3_time / n_rounds
        for r in range(n_rounds):
            if time.monotonic() >= deadline:
                break
            a_peel, obj_peel = peel_step(global_best_a, per_round * 0.5, rng,
                                          n_peels=300)
            if obj_peel < global_best_obj:
                global_best_obj = obj_peel
                global_best_a = a_peel.copy()
                print("  Peel round %d: obj=%.6f" % (r + 1, obj_peel))
            else:
                print("  Peel round %d: obj=%.6f" % (r + 1, obj_peel))

            remaining3 = deadline - time.monotonic()
            if remaining3 > 15:
                a_f, obj_f = run_fista(global_best_a, remaining3 * 0.5, rng,
                                        beta_init=100, beta_rate=0.008,
                                        beta_max=100000, step_scale=0.5,
                                        use_backtrack=True)
                if obj_f < global_best_obj:
                    global_best_obj = obj_f
                    global_best_a = a_f.copy()
                print("  FISTA after peel %d: obj=%.6f" % (r + 1, obj_f))

    # === Phase 4: Multi-scale upsample (20% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.20
    if p4_time > 60 and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        print("\n=== Phase 4: Upsample to n=%d (budget %.0fs) ===" % (n2, p4_time))

        # Try both interpolation and zero-stuffing
        a_up1 = upsample_interp(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up1 = a_up1 * (S2 / a_up1.sum())
        obj_up1 = compute_obj(a_up1)

        a_up2 = upsample_zero(global_best_a, factor=2)
        a_up2 = a_up2 * (S2 / a_up2.sum())
        obj_up2 = compute_obj(a_up2)

        print("  Interp upsample: obj=%.6f" % obj_up1)
        print("  Zero upsample: obj=%.6f" % obj_up2)

        a_up_best = a_up1 if obj_up1 <= obj_up2 else a_up2
        obj_up_best = min(obj_up1, obj_up2)
        if obj_up_best < global_best_obj:
            global_best_obj = obj_up_best
            global_best_a = a_up_best.copy()

        # Optimize the upsampled version
        rem4 = deadline - time.monotonic()
        if rem4 > 20:
            a_up_opt, obj_up_opt = run_fista(global_best_a, rem4 * 0.5, rng,
                                              beta_init=5, beta_rate=0.012,
                                              beta_max=30000, step_scale=0.8,
                                              use_backtrack=False)
            print("  FISTA upsampled: obj=%.6f" % obj_up_opt)
            if obj_up_opt < global_best_obj:
                global_best_obj = obj_up_opt
                global_best_a = a_up_opt.copy()

            rem4b = deadline - time.monotonic()
            if rem4b > 15:
                a_peel_up, obj_peel_up = peel_step(global_best_a, rem4b * 0.6, rng,
                                                    n_peels=200)
                print("  Peel upsampled: obj=%.6f" % obj_peel_up)
                if obj_peel_up < global_best_obj:
                    global_best_obj = obj_peel_up
                    global_best_a = a_peel_up.copy()

    # === Phase 5: Frank-Wolfe refinement (8% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.08
    if p5_time > 20:
        print("\n=== Phase 5: Frank-Wolfe (budget %.0fs) ===" % p5_time)
        a_fw, obj_fw = frank_wolfe_refine(global_best_a, p5_time, rng, n_active=8)
        print("  Frank-Wolfe: obj=%.6f" % obj_fw)
        if obj_fw < global_best_obj:
            global_best_obj = obj_fw
            global_best_a = a_fw.copy()

    # === Phase 6: Final high-precision polish (remaining time) ===
    remaining = deadline - time.monotonic()
    p6_time = remaining * 0.85
    if p6_time > 10:
        print("\n=== Phase 6: Final polish (budget %.0fs) ===" % p6_time)

        # Very high beta FISTA
        a_pol, obj_pol = run_fista(global_best_a, p6_time * 0.6, rng,
                                    beta_init=500, beta_rate=0.003,
                                    beta_max=500000, step_scale=0.3,
                                    use_backtrack=True)
        print("  High-beta FISTA: obj=%.6f" % obj_pol)
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()

        remaining6 = deadline - time.monotonic()
        if remaining6 > 10:
            a_pol2, obj_pol2 = run_fista(global_best_a, remaining6 * 0.5, rng,
                                          beta_init=2000, beta_rate=0.001,
                                          beta_max=1000000, step_scale=0.15,
                                          use_backtrack=True)
            print("  Ultra-high beta: obj=%.6f" % obj_pol2)
            if obj_pol2 < global_best_obj:
                global_best_obj = obj_pol2
                global_best_a = a_pol2.copy()

        # Final peel
        remaining6b = deadline - time.monotonic()
        if remaining6b > 10:
            a_final_peel, obj_final_peel = peel_step(global_best_a, remaining6b * 0.7,
                                                      rng, n_peels=500)
            print("  Final peel: obj=%.6f" % obj_final_peel)
            if obj_final_peel < global_best_obj:
                global_best_obj = obj_final_peel
                global_best_a = a_final_peel.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.6f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
