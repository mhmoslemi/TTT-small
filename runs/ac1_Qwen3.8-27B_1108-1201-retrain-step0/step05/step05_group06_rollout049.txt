        # Randomly select coordinates to try
        n_try = min(50, n // 100)
        indices = rng.choice(n, size=n_try, replace=False)

        improved = False
        for i in indices:
            orig_val = a[i]
            if orig_val < 1e-15:
                continue

            # Try reducing a[i]
            for frac in [0.1, 0.05, 0.01]:
                test_a = a.copy()
                test_a[i] = orig_val * (1.0 - frac)
                # Redistribute to maintain sum
                deficit = orig_val * frac
                test_a = test_a + deficit / n
                tc = fft_convolve_self(test_a)
                if tc.max() < best_conv - 1e-12:
                    a = test_a
                    c = tc
                    best_conv = tc.max()
                    best_a = a.copy()
                    improved = True
                    break

            # Try increasing a[i] (steal from others)
            if not improved and orig_val > 0:
                for frac in [0.05, 0.02]:
                    test_a = a.copy()
                    test_a[i] = orig_val * (1.0 + frac)
                    surplus = orig_val * frac
                    test_a = test_a - surplus / n
                    test_a = np.maximum(test_a, 0)
                    test_a = normalize_seq(test_a, S)
                    tc = fft_convolve_self(test_a)
                    if tc.max() < best_conv - 1e-12:
                        a = test_a
                        c = tc
                        best_conv = tc.max()
                        best_a = a.copy()
                        break

    return best_a


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)


def make_random_init(n, rng):
    a = rng.random(n)
    a = 0.3 + 0.7 * a
    # Add some structure
    freq = rng.integers(1, 5)
    phase = rng.uniform(0, 2 * np.pi)
    a = a * (1.0 + 0.3 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    return np.maximum(a, 0.01)


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)
