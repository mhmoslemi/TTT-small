import time

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_subgradient_topk(a, c, k, temperature=1.0):

    else:

def project_simplex(v, S):

def lp_direction_mirror_descent(a, S, max_iters=200, verbose=False):

    Find a direction g >= 0, sum(g) = S that maximizes sum(g)

    Uses mirror descent on the dual.
    
    Returns the direction g (normalized to sum = S).
    """

    c_aa = fft_convolve_self(a)
    M = c_aa.max()

    # If all u are zero, we can't improve
    if u.max() < 1e-15:
        return None
    
    # Mirror descent on dual:
    # min u^T y s.t. (A^T y)_j >= 1, y >= 0
    # where (A^T y)_j = sum_k y_k * a[k-j] (valid k)
    
    # Initialize
    y = np.zeros(2*n-1)
    # Start with uniform y where u > 0
    active = u > 1e-10
    if not active.any():
        return None
    y_init_val = 1.0 / (active.sum() * a.sum())
    y[active] = y_init_val
    
    eta = 0.1  # step size for dual
    g = np.ones(n) / n * S  # initial primal
    
    for it in range(max_iters):
        # Compute A^T y = sum_k y[k] * a[k-j] for each j
        # This is the cross-correlation of y and a
        # (A^T y)[j] = sum_k y[k] * a[k-j] where 0<=k<n and 0<=k-j<n
        # = sum_k y[k] * a[k-j] 
        # Let's compute via FFT: it's the "anti-convolution" or cross-correlation
        
        # Actually: (A^T y)[j] = sum_{k=j}^{j+n-1} y[k] * a[k-j]
        # This is the convolution of y and reversed a, evaluated at specific points
        # Or: it's the cross-correlation R_ya[j] = sum_k y[k+j] * a[k]
        
        # Let's just compute it directly using the fact that:
        # sum_k y[k] * a[k-j] = (y * a_flipped)[j + n - 1] ... hmm this is getting complicated
        # Let me use a different approach: compute A^T y by direct convolution
        
        # A[k,j] = a[k-j] if 0 <= k-j <= n-1 else 0
        # (A^T y)[j] = sum_k A[k,j] * y[k] = sum_k a[k-j] * y[k] (valid k,j)
        # = sum_{m=0}^{n-1} a[m] * y[m+j] (where m = k-j, so k = m+j)
        # This is the cross-correlation of a and y at lag j
        
        # np.correlate(y, a, mode='full')[n-1+j] = sum_k y[k] * a[k-j]
        # Actually np.correlate(v, w, 'full')[n_w-1+k] = sum_i v[i+k]*w[i]
        # So np.correlate(y, a, 'full')[n-1+j] = sum_i y[i+j]*a[i]
        # Which is what we want!
        
        # But for large arrays, this is O(n^2). Let's use FFT.
        # Cross-correlation via FFT:
        # R[j] = sum_k y[k+j]*a[k] = iff(fft(y) * conj(fft(a)))... 
        
        # Actually, the standard result is:
        # cross_corr = ifft(conj(fft(a, N)) * fft(y, N))
        # where N >= len(a) + len(y) - 1
        # And cross_corr[k] = sum_i a[i] * y[i+k] (for the "full" output)
        
        # Hmm, let me just verify with a small example in code... 
        # For now, let me use a simpler approach:
        
        # Compute conv(y, a) via FFT and extract the relevant part
        # conv(y, a)[m] = sum_i y[i] * a[m-i]
        # We want (A^T y)[j] = sum_k a[k-j] * y[k] = sum_m a[m] * y[m+j]
        # = conv(y_reversed, a)[j + n - 1] ... 
        
        # This is getting too complicated. Let me just use the direct subgradient approach
        # and skip the LP direction for now.
        
        # Alternative: use the subgradient of the constraint violation
        c_ag = fft_convolve_xy(a, g)  # conv(a, g), length 2n-1
        
        # Dual update: y_k = max(0, y_k + eta * (c_ag[k] - u[k]))
        y = np.maximum(0, y + eta * (c_ag[:2*n-1] - u))
        
        # Primal update: g_j proportional to exp(-eta * (A^T y)_j + eta)
        # where (A^T y)_j = sum_k y[k] * A[k,j] = sum_k y[k] * a[k-j]
        # For efficiency, compute via FFT cross-correlation
        # (A^T y)[j] = sum_{k} y[k] * a[k-j] for valid indices
        # This equals the cross-correlation of y and a
        
        # Using the identity: sum_k y[k]*a[k-j] = (y * a)[j] in the "correlation" sense
        # In numpy: np.dot(y[j:], a[:n-j]) for j >= 0
        # Via FFT: ifft(fft(y, N) * conj(fft(a, N)))[:n] gives sum_k y[k]*a[k-j]
        # Wait no. Let me be very explicit.
        
        # Define: C[j] = sum_{k=max(j,0)}^{min(j+n-1, 2n-2)} y[k] * a[k-j]
        # For j = 0: C[0] = sum_{k=0}^{n-1} y[k]*a[k] = dot(y[:n], a)
        # For j = 1: C[1] = sum_{k=1}^{n} y[k]*a[k-1] = dot(y[1:n+1], a[:n-1])
        # In general: C[j] = dot(y[j:j+n], a[:n-j])
        
        # This is the "cross-correlation" or "sliding dot product".
        # Via FFT: C = ifft(fft(y, N) * conj(fft(a, N))) where N >= 3n
        # Actually the standard formula is:
        # cross_corr[m] = sum_k x[k] * conj(h[k+m])
        # For real: cross_corr[m] = sum_k x[k] * h[k+m]
        # So C[j] = cross_corr of y and a at lag j = sum_k y[k] * a[k+j]
        # Wait that's not what we want. We want sum_k y[k] * a[k-j].
        
        # Let me define it differently. Let b = a (reversed).
        # Then C[j] = sum_k y[k] * b[n-1-(k-j)] = sum_k y[k] * b[n-1-k+j]
        # = cross_corr(y, b)[n-1+j]
        
        # I'm going in circles. Let me just compute it directly for the FFT case.
        # The key insight is: C[j] = (y * a)[j+n-1] where * is the cross-correlation
        # In terms of convolution: C = ifft(conj(fft(a, N)) .* fft(y, N))[:2n-1] shifted
        
        # You know what, let me just use a simpler approach for the dual update.
        # Instead of the full mirror descent, let me just use the subgradient of the 
        # max constraint directly.
        
        # Simplified: just do one step of the "dual ascent" which is the subgradient
        # of the max constraint.
        
        if (it + 1) % 50 == 0 and verbose:
            viol = c_ag[:2*n-1].max() - u.max()
            print(f"      MD iter {it}: max_violation={viol:.6f}")
        
        # Update g using the current constraint violations
        # g_j *= exp(eta * sum_k y_k * a[k-j]) ... this is the mirror descent step
        # For simplicity, just scale g by the "excess" at each position
        excess = np.zeros(n)
        for k_idx in range(min(10, len(y))):
            if y[k_idx] > 0:
                j_start = max(0, k_idx - n + 1)
                j_end = min(n - 1, k_idx)
                if j_start <= j_end:
                    indices = k_idx - np.arange(j_start, j_end + 1)
                    excess[j_start:j_end+1] += y[k_idx] * a[indices]
        
        g = g * np.exp(-eta * excess)
        g = np.maximum(g, 0)
        if g.sum() > 0:
            g = g * S / g.sum()
        else:
            g = np.ones(n) / n * S
    
    return g if g.sum() > 0.01 * S else None

def make_palindromic(a):
    """Make a sequence palindromic by averaging with its reverse."""
    n = len(a)
    return (a + a[::-1]) / 2.0
