def fft_convolve_self(a):
    n = len(a)

    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def compute_lse_weights(c, k, beta):
    """Compute top-k LSE value and weights."""
    n = len(c)
    k = min(k, n)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    # Prevent overflow
    max_scaled = scaled.max()
    if max_scaled > 500:
        scale = 500.0 / max_scaled
        beta_eff = beta * scale
        scaled = beta_eff * (c_top - cm)
    else:
        beta_eff = beta
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    w_full = np.zeros(n)
    w_full[idx] = w
    return w_full, idx

def compute_gradient_a(a, w_full):
    """Compute gradient of weighted sum of c w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def compute_gradient_b(a, grad_a):
    """Convert gradient w.r.t. a to gradient w.r.t. log-space b."""
    h = np.dot(a, grad_a)
    return a * (grad_a - h)

def softmax_b_to_a(b, S):
    """Convert log-space b to a = S * softmax(b)."""
    b_shifted = b - b.max()
    e = np.exp(b_shifted)
    a = S * e / e.sum()
    return a

def run_logspace_fista(a_init, S, max_time, rng, beta_init=20.0, beta_rate=0.002, beta_max=5000.0, k_frac=0.08):
    """FISTA in log-space parametrization."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    # Initialize b from a: b = log(a) (approximately)
    b = np.log(np.maximum(a, 1e-30))
    b = b - b.mean()  # Center for numerical stability
    
    y_b = b.copy()
    y_a = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0
    
    k = max(4, int(k_frac * (2 * n - 1)))
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y_a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y_a.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1
        
        w_full, idx = compute_lse_weights(c, k, beta)
        grad_a = compute_gradient_a(y_a, w_full)
        grad_b = compute_gradient_b(y_a, grad_a)
        
        gnorm = np.linalg.norm(grad_b)
        if gnorm < 1e-14:
            break
        
        # Step in b-space
        step = 1.0 / (beta * 0.5)  # Adaptive step
        b_new = y_b - step * grad_b
        # Numerical stability: recenter b
        b_new = b_new - b_new.mean()
        a_new = softmax_b_to_a(b_new, S)
        
        # Check if this is an improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() >= cur_max + 1e-10:
            # Try smaller step
            for _ in range(5):
                step *= 0.5
                b_new = y_b - step * grad_b
                b_new = b_new - b_new.mean()
                a_new = softmax_b_to_a(b_new, S)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < cur_max + 1e-10:
                    break
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_b_new = b_new + ((t_val - 1.0) / t_new) * (b_new - b)
        y_b_new = y_b_new - y_b_new.mean()
        
        b = b_new
        y_b = y_b_new
        y_a = softmax_b_to_a(y_b, S)
        t_val = t_new
        
        # Periodic restart
        if stuck > 2000:
            noise = rng.normal(0, 0.05, n)
            y_b = y_b + noise
            y_b = y_b - y_b.mean()
            y_a = softmax_b_to_a(y_b, S)
            t_val = 1.0
            stuck = 0
        
        # Periodic recentering
        if it % 1000 == 0:
            b = b - b.mean()
            y_b = y_b - y_b.mean()
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

def run_fista_simplex(a_init, S, max_time, rng, beta_init=20.0, beta_rate=0.003, beta_max=5000.0, k_frac=0.08):
    """Standard FISTA with simplex projection (similar to previous code)."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0
    
    k = max(4, int(k_frac * (2 * n - 1)))
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1
        
        w_full, idx = compute_lse_weights(c, k, beta)
        grad = compute_gradient_a(y, w_full)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(8):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        
        a = x_new
        y = y_new
        t_val = t_new
        
        if stuck > 1500:
            noise_scale = 0.004 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0
        
        if it % 2000 == 0 and rng.random() < 0.2:
            y = spectral_perturb(y, 0.08, rng)
            t_val = 1.0
            stuck = 0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

def spectral_perturb(a, amount, rng):
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (a.sum() / s)

def run_polish(a_init, S, max_time, rng):
    """High-beta polishing with careful step control."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    k = max(4, int(0.02 * (2 * n - 1)))
    it = 0
    step_scale = 1.0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(50000.0, 500.0 * (1.0 + it / 500.0))
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        w_full, idx = compute_lse_weights(c, k, beta)
        grad = compute_gradient_a(a, w_full)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        base_step = 0.3 / (1.0 + 0.0005 * it)
        step = base_step * step_scale
        
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() < cur_max:
            a = a_new
            step_scale = min(2.0, step_scale * 1.03)
        else:
            step_scale = max(0.01, step_scale * 0.5)
            a = a_new
        
        if it % 3000 == 0 and (time.monotonic() - start) / max_time < 0.5:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best

def resample_sequence(a, n_target):
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)

def run_adam(a_init, S, max_time, rng, lr_init=0.02, beta_adam=50.0, beta_rate=0.01):
    """Adam optimizer with adaptive LSE gradient."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    
    k = max(4, int(0.06 * (2 * n - 1)))
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        beta = min(3000.0, beta_adam * np.exp(beta_rate * t))
        w_full, idx = compute_lse_weights(c, k, beta)
        grad = compute_gradient_a(a, w_full)
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        progress = min(1.0, (time.monotonic() - start) / max_time)
        lr = lr_init * (1.0 - 0.8 * progress)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = project_simplex(a_new, S)
        
        if t % 5000 == 0 and rng.random() < 0.15:
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                m = np.zeros(n)
                v = np.zeros(n)
    
    return best_a, best
