def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences using exact subgradient
method with adaptive active set, backtracking line search, momentum, and multi-scale
optimization. Key innovations: (1) exact subgradient instead of LSE smoothing for
higher accuracy near the optimum, (2) adaptive active set that grows/shrinks based
on the spectral gap, (3) multi-resolution strategy to escape local minima.

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection on Lagrange multiplier."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    # Bisection method for finding theta
    lo = v.min() - S
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.maximum(v - mid, 0.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    result = np.maximum(v - theta, 0.0)
    # Fix exact sum
    diff = S - result.sum()
    if abs(diff) > 1e-15 and result.sum() > 0:
        mask = result > 1e-15
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.maximum(result, 0.0)
    return result

def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def compute_subgradient(a, w, fsize):
    """
    Compute gradient of sum_k w[k]*c[k] where c = convolve(a,a).
    w has length 2n-1, result has length n.
    """
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:len(a)]
    return grad

def objective_value(a):
    """Compute the objective value for a (assumes sum(a) = sqrt(2n))."""
    n = len(a)
    c = fft_convolve_self(a)
    return c.max()

def run_subgradient_descent(a_init, max_time, rng, verbose=True):
    """
    Exact subgradient method with adaptive active set and backtracking line search.
    Uses momentum (heavy ball) with adaptive restarts.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n).bit_length()  # Need at least 2n-1 + n - 1 = 3n-2, use next power of 2 above 3n
    
    # Actually, for the subgradient computation, w has length 2n-1 and a has length n.
    # The cross-correlation needs fsize >= (2n-1) + n - 1 = 3n - 2
    fsize = 1 << (3 * n - 2 - 1).bit_length() if 3*n - 2 > 1 else 1
    
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    # Momentum
    v = np.zeros(n)
    mu = 0.9  # momentum coefficient
    
    it = 0
    last_improve = 0
    stagnant = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            last_improve = it
            stagnant = 0
        else:
            stagnant += 1
        
        # Adaptive active set: include lags within (1 - 1/it^0.3) of max
        # This starts with a wider set and narrows over time
        gap = c.max() - c
        # Use a threshold that shrinks over time
        threshold_frac = max(0.001, 1.0 / (50 + it * 0.1))
        threshold = c.max() * (1 - threshold_frac)
        active = c >= threshold
        if active.sum() < 1:
            active[np.argmax(c)] = True
        
        # Uniform weight over active set
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / active.sum()
        
        # Compute subgradient
        grad = compute_subgradient(a, w, fsize)
        
        # Backtracking line search with momentum
        # Start with a reasonable step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size based on gradient recovery test
        # Use the fact that f(a') <= f(a) + <g, a'-a> + (1/(2*wr))||a'-a||^2
        # where wr is a trust radius. We'll use backtracking.
        
        # Update momentum
        v = mu * v + (1 - mu) * grad
        
        # Try different step sizes
        base_step = 1.0 / gnorm
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = base_step * factor
            a_candidate = a - step * v
            a_candidate = project_simplex(a_candidate, S)
            
            c_candidate = fft_convolve_self(a_candidate)
            cand_max = c_candidate.max()
            
            # Armijo condition: f(a') <= f(a) - c1 * step * ||g||^2
            # For subgradient: f(a') <= f(a) + <g, a'-a> + epsilon
            # Simplified: just require improvement
            if cand_max < cur_max * (1 - 1e-7):
                a = a_candidate
                improved = True
                break
        
        if not improved:
            # No improvement, reduce momentum and try small step
            mu = max(0.5, mu * 0.95)
            a = project_simplex(a - 0.001 * base_step * grad, S)
        
        # Periodic noise injection
        if stagnant > 100 and it % 50 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 1000.0)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                stagnant = 0
        
        # Momentum restart
        if stagnant > 200:
            v = np.zeros(n)
            mu = 0.9
            stagnant = 0
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print("    subgrad iter %d: obj=%.8f, best=%.8f, t=%.1fs" % (it, cur_max, best, elapsed))
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_lse_optimization(a_init, max_time, rng, beta_start=100.0, beta_end=20000.0):
    """
    LSE-smoothed optimization with FISTA. Used for broader exploration.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    
    fsize_conv = 1 << (2 * n - 2).bit_length()
    fsize_grad = 1 << (3 * n - 2 - 1).bit_length()
    
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    y = a.copy()
    t_mom = 1.0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta
        frac = min(1.0, it / 200.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
        
        # LSE gradient
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        
        W = fft(w, fsize_grad)
        A = fft(y, fsize_grad)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / beta
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic noise
        if it % 500 == 0:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_mom = 1.0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

def multi_scale_optimize(a_init, n, max_time, rng):
    """
    Multi-scale optimization: work at half resolution, then refine at full resolution.
    """
    n_half = n // 2
    S_half = np.sqrt(2.0 * n_half)
    
    # Downsample by averaging pairs
    a_half = np.zeros(n_half)
    for i in range(n_half):
        a_half[i] = (a_init[2*i] + a_init[2*i+1]) / 2.0
    a_half = normalize_sum(a_half, n_half)
    
    # Optimize at half resolution
    time_half = min(60.0, max_time * 0.3)
    a_half_opt, _ = run_subgradient_descent(a_half, time_half, rng, verbose=False)
    
    # Upsample: interpolate
    t_old = np.linspace(0, 1, n_half)
    t_new = np.linspace(0, 1, n)
    a_full = np.interp(t_new, t_old, a_half_opt)
    a_full = normalize_sum(a_full, n)
    
    # Refine at full resolution
    remaining = max_time - (time.monotonic() - (time.monotonic() - time_half))
    # Actually let's just track time properly
    return a_full


def peak_reduction(a, n, S, max_time, rng):
    """
    Targeted peak reduction: identify the top peaks and use a small quadratic
    subproblem to reduce them while maintaining the sum constraint.
    """
    a = a.copy()
    start = time.monotonic()
    best = objective_value(a)
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Find the top peaks
        top_k = np.argsort(c)[::-1][:5]
        
        # For each peak, compute which elements of a contribute most
        # and reduce them slightly
        reduction = np.zeros(n)
        for k in top_k:
            # c[k] = sum_j a[j] * a[k-j]
            # The "responsible" elements are those with large a[j]*a[k-j]
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo > hi:
                continue
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            contrib_max = contrib.max()
            if contrib_max < 1e-15:
                continue
            # Weight by contribution
            weight = max(0, c[k] - cur_max * 0.999) / contrib_max
            np.add.at(reduction, js, contrib * weight + contrib * 0.001)  # small base weight
        
        # Apply reduction: multiply by (1 - alpha * normalized_reduction)
        if reduction.max() < 1e-15:
            break
        
        alpha = 0.01 / reduction.max()
        a = a * np.exp(-alpha * reduction)
        a = project_simplex(a, S)
    
    return best_a, best

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences using exact subgradient
method with adaptive active set, backtracking line search, momentum, and multi-scale
optimization. Key innovations: (1) exact subgradient instead of LSE smoothing for
higher accuracy near the optimum, (2) adaptive active set that narrows over time,
(3) multi-phase strategy balancing exploration and refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w + 1e-14
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    diff = S - result.sum()
    if abs(diff) > 1e-14:
        mask = result > 1e-14
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.maximum(result, 0.0)
    return result


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_subgradient(a, w, fsize):
    """
    Compute gradient of sum_k w[k]*c[k] where c = convolve(a,a, full).
    w has length 2n-1, a has length n, result has length n.
    """
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:len(a)]
    return grad


def run_subgradient(a_init, max_time, rng, verbose=False):
    """
    Exact subgradient method with adaptive active set, momentum, and backtracking.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    # fsize for subgradient: w(2n-1) cross-correlated with a(n) needs size >= 3n-2
    fsize = 1 << (3 * n - 3).bit_length()

    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    v = np.zeros(n)
    mu = 0.7
    it = 0
    stagnant = 0
    step_base = 1.0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if time.monotonic() - start > max_time:
            break

        # Adaptive active set: fraction of max to include
        # Starts wider, narrows over time
        frac = min(1.0, it / 500.0)
        threshold_frac = 0.01 * (1.0 - frac) + 0.0005 * frac
        threshold = c.max() * (1.0 - threshold_frac)
        active = c >= threshold
        if active.sum() < 1:
            active[np.argmax(c)] = True
        if active.sum() > 200:
            # Keep only top 200

            active[top_idx] = True

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / active.sum()

        grad = compute_subgradient(a, w, fsize)

        best_a = a.copy()

    return best_a, best


def run_lse_fista(a_init, max_time, rng, beta_start=50.0, beta_end=10000.0):
    """LSE-smoothed FISTA for broader exploration."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)

    it = 0

        it += 1

        frac = min(1.0, it / 300.0)
        beta = beta_start * (beta_end / beta_start) ** frac

        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()

        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot

        W = fft(w, fsize_grad)
        A = fft(y, fsize_grad)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)

        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)

        best_a = a.copy()

def run_peak_shave(a_init, max_time, rng):
    """Surgical peak shaving via multiplicative updates on responsible elements."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)

    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

        it += 1
        c = fft_convolve_self(a)

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Top 8 peaks
        top_idx = np.argsort(c)[::-1][:8]

        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo > hi:
                continue
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            weight = max(0.0, c[k] - cur_max * 0.999)
            np.add.at(resp, js, contrib * weight)
            # Also add small base weight for all contributors
            np.add.at(resp, js, contrib * 0.0001)

        rm = resp.max()
        if rm < 1e-20:
            break

        lr = 0.02 / rm
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)

    return best_a, best

            print("  Trying init %d (score=%.4f), %.0fs each" % (idx, score, t_per))
            ai = normalize_sum(inits[idx], n)
            ai_opt, vi = run_subgradient(ai, t_per, rng, verbose=False)
            ci = fft_convolve_self(ai_opt)
            if ci.max() < global_best:
                global_best = ci.max()
                global_a = ai_opt.copy()
                print("    New best from multi-start: %.8f" % global_best)
            a_current = global_a.copy()

    # === Phase 6: Final LSE + subgradient polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        # Quick high-beta LSE
        t6a = min(remaining * 0.5, 300.0)
        if t6a > 20:
            print("\n=== Phase 6a: High-beta LSE polish (%.0fs) ===" % t6a)
            a6, v6 = run_lse_fista(a_current, t6a, rng, beta_start=2000.0, beta_end=50000.0)
            c6 = fft_convolve_self(a6)
            if c6.max() < global_best:
                global_best = c6.max()
                global_a = a6.copy()
                print("  Phase 6a best: %.8f" % global_best)
            a_current = global_a.copy()

        # Final subgradient
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print("\n=== Phase 6b: Final subgradient polish (%.0fs) ===" % (remaining * 0.9))
            a6b, v6b = run_subgradient(a_current, remaining * 0.9, rng, verbose=True)
            c6b = fft_convolve_self(a6b)
            if c6b.max() < global_best:
                global_best = c6b.max()
                global_a = a6b.copy()
                print("  Phase 6b best: %.8f" % global_best)

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\n" + "=" * 50)
    print("FINAL: evaluate_sequence = %.8f" % (ver if np.isfinite(ver) else 999.0))
    print("Internal best: %.8f, length=%d" % (global_best, len(global_a)))
    print("=" * 50)

    if not np.isfinite(ver):
        print("WARNING: evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    return [float(x) for x in global_a]
