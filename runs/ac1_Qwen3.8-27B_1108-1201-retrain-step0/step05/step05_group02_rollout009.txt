        for j in range(lo, hi + 1):
            j2 = k_star - j
            if 0 <= j2 < n:
                contrib[j] = a[j2]
                contrib[j2] = max(contrib[j2], a[j])

        # Try top-contributing sources
        src_rank = np.argsort(contrib * np.maximum(a, 1e-10))[::-1]
        n_src = min(8, n // 4)

        improved = False
        for j in src_rank[:n_src]:
            if a[j] < 1e-10:
                continue
            delta = a[j] * 0.03
            if delta < 1e-12:
                continue

            # Find best target: coordinate that minimizes new max
            # Approximate by checking a few random targets
            best_new_max = cur_max
            best_t = -1
            for _ in range(20):
                t = int(rng.integers(0, n))
                if t == j:
                    continue
                # Quick estimate: check effect near the peak
                # c changes at positions j+k and t+k for k=0..n-1
                # Focus on whether new value at j+k or t+k exceeds cur_max
                jk = j + k_star
                tk = t + k_star
                if jk < 2 * n - 1:
                    new_jk = c[jk] - delta * a[k_star - j] if (0 <= k_star - j < n) else c[jk]
                    if new_jk > best_new_max and best_new_max < cur_max:
                        best_new_max = new_jk
                if tk < 2 * n - 1:
                    # effect of adding delta at t
                    pass

                # Simple heuristic: prefer targets far from peak-contributing region
                if not (lo <= t <= hi):
                    break
                if _ < 19:
                    continue
            else:
                t = int(rng.integers(0, n))
                while t == j or (lo <= t <= hi):
                    t = int(rng.integers(0, n))

            # Full check
            new_c = c.copy()
            for k2 in range(n):
                new_c[j + k2] -= delta * a[k2]
                new_c[t + k2] += delta * a[k2]
            new_max = new_c.max()

            if new_max < cur_max * (1 - 1e-5):
                a[j] -= delta
                a[t] += delta
                c = new_c
                improved = True
                break

        if not improved and stuck > 500:
            # Structured perturbation
            if stuck % 1000 < 100:
                # Block swap
                bs = max(1, n // 16)
                b1 = int(rng.integers(0, n - 2 * bs))
                b2 = int(rng.integers(0, n - 2 * bs))
                if b1 != b2:
                    a[[slice(b1, b1 + bs), slice(b2, b2 + bs)]] = a[[slice(b2, b2 + bs), slice(b1, b1 + bs)]]
                    a = np.maximum(a, 0)
                    a = project_simplex(a, S)
                    c = fft_convolve_self(a)
            else:
                noise_scale = max(1e-6, 0.005 * S / np.sqrt(n) * np.exp(-stuck / 10000))
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)

        if it % 200 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def fista_lse(a_init, max_time, rng, beta_init=20.0, beta_rate=0.008, beta_max=5000.0):
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / max(beta_init, 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        val, grad, c = lse_value_grad_a(y, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        x_new = project_simplex(y - step * grad, S)
        obj_new = compute_obj(x_new)

        bt = 0
        while obj_new > cur_obj + 1e-12 and step > 1e-14 and bt < 4:
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
            obj_new = compute_obj(x_new)
            bt += 1
        if bt <= 1:
            step = min(step * 1.02, 2.0 / max(beta, 1.0))

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        if it - last_improve > 2000:
            t_mom = 1.0
            ns = max(1e-6, 0.003 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, ns, n), S)

    return best_a, best_obj


def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-3.0 * (tt - 0.5) ** 2 / (0.2 ** 2))
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "flat":
        a = np.ones(n)
    else:
        a = 0.3 + 0.3 * np.sin(2 * np.pi * 5 * tt) + 0.2 * np.cos(2 * np.pi * 13 * tt + 0.5)
        a = np.abs(a)
    a = np.maximum(a, 0.01)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    # Get starting point
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n_default = 8192
        a_start = make_init(n_default, "cosine", rng)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.6f" % (n0, obj0))

    global_best_obj = obj0

    remaining = deadline - time.monotonic()

        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

    # === Phase 2: L-BFGS with high beta (20% time) ===
    remaining = deadline - time.monotonic()

    if p2_time > 30:

        a_opt, obj_opt = run_lbfgs(global_best_a, p2_time, beta=200, rng=rng)
        print("  L-BFGS beta=200: obj=%.6f" % obj_opt)

        a_cd, obj_cd = enhanced_coord_descent(global_best_a, p3_time, rng)

        if obj_cd < global_best_obj:

            global_best_a = a_cd.copy()

    # === Phase 4: Upsample to larger n (15% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.15
    if p4_time > 30 and len(global_best_a) <= 16384:
        n2 = len(global_best_a) * 2
        print("\n=== Phase 4: Upsample to n=%d (budget %.0fs) ===" % (n2, p4_time))
        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = a_up * (S2 / a_up.sum())
        obj_up = compute_obj(a_up)
        print("  Upsampled initial: obj=%.6f" % obj_up)
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up.copy()

        a_up_opt, obj_up_opt = run_lbfgs(a_up, p4_time * 0.7, beta=20, rng=rng)
        print("  L-BFGS upsampled: obj=%.6f" % obj_up_obj if False else "  L-BFGS upsampled: obj=%.6f" % obj_up_opt)
        if obj_up_opt < global_best_obj:
            global_best_obj = obj_up_opt
            global_best_a = a_up_opt.copy()

        remaining4 = deadline - time.monotonic()
        if remaining4 > 15:

    # === Phase 5: FISTA with high beta for final polish (10% time) ===

    remaining = deadline - time.monotonic()

    # Fallback
    n_fb = 8192
    a_fb = make_init(n_fb, "cosine", rng)
    final_list = [float(x) for x in a_fb]
    return final_list
