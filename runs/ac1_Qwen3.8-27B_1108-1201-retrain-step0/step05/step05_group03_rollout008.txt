"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-phase adaptive LSE-gradient optimization with:
1. FISTA with adaptive top-k LSE smoothing and aggressive backtracking
2. Dual-inspired peak targeting (identify top peaks, use them for subgradient)
3. Multi-restart with diverse initializations and multiple sequence lengths
4. Final high-precision polishing phase

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if v.sum() < 1e-30:
        return np.full(n, S / n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def project_simplex_fast(v, S):
    """Fast approximate projection: clip and rescale."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.full(len(v), S / len(v))
    return v * (S / s)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_topk_gradient(a, c, k, beta):
    """
    Compute gradient of top-k log-sum-exp smoothing of max of self-convolution.
    Returns (value, gradient) pair.
    """
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    val = cm + np.log(max(tot, 1e-300)) / beta
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return val, grad, idx


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def spectral_perturb(a, amount, rng):
    """Perturb the signal in the frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase_perturb = rng.normal(0, amount, fsize)
    phase_perturb[0] = 0
    A_new = A * np.exp(1j * phase_perturb)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, a.sum())


def spectral_flatten(a, alpha):
    """One step of spectral flattening to reduce autocorrelation peaks."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a
    ratio = power / max(mean_p, 1e-20)
    new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + ratio))
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / power)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def run_fista_phase(a_init, S, max_time, rng, phase=1):
    """
    FISTA with adaptive top-k LSE.
    Phase 1: moderate beta, larger k
    Phase 2: high beta, smaller k
    Phase 3: very high beta, minimal k, polishing
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0

    if phase == 1:
        beta_start, beta_end = 10.0, 200.0
        k_frac_start, k_frac_end = 0.12, 0.06
    elif phase == 2:
        beta_start, beta_end = 200.0, 2000.0
        k_frac_start, k_frac_end = 0.06, 0.02
    else:
        beta_start, beta_end = 2000.0, 20000.0
        k_frac_start, k_frac_end = 0.02, 0.005

    use_momentum = phase < 3

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / max(1, int(max_time * 30)))
        beta = beta_start * (beta_end / beta_start) ** progress
        k = max(8, int((k_frac_start + (k_frac_end - k_frac_start) * progress) * (2 * n - 1)))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        val, grad, idx = lse_topk_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step = 1.0 / (2.0 * beta * 0.05)
        step = min(step, 5.0)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-10:
                break
            step *= 0.6
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            momentum = (t_val - 1.0) / t_new
            # Limit momentum to prevent instability
            momentum = min(momentum, 0.9)
            y_new = x_new + momentum * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        # Restart when stuck
        restart_threshold = 500 if phase < 3 else 200
        if stuck > restart_threshold:
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 + phase * 0.2)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

        # Periodic spectral perturbation (only in phase 1)
        if phase == 1 and it % 3000 == 0 and rng.random() < 0.15:
            y = spectral_perturb(y, 0.05, rng)
            t_val = 1.0
            stuck = 0

        # Periodic spectral flattening (phase 1 and 2)
        if phase < 3 and it % 5000 == 0 and rng.random() < 0.1:
            y = spectral_flatten(y, 0.03)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_peak_targeting(a_init, S, max_time, rng):
    """
    Peak targeting: directly reduce the top peaks of the convolution.
    For each top peak at lag k, the gradient contribution is 2*f(reflected at k).
    We take steps that specifically target the worst peaks.
    """
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Find top peaks
        n_peaks = max(10, min(100, n // 64))
        peak_indices = np.argsort(c)[::-1][:n_peaks]
        peak_values = c[peak_indices]

        # Build gradient from top peaks with exponential weighting
        max_pk = peak_values.max()
        weights = np.exp(-(peak_values - peak_values.min()) / max(peak_values.max() - peak_values.min() + 1e-10, 1e-10) * 2.0)
        weights /= weights.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for w_i, k_i in zip(weights, peak_indices):
            # d/d f(m) of (f*f)(k_i) = 2 * f(k_i - m) for valid indices
            # i.e., grad[m] += 2 * w_i * f(k_i - m) for 0 <= k_i - m < n
            lo = max(0, int(k_i) - (n - 1))
            hi = min(n - 1, int(k_i))
            if lo <= hi:
                # f[k_i - m] for m in [lo, hi] means indices [k_i-hi, k_i-lo]
                flipped = f = a[k_i - hi:k_i - lo + 1][::-1]
                grad[lo:hi + 1] += 2.0 * w_i * flipped

        # Also add a small uniform component to maintain spread
        grad += 0.01 * np.ones(n) / n

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        base_step = 0.5 / (it + 10)
        step = base_step
        a_new = project_simplex(a - step * grad, S)
        for _ in range(6):
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max + 1e-12:
                break
            step *= 0.5
            a_new = project_simplex(a - step * grad, S)

        a = a_new

        # Occasional perturbation
        if it % 1000 == 0 and (time.monotonic() - start) / max_time < 0.6:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def run_adam_phase(a_init, S, max_time, rng):
    """Adam optimizer with LSE gradient and adaptive learning rate."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    while time.monotonic() - start < max_time:
        t += 1
        progress = min(1.0, t / max(1, int(max_time * 10)))
        beta = 50.0 * (1000.0 / 50.0) ** progress
        k = max(8, int(0.05 * (2 * n - 1) * (1.0 - 0.5 * progress)))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        val, grad, _ = lse_topk_gradient(a, c, k, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Learning rate with decay
        lr = 0.02 * (1.0 - 0.8 * progress) / (1.0 + 0.01 * t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = project_simplex(a_new, S)

        # Periodic restart
        if t % 3000 == 0 and rng.random() < 0.1 and progress < 0.7:
            noise_scale = 0.002 * S / np.sqrt(n) * (1.0 - progress)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            m = np.zeros(n)
            v = np.zeros(n)

    return best_a, best


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)

    remaining = deadline - time.monotonic()
    if remaining > 80:
        n3 = 4096
        t_budget = min(remaining * 0.15, 300)
        print(f"\n=== n=4096, budget={t_budget:.0f}s ===")
        a_start = resample_sequence(global_a, n3) if global_a is not None else None
        a_opt, obj = try_length(n3, a_start, t_budget, rng, label="4k")
        update_global(a_opt, n3, "4k")

    # Fourth: try n=16384 if time permits
    remaining = deadline - time.monotonic()
    if remaining > 150 and global_a is not None:
        n4 = 16384
        t_budget = min(remaining * 0.20, 400)
        print(f"\n=== n=16384, budget={t_budget:.0f}s ===")
        a_start = resample_sequence(global_a, n4)
        a_opt, obj = try_length(n4, a_start, t_budget, rng, label="16k")
        update_global(a_opt, n4, "16k")

    # Final intensive polishing on the best found
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print(f"\n=== Final Polish: n={n_best}, budget={remaining:.0f}s ===")

        # Deep FISTA polish
        t_fin = remaining * 0.5
        if t_fin > 20:
            a_fin, val_fin = run_fista_phase(global_a, S_best, t_fin, rng, phase=3)
            obj_fin = 2.0 * n_best * val_fin / (a_fin.sum() ** 2)
            if obj_fin < global_best:
                global_best = obj_fin
                global_a = a_fin.copy()
                print(f"  Final FISTA3: obj={global_best:.6f}")

        # Peak targeting polish
        remaining2 = deadline - time.monotonic() - 10
        if remaining2 > 20:
            a_pk, val_pk = run_peak_targeting(global_a, S_best, remaining2, rng)
            obj_pk = 2.0 * n_best * val_pk / (a_pk.sum() ** 2)
            if obj_pk < global_best:
                global_best = obj_pk
                global_a = a_pk.copy()
                print(f"  Final Peak: obj={global_best:.6f}")

    # Verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            print(f"\nVerification via evaluate_sequence: {ver:.6f}")
            if not np.isfinite(ver):
                print("  evaluate_sequence rejected! Using fallback.")
                global_a = np.ones(256) * (np.sqrt(512.0) / 256)
                final_list = [float(x) for x in global_a]
        except Exception as e:
            print(f"  Verification error: {e}")
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]

    elapsed = time.monotonic() - start_total
    print(f"\nTotal time: {elapsed:.1f}s, best obj={global_best:.6f}, length={len(global_a)}")
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
