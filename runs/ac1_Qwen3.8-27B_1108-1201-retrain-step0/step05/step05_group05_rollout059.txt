"""

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import splu, spsolve


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient_topk(a, c, beta, K=None):
    """
    Gradient of LSE-smoothed max using only top-K lags.
    """
    n = len(a)
    if K is None or K >= len(c):
        K = len(c)
    # Find top K lags
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
    else:
        top_idx = np.arange(len(c))
    c_top = c[top_idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Scatter back to full vector
    w_full = np.zeros(len(c))
    w_full[top_idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def lse_value(c, beta, K=None):
    """LSE-smoothed approximation of max(c)."""
    if K is not None and K < len(c):
        top = np.sort(c)[::-1][:K]
    else:
        top = c
    cm = top.max()
    return cm + np.log(np.sum(np.exp(beta * (top - cm)))) / beta


def run_lse_fista(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.003,
                  beta_max=20000.0, K_init=None, K_min=8,
                  perturbation_period=3000):
    """
    FISTA with LSE smoothing and adaptive top-K focusing.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    if K_init is None:
        K_init = min(n, 200)

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        # Adaptive K: start with many lags, focus to fewer
        K = max(K_min, int(K_init * np.exp(-3.0 * progress)))
        K = min(K, len(c0))

        c = fft_convolve_self(y)
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient_topk(y, c, beta, K=K)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        step = 1.0 / (3.0 * beta)

        # Stuck detection
        if it - last_improve > 3000:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 5000.0)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if perturbation_period > 0 and it % perturbation_period == 0:
            decay = np.exp(-it / (20.0 * perturbation_period))
            noise_scale = 0.002 * S / np.sqrt(n) * decay

                y = project_simplex(y + rng.normal(0, noise_scale, n), S)

    return best_a, c_final.max()


def kkt_solve(a, S, tol=1e-4, max_active=50):
    """
    Given current a, find the KKT-optimal a by:
    1. Identify active lags (top K).
    2. Compute optimal weights lambda on active lags.
    3. Solve the linear system for the KKT-optimal a.
    """
    n = len(a)
    c = fft_convolve_self(a)
    cmax = c.max()
    
    # Active set: lags within tol of max
    active = np.where(c >= cmax - tol * cmax)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    active = active[:max_active]
    
    # Weights: proportional to c values on active set
    lam = c[active]
    lam_sum = lam.sum()
    if lam_sum < 1e-30:
        return a.copy(), cmax
    lam = lam / lam_sum
    
    # Build the linear system: sum_k lambda_k * 2 * a_{k-m} = mu for all m
    # M[m, j] = 2 * lambda_k where k = m + j is in active set
    # We solve M @ a_new = mu * ones, but mu is unknown.
    # Instead, we solve: M @ a_new = ones (and normalize later).
    # Or better: M @ a_new = mean(M @ a) * ones (to make the "gradient" flat).
    
    # Construct M implicitly: (M @ v)[m] = sum_j M[m,j] * v[j]
    # M[m,j] = 2*lam[i] if m+j == active[i] for some i
    # So (M @ v)[m] = sum_i 2*lam[i] * v[active[i] - m] (if 0 <= active[i]-m < n)
    
    def Mv(v):
        """Apply M to vector v without forming M explicitly."""
        result = np.zeros(n)
        for i, k in enumerate(active):
            # M[m, k-m] += 2*lam[i] for m in range such that 0 <= k-m < n
            # i.e., m in range [max(0, k-n+1), min(k, n-1)]
            m_start = max(0, k - n + 1)
            m_end = min(k, n - 1)
            if m_start <= m_end:
                # v[k-m] for m in [m_start, m_end]
                # k-m ranges from k-m_start to k-m_end
                idx_start = k - m_end
                idx_end = k - m_start
                result[m_start:m_end+1] += 2 * lam[i] * v[idx_start:idx_end+1][::-1]
        return result
    
    # We want to solve: M @ a = mu * 1, but mu is free.
    # Equivalently: (M - mu * J) @ a = 0 where J is the matrix with 1/M @ 1 in each row... 
    # Actually, let's just solve M @ a = 1 (all ones) and normalize.
    # This makes the "gradient" proportional to a constant vector.
    
    # Use iterative method: power iteration or conjugate gradient on M
    # M might not be symmetric, so use a general method.
    # Actually, M is a "banded" matrix with a specific structure.
    # Let's use a simple iterative approach: a_{new} = normalize(M @ a_old)
    # This is a fixed-point iteration that converges to the principal eigenvector of M.
    
    # Alternative: just do a few steps of "a = project_simplex(M @ a / (M @ a).sum() * S, S)"
    # This makes the "gradient" more uniform.
    
    # Actually, the cleanest approach: 
    # The KKT condition is: M @ a = mu * 1 (on the support).
    # If the support is full: M @ a = mu * 1.
    # This is a homogeneous system (up to the constant mu).
    # We can solve it by finding the eigenvector of M corresponding to the eigenvalue mu.
    # Specifically, M @ a = mu * 1 means (M - mu * 1 * w^T) @ a = 0 for some w.
    # This is not a standard eigenvalue problem.
    
    # Let me just use the iterative approach:
    # a_{new} = project_simplex(M @ a, S)
    # This should drive a towards the KKT condition.
    
    a_work = a.copy()
    for _ in range(5):
        a_work = Mv(a_work)
        a_work = np.maximum(a_work, 0)
        s = a_work.sum()
        if s < 1e-15:
            break
        a_work = a_work * (S / s)
    
    return a_work, cmax


def run_kkt_iteration(a_init, S, max_time, rng, tol=1e-4):
    """
    Iterate the KKT active-set method:
    1. Find active lags.
    2. Compute KKT-optimal a.
    3. Blend with current a.
    4. Repeat.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        a_new, _ = kkt_solve(a, S, tol=tol, max_active=30)
        
        # Blend
        alpha = 0.3 + 0.5 * (it / max(1, it + 10))
        a_blend = (1 - alpha) * a + alpha * a_new
        a_blend = project_simplex(a_blend, S)
        
        c = fft_convolve_self(a_blend)
        if c.max() < best:
            best = c.max()
            best_a = a_blend.copy()
            last_improve = it
        else:
            a_blend = a  # Don't update if no improvement
        
        a = a_blend
        
        if it - last_improve > 2000:
            # Perturb
            noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 1000.0)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            else:
                break
            last_improve = it
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Windows
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.01)
    inits.append(np.exp(-8.0 * (tt - 0.5) ** 2) + 0.01)
    
    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * tt + 0.7) + 0.2 * np.cos(10 * np.pi * tt), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.sin(6 * np.pi * tt) + 0.15 * np.cos(14 * np.pi * tt + 0.3), 0.01))
    inits.append(np.maximum(1.0 + 0.2 * np.cos(8 * np.pi * tt) + 0.15 * np.sin(20 * np.pi * tt + 1.0), 0.01))
    
    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.normal(0.8, 0.4, n)) + 0.05)
    
    # "Noise-like" (flat spectrum approximation)
    noise = rng.standard_normal(n)
    noise = np.abs(noise) + 0.1  # Make non-negative
    inits.append(noise)
    
    # Block structures
    for nb in [4, 8, 16]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.3, 1.5, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n] + 0.01)
    
    return inits


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)
