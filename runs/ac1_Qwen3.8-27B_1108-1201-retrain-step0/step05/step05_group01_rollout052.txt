    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    a = a.copy()
    S = a.sum()

        A = fft(a, fsize)
        mag = np.abs(A)
        # Target: flat magnitude
        target_mag = mag.sum() / fsize
        # Scale
        mask = mag > 1e-10
        A_new = A.copy()
        A_new[mask] = A_new[mask] * (target_mag / mag[mask])
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0)
        a_new = a_new * (S / a_new.sum())
        a = a_new
    return a

- Apply T(w) to a vector: this is a convolution, doable in O(n log n)
- So each CG step is O(n log n)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Improved multi-phase: FISTA-LSE with adaptive beta, L-BFGS-B polishing,
BB-step subgradient refinement, and multi-n exploration.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista(a_init, max_time, rng, beta_start=20.0, beta_end=500.0,
              total_iters_hint=5000):
    """
    FISTA with LSE, linearly increasing beta.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        
        # Linearly ramp beta
        frac = min(1.0, it / total_iters_hint)
        beta = beta_start + (beta_end - beta_start) * frac
        beta = min(beta, 2000.0)
        step = 1.0 / beta

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # If stuck for too long, restart with higher beta
        if stuck_count > 3000:
            beta = min(beta * 2, 5000.0)
            step = 1.0 / beta
            stuck_count = 0
            t_mom = 1.0
            y = best_a.copy()

        # Gradient via LSE
        w = lse_weights(c, beta)
        grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum reset to avoid overshooting
        if it % 3000 == 0:
            t_mom = 1.0

    return best_a, best_conv


def run_fista_topk(a_init, max_time, rng):
    """
    FISTA using top-k gradient (exact subgradient approximation).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        
        # Adaptive top-k
        if stuck < 200:
            k = 1
        elif stuck < 1000:
            k = 3
        elif stuck < 5000:
            k = 8
        else:
            k = min(32, 8 + (stuck - 5000) // 500)

        # Get top-k lags
        k_actual = min(k, len(c))
        idx = np.argsort(c)[::-1][:k_actual]
        vals = c[idx]
        # Weight by proximity to max
        w_vals = np.ones(k_actual)
        w_full = np.zeros(len(c))
        w_full[idx] = w_vals / w_vals.sum()
        
        grad = conv_gradient(y, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size (BB-inspired)
        step = S / gnorm
        step = min(step, 2.0 / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 5000 and it % 500 == 0:
            # Small random perturbation
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0

    return best_a, best_conv


def run_lbfgs(a_init, max_time, rng):
    """
    L-BFGS-B polishing with multiple beta levels.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    # Progressive beta levels
    beta_schedule = [30.0, 80.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]
    
    for beta in beta_schedule:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break

        c = fft_convolve_self(a)
        cm = c.max()

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            obj = cm_local + np.log(e.sum()) / beta
            # Gradient
            w = e / e.sum()
            grad = conv_gradient(x, w)
            return obj, grad

        t0 = time.monotonic()
        max_iter = min(200, max(30, int(remaining * 5)))

        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': max_iter, 'maxls': 40,
                     'ftol': 1e-16, 'gtol': 1e-12, 'iprint': -1}
        )

        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        if a_new.sum() > 1e-15:
            a_new = a_new * (S / a_new.sum())
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
                print("    L-BFGS beta=%.0f: obj=%.10f (t=%.1fs)" % (beta, obj_new, elapsed))
            else:
                print("    L-BFGS beta=%.0f: obj=%.10f (no improvement)" % (beta, obj_new))

    return best_a, best_obj


def gs_flat_spectrum(a_init, n_iters, rng):
    """Gerchberg-Saxton flat spectrum method."""
    n = len(a_init)
    S = a_init.sum()
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = a * (S / a.sum())
    
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    for i in range(n_iters):
        A = fft(a, fsize)
        mag = np.abs(A)
        target = mag.mean()
        mask = mag > 1e-10
        A_new = A.copy()
        A_new[mask] = A_new[mask] * (target / mag[mask])
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() > 1e-10:
            a = a_new * (S / a_new.sum())
    
    return a


def run_bb_subgradient(a_init, max_time, rng):
    """
    Barzilai-Borwein step size subgradient method.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    
    x_prev = a.copy()
    g_prev = None
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        
        # Determine active set size
        if stuck < 50:
            k = 1
        elif stuck < 200:
            k = 2
        elif stuck < 1000:
            k = 4
        elif stuck < 5000:
            k = 8
        else:
            k = min(16, 8 + (stuck - 5000) // 1000)

        idx = np.argsort(c)[::-1][:k]
        w_full = np.zeros(len(c))
        w_full[idx] = 1.0 / k
        grad = conv_gradient(a, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # BB step size
        if g_prev is not None:
            s_vec = a - x_prev
            y_vec = grad - g_prev
            s_dot_y = np.dot(s_vec, y_vec)
            yy = np.dot(y_vec, y_vec)
            if yy > 1e-10 and abs(s_dot_y) > 1e-10:
                step_bb = s_dot_y / yy
                if step_bb > 0 and step_bb < 10.0 / gnorm:
                    step = step_bb
                else:
                    step = 1.0 / gnorm
            else:
                step = 1.0 / gnorm
        else:
            step = 1.0 / gnorm

        step = min(step, S / (0.5 * gnorm))

        x_prev = a.copy()
        g_prev = grad.copy()
        
        a_new = project_simplex(a - step * grad, S)
        a = a_new

    return best_a, best_conv

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase: FISTA-LSE with adaptive beta ramp, L-BFGS-B progressive polishing,
top-k active-set FISTA, BB-step subgradient, and multi-length exploration.
Key improvements: higher beta ceilings, multiplicative update option,
progressive L-BFGS with tighter tolerances, and n-exploration.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_lse(a_init, max_time, rng, beta_start=10.0, beta_end=500.0,
                  iter_hint=4000, momentum_reset_period=2000):
    """FISTA with LSE and linearly ramping beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / iter_hint)
        beta = beta_start + (beta_end - beta_start) * frac
        beta = min(beta, 10000.0)
        step = 1.0 / beta

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        # If stuck, boost beta and reset
        if stuck > 2000:
            beta = min(beta * 3, 20000.0)
            step = 1.0 / beta
            y = best_a.copy()
            t_mom = 1.0
            stuck = 0

        w = lse_weights(c, beta)
        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % momentum_reset_period == 0:
            t_mom = 1.0

    return best_a, best_conv


def run_fista_multiplicative(a_init, max_time, rng, beta_start=50.0, beta_end=2000.0):
    """
    FISTA variant using multiplicative (log-space) updates instead of Euclidean projection.
    a[i] *= exp(-step * grad[i]), then normalize.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / 3000.0)
        beta = beta_start + (beta_end - beta_start) * frac
        beta = min(beta, 10000.0)
        step = 1.0 / beta

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        if stuck > 3000:
            beta = min(beta * 2, 20000.0)
            step = 1.0 / beta
            y = best_a.copy()
            t_mom = 1.0
            stuck = 0

        w = lse_weights(c, beta)
        grad = conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Multiplicative update
        log_a = np.log(y + 1e-30)
        log_a_new = log_a - step * grad
        # FISTA momentum in log-space
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        log_y = log_a_new + ((t_mom - 1.0) / t_new) * (log_a_new - log_a)
        y = np.exp(log_y - log_y.max())
        y = y * (S / y.sum())
        y = np.maximum(y, 1e-10)
        a = y.copy()
        t_mom = t_new

        if it % 3000 == 0:
            t_mom = 1.0

    return best_a, best_conv


def run_lbfgs_progressive(a_init, max_time, rng):
    """
    L-BFGS-B with progressive beta increase and tight tolerances.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    beta_schedule = [50.0, 150.0, 400.0, 1000.0, 3000.0, 8000.0, 15000.0]

    for beta in beta_schedule:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 15:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            obj = cm_local + np.log(e.sum()) / beta
            w = e / e.sum()
            grad = conv_gradient(x, w)
            return obj, grad

        t0 = time.monotonic()
        max_iter = min(300, max(50, int(remaining * 3)))

        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': max_iter, 'maxls': 50,
                     'ftol': 1e-18, 'gtol': 1e-14, 'iprint': -1}
        )

        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        if a_new.sum() > 1e-15:
            a_new = a_new * (S / a_new.sum())
            obj_new = compute_obj(a_new)
            improved = obj_new < best_obj - 1e-12
            if improved:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
            print("    L-BFGS b=%5.0f: obj=%.10f %s (t=%.1fs, niter=%d)" %
                  (beta, obj_new, "IMPROVED" if improved else "", elapsed, result.nit))

    return best_a, best_obj


def run_active_subgrad(a_init, max_time, rng):
    """
    Subgradient method on the active set with BB step size and periodic exploration.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    x_prev = a.copy()
    g_prev = None
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive active set size
        if stuck < 30:
            k = 1
        elif stuck < 100:
            k = 2
        elif stuck < 500:
            k = 4
        elif stuck < 2000:
            k = 8
        elif stuck < 8000:
            k = 16
        else:
            k = min(32, 16 + (stuck - 8000) // 1000)

        idx = np.argsort(c)[::-1][:k]
        w_full = np.zeros(len(c))
        w_full[idx] = 1.0 / k
        grad = conv_gradient(a, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # BB step size
        if g_prev is not None and it > 2:
            s_vec = a - x_prev
            y_vec = grad - g_prev
            s_dot_y = np.dot(s_vec, y_vec)
            yy = np.dot(y_vec, y_vec)
            if yy > 1e-12 and abs(s_dot_y) > 1e-12:
                step_bb = s_dot_y / yy
                if 0.01 / gnorm < step_bb < 5.0 / gnorm:
                    step = step_bb
                else:
                    step = 0.5 / gnorm
            else:
                step = 0.5 / gnorm
        else:
            step = 0.5 / gnorm

        step = min(step, S / (0.3 * gnorm + 1e-10))

        x_prev = a.copy()
        g_prev = grad.copy()
        a = project_simplex(a - step * grad, S)

        # Periodic exploration: perturb with a random lag's subgradient
        if stuck > 3000 and it % 200 == 0:
            random_idx = rng.integers(0, len(c))
            w_rand = np.zeros(len(c))
            w_rand[random_idx] = 1.0
            grad_rand = conv_gradient(a, w_rand)
            gr_norm = np.linalg.norm(grad_rand)
            if gr_norm > 1e-10:
                explore_step = 0.1 / gr_norm
                a = project_simplex(a - explore_step * grad_rand, S)

    return best_a, best_conv


def run_hybrid_fista(a_init, max_time, rng):
    """
    FISTA that alternates between LSE gradient and exact active-set subgradient.
    Uses DCT-domain structured noise when stuck.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Choose gradient type based on stuck count
        if stuck < 500:
            beta = 20 + 0.1 * it
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
            step = 1.0 / beta
        elif stuck < 3000:
            # Blend LSE with active set
            beta = 100 + 0.05 * stuck
            w_lse = lse_weights(c, beta)
            k = min(8, 1 + stuck // 500)
            idx = np.argsort(c)[::-1][:k]
            w_act = np.zeros(len(c))
            w_act[idx] = 1.0 / k
            blend = min(0.5, stuck / 3000.0)
            w = (1 - blend) * w_lse + blend * w_act
            grad = conv_gradient(y, w)
            step = 1.0 / (beta * (1 + 0.01 * stuck))
        else:
            # Pure active set with widening tolerance
            tol = min(0.05, 0.002 * (stuck - 3000) / 1000.0)
            active = np.where(c >= cur_max * (1.0 - tol))[0]
            k_use = min(len(active), 16)
            w = np.zeros(len(c))
            w[active[:k_use]] = 1.0 / k_use
            grad = conv_gradient(y, w)
            gnorm = np.linalg.norm(grad)
            step = 1.0 / (gnorm * (1 + 0.001 * (stuck - 3000)))

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # DCT-domain noise when very stuck
        if stuck > 5000 and it % 300 == 0:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-min(stuck, 20000) / 15000.0)
            if noise_scale > 1e-8:
                noise = rng.normal(0, 1.0, n)
                F = fft(noise)
                cutoff = max(3, n // 100)
                F[cutoff:-cutoff] *= 0.05
                noise_smooth = np.real(ifft(F))[:n]
                nn = np.linalg.norm(noise_smooth)
                if nn > 1e-10:
                    noise_smooth = noise_smooth / nn * noise_scale
                    y = project_simplex(y + noise_smooth, S)
                    t_mom = 1.0

        if it % 4000 == 0:
            t_mom = 1.0

    return best_a, best_conv

            global_best_a = a2.copy()

    # === Phase 3: L-BFGS-B progressive (25% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3 = remaining * 0.25
        print("\n=== P3: L-BFGS-B progressive (%.0fs) ===" % p3)
        t0 = time.monotonic()
        a3, obj3 = run_lbfgs_progressive(global_best_a, p3, rng)
        print("  Best from L-BFGS: obj=%.10f (t=%.0fs)" % (obj3, time.monotonic() - t0))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Active-set subgradient (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p4 = remaining * 0.15
        print("\n=== P4: Active-set subgrad (%.0fs) ===" % p4)
        t0 = time.monotonic()

            global_best_a = a4.copy()

    # === Phase 5: Hybrid FISTA (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5 = remaining * 0.10
        print("\n=== P5: Hybrid FISTA (%.0fs) ===" % p5)
        t0 = time.monotonic()

        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Final L-BFGS-B squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== P6: Final squeeze (%.0fs) ===" % remaining)
        t0 = time.monotonic()
        a6, obj6 = run_lbfgs_progressive(global_best_a, remaining * 0.9, rng)
        print("  Final: obj=%.10f (t=%.0fs)" % (obj6, time.monotonic() - t0))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nVerified: %.10f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:

            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective: %.10f" % global_best_obj)
