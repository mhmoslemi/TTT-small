"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: Multi-phase optimization with convolution-aware basin hopping:
1. Diverse initialization with fast FISTA screening
2. Basin hopping with convolution-targeted mutations and simulated annealing
3. Peak-shaving polish with adaptive top-k gradients
4. Final KKT-guided refinement with very high smoothing

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_gradient(a, c, k, fsize, A_cached=None):
    """
    Compute the subgradient from the top-k lags of the convolution.
    The subgradient at the max is 2*(a convolved with delta at the peak lag).
    For a mix of top-k lags, we weight them by their excess above the mean.
    """
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    # Weight by excess above mean
    mean_c = c.mean()
    weights = np.maximum(c[idx] - mean_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        # Uniform weights on top-k
        weights = np.ones(k) / k
    else:
        weights = weights / wsum
    
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    
    if A_cached is None:
        A_cached = fft(a, fsize)
    
    W = fft(w_full, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A_cached)))[:n]
    return grad


def lse_gradient_v2(a, c, beta, fsize, A_cached=None):
    """LSE gradient with caching."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clamp to avoid overflow
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    
    if A_cached is None:
        A_cached = fft(a, fsize)
    
    W = fft(w, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A_cached)))[:n]
    return grad


def mirror_descent_step(a, grad, eta, S):
    """KL-proximal (mirror descent) step."""
    g_max = np.max(np.abs(grad))
    if g_max < 1e-30:
        return a.copy()
    scaled = -np.clip(eta * grad / g_max, -30, 30)
    a_new = a * np.exp(scaled)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new = a_new * (S / s)
    return np.maximum(a_new, 0.0)


def fourier_phase_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def convolution_target_mutate(a, rng, n_peaks=5):
    """
    Mutation that targets specific peaks of the convolution.
    For each peak lag k, we try to reduce its contribution by moving
    mass from positions that feed into that lag.
    """
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    
    # Find top peaks
    idx = np.argsort(c)[::-1][:n_peaks]
    
    a_new = a.copy()
    for k in idx:
        # The convolution at lag k is sum_i a[i]*a[k-i]
        # To reduce it, we can slightly reduce a[i] for i that contribute significantly
        # and add that mass to positions that don't contribute to this lag
        contrib = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n and j != i:
                contrib[i] = a[i] * a[j]
            elif j == i:
                contrib[i] = a[i] * a[j]
        
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            continue
        
        # Fraction to remove from each contributing position
        epsilon = 0.01 * S / n  # Small perturbation
        # Reduce top contributors
        top_i = np.argsort(contrib)[::-1][:max(1, n // 50)]
        for i in top_i:
            reduce = min(a[i] * 0.01, epsilon / max(len(top_i), 1))
            a_new[i] -= reduce
            # Add to a random position that doesn't contribute to lag k
            # Position j contributes if a[j]*a[k-j] > 0, i.e., k-j is in [0,n-1]
            candidates = [j for j in range(n) if not (0 <= k-j < n)]
            if candidates:
                j = rng.integers(len(candidates))
                a_new[candidates[j]] += reduce
            else:
                a_new[rng.integers(n)] += reduce
    
    return project_simplex(a_new, S)

    weights = np.maximum(c[idx] - mean_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    if A_cached is None:
        A_cached = fft(a, fsize)
    W = fft(w_full, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A_cached)))[:n]
    return grad


def lse_gradient(a, c, beta, fsize, A_cached=None):
    """LSE-smoothed max gradient."""
    n = len(a)
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -500, 0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    if A_cached is None:
        A_cached = fft(a, fsize)
    W = fft(w, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A_cached)))[:n]
    return grad


def mirror_descent_step(a, grad, eta, S):
    """KL-proximal (mirror descent) step."""
    g_max = np.max(np.abs(grad))
    if g_max < 1e-30:
        return a.copy()
    scaled = -np.clip(eta * grad / g_max, -30, 30)
    a_new = a * np.exp(scaled)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new = a_new * (S / s)
    return np.maximum(a_new, 0.0)


def fourier_phase_mutate(a, rng, scale=0.1):
    """Fourier-domain phase mutation."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * fsize)
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def fourier_crossover(a1, a2, rng):
    """Fourier-domain crossover."""
    n = len(a1)
    S = a1.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)
    r = rng.random(fsize)
    alpha = 0.3 + 0.4 * r
    new_mag = alpha * np.abs(A1) + (1 - alpha) * np.abs(A2)
    new_phase = np.where(r < 0.5, np.angle(A1), np.angle(A2))
    new_A = new_mag * np.exp(1j * new_phase)
    new_A[0] = A1[0]
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def subgradient_lag(a, k, n):
    """Compute the subgradient column for a single convolution lag k."""
    sg = np.zeros(n)
    # d conv[k]/d a[j] = a[k-j] + a[k-j] (if both valid and j != k-j)
    #                   = a[k-j] (if j == k-j)
    # Simplified: sg[j] = 2*a[k-j] if 0<=k-j<n, with correction if 2j==k
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    for j in range(lo, hi + 1):
        sg[j] = 2.0 * a[k - j]
    # Correction for the self-term where j = k-j i.e. 2j = k
    if k % 2 == 0:
        j_mid = k // 2
        if 0 <= j_mid < n:
            sg[j_mid] = a[j_mid]
    return sg


def subgradient_lag_vectorized(a, k, n):
    """Vectorized subgradient for lag k."""
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    j_range = np.arange(lo, hi + 1)
    sg = np.zeros(n)
    sg[j_range] = 2.0 * a[k - j_range]
    if k % 2 == 0:
        j_mid = k // 2
        if 0 <= j_mid < n:
            sg[j_mid] = a[j_mid]
    return sg


def conv_target_mutate(a, rng, n_peaks=4, strength=0.03):
    """
    Convolution-targeted mutation: use subgradients at the top peaks
    to guide a descent step that specifically reduces those peaks.
    """
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    idx = np.argsort(c)[::-1][:n_peaks]
    
    # Combine subgradients from top peaks, weighted by excess
    mean_c = c.mean()
    weights = np.maximum(c[idx] - mean_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(n_peaks) / n_peaks
    else:
        weights = weights / wsum
    
    combined_sg = np.zeros(n)
    for i, k in enumerate(idx):
        combined_sg += weights[i] * subgradient_lag_vectorized(a, k, n)
    
    # Take a step in the negative subgradient direction
    sg_norm = np.linalg.norm(combined_sg)
    if sg_norm < 1e-15:
        return a.copy()
    
    step = strength * S
    a_new = a - step * combined_sg / sg_norm
    return project_simplex(a_new, S)


def run_fista(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.005,
              beta_max=5000.0, use_topk=False, topk_init=8, topk_max=64,
              mixed_mode=False):
    """
    FISTA with adaptive smoothing. Supports LSE, top-k, and mixed modes.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    fsize = 1 << (2 * n - 1).bit_length()
    A_cached = fft(a, fsize)
    
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        
        # Recompute cached FFT periodically
        if it % 5 == 1:
            A_cached = fft(y, fsize)
        
        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = y.copy()
            last_improve = it

        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        # Choose gradient
        if mixed_mode and it % 4 == 0:
            k = min(topk_max, topk_init + it // 100)
            grad = topk_gradient(y, c, k, fsize, A_cached)
            gnorm = np.linalg.norm(grad)
            step = 1.0 * S / (n * (1.0 + 0.002 * it)) if gnorm < 1e-14 else \
                   1.5 * S / (n * (gnorm / (S / n) + 0.01))
        elif use_topk:
            k = min(topk_max, topk_init + it // 200)
            grad = topk_gradient(y, c, k, fsize, A_cached)
            gnorm = np.linalg.norm(grad)
            step = 1.5 * S / (n * (gnorm / (S / n) + 0.01))
        else:
            grad = lse_gradient(y, c, beta, fsize, A_cached)
            step = 1.0 / (3.0 * beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Stuck detection
        if it - last_improve > 1500:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 3000.0)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            else:
                y = mirror_descent_step(y, grad, 0.01, S)
            t_mom = 1.0
            last_improve = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_peak_shave(a_init, S, max_time, rng):
    """
    Peak-shaving: iteratively target worst peaks with adaptive top-k
    and backtracking line search.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    fsize = 1 << (2 * n - 1).bit_length()
    
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    
    y = a.copy()
    t_mom = 1.0
    it = 0
    last_improve = 0
    A_cached = fft(y, fsize)

    while time.monotonic() - start < max_time:
        it += 1
        if it % 5 == 1:
            A_cached = fft(y, fsize)
        
        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = y.copy()
            last_improve = it
        
        if it - last_improve > 2500:
            noise_scale = 0.004 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue
        
        progress = (time.monotonic() - start) / max_time
        k = min(2 + int(98 * progress), n // 8)
        
        grad = topk_gradient(y, c, k, fsize, A_cached)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step with backtracking
        step = 2.0 * S / (n * (gnorm / (S / n) + 0.005))
        
        for _ in range(3):
            x_new = project_simplex(y - step * grad, S)
            c_check = fft_convolve_self(x_new)
            if c_check.max() <= cmax * 1.0001:
                break
            step *= 0.5
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_basin_hopping(a_init, S, max_time, rng):
    """
    Basin hopping with convolution-targeted mutations and SA acceptance.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    current_val = best
    
    opt_time = max(12.0, max_time / 25.0)
    
    T_start = 0.3 * best
    T_end = 0.005 * best
    
    population = [(best, a.copy())]
    iteration = 0

    while time.monotonic() - start < max_time - opt_time:
        iteration += 1
        elapsed = time.monotonic() - start
        total_iters_est = int(max_time / opt_time)
        progress = min(1.0, iteration / max(1, total_iters_est))
        T_current = T_start * (T_end / T_start) ** progress
        
        # Choose mutation
        r = rng.random()
        if len(population) >= 2 and r < 0.2:
            i1 = rng.integers(len(population))
            i2 = rng.integers(len(population))
            if i1 != i2:
                a = fourier_crossover(population[i1][1], population[i2][1], rng)
            else:
                a = fourier_phase_mutate(a, rng, 0.05 + 0.15 * (1 - progress))
        elif r < 0.5:
            # Convolution-targeted mutation
            n_peaks = 2 + int(4 * (1 - progress))
            a = conv_target_mutate(a, rng, n_peaks=n_peaks,
                                    strength=0.02 + 0.03 * (1 - progress))
        else:
            scale = 0.03 + 0.2 * (1 - progress) * rng.random()
            a = fourier_phase_mutate(a, rng, scale)
        
        # Local optimization
        t_remaining = max_time - (time.monotonic() - start) - 20
        t_alloc = min(opt_time, t_remaining)
        if t_alloc < 5:
            break
        
        use_topk = (iteration > 3)
        a_opt, val = run_fista(a, S, t_alloc, rng,
                                beta_init=8 + 15 * min(iteration, 15),
                                beta_rate=0.004,
                                beta_max=2000 + 200 * min(iteration, 20),
                                use_topk=use_topk,
                                topk_init=4, topk_max=32,
                                mixed_mode=True)
        
        # SA acceptance
        delta = val - current_val
        if delta < 0 or rng.random() < np.exp(-delta / max(T_current, 1e-10)):
            current_val = val
        
        if val < best:
            best = val
            best_a = a_opt.copy()
            obj = 2.0 * n * val / (S * S)
            print("    BH iter %d: max_b=%.8f obj=%.6f" % (iteration, val, obj))
        
        population.append((val, a_opt.copy()))
        if len(population) > 6:
            population.sort(key=lambda x: x[0])
            population = population[:6]

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_high_beta_polish(a_init, S, max_time, rng):
    """Final polish with very high beta LSE and alternating top-k."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    fsize = 1 << (2 * n - 1).bit_length()
    
    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()
    
    y = a.copy()
    t_mom = 1.0
    it = 0
    last_improve = 0
    A_cached = fft(y, fsize)

    while time.monotonic() - start < max_time:
        it += 1
        if it % 3 == 1:
            A_cached = fft(y, fsize)
        
        c = fft_convolve_self(y)
        cmax = c.max()
        if cmax < best:
            best = cmax
            best_a = y.copy()
            last_improve = it
        
        if it - last_improve > 2000:
            # Very small perturbation
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            continue
        
        # Alternate between high-beta LSE and top-k
        progress = (time.monotonic() - start) / max_time
        if it % 2 == 0:
            beta = 200.0 * (1.0 + 200.0 * progress)
            beta = min(beta, 50000.0)
            grad = lse_gradient(y, c, beta, fsize, A_cached)
            step = 1.0 / (4.0 * beta)
        else:
            k = min(4 + int(60 * progress), n // 16)
            grad = topk_gradient(y, c, k, fsize, A_cached)
            gnorm = np.linalg.norm(grad)
            step = 0.5 * S / (n * (gnorm / (S / n) + 0.001)) if gnorm > 1e-14 else 0.0
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    if given_seq is not None:
        g = np.clip(given_seq, 0, 1000)
        if g.sum() > 0:
            inits.append(g)
    
    # Various window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.54 + 0.46 * np.cos(2 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.sin(np.pi * tt) ** 4 + 0.01)
    inits.append(np.exp(-6.0 * (tt - 0.5) ** 2 / 0.1) + 0.01)
    
    # Multi-frequency
    inits.append(np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * tt + 0.7) +
                            0.2 * np.cos(10 * np.pi * tt), 0.01))
    inits.append(np.maximum(1.0 + 0.35 * np.sin(6 * np.pi * tt + 0.2) +
                            0.15 * np.cos(16 * np.pi * tt), 0.01))
    inits.append(np.maximum(1.0 + 0.25 * np.sin(2 * np.pi * tt) -
                            0.2 * np.cos(8 * np.pi * tt), 0.01))
    
    # Random
    inits.append(rng.random(n) + 0.05)
    inits.append(np.abs(rng.normal(0.9, 0.3, n)) + 0.02)
    
    # Sparse + dense mix
    base = np.ones(n) * 0.1
    n_sparse = n // 4
    sparsity = rng.random(n)
    base[sparsity > 0.75] += rng.uniform(0.5, 2.0, n_sparse)
    inits.append(base)
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 30
    
    print("Starting optimization, budget=%.0fs" % budget_s)
    
    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass
    
    # Choose n
    n = 8192
    S = np.sqrt(2.0 * n)
    global_best = np.inf
    global_a = None
    
    def check_update(a, val):
        nonlocal global_best, global_a
        obj = 2.0 * n * val / (a.sum() ** 2)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            return True
        return False
    
    # Phase 1: Quick screening with diverse initializations
    print("\n=== Phase 1: Quick screening ===")
    p1_budget = min(200.0, (deadline - total_start) * 0.1)
    inits = generate_inits(n, rng, given_seq=given)
    
    # If we have the given sequence, put it first and give it more time
    n_screen = min(6, len(inits))
    candidates = []
    
    for i in range(n_screen):
        t_elapsed = time.monotonic() - total_start
        remaining = deadline - t_elapsed
        if remaining < p1_budget * 0.3:
            break
        t_alloc = min(p1_budget / (n_screen - i), remaining * 0.15)
        if t_alloc < 10:
            break
        
        if i == 0 and given is not None:
            # Give the given sequence a longer optimization
            t_alloc = min(80.0, remaining * 0.3)
        
        a_opt, val = run_fista(inits[i], S, t_alloc, rng,
                                beta_init=5.0, beta_rate=0.008, beta_max=2000,
                                use_topk=False)
        obj = 2.0 * n * val / (a_opt.sum() ** 2)
        candidates.append((obj, a_opt.copy()))
        print("  Screen %d: obj=%.6f (max_b=%.6f)" % (i, obj, val))
        check_update(a_opt, val)
    
    candidates.sort(key=lambda x: x[0])
    print("  Best after screening: obj=%.6f" % global_best)
    
    # Phase 2: Basin hopping
    print("\n=== Phase 2: Basin hopping ===")
    bh_starts = min(3, len(candidates))
    for i in range(bh_starts):
        t_elapsed = time.monotonic() - total_start
        remaining = deadline - t_elapsed
        if remaining < 200:
            break
        t_alloc = min(remaining * (0.8 / (bh_starts - i + 1)), 500.0)
        if t_alloc < 60:
            break
        print("  BH from candidate %d (obj=%.6f), alloc=%.0fs" %
              (i, candidates[i][0], t_alloc))
        a_bh, val_bh = run_basin_hopping(candidates[i][1], S, t_alloc, rng)
        check_update(a_bh, val_bh)
        print("  BH %d done: obj=%.6f" % (i, 2.0 * n * val_bh / (a_bh.sum() ** 2)))
    
    # Phase 3: Peak-shaving polish
    print("\n=== Phase 3: Peak-shaving polish ===")
    if global_a is not None:
        t_elapsed = time.monotonic() - total_start
        remaining = deadline - t_elapsed
        p3_budget = min(remaining * 0.4, 500.0)
        
        n_sub = 2
        for sub in range(n_sub):
            t_remaining = deadline - time.monotonic()
            if t_remaining < 100:
                break
            t_alloc = min(p3_budget / (n_sub - sub), t_remaining * 0.4)
            if t_alloc < 30:
                break
            print("  Peak-shave sub %d/%d, t=%.0fs" % (sub + 1, n_sub, t_alloc))
            a_p, val_p = run_peak_shave(global_a, S, t_alloc, rng)
            check_update(a_p, val_p)
            print("  After sub %d: obj=%.6f" % (sub + 1, global_best))
    
    # Phase 4: High-beta final polish
    print("\n=== Phase 4: High-beta polish ===")
    if global_a is not None:
        t_elapsed = time.monotonic() - total_start
        remaining = deadline - t_elapsed - 10
        if remaining > 30:
            print("  High-beta polish, t=%.0fs" % remaining)
            a_hb, val_hb = run_high_beta_polish(global_a, S, remaining, rng)
            check_update(a_hb, val_hb)
            print("  Final: obj=%.6f" % global_best)
    
    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(n) * S / n
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(n) * S / n
        final_list = [float(x) for x in global_a]
    
    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.6f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
