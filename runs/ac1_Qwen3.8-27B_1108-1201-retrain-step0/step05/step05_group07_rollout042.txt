import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def project_simplex(v, S):
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def softmax(x):
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    return exp_x / exp_x.sum()

def lse_grad_softmax(x, beta, c=None, a=None):
    """
    Compute LSE objective and gradient w.r.t. x (softmax parameterization).
    Returns (lse_value, grad_x, a, c, w)
    """
    n = len(x)
    if a is None:
        a = softmax(x)
    if c is None:
        c = fft_convolve_self(a)
    
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 400)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    
    lse_val = cm + np.log(tot) / beta
    
    # Compute da_f = 2 * (conv(a,w)[m] + (a_rev * w)[n-1+m])
    fsize = 1 << (3 * n - 1).bit_length()  # need at least 3n-2
    
    A = fft(a, fsize)
    W = fft(w, fsize)
    conv1 = np.real(ifft(A * W))  # a * w, length 3n-2 (padded to fsize)
    
    a_rev = a[::-1]
    A_rev = fft(a_rev, fsize)
    conv2 = np.real(ifft(A_rev * W))  # a_rev * w, length 3n-2
    
    da_f = 2.0 * (conv1[:n] + conv2[n-1:2*n-1])
    
    # Gradient w.r.t. x: dx_f = a * (da_f - dot(da_f, a))
    dot_da_a = np.dot(da_f, a)
    dx_f = a * (da_f - dot_da_a)
    
    return lse_val, dx_f, a, c, w

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Uses softmax parameterization with LSE-smoothed objective optimized via L-BFGS,
multi-phase beta annealing, peak-targeted coordinate refinement, and length exploration.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_objective_direct(a):
    """Compute 2n*max(conv)/sum^2 directly."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s


def to_log_params(a):
    """Convert non-negative sequence to log-space parameters for softmax."""
    a = np.maximum(a, 1e-15)
    return np.log(a)


def lse_objective_and_grad(x, beta, n):
    """
    Compute LSE(beta * conv(a,a)) and its gradient w.r.t. x where a=softmax(x).
    Returns (lse_value, grad_x).
    """
    a = softmax(x)
    c = fft_convolve_self(a)

    cm = c.max()
    scaled = np.clip(beta * (c - cm), -500, 400)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(n)
    w = e / tot

    lse_val = cm + np.log(tot) / beta

    # Gradient: da_f[m] = 2 * (conv(a,w)[m] + (a_rev * w)[n-1+m])
    fsize = 1 << (3 * n - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    conv1 = np.real(ifft(A * W))

    a_rev = a[::-1]
    A_rev = fft(a_rev, fsize)
    conv2 = np.real(ifft(A_rev * W))

    da_f = 2.0 * (conv1[:n] + conv2[n - 1:2 * n - 1])

    # dx_f = a * (da_f - dot(da_f, a))
    dot_da_a = np.dot(da_f, a)
    dx_f = a * (da_f - dot_da_a)

    return lse_val, dx_f


def peak_targeted_reduce(a, c, num_peaks=10, num_iters=50, lr_scale=1.0):
    """
    Targeted reduction of the top peak lags in the convolution.
    Modifies a in-place (returns new a). Uses coordinate-style updates.
    """
    n = len(a)
    a_work = a.copy()
    c_work = c.copy()

    for it in range(num_iters):
        # Find top peak lag
        peak_idx = int(np.argmax(c_work))
        peak_val = c_work[peak_idx]

        # Find which coordinates contribute most to this peak
        # c[peak_idx] = sum_j a[j] * a[peak_idx - j]
        contrib = np.zeros(n)
        j_start = max(0, peak_idx - n + 1)
        j_end = min(n - 1, peak_idx)
        if j_start <= j_end:
            indices = peak_idx - np.arange(j_start, j_end + 1)
            contrib[j_start:j_end + 1] = a_work[indices]

        # Reduce the largest contributors
        top_k = min(5, n)
        top_indices = np.argsort(contrib)[::-1][:top_k]

        for idx in top_indices:
            old_val = a_work[idx]
            if old_val < 1e-15:
                continue
            # Reduce by a fraction
            reduction = lr_scale * 0.01 * old_val
            new_val = max(0.0, old_val - reduction)
            delta = new_val - old_val
            if abs(delta) < 1e-18:
                continue

            # Update convolution incrementally
            # c[k] changes by delta * (a[k-idx] + a[idx-k]) for k != 2*idx
            # c[2*idx] changes by delta^2 (correction)
            k_range1 = np.arange(max(0, idx), min(2 * n - 1, idx + n))
            c_work[k_range1] += delta * a_work[k_range1 - idx]

            k_range2 = np.arange(max(0, idx - n + 1), min(2 * n - 1, idx + 1))
            c_work[k_range2] += delta * a_work[idx - k_range2]

            if 2 * idx < 2 * n - 1:
                c_work[2 * idx] += delta * delta

            a_work[idx] = new_val

    return a_work


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def run_lbfgs_phase(x_init, beta, n, max_time, maxiter=20000):
    """Run L-BFGS-B on the LSE objective for a given beta."""
    start = time.monotonic()
    best_obj = compute_objective_direct(softmax(x_init))
    best_x = x_init.copy()
    best_a = softmax(x_init)
    eval_count = [0]

    def obj_grad(x):
        eval_count[0] += 1
        val, grad = lse_objective_and_grad(x, beta, n)
        return val, grad

    def callback(xk):
        a = softmax(xk)
        obj = compute_objective_direct(a)
        if obj < best_obj:
            nonlocal best_obj, best_x, best_a
            best_obj = obj
            best_x = xk.copy()
            best_a = a.copy()

    try:
        res = minimize(obj_grad, x_init, method='L-BFGS-B', jac=True,
                       callback=callback,
                       options={'maxiter': maxiter, 'gtol': 1e-8,
                                'maxfun': 100000, 'iprint': -1})
    except Exception as e:
        print(f"    L-BFGS error: {e}")

    elapsed = time.monotonic() - start
    print(f"    beta={beta:.0f}, time={elapsed:.1f}s, evals={eval_count[0]}, best_obj={best_obj:.6f}")
    return best_a, best_obj, best_x

def coordinate_refine(a_init, max_time, rng, n_peaks=5, verbose=True):
    """
    Coordinate descent refinement targeting peak lags.
    Normalizes to sum=1.
    """
    a = project_simplex(a_init.copy(), 1.0)
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    start = time.monotonic()
    iteration = 0

    while time.monotonic() - start < max_time:
        iteration += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # Get top peak lags
        top_peaks = np.argsort(c)[::-1][:n_peaks]

        # For each peak, try to reduce it by adjusting contributing coordinates
        for pk in top_peaks:
            pk = int(pk)
            # Contributions: a[j] * a[pk-j] for valid j
            j_start = max(0, pk - n + 1)
            j_end = min(n - 1, pk)
            if j_start > j_end:
                continue

            # Find the largest contributor
            contrib = a[j_start:j_end + 1] * a[pk - np.arange(j_start, j_end + 1)]
            local_max_idx = np.argmax(contrib)
            global_idx = j_start + local_max_idx

            old_val = a[global_idx]
            if old_val < 1e-14:
                continue

            # Binary search for the value that reduces this peak
            # while not increasing other peaks too much
            # Simple: reduce by a small fraction
            frac = 0.005 + 0.01 * (c[pk] - c.mean()) / max(c.std(), 1e-15)
            frac = min(frac, 0.05)
            new_val = old_val * (1.0 - frac)

            delta = new_val - old_val
            # Update convolution
            ks1 = np.arange(max(0, global_idx), min(2 * n - 1, global_idx + n))
            c[ks1] += delta * a[ks1 - global_idx]
            ks2 = np.arange(max(0, global_idx - n + 1), min(2 * n - 1, global_idx + 1))
            c[ks2] += delta * a[global_idx - ks2]
            if 2 * global_idx < 2 * n - 1:
                c[2 * global_idx] += delta * delta
            a[global_idx] = new_val

        # Also try to increase small values that help flatten
        if iteration % 50 == 0:
            small_mask = a < 0.01 * a.mean()
            if np.any(small_mask):
                idxs = np.where(small_mask)[0]
                pick = rng.choice(idxs, size=min(10, len(idxs)), replace=False)
                for idx in pick:
                    boost = 0.001 * a.mean()
                    delta = boost
                    a[idx] += delta
                    ks1 = np.arange(max(0, int(idx)), min(2 * n - 1, int(idx) + n))
                    c[ks1] += delta * a[ks1 - int(idx)]
                    ks2 = np.arange(max(0, int(idx) - n + 1), min(2 * n - 1, int(idx) + 1))
                    c[ks2] += delta * a[int(idx) - ks2]
                    if 2 * int(idx) < 2 * n - 1:
                        c[2 * int(idx)] += delta * delta

        # Renormalize periodically to avoid drift
        if iteration % 200 == 0:
            a = project_simplex(a, 1.0)
            c = fft_convolve_self(a)

        # Noise injection if stuck
        if iteration % 500 == 0:
            noise = rng.normal(0, 0.001 * a.mean(), n)
            a = project_simplex(a + noise, 1.0)
            c = fft_convolve_self(a)

        if verbose and iteration % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"    coord_it={iteration}, t={elapsed:.0f}s, obj={best_obj:.6f}, "
                  f"cur_max={cur_max:.2e}, spread={c.max()/c.mean():.3f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. L-BFGS with LSE smoothing (increasing beta)
    2. Peak-targeted coordinate refinement
    3. Length exploration
    """
    rng = np.random.default_rng(seed)
    abs_deadline = time.monotonic() + budget_s
    deadline = abs_deadline - 30  # Leave buffer for final evaluation

    print(f"Starting optimization, budget={budget_s}s")

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0.01 and len(a_provided) >= 64:
            print(f"Using provided sequence: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_global(a, msg=""):
        nonlocal global_best, global_a
        if a is None or len(a) < 2:
            return
        a_clean = np.clip(a, 0, 1000)
        if a_clean.sum() < 0.01:
            return
        obj = compute_objective_direct(a_clean)
        if obj < global_best:
            global_best = obj
            global_a = a_clean.copy()
            print(f"  ★ NEW GLOBAL BEST: obj={obj:.6f} ({msg})")

    # === Phase 1: L-BFGS optimization on provided length ===
    if a_provided is not None:
        n1 = len(a_provided)
        S1 = a_provided.sum()
        a1_norm = a_provided / S1  # Normalize to sum=1

        t1 = min(800, (deadline - time.monotonic()) * 0.40)
        print(f"\nPhase 1: L-BFGS LSE optimization, n={n1}, time={t1:.0f}s")
        a_opt, obj_opt = run_scipy_smoothed(a1_norm, n1, t1, rng)
        update_global(a_opt, "Phase 1 L-BFGS")

        # === Phase 2: Coordinate refinement ===
        remaining = deadline - time.monotonic()
        t2 = min(400, remaining * 0.30)
        if t2 > 30 and global_a is not None:
            print(f"\nPhase 2: Coordinate refinement, time={t2:.0f}s")
            a_coord, obj_coord = coordinate_refine(global_a, t2, rng, n_peaks=8)
            update_global(a_coord, "Phase 2 coordinate")

            # Re-run L-BFGS from the coordinate result with high beta
            remaining = deadline - time.monotonic()
            t2b = min(200, remaining * 0.30)
            if t2b > 20:
                print(f"\nPhase 2b: Post-coordinate L-BFGS polish")
                x2 = to_log_params(project_simplex(global_a, 1.0))
                a_pol, obj_pol = run_scipy_smoothed(global_a, len(global_a), t2b, rng)
                update_global(a_pol, "Phase 2b polish")

    else:
        # No provided sequence, generate from scratch
        n1 = 8192
        a1 = np.ones(n1) / n1
        print(f"\nNo provided sequence, starting with uniform n={n1}")

        t1 = min(600, (deadline - time.monotonic()) * 0.35)
        print(f"\nPhase 1: L-BFGS from uniform, n={n1}, time={t1:.0f}s")
        a_opt, obj_opt = run_scipy_smoothed(a1, n1, t1, rng)
        update_global(a_opt, "Phase 1 from uniform")

    # === Phase 3: Try alternative lengths ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        lengths_to_try = [16384, 4096]
        for n_try in lengths_to_try:
            if time.monotonic() > deadline - 60:
                break
            if global_a is not None and len(global_a) == n_try:
                continue

            t_alloc = min(300, (deadline - time.monotonic()) * 0.25)
            if t_alloc < 60:
                continue

            print(f"\nPhase 3: Trying n={n_try}, time={t_alloc:.0f}s")
            if global_a is not None and len(global_a) > 64:
                # Interpolate global best to new length
                t_old = np.linspace(0, 1, len(global_a))
                t_new = np.linspace(0, 1, n_try)
                a_init = np.interp(t_new, t_old, global_a)
                a_init = np.maximum(a_init, 0)
                a_init = project_simplex(a_init, 1.0)
            else:
                a_init = np.ones(n_try) / n_try

            a_opt, obj_opt = run_scipy_smoothed(a_init, n_try, t_alloc * 0.7, rng)
            update_global(a_opt, f"Phase 3 n={n_try}")

            # Quick coordinate refinement
            rem = deadline - time.monotonic()
            if rem > 30:
                a_cr, obj_cr = coordinate_refine(a_opt, min(120, rem * 0.5), rng, n_peaks=5)
                update_global(a_cr, f"Phase 3 n={n_try} coord")

    # === Phase 4: Final intensive refinement on global best ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        n = len(global_a)
        print(f"\nPhase 4: Final refinement, n={n}, time={remaining:.0f}s")

        # Alternate between L-BFGS (high beta) and coordinate refinement
        a_curr = global_a.copy()
        for phase_i in range(6):
            if time.monotonic() > deadline - 15:
                break
            t_phase = min(remaining / (6 - phase_i), deadline - time.monotonic() - 15)
            if t_phase < 10:
                break

            if phase_i % 2 == 0:
                # L-BFGS with very high beta
                x_curr = to_log_params(project_simplex(a_curr, 1.0))
                a_curr, _, _ = run_lbfgs_phase(x_curr, 10000, n, t_phase, maxiter=3000)
            else:
                # Coordinate refinement
                a_curr, _ = coordinate_refine(a_curr, t_phase, rng, n_peaks=10)

            update_global(a_curr, f"Phase 4 iter {phase_i}")
            a_curr = global_a.copy()  # Always start from global best

        # Final perturbations
        remaining = deadline - time.monotonic()
        if remaining > 20:
            for ri in range(3):
                if time.monotonic() > deadline - 10:
                    break
                noise_scale = 0.01 * global_a.mean()
                a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n), 1.0)
                t_per = min(60, (deadline - time.monotonic() - 10) / (3 - ri))
                if t_per > 10:
                    a_cr, _ = coordinate_refine(a_pert, t_per, rng, n_peaks=5)
                    update_global(a_cr, f"Phase 4 perturb {ri}")

    # === Final verification and return ===
    if global_a is None:
        print("WARNING: No valid solution, using fallback")
        global_a = np.ones(256) / 256.0
        global_best = compute_objective_direct(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) / 256.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: len={n_final}, internal_obj={global_best:.6f}")
    print(f"evaluate_sequence verdict: {ver:.6f}")
    print(f"{'='*60}")

    return final_list
