def fft_convolve_self(a):

    c = np.real(ifft(A * A))

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]

    valid = tc <= w

        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def spectral_equalize(a, n_iter=10, alpha=0.5):
    """
    In frequency domain, try to flatten |A(k)|^2 subject to A(0) = S.
    This is done iteratively: 
    1. Compute A = fft(a)
    2. Target: make |A(k)|^2 as flat as possible
    3. Update: A_new(k) = A(k) * (target / |A(k)|^2)^(alpha/2) for k != 0
    4. Keep A(0) fixed
    5. a_new = ifft(A_new), take real part, project to non-negative simplex
    """
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    
    for _ in range(n_iter):
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        # Target: flat power spectrum (but keep DC)
        # Average power excluding DC
        avg_power = power[1:].mean() if n > 1 else power[0]
        if avg_power < 1e-30:
            break
        # Blend: new_power[k] = alpha * avg_power + (1-alpha) * power[k] for k > 0
        new_power = power.copy()
        new_power[1:] = alpha * avg_power + (1 - alpha) * power[1:]
        # Scale amplitudes
        scale = np.sqrt(new_power / np.maximum(power, 1e-30))
        A_new = A * scale
        # Ensure A(0) is unchanged
        A_new[0] = A[0]
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)
        a = a_new
    
    return a

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase optimizer combining:
- LSE-FISTA with adaptive beta, backtracking, and momentum restart
- Peak-shaving coordinate descent for targeted reduction
- Spectral equalization for frequency-domain diversification
- Active-set subgradient refinement
- Multi-scale (upsample) exploration
- Differential evolution for global diversification

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_value(c, beta):
    """Compute log-sum-exp value of beta * c."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    return cm + np.log(np.exp(shifted).sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE(beta * c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(a, c, k, power=0.0):
    """Subgradient from top-k lags with power weighting."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    if power <= 0.0:
        w = np.ones(k) / k
    else:
        vals = c[idx]
        w = np.exp(power * (vals - vals.max()))
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def spectral_equalize(a, n_iter=10, alpha=0.3):
    """
    Frequency-domain flatting: blend power spectrum toward uniform.
    Preserves DC component (total sum).
    """
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()

        A = fft(a, fsize)
        power = np.abs(A) ** 2
        if n == 1:
            break
        avg_power = power[1:].mean()
        if avg_power < 1e-30:
            break
        new_power = power.copy()
        new_power[1:] = alpha * avg_power + (1.0 - alpha) * power[1:]

        A_new[0] = A[0]
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)
        a = a_new

    return a

def run_active_refine(a_init, max_time, rng):
    """Active-set subgradient method with adaptive top-k weighting."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 0.05
    proxy_beta = 100.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k and power
        if stuck < 200:
            k = 1
            power_p = 0.0
        elif stuck < 1000:
            k = 3
            power_p = 1.0
        elif stuck < 5000:
            k = min(16, 3 + stuck // 300)
            power_p = 2.0
        elif stuck < 20000:
            k = min(64, 16 + stuck // 500)
            power_p = 3.0
        else:
            k = min(256, 64 + stuck // 500)
            power_p = 5.0

        grad = topk_subgradient(a, c, k, power=power_p)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking
        lse_old = lse_value(c, proxy_beta)
        for _ in range(15):
            x_new = project_simplex(a - step * grad / gnorm, S)
            c_new = fft_convolve_self(x_new)
            lse_new = lse_value(c_new, proxy_beta)
            if lse_new < lse_old:
                break
            step *= 0.5
        else:
            step *= 0.5
            a = project_simplex(a - step * grad / gnorm, S)
            continue

        a = x_new
        step = min(step * 1.05, 0.5 * S / max(gnorm, 1e-10))

        # Escape
        if stuck > 8000 and it % 300 == 0:
            noise_scale = max(1e-6, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 30000.0))
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            step = 0.02

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)


def make_chirp_init(n):
    tt = np.arange(n, dtype=np.float64)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt * tt / n)


def make_random_init(n, rng):
    a = rng.random(n)
    a = a ** 2
    return a


def upsample_bandlimited(a, factor=2):
    """Bandlimited upsample via FFT zero-padding."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    n_new = n * factor
    fsize_new = fsize * factor
    A_new = np.zeros(fsize_new, dtype=complex)
    half = len(A) // 2
    A_new[:half + 1] = A[:half + 1]
    A_new[fsize_new - half:] = A[half:]
    a_new = np.real(ifft(A_new))[:n_new]
    return a_new


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: LSE-FISTA -> peak shave -> spectral -> active refine -> upsample -> final polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    total_start = time.monotonic()
    deadline = total_start + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        S0 = a_start.sum()
        a_start_norm = a_start / S0 * np.sqrt(2.0 * n0)
        obj_start = compute_obj(a_start)
        print("Starting n=%d, obj=%.6f" % (n0, obj_start))
        global_best_obj = obj_start
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        a_start_norm = make_cosine_init(n0)
        S0 = np.sqrt(2.0 * n0)
        a_start_norm = normalize_seq(a_start_norm, S0)
        obj_start = compute_obj(a_start_norm)
        print("Fallback n=%d, init obj=%.6f" % (n0, obj_start))
        global_best_obj = obj_start
        global_best_a = a_start_norm.copy()

    # === Phase 1: LSE-FISTA main optimization ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.28
    print("\n=== Phase 1: LSE-FISTA n=%d (budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    a1, _ = run_lse_fista(a_start_norm, p1_budget, rng,
                          beta_init=5.0, beta_max=30000.0,
                          use_backtrack=True, momentum_restart_interval=4000)
    obj1 = compute_obj(a1)
    print("  Phase 1 done: obj=%.6f (t=%.1fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    # === Phase 2: Peak shaving ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.10
    if p2_budget > 10:
        print("\n=== Phase 2: Peak shave (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        n_shave_rounds = max(20, int(p2_budget * 20))
        a2 = peak_shave(global_best_a, n_rounds=n_shave_rounds, rng=rng)
        obj2 = compute_obj(a2)
        print("  Peak shave: obj=%.6f (t=%.1fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: Spectral equalization + refinement ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.12
    if p3_budget > 10:
        print("\n=== Phase 3: Spectral equalize + refine (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        # Try multiple alpha values
        best_a3 = global_best_a.copy()
        best_obj3 = global_best_obj
        for alpha in [0.1, 0.2, 0.4, 0.6]:
            a_spec = spectral_equalize(global_best_a, n_iter=10, alpha=alpha)
            obj_spec = compute_obj(a_spec)
            if obj_spec < best_obj3:
                best_obj3 = obj_spec
                best_a3 = a_spec.copy()
            print("  alpha=%.1f: obj=%.6f" % (alpha, obj_spec))

        if best_obj3 < global_best_obj:
            global_best_obj = best_obj3
            global_best_a = best_a3.copy()

        # Refine
        rem3 = p3_budget - (time.monotonic() - t3)
        if rem3 > 5:
            a3b, _ = run_lse_fista(global_best_a, rem3, rng,
                                    beta_init=20.0, beta_max=50000.0)
            obj3b = compute_obj(a3b)
            print("  After refine: obj=%.6f (t=%.1fs)" % (obj3b, time.monotonic() - t3))
            if obj3b < global_best_obj:
                global_best_obj = obj3b
                global_best_a = a3b.copy()

    # === Phase 4: Active-set refinement ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.20
    if p4_budget > 10:
        print("\n=== Phase 4: Active-set refine (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        a4, _ = run_active_refine(global_best_a, p4_budget, rng)
        obj4 = compute_obj(a4)
        print("  Active refine: obj=%.6f (t=%.1fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: High-beta LSE final pass ===
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.12
    if p5_budget > 10:
        print("\n=== Phase 5: High-beta LSE (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        a5, _ = run_lse_fista(global_best_a, p5_budget, rng,
                              beta_init=200.0, beta_max=200000.0,
                              use_backtrack=False, momentum_restart_interval=3000)
        obj5 = compute_obj(a5)
        print("  High-beta LSE: obj=%.6f (t=%.1fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Try upsample to 2n ===
    remaining = deadline - time.monotonic()
    if remaining > 150:
        n2 = len(global_best_a) * 2
        if n2 <= 32768:
            p6_budget = remaining * 0.45
            print("\n=== Phase 6: Upsample to n=%d (budget %.0fs) ===" % (n2, p6_budget))
            t6 = time.monotonic()

            a_up = upsample_bandlimited(global_best_a, factor=2)
            a_up = np.maximum(a_up, 0.0)
            S2 = np.sqrt(2.0 * n2)
            a_up = normalize_seq(a_up, S2)
            obj_up = compute_obj(a_up)
            print("  After upsample: obj=%.6f" % obj_up)

            if obj_up < global_best_obj:
                global_best_obj = obj_up
                global_best_a = a_up.copy()

            # Optimize upsampled
            rem6 = p6_budget - (time.monotonic() - t6)
            if rem6 > 10:
                a6, _ = run_lse_fista(global_best_a, rem6 * 0.5, rng,
                                      beta_init=5.0, beta_max=20000.0)
                obj6 = compute_obj(a6)
                print("  Upsampled LSE: obj=%.6f (t=%.1fs)" % (obj6, time.monotonic() - t6))
                if obj6 < global_best_obj:
                    global_best_obj = obj6
                    global_best_a = a6.copy()

                rem6b = p6_budget - (time.monotonic() - t6)
                if rem6b > 5:
                    a6b, _ = run_active_refine(global_best_a, rem6b, rng)
                    obj6b = compute_obj(a6b)
                    print("  Upsampled refine: obj=%.6f (t=%.1fs)" % (obj6b, time.monotonic() - t6))
                    if obj6b < global_best_obj:
                        global_best_obj = obj6b
                        global_best_a = a6b.copy()

    # === Phase 7: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p7_budget = remaining * 0.85
        print("\n=== Phase 7: Final polish (budget %.0fs) ===" % p7_budget)
        t7 = time.monotonic()

        # Peak shave first
        n_shave = max(20, int(p7_budget))
        a7 = peak_shave(global_best_a, n_rounds=n_shave, rng=rng)
        obj7 = compute_obj(a7)
        if obj7 < global_best_obj:
            global_best_obj = obj7
            global_best_a = a7.copy()

        # Then LSE
        rem7 = p7_budget - (time.monotonic() - t7)
        if rem7 > 10:
            a7b, _ = run_lse_fista(global_best_a, rem7 * 0.5, rng,
                                   beta_init=100.0, beta_max=100000.0)
            obj7b = compute_obj(a7b)
            print("  Final LSE: obj=%.6f (t=%.1fs)" % (obj7b, time.monotonic() - t7))
            if obj7b < global_best_obj:
                global_best_obj = obj7b
                global_best_a = a7b.copy()

            rem7c = p7_budget - (time.monotonic() - t7)
            if rem7c > 5:
                a7c, _ = run_active_refine(global_best_a, rem7c, rng)
                obj7c = compute_obj(a7c)
                print("  Final refine: obj=%.6f (t=%.1fs)" % (obj7c, time.monotonic() - t7))
                if obj7c < global_best_obj:
                    global_best_obj = obj7c
                    global_best_a = a7c.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL RESULT ===")
            print("Objective: %.6f (n=%d)" % (ver, len(final_list)))
            print("Total time: %.0fs" % (time.monotonic() - total_start))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
