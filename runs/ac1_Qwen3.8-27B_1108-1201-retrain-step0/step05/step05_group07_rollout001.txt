import numpy as np

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_smoothed_max(c, beta):
    """Compute LSE approximation of max(c) and weights."""
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.ones(len(c)) / len(c)
    lse = cm + np.log(tot) / beta
    w = e / tot
    return lse, w


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of self-convolution."""
    n = len(a)
    _, w = lse_smoothed_max(c, beta)

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_full = 2.0 * np.real(ifft(W * np.conj(A)))
    return grad_full[:n]


def subgradient_topk(a, c, k):

def hybrid_subgradient_lse(a0, S, max_time, rng, verbose=True):

    """

    mu = 0.9  # momentum coefficient

    it = 0
    best_obj = fft_convolve_self(a).max()
    best_a = a.copy()

        it += 1
        c = fft_convolve_self(a)

            best_obj = cur_max
            best_a = a.copy()

            stuck += 1
        
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        # Choose method: LSE for most of the time, exact subgradient near the end

            grad = subgradient_topk(a, c, k)

        if gnorm < 1e-15:
            break
        
        # Step size with backtracking

                a = a_try
                accepted = True

            velocity = np.zeros(n)

            a = project_simplex(a + rng.normal(0, scale, n), S)

def structured_init(n, rng, mode=0):
    """Generate structured initial sequences."""
    t = np.linspace(0, 1, n, endpoint=False)
    if mode == 0:
        # Raised cosine
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) + 0.01
    elif mode == 1:
        # Hann window
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) + 0.01
    elif mode == 2:
        # Hamming window
        a = 0.54 + 0.46 * np.cos(2.0 * np.pi * t) + 0.01
    elif mode == 3:
        # Two-frequency
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 2 * t) + 0.2 * np.cos(2.0 * np.pi * 5 * t + 0.5)
        a = np.abs(a) + 0.01
    elif mode == 4:
        # Gaussian-like
        a = np.exp(-2.0 * (t - 0.5) ** 2) + 0.1
    elif mode == 5:
        # Random smoothed
        r = rng.random(n)
        # Simple boxcar smoothing
        kernel = np.ones(8) / 8.0
        a = np.convolve(r, kernel, mode='same') + 0.01
    elif mode == 6:
        # Alternating with decay
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t * 3) * np.exp(-3.0 * np.abs(t - 0.5))
        a = np.abs(a) + 0.01
    else:
        a = np.ones(n) * 0.1
    
    return np.maximum(a, 0.0)
