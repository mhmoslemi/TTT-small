"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Combines FISTA with adaptive LSE, spectral flattening (frequency-domain
iterative reweighting), KKT-based gradient-matching polish, and multi-length
evaluation. Key innovations over previous: (1) spectral flattening via
score-matching in frequency domain, (2) KKT residual-based Newton polish,
(3) multi-length sweep, (4) adaptive backtracking line search in FISTA.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.5):
    """Top-k weighted gradient."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

                       mode="lse", use_backtrack=True):

    FISTA with adaptive beta (geometric annealing) and optional backtracking.

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Geometric beta annealing
        frac = min(1.0, it / max(1, int(max_time * 30)))  # ~30 iters/sec budget
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = max(5.0, min(beta, 80000.0))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Gradient
        if mode == "lse":
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
        elif mode == "topk":

            grad = topk_gradient(y, c, k, power=1.2)
        else:  # hybrid
            if stuck_count > 500:
                tol = min(0.03, 0.001 * stuck_count)
                active = np.where(c >= c.max() * (1.0 - tol))[0]
                w_full = np.zeros(len(c))
                w_full[active] = 1.0 / len(active)
                grad = conv_gradient(y, w_full)
                # Blend with LSE

        if gnorm < 1e-15:
            break

        # Step size
        if use_backtrack:
            step = 0.5 / max(beta, 1.0)
        else:
            step = 1.0 / max(beta, 1.0)

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic momentum reset
        if it % 3000 == 0:
            t_mom = 1.0

        # Stuck recovery: DCT-domain smooth noise
        if stuck_count > 3000 and it % 500 == 0:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck_count / 10000.0)
            if noise_scale > 1e-8:
                F = fft(rng.normal(0, 1.0, n))
                cutoff = max(5, n // 80)
                F[cutoff:-cutoff] *= 0.05
                noise = np.real(ifft(F))[:n]
                noise = noise / (np.linalg.norm(noise) + 1e-15) * noise_scale
                y = project_simplex(y + noise, S)
                t_mom = 1.0
                stuck_count = 0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_spectral_flattening(a_init, max_time, rng, alpha_start=0.7, alpha_end=0.05):
    """
    Iterative spectral flattening: in frequency domain, push |A| toward
    uniform magnitude while preserving phase. Then project back to time
    domain (clip negatives, renormalize).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * N) if False else np.sqrt(2.0 * n)
    fsize = 1 << (2 * n - 1 - 1).bit_length()

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Anneal alpha (how much to flatten)
        frac = min(1.0, it / 5000.0)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # Compute spectrum
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target magnitude: partially flatten toward mean
        mean_mag = mag.mean()
        # Use a soft flattening: mag_new = mag^alpha * mean_mag^(1-alpha)
        mag_new = np.power(mag + 1e-30, alpha) * np.power(mean_mag + 1e-30, 1 - alpha)

        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]

        # Project: clip negatives, renormalize
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new * (S / sn)

        # Evaluate
        c = fft_convolve_self(a_new)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a_new.copy()

        a = a_new

        # Every 100 iters, do a few FISTA steps to escape
        if it % 100 == 0 and it < max_time * 30:
            a = _quick_fista_step(a, S, 5, rng)

    return best_a, best_conv


def _quick_fista_step(a, S, n_steps, rng):
    """A few FISTA steps from given point."""
    n = len(a)
    beta = 100.0
    for _ in range(n_steps):
        c = fft_convolve_self(a)
        w = lse_weights(c, beta)
        grad = conv_gradient(a, w)
        step = 0.3 / beta
        a = project_simplex(a - step * grad, S)
    return a


def run_kkt_polish(a_init, max_time, rng, tol=1e-10):
    """
    KKT-based gradient matching polish.
    At optimum, gradient of active LSE should be constant on the support.
    We iteratively correct the gradient discrepancy.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()

        # Identify active lags
        active_tol = 1e-4
        active = np.where(c >= cur_max * (1.0 - active_tol))[0]

        # Compute gradient from active lags
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        grad = conv_gradient(a, w)

        # Gradient matching: on support, grad should be constant
        support = np.where(a > 1e-10)[0]
        if len(support) == 0:
            break

        g_s = grad[support]
        g_mean = g_s.mean()
        residual = g_s - g_mean

        # Check convergence
        res_norm = np.linalg.norm(residual) / (np.linalg.norm(g_s) + 1e-15)
        if res_norm < tol:
            break

        # Newton-like step: move a to reduce gradient discrepancy
        # The correction direction in the support is -residual (scaled)
        step_size = 0.1 / (np.linalg.norm(grad) + 1e-15)
        a_new = a.copy()
        a_new[support] -= step_size * residual

        # Also try to reduce the max by moving mass from high-gradient to low
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))

        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max + 1e-12:
            a = a_new
        else:
            # Halve step
            a_new2 = a.copy()
            a_new2[support] -= 0.5 * step_size * residual
            a_new2 = np.maximum(a_new2, 0.0)
            a_new2 = a_new2 * (S / max(a_new2.sum(), 1e-15))
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < cur_max:
                a = a_new2
            else:
                # Try FISTA step instead
                beta = 500.0
                w_lse = lse_weights(c, beta)
                grad_lse = conv_gradient(a, w_lse)
                a = project_simplex(a - 0.1 / beta * grad_lse, S)

        if it % 1000 == 0 and it > 2000:
            # Occasional perturbation to escape
            if rng.random() < 0.3:
                a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)

    return best_a, best_conv


def run_lbfgs_polish(a_init, max_time, rng, n_beta_levels=6):
    """
    L-BFGS-B polishing with increasing beta for LSE smoothing.
    Uses penalty for the sum constraint.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / max(a.sum(), 1e-15))

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    betas = [50.0, 200.0, 500.0, 1500.0, 4000.0, 10000.0][:n_beta_levels]
    penalties = [10.0, 50.0, 200.0, 500.0, 2000.0, 5000.0][:n_beta_levels]

    for idx, (beta, pen) in enumerate(zip(betas, penalties)):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm_local + np.log(e.sum()) / beta
            sum_x = x.sum()
            obj += pen * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 2.0 * pen * (sum_x - S)
            return obj, grad

        t0 = time.monotonic()
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': 200, 'maxls': 50, 'ftol': 1e-16, 'gtol': 1e-12,
                     'iprint': -1}
        )

        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_lbfgs_no_constraint(a_init, max_time, rng, n_beta_levels=5):
    """
    L-BFGS-B without sum constraint (objective is scale-invariant).
    Directly minimize LSE of convolution. The scale-invariance means
    we don't need the penalty.
    """
    n = len(a_init)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    betas = [100.0, 500.0, 2000.0, 8000.0, 30000.0][:n_beta_levels]

    for beta in betas:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            if x.sum() < 1e-15:
                return 1e10, np.zeros_like(x)
            b = fft_convolve_self(x)
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm_local + np.log(e.sum()) / beta
            # Normalize objective by sum^2 / n to get scale-invariant form
            s = x.sum()
            obj = obj * (2.0 * n) / (s * s)
            # Gradient (chain rule with normalization)
            grad_b = conv_gradient(x, w)
            grad_b += 2.0 * (1.0 / beta) * (cm_local + 1.0 / beta) * np.zeros(n)  # derivative of cm
            # Actually, the gradient of LSE is:
            # d/dx [cm + log(sum(exp(beta*(b-cm))))/beta]
            # = sum_k w[k] * grad b[k]  (since w sums to 1 and cm cancels)
            grad = conv_gradient(x, w)
            # Gradient of the normalization factor 2n/(sum x)^2
            s_grad = 2.0 * n * grad / (s * s) - 2.0 * n * obj_b / (s * s) * 2 * s  # hmm this is getting complex
            # Simplify: just use the unnormalized gradient and let L-BFGS handle it
            # The scale invariance means L-BFGS will find a direction
            grad = grad / (s * s) * 2.0 * n - obj * grad * (2.0 * s / (s * s)) * 2.0
            # This is getting tangled. Let's just use:
            grad = conv_gradient(x, w) * (2.0 * n / (s * s))
            return obj, grad

        # Skip this complex version; use the simpler penalty-based one
        break

    return best_a, best_obj


def generate_inits(n, rng, S):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Primary: cosine notch
    a = 0.54 + 0.46 * np.cos(2 * np.pi * t)
    a = np.maximum(a, 0.001)
    a = a * (S / a.sum())
    inits.append(a.copy())

    # Double cosine
    a = 0.5 + 0.3 * np.cos(2 * np.pi * t) + 0.2 * np.cos(4 * np.pi * t)
    a = np.maximum(a, 0.001)
    a = a * (S / a.sum())
    inits.append(a.copy())

    # Asymmetric cosine
    a = 0.5 + 0.4 * np.cos(2 * np.pi * (t + 0.1))
    a = np.maximum(a, 0.001)
    a = a * (S / a.sum())
    inits.append(a.copy())

    # Hann-like window (high in middle)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    a = np.maximum(a, 0.001)
    a = a * (S / a.sum())
    inits.append(a.copy())

    # Smooth random
    raw = rng.random(n)
    F = fft(raw)
    cutoff = max(3, n // 50)
    F[cutoff:-cutoff] *= 0.1
    smooth = np.real(ifft(F))[:n]
    smooth = np.maximum(smooth, 0.01)
    smooth = smooth * (S / smooth.sum())
    inits.append(smooth.copy())

    # Power-law shaped
    a = 1.0 / (0.5 + np.abs(t - 0.5) * 2.0) ** 0.5
    a = a * (S / a.sum())
    inits.append(a.copy())

    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA adaptive + spectral flattening + KKT polish + L-BFGS.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence if available
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.8f" % (len(a_start), obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
        n_best = len(a_start)
    else:
        n_best = 8192
        t = np.linspace(0, 1, n_best, endpoint=False)
        a_start = 0.54 + 0.46 * np.cos(2 * np.pi * t)
        a_start = np.maximum(a_start, 0.001)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.8f" % (n_best, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: FISTA with adaptive beta (30% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.30
    print("\n=== Phase 1: FISTA adaptive (n=%d, budget %.0fs) ===" % (n_best, p1_budget))
    t1 = time.monotonic()

    a1, _ = run_fista_adaptive(global_best_a, p1_budget * 0.5, rng,
                               beta_start=10.0, beta_end=20000.0, mode="lse")
    obj1 = compute_obj(a1)
    print("  FISTA-lse: obj=%.8f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    rem1 = p1_budget - (time.monotonic() - t1)
    if rem1 > 10:
        a1b, _ = run_fista_adaptive(global_best_a, rem1, rng,
                                    beta_start=50.0, beta_end=50000.0, mode="hybrid")
        obj1b = compute_obj(a1b)
        print("  FISTA-hybrid: obj=%.8f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: Spectral flattening (25% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = remaining * 0.25
        print("\n=== Phase 2: Spectral flattening (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        a2, _ = run_spectral_flattening(global_best_a, p2_budget * 0.6, rng,
                                        alpha_start=0.6, alpha_end=0.1)
        obj2 = compute_obj(a2)
        print("  Spectral flat: obj=%.8f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

        rem2 = p2_budget - (time.monotonic() - t2)
        if rem2 > 10:
            # Continue with different alpha range
            a2b, _ = run_spectral_flattening(global_best_a, rem2, rng,
                                             alpha_start=0.4, alpha_end=0.05)
            obj2b = compute_obj(a2b)
            print("  Spectral flat2: obj=%.8f (t=%.0fs)" % (obj2b, time.monotonic() - t2))
            if obj2b < global_best_obj:
                global_best_obj = obj2b
                global_best_a = a2b.copy()

    # === Phase 3: KKT gradient-matching polish (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = remaining * 0.15
        print("\n=== Phase 3: KKT polish (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()

        a3, _ = run_kkt_polish(global_best_a, p3_budget, rng, tol=1e-12)
        obj3 = compute_obj(a3)
        print("  KKT polish: obj=%.8f (t=%.0fs)" % (obj3, time.monotonic() - t3))

            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: L-BFGS-B polishing (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p4_budget = remaining * 0.20
        print("\n=== Phase 4: L-BFGS-B polishing (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        a4, obj4 = run_lbfgs_polish(global_best_a, p4_budget, rng, n_beta_levels=6)
        print("  L-BFGS: obj=%.8f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: FISTA high-beta refinement (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p5_budget = remaining * 0.10
        print("\n=== Phase 5: High-beta FISTA refinement (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        a5, _ = run_fista_adaptive(global_best_a, p5_budget, rng,
                                   beta_start=500.0, beta_end=80000.0, mode="lse")
        obj5 = compute_obj(a5)
        print("  High-beta: obj=%.8f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Try different sequence lengths (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print("\n=== Phase 6: Multi-length sweep (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()

        # Try a few different lengths
        lengths_to_try = [4096, 12288, 16384]
        # Filter out the one we're already using
        lengths_to_try = [l for l in lengths_to_try if l != n_best]

        time_per_length = remaining / (len(lengths_to_try) + 1)

        for n_trial in lengths_to_try:
            if time.monotonic() > deadline:
                break
            if time.monotonic() - t6 > remaining:
                break

            S_trial = np.sqrt(2.0 * n_trial)
            inits = generate_inits(n_trial, rng, S_trial)

            # Quick FISTA from best init
            best_trial_obj = np.inf
            best_trial_a = None
            time_per_init = min(20.0, time_per_length / len(inits))

            for init in inits:
                if time.monotonic() > deadline:
                    break
                a_t, _ = run_fista_adaptive(init, min(time_per_init, 30.0), rng,
                                            beta_start=20.0, beta_end=10000.0, mode="lse")
                obj_t = compute_obj(a_t)
                if obj_t < best_trial_obj:
                    best_trial_obj = obj_t
                    best_trial_a = a_t.copy()

            # KKT polish the best
            if best_trial_a is not None and best_trial_obj < global_best_obj * 1.001:
                rem = deadline - time.monotonic()
                if rem > 20:
                    a_p, _ = run_kkt_polish(best_trial_a, min(rem * 0.5, 60.0), rng)
                    obj_p = compute_obj(a_p)
                    if obj_p < best_trial_obj:
                        best_trial_obj = obj_p
                        best_trial_a = a_p

                    # L-BFGS squeeze
                    rem2 = deadline - time.monotonic()
                    if rem2 > 20:
                        a_l, obj_l = run_lbfgs_polish(best_trial_a, min(rem2 * 0.5, 60.0), rng,
                                                      n_beta_levels=3)
                        if obj_l < best_trial_obj:
                            best_trial_obj = obj_l
                            best_trial_a = a_l

            print("  n=%d: best obj=%.8f" % (n_trial, best_trial_obj))
            if best_trial_a is not None and best_trial_obj < global_best_obj:
                global_best_obj = best_trial_obj
                global_best_a = best_trial_a.copy()
                print("  -> NEW BEST!")

    # === Phase 7: Final intensive polish (whatever time remains) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 7: Final polish (budget %.0fs) ===" % remaining)
        t7 = time.monotonic()

        # KKT polish
        a7, _ = run_kkt_polish(global_best_a, remaining * 0.4, rng, tol=1e-13)
        obj7 = compute_obj(a7)
        if obj7 < global_best_obj:
            global_best_obj = obj7
            global_best_a = a7.copy()
            print("  KKT final: obj=%.8f" % obj7)

        # L-BFGS final
        rem7 = deadline - time.monotonic()
        if rem7 > 15:
            a7b, obj7b = run_lbfgs_polish(global_best_a, rem7 * 0.7, rng, n_beta_levels=3)
            if obj7b < global_best_obj:
                global_best_obj = obj7b
                global_best_a = a7b.copy()
                print("  L-BFGS final: obj=%.8f" % obj7b)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.10f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:

            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.10f" % global_best_obj)
    return final_list
