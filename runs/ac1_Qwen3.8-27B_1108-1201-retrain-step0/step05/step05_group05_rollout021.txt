"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: Multi-phase optimization with:
1. Spectral initialization (ideal flat convolution via polynomial square root)
2. B-spline and window-based diverse initializations
3. FISTA with adaptive LSE smoothing and momentum
4. Guided local search (pair swaps with subgradient sensitivity)
5. Periodic spectral phase reshaping
6. Multi-start strategy with different sequence lengths
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_from_conv(a, c):
    """Compute objective given a and its self-convolution c."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta_eff = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta_eff * (c - cm)
        beta = beta_eff
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_topk(a, c, k=8):
    """Subgradient from top-k lags with geometric weighting."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    # Geometric weights: more weight on the top
    weights = np.power(0.85, np.arange(k))
    weights = weights / weights.sum()
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def sensitivity_vector(a, c, k=16):
    """Compute how much each a[i] contributes to the top-k peaks."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    weights = np.ones(k) / k
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    # sensitivity[i] = sum_k w_full[k] * a[k-i] (for valid k-i)
    # This is the cross-correlation of w_full with a
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    # ifft(conj(W) * A) gives cross-correlation
    sens = np.real(ifft(np.conj(W) * A))[:n]
    return sens


def local_search_pair_swap(a, c, S, n_trials=200, rng=None):
    """
    Local search: try pair swaps guided by sensitivity.
    Returns improved (a, c, obj) or original if no improvement found.
    """
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()
    
    obj_cur = objective_from_conv(a, c)
    sens = sensitivity_vector(a, c, k=16)
    
    # Sort by sensitivity: high sens = contributes more to peaks
    order = np.argsort(sens)[::-1]
    top_positions = order[:min(200, n // 10)]
    bot_positions = order[-min(200, n // 10):]
    
    best_obj = obj_cur
    best_a = a.copy()
    fsize = 1 << (2 * n - 1).bit_length()
    
    for _ in range(n_trials):
        i = rng.integers(len(top_positions))
        j = rng.integers(len(bot_positions))
        i = top_positions[i]
        j = bot_positions[j]
        
        if i == j:
            continue
        if a[i] < 1e-10:
            continue
        
        # Try moving fraction of a[i] to a[j]
        # Try several fractions
        for frac in [0.01, 0.05, 0.1, 0.2, 0.33, 0.5]:
            delta = a[i] * frac
            if delta < 1e-15:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            c_new = fft_convolve_self(a_new)
            obj_new = objective_from_conv(a_new, c_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
    
    if best_obj < obj_cur:
        c_new = fft_convolve_self(best_a)
        return best_a, c_new, best_obj
    return a, c, obj_cur


def spectral_init_flat(n, S):
    """
    Initialize via spectral factorization for ideally flat convolution.
    Compute A[m] = sqrt(DFT(ideal_b)) where ideal_b is flat.
    """
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    # Ideal flat b
    b_ideal = np.zeros(fsize)
    c_val = S * S / size
    b_ideal[:size] = c_val
    
    B = fft(b_ideal)
    # Take square root (principal branch)
    # Need to handle the phase carefully
    A = np.sqrt(B)  # principal square root
    
    a = np.real(ifft(A, fsize))[:n]
    a = np.maximum(a, 0.0)
    a = normalize_to_sum(a, S)
    return a


def bspline_init(n, S, order=4):
    """Initialize with a discrete B-spline of given order."""
    # Build B-spline by repeated convolution of [1, 1]
    kernel = np.array([1.0, 1.0])
    for _ in range(order - 1):
        kernel = np.convolve(kernel, [1.0, 1.0])
    
    # Scale and truncate/pad to length n
    ks = len(kernel)
    if ks < n:
        # Center the kernel in the sequence
        pad = (n - ks) // 2
        a = np.zeros(n)
        a[pad:pad + ks] = kernel
    else:
        # Truncate
        start = (ks - n) // 2
        a = kernel[start:start + n].astype(np.float64)
    
    a = normalize_to_sum(a, S)
    return a


def cosine_window_init(n, S, alpha=1.0):
    """Initialize with a cosine window."""
    t = np.arange(n) / n
    a = np.maximum(0.5 * (1.0 + np.cos(2.0 * np.pi * t)), 0.0) + 0.01
    a = normalize_to_sum(a, S)
    return a


def gaussian_init(n, S, sigma_frac=0.3):
    """Initialize with a Gaussian window."""
    t = np.arange(n) / n
    sigma = sigma_frac
    a = np.exp(-0.5 * ((t - 0.5) / sigma) ** 2) + 0.001
    a = normalize_to_sum(a, S)
    return a


def random_init(n, S, rng):
    """Random initialization."""
    a = rng.random(n) + 0.1
    a = normalize_to_sum(a, S)
    return a


def fourier_phase_perturb(a, rng, scale=0.05):
    """Perturb the Fourier phase of a."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    noise = rng.normal(0, scale, fsize)
    new_A = mag * np.exp(1j * (phase + noise))
    new_A[0] = A[0]  # Preserve DC
    a_new = np.real(ifft(new_A, fsize))[:n]
    return project_simplex(a_new, S)


def run_fista_lse(a_init, S, max_time, rng, beta_init=10.0, beta_rate=0.003,
                  beta_max=10000.0, stuck_threshold=3000, use_topk=False, topk_k=8):
    """
    FISTA with LSE or top-k subgradient.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        if c.max() < best - 1e-14:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        if use_topk:
            grad = subgradient_topk(y, c, k=topk_k)
        else:
            grad = lse_gradient(y, c, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        if use_topk:
            step = 1.0 / (gnorm / (S / n) + 1e-10) * (S / n) * 0.5
        else:
            step = 1.0 / (3.0 * beta)

        # Stuck detection
        if it - last_improve > stuck_threshold:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 10000.0)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            else:
                y = fourier_phase_perturb(y, rng, scale=0.02)
            t_mom = 1.0
            last_improve = it
            continue

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_combined_optimize(a_init, S, max_time, rng):
    """
    Combined optimizer: FISTA + local search alternation.
    """
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best = objective_from_conv(a, c)
    best_a = a.copy()
    
    it = 0
    fista_iter = 0
    last_improve_iter = 0
    fista_period = 500  # FISTA iterations before local search
    ls_period = 500  # Local search trials
    
    while time.monotonic() - start < max_time:
        # Check time
        elapsed = time.monotonic() - start
        if elapsed > max_time - 5:
            break
        
        # FISTA phase
        remaining = max_time - elapsed
        fista_time = min(remaining * 0.7, 60.0)
        if fista_time < 5:
            break
        
        use_topk = (fista_iter > 2000)
        a_opt, val = run_fista_lse(a, S, fista_time, rng,
                                    beta_init=10.0 + 5.0 * (fista_iter / 1000),
                                    beta_rate=0.002,
                                    beta_max=2000.0 + 1000.0 * min(fista_iter / 1000, 5.0),
                                    stuck_threshold=2000,
                                    use_topk=use_topk,
                                    topk_k=8 + (fista_iter // 500))
        
        c = fft_convolve_self(a_opt)
        obj = objective_from_conv(a_opt, c)
        if obj < best - 1e-12:
            best = obj
            best_a = a_opt.copy()
            last_improve_iter = fista_iter
        
        fista_iter += 1
        a = a_opt
        
        # Local search phase
        ls_time = min(remaining * 0.3, 30.0)
        if ls_time < 3:
            continue
        
        n_trials = int(ls_time * 100)  # Scale trials with available time
        n_trials = max(50, min(n_trials, 500))
        a_ls, c_ls, obj_ls = local_search_pair_swap(a, c, S, n_trials=n_trials, rng=rng)
        
        if obj_ls < best - 1e-12:
            best = obj_ls
            best_a = a_ls.copy()
            last_improve_iter = fista_iter
            a = a_ls
            c = c_ls
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_high_beta_polish(a_init, S, max_time, rng):
    """Final polishing with very high beta LSE."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()
    c = fft_convove_self(a) if False else fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    y = a.copy()
    t_mom = 1.0
    it = 0
    last_improve = 0
    beta = 500.0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        beta = 500.0 + 20000.0 * progress * progress
        
        c = fft_convolve_self(y)
        if c.max() < best - 1e-14:
            best = c.max()
            best_a = y.copy()
            last_improve = it
        
        if it - last_improve > 5000:
            y = fourier_phase_perturb(y, rng, scale=0.01)
            t_mom = 1.0
            last_improve = it
            continue
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (3.0 * beta)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    S = np.sqrt(2.0 * n)
    inits = []
    
    if given_seq is not None:
        a = np.clip(given_seq, 0, 1000).astype(np.float64)
        if a.sum() > 0.01:
            inits.append(normalize_to_sum(a, S))
    
    # Spectral init
    inits.append(spectral_init_flat(n, S))
    
    # B-splines
    for order in [3, 4, 6, 8, 12]:
        inits.append(bspline_init(n, S, order=order))
    
    # Windows
    inits.append(cosine_window_init(n, S))
    inits.append(gaussian_init(n, S, sigma_frac=0.25))
    inits.append(gaussian_init(n, S, sigma_frac=0.4))
    inits.append(gaussian_init(n, S, sigma_frac=0.15))
    
    # Random
    inits.append(random_init(n, S, rng))
    inits.append(random_init(n, S, rng))
    
    return inits


def propose_candidate(seed=42, budget_s=2340, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.
    
    Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + budget_s - 20
    
    print("Starting optimization, budget=%.0fs" % budget_s)
    
    # Try to get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print("Given sequence: length=%d, initial obj=%.6f" % (
                len(given), objective_from_conv(given, fft_convolve_self(given))))
    except Exception:
        pass
    
    # Try multiple n values
    n_candidates = [8192, 16384, 4096]
    
    global_best = np.inf
    global_a = None
    global_n = 0
    
    def update_global(a, val_c):
        nonlocal global_best, global_a, global_n
        n = len(a)
        S = a.sum()
        obj = 2.0 * n * val_c / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            global_n = n
            print("  NEW GLOBAL BEST: obj=%.8f (n=%d)" % (obj, n))
            return True
        return False
    
    # Phase 1: Quick screening across different n and inits
    print("\n=== Phase 1: Quick screening ===")
    p1_start = time.monotonic()
    p1_budget = min(200.0, (deadline - p1_start) * 0.15)
    
    all_candidates = []
    
    for n in n_candidates:
        S = np.sqrt(2.0 * n)
        inits = generate_inits(n, rng, given_seq=None)
        # If we have a given sequence, resize it
        if given is not None and len(given) != n:
            # Interpolate
            from numpy import interp
            old_t = np.linspace(0, 1, len(given))
            new_t = np.linspace(0, 1, n)
            a_given = interp(new_t, old_t, given)
            inits.append(normalize_to_sum(a_given, S))
        
        # Quick FISTA on each init
        per_init_time = p1_budget / (len(inits) * len(n_candidates))
        per_init_time = max(5.0, per_init_time)
        
        for i, a_init in enumerate(inits):
            if time.monotonic() > deadline - (deadline - total_start) * 0.7:
                break
            t_alloc = min(per_init_time, deadline - time.monotonic() - 120)
            if t_alloc < 3:
                break
            
            a_opt, val = run_fista_lse(a_init, S, t_alloc, rng,
                                        beta_init=5.0, beta_rate=0.01, beta_max=2000,
                                        stuck_threshold=1000, use_topk=False)
            c = fft_convolve_self(a_opt)
            obj = objective_from_conv(a_opt, c)
            all_candidates.append((obj, a_opt.copy(), n))
            update_global(a_opt, c.max())
    
    all_candidates.sort(key=lambda x: x[0])
    print("Phase 1 done. Best obj=%.8f" % global_best)
    print("Top candidates:")
    for i, (obj, a, n) in enumerate(all_candidates[:5]):
        print("  %d: obj=%.8f, n=%d" % (i, obj, n))
    
    # Phase 2: Main optimization on top candidates
    print("\n=== Phase 2: Main optimization ===")
    p2_start = time.monotonic()
    p2_budget = (deadline - p2_start) * 0.55
    p2_budget = min(p2_budget, 1200.0)
    
    n_main = min(3, len(all_candidates))
    for i in range(n_main):
        if time.monotonic() > deadline - (deadline - total_start) * 0.3:
            break
        t_alloc = (deadline - time.monotonic() - (deadline - total_start) * 0.3) / (n_main - i)
        t_alloc = min(t_alloc, p2_budget)
        if t_alloc < 30:
            break
        
        obj_init, a_init, n = all_candidates[i]
        S = np.sqrt(2.0 * n)
        print("  Main opt %d: n=%d, init obj=%.8f, alloc=%.0fs" % (i, n, obj_init, t_alloc))
        
        a_opt, val = run_combined_optimize(a_init, S, t_alloc, rng)
        c = fft_convolve_self(a_opt)
        update_global(a_opt, c.max())
    
    # Phase 3: High-beta polish
    print("\n=== Phase 3: High-beta polish ===")
    p3_start = time.monotonic()
    p3_budget = min(400.0, (deadline - p3_start) * 0.4)
    
    if global_a is not None and p3_budget > 30:
        # Polish with high-beta FISTA
        seg1 = p3_budget * 0.6
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a_hb, val_hb = run_high_beta_polish(global_a, S, seg1, rng)
        c_hb = fft_convolve_self(a_hb)
        update_global(a_hb, c_hb.max())
        print("  After high-beta polish: obj=%.8f" % global_best)
        
        # Another round of combined optimization
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            a_co, val_co = run_combined_optimize(global_a, S, remaining, rng)
            c_co = fft_convolve_self(a_co)
            update_global(a_co, c_co.max())
            print("  After final combined: obj=%.8f" % global_best)
    
    # Phase 4: Final local search
    print("\n=== Phase 4: Final local search ===")
    if global_a is not None:
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            n = len(global_a)
            S = np.sqrt(2.0 * n)
            c = fft_convolve_self(global_a)
            obj = objective_from_conv(global_a, c)
            
            # Multiple rounds of local search
            n_rounds = int(remaining / 30)
            n_rounds = max(1, min(n_rounds, 10))
            for r in range(n_rounds):
                if time.monotonic() > deadline - 5:
                    break
                a_ls, c_ls, obj_ls = local_search_pair_swap(
                    global_a, c, S,
                    n_trials=min(1000, int((deadline - time.monotonic() - 5) * 50)),
                    rng=rng)
                if obj_ls < obj:
                    global_a = a_ls
                    c = c_ls
                    obj = obj_ls
                    global_best = obj
                    print("  LS round %d: obj=%.8f" % (r, obj))
    
    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(len(global_a)) * np.sqrt(2.0 * len(global_a)) / len(global_a)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: obj=%.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            n = len(global_a)
            global_a = np.ones(n) * np.sqrt(2.0 * n) / n
            final_list = [float(x) for x in global_a]
    else:
        n = 8192
        global_a = np.ones(n) * np.sqrt(2.0 * n) / n
        final_list = [float(x) for x in global_a]
    
    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final obj=%.8f, n=%d" % (elapsed, global_best, len(global_a)))
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
