This is a "minimax quadratic" problem, which is NP-hard in general but can be approximated.

**New approach idea: Use the KKT conditions to guide the optimization.**

At the optimum, if the max is achieved at lags $k_1, k_2, \ldots, k_m$, then:
- $\nabla (a^T H_{k_j} a) = \lambda \mathbf{1} + \mu$ for each active $k_j$
- where $\lambda$ is the Lagrange multiplier for $\sum a = 1$ and $\mu \ge 0$ is for $a \ge 0$

This means that at the optimum, the sum of the gradients of the active quadratic forms should be "flat" (constant) on the support of $a$.

**Algorithm: Active-set gradient balancing**
1. Find the active set (top k lags where $c[k] \approx \max c$)
2. Compute the combined gradient: $g = \sum_{j \in \text{active}} \nabla (a^T H_{k_j} a)$
3. The ideal update is to make $g$ as flat as possible on the support
4. This gives us a direction to move

This is essentially what LSE does, but with the twist that we can be smarter about the "active set" identification.

def fft_convolve_self(a):

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    if s < 1e-15:

    return a * (S / s)

    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]

    scaled = beta * (c_top - cm)

    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def run_peak_shave(a_init, S, max_time, rng):
    """Targeted peak shaving: identify peaks and move mass to reduce them."""
    n = len(a_init)
    a = normalize_seq(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Find top peaks
        n_peaks = min(20, 2 * n - 1)
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[peak_idx]
        
        # Compute contribution of each position to the peaks
        # grad[i] = 2 * sum over peaks of a[peak - i]
        contrib = np.zeros(n)
        for k_val in peak_idx[:10]:  # Top 10 peaks

            # Simple: just use unweighted sum for top peaks
            lo = max(0, k_val - n + 1)
            hi = min(n - 1, k_val)

                contrib[i] += a[k_val - i]
        
        # Move mass from high-contribution to low-contribution positions
        # Find top and bottom contributors
        active = a > 1e-10
        if not np.any(active):
            break
        
        contrib_active = contrib[active]
        a_active = a[active]
        
        # Identify positions to decrease (high contrib, large a) and increase (low contrib)
        if len(contrib_active) > 1:
            # Top 5% to decrease, bottom 20% to increase
            thresh_dec = np.percentile(contrib_active, 95)
            thresh_inc = np.percentile(contrib_active, 20)
            
            dec_mask = np.zeros(n, dtype=bool)
            inc_mask = np.zeros(n, dtype=bool)
            dec_mask[active] = contrib_active > thresh_dec
            inc_mask[active] = contrib_active < thresh_inc
            
            if np.any(dec_mask) and np.any(inc_mask):
                # Compute how much to move
                total_dec = a[dec_mask].sum() * 0.01  # Move 1% of mass from dec positions
                n_inc = inc_mask.sum()
                move_per_inc = total_dec / n_inc
                
                a_new = a.copy()
                a_new[dec_mask] -= total_dec / dec_mask.sum()
                a_new[dec_mask] = np.maximum(a_new[dec_mask], 0)
                a_new[inc_mask] += move_per_inc
                a_new = normalize_seq(a_new, S)
                
                c_new = fft_convolve_self(a_new)
                if c_new.max() < cur_max:
                    a = a_new
                else:
                    # Try smaller move
                    a_new[dec_mask] = a[dec_mask] - total_dec / (2 * dec_mask.sum())
                    a_new[dec_mask] = np.maximum(a_new[dec_mask], 0)
                    a_new[inc_mask] = a[inc_mask] + move_per_inc / 2
                    a_new = normalize_seq(a_new, S)
                    c_new = fft_convolve_self(a_new)
                    if c_new.max() < cur_max:
                        a = a_new
                    else:
                        # No improvement, add small noise
                        a = project_simplex(a + rng.normal(0, 0.0005 * S / np.sqrt(n), n), S)

def run_fine_polish(a_init, S, max_time, rng):
    """Very fine polishing with high beta and small steps."""
    n = len(a_init)
    a = normalize_seq(a_init.copy(), S)
    start = time.monotonic()

    best_a = a.copy()

    step = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        k = min(10, 2 * n - 1)
        beta = min(50000.0, 500.0 * (1.0 + 0.001 * it))

        if cur_max < best:
            best = cur_max

        gnorm = np.linalg.norm(grad)

        hi = min(n - 1, k_val)

    G = fft(g_proj, fft_size)

    Q = sum(auto[k_val] for k_val in k_active) / len(k_active)

    step = 1.0 / (np.max(np.abs(g_proj)) + 1e-10)
    if Q > 0:

    a_new = project_simplex(a - step * g, S)
    return a_new

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.
Key innovation over previous: Newton-corrected gradient steps that account for
curvature of active autocorrelation peaks, plus iterated local search with
large perturbations to escape shallow basins. Multi-phase: FISTA → Newton → Polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_gradient(a, c, k, beta):
    """Top-k LSE gradient of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    scaled = np.clip(scaled, -500.0, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def autocorr_at_lags(g, lags, n):
    """Compute autocorrelation of g at specified lags via FFT."""
    fsize = 1 << (2 * n).bit_length()
    G = fft(g, fsize)
    auto = np.real(ifft(G * np.conj(G), fsize))[:2 * n - 1]
    result = np.zeros(len(lags))
    for j, lag in enumerate(lags):
        if 0 <= lag < len(auto):
            result[j] = auto[lag]
        elif lag >= len(auto) and lag < 2 * n:
            result[j] = auto[2 * n - 1 - lag]
    return result


def run_fista(a_init, S, max_time, rng,
              beta_init=10.0, beta_rate=0.004, beta_max=8000.0,
              k_init=2, k_max=60, k_grow=400):
    """FISTA with adaptive top-k and backtracking line search."""
    n = len(a_init)
    a = normalize_seq(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    no_improve = 0
    step_base = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        k = min(k_max, k_init + it // k_grow)
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        grad = lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        alpha = step_base
        x_new = project_simplex(y - alpha * grad / gnorm, S)
        for _ in range(12):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-12 * max(cur_max, 1e-10):
                break
            alpha *= 0.5
            x_new = project_simplex(y - alpha * grad / gnorm, S)
        step_base = min(alpha * 1.05, 1.0 / (2.0 * beta * 0.005 + 1e-10))

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        # Damp momentum when stuck
        momentum = (t_val - 1.0) / t_new
        if no_improve > 300:
            momentum *= 0.5
        elif no_improve > 100:
            momentum *= 0.7
        y_new = x_new + momentum * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = project_simplex(y_new, S) if momentum * (x_new - a) > 0 else y_new
        t_val = t_new

        if it % 4000 == 0 and no_improve > 300:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            a = y.copy()
            t_val = 1.0

    return best_a, best


def run_newton_steps(a_init, S, max_time, rng, n_active=15, nls_mode=False):
    """
    Gradient steps with Newton (curvature) correction for active peaks.
    The step size accounts for the Hessian of the peak functional,
    preventing overshooting in high-curvature directions.
    """
    n = len(a_init)
    a = normalize_seq(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0
    fsize = 1 << (2 * n).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Identify active peaks (top n_active)
        peak_idx = np.argsort(c)[::-1][:n_active]
        # Only truly active (within 2% of max)
        threshold = cur_max * 0.98
        active_lags = peak_idx[c[peak_idx] >= threshold]
        if len(active_lags) == 0:
            active_lags = peak_idx[:5]

        # Compute gradient: g[i] = 2 * sum_{k in active} a[k-i] (with range check)
        # Use FFT-based cross-correlation
        w = np.zeros(2 * n - 1)
        for lag in active_lags:
            if 0 <= lag < 2 * n - 1:
                w[lag] = 1.0
        w /= len(active_lags)

        W = fft(w, fsize)
        A = fft(a, fsize)
        g = 2.0 * np.real(ifft(W * A))[:n]

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Newton correction: compute curvature Q = (1/m) * sum_k g^T H_k g
        # g^T H_k g = 2 * autocorr(g)[k]
        autoc_g = autocorr_at_lags(g, active_lags, n)
        Q = 2.0 * np.mean(autoc_g)

        # Step size: balance gradient norm vs curvature
        grad_dir = g / gnorm
        # Linear decrease rate
        lin_rate = np.min(autocorr_at_lags(g, active_lags, n)) * 2.0 / gnorm

        if Q > 1e-12:
            step = min(1.0 / gnorm, 0.5 * lin_rate / Q) if lin_rate > 0 else 1.0 / gnorm
        else:
            step = 1.0 / gnorm
        step = max(step, 1e-10)
        step = min(step, 10.0 / gnorm)

        # Take the step with backtracking
        a_new = project_simplex(a - step * g, S)
        for _ in range(8):
            c_new = fft_convolve_self(a_new)
            if c_new.max() <= cur_max + 1e-12 * cur_max:
                break
            step *= 0.4
            a_new = project_simplex(a - step * g, S)

        a = a_new

        # Periodic perturbation to escape
        if stuck > 500 and it % 100 == 0:
            scale = 0.003 * S / np.sqrt(n) * min(1.0, stuck / 2000.0)
            a = project_simplex(a + rng.normal(0, scale, n), S)
            stuck = 0

        # Larger perturbation every so often (iterated local search)
        if it % 5000 == 0 and (time.monotonic() - start) / max_time < 0.6:
            # Perturb a random subset of coordinates
            frac = 0.1
            mask = rng.random(n) < frac
            a[mask] *= rng.uniform(0.5, 1.5, mask.sum())
            a = normalize_seq(a, S)
            stuck = 0

    return best_a, best
