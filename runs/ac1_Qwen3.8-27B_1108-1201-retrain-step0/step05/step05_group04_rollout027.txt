from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

   - If no progress after many steps, lower M (accept the current max as the new target).

This is essentially a "projected subgradient" method with a "reduction of target" heuristic. It's different from LSE in that it uses a hard threshold (only lags above M contribute to the gradient).

The advantage: it focuses all effort on the most problematic lags, which can be more efficient than LSE when only a few lags are near the maximum.

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Multi-phase optimization combining LSE-FISTA, target-level subgradient
method, active-set refinement, and multi-length exploration.
Novel contributions: (1) Target-level subgradient with adaptive reduction,
(2) Structured perturbations based on convolution peak analysis,
(3) Conjugate-gradient-inspired direction updates.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient_and_value(a, c, beta):
    """Compute LSE-smoothed gradient and value of max(c)."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    # Numerical stability
    shifted_max = shifted.max()
    e = np.exp(shifted - shifted_max)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + (shifted_max + np.log(tot)) / beta
    return grad, lse_val


def subgradient_at_target(a, c, target):
    """Compute subgradient of max(c) focusing on lags above target."""
    n = len(a)
    cm = c.max()
    if target >= cm:
        # All lags are below target; use the argmax lag
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        # Weight lags above target by their excess
        excess = np.maximum(c - target, 0.0)
        tot = excess.sum()
        if tot < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = excess / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_lse_fista_v2(a_init, n, S, max_time, rng,
                     beta_start=50.0, beta_end=20000.0, beta_steps=200,
                     momentum_restart_gap=300):
    """
    Improved FISTA with adaptive LSE smoothing.
    Key improvements: (1) Better beta schedule, (2) Adaptive step size,
    (3) Gradient norm-based restart, (4) Periodic re-orthogonalization.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    prev_grad = None
    grad_norm_history = []
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta: smooth sigmoid schedule
        frac = min(1.0, it / max(1, beta_steps))
        # Use a smoother schedule: beta = beta_start + (beta_end - beta_start) * frac^2
        beta = beta_start + (beta_end - beta_start) * (frac ** 1.5)
        beta = min(beta, 80000.0)
        
        # Compute convolution
        c = fft_convolve_self(y)
        cur_true_max = c.max()
        
        if cur_true_max < best:
            best = cur_true_max
            best_a = y.copy()
            last_improve_it = it
        
        # Compute LSE gradient
        grad, lse_val = lse_gradient_and_value(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size: base is 1/beta, adjusted by gradient norm
        step = 1.0 / beta
        
        # Stagnation handling
        stagnant = it - last_improve_it
        if stagnant > momentum_restart_gap:
            t_mom = 1.0
            # Structured perturbation based on convolution peaks
            c_peaks = np.argsort(c)[::-1][:5]
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 500.0)
            if noise_scale > 1e-10:
                # Move mass away from peak-contributing positions
                pert = np.zeros(n)
                for k in c_peaks:
                    lo = max(0, k - n + 1)
                    hi = min(k, n - 1)
                    pert[lo:hi+1] += c[k]
                pert = pert - pert.mean()
                y = project_simplex(y + noise_scale * pert / (np.std(pert) + 1e-10), S)
            last_improve_it = it
            step *= 0.5
        
        # FISTA update with gradient clipping
        grad_clipped = np.clip(grad, -50.0 * S / n, 50.0 * S / n)
        x_new = project_simplex(y - step * grad_clipped, S)
        
        # Nesterov momentum with restart
        grad_dot = 0.0
        if prev_grad is not None:
            grad_dot = np.dot(grad, prev_grad)
        if grad_dot > 0:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        else:
            t_new = 1.0  # Restart momentum
        
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        prev_grad = grad
        
        # Periodic full check
        if it % 500 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() < best:
                best = c_check.max()
                best_a = a.copy()
                last_improve_it = it
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_target_level(a_init, n, S, max_time, rng):
    """
    Target-level subgradient method.
    Maintains a target level M and tries to drive max(f*f) below it.
    When successful, reduces M. When stuck, slightly raises M and retries.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    M = c.max()  # Initial target
    it = 0
    stuck_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Update target
        if cur_max <= M:
            # Below target: reduce it
            M = cur_max * (1.0 - 0.001 * max(0.1, 50.0 / (it + 1)))
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count > 100:
                # Stuck: raise target slightly and try different approach
                M = cur_max * 1.001
                stuck_count = 0
                # Add small noise
                noise_scale = 0.001 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
        
        # Compute subgradient
        grad = subgradient_at_target(a, c, M)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = 0.1 * S / (n * gnorm + 1e-10)
        base_obj = cur_max
        improved = False
        for _ in range(6):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            test_obj = c_test.max()
            if test_obj < base_obj * (1 - 1e-7):
                a = a_test
                improved = True
                break
            step *= 0.5
        
        if not improved and it % 10 == 0:
            # Very small step
            a = project_simplex(a - step * 0.01 * grad, S)
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_conjugate_refinement(a_init, n, S, max_time, rng):
    """
    Conjugate-gradient-inspired refinement using active-set gradients.
    Maintains a direction memory and combines gradients from multiple iterations.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    d = None  # Previous direction
    beta_cg = 0.0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Active set: top lags within 5% of max
        cm = c.max()
        threshold = cm * 0.95
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        w = np.zeros(len(c))
        # Weight by excess
        w[active] = np.maximum(c[active] - threshold, 1e-10)
        w = w / w.sum()
        
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Project gradient onto tangent space of simplex
        # (subtract mean to maintain sum constraint approximately)
        g_tangent = g - g.mean()
        gnorm = np.linalg.norm(g_tangent)
        if gnorm < 1e-14:
            break
        
        g_tangent /= gnorm
        
        # Conjugate gradient direction
        if d is not None:
            # Fletcher-Reeves beta
            beta_cg = np.dot(g_tangent, d) / (np.linalg.norm(d) ** 2 + 1e-10)
            beta_cg = max(0.0, beta_cg)
            search_dir = -g_tangent + beta_cg * d
        else:
            search_dir = -g_tangent
        
        # Ensure the direction is "feasible" (points into the simplex)
        # Project the direction
        search_dir = search_dir - search_dir.mean()
        
        # Line search
        step = 0.01 * S / n
        base_obj = cur_max
        improved = False
        for _ in range(8):
            a_test = project_simplex(a + step * search_dir, S)
            c_test = fft_convolve_self(a_test)
            test_obj = c_test.max()
            if test_obj < base_obj * (1 - 1e-7):
                a = a_test
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # Try negative direction
            for _ in range(4):
                a_test = project_simplex(a - step * search_dir, S)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < base_obj * (1 - 1e-7):
                    a = a_test
                    improved = True
                    break
                step *= 0.5
            if not improved:
                d = None  # Reset conjugate direction
                continue
        
        d = search_dir
        
        # Periodic noise
        if it % 500 == 0:
            noise_scale = 0.0005 * S / np.sqrt(n) * np.exp(-it / 3000.0)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                d = None
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_peak_shaving_v2(a_init, n, S, max_time, rng):
    """
    Improved peak shaving with adaptive learning rate and multi-scale peaks.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()
        
        if cm < best:
            best = cm
            best_a = a.copy()
        
        # Get top peaks (adaptive number based on how spread out they are)
        sorted_c = np.sort(c)[::-1]
        n_peaks = max(3, min(20, int(np.sum(sorted_c > cm * 0.9))))
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Compute responsibility
        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            weight = max(0, c[k] - cm * 0.98)
            resp[js] += contrib * weight
        
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        
        # Adaptive learning rate
        baseline = 0.01
        lr = baseline * S / (n * resp_max)
        
        # Multiplicative update
        a_new = a * np.exp(-lr * resp)
        a_new = project_simplex(a_new, S)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cm:
            a = a_new
        else:
            # Try smaller step
            a_half = a * np.exp(-0.5 * lr * resp)
            a_half = project_simplex(a_half, S)
            c_half = fft_convolve_self(a_half)
            if c_half.max() < cm:
                a = a_half
            else:
                # Try even smaller
                a_quarter = a * np.exp(-0.25 * lr * resp)
                a_quarter = project_simplex(a_quarter, S)
                c_q = fft_convolve_self(a_quarter)
                if c_q.max() < cm:
                    a = a_quarter
                else:
                    # Give up this step, add noise
                    noise_scale = 0.001 * S / np.sqrt(n)
                    a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

import numpy as np

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient_and_value(a, c, beta):
    """Compute LSE-smoothed gradient and value of max(c)."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted_max = shifted.max()
    e = np.exp(shifted - shifted_max)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + (shifted_max + np.log(tot)) / beta
    return grad, lse_val


def subgradient_at_target(a, c, target):
    """Compute subgradient focusing on lags above target."""
    n = len(a)
    cm = c.max()
    if target >= cm:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        excess = np.maximum(c - target, 0.0)
        tot = excess.sum()
        if tot < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = excess / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def active_set_gradient(a, c, frac=0.05):
    """Gradient using top (1-frac) quantile of lags."""
    n = len(a)
    cm = c.max()
    threshold = cm - cm * frac
    w = np.where(c >= threshold, 1.0, 0.0)
    tot = w.sum()
    if tot < 1:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        tot = 1.0
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def fourier_parameterize(coeffs, n):
    """Generate sequence from Fourier coefficients (always non-negative via softplus)."""
    t = np.arange(n, dtype=np.float64)
    val = np.zeros(n)
    num_harmonics = (len(coeffs) - 1) // 2
    val += coeffs[0]
    for k in range(num_harmonics):
        a_k = coeffs[2 * k + 1]
        b_k = coeffs[2 * k + 2]
        val += a_k * np.cos(2.0 * np.pi * k * t / n) + b_k * np.sin(2.0 * np.pi * k * t / n)
    # Softplus for non-negativity
    val = np.log1p(np.exp(np.clip(val, -50, 50)))
    return val

            val += coeffs_vec[2 * k + 1] * np.cos(2.0 * np.pi * k * t / n)
            val += coeffs_vec[2 * k + 2] * np.sin(2.0 * np.pi * k * t / n)
        val = np.log1p(np.exp(np.clip(val, -50, 50)))
        val = np.maximum(val, 1e-10)
        s = val.sum()
        if s < 1e-10:
            return 1e10
        val = val * S / s
        c = fft_convolve_self(val)
        return c.max()

    def fourier_seq(coeffs_vec):
        val = np.zeros(n)
        val += coeffs_vec[0]
        for k in range(num_harmonics):
            val += coeffs_vec[2 * k + 1] * np.cos(2.0 * np.pi * k * t / n)
            val += coeffs_vec[2 * k + 2] * np.sin(2.0 * np.pi * k * t / n)
        val = np.log1p(np.exp(np.clip(val, -50, 50)))
        val = np.maximum(val, 1e-10)
        s = val.sum()
        if s < 1e-10:
            return np.ones(n) * S / n
        return val * S / s

    # Nelder-Mead for the low-dimensional problem
    from scipy.optimize import minimize
    x0 = coeffs.copy()
    result = minimize(fourier_obj, x0, method='Nelder-Mead',
                      options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10,
                               'adaptive': True})
    coeffs_opt = result.x if result.success else x0

    a_fm = fourier_seq(coeffs_opt)
    c_fm = fft_convolve_self(a_fm)
    val_fm = c_fm.max()
    print("    Fourier init: %.8f" % val_fm)

    if val_fm < best:
        best = val_fm
        best_a = a_fm.copy()

    # Now do gradient refinement in Fourier space
    # Use finite differences for gradient (low-dimensional)
    c = 1e-4
    for fi in range(300):
        if time.monotonic() - start > max_time:
            break
        # Compute gradient by finite differences
        grad = np.zeros(num_params)
        for i in range(num_params):
            cp = coeffs_opt.copy()
            cm_ = coeffs_opt.copy()
            cp[i] += c
            cm_[i] -= c
            grad[i] = (fourier_obj(cp) - fourier_obj(cm_)) / (2 * c)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        # Adaptive step
        step = 0.1 / (gnorm + 1e-10)
        improved = False
        for _ in range(5):
            coeffs_new = coeffs_opt - step * grad
            obj_new = fourier_obj(coeffs_new)
            if obj_new < fourier_obj(coeffs_opt) * (1 - 1e-10):
                coeffs_opt = coeffs_new
                improved = True
                break
            step *= 0.5

        if not improved:
            break

        a_fm = fourier_seq(coeffs_opt)
        c_fm = fft_convolve_self(a_fm)
        if c_fm.max() < best:
            best = c_fm.max()
            best_a = a_fm.copy()

    return best_a, best


def run_knockout_coord_descent(a_init, n, S, max_time, rng):
    """
    Coordinate-pair descent: swap mass between pairs of positions to reduce peak.
    More effective than single-coordinate for this problem.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Find the peak lag
        k_peak = np.argmax(c)

        # Elements that contribute to c[k_peak]: a[j] * a[k_peak - j]
        lo = max(0, k_peak - n + 1)
        hi = min(k_peak, n - 1)

        # Find the pair with maximum product
        js = np.arange(lo, hi + 1)
        partners = k_peak - js
        products = a[js] * a[partners]
        max_prod_idx = np.argmax(products)

        if products[max_prod_idx] < 1e-10:
            break

        # Move mass from the larger of the pair to somewhere else
        j1 = js[max_prod_idx]
        j2 = partners[max_prod_idx]

        if a[j1] >= a[j2]:
            donor = j1
        else:
            donor = j2

        # Find a good recipient: position that minimizes increase in max convolution
        # Heuristic: position far from the donor in the "convolution sense"
        # Try a few candidate positions
        candidates = []
        for offset in [n // 4, n // 3, n // 2, 3 * n // 4]:
            rec = (donor + offset) % n
            if rec != donor:
                candidates.append(rec)
        candidates = list(set(candidates))
        if not candidates:
            candidates = [(donor + 1) % n]

        best_recipient = candidates[0]
        best_test_val = cur_max

        # Try small transfers to each candidate
        for rec in candidates:
            for frac in [0.05, 0.02, 0.01]:
                delta = a[donor] * frac
                if delta < 1e-10:
                    continue
                a_test = a.copy()
                a_test[donor] -= delta
                a_test[rec] += delta
                a_test = project_simplex(a_test, S)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < best_test_val:
                    best_test_val = c_test.max()
                    best_recipient = rec
                    # Accept the best fraction
                    a_try = a_test
                    if c_test.max() < cur_max:
                        a = a_test
                        break
            else:
                continue
            break

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def propose_candidate(seed=42, budget_s=2100, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Phase 1: Load and quick-refine starting sequence.
    Phase 2: Fourier-parameterized low-dim optimization for shape discovery.
    Phase 3: Main LSE-FISTA refinement.
    Phase 4: Target-level subgradient.
    Phase 5: Conjugate-direction refinement.
    Phase 6: Peak shaving.
    Phase 7: High-beta LSE polish.
    Phase 8: Multi-length exploration.
    Phase 9: Structured restarts.
    Phase 10: Final coordinate descent and verification.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    print("Budget: %.0fs, deadline in %.0fs" % (budget_s, budget_s - safety))

    # Load starting sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_best(a):
        nonlocal global_best, global_a
        n = len(a)
        S = np.sqrt(2.0 * n)
        a = normalize_sum(a, n)
        c = fft_convolve_self(a)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a.copy()
            print("  >> New best: %.8f (n=%d)" % (val, n))
        return val

    # === Phase 1: Quick LSE-FISTA on given (warm start) ===
    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        remaining = deadline - time.monotonic()
        t1 = min(180.0, remaining * 0.1)
        a_norm = normalize_sum(given, n_g)
        c_init = fft_convolve_self(a_norm)
        print("=== Phase 1: LSE-FISTA warm start (n=%d, %.0fs, init=%.6f) ===" %
              (n_g, t1, c_init.max()))
        a1, v1 = run_lse_fista(a_norm, n_g, S_g, t1, rng,
                                beta_start=500.0, beta_end=30000.0, beta_steps=80)
        update_best(a1)
    else:
        print("No starting sequence found, using uniform.")
        n_g = 8192
        S_g = np.sqrt(2.0 * n_g)
        given = np.ones(n_g) * S_g / n_g
        global_a = given.copy()
        global_best = fft_convolve_self(given).max()

    # === Phase 2: Fourier parameterized optimization ===
    if global_a is not None:
        n_f = len(global_a)
        S_f = np.sqrt(2.0 * n_f)
        remaining = max(1.0, deadline - time.monotonic())
        t2 = min(120.0, remaining * 0.08)
        print("=== Phase 2: Fourier optimization (n=%d, %.0fs) ===" % (n_f, t2))
        a_f, v_f = run_fourier_opt(global_a, n_f, S_f, t2, rng, num_harmonics=8)
        update_best(a_f)

    # === Phase 3: Main LSE-FISTA ===
    if global_a is not None:
        n_m = len(global_a)
        S_m = np.sqrt(2.0 * n_m)
        remaining = max(1.0, deadline - time.monotonic())
        t3 = remaining * 0.2
        print("=== Phase 3: Main LSE-FISTA (n=%d, %.0fs) ===" % (n_m, t3))
        a3, v3 = run_lse_fista(global_a, n_m, S_m, t3, rng,
                                beta_start=200.0, beta_end=50000.0, beta_steps=150)
        update_best(a3)

    # === Phase 4: Target-level subgradient ===
    if global_a is not None:
        n_t = len(global_a)
        S_t = np.sqrt(2.0 * n_t)
        remaining = max(1.0, deadline - time.monotonic())
        t4 = remaining * 0.15
        print("=== Phase 4: Target-level (n=%d, %.0fs) ===" % (n_t, t4))
        a4, v4 = run_target_level(global_a, n_t, S_t, t4, rng)
        update_best(a4)

    # === Phase 5: Conjugate refinement ===
    if global_a is not None:
        n_c = len(global_a)
        S_c = np.sqrt(2.0 * n_c)
        remaining = max(1.0, deadline - time.monotonic())
        t5 = remaining * 0.15
        print("=== Phase 5: Conjugate refinement (n=%d, %.0fs) ===" % (n_c, t5))
        a5, v5 = run_conjugate_refinement(global_a, n_c, S_c, t5, rng)
        update_best(a5)

    # === Phase 6: Peak shaving ===
    if global_a is not None:
        n_p = len(global_a)
        S_p = np.sqrt(2.0 * n_p)
        remaining = max(1.0, deadline - time.monotonic())
        t6 = remaining * 0.12
        print("=== Phase 6: Peak shaving (n=%d, %.0fs) ===" % (n_p, t6))
        a6, v6 = run_peak_shaving(global_a, n_p, S_p, t6, rng)
        update_best(a6)

    # === Phase 7: High-beta LSE polish ===
    if global_a is not None:
        n_h = len(global_a)
        S_h = np.sqrt(2.0 * n_h)
        remaining = max(1.0, deadline - time.monotonic())
        t7 = remaining * 0.12
        print("=== Phase 7: High-beta LSE (n=%d, %.0fs) ===" % (n_h, t7))
        a7, v7 = run_lse_fista(global_a, n_h, S_h, t7, rng,
                                beta_start=20000.0, beta_end=100000.0, beta_steps=30)
        update_best(a7)

    # === Phase 8: Multi-length exploration ===
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 150 and global_a is not None:
        n_large = len(global_a) * 2
        if n_large <= 32768:
            S_large = np.sqrt(2.0 * n_large)
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n_large)
            init_large = np.interp(t_new, t_old, global_a)
            init_large = normalize_sum(init_large, n_large)
            c_large = fft_convolve_self(init_large)
            t8 = remaining * 0.25
            if t8 > 80:
                print("=== Phase 8: Larger n=%d (init=%.6f, %.0fs) ===" %
                      (n_large, c_large.max(), t8))
                a8, v8 = run_lse_fista(init_large, n_large, S_large, t8 * 0.6, rng,
                                        beta_start=100.0, beta_end=20000.0, beta_steps=60)
                update_best(a8)
                if global_a is not None and len(global_a) == n_large:
                    remaining2 = max(0.0, deadline - time.monotonic())
                    t8b = remaining2 * 0.3
                    if t8b > 30:
                        a8b, v8b = run_target_level(global_a, n_large, S_large, t8b, rng)
                        update_best(a8b)

    # === Phase 9: Structured restarts ===
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 60 and global_a is not None:
        n_r = len(global_a)
        S_r = np.sqrt(2.0 * n_r)
        n_restarts = max(2, int(remaining / 40))
        t9 = remaining * 0.25
        per_restart = max(5.0, t9 / n_restarts)
        print("=== Phase 9: %d restarts (n=%d, %.0fs total) ===" %
              (n_restarts, n_r, t9))

        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            a_r = global_a.copy()
            if r % 4 == 0:
                # Gaussian noise
                noise_scale = 0.008 * S_r / np.sqrt(n_r) * (1.0 + r * 0.3)
                a_r = project_simplex(a_r + rng.normal(0, noise_scale, n_r), S_r)
            elif r % 4 == 1:
                # Block shift
                block = n_r // 8
                shift = rng.integers(block // 2, block)
                a_r2 = a_r.copy()
                a_r2[shift:shift + block] = a_r[:block] * 0.9
                a_r2[:shift] = a_r2[:shift] + (a_r[:block] - a_r2[shift:shift + block] * 10 / block)
                a_r = project_simplex(np.maximum(a_r2, 0), S_r)
            elif r % 4 == 2:
                # Frequency domain perturbation
                fsize = 1 << (2 * n_r - 1).bit_length()
                A = fft(a_r, fsize)
                mask = np.abs(A) > np.abs(A).max() * 0.01
                A[mask] *= (1.0 + 0.005 * rng.standard_normal(np.sum(mask)))
                a_r = np.real(ifft(A))[:n_r]
                a_r = project_simplex(np.maximum(a_r, 0), S_r)
            else:
                # Smooth perturbation (low-pass)
                noise = rng.normal(0, 0.005 * S_r / np.sqrt(n_r), n_r)
                # Low-pass filter
                noise_f = fft(noise)
                cutoff = n_r // 8
                noise_f[cutoff:-cutoff] = 0
                noise = np.real(ifft(noise_f))[:n_r]
                a_r = project_simplex(a_r + noise, S_r)

            a_r, v_r = run_lse_fista(a_r, n_r, S_r, per_restart, rng,
                                      beta_start=500.0, beta_end=30000.0, beta_steps=50)
            update_best(a_r)

    # === Phase 10: Final polish ===
    if global_a is not None:
        remaining = max(1.0, deadline - time.monotonic())
        if remaining > 15:
            n_final = len(global_a)
            S_final = np.sqrt(2.0 * n_final)
            t10a = remaining * 0.4
            t10b = remaining * 0.35
            t10c = remaining * 0.2
            print("=== Phase 10: Final polish (%.0fs total) ===" % remaining)

            print("  10a: Conjugate refinement")
            a10a, _ = run_conjugate_refinement(global_a, n_final, S_final, t10a, rng)
            update_best(a10a)

            if global_a is not None:
                print("  10b: Peak shaving")
                a10b, _ = run_peak_shaving(global_a, n_final, S_final, t10b, rng)
                update_best(a10b)

            if global_a is not None:
                print("  10c: High-beta LSE")
                a10c, _ = run_lse_fista(global_a, n_final, S_final, t10c, rng,
                                         beta_start=50000.0, beta_end=100000.0, beta_steps=20)
                update_best(a10c)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.8f, internal = %.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            global_best = 2.0
    else:
        print("WARNING: No valid result found, using uniform fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        global_best = 2.0

    result = np.clip(global_a, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.8f" % (len(result), global_best))
    return [float(x) for x in result]
