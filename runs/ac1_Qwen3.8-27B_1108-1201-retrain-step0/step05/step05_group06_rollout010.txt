    v = np.asarray(v, dtype=np.float64)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive LSE-FISTA with Armijo
backtracking, aggressive beta continuation, and exact-subgradient polishing.
Key improvements over previous:
1. Armijo backtracking for larger effective steps

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_s(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_and_grad(a, beta):
    """Compute LSE value, gradient, full convolution, and max of convolution."""
    c = fft_convolve_self(a)
    m = c.max()
    shifted = beta * (c - m)
    shifted -= shifted.max()
    e = np.exp(shifted)

    lse_val = m + np.log(tot) / beta

    n = len(a)

    A = fft(a, fsize)

                k_star = int(np.argmax(c))
                w2 = np.zeros(len(c))
                w2[k_star] = 1.0
                W2 = fft(w2, fsize)
                A2 = fft(a, fsize)
                grad2 = 2.0 * np.real(ifft(W2 * np.conj(A2)))[:n]
                gnorm2 = np.linalg.norm(grad2)
                if gnorm2 > 1e-14:
                    for trial_alpha in [0.001, 0.005, 0.01, 0.02]:
                        a_t = project_simplex(a - trial_alpha * grad2, S)
                        c_t = fft_convolve_self(a_t)
                        if c_t.max() < cur_max - 1e-14:
                            a = a_t
                            break

            # Perturbation when very stuck
            if stuck > 2000 and it % 100 == 0:
                noise_scale = max(1e-6, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 5000.0))
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    c_final = fft_convolve_self(a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
        best_a = a.copy()

    return best_a, best_conv


def run_mirror_descent(a_init, max_time, rng, beta_init=100.0):
    """
    Mirror descent (multiplicative weights) with LSE gradient.
    Different geometry than FISTA - explores different regions.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_to_s(a, S)

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    last_improve = 0
    it = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 200:
            beta = min(100000.0, beta * 1.03)

        # LSE gradient
        _, grad, _, _ = lse_and_grad(a, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Mirror descent step: a_j *= exp(-step * grad_j)
        step = 0.1 / (beta * 0.01 + 1.0)
        step = min(step, 0.05)

        a_new = a * np.exp(-step * grad)
        a_new = normalize_to_s(a_new, S)

        # Accept if it helps
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max - 1e-16:
            a = a_new
        else:
            # Try smaller step
            a_half = a * np.exp(-step * 0.25 * grad)
            a_half = normalize_to_s(a_half, S)
            c_half = fft_convolve_self(a_half)
            if c_half.max() < cur_max - 1e-16:
                a = a_half
            else:
                # Tiny step
                a = normalize_to_s(a * np.exp(-step * 0.01 * grad), S)

        if stuck > 5000 and it % 200 == 0:
            # Multiplicative noise
            noise = rng.normal(0, 0.01, n)
            a = normalize_to_s(a * np.exp(noise), S)

    c_final = fft_convolve_self(a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
        best_a = a.copy()

    return best_a, best_conv


def make_cosine_init(n):
    """Raised cosine initialization."""
    t = np.linspace(0, 1, n)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t)) + 1e-6


def make_flat_init(n):
    """Uniform initialization."""
    return np.ones(n)


def make_bell_init(n, width=0.3):
    """Gaussian bell initialization."""
    t = np.linspace(0, 1, n)
    return np.exp(-(t - 0.5) ** 2 / (2 * width * width)) + 1e-6


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: LSE-FISTA (increasing beta) -> mirror descent -> exact polish -> final.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        print("Using provided sequence, n=%d" % n0)
    else:
        n0 = 8192
        a_start = make_cosine_init(n0)
        print("Using cosine init, n=%d" % n0)

    S = np.sqrt(2.0 * n0)
    a_init = normalize_to_s(a_start, S)

    # Initial evaluation
    c_init = fft_convolve_self(a_init)
    init_max = c_init.max()
    init_obj = 2.0 * n0 * init_max / (a_init.sum() ** 2)
    print("Initial: max_conv=%.8f, obj=%.8f" % (init_max, init_obj))
    global_best_obj = init_obj
    global_best_a = a_init.copy()

    # Also try alternative initializations quickly
    alts = [
        ("cosine", make_cosine_init(n0)),
        ("flat", make_flat_init(n0)),
        ("bell", make_bell_init(n0, 0.25)),
        ("bell_wide", make_bell_init(n0, 0.4)),
    ]
    for name, alt in alts:
        alt_norm = normalize_to_s(alt, S)
        c_alt = fft_convolve_self(alt_norm)
        alt_obj = 2.0 * n0 * c_alt.max() / (alt_norm.sum() ** 2)
        if alt_obj < global_best_obj:
            global_best_obj = alt_obj
            global_best_a = alt_norm.copy()
            a_init = alt_norm.copy()
            print("  Better alt '%s': obj=%.8f" % (name, alt_obj))

    print("Best start: obj=%.8f" % global_best_obj)

    # === Phase 1: Quick warmup (low beta) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.05
    if p1_budget > 5:
        print("\n--- Phase 1: Warmup beta=5 (%.0fs) ---" % p1_budget)
        t = time.monotonic()
        a1, conv1 = run_armijo_fista(global_best_a, p1_budget, rng,
                                      beta_init=5.0, beta_growth=1.01, beta_max=500.0)
        obj1 = 2.0 * n0 * conv1 / (a1.sum() ** 2)
        print("  conv=%.8f obj=%.8f (%.1fs)" % (conv1, obj1, time.monotonic() - t))
        if obj1 < global_best_obj:
            global_best_obj = obj1
            global_best_a = a1.copy()

    # === Phase 2: Main LSE-FISTA (medium beta) ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.30
    if p2_budget > 20:
        print("\n--- Phase 2: Main LSE beta=50 (%.0fs) ---" % p2_budget)
        t = time.monotonic()
        a2, conv2 = run_armijo_fista(global_best_a, p2_budget, rng,
                                      beta_init=50.0, beta_growth=1.04, beta_max=50000.0)
        obj2 = 2.0 * n0 * conv2 / (a2.sum() ** 2)
        print("  conv=%.8f obj=%.8f (%.1fs)" % (conv2, obj2, time.monotonic() - t))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: High beta LSE-FISTA ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.25
    if p3_budget > 20:
        print("\n--- Phase 3: High LSE beta=500 (%.0fs) ---" % p3_budget)
        t = time.monotonic()
        a3, conv3 = run_armijo_fista(global_best_a, p3_budget, rng,
                                      beta_init=500.0, beta_growth=1.06, beta_max=500000.0)
        obj3 = 2.0 * n0 * conv3 / (a3.sum() ** 2)
        print("  conv=%.8f obj=%.8f (%.1fs)" % (conv3, obj3, time.monotonic() - t))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Mirror descent (different geometry) ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.15
    if p4_budget > 15:
        print("\n--- Phase 4: Mirror descent (%.0fs) ---" % p4_budget)
        t = time.monotonic()
        a4, conv4 = run_mirror_descent(global_best_a, p4_budget, rng, beta_init=200.0)
        obj4 = 2.0 * n0 * conv4 / (a4.sum() ** 2)
        print("  conv=%.8f obj=%.8f (%.1fs)" % (conv4, obj4, time.monotonic() - t))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Exact subgradient polishing ===
    remaining = deadline - time.monotonic()
    p5_budget = remaining * 0.20
    if p5_budget > 15:
        print("\n--- Phase 5: Exact polish (%.0fs) ---" % p5_budget)
        t = time.monotonic()
        a5, conv5 = run_exact_polish(global_best_a, p5_budget, rng)
        obj5 = 2.0 * n0 * conv5 / (a5.sum() ** 2)
        print("  conv=%.8f obj=%.8f (%.1fs)" % (conv5, obj5, time.monotonic() - t))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Final alternating ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p6_budget = remaining * 0.85
        print("\n--- Phase 6: Final (%.0fs) ---" % p6_budget)
        t = time.monotonic()

        # Alternate between high-beta LSE and exact polish
        n_rounds = max(2, int(p6_budget / 80))
        round_time = p6_budget / n_rounds
        for i in range(n_rounds):
            if time.monotonic() >= deadline - 5:
                break
            if i % 2 == 0:
                a_cur, conv_cur = run_armijo_fista(global_best_a, round_time, rng,
                                                    beta_init=2000.0 * (1 + i * 2),
                                                    beta_growth=1.08, beta_max=1000000.0)
            else:
                a_cur, conv_cur = run_exact_polish(global_best_a, round_time, rng)

            obj_cur = 2.0 * n0 * conv_cur / (a_cur.sum() ** 2)
            improved = " *NEW*" if obj_cur < global_best_obj else ""
            print("  Round %d: conv=%.8f obj=%.8f%s" % (i, conv_cur, obj_cur, improved))
            if obj_cur < global_best_obj:
                global_best_obj = obj_cur
                global_best_a = a_cur.copy()

        print("  Phase 6 done in %.1fs" % (time.monotonic() - t))

    # Final result
    print("\n=== FINAL: obj=%.8f, n=%d ===" % (global_best_obj, len(global_best_a)))
    final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]

    # Verify
    ver = evaluate_sequence(final_list)
    if not np.isfinite(ver):
        print("WARNING: evaluate rejected! Using fallback.")
        n_fb = n0
        fallback = make_cosine_init(n_fb)
        fallback = normalize_to_s(fallback, np.sqrt(2.0 * n_fb))
        final_list = [float(x) for x in fallback]

    return final_list
