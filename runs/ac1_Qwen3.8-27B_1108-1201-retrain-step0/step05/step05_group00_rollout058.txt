def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:

    else:
        rho = int(np.max(np.where(valid)[0]) + 1)

"""
Minimize 2*n*max_conv/sum^2 using adaptive top-K/power-sum projected gradient
with spectral restarts. Tries multiple sequence lengths (8192, 16384) to find
the best ratio. Uses the given sequence as a warm start when possible.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def compute_conv(a, N):
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    v = v.copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def power_sum_grad(a, c, p, n, N2, p_weights):
    """Gradient of (sum c[k]^p) w.r.t. a.
    p_weights[k] = c[k]^{p-1} (precomputed).
    grad[m] = 2*p * sum_k p_weights[k] * (a[k-m] + a[m-k])
    """
    w = p_weights  # w[k] = c[k]^{p-1} for k=0..2n-2
    Af = fft(a, N2)
    Wf = fft(w, N2)
    u = np.real(ifft(Wf * np.conj(Af)))[:n]
    v = np.real(ifft(Af * Wf))[:n]
    return 2.0 * p * (u + v)


def topk_grad(a, w, n, N2):
    """Gradient of sum of top-K c values. w[k]=1 for top-K indices."""
    Af = fft(a, N2)
    Wf = fft(w, N2)
    u = np.real(ifft(Wf * np.conj(Af)))[:n]
    v = np.real(ifft(Af * Wf))[:n]
    return 2.0 * (u + v)


def spectral_restart(a, N, rng, frac=0.0):
    """Randomize fraction of FFT phases. frac=0 means full randomization."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    if frac <= 0:
        phase = rng.uniform(0, 2 * np.pi, N)
    else:
        phase = np.angle(A)
        n_rand = max(1, int(N * frac))
        rand_idx = rng.choice(N, size=n_rand, replace=False)
        phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def normalize(a, n, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * (S / n)
    return a * (S / s)


def optimize_sequence(a_init, n, S, N, N2, time_budget, rng, label=""):
    """Main optimization loop using adaptive power-sum / top-K gradient."""
    a = normalize(a_init, n, S).copy()
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    
    # Adaptive parameters
    p_power = 4.0  # Start with L_4 norm (smooth)
    p_max = 100.0
    it = 0
    stall = 0
    step_scale = 1.0
    
    # Nesterov momentum
    y = a.copy()
    t_mom = 1.0
    
    while time.monotonic() - start_time < time_budget:
        it += 1
        
        # Increase p over time (from smooth to sharp)
        progress = min(1.0, it / 5000.0)
        p_current = p_power + (p_max - p_power) * progress
        use_topk = p_current > 50.0
        
        # Compute convolution at y (Nesterov lookahead)
        c, A = compute_conv(y, N)
        cmax = c.max()
        
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute gradient
        if use_topk:
            K = max(1, int(n / (4 * progress + 1)))
            K = min(K, 2 * n - 1)
            top_idx = np.argsort(c)[::-1][:K]
            w = np.zeros(2 * n - 1)
            w[top_idx] = 1.0
            grad = topk_grad(y, w, n, N2)
        else:
            # Power sum gradient
            c_clipped = np.maximum(c, 1e-15)
            p_weights = c_clipped ** (p_current - 1)
            grad = power_sum_grad(y, c, p_current, n, N2, p_weights)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Step size
        step = step_scale * S / (gnorm * np.sqrt(n))
        step = min(step, 2.0 * S / n)  # Cap step size
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        x_new = project_simplex(y - step * grad, S)
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-normalize y to maintain sum constraint for next iteration
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y * (S / s_y)
        a = x_new
        t_mom = t_new
        
        # Periodic restart when stuck

            if rng.random() < 0.2:
                a = spectral_restart(a, N, rng, frac=0.3)
                y = a.copy()
                t_mom = 1.0
                stall = 0
            elif rng.random() < 0.1:
                a = spectral_restart(a, N, rng, frac=0.0)
                y = a.copy()
                t_mom = 1.0
                stall = 0
        
        # Adaptive step scale
        if it % 200 == 0:
            # Check progress
            c_check, _ = compute_conv(a, N)
            if c_check.max() < best_cmax:
                step_scale = min(2.0, step_scale * 1.1)
            else:
                step_scale = max(0.1, step_scale * 0.8)
    
    # Final verification
    c, _ = compute_conv(best_a, N)
    return best_a, c.max()
