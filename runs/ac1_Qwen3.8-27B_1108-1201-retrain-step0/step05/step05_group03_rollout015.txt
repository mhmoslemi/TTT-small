"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-start optimization with diverse initializations,
FISTA with top-k LSE smoothing, spectral flattening, and adaptive parameters.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta_eff = scale
        else:
            beta_eff = beta
        scaled = beta_eff * (c_top - cm)
    else:
        beta_eff = beta
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista(a_init, S, max_time, rng,
              beta_init=20.0, beta_rate=0.005, beta_max=10000.0,
              k_frac=0.10, use_momentum=True):
    """FISTA with top-k LSE, backtracking, and momentum."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0

    k = max(4, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        # Periodic restart when stuck
        if stuck > 1000:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def spectral_flatten(a, alpha, rng):
    """One step of spectral flattening."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a
    ratio = power / max(mean_p, 1e-20)
    new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + ratio))
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / power)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def resample_sequence(a, n_target):
    """Resample a sequence to a different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def generate_diverse_starts(n, S, rng):

        A = rng.normal(0, 1, fsize) + 1j * rng.normal(0, 1, fsize)

            A[fsize - i] = np.conj(A[i])

        x = np.convolve(x, kernel, mode='same')
        starts.append(normalize(x, S))

    for p in [0.5, 1.0, 2.0]:
        x = np.power(np.arange(1, n+1), -p)

def run_adam(a_init, S, max_time, rng,
             lr_init=0.01, beta_lse=50.0, beta_rate=0.01):
    """Adam optimizer with LSE gradient."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    k = max(4, int(0.08 * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()

        beta = min(5000.0, beta_lse * np.exp(beta_rate * t))
        grad = topk_lse_gradient(a, c, k, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        progress = min(1.0, (time.monotonic() - start) / max_time)
        lr = lr_init * (1.0 - 0.8 * progress)
        if lr < 1e-6:
            break

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)

        if t % 5000 == 0 and rng.random() < 0.1:
            noise_scale = 0.002 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                m = np.zeros(n)
                v = np.zeros(n)

    return best_a, best

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-start optimization with diverse spectral/structured initializations,
FISTA with adaptive top-k LSE smoothing, and multi-scale (n=8192, n=16384) search.
Key improvements over baseline: diverse start generation, adaptive k scheduling,
spectral-domain perturbations, and focused polishing phase.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    diff = c_top - cm
    scaled = beta * diff
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, diff.max())
        if scale > 0:
            beta = scale
            scaled = beta * diff
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_obj(a, n):
    """Compute objective 2n*max(conv)/sum^2."""
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        # Adaptive k: start wide, narrow down
        progress = min(1.0, it / 50000.0)
        k_frac = k_frac_init - (k_frac_init - k_frac_min) * progress
        k = max(4, int(k_frac * (2 * n - 1)))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-14:
            best = cur_max
            best_a = y.copy()
            stuck = 0
        else:
            stuck += 1

        grad = topk_lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / max(2.0 * beta * 0.08, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(12):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-13:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            momentum_coef = (t_val - 1.0) / t_new
            y_new = x_new + momentum_coef * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        # Perturbation to escape local minima
        if stuck > perturb_every and rng.random() < 0.3:
            noise_scale = 0.004 * S / np.sqrt(n) * (1.0 + 0.5 * np.sin(0.01 * stuck))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_adam(a_init, S, max_time, rng,
             lr_init=0.01, beta_lse=40.0, beta_rate=0.008):
    """Adam with LSE gradient and adaptive learning rate."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    k = max(4, int(0.08 * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best - 1e-14:
            best = cur_max
            best_a = a.copy()

        beta = min(5000.0, beta_lse * np.exp(beta_rate * t))
        grad = topk_lse_gradient(a, c, k, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        progress = min(1.0, (time.monotonic() - start) / max_time)
        lr = lr_init * (1.0 - 0.85 * progress)
        if lr < 1e-7:
            break

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_new, S)

        if t % 4000 == 0 and rng.random() < 0.12:
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                m = np.zeros(n)
                v = np.zeros(n)
    return best_a, best


def spectral_flatten(a, alpha, rng):
    """Spectral flattening: reduce peaks in power spectrum."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    mean_p = power.mean()
    if mean_p < 1e-20:
        return a
    ratio = power / max(mean_p, 1e-20)
    new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + ratio))
    new_power = np.maximum(new_power, 1e-30)
    A_new = A * np.sqrt(new_power / power)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def spectral_perturb(a, amount, rng):
    """Perturb phase in frequency domain."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    phase = rng.normal(0, amount, fsize)
    phase[0] = 0.0
    A_new = A * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new, S)


def resample_sequence(a, n_target):
    """Resample sequence to different length."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def generate_starts(n, S, rng):
    """Generate diverse starting sequences for optimization."""
    starts = []

    # Uniform
    starts.append(np.ones(n) * (S / n))

    # Random non-negative (Gaussian clipped)
    for trial in range(4):
        x = rng.normal(0, 1, n)
        x = np.clip(x, 0, None)
        if x.sum() > 0:
            starts.append(normalize(x, S))

    # Spectral white noise (flat power spectrum, non-negative)
    for trial in range(3):
        fsize = 1 << (2 * n - 1).bit_length()
        A = rng.normal(0, 1, fsize) + 1j * rng.normal(0, 1, fsize)
        for i in range(1, fsize // 2 + 1):
            A[fsize - i] = np.conj(A[i])
        if fsize % 2 == 0:
            A[fsize // 2] = np.abs(A[fsize // 2])
        a = np.real(ifft(A, fsize))[:n]
        a = np.clip(a, 0, None)
        if a.sum() > 1e-10:
            starts.append(normalize(a, S))

    # Smooth random (low-pass filtered noise)
    for trial in range(3):
        x = rng.normal(0, 1, n)
        # Simple moving average smoothing
        kernel_size = max(1, n // 100)
        kernel = np.ones(kernel_size) / kernel_size
        x = np.convolve(x, kernel, mode='same')
        x = np.clip(x, 0, None)
        if x.sum() > 1e-10:
            starts.append(normalize(x, S))

    # Power law / decay profiles
    for p in [0.3, 0.7, 1.5, 3.0]:
        x = np.power(np.linspace(0.01, 1.0, n), p)
        starts.append(normalize(x, S))
        x2 = np.power(np.linspace(1.0, 0.01, n), p)
        starts.append(normalize(x2, S))

    # Sinusoidal modulated
    for freq in [1, 2, 3, 5]:
        for amp in [0.3, 0.6, 0.9]:
            x = 1.0 + amp * np.sin(2 * np.pi * freq * np.arange(n) / n)
            x = np.clip(x, 0, None)
            if x.sum() > 0:
                starts.append(normalize(x, S))

    # Sparse peaks + smoothing
    for trial in range(3):
        x = np.zeros(n)
        n_peaks = rng.integers(max(5, n // 20), max(10, n // 5))
        positions = rng.choice(n, size=n_peaks, replace=False)
        x[positions] = 1.0
        ks = max(3, n // 50)
        kernel = np.ones(ks) / ks
        x = np.convolve(x, kernel, mode='same')
        if x.sum() > 0:
            starts.append(normalize(x, S))

    return starts


def propose_candidate(budget_s=2400):
    """
    Multi-start search minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses diverse initializations, FISTA with adaptive LSE, and multi-length search.
    """
    rng = np.random.default_rng(777)
    start_total = time.monotonic()
    deadline = start_total + budget_s - 25.0

    print("=== Multi-start optimization, budget=%.0fs ===" % budget_s)

    # Try to load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given = np.clip(given, 0, 1000)
            if given.sum() > 0:
                print("Loaded provided sequence, length=%d, sum=%.4f" % (len(given), given.sum()))
            else:
                given = None
    except Exception:
        pass

    global_best_obj = np.inf
    global_a = None
    global_n = 0

    def check_update(a, n):
        nonlocal global_best_obj, global_a, global_n
        if global_a is not None and len(global_a) == n:
            prev_diff = np.max(np.abs(a - global_a))
        else:
            prev_diff = np.inf
        obj = compute_obj(a, n)
        if obj < global_best_obj - 1e-9:
            global_best_obj = obj
            global_a = a.copy()
            global_n = n
            print("  *** NEW BEST: obj=%.6f (n=%d) ***" % (obj, n))
        elif obj < global_best_obj and prev_diff > 1e-6:
            # Same objective but different sequence - keep the current best
            pass
        return obj

    S = 1.0  # Scale invariant normalization

    # ============ PHASE 1: Multi-start at n=8192 ============
    n1 = 8192
    phase1_budget = (deadline - time.monotonic()) * 0.42
    print("\n[Phase 1] Multi-start FISTA, n=%d, budget=%.0fs" % (n1, phase1_budget))

    starts = generate_starts(n1, S, rng)

    # Add given sequence if available
    if given is not None:
        if len(given) == n1:
            starts.insert(0, normalize(given.copy(), S))
        else:
            starts.insert(0, normalize(resample_sequence(given, n1), S))
        print("  Added given sequence as primary start")

    # Quick scoring
    scores = []
    for i, s in enumerate(starts):
        c = fft_convolve_self(s)
        scores.append((c.max(), i))
    scores.sort()

    print("  %d starts generated. Top 5 scores:" % len(starts))
    for sc, idx in scores[:5]:
        print("    start[%d]: max_conv=%.8f -> obj=%.6f" % (idx, sc, 2.0 * n1 * sc / S**2))

    # Run FISTA from top starts
    n_starts_to_try = min(20, len(starts))
    time_per_start = phase1_budget / n_starts_to_try

    for rank in range(n_starts_to_try):
        elapsed = time.monotonic() - start_total
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        _, idx = scores[rank]
        s = starts[idx]
        t_alloc = min(time_per_start * 1.5, remaining * 0.9 / max(1, n_starts_to_try - rank))
        if t_alloc < 5:
            break

        a_opt, val_opt = run_fista(s, S, t_alloc, rng,
                                    beta_init=15.0, beta_rate=0.004,
                                    beta_max=8000.0, k_frac_init=0.10,
                                    k_frac_min=0.03, perturb_every=1200)
        obj = 2.0 * n1 * val_opt / S**2
        if rank < 5 or obj < global_best_obj:
            print("  start[%d] rank=%d: obj=%.6f" % (idx, rank, obj))
        check_update(a_opt, n1)

    print("  Phase 1 best: obj=%.6f" % global_best_obj)

    # ============ PHASE 2: Try n=16384 ============
    remaining = deadline - time.monotonic()
    if remaining > 250 and global_a is not None:
        n2 = 16384
        phase2_budget = remaining * 0.25
        print("\n[Phase 2] n=%d exploration, budget=%.0fs" % (n2, phase2_budget))

        starts2 = []
        # Resample best
        starts2.append(normalize(resample_sequence(global_a, n2), S))
        # Add some diverse starts
        extra = generate_starts(n2, S, rng)
        starts2.extend(extra[:6])

        for i, s in enumerate(starts2[:8]):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = phase2_budget / max(1, 8 - i) * 0.8
            if t_alloc < 5:
                break
            a_opt, val_opt = run_fista(s, S, t_alloc, rng,
                                        beta_init=15.0, beta_rate=0.003,
                                        beta_max=6000.0, k_frac_init=0.08,
                                        k_frac_min=0.025, perturb_every=1500)
            obj = 2.0 * n2 * val_opt / S**2
            print("  n=%d start[%d]: obj=%.6f" % (n2, i, obj))
            check_update(a_opt, n2)

    # ============ PHASE 3: Intensive optimization ============
    remaining = deadline - time.monotonic()
    if remaining > 80 and global_a is not None:
        n_best = len(global_a)
        print("\n[Phase 3] Intensive optimization, n=%d, budget=%.0fs" % (n_best, remaining))

        # 3a: FISTA with higher initial beta
        t3a = remaining * 0.35
        a_opt, val_opt = run_fista(global_a, S, t3a, rng,
                                    beta_init=80.0, beta_rate=0.003,
                                    beta_max=15000.0, k_frac_init=0.06,
                                    k_frac_min=0.015, perturb_every=2000)
        obj = 2.0 * n_best * val_opt / S**2
        print("  Intensive FISTA: obj=%.6f" % obj)
        check_update(a_opt, n_best)

        # 3b: Adam refinement
        remaining2 = deadline - time.monotonic()
        if remaining2 > 40:
            t3b = remaining2 * 0.35
            a_opt, val_opt = run_adam(global_a, S, t3b, rng,
                                       lr_init=0.005, beta_lse=150.0,
                                       beta_rate=0.006)
            obj = 2.0 * n_best * val_opt / S**2
            print("  Adam: obj=%.6f" % obj)
            check_update(a_opt, n_best)

        # 3c: Spectral flatten + re-optimize
        remaining3 = deadline - time.monotonic()
        if remaining3 > 40:
            a_flat = spectral_flatten(global_a, 0.15, rng)
            a_pert = spectral_perturb(a_flat, 0.05, rng)
            t3c = remaining3 * 0.5
            a_opt, val_opt = run_fista(a_pert, S, t3c, rng,
                                        beta_init=80.0, beta_rate=0.003,
                                        beta_max=15000.0, k_frac_init=0.05,
                                        k_frac_min=0.015, perturb_every=2000)
            obj = 2.0 * n_best * val_opt / S**2
            print("  Spectral+Perturb+FISTA: obj=%.6f" % obj)
            check_update(a_opt, n_best)

    # ============ PHASE 4: Final polish ============
    remaining = deadline - time.monotonic()
    if remaining > 25 and global_a is not None:
        n_best = len(global_a)
        print("\n[Phase 4] Final polish, budget=%.0fs" % remaining)

        # Very high beta, no momentum, small k
        a_opt, val_opt = run_fista(global_a, S, remaining - 10, rng,
                                    beta_init=500.0, beta_rate=0.001,
                                    beta_max=50000.0, k_frac_init=0.02,
                                    k_frac_min=0.005, use_momentum=False,
                                    perturb_every=99999)
        obj = 2.0 * n_best * val_opt / S**2
        print("  Polish: obj=%.6f" % obj)
        check_update(a_opt, n_best)

    # ============ Final verification and return ============
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\n=== FINAL ===")
        print("Length: %d" % len(global_a))
        print("Internal obj: %.6f" % global_best_obj)
        print("evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver) or ver > 2.0:
            print("  WARNING: evaluation rejected or poor, using fallback")
            global_a = np.ones(256) * (1.0 / 256)
            final_list = [float(x) for x in global_a]
            print("  Fallback eval: %.6f" % evaluate_sequence(final_list))
    else:
        global_a = np.ones(256) * (1.0 / 256)
        final_list = [float(x) for x in global_a]
        print("\n=== FALLBACK: uniform n=256, obj=%.6f ===" % evaluate_sequence(final_list))

    total_time = time.monotonic() - start_total
    print("Total time: %.1fs" % total_time)
    return [float(x) for x in global_a]
