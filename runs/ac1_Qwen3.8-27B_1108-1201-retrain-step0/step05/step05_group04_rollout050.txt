Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: LP-based Frank-Wolfe direction finding combined with LSE-FISTA smoothing
and multiplicative weights fine-tuning. The LP at each step finds the best direction
by maximizing sum(b) subject to cross-convolution constraints (f*b)(k) <= M at active lags.
This naturally pushes mass to regions where f is small, reducing the self-convolution peak.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta

        top_idx = np.argsort(c)[::-1][:n_peaks]

        if resp_max < 1e-30:
            break

        # Exponentiated gradient with adaptive learning rate
        lr = 0.005 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = normalize_sum(a, n)

        # Periodic noise

        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking
        step = 0.01
        improved = False
        for _ in range(10):
            a_test = normalize_sum(a - step * grad, n)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur_max * (1 - 1e-7):
                a = a_test
                improved = True
                break
            step *= 0.5

        if not improved:
            step = 0.0005 / (1 + it * 0.0005)
            a = normalize_sum(a - step * grad, n)

        # Periodic perturbation
        if it % 100 == 0:
            noise_scale = 0.0005 * S / np.sqrt(n) * np.exp(-it / 3000.0)
            a = normalize_sum(a + rng.normal(0, noise_scale, n), n)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Uses LP-based Frank-Wolfe direction finding as the main engine,
    combined with LSE-FISTA for initialization and multiplicative weights
    for fine-tuning.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 90.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    print("Budget: %.0fs, safety margin: %.0fs" % (budget_s, safety))

    # Load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_sequence(a):
        nonlocal global_best, global_a
        n = len(a)
        S = np.sqrt(2.0 * n)
        a = normalize_sum(a, n)
        c = fft_convolve_self(a)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a.copy()
            print("  New best: %.8f (n=%d)" % (val, n))
            return True
        return False

    # === Phase 1: Quick LSE-FISTA on given sequence ===
    if given is not None:
        n_g = len(given)
        t1 = min(45.0, (deadline - time.monotonic()) * 0.05)
        print("=== Phase 1: LSE-FISTA (n=%d, %.0fs) ===" % (n_g, t1))
        a1 = run_lse_fista(given, n_g, t1, rng,
                           beta_start=100.0, beta_end=5000.0, beta_steps=50)
        try_sequence(a1)

    # If no given, create a good initial
    if global_a is None:
        n_init = 8192
        t = np.linspace(0, 1, n_init)
        # Asymmetric raised cosine
        a_init = 0.5 * (1 + np.cos(np.pi * (t - 0.35) * 2))
        a_init = np.maximum(a_init, 1e-10)
        global_a = a_init
        global_best = np.inf
        try_sequence(a_init)

    # === Phase 2: Main LP-FW optimization ===
    if global_a is not None:
        n_main = len(global_a)
        remaining = deadline - time.monotonic()
        t2 = remaining * 0.5
        print("=== Phase 2: LP-FW (n=%d, %.0fs) ===" % (n_main, t2))
        a2 = run_lp_fw(global_a, n_main, t2, rng, K=30, K_max=60)
        try_sequence(a2)

    # === Phase 3: More LP-FW with different K ===
    if global_a is not None:
        n_main = len(global_a)
        remaining = deadline - time.monotonic()
        t3 = remaining * 0.3
        if t3 > 30:
            print("=== Phase 3: LP-FW K=15 (n=%d, %.0fs) ===" % (n_main, t3))
            a3 = run_lp_fw(global_a, n_main, t3, rng, K=15, K_max=40)
            try_sequence(a3)

    # === Phase 4: Multiplicative weights fine-tuning ===
    if global_a is not None:
        n_p = len(global_a)
        remaining = deadline - time.monotonic()
        t4 = remaining * 0.35
        if t4 > 15:
            print("=== Phase 4: Mult weights (n=%d, %.0fs) ===" % (n_p, t4))
            a4 = multiplicative_weights(global_a, n_p, t4, rng, n_peaks=15)
            try_sequence(a4)

    # === Phase 5: Active-set refinement ===
    if global_a is not None:
        n_f = len(global_a)
        remaining = deadline - time.monotonic()
        t5 = remaining * 0.5
        if t5 > 10:
            print("=== Phase 5: Active-set refinement (n=%d, %.0fs) ===" % (n_f, t5))
            a5 = active_set_refinement(global_a, n_f, t5, rng)
            try_sequence(a5)

    # === Phase 6: Final LSE-FISTA polish ===
    if global_a is not None:
        n_f2 = len(global_a)
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print("=== Phase 6: Final LSE polish (%.0fs) ===" % remaining)
            a6 = run_lse_fista(global_a, n_f2, remaining * 0.7, rng,
                               beta_start=2000.0, beta_end=50000.0, beta_steps=40)
            try_sequence(a6)

    # === Phase 7: Try a fresh LP-FW from scratch with different init ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            n_try = len(global_a)
            # Create a perturbed version
            perturbed = global_a * (1 + 0.1 * rng.normal(0, 1, n_try))
            perturbed = np.maximum(perturbed, 0)
            t7 = remaining * 0.5
            print("=== Phase 7: Perturbed LP-FW (%.0fs) ===" % t7)
            a7 = run_lp_fw(perturbed, n_try, t7, rng, K=25, K_max=50)
            try_sequence(a7)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence=%.8f, internal=%.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using uniform fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result found, using uniform fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
