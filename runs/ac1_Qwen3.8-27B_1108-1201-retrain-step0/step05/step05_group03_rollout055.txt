"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Uses multi-phase FISTA with adaptive top-k LSE smoothing, peak-directed
subgradient polishing, and multi-length exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Scale array to have sum S, maintaining non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_grad(a, c, k, beta):
    """Gradient of top-k LSE of self-convolution w.r.t. a."""
    n = len(a)
    conv_len = len(c)
    k = min(k, conv_len)
    
    # Find top-k indices efficiently
    if k < conv_len:
        top_indices = np.argpartition(c, -k)[-k:]
    else:
        top_indices = np.arange(conv_len)
    
    c_top = c[top_indices]
    cm = c_top.max()
    scaled = np.clip(beta * (c_top - cm), -500.0, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    
    w_full = np.zeros(conv_len)
    w_full[top_indices] = w
    
    # FFT size for gradient computation
    fsize = 1 << (conv_len + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def resample(a, n_target):
    """Resample sequence to target length using linear interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)

def flat_spectrum_init(n, rng, n_iter=200):
    """Initialize using alternating projection for flat power spectrum."""
    S = np.sqrt(2.0 * n)
    a = rng.random(n) * S / n
    a = normalize_to_sum(a, S)
    
    fsize = 1 << (2 * n - 1).bit_length()
    
    for _ in range(n_iter):
        A = fft(a, fsize)
        mag = np.abs(A)
        # Target: flat magnitude (with DC component preserved for sum constraint)
        target_mag = np.full(fsize, mag.mean())
        # Preserve DC for sum constraint
        target_mag[0] = mag[0]
        # Update phase, fix magnitude
        A_new = A / mag * target_mag
        A_new[0] = 0  # Remove DC (we'll handle sum separately)
        a_new = np.real(ifft(A_new, fsize))[:n]
        a = np.maximum(a_new, 0.0)
        a = normalize_to_sum(a, S)
    
    return a

**Alternative: Use the duality between time and frequency domains**

The objective is 2n * max_k c[k] / S^2 where c = a * a.

In the frequency domain: c[k] = (1/2π) ∫ |A(ω)|^2 e^{ikω} dω (continuous case)
Or for discrete: c[k] = (1/N_f) sum_m |A[m]|^2 exp(2πikm/N_f)

So max_k c[k] is the maximum of the inverse DFT of |A|^2.

We want to minimize the peak of the IFFT of |A|^2, subject to:
- a >= 0 (time domain constraint)
- sum(a) = S (which fixes A[0] = S)

This is a non-convex problem (the a >= 0 constraint is non-convex in the frequency domain).

One approach: alternately optimize in time and frequency domains.
- Frequency step: given |A|^2, redistribute it to minimize the peak of its IFFT
- Time step: given a, project onto a >= 0 and sum(a) = S

The frequency step is: minimize max_k IFFT(|A|^2)[k] subject to |A[0]|^2 fixed (= S^2) and |A|^2 >= 0.

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Multi-phase optimization: FISTA with adaptive top-k LSE, spectral equalization,
peak-directed subgradient polishing, and multi-length exploration.
Uses parent sequence as primary starting point with multiple initialization strategies.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_grad(a, c, k, beta):
    n = len(a)
    conv_len = len(c)
    k = min(k, conv_len)
    if k < conv_len:
        top_indices = np.argpartition(c, -k)[-k:]
    else:
        top_indices = np.arange(conv_len)
    c_top = c[top_indices]
    cm = c_top.max()
    scaled = np.clip(beta * (c_top - cm), -500.0, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)
    w_full = np.zeros(conv_len)
    w_full[top_indices] = w
    fsize = 1 << (conv_len + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def resample_seq(a, n_target):
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def spectral_equalize(a, S, alpha=0.15):
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    non_dc_mask = np.ones(fsize, dtype=bool)
    non_dc_mask[0] = False
    non_dc_power = power[non_dc_mask]
    non_dc_power = non_dc_power[non_dc_power > 1e-30]
    if len(non_dc_power) == 0:
        return a
    target = np.geomean(non_dc_power)
    target_power = np.full(fsize, target)
    target_power[0] = power[0]
    new_power = power + alpha * (target_power - power)
    new_power = np.maximum(new_power, 1e-30)
    mag = np.abs(A)
    phase = np.ones(fsize, dtype=np.complex128)
    nonzero = mag > 1e-30
    phase[nonzero] = A[nonzero] / mag[nonzero]
    A_new = phase * np.sqrt(new_power)
    A_new[0] = A[0]
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.maximum(a_new, 0.0)
    return normalize_to_sum(a_new, S)


def run_fista(a_init, S, time_limit, rng, beta_start=20.0, beta_end=15000.0,

              spec_eq_interval=1500, spec_eq_prob=0.3):

    y = a.copy()
    t_val = 1.0

    c = fft_convolve_self(a)

    best_a = a.copy()

        beta = beta_start * (beta_end / beta_start) ** progress

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = y.copy()
            last_improve_it = it

        a_alt = resample_seq(global_a, n_primary)
        a_alt = normalize_to_sum(a_alt, S1)
        # Perturb more aggressively
        noise = 0.02 * S1 / np.sqrt(n_primary)
        a_alt = project_simplex(a_alt + rng2.normal(0, noise, n_primary), S1)
        do_eval(a_alt, n_primary)

        t5 = remaining * 0.5
        if t5 > 20:
            a_opt, _ = run_fista(a_alt, S1, t5, rng2,
                                 beta_start=50.0, beta_end=30000.0,
                                 k_frac_start=0.04, k_frac_end=0.002,
                                 use_momentum=True, spec_eq_interval=1000,
                                 spec_eq_prob=0.3)
            do_eval(a_opt, n_primary)

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nFinal verification via evaluate_sequence: %.6f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256) * (np.sqrt(512.0) / 256)
    else:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)

    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    print("Returning sequence of length %d, best obj=%.6f" % (len(global_a), global_best))
    return final_list
