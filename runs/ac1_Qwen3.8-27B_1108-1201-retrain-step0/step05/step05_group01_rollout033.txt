"""
Minimize 2*n*max_conv/sum^2 using MWU, Frank-Wolfe, and FISTA combination.
Key innovations: multiplicative weight updates for natural non-negativity,
adaptive smoothing with annealing, symmetry exploitation, and frequency-domain
perturbations for escaping local minima.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def run_mwu(a_init, max_time, rng, beta_start=5.0, beta_end=500.0,
            eta_scale=1.0, mode="anneal"):
    """
    Multiplicative Weight Updates for minimizing LSE(conv(a,a)).
    a_j <- a_j * exp(-eta * grad_j), then normalize to sum = S.
    This naturally maintains non-negativity.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Annealing schedule for beta
        frac = min(1.0, it / (max_time * 100))  # approximate
        if mode == "anneal":
            beta = beta_start * (beta_end / beta_start) ** frac
        elif mode == "oscillate":
            beta = (beta_start + beta_end) / 2 * (1 + 0.5 * np.sin(2 * np.pi * it / 500))
            beta = max(beta_start, min(beta, beta_end))
        else:
            beta = beta_start

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve

        # Get weights
        w = lse_weights(c, beta)
        grad = conv_gradient(a, w)

        # Learning rate for MWU
        # grad is roughly O(S) in magnitude
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        eta = eta_scale / (gmax * np.sqrt(it + 1))
        eta = min(eta, 0.5 / max(a.max(), 1e-10))  # prevent overflow

        # MWU update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a = log_a - eta * grad
        # Normalize to keep sum = S
        log_max = log_a.max()
        log_a = log_a - log_max
        a_new = np.exp(log_a)
        a_new = a_new * (S / a_new.sum())
        a_new = np.clip(a_new, 0, 1000)
        
        if a_new.sum() < 1e-10 or not np.all(np.isfinite(a_new)):
            break
        a = a_new

        # Periodic perturbation when stuck
        if stuck > 500 and it % 100 == 0:
            noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-stuck / 2000.0)
            if noise_scale > 1e-8:
                # Frequency-domain smooth noise
                noise = rng.normal(0, 1.0, n)
                F = fft(noise)
                cutoff = max(3, n // 100)
                F[cutoff:-cutoff] *= 0.1
                noise_smooth = np.real(ifft(F))[:n]
                noise_smooth = noise_smooth / (np.linalg.norm(noise_smooth) + 1e-15) * noise_scale
                a = project_simplex(a + noise_smooth, S)
                last_improve = it  # Reset stuck counter

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_frank_wolfe(a_init, max_time, rng, beta_start=10.0, beta_end=200.0):
    """
    Frank-Wolfe (Conditional Gradient) method.
    At each step: compute gradient, find vertex of simplex closest to -grad,
    interpolate with step size.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        frac = min(1.0, it / max(1, int(max_time * 50)))
        beta = beta_start * (beta_end / beta_start) ** frac

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        w = lse_weights(c, beta)
        grad = conv_gradient(a, w)

        # Frank-Wolfe: linear oracle gives vertex
        # v = S * e_{argmin(grad)}
        j_min = np.argmin(grad)
        v = np.zeros(n)
        v[j_min] = S

        # Step size: FW uses t_k = 2/(k+2) or line search
        t = 2.0 / (it + 2.0)
        if stuck > 200:
            t = min(t + 0.01, 0.3)  # Slightly larger steps when stuck

        a_new = (1 - t) * a + t * v
        a_new = np.maximum(a_new, 0)
        a_new = a_new * (S / a_new.sum())
        a = a_new

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_fista(a_init, max_time, rng, beta_start=10.0, beta_end=500.0):
    """Standard FISTA with annealed beta."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        frac = min(1.0, it / max(1, int(max_time * 60)))
        beta = beta_start * (beta_end / beta_start) ** frac
        step = 1.0 / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        w = lse_weights(c, beta)
        grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = step_scale = 1.0 / max(beta, 1.0)
        if stuck > 1000:
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 3000 == 0:
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_symmetric_mwu(a_init, max_time, rng, beta_start=10.0, beta_end=500.0):
    """
    MWU with symmetry constraint: a_j = a_{n-1-j}.
    This halves the effective dimension and can find better solutions.
    """
    n = len(a_init)
    half = n // 2
    S = np.sqrt(2.0 * n)
    
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    # Enforce symmetry
    a_sym = 0.5 * (a + a[::-1])
    a_sym = a_sym * (S / a_sym.sum())
    
    # Work with the first half
    a_half = a_sym[:half].copy()
    
    start = time.monotonic()
    best_conv = np.inf
    best_a = a_sym.copy()
    it = 0
    last_improve = 0

    def full_sequence(half_vals):
        """Reconstruct full symmetric sequence from first half."""
        if n % 2 == 0:
            full = np.concatenate([half_vals, half_vals[::-1]])
        else:
            full = np.concatenate([half_vals, half_vals[-2::-1]])
        return full

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, int(max_time * 60)))
        beta = beta_start * (beta_end / beta_start) ** frac

        a_full = full_sequence(a_half)
        a_full = a_full * (S / a_full.sum())

        c = fft_convolve_self(a_full)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a_full.copy()
            last_improve = it

        stuck = it - last_improve
        w = lse_weights(c, beta)
        grad_full = conv_gradient(a_full, w)
        # Project gradient back to half
        grad_half = 0.5 * (grad_full[:half] + grad_full[::-1][:half])
        if n % 2 != 0:
            # Adjust for odd length
            pass

        gmax = np.abs(grad_half).max()
        if gmax < 1e-15:
            break
        eta = 1.0 / (gmax * np.sqrt(it + 1))
        eta = min(eta, 0.3 / max(a_half.max(), 1e-10))

        log_half = np.log(np.maximum(a_half, 1e-300))
        log_half = log_half - eta * grad_half
        log_max = log_half.max()
        log_half = log_half - log_max
        a_half = np.exp(log_half)
        
        if a_half.sum() < 1e-10 or not np.all(np.isfinite(a_half)):
            break

        if stuck > 300 and it % 200 == 0:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck / 3000.0)
            if noise_scale > 1e-8:
                a_half = np.maximum(a_half + rng.normal(0, noise_scale, half), 0)

    return best_a, best_conv

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y).
    Returns 2 * cross-correlation of w and y."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def exact_subgradient(y, c, top_k=1, tol_frac=0.0):
    """Exact subgradient using active set (top-k or tolerance-based)."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        idx = np.argsort(c)[::-1][:top_k]
        active = idx
    k = min(len(active), top_k) if top_k > 0 else len(active)
    active = active[:max(k, 1)]
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return conv_gradient(y, w_full)


def lse_value(c, beta):
    """Compute LSE value."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    return cm + np.log(np.exp(shifted).sum()) / beta


def run_fista_bt(a_init, max_time, rng, beta_start=5.0, beta_end=1000.0,
                 use_active=False, active_threshold=200.0):
    """
    FISTA with backtracking line search and adaptive beta annealing.
    At high beta, optionally switches to exact active-set subgradient.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / beta_start

    while time.monotonic() - start < max_time:
        it += 1

        # Annealing: exponential ramp
        frac = min(1.0, it / max(1.0, max_time * 40.0))
        beta = beta_start * (beta_end / beta_start) ** (frac ** 0.7)
        beta = max(beta_start, min(beta, 10000.0))

        c = fft_convolve_self(y)
        cur_obj_val = 2.0 * n * c.max() / (S * S)
        if cur_obj_val < best_obj - 1e-12:
            best_obj = cur_obj_val
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Choose gradient method
        if use_active and beta > active_threshold:
            # Use active set: top lags within tolerance of max
            max_val = c.max()
            tol = max(0.001, 0.01 * min(1.0, stuck / 500.0))
            active = np.where(c >= max_val * (1.0 - tol))[0]
            if len(active) > 64:
                idx = np.argsort(c)[::-1][:64]
                active = idx
            w_full = np.zeros(len(c))
            w_full[active] = 1.0 / len(active)
            grad = conv_gradient(y, w_full)
        else:
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step = min(step * 1.5, 2.0 / beta)
        c_y = c.copy()
        backtracks = 0
        while backtracks < 20:
            x_try = project_simplex(y - step * grad, S)
            c_try = fft_convolve_self(x_try)
            # Check sufficient decrease (Armijo-like)
            descent = np.dot(grad, y - x_try)
            if 2.0 * n * c_try.max() / (S * S) <= cur_obj_val - 0.1 * descent / (S * S):
                break
            step *= 0.5
            backtracks += 1
        else:
            x_try = project_simplex(y - step * grad, S)

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_try + ((t_mom - 1.0) / t_new) * (x_try - a)
        y = np.maximum(y, 0.0)
        a = x_try
        t_mom = t_new

        # Reset momentum periodically or when stuck
        if it % 2000 == 0 or (stuck > 3000 and it % 500 == 0):
            t_mom = 1.0

        # Perturbation when very stuck
        if stuck > 5000 and it % 300 == 0:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
            if noise_scale > 1e-9:
                noise = rng.normal(0, 1.0, n)
                F = fft(noise)
                cutoff = max(3, n // 200)
                F[cutoff:-cutoff] *= 0.05
                noise_smooth = np.real(ifft(F))[:n]
                noise_smooth = noise_smooth / (np.linalg.norm(noise_smooth) + 1e-15) * noise_scale
                y = project_simplex(y + noise_smooth, S)
                t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    obj_final = 2.0 * n * c_final.max() / (best_a.sum() ** 2)
    return best_a, obj_final


def run_mwu(a_init, max_time, rng, beta_start=5.0, beta_end=500.0):
    """
    Multiplicative Weight Updates. Naturally maintains non-negativity.
    Uses log-space updates to avoid underflow.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1.0, max_time * 50.0))
        beta = beta_start * (beta_end / beta_start) ** frac

        c = fft_convolve_self(a)
        cur_obj_val = 2.0 * n * c.max() / (S * S)
        if cur_obj_val < best_obj - 1e-12:
            best_obj = cur_obj_val
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        w = lse_weights(c, beta)
        grad = conv_gradient(a, w)

        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break

        # Adaptive learning rate
        eta = 1.0 / (gmax * (1.0 + 0.01 * it))
        eta = min(eta, 0.3 / max(a.max(), 1e-10))

        # Log-space MWU update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a -= eta * grad
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10 or not np.all(np.isfinite(a_new)):
            break
        a = a_new * (S / a_new_sum)

        # Perturbation when stuck
        if stuck > 400 and it % 200 == 0:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stuck / 3000.0)
            if noise_scale > 1e-9:
                noise = rng.normal(0, 1.0, n)
                F = fft(noise)
                cutoff = max(3, n // 100)
                F[cutoff:-cutoff] *= 0.1
                noise_smooth = np.real(ifft(F))[:n]
                noise_smooth = noise_smooth / (np.linalg.norm(noise_smooth) + 1e-15) * noise_scale
                a = project_simplex(a + noise_smooth, S)
                last_improve = it

    return best_a, best_obj


def run_frank_wolfe(a_init, max_time, rng, beta_start=5.0, beta_end=300.0):
    """Frank-Wolfe (Conditional Gradient) method."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1.0, max_time * 30.0))
        beta = beta_start * (beta_end / beta_start) ** frac

        c = fft_convolve_self(a)
        cur_obj_val = 2.0 * n * c.max() / (S * S)
        if cur_obj_val < best_obj - 1e-12:
            best_obj = cur_obj_val
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        w = lse_weights(c, beta)
        grad = conv_gradient(a, w)

        # FW: move toward vertex with min gradient
        j_min = np.argmin(grad)
        v = np.zeros(n)
        v[j_min] = S

        # Step size: standard FW is 2/(k+2), but use larger when stuck
        t = 2.0 / (it + 2.0)
        if stuck > 100:
            t = min(0.1, t + 0.02)

        a_new = (1 - t) * a + t * v
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum > 1e-10:
            a = a_new * (S / a_new_sum)

    return best_a, best_obj


def run_symmetric_fista(a_init, max_time, rng, beta_start=10.0, beta_end=500.0):
    """FISTA with symmetry constraint a_j = a_{n-1-j}."""
    n = len(a_init)
    half = n // 2
    S = np.sqrt(2.0 * n)

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    # Enforce initial symmetry
    a = 0.5 * (a + a[::-1])
    a = a * (S / a.sum())
    a_half = a[:half].copy()

    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    t_mom = 1.0
    a_prev_half = a_half.copy()

    def reconstruct(half_vals):
        if n % 2 == 0:
            full = np.concatenate([half_vals, half_vals[::-1]])
        else:
            full = np.concatenate([half_vals, half_vals[-2::-1]])
        return full

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1.0, max_time * 40.0))
        beta = beta_start * (beta_end / beta_start) ** frac
        step = 1.0 / max(beta, 1.0)

        a_full = reconstruct(a_half)
        a_full = a_full * (S / a_full.sum())
        c = fft_convolve_self(a_full)
        cur_obj_val = 2.0 * n * c.max() / (S * S)
        if cur_obj_val < best_obj - 1e-12:
            best_obj = cur_obj_val
            best_a = a_full.copy()
            last_improve = it

        stuck = it - last_improve
        w = lse_weights(c, beta)
        grad_full = conv_gradient(a_full, w)

        # Project gradient to symmetric half
        if n % 2 == 0:
            grad_half = 0.5 * (grad_full[:half] + grad_full[n-1:half-1:-1])
        else:
            grad_half = 0.5 * (grad_full[:half] + grad_full[n-1:half-1:-1])

        gnorm = np.linalg.norm(grad_half)
        if gnorm < 1e-15:
            break

        # Update in half-space
        a_half_new = np.maximum(a_half - step * grad_half, 0)
        # Normalize full sequence
        full_new = reconstruct(a_half_new)
        full_new_sum = full_new.sum()
        if full_new_sum < 1e-10:
            break
        scale = S / full_new_sum
        a_half = a_half_new * scale

        # Nesterov
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_half = a_half + ((t_mom - 1.0) / t_new) * (a_half - a_prev_half)
        y_half = np.maximum(y_half, 0)
        a_prev_half = a_half
        a_half = y_half
        t_mom = t_new

        if it % 2000 == 0:
            t_mom = 1.0

    return best_a, best_obj


def run_active_polish(a_init, max_time, rng):
    """Final polishing with exact active-set subgradient and adaptive top-k."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_obj_val = 2.0 * n * c.max() / (S * S)
        if cur_obj_val < best_obj - 1e-12:
            best_obj = cur_obj_val
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive active set size
        if stuck < 50:
            top_k = 1
            tol_frac = 0.0
        elif stuck < 200:
            top_k = 3
            tol_frac = 0.001
        elif stuck < 1000:
            top_k = min(16, 4 + stuck // 100)
            tol_frac = 0.003
        else:
            top_k = min(64, 16 + stuck // 100)
            tol_frac = 0.01

        max_val = c.max()
        if tol_frac > 0:
            active = np.where(c >= max_val * (1.0 - tol_frac))[0]
            if len(active) > top_k:
                active = np.argsort(c)[::-1][:top_k]
        else:
            active = np.array([int(np.argmax(c))])

        w_full = np.zeros(len(c))
        w_full[active] = 1.0 / len(active)
        grad = conv_gradient(y, w_full)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Careful step size
        step = S / (gnorm * (1.0 + 0.001 * it))
        step = min(step, 0.5 * S / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 2000 and it % 500 == 0:
            noise_scale = max(1e-7, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 5000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    return best_a, best_obj


def generate_frequential_init(n, rng, S):
    """Generate initial sequence with controlled frequency content."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Various cosine-based
    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            a = 1.0 - amp + amp * np.cos(2 * np.pi * freq * t)
            a = np.maximum(a, 0.01)
            a = a * (S / a.sum())
            inits.append(a.copy())

    # Smooth random in freq domain
    for _ in range(4):
        F = rng.normal(0, 1, n) + 1j * rng.normal(0, 1, n)
        # Emphasize low frequencies
        for k in range(1, n // 4):
            F[k] *= 0.8 ** k
            F[-k] = np.conj(F[k])
        a = np.real(ifft(F))[:n]
        a = np.maximum(a, 0.0)
        if a.sum() > 1e-10:
            a = a * (S / a.sum())
            inits.append(a.copy())

    return inits

        a_try, obj_try = run_fista_bt(a_try, min(remaining * 0.5, 80.0), rng,
                                       beta_start=3.0, beta_end=200.0)
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_try, obj_try = run_fista_bt(a_try, min(remaining * 0.5, 50.0), rng,
                                           beta_start=50.0, beta_end=2000.0,
                                           use_active=True, active_threshold=100.0)
            remaining = deadline - time.monotonic()
        if remaining > 10:
            a_try, obj_try = run_active_polish(a_try, min(remaining * 0.5, 30.0), rng)

        print("  n=16384: obj=%.8f (t=%.0fs)" % (obj_try, time.monotonic() - t0))
        if obj_try < global_best_obj:
            global_best_obj = obj_try
            global_best_a = a_try.copy()
            print("  *** New best from n=16384! ***")

    # === Phase 8: Final polish with best found (5%) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 8: Final polish (budget %.0fs) ===" % remaining)
        t0 = time.monotonic()
        a8, obj8 = run_active_polish(global_best_a, remaining * 0.7, rng)
        if obj8 < global_best_obj:
            global_best_obj = obj8
            global_best_a = a8.copy()
        print("  Final: obj=%.8f (t=%.0fs)" % (obj8, time.monotonic() - t0))

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
