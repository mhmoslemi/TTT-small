- When `p = i`: contributes `a_i * a_{j-i}` (if `0 <= j-i < n`)
- When `j-p = i`, i.e., `p = j-i`: contributes `a_{j-i} * a_i` (if `0 <= j-i < n`)

If `i ≠ j-i` (i.e., `j ≠ 2i`), these are two different terms in the sum, so the total contribution is `2 * a_i * a_{j-i}`.
If `i = j-i` (i.e., `j = 2i`), it's one term: `a_i^2`.

So:
- For `j = 2i`: `(a*a)_j = a_i^2 + C0_j` where `C0_j` doesn't involve `a_i`.
- For `j ≠ 2i` but `i` is in the valid range for lag `j`: `(a*a)_j = 2 * a_i * a_{j-i} + C0_j`.
- For other j: `(a*a)_j = C0_j` (constant).

So the max convex function in `a_i` is:
`max(j in active) [quadratic or linear in a_i]`

where active j are those where `a_i` appears.

This is a 1D problem! We can solve it exactly:
- Find the value of `a_i >= 0` that minimizes `max_j f_j(a_i)` where each `f_j` is either `a_i^2 + c0` or `2*b_i * a_i + c0` (linear in `a_i`).

The minimum of the max of linear and one quadratic function can be found by:
1. The quadratic `a_i^2 + c0` has minimum at `a_i = 0` with value `c0`.
2. Each linear `2*b_i * a_i + c0` has its own behavior.
3. The max of all these is a convex piecewise-linear-plus-quadratic function.
4. We can find the minimum by bisection or by solving the KKT conditions.

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Improved approach with:
- FISTA with exact subgradient (not LSE) for fast convergence
- Pure subgradient with decreasing step for precise convergence
- 1D coordinate descent polishing for final improvement
- Multi-scale upsample to exploit larger n
- Adaptive escaping when stuck

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def exact_subgradient(a, conv):
    """Exact subgradient from the argmax of convolution."""
    n = len(a)
    j = int(np.argmax(conv))
    # grad_i = 2 * conv_j's derivative w.r.t. a_i
    # (a*a)_j = sum_p a_p * a_{j-p}
    # d/d a_i = a_{j-i} + a_{j-i} = 2*a_{j-i} if i != j-i
    #          = 2*a_i if j = 2i
    # Wait: d/d a_i of sum_p a_p * a_{j-p}
    # The terms involving a_i are:
    #   p=i: a_i * a_{j-i}, derivative = a_{j-i}
    #   p=j-i: a_{j-i} * a_i, derivative = a_{j-i}
    # If j-i != i (i.e., j != 2i): total derivative = 2*a_{j-i}
    # If j-i == i (i.e., j == 2i): it's the same term, derivative = 2*a_i
    # Wait no. If j = 2i, then the term is a_i * a_i = a_i^2, and its derivative is 2*a_i.
    # If j != 2i, the two terms a_i*a_{j-i} and a_{j-i}*a_i are different (different p),
    # so the derivative is a_{j-i} + a_{j-i} = 2*a_{j-i}.
    # In both cases, the derivative is 2 * a[j-i] (since when j=2i, a[j-i] = a[i]).
    # But we need j-i to be a valid index: 0 <= j-i < n.
    
    grad = np.zeros(n)
    # j-i must be in [0, n-1], so i must be in [j-n+1, j] intersect [0, n-1]
    i_lo = max(0, j - n + 1)
    i_hi = min(n - 1, j)
    grad[i_lo:i_hi+1] = 2.0 * a[j - np.arange(i_lo, i_hi + 1)]
    return grad


def multi_active_subgradient(a, conv, k=5, tol=0.01):
    """Subgradient from top-k active lags."""
    n = len(a)
    max_val = conv.max()
    # Find lags within tolerance of max
    active = np.where(conv >= max_val * (1 - tol))[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    k = min(k, len(active))
    # Weight by how close to max
    weights = conv[active] / conv[active].sum()
    
    grad = np.zeros(n)
    for j in active[:k]:
        i_lo = max(0, j - n + 1)
        i_hi = min(n - 1, j)
        grad[i_lo:i_hi+1] += weights[active == j] * 2.0 * a[j - np.arange(i_lo, i_hi + 1)]
    return grad


def run_fista_subgrad(a_init, max_time, rng, use_multi=False, k=5, tol=0.01):
    """FISTA with exact/multi-active subgradient."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        conv_y = fft_convolve_self(y)
        cur_max = conv_y.max()
        
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Compute gradient
        if use_multi and stuck > 100:
            grad = multi_active_subgradient(y, conv_y, k=k, tol=tol)
        else:
            grad = exact_subgradient(y, conv_y)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        # Lipschitz constant bound: L <= 4 * max(a)
        # (since the Hessian of (a*a)_j has entries bounded by 2*max(a))
        L = 4.0 * np.max(a) + 1e-10
        step = 1.0 / L
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Restart when stuck
        if stuck > 3000:
            t_mom = 1.0
            noise_scale = 0.01 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                a = y.copy()
            last_improve = it
        
        # Periodic restart
        if it % 10000 == 0:
            t_mom = 1.0
    
    return best_a, best_conv


def run_subgrad_decreasing(a_init, max_time, rng):
    """Subgradient method with decreasing step size (no momentum)."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    
    # Step size: alpha_t = alpha_0 / sqrt(t)
    alpha_0 = S * 0.5
    
    while time.monotonic() - start < max_time:
        it += 1
        conv = fft_convolve_self(a)
        cur_max = conv.max()
        
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = a.copy()
        
        grad = exact_subgradient(a, conv)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        alpha = alpha_0 / np.sqrt(it)
        # Adaptive: if no improvement recently, use smaller step
        a_new = project_simplex(a - alpha * grad, S)
        
        # Backtracking check
        conv_new = fft_convolve_self(a_new)
        if conv_new.max() > cur_max * (1 + 0.1):
            # Step was too large, try smaller
            a_new = project_simplex(a - alpha * 0.1 * grad, S)
        
        a = a_new
    
    return best_a, best_conv

    best_conv = conv.max()
    best_a = a.copy()

        improved_this_pass = 0

            order = rng.permutation(n)
        else:
            order = np.arange(n)

                break

            #   - Unaffected conv values (constant): max_unaffected

            #   - For j in [j_lo, j_hi], j != 2i: (conv[j] - 2*a[i]*a[j-i]) + 2*t*a[j-i]
            
            # Max of unaffected
            if j_lo > 0 or j_hi < 2*n-2:
                mask = np.zeros(2*n-1, dtype=bool)
                mask[j_lo:j_hi+1] = True
                max_unaffected = conv[~mask].max() if np.any(~mask) else 0.0
            else:
                max_unaffected = 0.0
            
            # Current value
            ai = a[i]
            
            # Try candidate values
            candidates = [0.0, ai]
            # Gradient-based candidates
            # Subgradient at current point from active lags
            active_in_range = conv[j_lo:j_hi+1] >= cur_max * 0.999
            if np.any(active_in_range):
                active_j = np.where(active_in_range)[0] + j_lo
                # Compute subgradient contribution
                sg = 0.0
                for j in active_j:
                    other_idx = j - i
                    if 0 <= other_idx < n:
                        if j == 2 * i:
                            sg += 2.0 * ai
                        else:
                            sg += 2.0 * a[other_idx]
                if sg > 0:
                    step_1d = min(ai, 0.1 * cur_max / max(sg, 1e-15))
                    candidates.append(max(0.0, ai - step_1d))
                    candidates.append(max(0.0, ai - 0.5 * step_1d))
                elif sg < 0:
                    steps = [0.01 * S/n, 0.001 * S/n]
                    for s in steps:
                        candidates.append(ai + s)
            
            # Also try: the value that makes the quadratic equal to max_unaffected
            if 2 * i <= 2 * n - 2:
                c_quad = conv[2*i] - ai*ai
                if max_unaffected > c_quad:
                    t_eq = np.sqrt(max(0, max_unaffected - c_quad))
                    candidates.append(t_eq)
            
            # Evaluate candidates
            best_t = ai
            best_g = cur_max
            
            for t in candidates:
                t = max(0.0, min(t, 1000.0))
                # Compute g(t) = max of affected and unaffected
                g_val = max_unaffected
                # Quadratic term
                if 2 * i <= 2 * n - 2:
                    g_val = max(g_val, (conv[2*i] - ai*ai) + t*t)
                # Linear terms
                for j in range(j_lo, j_hi + 1):
                    if j == 2 * i:
                        continue
                    other_idx = j - i
                    if 0 <= other_idx < n:
                        val = (conv[j] - 2*ai*a[other_idx]) + 2*t*a[other_idx]
                        g_val = max(g_val, val)
                
                if g_val < best_g:
                    best_g = g_val
                    best_t = t
            
            # Update if improved
            if best_t != ai and best_g < cur_max - 1e-15:
                # Update conv efficiently
                ai_old = a[i]
                a[i] = best_t
                # Update affected conv values
                delta = best_t - ai_old
                for j in range(j_lo, j_hi + 1):
                    other_idx = j - i
                    if 0 <= other_idx < n:
                        if j == 2 * i:
                            conv[j] += (best_t*best_t - ai_old*ai_old)
                        else:
                            conv[j] += 2*delta*a[other_idx]
                cur_max_new = conv.max()
                if cur_max_new < best_conv - 1e-15:
                    best_conv = cur_max_new
                    best_a = a.copy()
                improved_this_pass += 1
            
            # Renormalize periodically (to maintain sum = S)
            # Actually, changing one coordinate changes the sum.
            # We should renormalize, but that changes all coordinates.
            # For now, skip renormalization within a pass and do it at the end.
        
        # Renormalize at end of pass
        a = normalize_seq(a, S)
        conv = fft_convolve_self(a)
        cur_max = conv.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = a.copy()
        
        # Check time
        if time.monotonic() - start > max_time - 5:
            break
    
    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.54 + 0.46 * np.cos(2 * np.pi * tt)
