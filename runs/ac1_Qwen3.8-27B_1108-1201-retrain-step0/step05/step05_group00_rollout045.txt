def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_grad(a, N, beta):
    """Gradient of LSE-smoothed peak of conv(a,a)."""
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    if beta < 1e-6:
        weights = np.ones(len(c)) / len(c)
        lse_val = c_max
    else:
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:

    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max

def exp_grad_step(a, grad, N, S, eta):
    """Exponentiated gradient (multiplicative) update step."""
    n = len(a)
    # a_i <- a_i * exp(-eta * grad_i), then renormalize
    # Use log-domain for stability
    log_a = np.where(a > 1e-30, np.log(a), -20.0)
    log_a_new = log_a - eta * grad
    # Renormalize: sum(exp(log_a_new)) = S
    # Use log-sum-exp for stability
    m = log_a_new.max()
    exp_shifted = np.exp(log_a_new - m)
    s = exp_shifted.sum()
    if s < 1e-300:
        return a.copy()
    a_new = exp_shifted * (S / s)
    a_new = np.clip(a_new, 0.0, 1000.0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new * (S / sn)


def lbfgs_dir(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i

    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def spectral_restart(a, N, rng, frac=1.0):
    """Randomize a fraction of FFT phases."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def peak_bias_restart(a, N, rng, c, strength=0.3):
    """Restart biased towards reducing the current peak."""
    n = len(a)
    S = a.sum()
    k_star = int(np.argmax(c))
    # Compute gradient of c(k_star) w.r.t. a
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    grad_peak = np.zeros(n)
    for i in range(lo, hi + 1):
        j = k_star - i
        grad_peak[i] += a[j]
    # Perturb in direction that reduces the peak
    perturbation = strength * grad_peak / max(1.0, np.linalg.norm(grad_peak))
    a_perturbed = a - perturbation
    a_perturbed = np.maximum(a_perturbed, 0.0)
    # Also add spectral randomization
    A = fft(a_perturbed, N)
    mag = np.abs(A)
    phase = np.angle(A)
    n_rand = max(1, int(N * 0.3))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def safe_normalize(a, n, S):
    """Clamp and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_lbfgs_block(a_init, n, S, N, max_time, rng, beta_init=5.0, beta_max=1000.0,
                    m_lbfgs=12):
    """L-BFGS on LSE with adaptive beta and line search."""
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []

    g, f_val, c, c_max = lse_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()

    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        if it % 100 == 0:
            beta = min(beta_max, beta * 1.15)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            g, f_val, c, c_max = lse_grad(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

        if it % 50 == 0 and len(s_hist) >= m_lbfgs:
            s_hist = s_hist[-m_lbfgs:]
            y_hist = y_hist[-m_lbfgs:]

        direction = lbfgs_dir(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        best_ls = c_max
        for ls in range(15):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls, _ = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls < best_ls:
                best_ls = cm_ls
                a_new = candidate
                cmax_new = cm_ls
            if cm_ls <= c_max * 1.01 or ls == 14:
                break
            alpha_ls *= 0.5

        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()

        g_new, f_new, c_nw, cmax_nw = lse_grad(a_new, N, beta)

        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * ss and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = a_new
        g = g_new
        c_max = cmax_nw

        if it % 300 == 0 and rng.random() < 0.3:
            a = spectral_restart(a, N, rng, frac=0.25)
            s_hist = []
            y_hist = []
            g, f_val, c, c_max = lse_grad(a, N, beta)

    return best_a, best_cmax


def exp_grad_block(a_init, n, S, N, max_time, rng):
    """Exponentiated gradient method with adaptive step size."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    c, A = compute_conv(a, N)
    c_max = c.max()
    best_cmax = c_max
    eta = 1.0 / max(1.0, c_max * 0.1)
    start_time = time.monotonic()
    it = 0
    patience = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        if it % 50 == 0:
            g, f_val, c, c_max = lse_grad(a, N, 20.0)
            if c_max < best_cmax - 1e-10:
                best_cmax = c_max
                best_a = a.copy()
                patience = 0
                eta = min(eta * 1.2, S / (n * 0.1))
            else:
                patience += 1
                if patience > 5:
                    eta *= 0.85
                    patience = 0
                    if rng.random() < 0.3:
                        a = spectral_restart(a, N, rng, frac=0.2)
        else:
            # Use power-sum gradient for more peak-focused updates
            g, c, c_max = power_sum_grad(a, N, p=20)
            if c_max < best_cmax - 1e-10:
                best_cmax = c_max
                best_a = a.copy()
                patience = 0

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-16:
            break

        a = exp_grad_step(a, g / gnorm, N, S, eta)

        if it % 200 == 0:
            c_chk, _ = compute_conv(a, N)
            if c_chk.max() < best_cmax:
                best_cmax = c_chk.max()
                best_a = a.copy()

    return best_a, best_cmax


def nesterov_toppk_block(a_init, n, S, N, max_time, rng, top_k_init=5, top_k_max=200):
    """Nesterov top-K subgradient with adaptive K for polishing."""
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0
    mu = 0.9  # momentum

    while time.monotonic() - start_time < max_time:
        it += 1
        top_k = min(top_k_max, max(1, top_k_init + it // 100))
        c, A = compute_conv(y, N)
        cur = c.max()
        if cur < best_cmax:
            best_cmax = cur
            best_a = y.copy()

        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        fsize = 1 << (3 * n - 2).bit_length()
        w_full = np.zeros(fsize)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]
        W = fft(w_full, fsize)
        Af = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        step = 3.0 * S / (n * (1.0 + 0.005 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 100 == 0:
            c_x, _ = compute_conv(x_new, N)
            if c_x.max() < best_cmax:
                best_cmax = c_x.max()
                best_a = x_new.copy()

        if it % 500 == 0 and rng.random() < 0.1:
            c_y, _ = compute_conv(y, N)
            y = peak_bias_restart(y, N, rng, c_y, strength=0.2)
            t = 1.0

    return best_a, best_cmax


def anneal_local_search(a_init, n, S, N, max_time, rng):
    """Simulated annealing on the peak value."""
    a = safe_normalize(a_init, n, S)
    c, _ = compute_conv(a, N)
    c_max = c.max()
    best_a = a.copy()
    best_cmax = c_max
    T = c_max * 0.1
    T_min = c_max * 0.001
    start_time = time.monotonic()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        # Random perturbation
        noise = rng.standard_normal(n) * (S / n) * 0.05
        a_new = project_simplex(a + noise, S)
        c_new, _ = compute_conv(a_new, N)
        c_max_new = c_new.max()

        # Acceptance criterion
        delta = c_max_new - c_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            c_max = c_max_new
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

        # Cool down
        if it % 10 == 0:
            T = max(T_min, T * 0.97)

        # Periodic restart from best
        if it % 200 == 0:
            a = best_a + rng.standard_normal(n) * (S / n) * 0.02
            a = project_simplex(a, S)
            c, _ = compute_conv(a, N)
            c_max = c.max()

    return best_a, best_cmax


def make_init_hann(n, rng):
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    return a + 0.01 * rng.standard_normal(n)


def make_init_blackman(n, rng):
    t = np.arange(n) / max(1, n - 1)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    return np.clip(a, 0.0, None)


def make_init_gauss(n, rng):
    t = np.linspace(-3, 3, n)
    a = np.exp(-t**2)
    return a + 0.01 * rng.standard_normal(n)


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_multi_freq(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for freq in [3, 7, 13, 21, 47]:
        a += 0.12 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    i = np.arange(n)
    c = rng.integers(1, n)
    phase = -np.pi * i * (2 * i + 1 - c) / n
    a = 1.0 + 0.7 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_fourier(n, rng, N):
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_smooth_taper(n, rng):
    """Smooth taper with random interior modulation."""
    t = np.linspace(0, 1, n)
    base = 4.0 * t * (1 - t)  # parabola, peak=1 at center
    mod = 1.0 + 0.3 * np.sin(2 * np.pi * 7 * t + rng.random() * 2 * np.pi)
    a = base * mod
    return np.clip(a, 0.0, None)


def optimize_for_n(n, S, N, time_budget, rng, given_seq=None):
    """Full optimization pipeline for a given sequence length."""
    start = time.monotonic()
    best_a = None
    best_cmax = np.inf

    def update(a_cand, label=""):
        nonlocal best_a, best_cmax
        a_c = safe_normalize(a_cand, n, S)
        c, _ = compute_conv(a_c, N)
        sc = c.max()
        if sc < best_cmax:
            best_cmax = sc
            best_a = a_c.copy()
            score = 2.0 * n * sc / (S * S)
            if label:
                print("    [%s n=%d] peak=%.6f score=%.6f" % (label, n, sc, score))
        return sc

    elapsed = time.monotonic() - start

    # Phase 1: Multi-start (20% of budget)
    phase1_time = time_budget * 0.20 - elapsed
    if phase1_time < 5:
        return best_a, best_cmax
    p1_start = time.monotonic()

    inits = []
    if given_seq is not None and len(given_seq) == n:
        inits.append(("given", np.clip(given_seq, 0, 1000).astype(np.float64)))
    inits.append(("rand", make_init_rand(n, rng)))
    inits.append(("hann", make_init_hann(n, rng)))
    inits.append(("blackman", make_init_blackman(n, rng)))
    inits.append(("gauss", make_init_gauss(n, rng)))
    inits.append(("chirp", make_init_chirp(n, rng)))
    inits.append(("multifreq", make_init_multi_freq(n, rng)))
    inits.append(("zadoff", make_init_zadoff(n, rng)))
    inits.append(("fourier", make_init_fourier(n, rng, N)))
    inits.append(("taper", make_init_smooth_taper(n, rng)))

    per_init = (p1_start + phase1_time - time.monotonic()) / max(1, len(inits))
    per_init = max(2.0, min(per_init, 30.0))

    for name, arr in inits:
        if time.monotonic() > p1_start + phase1_time:
            break
        a_norm = safe_normalize(arr, n, S)
        t0 = time.monotonic()
        a_l, sc_l = lse_lbfgs_block(a_norm, n, S, N, min(per_init, 25.0), rng,
                                     beta_init=5.0, beta_max=100.0, m_lbfgs=8)
        update(a_l, name + "/LSE")

    if best_a is None:
        best_a = np.ones(n) * S / n
        best_cmax = n  # peak of all-ones

    # Phase 2: Focused LSE-LBFGS (30% of budget)
    phase2_time = time_budget * 0.30
    p2_start = time.monotonic()
    a_curr = best_a.copy()
    stall = 0

    while time.monotonic() - p2_start < phase2_time:
        block_time = min(200.0, p2_start + phase2_time - time.monotonic())
        if block_time < 10:
            break
        beta_s = 8.0 + 4.0 * min(stall, 8)
        a_new, sc_new = lse_lbfgs_block(a_curr, n, S, N, block_time, rng,
                                         beta_init=beta_s, beta_max=1200.0, m_lbfgs=12)
        if sc_new < best_cmax - 1e-9:
            best_cmax = sc_new
            best_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall == 2:
                c_cur, _ = compute_conv(a_curr, N)
                a_curr = peak_bias_restart(a_curr, N, rng, c_cur, strength=0.25)
                stall = 0
            elif stall >= 4:
                a_curr = spectral_restart(a_curr, N, rng, frac=0.5)
                stall = 0

    update(best_a, "P2_done")

    # Phase 3: Nesterov top-K (25% of budget)
    phase3_time = time_budget * 0.25
    p3_start = time.monotonic()
    a_ng, sc_ng = nesterov_toppk_block(best_a, n, S, N, phase3_time, rng,
                                        top_k_init=3, top_k_max=150)
    update(a_ng, "nesterov")

    # Phase 4: Exponentiated gradient (10% of budget)
    phase4_time = time_budget * 0.10
    p4_start = time.monotonic()
    a_eg, sc_eg = exp_grad_block(best_a, n, S, N, phase4_time, rng)
    update(a_eg, "exp_grad")

    # Phase 5: Final high-beta LSE (10% of budget)
    remaining5 = max(5.0, time_budget - (time.monotonic() - start) - 10)
    if remaining5 > 10:
        a_fin, sc_fin = lse_lbfgs_block(best_a, n, S, N, remaining5, rng,
                                         beta_init=50.0, beta_max=2000.0, m_lbfgs=15)
        update(a_fin, "final_LSE")

    # Phase 6: Annealing local search (remaining time)
    remaining6 = max(5.0, time_budget - (time.monotonic() - start) - 5)
    if remaining6 > 10:
        a_an, sc_an = anneal_local_search(best_a, n, S, N, remaining6, rng)
        update(a_an, "anneal")

    return best_a, best_cmax


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_b/sum_a^2.
    Uses multi-length exploration with LSE-LBFGS, Nesterov top-K,
    exponentiated gradient, peak-biased restarts, and annealing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    print("Total budget: %.0f s" % budget_s)

    # Try to use given sequence as initial
    given = None
    try:
        g = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if g.size >= 512:
            given = g
            print("Using given sequence of length %d" % g.size)
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def global_update(a_cand, n_cand):
        nonlocal global_best, global_a
        if a_cand is None:
            return
        a_c = safe_normalize(a_cand, len(a_cand), np.sqrt(2.0 * len(a_cand)))
        n_c = len(a_c)
        N_c = 1 << (2 * n_c - 1).bit_length()
        c, _ = compute_conv(a_c, N_c)
        sc = c.max()
        score = 2.0 * n_c * sc / (float(n_c) * np.sqrt(2.0 * n_c) ** 2)
        if score < global_best:
            global_best = score
            global_a = a_c.copy()
            print("  ** GLOBAL BEST: n=%d score=%.6f" % (n_c, score))

    # Decide which n values to try
    # Prioritize n=8192 (proven to work well) and try n=16384 if time allows
    n_options = [8192]
    if given is not None:
        for candidate_n in [4096, 8192, 16384]:
            if abs(len(given) - candidate_n) / candidate_n < 0.1:
                if candidate_n not in n_options:
                    n_options.insert(0, candidate_n)

    # Time allocation: give more time to n=8192, less to others
    time_alloc = {}
    total_n = sum(n_options)
    for n in n_options:
        if n == 8192:
            time_alloc[n] = 0.6
        elif n == 16384:
            time_alloc[n] = 0.25
        else:
            time_alloc[n] = 0.15
    # Normalize
    total_alloc = sum(time_alloc.values())
    time_alloc = {k: v / total_alloc * (budget_s - 60) for k, v in time_alloc.items()}

    for n in n_options:
        if time.monotonic() > deadline - 40:
            break
        S = np.sqrt(2.0 * float(n))
        N = 1 << (2 * n - 1).bit_length()
        print("\n=== Optimizing n=%d, S=%.4f, N=%d, time=%.0fs ===" % (n, S, N, time_alloc[n]))

        a_best, cmax_best = optimize_for_n(n, S, N, time_alloc[n], rng, given_seq=given)
        if a_best is not None:
            global_update(a_best, n)

    # Final: one more focused push on the best n
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        N_best = 1 << (2 * n_best - 1).bit_length()
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            print("\n=== Final push on n=%d, remaining=%.0fs ===" % (n_best, remaining))
            a_fin, sc_fin = lse_lbfgs_block(global_a, n_best, S_best, N_best,
                                             min(remaining, 120.0), rng,
                                             beta_init=80.0, beta_max=2000.0, m_lbfgs=15)
            global_update(a_fin, n_best)
            remaining2 = deadline - time.monotonic() - 5
            if remaining2 > 20:
                a_ng, sc_ng = nesterov_toppk_block(global_a, n_best, S_best, N_best,
                                                    min(remaining2, 80.0), rng,
                                                    top_k_init=2, top_k_max=50)
                global_update(a_ng, n_best)

    # Verification and return
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nVerification score: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("\nVerification failed, using fallback")
            global_a = np.ones(256, dtype=np.float64)

    else:

    print("\nReturning length %d, best internal score=%.6f" % (len(global_a), global_best))
