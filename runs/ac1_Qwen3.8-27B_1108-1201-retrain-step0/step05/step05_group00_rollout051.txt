from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a, N):

    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

    active_set = list(np.argsort(c)[::-1][:max_active])

        # Check for violations
        fg = np.convolve(f, g)[:2*n-1]
        violations = np.where(fg > M + 1e-10)[0]
        if len(violations) == 0:
            break
        # Add the most violated
        new_idx = violations[np.argmax(fg[violations])]
        if new_idx not in active_set:
            active_set.append(int(new_idx))
            if len(active_set) > 2 * max_active:

                fg_vals = fg[active_set]
                sorted_idx = np.argsort(fg_vals)[::-1][:max_active]
                active_set = [active_set[i] for i in sorted_idx]
        else:
            break
    
    return g, M, c

    f = np.clip(f, 0, 1000)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) * S / n
    else:
        f = f * (S / s)
    
    best_f = f.copy()
    best_score = np.inf
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        g, M, c = fw_iteration(f, n, S, N, max_active=20, max_lp_iters=5)
        
        # Rescale g to have sum S
        gs = g.sum()
        if gs < 1e-15:
            g = f.copy()
        else:
            g = g * (S / gs)
        
        # Frank-Wolfe step size
        t_fw = 2.0 / (it + 2)
        
        f_new = (1 - t_fw) * f + t_fw * g
        
        # Project
        f_new = project_simplex(f_new, S)
        
        c_new = compute_conv(f_new, N)
        score_new = 2 * n * c_new.max() / (S * S)
        
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
        
        f = f_new
    
    return best_f, best_score

Minimize 2*n*max_conv/sum^2 using Frank-Wolfe with LP oracle + Nesterov polishing.

Key innovations:
1. Frank-Wolfe with LP oracle (constraint generation): directly optimizes the peak
   without LSE bias, using a separation oracle for the linearized constraint.
2. Peak surgery: targeted mass redistribution for the top autocorrelation peaks.
3. Adaptive step sizes with line search.
4. Multi-scale approach: fast local optimization + slow global exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns (2n-1) values."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def fp_cross(f, g, N):
    """Cross-correlation (f*g) via FFT. Returns (2n-1) values."""
    n = len(f)
    f_pad = np.zeros(N)
    f_pad[:n] = f
    g_pad = np.zeros(N)
    g_pad[:n] = g
    F = fft(f_pad)
    G = fft(g_pad)
    c = np.real(ifft(F * G))[:2 * n - 1]
    return c

    Returns g or None if infeasible.

    m = len(active_lags)
    # Build constraint matrix for active lags
    # (f*g)_k = sum_j f[k-j] * g[j] for valid j (0 <= k-j < n, 0 <= j < n)
    A_rows = []
    for k in active_lags:
        row = np.zeros(n)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            i = k - j
            if 0 <= i < n:
                row[j] = f[i]
        A_rows.append(row)
    
    A_mat = np.array(A_rows) if m > 0 else np.zeros((0, n))
    b_mat = np.full(m, M)
    
    # LP: min -sum(g) s.t. A_mat @ g <= b_mat, -g <= 0
    c_obj = -np.ones(n)
    if m > 0:
        A_ub = np.vstack([A_mat, -np.eye(n)])
        b_ub = np.concatenate([b_mat, np.zeros(n)])
    else:
        A_ub = -np.eye(n)
        b_ub = np.zeros(n)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': 30})
    
    if res.success:
        return res.x
    elif res.status == 9:  # termination requested (time limit)
        # Return current iterate if available
        if hasattr(res, 'x') and res.x is not None:
            return res.x
        return None
    else:
        return None


def fw_with_lp(f_init, n, S, N, max_time, rng, max_active=30, max_subiter=8):
    """
    Frank-Wolfe method with LP oracle.
    """
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = np.clip(f, 0, 1000)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) * S / n
    else:
        f = f * (S / s)
    
    best_f = f.copy()
    best_peak = np.inf
    start = time.monotonic()
    it = 0
    stall_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Compute current convolution
        c = compute_conv(f, N)
        cur_peak = c.max()
        if cur_peak < best_peak:
            best_peak = cur_peak
            best_f = f.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Determine active set: top peaks
        top_k = min(max_active, 5 + it // 10)
        active_lags = list(np.argsort(c)[::-1][:top_k])
        
        # Solve LP with constraint generation
        M = cur_peak
        g = None
        for sub in range(max_subiter):
            g = solve_fw_lp(f, M, active_lags, n)
            if g is None:
                break
            # Check for new violations
            fg = fp_cross(f, g, N)
            violations = fg > M + 1e-10
            if not np.any(violations):
                break
            # Add the worst violation
            viol_idx = np.where(violations)[0]
            worst = viol_idx[np.argmax(fg[viol_idx])]
            if int(worst) not in active_lags:
                active_lags.append(int(worst))
                if len(active_lags) > 2 * max_active:
                    # Keep only the worst ones
                    fg_active = fg[active_lags]
                    keep = np.argsort(fg_active)[::-1][:max_active]
                    active_lags = [active_lags[i] for i in keep]
            else:
                break
        
        if g is None:
            # LP failed, use subgradient step instead
            d = np.zeros(n)
            for k in active_lags[:5]:
                d += 2.0 * np.roll(f, k)  # This is wrong, let me fix
            # Actually, the subgradient of (f*f)_k at f is 2 * shift_k(f)
            # where shift_k(f)_i = f[k-i] (with out-of-range = 0)
            d = np.zeros(n)
            for k in active_lags[:5]:
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        d[i] += 2.0 * f[j]
            g = project_simplex(f - f.sum() / (n * cur_peak) * d, S)
        
        # Rescale g
        gs = g.sum()
        if gs < 1e-15:
            g = f.copy()
        else:
            g = g * (S / gs)
        
        # Line search for step size
        best_t = 2.0 / (it + 2)
        best_val = cur_peak
        for t_try in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_try = (1 - t_try) * f + t_try * g
            f_try = project_simplex(f_try, S)
            c_try = compute_conv(f_try, N)
            if c_try.max() < best_val:
                best_val = c_try.max()
                best_t = t_try
        
        f_new = (1 - best_t) * f + best_t * g
        f = project_simplex(f_new, S)
        
        # Periodic restart
        if stall_count > 20 and rng.random() < 0.3:
            f = spectral_partial_restart(f, N, rng, frac=0.3)
            stall_count = 0
    
    return best_f, best_peak


def spectral_partial_restart(a, N, rng, frac=0.3):
    """Randomize a fraction of the FFT phases."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = old_phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)
