"""
Search for non-negative step-function heights minimizing evaluate_sequence.

The method combines FFT-based smooth conv-peak optimization with several
complementary mechanisms:
  - LSE-smoothed L-BFGS on the projection simplex,
  - multiplicative / natural-gradient exponential-fitting blocks,
  - top-k subgradient/Nesterov polishing,
  - cosine and structured initializations,
  - symmetrization as a candidate transformation,
  - spectral phase restarts,
  - multi-length search and final direct evaluation when available.

Only top-level helper functions are used. No lambda functions are used.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fft_len(n):
    return next_pow2(2 * int(n) - 1)


def normalize_shape(a, n, S):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size < n:
        tmp = np.zeros(n, dtype=np.float64)
        tmp[:a.size] = a
        a = tmp
    elif a.size > n:
        a = a[:n]
    a = np.clip(a, 0.0, 1000.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.full(n, S / float(n), dtype=np.float64)
    return a * (S / s)


def symmetrize_full(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    return 0.5 * (a + a[::-1])


def project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    n = v.size
    if n == 0:
        return v
    if S <= 0.0:
        return np.zeros(n, dtype=np.float64)
    v = np.nan_to_num(v, nan=0.0, posinf=1e100, neginf=0.0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w + 1e-12
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0]) + 1)
    else:
        rho = n
    theta = (cumsum[rho - 1] - S) / float(rho)
    return np.maximum(v - theta, 0.0)


def project_symmetric_simplex(u, n, S):
    u = np.asarray(u, dtype=np.float64).reshape(-1)
    if u.size < n:
        tmp = np.zeros(n, dtype=np.float64)
        tmp[:u.size] = u
        u = tmp
    elif u.size > n:
        u = u[:n]
    u = np.nan_to_num(u, nan=0.0, posinf=1e100, neginf=0.0)
    if S <= 0.0:
        return np.zeros(n, dtype=np.float64)

    m = (n + 1) // 2
    v = 0.5 * (u[:m] + u[n - 1::-1][:m])
    weights = 2.0 * np.ones(m, dtype=np.float64)
    if n % 2 == 1:
        weights[n // 2] = 1.0

    order = np.argsort(v)[::-1]
    vs = v[order]
    ws = weights[order]
    cumW = np.cumsum(ws)
    cumWV = np.cumsum(vs * ws)
    theta_cand = (cumWV - S) / cumW

    ok = theta_cand <= vs + 1e-12
    if m > 1:
        next_vals = vs[1:]
        ok[:m - 1] = ok[:m - 1] & (theta_cand[:m - 1] >= next_vals - 1e-12)

    if np.any(ok):
        rho = int(np.max(np.where(ok)[0]) + 1)
    else:
        rho = m
    theta = theta_cand[rho - 1]

    x = np.maximum(v - theta, 0.0)
    s = float(np.dot(weights, x))
    if s <= 0.0:
        return np.full(n, S / float(n), dtype=np.float64)
    x *= S / s

    a = np.empty(n, dtype=np.float64)
    a[:m] = x
    a[n - 1::-1] = x
    s2 = float(a.sum())
    if s2 > 0.0:
        a *= S / s2
    else:
        a = np.full(n, S / float(n), dtype=np.float64)
    return a


def compute_conv(a, N):
    n = int(a.size)
    ap = np.zeros(N, dtype=np.float64)
    ap[:n] = a
    A = fft(ap)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c = np.maximum(c, 0.0)
    c = np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)
    return c, A


def score_conv(a, n, S, N=None):
    if N is None:
        N = fft_len(n)
    c, _ = compute_conv(a, N)
    peak = float(c.max())
    return float(2.0 * n * peak / (S * S))


def better_with_sym(a, n, S, N=None):
    a = normalize_shape(a, n, S)
    sc = score_conv(a, n, S, N)
    b = project_symmetric_simplex(symmetrize_full(a), n, S)
    scb = score_conv(b, n, S, N)
    if scb < sc:
        return b, scb
    return a, sc


def lse_weights(c, beta):
    cmax = float(c.max())
    if beta <= 1e-8:
        weights = np.full_like(c, 1.0 / float(c.size), dtype=np.float64)
        lse = float(c.mean())
        return weights, lse, cmax
    shifted = beta * (c - cmax)
    shifted = np.clip(shifted, -745.0, 0.0)
    exps = np.exp(shifted)
    total = float(exps.sum())
    if total <= 0.0:
        idx = int(np.argmax(c))
        weights = np.zeros_like(c, dtype=np.float64)
        weights[idx] = 1.0
        return weights, cmax, cmax
    weights = exps / total
    lse = cmax + np.log(total) / beta
    return weights, float(lse), cmax


def grad_from_weights(w, A, N, n):
    W = fft(w, N)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
    return g


def lbfgs_dir(s_hist, y_hist, g):
    m = len(s_hist)
    n = len(g)
    gn = float(np.linalg.norm(g))
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n, dtype=np.float64)
        return -g / gn

    q = g.copy()
    alphas = np.zeros(m, dtype=np.float64)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = float(np.dot(y_i, y_i))
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = float(np.dot(s_i, q)) / yy
            q = q - alphas[i] * y_i

    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = float(np.dot(y_last, y_last))
    if yy_last > 1e-30:
        gamma_use = max(1e-8, float(np.dot(s_last, y_last)) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q

    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = float(np.dot(y_i, y_i))
        if yy < 1e-30:
            continue
        beta_i = float(np.dot(s_i, y_i)) / yy
        r = r + (alphas[i] - beta_i) * y_i

    direction = -r
    if float(np.dot(g, direction)) > 0.0:
        direction = -g
    return direction


def spectral_restart(a, N, rng, frac, use_sym):
    n = int(a.size)
    S = float(a.sum())
    A = fft(a, N)
    mag = np.abs(A)
    ph = np.angle(A)

    nrand = max(1, int(N * frac))
    if nrand >= N:
        ph_new = rng.uniform(0.0, 2.0 * np.pi, N)
    else:
        idx = rng.choice(N, size=nrand, replace=False)
        ph_new = ph.copy()
        ph_new[idx] = rng.uniform(0.0, 2.0 * np.pi, nrand)

    A_new = mag * np.exp(1j * ph_new)
    x = np.real(ifft(A_new))[:n]
    x = x - float(x.min())
    x = np.maximum(x, 0.0)
    if float(x.sum()) <= 1e-15:
        x = np.abs(x)

    x = normalize_shape(x, n, S)
    if use_sym:
        x = project_symmetric_simplex(symmetrize_full(x), n, S)
    else:
        x = project_simplex(x, S)
    return x


def cosine_search_init(n, S, N):
    best_a = np.full(n, S / float(n), dtype=np.float64)
    best_score = score_conv(best_a, n, S, N)

    t = np.linspace(0.0, 1.0, n, endpoint=False)
    freqs = (0.5, 1.0, 2.0, 3.0, 5.0, 8.0, 13.0, 21.0, 34.0)
    rss = (0.0, 0.35, 0.65, 0.85, 0.95, 0.99)
    phis = (0.0, np.pi / 4.0, np.pi / 2.0, 3.0 * np.pi / 4.0, np.pi)

    for f in freqs:
        base = 2.0 * np.pi * f * t
        for r in rss:
            for ph in phis:
                arr = 1.0 + r * np.sin(base + ph)
                mn = float(arr.min())
                if mn < 0.0:
                    if mn > -1e-10:
                        arr = np.maximum(arr, 0.0)
                    else:
                        continue
                arr = normalize_shape(arr, n, S)
                sc = score_conv(arr, n, S, N)
                if sc < best_score:
                    best_score = sc
                    best_a = arr.copy()

    i = np.arange(n, dtype=np.float64)
    for p in (0.25, 0.5, 1.0, 2.0, 4.0):
        arr = np.power(np.sin(np.pi * (i + 0.5) / float(n)), p) + 1e-8
        arr = normalize_shape(arr, n, S)
        sc = score_conv(arr, n, S, N)
        if sc < best_score:
            best_score = sc
            best_a = arr.copy()

    return best_a, float(best_score)


def make_initializations(n, S, N, rng, parent):
    inits = []

    if parent is not None and parent.size == n:
        inits.append(normalize_shape(parent, n, S))
        inits.append(normalize_shape(symmetrize_full(parent), n, S))

    inits.append(np.full(n, S / float(n), dtype=np.float64))
    inits.append(normalize_shape(rng.random(n) + 0.05, n, S))

    t = np.linspace(0.0, 1.0, n, endpoint=False)

    chirp = 0.5 + 0.4 * np.sin(2.0 * np.pi * (5.0 * t + t * t)) + 0.1 * rng.standard_normal(n)
    inits.append(normalize_shape(chirp, n, S))

    if n > 1:
        c = int(rng.integers(1, n))
    else:
        c = 1
    i = np.arange(n, dtype=np.float64)
    phase = -np.pi * i * (2.0 * i + 1.0 - float(c)) / float(n)
    zad = 1.0 + 0.7 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    inits.append(normalize_shape(zad, n, S))

    mf = 1.0
    for f in (2, 5, 9, 17, 29, 47):
        mf = mf + 0.12 * np.sin(2.0 * np.pi * f * t + rng.uniform(0.0, 2.0 * np.pi))
    inits.append(normalize_shape(mf, n, S))

    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    ph = rng.uniform(0.0, 2.0 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * ph)
    x = np.abs(np.real(ifft(A)))[:n] + 0.05
    inits.append(normalize_shape(x, n, S))

    b = (rng.random(n) > 0.4).astype(np.float64) * (0.5 + 0.5 * rng.random(n))
    inits.append(normalize_shape(b, n, S))

    for p in (0.5, 1.0, 2.0):
        arr = np.power(np.sin(np.pi * (i + 0.5) / float(n)), p) + 1e-8
        inits.append(normalize_shape(arr, n, S))

    raw = inits[:]
    for arr in raw[:6]:
        inits.append(normalize_shape(symmetrize_full(arr), n, S))

    return inits


def ef_block(a, n, S, N, budget, rng, beta_start=1.0, beta_max=300.0, use_symmetric=False):
    a = normalize_shape(a, n, S)
    if use_symmetric:
        a = project_symmetric_simplex(symmetrize_full(a), n, S)

    c, A = compute_conv(a, N)
    best_peak = float(c.max())
    best_score = float(2.0 * n * best_peak / (S * S))
    best_a = a.copy()

    beta = float(beta_start)
    vel = np.zeros(n, dtype=np.float64)
    start = time.monotonic()
    it = 0
    no_prog = 0

    while time.monotonic() - start < budget:
        it += 1
        w_cur, lse_cur, cmax_now = lse_weights(c, beta)
        cmax = float(c.max())

        if cmax < best_peak - 1e-12:
            best_peak = cmax
            best_score = float(2.0 * n * cmax / (S * S))
            best_a = a.copy()
            no_prog = 0
        else:
            no_prog += 1

        if use_symmetric:
            w_cur = 0.5 * (w_cur + w_cur[::-1])
        g = grad_from_weights(w_cur, A, N, n)
        if use_symmetric:
            g = 0.5 * (g + g[::-1])

        gmax = float(g.max())
        if gmax <= 1e-15:
            break
        g_norm = g / gmax

        gamma = 0.5 + 0.4 * min(1.0, it / 200.0)
        vel = gamma * vel + (1.0 - gamma) * g_norm
        vmax = float(vel.max())
        if vmax <= 1e-15:
            break
        v = vel / vmax

        accepted = False
        trial_best = None
        for eta in (0.8, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
            factor = np.exp(-eta * v)
            cand = a * factor
            if use_symmetric:
                cand = project_symmetric_simplex(cand, n, S)
            else:
                cand = project_simplex(cand, S)

            c_t, A_t = compute_conv(cand, N)
            p_t = float(c_t.max())
            w_t, lse_t, _ = lse_weights(c_t, beta)

            if trial_best is None or p_t < trial_best[0]:
                trial_best = (p_t, lse_t, cand, A_t, c_t)

            if p_t < cmax - 1e-12 or lse_t < lse_cur - 1e-12:
                accepted = True
                new_a = cand
                new_A = A_t
                new_c = c_t
                new_peak = p_t
                new_lse = lse_t
                break

        if not accepted:
            if trial_best is not None and trial_best[1] <= lse_cur + 1e-9:
                new_a = trial_best[2]
                new_A = trial_best[3]
                new_c = trial_best[4]
                new_peak = trial_best[0]
                new_lse = trial_best[1]
            else:
                noise = np.exp(0.02 * rng.standard_normal(n))
                a = a * noise
                if use_symmetric:
                    a = project_symmetric_simplex(symmetrize_full(a), n, S)
                else:
                    a = project_simplex(a, S)
                vel[:] = 0.0
                c, A = compute_conv(a, N)
                beta = min(float(beta_max), beta * 1.25)
                if no_prog > 45:
                    break
                continue

        a = new_a
        c = new_c
        A = new_A

        if new_peak < best_peak - 1e-12:
            best_peak = new_peak
            best_score = float(2.0 * n * new_peak / (S * S))
            best_a = a.copy()
            no_prog = 0

        if it % 25 == 0:
            if no_prog < 5:
                beta = min(float(beta_max), beta * 1.12)
            else:
                beta = max(1.0, beta * 0.82)

        if it % 500 == 0 and rng.random() < 0.12:
            a = spectral_restart(a, N, rng, 0.35, use_symmetric)
            vel[:] = 0.0
            c, A = compute_conv(a, N)
            beta = max(1.0, beta * 0.6)

    return best_a, best_score


def lbfgs_block(a, n, S, N, budget, rng, beta_start=5.0, beta_max=500.0,
                use_symmetric=False, m_lbfgs=12, multi_beta=True):
    a = normalize_shape(a, n, S)
    if use_symmetric:
        a = project_symmetric_simplex(symmetrize_full(a), n, S)

    c, A = compute_conv(a, N)
    best_peak = float(c.max())
    best_score = float(2.0 * n * best_peak / (S * S))
    best_a = a.copy()

    beta = float(beta_start)
    s_hist = []
    y_hist = []
    start = time.monotonic()
    it = 0
    no_prog = 0

    while time.monotonic() - start < budget:
        it += 1

        w_cur, lse_cur, _ = lse_weights(c, beta)
        cmax = float(c.max())

        if cmax < best_peak - 1e-12:
            best_peak = cmax
            best_score = float(2.0 * n * cmax / (S * S))
            best_a = a.copy()
            no_prog = 0
        else:
            no_prog += 1

        if it % 35 == 0:
            if no_prog < 8:
                beta = min(float(beta_max), beta * 1.13)
            elif no_prog > 25:
                beta = max(1.0, beta * 0.72)

        if use_symmetric:
            w_cur = 0.5 * (w_cur + w_cur[::-1])

        if multi_beta:
            betas = (max(1.0, beta / 3.0), beta, min(2.0 * float(beta_max), beta * 3.0))
            grad = None
            for bb in betas:
                wt, _, _ = lse_weights(c, bb)
                if use_symmetric:
                    wt = 0.5 * (wt + wt[::-1])
                gt = grad_from_weights(wt, A, N, n)
                if use_symmetric:
                    gt = 0.5 * (gt + gt[::-1])
                if grad is None:
                    grad = gt.copy()
                else:
                    grad += gt
            grad *= 1.0 / 3.0
        else:
            grad = grad_from_weights(w_cur, A, N, n)
            if use_symmetric:
                grad = 0.5 * (grad + grad[::-1])

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        direction = lbfgs_dir(s_hist, y_hist, grad)
        dnorm = float(np.linalg.norm(direction))
        if dnorm < 1e-15:
            direction = -grad
            dnorm = gnorm

        alpha = 20.0 * S / (float(n) * max(1.0, dnorm))
        accepted = False
        trial_best = None

        for _ in range(10):
            cand = a + alpha * direction
            if use_symmetric:
                cand = project_symmetric_simplex(cand, n, S)
            else:
                cand = project_simplex(cand, S)

            c_t, A_t = compute_conv(cand, N)
            p_t = float(c_t.max())
            w_t, lse_t, _ = lse_weights(c_t, beta)

            if trial_best is None or p_t < trial_best[0] or (
                    p_t == trial_best[0] and lse_t < trial_best[1]):
                trial_best = (p_t, lse_t, cand, A_t, c_t)

            if p_t < cmax - 1e-12 or lse_t < lse_cur - 1e-12:
                accepted = True
                new_a = cand
                new_A = A_t
                new_c = c_t
                new_peak = p_t
                new_lse = lse_t
                break

            alpha *= 0.5

        if not accepted:
            if trial_best is not None and trial_best[1] <= lse_cur + 1e-9:
                new_a = trial_best[2]
                new_A = trial_best[3]
                new_c = trial_best[4]
                new_peak = trial_best[0]
                new_lse = trial_best[1]
                accepted = True

        if not accepted:
            a = a + 1e-4 * (S / float(n)) * rng.standard_normal(n)
            if use_symmetric:
                a = project_symmetric_simplex(symmetrize_full(a), n, S)
            else:
                a = project_simplex(a, S)
            c, A = compute_conv(a, N)
            s_hist = []
            y_hist = []
            if no_prog > 80:
                break
            continue

        if multi_beta:
            betas = (max(1.0, beta / 3.0), beta, min(2.0 * float(beta_max), beta * 3.0))
            grad_new = None
            for bb in betas:
                wt, _, _ = lse_weights(new_c, bb)
                if use_symmetric:
                    wt = 0.5 * (wt + wt[::-1])
                gt = grad_from_weights(wt, new_A, N, n)
                if use_symmetric:
                    gt = 0.5 * (gt + gt[::-1])
                if grad_new is None:
                    grad_new = gt.copy()
                else:
                    grad_new += gt
            grad_new *= 1.0 / 3.0
        else:
            wt, _, _ = lse_weights(new_c, beta)
            if use_symmetric:
                wt = 0.5 * (wt + wt[::-1])
            grad_new = grad_from_weights(wt, new_A, N, n)
            if use_symmetric:
                grad_new = 0.5 * (grad_new + grad_new[::-1])

        s_k = new_a - a
        y_k = grad_new - grad
        ss = float(np.dot(s_k, s_k))
        if ss > 1e-20:
            sy = float(np.dot(s_k, y_k))
            if sy > 1e-14 * ss:
                s_hist.append(s_k)
                y_hist.append(y_k)
                if len(s_hist) > m_lbfgs:
                    s_hist.pop(0)
                    y_hist.pop(0)

        a = new_a
        c = new_c
        A = new_A

        if new_peak < best_peak - 1e-12:
            best_peak = new_peak
            best_score = float(2.0 * n * new_peak / (S * S))
            best_a = a.copy()
            no_prog = 0

        if it % 350 == 0 and rng.random() < 0.18:
            a = spectral_restart(a, N, rng, 0.25, use_symmetric)
            c, A = compute_conv(a, N)
            s_hist = []
            y_hist = []
            beta = max(1.0, beta * 0.7)

    return best_a, best_score


def nesterov_block(a, n, S, N, budget, rng, top_k_init=16, top_k_max=512,
                   use_symmetric=False, beta_top=20.0):
    a = normalize_shape(a, n, S)
    if use_symmetric:
        a = project_symmetric_simplex(symmetrize_full(a), n, S)

    x = a.copy()
    y = a.copy()
    t = 1.0

    c, A = compute_conv(y, N)
    best_peak = float(c.max())
    best_score = float(2.0 * n * best_peak / (S * S))
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    no_prog = 0

    while time.monotonic() - start < budget:
        it += 1
        c, A = compute_conv(y, N)
        peak = float(c.max())
        if peak < best_peak - 1e-12:
            best_peak = peak
            best_score = float(2.0 * n * peak / (S * S))
            best_a = y.copy()
            no_prog = 0
        else:
            no_prog += 1

        K = min(top_k_max, top_k_init + it // 20)
        K = max(1, min(K, len(c)))

        idx = np.argpartition(c, K - 1)[-K:]
        top = c[idx]
        order = np.argsort(top)[::-1]
        idx = idx[order]
        top = top[order]

        shift = np.clip(beta_top * (top - top[0]), -745.0, 0.0)
        ew = np.exp(shift)
        ew_sum = float(ew.sum())
        if ew_sum <= 1e-300:
            break
        w = np.zeros_like(c, dtype=np.float64)
        w[idx] = ew / ew_sum

        if use_symmetric:
            w = 0.5 * (w + w[::-1])
        g = grad_from_weights(w, A, N, n)
        if use_symmetric:
            g = 0.5 * (g + g[::-1])

        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-15:
            break
        g_dir = g / gnorm
        step_base = 10.0 * S / (float(n) * max(1.0, gnorm))

        accepted = False
        trial_best = None
        for factor in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
            cand = y - step_base * factor * g_dir
            if use_symmetric:
                cand = project_symmetric_simplex(cand, n, S)
            else:
                cand = project_simplex(cand, S)

            c_t, A_t = compute_conv(cand, N)
            p_t = float(c_t.max())
            if trial_best is None or p_t < trial_best[0]:
                trial_best = (p_t, cand, A_t, c_t)
            if p_t < peak - 1e-12:
                accepted = True
                new_x = trial_best[1]
                new_peak = trial_best[0]
                break

        if not accepted:
            if trial_best is not None and trial_best[0] <= peak + 1e-9:
                new_x = trial_best[1]
                new_peak = trial_best[0]
            else:
                y = y * np.exp(0.02 * rng.standard_normal(n))
                if use_symmetric:
                    y = project_symmetric_simplex(symmetrize_full(y), n, S)
                else:
                    y = project_simplex(y, S)
                x = y.copy()
                t = 1.0
                no_prog += 1
                if no_prog > 60:
                    break
                continue

        if new_peak < best_peak - 1e-12:
            best_peak = new_peak
            best_score = float(2.0 * n * new_peak / (S * S))
            best_a = new_x.copy()
            no_prog = 0

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        beta_m = (t - 1.0) / t_new
        y = new_x + beta_m * (new_x - x)
        if use_symmetric:
            y = project_symmetric_simplex(y, n, S)
        else:
            y = project_simplex(y, S)
        x = new_x
        t = t_new

    return best_a, best_score


def direct_score_of(a):
    try:
        lst = [float(x) for x in a]
        val = evaluate_sequence(lst)
    except Exception:
        return np.inf
    try:
        val = float(val)
    except Exception:
        return np.inf
    if np.isfinite(val):
        return val
    return np.inf


def search_length(n, parent_arr, rng, budget):
    n = int(n)
    if n < 2:
        n = 2
    S = float(np.sqrt(2.0 * float(n)))
    N = fft_len(n)

    if budget < 5.0:
        a = np.full(n, S / float(n), dtype=np.float64)
        return a, score_conv(a, n, S, N)

    t_start = time.monotonic()
    t_end = t_start + max(0.5, float(budget))

    best_a = np.full(n, S / float(n), dtype=np.float64)
    best_score = score_conv(best_a, n, S, N)

    a_cos, sc_cos = cosine_search_init(n, S, N)
    if sc_cos < best_score:
        best_score = sc_cos
        best_a = a_cos.copy()

    quick_end = min(t_end - 20.0, t_start + max(20.0, budget * 0.25))
    inits = make_initializations(n, S, N, rng, parent_arr)
    max_inits = 8 if n <= 10000 else 5
    if len(inits) > max_inits:
        idxs = rng.choice(len(inits), size=max_inits, replace=False)
        inits = [inits[int(i)] for i in idxs]

    for arr in inits:
        if time.monotonic() > quick_end:
            break
        now = time.monotonic()
        ef_time = min(18.0, max(3.0, (quick_end - now) * 0.55))
        a_e, sc_e = ef_block(arr, n, S, N, ef_time, rng,
                             beta_start=2.0, beta_max=120.0, use_symmetric=False)
        a_e, sc_e = better_with_sym(a_e, n, S, N)
        if sc_e < best_score:
            best_score = sc_e
            best_a = a_e.copy()

        now = time.monotonic()
        lb_time = min(12.0, max(2.0, quick_end - now - 1.0))
        a_l, sc_l = lbfgs_block(a_e, n, S, N, lb_time, rng,
                                beta_start=5.0, beta_max=250.0, use_symmetric=False,
                                multi_beta=True)
        a_l, sc_l = better_with_sym(a_l, n, S, N)
        if sc_l < best_score:
            best_score = sc_l
            best_a = a_l.copy()

    stage = 0
    stall = 0
    while time.monotonic() < t_end - 5.0:
        remaining = t_end - time.monotonic()
        if remaining < 15.0:
            break
        block_time = min(55.0, max(8.0, remaining * 0.28))

        beta_start = 4.0 + 3.0 * min(stage, 12) + 6.0 * min(stall, 6)
        beta_max = 200.0 + 120.0 * min(stage, 12)

        mod = stage % 5
        if mod == 0:
            a_new, sc_new = ef_block(best_a, n, S, N, block_time, rng,
                                     beta_start=beta_start, beta_max=beta_max,
                                     use_symmetric=False)
        elif mod == 1:
            a_new, sc_new = lbfgs_block(best_a, n, S, N, block_time, rng,
                                        beta_start=beta_start, beta_max=beta_max,
                                        use_symmetric=True, m_lbfgs=14, multi_beta=True)
        elif mod == 2:
            a_new, sc_new = nesterov_block(best_a, n, S, N, block_time, rng,
                                           top_k_init=16, top_k_max=512,
                                           use_symmetric=False)
        elif mod == 3:
            a_new, sc_new = ef_block(best_a, n, S, N, block_time, rng,
                                     beta_start=max(1.0, beta_start / 2.0),
                                     beta_max=beta_max, use_symmetric=False)
        else:
            a_new, sc_new = lbfgs_block(best_a, n, S, N, block_time, rng,
                                        beta_start=max(5.0, beta_start * 1.5),
                                        beta_max=min(1200.0, beta_max * 2.0),
                                        use_symmetric=False, m_lbfgs=14, multi_beta=False)

        a_new, sc_new = better_with_sym(a_new, n, S, N)
        if sc_new < best_score - 1e-10:
            best_a = a_new
            best_score = sc_new
            stall = 0
            print("  length=%d score=%.6f" % (n, sc_new))
        else:
            stall += 1
            if stall == 2:
                best_a = spectral_restart(best_a, N, rng, 0.35, use_symmetric=False)
                best_a, best_score = better_with_sym(best_a, n, S, N)
            elif stall >= 4:
                arr = normalize_shape(rng.random(n) + 0.1, n, S)
                a_new, sc_new = ef_block(arr, n, S, N, 25.0, rng,
                                         beta_start=1.5, beta_max=150.0,
                                         use_symmetric=False)
                a_new, sc_new = better_with_sym(a_new, n, S, N)
                if sc_new < best_score:
                    best_a = a_new
                    best_score = sc_new
                stall = 0

        stage += 1

    return best_a, float(best_score)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    total_budget = float(budget_s)
    deadline = t0 + max(0.0, total_budget - 45.0)
    rng = np.random.default_rng(seed)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        p = np.clip(p, 0.0, 1000.0)
        if p.size >= 512 and p.size <= 32768:
            parent = p
    except Exception:
        parent = None

    n_main = 8192
    if parent is not None and parent.size >= 8192:
        n_main = min(16384, int(parent.size))
    n_sec = 8192 if n_main >= 16384 else 16384

    print("lengths: main=%d sec=%d parent=%s" % (n_main, n_sec, parent is not None))

    best_main = None
    best_main_score = np.inf
    best_sec = None
    best_sec_score = np.inf

    parent_slice = None
    if parent is not None and int(parent.size) == n_main:
        parent_slice = parent

    b_main = max(60.0, (deadline - time.monotonic()) * 0.55)
    a_m, sc_m = search_length(n_main, parent_slice, rng, b_main)
    best_main = a_m
    best_main_score = sc_m
    print("main best=%.6f" % best_main_score)

    remaining = deadline - time.monotonic()
    if remaining > 90.0:
        b_sec = max(30.0, min(remaining - 80.0, (deadline - t0 - total_budget * 0.45) * 1.0))
        b_sec = max(30.0, min(remaining - 60.0, 900.0))
        parent_slice_sec = None
        if parent is not None and int(parent.size) == n_sec:
            parent_slice_sec = parent
        a_s, sc_s = search_length(n_sec, parent_slice_sec, rng, b_sec)
        best_sec = a_s
        best_sec_score = sc_s
        print("sec best=%.6f" % best_sec_score)

    candidates = []
    if best_main is not None:
        candidates.append((best_main, int(len(best_main)), best_main_score))
    if best_sec is not None:
        candidates.append((best_sec, int(len(best_sec)), best_sec_score))

    if parent is not None and parent.size >= 256:
        n_p = int(parent.size)
        S_p = float(np.sqrt(2.0 * float(n_p)))
        parent_norm = normalize_shape(parent, n_p, S_p)
        sc_p = score_conv(parent_norm, n_p, S_p, fft_len(n_p))
        candidates.append((parent_norm, n_p, sc_p))

    for extra_n in (2048, 4096, 8192, 16384):
        Sx = float(np.sqrt(2.0 * float(extra_n)))
        ax = np.full(extra_n, Sx / float(extra_n), dtype=np.float64)
        scx = score_conv(ax, extra_n, Sx, fft_len(extra_n))
        candidates.append((ax, extra_n, scx))

    final_a = None
    final_n = -1
    final_score = np.inf
    for a_cand, n_cand, sc_cand in candidates:
        if sc_cand < final_score:
            final_score = sc_cand
            final_a = normalize_shape(a_cand, n_cand, float(np.sqrt(2.0 * float(n_cand))))
            final_n = int(n_cand)

    if final_a is None:
        final_n = 8192
        S_final = float(np.sqrt(2.0 * float(final_n)))
        final_a = np.full(final_n, S_final / float(final_n), dtype=np.float64)
        final_score = 2.0

    remaining = deadline - time.monotonic() - 25.0
    if remaining > 50.0 and final_a is not None:
        S_final = float(np.sqrt(2.0 * float(final_n)))
        N_final = fft_len(final_n)
        print("final polish n=%d score=%.6f remaining=%.0fs" % (final_n, final_score, remaining))

        half = max(30.0, min(180.0, remaining * 0.45))
        a_pol, sc_pol = lbfgs_block(final_a, final_n, S_final, N_final, half, rng,
                                    beta_start=40.0, beta_max=900.0,
                                    use_symmetric=True, m_lbfgs=16, multi_beta=True)
        a_pol, sc_pol = better_with_sym(a_pol, final_n, S_final, N_final)
        if sc_pol < final_score:
            final_a = a_pol
            final_score = sc_pol

        remaining = deadline - time.monotonic() - 25.0
        if remaining > 30.0:
            etime = max(20.0, min(80.0, remaining * 0.45))
            a_pol, sc_pol = ef_block(final_a, final_n, S_final, N_final, etime, rng,
                                     beta_start=10.0, beta_max=500.0, use_symmetric=False)
            a_pol, sc_pol = better_with_sym(a_pol, final_n, S_final, N_final)
            if sc_pol < final_score:
                final_a = a_pol
                final_score = sc_pol

        remaining = deadline - time.monotonic() - 20.0
        if remaining > 25.0:
            ntime = max(15.0, min(50.0, remaining * 0.5))
            a_pol, sc_pol = nesterov_block(final_a, final_n, S_final, N_final, ntime, rng,
                                           top_k_init=32, top_k_max=700, use_symmetric=False)
            a_pol, sc_pol = better_with_sym(a_pol, final_n, S_final, N_final)
            if sc_pol < final_score:
                final_a = a_pol
                final_score = sc_pol

    # Direct verification / black-box selection if possible.
    direct_best = np.inf
    direct_best_a = None
    checked = 0
    ordered = []
    seen = []

    for a_cand, n_cand, sc_cand in candidates:
        key = n_cand
        if key not in seen:
            seen.append(key)
        ordered.append((sc_cand, a_cand, n_cand))
    ordered.sort(key=lambda z: z[0])

    for sc_cand, a_cand, n_cand in ordered:
        if checked >= 4:
            break
        if time.monotonic() > deadline - 15.0:
            break
        if n_cand > 16384:
            continue
        S_cand = float(np.sqrt(2.0 * float(n_cand)))
        a_cand = normalize_shape(a_cand, n_cand, S_cand)
        val = direct_score_of(a_cand)
        checked += 1
        if np.isfinite(val) and val < direct_best:
            direct_best = val
            direct_best_a = a_cand
            print("direct check length=%d direct=%.6f" % (n_cand, val))

    if direct_best_a is not None and np.isfinite(direct_best):
        final_a = direct_best_a
        final_score = direct_best

    final_a = normalize_shape(final_a, len(final_a), float(np.sqrt(2.0 * float(len(final_a)))))
    out = [float(x) for x in final_a]

    try:
        ver = direct_score_of(out)
        print("final direct verify=%.6f" % ver)
    except Exception:
        pass

    print("return length=%d internal=%.6f" % (len(out), final_score))
    return out
