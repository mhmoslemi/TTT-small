            j_max = min(n - 1, ki)
            if j_min <= j_max:
                idx = np.arange(j_min, j_max + 1)
                grad[idx] += 2.0 * wi * a[ki - idx]
    return grad


def lse_smooth_max(c, beta):
    """Compute LSE-smoothed maximum and weights."""
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 350:
        scale_factor = 350.0 / max(1e-15, (c - cm).max())
        scaled = scale_factor * (c - cm)
        beta = scale_factor
    e = np.exp(scaled)
    tot = e.sum()
    w = e / tot
    val = cm + np.log(tot) / beta
    return val, w


def compute_lse_grad(a, c, beta):
    """Gradient of LSE-smoothed max conv."""
    n = len(a)
    _, w = lse_smooth_max(c, beta)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def obj_from_conv(n, c_max, S):
    """Compute objective from convolution max and sum."""
    return 2.0 * n * c_max / (S * S)


def subgradient_optimize(a0, max_time, rng, verbose=True):
    """
    Nesterov-accelerated subgradient descent on the simplex.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=S.
    """
    n = len(a0)
    S = a0.sum()
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best_val = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 500
    
    # Momentum state
    momentum = np.zeros(n)
    mu = 0.7  # momentum coefficient
    
    # Adaptive parameters
    alpha = 1.0  # step size scaling
    k_cur = 8
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        
        # Adapt k: grow from 8 to ~80
        k = int(8 + 72 * progress ** 0.7)
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_val - 1e-14:
            best_val = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > patience:
            # Inject noise
            noise_scale = min(0.05, 0.003 * (stuck / patience)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            momentum *= 0.1
            stuck = 0
            patience = int(patience * 1.2)
        
        # Compute subgradient
        grad = compute_subgrad_topk(a, c, k)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        # Gap-based step size
        gap = cur_max - c.mean()
        if gap < 1e-14:
            break
        step = alpha * gap / (gn * gn)
        step = min(step, 3.0 / gn)
        
        # Nesterov-style momentum
        momentum = mu * momentum - step * grad
        
        # Try the step with line search
        a_try = project_simplex(a + momentum, S)
        c_try = fft_convolve_self(a_try)
        
        if c_try.max() <= cur_max * (1.0 + 1e-10):
            a = a_try
        else:
            # Backtrack
            for bt in range(5):
                momentum = momentum * 0.5
                a_try = project_simplex(a + momentum, S)
                c_try = fft_convolve_self(a_try)
                if c_try.max() <= cur_max * (1.0 + 1e-10):
                    a = a_try
                    break
            else:
                # Very small step
                a = project_simplex(a - 0.001 * grad / gn, S)
                momentum = mu * momentum
        
        # Periodically decay momentum when progress is good
        if stuck == 0 and it % 100 == 0:
            mu = min(0.9, mu + 0.01)
        if stuck > 0:
            mu = max(0.3, mu - 0.005)
        
        if verbose and it % 2000 == 0:
            cur_obj = obj_from_conv(n, best_val, S)
            print(f"  it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, max={best_val:.8f}, k={k}, stuck={stuck}")
    
    return best_a, best_val


def lse_optimize(a0, max_time, rng, beta_start=20.0, beta_end=5000.0, verbose=True):
    """LSE-smoothed gradient descent for polishing."""
    n = len(a0)
    S = a0.sum()
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best_val = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        
        # Exponential beta schedule
        beta = beta_start * (beta_end / beta_start) ** progress
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_val - 1e-14:
            best_val = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 300:
            # Small perturbation
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 - progress * 0.5)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            continue
        
        grad = compute_lse_grad(a, c, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        # Step size for smooth optimization
        step = 1.0 / (1.5 * beta)
        step = min(step, 2.0 / gn)
        
        a_try = project_simplex(a - step * grad, S)
        c_try = fft_convolve_self(a_try)
        
        if c_try.max() <= cur_max + 1e-12:
            a = a_try
        else:
            step *= 0.5
            a_try = project_simplex(a - step * grad, S)
            a = a_try
        
        if verbose and it % 1000 == 0:
            cur_obj = obj_from_conv(n, best_val, S)
            print(f"  it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, beta={beta:.0f}")
    
    return best_a, best_val


def coordinate_polish(a0, max_time, rng, n_candidates=50):
    """
    Coordinate-wise line search polishing.
    For randomly selected coordinates, try to find the optimal value
    given all other coordinates fixed.
    """
    n = len(a0)
    S = a0.sum()
    a = project_simplex(a0, S).copy()
    c = fft_convolve_self(a)
    best_val = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        # Select n_candidates random coordinates
        coords = rng.integers(0, n, size=n_candidates)
        
        for p in coords:
            if time.monotonic() - start > max_time - 0.1:
                break
            
            # Try several multiplicative changes to a[p]
            if a[p] < 1e-15:
                # Try setting to small positive values
                candidates = [S * 1e-4, S * 5e-5, S * 1e-5]
            else:
                candidates = [a[p] * f for f in [0.5, 0.7, 1.3, 1.5, 2.0, 3.0]]
            
            best_local = None
            best_local_val = best_val
            
            for cv in candidates:
                a[p] = max(0.0, cv)
                S_new = a.sum()
                if S_new < 1e-10:
                    a[p] = 0.0
                    continue
                # Renormalize
                a_renorm = a * (S / S_new)
                c_try = fft_convolve_self(a_renorm)
                if c_try.max() < best_local_val:
                    best_local_val = c_try.max()
                    best_local = a_renorm.copy()
            
            if best_local is not None:
                a = best_local
                if a.sum() != S:
                    a = project_simplex(a, S)
                if best_local_val < best_val - 1e-14:
                    best_val = best_local_val
                    best_a = a.copy()
            else:
                # Restore original
                a[p] = 0.0  # was already 0 or we'll re-set
                # Actually we need to restore properly - let me redo this
        
        it += 1
    
    return best_a, best_val


def generate_init(n, base=None):
    """Generate initial sequence."""
    if base is not None and len(base) > 1:
        t_old = np.linspace(0, 1, len(base))
        t_new = np.linspace(0, 1, n)
        a = np.interp(t_new, t_old, base)
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            return a
    # Raised cosine window
    tt = np.linspace(0, np.pi, n)
    a = 0.5 * (1 + np.cos(tt)) + 0.01
    return a
