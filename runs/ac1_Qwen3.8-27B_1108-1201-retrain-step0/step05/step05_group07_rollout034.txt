"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses LSE-smoothed gradient descent with Adam optimizer, adaptive beta schedule,
structured cosine-basis noise, multi-length exploration, multi-scale gradients,
and final high-precision polishing with very high beta.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_lse_weights(c, beta):
    """Compute softmax weights for LSE smoothing."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500.0, 0.0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        k = min(10, len(c))
        top = np.argsort(c)[::-1][:k]
        w = np.zeros(len(c))
        w[top] = 1.0 / k
        return w
    return e / tot


def compute_gradient_from_weights(a, p):
    """Compute gradient of LSE(c) w.r.t. a given LSE weights p."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    P = fft(p, fsize)
    A_rev = fft(a[::-1], fsize)
    corr = np.real(ifft(P * A_rev))
    return 2.0 * corr[n - 1:2 * n - 1]


def compute_multi_scale_gradient(a, c, betas):
    """Compute multi-scale gradient combining multiple beta values."""
    n = len(a)
    grad = np.zeros(n)
    total_weight = 0.0
    for beta in betas:
        p = compute_lse_weights(c, beta)
        g = compute_gradient_from_weights(a, p)
        w = 1.0 / (1.0 + 0.001 * beta)
        grad += w * g
        total_weight += w
    if total_weight > 0:
        grad /= total_weight
    return grad


def cosine_noise(n, rng, num_modes=30):
    """Generate structured noise using cosine basis (low-frequency dominant)."""
    t = np.linspace(0, 2.0 * np.pi, n, endpoint=False)
    noise = np.zeros(n)
    nm = min(num_modes, n // 2)
    for m in range(1, nm + 1):
        coeff = rng.normal(0, 1.0 / m)
        noise += coeff * np.cos(m * t)
        if m < nm:
            noise += coeff * np.sin(m * t)
    return noise


def compute_obj(a):
    """Compute objective value 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def optimize_core(a0, S, max_time, rng, beta_start, beta_end,
                  lr_init=0.1, use_multi_scale=False, verbose=True,
                  noise_period=300, label=""):
    """Core LSE-smoothed gradient descent with Adam and structured noise."""
    n = len(a0)
    a = project_simplex(a0, S)

    best_obj = compute_obj(a)
    best_a = a.copy()
    c = fft_convolve_self(a)
    cur_max = c.max()

    start = time.monotonic()
    it = 0
    stuck = 0

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    lr = lr_init

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress

        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Compute gradient
        if use_multi_scale and it < max_time / 3:
            betas = [beta * 0.1, beta, beta * 10.0]
            betas = [max(5.0, b) for b in betas]
            g = compute_multi_scale_gradient(a, c, betas)
        else:
            p = compute_lse_weights(c, beta)
            g = compute_gradient_from_weights(a, p)

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Adam update
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)

        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

        # Line search with backtracking
        accepted = False
        for _ in range(6):
            a_new = project_simplex(a - step_vec, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() <= cur_max * (1.0 + 1e-10):
                a = a_new
                accepted = True
                break
            step_vec *= 0.5

            # Tiny fallback step
            a = project_simplex(a - 1e-7 * g / gnorm, S)

        # Adaptive learning rate
        if stuck > 50:
            lr = max(1e-6, lr * 0.99)
        if stuck > noise_period:
            # Structured noise injection
            scale = min(0.05, 0.003 * (stuck - noise_period) / 100.0)

            noise = cosine_noise(n, rng, num_modes=min(40, n // 8))
            noise = noise - noise.mean()
            a = project_simplex(a + scale * noise, S)
            stuck = 0
            lr = lr_init * 0.5
            m[:] = 0.0
            v[:] = 0.0

        if verbose and it % 500 == 0:
            print(f"    {label} it={it}, t={elapsed:.0f}s, beta={beta:.0f}, "
                  f"obj={best_obj:.6f}, stuck={stuck}")

    return best_a, best_obj


def interpolate_sequence(src, target_len):
    """Interpolate a sequence to a new length."""
    n = len(src)
    if n == 1:
        return np.full(target_len, src[0])
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, target_len)
    return np.interp(t_new, t_old, np.maximum(src, 0.0))
