"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Exact coordinate descent with:
1. 1D convex optimization for each coordinate (exact solution)
2. Multiple sweeps with randomized ordering
3. Periodic normalization and perturbation
4. Multiple restarts with different initializations
5. Frequency-domain spectral shaping as initialization

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def optimize_coordinate(b, a, j):
    """Find optimal a[j] given all other coordinates fixed.
    Returns the new optimal value of a[j].
    """
    n = len(a)
    aj = a[j]
    
    # Affected lags: k in [j, j+n-1] clipped to [0, 2n-2]
    lo = max(0, j)
    hi = min(2 * n - 2, j + n - 1)
    
    # p[k] = 2*a[k-j] for k in [lo, hi], k != 2j
    # For k = 2j, we have quadratic term
    k_range = np.arange(lo, hi + 1)
    p_seg = 2.0 * a[k_range - j]
    
    # Build the full function evaluation
    # f(delta) = max(
    #   max_{k in [lo,hi], k != 2j} (b[k] + p[k]*delta),
    #   b[2j] + 2*aj*delta + delta^2  (if 2j in [0, 2n-2]),
    #   max_{k not in [lo,hi]} b[k]
    # )
    
    # Precompute the max outside the affected range
    if lo > 0 or hi < 2 * n - 2:
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[lo:hi + 1] = False
        b_outside_max = b[mask].max() if mask.any() else -np.inf
    else:
        b_outside_max = -np.inf
    
    # For the affected range, b[k] + 2*delta*a[k-j] (linear part)
    b_seg = b[lo:hi + 1]
    
    # Handle the quadratic term at k=2j
    has_quad = (0 <= 2 * j <= 2 * n - 2) and (lo <= 2 * j <= hi)
    if has_quad:
        quad_b = b[2 * j]
        quad_coeff = 2.0 * aj  # coefficient of delta in the quadratic term at k=2j
        # The linear part at k=2j is already included in b_seg + p_seg*delta
        # But we need to add delta^2
    
    def f_of_delta(delta):
        # Linear part over affected range
        vals_seg = b_seg + p_seg * delta
        fval = vals_seg.max()
        # Quadratic correction at k=2j
        if has_quad:
            quad_val = quad_b + quad_coeff * delta + delta * delta
            fval = max(fval, quad_val)
        # Outside max
        fval = max(fval, b_outside_max)
        return fval
    
    def subgradient_at(delta):
        # Returns (min_subgrad, max_subgrad)
        vals_seg = b_seg + p_seg * delta
        fval = vals_seg.max()
        if has_quad:
            quad_val = quad_b + quad_coeff * delta + delta * delta
            fval = max(fval, quad_val)
        fval = max(fval, b_outside_max)
        
        # Subgradients from active constraints
        sgrads = []
        active_seg = vals_seg >= fval - 1e-12 * max(1.0, abs(fval))
        if active_seg.any():
            sgrads.append(p_seg[active_seg].min())
            sgrads.append(p_seg[active_seg].max())
        if has_quad:
            quad_val = quad_b + quad_coeff * delta + delta * delta
            if quad_val >= fval - 1e-12 * max(1.0, abs(fval)):
                sgrad_vals = quad_coeff + 2 * delta
                sgrads.append(sgrad_vals)
                sgrads.append(sgrad_vals)
        if b_outside_max >= fval - 1e-12 * max(1.0, abs(fval)):
            sgrads.append(0.0)
            sgrads.append(0.0)
        
        if not sgrads:
            return 0.0, 0.0
        return min(sgrads), max(sgrads)
    
    # Bisection on the subgradient
    delta_lo = -aj  # a[j] + delta >= 0
    delta_hi = max(aj * 2, 1.0)  # Upper bound
    
    # Check delta = 0
    sg_lo, sg_hi = subgradient_at(0.0)
    if sg_lo <= 0 <= sg_hi:
        return aj  # Already optimal
    
    # Narrow the search
    if sg_lo > 0:
        # Need to decrease delta
        delta_hi = 0.0
    else:
        # Need to increase delta  
        delta_lo = 0.0
    
    # Bisection
    delta_opt = 0.0
    for _ in range(50):

        sg_lo, sg_hi = subgradient_at(delta_mid)
        if sg_lo <= 0 <= sg_hi:
            delta_opt = delta_mid
            break
        if sg_lo > 0:
            delta_hi = delta_mid
        else:
            delta_lo = delta_mid
        delta_opt = delta_mid
    
    return aj + delta_opt


def coordinate_descent_sweep(a, b, rng, damping=1.0):
    """One sweep of coordinate descent. Returns updated a and b."""
    n = len(a)
    a = a.copy()
    order = rng.permutation(n)
    
    for j in order:
        new_aj = optimize_coordinate(b, a, j)
        delta = (new_aj - a[j]) * damping
        if abs(delta) > 1e-14:
            a[j] = new_aj  # Use full update (damping might not be needed with exact 1D)
            # Update b
            lo = max(0, j)
            hi = min(2 * n - 2, j + n - 1)
            b[lo:hi + 1] += 2.0 * delta * a[lo - j:hi + 1 - j]
            if lo <= 2 * j <= hi:
                b[2 * j] += delta * delta
    
    return a, b


def run_coordinate_descent(a_init, n_sweeps, rng, normalize_sum=None):
    """Run multiple sweeps of coordinate descent."""
    n = len(a_init)
    a = a_init.copy()
    
    if normalize_sum is not None:
        a = normalize(a, normalize_sum)
    
    b = fft_convolve_self(a)
    best_obj = 2.0 * n * b.max() / (a.sum() ** 2)
    best_a = a.copy()
    
    for sweep in range(n_sweeps):
        a, b = coordinate_descent_sweep(a, b, rng)
        
        # Renormalize
        if normalize_sum is not None:
            old_sum = a.sum()
            a = normalize(a, normalize_sum)
            scale = a.sum() / max(old_sum, 1e-15)
            b = b * scale * scale
        
        obj = 2.0 * n * b.max() / (a.sum() ** 2)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def spectral_init(n, rng, target_sum):
    """Initialize using spectral shaping."""
    # Start with random signal
    a = rng.random(n)
    a = normalize(a, target_sum)
    
    # Flatten spectrum
    fsize = 1 << (2 * n - 1).bit_length()
    for _ in range(20):
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        mean_p = power.mean()
        # Compress dynamic range
        new_power = power ** 0.7 * mean_p ** 0.3
        A_new = A * np.sqrt(new_power / np.maximum(power, 1e-30))
        a_new = np.real(ifft(A_new, fsize))[:n]
        a = np.clip(a_new, 0, None)
        a = normalize(a, target_sum)
    
    return a


def cosine_init(n, target_sum):
    """Cosine window initialization."""
    a = np.cos(np.pi * np.arange(n) / n) ** 2
    a = np.clip(a, 0, None)
    return normalize(a, target_sum)


def uniform_init(n, target_sum):
    """Uniform initialization."""

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def optimize_coordinate(b, a, j):

    aj = a[j]
    lo = max(0, j)
    hi = min(2 * n - 2, j + n - 1)

    p_seg = 2.0 * a[k_range - j]

    if has_quad:
        quad_b = b[2 * j]
        quad_coeff = 2.0 * aj
    
    if lo > 0 or hi < 2 * n - 2:
        # Compute max outside affected range
        b_outside_max = max(b[:lo].max() if lo > 0 else -np.inf,
                           b[hi+1:].max() if hi < 2*n-2 else -np.inf)

        b_outside_max = -np.inf

    fval = vals_seg.max()
    if has_quad:

    fval = max(fval, b_outside_max)

    sgrad_max = -np.inf
    tol = 1e-12 * max(1.0, abs(fval))
    active = vals_seg >= fval - tol
    if active.any():

        sgrad_max = max(sgrad_max, p_seg[active].max())
    if has_quad and quad_b >= fval - tol:
        sgrad_min = min(sgrad_min, quad_coeff)
        sgrad_max = max(sgrad_max, quad_coeff)
    if b_outside_max >= fval - tol:
        sgrad_min = min(sgrad_min, 0.0)
        sgrad_max = max(sgrad_max, 0.0)

        return aj

        delta_lo = -aj
        delta_hi = 0.0
    else:
        delta_lo = 0.0
        delta_hi = max(aj, 0.01)

    # So the function is bounded above by delta^2 + ...
    # A safe upper bound: where delta^2 > fval + something

    best_fval = fval

        vals_seg = b_seg + p_seg * delta_mid
        fval = vals_seg.max()

            quad_val = quad_b + quad_coeff * delta_mid + delta_mid * delta_mid
            fval = max(fval, quad_val)
        fval = max(fval, b_outside_max)
        
        if fval < best_fval:
            best_fval = fval
            best_delta = delta_mid
        
        # Subgradient
        tol = 1e-12 * max(1.0, abs(fval))
        sgrad_min = np.inf
        sgrad_max = -np.inf
        active = vals_seg >= fval - tol
        if active.any():
            sgrad_min = min(sgrad_min, p_seg[active].min())
            sgrad_max = max(sgrad_max, p_seg[active].max())
        if has_quad:

                sgrad_max = max(sgrad_max, s)

def cd_sweep(a, b, rng, n_iter_pairs=None):
    """One sweep of coordinate descent."""
    n = len(a)
    a = a.copy()
    order = rng.permutation(n)
    
    for j in order:
        new_aj = optimize_coordinate(b, a, j)
        delta = new_aj - a[j]

            lo = max(0, j)
            hi = min(2 * n - 2, j + n - 1)
            b[lo:hi + 1] += 2.0 * delta * a[lo - j:hi + 1 - j]
            if lo <= 2 * j <= hi:
                b[2 * j] += delta * delta
    
    return a, b
