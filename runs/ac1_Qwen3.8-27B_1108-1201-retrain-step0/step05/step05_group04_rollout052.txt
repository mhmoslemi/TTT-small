"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Multi-direction active-set gradient with exact line search,
structured perturbations for local exploration, and final polish.
Key innovation: Instead of LSE smoothing, we directly optimize using the
exact subgradient of the max function, trying multiple active-set configurations
and using bisection line search for accuracy.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix numerical issues
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference among positive elements
        pos = result > 0
        n_pos = pos.sum()
        if n_pos > 0:
            result[pos] += diff / n_pos
            result = np.maximum(result, 0.0)
    return result


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_gradient_active(a, c, active_lags, weights=None):
    """
    Compute the exact subgradient of max(c) using given active lags.
    grad[i] = sum_k w_k * 2 * a[k-i] (for 0<=k-i<n)
    """
    n = len(a)
    if weights is None:
        weights = np.ones(len(active_lags)) / len(active_lags)
    
    grad = np.zeros(n)
    for k, w in zip(active_lags, weights):
        # grad[i] += w * 2 * a[k-i] for valid i (0 <= i < n, 0 <= k-i < n)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            partners = k - js
            grad[js] += 2.0 * w * a[partners]
    return grad


def exact_line_search(a, grad, S, n, c_max, max_iter=50):
    """
    Bisection line search: find the largest step such that
    max(conv(a - step*grad)) < c_max.
    """
    lo = 0.0
    hi = 1.0
    
    # First, find an upper bound where objective increases
    while hi < 100.0:
        a_test = project_simplex(a - hi * grad, S)
        c_test = fft_convolve_self(a_test)
        if c_test.max() > c_max:
            break
        lo = hi
        hi *= 2
    
    # Bisection
    for _ in range(max_iter):
        mid = 0.5 * (lo + hi)
        a_test = project_simplex(a - mid * grad, S)
        c_test = fft_convolve_self(a_test)
        if c_test.max() > c_max:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-10:
            break
    
    a_new = project_simplex(a - lo * grad, S)
    c_new = fft_convolve_self(a_new)
    return a_new, c_new.max(), lo

def quick_line_search(a, grad, S, c_max, max_iter=20):
    """
    Quick line search: find a step that decreases the objective.

def lse_fista_fast(a_init, n, S, max_time, rng,
                   beta_start=100.0, beta_end=20000.0, beta_steps=100):
    """
    Fast LSE-FISTA for initial optimization.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()

    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
        
        # LSE gradient
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / beta
        
        # Stagnation handling
        stagnant = it - last_improve
        if stagnant > 200:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 10000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            step *= 0.5
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

def responsibility_shave(a, n, S, max_time, rng, n_peaks=5):
    """
    Responsibility-based peak shaving: for each top peak, compute which
    a[i] contribute most, and reduce them.
    """
    a = normalize_sum(a, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        c_max = c.max()
        
        if c_max < best:
            best = c_max
            best_a = a.copy()
        
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # Compute responsibility
        resp = np.zeros(n)
        for k in top_idx:
            weight = max(0.0, c[k] - c_max * 0.95)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            resp[js] += weights_for(k, weight, contrib)
        
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        
        # Multiplicative update
        lr = 0.01 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def weights_for(k, weight, contrib):
    Helper function to compute weighted contributions.
    return contrib * weight

Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Exact active-set subgradient optimization with adaptive line search,
multiple active-set configurations, and structured local search.
Key insight: The subgradient of max(conv(a,a)) at lag k* is 2*a[k*-i],
which means the gradient has the same shape as a (reflected). We exploit
this structure for efficient line search and direction selection.

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix sum
    diff = S - result.sum()
    pos = result > 0
    npos = pos.sum()
    if npos > 0 and abs(diff) > 1e-12:
        result[pos] += diff / npos
        result = np.maximum(result, 0.0)
    return result


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def subgradient_at_lag(a, k):
    """Exact subgradient of c[k] with respect to a: 2*a[k-i] for valid i."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        js = np.arange(lo, hi + 1)
        partners = k - js
        grad[js] = 2.0 * a[partners]
    return grad


def weighted_subgradient(a, lags, weights):
    """Weighted combination of subgradients at multiple lags."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(lags, weights):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            partners = k - js
            grad[js] += 2.0 * w * a[partners]
    return grad


def line_search(a, grad, S, obj_current, max_trials=15):
    """
    Find the best step among a set of candidate steps.
    Returns (best_a, best_obj).

        return a.copy(), obj_current
