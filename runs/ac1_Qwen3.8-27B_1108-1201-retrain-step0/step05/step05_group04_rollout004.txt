"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Adaptive hybrid optimization combining:
1. LSE-FISTA with backtracking line search on true objective
2. Blended active-set gradient for the "last mile"
3. Gerchberg-Saxton spectral diversification
4. Multi-length exploration with intensive peak shaving
Key insight: Normalize sum(f)=sqrt(2n), then minimize max(f*f) directly.
The spectral approach explores a fundamentally different basin via alternating
projections between non-negativity (time) and flat-spectrum (frequency).

import numpy as np
import time
from scipy.fft import fft, ifft


# === Top-level helper functions ===

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c). Returns (grad, lse_value)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def active_set_gradient(a, c, n_active=20, threshold_frac=0.002):
    """Gradient using hard active set (top lags)."""
    n = len(a)
    cm = c.max()
    threshold = cm - threshold_frac * cm
    w = np.where(c >= threshold, 1.0, 0.0)
    # Also ensure at least n_active lags are included
    if w.sum() < n_active:
        top_idx = np.argsort(c)[::-1][:n_active]
        w[top_idx] = 1.0
    tot = w.sum()
    if tot < 1:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        tot = 1.0
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def blended_gradient(a, c, beta, blend):
    """Blend LSE gradient and active-set gradient."""
    grad_lse, _ = lse_gradient(a, c, beta)
    grad_as = active_set_gradient(a, c, n_active=max(10, len(a) // 512))
    return (1.0 - blend) * grad_lse + blend * grad_as


def gs_diversify(a, n, S, n_iters=2000, compression=0.3, rng=None):
    """
    Gerchberg-Saxton iterations: alternate between projecting to a>=0 (time)
    and compressing |FFT(a)| toward flat (frequency).
    """
    a = a.copy()
    fsize = 1 << (2 * n - 2).bit_length()
    
    for i in range(n_iters):
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress magnitude toward geometric mean
        log_mag = np.log(np.maximum(mag, 1e-30))
        target_log = np.mean(log_mag)
        log_mag_new = log_mag * (1.0 - compression) + target_log * compression
        mag_new = np.exp(log_mag_new)
        
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fsize))[:n]
        
        # Project to non-negative, normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new * (S / s)
        
        a = a_new
    
    return a

            step = min(step * 1.05, 2.0 / beta)
        else:
            # Reject, shrink step
            step *= 0.5
            if step < 1e-10 / max(1, beta):
                # Try a direct subgradient step with tiny step
                a = project_simplex(y - (1.0 / (beta + 1)) * grad, S)
                step = 1.0 / (beta + 1)

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = a + ((t_mom - 1.0) / t_new) * (a - (y - ((t_new - 1.0) / t_new) * (y - a) - (y - a)))
        # Simplified: standard FISTA
        y = project_simplex(a + ((t_mom - 1.0) / t_new) * (a - a), S) if False else y
        # Let me redo this properly
        # Standard FISTA: y_{k+1} = x_{k+1} + (t_k - 1)/(t_{k+1}) * (x_{k+1} - x_k)
        x_old = a  # this is x_{k+1} after projection
        # Actually let me restructure
        t_mom = t_new

        # Periodic re-evaluation
        if it % 200 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() < best:
                best = c_check.max()
                best_a = a.copy()

    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def lse_fista_v2(a_init, n, S, max_time, rng,
                 beta_start=50.0, beta_end=30000.0, beta_steps=300):
    """
    Cleaner FISTA implementation with proper momentum, backtracking,
    and blended gradients for the last mile.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step_size = 1.0 / beta_start

    while time.monotonic() - start < max_time:
        it += 1

        frac = min(1.0, it / max(1, beta_steps))
        beta = min(beta_start * (beta_end / beta_start) ** frac, 80000.0)

        c_y = fft_convolve_self(y)
        cur_max = c_y.max()

        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        # Compute gradient
        blend = min(0.6, max(0.0, (frac - 0.5) * 1.2))
        if blend > 0.01:
            grad_lse, _ = lse_gradient(y, c_y, beta)
            grad_as = active_set_gradient(y, c_y, n_active=max(10, n // 256))
            grad = (1.0 - blend) * grad_lse + blend * grad_as
        else:
            grad, _ = lse_gradient(y, c_y, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Normalize gradient to have unit norm for step control
        grad_unit = grad / gnorm

        # Backtracking
        step = min(step_size, 1.0 / beta)
        improved = False
        for _ in range(10):
            x_new = project_simplex(y - step * grad_unit * gnorm, S)
            c_x = fft_convolve_self(x_new)
            if c_x.max() < cur_max * (1.0 - 1e-8) or step < 1e-12:
                improved = True
                break
            step *= 0.5

        if not improved:
            x_new = y  # No improvement, stay
            step *= 0.5
        else:
            step_size = min(step * 1.1, 2.0 / beta)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_mom = t_new

        # Stagnation handling
        stagnant = it - last_improve
        if stagnant > 500:
            t_mom = 1.0
            y = a.copy()
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stagnant / 20000.0)
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            step_size = min(step_size * 0.5, 1.0 / beta)

    # Final
    c_f = fft_convolve_self(a)
    if c_f.max() < best:
        best = c_f.max()
        best_a = a.copy()

    return best_a, best


def active_set_refine(a_init, n, S, max_time, rng, n_active_top=30):
    """
    Pure active-set gradient descent with backtracking.
    For the final "last mile" when LSE is too smooth.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    step = 0.01

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Active set gradient with adaptive threshold
        cm = cur_max
        threshold = cm - 0.003 * cm  # lags within 0.3% of max
        w = np.where(c >= threshold, 1.0, 0.0)
        if w.sum() < n_active_top:
            top_idx = np.argsort(c)[::-1][:n_active_top]
            w[top_idx] = np.maximum(w[top_idx], 1.0)
        tot = w.sum()
        w /= tot

        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        grad_unit = grad / gnorm

        # Backtracking
        improved = False
        for _ in range(15):
            a_trial = project_simplex(a - step * grad_unit * gnorm, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_max * (1.0 - 1e-9):
                a = a_trial
                improved = True
                break
            step *= 0.5

        if not improved:
            step *= 0.3
            if step < 1e-14:
                # Tiny random perturbation
                a = project_simplex(a + rng.normal(0, 1e-6 * S / n, n), S)
                step = 0.01
        else:
            step = min(step * 1.15, 0.1)

        # Periodic noise for exploration
        if it % 500 == 0:
            ns = 0.001 * S / np.sqrt(n) * np.exp(-it / 5000.0)
            if ns > 1e-10:
                a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best


def peak_shaving_direct(a, n, S, max_time, rng, n_peaks=10):
    """
    Direct peak shaving: for each peak lag, reduce the contributions.
    Uses multiplicative updates on high-responsibility elements.
    """
    a = a.copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cm = c.max()
        if cm < best:
            best = cm
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:n_peaks]

        # Compute responsibility for each a[j]
        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            # Weight by excess over target (average = 2n/(2n-1) ~ 1)
            weight = max(0.0, c[k] - 1.0) ** 2
            np.add.at(resp, js, contrib * weight)

        resp_max = resp.max()
        if resp_max < 1e-30:
            break

        lr = 0.01 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)

    return best_a, best


def diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n)

    # Asymmetric raised cosine
    for cf in [0.2, 0.25, 0.3, 0.35, 0.4, 0.45, 0.5]:
        for wf in [0.3, 0.4, 0.5, 0.6]:
            center = cf
            width = wf / 2.0
            x = (t - center) / width
            a = np.zeros(n)
            mask = np.abs(x) < 1
            a[mask] = 0.5 * (1 + np.cos(np.pi * x[mask]))
            inits.append(np.maximum(a, 0.0))

    # Gaussian bumps
    for sigma in [0.08, 0.12, 0.15, 0.2, 0.25]:
        for mu in [0.25, 0.35, 0.45, 0.55]:
            a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
            inits.append(a)

    # Power laws
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(t ** p)
        inits.append((1 - t) ** p)
        # Translated
        for shift in [0.2, 0.3]:
            a = np.maximum(t - shift, 0) ** p
            inits.append(a)

    # Sine-based
    for k in [1, 2, 3]:
        for phase in [0, 0.3, 0.5]:
            a = np.maximum(0.5 + 0.5 * np.sin(2 * np.pi * k * t + phase * np.pi), 0)
            inits.append(a)

    # Bimodal
    for gap in [0.15, 0.2, 0.3]:
        for width in [0.08, 0.12]:
            a = np.exp(-0.5 * ((t - 0.3) / width) ** 2) + np.exp(-0.5 * ((t - 0.3 - gap) / width) ** 2)
            inits.append(a)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Uses multi-phase strategy: warm start, adaptive LSE-FISTA, GS diversification,
    active-set refinement, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 90.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))

    # Load previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded given: n=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_seq(a):
        nonlocal global_best, global_a
        if a is None:
            return
        n = len(a)
        S = np.sqrt(2.0 * n)
        a = normalize_sum(a, n)
        c = fft_convolve_self(a)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a.copy()
            print("  NEW BEST: %.8f (n=%d, t=%.0fs)" % (val, n, time.monotonic() - (deadline - budget_s + safety)))

    # === Phase 1: Quick warm start on given (60s) ===
    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        t1 = min(60.0, max(10.0, (deadline - time.monotonic()) * 0.05))
        print("=== Phase 1: Warm start (n=%d, %.0fs) ===" % (n_g, t1))
        c_g = fft_convolve_self(normalize_sum(given, n_g))
        print("  Given max: %.8f" % c_g.max())
        a1, _ = lse_fista_v2(given, n_g, S_g, t1, rng,
                             beta_start=200.0, beta_end=10000.0, beta_steps=100)
        try_seq(a1)

    # === Phase 2: Main LSE-FISTA on best (600s) ===
    if global_a is not None:
        n_m = len(global_a)
        S_m = np.sqrt(2.0 * n_m)
        remaining = deadline - time.monotonic()
        t2 = min(600.0, remaining * 0.35)
        print("=== Phase 2: Main LSE-FISTA (n=%d, %.0fs) ===" % (n_m, t2))
        a2, _ = lse_fista_v2(global_a, n_m, S_m, t2, rng,
                             beta_start=100.0, beta_end=50000.0, beta_steps=400)
        try_seq(a2)

    # === Phase 3: GS diversification (300s) ===
    if global_a is not None and (deadline - time.monotonic()) > 100:
        n_gs = len(global_a)
        S_gs = np.sqrt(2.0 * n_gs)
        remaining = deadline - time.monotonic()
        t3 = min(300.0, remaining * 0.2)
        print("=== Phase 3: GS diversification (n=%d, %.0fs) ===" % (n_gs, t3))

        # Generate several GS variants
        n_variants = 4
        per_variant_gs = max(1, int(n_iters_split := t3 * 0.6 / n_variants))
        per_variant_opt = max(5.0, t3 * 0.35 / n_variants)
        t3_per_gs = t3 * 0.6 / n_variants
        t3_per_opt = t3 * 0.35 / n_variants

        for v in range(n_variants):
            if time.monotonic() > deadline - 30:
                break
            comp = 0.15 + 0.15 * v  # varying compression
            print("  GS variant %d (compression=%.2f)" % (v, comp))
            a_gs = gs_diversify(global_a, n_gs, S_gs, n_iters=int(t3_per_gs * 50), compression=comp, rng=rng)
            # Quick LSE polish
            a_pol, _ = lse_fista_v2(a_gs, n_gs, S_gs, t3_per_opt, rng,
                                    beta_start=500.0, beta_end=20000.0, beta_steps=50)
            try_seq(a_pol)

    # === Phase 4: Active-set refinement (400s) ===
    if global_a is not None and (deadline - time.monotonic()) > 50:
        n_as = len(global_a)
        S_as = np.sqrt(2.0 * n_as)
        remaining = deadline - time.monotonic()
        t4 = min(400.0, remaining * 0.3)
        print("=== Phase 4: Active-set refinement (n=%d, %.0fs) ===" % (n_as, t4))
        a4, _ = active_set_refine(global_a, n_as, S_as, t4, rng, n_active_top=40)
        try_seq(a4)

    # === Phase 5: Peak shaving (200s) ===
    if global_a is not None and (deadline - time.monotonic()) > 30:
        n_ps = len(global_a)
        S_ps = np.sqrt(2.0 * n_ps)
        remaining = deadline - time.monotonic()
        t5 = min(200.0, remaining * 0.25)
        print("=== Phase 5: Peak shaving (n=%d, %.0fs) ===" % (n_ps, t5))
        a5, _ = peak_shaving_direct(global_a, n_ps, S_ps, t5 * 0.7, rng, n_peaks=12)
        try_seq(a5)
        # Polish after shaving
        if (deadline - time.monotonic()) > 20:
            a5b, _ = lse_fista_v2(a5, n_ps, S_ps, min(60.0, deadline - time.monotonic() - 10), rng,
                                  beta_start=2000.0, beta_end=50000.0, beta_steps=30)
            try_seq(a5b)

    # === Phase 6: Try longer sequence (300s) ===
    if global_a is not None and (deadline - time.monotonic()) > 150:
        n_ext = len(global_a) * 2
        if n_ext <= 32768:
            S_ext = np.sqrt(2.0 * n_ext)
            remaining = deadline - time.monotonic()
            t6 = min(300.0, remaining * 0.4)
            print("=== Phase 6: Extended n=%d (%.0fs) ===" % (n_ext, t6))
            # Interpolate
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n_ext)
            init_ext = np.interp(t_new, t_old, global_a)
            init_ext = normalize_sum(init_ext, n_ext)
            a6, _ = lse_fista_v2(init_ext, n_ext, S_ext, t6 * 0.7, rng,
                                 beta_start=100.0, beta_end=30000.0, beta_steps=200)
            try_seq(a6)
            # Active set refine on extended
            if (deadline - time.monotonic()) > 20:
                a6b, _ = active_set_refine(a6, n_ext, S_ext, min(60.0, deadline - time.monotonic() - 10), rng)
                try_seq(a6b)

    # === Phase 7: Diverse initializations (200s) ===
    if (deadline - time.monotonic()) > 100:
        n_div = 8192
        S_div = np.sqrt(2.0 * n_div)
        remaining = deadline - time.monotonic()
        t7 = min(200.0, remaining * 0.4)
        print("=== Phase 7: Diverse inits (n=%d, %.0fs) ===" % (n_div, t7))

        inits = diverse_inits(n_div, rng)
        # Score initializations
        scores = []
        for i, init in enumerate(inits):
            init_n = normalize_sum(init, n_div)
            c_init = fft_convolve_self(init_n)
            scores.append((c_init.max(), i))
        scores.sort()

        # Try top 4 diverse initializations
        per_init = max(10.0, t7 / 5.0)
        tried = set()
        count = 0
        for score, idx in scores:
            if count >= 4 or time.monotonic() > deadline - 30:
                break
            if idx in tried:
                continue
            tried.add(idx)
            count += 1
            print("  Diverse init %d (score=%.4f)" % (idx, score))
            ai, _ = lse_fista_v2(inits[idx], n_div, S_div, per_init, rng,
                                 beta_start=50.0, beta_end=20000.0, beta_steps=100)
            try_seq(ai)

    # === Final active-set polish ===
    if global_a is not None and (deadline - time.monotonic()) > 5:
        n_f = len(global_a)
        S_f = np.sqrt(2.0 * n_f)
        t_f = max(2.0, deadline - time.monotonic() - 5)
        print("=== Final polish: active-set (%.0fs) ===" % t_f)
        af, _ = active_set_refine(global_a, n_f, S_f, t_f, rng, n_active_top=50)
        try_seq(af)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate=%.8f, internal=%.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Fallback to uniform.")
            n_fb = 256
            global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
    else:
        print("WARNING: No valid result. Using uniform fallback.")
        n_fb = 256
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 256
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)

    print("RETURNING: n=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
