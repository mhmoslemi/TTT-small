```python
def lp_step(a, M, active_lags, time_limit=5.0):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)[k] <= M for k in active_lags.
    Returns the optimal b (or None if infeasible/timeout).
    """
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    n = len(a)
    m = len(active_lags)
    
    # Build constraint matrix: for each lag k, (a*b)[k] = sum_j a[j]*b[k-j] <= M
    # Row i corresponds to lag active_lags[i]
    # Column j corresponds to b_j
    
    # For lag k: (a*b)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * b[k-j]
    # Let l = k-j, so j = k-l, and we need 0 <= l < n and 0 <= k-l < n
    # So l ranges from max(0, k-n+1) to min(k, n-1)
    # The coefficient of b[l] in row i is a[k-l] = a[active_lags[i] - l]
    
    rows = []
    cols = []
    vals = []
    
    for i, k in enumerate(active_lags):
        l_min = max(0, k - n + 1)
        l_max = min(k, n - 1)
        for l in range(l_min, l_max + 1):
            rows.append(i)
            cols.append(l)
            vals.append(a[k - l])
    
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    b_vec = np.full(m, M)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    
    if res.success:
        return res.x
    return None
```

```python
"""
Minimize 2*n*max(f*f)/sum(f)^2 using LP-based direction finding combined with
LSE-FISTA and surgical peak refinement. Key innovation: exact active-set LP for
descent direction, frequency-domain subgradient computation, and multi-scale
exploration to achieve sub-1.503 objective values.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix, lil_matrix
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full cross-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_subgradient_fft(a, lags, weights, fsize=None):
    """
    Compute sum_k weights[k] * grad of (a*a)[k] w.r.t. a, using FFT.
    grad_k[m] = 2*a[k-m] if 0 <= k-m < n, else 0.
    In frequency domain: FFT(grad_k) = 2 * A * exp(-2*pi*i*freq*k/fsize)
    Actually it's a cross-correlation, so: FFT(grad_k)[f] = 2 * A[f] * conj(exp(2*pi*i*f*k/fsize))
    = 2 * A[f] * exp(-2*pi*i*f*k/fsize)
    """
    n = len(a)
    if fsize is None:
        size = 2 * n - 1
        fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    freqs = np.arange(fsize)
    # grad = sum_k weights[k] * 2 * ifft(A * exp(-2*pi*i*freqs*k/fsize))
    # = 2 * ifft(A * sum_k weights[k] * exp(-2*pi*i*freqs*k/fsize))
    W = np.zeros(fsize, dtype=np.complex128)
    for k, w in zip(lags, weights):
        if w != 0.0:
            W += w * np.exp(-2j * np.pi * freqs * k / fsize)
    grad = 2.0 * np.real(ifft(A * W))[:n]
    return grad


def lse_subgradient(a, c, beta, fsize=None):
    """Compute LSE gradient (soft subgradient)."""
    n = len(a)
    if fsize is None:
        size = 2 * n - 1
        fsize = 1 << (size - 1).bit_length()
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Use top lags only for efficiency
    n_top = min(len(c), 100)
    idx = np.argsort(w)[::-1][:n_top]
    w_top = w[idx]
    w_top = w_top / w_top.sum()
    grad = compute_subgradient_fft(a, idx, w_top, fsize)
    lse_val = cm + np.log(tot) / beta
    return grad, lse_val


def solve_lp_direction(a, M, active_lags, time_limit=10.0):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)[k] <= M for k in active_lags.
    Returns optimal b or None.
    """
    n = len(a)
    m = len(active_lags)
    if m == 0:
        return None

    rows = []
    cols = []
    vals = []

    for i, k in enumerate(active_lags):
        l_min = max(0, k - n + 1)
        l_max = min(k, n - 1)
        if l_min > l_max:
            continue
        for l in range(l_min, l_max + 1):
            rows.append(i)
            cols.append(l)
            vals.append(a[k - l])

    if len(vals) == 0:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    b_vec = np.full(m, M)
    c_obj = -np.ones(n)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': time_limit,
                                                'presolve': True,
                                                'dual_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_direction_capped(a, M, active_lags, cap, time_limit=10.0):
    """
    Solve: max sum(b) s.t. b >= 0, sum(b) <= cap, (a*b)[k] <= M for k in active_lags.
    """
    n = len(a)
    m = len(active_lags)
    if m == 0:
        return None

    rows = []
    cols = []
    vals = []

    for i, k in enumerate(active_lags):
        l_min = max(0, k - n + 1)
        l_max = min(k, n - 1)
        if l_min > l_max:
            continue
        for l in range(l_min, l_max + 1):
            rows.append(i)
            cols.append(l)
            vals.append(a[k - l])

    if len(vals) == 0:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    b_vec = np.full(m, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def run_lp_optimize(a_init, n, max_time, rng, verbose=True):
    """
    LP-based iterative optimization.
    At each step: find active lags, solve LP for direction, line search.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    stagnant = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()

        if M < best:
            best = M
            best_a = a.copy()
            stagnant = 0
            if verbose and it % 20 == 0:
                elapsed = time.monotonic() - start
                print("    LP iter %d: M=%.10f, elapsed=%.0fs" % (it, M, elapsed))
        else:
            stagnant += 1

        # Find active lags (within tolerance of max)
        tol = max(1e-12, 1e-8 * M)
        active = np.where(c >= M - tol)[0]

        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Solve LP
        b = None
        if len(active) <= 50:
            b = solve_lp_direction(a, M, active, time_limit=5.0)
        if b is None or b.sum() < 1e-10:
            # Try with cap
            if len(active) <= 100:
                b = solve_lp_direction_capped(a, M, active, cap=10.0 * S, time_limit=5.0)

        if b is not None and b.sum() > 1e-10:
            g = b * S / b.sum()
            # Line search
            best_t = 0.0
            best_val = M
            for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_try = (1.0 - t) * a + t * g
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try * S / s
                c_try = fft_convolve_self(a_try)
                val = c_try.max()
                if val < best_val:
                    best_val = val
                    best_t = t

            if best_t > 0:
                a = (1.0 - best_t) * a + best_t * g
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-10:
                    a = a * S / s
            else:
                stagnant += 1
        else:
            # LP failed, use LSE gradient
            beta = 500.0 + 100.0 * it
            grad, _ = lse_subgradient(a, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad = grad / gnorm * gnorm
            a = project_simplex(a - 0.01 * grad / (gnorm + 1e-15), S)
            stagnant += 1

        # Stagnation handling
        if stagnant > 30:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-stagnant / 200.0)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stagnant = 0

        # Convergence check
        if it > 100 and (best - c.max()) > 1e-15:
            pass  # still improving

    # Final
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_lse_fista(a_init, n, max_time, rng, beta_start=50.0, beta_end=20000.0,
                  beta_steps=200, verbose=True):
    """FISTA with adaptive LSE smoothing."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    fsize = 1 << (2 * n).bit_length()

    while time.monotonic() - start < max_time:
        it += 1

        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        grad, _ = lse_subgradient(y, c, beta, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta * 0.9

        stagnant = it - last_improve
        if stagnant > 200:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 5000.0)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print("    FISTA iter %d: best=%.10f, elapsed=%.0fs" % (it, best, elapsed))

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def surgical_peak_refinement(a, n, max_time, rng, verbose=True):
    """
    Targeted peak reduction: identify top contributing pairs and adjust them
    to reduce the maximum convolution value.
    """
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()

        if M < best:
            best = M
            best_a = a.copy()

        # Find the peak lag
        k_peak = int(np.argmax(c))

        # Find top contributing pairs for this lag
        m_min = max(0, k_peak - n + 1)
        m_max = min(k_peak, n - 1)
        if m_min > m_max:
            break

        indices = np.arange(m_min, m_max + 1)
        products = a[indices] * a[k_peak - indices]
        total = products.sum()

        if total < 1e-15:
            break

        # Top contributors
        top_count = min(20, len(indices))
        top_local = np.argsort(products)[::-1][:top_count]
        top_idxs = indices[top_local]
        top_products = products[top_local]

        # For each top pair (j, k_peak-j), try to rebalance
        # Reduce the larger one slightly, increase the smaller one
        improved = False
        for idx in top_idxs:
            partner = k_peak - idx
            if partner < 0 or partner >= n or partner == idx:
                continue
            a_j = a[idx]
            a_p = a[partner]
            if a_j < 1e-15 or a_p < 1e-15:
                continue

            # Try: reduce max(a_j, a_p) by delta, increase min by delta
            # This keeps sum the same but reduces the product
            delta = 0.01 * min(a_j, a_p)
            if a_j >= a_p:
                a_try = a.copy()
                a_try[idx] -= delta
                a_try[partner] += delta
            else:
                a_try = a.copy()
                a_try[partner] -= delta
                a_try[idx] += delta

            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try * S / s
            c_try = fft_convolve_self(a_try)
            if c_try.max() < M * (1 - 1e-10):
                a = a_try
                improved = True
                break

        if not improved:
            # Try small random perturbations focused on peak region
            noise = np.zeros(n)
            noise[top_idxs] = rng.normal(0, 0.001 * S / np.sqrt(n), len(top_idxs))
            a_try = project_simplex(a + noise, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < M:
                a = a_try
            else:
                # Try LSE step
                beta = 1000.0
                grad, _ = lse_subgradient(a, c, beta)
                for lr in [0.001, 0.0005, 0.0001]:
                    a_try = project_simplex(a - lr * grad, S)
                    c_try = fft_convolve_self(a_try)
                    if c_try.max() < M:
                        a = a_try
                        break

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print("    Surgical iter %d: M=%.10f, elapsed=%.0fs" % (it, M, elapsed))

    return best_a, best


def generate_raised_cosine(n, center, width):
    """Generate a raised cosine bump."""
    t = np.linspace(0, 1, n)
    x = (t - center) / (width / 2.0)
    a = np.ones(n)
    mask = np.abs(x) < 1.0
    a[mask] = 0.5 * (1.0 + np.cos(np.pi * x[mask]))
    a[~mask] = 0.0
    return np.maximum(a, 1e-10)


def try_sequence(a, global_state):
    """Evaluate and track best sequence."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a, n)
    c = fft_convolve_self(a)
    val = c.max()
    if val < global_state['best']:
        global_state['best'] = val
        global_state['a'] = a.copy()
        print("  New best: %.10f (n=%d)" % (val, n))
    return val


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Uses LP-based direction finding, LSE-FISTA, and surgical peak refinement.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 30.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))

    global_state = {'best': np.inf, 'a': None}

    # Load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded given sequence: length=%d" % len(given))
    except Exception:
        pass

    if given is None:
        given = np.ones(4096, dtype=np.float64)
        print("WARNING: No given sequence, using uniform fallback")

    n_given = len(given)

    # === Phase 1: Quick LSE-FISTA on given ===
    remaining = deadline - time.monotonic()
    t1 = min(120.0, remaining * 0.08)
    print("=== Phase 1: LSE-FISTA on given (n=%d, %.0fs) ===" % (n_given, t1))
    a1, v1 = run_lse_fista(given, n_given, t1, rng,
                           beta_start=100.0, beta_end=5000.0, beta_steps=100)
    try_sequence(a1, global_state)

    # === Phase 2: LP-based optimization on best ===
    if global_state['a'] is not None:
        remaining = deadline - time.monotonic()
        t2 = remaining * 0.45
        n_lp = len(global_state['a'])
        print("=== Phase 2: LP optimization (n=%d, %.0fs) ===" % (n_lp, t2))
        a2, v2 = run_lp_optimize(global_state['a'], n_lp, t2, rng)
        try_sequence(a2, global_state)

    # === Phase 3: Surgical peak refinement ===
    if global_state['a'] is not None:
        remaining = deadline - time.monotonic()
        t3 = remaining * 0.2
        n_s = len(global_state['a'])
        print("=== Phase 3: Surgical refinement (n=%d, %.0fs) ===" % (n_s, t3))
        a3, v3 = surgical_peak_refinement(global_state['a'], n_s, t3, rng)
        try_sequence(a3, global_state)

    # === Phase 4: Try larger n ===
    if global_state['a'] is not None:
        remaining = deadline - time.monotonic()
        if remaining > 180:
            n_large = len(global_state['a']) * 2
            if n_large <= 32768:
                S_large = np.sqrt(2.0 * n_large)
                t_old = np.linspace(0, 1, len(global_state['a']))
                t_new = np.linspace(0, 1, n_large)
                init_large = np.interp(t_new, t_old, global_state['a'])
                init_large = normalize_sum(init_large, n_large)

                t4 = remaining * 0.3
                print("=== Phase 4: Larger n=%d (%.0fs) ===" % (n_large, t4))

                # Quick LSE first
                t4a = t4 * 0.5
                a4, v4 = run_lse_fista(init_large, n_large, t4a, rng,
                                       beta_start=50.0, beta_end=10000.0, beta_steps=100)
                try_sequence(a4, global_state)

                # Then LP
                remaining4 = deadline - time.monotonic()
                if remaining4 > 30 and global_state['a'] is not None:
                    n_cur = len(global_state['a'])
                    if n_cur == n_large or global_state['best'] < v4:
                        t4b = min(remaining4 * 0.5, 120.0)
                        a4b, v4b = run_lp_optimize(global_state['a'], n_cur, t4b, rng)
                        try_sequence(a4b, global_state)

    # === Phase 5: Try different initializations at original n ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_state['a'] is not None:
        n_try = len(global_state['a'])
        t5 = remaining * 0.25
        print("=== Phase 5: Diverse inits (n=%d, %.0fs) ===" % (n_try, t5))

        inits = []
        t = np.linspace(0, 1, n_try)
        # Asymmetric raised cosines
        for cf in [0.3, 0.35, 0.4, 0.45, 0.5]:
            for wf in [0.4, 0.5, 0.6]:
                inits.append(generate_raised_cosine(n_try, cf, wf))
        # Gaussian bumps
        for mu in [0.35, 0.4, 0.45, 0.5]:
            for sig in [0.12, 0.15, 0.2]:
                inits.append(np.exp(-0.5 * ((t - mu) / sig) ** 2))

        # Score inits
        scores = []
        for i, init in enumerate(inits):
            init = normalize_sum(init, n_try)
            c = fft_convolve_self(init)
            scores.append((c.max(), i))
        scores.sort()

        per_init = max(5.0, t5 / min(8, len(scores)))
        improved_count = 0
        for score, idx in scores[:8]:
            if time.monotonic() > deadline - 30:
                break
            if per_init < 5:
                break
            init = inits[idx]
            print("  Trying init %d (initial max=%.4f)" % (idx, score))
            ai, vi = run_lse_fista(init, n_try, per_init, rng,
                                   beta_start=30.0, beta_end=20000.0, beta_steps=80)
            before = global_state['best']
            try_sequence(ai, global_state)
            if global_state['best'] < before:
                improved_count += 1

    # === Phase 6: Final LP polish on absolute best ===
    if global_state['a'] is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            n_f = len(global_state['a'])
            t6 = remaining * 0.7
            print("=== Phase 6: Final LP polish (n=%d, %.0fs) ===" % (n_f, t6))
            a6, v6 = run_lp_optimize(global_state['a'], n_f, t6, rng)
            try_sequence(a6, global_state)

    # === Phase 7: Ultra-fine peak shaving ===
    if global_state['a'] is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_p = len(global_state['a'])
            t7 = remaining * 0.8
            print("=== Phase 7: Ultra-fine refinement (%.0fs) ===" % t7)
            a7, v7 = surgical_peak_refinement(global_state['a'], n_p, t7, rng)
            try_sequence(a7, global_state)

    # Final verification
    if global_state['a'] is not None:
        final_a = np.clip(global_state['a'], 0.0, 1000.0)
        final_list = [float(x) for x in final_a]
        try:
            ver = evaluate_sequence(final_list)
            print("Final: evaluate_sequence = %.10f, internal = %.10f (n=%d)" %
                  (ver if np.isfinite(ver) else 999.0, global_state['best'], len(final_a)))
            if not np.isfinite(ver):
                print("  evaluate_sequence rejected! Using fallback.")
                global_state['a'] = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
                global_state['best'] = 2.0
        except Exception as e:
            print("  Evaluation error: %s" % str(e))
    else:
        print("WARNING: No valid result, using uniform fallback.")
        global_state['a'] = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        global_state['best'] = 2.0

    global_state['a'] = np.clip(global_state['a'], 0.0, 1000.0)
    if global_state['a'].sum() < 0.01:
        global_state['a'] = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.10f" % (len(global_state['a']), global_state['best']))
    return [float(x) for x in global_state['a']]
```
