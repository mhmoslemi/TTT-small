from scipy.fft import fft, ifft

def fft_full_convolve(a):
    """Compute full self-convolution via FFT for a sequence of length n."""
    n = len(a)
    A = fft(a, n + n)
    conv_result = np.real(ifft(A * np.conj(A), n + n - 1))
    return conv_result[:2 * n - 1]

def project_simplex_from_sum(v, S):
    """Project a vector onto the simplex with sum S and non-negative elements."""
    v = np.asarray(v, dtype=np.float64)
    s = np.sum(v)
    if s < 1e-15:
        return np.clip(v, 0, 1000) * S / np.clip(np.sum(np.clip(v, 0, 1000)), 1e-15, None)
    scale = S / s
    new_v = np.clip(np.asarray(v) * scale, 0, 1000)
    return new_v * (S / np.sum(new_v))

def compute_normalized_objective(a):
    """Compute the normalized objective 2*n*max_conv / (sum(a))^2 with clipping."""
    n = len(a)

    return 2.0 * n * conv.max() / (s * s), [float(x) for x in clipped]
