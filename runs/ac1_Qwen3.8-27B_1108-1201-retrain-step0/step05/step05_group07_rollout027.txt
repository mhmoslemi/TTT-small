    """Compute gradient of LSE-smoothed max of self-convolution."""
    n = len(a)
    c = fft_convolve_self(a)
    # LSE weights
    scaled = beta * (c - c.max())
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)

    Ar = fft(ar, fsize)
    conv_res = np.real(ifft(W * Ar))
    grad = 2.0 * conv_res[n-1 : 2*n-1]

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses log-parameterization (a=exp(v)) with LSE-smoothed gradients,
adaptive beta scheduling, multi-length exploration, and periodic restarts.
Key improvements over previous: unconstrained optimization in log-space
avoids simplex projection artifacts, and the LSE gradient is computed
efficiently via a single FFT-based cross-correlation.
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_smoothed_obj_and_grad(a, beta):
    """
    Compute LSE-smoothed objective and gradient w.r.t. a.
    Returns (lse_value, grad_a) where grad_a has the same shape as a.
    """
    n = len(a)
    c = fft_convolve_self(a)
    cmax = c.max()
    
    # LSE value
    scaled = beta * (c - cmax)
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)
    lse_val = cmax + np.log(e.sum()) / beta
    
    # Weights
    w = e / e.sum()
    
    # Gradient: grad[i] = 2 * conv(w, a[::-1])[i + n - 1]
    ar = a[::-1]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    Ar = fft(ar, fsize)
    conv_res = np.real(ifft(W * Ar))
    grad = 2.0 * conv_res[n-1 : 2*n-1]
    
    return lse_val, grad, c


def obj_value(a):
    """Compute the actual objective."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def optimize_multiplicative(a0, S_target, max_time, rng, beta_start=5.0, beta_end=10000.0, verbose=True):
    """
    Multiplicative update optimization.
    a[i] <- a[i] * exp(-eta * grad[i]), then normalize.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a * S_target / a.sum()
    
    best_obj = np.inf
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 300
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = beta_start * np.exp(np.log(beta_end / beta_start) * progress)
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        S = a.sum()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > patience:
            # Add relative noise
            noise = 1.0 + rng.normal(0, min(0.1, 0.02 * stuck / patience), n)
            a = a * np.maximum(noise, 1e-5)
            a = a * S_target / a.sum()
            stuck = 0
            patience = int(patience * 1.2)
            continue
        
        # LSE weights
        scaled = beta * (c - cur_max)
        scaled = np.clip(scaled, -500, 0)
        e = np.exp(scaled)
        w = e / e.sum()
        
        # Gradient w.r.t. a
        ar = a[::-1]
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        Ar = fft(ar, fsize)
        conv_res = np.real(ifft(W * Ar))
        grad = 2.0 * conv_res[n-1 : 2*n-1]
        
        # Normalize gradient for multiplicative update
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        gap = cur_max - c.mean()
        eta = gap / (S * gnorm + 1e-10)
        eta = min(eta, 0.5)
        
        # Multiplicative update
        log_factor = np.clip(-eta * grad, -5, 5)
        a = a * np.exp(log_factor)
        a = np.maximum(a, 1e-10)
        a = a * S_target / a.sum()
        
        if verbose and it % 2000 == 0:
            print(f"      it={it}, t={elapsed:.0f}s, beta={beta:.1f}, obj={best_obj:.6f}")

    old_n = len(a)
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses log-parameterization (a=exp(v)) with LSE-smmoothed gradients computed
via FFT cross-correlation, adaptive beta scheduling, multiplicative updates,

optimization in log-space avoids simplex projection artifacts, and gradient is
computed via a single FFT-based cross-correlation for efficiency.

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_a(a, w):
    """
    Compute gradient of LSE-smoothed max conv w.r.t. a.
    w: LSE weights over convolution positions (length 2n-1).
    grad[i] = 2 * conv(w, a[::-1])[i + n - 1]
    """
    n = len(a)
    ar = a[::-1]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    Ar = fft(ar, fsize)
    conv_res = np.real(ifft(W * Ar))
    return 2.0 * conv_res[n - 1:2 * n - 1]


def compute_lse_weights(c, beta):
    """Compute LSE (softmax) weights over convolution values."""
    cmax = c.max()
    scaled = beta * (c - cmax)
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / s

def generate_smooth_init(n, rng):
    """Generate a smooth non-negative initial sequence."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.05
    # Add small random perturbation
    a = a * (1 + 0.05 * rng.standard_normal(n))
    a = np.maximum(a, 0.01)
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing
    2n*max(conv(a,a))/(sum(a))^2.
    Phases: log-param optimization, multiplicative refinement,
    multi-length exploration, final high-beta polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    print(f"Starting optimization, budget={budget_s}s")

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 256:
            print(f"Using provided sequence: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def do_update(a, label):
        nonlocal global_best, global_a
        if a is None or len(a) < 2:
            return
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  NEW BEST: obj={obj:.6f} [{label}] n={len(a)}")

    # === Phase 1: Primary optimization on provided sequence ===
    if a_provided is not None:
        n1 = len(a_provided)
        print(f"\nPhase 1: Log-param optimization on provided, n={n1}")
        t1 = min(500, (deadline - time.monotonic()) * 0.25)
        a_opt, _ = optimize_log_param(a_provided, t1, rng,
                                       beta_start=5.0, beta_end=5000.0, verbose=True)
        do_update(a_opt, "P1 log-param")

        # Phase 1b: Multiplicative refinement
        remaining = deadline - time.monotonic()
        t1b = min(300, remaining * 0.2)
        if t1b > 30 and global_a is not None:
            n1b = len(global_a)
            print(f"Phase 1b: Multiplicative refinement, n={n1b}")
            a_opt2, _ = optimize_multiplicative(global_a, t1b, rng,
                                                 beta_start=20.0, beta_end=15000.0, verbose=False)
            do_update(a_opt2, "P1b multiplicative")

    # === Phase 2: Multi-length exploration ===
    lengths_to_try = [4096, 8192, 16384]
    if global_a is not None:
        base_n = len(global_a)
        candidates = [base_n, base_n // 2, base_n * 2, 4096, 8192, 16384, 32768]
        candidates = [x for x in candidates if 1024 <= x <= 32768]
        lengths_to_try = sorted(set(candidates), key=lambda x: abs(x - base_n))

    for target_n in lengths_to_try[:5]:
        if time.monotonic() > deadline - 80:
            break
        if global_a is not None and target_n == len(global_a):
            continue

        S = np.sqrt(2.0 * target_n)
        t_alloc = min(250, (deadline - time.monotonic()) * 0.2)
        if t_alloc < 40:
            continue

        print(f"\nPhase 2: Exploring n={target_n}, time={t_alloc:.0f}s")

        # Initialize from global best (resampled) or smooth init
        if global_a is not None:
            init = resample_sequence(global_a, target_n)
        else:
            init = generate_smooth_init(target_n, rng)
        init = np.maximum(init, 1e-10)

        # Log-param optimization
        t_log = t_alloc * 0.55
        a_opt, _ = optimize_log_param(init, t_log, rng,
                                       beta_start=5.0, beta_end=5000.0, verbose=False)
        do_update(a_opt, f"P2 n={target_n} log")

        rem = deadline - time.monotonic()
        if rem > 40:
            t_mult = min(100, rem * 0.35)
            best_for_this = a_opt if global_a is None or len(a_opt) == target_n else global_a
            a_opt2, _ = optimize_multiplicative(a_opt, t_mult, rng,
                                                 beta_start=50.0, beta_end=30000.0, verbose=False)
            do_update(a_opt2, f"P2 n={target_n} mult")

    # === Phase 3: Intensive refinement with alternating methods ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 80:
        n = len(global_a)
        print(f"\nPhase 3: Intensive refinement, n={n}, remaining={remaining:.0f}s")

        a_curr = global_a.copy()
        n_rounds = min(8, int(remaining / 60) + 1)
        t_per = (remaining - 20) / n_rounds

        for round_i in range(n_rounds):
            if time.monotonic() > deadline - 15:
                break
            t_phase = min(t_per, deadline - time.monotonic() - 10)
            if t_phase < 15:
                break

            strategy = round_i % 4
            if strategy == 0:
                a_curr, _ = optimize_log_param(a_curr, t_phase, rng,
                                                beta_start=50.0, beta_end=50000.0, verbose=False)
            elif strategy == 1:
                a_curr, _ = optimize_multiplicative(a_curr, t_phase, rng,
                                                     beta_start=50.0, beta_end=50000.0, verbose=False)
            elif strategy == 2:
                # Perturb then log-optimize
                noise_scale = 0.03 * (1.0 - round_i / max(1, n_rounds))
                a_pert = a_curr * np.exp(rng.normal(0, noise_scale, n))
                a_pert = np.maximum(a_pert, 1e-10)
                a_curr, _ = optimize_log_param(a_pert, t_phase, rng,
                                                beta_start=20.0, beta_end=30000.0, verbose=False)
            else:
                # Larger perturbation + hybrid
                noise_scale = 0.08 * (1.0 - round_i / max(1, n_rounds))
                a_pert = a_curr * np.exp(rng.normal(0, noise_scale, n))
                a_pert = np.maximum(a_pert, 1e-10)
                a_curr, _ = optimize_hybrid(a_pert, t_phase, rng,
                                             beta_start=10.0, beta_end=20000.0, verbose=False)

            do_update(a_curr, f"P3 round {round_i} strat={strategy}")

    # === Phase 4: Final high-beta polish ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        n = len(global_a)
        print(f"\nPhase 4: Final high-beta polish, n={n}, remaining={remaining:.0f}s")
        t_pol = min(remaining - 10, 180)
        a_pol, _ = optimize_log_param(global_a, t_pol, rng,
                                       beta_start=200.0, beta_end=100000.0, verbose=False)
        do_update(a_pol, "P4 final polish")

        # One more multiplicative pass
        rem2 = deadline - time.monotonic()
        if rem2 > 20:
            a_pol2, _ = optimize_multiplicative(a_pol, min(rem2 - 5, 100), rng,
                                                 beta_start=500.0, beta_end=100000.0, verbose=False)
            do_update(a_pol2, "P4b final mult")

    # === Return ===
    if global_a is None:
        print("WARNING: No valid solution found, using fallback")
        global_a = np.ones(256)
        global_best = obj_value(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print(f"\n{'=' * 60}")
    print(f"FINAL: len={len(global_a)}, internal_obj={global_best:.6f}")
    print(f"evaluate_sequence verdict: {ver:.6f}")
    print(f"{'=' * 60}")

    return final_list
