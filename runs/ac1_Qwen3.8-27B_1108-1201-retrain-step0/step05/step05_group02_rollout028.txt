Strategy: 
1. Start from parent or diverse initializations
2. Mirror descent (multiplicative weights) with exact top-k subgradient
3. FISTA with log-sum-exp smoothing for refinement
4. Coordinate descent for fine polish

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_topk(a, c, k):
    """Compute subgradient from top-k lags of the convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    g = np.zeros(n)
    for kk in idx:
        kk = int(kk)
        # dc[kk]/da[j] = 2*a[kk-j] if 0<=kk-j<n
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if hi >= lo:
            g[lo:hi+1] += 2.0 * a[kk - hi:kk - lo + 1][::-1]
    g /= k
    return g

def lse_fista(a_init, max_time, rng, beta_init=50.0, beta_max=50000.0):

    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    def lse_grad(y, beta):
        c = fft_convolve_self(y)
        cm = c.max()
        shifted = beta * (c - cm)
        shifted = shifted - shifted.max()
        e = np.exp(shifted)
        tot = e.sum()
        if tot < 1e-300:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / tot
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        return w, grad, c
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.005 ** it))

            best_a = y.copy()
            last_improve = it

            break

Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy:
1. Multi-start diverse initialization with quick screening
2. Mirror descent (multiplicative weights) with exact top-k subgradient
3. FISTA with log-sum-exp smoothing at multiple beta levels
4. Simplified coordinate descent for local refinement
5. Final high-precision polish

Key innovation: Uses the exact subgradient structure (shifted signal) 
and multiplicative weight updates naturally suited to the simplex constraint.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_topk(a, c, k):
    """Compute exact subgradient from top-k lags using the shifted-signal structure."""
    n = len(a)
    k = min(k, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    g = np.zeros(n)
    for kk in top_idx:
        kk = int(kk)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if hi >= lo:
            g[lo:hi + 1] += 2.0 * a[kk - hi:kk - lo + 1][::-1]
    g /= k
    return g


def lse_subgradient(a, beta):
    """Compute LSE-based smooth subgradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + np.log(max(tot, 1e-300)) / beta
    return lse_val, grad, c, w

            y = project_simplex(y + rng.normal(0, ns, n), S)

    return best_a, best_obj


def run_coord_descent(a_init, max_time, rng, max_pairs=8):
    """Simplified coordinate descent via mass transfer."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 15000:
            break

        # Find top lags and their contributing coordinates
        top_k = min(3, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]

        multi_contrib = np.zeros(n)
        for kk in top_idx:
            kk = int(kk)
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            if hi >= lo:
                multi_contrib[lo:hi + 1] += a[kk - hi:kk - lo + 1][::-1]

        # Source: high contribution, large value
        # Target: low contribution, small value
        src_scores = multi_contrib * np.maximum(a, 0.001)
        tgt_scores = -multi_contrib * (S / n - a + 0.001)

        src_candidates = np.argsort(src_scores)[::-1][:max_pairs]
        tgt_candidates = np.argsort(tgt_scores)[:max_pairs]

        best_delta_obj = cur_obj + 1e-15
        best_move = None

        for j in src_candidates[:max_pairs // 2]:
            for l in tgt_candidates[:max_pairs // 2]:
                j, l = int(j), int(l)
                if j == l or a[j] < 1e-10:
                    continue
                delta = a[j] * 0.05
                if delta < 1e-12:
                    continue

                # Compute change in c
                delta_c = np.zeros(2 * n - 1)
                js = j
                je = min(2 * n - 2, j + n - 1)
                lo_j = max(0, js - j)
                hi_j = min(n - 1, je - j)
                if hi_j >= lo_j:
                    delta_c[js:je + 1] -= delta * a[lo_j:hi_j + 1]

                ls = l
                le = min(2 * n - 2, l + n - 1)
                lo_l = max(0, ls - l)
                hi_l = min(n - 1, le - l)
                if hi_l >= lo_l:
                    delta_c[ls:le + 1] += delta * a[lo_l:hi_l + 1]

                new_max = np.max(c + delta_c)
                new_obj = 2.0 * n * new_max / (S * S)
                if new_obj < best_delta_obj:
                    best_delta_obj = new_obj
                    best_move = (j, l, delta)

        if best_move is not None:
            j, l, delta = best_move
            a[j] -= delta
            a[l] += delta
            # Update c
            js = j
            je = min(2 * n - 2, j + n - 1)
            lo_j = max(0, js - j)
            hi_j = min(n - 1, je - j)
            if hi_j >= lo_j:
                c[js:je + 1] -= delta * a[lo_j:hi_j + 1]
            ls = l
            le = min(2 * n - 2, l + n - 1)
            lo_l = max(0, ls - l)
            hi_l = min(n - 1, le - l)
            if hi_l >= lo_l:
                c[ls:le + 1] += delta * a[lo_l:hi_l + 1]
        else:
            if stuck > 500:
                idx = int(rng.integers(0, n))
                delta = a[idx] * 0.01
                if delta > 1e-12:
                    a[idx] -= delta
                    a[int(rng.integers(0, n))] += delta
                    c = fft_convolve_self(a)

        if it % 200 == 0:
            c = fft_convolve_self(a)

    return best_a, best_obj


def run_frank_wolfe(a_init, max_time, rng):
    """Frank-Wolfe (conditional gradient) method."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    last_improve = 0
    it = 0
    k = 1

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 5000:
            break
        k = min(32, 1 + stuck // 200)

        g = subgradient_topk(a, c, k)
        # Linear minimization over simplex: argmin g[i] -> vertex S*e_{i*}
        i_star = int(np.argmin(g))

        # Try multiple step sizes
        best_t = 0.0
        best_obj_t = cur_obj
        for t in [1.0, 0.5, 0.33, 0.2, 0.1, 0.05]:
            a_trial = (1 - t) * a + t * S * np.zeros(n)
            a_trial[i_star] += t * S
            a_trial = np.maximum(a_trial, 0)
            ts = a_trial.sum()
            if ts < 1e-15:
                continue
            a_trial = a_trial * (S / ts)
            obj_t = compute_obj(a_trial)
            if obj_t < best_obj_t:
                best_obj_t = obj_t
                best_t = t

        if best_t > 0:
            a = (1 - best_t) * a + best_t * S * np.zeros(n)
            a[i_star] += best_t * S
            a = np.maximum(a, 0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
        elif stuck > 500:
            ns = max(1e-5, 0.005 * S / np.sqrt(n) * np.exp(-stuck / 3000))
            a = project_simplex(a + rng.normal(0, ns, n), S)
            c = fft_convolve_self(a)

    return best_a, best_obj


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.15 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 4 * tt + 0.7)
    elif method == "multi_tone":
        a = (0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt) +
             0.2 * np.sin(2 * np.pi * 7 * tt + 1.3) +
             0.15 * np.sin(2 * np.pi * 11 * tt + 2.1))
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 2.0 * np.abs(tt - 0.5))
    elif method == "white":
        a = np.ones(n)
    elif method == "triangle":
        a = np.abs(np.sin(np.pi * tt))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search combining mirror descent, FISTA, Frank-Wolfe,
    and coordinate descent with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point from parent
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    n0 = len(a_start) if a_start is not None else 8192

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Parent: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        a_start = make_init(n0, "cosine", rng)
        obj0 = compute_obj(a_start)
        print("Initial: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: Diverse initialization + quick MD screening (12% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.12
    if p1_time > 25:
        print("\n=== Phase 1: Diverse MD screening (budget %.0fs) ===" % p1_time)
        methods = ["gaussian", "multi_tone", "random", "triangle", "chirp"]
        per_time = p1_time / len(methods)

        for method in methods:
            if time.monotonic() >= deadline:
                break
            a_div = make_init(n0, method, rng)
            a_opt, obj_opt = run_mirror_descent(a_div, per_time * 0.8, rng,
                                                eta_init=0.03, k_init=1, k_max=32)
            tag = " ***BEST***" if obj_opt < global_best_obj else ""
            print("  %s: obj=%.6f%s" % (method, obj_opt, tag))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 2: Intensive Mirror Descent (30% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.30
    if p2_time > 30:
        print("\n=== Phase 2: Intensive Mirror Descent (budget %.0fs) ===" % p2_time)

        configs = [
            (0.02, 1, 64),
            (0.05, 1, 128),
            (0.1, 2, 256),
        ]
        per_cfg = p2_time / len(configs)
        for eta_i, k_i, k_m in configs:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            a_opt, obj_opt = run_mirror_descent(global_best_a, min(per_cfg, rem - 5),
                                                rng, eta_init=eta_i, k_init=k_i, k_max=k_m)
            tag = " ***BEST***" if obj_opt < global_best_obj else ""
            print("  MD(eta=%.2f,k=%d..%d): obj=%.6f%s" % (eta_i, k_i, k_m, obj_opt, tag))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 3: FISTA with LSE (20% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.20
    if p3_time > 25:
        print("\n=== Phase 3: FISTA LSE (budget %.0fs) ===" % p3_time)

        beta_configs = [(20, 0.005, 20000), (100, 0.003, 80000), (500, 0.002, 200000)]
        per_cfg = p3_time / len(beta_configs)
        for bi, br, bm in beta_configs:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            a_opt, obj_opt = run_fista_lse(global_best_a, min(per_cfg, rem - 5), rng,
                                           beta_init=bi, beta_rate=br, beta_max=bm)
            tag = " ***BEST***" if obj_opt < global_best_obj else ""
            print("  FISTA(bi=%d): obj=%.6f%s" % (bi, obj_opt, tag))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 4: Coordinate Descent (20% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.20
    if p4_time > 20:
        print("\n=== Phase 4: Coordinate Descent (budget %.0fs) ===" % p4_time)
        a_cd, obj_cd = run_coord_descent(global_best_a, p4_time * 0.6, rng, max_pairs=8)
        tag = " ***BEST***" if obj_cd < global_best_obj else ""
        print("  CD: obj=%.6f%s" % (obj_cd, tag))
        if obj_cd < global_best_obj:
            global_best_obj = obj_cd
            global_best_a = a_cd.copy()

        # Follow up with MD on CD result
        rem = deadline - time.monotonic()
        if rem > 15:
            a_md, obj_md = run_mirror_descent(global_best_a, rem * 0.6, rng,
                                              eta_init=0.01, k_init=2, k_max=64)
            tag = " ***BEST***" if obj_md < global_best_obj else ""
            print("  MD after CD: obj=%.6f%s" % (obj_md, tag))
            if obj_md < global_best_obj:
                global_best_obj = obj_md
                global_best_a = a_md.copy()

    # === Phase 5: Frank-Wolfe + final polish (18% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.18
    if p5_time > 15:
        print("\n=== Phase 5: FW + Final Polish (budget %.0fs) ===" % p5_time)

        # Frank-Wolfe
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fw, obj_fw = run_frank_wolfe(global_best_a, rem * 0.35, rng)
            tag = " ***BEST***" if obj_fw < global_best_obj else ""
            print("  FW: obj=%.6f%s" % (obj_fw, tag))
            if obj_fw < global_best_obj:
                global_best_obj = obj_fw
                global_best_a = a_fw.copy()

        # Final fine MD
        rem = deadline - time.monotonic()
        if rem > 10:
            a_pol, obj_pol = run_mirror_descent(global_best_a, rem * 0.45, rng,
                                                eta_init=0.005, k_init=4, k_max=512)
            tag = " ***BEST***" if obj_pol < global_best_obj else ""
            print("  Fine MD: obj=%.6f%s" % (obj_pol, tag))
            if obj_pol < global_best_obj:
                global_best_obj = obj_pol
                global_best_a = a_pol.copy()

        # Final high-beta FISTA
        rem = deadline - time.monotonic()
        if rem > 10:
            a_f2, obj_f2 = run_fista_lse(global_best_a, rem, rng,
                                         beta_init=2000, beta_rate=0.001, beta_max=500000)
            tag = " ***BEST***" if obj_f2 < global_best_obj else ""
            print("  High-beta FISTA: obj=%.6f%s" % (obj_f2, tag))
            if obj_f2 < global_best_obj:
                global_best_obj = obj_f2
                global_best_a = a_f2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.6f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
