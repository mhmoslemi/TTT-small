Strategy: RU algorithm with reduced LP (active-set) for main optimization,
followed by LSE-FISTA polish and peak shaving.
Key insight: The LP subproblem in the RU iteration has a small number of variables
(the active set size) when the convolution is already well-optimized, making it
fast to solve. This allows many iterations within the time budget.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import sys

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def solve_dual_lp(a, active_lags, M):
    """
    Solve the reduced dual LP:
    min M * sum(y_k) s.t. sum_{k in active} y_k * a_{k-j} >= 1 for all j, y >= 0
    
    Returns y (length len(active)) or None if infeasible.
    """
    n = len(a)
    m = len(active_lags)
    
    # Build constraint matrix A_mat: A_mat[j, i] = a[active_lags[i] - j] if 0 <= active_lags[i] - j < n
    # This is n x m
    A_mat = np.zeros((n, m))
    for i, k in enumerate(active_lags):
        # For j = 0, ..., n-1: a[k-j] is valid when 0 <= k-j < n
        # i.e., max(0, k-n+1) <= j <= min(k, n-1)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            # j = lo, lo+1, ..., hi
            # a[k-j] = a[k-lo], a[k-lo-1], ..., a[k-hi]
            # = a[k-hi : k-lo+1][::-1]
            A_mat[lo:hi+1, i] = a[lo:hi+1][::-1]
    
    # LP: min M * sum(y) s.t. A_mat @ y >= 1, y >= 0
    # Convert to standard form for linprog: min c^T y s.t. A_ub @ y <= b_ub, y >= 0
    c_obj = M * np.ones(m)
    A_ub = -A_mat
    b_ub = -np.ones(n)
    
    try:

                      bounds=[(0, None)] * m, maxiter=1000)
        if res.status == 0:
            return res.x
        elif res.status == 2:  # infeasible
            return None
        else:

    except Exception:

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Combine RU algorithm (LP-based active set iteration) for coarse optimization,
exact subgradient refinement for the "last mile," and adaptive LSE-FISTA polish.
The key insight is that the LP subproblem in RU has few variables (active set size),
making it fast to solve and allowing many iterations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_correlate(a, b):
    """Full cross-correlation: result[k] = sum_j a[j] * b[k-j] (valid range)."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix numerical issues
    s = result.sum()
    if s > 0:
        result = result * (S / s)
    return result


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def build_dual_matrix(a, active_lags):
    """
    Build n x m matrix where A[j, i] = a[active_lags[i] - j] if 0 <= active_lags[i]-j < n.
    This represents the cross-correlation structure for the reduced dual LP.
    """
    n = len(a)
    m = len(active_lags)
    A = np.zeros((n, m), dtype=np.float64)
    for i in range(m):
        k = int(active_lags[i])
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[lo:hi + 1, i] = a[hi:lo - 1: -1]
    return A


def solve_ru_subproblem(a, n, active_lags, M):
    """
    Solve the reduced dual LP to get a direction for the RU update.
    min M * sum(y) s.t. A^T y >= 1, y >= 0
    Returns primal direction b (non-negative vector of length n) or None.
    """
    m = len(active_lags)
    if m == 0:
        return None
    if m > 300:
        # Take the top 300
        c = fft_convolve_self(a)
        active_lags = active_lags[np.argsort(c[active_lags])[-300:]]
        m = 300

    A_mat = build_dual_matrix(a, active_lags)

    # LP: min c^T y s.t. A_ub @ y <= b_ub, 0 <= y
    c_obj = M * np.ones(m)
    A_ub = -A_mat
    b_ub = -np.ones(n)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs',
                      bounds=[(0, None)] * m)
        if res.status == 0:
            y_star = res.x
            h = A_mat @ y_star
            # Complementary slackness: b_j > 0 where h_j ≈ 1
            tol = 1e-4
            b = (h <= 1.0 + tol).astype(np.float64)
            if b.sum() < 1:
                b = np.ones(n, dtype=np.float64)
            return b
        else:
            return None
    except Exception:
        return None


def ru_step(a, n, S, active_tol=0.003, max_active=200):
    """
    One step of the RU algorithm.
    Returns the updated sequence.
    """
    c = fft_convolve_self(a)
    M = c.max()

    # Find active set
    threshold = M - active_tol
    active_mask = c >= threshold
    active_lags = np.where(active_mask)[0]

    if len(active_lags) == 0:
        active_lags = np.array([np.argmax(c)])

    if len(active_lags) > max_active:
        # Keep only the top max_active
        idx = np.argsort(c[active_lags])[::-1][:max_active]
        active_lags = active_lags[idx]

    # Solve subproblem
    b = solve_ru_subproblem(a, n, active_lags, M)

    if b is None or b.sum() < 1e-10:
        return a

    # Rescale b to have sum S
    b_scaled = b * (S / b.sum())
    b_scaled = np.maximum(b_scaled, 0.0)
    b_scaled = project_simplex(b_scaled, S)

    # Update with line search
    best_a = a.copy()
    best_val = M
    for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
        a_new = (1 - t) * a + t * b_scaled
        a_new = project_simplex(a_new, S)
        c_new = fft_convolve_self(a_new)
        val_new = c_new.max()
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()

    return best_a


def subgradient_refine(a, n, S, max_time, rng, n_active_start=1, n_active_end=20):
    """
    Exact subgradient refinement using adaptive active set.
    Computes the exact subgradient of max(conv(a,a)) and uses it for refinement.
    """
    a = a.copy()
    start = time.monotonic()
    best_val = np.inf
    best_a = a.copy()
    it = 0
    stagnation = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_val = c.max()

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 500:
            # Escape: inject noise
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnation / 10000.0)
            if noise_scale < 1e-10:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stagnation = 0
            continue

        # Adaptive active set size
        progress = min(1.0, it / 5000.0)
        n_active = int(n_active_start + (n_active_end - n_active_start) * progress)

        # Find active lags
        cm = c.max()
        if n_active == 1:
            active = np.array([np.argmax(c)])
        else:
            # Top n_active lags within tolerance
            tol = 0.01 / (1 + it * 0.001)
            active = np.where(c >= cm - tol)[0]
            if len(active) > n_active:
                active = active[np.argsort(c[active])[-n_active:]]
            if len(active) == 0:
                active = np.array([np.argmax(c)])

        # Compute subgradient: average of gradients at active lags
        # Gradient of (a*a)_k w.r.t. a is 2 * a[k-::-1] (reversed and shifted)
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a[hi:lo - 1: -1]
        grad /= len(active)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: adaptive
        # Use Armijo-like condition
        base = cur_val
        step = S / (gnorm + 1e-10) * 0.1
        improved = False
        for shrink in range(10):
            a_new = project_simplex(a - step * grad / (step + 1e-10) * step, S)
            # Simpler: just use a fixed small step
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < base * (1 - 1e-8):
                a = a_new
                improved = True
                break
            step *= 0.5

        if not improved:
            # Very small step
            a = project_simplex(a - 1e-6 * grad / (np.linalg.norm(grad) + 1e-10), S)

    return best_a, best_val


def lse_fista(a, n, S, max_time, rng, beta_start=100.0, beta_end=30000.0, beta_steps=100):
    """LSE-FISTA for polishing."""
    a_work = a.copy()
    y = a_work.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_val = np.inf
    best_a = a_work.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 80000.0)

        c = fft_convolve_self(y)
        cur_val = c.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = y.copy()
            last_improve = it

        # LSE gradient
        cm = c.max()
        # Scale to avoid overflow
        scaled = beta * (c - cm)
        scaled = np.minimum(scaled, 500.0)
        e = np.exp(scaled)
        tot = e.sum()
        if tot < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = e / tot

        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta

        # Stagnation
        stagnant = it - last_improve
        if stagnant > 300:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 20000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a_work)
        y = np.maximum(y, 0.0)
        a_work = x_new
        t_mom = t_new

    # Final check
    c_final = fft_convolve_self(a_work)
    if c_final.max() < best_val:
        best_val = c_final.max()
        best_a = a_work.copy()

    return best_a, best_val


def peak_shave(a, n, S, n_steps=200, lr_base=0.003):
    """Surgical peak shaving: reduce elements that contribute most to the peak."""
    a = a.copy()
    for step_i in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()

        # Responsibility: for each element j, how much does it contribute to top peaks?
        top_k = np.argmax(c)
        # For peak at top_k, contribution of a[j] is a[j]*a[top_k-j]
        lo = max(0, top_k - n + 1)
        hi = min(top_k, n - 1)
        js = np.arange(lo, hi + 1)
        partners = top_k - js
        contrib = a[js] * a[partners]

        # Also consider second peak
        c_second = c.copy()
        c_second[top_k] = -np.inf
        top_k2 = np.argmax(c_second)
        lo2 = max(0, top_k2 - n + 1)
        hi2 = min(top_k2, n - 1)
        js2 = np.arange(lo2, hi2 + 1)
        partners2 = top_k2 - js2
        contrib2 = a[js2] * a[partners2]

        resp = np.zeros(n)
        np.add.at(resp, js, contrib)
        np.add.at(resp, js2, contrib2 * 0.5)

        resp_max = resp.max()
        if resp_max < 1e-30:
            break

        lr = lr_base * (1.0 + step_i * 0.01)
        a = a * np.exp(-lr * resp / resp_max)
        a = project_simplex(a, S)

    return a

        a_new = ru_step(a, n, S, active_tol=0.002 + it * 0.00001, max_active=150)
        c_new = fft_convolve_self(a_new)
        val_new = c_new.max()

        if val_new < ru_best_val - 1e-10:
            ru_best_val = val_new
            ru_stagnation = 0
            a = a_new.copy()
            if val_new < global_best:
                global_best = val_new
                global_a = a.copy()
                print("  RU iter %d: new global best %.8f" % (it, global_best))
        else:
            ru_stagnation += 1
            # Even if not improving globally, accept if locally better
            if val_new < global_best + 0.001:
                a = a_new.copy()

        if ru_stagnation > 200:
            # Restart with perturbation
            noise = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)
            ru_stagnation = 0
            if it % 100 == 0:
                print("  RU iter %d: stagnation restart (val=%.8f)" % (it, c_new.max()))

        if it % 100 == 0:
            print("  RU iter %d: val=%.8f" % (it, c_new.max()))

    # === Phase 2: Subgradient refinement (25% of budget) ===
    remaining = deadline - time.monotonic()
    t_sub = remaining * 0.25
    print("=== Phase 2: Subgradient refinement (%.0fs) ===" % t_sub)
    a_ref, val_ref = subgradient_refine(global_a, n, S, t_sub, rng)
    if val_ref < global_best:
        global_best = val_ref
        global_a = a_ref.copy()
        print("  Subgradient: new best %.8f" % global_best)
    else:
        print("  Subgradient: best %.8f (no improvement)" % val_ref)

    # === Phase 3: LSE-FISTA polish (20% of budget) ===
    remaining = deadline - time.monotonic()
    t_lse = remaining * 0.25
    print("=== Phase 3: LSE-FISTA polish (%.0fs) ===" % t_lse)
    a_lse, val_lse = lse_fista(global_a, n, S, t_lse, rng,
                                beta_start=500.0, beta_end=50000.0, beta_steps=80)
    if val_lse < global_best:
        global_best = val_lse
        global_a = a_lse.copy()
        print("  LSE: new best %.8f" % global_best)
    else:
        print("  LSE: best %.8f (no improvement)" % val_lse)

    # === Phase 4: Multi-start search (10% of budget) ===
    remaining = deadline - time.monotonic()
    t_ms = remaining * 0.3
    if t_ms > 30:
        print("=== Phase 4: Multi-start search (%.0fs) ===" % t_ms)
        a_ms, val_ms = multi_start_search(global_a, n, S, t_ms, rng, n_starts=4)
        if val_ms < global_best:
            global_best = val_ms
            global_a = a_ms.copy()
            print("  Multi-start: new best %.8f" % global_best)

    # === Phase 5: Peak shaving + final polish (remaining) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("=== Phase 5: Peak shaving + final polish (%.0fs) ===" % remaining)
        a_shave = peak_shave(global_a, n, S, n_steps=100, lr_base=0.002)
        val_shave = fft_convolve_self(a_shave).max()
        if val_shave < global_best:
            global_best = val_shave
            global_a = a_shave.copy()
            print("  Shaved: new best %.8f" % global_best)

        # Final LSE after shaving
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            a_final, val_final = lse_fista(global_a, n, S, remaining2 * 0.6, rng,
                                            beta_start=5000.0, beta_end=80000.0, beta_steps=40)
            if val_final < global_best:
                global_best = val_final
                global_a = a_final.copy()
                print("  Final LSE: new best %.8f" % global_best)

    # Final verification
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    ver = evaluate_sequence(final_list)
    print("Final: evaluate_sequence = %.8f, internal = %.8f (n=%d)" %
          (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
