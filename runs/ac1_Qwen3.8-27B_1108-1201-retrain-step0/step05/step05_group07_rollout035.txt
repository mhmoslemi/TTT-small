```python
import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    # Project v onto {x: x >= 0, sum(x) = S}
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, c, beta):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a.
    c = conv(a,a), length 2n-1.
    Returns gradient of length n.
    """
    n = len(a)
    # Softmax weights over lags
    c_max = c.max()
    scaled = beta * (c - c_max)
    e = np.exp(np.clip(scaled, -50, 50))
    w = e / e.sum()
    
    # Compute grad[j] = sum_m a[m] * (w[m+j] + w[j-m])
    # using FFT
    # w is defined on [0, 2n-2]
    # S1[j] = sum_m a[m]*w[m+j] = cross-correlation
    # S2[j] = sum_m a[m]*w[j-m] = convolution (truncated)
    
    # For S1: ifft(conj(fft(a, L)) * fft(w, L))[j]
    # For S2: ifft(fft(a, L) * fft(w, L))[j]
    # Need L >= 3n-2 for no circular aliasing
    
    L = 1 << (3*n - 2).bit_length()  # next power of 2 >= 3n-2
    
    A = fft(a, L)
    W = fft(w, L)
    
    S1 = np.real(ifft(np.conj(A) * W))[:n]
    S2 = np.real(ifft(A * W))[:n]
    
    return S1 + S2

def obj_from_conv(n, max_b, S):
```

```python
def lse_grad(a, c, beta):
    """Gradient of LSE-smoothed max of c w.r.t. a."""
    n = len(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    e = np.exp(np.clip(scaled, -50, 50))
    w = e / e.sum()
    
    L = 1 << (3*n - 2).bit_length()
    A = fft(a, L)
    W = fft(w, L)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad
```

```python
    c = fft_convolve_self(a)

        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best - 1e-14:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Adaptive beta
        progress = (time.monotonic() - start) / max_time
        beta = beta_init * np.exp(np.log(beta_final/beta_init) * progress)
        # Or: beta = beta_init + (beta_final - beta_init) * progress^2
        
        grad = lse_grad(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: gap-based
        gap = cur_max - c.mean()
        if gap < 1e-14:
            break
        
        step = gap / (gnorm * gnorm)
        step = min(step, 2.0 / gnorm)
        
        # Backtracking line search
        accepted = False
        for _ in range(20):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max + 1e-12:
                a = a_try
                accepted = True
                break
            step *= 0.5
        
        if not accepted:
            # Try multiplicative update as fallback
            a_new = a * np.exp(-0.01 * grad / max(1, gnorm))
            a = project_simplex(a_new, S)

        if stuck > patience:
            noise_scale = min(0.05, 0.001 * stuck / patience) * S / np.sqrt(n)

        if it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    it={it}, t={elapsed:.0f}s, max={best:.8f}, beta={beta:.1f}")
```

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses LSE-smoothed gradient descent with adaptive beta, multiplicative updates,
multiple initializations, and multi-scale refinement.
"""
import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_grad(a, c, beta):
    n = len(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    e = np.exp(np.clip(scaled, -50, 50))
    w = e / e.sum()
    L = 1 << (3 * n - 2).bit_length()
    A = fft(a, L)
    W = fft(w, L)
    return 2.0 * np.real(ifft(np.conj(A) * W))[:n]

def optimize_LSE(a0, S, max_time, rng, beta_init=20.0, beta_max=10000.0, verbose=True):
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 300
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best - 1e-14:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = beta_init * (beta_max / beta_init) ** progress
        
        grad = lse_grad(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        gap = cur_max - c.mean()
        if gap < 1e-13:
            break
        
        # Step size
        step = gap / (gnorm * gnorm)
        step = min(step, 3.0 / gnorm)
        
        # Backtracking
        accepted = False
        for _ in range(15):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max + 1e-12:
                a = a_try
                accepted = True
                break
            step *= 0.4
        
        if not accepted:
            # Multiplicative update fallback
            eta = 0.01
            a_new = a * np.exp(-eta * grad / max(1.0, gnorm))
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-10:
                a = a_new * (S / s)
            else:
                a = a  # keep current
        
        # Noise injection
        if stuck > patience:
            scale = min(0.03, 0.002 * stuck / patience) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, scale, n), S)
            stuck = 0
            patience = int(patience * 1.15)
        
        if verbose and it % 2000 == 0:
            print(f"    it={it}, t={elapsed:.0f}s, max={best:.8f}, beta={beta:.0f}, stuck={stuck}")
    
    return best_a, best

    deadline = time.monotonic() + max(0, budget_s - 30)
    
    print(f"Starting with budget={budget_s}s")
    
    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 100:
            print(f"Provided: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except:
        pass
    
    global_best = np.inf
    global_a = None
    
    def update_global(a, msg=""):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  NEW BEST: obj={obj:.6f} ({msg})")
    
    # Phase 1: Optimize provided sequence
    if a_provided is not None:
        n = len(a_provided)
        S = np.sqrt(2.0 * n)
        a = a_provided * (S / a_provided.sum())
        
        t1 = min(600, (deadline - time.monotonic()) * 0.3)
        print(f"\nPhase 1: LSE optimize, n={n}, S={S:.2f}, time={t1:.0f}s")
        a_opt, _ = optimize_LSE(a, S, t1, rng, beta_init=10, beta_max=5000, verbose=True)
        update_global(a_opt, "Phase 1")
    
    # Phase 2: Try different initializations at same length
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        
        initializations = []
        # Hann window
        t = np.linspace(0, 1, n)
        initializations.append(0.5 * (1 + np.cos(2*np.pi*t)))
        # Inverted Hann
        initializations.append(0.5 * (1 - np.cos(2*np.pi*t)))
        # Raised cosine (different)
        initializations.append(np.sin(np.pi * t) ** 2)
        # Uniform
        initializations.append(np.ones(n))
        # Bimodal
        initializations.append(np.exp(-((t-0.5)**2) / (2*0.1**2)))
        
        remaining = deadline - time.monotonic()
        t_per = min(200, remaining * 0.15)
        
        for i, init in enumerate(initializations[:3]):  # Try top 3
            if time.monotonic() > deadline - 60:
                break
            if t_per < 20:
                break
            print(f"\nPhase 2.{i}: Init={i}, n={n}, time={t_per:.0f}s")
            a_init = project_simplex(init, S)
            a_opt, _ = optimize_LSE(a_init, S, t_per, rng, beta_init=5, beta_max=3000, verbose=False)
            update_global(a_opt, f"Phase 2 init {i}")
    
    # Phase 3: Try different lengths
    for n_try in [4096, 16384]:
        if time.monotonic() > deadline - 120:
            break
        if global_a is not None and len(global_a) == n_try:
            continue
        S = np.sqrt(2.0 * n_try)
        t_alloc = min(300, (deadline - time.monotonic()) * 0.2)
        if t_alloc < 30:
            continue
        print(f"\nPhase 3: n={n_try}, time={t_alloc:.0f}s")
        # Interpolate from global best
        if global_a is not None:
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n_try)
            a_init = np.interp(t_new, t_old, global_a)
        else:
            a_init = np.ones(n_try)
        a_init = project_simplex(np.maximum(a_init, 0), S)
        a_opt, _ = optimize_LSE(a_init, S, t_alloc, rng, beta_init=10, beta_max=5000, verbose=False)
        update_global(a_opt, f"Phase 3 n={n_try}")
