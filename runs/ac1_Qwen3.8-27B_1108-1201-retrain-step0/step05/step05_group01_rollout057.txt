"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses exact active-set subgradients with FISTA, Adam variant,
multi-n search, periodic perturbations, and L-BFGS polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_conv(a, b):
    """Full linear convolution via FFT."""
    size = len(a) + len(b) - 1
    fsize = 1 << (size - 1).bit_length()
    return np.real(ifft(fft(a, fsize) * fft(b, fsize)))[:size]


def self_conv(a):
    """Full self-convolution via FFT."""
    return fft_conv(a, a)


def simplex_project(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    # Standard simplex projection
    if S <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * c.max() / (s * s)


def active_subgradient(y, c, tol=0.0):
    """Compute subgradient from active (near-max) lags.
    Returns gradient vector of length n."""
    n = len(y)
    M = c.max()
    if tol > 0:
        active = np.where(c >= M * (1.0 - tol))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    y_rev = y[::-1]
    conv_wy = fft_conv(w, y_rev)
    return 2.0 * conv_wy[n-1:2*n-1]


def lse_subgradient(y, c, beta):
    """Compute LSE-weighted subgradient."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    w = e / e.sum()
    y_rev = y[::-1]
    conv_wy = fft_conv(w, y_rev)
    return 2.0 * conv_wy[n-1:2*n-1], cm + np.log(e.sum()) / beta


def run_fista_exact(a_init, budget, rng, beta0=50.0, use_adam=False):
    """FISTA with exact active-set subgradient and adaptive step."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / beta0
    
    # Adam state
    if use_adam:
        m_v = np.zeros(n)
        v_v = np.zeros(n)
        lr = 0.01
    
    while time.monotonic() - start < budget:
        it += 1
        c = self_conv(y)
        M = c.max()
        cur_obj = 2.0 * n * M / (S * S)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Choose subgradient
        if stuck < 200:
            grad = active_subgradient(y, c, tol=0.0)
        elif stuck < 1000:
            grad = active_subgradient(y, c, tol=0.001 * stuck)
        else:
            grad, _ = lse_subgradient(y, c, max(10.0, stuck * 0.1))
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        if use_adam:
            m_v = 0.9 * m_v + 0.1 * grad
            v_v = 0.999 * v_v + 0.001 * grad**2
            m_hat = m_v / (1 - 0.9**it)
            v_hat = v_v / (1 - 0.999**it)
            y_new = y - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            y_new = simplex_project(y_new, S)
            # Periodic lr decay
            if it % 2000 == 0:
                lr *= 0.5
        else:
            # Adaptive step with backtracking
            if stuck > 500:
                step *= 0.95  # Shrink step when stuck
            elif stuck < 100:
                step = min(step * 1.01, 1.0 / 5.0)  # Grow step
            
            y_new = simplex_project(y - step * grad, S)
            
            # Nesterov momentum
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2))
            y_next = y_new + ((t_mom - 1.0) / t_new) * (y_new - a)
            y_next = np.maximum(y_next, 0.0)
            a = y_new
            y = y_next
            t_mom = t_new
        
        # Periodic perturbation when very stuck
        if stuck > 3000 and it % 500 == 0:
            noise_scale = max(0.001, 0.01 * np.exp(-stuck / 10000.0)) * S / np.sqrt(n)
            y = simplex_project(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            a = y.copy()
        
        # Momentum reset
        if it % 3000 == 0:
            t_mom = 1.0
    
    return best_a, best_obj


def run_adam(a_init, budget, rng):

    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    lr = 0.05
    
    while time.monotonic() - start < budget:
        it += 1
        c = self_conv(a)
        M = c.max()
        cur_obj = 2.0 * n * M / (S * S)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        if stuck < 500:
            grad = active_subgradient(a, c, tol=0.0)
        else:
            grad, _ = lse_subgradient(a, c, max(10.0, stuck * 0.05))
        
        m_v = 0.9 * m_v + 0.1 * grad
        v_v = 0.999 * v_v + 0.001 * grad**2
        m_hat = m_v / (1 - 0.9**it)
        v_hat = v_v / (1 - 0.999**it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = simplex_project(a_new, S)
        
        # Adaptive learning rate
        if stuck > 1000:
            lr *= 0.99
        elif stuck > 5000:
            # Large jump
            noise_scale = 0.01 * S / np.sqrt(n)
            a = simplex_project(a + rng.normal(0, noise_scale, n), S)
            m_v = np.zeros(n)
            v_v = np.zeros(n)
        
        if it % 10000 == 0:
            lr = max(0.001, lr * 0.5)
    
    return best_a, best_obj


def run_lbfgs(a_init, budget, rng):
    """L-BFGS-B polish with LSE objective."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    betas = [50.0, 200.0, 1000.0]
    for beta in betas:
        if time.monotonic() - start > budget:
            break
        
        def obj_grad(x):
            x = np.maximum(x, 0.0)
            c = self_conv(x)
            cm = c.max()
            shifted = beta * (c - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / beta
            # Add penalty for sum constraint
            sum_x = x.sum()
            pen = 10.0
            obj += pen * (sum_x - S)**2
            grad = active_subgradient(x, c, tol=0.0)
            # Actually use LSE gradient
            y_rev = x[::-1]
            conv_wy = fft_conv(w, y_rev)
            grad = 2.0 * conv_wy[n-1:2*n-1]
            grad += 2.0 * pen * (sum_x - S)
            return obj, grad
        
        t0 = time.monotonic()
        result = scipy_minimize(obj_grad, a.copy(), method='L-BFGS-B',
                                  jac=True, bounds=[(0, 1000)]*n,
                                  options={'maxiter': 50, 'maxls': 20,
                                           'ftol': 1e-14, 'gtol': 1e-10,
                                           'iprint': -1})
        a_new = np.clip(result.x, 0, 1000)
        a_new = a_new * (S / max(a_new.sum(), 1e-15))
        obj_new = compute_obj(a_new)
        
        print("    L-BFGS beta=%.0f: obj=%.8f (t=%.1fs)" % (beta, obj_new, time.monotonic()-t0))
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_obj

    """Find Frank-Wolfe direction: max sum(g) s.t. f*g <= M, g >= 0.
    Uses only top-K binding constraints for tractability."""
    n = len(f)
    c = self_conv(f)
    # Find top-K binding constraints
    max_val = c.max()

    # Actually, we want constraints that are most "binding" for the direction
    # The most binding are those where f*f is closest to M
    active = np.argsort(c)[::-1][:K]
    
    # Build LP: maximize sum(g) subject to:
    # For each k in active: (f*g)[k] <= M
    # g >= 0
    
    # (f*g)[k] = sum_i f[i] * g[k-i] (for valid i, k-i)
    # This is a linear constraint on g
    
    # Build constraint matrix A such that A @ g <= M * ones(K)
    A = np.zeros((K, n))
    for idx, k in enumerate(active):
        # (f*g)[k] = sum_i f[i] * g[k-i] where 0<=k-i<n
        # = sum_i f[i] * g[k-i]
        # For g[j]: coefficient is f[k-j] (if 0 <= k-j < n)
        for j in range(n):
            if 0 <= k - j < n:
                A[idx, j] = f[k - j]
    
    # Solve LP: maximize c^T g subject to A g <= b, g >= 0
    # In scipy linprog: minimize -c^T g subject to A g <= b, -I g <= 0 (or bounds)
    c_obj = -np.ones(n)  # Maximize sum(g) = minimize -sum(g)
    b = np.full(K, M)
    from scipy.optimize import linprog
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if result.status != 0:
        return None
    
    return result.x

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Combines Frank-Wolfe (LP oracle with top-K binding constraints),
FISTA with exact active-set subgradients, adaptive step sizing,
and L-BFGS polishing. Multi-n search with adaptive time allocation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize, linprog


def fft_conv(a, b):
    """Full linear convolution via FFT."""
    size = len(a) + len(b) - 1
    fsize = 1 << (size - 1).bit_length()
    return np.real(ifft(fft(a, fsize) * fft(b, fsize)))[:size]


def self_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:2 * n - 1]


def simplex_project(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection on threshold."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find rho: largest index where u[rho] > (cssv[rho] - S) / (rho+1)
    idx = np.arange(1, n + 1, dtype=np.float64)
    rho_candidates = np.where(u * idx > (cssv - S))[0]
    if len(rho_candidates) == 0:
        return np.maximum(v - v.max(), 0.0) * S / max(v.max() * n, 1e-300)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * c.max() / (s * s)


def compute_grad_from_weights(y, w):
    """Gradient of sum_k w[k]*(y*y)[k] w.r.t. y.
    grad[j] = 2 * sum_k w[k] * y[k-j] (valid indices)
    = 2 * (w * y_rev)[n-1 + j] where y_rev = y[::-1]
    """
    n = len(y)
    y_rev = y[::-1]
    conv_wy = fft_conv(w, y_rev)
    return 2.0 * conv_wy[n - 1:2 * n - 1]


def active_subgradient(y, c, tol_frac=0.0):
    """Subgradient using lags within tolerance of max."""
    M = c.max()
    if M < 1e-30:
        return np.zeros(len(y))
    if tol_frac > 0:
        active = np.where(c >= M * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    return compute_grad_from_weights(y, w)


def lse_subgradient(y, c, beta):
    """LSE-smoothed subgradient and objective."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    w = e / e.sum()
    grad = compute_grad_from_weights(y, w)
    obj = cm + np.log(e.sum()) / beta
    return grad, obj


def solve_fw_direction(f, M, K=15):
    """Solve Frank-Wolfe LP: max sum(g) s.t. (f*g)[k] <= M for top-K lags, g >= 0."""
    n = len(f)
    c = self_conv(f)
    # Top-K lags (most binding)
    top_k = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix: A[idx, j] = f[k - j] if 0 <= k-j < n
    A = np.zeros((K, n))
    for idx, k in enumerate(top_k):
        # For g[j]: coefficient is f[k-j] if 0 <= k-j < n
        valid_j = np.arange(max(0, k - n + 1), min(n, k + 1))
        A[idx, valid_j] = f[k - valid_j]
    
    # LP: minimize -sum(g) subject to A*g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'presolve': True, 'time_limit': 60})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None

def run_fista_adaptive(a_init, budget, rng, mode="exact"):

        a = a * (S / s)

    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 0.05  # Initial step size

        it += 1
        c = self_conv(y)
        M = c.max()
        cur_obj = 2.0 * n * M / (S * S)
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Choose gradient based on stuck-ness
        if mode == "exact" or stuck < 300:
            tol = 0.0 if stuck < 100 else min(0.05, 0.0005 * stuck)
            grad = active_subgradient(y, c, tol_frac=tol)
        else:

        if stuck < 50:

        elif stuck < 500:
            step = step * 0.999
        else:
            step = step * 0.995
        step = max(step, 0.001)
        
        # FISTA update
        x_new = simplex_project(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Ensure sum is correct

        if sy > 1e-15:
            y = y * (S / sy)
        a = x_new
        t_mom = t_new
        
        # Periodic momentum reset
        if it % 2000 == 0:
            t_mom = 1.0
        
        # Large perturbation when very stuck
        if stuck > 5000 and it % 1000 == 0:
            noise_scale = max(0.002, 0.02 * np.exp(-stuck / 20000.0)) * S / np.sqrt(n)
            y = simplex_project(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            a = y.copy()
            if it % 5000 == 0:
                print("    [FISTA] it=%d obj=%.8f (perturbed, stuck=%d)" % (it, best_obj, stuck))

    lr = 0.02
    
    print("    [Adam] Start: obj=%.8f" % best_obj)
    
    while time.monotonic() - start < budget:
        it += 1
        c = self_conv(a)
        M = c.max()
        cur_obj = 2.0 * n * M / (S * S)
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        if stuck < 300:
            grad = active_subgradient(a, c, tol_frac=0.0)
        else:
            beta = min(2000.0, 10.0 + 0.02 * stuck)
            grad, _ = lse_subgradient(a, c, beta)
        
        m_v = 0.9 * m_v + 0.1 * grad
        v_v = 0.999 * v_v + 0.001 * grad * grad
        m_hat = m_v / (1.0 - 0.9 ** it)
        v_hat = v_v / (1.0 - 0.999 ** it)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = simplex_project(a_new, S)
        
        # Adaptive LR
        if stuck > 2000:
            lr *= 0.995
        if stuck > 8000 and it % 2000 == 0:
            noise_scale = 0.005 * S / np.sqrt(n)
            a = simplex_project(a + rng.normal(0, noise_scale, n), S)
            m_v.fill(0)
            v_v.fill(0)
        
        lr = max(lr, 0.0005)
        
        if it % 5000 == 0:
            print("    [Adam] it=%d obj=%.8f lr=%.5f" % (it, best_obj, lr))
    
    c_final = self_conv(a)
    final_obj = 2.0 * n * c_final.max() / (S * S)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()
    
    print("    [Adam] Done: obj=%.8f (iters=%d, t=%.0fs)" % (best_obj, it, time.monotonic() - start))
    return best_a, best_obj


def run_lbfgs_polish(a_init, budget, rng):
    """L-BFGS-B with LSE objective and sum penalty."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    configs = [(30.0, 5.0), (100.0, 10.0), (500.0, 20.0)]
    
    for beta, pen in configs:
        if time.monotonic() - start > budget:
            break
        remaining = budget - (time.monotonic() - start)
        if remaining < 10:
            break
        
        def obj_grad(x):
            x = np.maximum(x, 0.0)
            c = self_conv(x)
            cm = c.max()
            shifted = beta * (c - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / beta
            sum_x = x.sum()
            obj += pen * (sum_x - S) ** 2
            grad = compute_grad_from_weights(x, w)
            grad += 2.0 * pen * (sum_x - S)
            return obj, grad
        
        t0 = time.monotonic()
        try:
            result = scipy_minimize(obj_grad, a.copy(), method='L-BFGS-B',
                                     jac=True, bounds=[(0, 1000)] * n,
                                     options={'maxiter': 30, 'maxls': 15,
                                              'ftol': 1e-15, 'gtol': 1e-10,
                                              'iprint': -1})
            a_new = np.clip(result.x, 0, 1000)
            a_new = a_new * (S / max(a_new.sum(), 1e-15))
            obj_new = compute_obj(a_new)
            print("    [L-BFGS] beta=%.0f: obj=%.8f (t=%.1fs, nit=%d)" %
                  (beta, obj_new, time.monotonic() - t0, result.nit))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new.copy()
        except Exception as ex:
            print("    [L-BFGS] error: %s" % str(ex))
    
    return best_a, best_obj


def run_coordinate_descent(a_init, budget, rng):
    """Pairwise coordinate descent: move mass between pairs to reduce peak."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    print("    [CoordDesc] Start: obj=%.8f" % best_obj)
    
    # Build list of positions sorted by value (we'll move from high to low)
    while time.monotonic() - start < budget:
        it += 1
        c = self_conv(a)
        M = c.max()
        cur_obj = 2.0 * n * M / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        stuck = it - last_improve
        if stuck > 500:
            break
        
        # Compute gradient
        tol = 0.0 if stuck < 50 else min(0.03, 0.0001 * stuck)
        grad = active_subgradient(a, c, tol_frac=tol)
        
        # Find best pair: source (positive grad) and target (negative grad)
        # Use top-k for efficiency
        k = max(10, n // 100)
        pos_idx = np.argsort(grad)[::-1][:k]  # Most positive (want to decrease)
        neg_idx = np.argsort(grad)[:k]  # Most negative (want to increase)
        
        best_improvement = 0
        best_move = None
        
        # Try top pairs
        for i in pos_idx[:min(5, len(pos_idx))]:
            if a[i] < 1e-10:
                continue
            for j in neg_idx[:min(5, len(neg_idx))]:
                if i == j:
                    continue
                # Move small amount from i to j
                delta = min(a[i] * 0.01, S * 0.001)
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                # Quick objective check
                c_try = self_conv(a_try)
                obj_try = 2.0 * n * c_try.max() / (S * S)
                if obj_try < cur_obj - 1e-14:
                    improvement = cur_obj - obj_try
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_move = (i, j, delta)
        
        if best_move is not None:
            i, j, delta = best_move
            # Try to increase delta
            for mult in [5, 10, 20, 50]:
                d2 = min(delta * mult, a[i])
                a_try = a.copy()
                a_try[i] -= d2
                a_try[j] += d2
                c_try = self_conv(a_try)
                obj_try = 2.0 * n * c_try.max() / (S * S)
                if obj_try < cur_obj - 1e-14:
                    delta = d2
                else:
                    break
            a[i] -= delta
            a[j] += delta
        else:
            # No improvement found, add small noise
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-stuck / 5000.0)
            if noise_scale > 1e-10:
                a = simplex_project(a + rng.normal(0, noise_scale, n), S)
            else:
                break
    
    c_final = self_conv(a)
    final_obj = 2.0 * n * c_final.max() / (S * S)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()
    
    print("    [CoordDesc] Done: obj=%.8f (iters=%d, t=%.0fs)" % (best_obj, it, time.monotonic() - start))
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-method: Frank-Wolfe (LP), FISTA (exact subgradient),
    Adam, coordinate descent, and L-BFGS polish. Multi-n search.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)
    
    global_best_obj = np.inf
    global_best_a = None
    
    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None
    
    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Initial: n=%d, obj=%.8f" % (len(a_start), obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n_fb = 8192
        t = np.linspace(0, 1, n_fb, endpoint=False)
        a_start = 0.54 + 0.46 * np.cos(2 * np.pi * t)
        obj0 = compute_obj(a_start)
        print("Fallback init: n=%d, obj=%.8f" % (n_fb, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    
    # Determine n values to try
    n_values = []
    if a_start is not None:
        n_values.append(len(a_start))
    for n_try in [8192, 12288, 16384]:
        if n_try not in n_values:
            n_values.append(n_try)
    
    print("N values to try: %s" % n_values)
    
    # Allocate time: give more time to the n that already has the best result
    n_allocations = {}
    base_time = (deadline - time.monotonic()) * 0.85 / len(n_values)
    for n in n_values:
        n_allocations[n] = base_time
    # Extra time for the n with current best
    if a_start is not None and len(a_start) in n_allocations:
        n_allocations[len(a_start)] += base_time * 0.5
    
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 180:
            break
        
        n_budget = min(n_allocations.get(n, remaining * 0.3), remaining * 0.7)
        if n_budget < 120:
            continue
        
        print("\n========== n=%d (budget %.0fs) ==========" % (n, n_budget))
        S = np.sqrt(2.0 * n)
        
        # Generate initial points
        inits = []
        
        # Resized from a_start
        if a_start is not None and len(a_start) == n:
            inits.append(a_start.copy())
        elif a_start is not None:
            t_new = np.linspace(0, 1, n, endpoint=False)
            t_old = np.linspace(0, 1, len(a_start), endpoint=False)
            a_r = np.interp(t_new, t_old, a_start)
            a_r = np.maximum(a_r, 0)
            if a_r.sum() > 1e-15:
                a_r = a_r * (S / a_r.sum())
                inits.append(a_r)
        
        # Current global best (padded or truncated)
        if global_best_a is not None:
            if len(global_best_a) == n:
                inits.append(global_best_a.copy())
            else:
                # Resample
                t_new = np.linspace(0, 1, n, endpoint=False)
                t_old = np.linspace(0, 1, len(global_best_a), endpoint=False)
                a_g = np.interp(t_new, t_old, global_best_a)
                a_g = np.maximum(a_g, 0)
                if a_g.sum() > 1e-15:
                    a_g = a_g * (S / a_g.sum())
                    inits.append(a_g)
        
        # Cosine
        t = np.linspace(0, 1, n, endpoint=False)
        a_cos = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a_cos = np.maximum(a_cos, 0.01)
        a_cos = a_cos * (S / a_cos.sum())
        inits.append(a_cos)
        
        # Limit to 2 inits to save time
        inits = inits[:2]
        
        # Strategy: FW -> FISTA -> CoordDesc -> L-BFGS
        time_fw = n_budget * 0.30
        time_fista = n_budget * 0.35
        time_coord = n_budget * 0.20
        time_lbfgs = n_budget * 0.15
        
        for init_idx, init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            time_left = deadline - time.monotonic()
            if time_left < 60:
                break
            
            print("  Init %d:" % init_idx)
            
            # Phase 1: Frank-Wolfe
            if time_fw > 30:
                t0 = time.monotonic()
                a_fw, obj_fw = run_frank_wolfe(init, min(time_fw, time_left * 0.3), rng, K=12)
