def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]

    idx = np.arange(1, n + 1, dtype=np.float64)

    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def multi_term_descent(a, max_time, rng, K_init=3, K_max=20):
    """
    Direct multi-term gradient descent.
    At each step, targets the top K terms of the convolution.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0, np.inf).copy()
    a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    last_improve = 0
    K = K_init
    
    while time.monotonic() - start < max_time:
        it += 1
        stuck = it - last_improve
        
        # Adapt K based on stagnation
        if stuck < 100:
            K = K_init
        elif stuck < 500:
            K = min(K_max, K_init + stuck // 100)
        elif stuck < 2000:
            K = K_max
        else:
            # Reset with perturbation
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            c = fft_convolve_self(a)
            last_improve = it
            K = K_init
            if stuck > 5000:
                break
        
        # Find top K terms
        top_idx = np.argsort(c)[::-1][:K]
        M = c[top_idx[0]]
        
        # Compute multi-term gradient
        grad = np.zeros(n)
        for kk in top_idx:
            kk = int(kk)
            lo = max(0, kk - n + 1)
            hi = min(n - 1, kk)
            if hi >= lo:
                grad[lo:hi+1] += a[kk - hi:kk - lo + 1][::-1]
        # Multiply by 2 (gradient of b[k] w.r.t. a is 2*a[k-j])
        grad *= 2.0
        
        # Center the gradient (ensure sum = 0)
        grad -= np.mean(grad)
        
        # Compute step size
        gnorm = np.sqrt(np.sum(grad**2))
        if gnorm < 1e-15:
            break
        
        # Barzili-Borwein type step size
        # Or simple: step = alpha * S / gnorm
        step = S / (gnorm * (1.0 + 0.01 * stuck))
        
        # Backtracking line search
        t = step
        for _ in range(10):
            a_trial = project_simplex(a - t * grad, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = 2.0 * n * c_trial.max() / (S * S)
            if obj_trial < best_obj + 1e-12:
                break
            t *= 0.5
        else:
            # No improvement; try a random perturbation
            if stuck > 50:
                idx1 = rng.integers(0, n)
                idx2 = rng.integers(0, n)
                if idx1 != idx2:
                    delta = min(a[idx1], 0.01 * S / n)
                    if delta > 1e-15:
                        a[idx1] -= delta
                        a[idx2] += delta
                        c = fft_convolve_self(a)
            continue
        
        a = a_trial
        c = c_trial
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
    
    return best_a, best_obj

    lo2 = max(0, l)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute the objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def compute_obj_from_conv(a, c):
    """Compute objective given precomputed convolution."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def lse_grad(a, beta):
    """Compute LSE(beta * conv(a,a)) value and gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        lse_val = cm
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j]
    size_need = len(w) + n - 1
    fsize = 1 << (size_need - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    cc = np.real(ifft(W * np.conj(A)))
    grad = 2.0 * cc[:n]
    return lse_val, grad, c, w


def topk_subgradient(a, c, k):
    """Compute subgradient from top-k max terms of convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    size_need = len(w_full) + n - 1
    fsize = 1 << (size_need - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    cc = np.real(ifft(W * np.conj(A)))
    return 2.0 * cc[:n]

    y = a.copy()

        a = rng.random(n) + 0.05
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 1.5 * np.abs(tt - 0.5))
    elif method == "trapezoid":
        w = 0.3
        a = np.ones(n)
        edge = int(w * n)
        if edge > 0:
            a[:edge] = np.linspace(0.3, 1.0, edge)
            a[-edge:] = np.linspace(1.0, 0.3, edge)
    elif method == "bimodal":
        a = np.exp(-5.0 * (tt - 0.25) ** 2 / (0.1 ** 2)) + np.exp(-5.0 * (tt - 0.75) ** 2 / (0.1 ** 2))
    elif method == "asymmetric":
        a = np.exp(-2.0 * (tt - 0.3) ** 2 / (0.2 ** 2))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: diverse init -> FISTA-LSE -> exact subgradient -> pairwise -> multi-n.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point from height_sequence_1 if available
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Starting from height_sequence_1: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        a_start = make_init(n0, "cosine", rng)
        obj0 = compute_obj(a_start)
        print("No parent sequence. Starting fresh: n=%d, obj=%.6f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: Diverse initialization screening (10% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.10
    if p1_time > 30:
        print("\n=== Phase 1: Diverse starts (budget %.0fs) ===" % p1_time)
        methods = ["gaussian", "trapezoid", "bimodal", "asymmetric", "multi_tone"]
        per_time = p1_time / len(methods)

        for method in methods:
            if time.monotonic() >= deadline:
                break
            a_div = make_init(n0, method, rng)
            a_opt, obj_opt = run_fista_lse(a_div, per_time * 0.7, rng,
                                            beta_init=10, beta_rate=0.02,
                                            beta_max=5000, step_scale=0.8,
                                            use_backtrack=False)
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()
                print("  %s: obj=%.6f ***NEW BEST***" % (method, obj_opt))
            else:
                print("  %s: obj=%.6f" % (method, obj_opt))

    # === Phase 2: Intensive FISTA-LSE (25% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.25
    if p2_time > 45:
        print("\n=== Phase 2: FISTA-LSE (budget %.0fs) ===" % p2_time)

        variants = [
            (20, 0.015, 10000, 1.0, True),
            (50, 0.01, 30000, 0.7, True),
            (200, 0.005, 100000, 0.4, False),
        ]
        for i, (bi, br, bm, ss, bt) in enumerate(variants):
            if time.monotonic() >= deadline:
                break
            rem = min(p2_time * 0.35, deadline - time.monotonic())
            if rem < 5:
                break
            a_opt, obj_opt = run_fista_lse(global_best_a, rem, rng,
                                            beta_init=bi, beta_rate=br,
                                            beta_max=bm, step_scale=ss,
                                            use_backtrack=bt)
            print("  FISTA%d: obj=%.6f" % (i + 1, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 3: Exact subgradient descent (30% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.30
    if p3_time > 45:
        print("\n=== Phase 3: Exact subgradient (budget %.0fs) ===" % p3_time)
        a_sg, obj_sg = run_exact_subgrad(global_best_a, p3_time * 0.6, rng)
        print("  Subgrad: obj=%.6f" % obj_sg)
        if obj_sg < global_best_obj:
            global_best_obj = obj_sg
            global_best_a = a_sg.copy()

        remaining3 = deadline - time.monotonic()
        if remaining3 > 15:
            # Follow up with FISTA at high beta
            a_f2, obj_f2 = run_fista_lse(global_best_a, remaining3 * 0.6, rng,
                                          beta_init=500, beta_rate=0.005,
                                          beta_max=200000, step_scale=0.3,
                                          use_backtrack=True)
            print("  FISTA high-beta: obj=%.6f" % obj_f2)
            if obj_f2 < global_best_obj:
                global_best_obj = obj_f2
                global_best_a = a_f2.copy()

    # === Phase 4: Pairwise optimal transfer (20% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.20
    if p4_time > 30:
        print("\n=== Phase 4: Pairwise transfer (budget %.0fs) ===" % p4_time)
        a_pw, obj_pw = pairwise_optimal(global_best_a, p4_time, rng)
        print("  Pairwise: obj=%.6f" % obj_pw)
        if obj_pw < global_best_obj:
            global_best_obj = obj_pw
            global_best_a = a_pw.copy()

    # === Phase 5: Try larger n (10% time) ===
    remaining = deadline - time.monotonic()
    p5_time = remaining * 0.10
    if p5_time > 30 and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        print("\n=== Phase 5: Upsample to n=%d (budget %.0fs) ===" % (n2, p5_time))
        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = a_up * (S2 / a_up.sum())
        obj_up = compute_obj(a_up)
        print("  Upsampled initial: obj=%.6f" % obj_up)
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up.copy()

        a_up_opt, obj_up_opt = run_fista_lse(a_up, p5_time * 0.5, rng,
                                              beta_init=5, beta_rate=0.01,
                                              beta_max=5000, step_scale=1.0,
                                              use_backtrack=False)
        print("  FISTA upsampled: obj=%.6f" % obj_up_opt)
        if obj_up_opt < global_best_obj:
            global_best_obj = obj_up_opt
            global_best_a = a_up_opt.copy()

        remaining5 = deadline - time.monotonic()
        if remaining5 > 10:
            a_up_sg, obj_up_sg = run_exact_subgrad(global_best_a, remaining5 * 0.7, rng)
            print("  Subgrad upsampled: obj=%.6f" % obj_up_sg)
            if obj_up_sg < global_best_obj:
                global_best_obj = obj_up_sg
                global_best_a = a_up_sg.copy()

    # === Phase 6: Final polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 5:
        print("\n=== Phase 6: Final polish (budget %.0fs) ===" % remaining)
        # One more round of exact subgradient
        a_fin, obj_fin = run_exact_subgrad(global_best_a, remaining * 0.5, rng)
        print("  Final subgrad: obj=%.6f" % obj_fin)
        if obj_fin < global_best_obj:
            global_best_obj = obj_fin
            global_best_a = a_fin.copy()

        remaining6 = deadline - time.monotonic()
        if remaining6 > 3:
            a_fin2, obj_fin2 = run_fista_lse(global_best_a, remaining6, rng,
                                              beta_init=1000, beta_rate=0.002,
                                              beta_max=500000, step_scale=0.2,
                                              use_backtrack=True)
            print("  Final FISTA: obj=%.6f" % obj_fin2)
            if obj_fin2 < global_best_obj:
                global_best_obj = obj_fin2
                global_best_a = a_fin2.copy()

    # Final verification
    print("\n=== Final Result ===")
    print("Best objective (internal): %.6f, n=%d" % (global_best_obj, len(global_best_a)))

    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        # Verify with evaluate_sequence
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified: obj=%.6f" % ver)
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
