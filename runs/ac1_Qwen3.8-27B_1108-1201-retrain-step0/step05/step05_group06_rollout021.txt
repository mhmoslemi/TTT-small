        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def cross_corr_fft(d, a, N):
    D = fft(d, N)
    A = fft(a, N)
    return 2.0 * np.real(ifft(D * np.conj(A)))[:len(a)]


def lse_gradient(a, c, beta, N):
    shifted = beta * (c - c.max())
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(len(a))
    d = e / tot
    return cross_corr_fft(d, a, N)


def pnorm_gradient(a, c, p, N):
    c_safe = np.maximum(c, 1e-30)
    # Normalize to avoid overflow: use (c/cmax)^(p-1) * cmax^(p-1)
    # But for direction, scaling doesn't matter
    cmax = c.max()
    if cmax < 1e-30:
        return np.zeros(len(a))
    d = np.power(c_safe / cmax, p - 1.0)
    # Scale back for correct magnitude (optional, helps with step size)
    d = d * cmax ** (p - 1.0)
    return cross_corr_fft(d, a, N)


def make_hann(n):
    tt = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * tt / n))


def make_cosine_linear(n):
    tt = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * tt / n))


def make_chirp(n):
    tt = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(np.pi * tt * tt / (2.0 * n)))


def make_smooth_random(n, rng):
    ncomp = 6
    freqs = rng.uniform(0.2, 8.0, ncomp)
    amps = rng.uniform(0.3, 1.0, ncomp)
    phases = rng.uniform(0, 2 * np.pi, ncomp)
    tt = np.arange(n, dtype=np.float64)
    a = np.ones(n) * 2.0
    for f, amp, ph in zip(freqs, amps, phases):
        a += amp * np.cos(2.0 * np.pi * f * tt / n + ph)
    return np.maximum(a, 0.0)


def upsample_factor2(a):
    n = len(a)
    t_old = np.arange(n, dtype=np.float64)
    t_new = np.arange(2 * n, dtype=np.float64) * 0.5
    return np.interp(t_new, t_old, a)


def upsample_zeros(a):
    n = len(a)
    a_new = np.zeros(2 * n)
    a_new[0::2] = a
    return a_new


def fista_backtrack(a_init, budget, rng, beta_scheme="adaptive", verbose=True):
    """
    FISTA with backtracking line search.
    beta_scheme: 'adaptive' for LSE with increasing beta,
                 'pnorm' for p-norm with increasing p,
                 'hybrid' for combined
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    N = next_pow2(2 * n - 1)

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0

    start_time = time.monotonic()
    deadline = start_time + budget
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = 5.0
    p_cur = 4
    step = 1.0
    last_verbose = 0

    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start_time
        frac = elapsed / max(budget, 1e-10)

        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive parameters
        if beta_scheme == "adaptive":
            if frac < 0.4:
                beta = 5.0 + 45.0 * (frac / 0.4)
            else:
                beta = 50.0 + 950.0 * ((frac - 0.4) / 0.6) ** 0.7
        elif beta_scheme == "pnorm":
            if frac < 0.5:
                p_cur = int(4 + 12 * (frac / 0.5))
            else:
                p_cur = int(16 + 48 * ((frac - 0.5) / 0.5))
        else:  # hybrid
            if frac < 0.3:
                p_cur = int(4 + 12 * (frac / 0.3))
            else:
                beta = 10.0 + 190.0 * ((frac - 0.3) / 0.7) ** 0.5

        # Compute gradient
        if beta_scheme == "pnorm" or (beta_scheme == "hybrid" and frac < 0.3):
            grad = pnorm_gradient(y, c, p_cur, N)
        else:
            grad = lse_gradient(y, c, beta, N)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search (simplified: check every 10th iteration)
        if it % 10 == 1:
            # Try the step, check if objective decreased
            x_try = project_simplex(y - step * grad, S)
            c_try = fft_convolve_self(x_try)
            obj_try = 2.0 * n * c_try.max() / (S * S)
            if obj_try > cur_obj * 1.01 + 1e-10:
                step *= 0.5
            elif obj_try < cur_obj * 0.99:
                step = min(step * 1.2, gnorm / (0.1 * gnorm + 1e-10))
        else:
            # Use current step
            x_new = project_simplex(y - step * grad, S)
            # FISTA momentum
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new
            continue

        # Backtracking accepted, do FISTA step
        x_new = x_try
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Randomization for escape
        if stuck > 2000 and it % 50 == 0:
            noise_scale = max(1e-6, 0.01 * S / np.sqrt(n) * np.exp(-stuck / 30000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            step = gnorm / (0.5 * gnorm + 1e-10)

        # Periodic momentum restart
        if it % 5000 == 0:
            t_mom = 1.0

        if verbose and it - last_verbose > 20000:
            print("    FISTA it=%d obj=%.6f frac=%.2f beta=%.1f p=%d step=%.4f" % (
                it, best_obj, frac, beta, p_cur, step))
            last_verbose = it

    return best_a, best_obj


def mirror_descent_opt(a_init, budget, rng, beta_init=10.0, verbose=True):
    """Mirror descent (multiplicative weights) with LSE gradient."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    N = next_pow2(2 * n - 1)

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)

    start_time = time.monotonic()
    deadline = start_time + budget
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    last_verbose = 0

    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start_time
        frac = elapsed / max(budget, 1e-10)

        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 500:
            beta = min(5000.0, beta * 1.01)

        grad = lse_gradient(a, c, beta, N)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Mirror descent step
        step = 1.0 / (beta * gnorm + 1.0)
        step = min(step, 0.05)

        a_new = a * np.exp(-step * grad)
        a_new = normalize_seq(a_new, S)

        # Accept if improved or periodically
        c_check = fft_convolve_self(a_new)
        if c_check.max() < cur_max or it % 50 == 0:
            a = a_new
        else:
            # Try half step
            a_half = a * np.exp(-step * 0.5 * grad)
            a_half = normalize_seq(a_half, S)
            c_half = fft_convolve_self(a_half)
            if c_half.max() < cur_max:
                a = a_half

        # Randomization
        if stuck > 5000 and it % 200 == 0:
            noise = rng.normal(0, 0.02, n)
            a = normalize_seq(a * np.exp(noise * 0.05), S)
            last_improve = it

        if verbose and it - last_verbose > 10000:
            print("    MD it=%d obj=%.6f frac=%.2f beta=%.1f" % (it, best_obj, frac, beta))
            last_verbose = it

    return best_a, best_obj


def coord_descent_finishing(a_init, budget, rng, verbose=True):
    """
    Stochastic coordinate descent with sign test.
    For each random coordinate, try to decrease the objective.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    N = next_pow2(2 * n - 1)

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    c = fft_convolve_self(a)

    start_time = time.monotonic()
    deadline = start_time + budget
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_verbose = 0
    accept_count = 0

    while time.monotonic() < deadline:
        it += 1

        # Pick random coordinate
        j = rng.integers(0, n)

        # Compute gradient component at j
        # grad_j = 2 * sum_k d[k] * a[k-j]
        # Use LSE weights for direction
        beta = 50.0
        shifted = beta * (c - c.max())
        shifted -= shifted.max()
        e = np.exp(shifted)
        tot = e.sum()
        if tot > 1e-300:
            w = e / tot
        else:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0

        # grad_j = 2 * sum_k w[k] * a[k-j] where 0 <= k-j < n
        lo = max(0, j)
        hi = min(len(c), j + n)
        if lo < hi:
            grad_j = 2.0 * np.sum(w[lo:hi] * a[lo - j:hi - j])
        else:
            grad_j = 0.0

        # Try moving in the negative gradient direction
        # delta = -alpha * grad_j, with a[j] + delta >= 0
        alpha = 0.01 * S / (np.abs(grad_j) + 1e-10)
        delta = -alpha * grad_j
        if a[j] + delta < 0:
            delta = -a[j] * 0.5

        # Apply and renormalize (approximate: adjust one other coordinate)
        a_try = a.copy()
        a_try[j] += delta
        # Renormalize: adjust a random other coordinate
        if delta != 0:
            k = rng.integers(0, n)
            if k == j:
                k = (j + 1) % n
            a_try[k] -= delta
            a_try[k] = max(a_try[k], 0)
            a_try = normalize_seq(a_try, S)

        # Check improvement
        c_try = fft_convolve_self(a_try)
        if c_try.max() < c.max() - 1e-12:
            a = a_try
            c = c_try
            accept_count += 1
            cur_obj = 2.0 * n * c.max() / (S * S)
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()

        if verbose and it - last_verbose > 5000:
            rate = accept_count / max(it - last_verbose, 1)
            print("    CD it=%d obj=%.6f accept_rate=%.3f" % (it, best_obj, rate))
            last_verbose = it
            accept_count = 0

    return best_a, best_obj


def active_set_polish(a_init, budget, rng, verbose=True):
    """
    Polishing using top-k subgradient (focus on the k largest convolution values).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    N = next_pow2(2 * n - 1)

    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0

    start_time = time.monotonic()
    deadline = start_time + budget
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    last_verbose = 0

    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k: focus on more elements as we get stuck
        if stuck < 500:
            k = 1
        elif stuck < 2000:
            k = min(4, 1 + stuck // 500)
        elif stuck < 8000:
            k = min(16, 4 + (stuck - 2000) // 1000)
        else:
            k = min(64, 16 + (stuck - 8000) // 2000)

        # Top-k weights
        idx = np.argsort(c)[::-1][:k]
        w = np.zeros(len(c))
        w[idx] = 1.0 / k

        grad = cross_corr_fft(w, y, N)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = min(1.0 / gnorm, 0.01 * S / gnorm)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 5000 and it % 100 == 0:
            noise_scale = max(1e-6, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 50000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it

        if it % 5000 == 0:
            t_mom = 1.0

        if verbose and it - last_verbose > 10000:
            print("    AS it=%d obj=%.6f k=%d" % (it, best_obj, k))
            last_verbose = it

    return best_a, best_obj


def propose_candidate():
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: diverse init -> FISTA BT -> MD diversify -> CD finish.
    Primary work at n=16384.
    """
    rng = np.random.default_rng(12345)
    total_budget = 2340.0  # leave 60s buffer
    t0 = time.monotonic()
    deadline = t0 + total_budget

    global_best_obj = np.inf
    global_best_a = None

    def update_best(a, label=""):
        nonlocal global_best_obj, global_best_a
        obj = compute_obj(a)
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a.copy()
            print("  NEW BEST [%s]: %.6f (n=%d) [t=%.0fs]" % (label, obj, len(a), time.monotonic() - t0))
        return obj

    # === Step 1: Generate starting points ===
    print("=== Generating starting points ===")
    n16 = 16384
    S16 = np.sqrt(2.0 * n16)

    candidates = []
    names = []

    # From previous best (n=8192)
    try:
        a_prev = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_prev = np.clip(a_prev, 0, 1000)
        if a_prev.sum() > 0.01:
            # Direct use at n=8192
            candidates.append(normalize_seq(a_prev, np.sqrt(2.0 * len(a_prev))))
            names.append("prev_8192")
            # Upsampled to 16384 (interpolation)
            a_up1 = upsample_factor2(a_prev)
            candidates.append(normalize_seq(a_up1, S16))
            names.append("prev_interp_16384")
            # Upsampled to 16384 (zero-stuff)
            a_up2 = upsample_zeros(a_prev)
            candidates.append(normalize_seq(a_up2, S16))
            names.append("prev_zero_16384")
    except Exception:
        pass

    # Fresh n=16384 initializations
    candidates.append(normalize_seq(make_hann(n16), S16))
    names.append("hann_16384")

    candidates.append(normalize_seq(make_chirp(n16), S16))
    names.append("chirp_16384")

    candidates.append(normalize_seq(make_smooth_random(n16, rng), S16))
    names.append("smooth_rand1")

    candidates.append(normalize_seq(make_smooth_random(n16, rng), S16))
    names.append("smooth_rand2")

    candidates.append(normalize_seq(make_smooth_random(n16, rng), S16))
    names.append("smooth_rand3")

    # Quick evaluation
    print("\n=== Quick evaluation ===")
    best_init = None
    best_init_obj = np.inf
    for a_c, name in zip(candidates, names):
        obj = compute_obj(a_c)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_c
        update_best(a_c, name)
        print("  %s: %.6f" % (name, obj))

    print("Best init: %.6f" % best_init_obj)

    # Select top 3 diverse candidates for parallel-ish optimization
    # (sequential but with different starting points)
    scored = []
    for a_c, name in zip(candidates, names):
        obj = compute_obj(a_c)
        scored.append((obj, a_c, name))
    scored.sort(key=lambda x: x[0])
    top_candidates = scored[:3]  # Top 3

    # === Step 2: Main FISTA optimization on best candidate ===
    remaining = deadline - time.monotonic()
    print("\n=== Phase 1: FISTA backtracking (budget=%.0fs) ===" % remaining * 0.4)
    p1_time = remaining * 0.35
    a_f1, obj_f1 = fista_backtrack(top_candidates[0][1], p1_time, rng,
                                    beta_scheme="adaptive", verbose=True)
    update_best(a_f1, "FISTA_main")

    # === Step 3: FISTA on second candidate with p-norm ===
    remaining = deadline - time.monotonic()
    print("\n=== Phase 2: FISTA p-norm on candidate 2 (budget=%.0fs) ===" % remaining * 0.25)
    p2_time = remaining * 0.2
    if len(top_candidates) > 1:
        a_f2, obj_f2 = fista_backtrack(top_candidates[1][1], p2_time, rng,
                                        beta_scheme="pnorm", verbose=True)
        update_best(a_f2, "FISTA_pnorm")

    # === Step 4: Continue on best with LSE high-beta ===
    remaining = deadline - time.monotonic()
    print("\n=== Phase 3: LSE high-beta refinement (budget=%.0fs) ===" % remaining * 0.25)
    p3_time = remaining * 0.2
    if global_best_a is not None:
        a_f3, obj_f3 = fista_backtrack(global_best_a, p3_time, rng,
                                        beta_scheme="adaptive", verbose=True)
        update_best(a_f3, "LSE_high")

    # === Step 5: Mirror descent for diversification ===
    remaining = deadline - time.monotonic()
    print("\n=== Phase 4: Mirror descent (budget=%.0fs) ===" % remaining * 0.15)
    p4_time = remaining * 0.12
    if global_best_a is not None:
        a_md, obj_md = mirror_descent_opt(global_best_a, p4_time, rng,
                                           beta_init=20.0, verbose=True)
        update_best(a_md, "MirrorDescend")

    # === Step 6: Active set polish ===
    remaining = deadline - time.monotonic()
    print("\n=== Phase 5: Active set polish (budget=%.0fs) ===" % remaining * 0.15)
    p5_time = remaining * 0.12
    if global_best_a is not None:
        a_as, obj_as = active_set_polish(global_best_a, p5_time, rng, verbose=True)
        update_best(a_as, "ActiveSet")

    # === Step 7: Coordinate descent finishing ===
    remaining = deadline - time.monotonic()
    print("\n=== Phase 6: Coordinate descent finishing (budget=%.0fs) ===" % remaining * 0.2)
    p6_time = remaining * 0.18
    if global_best_a is not None:
        a_cd, obj_cd = coord_descent_finishing(global_best_a, p6_time, rng, verbose=True)
        update_best(a_cd, "CoordDescend")

    # === Step 8: Final FISTA on best ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_best_a is not None:
        print("\n=== Phase 7: Final FISTA (budget=%.0fs) ===" % remaining * 0.7)
        p7_time = remaining * 0.65
        a_fin, obj_fin = fista_backtrack(global_best_a, p7_time, rng,
                                          beta_scheme="adaptive", verbose=True)
        update_best(a_fin, "FinalFISTA")

    # === Final verification ===
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL RESULT ===")
            print("Objective: %.6f (n=%d)" % (ver, len(final_list)))
            print("Total time: %.0fs" % (time.monotonic() - t0))
        else:
            print("WARNING: evaluate_sequence rejected! Falling back.")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
