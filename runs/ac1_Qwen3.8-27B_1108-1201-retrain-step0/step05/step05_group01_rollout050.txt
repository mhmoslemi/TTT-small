import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full cross-convolution a*b via FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def compute_max_conv(a):
    """Compute max of self-convolution."""
    c = fft_convolve_self(a)
    return c.max()


def lse_weights(c, beta):
    """Softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient_self(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.5):
    """Top-k weighted gradient for self-convolution."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient_self(y, w_full)


def cross_grad_wrt_b(a, b, k):
    """
    Gradient of conv(a,b)[k] w.r.t. b, where len(a)=len(b)=n.
    conv(a,b)[k] = sum_i a[i]*b[k-i], so d/db[j] = a[k-j] if 0<=k-j<n.

    n = len(b)
    na = len(a)
    j_lo = max(0, k - na + 1)
    j_hi = min(k, n - 1)
    grad = np.zeros(n)
    if j_lo <= j_hi:
        js = np.arange(j_lo, j_hi + 1)
        grad[js] = a[k - js]
    return grad


def run_alternating(a_init, budget, rng, inner_iters=200, outer_steps=50):
    """
    LP-inspired alternating method.
    Alternates between:
      - Optimizing b (given a) to minimize max_k conv(a,b)[k] s.t. b>=0, sum(b)=S
      - Updating a towards the optimal b
    Uses projected subgradient with Polyak step for inner optimization.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    lower_bound = 2.0 * n / (2 * n - 1)  # theoretical lower bound on max_conv under normalization

    outer_it = 0
    while time.monotonic() - start < budget and outer_it < outer_steps * 10:
        outer_it += 1

        # Compute current max
        c_aa = fft_convolve_self(a)
        M = c_aa.max()
        cur_obj = 2.0 * n * M / S * S  # = M since S^2 = 2n
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()

        # Inner: optimize b to minimize max_k conv(a,b)[k] s.t. b>=0, sum(b)=S
        b = a.copy()  # Initialize b at current a
        b_obj = np.inf
        for inner_it in range(inner_iters):
            if time.monotonic() - start > budget:
                break
            c_ab = fft_cross_convolve(a, b)
            cur_max = c_ab.max()
            if cur_max < b_obj - 1e-14:
                b_obj = cur_max

            # Polyak step: eta = (current - lower_bound) / ||grad||^2
            k_star = int(np.argmax(c_ab))
            grad = cross_grad_wrt_b(a, b, k_star)
            gnorm2 = np.dot(grad, grad)
            if gnorm2 < 1e-30:
                break
            eta = (cur_max - lower_bound) / gnorm2
            eta = min(eta, S / (np.linalg.norm(grad) + 1e-15))
            eta = min(eta, S)  # cap step size

            b_new = project_simplex(b - eta * grad, S)
            if np.linalg.norm(b_new - b) < 1e-10 * S:
                b = b_new
                break
            b = b_new

        # Update a towards b
        t = 0.3 + 0.4 * (1.0 - np.exp(-outer_it / 10.0))
        a_new = (1.0 - t) * a + t * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new * (S / s_new)
        else:
            a = b.copy()

        # Check overall progress
        c_check = fft_convolve_self(a)
        obj_check = c_check.max()
        if obj_check < best_obj - 1e-12:
            best_obj = obj_check
            best_a = a.copy()

    return best_a, best_obj


def run_mirror_descent(a_init, budget, rng, beta_start=50.0):
    """
    Multiplicative weight updates (mirror descent with negative entropy).
    Different from FISTA: uses exponential updates instead of Euclidean projection.
    a_new[i] = a[i] * exp(-eta * grad[i]), then rescaled to sum S.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    # Ensure no zeros for multiplicative updates
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())

    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    momentum = a.copy()

    while time.monotonic() - start < budget:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_obj - 1e-12:
            best_obj = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        beta = beta_start * (1.0 + 0.01 * it)
        if stuck > 2000:
            beta *= 0.5  # Reduce smoothing when stuck to get sharper signal

        w = lse_weights(c, beta)
        grad = conv_gradient_self(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        eta = S / (gnorm * (1.0 + 0.01 * it))
        eta = min(eta, 0.1)

        # Multiplicative update
        log_update = -eta * grad
        log_update -= log_update.max()  # numerical stability
        multiplier = np.exp(log_update)
        a_new = a * multiplier
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * (S / s_new)
        a_new = np.maximum(a_new, 1e-15)

        # Momentum (Nesterov-like for multiplicative)
        a_new = 0.7 * a_new + 0.3 * momentum
        momentum = a_new
        a = a_new * (S / a_new.sum())

        # Periodic perturbation when stuck
        if stuck > 3000 and it % 200 == 0:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            a = np.maximum(a + noise, 1e-10)
            a = a * (S / a.sum())
            momentum = a.copy()

    best_a = np.maximum(best_a, 0.0)
    best_a = best_a * (S / best_a.sum())
    return best_a, compute_obj(best_a)


def run_fista(a_init, budget, rng, beta_start=10.0, mode="lse"):
    """Standard FISTA with LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < budget:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        beta = beta_start * (1.0 + 0.005 * it)
        beta = min(beta, 50000.0)
        step = 1.0 / max(beta, 1.0)

        if mode == "topk" and stuck > 500:
            k = max(4, min(128, n // 16))
            grad = topk_gradient(y, c, k, power=1.5)
        else:
            w = lse_weights(c, beta)
            grad = conv_gradient_self(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if stuck > 2000:
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck / 10000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 5000 == 0:
            t_mom = 1.0

    best_a = np.maximum(best_a, 0.0)
    best_a = best_a * (S / best_a.sum())
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_lbfgs_polish(a_init, budget, rng):
    """
    Aggressive L-BFGS-B polishing with LSE smoothing and sum penalty.
    Multiple rounds with increasing beta and more iterations.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    # More aggressive schedule: higher betas, more iterations
    rounds = [
        (30.0, 10.0, 300),
        (100.0, 20.0, 300),
        (300.0, 50.0, 200),
        (1000.0, 100.0, 150),
        (3000.0, 200.0, 100),
    ]

    for beta, pen, max_iter in rounds:
        if time.monotonic() - start > budget:
            break
        remaining = budget - (time.monotonic() - start)
        if remaining < 10:
            break

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm = b.max()
            shifted = beta * (b - cm)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            obj = cm + np.log(e.sum()) / beta
            sum_x = x.sum()
            obj += pen * (sum_x - S) ** 2
            grad = conv_gradient_self(x, w)
            grad += 2.0 * pen * (sum_x - S) * np.ones(n)
            return obj, grad

        t0 = time.monotonic()
        result = scipy_minimize(
            obj_and_grad, a.copy(), method='L-BFGS-B',
            jac=True, bounds=[(0, 1000)] * n,
            options={'maxiter': max_iter, 'maxls': 50, 'ftol': 1e-15,
                     'gtol': 1e-12, 'iprint': -1}
        )
        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * (S / s_new)
        obj_new = compute_obj(a_new)
        print("    L-BFGS beta=%.0f pen=%.0f: obj=%.8f (t=%.1fs, it=%d)" %
              (beta, pen, obj_new, elapsed, result.nit))

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_hybrid_polish(a_init, budget, rng):
    """
    Hybrid: alternate between FISTA (topk mode) and L-BFGS in short bursts.
    This combines the global exploration of FISTA with local precision of L-BFGS.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    burst_time = 30.0  # 30 seconds per burst
    cycle = 0
    while time.monotonic() - start < budget:
        remaining = budget - (time.monotonic() - start)
        if remaining < 5:
            break

        t = min(burst_time, remaining)
        if cycle % 3 == 0:
            # FISTA topk
            a_new, obj_new = run_fista(a, t, rng, beta_start=50.0, mode="topk")
        elif cycle % 3 == 1:
            # L-BFGS
            a_new, obj_new = run_lbfgs_polish(a, t, rng)
        else:
            # Mirror descent
            a_new, obj_new = run_mirror_descent(a, t, rng, beta_start=30.0)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            print("    Hybrid cycle %d (mode=%d): obj=%.8f" % (cycle, cycle % 3, obj_new))
        else:
            # If no improvement, try a perturbation
            if best_obj < 1.51:
                noise = rng.normal(0, 0.005 / np.sqrt(n), n)
                a_pert = np.maximum(a + noise, 0)
                a_pert = a_pert * (S / a_pert.sum())
                # Quick FISTA from perturbed
                a_try, obj_try = run_fista(a_pert, min(10, remaining), rng, beta_start=20.0)
                if obj_try < best_obj - 1e-12:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    print("    Hybrid perturbation success: obj=%.8f" % obj_try)

        cycle += 1

    return best_a, best_obj


def optimize_at_n(a_init, n, budget, rng):
    """Full optimization pipeline at a given n."""
    if a_init is not None and len(a_init) == n:
        a = a_init.copy()
    else:
        # Initialize: cosine bump
        t = np.linspace(0, 1, n, endpoint=False)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 0.01)

    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    obj = compute_obj(a)
    print("  n=%d init: obj=%.8f" % (n, obj))

    # Phase 1: Alternating method
    rem = budget - 0
    t1 = rem * 0.25
    if t1 > 10:
        a1, obj1 = run_alternating(a, t1, rng)
        if obj1 < obj:
            a, obj = a1, obj1
            print("  n=%d after alternating: obj=%.8f" % (n, obj))

    # Phase 2: FISTA
    rem = budget - (0.25 * budget)
    t2 = rem * 0.30
    if t2 > 10:
        a2, obj2 = run_fista(a, t2, rng, beta_start=20.0)
        if obj2 < obj:
            a, obj = a2, obj2
            print("  n=%d after FISTA: obj=%.8f" % (n, obj))

    # Phase 3: Mirror descent
    rem = budget - (0.55 * budget)
    t3 = rem * 0.25
    if t3 > 10:
        a3, obj3 = run_mirror_descent(a, t3, rng, beta_start=30.0)
        if obj3 < obj:
            a, obj = a3, obj3
            print("  n=%d after mirror: obj=%.8f" % (n, obj))

    # Phase 4: L-BFGS polish
    rem = budget - (0.80 * budget)
    t4 = rem * 0.8
    if t4 > 10:
        a4, obj4 = run_lbfgs_polish(a, t4, rng)
        if obj4 < obj:
            a, obj = a4, obj4
            print("  n=%d after L-BFGS: obj=%.8f" % (n, obj))

    # Phase 5: Hybrid final
    rem = budget - 0.95 * budget
    if rem > 10:
        a5, obj5 = run_hybrid_polish(a, rem, rng)
        if obj5 < obj:
            a, obj = a5, obj5
            print("  n=%d after hybrid: obj=%.8f" % (n, obj))

    return a, obj


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-method with alternating LP-style, mirror descent, FISTA, L-BFGS.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None
    global_best_n = 0

    # Load initial sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.8f" % (n0, obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
        global_best_n = n0
    else:
        n0 = 8192
        t = np.linspace(0, 1, n0, endpoint=False)
        a_start = 0.54 + 0.46 * np.cos(2 * np.pi * t)
        global_best_obj = compute_obj(a_start)
        global_best_a = a_start.copy()
        global_best_n = n0
        print("Fallback init: n=%d, obj=%.8f" % (n0, global_best_obj))

    # === Phase 1: Optimize at starting n (40% time) ===
    remaining = deadline - time.monotonic()
    p1 = remaining * 0.40
    print("\n=== Phase 1: Full pipeline at n=%d (budget %.0fs) ===" % (global_best_n, p1))
    t1 = time.monotonic()

    if global_best_n <= 16384:
        a1, obj1 = optimize_at_n(global_best_a, global_best_n, p1, rng)
        if obj1 < global_best_obj:
            global_best_obj = obj1
            global_best_a = a1.copy()
            print("Phase 1 result: obj=%.8f (improved!)" % obj1)
        else:
            print("Phase 1 result: obj=%.8f (no improvement)" % obj1)
    else:
        # Too large for full pipeline, just do FISTA + L-BFGS
        a1, obj1 = run_fista(global_best_a, p1 * 0.6, rng, beta_start=20.0)
        if obj1 < global_best_obj:
            global_best_obj = obj1
            global_best_a = a1.copy()
        rem1 = p1 - (time.monotonic() - t1)
        if rem1 > 10:
            a1b, obj1b = run_lbfgs_polish(global_best_a, rem1, rng)
            if obj1b < global_best_obj:
                global_best_obj = obj1b
                global_best_a = a1b.copy()

    # === Phase 2: Try different n values (30% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2 = remaining * 0.30
        print("\n=== Phase 2: Multi-n exploration (budget %.0fs) ===" % p2)
        t2 = time.monotonic()

        # Try n values different from current best
        n_candidates = [4096, 8192, 16384]
        n_candidates = [n for n in n_candidates if n != global_best_n]
        # Add current best n
        n_candidates.insert(0, global_best_n)
        # Remove duplicates
        seen = set()
        n_unique = []
        for n in n_candidates:
            if n not in seen:
                seen.add(n)
                n_unique.append(n)

        # Allocate time proportionally (more time for n close to current best)
        n_weights = []
        for n in n_unique:
            if n == global_best_n:
                n_weights.append(2.0)
            else:
                n_weights.append(1.0)
        total_w = sum(n_weights)
        n_times = [p2 * w / total_w for w in n_weights]

        for idx, (n_try, t_alloc) in enumerate(zip(n_unique, n_times)):
            if time.monotonic() > deadline:
                break
            if t_alloc < 15:
                continue

            print("  Trying n=%d (budget %.0fs)..." % (n_try, t_alloc))
            # Initialize from best known (resample if n changed)
            if n_try == len(global_best_a):
                init = global_best_a.copy()
            else:
                # Resample: interpolate from current best
                from numpy import interp
                x_old = np.linspace(0, 1, len(global_best_a))
                x_new = np.linspace(0, 1, n_try)
                init = interp(x_new, x_old, global_best_a)
                init = np.maximum(init, 0)

            a_try, obj_try = optimize_at_n(init, n_try, t_alloc, rng)
            if obj_try < global_best_obj:
                global_best_obj = obj_try
                global_best_a = a_try.copy()
                global_best_n = n_try
                print("  *** n=%d improved! obj=%.8f" % (n_try, obj_try))
            else:
                print("  n=%d: obj=%.8f (no improvement)" % (n_try, obj_try))

    # === Phase 3: Hybrid polish at best n (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3 = remaining * 0.20
        print("\n=== Phase 3: Hybrid polish (budget %.0fs) ===" % p3)
        t3 = time.monotonic()

        a3, obj3 = run_hybrid_polish(global_best_a, p3, rng)
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()
            print("Phase 3 result: obj=%.8f (improved!)" % obj3)
        else:
            print("Phase 3 result: obj=%.8f" % obj3)

    # === Phase 4: Final L-BFGS squeeze (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p4 = remaining * 0.10
        print("\n=== Phase 4: Final L-BFGS squeeze (budget %.0fs) ===" % p4)
        a4, obj4 = run_lbfgs_polish(global_best_a, p4, rng)
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()
            print("Phase 4 result: obj=%.8f (improved!)" % obj4)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f (n=%d)" % (global_best_obj, len(final_list)))
    return final_list
