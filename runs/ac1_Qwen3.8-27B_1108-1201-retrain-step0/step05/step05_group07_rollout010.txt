Uses exponential-family (mirror descent) optimization with adaptive
temperature, FISTA acceleration, and LSE polishing.

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, c, beta):

        step = 1.0 / (2.0 * beta)
        # Cap step to prevent divergence
        step = min(step, 1.5 / max(gnorm, 1e-15))

        # Proximal step (simplex projection)
        a_new = project_simplex(y - step * grad_y, S)

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        y = y * (S / max(y.sum(), 1e-30))
        t_mom = t_new

        a = a_new

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_c - 1e-14:
            best_c = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Reset momentum and add noise when stuck
        if stuck > 400:
            y = a.copy()
            t_mom = 1.0
            noise_scale = 0.01 * S / np.sqrt(n) * (1.0 - 0.5 * progress)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                y = a.copy()
            stuck = 0

    return best_a, best_c


def lse_polish(a0, S, max_time, rng, beta_init=200.0, beta_max=50000.0):
    """
    Intensive LSE polishing with adaptive beta and careful line search.
    Designed for final convergence when close to optimum.
    """
    n = len(a0)
    a = project_simplex(a0, S)

    c = fft_convolve_self(a)
    best_c = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    best_step = 1.0 / (2.0 * beta_init)

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        # Geometrically increasing beta
        beta = min(beta_max, beta_init * np.exp(0.003 * it))

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_c - 1e-14:
            best_c = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 300 and progress > 0.7:
            break

        if stuck > 100:
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 - progress)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            continue

        grad = compute_lse_gradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step: use best known good step, shrink if needed
        step = min(best_step, 1.0 / (2.0 * beta * max(1.0, gnorm / (S / n))))
        step = min(step, 3.0 / max(gnorm * gnorm, 1e-30))

        accepted = False
        for _ in range(12):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < cur_max - 1e-16:
                a = a_try
                best_step = step
                accepted = True
                break
            step *= 0.5

        if not accepted and step < 1e-18:
            break

    return best_a, best_c


def accelerated_subgradient(a0, S, max_time, rng):
    """
    Accelerated subgradient with top-k gradient and Nesterov-like momentum.
    """
    n = len(a0)
    a = project_simplex(a0, S)
    y = a.copy()
    t_mom = 1.0

    c = fft_convolve_self(a)
    best_c = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        k = int(4 + 60 * progress ** 0.7)

        c_y = fft_convolve_self(y)
        grad_y = compute_subgradient_topk(y, c_y, k)
        gnorm = np.linalg.norm(grad_y)
        if gnorm < 1e-15:
            break

        gap = c_y.max() - c_y.mean()
        gap = max(gap, 1e-15)

        step = 2.0 * gap / (gnorm * gnorm)
        step = min(step, 4.0 / max(gnorm, 1e-15))

        a_new = project_simplex(y - step * grad_y, S)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        y = y * (S / max(y.sum(), 1e-30))
        t_mom = t_new

        a = a_new

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_c - 1e-14:
            best_c = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 500:
            y = a.copy()
            t_mom = 1.0
            noise_scale = 0.015 * S / np.sqrt(n) * (1.0 - 0.5 * progress)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                y = a.copy()
            stuck = 0

    return best_a, best_c


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Multi-phase hybrid: mirror descent, accelerated subgradient, FISTA+LSE,
    intensive LSE polishing, and perturbation-based exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    print(f"Starting optimization, budget={budget_s}s")

    # Load provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() > 0 and len(a_provided) >= 256:
            print(f"Using provided sequence: len={len(a_provided)}, sum={a_provided.sum():.2f}")
        else:
            a_provided = None
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def update_global(a, msg=""):
        nonlocal global_best, global_a
        S = a.sum()
        if S < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * len(a) * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print(f"  NEW GLOBAL BEST: obj={obj:.6f} ({msg})")

    # Setup
    if a_provided is not None:
        n = len(a_provided)
        S = np.sqrt(2.0 * n)
        a_init = a_provided * (S / a_provided.sum())
    else:
        n = 8192
        S = np.sqrt(2.0 * n)
        tt = np.linspace(0, 1, n, endpoint=False)
        a_init = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.1
        a_init = a_init * (S / a_init.sum())

    update_global(a_init, "initial")
    print(f"Initial objective: {global_best:.6f}")

    # === Phase 1: Mirror Descent (30% time) ===
    remaining = deadline - time.monotonic()
    t1 = remaining * 0.30
    if t1 > 20:
        print(f"\nPhase 1: Mirror Descent, n={n}, time={t1:.0f}s")
        a_opt, _ = mirror_descent(a_init, S, t1, rng, beta_init=0.3, beta_growth=1.0006)
        update_global(a_opt, "Phase 1 mirror descent")

    # === Phase 2: Accelerated Subgradient (25% time) ===
    remaining = deadline - time.monotonic()
    t2 = remaining * 0.28
    if t2 > 20:
        print(f"Phase 2: Accelerated Subgradient, time={t2:.0f}s")
        a_opt, _ = accelerated_subgradient(global_a, S, t2, rng)
        update_global(a_opt, "Phase 2 accel subgrad")

    # === Phase 3: FISTA + LSE (25% time) ===
    remaining = deadline - time.monotonic()
    t3 = remaining * 0.30
    if t3 > 20:
        print(f"Phase 3: FISTA+LSE, time={t3:.0f}s")
        a_opt, _ = fista_lse(global_a, S, t3, rng, beta_init=5.0, beta_growth=1.003)
        update_global(a_opt, "Phase 3 FISTA")

    # === Phase 4: Intensive LSE Polish (15% time) ===
    remaining = deadline - time.monotonic()
    t4 = remaining * 0.40
    if t4 > 15:
        print(f"Phase 4: LSE Polish, time={t4:.0f}s")
        a_opt, _ = lse_polish(global_a, S, t4, rng, beta_init=100.0, beta_max=40000.0)
        update_global(a_opt, "Phase 4 LSE polish")

    # === Phase 5: Perturbation + quick re-optimize (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n_restarts = min(5, int(remaining / 40))
        t_per = (remaining - 10) / max(n_restarts, 1)
        print(f"\nPhase 5: {n_restarts} perturbations+refines, {t_per:.0f}s each")
        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            noise_scale = 0.015 * S / np.sqrt(n) * (1.0 + 0.5 * ri)
            a_pert = project_simplex(global_a + rng.normal(0, noise_scale, n), S)
            # Quick FISTA on perturbed
            a_opt, _ = fista_lse(a_pert, S, min(t_per * 0.6, 120), rng,
                                  beta_init=20.0, beta_growth=1.005)
            update_global(a_opt, f"Phase 5 restart {ri}")
            # Quick LSE polish
            rem = deadline - time.monotonic()
            if rem > 15:
                a_opt2, _ = lse_polish(global_a, S, min(rem - 10, 60), rng,
                                        beta_init=300.0, beta_max=30000.0)
                update_global(a_opt2, f"Phase 5 polish {ri}")

    # === Phase 6: Final ultra-fine polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10 and global_a is not None:
        print(f"\nPhase 6: Final ultra-fine polish, time={remaining:.0f}s")
        a_opt, _ = lse_polish(global_a, S, remaining - 5, rng,
                               beta_init=1000.0, beta_max=80000.0)
        update_global(a_opt, "Phase 6 final")

    # Final verification
    if global_a is None:
        print("WARNING: No valid solution found")
        global_a = np.ones(256) * 1.0
        global_best = obj_value(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: len={len(global_a)}, internal_obj={global_best:.6f}")
    print(f"evaluate_sequence verdict: {ver:.6f}")
    print(f"{'='*60}")

    return final_list
