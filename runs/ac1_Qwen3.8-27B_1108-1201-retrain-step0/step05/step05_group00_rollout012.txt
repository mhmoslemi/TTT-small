"""
Minimize 2*n*max_conv/sum^2 using adaptive L-BFGS on LSE + top-K peak chasing + peak surgery.

Key innovations:
1. Top-K weighted gradient: combines LSE smoothing with explicit top-K peak focusing
2. Peak surgery: exact gradient of the maximum peak with geometric step decay
3. Non-palindromic initializations (triangular, exponential) that exploit the
   freedom to make b[n-1] < (sum)^2/n
4. Adaptive beta schedule that ramps aggressively
5. Multi-n exploration (4096, 8192, 16384) with time allocation

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c of length 2n-1."""
    n = len(a)
    if N < 2 * n - 1:
        N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def safe_normalize(a, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        n = len(a)
        return np.ones(n) * S / n
    return a * (S / s)


def lse_grad(a, N, beta):
    """Gradient of LSE-smoothed peak. Returns (grad, c, c_max)."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    if beta < 1e-6:
        weights = np.ones(len(c)) / len(c)
    else:
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
        else:
            weights = exp_bc / total
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, c, c_max


def topk_lse_grad(a, N, beta, top_k):
    """Combined gradient: LSE + extra weight on top-K peaks. Returns (grad, c, c_max)."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    if beta < 1e-6:
        weights = np.ones(len(c)) / len(c)
    else:
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
        else:
            weights = exp_bc / total
    # Add extra weight to top-K peaks
    if top_k > 0 and top_k < len(c):
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        tv_max = top_vals[0]
        if tv_max > 0:
            extra_w = (top_vals - c_max * 0.5) / (top_vals.sum() + 1e-30)
            extra_w = np.maximum(extra_w, 0)
            top_w = np.zeros(len(c))
            top_w[top_idx] = extra_w
            weights = weights + 0.3 * top_w
            weights = weights / weights.sum()
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, c, c_max


def peak_gradient(a, k_star):
    """Exact gradient of b[k_star] = sum_i a[i]*a[k_star-i] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for j in range(n):
        m = k_star - j
        if 0 <= m < n:
            grad[j] = 2.0 * a[m]
    return grad


def peak_surgery(a, n, S, N, max_iters, step_init, rng):
    """Exact peak gradient descent with geometric step decay."""
    a = safe_normalize(a, S).copy()
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    step = step_init
    for it in range(max_iters):
        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
        k_star = int(np.argmax(c))
        grad = peak_gradient(a, k_star)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        grad = grad / gnorm
        a_new = project_simplex(a - step * grad, S)
        c_new = compute_conv(a_new, N)
        if c_new.max() <= c_max or it > max_iters // 2:
            a = a_new
            step *= 0.98
        else:
            step *= 0.5
            if step < 1e-12:
                break
    return best_a, best_cmax


def lbfgs_two_loop(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    gn = np.linalg.norm(g)
    if m == 0 or gn < 1e-18:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def spectral_restart(a, N, rng, frac=1.0):
    """Randomize FFT phases (fraction frac) while preserving magnitudes."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    old_phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    new_phase = old_phase.copy()
    if n_rand < N:
        rand_idx = rng.choice(N, size=n_rand, replace=False)
        new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    else:
        new_phase = rng.uniform(0, 2 * np.pi, N)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_triangle(n, rng, flip=False):
    """Triangular (non-palindromic) initialization."""
    if flip:
        a = np.linspace(n, 1, n)
    else:
        a = np.linspace(1, n, n)
    a = a + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_exp(n, rng, decay=0.5):
    """Exponential decay initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-decay * t / n)
    a = a + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    L = n
    i = np.arange(L)
    c = rng.integers(1, L)
    phase = -np.pi * i * (2 * i + 1 - c) / L
    a = 1.0 + 0.7 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_multi_freq(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for freq in [3, 7, 13, 21, 47]:
        a += 0.12 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    return np.clip(a, 0.01, None)


def make_init_fourier(n, rng, N):
    P = rng.exponential(1.0, N)
    P[0] = float(n)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def make_init_binary(n, rng):
    a = (rng.random(n) > 0.4).astype(np.float64) * (0.5 + 0.5 * rng.random(n))
    return np.clip(a, 0.0, None)


def make_init_two_humped(n, rng):
    """Two asymmetric humps - exploits non-palindromic freedom."""
    t = np.arange(n, dtype=np.float64) / n
    a = 1.5 * np.exp(-20 * (t - 0.3) ** 2) + 0.8 * np.exp(-20 * (t - 0.7) ** 2) + 0.05
    a = a + 0.03 * rng.standard_normal(n)

def lbfgs_lse_block(a_init, n, S, N, max_time, rng, beta_init, beta_max,

        a_new = a
        cmax_new = c_max
        for ls in range(10):

        c_max = cmax_nw

            y_hist = []

def peak_surgery_phase(a, n, S, N, max_time, rng):

def deflation_init(n, S, N, n_iters, rng):
    """Constructive initialization by iteratively reducing the peak."""
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c = compute_conv(a, N)
        k_star = int(np.argmax(c))
        contrib = np.zeros(n)
        for j in range(n):
            m = k_star - j
            if 0 <= m < n:
                contrib[j] = 2.0 * a[m]
        if np.max(contrib) < 1e-15:
            break
        i_max = int(np.argmax(contrib))
        i_min = int(np.argmin(contrib))
        if i_max == i_min:
            break
        transfer = min(0.02 * a[i_max], 0.3 * a[i_max])
        if transfer < 1e-15:
            break
        a[i_max] -= transfer
        a[i_min] += transfer
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break
        a_norm = safe_normalize(arr, S)
        t0 = time.monotonic()
        rem = max(1.0, per_init - (time.monotonic() - t0) - 0.5)
        a_l, sc_l = lbfgs_lse_block(a_norm, n_main, S, N, min(rem, 50.0), rng,
                                     beta_init=3.0, beta_max=100.0, m_lbfgs=10,
                                     top_k=0, use_topk_grad=False)
        check_and_update(a_l, name + "/LBFGS")

    print("  Phase 1 best: %.6f" % global_best)

    # Phase 2: Focused optimization with top-K (35% of main time)
    print("Phase 2: Focused L-BFGS with top-K gradient")
    if global_a is None:
        global_a = np.ones(n_main) * S / n_main
        global_best = 2.0

    t2 = time.monotonic()
    phase2_time = 0.35 * (deadline - t2)
    a_curr = global_a.copy()
    stall = 0
    block_count = 0

    while time.monotonic() < t2 + phase2_time:
        if time.monotonic() > deadline - 200:
            break
        block_time = min(120.0, (t2 + phase2_time) - time.monotonic())
        if block_time < 10:
            break
        beta_start = 5.0 + 2.0 * min(stall, 10)
        beta_end = min(500.0, 30.0 * (1 + 0.5 * stall))
        # Alternate between LSE-only and top-K gradient
        use_topk = (block_count % 2 == 1)
        a_new, sc_new = lbfgs_lse_block(a_curr, n_main, S, N, block_time, rng,
                                         beta_init=beta_start, beta_max=beta_end,
                                         m_lbfgs=14, top_k=5, use_topk_grad=use_topk)
        score_new = 2.0 * n_main * sc_new / (S * S)
        if score_new < global_best - 1e-9:
            global_best = score_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            print("  [P2-%d] score=%.6f (beta %.0f-%.0f, topk=%s)" %
                  (block_count, score_new, beta_start, beta_end, use_topk))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall == 2:
                a_curr = spectral_restart(a_curr, N, rng, frac=0.4)
                stall = 0
            elif stall >= 4:
                a_curr = spectral_restart(a_curr, N, rng, frac=0.7)
                stall = 0
        block_count += 1

    print("  Phase 2 best: %.6f" % global_best)

    # Phase 3: Peak surgery polishing (25% of main time)
    print("Phase 3: Peak surgery polishing")
    t3 = time.monotonic()
    phase3_time = 0.25 * (deadline - t3)
    if phase3_time > 20:
        a_surg, sc_surg = peak_surgery_phase(global_a, n_main, S, N, phase3_time, rng)
        score_surg = 2.0 * n_main * sc_surg / (S * S)
        if score_surg < global_best:
            global_best = score_surg
            global_a = a_surg.copy()
            print("  [P3] peak surgery score=%.6f" % score_surg)
        else:
            print("  [P3] no improvement, best stays %.6f" % global_best)

    # Phase 4: Final high-beta L-BFGS + surgery alternation (15% of main time)
    print("Phase 4: Final refinement")
    t4 = time.monotonic()
    phase4_time = 0.15 * (deadline - t4)
    if phase4_time > 20:
        half = phase4_time / 2.0
        # High-beta L-BFGS
        a_fin, sc_fin = lbfgs_lse_block(global_a, n_main, S, N, min(half, 300.0), rng,
                                         beta_init=30.0, beta_max=800.0,
                                         m_lbfgs=16, top_k=3, use_topk_grad=True)
        score_fin = 2.0 * n_main * sc_fin / (S * S)
        if score_fin < global_best:
            global_best = score_fin
            global_a = a_fin.copy()
            print("  [P4a] L-BFGS score=%.6f" % score_fin)
        # Surgery
        if deadline - time.monotonic() > 15:
            a_s2, sc_s2 = peak_surgery_phase(global_a, n_main, S, N,
                                              min(deadline - time.monotonic() - 10, 150.0), rng)
            score_s2 = 2.0 * n_main * sc_s2 / (S * S)
            if score_s2 < global_best:
                global_best = score_s2
                global_a = a_s2.copy()
                print("  [P4b] surgery score=%.6f" % score_s2)

    # ===== Try n=4096 if time remains =====
    t5 = time.monotonic()
    remaining = deadline - t5 - 10
    if remaining > 60:
        n_alt = 4096
        S_alt = np.sqrt(2.0 * float(n_alt))
        N_alt = 1 << (2 * n_alt - 1).bit_length()
        print("=== Alt: n=%d, time=%.0fs ===" % (n_alt, remaining))
        inits_alt = [
            ("rand4k", make_init_rand(n_alt, rng)),
            ("tri4k", make_init_triangle(n_alt, rng, flip=False)),
            ("exp4k", make_init_exp(n_alt, rng, decay=1.0)),
            ("zadoff4k", make_init_zadoff(n_alt, rng)),
        ]
        per_alt = remaining / len(inits_alt)
        for name, arr in inits_alt:
            if time.monotonic() > deadline - 20:
                break
            a_norm = safe_normalize(arr, S_alt)
            a_l, sc_l = lbfgs_lse_block(a_norm, n_alt, S_alt, N_alt, min(per_alt, 40.0), rng,
                                         beta_init=3.0, beta_max=100.0, m_lbfgs=10)
            check_and_update(a_l, name + "/LBFGS")
            # Quick surgery
            if deadline - time.monotonic() > 10:
                a_s, sc_s = peak_surgery_phase(a_l, n_alt, S_alt, N_alt,
                                                min(10.0, (deadline - time.monotonic()) / 2), rng)
                check_and_update(a_s, name + "/surgery")

    # ===== Try n=16384 if time remains =====
    t6 = time.monotonic()
    remaining6 = deadline - t6 - 10
    if remaining6 > 90:
        n_big = 16384
        S_big = np.sqrt(2.0 * float(n_big))
        N_big = 1 << (2 * n_big - 1).bit_length()
        print("=== Big: n=%d, time=%.0fs ===" % (n_big, remaining6))
        inits_big = [
            ("rand16k", make_init_rand(n_big, rng)),
            ("exp16k", make_init_exp(n_big, rng, decay=0.5)),
            ("chirp16k", make_init_chirp(n_big, rng)),
        ]
        per_big = remaining6 / len(inits_big)
        for name, arr in inits_big:
            if time.monotonic() > deadline - 15:
                break
            a_norm = safe_normalize(arr, S_big)
            a_l, sc_l = lbfgs_lse_block(a_norm, n_big, S_big, N_big, min(per_big, 50.0), rng,
                                         beta_init=3.0, beta_max=80.0, m_lbfgs=10)
            check_and_update(a_l, name + "/LBFGS")

    # ===== Final verification =====
    print("=== Final verification ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
            ver = evaluate_sequence([float(x) for x in global_a])
            print("  Fallback score: %.6f" % ver)
    else:
        global_a = np.ones(256, dtype=np.float64)
        ver = evaluate_sequence([float(x) for x in global_a])
        print("  Fallback score: %.6f" % ver)

    print("Returning length %d, best score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
