The KKT condition for minimizing `f(a) = c[k*]` subject to `sum(a) = S` and `a >= 0`:

- `mu[j] >= 0`, `a[j] >= 0`, `mu[j]*a[j] = 0`
- `sum(a) = S`

So for j where a[j] > 0: `2*a[k*-j] = lambda` (constant)
For j where a[j] = 0: `2*a[k*-j] <= lambda`

This means that the optimal sequence, when restricted to its support, is such that `a[k*-j]` is constant for all j in the support. In other words, the reversed sequence (reflected around k*/2) is constant on the support!

This is a very strong constraint. If k* = n-1 (center), then `a[n-1-j] = const` for all j in support. This means the sequence is constant (on its support) when reversed, i.e., the sequence itself is constant on its support.

But wait, this is only for a single active lag. In practice, multiple lags are active (close to the max), and the KKT conditions involve the average gradient over all active lags.

If the active set is {k_1, k_2, ..., k_m}, then the condition becomes:
`(2/m) * sum_i a[k_i - j] = lambda` for all j in support.

This is a more complex condition. It says that the average of the sequence shifted by different amounts (corresponding to the active lags) should be constant on the support.

This gives us a way to check if we're at a local optimum: compute the active set gradient and check if it's constant on the support.

1. Compute the active set
2. Compute the active set gradient
3. Check if gradient is constant on support (KKT condition)
4. If not, perform a step in the direction that reduces the violation

Minimize 2*n*max(f*f)/sum(f)^2 using KKT-informed gradient optimization.
Key insight: At the optimum, the active-set gradient should be constant on the support.
We exploit this for adaptive step sizing and convergence detection.

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def active_set_gradient(a, c, threshold_abs=0.01):
    """Compute average gradient over active set."""
    n = len(a)
    cm = c.max()
    active = np.where(c >= cm - threshold_abs)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            grad[lo:hi+1] += 2.0 * a[k-hi : k-lo+1][::-1]
    grad /= len(active)
    return grad, active

def kkt_violation(a, c, threshold_abs=0.01):
    """Check how close we are to KKT conditions."""
    n = len(a)
    grad, active = active_set_gradient(a, c, threshold_abs)
    support = a > 1e-10
    if not np.any(support):
        return np.inf
    g_support = grad[support]
    # KKT: gradient should be constant on support
    return g_support.max() - g_support.min()

def optimize_sequence(a_init, n, S, max_time, rng, 
                       step_init=0.5, threshold_init=0.01,
                       momentum_init=0.9, noise_scale_init=0.01):

    Main optimization loop using active-set gradient with Nesterov momentum.
    """
    a = project_simplex(a_init.copy(), S)
    y = a.copy()  # Nesterov y
    t_mom = 1.0
    momentum = momentum_init
    threshold = threshold_init
    step = step_init
    noise_scale = noise_scale_init
    
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        t_elapsed = time.monotonic() - start
        progress = t_elapsed / max_time
        
        # Compute convolution
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
        
        # Compute gradient
        grad, active = active_set_gradient(y, c, threshold)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Adaptive threshold: increase as we converge
        threshold = threshold_init * (1 + 10 * progress)
        
        # Adaptive noise: decrease over time
        noise_s = noise_scale_init * S / np.sqrt(n) * np.exp(-5 * progress)
        
        # Nesterov step
        x_new = project_simplex(y - step * grad, S)
        
        # Momentum update
        if it - last_improve > 200:
            # Restart momentum
            t_mom = 1.0
            momentum = momentum_init
            # Inject noise
            if noise_s > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_s, n), S)
            last_improve = it
            step = step_init * (1 + 5 * progress)  # Increase step as we go
        else:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new
        
        # Occasionally check KKT violation
        if it % 500 == 0:
            kkt = kkt_violation(y, c, threshold)
            if kkt < 0.01:
                print("    KKT violation small: %.6f at iter %d" % (kkt, it))
    
    # Final
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

Minimize 2*n*max(f*f)/sum(f)^2 using active-set subgradient optimization with
Nesterov acceleration, KKT-informed adaptive stepping, and basin hopping.
Key insight: At KKT optimum, the active-set gradient is constant on the support.
We use this for convergence detection and adaptive step sizing.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def active_set_gradient(a, c, threshold_abs):
    """Compute average gradient over active set (lags within threshold of max)."""
    n = len(a)
    cm = c.max()
    active = np.where(c >= cm - threshold_abs)[0]
    if len(active) == 0:

        hi = min(k, n - 1)
        if lo <= hi:

def run_nesterov_active(a_init, n, S, max_time, rng,

    """
    Nesterov-accelerated gradient descent using active-set subgradient.

    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t_mom = 1.0
    step = step_init
    threshold = thresh_init
    noise_scale = noise_init * S / np.sqrt(n)

    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

        c = fft_convolve_self(y)

            best_a = y.copy()

        # Adaptive noise: decay over time
        cur_noise = noise_init * S / np.sqrt(n) * np.exp(-4.0 * progress)

        # Step adaptation: increase step as we approach convergence (bolder steps)
        cur_step = step_init * (1.0 + 3.0 * progress * progress)

def basin_hopping(a_init, n, S, max_time, rng,
                  n_inner=50, perturb_scale=0.02):
    """
    Basin hopping: perturb, optimize briefly, accept if better.
    """
    a = project_simplex(a_init.copy(), S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    c0 = fft_convolve_self(a)
    best = c0.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(max_time, 1.0))

        # Perturb
        scale = perturb_scale * S / np.sqrt(n) * np.exp(-2.0 * progress)
        if scale < 1e-10:
            break
        a_pert = project_simplex(a + rng.normal(0, scale, n), S)

        # Short inner optimization (Nesterov)
        a_new, _ = run_nesterov_active(a_pert, n, S, min(2.0, max_time * 0.01), rng,
                                       step_init=0.3, thresh_init=0.01,
                                       noise_init=0.0, restart_gap=200)

        c_new = fft_convolve_self(a_new)
        val = c_new.max()

        if val < best:
            best = val
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Accept with Metropolis criterion (cooled)
            T = 0.01 * (1.0 - progress)
            if T > 1e-6 and rng.random() < np.exp(-(val - best) / max(T, 1e-10)):
                a = a_new.copy()

    print("Budget: %.0fs, safety: %.0fs, deadline in %.0fs" % (budget_s, safety, budget_s - safety))

    # Load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given = np.clip(given, 0, 1000)
        if given.size >= 256 and given.sum() > 0.01:
            print("Loaded given sequence: length=%d, sum=%.4f" % (len(given), given.sum()))
        else:
            given = None
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def evaluate_internal(a):
        """Compute objective value for normalized sequence."""
        n = len(a)
        S = np.sqrt(2.0 * n)
        a_norm = project_simplex(a, S)
        c = fft_convolve_self(a_norm)
        return c.max()

    def try_sequence(a):
        nonlocal global_best, global_a
        val = evaluate_internal(a)
        if val < global_best:
            global_best = val
            global_a = a.copy()
            print("  NEW BEST: %.8f (n=%d)" % (val, len(a)))
            return True
        return False

    # === Phase 1: Quick optimization of given sequence ===
    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        t1 = min(90.0, (deadline - time.monotonic()) * 0.15)
        print("=== Phase 1: Nesterov on given (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_nesterov_active(given, n_g, S_g, t1, rng,
                                     step_init=0.3, thresh_init=0.005,
                                     noise_init=0.003, restart_gap=100)
        try_sequence(a1)

        # Also try line search descent
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t1b = min(30.0, remaining * 0.1)
            print("=== Phase 1b: Line search on given (%.0fs) ===" % t1b)
            a1b, v1b = run_line_search_descent(given, n_g, S_g, t1b, rng,
                                               step_init=0.1, thresh_init=0.01)
            try_sequence(a1b)

    # === Phase 2: Main optimization with best found so far ===
    if global_a is not None:
        n_main = len(global_a)
        S_main = np.sqrt(2.0 * n_main)
        remaining = deadline - time.monotonic()
        t2 = remaining * 0.35
        print("=== Phase 2: Main Nesterov (n=%d, %.0fs) ===" % (n_main, t2))
        a2, v2 = run_nesterov_active(global_a, n_main, S_main, t2, rng,
                                     step_init=0.5, thresh_init=0.003,
                                     noise_init=0.004, restart_gap=120)
        try_sequence(a2)

        # Follow with line search
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t2b = remaining * 0.15
            print("=== Phase 2b: Line search refine (%.0fs) ===" % t2b)
            a2b, _ = run_line_search_descent(global_a, n_main, S_main, t2b, rng,
                                             step_init=0.05, thresh_init=0.005)
            try_sequence(a2b)

    # === Phase 3: Try diverse initializations ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n_try = 8192
        S_try = np.sqrt(2.0 * n_try)
        t3 = remaining * 0.3
        print("=== Phase 3: Diverse inits (n=%d, %.0fs) ===" % (n_try, t3))

        inits = generate_inits(n_try, rng)

        # Quick score all inits
        init_scores = []
        for i, init in enumerate(inits):
            init_norm = project_simplex(init, S_try)
            c = fft_convolve_self(init_norm)
            init_scores.append((c.max(), i))
        init_scores.sort()

        print("  Top init score: %.6f, bottom: %.6f (of %d inits)" %
              (init_scores[0][0], init_scores[-1][0], len(inits_scores) if len(inits_scores) > 0 else 1))

        # Try top inits
        best_init_val = np.inf
        best_init_a = None
        per_init_time = max(2.0, t3 / 10)

        for rank, (score, idx) in enumerate(init_scores[:8]):
            if time.monotonic() > deadline - 60:
                break
            init = inits[idx]
            init_norm = project_simplex(init, S_try)
            print("  Init %d (score=%.6f), optimizing for %.0fs..." % (idx, score, per_init_time))
            ai, vi = run_nesterov_active(init_norm, n_try, S_try, per_init_time, rng,
                                         step_init=0.3, thresh_init=0.005,
                                         noise_init=0.003, restart_gap=100)
            if try_sequence(ai):
                best_init_a = ai.copy()
                best_init_val = vi

    # === Phase 4: Basin hopping on best ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_bh = len(global_a)
            S_bh = np.sqrt(2.0 * n_bh)
            t4 = remaining * 0.4
            print("=== Phase 4: Basin hopping (n=%d, %.0fs) ===" % (n_bh, t4))
            a4, v4 = basin_hopping(global_a, n_bh, S_bh, t4, rng,
                                   n_inner=30, perturb_scale=0.015)
            try_sequence(a4)

    # === Phase 5: Final polish with very tight threshold ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_f = len(global_a)
            S_f = np.sqrt(2.0 * n_f)
            t5 = remaining * 0.6
            print("=== Phase 5: Final polish (%.0fs) ===" % t5)
            a5, v5 = run_nesterov_active(global_a, n_f, S_f, t5, rng,
                                         step_init=0.2, thresh_init=0.001,
                                         noise_init=0.001, restart_gap=500)
            try_sequence(a5)

            # Very tight active set refinement
            remaining2 = deadline - time.monotonic()
            if remaining2 > 5:
                a5b, _ = run_line_search_descent(global_a, n_f, S_f, remaining2 * 0.5, rng,
                                                 step_init=0.01, thresh_init=0.001)
                try_sequence(a5b)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.8f, internal = %.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result, using uniform fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
