                last_improve = it

        if it % 5000 == 0:
            t_mom = 1.0

    return best_a, best_maxc


def run_exact_polish(a_init, max_time, rng):
    """
    Exact subgradient polishing with backtracking.
    Uses the true max (not LSE) for convergence checking.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(clip_nonneg(a_init), S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_maxc = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 0.5 / (np.max(a) + 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve(y)
        cur_maxc = c.max()
        if cur_maxc < best_maxc - 1e-14:
            best_maxc = cur_maxc
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive active set size
        if stuck < 50:
            k = 1
        elif stuck < 200:
            k = 2
        elif stuck < 1000:
            k = min(4, 2 + stuck // 100)
        elif stuck < 5000:
            k = min(16, 4 + stuck // 500)
        else:
            k = min(64, 16 + stuck // 500)

        if k == 1:
            w = exact_subgrad_weights(c)
        else:
            w = topk_weights(c, k)

        grad = grad_from_weights(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking step
        step = min(step * 1.01, 1.0 / (gnorm * 0.5 + 0.1))
        x_new = None
        for _bt in range(8):
            x_trial = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y_trial = x_trial + ((t_mom - 1.0) / t_new) * (x_trial - a)
            y_trial = np.maximum(y_trial, 0.0)
            c_trial = fft_convolve(y_trial)
            if c_trial.max() <= cur_maxc * (1 + 1e-8) or _bt >= 3:
                x_new = x_trial
                break
            step *= 0.5

        if x_new is not None:
            y = y_trial
            a = x_new
            t_mom = t_new
        else:
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

        if stuck > 5000 and it % 500 == 0:
            t_mom = 1.0
            noise_scale = max(1e-8, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 30000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a, best_maxc


def run_coord_descent(a_init, max_time, rng):
    """
    Cyclic coordinate descent: for each coordinate, optimize it in 1D
    while keeping others fixed. Exploits that the max-convolution as a
    function of a single coordinate is the max of linear+quadratic terms.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(clip_nonneg(a_init), S).copy()
    start = time.monotonic()
    best_maxc = np.inf
    best_a = a.copy()
    c = fft_convolve(a)
    best_maxc = c.max()

    pass_num = 0
    while time.monotonic() - start < max_time:
        pass_num += 1
        improved = False
        # Randomize the starting point each pass to avoid cycling
        idx = np.random.default_rng(pass_num * 12345 + 7).permutation(n)
        
        for i in idx:
            if time.monotonic() - start > max_time:
                break
            # Compute t_k = c[k] - contributions from a[i]
            # c[k] = sum_j a[j]*a[k-j]; terms involving a[i]:
            #   j=i: a[i]*a[k-i] if 0<=k-i<n
            #   k-j=i => j=k-i: a[k-i]*a[i] if 0<=k-i<n and k-i!=i
            # So contribution = a[i]*a[k-i] + a[k-i]*a[i] = 2*a[i]*a[k-i] (if k-i valid and k-i!=i)
            #                 or a[i]^2 (if k-i==i, i.e., k=2i)
            
            ai = a[i]
            if ai < 1e-15:
                continue

            # Compute the "removal" vector
            # For each k, the contribution of a[i] to c[k]:
            # k ranges from 0 to 2n-2
            # a[i] contributes to c[k] when k-i is in [0,n-1] (as the second factor)
            #   or i is in [0,n-1] and k-i is in [0,n-1] (as the first factor)
            # Actually c[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j]*a[k-j]
            # a[i] appears when j=i (needs 0<=i<n and 0<=k-i<n) or k-j=i (needs j=k-i in range)
            
            # Contribution to c[k] from a[i]:
            # If k-i is in [0, n-1]: a[i] * a[k-i] (j=i term)
            # If k-i is in [0, n-1] and k-i != i: a[k-i] * a[i] (j=k-i term, same as above)
            # If k = 2i: a[i] * a[i] = a[i]^2
            # So total: if k-i in [0,n-1]: 2*a[i]*a[k-i], except k=2i gives a[i]^2

            # Build the linear coefficients: c[k] = alpha_k * x + beta_k * x^2 + t_k
            # where x = a[i] (the variable we're optimizing)
            # alpha_k = 2*a[k-i] if 0<=k-i<n and k!=2i, else 0
            # beta_k = 1 if k==2i, else 0
            # t_k = c[k] - alpha_k * ai - beta_k * ai^2

            # For the 1D optimization: minimize max_k (alpha_k * x + beta_k * x^2 + t_k) over x >= 0
            # Also need to maintain sum constraint: if we change a[i] to x, we need to
            # redistribute (ai - x) to other coordinates. For simplicity, let's just
            # optimize a[i] freely and then renormalize at the end of each pass.
            
            # Actually, for correctness with the sum constraint, let's just optimize
            # freely and renormalize periodically. The effect is small for large n.
            
            # Get the relevant a values: a[k-i] for k such that 0<=k-i<n
            # k ranges from i to i+n-1 (so k-i ranges from 0 to n-1)
            k_start = i
            k_end = i + n - 1
            
            # alpha for these k values: 2*a[k-i] for k != 2i, 0 for k=2i
            alpha = 2.0 * a[np.arange(0, n)]  # a[k-i] for k-i = 0..n-1
            # For k = 2i (if in range), set alpha[2i-k_start] = 0 and beta = 1
            if 2 * i >= k_start and 2 * i <= k_end:
                alpha[2 * i - k_start] = 0.0
            
            # t_k for these k: c[k] - alpha_k * ai - (ai^2 if k==2i)
            t_vals = c[k_start:k_end+1].copy()
            t_vals = t_vals - alpha * ai
            if 2 * i >= k_start and 2 * i <= k_end:
                t_vals[2 * i - k_start] -= ai * ai  # already removed via alpha=0, so just the ai^2 term
                # Wait: original c[k] includes the term. We subtracted alpha*ai = 0 for k=2i.
                # But the actual contribution at k=2i is ai^2 (not 2*ai*ai).
                # So t[2i-k_start] should be c[2i] - ai^2.
                # We subtracted 0*ai = 0, so t = c[2i]. We need to subtract ai^2.
                # Already done above.
            
            # For k outside [k_start, k_end]: t_k = c[k] (no contribution from a[i])
            # These are "constant" terms in the 1D problem.
            # The max over ALL k includes these constants.
            # max_const = max(c) over k not in [k_start, k_end]
            if k_start > 0:
                const_part = c[:k_start]
            else:
                const_part = np.array([])
            if k_end < 2 * n - 2:
                const_part2 = c[k_end+1:]
            else:
                const_part2 = np.array([])
            if len(const_part) > 0 and len(const_part2) > 0:
                max_const = max(const_part.max(), const_part2.max())
            elif len(const_part) > 0:
                max_const = const_part.max()
            elif len(const_part2) > 0:
                max_const = const_part2.max()
            else:
                max_const = -np.inf

            # Now solve: min_{x>=0} max( max_const, max_{j=0..n-1} (alpha[j]*x + (1 if j==2i-k_start else 0)*x^2 + t_vals[j]) )
            # The objective is convex in x. Use bisection on the subgradient.
            
            # At x=0: value = max(max_const, max(t_vals))
            # The subgradient at x=0 from the active terms:
            #   from linear terms: alpha[j] for active j
            #   from quadratic: 0 (derivative of x^2 at x=0 is 0)
            #   from constant: 0
            
            # Let's use golden section search for simplicity (works for any unimodal function)
            # Actually, bisection on the derivative is more efficient.
            
            # g(x) = max(max_const, max_j f_j(x)) where f_j(x) = alpha[j]*x + delta_j*x^2 + t_vals[j]
            # g is convex. g'(x) is the subgradient.
            # At optimum: 0 is in the subgradient of g.
            
            # Simple approach: ternary search on [0, x_max]
            # x_max: a reasonable upper bound, say 2*ai (we don't want to increase too much)
            x_lo = 0.0
            x_hi = max(2.0 * ai, ai + S / n)
            
            # Evaluate g(x)
            def g_at(x):
                vals = alpha * x + t_vals
                # Add quadratic term
                if 2 * i >= k_start and 2 * i <= k_end:
                    vals[2 * i - k_start] += x * x
                return max(max_const, vals.max())
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_gs = x_lo
            b_gs = x_hi
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            fc = g_at(c_gs)
            fd = g_at(d_gs)
            for _ in range(40):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = g_at(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = g_at(d_gs)
            
            x_opt = (a_gs + b_gs) / 2.0
            x_opt = max(0.0, x_opt)
            
            if abs(x_opt - ai) > 1e-10:
                a[i] = x_opt
                improved = True
        
        # Renormalize after each full pass
        a = normalize_seq(a, S)
        c = fft_convolve(a)
        cur_maxc = c.max()
        if cur_maxc < best_maxc:
            best_maxc = cur_maxc
            best_a = a.copy()
        
        if not improved and best_maxc < c.max() + 1e-12:
            break

    return best_a, best_maxc


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.54 + 0.46 * np.cos(2 * np.pi * tt)


def make_random_phase_init(n, rng):
    """Initialize from random phases (gives approximately flat power spectrum)."""
    phases = rng.uniform(0, 2 * np.pi, n)
    a = np.real(ifft(np.exp(1j * phases)))
    a = np.abs(a) + 0.01  # Make non-negative with small floor
    return a


def make_gaussian_init(n):
    tt = np.arange(n, dtype=np.float64)
    center = n / 2.0
    sigma = n / 6.0
    return np.exp(-0.5 * ((tt - center) / sigma) ** 2)


def run_fista_simple(a_init, max_time, rng, beta_start=3.0, beta_end=1000.0):
    """
    Cleaner FISTA with proper backtracking and adaptive beta.
    This is the main workhorse.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(clip_nonneg(a_init), S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_maxc = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 0.3 / (beta_start * np.max(a) + np.max(a) + 1.0)
    step = max(step, 1e-10)

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max_time)

        # Beta schedule: quadratic ramp after 50% of time
        if frac < 0.5:
            beta = beta_start * (1.0 + frac / 0.5)
        else:
            pos = (frac - 0.5) / 0.5
            beta = beta_start * 2.0 + (beta_end - beta_start * 2.0) * pos ** 2
        beta = max(beta, beta_start)

        c = fft_convolve(y)
        cur_maxc = c.max()
        if cur_maxc < best_maxc - 1e-14:
            best_maxc = cur_maxc
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Gradient via LSE
        w = lse_weights(c, beta)
        grad = grad_from_weights(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: adaptive
        L_est = 2.0 * beta * np.max(a) + 2.0 * np.max(a)
        step_max = 0.9 / max(L_est, 1e-10)
        step = min(step * 1.005, step_max)  # Slowly increase
        step = max(step, 1e-12)

        # FISTA iteration with backtracking
        accepted = False
        for _bt in range(8):
            x_trial = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y_trial = x_trial + ((t_mom - 1.0) / t_new) * (x_trial - a)
            y_trial = np.maximum(y_trial, 0.0)

            # Check improvement using LSE (fast proxy)
            c_trial = fft_convolve(y_trial)
            lse_trial = lse_value(c_trial, beta)
            lse_cur = lse_value(c, beta)

            if lse_trial <= lse_cur + 1e-12:
                accepted = True
                break
            step *= 0.5

        if accepted:
            a = x_trial
            y = y_trial
            t_mom = t_new
        else:
            # Very small step
            step *= 0.5
            a = project_simplex(y - step * grad, S)
            t_mom = 1.0  # Restart momentum if stuck

        # Restart strategy
        if stuck > 2000:
            t_mom = 1.0
        if stuck > 10000 and it % 1000 == 0:
            noise_scale = max(1e-9, 0.002 * S / np.sqrt(n) * np.exp(-stuck / 30000.0))
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a, best_maxc


def propose_candidate(budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: diverse inits -> FISTA-LSE -> exact polish -> coordinate descent.
    """
    rng = np.random.default_rng(42)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get provided sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    # Determine base n
    if a_start is not None:
        n0 = len(a_start)
    else:
        n0 = 8192
        a_start = make_cosine_init(n0)

    print("Base n = %d" % n0)

    n = n0
    S = np.sqrt(2.0 * n)

    # Generate diverse initializations
    inits = []
    names = []

    if a_start is not None:
        inits.append(normalize_seq(a_start, S))
        names.append("provided")

    inits.append(normalize_seq(make_cosine_init(n), S))
    names.append("cosine")

    inits.append(normalize_seq(make_random_phase_init(n, rng), S))
    names.append("randphase")

    inits.append(normalize_seq(make_gaussian_init(n), S))
    names.append("gaussian")

    inits.append(normalize_seq(rng.random(n) ** 0.5, S))
    names.append("random")

    # Quick evaluation
    print("\n=== Init evaluation ===")
    best_init_idx = 0
    best_init_obj = np.inf
    init_objs = []
    for i, (a_init, name) in enumerate(zip(inits, names)):
        obj = compute_obj_normalized(a_init)
        init_objs.append(obj)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init_idx = i
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a_init.copy()
        print("  %s: obj=%.6f" % (name, obj))

    print("Best init: %s with obj=%.6f" % (names[best_init_idx], best_init_obj))

    # === Phase 1: Run FISTA-LSE on top 3 initializations ===
    remaining = deadline - time.monotonic()
    top3 = np.argsort(init_objs)[:3]
    per_init_budget = remaining * 0.45 / len(top3)

    print("\n=== Phase 1: FISTA-LSE on top inits (budget %.0fs each) ===" % per_init_budget)
    t1_start = time.monotonic()

    for init_idx in top3:
        if time.monotonic() > deadline:
            break
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        alloc = min(per_init_budget, rem * 0.6)
        if alloc < 10:
            break

        print("  Running on %s..." % names[init_idx])
        a_res, mc_res = run_fista_simple(inits[init_idx], alloc, rng,
                                          beta_start=3.0, beta_end=500.0)
        obj_res = compute_obj_normalized(a_res)
        print("    %s -> obj=%.6f (maxc=%.6f, t=%.0fs)" % (
            names[init_idx], obj_res, mc_res, time.monotonic() - t1_start))
        if obj_res < global_best_obj:
            global_best_obj = obj_res
            global_best_a = a_res.copy()

    # === Phase 2: Extended FISTA with higher beta on best ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = remaining * 0.30
        print("\n=== Phase 2: High-beta FISTA (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()
        a_res, mc_res = run_fista_simple(global_best_a, p2_budget, rng,
                                          beta_start=50.0, beta_end=5000.0)
        obj_res = compute_obj_normalized(a_res)
        print("  FISTA-high: obj=%.6f (t=%.0fs)" % (obj_res, time.monotonic() - t2))
        if obj_res < global_best_obj:
            global_best_obj = obj_res
            global_best_a = a_res.copy()

    # === Phase 3: Exact subgradient polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_budget = remaining * 0.20
        print("\n=== Phase 3: Exact subgradient polish (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()
        a_res, mc_res = run_exact_polish(global_best_a, p3_budget, rng)
        obj_res = compute_obj_normalized(a_res)
        print("  Exact: obj=%.6f (t=%.0fs)" % (obj_res, time.monotonic() - t3))
        if obj_res < global_best_obj:
            global_best_obj = obj_res
            global_best_a = a_res.copy()

    # === Phase 4: Coordinate descent polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p4_budget = remaining * 0.25
        print("\n=== Phase 4: Coordinate descent (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        a_res, mc_res = run_coord_descent(global_best_a, p4_budget, rng)
        obj_res = compute_obj_normalized(a_res)
        print("  CoordDesc: obj=%.6f (t=%.0fs)" % (obj_res, time.monotonic() - t4))
        if obj_res < global_best_obj:
            global_best_obj = obj_res
            global_best_a = a_res.copy()

    # === Phase 5: Final exact polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = remaining * 0.50
        print("\n=== Phase 5: Final exact polish (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()
        a_res, mc_res = run_exact_polish(global_best_a, p5_budget, rng)
        obj_res = compute_obj_normalized(a_res)
        print("  Final: obj=%.6f (t=%.0fs)" % (obj_res, time.monotonic() - t5))
        if obj_res < global_best_obj:
            global_best_obj = obj_res
            global_best_a = a_res.copy()

    # Final coordinate descent if time permits
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(global_best_a) <= 10000:
        p6_budget = min(remaining * 0.5, 120)
        print("\n=== Phase 6: Final coord desc (budget %.0fs) ===" % p6_budget)
        t6 = time.monotonic()
        a_res, mc_res = run_coord_descent(global_best_a, p6_budget, rng)
        obj_res = compute_obj_normalized(a_res)
        print("  CoordDesc2: obj=%.6f (t=%.0fs)" % (obj_res, time.monotonic() - t6))
        if obj_res < global_best_obj:
            global_best_obj = obj_res
            global_best_a = a_res.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL RESULT ===")
            print("Objective: %.6f (n=%d)" % (ver, len(final_list)))
        else:
            print("WARNING: evaluate_sequence rejected")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
