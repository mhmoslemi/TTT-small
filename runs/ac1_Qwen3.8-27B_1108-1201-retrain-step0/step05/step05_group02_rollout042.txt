```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase: FISTA screening, peak-chasing, fine polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_conv(a)
    return 2.0 * n * c.max() / (s * s)


def grad_max_lag(a, k, c):
    """Gradient of c[k] w.r.t. a, where c = convolve(a,a)."""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= k - j < n:
            val += a[k - j]
        if 0 <= j - k < n:
            val += a[j - k]
        g[j] = val
    return g
```

```python
"""
```

```python
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_conv(a)
    return 2.0 * n * c.max() / (s * s)


def grad_single_lag(a, k):
    """Exact gradient of c[k] w.r.t. a. O(n) computation."""
    n = len(a)
    g = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_hi >= j_lo:
        idx = np.arange(j_lo, j_hi + 1)
        g[idx] = 2.0 * a[k - idx]
    return g


def lse_grad(a, beta, c):
    """Compute LSE value and gradient using precomputed convolution."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        lse_val = cm
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    # Gradient = 2 * (w * a) middle n
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, w


def run_fista(a_init, max_time, rng, beta_init=20.0, beta_rate=0.008,
              beta_max=50000.0, step_scale=1.0, use_backtrack=True):
    """FISTA with adaptive LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / max(beta_init, 1.0)
    c = fft_conv(y)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, w = lse_grad(y, beta, c)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        if use_backtrack:
            step_local = step
            x_trial = project_simplex(y - step_local * grad, S)
            obj_trial = compute_obj(x_trial)
            bt = 0
            while obj_trial > cur_obj + 1e-12 and step_local > 1e-16 and bt < 6:
                step_local *= 0.5
                x_trial = project_simplex(y - step_local * grad, S)
                obj_trial = compute_obj(x_trial)
                bt += 1
            if bt > 3:
                step = step_local * 0.7
            else:
                step = min(step * 1.02, 3.0 / max(beta, 1.0))
            x_new = x_trial
        else:
            step = step_scale / max(beta, 1.0)
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:

        a = x_new
        t_mom = t_new

        c = fft_conv(y)

            noise_scale = max(1e-7, 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 3000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)

def run_peak_chase(a_init, max_time, rng, verbose=True):
    """
    Peak-chasing: iteratively minimize the current max lag using exact gradient.
    Uses backtracking line search and periodic multi-lag consideration.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_conv(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0 / max(c.max() * n / (S * S), 1e-10)

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 15000:
            break

        # Find top lags
        top_k = min(5, 2 * n - 1)
        if stuck < 500:
            use_k = 1
        elif stuck < 2000:
            use_k = 2
        else:
            use_k = min(top_k, 2 + stuck // 1000)

        top_idx = np.argsort(c)[::-1][:use_k]

        # Average gradient over top lags (weighted by how close to max)
        grad = np.zeros(n)
        for kk in top_idx:
            grad += grad_single_lag(a, int(kk))
        grad /= use_k

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            # Try a random perturbation
            idx = rng.integers(0, n)
            delta = a[idx] * 0.01
            if delta > 1e-12:
                a[idx] -= delta
                a[rng.integers(0, n)] += delta
                a = np.maximum(a, 0)
                a = a * (S / a.sum())
                c = fft_conv(a)
            continue

        # Backtracking line search
        step_local = step
        found = False
        for bt in range(8):
            a_new = project_simplex(a - step_local * grad, S)
            c_new = fft_conv(a_new)
            new_max = c_new.max()
            if new_max < cur_max - 1e-14:
                a = a_new
                c = c_new
                found = True
                break
            step_local *= 0.5
        else:
            # No improvement; try individual lags
            found = False
            for kk in top_idx:
                g_kk = grad_single_lag(a, int(kk))
                gn = np.linalg.norm(g_kk)
                if gn < 1e-15:
                    continue
                step_kk = S / (gn * np.sqrt(n))
                for bt2 in range(5):
                    a_try = project_simplex(a - step_kk * g_kk, S)
                    c_try = fft_conv(a_try)
                    if c_try.max() < cur_max - 1e-14:
                        a = a_try
                        c = c_try
                        found = True
                        step = step_kk * 0.8
                        break
                    step_kk *= 0.5
                if found:
                    break

        if not found:
            step *= 0.5
            if step < 1e-12:
                # Add noise to escape
                ns = max(1e-6, 0.001 * S / np.sqrt(n))
                a = project_simplex(a + rng.normal(0, ns, n), S)
                c = fft_conv(a)
                step = S / (n * max(c.max(), 1e-10))

        # Adaptive step adjustment
        if found:
            step = min(step * 1.01, 3.0 * S / (gnorm * np.sqrt(n)))

    return best_a, best_obj

                    break
            if not improved:
                # Try a random small perturbation
                if it % 100 == 0:
                    ns = max(1e-8, 1e-5 * S / np.sqrt(n) * np.exp(-stuck / 10000.0))
                    a = project_simplex(a + rng.normal(0, ns, n), S)
                    c = fft_conv(a)

    return best_a, best_obj


def run_coord_descent(a_init, max_time, rng):
    """
    Coordinate descent: transfer mass from high-peak-contributing coords
    to low-peak-contributing coords.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_conv(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > 30000:
            break

        k_star = int(np.argmax(c))
        # Contribution of each coordinate to c[k_star]
        # c[k_star] = sum_i a[i]*a[k_star-i], so coord j contributes a[j]*a[k_star-j]
        contrib = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        idx = np.arange(j_lo, j_hi + 1)
        contrib[idx] = a[idx] * a[k_star - idx]

        # Also consider second peak
        k2 = int(np.argsort(c)[::-1][1]) if len(c) > 1 else k_star
        contrib2 = np.zeros(n)
        j_lo2 = max(0, k2 - n + 1)
        j_hi2 = min(n - 1, k2)
        if j_hi2 >= j_lo2:
            idx2 = np.arange(j_lo2, j_hi2 + 1)
            contrib2[idx2] = a[idx2] * a[k2 - idx2]

        total_contrib = contrib + 0.5 * contrib2

        # Source: high contribution, large value
        # Target: low contribution, room to grow
        nonzero_mask = a > 1e-10
        src_scores = total_contrib * a
        tgt_scores = -total_contrib * (1.0 - a / max(a.max(), 1e-10))

        n_try = min(10, n)
        src_cands = np.argsort(src_scores)[::-1][:n_try]
        tgt_cands = np.argsort(tgt_scores)[:n_try]

        best_new_max = cur_max + 1e-12
        best_move = None

        for j in src_cands[:n_try]:
            for l in tgt_cands[:n_try]:
                if j == l or a[j] < 1e-10:
                    continue
                # Try different transfer amounts
                for frac in [0.02, 0.05, 0.1]:
                    delta = a[j] * frac
                    if delta < 1e-12:
                        continue
                    # Approximate effect on c
                    # c changes at lags affected by a[j] and a[l]
                    # For lag k: delta_c[k] = -delta*a[k-j] + delta*a[k-l] (approx)
                    # Focus on the peak lag
                    dc_peak = -delta * (a[k_star - j] if 0 <= k_star - j < n else 0)
                    dc_peak += delta * (a[k_star - l] if 0 <= k_star - l < n else 0)
                    # Also check if other lags might increase
                    # Approximate: check a few top lags
                    ok = True
                    for kk in np.argsort(c)[::-1][:3]:
                        dc_kk = -delta * (a[int(kk) - j] if 0 <= int(kk) - j < n else 0)
                        dc_kk += delta * (a[int(kk) - l] if 0 <= int(kk) - l < n else 0)
                        if c[int(kk)] + dc_kk > best_new_max:
                            ok = False
                            break
                    if ok and c[k_star] + dc_peak < best_new_max:
                        best_new_max = c[k_star] + dc_peak
                        best_move = (j, l, delta)

        if best_move is not None:
            j, l, delta = best_move
            a[j] -= delta
            a[l] += delta
            a = np.maximum(a, 0.0)
            # Update c approximately (will be corrected by periodic recompute)
            c = fft_conv(a)
        else:
            if stuck > 5000 and it % 50 == 0:
                ns = max(1e-7, 0.0005 * S / np.sqrt(n) * np.exp(-stuck / 10000))
                a = project_simplex(a + rng.normal(0, ns, n), S)
                c = fft_conv(a)

        if it % 50 == 0:
            c = fft_conv(a)

    return best_a, best_obj


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-2.0 * (tt - 0.5) ** 2 / (0.15 ** 2))
    elif method == "chirp":
        a = 0.5 + 0.5 * np.sin(2 * np.pi * 4 * tt + 0.7)
    elif method == "multi_tone":
        a = (0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt)
             + 0.2 * np.sin(2 * np.pi * 7 * tt + 1.3)
             + 0.15 * np.sin(2 * np.pi * 11 * tt + 2.1))
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.1
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 2.0 * np.abs(tt - 0.5))
    elif method == "flat":
        a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def upsample(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA screening, peak-chasing, coordinate descent, fine polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(1.0, budget_s - 20.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get starting point from height_sequence_1 if available
    a_start = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        raw = np.clip(raw, 0, 1000)
        if raw.sum() >= 0.01:
            a_start = raw
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        a_start = make_init(n, "flat", rng)

    n0 = len(a_start)
    S0 = np.sqrt(2.0 * n0)
    a_start = a_start * (S0 / a_start.sum())
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.8f" % (n0, obj0))
    global_best_obj = obj0
    global_best_a = a_start.copy()

    # === Phase 1: Quick FISTA screening with diverse parameters (15% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.15
    if p1_time > 30:
        print("\n=== Phase 1: FISTA screening (budget %.0fs) ===" % p1_time)
        configs = [
            (10, 0.01, 5000, 0.8, True),
            (50, 0.005, 20000, 0.6, True),
            (200, 0.003, 50000, 0.4, False),
            (5, 0.015, 3000, 1.2, True),
        ]
        per_time = p1_time / len(configs)
        for i, (bi, br, bm, ss, bt) in enumerate(configs):
            if time.monotonic() >= deadline:
                break
            rem = min(per_time, deadline - time.monotonic())
            if rem < 3:
                break
            a_opt, obj_opt = run_fista(global_best_a, rem, rng,
                                        beta_init=bi, beta_rate=br,
                                        beta_max=bm, step_scale=ss,
                                        use_backtrack=bt)
            tag = " ***" if obj_opt < global_best_obj else ""
            print("  FISTA cfg%d: obj=%.8f%s" % (i + 1, obj_opt, tag))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # === Phase 2: Peak-chasing (35% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.35
    if p2_time > 30:
        print("\n=== Phase 2: Peak-chasing (budget %.0fs) ===" % p2_time)
        a_pc, obj_pc = run_peak_chase(global_best_a, p2_time, rng)
        tag = " ***" if obj_pc < global_best_obj else ""
        print("  Peak chase: obj=%.8f%s" % (obj_pc, tag))
        if obj_pc < global_best_obj:
            global_best_obj = obj_pc
            global_best_a = a_pc.copy()

        # Follow up with FISTA on peak-chase result
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_f, obj_f = run_fista(global_best_a, remaining2 * 0.5, rng,
                                    beta_init=100, beta_rate=0.005,
                                    beta_max=100000, step_scale=0.5,
                                    use_backtrack=True)
            tag = " ***" if obj_f < global_best_obj else ""
            print("  FISTA after PC: obj=%.8f%s" % (obj_f, tag))
            if obj_f < global_best_obj:
                global_best_obj = obj_f
                global_best_a = a_f.copy()

    # === Phase 3: Coordinate descent (20% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.20
    if p3_time > 30:
        print("\n=== Phase 3: Coordinate descent (budget %.0fs) ===" % p3_time)
        a_cd, obj_cd = run_coord_descent(global_best_a, p3_time, rng)
        tag = " ***" if obj_cd < global_best_obj else ""
        print("  Coord descent: obj=%.8f%s" % (obj_cd, tag))
        if obj_cd < global_best_obj:
            global_best_obj = obj_cd
            global_best_a = a_cd.copy()

    # === Phase 4: Fine polish (25% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.25
    if p4_time > 20:
        print("\n=== Phase 4: Fine polish (budget %.0fs) ===" % p4_time)
        a_pol, obj_pol = run_fine_polish(global_best_a, p4_time, rng)
        tag = " ***" if obj_pol < global_best_obj else ""
        print("  Fine polish: obj=%.8f%s" % (obj_pol, tag))
        if obj_pol < global_best_obj:
            global_best_obj = obj_pol
            global_best_a = a_pol.copy()

    # === Phase 5: Final peak-chase + polish cycle (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 5: Final cycle (budget %.0fs) ===" % remaining)
        # Peak chase
        t1 = remaining * 0.4
        a_pc2, obj_pc2 = run_peak_chase(global_best_a, t1, rng)
        tag = " ***" if obj_pc2 < global_best_obj else ""
        print("  Peak chase 2: obj=%.8f%s" % (obj_pc2, tag))
        if obj_pc2 < global_best_obj:
            global_best_obj = obj_pc2
            global_best_a = a_pc2.copy()

        # Fine polish
        remaining5 = deadline - time.monotonic()
        if remaining5 > 10:
            a_pol2, obj_pol2 = run_fine_polish(global_best_a, remaining5, rng)
            tag = " ***" if obj_pol2 < global_best_obj else ""
            print("  Fine polish 2: obj=%.8f%s" % (obj_pol2, tag))
            if obj_pol2 < global_best_obj:
                global_best_obj = obj_pol2
                global_best_a = a_pol2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.8f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
```
