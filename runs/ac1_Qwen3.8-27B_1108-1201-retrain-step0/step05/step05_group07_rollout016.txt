4. Final coordinate-wise refinement

import numpy as np
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution via FFT."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # grad = 2 * correlation of w with a (reversed)
    # (a*a)_k gradient w.r.t. a is 2 * (flip-a shifted by k)
    # Sum over k: sum_k w_k * 2 * (flip-a)[k - (n-1) + offset]
    # This is 2 * ifft(fft(w) * conj(fft(a)))[:n] -- circular correlation
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def mirror_descent_step(a, c, S, eta, beta):
    """Mirror descent step using entropy regularizer."""
    n = len(a)
    # Compute subgradient via LSE
    grad = compute_lse_gradient(a, c, beta)
    # Remove mean component (tangent to simplex)
    grad = grad - grad.mean()
    # Multiplicative update
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - eta * grad
    # Renormalize
    a_new = np.exp(log_a_new - log_a_new.max())
    a_new *= S / a_new.sum()
    return a_new

def optimize_subgradient_momentum(a0, S, max_time, rng, beta_init=20.0, beta_growth=0.005, momentum_init=0.9):
    """
    Subgradient optimization with Nesterov momentum and BB step size.
    """
    n = len(a0)
    a = project_simplex(a0, S).copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    # Momentum state
    v = np.zeros(n)
    momentum = momentum_init
    momentum_decay = 0.9999
    
    # BB step size state
    prev_a = a.copy()
    prev_grad = None
    bb_step = None
    
    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 500
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = min(5000.0, beta_init * np.exp(beta_growth * it))
        
        grad = compute_lse_gradient(a, c, beta)
        grad = grad - grad.mean()  # Project onto tangent space
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # BB step size
        if prev_grad is not None and bb_step is None:
            s = a - prev_a
            y = grad - prev_grad
            sy = s @ y
            yy = y @ y
            if yy > 1e-20:
                bb_step = max(0.01, min(10.0, sy / yy))
        
        if bb_step is not None:
            step = bb_step
        else:
            step = 2.0 * (cur_max - c.mean()) / (gnorm * gnorm)
            step = min(step, 3.0 / gnorm)
        
        # Nesterov momentum
        v = momentum * v + grad
        v_norm = np.linalg.norm(v)
        if v_norm > 0:
            v = v / v_norm * step  # Scale to step size
        else:
            v = np.zeros(n)
        
        a_trial = project_simplex(a - v, S)
        
        # Line search
        c_trial = fft_convolve_self(a_trial)
        if c_trial.max() > cur_max * (1.0 + 1e-8):
            # Try smaller step
            for factor in [0.5, 0.25, 0.1, 0.05]:
                a_trial = project_simplex(a - factor * v, S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() <= cur_max * (1.0 + 1e-8):
                    break
        
        prev_a = a.copy()
        prev_grad = grad.copy()
        a = a_trial
        momentum = max(0.5, momentum * momentum_decay)
        
        # Periodic restart
        if stuck > patience:
            noise_scale = min(0.05, 0.01 * (stuck / patience)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            v = np.zeros(n)
            stuck = 0
            patience = int(patience * 1.2)
            bb_step = None
            prev_grad = None
        
        if it % 2000 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"    it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, beta={beta:.0f}, step={step:.4f}")
    
    return best_a, best

    """
    n = len(a0)

        else:
            stuck += 1

def generate_exponential_init(n, rng, decay_rate):
    """Generate exponential decay initialization."""
    t = np.arange(n)

def generate_gaussian_init(n, rng, center=0.5, width=0.15):
    """Generate Gaussian-shaped initialization."""

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Multi-phase approach combining:
1. LSE-smoothed subgradient with Nesterov momentum and BB step size
2. Mirror descent (multiplicative weights) for exploration
3. Adaptive beta schedule with periodic restarts
4. Final high-beta polishing
5. Multi-initialization and multi-length exploration
"""
import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution via FFT cross-correlation."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    # Cross-correlation: grad_j = 2 * sum_m w_{j+m} * a_m
    # = 2 * real(ifft(fft(w, fsize) * conj(fft(a, fsize))))[:n]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def obj_value(a):
    """Compute objective 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def optimize_momentum_subgrad(a0, S, max_time, rng, beta_init=20.0, beta_growth=0.004,
                              momentum_init=0.85, verbose=False):
    """
    Subgradient optimization with Nesterov momentum and adaptive step size.
    Uses LSE smoothing with increasing beta.
    """
    n = len(a0)
    a = project_simplex(a0, S).copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    v = np.zeros(n)  # momentum velocity
    momentum = momentum_init

    # BB step tracking
    prev_a = a.copy()
    prev_grad = None
    bb_step = None

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 400

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = min(8000.0, beta_init * np.exp(beta_growth * it))

        grad = compute_lse_gradient(a, c, beta)
        # Project gradient onto tangent space of simplex
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Compute step size
        if prev_grad is not None and (bb_step is None or it % 200 == 0):
            s = a - prev_a
            y = grad - prev_grad
            sy = s @ y
            yy = y @ y
            if yy > 1e-18 and sy > 0:
                bb_step = max(0.05, min(5.0, sy / yy))
            elif yy > 1e-18:
                bb_step = 1.0 / max(gnorm, 1e-10)

        if bb_step is not None:
            step = bb_step
        else:
            gap = max(0.0, cur_max - np.median(c))
            step = 2.0 * gap / (gnorm * gnorm)
            step = min(step, 4.0 / gnorm)

        # Nesterov-style momentum
        v = momentum * v + grad
        v_norm = np.linalg.norm(v)
        if v_norm > gnorm * 0.1:
            # Scale velocity to approximate step size
            scale = step / v_norm
            if scale > 5.0:
                scale = 5.0
            direction = v * scale
        else:
            direction = step * grad / gnorm

        a_trial = project_simplex(a - direction, S)

        # Backtracking line search
        c_trial = fft_convolve_self(a_trial)
        trial_max = c_trial.max()
        if trial_max > cur_max * (1.0 + 1e-9):
            for factor in [0.5, 0.25, 0.125, 0.0625]:
                a_trial = project_simplex(a - factor * direction, S)
                c_trial = fft_convolve_self(a_trial)
                trial_max = c_trial.max()
                if trial_max <= cur_max * (1.0 + 1e-9):
                    break
            if trial_max > cur_max * (1.0 + 1e-9):
                # Accept tiny step
                a_trial = project_simplex(a - 0.01 * direction, S)

        prev_a = a.copy()
        prev_grad = grad.copy()
        a = a_trial
        momentum = max(0.5, momentum * 0.99995)

        # Periodic restart when stuck
        if stuck > patience:
            noise_scale = min(0.04, 0.005 * (stuck / patience)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            v = np.zeros(n)
            stuck = 0
            patience = int(patience * 1.15)
            bb_step = None
            prev_grad = None

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"      it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}, beta={beta:.0f}")

    return best_a, best


def optimize_mirror_gradient(a0, S, max_time, rng, verbose=False):
    """
    Mixed optimization: alternates between mirror descent (multiplicative)
    and projected gradient steps.
    """
    n = len(a0)
    a = project_simplex(a0, S).copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 300

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = min(4000.0, 15.0 * np.exp(0.008 * it))

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        grad = compute_lse_gradient(a, c, beta)
        grad_centered = grad - grad.mean()
        gnorm = np.linalg.norm(grad_centered)
        if gnorm < 1e-14:
            break

        # Alternate: mirror descent (multiplicative) every 4th step, else projected
        if it % 4 == 0 and progress < 0.6:
            # Mirror descent step
            eta = 0.05 / (1.0 + 0.0005 * it)
            log_a = np.log(np.maximum(a, 1e-300))
            log_a_new = log_a - eta * grad_centered
            # Renormalize to maintain sum = S
            a_new = np.exp(log_a_new - log_a_new.max())
            a_new *= S / a_new.sum()
            a = a_new
        else:
            # Projected gradient step
            step = 2.0 * max(0.0, cur_max - np.median(c)) / (gnorm * gnorm)
            step = min(step, 4.0 / gnorm)
            a = project_simplex(a - step * grad_centered, S)

        # Restart if stuck
        if stuck > patience:
            noise_scale = 0.015 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.1)

        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - start
            cur_obj = 2.0 * n * best / (S * S)
            print(f"      it={it}, t={elapsed:.0f}s, obj={cur_obj:.6f}")

    return best_a, best


def optimize_high_beta_polish(a0, S, max_time, rng, beta_init=2000.0):
    """Final polishing with very high beta LSE and small steps."""
    n = len(a0)
    a = project_simplex(a0, S).copy()
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(50000.0, beta_init * (1.0 + 0.01 * it))

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 300:
            # Tiny perturbation to escape
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            continue

        grad = compute_lse_gradient(a, c, beta)
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (1.0 * beta)
        step = min(step, 2.0 / gnorm)
        step = min(step, 0.1 / gnorm)  # Very conservative

        a_try = project_simplex(a - step * grad, S)
        c_try = fft_convolve_self(a_try)
        if c_try.max() <= cur_max * (1.0 + 1e-10):
            a = a_try
        else:
            a = project_simplex(a - 0.5 * step * grad, S)

    return best_a, best


def generate_init_from_base(base, n, rng):
    """Interpolate a base sequence to length n."""
    if base is None or len(base) < 2:
        t = np.linspace(0, 1, n, endpoint=False)
        return 0.5 * (1 + np.cos(2 * np.pi * t)) + 0.05
    t_old = np.linspace(0, 1, len(base))
    t_new = np.linspace(0, 1, n)
    a = np.interp(t_new, t_old, base)
    return np.maximum(a, 0.0)


def generate_ramp_init(n):
    """Asymmetric ramp initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    return (1.0 - t) ** 1.5 + 0.01


def generate_exp_init(n, rate=1.0):
    """Exponential decay initialization."""
    t = np.arange(n, dtype=np.float64) / n
    return np.exp(-rate * t) + 0.01


def generate_gauss_init(n, center=0.3, width=0.15):
    """Gaussian bump initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    return np.exp(-0.5 * ((t - center) / width) ** 2) + 0.005
