import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def subgradient_at_lag(a, k):
    """Compute 2*shift_flip(a, k): the subgradient of b_k w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    # grad[m] = 2*a[k-m] if 0 <= k-m < n
    valid_m = np.arange(max(0, k - n + 1), min(n, k + 1))
    if len(valid_m) > 0:
        grad[valid_m] = 2.0 * a[k - valid_m]
    return grad

def lse_loss_and_grad(a, beta):
    """Compute LSE of b and its gradient."""
    n = len(a)
    b = fft_convolve_self(a)
    cm = b.max()
    shifted = beta * (b - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        lse = cm
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        lse = cm + np.log(tot) / beta
        w = e / tot
    # Gradient: dw/db * db/da, where dw/db = w * (beta - w) for LSE
    # Actually, for LSE(x) = (1/beta)*log(sum(exp(beta*x_i))), 
    # dLSE/dx_j = (beta*exp(beta*x_j)) / sum(exp(beta*x_i)) = w_j
    # So dLSE/da = sum_k w_k * db_k/da
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization combining:
1. Active-set subgradient with backtracking line search
2. LSE-smoothed FISTA with adaptive beta
3. Greedy per-active-lag coordinate descent
4. Diversification via structured perturbations
5. Final fine-tuning

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def subgradient_at_lag(a, k):
    """Subgradient of b_k = (a*a)_k w.r.t. a. Result: 2*shift_flip(a, k)."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        grad[m] = 2.0 * a[k - m]
    return grad


def lse_grad(a, b, beta):
    """Gradient of LSE_beta(b) w.r.t. a."""
    n = len(a)
    cm = b.max()
    shifted = beta * (b - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def backtracking_step(a, grad, S, f_current, gamma=0.5, rho=0.7):
    """Backtracking line search. Returns (new_a, new_step, improved)."""
    step = 1.0 / (np.linalg.norm(grad) + 1e-10)
    for _ in range(30):
        a_try = project_simplex(a - step * grad, S)
        b_try = fft_convolve_self(a_try)
        f_try = b_try.max()
        # Armijo condition: f_try <= f_current - gamma * step * ||grad||^2
        if f_try <= f_current - gamma * step * np.dot(grad, a - a) :
            break
        # Simpler: just check if f_try < f_current
        if f_try < f_current:
            break
        step *= rho
    else:
        return a, step, False
    return a_try, step, f_try < f_current

            best_obj = M
            best_a = a.copy()

def run_lse_fista(a_init, max_time, rng, beta_init=10.0, beta_max=50000.0):
    """LSE-smoothed FISTA."""

    S = np.sqrt(2.0 * n)
    a = normalize_seq(np.clip(a_init, 0, 1000), S).copy()
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()

            break

        step = max(step, 1e-14)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))

def run_diversify(a_init, max_time, rng):
