def solve_lp_direction(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build constraint: (f*g)[k] <= M for k = 0, ..., 2n-2
    # This is a sparse linear constraint
    # A @ g <= M
    
    # Instead of building the full matrix, use cvxpy's expression
    # f*g = convolution, which in cvxpy would be...
    # Actually, cvxpy doesn't have a built-in convolution.
    # We need to express it as a matrix multiplication.
    
    # The convolution matrix A is (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    
    # For efficiency, let's build this as a sparse matrix
    import scipy.sparse as sp
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    objective = cp.Maximize(cp.sum(g))

    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CLARABEL)
    
    return g.value

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Combined approach: FISTA with LSE smoothing, LP-based direction finding
with top-K constraints, multi-scale warm start, and L-BFGS polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize
import scipy.sparse as sp
try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax/LSE weights for smoothed max gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def topk_gradient(y, c, k, power=1.0):
    """Top-k weighted gradient for focus on active constraints."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power > 1:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
    else:
        w = np.ones(k)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    return conv_gradient(y, w_full)


def compute_obj(a):
    """Compute the actual objective value 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def build_conv_matrix_indices(n):
    """Precompute row/col indices for the convolution matrix (2n-1) x n."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_direction(f, M, top_k=50, max_time=30.0):
    """
    Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for top_k lags.
    Returns g or None if infeasible/too slow.
    """
    if not HAS_CVXPY:
        return None

    n = len(f)
    b = fft_convolve_self(f)
    # Get top_k lags
    top_idx = np.argsort(b)[::-1][:top_k]
    top_idx = np.sort(top_idx)

    time_start = time.monotonic()

    try:
        g = cp.Variable(n, nonneg=True)
        constraints = []

        # Build constraints for top lags using sparse rows
        for k in top_idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            coeffs = f[k - j_min:(k - j_max) + 1][::-1]
            indices = np.arange(j_min, j_max + 1)
            # Sum of f[k-j]*g[j] for j in [j_min, j_max]
            expr = cp.sum(cp.multiply(coeffs, g[indices]))
            constraints.append(expr <= M)

        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)

        solver = cp.CLARABEL if cp.CLARABEL in cp.installed_solvers() else cp.SCS
        prob.solve(solver=solver, verbose=False)

        if time.monotonic() - time_start > max_time:
            return None

        if prob.status in ['optimal', 'optimal_inaccurate']:
            g_val = np.array(g.value, dtype=np.float64)
            g_val = np.maximum(g_val, 0.0)
            if g_val.sum() > 1e-10:
                return g_val
        return None
    except Exception:
        return None


def run_fista_adaptive(a_init, max_time, rng, beta_start=10.0, beta_end=5000.0,
                       mode='anneal', step_scale=1.0):
    """
    FISTA with adaptive annealing beta schedule.
    Beta grows exponentially, with periodic drops to escape local minima.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: exponential growth with periodic resets
        progress = min(1.0, it / max(1000, max_time * 5))
        if mode == 'anneal':
            beta = beta_start * (beta_end / beta_start) ** progress
            # Periodic drop every 3000 iterations
            if it % 3000 == 0 and it > 1000:
                beta = max(beta_start, beta * 0.1)
        elif mode == 'osc':
            base = beta_start * (1.0 + 0.005 * it)
            osc = np.sin(2.0 * np.pi * it / 2000)
            beta = base * (1.0 + 0.5 * osc)
        else:
            beta = beta_start * (1.0 + 0.002 * it)

        beta = max(5.0, min(beta, 80000.0))
        step = step_scale / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Adaptive gradient: start with LSE, switch to top-k when stuck
        if stuck_count < 500:
            w = lse_weights(c, beta)
            grad = conv_gradient(y, w)
        elif stuck_count < 2000:
            k = min(64, 8 + stuck_count // 100)
            grad = topk_gradient(y, c, k, power=1.2)
        elif stuck_count < 5000:
            k = min(256, 64 + stuck_count // 100)
            grad = topk_gradient(y, c, k, power=1.0)
        else:
            # Very stuck: exact subgradient on all near-max
            tol = 0.01
            active = np.where(c >= c.max() * (1.0 - tol))[0]
            w_full = np.zeros(len(c))
            w_full[active] = 1.0 / len(active)
            grad = conv_gradient(y, w_full)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search (simplified)
        alpha = step
        for _ in range(3):
            x_try = project_simplex(y - alpha * grad, S)
            c_try = fft_convolve_self(x_try)
            if c_try.max() < cur_max + 0.01 * cur_max:
                break
            alpha *= 0.5

        # FISTA momentum update
        x_new = project_simplex(y - alpha * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Reset momentum periodically or when stuck
        if it % 4000 == 0 or stuck_count > 3000:
            t_mom = 1.0
            if stuck_count > 5000:
                # Add small structured noise
                noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
                y = project_simplex(y + noise, S)
                t_mom = 1.0
                stuck_count = 0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_lp_iteration(a_init, max_time, rng, top_k_start=30, top_k_end=200,
                     max_lp_time=15.0):
    """
    LP-based direction finding iteration.
    At each step, solve an LP to find the best direction g,
    then blend with current solution.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive top-k
        progress = min(1.0, it / 50.0)
        top_k = int(top_k_start + progress * (top_k_end - top_k_start))

        c = fft_convolve_self(a)
        M = c.max()

        # Solve LP
        g = solve_lp_direction(a, M, top_k=top_k, max_time=max_lp_time)

        if g is None:
            # Fallback: use top-k gradient step
            grad = topk_gradient(a, c, min(32, top_k), power=1.0)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = S / (gnorm * (1.0 + 0.01 * it))
            step = min(step, S / (0.5 * gnorm))
            a_new = project_simplex(a - step * grad, S)
        else:
            # Blend with LP solution
            g_sum = g.sum()
            if g_sum < S * 0.5:
                # LP didn't improve much, use gradient fallback
                grad = topk_gradient(a, c, min(32, top_k), power=1.0)
                gnorm = np.linalg.norm(grad)
                if gnorm < 1e-15:
                    break
                step = S / (gnorm * (1.0 + 0.01 * it))
                step = min(step, S / (0.5 * gnorm))
                a_new = project_simplex(a - step * grad, S)
            else:
                # Try different blend ratios
                best_blend_obj = np.inf
                a_new = a
                for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
                    g_norm = g * (S / g_sum)
                    candidate = (1 - t) * a + t * g_norm
                    candidate = np.maximum(candidate, 0)
                    cs = candidate.sum()
                    if cs < 1e-10:
                        continue
                    candidate = candidate * (S / cs)
                    obj_c = compute_obj(candidate)
                    if obj_c < best_blend_obj:
                        best_blend_obj = obj_c
                        a_new = candidate

        # Check improvement
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                # Restart from best with perturbation
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = project_simplex(best_a + noise, S)
                no_improve = 0

        if it % 10 == 0:
            elapsed = time.monotonic() - start
            print("    LP-iter %d: obj=%.8f (t=%.0fs, top_k=%d)" % (it, best_obj, elapsed, top_k))

    return best_a, best_obj


def run_lbfgs_polish(a_init, max_time, rng, n_levels=4):
    """L-BFGS-B polishing with increasing beta and penalty."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    configs = [
        (50.0, 5.0),
        (200.0, 10.0),
        (500.0, 20.0),
        (1000.0, 50.0),
        (2000.0, 100.0),
    ][:n_levels]

    for idx, (beta, pen) in enumerate(configs):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break

        c = fft_convolve_self(a)

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            w = e / e.sum()
            sm_val = cm_local + np.log(e.sum()) / beta
            sum_x = x.sum()
            obj = sm_val + pen * (sum_x - S) ** 2
            grad = conv_gradient(x, w)
            grad += 2.0 * pen * (sum_x - S)
            return obj, grad

        t0 = time.monotonic()
        try:
            result = scipy_minimize(
                obj_and_grad, a.copy(), method='L-BFGS-B',
                jac=True, bounds=[(0, 1000)] * n,
                options={'maxiter': 80, 'maxls': 20, 'ftol': 1e-14,
                         'gtol': 1e-12, 'iprint': -1}
            )
            a_new = np.clip(result.x, 0, 1000)
            a_new = a_new * (S / max(a_new.sum(), 1e-15))
            obj_new = compute_obj(a_new)
        except Exception:
            a_new = a.copy()
            obj_new = best_obj

        elapsed = time.monotonic() - t0
        print("    L-BFGS beta=%.0f pen=%.0f: obj=%.8f (t=%.1fs, iters=%d)" %
              (beta, pen, obj_new, elapsed, result.nit if 'result' in dir() else 0))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_active_polish(a_init, max_time, rng):
    """Intensive active-set polishing with exact subgradient."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive active set size
        if stuck < 50:
            tol_frac = 0.0
        elif stuck < 200:
            tol_frac = 0.001
        elif stuck < 500:
            tol_frac = 0.005
        elif stuck < 1500:
            tol_frac = 0.01
        else:
            tol_frac = 0.02

        active = np.where(c >= cur_max * (1.0 - tol_frac))[0]
        if len(active) > 200:
            # Keep only top 200
            top200 = np.argsort(c)[::-1][:200]
            active = top200

        w_full = np.zeros(len(c))
        w_full[active] = 1.0 / len(active)
        grad = conv_gradient(y, w_full)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = S / (gnorm * (1.0 + 0.005 * it))
        step = min(step, 2.0 * S / gnorm)
        step = max(step, 1e-8 * S / gnorm)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart when very stuck
        if stuck > 2000 and it % 200 == 0:
            noise_scale = max(1e-7, 0.0005 * S / np.sqrt(n) * np.exp(-stuck / 5000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample(a, n_new):
    """Upsample using interpolation + renormalization."""
    n_old = len(a)
    t_old = np.arange(n_old)
    t_new = np.linspace(0, n_old - 1, n_new)
    a_new = np.interp(t_new, t_old, a)
    a_new = np.maximum(a_new, 0)
    S_new = np.sqrt(2.0 * n_new)
    a_new = a_new * (S_new / max(a_new.sum(), 1e-15))
    return a_new


def generate_parametric_init(n, rng, S):
    """Generate a parametric initial sequence with cosine harmonics."""
    t = np.linspace(0, 1, n, endpoint=False)
    # Random harmonics
    a = np.ones(n) * 0.5
    for i in range(5):
        freq = rng.integers(1, 6)
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a

        print("  Active polish: obj=%.8f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: L-BFGS final squeeze (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = remaining * 0.10
        print("\n=== Phase 5: L-BFGS squeeze (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        a5, obj5 = run_lbfgs_polish(global_best_a, p5_budget, rng, n_levels=3)
        print("  Final squeeze: obj=%.8f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Try larger n (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p6_budget = min(remaining * 0.10, 120.0)
        n_large = 16384
        print("\n=== Phase 6: Larger n exploration (n=%d, budget %.0fs) ===" % (n_large, p6_budget))
        t6 = time.monotonic()

        # Upsample current best to larger n
        a_large = upsample(global_best_a, n_large)
        obj_large_init = compute_obj(a_large)
        print("  Upsampled init: obj=%.8f" % obj_large_init)

        a6, _ = run_fista_adaptive(a_large, p6_budget * 0.7, rng,
                                    beta_start=20.0, beta_end=20000.0,
                                    mode='anneal', step_scale=0.8)
        obj6 = compute_obj(a6)
        print("  Larger-n FISTA: obj=%.8f (t=%.0fs)" % (obj6, time.monotonic() - t6))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()
            print("  *** Improved with larger n! ***")

        rem6 = p6_budget - (time.monotonic() - t6)
        if rem6 > 15 and obj6 < global_best_obj + 1e-6:
            a6b, _ = run_active_polish(a6, rem6, rng)
            obj6b = compute_obj(a6b)
            print("  Larger-n polish: obj=%.8f" % obj6b)
            if obj6b < global_best_obj:
                global_best_obj = obj6b
                global_best_a = a6b.copy()

    # === Phase 7: Final precision polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 7: Final precision polish (budget %.0fs) ===" % remaining)
        t7 = time.monotonic()
        a7, obj7 = run_lbfgs_polish(global_best_a, remaining * 0.7, rng, n_levels=2)
        print("  Precision: obj=%.8f (t=%.0fs)" % (obj7, time.monotonic() - t7))
        if obj7 < global_best_obj:
            global_best_obj = obj7
            global_best_a = a7.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.8f (n=%d)" % (ver, len(final_list)))
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            S_fb = np.sqrt(2.0 * n_fb)
            global_best_a = np.ones(n_fb) * (S_fb / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        S_fb = np.sqrt(2.0 * n_fb)
        global_best_a = np.ones(n_fb) * (S_fb / n_fb)
        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.8f" % global_best_obj)
    return final_list
