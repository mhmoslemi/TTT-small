import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def grad_c_single(a, k):
    """Gradient of c[k] w.r.t. a. c[k] = sum_i a[i]*a[k-i]."""
    n = len(a)
    # grad = 2 * (a shifted by k), with appropriate truncation
    # More precisely, grad[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    # But since c[k] = sum_{i+j=k} a[i]*a[j], and both (i,j) and (j,i) appear:
    # grad[i] = a[k-i] (if valid) + a[i-k] (if valid and i != k-i)
    # Actually for the linear convolution, c[k] = sum_{i: max(0,k-n+1)<=i<=min(k,n-1)} a[i]*a[k-i]
    # grad[i] = a[k-i] if i is in the range and k-i is valid, times 2 if i != k-i
    # Simpler: use the FFT approach
    e = np.zeros(2 * n - 1)
    e[k] = 1.0
    fsize = 1 << (2 * n - 1 + n - 1).bit_length()
    E = fft(e, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(E * np.conj(A)))[:n]


def grad_c_weighted(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def hankel_matvec(a, w):
    """Compute A*a where A[i,j] = w[i+j]. Uses FFT."""
    n = len(a)
    a_rev = a[::-1]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    AR = fft(a_rev, fsize)
    conv_full = np.real(ifft(W * AR))
    return conv_full[n - 1: n - 1 + n]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_smooth(c, beta):
    """Smoothed LSE approximation of max(c)."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    return cm + np.log(e.sum()) / beta


def lse_grad(a, c, beta):
    """Gradient of LSE max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    w = e / e.sum()
    return grad_c_weighted(a, w)


def compute_obj_normalized(a):
    """Compute max(c) for normalized a (sum = sqrt(2n))."""
    c = fft_convolve_self(a)
    return c.max()
