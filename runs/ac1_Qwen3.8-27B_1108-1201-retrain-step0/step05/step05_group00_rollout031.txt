import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    """Self-convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c

    n = len(a)

    A = fft(a_pad)

def peak_weighted_grad(a, c, N, n, top_k, temperature):
    """
    Compute gradient of a peak-weighted objective.
    weight[k] = exp((c[k] - c_max) / temperature)
    grad = sum_k weight[k] * d(c[k])/d(a)
    """
    c_max = c.max()
    weights = np.exp((c - c_max) / temperature)
    # Focus on top-k peaks
    if top_k < len(c):
        top_idx = np.argsort(c)[::-1][:top_k]
        mask = np.zeros(len(c))
        mask[top_idx] = 1.0
        weights *= mask
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
        w_sum = 1.0
    weights /= w_sum
    
    # Gradient: d(c[k])/d(a[j]) = a[k-j] + a[j] (for valid indices)
    # Using FFT: grad = 2 * real(ifft(W * conj(A)))[:n]
    # where W = fft(weights)
    fsize = N
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_shaping_restart(a, N, rng, flatten_power=2.0):
    """
    Restart by flattening the magnitude spectrum.
    New magnitude = old_mag^alpha * mean_mag^(1-alpha) for alpha < 1.
    This reduces spectral peaks while preserving some structure.
    """
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    mean_mag = mag.mean()
    
    # Flatten: new_mag = mag^alpha * mean_mag^(1-alpha)
    alpha = 1.0 / flatten_power  # alpha < 1 means flattening
    new_mag = np.power(np.maximum(mag, 1e-10), alpha) * np.power(mean_mag, 1 - alpha)
    
    # Keep original phase for DC and a random fraction of others
    phase = np.angle(A)
    n_random = N // 2
    rand_idx = rng.choice(N, size=n_random, replace=False)
    new_phase = phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_random)
    
    A_new = new_mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def adaptive_peak_descent(a_init, n, S, N, max_time, rng,

    temp = initial_temp
    top_k = initial_top_k

        it += 1

        # If peaks are concentrated (low entropy of normalized peaks), reduce top_k
        # If peaks are spread out, increase top_k

        entropy = -np.sum(c_norm * np.log(np.maximum(c_norm, 1e-30)))
        max_entropy = np.log(len(c))

        grad = peak_weighted_grad(a, c, N, n, top_k, temp)

"""
Minimize 2*n*max_conv/sum^2 using adaptive L-BFGS with peak-focused subgradient
polishing, spectral shaping restarts, and multi-resolution strategies.

Key innovations over previous approach:
1. Adaptive peak-weighted gradients: dynamically adjusts which peaks to focus on
   based on the current peak distribution (entropy-based).
2. Spectral shaping restarts: explicitly flattens the magnitude spectrum rather
   than randomizing phases, creating more "flat-autocorrelation" structures.
3. Multi-peak subgradient polishing: blends subgradients from top-K peaks with
   adaptive weights to avoid see-sawing between peaks.
4. Trust-region subgradient steps: limits step size to prevent increasing
   secondary peaks, enabling safer convergence at the end.
5. Coarse-to-fine multi-resolution: optimize a downsampled version first for
   better global structure, then refine at full resolution.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_grad(a, N, n, beta):
    """Gradient of LSE-smoothed peak of self-convolution."""
    c = compute_conv(a, N)
    c_max = c.max()
    bc = beta * (c - c_max)
    exp_bc = np.exp(bc)
    total = exp_bc.sum()
    if total < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
    else:
        weights = exp_bc / total
    fval = c_max + np.log(total) / beta
    W = fft(weights, N)
    A = fft(a, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, fval, c, c_max


def peak_weighted_grad(a, c, N, n, top_k, temperature):
    """Gradient with adaptive peak weighting based on temperature."""
    c_max = c.max()
    weights = np.exp((c - c_max) / max(temperature, 1e-10))
    if top_k < len(c):
        top_idx = np.argsort(c)[::-1][:top_k]
        mask = np.zeros(len(c))
        mask[top_idx] = 1.0
        weights = weights * mask
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
        w_sum = 1.0
    weights /= w_sum
    W = fft(weights, N)
    A = fft(a, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def multi_peak_subgrad(a, c, N, n, top_k, blend_weights=None):
    """
    Compute blended subgradient from top-K peaks.
    Each peak's subgradient is weighted by its excess over the minimum of the top-K.
    """
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    min_top = top_vals[-1]
    if blend_weights is None:
        excess = top_vals - min_top
        w_sum = excess.sum()
        if w_sum < 1e-30:
            blend_weights = np.ones(top_k) / top_k
        else:
            blend_weights = excess / w_sum
    grad = np.zeros(n)
    for idx, w in zip(top_idx, blend_weights):
        k = int(idx)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n:
                grad[i] += w * (a[i] + a[j])
    return grad


def spectral_shaping_restart(a, N, rng, flatten_alpha=0.7, phase_rand_frac=0.3):
    """
    Restart by flattening the magnitude spectrum and partially randomizing phases.
    lower flatten_alpha = more flattening.
    """
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    mean_mag = np.mean(mag)
    new_mag = np.power(np.maximum(mag, 1e-10), flatten_alpha) * np.power(mean_mag, 1 - flatten_alpha)
    phase = np.angle(A)
    n_rand = max(1, int(N * phase_rand_frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = new_mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def spectral_phase_restart(a, N, rng, frac=0.4):
    """Randomize a fraction of FFT phases preserving magnitudes."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    new_phase = phase.copy()
    new_phase[rand_idx] = rng.uniform(0, 2 * np.pi, n_rand)
    A_new = mag * np.exp(1j * new_phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def lbfgs_block(a_init, n, S, N, max_time, rng, beta_init=8.0, beta_max=200.0, m_lbfgs=12):
    """
    L-BFGS on LSE-smoothed objective with adaptive beta and line search.
    """
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    s_hist = []
    y_hist = []
    start_time = time.monotonic()
    it = 0

    g, _, c, c_max = lse_grad(a, N, n, beta)

    while time.monotonic() - start_time < max_time:
        it += 1

        if it % 80 == 0:
            beta = min(beta_max, beta * 1.06)
            g, _, c, c_max = lse_grad(a, N, n, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()

        direction = lbfgs_dir_func(s_hist, y_hist, g, m_lbfgs)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break

        alpha = 1.5 * S / (n * dir_norm)
        best_candidate = a
        best_cm = c_max
        for ls in range(14):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max:
                best_candidate = candidate
                best_cm = cm_ls
                break
            best_candidate = candidate
            best_cm = cm_ls
            alpha *= 0.55

        if best_cm < best_cmax:
            best_cmax = best_cm
            best_a = best_candidate.copy()

        g_new, _, c_new, c_max_new = lse_grad(best_candidate, N, n, beta)
        s_k = best_candidate - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * max(ss, 1e-20) and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]

        a = best_candidate
        g = g_new
        c_max = c_max_new

    return best_a, best_cmax


def lbfgs_dir_func(s_hist, y_hist, g, m):
    """L-BFGS two-loop recursion (top-level function)."""
    n = len(g)
    m_actual = min(len(s_hist), m)
    gn = np.linalg.norm(g)
    if m_actual == 0:
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn

    s_h = s_hist[-m_actual:]
    y_h = y_hist[-m_actual:]

    q = g.copy()
    alphas = np.zeros(m_actual)
    for i in range(m_actual - 1, -1, -1):
        s_i = s_h[i]
        y_i = y_h[i]
        yy = np.dot(y_i, y_i)
        if yy > 1e-30:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i

    s_last = s_h[-1]
    y_last = y_h[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    for i in range(m_actual):
        s_i = s_h[i]
        y_i = y_h[i]
        yy = np.dot(y_i, y_i)
        if yy > 1e-30:
            beta_i = np.dot(s_i, y_i) / yy
            r = r + (alphas[i] - beta_i) * y_i

    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def polish_block(a_init, n, S, N, max_time, rng, use_multi_peak=True):
    """
    Polishing using adaptive peak-weighted gradient with momentum.
    Switches between LSE and exact subgradient based on peak concentration.
    """
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    momentum = np.zeros(n)
    beta_mom = 0.85
    start_time = time.monotonic()
    it = 0
    stagnant = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        c = compute_conv(a, N)
        c_max = c.max()

        if c_max < best_cmax - 1e-10:
            best_cmax = c_max
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 1000:
            break

        # Determine peak concentration
        c_sorted = np.sort(c)[::-1]
        top_sum = c_sorted[:10].sum()
        total_sum = c.sum()
        if total_sum < 1e-30:
            break
        concentration = top_sum / total_sum  # 1 = all in top 10, ~1/len(c) = uniform

        if concentration > 0.5 and use_multi_peak:
            # Peaks are concentrated: use multi-peak subgradient
            top_k = max(3, int(10 * concentration))
            grad = multi_peak_subgrad(a, c, N, n, top_k)
        else:
            # Peaks are spread: use LSE gradient
            beta = 5.0 + 30.0 * concentration
            grad, _, _, _ = lse_grad(a, N, n, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        direction = -momentum
        direction = direction - np.mean(direction)

        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-18:
            break

        alpha = S / (n * dnorm)
        best_candidate = a
        best_cm = c_max
        for ls in range(12):
            candidate = project_simplex(a + alpha * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max:
                best_candidate = candidate
                best_cm = cm_ls
                break
            best_candidate = candidate
            best_cm = cm_ls
            alpha *= 0.6

        a = best_candidate

        if stagnant > 200 and stagnant % 200 == 0:
            a = spectral_shaping_restart(a, N, rng, flatten_alpha=0.6, phase_rand_frac=0.2)
            momentum = np.zeros(n)
            stagnant = 0

    return best_a, best_cmax


def subgrad_polish(a_init, n, S, N, max_time, rng):
    """
    Ultra-fine polishing using exact subgradient on the single worst peak
    with careful trust-region steps.
    """
    a = project_simplex(np.asarray(a_init, dtype=np.float64), S)
    best_a = a.copy()
    best_cmax = compute_conv(a, N).max()
    start_time = time.monotonic()
    it = 0
    stagnant = 0
    last_alpha = S / (n * n)

    while time.monotonic() - start_time < max_time:
        it += 1

        c = compute_conv(a, N)
        c_max = c.max()

        if c_max < best_cmax - 1e-12:
            best_cmax = c_max
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 200:
            # Try a small perturbation
            if stagnant % 100 == 0:
                perturb_scale = 0.005 * S / n
                a = a + perturb_scale * rng.standard_normal(n)
                a = project_simplex(a, S)
                stagnant = 0
            elif stagnant > 500:
                break

        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = a[i] + a[j]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        direction = -grad
        direction = direction - np.mean(direction)
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-18:
            break

        # Trust region: ensure we don't increase secondary peaks too much
        # Find max alpha such that a + alpha*direction >= 0
        neg_mask = direction < 0
        if np.any(neg_mask):
            max_alpha_feasible = np.min(a[neg_mask] / (-direction[neg_mask]))
        else:
            max_alpha_feasible = np.inf

        base_alpha = min(last_alpha * 2.0, S / (n * dnorm))
        best_candidate = a
        best_cm = c_max

        for factor in [1.0, 0.7, 0.5, 0.3, 0.15, 0.07, 0.03]:
            alpha_try = min(base_alpha * factor, max_alpha_feasible * 0.99)
            if alpha_try <= 1e-15:
                continue
            candidate = a + alpha_try * direction
            candidate = np.maximum(candidate, 0.0)
            candidate = project_simplex(candidate, S)
            c_try = compute_conv(candidate, N)
            cm_try = c_try.max()
            if cm_try < best_cm:
                best_cm = cm_try
                best_candidate = candidate
                last_alpha = alpha_try

        a = best_candidate

    return best_a, best_cmax


def make_init_rand(n, rng):
    return rng.random(n) + 0.05


def make_init_chirp(n, rng):
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (3 * tt + 0.5 * tt * tt)) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_zadoff(n, rng):
    i = np.arange(n)
    c = rng.integers(1, n)
    phase = -np.pi * i * (2 * i + 1 - c) / n
    a = 1.0 + 0.5 * np.cos(phase) + 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_fourier(n, rng, N):
    P = rng.exponential(1.0, N)
    P = np.maximum(P, 1e-6)
    phase = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(P) * np.exp(1j * phase)
    a_full = np.real(ifft(A))[:n]
    return np.abs(a_full) + 0.01


def coarse_to_fine(a_coarse, factor, n, S):
    """Upsample a coarse solution to full resolution by interpolation."""
    n_coarse = len(a_coarse)
    n_fine = n
    # Simple linear interpolation
    x_coarse = np.linspace(0, 1, n_coarse, endpoint=False)
    x_fine = np.linspace(0, 1, n_fine, endpoint=False)
    a_fine = np.interp(x_fine, x_coarse, a_coarse)
    return project_simplex(a_fine, S)

            score = 2 * n * sc_new / (S * S)
            print("  [P3] score=%.8f (stall=%d)" % (score, stall))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall >= 2:
                a_curr = spectral_shaping_restart(a_curr, N, rng, flatten_alpha=0.5, phase_rand_frac=0.2)
                stall = 0

    # === Phase 4: Ultra-fine subgradient polishing (15% of budget) ===
    print("=== Phase 4: Ultra-fine subgradient polishing ===")
    t4 = time.monotonic()
    phase4_end = min(t4 + (deadline - t4) * 0.35, deadline)

    a_curr = global_a.copy()
    stall = 0

    while time.monotonic() < phase4_end:
        block_time = min(80.0, phase4_end - time.monotonic())
        if block_time < 10:
            break

        a_new, sc_new = subgrad_polish(a_curr, n, S, N, block_time, rng)
        if sc_new < global_best - 1e-10:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            score = 2 * n * sc_new / (S * S)
            print("  [P4] score=%.8f" % score)
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall >= 2:
                # Very small perturbation
                a_curr = a_curr + 0.002 * S / n * rng.standard_normal(n)
                a_curr = project_simplex(a_curr, S)
                stall = 0

    # === Phase 5: Final verification and short L-BFGS ===
    remaining = deadline - time.monotonic() - 10
    if remaining > 20:
        print("=== Phase 5: Final L-BFGS ===")
        a_fin, sc_fin = lbfgs_block(global_a, n, S, N, min(remaining, 150.0), rng,
                                     beta_init=30.0, beta_max=500.0, m_lbfgs=15)
        update_best(a_fin, "final_LBFGS")

    # Final verification
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification score: %.8f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(512, dtype=np.float64)

    final_score = 2.0 * len(global_a) * global_best / (S * S) if global_best < np.inf else float('inf')
    print("Returning length %d, peak=%.8f, approx score=%.8f" % (len(global_a), global_best, final_score))
    return [float(x) for x in global_a]
