import numpy as np

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def lse_smooth_max(c, beta):
    """Compute LSE approximation of max and its gradient w.r.t. the convolution output."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(len(c))
    lse_val = cm + np.log(tot) / beta
    w = e / tot
    return lse_val, w

def grad_of_lse_conv(a, w, beta):
    """Gradient of beta * LSE(convolve(a,a)) w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def support_reduction(a, max_time, rng):
    """Try zeroing out small entries and re-optimizing."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a_work = a.copy()
    best_obj = compute_obj(a_work)
    best_a = a_work.copy()
    
    start = time.monotonic()
    
    # Sort by magnitude
    magnitudes = a_work.copy()
    
    # Try various sparsity levels
    for frac in [0.1, 0.2, 0.3, 0.5]:
        if time.monotonic() - start > max_time:
            break
        threshold = np.quantile(magnitudes[magnitudes > 0], frac) * 0.1
        mask = a_work >= threshold
        if mask.sum() < n * 0.3:  # Don't zero out too much
            continue
        a_sparse = a_work.copy()
        a_sparse[~mask] = 0
        a_sparse = a_sparse * (S / a_sparse.sum())
        obj = compute_obj(a_sparse)
        if obj < best_obj:
            best_obj = obj
            best_a = a_sparse.copy()
    
    return best_a, best_obj

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)

    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta with plateau
        if it < 1000:
            beta = beta_init * np.exp(0.02 * it)
        elif it < 5000:
            beta = min(beta_plateau, beta_init * np.exp(0.02 * 1000) * (1.0 + 0.0005 * (it - 1000)))
        else:
            beta = min(beta_max, beta_plateau * np.exp(0.003 * (it - 5000)))

        c = fft_convolve_self(y)
        w = lse_weights(c, beta)
        grad = grad_conv_lse(y, w)

        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step = 2.0 / max(beta, 1.0)
        x_trial = project_simplex(y - step * grad, S)
        obj_trial = compute_obj(x_trial)
        bt = 0
        while obj_trial > cur_obj + 1e-12 and step > 1e-16 and bt < 10:
            step *= 0.5
            x_trial = project_simplex(y - step * grad, S)
            obj_trial = compute_obj(x_trial)
            bt += 1
        x_new = x_trial

        # Momentum with periodic restart
        stuck = it - last_improve
        if stuck > restart_every:
            t_mom = 1.0

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

        # Fourier-based structured noise when very stuck
        if stuck > noise_every and it % 100 == 0:
            # Generate noise in frequency domain (smooth perturbation)
            n_freq = min(n // 4, 20)
            freqs = rng.integers(1, n // 2, size=n_freq)
            phases = rng.uniform(0, 2 * np.pi, size=n_freq)
            amps = rng.uniform(0.001, 0.01, size=n_freq) * S / np.sqrt(n)
            noise = np.zeros(n)
            for f, p, amp in zip(freqs, phases, amps):
                noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + p)
            y = project_simplex(y + noise, S)
            t_mom = 1.0

    return best_a, best_obj


def lp_direction_finding(a, max_time_lp=20.0):
    """
    Find a good direction using LP with top-K active constraints.
    Solves: max sum(g) s.t. g >= 0, (a*g)_k <= M for top-K lags.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    c = fft_convolve_self(a)
    M = c.max()

    # Use top-K lags as constraints
    K = min(8, 2 * n - 1)
    top_lags = np.argsort(c)[::-1][:K]

    # Build constraint matrix A_eq for (a*g)_k <= M
    # (a*g)_k = sum_j a[j] * g[k-j] for valid j
    # Variables: g_0, ..., g_{n-1}
    # For each lag k, constraint: sum_{j valid} a[j] * g[k-j] <= M

    # For efficiency, build row by row
    A_ub = np.zeros((K, n))
    b_ub = np.full(K, M)

    for i, k in enumerate(top_lags):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[i, j] = a[j]

    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)

    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': max_time_lp, 'presolve': True})
        if result.status == 0 and result.x is not None:
            g = np.maximum(result.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-15:
                return g
    except Exception:
        pass

    # Fallback: heuristic direction based on contributions to peak
    peak_lag = int(np.argmax(c))
    contrib = np.zeros(n)
    lo = max(0, peak_lag - n + 1)
    hi = min(n - 1, peak_lag)
    for j in range(lo, hi + 1):
        contrib[j] = a[j] * a[peak_lag - j]

    # Direction: reduce high-contrib, increase low-contrib
    median_c = np.median(contrib[contrib > 0]) if np.any(contrib > 0) else 0
    g = np.ones(n)
    g[contrib > median_c * 2] = 0.1
    g[contrib < median_c * 0.5] = 2.0
    return g


def multiplicative_descent(a_init, max_time, rng, alpha_init=0.1, decay=0.999):
    """
    Mirror descent with multiplicative updates.
    Natural for non-negative variables: a_j <- a_j * exp(-step * grad_j).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 1e-10, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    alpha = alpha_init

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
            alpha = min(alpha * 1.02, 0.5)
        else:
            alpha *= decay

        # Top-k subgradient
        stuck = it - last_improve
        k_top = min(4, 2 * n - 1) if stuck < 1000 else min(16, 4 + stuck // 500)
        top_idx = np.argsort(c)[::-1][:k_top]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / k_top

        grad = grad_conv_lse(a, w)

        # Multiplicative update
        step = alpha / (np.max(np.abs(grad)) + 1e-15)
        step = min(step, 1.0)
        a = a * np.exp(-step * grad)
        a = np.maximum(a, 1e-15)
        a = a * (S / a.sum())

        if stuck > 20000:
            break

    return best_a, best_obj


def fourier_restart(a, rng, n_modes=10, amplitude=0.05):
    """Add structured Fourier noise and reproject."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    t = np.arange(n)
    noise = np.zeros(n)
    for _ in range(n_modes):
        f = rng.integers(1, n // 2)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 1.0) * amplitude * S / np.sqrt(n)
        noise += amp * np.cos(2 * np.pi * f * t / n + phase)
    a_new = project_simplex(a + noise, S)
    return a_new


def make_diverse_inits(n, rng, count=6):
    """Create diverse initial sequences."""
    S = np.sqrt(2.0 * n)
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []

    # Cosine
    a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    inits.append(np.maximum(a, 0.01) * (S / np.maximum(a, 0.01).sum()))

    # Multi-tone
    a = 0.3 + 0.25 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.sin(2 * np.pi * 7 * t + 1.3) + 0.15 * np.sin(2 * np.pi * 11 * t + 2.1)
    a = np.abs(a)
    inits.append(a * (S / a.sum()))

    # Random
    a = rng.random(n) + 0.05
    inits.append(a * (S / a.sum()))

    # Chirp
    a = 0.5 + 0.5 * np.sin(2 * np.pi * 5 * t + 0.3)
    inits.append(np.maximum(a, 0.01) * (S / np.maximum(a, 0.01).sum()))

    # Two-hump
    a = np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2)
    inits.append(a * (S / a.sum()))

    # Random Fourier
    a = np.ones(n)
    for _ in range(5):
        f = rng.integers(1, n // 4)
        phase = rng.uniform(0, 2 * np.pi)
        a += 0.3 * np.cos(2 * np.pi * f * t / 1.0 + phase)
    a = np.maximum(a, 0.01)
    inits.append(a * (S / a.sum()))

    return inits


def upsmooth(a, factor=2):
    """Upsample by interpolation and slight smoothing."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    a_up = np.interp(t_new, t_old, a)
    return a_up


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Try to get parent
    a_parent = None
    try:
        a_parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_parent = np.clip(a_parent, 0, 1000)
        if a_parent.sum() < 0.01:
            a_parent = None
    except Exception:
        a_parent = None

    # ===== Phase 1: Multi-n exploration (25% time) =====
    p1_time = (deadline - time.monotonic()) * 0.25
    if p1_time > 60:
        print("=== Phase 1: Multi-n exploration (budget %.0fs) ===" % p1_time)

        n_candidates = [8192]
        if a_parent is not None:
            n_candidates = [len(a_parent)]
            for nn in [4096, 16384]:
                if nn != len(a_parent):
                    n_candidates.append(nn)
        else:
            n_candidates = [4096, 8192, 16384]

        for n_try in n_candidates:
            if time.monotonic() >= deadline:
                break
            rem = deadline - time.monotonic()
            if rem < 20:
                break

            # Get best initialization for this n
            if a_parent is not None and n_try == len(a_parent):
                a_init = a_parent.copy()
                best_init_obj = compute_obj(a_init)
            else:
                inits = make_diverse_inits(n_try, rng, count=4)
                best_init_obj = np.inf
                a_init = inits[0]
                for inp in inits:
                    obj = compute_obj(inp)
                    if obj < best_init_obj:
                        best_init_obj = obj
                        a_init = inp
                print("  n=%d: best init obj=%.6f" % (n_try, best_init_obj))

            S = np.sqrt(2.0 * n_try)
            a_init = a_init * (S / a_init.sum())

            # Run FISTA
            t_alloc = min(rem * 0.6, p1_time * 0.6 / len(n_candidates))
            if t_alloc > 15:
                a_opt, obj_opt = run_fista_v2(a_init, t_alloc, rng,
                                               beta_init=20, beta_max=30000,
                                               beta_plateau=3000, restart_every=2000)
                print("  n=%d: FISTA obj=%.6f" % (n_try, obj_opt))
                if obj_opt < global_best_obj:
                    global_best_obj = obj_opt
                    global_best_a = a_opt.copy()

    # Update with parent if available
    if a_parent is not None:
        obj_p = compute_obj(a_parent)
        if obj_p < global_best_obj:
            global_best_obj = obj_p
            global_best_a = a_parent.copy()
            print("  Parent adopted: obj=%.6f" % obj_p)

    if global_best_a is None:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        global_best_obj = compute_obj(global_best_a)

    print("After Phase 1: best obj=%.6f, n=%d" % (global_best_obj, len(global_best_a)))

    # ===== Phase 2: Intensive FISTA with LP guidance (30% time) =====
    p2_time = (deadline - time.monotonic()) * 0.30
    if p2_time > 40:
        print("=== Phase 2: Intensive optimization (budget %.0fs) ===" % p2_time)

        n = len(global_best_a)

        # Run FISTA with different configs
        configs = [
            (15, 40000, 4000, 2000, 5000),
            (100, 80000, 8000, 3000, 8000),
            (500, 200000, 15000, 4000, 10000),
        ]

        for i, (bi, bm, bp, ri, ne) in enumerate(configs):
            if time.monotonic() >= deadline:
                break
            rem = min(p2_time * 0.35, deadline - time.monotonic())
            if rem < 10:
                break
            a_opt, obj_opt = run_fista_v2(global_best_a, rem, rng,
                                           beta_init=bi, beta_max=bm,
                                           beta_plateau=bp, restart_every=ri,
                                           noise_every=ne)
            print("  FISTA%d: obj=%.6f" % (i + 1, obj_opt))
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

        # LP-guided step
        rem = deadline - time.monotonic()
        if rem > 15 and len(global_best_a) <= 8192:
            g = lp_direction_finding(global_best_a, max_time_lp=min(rem * 0.5, 20.0))
            if g is not None:
                S = np.sqrt(2.0 * n)
                g = g * (S / g.sum())
                # Blend
                for t_blend in [0.1, 0.2, 0.3, 0.5]:
                    a_blend = (1 - t_blend) * global_best_a + t_blend * g
                    a_blend = np.maximum(a_blend, 0)
                    a_blend = a_blend * (S / a_blend.sum())
                    obj_blend = compute_obj(a_blend)
                    if obj_blend < global_best_obj:
                        global_best_obj = obj_blend
                        global_best_a = a_blend
                        print("  LP blend t=%.1f: obj=%.6f ***NEW BEST***" % (t_blend, obj_blend))

        # FISTA after LP
        rem = deadline - time.monotonic()
        if rem > 15:
            a_opt, obj_opt = run_fista_v2(global_best_a, rem * 0.7, rng,
                                           beta_init=50, beta_max=50000,
                                           beta_plateau=5000, restart_every=2000)
            print("  FISTA after LP: obj=%.6f" % obj_opt)
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # ===== Phase 3: Multiplicative descent (20% time) =====
    p3_time = (deadline - time.monotonic()) * 0.20
    if p3_time > 30:
        print("=== Phase 3: Multiplicative descent (budget %.0fs) ===" % p3_time)
        a_mult, obj_mult = multiplicative_descent(global_best_a, p3_time * 0.7, rng,
                                                   alpha_init=0.05, decay=0.9995)
        print("  Multiplicative: obj=%.6f" % obj_mult)
        if obj_mult < global_best_obj:
            global_best_obj = obj_mult
            global_best_a = a_mult.copy()

        # FISTA polish after multiplicative
        rem = deadline - time.monotonic()
        if rem > 15:
            a_opt, obj_opt = run_fista_v2(global_best_a, rem * 0.8, rng,
                                           beta_init=200, beta_max=100000,
                                           beta_plateau=10000, restart_every=3000)
            print("  FISTA after mult: obj=%.6f" % obj_opt)
            if obj_opt < global_best_obj:
                global_best_obj = obj_opt
                global_best_a = a_opt.copy()

    # ===== Phase 4: Upsample and optimize (15% time) =====
    p4_time = (deadline - time.monotonic()) * 0.15
    if p4_time > 30 and len(global_best_a) <= 8192:
        print("=== Phase 4: Upsample to n=%d ===" % (len(global_best_a) * 2))
        a_up = upsmooth(global_best_a, factor=2)
        n_up = len(a_up)
        S_up = np.sqrt(2.0 * n_up)
        a_up = a_up * (S_up / a_up.sum())
        obj_up = compute_obj(a_up)
        print("  Upsampled: obj=%.6f" % obj_up)
        if obj_up < global_best_obj:
            global_best_obj = obj_up
            global_best_a = a_up.copy()

        a_opt, obj_opt = run_fista_v2(a_up, p4_time * 0.7, rng,
                                       beta_init=5, beta_max=20000,
                                       beta_plateau=2000, restart_every=2000,
                                       noise_every=3000)
        print("  FISTA upsampled: obj=%.6f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

    # ===== Phase 5: Final high-beta polish (10% time) =====
    p5_time = (deadline - time.monotonic()) * 0.10
    if p5_time > 10:
        print("=== Phase 5: Final polish (budget %.0fs) ===" % p5_time)
        a_opt, obj_opt = run_fista_v2(global_best_a, p5_time * 0.6, rng,
                                       beta_init=1000, beta_max=300000,
                                       beta_plateau=50000, restart_every=1000,
                                       noise_every=2000)
        print("  High-beta polish: obj=%.6f" % obj_opt)
        if obj_opt < global_best_obj:
            global_best_obj = obj_opt
            global_best_a = a_opt.copy()

        # One more round with restart
        rem = deadline - time.monotonic()
        if rem > 10:
            a_restart = fourier_restart(global_best_a, rng, n_modes=5, amplitude=0.02)
            a_opt2, obj_opt2 = run_fista_v2(a_restart, rem, rng,
                                             beta_init=500, beta_max=200000,
                                             beta_plateau=20000, restart_every=1500)
            print("  Restart+polish: obj=%.6f" % obj_opt2)
            if obj_opt2 < global_best_obj:
                global_best_obj = obj_opt2
                global_best_a = a_opt2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified: obj=%.6f, n=%d" % (ver, len(final_list)))
        else:
            print("WARNING: invalid result, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
