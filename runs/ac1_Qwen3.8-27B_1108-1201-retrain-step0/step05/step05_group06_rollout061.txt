```python
"""
Flat-autocorrelation (minimum peak/mean) step-function search.

The objective f(a)=2*n*max(a*a)/sum(a)^2 is scale invariant and equals the
peak-to-average ratio of the linear self-convolution (a*a).  Minimizing it is
equivalent to making the autocorrelation as flat as possible with a>=0.

Algorithm (imperative, no closures):
  * Fast warmup: projected gradient on a log-sum-exp surrogate (top-lag
    smooth max) with backtracking.
  * Polishing engine (the differentiator): Successive Convex Approximation
    (SCA) that, each step, linearizes the top active lags of the convolution
    and solves a small LP to compute the best "flattening" step (sum-preserving,
    a>=0), then line-searches that step on the true objective.  Adaptive
    active-set growth guards non-active lags from blowing up.
  * Multiple diverse initializations and several lengths; always keep the best.
"""

import math
import time
import numpy as np
from scipy.fft import fft, ifft
import scipy.optimize as sco

S_CONST = 1.0


def _nextpow2(x):
    if x <= 1:
        return 1
    r = 1
    while r < x:
        r *= 2
    return r


def conv(a, b=None):
    n = len(a)
    size = 2 * n - 1
    N = _nextpow2(size)
    A = fft(a, N)
    if b is None:
        c = ifft(A * A)
    else:
        B = fft(b, N)
        c = ifft(A * B)
    return np.real(c)[:size]


def objective(a):
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s <= 1e-30:
        return np.inf
    return float(2.0 * len(a) * conv(a).max() / (s * s))


def normalize(x, S):
    x = np.maximum(np.asarray(x, dtype=np.float64), 0.0)
    s = x.sum()
    if s <= 1e-15:
        return np.full(len(x), S / len(x), dtype=np.float64)
    return (x * (S / s)).copy()


def project_simplex(v, S):
    n = len(v)
    u = np.sort(v)[::-1]
    cum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    valid = (u - (cum - S) / idx) > 0.0
    if np.any(valid):
        rho = int(np.max(np.where(valid)[0]))
    else:
        rho = n - 1
    theta = (cum[rho] - S) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    sw = w.sum()
    if sw < S - 1e-9:
        w = w + (S - sw) / n
    return w


def topk_idx(c, k):
    k = int(min(k, len(c)))
    if k <= 0:
        return np.empty(0, dtype=int)
    return np.argpartition(c, -k)[-k:]


def build_rows(a, idx):
    n = len(a)
    ar = a[::-1]
    rows = np.zeros((len(idx), n), dtype=np.float64)
    for t in range(len(idx)):
        k = int(idx[t])
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        s0 = n - 1 - k + lo
        e0 = n - 1 - k + hi + 1
        rows[t, lo:hi + 1] = 2.0 * ar[s0:e0]
    return rows


def make_inits(n, rng, prov=None):
    S = S_CONST
    out = []
    t = np.arange(n, dtype=np.float64)
    tiny = rng.normal(0, 2e-4, n)
    out.append(('uniform', normalize(np.ones(n) + tiny, S)))
    out.append(('ramp_up', normalize(t + 1.0, S)))
    out.append(('ramp_down', normalize(n - t, S)))
    out.append(('chirp', normalize(0.5 + 0.5 * np.cos(2.0 * np.pi * (t * t) / n), S)))
    ra = np.abs(rng.normal(0.0, 1.0, n))
    out.append(('randA', normalize(ra, S)))
    rb = rng.random(n) ** 2
    out.append(('randB', normalize(rb, S)))
    head = (t < 0.40 * n).astype(float) + 0.03 + 0.02 * rng.random(n)
    out.append(('head', normalize(head, S)))
    tail = (t > 0.55 * n).astype(float) + 0.03 + 0.02 * rng.random(n)
    out.append(('tail', normalize(tail, S)))
    if prov is not None and len(prov) == n:
        out.append(('provided', normalize(prov, S)))
    return out


def rank_inits(inits):
    ranked = []
    for nm, a in inits:
        f = objective(a)
        ranked.append((f, nm, a))
    ranked.sort(key=lambda z: z[0])
    return ranked


def lse_of_conv(c, beta):
    m = c.max()
    return float(m + (1.0 / beta) * math.log(np.exp(beta * (c - m)).sum() + 1e-300))


def coarse(a0, S, budget, rng, K=32, beta0=8.0, beta_max=60.0):
    n = len(a0)
    a = a0.copy()
    start = time.monotonic()
    end = start + budget
    best = a.copy()
    bestf = objective(a)
    beta = beta0
    L = 2.0
    it = 0
    last_imp = 0
    while time.monotonic() < end:
        c = conv(a)
        base = lse_of_conv(c, beta)
        idx = topk_idx(c, K)
        vals = c[idx]
        w = np.exp(beta * (vals - vals.max()))
        w = w / w.sum()
        g = np.zeros(n, dtype=np.float64)
        rows = build_rows(a, idx)
        g = w @ rows
        accept = False
        x = None
        for _ in range(8):
            x = project_simplex(a - g / L, S)
            cx = conv(x)
            if lse_of_conv(cx, beta) <= base + np.dot(g, x - a) + 0.5 * (1.0 / L) * np.dot(x - a, x - a) + 1e-10:
                accept = True
                break
            L *= 2.0
        a = x

            L = 2.0
    return best, bestf


def cap_floats(v):
    v = np.clip(v, -1e6, 1e6)
    return np.nan_to_num(v, nan=0.0, posinf=1e6, neginf=-1e6)
```

```python
"""
Flat-autocorrelation (minimum peak/mean) step-function search.

f(a)=2*n*max(a*a)/sum(a)^2 is scale invariant and equals the peak-to-average
ratio of the linear self-convolution (a*a).  Minimizing it = flattening the
autocorrelation of a nonnegative sequence.

Pipeline (top-level helpers, no closures / lambdas):
  * coarse warmup: projected gradient on a smoothed top-lag (log-sum-exp)
    surrogate with backtracking;
  * SCA polish: linearize the top active conv lags, solve a small LP for the
    best sum-preserving flattening direction (a>=0), line-search it on the
    true objective; adaptive active-set growth guards other lags; trust-region
    shrinks (with occasional kicks) on stall;
  * diverse initializations x several lengths; keep the global best.
"""

import math
import time
import numpy as np
import scipy.optimize as sco
from scipy.fft import fft, ifft

S_CONST = 1.0


def _nextpow2(x):
    if x <= 1:
        return 1
    r = 1
    while r < x:
        r *= 2
    return r


def conv(a, b=None):
    n = len(a)
    size = 2 * n - 1
    N = _nextpow2(size)
    A = fft(a, N)
    if b is None:
        c = ifft(A * A)
    else:
        c = ifft(A * fft(b, N))
    return np.real(c)[:size]


def objective(a):
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s <= 1e-30:
        return np.inf
    return float(2.0 * len(a) * conv(a).max() / (s * s))


def normalize(x, S):
    x = np.maximum(np.asarray(x, dtype=np.float64), 0.0)
    s = x.sum()
    if s <= 1e-15:
        return np.full(len(x), S / len(x), dtype=np.float64)
    return (x * (S / s)).copy()


def project_simplex(v, S):
    n = len(v)
    u = np.sort(v)[::-1]
    cum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    valid = (u - (cum - S) / idx) > 0.0
    rho = int(np.max(np.where(valid)[0])) if np.any(valid) else (n - 1)
    theta = (cum[rho] - S) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    if w.sum() < S - 1e-9:
        w = w + (S - w.sum()) / n
    return w


def topk_idx(c, k):
    k = int(min(k, len(c)))
    return np.argpartition(c, -k)[-k:] if k > 0 else np.empty(0, dtype=int)


def build_rows(a, idx):
    n = len(a)
    ar = a[::-1]
    rows = np.zeros((len(idx), n), dtype=np.float64)
    for t in range(len(idx)):
        k = int(idx[t])
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        rows[t, lo:hi + 1] = 2.0 * ar[n - 1 - k + lo: n - 1 - k + hi + 1]
    return rows


def make_inits(n, rng, prov=None):
    S = S_CONST
    t = np.arange(n, dtype=np.float64)
    out = [
        ('uniform', normalize(np.ones(n) + rng.normal(0, 2e-4, n), S)),
        ('ramp_up', normalize(t + 1.0, S)),
        ('ramp_down', normalize(n - t, S)),
        ('chirp', normalize(0.5 + 0.5 * np.cos(2.0 * np.pi * (t * t) / n), S)),
        ('randA', normalize(np.abs(rng.normal(0.0, 1.0, n)), S)),
        ('randB', normalize(rng.random(n) ** 2, S)),
        ('head', normalize((t < 0.40 * n).astype(float) + 0.03 + 0.02 * rng.random(n), S)),
        ('tail', normalize((t > 0.55 * n).astype(float) + 0.03 + 0.02 * rng.random(n), S)),
    ]
    if prov is not None and len(prov) == n:
        out.append(('provided', normalize(prov, S)))
    return out


def rank_inits(inits):
    ranked = [(objective(a), nm, a) for (nm, a) in inits]
    ranked.sort(key=lambda z: z[0])
    return ranked


def lse_of(c, beta):
    m = c.max()
    return float(m + (1.0 / beta) * math.log(np.exp(beta * (c - m)).sum() + 1e-300))


def coarse(a0, S, budget, rng, K=32, beta0=8.0, beta_max=60.0):
    n = len(a0)
    a = a0.copy()
    end = time.monotonic() + budget
    best, bestf = a.copy(), objective(a)
    beta, L, it, last = beta0, 2.0, 0, 0
    while time.monotonic() < end:
        c = conv(a)
        base = lse_of(c, beta)
        idx = topk_idx(c, K)
        w = np.exp(beta * (c[idx] - c[idx].max()))
        w = w / w.sum()
        g = w @ build_rows(a, idx)
        x = a
        for _ in range(8):

            if lse_of(conv(x), beta) <= base + np.dot(g, x - a) + 0.5 * (1.0 / L) * np.dot(x - a, x - a) + 1e-10:
                break
            L *= 2.0
        a = x

            c = conv(a)

    return best, bestf


def _cap(v):
    return np.nan_to_num(np.clip(v, -1e6, 1e6), nan=0.0, posinf=1e6, neginf=-1e6)


def _solve_lp(a, c, aset, delta):
    n = len(a)
    idx = np.asarray(aset, dtype=int)
    A_ub = np.hstack([build_rows(a, idx), -np.ones((len(idx), 1))])
    co = np.zeros(n + 1)

    A_eq = np.ones((1, n + 1)); A_eq[0, n] = 0.0
    B = list(zip(_cap(-a), np.full(n, float(delta))))
    B.append((None, None))
    try:
        res = sco.linprog(co, A_ub=A_ub, b_ub=-c[idx].copy(), A_eq=A_eq,
                          b_eq=np.zeros(1), bounds=B, method='highs',
                          options={'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None or not np.isfinite(res.x).all():
        return None
    return res.x[:n]
```

```python
Flat-autocorrelation (minimum peak/mean) step-function search.

ratio of the linear self-convolution (a*a).  Minimizing it is equivalent to

    if x <= 1:
        return 1
    r = 1
    while r < x:
        r *= 2
    return r


def conv(a, b=None):
    n = len(a)
    size = 2 * n - 1
    N = _nextpow2(size)
    A = fft(a, N)
    if b is None:
        c = ifft(A * A)
    else:
        c = ifft(A * fft(b, N))
    return np.real(c)[:size]

    w = np.maximum(v - theta, 0.0)
    if w.sum() < S - 1e-9:
        w = w + (S - w.sum()) / n
    return w


def topk_idx(c, k):
    k = int(min(k, len(c)))
    return np.argpartition(c, -k)[-k:] if k > 0 else np.empty(0, dtype=int)


def build_rows(a, idx):
    """Row k of gradient dc_k/da: 2*a_{k-r} (valid r)."""
    n = len(a)
    ar = a[::-1]
    rows = np.zeros((len(idx), n), dtype=np.float64)
    for t in range(len(idx)):
        k = int(idx[t])
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        rows[t, lo:hi + 1] = 2.0 * ar[n - 1 - k + lo: n - 1 - k + hi + 1]
    return rows


def make_inits(n, rng, prov=None):
    S = S_CONST
    t = np.arange(n, dtype=np.float64)
    out = [
        ('uniform', normalize(np.ones(n) + rng.normal(0, 2e-4, n), S)),
        ('ramp_up', normalize(t + 1.0, S)),
        ('ramp_down', normalize(n - t, S)),
        ('chirp', normalize(0.5 + 0.5 * np.cos(2.0 * np.pi * (t * t) / n), S)),
        ('randA', normalize(np.abs(rng.normal(0.0, 1.0, n)), S)),
        ('randB', normalize(rng.random(n) ** 2, S)),
        ('head', normalize((t < 0.40 * n).astype(float) + 0.03 + 0.02 * rng.random(n), S)),
        ('tail', normalize((t > 0.55 * n).astype(float) + 0.03 + 0.02 * rng.random(n), S)),
    ]
    if prov is not None and len(prov) == n:
        out.append(('provided', normalize(prov, S)))
    return out


def rank_inits(inits):
    ranked = [(objective(a), nm, a) for (nm, a) in inits]
    ranked.sort(key=lambda z: z[0])
    return ranked


def lse_of(c, beta):
    m = c.max()
    return float(m + (1.0 / beta) * math.log(np.exp(beta * (c - m)).sum() + 1e-300))


def coarse(a0, S, budget, rng, K=32, beta0=8.0, beta_max=60.0):
    n = len(a0)
    a = a0.copy()
    end = time.monotonic() + budget
    best, c = a.copy(), conv(a)
    bestf = 2.0 * n * c.max()
    beta, L, it, last = beta0, 2.0, 0, 0
    while time.monotonic() < end:
        c = conv(a)
        base = lse_of(c, beta)
        idx = topk_idx(c, K)
        w = np.exp(beta * (c[idx] - c[idx].max()))
        w = w / w.sum()
        g = w @ build_rows(a, idx)
        x = a
        for _ in range(8):
            x = project_simplex(a - g / L, S)
            if lse_of(conv(x), beta) <= base + np.dot(g, x - a) + 0.5 * (1.0 / L) * np.dot(x - a, x - a) + 1e-10:
                break
            L *= 2.0
        a = x
        it += 1
        ca = conv(a)
        if 2.0 * n * ca.max() < bestf:
            bestf, best = 2.0 * n * ca.max(), a.copy()
            last = it
        if it - last > 400:
            beta = min(beta_max, beta * 1.25)
        if it % 400 == 0:
            beta, L = max(beta0, beta * 0.75), 2.0
    return best, bestf


def _cap(v):
    return np.nan_to_num(np.clip(v, -1e6, 1e6), nan=0.0, posinf=1e6, neginf=-1e6)


def _solve_lp(a, c, aset, delta):
    n = len(a)
    idx = np.asarray(aset, dtype=int)
    A_ub = np.hstack([build_rows(a, idx), -np.ones((len(idx), 1))])
    co = np.zeros(n + 1)
    co[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    B = list(zip(_cap(-a), np.full(n, float(delta))))
    B.append((None, None))
    try:
        res = sco.linprog(co, A_ub=A_ub, b_ub=-c[idx].copy(), A_eq=A_eq,
                          b_eq=np.zeros(1), bounds=B, method='highs',
                          options={'presolve': True})
    except Exception:
        return None
    if res.status != 0 or res.x is None or not np.isfinite(res.x).all():
        return None
    return res.x[:n]


def sca_polish(a0, S, budget, rng, K0=192, Kmax=512, delta0=None):
    n = len(a0)
    mean = S / n
    a = normalize(a0, S)
    c = conv(a)
    end = time.monotonic() + budget
    best, bestf = a.copy(), 2.0 * n * c.max()
    delta = float(delta0) if delta0 is not None else 3.0 * mean
    K, it, fai = K0, 0, 0
    while time.monotonic() < end:
        fp = 2.0 * n * c.max()
        if fp < bestf:
            bestf, best = fp, a.copy()
        it0 = time.monotonic()
        idx = [int(i) for i in topk_idx(c, K)]
        base = set(idx)
        aset = idx
        d, ok = None, False
        for _ in range(3):
            d = _solve_lp(a, c, aset, delta)
            if d is None:
                ok = False
                break
            la = c + 2.0 * conv(a, d)
            gk = int(np.argmax(la))
            if la[gk] > c.max() + 1e-12 and gk not in base:
                aset.append(gk)
                base.add(gk)
                continue
            ok = True
            break
        it += 1
        per = time.monotonic() - it0
        if not ok:
            fai += 1
            delta = max(0.05 * mean, delta * 0.7)
            if fai >= 40:
                fai = 0
                kick = rng.normal(0.0, 0.03 * mean, n)
                kick -= kick.mean()
                ak = normalize(np.maximum(a + kick, 0.0), S)
                ca = conv(ak)
                if 2.0 * n * ca.max() < fp:
                    a, c = ak, ca
            if it % 25 == 0:
                print("  [SCA-fail] n=%d it=%d best=%.6f delta=%.3g K=%d" % (n, it, bestf, delta, K))
            continue
        if np.linalg.norm(d) < 1e-13:
            delta = max(0.02 * mean, delta * 0.7)
            continue
        fs, bestcand = [], None
        for gfac in (1.0, 0.5, 0.25, 0.125):
            a2 = a + gfac * d
            if a2.min() < -1e-9:
                continue
            a2 = np.maximum(a2, 0.0)
            ca2 = conv(a2)
            f2 = 2.0 * n * ca2.max()
            fs.append(f2)
            if bestcand is None or f2 < bestcand[0]:
                bestcand = (f2, a2, ca2)
        if bestcand is not None and bestcand[0] < fp - 1e-13:
            a = bestcand[1]
            a = a / max(a.sum(), 1e-30)
            c = bestcand[2]
        if it % 50 == 0:
            if per < 0.05 and K < Kmax:
                K = min(Kmax, K + 16)
            elif per > 0.28 and K > 96:
                K = max(96, K - 16)
            if bestf < 1.51 and delta > 0.2 * mean:
                delta = delta * 0.88
            print("  [SCA] n=%d it=%d best=%.6f K=%d delta=%.3g" % (n, it, bestf, K, delta))
    return best, bestf


def explore_n(n, budget, rng, prov=None, global_best=None):
    inits = make_inits(n, rng, prov)
    if global_best is not None and len(global_best) == n:
        inits.append(('globalbest', normalize(global_best, S_CONST)))
    ranked = rank_inits(inits)
    top = ranked[:2]
    for f, nm, a in top:
        if f < 1.52:
            top = top[:1]
            break
    mb_f, mb = np.inf, top[0][2].copy()
    fb = min(110.0, 0.28 * budget)
    for f, nm, a in top:
        af, bf = coarse(a, S_CONST, max(8.0, fb / len(top)), rng)
        if bf < mb_f:
            mb_f, mb = bf, af
    sb = budget - fb - 15.0
    if sb > 20:
        asca, bf2 = sca_polish(mb, S_CONST, sb, rng)
        if bf2 < mb_f:
            mb_f, mb = bf2, asca
    print("  explore n=%d best=%.6f" % (n, mb_f))
    return mb, mb_f


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    usable = max(120.0, budget_s - 70.0)
    t0 = time.monotonic()
    print("=== flat-autocorrelation SCA search (budget=%.0fs) ===" % budget_s)

    prov = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        p = np.clip(p, 0.0, 1000.0)
        if p.sum() > 0.01:
            prov = p
    except Exception:
        prov = None

    nN = 8192
    if prov is not None and len(prov) == nN:
        a_init = normalize(prov, S_CONST)
    else:
        a_init = normalize(np.arange(nN, dtype=np.float64) + 1.0, S_CONST)
    best_a, best_f = a_init.copy(), objective(a_init)
    print("initial n=%d f=%.6f" % (nN, best_f))

    # Phase 1: polish the best-known seed (the main improvement engine).
    b1 = 0.30 * usable
    s1, f1 = sca_polish(a_init, S_CONST, 0.55 * b1, rng, delta0=3.0 * (S_CONST / nN))
    if f1 < best_f:
        best_f, best_a = f1, s1
    print("phase1(provided) best=%.6f t=%.0fs" % (best_f, time.monotonic() - t0))
    if prov is not None and time.monotonic() - t0 < budget_s - 160:
        rk = rank_inits(make_inits(nN, rng, prov))
        other = rk[0][2]
        s2, f2 = sca_polish(normalize(best_a, S_CONST), S_CONST, 0.22 * b1, rng)
        if f2 < best_f:
            best_f, best_a = f2, s2
        s3, f3 = sca_polish(other, S_CONST, max(60.0, 0.23 * b1), rng)
        if f3 < best_f:
            best_f, best_a = f3, s3
    print("phase1 done best=%.6f t=%.0fs" % (best_f, time.monotonic() - t0))

    # Phase 2: diverse exploration across lengths / initializations.
    b2 = 0.55 * usable
    for n, fr in [(2048, 0.10), (4096, 0.20), (8192, 0.44), (16384, 0.26)]:
        if time.monotonic() - t0 > budget_s - 130:
            break
        bsub = min(b2 * fr, max(0.0, (budget_s - 130) - (time.monotonic() - t0)))
        if bsub < 60:
            continue
        try:
            aa, ff = explore_n(n, bsub, rng, prov, global_best=best_a)
            if ff < best_f:
                best_f, best_a = ff, aa
        except Exception as exc:
            print("  explore_n n=%d error: %r" % (n, exc))
    print("phase2 done best=%.6f t=%.0fs" % (best_f, time.monotonic() - t0))

    # Phase 3: final SCA polish on the global best.
    t_left = (budget_s - 70) - (time.monotonic() - t0)
    if t_left > 45:
        sf, ff = sca_polish(normalize(best_a, S_CONST), S_CONST, t_left, rng,
                            K0=256, Kmax=512, delta0=1.2 * (S_CONST / best_a.size))
        if ff < best_f:
            best_f, best_a = ff, sf
    print("phase3 done best=%.6f t=%.0fs" % (best_f, time.monotonic() - t0))

    final = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0)
    final = np.nan_to_num(final, nan=0.0, posinf=1000.0, neginf=0.0)
    if final.sum() < 0.05:
        final = np.full(8192, S_CONST / 8192)
    out = [float(v) for v in final]
    try:
        print("=== FINAL evaluate_sequence => %.6f (n=%d) ===" % (evaluate_sequence(out), len(out)))
    except Exception as exc:
        print("final check skipped:", exc)
    return out
```
