import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def fourier_sqrt_step(a, alpha, S):
    """One step of the Fourier convolution square root method."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    # Current convolution
    A = fft(a, fsize)
    B = A * A
    b = np.real(ifft(B))[:size]
    
    # Target: move towards flat
    b_mean = b.sum() / size  # = S^2 / (2n-1)
    b_target = alpha * b + (1 - alpha) * b_mean
    
    # Compute square root
    Bt = fft(b_target, fsize)
    # Ensure non-negative real part (PSD condition)
    Bt_real = np.real(Bt)
    Bt_clipped = np.maximum(Bt_real, 0.0)
    A_new = np.sqrt(Bt_clipped)
    
    # Preserve DC
    A_new[0] = A[0]
    
    a_new = np.real(ifft(A_new, fsize))[:n]
    return project_simplex(a_new, S), b

def kkt_step(a, S, k_top=32, beta=50.0):
    """KKT-guided subgradient step with adaptive top-k weights."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    b = fft_convolve(a)
    bmax = b.max()
    bmean = b.sum() / size
    
    # Adaptive top-k
    # Use weights proportional to smooth "excess" above mean
    scaled = beta * (b - bmax)
    e = np.exp(scaled - scaled.max())  # stable softmax-like
    w = e / e.sum()
    
    # Compute gradient: g_j = sum_k w_k * (a_{k-j} + a_{j-k})
    # This is: 2 * ifft(fft(w_padded) * conj(fft(a_padded)))[:n]
    W = fft(w, fsize)
    A = fft(a, fsize)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return g, bmax, b

def adaptive_kkt_optimize(a, S, max_time, rng, 
                          k_top_init=16, k_top_max=128,
                          beta_init=10.0, beta_max=5000.0):
    """Adaptive KKT optimization with momentum and line search."""
    n = len(a)
    a = normalize(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    
    b0 = fft_convolve(a)
    best = b0.max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = elapsed / max_time
        
        beta = min(beta_max, beta_init * np.exp(0.01 * it))
        k_top = min(k_top_max, k_top_init + int((k_top_max - k_top_init) * progress))
        
        g, bmax, b = kkt_step(y, S, k_top, beta)
        
        if bmax < best:
            best = bmax
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        step = 2.0 * (best - bmax + 1e-10) / (gnorm * gnorm + 1e-20)
        step = min(step, 1.0)  # cap step size
        
        # Backtracking
        for _ in range(10):
            x_new = project_simplex(y - step * g, S)
            b_new = fft_convolve(x_new)
            if b_new.max() < bmax + 1e-12:
                break
            step *= 0.5
        
        # Momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Stuck detection
        if it - last_improve > 5000:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
            last_improve = it
    
    return best_a, best

def fourier_sqrt_optimize(a, S, max_time, rng, alpha_init=0.95, alpha_min=0.01):
    """Optimize using the Fourier square root fixed-point method."""
    n = len(a)
    a = normalize(a, S)
    start = time.monotonic()
    
    b0 = fft_convolve(a)
    best = b0.max()
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        alpha = alpha_init * (alpha_min / alpha_init) ** progress
        
        a_new, b = fourier_sqrt_step(a, alpha, S)
        bmax = b.max()
        
        if bmax < best:
            best = bmax
            best_a = a_new.copy()
        
        # Over-relaxation
        t = 0.5 + 0.5 * np.exp(-it / 100.0)
        a = (1 - t) * a + t * a_new
        a = project_simplex(a, S)
    
    return best_a, best

def spectral_perturb(a, S, rng, scale=0.3):
    """Structured spectral perturbation: modify a few random frequency components."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    A = fft(a, fsize)
    # Pick a few random frequencies to perturb
    n_perturb = max(1, n // 64)
    freqs = rng.choice(range(1, fsize // 2), size=min(n_perturb, fsize // 2 - 1), replace=False)
    
    for f in freqs:
        # Perturb the magnitude and/or phase
        mag_scale = 1.0 + rng.normal(0, scale)
        phase_shift = rng.normal(0, scale * 0.5)
        A[f] = A[f] * mag_scale * np.exp(1j * phase_shift)
        A[fsize - f] = A[fsize - f] * mag_scale * np.exp(1j * phase_shift)
    
    A[0] = A[0]  # Preserve DC
    a_new = np.real(ifft(A, fsize))[:n]
    return project_simplex(a_new, S)

    n = len(a)

        progress = elapsed / max_time
        
        # Choose method: FSR early, KKT later
        r = rng.random()
        if progress < 0.5 and r < 0.6:

                                                alpha_init=0.9, alpha_min=0.1)
        else:

            best = val
            best_a = a_cur.copy()
            print("    BH: new best max_b=%.8f (obj=%.6f)" % (val, 2*n*val/(a.sum()**2)))
