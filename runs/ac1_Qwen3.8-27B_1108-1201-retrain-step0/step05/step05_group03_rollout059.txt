"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Key insight: The central binomial coefficient sequence a_i = C(2i,i)/4^i has a 
nearly flat self-convolution (exactly 1 for the first n lags in the infinite case),
giving an asymptotic objective of pi/2 ≈ 1.5708. We use this as our starting point
and apply FISTA with LSE-smoothed max, multiple tapers, and adaptive refinement 
to beat this bound. We also explore different sequence lengths and parameterizations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def topk_lse_value_and_gradient(a, c, k, beta):
    """Compute top-k LSE value and gradient w.r.t. a."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 400:
        scale = 400.0 / max(1e-15, (c_top - cm).max())
        if scale > 0:
            beta = scale
            scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    w = e / max(tot, 1e-300)

    w_full = np.zeros(len(c))
    w_full[idx] = w

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    val = cm + np.log(max(tot, 1e-300)) / beta
    return val, grad


def objective_value(a):
    """Compute the objective directly."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (s * s)


def central_binomial_seq(n):
    """Generate central binomial coefficient sequence C(2i,i)/4^i for i=0..n-1."""
    a = np.zeros(n)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2.0 * (2.0 * i - 1.0)) / (4.0 * i)
    return a


def tapered_seq(n, alpha=0.5, beta_taper=0.0, p=1.0):
    """Generate sequence (i+1)^(-alpha) * (1-(i/n)^p)^beta_taper."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** (-alpha)
    if beta_taper > 0:
        taper = 1.0 - (i / max(n - 1, 1)) ** p
        taper = np.maximum(taper, 0.0)
        a = a * taper ** beta_taper
    return a


def run_fista(a_init, S, max_time, rng,
              beta_init=30.0, beta_rate=0.004, beta_max=8000.0,
              k_frac=0.10, use_momentum=True, restart_interval=3000):
    """FISTA with top-k LSE, backtracking, momentum, and periodic restarts."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    last_improve = 0
    stuck = 0

    k = max(8, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best - 1e-15:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            stuck = 0
        else:
            stuck += 1

        if it % 500 == 1:
            progress = (time.monotonic() - start) / max_time
            print(f"    FISTA it={it}, max={best:.10f}, obj={2*n*best/S**2:.8f}, progress={progress:.2f}")

        val, grad = topk_lse_value_and_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step = 1.0 / max(2.0 * beta * 0.1, 1e-10)
        x_new = project_simplex(y - step * grad, S)
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max + 1e-12:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
            y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        else:
            y_new = x_new.copy()
        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new if use_momentum else 1.0

        # Periodic restart when stuck
        if stuck > restart_interval:
            progress = min(1.0, (time.monotonic() - start) / max_time)
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 - 0.8 * progress)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_val = 1.0
            stuck = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_adaptive_fista(a_init, S, max_time, rng):
    """Adaptive FISTA with multiple beta strategies and k values."""
    n = len(a_init)
    best_a = a_init.copy()
    best_val = fft_convolve_self(a_init).max()
    start = time.monotonic()

    # Phase 1: Moderate beta, wide k
    t1 = max_time * 0.40
    if t1 > 20:
        a1, v1 = run_fista(a_init, S, t1, rng,
                           beta_init=20.0, beta_rate=0.005, beta_max=5000.0,
                           k_frac=0.12, use_momentum=True, restart_interval=2000)
        if v1 < best_val:
            best_val = v1
            best_a = a1

    # Phase 2: Higher beta, narrower k
    remaining = max_time - (time.monotonic() - start)
    t2 = remaining * 0.35
    if t2 > 20 and time.monotonic() < start + max_time - 30:
        a2, v2 = run_fista(best_a, S, t2, rng,
                           beta_init=100.0, beta_rate=0.003, beta_max=20000.0,
                           k_frac=0.05, use_momentum=True, restart_interval=3000)
        if v2 < best_val:
            best_val = v2
            best_a = a2

    # Phase 3: Very high beta, very narrow k (polishing)
    remaining = max_time - (time.monotonic() - start)
    t3 = remaining * 0.25
    if t3 > 15:
        a3, v3 = run_fista(best_a, S, t3, rng,
                           beta_init=500.0, beta_rate=0.002, beta_max=50000.0,
                           k_frac=0.02, use_momentum=False, restart_interval=5000)
        if v3 < best_val:
            best_val = v3
            best_a = a3

    return best_a, best_val


def run_tail_equalize(a_init, S, max_time, rng):
    """Specifically target the tail of the convolution (lags n to 2n-2) to increase them."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    c = fft_convolve_self(a)
    tail = c[n:]  # lags n to 2n-2
    tail_max = tail.max()

    # We want to increase the tail values relative to the peak.
    # Gradient approach: push weight towards tail lags.
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Weight the tail lags more heavily to reduce their contribution to the max
        # Actually, we want to INCREASE the tail, which means we want to REDUCE the peak
        # relative to the tail. Use a non-uniform LSE weighting.
        
        # Strategy: use a weighted max that emphasizes the tail
        weights = np.ones(2 * n - 1)
        # Increase weight for tail lags where conv is below the max
        mask = c < cur_max * 0.95
        weights[mask] = 1.0
        # Decrease weight for the top few peaks
        top_idx = np.argsort(c)[::-1][:max(4, n // 200)]
        weights[top_idx] = 0.3

        # Compute weighted gradient
        k = max(8, int(0.08 * (2 * n - 1)))
        idx = np.argsort(c)[::-1][:k]
        c_top = c[idx]
        cm = c_top.max()
        beta = 50.0
        scaled = beta * (c_top - cm)
        scaled = np.clip(scaled, -400, 400)
        e = np.exp(scaled)
        tot = e.sum()
        w = e / max(tot, 1e-300)
        
        # Apply tail weighting
        w_weighted = w * weights[idx]
        w_weighted = w_weighted / max(w_weighted.sum(), 1e-300)
        
        w_full = np.zeros(2 * n - 1)
        w_full[idx] = w_weighted
        
        fsize = 1 << (2 * n - 1 + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        step = 0.3
        a_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller step
            a_new = project_simplex(a - step * 0.3 * grad, S)
            a = a_new

        if it % 2000 == 0:
            obj = 2 * n * best / S ** 2
            print(f"    Tail-eq it={it}, obj={obj:.8f}")

    return best_a, best


def spectral_refine(a, S, rng, steps=50, alpha=0.03):
    """Spectral domain refinement: equalize the power spectrum to flatten convolution."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    
    for _ in range(steps):
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        mean_p = power.mean()
        if mean_p < 1e-20:
            break
        ratio = power / mean_p
        # Compress dynamic range
        new_power = power * (1.0 + alpha * (1.0 - ratio) / (1.0 + 0.5 * ratio))
        new_power = np.maximum(new_power, 1e-30)
        A_new = A * np.sqrt(new_power / np.maximum(power, 1e-30))
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.clip(a_new, 0, None)
        a = normalize(a_new, S)
    
    return a


def propose_candidate(budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses central binomial initialization with multi-phase FISTA optimization.
    """
    rng = np.random.default_rng(12345)
    start_total = time.monotonic()
    deadline = start_total + budget_s - 30.0

    print(f"Starting optimization, budget={budget_s}s")

    # Get given sequence if available
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
            print(f"Using given sequence of length {len(given)}, sum={given.sum():.4f}")
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_sequence(a_test, desc=""):
        nonlocal global_best, global_a
        a_test = np.clip(a_test, 0.0, 1000.0)
        s = a_test.sum()
        if s < 0.01:
            return
        n = len(a_test)
        c = fft_convolve_self(a_test)
        obj = 2.0 * n * c.max() / (s * s)
        if obj < global_best:
            global_best = obj
            global_a = a_test.copy()
            print(f"  *** New best ({desc}): obj={obj:.8f}, n={n} ***")
        return obj

    # === PHASE 1: Central binomial with various n ===
    print("\n=== Phase 1: Central binomial initialization ===")
    
    for n in [4096, 8192, 16384]:
        a_cb = central_binomial_seq(n)
        S = a_cb.sum()
        obj = try_sequence(a_cb, f"CB n={n}")
        print(f"  CB n={n}: obj={obj:.8f}")

    # === PHASE 2: Optimized central binomial (main run) ===
    print("\n=== Phase 2: Main optimization on best initial ===")
    
    # Use the best CB sequence as starting point
    n_main = 8192
    a_start = central_binomial_seq(n_main)
    S_main = a_start.sum()
    
    t_main = min(1200.0, deadline - time.monotonic() - 200.0)
    t_main = max(t_main, 300.0)
    print(f"  Main optimization: n={n_main}, time={t_main:.0f}s")
    
    a_opt, val_opt = run_adaptive_fista(a_start, S_main, t_main, rng)
    obj_opt = try_sequence(a_opt, "main opt")
    print(f"  Main opt result: obj={obj_opt:.8f}")

    # === PHASE 3: Try given sequence (if available) ===
    if given is not None and time.monotonic() < deadline - 200:
        print("\n=== Phase 3: Given sequence optimization ===")
        n_g = len(given)
        a_g = normalize(given.copy(), S_main) if n_g == n_main else None
        
        if a_g is None:
            # Resample to n_main
            t_old = np.linspace(0, 1, n_g)
            t_new = np.linspace(0, 1, n_main)
            a_g = np.interp(t_new, t_old, given)
            a_g = normalize(a_g, S_main)
        
        t_g = min(400.0, deadline - time.monotonic() - 100.0)
        if t_g > 60:
            a_g_opt, val_g = run_adaptive_fista(a_g, S_main, t_g, rng)
            obj_g = try_sequence(a_g_opt, "given seq opt")
            print(f"  Given seq opt: obj={obj_g:.8f}")

    # === PHASE 4: Tapered sequence optimization ===
    if time.monotonic() < deadline - 150:
        print("\n=== Phase 4: Tapered sequences ===")
        
        for beta_taper in [0.3, 0.5, 1.0, 1.5]:
            a_t = tapered_seq(n_main, alpha=0.5, beta_taper=beta_taper, p=1.0)
            S_t = a_t.sum()
            obj_t0 = try_sequence(a_t, f"taper b={beta_taper}")
            
            t_t = min(200.0, deadline - time.monotonic() - 80.0)
            if t_t > 40:
                a_t_opt, val_t = run_fista(a_t, S_t, t_t, rng,
                                           beta_init=50.0, beta_rate=0.003,
                                           beta_max=10000.0, k_frac=0.08)
                obj_t = try_sequence(a_t_opt, f"taper opt b={beta_taper}")
    
    # === PHASE 5: Try different n values ===
    if time.monotonic() < deadline - 100:
        print("\n=== Phase 5: Different lengths ===")
        
        for n_alt in [2048, 16384, 32768]:
            if n_alt == n_main:
                continue
            a_alt = central_binomial_seq(n_alt)
            S_alt = a_alt.sum()
            
            t_alt = min(300.0, deadline - time.monotonic() - 60.0)
            if t_alt > 60 and time.monotonic() < deadline - 60:
                a_alt_opt, val_alt = run_adaptive_fista(a_alt, S_alt, t_alt, rng)
                obj_alt = try_sequence(a_alt_opt, f"alt n={n_alt}")
                
    # === PHASE 6: Tail equalization on best ===
    if global_a is not None and time.monotonic() < deadline - 60:
        print("\n=== Phase 6: Tail equalization ===")
        n_best = len(global_a)
        S_best = global_a.sum()
        t_te = min(300.0, deadline - time.monotonic() - 30.0)
        if t_te > 30:
            a_te, val_te = run_tail_equalize(global_a, S_best, t_te, rng)
            obj_te = try_sequence(a_te, "tail-eq")
    
    # === PHASE 7: Final polishing ===
    if global_a is not None and time.monotonic() < deadline - 20:
        print("\n=== Phase 7: Final polishing ===")
        n_best = len(global_a)
        S_best = global_a.sum()
        
        # Spectral refinement
        a_spec = spectral_refine(global_a.copy(), S_best, rng, steps=30, alpha=0.05)
        obj_spec = try_sequence(a_spec, "spectral")
        
        # Very high beta FISTA
        t_pol = max(20.0, deadline - time.monotonic() - 10.0)
        a_pol, val_pol = run_fista(global_a, S_best, t_pol, rng,
                                   beta_init=2000.0, beta_rate=0.001,
                                   beta_max=100000.0, k_frac=0.01,
                                   use_momentum=False, restart_interval=8000)
        obj_pol = try_sequence(a_pol, "final polish")
    
    # === Phase 8: Random restarts from best ===
    if global_a is not None and time.monotonic() < deadline - 20:
        print("\n=== Phase 8: Random restarts ===")
        n_best = len(global_a)
        S_best = global_a.sum()
        
        for restart in range(5):
            t_r = max(20.0, deadline - time.monotonic() - 10.0) / (5 - restart)
            if t_r < 20:
                break
            noise_scale = 0.005 * S_best / np.sqrt(n_best)
            a_r = project_simplex(global_a + rng.normal(0, noise_scale, n_best), S_best)
            a_r_opt, val_r = run_fista(a_r, S_best, t_r, rng,
                                       beta_init=100.0, beta_rate=0.003,
                                       beta_max=20000.0, k_frac=0.05)
            obj_r = try_sequence(a_r_opt, f"restart {restart}")
    
    # === Final verification ===
    print(f"\n=== Final Results ===")
    print(f"  Best objective: {global_best:.10f}")
    print(f"  Best sequence length: {0 if global_a is None else len(global_a)}")
    
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = central_binomial_seq(256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print(f"  Verification via evaluate_sequence: {ver:.10f}")
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = central_binomial_seq(4096)
            final_list = [float(x) for x in global_a]
    else:
        global_a = central_binomial_seq(4096)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print(f"  Fallback verification: {ver:.10f}")

    elapsed = time.monotonic() - start_total
    print(f"  Total time: {elapsed:.1f}s")
    print(f"  Returning sequence of length {len(global_a)}")
    
    return [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
