    # Select top-k lags with smooth weighting
    idx = np.argsort(c)[::-1][:k_top]
    # Weight by excess above (cmax - threshold)
    threshold = cmax - 0.05 * cmax  # lags within 5% of max
    weights = np.maximum(c[idx] - threshold, 0.0)
    if weights.sum() < 1e-15:
        weights = np.ones(k_top) / k_top
    else:
        weights = weights / weights.sum()
    
    w = np.zeros(2*n-1)
    for i, k in enumerate(idx):
        w[k] = weights[i]
    
    size = 1 << (2*n-1).bit_length()
    W = fft(w, size)
    A = fft(a, size)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return grad, cmax, c

def proximal_gradient_opt(a_init, S, max_time, rng, n_top_k=8):
    """Proximal gradient (FISTA) with exact subgradient and BB step size."""
    n = len(a_init)
    a = normalize_to_sum(a_init, S)
    c0 = fft_conv_self(a)
    best_val = c0.max()
    best_a = a.copy()
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    
    # BB step size parameters
    step_prev = 1.0 / (2.0 * np.linalg.norm(a) * 2.0)
    x_prev = a.copy()
    g_prev = None
    
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_conv_self(y)
        if c.max() < best_val:
            best_val = c.max()
            best_a = y.copy()
            last_improve = it
        
        grad, cmax, _ = subgradient_obj(y, k_top=n_top_k)
        
        # BB step size
        if g_prev is not None:
            sy = y - x_prev
            gs = grad - g_prev
            ys = sy @ gs
            ss = gs @ gs
            if ss > 1e-30 and np.abs(ys) > 1e-30:
                step_bb = ys / ss
                step = np.clip(step_bb, 1e-8, 10.0 / (cmax + 1e-10))
            else:
                step = step_prev
        else:
            step = 1.0 / (2.0 * cmax + 1e-10)
        
        # Backtracking line search
        for _ in range(20):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_conv_self(x_new)
            if c_new.max() <= cmax + 1e-10 * cmax:
                break
            step *= 0.5
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        
        # Update BB parameters
        x_prev = a.copy()
        # Recompute gradient at a for BB (or use the same grad as approximation)
        g_prev = grad
        a = x_new
        t_mom = t_new
        step_prev = step
        
        # Stuck handling
        if it - last_improve > 5000:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
            last_improve = it
            step_prev = 1.0 / (2.0 * cmax + 1e-10)
    
    return best_a, best_val

def polyak_subgradient(a_init, S, max_time, rng):

    a = normalize_to_sum(a_init, S)
    f_lower = 2.0 * n / (2.0 * n - 1)  # Lower bound on objective
    
    start = time.monotonic()
    c = fft_conv_self(a)
    best_val = c.max()  # In normalized terms (sum = S)
    best_a = a.copy()

        it += 1
        c = fft_conv_self(a)
        if c.max() < best_val:
            best_val = c.max()
            best_a = a.copy()
        
        obj = 2.0 * n * c.max() / (S * S)
        f_star_est = f_lower  # Use lower bound
        
        grad, cmax, _ = subgradient_obj(a, k_top=8)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Polyak step
        alpha = (obj - f_star_est) / (gnorm * gnorm)
        alpha = np.clip(alpha, 1e-10, 1.0)

        # Check if we improved; if not, halve the step
        c_new = fft_conv_self(a_new)
        if c_new.max() > cmax * 1.01:
            alpha *= 0.5
            a_new = project_simplex(a - alpha * grad, S)
        
        a = a_new
    
    return best_a, best_val

Let me also think about the "deflation" approach more carefully:

**Spectral Deflation:**
If the peak of the autocorrelation is at lag k*, we can "deflate" it by modifying the spectrum:
- In the frequency domain, the autocorrelation peak at lag k* corresponds to a "ripple" in |A(f)|^2
- To reduce the peak, we can reduce |A(f)| at frequencies that contribute to the ripple
- Specifically, the contribution of frequency f to the autocorrelation at lag k* is |A(f)|^2 * exp(i*2*pi*f*k*/n)
- To cancel the peak, we want to reduce |A(f)| where the phases align

This is similar to the "null-steering" approach in radar.

In practice:
1. Compute A = FFT(a)
2. Identify the peak lag k*
3. Compute the "ripple pattern": r(f) = Re(exp(i*2*pi*f*k*/n) * |A(f)|^2)
4. Reduce |A(f)| where r(f) > 0 (contributing to the peak)
5. Increase |A(f)| where r(f) < 0 (to maintain the sum constraint)
6. IFFT, clip to non-negative

Let me implement this:

```python
def spectral_deflation(a, n_iter=10, strength=0.1):
    """Reduce autocorrelation peaks by modifying the spectrum."""
    n = len(a)
    size = 1 << (2*n-1).bit_length()
    S = a.sum()
    
    a_work = a.copy()
    for _ in range(n_iter):
        A = fft(a_work, size)
        c = np.real(ifft(A * A))[:2*n-1]
        
        # Find peak lag
        k_peak = np.argmax(c)
        cmax = c[k_peak]
        target = c.sum() / (2*n-1)
        
        if cmax <= target * 1.01:
            break
        
        # Compute ripple pattern
        freqs = np.fft.fftfreq(size)
        ripple = np.real(np.exp(2j * np.pi * freqs * k_peak) * np.abs(A)**2)
        
        # Modify magnitude: reduce where ripple > 0, increase where ripple < 0
        factor = 1.0 - strength * ripple / (np.abs(ripple).max() + 1e-10)
        factor = np.maximum(factor, 0.5)  # Don't reduce too much
        new_mag = np.abs(A) * factor
        
        # Normalize to maintain sum
        # After modification, sum will change. We'll renormalize after IFFT.
        phase = np.angle(A)
        new_A = new_mag * np.exp(1j * phase)
        
        a_work = np.real(ifft(new_A, size))[:n]
        a_work = np.maximum(a_work, 0.0)
    
    return normalize_to_sum(a_work, S)
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
- Gerchberg-Saxton spectral shaping for global exploration
- Exact subgradient descent with backtracking for local optimization
- Spectral deflation for targeted peak reduction
- Simulated annealing with structured perturbations for escaping local minima
- Multi-start with diverse initializations and adaptive n selection
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = fft(a, size)
    c = np.real(ifft(A * A))
    return c[:2*n-1]


def obj_from_conv(c, n, S):
    """Compute objective from convolution."""
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if v.sum() <= S:
        # Need to add to reach S
        # This happens when v has many zeros
        diff = S - v.sum()
        v += diff / n
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def subgradient_max_conv(a, k_top=8, tol_frac=0.05):
    """Compute subgradient of max(conv(a,a)) using top-k active lags."""
    n = len(a)
    c = fft_conv_self(a)
    cmax = c.max()
    
    # Select active lags: those within tol_frac of max
    threshold = cmax * (1.0 - tol_frac)
    active = np.where(c >= threshold)[0]
    
    # Limit to top-k for efficiency
    if len(active) > k_top:
        active = active[np.argsort(c[active])[::-1][:k_top]]
    
    # Weight by excess above threshold
    weights = np.maximum(c[active] - threshold, 0.0)
    if weights.sum() < 1e-15:
        weights = np.ones(len(active)) / len(active)
    else:
        weights = weights / weights.sum()
    
    w = np.zeros(2*n-1)
    w[active] = weights
    
    size = 1 << (2*n-1).bit_length()
    W = fft(w, size)
    A = fft(a, size)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return grad, cmax, c


def gs_flat_iteration(a, S, n_iter=100, alpha=0.3):
    """GS iteration to flatten autocorrelation."""
    n = len(a)
    size = 1 << (2*n-1).bit_length()

    a_work[:n] = a.copy()

        A = fft(a_work)
        R = np.real(ifft(A * A))
        lin_R = R[:2*n-1]
        target = lin_R.sum() / (2*n-1)
        
        ratio = target / np.maximum(R, 1e-10)
        new_mag = (
