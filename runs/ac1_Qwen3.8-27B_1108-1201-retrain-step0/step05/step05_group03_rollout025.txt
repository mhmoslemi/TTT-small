```python
def lp_direction(f, M, n):
    """Find g >= 0 maximizing sum(g) subject to (f*g)[i] <= M for all i."""
    # Build the convolution constraint matrix
    # A[i,j] = f[i-j] if 0 <= i-j < n, else 0
    # This is a Toeplitz matrix
    
    # Use scipy's toeplitz to build it
    # Column: f[0], 0, 0, ..., 0, f[1], f[2], ..., f[n-1]
    # Row: f[0], f[1], ..., f[n-1], 0, 0, ..., 0
    
    col = np.zeros(2*n-1)
    col[0] = f[0]
    col[n:] = f[1:]
    row = np.zeros(2*n-1)
    row[:n] = f
    
    A = scipy.linalg.toeplitz(col, row)
    # A[i,j] should be f[i-j] if 0 <= i-j < n
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    result = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n)
    return result.x
```

```python
def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def build_conv_matrix(f, n):
    """Build sparse convolution matrix A where (A@x)[i] = (f*x)[i]."""
    # A has shape (2n-1, n)
    # A[i,j] = f[i-j] if 0 <= i-j <= n-1, else 0
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for i in range(j, j + n):
            rows.append(i)
            cols.append(j)
            vals.append(f[i-j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```

```python
    # So A[i,0] = c[i] for i=0,...,2n-2
    # And A[0,j] = r[j] for j=0,...,n-1
    # For general (i,j): A[i,j] = c[i-j] if i>=j (since column 0 is c, and it's Toeplitz)
    #                    A[i,j] = r[i] ... no
    
    # A Toeplitz matrix has A[i,j] = t[i-j] where t[k] is the "diagonal" value.
    # The first column gives t[0], t[1], ..., t[2n-2]
    # The first row gives t[0], t[-1], ..., t[-(n-1)]
    # So t[k] for k >= 0 is c[k], and t[k] for k < 0 is r[-k]
    
    # I want A[i,j] = f[i-j] if 0 <= i-j <= n-1, else 0
    # So t[k] = f[k] for 0 <= k <= n-1, and t[k] = 0 otherwise
    # And t[-k] = f[-k]... but f is only defined for 0..n-1
    # I want t[k] = 0 for k < 0 (since f[i-j] = 0 when i-j < 0)
    
    # So: c = [f[0], f[1], ..., f[n-1], 0, ...(2n-2-n+1 zeros)]
    #     r = [f[0], 0, 0, ..., 0] (first row is just f[0] followed by zeros)
    
    col = np.zeros(2*n-1)
    col[:n] = f
    row = np.zeros(2*n-1)  # but we only need n elements for the row
    row[0] = f[0]
    # Actually for toeplitz, r only needs length n (number of columns)
    row_n = np.zeros(n)
    row_n[0] = f[0]
    
    A_dense = toeplitz(col, row_n)
    return csr_matrix(A_dense)
```

```python
def build_conv_matrix(f, n):
    """Build sparse matrix A of shape (2n-1, n) where A[i,j] = f[i-j] if 0<=i-j<n."""
    # Each column j has nonzeros at rows j through j+n-1
    # with values f[0] through f[n-1]
    n_rows = 2*n - 1
    n_cols = n
    # Diagonal offsets: for column j, the first nonzero is at row j
    # So the "main diagonal" starts at row 0 for col 0, row 1 for col 1, etc.
    # This is a lower Hessenberg-like matrix (actually it's the "convolution" pattern)
    
    # Use COO format
    # For each column j (0 to n-1):
    #   rows: j, j+1, ..., j+n-1
    #   values: f[0], f[1], ..., f[n-1]
    
    # Total nonzeros: n*n
    rows_idx = np.zeros(n*n, dtype=np.int32)
    cols_idx = np.zeros(n*n, dtype=np.int32)
    vals = np.zeros(n*n, dtype=np.float64)
    
    for j in range(n):
        start = j * n
        end = (j+1) * n
        rows_idx[start:end] = np.arange(j, j+n)
        cols_idx[start:end] = j
        vals[start:end] = f
    
    return csr_matrix((vals, (rows_idx, cols_idx)), shape=(n_rows, n_cols))
```

```python
def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize(a, S):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def lse_max_value_and_weights(c, k, beta):
    """Top-k LSE approximation of max."""
    n = len(c)
    k = min(k, n)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    val = cm + np.log(max(tot, 1e-300)) / beta
    w = e / max(tot, 1e-300)
    return val, w, idx

def lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE of self-convolution."""
    n = len(a)
    _, w, idx = lse_max_value_and_weights(c, k, beta)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def run_fista(a_init, S, max_time, rng, beta_init=10.0, beta_max=5000.0, k_frac=0.1):
    """FISTA with top-k LSE smoothing."""
    n = len(a_init)
    a = normalize(a_init.copy(), S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    k = max(4, int(k_frac * (2 * n - 1)))
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + it / 500.0))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
        
        grad = lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking
        step = 1.0 / max(0.1 * beta, 1e-10)
        for _ in range(10):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-10:
                break
            step *= 0.5
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val**2))
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_val = t_new
        
        # Periodic restart
        if it % 500 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_val = 1.0
    
    return best_a, best

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[i] <= M, g >= 0."""
    # Build constraint matrix
    # A[i,j] = f[i-j] if 0 <= i-j < n
    row_idx = []
    col_idx = []
    vals = []
    for j in range(n):
        for i in range(j, j + n):
            row_idx.append(i)
            col_idx.append(j)
            vals.append(f[i - j])
    A = csr_matrix((np.array(vals), (np.array(row_idx, dtype=np.int32), np.array(col_idx, dtype=np.int32))), shape=(2*n-1, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        return res.x
    return None
```

```python
1. FISTA with adaptive top-k LSE smoothing and periodic reduced-LP direction kicks

import numpy as np
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize(a, S):
    """Clip and rescale to sum = S, maintain non-negativity."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_max_value_and_weights(c, k, beta):
    """Compute top-k LSE value and weights."""
    n = len(c)
    k = min(k, n)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    scaled = beta * (c_top - cm)
    if scaled.max() > 500:
        scale = 500.0 / max(1e-15, (c_top - cm).max())
        beta = max(scale, 0.01)
        scaled = beta * (c_top - cm)
    e = np.exp(scaled)
    tot = e.sum()
    val = cm + np.log(max(tot, 1e-300)) / beta
    w = e / max(tot, 1e-300)
    return val, w, idx


def lse_gradient(a, c, k, beta):
    """Gradient of top-k LSE smooth max of self-convolution."""
    n = len(a)
    _, w, idx = lse_max_value_and_weights(c, k, beta)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def build_reduced_lp_matrix(f, hot_spots, n):
    """Build sparse constraint matrix for reduced LP. A[i,j] = f[hot_spots[i]-j] if valid."""
    k = len(hot_spots)
    j_idx = np.arange(n)
    h_minus_j = np.array(hot_spots, dtype=np.float64)[:, None] - j_idx[None, :].astype(np.float64)
    valid = (h_minus_j >= 0) & (h_minus_j < n)
    vals = np.where(valid, f[np.clip(h_minus_j.astype(int), 0, n - 1)], 0.0)
    # Zero out invalid entries (already done by np.where, but ensure)
    vals[~valid] = 0.0
    return csr_matrix(vals)


def solve_lp_reduced(f, M, hot_spots, n):
    """Solve reduced LP: max 1^T g s.t. (f*g)[h] <= M for h in hot_spots, g >= 0."""
    k = len(hot_spots)
    A = build_reduced_lp_matrix(f, hot_spots, n)
    c_obj = -np.ones(n)
    b_ub = np.full(k, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def solve_lp_full(f, M, n_lp):
    """Solve full LP at reduced dimension: max 1^T g s.t. (f*g)[i] <= M for all i, g >= 0."""
    fsize_mat = 2 * n_lp - 1
    j_idx = np.arange(n_lp)
    i_idx = np.arange(fsize_mat)
    # A[i,j] = f[i-j] if 0 <= i-j < n_lp
    h_minus_j = i_idx[:, None].astype(np.float64) - j_idx[None, :].astype(np.float64)
    valid = (h_minus_j >= 0) & (h_minus_j < n_lp)
    vals = np.where(valid, f[np.clip(h_minus_j.astype(int), 0, n_lp - 1)], 0.0)
    vals[~valid] = 0.0
    A = csr_matrix(vals)
    c_obj = -np.ones(n_lp)
    b_ub = np.full(fsize_mat, M)
    bounds = [(0.0, None)] * n_lp
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 60})
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def resample_sequence(a, n_target):
    """Resample a sequence to a different length using linear interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)


def run_fista_with_lp_kicks(a_init, S, max_time, rng,

    n = len(a_init)

    t_val = 1.0
    start = time.monotonic()
    best = fft_convolve_self(a).max()

    it = 0
    k = max(8, int(k_frac * (2 * n - 1)))

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_growth * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()

        grad = lse_gradient(y, c, k, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / max(0.05 * beta, 1e-10)
        x_new = y
        for _ in range(12):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max + 1e-10:
                break
            step *= 0.5

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))

        y_new = np.maximum(y_new, 0.0)
        a = x_new
        y = y_new
        t_val = t_new

        # Periodic LP kick: use reduced LP to find a directional nudge

        best_obj = M

def objective_from_maxb(maxb, n, a):
    """Compute objective from max convolution value."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * maxb / (s * s)

    print("\n=== Phase 1: FISTA + LP kicks (n=%d, %.0fs) ===" % (n_primary, t1))

    a_opt, val_opt = run_fista_with_lp_kicks(
        a_start, S, t1, rng,
        beta_init=8.0, beta_growth=0.0015, beta_max=4000.0,
        k_frac=0.08, lp_kick_interval=150, lp_num_spots=80
    )
    obj1 = objective_from_maxb(val_opt, n_primary, a_opt)
    if obj1 < global_best:
        global_best = obj1
        global_a = a_opt.copy()
        print("  Phase 1 result: obj=%.6f" % global_best)
    else:
        print("  Phase 1: no improvement (best=%.6f, got=%.6f)" % (global_best, obj1))

    # Phase 2: LP refinement at reduced dimension (25% of time)
    remaining = deadline - time.monotonic()
    t2 = remaining * 0.40
    if t2 > 60:
        print("\n=== Phase 2: LP refinement (n_lp=1024, %.0fs) ===" % t2)
        a_lp = run_lp_refinement(global_a, S, t2, rng, n_lp=1024)
        obj2 = objective_from_maxb(fft_convolve_self(a_lp).max(), n_primary, a_lp)
        if obj2 < global_best:
            global_best = obj2
            global_a = a_lp.copy()
            print("  Phase 2 result: obj=%.6f" % global_best)
        else:
            print("  Phase 2: no improvement (got=%.6f)" % obj2)

    # Phase 2b: Try LP at n_lp=2048 if time permits
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print("\n=== Phase 2b: LP refinement (n_lp=2048, %.0fs) ===" % (remaining * 0.3))
        t2b = remaining * 0.3
        a_lp2 = run_lp_refinement(global_a, S, t2b, rng, n_lp=2048)
        obj2b = objective_from_maxb(fft_convolve_self(a_lp2).max(), n_primary, a_lp2)
        if obj2b < global_best:
            global_best = obj2b
            global_a = a_lp2.copy()
            print("  Phase 2b result: obj=%.6f" % global_best)

    # Phase 3: High-beta polish (remaining time minus buffer)
    remaining = deadline - time.monotonic() - 10.0
    if remaining > 20:
        print("\n=== Phase 3: High-beta polish (%.0fs) ===" % remaining)
        a_pol, val_pol = run_high_beta_polish(global_a, S, remaining, rng)
        obj3 = objective_from_maxb(val_pol, n_primary, a_pol)
        if obj3 < global_best:
            global_best = obj3
            global_a = a_pol.copy()
            print("  Phase 3 result: obj=%.6f" % global_best)

    # Phase 4: Final short FISTA with very high beta
    remaining = deadline - time.monotonic() - 5.0
    if remaining > 10:
        print("\n=== Phase 4: Final FISTA (%.0fs) ===" % remaining)
        a_fin, val_fin = run_fista_with_lp_kicks(
            global_a, S, remaining, rng,
            beta_init=1000.0, beta_growth=0.0005, beta_max=100000.0,
            k_frac=0.02, lp_kick_interval=500, lp_num_spots=40
        )
        obj4 = objective_from_maxb(val_fin, n_primary, a_fin)
        if obj4 < global_best:
            global_best = obj4
            global_a = a_fin.copy()
            print("  Phase 4 result: obj=%.6f" % global_best)

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\nFinal verification via evaluate_sequence: %.6f" % ver)

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Using safe fallback.")
        final_list = [float(x) for x in np.ones(256) * (np.sqrt(512.0) / 256)]
        ver = evaluate_sequence(final_list)
        print("  Fallback evaluation: %.6f" % ver)

    print("Returning sequence of length %d, objective=%.6f" % (len(final_list), ver))
    return final_list
```
