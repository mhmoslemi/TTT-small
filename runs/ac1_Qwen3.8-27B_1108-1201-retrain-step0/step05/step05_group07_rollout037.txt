"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LSE gradient descent with momentum,
spectral domain refinement, and multi-start. Key innovations over previous approach:
- Nesterov momentum for faster convergence
- Spectral domain flattening steps (Gerchberg-Saxton inspired)
- Adaptive beta with cosine annealing and periodic restarts
- Multi-resolution initialization (coarse-to-fine)
"""

import numpy as np
from scipy.fft import fft, ifft
import time
import math

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.where(u - (css - S) / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        return np.ones(n) * S / n
    rho = rho[-1]
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_weights(c, beta):
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot

def lse_value(c, beta):
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    return cm + np.log(np.exp(scaled).sum()) / beta

def lse_gradient(a, w):
    """Gradient of LSE w.r.t. a, given LSE weights w."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return grad

def spectral_flatten_step(a, S, alpha=0.3):
    """Spectral domain step: pull |FFT(a)| toward its geometric mean."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Target magnitude: geometric mean (preserves total energy)
    log_mag = np.log(mag + 1e-30)
    target_mag = np.exp(np.mean(log_mag))
    
    # Blend
    new_mag = (1 - alpha) * mag + alpha * target_mag
    new_mag = np.maximum(new_mag, 0)
    
    a_new = np.real(ifft(new_mag * np.exp(1j * phase), fsize))[:n]
    a_new = np.maximum(a_new, 0)
    return project_simplex(a_new, S)

def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def lse_momentum_descent(a0, S, max_time, rng, beta_start=20.0, beta_end=5000.0,
                         mu_start=0.9, mu_end=0.3, verbose=True):
    """LSE gradient descent with heavy-ball momentum and adaptive step size."""
    n = len(a0)
    a = project_simplex(a0, S)
    v = np.zeros(n)
    
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        # Beta: cosine annealing
        beta = beta_start + (beta_end - beta_start) * (1 - math.cos(math.pi * progress)) / 2.0
        # Momentum: linear decay
        mu = mu_start + (mu_end - mu_start) * progress
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best - 1e-10:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # LSE weights and gradient
        w = lse_weights(c, beta)
        grad = lse_gradient(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size
        gap = cur_max - c.mean()
        if gap < 1e-12:
            break
        step = 2.0 * gap / (gnorm * gnorm)
        step = min(step, 4.0 / gnorm)
        
        # Heavy ball momentum
        v = mu * v + grad
        a_new = project_simplex(a - step * v, S)
        
        # Accept/reject
        c_new = fft_convolve_self(a_new)
        if c_new.max() <= cur_max + 1e-10 * cur_max:
            a = a_new
        else:
            # Try without momentum
            a_new2 = project_simplex(a - step * grad, S)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() <= cur_max + 1e-10 * cur_max:
                a = a_new2
                v = mu * v  # dampen momentum
            else:
                # Halve step
                a_new3 = project_simplex(a - 0.5 * step * grad, S)
                a = a_new3
                v = mu * v
        
        # Periodic spectral step
        if it % 300 == 299 and stuck < 50:
            a_spec = spectral_flatten_step(a, S, alpha=0.2)
            c_spec = fft_convolve_self(a_spec)

Minimize 2*n*max(conv(a,a))/(sum(a))^2 using:
- LSE-smoothed gradient descent with heavy-ball momentum
- Multiplicative weights (mirror descent) as alternative
- Spectral domain flattening (Gerchberg-Saxton inspired)
- Cosine-annealed beta with periodic restarts
- Thue-Morse and random initializations for diversity

import numpy as np
from scipy.fft import fft, ifft
import time
import math

# All helper functions at top level

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    rho_candidates = np.where(u - (css - S) / idx > 0)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) * S / n
    rho = rho_candidates[-1] + 1
    theta = (css[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_weights(c, beta):
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -400, 400)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot

def lse_grad(a, w):
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(np.conj(W) * A))[:n]

def spectral_flatten(a, S, alpha):
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    log_mag = np.log(mag + 1e-30)
    target = np.exp(np.mean(log_mag))
    new_mag = (1.0 - alpha) * mag + alpha * target
    a_new = np.real(ifft(new_mag * np.exp(1j * phase), fsize))[:n]
    return project_simplex(np.maximum(a_new, 0.0), S)

def mult_weights_update(a, grad, step, S):

    e = np.exp(scaled)
    a_new = a * e
    s = a_new.sum()

    return S * a_new / s

        t[i] = t[i >> 1] ^ (i & 1)
    a = 1.0 + 0.8 * (2.0 * t - 1.0)  # maps 0->0.2, 1->1.8
    return np.maximum(a, 0.0)

    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

        mu = mu_start + (mu_end - mu_start) * progress

        if gap < 1e-12:

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    it={it}, t={elapsed:.0f}s, beta={beta:.0f}, obj={2.0*n*best/(S*S):.6f}")
        
        it += 1
    
    return best_a, best

def upsample_seq(a, new_n):
    old_n = len(a)
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    return np.maximum(np.interp(t_new, t_old, a), 0.0)
