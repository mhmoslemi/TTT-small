import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

Actually, one more thought: let me also try a completely different approach - **proximal gradient with a majorized objective**.

The key idea: at each step, we have a current point a. The max function is poorly approximated by a single linearization. Instead, we can use a "majorizer" - a function that is ≥ the objective everywhere and = the objective at the current point. Minimizing the majorizer is easier and guaranteed to decrease the objective.

For f(a) = max_k c_k(a), a majorizer at point a₀ is:
M(a) = max_k c_k(a₀) + max_k (∇c_k(a₀))·(a - a₀) + (1/2)||a - a₀||²_L

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
- L-BFGS-B on LSE-smoothed objective with softmax parameterization
- Adaptive subgradient method with Nesterov averaging
- Multi-scale exploration (n and 2n)
Key improvements over previous: L-BFGS for fast smooth convergence, 
Nesterov averaging for subgradient phase, adaptive active-set tolerance.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_logsumexp(c, beta):
    """Compute LSE: (1/beta)*log(sum(exp(beta*c))) with numerical stability."""
    c_max = c.max()
    shifted = beta * (c - c_max)
    shifted -= shifted.max()
    e = np.exp(shifted)
    return c_max + np.log(e.sum()) / beta, e / e.sum()


def lse_and_grad_a(a, beta, M):
    """Compute LSE value and gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve_self(a)
    lse_val, p = lse_logsumexp(c, beta)
    P = fft(p, M)
    A = fft(a, M)
    grad = 2.0 * np.real(ifft(P * np.conj(A)))[:n]
    return lse_val, grad


def softmax_to_seq(b, S):
    """Convert log-parameters to sequence with sum = S."""
    b_shifted = b - b.max()
    e = np.exp(b_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(b), S / len(b))
    return S * e / s


def seq_to_softmax(a, S):
    """Convert sequence to log-parameters."""
    a_clipped = np.clip(a, 1e-300, None)
    return np.log(a_clipped / S)


def _lbfgs_func(b, beta, M, S, n):
    """Objective function for L-BFGS (softmax parameterization)."""
    a = softmax_to_seq(b, S)
    val, _ = lse_and_grad_a(a, beta, M)
    return val


def _lbfgs_jac(b, beta, M, S, n):
    """Gradient for L-BFGS (softmax parameterization)."""
    a = softmax_to_seq(b, S)
    _, grad_a = lse_and_grad_a(a, beta, M)
    # Chain rule for softmax: grad_b_j = a_j * (grad_a_j - sum(a*grad_a)/S)
    mean_w = np.dot(a, grad_a) / S
    grad_b = a * (grad_a - mean_w)
    return grad_b


def run_lbfgs(a_init, beta, max_time, maxiter=1000):
    """Run L-BFGS-B on LSE-smoothed objective with softmax parameterization."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    M = 1 << (2 * n).bit_length()
    
    a = np.clip(a_init, 1e-10, 1000)
    a = normalize_seq(a, S)
    b = seq_to_softmax(a, S)
    
    start = time.monotonic()
    
    # Callback to check time
    state = {'count': 0}
    
    def _callback(bk):
        state['count'] += 1
        if time.monotonic() - start > max_time * 0.9:
            return True  # Stop
        return None
    
    try:
        result = minimize(_lbfgs_func, b, jac=_lbfgs_jac,
                         args=(beta, M, S, n),
                         method='L-BFGS-B',
                         callback=_callback,
                         options={'maxiter': maxiter, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxls': 20,
                                   'maxfun': maxiter * 5})
        b_final = result.x
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        b_final = b
    
    a_final = softmax_to_seq(b_final, S)
    elapsed = time.monotonic() - start
    return a_final, elapsed


def run_subgradient_polish(a_init, max_time, rng):
    """Subgradient method with Nesterov averaging and adaptive active set."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(a_init, 0, 1000).copy()
    a = normalize_seq(a, S)
    
    M = 1 << (2 * n).bit_length()
    start = time.monotonic()
    
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    # Nesterov averaging
    avg_a = a.copy()
    t = 1.0
    
    # Momentum (FISTA-style)
    y = a.copy()
    t_mom = 1.0
    
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)  # = cur_max since sum = sqrt(2n)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it
        
        stuck = it - last_improve
        
        # Adaptive active set
        if stuck < 200:
            tol_frac = 0.0
            k = 1
        elif stuck < 1000:
            tol_frac = 0.001
            k = 3
        elif stuck < 5000:
            tol_frac = 0.005
            k = 10
        elif stuck < 20000:
            tol_frac = 0.01
            k = 30
        else:
            tol_frac = 0.02
            k = 100
        
        k = min(k, 2 * n - 1)
        
        # Get top-k lags
        idx = np.argsort(c)[::-1][:k]
        # Weight by gap to max
        gaps = c.max() - c[idx]
        weights = np.exp(-gaps / max(c.max() * tol_frac, 1e-10))
        weights /= weights.sum()
        
        # Build weight vector on convolution lags
        w_full = np.zeros(2 * n - 1)
        w_full[idx] = weights
        
        # Subgradient: 2 * cross_correlation(w_full, y)
        W = fft(w_full, M)
        Y = fft(y, M)
        grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size with backtracking
        step = S / (gnorm * (1.0 + 0.1 * np.log1p(it * 0.01)))
        step = max(step, 1e-12)
        
        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-normalize to handle numerical drift
        y_sum = y.sum()
        if y_sum > 0:
            y = y * (S / y_sum)
        a = x_new
        t_mom = t_new
        
        # Nesterov averaging
        t += 1.0
        avg_a = (t - 1.0) / t * avg_a + 1.0 / t * x_new
        
        # Check averaged iterate
        if it % 100 == 0:
            avg_obj = compute_obj(avg_a)
            if avg_obj < best_obj:
                best_obj = avg_obj
                best_a = avg_a.copy()
        
        # Restart when stuck
        if stuck > 5000 and it % 500 == 0:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck / 20000.0)
            if noise_scale > 1e-10:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it
        
        # Periodic momentum restart
        if it % 10000 == 0:
            t_mom = 1.0
    
    return best_a, best_obj


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.54 + 0.46 * np.cos(2 * np.pi * tt)


def make_chirp_init(n):
    tt = np.arange(n, dtype=np.float64)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt * tt / n)


def make_random_init(n, rng):
    a = rng.random(n)
    return a ** 2


def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)
