    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def conv_peak_info(a, N):
    """Compute convolution and return (c, c_max, k_max, A)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()
    k_max = int(np.argmax(c))
    return c, c_max, k_max, A


def topk_gradient(a, N, k_max, top_k, c, A):
    """Compute gradient of top-k weighted peak objective."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    # Use softmax weighting over top-k
    beta_k = 10.0  # focus weight
    weights = np.exp(beta_k * (top_vals - top_vals[0]))
    weights /= weights.sum()
    
    # Build the weight vector for the convolution
    w_full = np.zeros(N)
    for i, k in enumerate(top_idx):
        w_full[int(k)] += weights[i]
    
    W = fft(w_full)
    fsize = N
    Af = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad


def coordinate_descent_step(a, c, k_max, n, S, step_size=0.01):
    """
    One coordinate descent step: find the coordinate that most contributes
    to the peak and transfer mass to a "safe" coordinate.
    """
    # Contribution to peak at k_max
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    
    # Responsibility: how much each coordinate contributes to the peak
    resp = np.zeros(n)
    for i in range(lo, hi + 1):
        j = k_max - i
        if 0 <= j < n:
            resp[i] += a[j]
            if i != j:
                resp[j] += a[i]
    
    # Find the coordinate with highest responsibility that has significant mass
    candidates = np.where(resp * a > 0.01 * a.max())[0]
    if len(candidates) == 0:
        return a
    
    # Score: responsibility * value (reducing this coordinate reduces the peak most)
    scores = resp * a
    # Weight toward coordinates that are "safe" to reduce (low contribution to other peaks)
    
    # Find top few responsible coordinates
    top_resp_idx = np.argsort(scores)[::-1][:min(10, len(candidates))]
    
    # Find safe targets: coordinates with low overall convolution contribution
    # Approximate: coordinates where a[i] is small
    # Better: coordinates that don't contribute much to any near-peak position
    top_peaks = np.argsort(c)[::-1][:5]
    unsafe = np.zeros(n)
    for kp in top_peaks:
        lo_p = max(0, kp - n + 1)
        hi_p = min(n - 1, kp)
        for i in range(lo_p, hi_p + 1):
            unsafe[i] += 1.0
            j = kp - i
            if 0 <= j < n and j != i:
                unsafe[j] += 1.0
    
    # Safe targets: low a[i] relative to their unsafe score... actually,
    # we want to ADD to coordinates that are far from peaks
    # Let's just pick coordinates with small values
    target_cand = np.where(a < 0.3 * a.mean())[0]
    if len(target_cand) == 0:
        target_cand = np.where(a < a.mean())[0]
    if len(target_cand) == 0:
        return a
    
    # Pick the source: highest responsibility * value
    src = top_resp_idx[0]
    # Pick the target: random from safe candidates
    tgt = target_cand[np.random.randint(len(target_cand))]
    
    if src == tgt:
        return a
    
    # Transfer
    transfer = min(step_size * a[src], 0.5 * a[src])
    if transfer < 1e-15:
        return a
    
    a_new = a.copy()
    a_new[src] -= transfer
    a_new[tgt] += transfer
    
    # Verify it actually helps
    c_new = np.real(ifft(fft(a_new, N) ** 2))[:2*n-1]
    if c_new.max() > c.max() and np.random.random() > 0.3:
        return a  # reject if it makes things worse (with some probability of accepting)
    
    # Renormalize
    s = a_new.sum()
    if s > 1e-15:
        a_new *= S / s
    
    return a_new


def safe_normalize(a, n, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def optimize_sequence(a_init, n, S, N, max_time, rng, 
                       beta_init=20.0, beta_max=500.0,
                       top_k_init=1, top_k_max=64):
    """
    Main optimization loop combining top-K gradient with coordinate descent.
    """
    a = safe_normalize(a_init, n, S)
    c, c_max, k_max, A = conv_peak_info(a, N)
    best_a = a.copy()
    best_cmax = c_max
    beta = beta_init
    top_k = top_k_init
    
    start_time = time.monotonic()
    it = 0
    stall_count = 0
    step_scale = S / (n * 0.5)
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        # Check time
        if it % 50 == 0:
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
            
            # Adaptive parameters
            if stall_count > 5:
                beta = min(beta_max, beta * 1.5)
                top_k = min(top_k_max, top_k + 2)
            elif stall_count > 15:
                # Diversification: perturb the sequence
                noise = rng.normal(0, 0.05 * a.mean(), n)
                a = safe_normalize(a + noise, n, S)
                c, c_max, k_max, A = conv_peak_info(a, N)
                stall_count = 0
            
            # Dampen step size as we converge
            step_scale = S / (n * (0.5 + 0.1 * min(it, 100)))
        
        # Compute gradient
        grad = topk_gradient(a, N, k_max, top_k, c, A)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Gradient step with line search
        alpha = step_scale / gnorm
        a_new = a
        cmax_new = c_max
        for ls in range(8):
            candidate = project_simplex(a - alpha * grad, S)
            c_ls = np.real(ifft(fft(candidate, N) ** 2))[:2*n-1]
            cm_ls = c_ls.max()
            if cm_ls <= c_max * (1 + 0.001) or ls == 7:
                a_new = candidate
                cmax_new = cm_ls
                if cm_ls <= c_max:
                    break
            alpha *= 0.5
        
        # Check if we improved
        if cmax_new < best_cmax - 1e-10:
            best_cmax = cmax_new
            best_a = a_new.copy()
            stall_count = 0
        
        a = a_new
        c, c_max, k_max, A = conv_peak_info(a, N)
        
        # Periodic coordinate descent
        if it % 100 == 0:
            for _ in range(3):
                a = coordinate_descent_step(a, c, k_max, n, S, step_scale * 0.1)
                c, c_max, k_max, A = conv_peak_info(a, N)
                if c_max < best_cmax:
                    best_cmax = c_max
                    best_a = a.copy()
    
    return best_a, best_cmax


def make_diverse_init(n, S, rng, mode='random'):
    """Create diverse initializations."""
    if mode == 'random':
        a = rng.random(n) + 0.01
    elif mode == 'smooth':
        t = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.sin(2 * np.pi * rng.integers(1, 20) * t) + 0.1 * rng.random(n)
    elif mode == 'bimodal':
        a = np.abs(np.sin(np.pi * np.linspace(0, 1, n))) ** rng.uniform(0.5, 3)
    elif mode == 'geometric':
        ratio = rng.uniform(0.95, 1.0)
        a = ratio ** np.arange(n)
    elif mode == 'fourier_rand':
        N = 1 << (2*n-1).bit_length()
        magnitudes = rng.exponential(1.0, N)
        phases = rng.uniform(0, 2*np.pi, N)
        A = magnitudes * np.exp(1j * phases)
        a = np.abs(np.real(ifft(A)))[:n] + 0.01
    else:
        a = rng.random(n) + 0.01
    return a

"""
Minimize 2*n*max_conv/sum^2 using adaptive LSE gradient descent with
coordinate descent peak-elimination and diversification restarts.

Key innovations over previous approach:
1. Direct coordinate descent on peak-contributing coordinates (peak elimination)
2. Multi-trail optimization with state exchange between trails
3. Adaptive top-K peak focusing with smooth transitions
4. Aggressive but safe line search with multiple scale attempts
5. Exploits structure: identifies and targets specific coordinate pairs
   responsible for the convolution peak

import numpy as np
import time
from scipy.fft import fft, ifft


def proj_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}. O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = w > (cs - S) / idx
    rho = int(np.sum(cond))
    if rho == 0:
        rho = 1
    theta = (cs[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Compute full linear convolution via FFT. Returns (c, A)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c, A


def lse_weights(c, beta, c_max):
    """Compute softmax weights for LSE gradient."""
    if beta < 1e-6:
        return np.ones(len(c)) / len(c)
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return w
    return exp_bc / total


def compute_grad(a, N, c, beta):
    """Compute gradient of LSE-smoothed peak. Returns grad (length n)."""
    n = len(a)
    c_max = c.max()
    w = lse_weights(c, beta, c_max)
    A = fft(a, N)
    W = fft(w, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def peak_responsibility(a, k_star, n):
    """Compute responsibility of each coordinate for the peak at k_star."""
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    resp = np.zeros(n)
    for i in range(lo, hi + 1):
        j = k_star - i
        if 0 <= j < n:
            resp[i] += a[j]
            if i != j:
                resp[j] += a[i]
    return resp


def coord_descent_peak(a, c, N, n, S, n_steps=5, rng=None):
    """
    Coordinate descent: directly reduce peak-contributing coordinates.
    Returns (a_new, max_improved).
    """
    if rng is None:
        rng = np.random.default_rng()
    a = a.copy()
    improved = False
    
    for step in range(n_steps):
        c_max = c.max()
        k_star = int(np.argmax(c))
        
        # Compute responsibility for the peak
        resp = peak_responsibility(a, k_star, n)
        
        # Find coordinates with high responsibility AND high value
        # These are the ones to reduce
        effectiveness = resp * a
        # Only consider coordinates with meaningful mass
        threshold = 0.01 * a.max()
        candidates = np.where(effectiveness > threshold)[0]
        
        if len(candidates) == 0:
            break
        
        # Pick the most effective coordinate to reduce
        src = candidates[np.argmax(effectiveness[candidates])]
        
        # Find safe targets: coordinates that contribute little to top peaks
        top_k = min(5, 2 * n - 1)
        top_peaks = np.argsort(c)[::-1][:top_k]
        
        # Compute a "safety score" for each coordinate
        # Low safety = safe to add mass here
        safety = np.ones(n) * n  # start high
        for kp in top_peaks:
            lo_p = max(0, kp - n + 1)
            hi_p = min(n - 1, kp)
            for i in range(lo_p, hi_p + 1):
                safety[i] -= 1.0
                j = kp - i
                if 0 <= j < n and j != i:
                    safety[j] -= 1.0
        
        safety = np.maximum(safety, 0.0)
        
        # Target: high safety, low current value (to avoid creating new peaks)
        target_score = safety * (a.max() - a + 0.001)
        # Exclude the source
        target_score[src] = -1
        targets = np.where(target_score > 0)[0]
        
        if len(targets) == 0:
            # Fallback: pick randomly from low-value coordinates
            low_val = np.where(a < 0.5 * a.mean())[0]
            if len(low_val) == 0:
                break
            tgt = low_val[rng.integers(len(low_val))]
        else:
            tgt = targets[rng.integers(len(targets))]
        
        if tgt == src:
            continue
        
        # Transfer mass
        transfer = min(0.05 * a[src], 0.3 * a[src])
        if transfer < 1e-15:
            break
        
        a[src] -= transfer
        a[tgt] += transfer
        
        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a *= S / s
        
        # Check if it helped
        c_new, _ = compute_conv(a, N)
        if c_new.max() <= c_max + 1e-6:
            c = c_new
            if c_new.max() < c_max - 1e-10:
                improved = True
        else:
            # Revert and try a smaller transfer
            a[src] += transfer
            a[tgt] -= transfer
            s = a.sum()
            if s > 1e-15:
                a *= S / s
            transfer *= 0.3
            if transfer < 1e-15:
                break
            a[src] -= transfer
            a[tgt] += transfer
            s = a.sum()
            if s > 1e-15:
                a *= S / s
            c_new, _ = compute_conv(a, N)
            if c_new.max() <= c_max + 1e-6:
                c = c_new
            else:
                a[src] += transfer
                a[tgt] -= transfer
                s = a.sum()
                if s > 1e-15:
                    a *= S / s
    
    return a, improved


def safe_normalize(a, n, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def optimize_single(a_init, n, S, N, max_time, rng,
                    beta_init=20.0, beta_max=800.0,
                    cd_interval=50, cd_steps=3):
    """
    Main optimization: LSE gradient descent with periodic coordinate descent.
    Returns (best_a, best_peak).
    """
    a = safe_normalize(a_init, n, S)
    c, A = compute_conv(a, N)
    best_a = a.copy()
    best_peak = c.max()
    beta = beta_init
    
    start = time.monotonic()
    it = 0
    stall = 0
    base_step = S / n / 2.0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Periodic parameter adaptation and stall detection
        if it % 30 == 0:
            if best_peak < c.max() + 1e-8:  # no improvement in recent iterations
                stall += 1
            else:
                stall = max(0, stall - 1)
            
            # Increase beta over time for sharper peak focus
            beta = min(beta_max, beta_init * (1.15 ** (it // 30)))
            
            # If very stalled, do a larger perturbation
            if stall >= 8:
                sigma = 0.05 * a.mean() * (1 + 0.5 * stall)
                a = safe_normalize(a + rng.normal(0, sigma, n), n, S)
                c, A = compute_conv(a, N)
                stall = 0
                beta = beta * 0.5  # reset beta after restart
        
        # Compute gradient
        grad = compute_grad(a, N, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        
        # Gradient step with multi-scale line search
        step = base_step / max(gnorm / (S / n), 0.1)
        a_new = a
        cmax_new = c.max()
        
        for scale in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            candidate = proj_simplex(a - step * scale * grad, S)
            c_ls, _ = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c.max() or scale == 0.05:
                a_new = candidate
                cmax_new = cm_ls
                break
        
        a = a_new
        c, A = compute_conv(a, N)
        
        # Track best
        if c.max() < best_peak - 1e-12:
            best_peak = c.max()
            best_a = a.copy()
        
        # Periodic coordinate descent
        if it % cd_interval == 0:
            a_cd, improved = coord_descent_peak(a, c, N, n, S, 
                                                 n_steps=cd_steps, rng=rng)
            if improved:
                c_cd, _ = compute_conv(a_cd, N)
                if c_cd.max() < best_peak:
                    best_peak = c_cd.max()
                    best_a = a_cd.copy()
                a = a_cd
    
    # Final coordinate descent polish
    for _ in range(3):
        a, improved = coord_descent_peak(a, c, N, n, S, n_steps=10, rng=rng)
        if improved:
            c, A = compute_conv(a, N)
            if c.max() < best_peak:
                best_peak = c.max()
                best_a = a.copy()
    
    return best_a, best_peak


def make_init_smooth(n, rng):
    """Smooth oscillatory initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for freq in rng.integers(2, 30, size=4):
        a += 0.2 * np.sin(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
    return np.clip(a, 0.01, None)


def make_init_fourier(n, rng, N):
    """Random Fourier initialization."""
    mags = rng.exponential(1.0, N)
    mags[0] = float(n)
    mags = np.maximum(mags, 1e-8)
    phases = rng.uniform(0, 2 * np.pi, N)
    A = np.sqrt(mags) * np.exp(1j * phases)
    a = np.abs(np.real(ifft(A)))[:n]
    return a + 0.01


def make_init_deflation(n, S, N, rng, n_iters=500):
    """Deflation from uniform: iteratively reduce peak by moving mass."""
    a = np.ones(n) * S / n
    for _ in range(n_iters):
        c, _ = compute_conv(a, N)
        k_star = int(np.argmax(c))
        resp = peak_responsibility(a, k_star, n)
        eff = resp * a
        src = int(np.argmax(eff))
        # Find target: minimum responsibility
        targets = np.where(resp < 0.3 * resp.max())[0]
        if len(targets) == 0:
            break
        tgt = int(targets[rng.integers(len(targets))])
        if src == tgt:
            continue
        transfer = min(0.02 * a[src], 0.5 * a[src])
        a[src] -= transfer
        a[tgt] += transfer
    return a

        a_res, peak_res = optimize_single(a_curr, n, S, N, bt, rng,
                                           beta_init=beta_s, beta_max=beta_m,
                                           cd_interval=30 + 10 * b,
                                           cd_steps=3 + b)
        score_res = 2.0 * n * peak_res / (S * S)
        if score_res < global_best_score - 1e-9:
            try_update(a_res)
            a_start = a_res.copy()
            stall_blocks = 0
        else:
            stall_blocks += 1
        print("  [P2-b%d] peak=%.6f score=%.6f (beta %.0f, stall=%d)" %
              (b, peak_res, score_res, beta_s, stall_blocks))
    
    print("Phase 2 done (%.1fs), best=%.6f" % (time.monotonic() - t1, global_best_score))
    
    # === Phase 3: Intensive coordinate descent polishing ===
    print("\n[Phase 3] Coordinate descent polishing")
    t2 = time.monotonic()
    p3_budget = min(500.0, (deadline - t2) * 0.22)
    
    if global_best_a is not None and len(global_best_a) == n and p3_budget > 20:
        a_cd = global_best_a.copy()
        c_cd, _ = compute_conv(a_cd, N)
        cd_best = c_cd.max()
        cd_best_a = a_cd.copy()
        cd_stall = 0
        
        cd_start = time.monotonic()
        cd_iter = 0
        while time.monotonic() - cd_start < p3_budget:
            cd_iter += 1
            a_cd, improved = coord_descent_peak(a_cd, c_cd, N, n, S,
                                                 n_steps=8, rng=rng)
            c_new, _ = compute_conv(a_cd, N)
            if c_new.max() < cd_best - 1e-12:
                cd_best = c_new.max()
                cd_best_a = a_cd.copy()
                cd_stall = 0
                if cd_iter % 50 == 0:
                    print("    CD iter=%d, peak=%.6f, score=%.6f" %
                          (cd_iter, cd_best, 2.0 * n * cd_best / (S * S)))
            else:
                cd_stall += 1
                if cd_stall >= 20:
                    # Perturb to escape
                    a_cd = safe_normalize(a_cd + 0.01 * a_cd.mean() * rng.standard_normal(n), n, S)
                    c_cd, _ = compute_conv(a_cd, N)
                    cd_stall = 0
            
            if cd_iter % 100 == 0:
                c_cd, _ = compute_conv(a_cd, N)
    
        try_update(cd_best_a)
    
    print("Phase 3 done (%.1fs), best=%.6f" % (time.monotonic() - t2, global_best_score))
    
    # === Phase 4: Final high-beta gradient polish ===
    print("\n[Phase 4] Final high-beta gradient polish")
    t3 = time.monotonic()
    remaining4 = deadline - t3 - 10
    if remaining4 > 30 and global_best_a is not None and len(global_best_a) == n:
        a_fin, peak_fin = optimize_single(global_best_a, n, S, N, remaining4, rng,
                                           beta_init=80.0, beta_max=1500.0,
                                           cd_interval=20, cd_steps=5)
        try_update(a_fin)
        print("  [P4] peak=%.6f, score=%.6f" % (peak_fin, 2.0 * n * peak_fin / (S * S)))
    
    # === Verification ===
    if global_best_a is not None:
        global_best_a = np.clip(global_best_a, 0.0, 1000.0)
        final_list = [float(x) for x in global_best_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verification: %.6f" % ver)
            print("Sequence length: %d" % len(final_list))
        else:
            print("WARNING: verification returned inf, using fallback")
            global_best_a = np.ones(256, dtype=np.float64)
            final_list = [1.0] * 256
            ver = evaluate_sequence(final_list)
            print("Fallback verification: %.6f" % ver)
    else:
        global_best_a = np.ones(256, dtype=np.float64)
        final_list = [1.0] * 256
        print("No valid solution found, using uniform fallback")
    
    print("Total time used: %.1fs" % (time.monotonic() - (deadline - budget_s + 20.0)))
    return final_list
