      def project_simplex(v, S):
          """Projectvonto{x>=0,sum(x)=S}."""
          v = np.asarray(v, dtype=np.float64)
          w = np.sort(v)[::-1]
          cumsum = np.cumsum(w)
          idx = np.arange(1, len(w)+1, dtype=np.float64)
          tc = (cumsum - S) / idx
          if not np.any(tc <= w):
              rho = len(w)
          else:
              rho = np.max(np.where(tc <= w)[0]) + 1
          theta = (cumsum[rho-1] - S) / rho
          return np.maximum(v - theta, 0.0)

def project_simplex(v, S):
    """Project all points onto x>=0 with sum constraint maintaining tight security."""
    v = np.asarray(v, dtype=np.float64)
    sorted_v = np.sort(v)[::-1]
