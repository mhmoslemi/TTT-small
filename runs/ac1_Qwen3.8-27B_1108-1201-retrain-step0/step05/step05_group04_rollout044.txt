    e = np.exp(beta * (c - cm))

    fsize = 1 << (len(w) + n - 1).bit_length()

    A = fft(a, fsize)

def run_subgradient_backtrack(a_init, n, S, max_time, rng,
                               n_top=15, rel_tol_start=1e-2, rel_tol_end=1e-5,
                               alpha_init=0.05, decay_factor=0.5):
    """
    Exact subgradient method with backtracking line search.
    Active set shrinks over time for increasing precision.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    alpha = alpha_init
    it = 0
    last_improve = 0
    total_time = max_time

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / total_time)

        # Adaptive active set: start broad, end narrow
        rel_tol = rel_tol_start * (rel_tol_end / rel_tol_start) ** frac
        n_top_cur = max(3, int(n_top * (1 - 0.5 * frac)))

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            last_improve = it

        K = find_active_set_strict(c, n_top_cur, rel_tol)
        sg = active_set_subgradient(a, K)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break

        # Backtracking
        alpha_test = alpha
        improved = False
        best_test_a = None
        best_test_val = cur_max
        for _ in range(12):
            a_test = project_simplex(a - alpha_test * sg, S)
            c_test = fft_convolve_self(a_test)
            test_val = c_test.max()
            if test_val < best_test_val:
                best_test_val = test_val
                best_test_a = a_test.copy()
            if test_val < cur_max - 1e-8:
                a = a_test
                alpha = min(alpha_test * 1.5, 1.0)
                improved = True
                break
            alpha_test *= decay_factor

        if not improved:
            if best_test_a is not None and best_test_val < cur_max:
                a = best_test_a
                alpha = max(alpha_test * 0.5, 1e-6)
            else:
                # Stuck - inject small perturbation
                noise_scale = 0.001 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                alpha = max(alpha * 0.5, 1e-6)
                last_improve = it

        # Periodic structured perturbation
        if it % 500 == 0:
            perturbation = 0.005 * S / np.sqrt(n)
            # Low-frequency sinusoidal perturbation
            t_arr = np.linspace(0, 1, n)
            freq = rng.integers(2, 10)
            phase = rng.uniform(0, 2 * np.pi)
            a = project_simplex(a + perturbation * np.sin(2 * np.pi * freq * t_arr + phase), S)
            last_improve = it

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_peak_shaving(a_init, n, S, max_time, rng):
    """
    Surgical peak shaving: iterate over peaks and reduce contributing elements.
    Uses a smarter strategy: for each peak position k, identify the pair (j, k-j)
    with maximum product and rebalance.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Get top peaks
        n_peaks = 5
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # For each peak, find the dominant contributing pair
        for k in top_idx:
            if c[k] < cur_max * 0.999:
                continue
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo > hi:
                continue
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            max_contrib_idx = np.argmax(contrib)
            j_star = js[max_contrib_idx]
            p_star = partners[max_contrib_idx]

            # Reduce the larger of the pair, increase a small element elsewhere
            if a[j_star] >= a[p_star]:
                donor = j_star
                val = a[donor]
            else:
                donor = p_star
                val = a[donor]

            # Find a low-value position to receive the mass
            low_positions = np.argsort(a)[:max(1, n // 100)]
            receiver = low_positions[rng.integers(len(low_positions))]

            # Transfer a small amount
            transfer = 0.01 * val
            a[donor] -= transfer
            a[receiver] += transfer
            a = np.maximum(a, 0.0)
            a = project_simplex(a, S)

        # Also try multiplicative reduction of high-responsibility elements
        if it % 10 == 0:
            resp = np.zeros(n)
            for k in top_idx[:3]:
                lo = max(0, k - n + 1)
                hi = min(k, n - 1)
                if lo > hi:
                    continue
                js = np.arange(lo, hi + 1)
                partners = k - js
                contrib = a[js] * a[partners]
                weight = max(0, c[k] - cur_max * 0.995)
                np.add.at(resp, js, contrib * weight)
            resp_max = resp.max()
            if resp_max > 1e-30:
                a = a * np.exp(-0.005 * resp / resp_max)
                a = project_simplex(a, S)

    return best_a, best


def generate_structured_perturbations(a, n, S, rng):
    """Generate structured perturbations of a."""
    perturbations = []
    t = np.linspace(0, 1, n)

    # Shift center of mass
    for shift in [0.02, -0.02, 0.05, -0.05]:
        idx = np.round((t + shift) * (n - 1)).astype(int)
        idx = np.clip(idx, 0, n - 1)
        perturbations.append(a[idx])

    # Add/remove sinusoidal component
    for freq in [2, 3, 5, 8]:
        for amp in [0.01, 0.03, 0.05]:
            for phase in [0, np.pi / 2, np.pi]:
                perturbations.append(a + amp * S / n * np.sin(2 * np.pi * freq * t + phase))

    # Scale different halves differently
    mid = n // 2
    for ratio in [1.1, 0.9, 1.2, 0.8]:
        new_a = a.copy()
        new_a[:mid] *= ratio
        perturbations.append(new_a)

    # Add gaussian envelope perturbation
    for sigma in [0.05, 0.1, 0.2]:
        for mu in [0.3, 0.5, 0.7]:
            pert = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
            perturbations.append(a + 0.02 * S / n * pert)

    # Random sparse perturbation
    for n_spikes in [10, 50, 200]:
        idx = rng.choice(n, size=n_spikes, replace=False)
        new_a = a.copy()
        new_a[idx] *= (1 + rng.normal(0, 0.1, n_spikes))
        perturbations.append(new_a)

    return perturbations


def quick_eval(a, n):
    """Quick evaluation of the objective."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a, n)
    c = fft_convolve_self(a)
    return c.max()


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search minimizing 2*n*max(f*f)/sum(f)^2.
    Phase 1: LSE-FISTA (bulk optimization)
    Phase 2: Exact subgradient with backtracking (precision)
    Phase 3: Structured perturbation + re-optimization (escape local minima)
    Phase 4: Surgical peak shaving (final polish)
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))

    # Load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded given sequence: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_sequence(a):
        nonlocal global_best, global_a
        n = len(a)
        S = np.sqrt(2.0 * n)
        a_norm = normalize_sum(a, n)
        c = fft_convolve_self(a_norm)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a_norm.copy()
            print("  New best: %.8f (n=%d)" % (val, n))

    # === Phase 1: LSE-FISTA on given sequence ===
    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        remaining = deadline - time.monotonic()
        t1 = min(remaining * 0.3, 300.0)
        print("=== Phase 1: LSE-FISTA (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_lse_fista(given, n_g, S_g, t1, rng,
                                beta_start=100.0, beta_end=10000.0, beta_steps=150)
        try_sequence(a1)
    else:
        n_g = 8192
        S_g = np.sqrt(2.0 * n_g)
        # Create initial sequence
        t = np.linspace(0, 1, n_g)
        a_init = np.exp(-0.5 * ((t - 0.4) / 0.25) ** 2)
        a_init = np.maximum(a_init, 1e-10)
        t1 = min(120.0, (deadline - time.monotonic()) * 0.2)
        print("=== Phase 1 (no given): LSE-FISTA (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_lse_fista(a_init, n_g, S_g, t1, rng,
                                beta_start=50.0, beta_end=5000.0, beta_steps=100)
        try_sequence(a1)

    # === Phase 2: Exact subgradient with backtracking ===
    if global_a is not None:
        n_m = len(global_a)
        S_m = np.sqrt(2.0 * n_m)
        remaining = deadline - time.monotonic()
        t2 = min(remaining * 0.3, 300.0)
        print("=== Phase 2: Subgradient+Backtrack (n=%d, %.0fs) ===" % (n_m, t2))
        a2, v2 = run_subgradient_backtrack(global_a, n_m, S_m, t2, rng,
                                            n_top=20, rel_tol_start=1e-2, rel_tol_end=1e-6,
                                            alpha_init=0.03, decay_factor=0.4)
        try_sequence(a2)

    # === Phase 3: Structured perturbation + re-optimization ===
    if global_a is not None:
        n_p = len(global_a)
        S_p = np.sqrt(2.0 * n_p)
        remaining = deadline - time.monotonic()
        print("=== Phase 3: Structured perturbations (n=%d, remaining %.0fs) ===" % (n_p, remaining))

        perturbations = generate_structured_perturbations(global_a, n_p, S_p, rng)

        # Quick screen all perturbations
        pert_scores = []
        for i, p in enumerate(perturbations):
            p = normalize_sum(p, n_p)
            score = quick_eval(p, n_p)
            pert_scores.append((score, i, p))

        pert_scores.sort(key=lambda x: x[0])

        # Optimize the top perturbations
        n_to_optimize = min(8, len(pert_scores))
        per_pert_time = max(3.0, (remaining * 0.35) / max(1, n_to_optimize))

        for rank, (score, idx, p) in enumerate(pert_scores[:n_to_optimize]):
            if time.monotonic() > deadline - 30:
                break
            if per_pert_time < 2:
                break
            if score >= global_best * 1.01:
                break  # Not promising enough
            print("  Pert %d (score=%.6f, improvement=%.4f%%): optimizing..." %
                  (idx, score, (global_best - score) / global_best * 100))
            # Quick LSE optimization
            a_opt, v_opt = run_lse_fista(p, n_p, S_p, per_pert_time, rng,
                                          beta_start=500.0, beta_end=20000.0, beta_steps=50)
            try_sequence(a_opt)

    # === Phase 4: Peak shaving ===
    if global_a is not None:
        n_s = len(global_a)
        S_s = np.sqrt(2.0 * n_s)
        remaining = deadline - time.monotonic()
        t4 = min(remaining * 0.2, 120.0)
        if t4 > 5:
            print("=== Phase 4: Peak shaving (n=%d, %.0fs) ===" % (n_s, t4))
            a_s, v_s = run_peak_shaving(global_a, n_s, S_s, t4, rng)
            try_sequence(a_s)

            # Quick LSE polish after shaving
            remaining2 = deadline - time.monotonic()
            if remaining2 > 15:
                a_s2, v_s2 = run_lse_fista(a_s, n_s, S_s, remaining2 * 0.4, rng,
                                            beta_start=5000.0, beta_end=30000.0, beta_steps=30)
                try_sequence(a_s2)

    # === Phase 5: Final subgradient polish ===
    if global_a is not None:
        n_f = len(global_a)
        S_f = np.sqrt(2.0 * n_f)
        remaining = deadline - time.monotonic()
        t5 = min(remaining * 0.3, 60.0)
        if t5 > 5:
            print("=== Phase 5: Final subgradient polish (%.0fs) ===" % t5)
            a_f, v_f = run_subgradient_backtrack(global_a, n_f, S_f, t5, rng,
                                                  n_top=5, rel_tol_start=1e-4, rel_tol_end=1e-8,
                                                  alpha_init=0.01, decay_factor=0.3)
            try_sequence(a_f)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.8f, internal = %.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result, using uniform fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
