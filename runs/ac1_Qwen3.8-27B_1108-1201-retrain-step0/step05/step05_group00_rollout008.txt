"""
Minimize 2*n*max_conv/sum^2 using LP-based iterative refinement with L-BFGS polishing.

Primary method: At each iteration, solve an LP to find the best perturbation direction
that keeps the cross-convolution bounded. This directly handles the non-smooth max
objective and provides reliable descent.

Secondary: L-BFGS on LSE for quick local improvements between LP solves.
Tertiary: Spectral phase randomization for escaping local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import lil_matrix
import cvxpy as cp


def compute_conv_full(a, N=0):
    n = len(a)
    if N < 2*n-1:
        N = 1 << (2*n-1).bit_length()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A*A))[:2*n-1]
    return c


def lse_grad(a, beta, N):
    n = len(a)
    c = compute_conv_full(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta*(c - c_max))
    total = exp_bc.sum()
    weights = exp_bc / total
    lse_val = c_max + np.log(total)/beta
    fsize = 1 << (3*n-2).bit_length()
    W = fft(weights.astype(np.complex128), fsize)
    Af = fft(a.astype(np.complex128), fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]
    return grad, lse_val, c, c_max


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def solve_lp(f, M, top_k_lags, solver=None):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for k in top_k_lags, g >= 0.
    Returns the optimal g, or None on failure.
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    K = len(top_k_lags)
    
    A = lil_matrix((K, n))
    for idx, k in enumerate(top_k_lags):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo > j_hi:
            continue
        for j in range(j_lo, j_hi+1):
            A[idx, j] = f[k-j]
    
    A = A.tocsr()
    constraints = [A @ g <= M * np.ones(K)]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    if solver is None:
        prob.solve()
    else:
        prob.solve(solver=solver)
    
    if g.value is not None:
        return np.asarray(g.value, dtype=np.float64).flatten()
    return None


def lbfgs_lse_block(a, n, S, N, max_time, rng, beta_init=5.0, beta_max=200.0, m_lbfgs=10):
    """L-BFGS on LSE-smoothed objective."""
    a = safe_normalize(a, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = compute_conv_full(a, N).max()
    s_hist = []
    y_hist = []
    
    g, f_val, c, c_max = lse_grad(a, beta, N)
    step_scale = 2.0 * S / (n * max(beta, 1.0))
    start_time = time.monotonic()
    it = 0
    
    s_last = None
    y_last = None
    gamma = 1.0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.15)
            step_scale = 2.0 * S / (n * max(beta, 1.0))
            g, f_val, c, c_max = lse_grad(a, beta, N)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
        
        # L-BFGS direction (simplified memory)
        if len(s_hist) == 0:
            direction = -g
        else:
            # Two-loop recursion
            m = len(s_hist)
            q = g.copy()
            alphas = np.zeros(m)
            for i in range(m-1, -1, -1):
                s_i = s_hist[i]
                y_i = y_hist[i]
                yy = np.dot(y_i, y_i)
                if yy > 1e-30:
                    alphas[i] = np.dot(s_i, q) / yy
                    q = q - alphas[i] * y_i
            if s_last is not None and y_last is not None:
                yy_l = np.dot(y_last, y_last)
                if yy_l > 1e-30:
                    gamma = max(1e-8, np.dot(s_last, y_last) / yy_l)
            r = gamma * q
            for i in range(m):
                s_i = s_hist[i]
                y_i = y_hist[i]
                yy = np.dot(y_i, y_i)
                if yy > 1e-30:
                    beta_i = np.dot(s_i, y_i) / yy
                    r = r + (alphas[i] - beta_i) * y_i
            direction = -r
            if np.dot(g, direction) > 0:
                direction = -g
        
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        # Line search
        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        for ls in range(10):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = compute_conv_full(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max or ls == 9:
                a_new = candidate
                cmax_new = cm_ls
                break
            alpha_ls *= 0.5
        
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
        
        g_new, _, c_nw, cmax_nw = lse_grad(a_new, beta, N)
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        ss = np.dot(s_k, s_k)
        if sy > 1e-14 * max(ss, 1e-30) and ss > 1e-20:
            s_hist.append(s_k)
            y_hist.append(y_k)
            s_last = s_k
            y_last = y_k
            if len(s_hist) > m_lbfgs:
                s_hist = s_hist[-m_lbfgs:]
                y_hist = y_hist[-m_lbfgs:]
        
        a = a_new
        g = g_new
        c_max = cmax_nw
        
        if it % 300 == 0 and rng.random() < 0.2:
            # Spectral partial restart
            a_N = fft(a, N)
            mag = np.abs(a_N)
            phase = np.angle(a_N)
            n_rand = max(1, int(N * 0.3))
            rand_idx = rng.choice(N, size=n_rand, replace=False)
            phase[rand_idx] = rng.uniform(0, 2*np.pi, n_rand)
            a_new_sp = np.real(ifft(mag * np.exp(1j*phase)))[:n]
            a_new_sp = np.maximum(a_new_sp, 0)
            s_sp = a_new_sp.sum()
            if s_sp > 1e-15:
                a = a_new_sp * (S / s_sp)
                s_hist = []
                y_hist = []
                g, _, c, c_max = lse_grad(a, beta, N)
    
    return best_a, best_cmax


def spectral_phase_restart(a, N, rng, frac=1.0):
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    n_rand = max(1, int(N * frac))
    rand_idx = rng.choice(N, size=n_rand, replace=False)
    phase[rand_idx] = rng.uniform(0, 2*np.pi, n_rand)
    a_new = np.real(ifft(mag * np.exp(1j*phase)))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def lp_iteration(f, M, K, S, n, N, solver=None):
    """
    One LP-based iteration.
    Returns (new_f, improved) where new_f has reduced peak.
    """
    c = compute_conv_full(f, N)
    top_k = np.argsort(c)[::-1][:K]
    
    g = solve_lp(f, M, top_k, solver=solver)
    if g is None:
        return f, False
    
    g = safe_normalize(g, n, S)
    
    # Line search
    best_alpha = 0.0
    best_peak = M
    for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9]:
        candidate = (1.0 - alpha) * f + alpha * g
        candidate = np.maximum(candidate, 0)
        s_cand = candidate.sum()
        if s_cand < 1e-15:
            continue
        candidate = candidate * (S / s_cand)
        c_cand = compute_conv_full(candidate, N)
        peak_cand = c_cand.max()
        if peak_cand < best_peak:
            best_peak = peak_cand
            best_alpha = alpha
    
    if best_alpha == 0.0:
        return f, False
    
    new_f = (1.0 - best_alpha) * f + best_alpha * g
    new_f = safe_normalize(new_f, n, S)
    return new_f, True
