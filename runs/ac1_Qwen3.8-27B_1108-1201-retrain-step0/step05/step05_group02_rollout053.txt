    # In the frequency domain: C = A^2 (element-wise, with zero padding to fs)
    # sum_k w[k] c[k] = real(ifft(w_fft) * c) ... no
    # = real(<w_fft, C> / fs) where <,> is complex inner product? No.
    # Actually: sum_k w[k] c[k] = real(sum_k w[k] * c[k]) (since both are real)
    # = real(ifft(w_fft) * c) ... no that's not how it works.
    # 
    # By Parseval: sum_k w[k] c[k] = (1/fs) * sum_f W[f] * conj(C[f]) ... no
    # Parseval: sum_k x[k] y[k] = (1/fs) sum_f X[f] conj(Y[f])
    # So sum_k w[k] c[k] = (1/fs) sum_f W[f] conj(C[f]) = (1/fs) sum_f W[f] conj(A[f]^2)
    # = (1/fs) sum_f W[f] conj(A[f])^2
    # 
    # d/d a[j] of this: A[f] = sum_i a[i] exp(-2pi i f i / fs)
    # d A[f] / d a[j] = exp(-2pi i f j / fs)
    # d conj(A[f]) / d a[j] = exp(+2pi i f j / fs) (since a is real)
    # 
    # d/da[j] (1/fs) sum_f W[f] conj(A[f])^2 = (1/fs) sum_f 2 W[f] conj(A[f]) exp(2pi i f j / fs)
    # = 2 * (1/fs) sum_f (W[f] conj(A[f])) exp(2pi i f j / fs)
    # = 2 * ifft(W * conj(A))[j]  (where ifft uses the convention with 1/fs normalization)
    # 
    # So grad = 2 * ifft(fft(w, fs) * conj(fft(a, fs)))[:n]
    
    fsize = next_fast_len(2 * n - 1)
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    # LSE value (approximate)
    if above.any():
        top_vals = c[above]
        lse_val = top_vals.max() + np.log(e.sum()) / beta
    else:
        lse_val = c.max()
    
    return lse_val, grad, c


def fista_active(a_init, max_time, rng, beta_init=50.0, k_top=5,

    """FISTA with active-set (top-k) LSE."""

        step_local = step

            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            if it - last_improve > 8000:

def fourier_equalize(a_init, max_time, rng):
    """
    Fourier-domain spectral equalization.
    Iteratively modify the amplitude spectrum to flatten the convolution,
    then project back to non-negative sequence.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    fsize = next_fast_len(2 * n - 1)
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        if it - last_improve > 5000:
            break
        
        # Compute current spectrum
        A = fft(a, fsize)
        # Target: we want |A|^2 to be more uniform (to flatten c)
        # The "ideal" |A|^2 for flat c would be constant
        # Use a soft target: |A|^2 target = mean(|A|^2) with some flattening
        amp2 = np.abs(A) ** 2
        mean_amp2 = amp2.mean()
        
        # Flattening: push amplitude towards the mean
        # Use a power compression: new_amp = (amp2^alpha)^(1/2)
        # alpha < 1 compresses dynamic range
        alpha = 0.7  # compression factor
        phase = np.angle(A)
        new_amp = (amp2 ** alpha) ** 0.5  # = amp2^0.35
        # Actually, let's use: new_A = A * (mean_amp2 / |A|^2)^gamma for gamma in (0, 0.5)
        gamma = 0.3
        ratio = mean_amp2 / np.maximum(amp2, 1e-30)
        new_amp2 = amp2 * ratio ** gamma
        new_A = np.sqrt(new_amp2) * np.exp(1j * phase)
        
        # But we need to preserve the DC component (sum constraint)
        # DC: A[0] = sum(a) = S (for real a)
        # We'll re-normalize after IFFT
        
        # IFFT to get the new sequence
        a_new = np.real(ifft(new_A))[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() > 1e-15:
            a_new = a_new * (S / a_new.sum())
        
        # Blend with current (to avoid oscillation)
        blend = 0.3 + 0.2 * np.sin(0.001 * it)  # oscillating blend
        a = (1 - blend) * a + blend * a_new
        a = np.maximum(a, 0.0)
        a = a * (S / a.sum())
    
    return best_a, best_obj


def projected_gradient(a_init, max_time, rng, k_subgrad=1):
    """
    Projected subgradient method with careful step size.
    Uses the exact subgradient from the argmax of the convolution.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    fsize = next_fast_len(2 * n - 1)
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        if it - last_improve > 5000:
            break
        
        # Subgradient from top-k lags
        k = min(k_subgrad + it // 1000, 20)
        top_idx = np.argsort(c)[::-1][:k]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / k
        
        # Gradient
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: use 2*f/(L*t) type rule
        step = 2.0 * (cur_obj - 1.0) / (gnorm * gnorm * (1.0 + 0.001 * it))
        step = min(step, 0.5 * S / gnorm)
        
        a_new = project_simplex(a - step * grad, S)
        obj_new = compute_obj(a_new)
        
        if obj_new < cur_obj:
            a = a_new
        else:
            # Try smaller step
            for factor in [0.5, 0.1, 0.01]:
                a_try = project_simplex(a - step * factor * grad, S)
                if compute_obj(a_try) < cur_obj:
                    a = a_try
                    break
    
    return best_a, best_obj


def make_init(n, method, rng):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "hann":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "flat_top":
        a = 1.0 - 0.36 * np.cos(2*np.pi*tt) - 0.24 * np.cos(4*np.pi*tt) + 0.10 * np.cos(6*np.pi*tt) + 0.04 * np.cos(8*np.pi*tt)
    elif method == "gaussian":
        a = np.exp(-3.0 * (tt - 0.5) ** 2 / (0.2 ** 2))
    elif method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "multi_tone":
        a = 0.3 + 0.25 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt + 1.3) + 0.15 * np.sin(2 * np.pi * 11 * tt + 2.1)
        a = np.abs(a)
    elif method == "random_smooth":
        a = rng.random(n)
        # Smooth with a small Gaussian kernel
        kernel = np.exp(-0.5 * (np.arange(-20, 21) / 5.0) ** 2)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 2.0 * np.abs(tt - 0.5))
    elif method == "triangle":
        a = 1.0 - 2.0 * np.abs(tt - 0.5)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def block_coordinate_descent(a_init, max_time, rng, block_size=128):
    """
    Block coordinate descent: optimize one block at a time.
    For each block, we can use the gradient information from the full
    convolution to guide the optimization.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        if it - last_improve > 3000:
            break
        
        # Pick a block (cycle through)
        n_blocks = n // block_size
        if n_blocks < 1:
            block_size = max(1, n // 4)
            n_blocks = n // block_size
        block_idx = it % n_blocks
        blo = block_idx * block_size
        bhi = min(blo + block_size, n)
        
        # Get the sub-gradient for this block
        # The gradient at positions [blo:bhi] tells us how to adjust this block
        # Use top-k lags for the subgradient
        k = min(5, 2*n-1)
        top_idx = np.argsort(c)[::-1][:k]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / k
        
        fsize = next_fast_len(2 * n - 1)
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad_full = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        grad_block = grad_full[blo:bhi]
        
        # Optimize this block: minimize subgradient objective subject to non-negativity
        # and keeping the sum of this block fixed (to maintain total sum)
        block_sum = a[blo:bhi].sum()
        
        # Simple: project the block onto the simplex with the current block sum
        step = 0.1
        for _ in range(10):  # Inner iterations for this block
            block_new = a[blo:bhi] - step * grad_block
            block_new = np.maximum(block_new, 0)
            if block_new.sum() > 1e-15:
                block_new = block_new * (block_sum / block_new.sum())
            a[blo:bhi] = block_new
            step *= 0.7
        
    return best_a, best_obj


def upsample(a, factor=2):
    """Upsample via interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)
