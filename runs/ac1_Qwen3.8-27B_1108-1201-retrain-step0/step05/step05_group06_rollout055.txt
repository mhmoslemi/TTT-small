import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 3000:
            t_mom = 1.0
            noise = 0.001 * S / np.sqrt(n) * np.exp(-stuck / 20000.0)
            if noise > 1e-10:
                y = project_simplex(y + rng.normal(0, noise, n), S)
            stuck = 0
            last_improve = it

    return best_a


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.54 + 0.46 * np.cos(2 * np.pi * tt)


def make_random_structured(n, rng):
    """Random init with some structure."""
    a = rng.random(n)
    # Apply a smooth filter
    kernel_size = max(3, n // 100)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: LSE FISTA -> Coordinate Descent -> Subgradient Polish -> Multi-n.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_total_start = time.monotonic()
    deadline = t_total_start + max(0.0, budget_s - 20.0)

    global_best_obj = np.inf
    global_best_a = None

    # Try to get provided sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        n0 = len(a_start)
    else:
        n0 = 8192
        a_start = make_cosine_init(n0)

    print("Base n = %d" % n0)

    # Quick evaluation of starting point
    obj_start = compute_obj(a_start)
    print("Starting obj: %.6f" % obj_start)
    if obj_start < global_best_obj:
        global_best_obj = obj_start
        global_best_a = a_start.copy()

    # === Phase 1: LSE FISTA warmup (20% of budget) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.20
    if p1_budget > 10:
        print("\n=== Phase 1: LSE FISTA warmup (%.0fs) ===" % p1_budget)
        t1 = time.monotonic()
        a1 = run_lse_fista(a_start, p1_budget, rng, beta_init=5.0, beta_rate=0.006, beta_max=10000.0)
        obj1 = compute_obj(a1)
        print("  LSE: obj=%.6f (t=%.1fs)" % (obj1, time.monotonic() - t1))
        if obj1 < global_best_obj:
            global_best_obj = obj1
            global_best_a = a1.copy()

    # === Phase 2: Coordinate descent (45% of budget) ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.45
    if p2_budget > 10:
        print("\n=== Phase 2: Coordinate Descent (%.0fs) ===" % p2_budget)
        t2 = time.monotonic()
        a2, obj2 = run_coord_descent(global_best_a, p2_budget, rng)
        obj2 = compute_obj(a2)
        print("  CoordDesc: obj=%.6f (t=%.1fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

    # === Phase 3: Subgradient polish (15% of budget) ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.15
    if p3_budget > 10:
        print("\n=== Phase 3: Subgradient Polish (%.0fs) ===" % p3_budget)
        t3 = time.monotonic()
        a3 = run_subgradient_polish(global_best_a, p3_budget, rng, beta_init=200.0)
        obj3 = compute_obj(a3)
        print("  Polish: obj=%.6f (t=%.1fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

    # === Phase 4: Second coordinate descent round (10% of budget) ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.10
    if p4_budget > 10:
        print("\n=== Phase 4: Coord Descent round 2 (%.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        a4, obj4 = run_coord_descent(global_best_a, p4_budget, rng, recompute_every=500)
        obj4 = compute_obj(a4)
        print("  CoordDesc2: obj=%.6f (t=%.1fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Multi-n exploration (10% of budget) ===
    remaining = deadline - time.monotonic()
    if remaining > 120:
        p5_budget = remaining * 0.60
        print("\n=== Phase 5: Multi-n exploration (%.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        # Try 2x upsample
        n_cur = len(global_best_a)
        n_try = n_cur * 2
        if n_try <= 32768:
            # Upsample
            t_old = np.linspace(0, 1, n_cur, endpoint=False)
            t_new = np.linspace(0, 1, n_try, endpoint=False)
            a_up = np.interp(t_new, t_old, global_best_a)
            S_up = np.sqrt(2.0 * n_try)
            a_up = normalize_seq(a_up, S_up)

            # Quick LSE on upsampled
            a_up_opt = run_lse_fista(a_up, p5_budget * 0.4, rng, beta_init=5.0, beta_rate=0.005, beta_max=5000.0)
            obj_up = compute_obj(a_up_opt)
            print("  Upsampled (n=%d): obj=%.6f" % (n_try, obj_up))
            if obj_up < global_best_obj:
                global_best_obj = obj_up
                global_best_a = a_up_opt.copy()

            # Coord descent on upsampled
            rem5 = p5_budget - (time.monotonic() - t5)
            if rem5 > 30:
                a_up2, _ = run_coord_descent(global_best_a, rem5, rng)
                obj_up2 = compute_obj(a_up2)
                print("  Upsampled CD (n=%d): obj=%.6f" % (n_try, obj_up2))
                if obj_up2 < global_best_obj:
                    global_best_obj = obj_up2
                    global_best_a = a_up2.copy()

    # === Phase 6: Final polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 6: Final polish (%.0fs) ===" % remaining)
        t6 = time.monotonic()
        a_fin, _ = run_coord_descent(global_best_a, remaining * 0.5, rng, recompute_every=200)
        obj_fin = compute_obj(a_fin)
        print("  Final1: obj=%.6f" % obj_fin)
        if obj_fin < global_best_obj:
            global_best_obj = obj_fin
            global_best_a = a_fin.copy()

        rem6 = remaining - (time.monotonic() - t6)
        if rem6 > 10:
            a_fin2 = run_subgradient_polish(global_best_a, rem6, rng, beta_init=500.0)
            obj_fin2 = compute_obj(a_fin2)
            print("  Final2: obj=%.6f" % obj_fin2)
            if obj_fin2 < global_best_obj:
                global_best_obj = obj_fin2
                global_best_a = a_fin2.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL RESULT ===")
            print("Objective: %.6f (n=%d)" % (ver, len(final_list)))
            print("Total time: %.0fs" % (time.monotonic() - t_total_start))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
