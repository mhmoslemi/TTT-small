def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_weighted_grad(a, q, N_cache=None):
    """Compute gradient of sum_k q_k * conv(a,a)[k] w.r.t. a."""
    n = len(a)

    A = fft(a, N_cache)
    Q = fft(q, N_cache)
    T = np.real(ifft(Q * np.conj(A)))[:n]
    return 2.0 * T

    tot = e.sum()

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_weighted_grad(a, q, fsize):
    """Gradient of sum_k q_k * conv(a,a)[k] w.r.t. a. Uses FFT."""
    A = fft(a, fsize)
    Q = fft(q, fsize)
    T = np.real(ifft(Q * np.conj(A)))[:len(a)]
    return 2.0 * T


def multiplicative_step(a, grad, S, lr):
    """Mirror descent step with negative entropy (multiplicative update)."""
    # Center the gradient to preserve sum
    grad_centered = grad - np.mean(grad)
    # Scale learning rate for stability
    max_grad = np.max(np.abs(grad_centered))
    if max_grad < 1e-30:
        return a.copy()
    effective_lr = min(lr, 1.0 / max_grad)
    log_update = np.exp(-effective_lr * grad_centered)
    a_new = a * log_update
    a_new_sum = a_new.sum()
    if a_new_sum < 1e-30:
        return a.copy()
    a_new = a_new / a_new_sum * S
    return a_new


def minimax_optimize(a0, S, max_time, rng, verbose=True):
    """
    Alternating minimax:
    - q weights lags by peak concentration (softmax of c)
    - a updated via multiplicative descent on weighted quadratic
    """
    n = len(a0)
    fsize = 1 << (2 * n - 1).bit_length()
    a = project_simplex(a0, S)

    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    # Initialize q concentrated on peaks
    beta_q = 3.0
    q = np.exp(-beta_q * c / max(c.max(), 1e-15))
    q = q / q.sum()

    start = time.monotonic()
    it = 0
    stuck = 0
    lr = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_max - 1e-13:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Update q: increase concentration over time
        beta_q_eff = 3.0 + 15.0 * progress
        c_range = c.max() - c.min()
        if c_range > 1e-30:
            q = np.exp(-beta_q_eff * (c - c.min()) / c_range)
        else:
            q = np.ones_like(c) / len(c)
        q_sum = q.sum()
        if q_sum < 1e-300:
            q = np.ones_like(c) / len(c)
        else:
            q = q / q_sum

        # Compute weighted gradient
        grad = compute_weighted_grad(a, q, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive learning rate
        weighted_obj = np.dot(q, c)
        gap = cur_max - weighted_obj
        lr = max(0.1, min(2.0, 2.0 * gap / max(gnorm * S, 1e-10)))

        # Multiplicative step
        a_try = multiplicative_step(a, grad, S, lr)
        c_try = fft_convolve_self(a_try)

        if c_try.max() <= cur_max * (1.0 + 1e-9):
            a = a_try
        else:
            # Try projected gradient
            step = min(1.0, 2.0 * gap / max(gnorm ** 2, 1e-30))
            step = min(step, 3.0 / gnorm)
            a_try2 = project_simplex(a - step * grad, S)
            c_try2 = fft_convolve_self(a_try2)
            if c_try2.max() <= cur_max * (1.0 + 1e-9):
                a = a_try2
            else:
                # Tiny step
                a = multiplicative_step(a, grad, S, lr * 0.01)

        # Stuck: inject structured noise
        if stuck > 150:
            noise_scale = min(0.015, 0.001 * (stuck - 150) / 100.0) * S / np.sqrt(n)
            # Targeted noise: perturb coordinates that contribute to peak lags
            peak_idx = np.argmax(c)
            noise = rng.normal(0, noise_scale, n)
            # Extra perturbation at positions that contribute to the peak
            j_start = max(0, peak_idx - n + 1)
            j_end = min(n - 1, peak_idx)
            if j_start <= j_end:
                noise[j_start:j_end + 1] *= 2.0
            a = project_simplex(a + noise, S)
            stuck = 0

        if verbose and it % 500 == 0:
            obj = 2.0 * n * best_max / (S * S)
            print(f"    Minimax it={it}, t={elapsed:.0f}s, obj={obj:.6f}, stuck={stuck}")

    return best_a, best_max


def lse_polish(a0, S, max_time, rng, beta_init=500.0, beta_max=100000.0, verbose=True):
    """High-beta LSE polishing for final convergence."""
    n = len(a0)
    fsize = 1 << (2 * n - 1).bit_length()
    a = project_simplex(a0, S)

    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stuck = 0
    prev_x = a.copy()
    prev_grad = None

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = min(beta_max, beta_init * np.exp(0.015 * it))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 400:
            noise_scale = 0.003 * S / np.sqrt(n) * max(0.0, 1.0 - progress)
            if noise_scale > 1e-12:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stuck = 0
            continue

        # Compute LSE gradient
        cm = c.max()
        scaled = beta * (c - cm)
        e = np.exp(np.clip(scaled, -500, 0))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        grad = compute_weighted_grad(a, w, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Barzilai-Borwein step size
        if prev_grad is not None:
            s = a - prev_x
            y = grad - prev_grad
            sy = np.dot(s, y)
            yy = np.dot(y, y)
            ss = np.dot(s, s)
            if yy > 1e-30 and ss > 1e-30:
                if sy > 1e-15:
                    step = ss / sy
                else:
                    step = sy / yy
                step = abs(step)
            else:
                step = 1.0 / gnorm
        else:
            step = 1.0 / gnorm

        step = min(step, 5.0 / max(gnorm, 1e-10))
        step = max(step, 1e-12)

        # Line search
        accepted = False
        for _ in range(10):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-10):
                a = a_try
                accepted = True
                break
            step *= 0.5

        if not accepted:
            a = project_simplex(a - 0.01 / gnorm * grad, S)

        prev_x = a.copy()
        prev_grad = grad.copy()

        if verbose and it % 1000 == 0:
            obj = 2.0 * n * best_max / (S * S)
            print(f"    LSE it={it}, t={elapsed:.0f}s, beta={beta:.0f}, obj={obj:.6f}")

    return best_a, best_max
