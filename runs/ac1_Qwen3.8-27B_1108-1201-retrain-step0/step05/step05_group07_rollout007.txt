import numpy as np

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def compute_lse_gradient(a, c, beta):
    """
    Compute gradient of LSE_beta(conv(a,a)) = (1/beta)*log(sum(exp(beta*conv_k)))
    w.r.t. a. Uses FFT for efficiency.
    
    grad[i] = cross_corr(a, w)[i] + conv(a, w)[i]
    where w = softmax(beta * c)
    """
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.clip(scaled, -500, 500)
    e = np.exp(scaled)
    w = e / e.sum()

    # conv(a, w) has length n + (2n-1) - 1 = 3n - 2
    # cross_corr(a, w)[i] = sum_j a[j]*w[j+i], also fits in 3n-2
    fsize = 1 << (3 * n - 1).bit_length()

    A = fft(a, fsize)
    W = fft(w, fsize)
    A_flip = fft(a[::-1], fsize)

                a_curr, _ = topk_subgradient_polish(a_curr, S, t_p, rng,
                                                    label=f"ref{pi}")
            update_global(a_curr, f"Phase4 iter{pi}")

    # === Phase 5: Multi-restart perturbation ===
    rem = deadline - time.monotonic()
    if global_a is not None and rem > 50:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        n_restarts = 3
        t_per = (rem - 20) / n_restarts

        print(f"\nPhase 5: {n_restarts} perturbations, {t_per:.0f}s each")
        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            # Different perturbation scales
            scales = [0.01, 0.03, 0.05]
            scale = scales[ri % len(scales)]
            noise = rng.normal(0, scale * S / np.sqrt(n), n)
            a_pert = project_simplex(global_a + noise, S)

            # Quick LSE + polish
            a_pert, _ = lse_optimize(a_pert, S, t_per * 0.6, rng,
                                     beta_start=50.0, beta_end=5000.0,
                                     momentum=0.9, label=f"restart{ri}")
            update_global(a_pert, f"Phase5 restart{ri}")

            rem = deadline - time.monotonic()
            if rem > 15:
                a_pert, _ = topk_subgradient_polish(a_pert, S, min(60, rem * 0.3),
                                                    rng, label=f"restart{ri}p")
                update_global(a_pert, f"Phase5 restart{ri} polish")

    # === Final verification ===
    if global_a is None:
        print("WARNING: No valid solution, using fallback")
        global_a = np.ones(256)
        global_best = obj_from_conv(256, fft_convolve_self(global_a).max(), global_a.sum())

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_list = [float(x) for x in global_a]
    n_final = len(global_a)
    S_final = global_a.sum()
    c_final = fft_convolve_self(global_a)
    print(f"\n{'='*60}")
    print(f"FINAL: len={n_final}, sum={S_final:.4f}, max_conv={c_final.max():.6f}")
    print(f"Internal objective: {2.0*n_final*c_final.max()/(S_final**2):.6f}")
    print(f"{'='*60}")

    return final_list
