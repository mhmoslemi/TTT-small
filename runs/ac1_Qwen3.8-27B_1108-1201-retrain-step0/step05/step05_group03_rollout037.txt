def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    rho = int(np.max(np.where(valid)[0]) + 1) if np.any(valid) else n
    theta = (cumsum[rho - 1] - S) / rho
    result = v - theta
    result[result < 0] = 0.0
    # Fix rounding: ensure sum is exactly S
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference among positive entries
        pos = result > 0
        if pos.any():
            result[pos] += diff / pos.sum()
            result[result < 0] = 0
    return result

    tot = e.sum()

    grad_c = np.zeros(size)
    grad_c[idx] = w

def lse_value(c, k, beta):

    idx = np.argpartition(c, -k)[-k:]
    c_top = c[idx]
    c_max = c_top.max()
    scaled = beta * (c_top - c_max)
    scaled = np.clip(scaled, -500.0, 0.0)
    return c_max + np.log(np.exp(scaled).sum()) / beta


def spectral_perturb(a, amount, rng):

    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()

        return a.copy()

def spectral_flatten(a, alpha, rng):

    n = len(a)

    mean_p = power.mean()

The function we're optimizing is `f(a) = LSE_beta(top_k(convolve(a, a)))`.

The gradient is `grad_a = 2 * IFFT(FFT(grad_c) * conj(FFT(a)))[:n]`.

The Lipschitz constant of the gradient: The Hessian of f w.r.t. a is `2 * H^T * W * H` where H is the convolution operator and W is the Hessian of LSE w.r.t. c (a diagonal matrix with entries bounded by beta).

So `||Hessian||_2 <= 2 * beta * ||H||_2^2`.

`||H||_2` is the operator norm of the convolution. For a non-negative sequence a, the convolution operator maps g to convolve(a, g). The operator norm is at most `sum(a) = S` (by the triangle inequality / Young's inequality for convolutions).

So `L <= 2 * beta * S^2 = 2 * beta * 2n = 4 * beta * n`.

With beta = 30 and n = 8192: L <= 4 * 30 * 8192 = 983040.

Step size: 1/L ≈ 1e-6. That's very small.

But this is a very loose bound. In practice, the actual Lipschitz constant is much smaller. Let me use adaptive step size.

A better approach: use the Armijo condition for line search. Start with a step and reduce if the function increase is too large.

Actually, for FISTA, the standard approach is:
1. Compute the gradient at y.
2. Take a step: x = prox_{S/L}(y - (1/L) * grad(y)) where prox is the projection onto the simplex.
3. Update y using momentum.

The step size 1/L should be based on a good estimate of L. Let me estimate L from the data:

After a few iterations, I can estimate L as:
`L_est = 2 * beta * max_eigen(H^T H)`

where `max_eigen(H^T H)` can be estimated from `||H * v||^2 / ||v||^2` for a few random vectors v.

Or more simply: `L_est = 2 * beta * S^2` (the worst case bound).

For a tighter bound, note that `||H||_2^2 = lambda_max(H^T H)`. For the convolution operator, `H^T H` is the matrix of the "autocorrelation of the autocorrelation", which is related to the power spectrum.

Actually, `H^T H` applied to a vector v gives: `(H^T H v)[i] = sum_{i,j} H[i,j]^2 * v[j]`... no that's not right.

Let me just use the bound `L = 2 * beta * S^2` and see if FISTA converges well. If the step is too small, I can increase it adaptively.

Actually, a better practical approach: use the **backtracking** line search in FISTA. At each iteration:
1. Try the current step size.
2. Check if the Armijo condition is satisfied.
3. If not, halve the step size.

The Armijo check requires one function evaluation (one FFT convolution), which adds cost but ensures robustness.

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
1. Fast FISTA with fixed step size, adaptive top-k LSE smoothing, and momentum
2. LP-based Frank-Wolfe restarts (using reduced active-set constraints) to escape local minima
3. Spectral shaping (phase perturbation + power equalization) as additional restarts
4. Multi-start with different initializations
5. Final high-precision polishing phase
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import coo_matrix
from scipy.optimize import linprog


def fconv_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    fsize = 1 << (2 * n).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:2 * n - 1]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    rho = np.max(np.where((cs - S) / np.arange(1, n + 1) <= w)[0]) + 1
    theta = (cs[rho - 1] - S) / rho
    r = np.maximum(v - theta, 0.0)
    d = S - r.sum()
    if abs(d) > 1e-12:
        pos = r > 0
        if pos.any():
            r[pos] += d / pos.sum()
            r[r < 0] = 0.0
    return r


def lse_grad(a, c, k, beta):
    """Gradient of top-k LSE of c=conv(a,a) w.r.t. a."""
    n = len(a)
    size = len(c)
    k = min(k, size)
    fsize = 1 << (size + n - 1).bit_length()
    idx = np.argpartition(c, size - k)[size - k:]
    ct = c[idx]
    cm = ct.max()
    sc = beta * (ct - cm)
    sc = np.clip(sc, -500.0, 0.0)
    e = np.exp(sc)
    w = e / e.sum()
    gc = np.zeros(size)
    gc[idx] = w
    GC = fft(gc, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(GC * A.conj()))[:n]


def lse_val(c, k, beta):
    """Top-k LSE value."""
    k = min(k, len(c))
    idx = np.argpartition(c, len(c) - k)[len(c) - k:]
    ct = c[idx]
    return ct.max() + np.log(np.exp(np.clip(beta * (ct - ct.max()), -500, 0)).sum()) / beta


def spectrum_flatten(a, alpha):
    """One step of spectral power equalization."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n).bit_length()
    A = fft(a, fsize)
    p = np.abs(A) ** 2
    mp = p.mean()
    if mp < 1e-20:
        return a.copy()
    r = p / max(mp, 1e-20)
    corr = 1.0 + alpha * (1.0 - r) / (1.0 + r)
    np_ = np.maximum(p * corr, 1e-30)
    A2 = A * np.sqrt(np_ / np.maximum(p, 1e-30))
    a2 = np.real(ifft(A2, fsize))[:n]
    a2 = np.clip(a2, 0, None)
    s = a2.sum()
    return a2 * (S / s) if s > 1e-15 else a.copy()


def spectrum_perturb(a, amt, rng):
    """Phase perturbation in frequency domain."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n).bit_length()
    A = fft(a, fsize)
    ph = rng.normal(0, amt, fsize)
    ph[0] = 0
    A2 = A * np.exp(1j * ph)
    a2 = np.real(ifft(A2, fsize))[:n]
    a2 = np.clip(a2, 0, None)
    s = a2.sum()
    return a2 * (S / s) if s > 1e-15 else a.copy()


def lp_fw_step(a, cur_max, S, n):
    """LP-based Frank-Wolfe step using reduced active-set constraints."""
    c = fconv_self(a)
    M = cur_max
    size = 2 * n - 1
    thresh = M * 0.80
    active = np.where(c >= thresh)[0]
    max_act = min(len(active), 300)
    if max_act < 5:
        active = np.argsort(c)[-10:]
        max_act = len(active)
    else:
        active = active[np.argsort(c[active])[::-1][:max_act]]

    rows_l, cols_l, vals_l = [], [], []
    for m in active:
        js = range(max(0, m - n + 1), min(m, n - 1) + 1)
        for j in js:
            rows_l.append(m)
            cols_l.append(j)
            vals_l.append(a[m - j])
    if not rows_l:
        return a.copy()
    H = coo_matrix((vals_l, (rows_l, cols_l)), shape=(len(active), n)).tocsr()

    c_obj = -np.ones(n)
    A_ub = H
    b_ub = M * np.ones(len(active))
    bounds = [(0.0, None)] * n
    A_eq = np.ones((1, n))
    b_eq = np.array([S])
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0 or res.x is None:
        return a.copy()
    g = np.clip(res.x, 0, None)
    s = g.sum()
    if s < 1e-10:
        return a.copy()
    return g * (S / s)


def resample_seq(a, n_t):
    """Resample sequence to target length."""
    n_s = len(a)
    if n_s == n_t:
        return a.copy()
    t_o = np.linspace(0, 1, n_s)
    t_n = np.linspace(0, 1, n_t)
    return np.interp(t_n, t_o, a)


def obj_val(a):
    """Compute the objective value."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * fconv_self(a).max() / (s * s)


def run_fista(a_init, S, max_t, rng, b0=30.0, bg=0.0015, bmax=5000.0, kf=0.08):
    """Fast FISTA with fixed step size and adaptive top-k LSE."""
    n = len(a_init)
    a = project_simplex(a_init.copy(), S)
    size = 2 * n - 1
    c = fconv_self(a).copy()
    bm = c.max()
    ba = a.copy()
    t = 1.0
    lr = 1.0 / (4.0 * b0 * S * S)
    start = time.monotonic()
    it = 0
    last_imp = 0

    while time.monotonic() - start < max_t:
        it += 1
        bk = min(bmax, b0 * (1.0 + bg * it))
        kk = max(3, int(kf * size))

        g = lse_grad(a, c, kk, bk)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        xn = project_simplex(a - lr * g, S)
        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        yn = xn + ((t - 1.0) / tn) * (xn - a)
        yn = np.maximum(yn, 0.0)
        a = xn
        t = tn

        if it % 32 == 0 or it < 50:
            c = fconv_self(a)
            cm = c.max()
            if cm < bm:
                bm = cm
                ba = a.copy()
                last_imp = it

        if it % 4096 == 0 and it - last_imp > 20000:
            t = 1.0
            ns = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, ns, n), S)
            c = fconv_self(a)
            last_imp = it

    c = fconv_self(a)
    if c.max() < bm:
        bm = c.max()
        ba = a.copy()
    return ba, bm

                a_s = spectrum_flatten(a_s, 0.06)
                a_s = project_simplex(a_s, SP)
                rem_sf = max(0.0, cyc_t - (time.monotonic() - st) - 2.0)
                if rem_sf > 5:
                    a4, v4 = run_fista(a_s, SP, rem_sf, rng, b0=40.0, bg=0.002,
                                       bmax=3000.0, kf=0.06)
                    if v4 < b_c:
                        a_c, b_c = a4.copy(), v4
                upd(a_c, b_c)
            except Exception:
                pass

    # Phase 3: final polish (20% remaining)
    rem = deadline - time.monotonic()
    if rem > 8 and ga is not None:
        n_f = len(ga)
        S_f = ga.sum()
        print("\n[P3] polish rem=%.0fs n=%d" % (rem, n_f))
        # Polynomial power domain
        rem_pd = max(0.0, rem * 0.35 - 5)
        if rem_pd > 5:
            try:
                a_pd, v_pd = run_pd_cycle(ga, S_f, rem_pd, rng, n_cycles=4)
                upd(a_pd, v_pd)
            except Exception:
                pass
        # FISTA with high beta
        rem_fp = max(0.0, (deadline - time.monotonic()) - 12)
        if rem_fp > 5:
            a_fp, v_fp = run_fista(ga, S_f, rem_fp, rng, b0=80.0, bg=0.0008,
                                   bmax=8000.0, kf=0.03)
            upd(a_fp, v_fp)
        # Polish
        rem_po = max(0.0, (deadline - time.monotonic()) - 8)
        if rem_po > 5:
            a_po, v_po = run_polish(ga, S_f, rem_po, rng)
            upd(a_po, v_po)

    # Verify
    if glist is None:
        glist = [float(x) for x in np.clip(ga, 0.0, 1000.0)] if ga is not None else [1.0]
    else:
        try:
            vv = evaluate_sequence(glist)
            if not np.isfinite(vv) or vv >= gb:
                raise ValueError("bad")
        except NameError:
            pass
        except Exception:
            fv = obj_val(np.array(glist))
            if not np.isfinite(fv):
                glist = [float(x) for x in np.ones(nP) * SP / nP]
    print("\n[done] obj=%.6f n=%d time=%.0fs" % (gb, len(glist), time.monotonic() - t0))
    return glist
