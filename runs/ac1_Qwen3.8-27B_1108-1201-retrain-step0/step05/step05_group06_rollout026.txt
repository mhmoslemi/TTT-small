import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def solve_lp(f, M, n):
    """
    Solve: maximize sum(b) subject to (f*b)[k] <= M, b >= 0
    The constraint (f*b)[k] = sum_{i: 0<=i<n, 0<=k-i<n} f[i] * b[k-i] <= M
    """
    # c = -1 for all variables (we minimize -sum(b))
    c_obj = -np.ones(n)
    
    # A_ub * b <= b_ub
    # For each k in [0, 2n-2]:
    #   sum_{i=max(0, k-n+1)}^{min(k, n-1)} f[i] * b[k-i] <= M
    #
    # Let j = k-i, so i = k-j. The constraint is:
    #   sum_{j=max(0, k-n+1)}^{min(k, n-1)} f[k-j] * b[j] <= M
    #
    # So the row for constraint k has:
    #   A[k, j] = f[k-j] if max(0, k-n+1) <= j <= min(k, n-1), else 0
    
    # Build the constraint matrix
    # Each row k has nonzeros at positions j where 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(k, n-1)
    
    # For efficiency, build row by row
    A_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        # For j in [j_start, j_end], A[k, j] = f[k - j]
        row[j_start:j_end+1] = f[k-j_end:k-j_start+1][::-1]
        A_rows.append(row)
    
    A_ub = np.array(A_rows)
    b_ub = np.full(2*n-1, M)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return np.zeros(n)

Minimize 2*n*max_conv/sum^2 using a hybrid LP-based and FISTA approach.
The LP-based direction finding directly targets the non-convex structure,

"""

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_seq(a, S):
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def lse_gradient(y, c, beta):

def solve_direction_lp(f, M, n):
    """
    Solve LP: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g (unnormalized).
    """

    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0, for k in [0, 2n-2]
    
    # Count nonzeros
    # For row k, nonzeros are at j where max(0, k-n+1) <= j <= min(k, n-1)
    # Number of nonzeros in row k = min(k, n-1) - max(0, k-n+1) + 1
    
    nnz_per_row = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        nnz_per_row.append(j_end - j_start + 1)
    
    total_nnz = sum(nnz_per_row)
    
    # Build sparse matrix
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    
    idx = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
