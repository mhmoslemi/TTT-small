def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

def lse_grad(a, beta):
    """Compute LSE value, gradient, and weights for the convolution."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    w = e / tot
    lse_val = cm + np.log(tot) / beta
    
    # Gradient: dLSE/da_j = 2 * sum_k w_k * a_{k-j} (with bounds)
    # c_k = sum_j a_j * a_{k-j}, so dc_k/da_j = a_{k-j} (if valid)
    # dLSE/da_j = sum_k w_k * dc_k/da_j = 2 * sum_k w_k * a_{k-j} (by symmetry)
    # Actually: c_k = sum_{j: valid} a_j * a_{k-j}
    # dc_k/da_j = a_{k-j} if 0 <= k-j < n, else 0
    # dLSE/da_j = sum_k w_k * a_{k-j} (for valid k-j)

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return lse_val, grad, c, w

                 beta_max=50000.0, eta_init=1.0, eta_decay=0.999):
    """Exponential gradient (mirror descent) with LSE."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    eta = eta_init
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        lse_val, grad, c, w = lse_grad(a, beta)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it
        
        # Exponential gradient update
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Scale gradient to have unit norm for stable updates
        grad_normed = grad / gnorm
        
        # Update: a_j <- a_j * exp(-eta * grad_j)
        # Use log to avoid underflow
        log_a = np.log(np.maximum(a, 1e-300))
        log_a -= eta * grad_normed
        # Renormalize
        log_a -= log_a.max()  # numerical stability
        a_new = np.exp(log_a)
        a_new = a_new * (S / a_new.sum())

            eta = eta_init * 0.5

    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def simplex_project(v, S):
    n = len(v)
    v = np.array(v, copy=True, dtype=np.float64)
    w = np.sort(v)[::-1]

    rho = np.arange(1, n+1, dtype=np.float64)

        theta = theta[i]

"""
Minimize 2*n*max_conv/sum^2 using a multi-phase approach combining:
1. Exponential gradient (mirror descent) with LSE smoothing - primary optimizer
2. FISTA with backtracking - secondary optimizer for stuck phases
3. Pairwise coordinate descent (mass transfer) - fine polish
4. Upscaling to larger n for more degrees of freedom

Key novelty: exponential gradient makes multiplicative updates, naturally
maintaining positivity and often converging faster than Euclidean-projected
methods for non-negative variables.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def simplex_project(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    rho = np.arange(1, n + 1, dtype=np.float64)
    theta = (cs - S) / rho
    valid = theta <= w
    if np.any(valid):
        i = int(np.max(np.where(valid)[0]))
        theta = theta[i]
    else:
        theta = theta[-1]
    return np.maximum(v - theta, 0.0)


def obj_val(a):
    """Compute objective 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_conv_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_grad(a, beta):
    """Compute LSE(beta*conv) value and gradient w.r.t. a."""
    n = len(a)
    c = fft_conv_self(a)
    cm = c.max()
    x = beta * (c - cm)
    x -= x.max()
    e = np.exp(x)
    Z = e.sum()
    w = e / Z
    val = cm + np.log(Z) / beta
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return val, grad, c, w


def run_exp_grad(a_init, max_time, rng, beta_init=80.0, beta_rate=0.003,
                 beta_max=80000.0, eta_init=None, eta_decay=0.9997):
    """Exponential gradient (mirror descent) with LSE smoothing."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    if eta_init is None:
        eta_init = 0.8 / (2.0 * S)

    start = time.monotonic()
    c = fft_conv_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_imp = 0
    eta = eta_init

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        _, grad, c, w = lse_grad(a, beta)
        cur_obj = 2.0 * n * c.max() / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_imp = it

        # Exponential gradient update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a -= eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new * (S / a_new.sum())
        a_new = np.clip(a_new, 0, 1000)

        if not np.all(np.isfinite(a_new)):
            break

        a = a_new
        eta *= eta_decay

        stuck = it - last_imp
        if stuck > 1000:
            # Periodic noise injection
            if stuck % 500 == 0:
                ns = max(1e-5, 0.005 * S / np.sqrt(n) * np.exp(-stuck / 8000.0))
                a = a + rng.normal(0, ns, n)
                a = np.maximum(a, 1e-15)
                a = a * (S / a.sum())
                eta = eta_init * 0.5
        elif stuck > 200:
            # Reduce eta when struggling
            eta = max(eta * 0.95, eta_init * 0.01)
        else:
            # Slightly increase eta when making progress
            eta = min(eta * 1.001, eta_init * 3.0)

        if stuck > 30000:
            break

    return best_a, best_obj


def run_fista_backtrack(a_init, max_time, rng, beta=200.0):
    """FISTA with backtracking line search for LSE objective."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    y = a.copy()
    t = 1.0
    step = 1.0 / (beta * 2.0 * S + 1.0)

    start = time.monotonic()
    c = fft_conv_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    it = 0
    last_imp = 0

    while time.monotonic() - start < max_time:
        it += 1
        _, grad, c, w = lse_grad(y, beta)
        cur_obj = 2.0 * n * c.max() / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            last_imp = it

        # Backtracking line search
        step_bt = step
        x_new = simplex_project(y - step_bt * grad, S)
        obj_new = 2.0 * n * fft_conv_self(x_new).max() / (S * S)
        bt = 0
        while obj_new > cur_obj + 1e-13 and step_bt > 1e-16 and bt < 8:
            step_bt *= 0.5
            x_new = simplex_project(y - step_bt * grad, S)
            obj_new = 2.0 * n * fft_conv_self(x_new).max() / (S * S)
            bt += 1

        if bt > 3:
            step = step_bt * 0.9
        else:
            step = min(step * 1.02, 4.0 / (beta * 2.0 * S + 1.0))

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t = t_new

        stuck = it - last_imp
        if stuck > 3000:
            t = 1.0
            ns = max(1e-5, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 10000.0))
            y = simplex_project(y + rng.normal(0, ns, n), S)

        if stuck > 40000:
            break

    return best_a, best_obj


def pair_cd(a_init, max_time, rng, top_k=5, pairs_per_iter=10):
    """Coordinate descent via pairwise mass transfers with exact max tracking."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    c = fft_conv_self(a)
    start = time.monotonic()
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    no_imp = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 80000:
            break

        # Contribution vector from top-k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        contrib = np.zeros(n)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                contrib[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]

        # Source: high contribution * high value
        # Target: low contribution
        src = np.argsort(contrib * a)[::-1][:pairs_per_iter]
        tgt = np.argsort(contrib)[:pairs_per_iter]

        made_move = False
        for j in src:
            if a[j] < 1e-12:
                continue
            for l in tgt:
                if j == l:
                    continue
                for frac in [0.03, 0.08, 0.15]:
                    delta = a[j] * frac
                    if delta < 1e-14:
                        continue

                    # Estimate new max checking only top lags + neighbors
                    new_max = c.max()
                    for k in top_idx:
                        k = int(k)
                        dc = 0.0
                        if 0 <= k - j < n:
                            dc -= 2.0 * delta * a[k - j]
                        if 0 <= k - l < n:
                            dc += 2.0 * delta * a[k - l]
                        # Cross correction
                        if 0 <= k - j < n and (k - j) == l:
                            dc += delta * (a[l] - a[j])
                        if 0 <= k - l < n and (k - l) == j:
                            dc += delta * (a[j] - a[l])
                        est = c[k] + dc
                        if est > new_max:
                            new_max = est

                    new_obj = 2.0 * n * new_max / (S * S)
                    if new_obj < best_obj - 1e-15:
                        a[j] -= delta
                        a[l] += delta
                        made_move = True
                        break
                if made_move:
                    break
            if made_move:
                break

        if not made_move:
            # Random perturbation when stuck
            if no_imp > 2000 and it % 100 == 0:
                idx = rng.integers(0, n)
                delta = a[idx] * 0.02
                if delta > 1e-13:
                    a[idx] -= delta
                    a[rng.integers(0, n)] += delta
                    c = fft_conv_self(a)
            elif it % 500 == 0:
                c = fft_conv_self(a)
        else:
            if it % 200 == 0:
                c = fft_conv_self(a)

    return best_a, best_obj


def upsample_seq(a, factor=2):
    """Upsample via linear interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def make_init(n, method, rng):
    """Create initial sequences of various types."""
    S = np.sqrt(2.0 * n)
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "cosine":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    elif method == "gaussian":
        a = np.exp(-4.0 * (tt - 0.5) ** 2 / (0.12 ** 2))
    elif method == "chirp":
        a = 0.4 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt + 1.0)
        a = np.abs(a)
    elif method == "random":
        a = rng.random(n) + 0.05
    elif method == "bump":
        a = np.maximum(0.01, 1.0 - 1.5 * np.abs(tt - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: exp-grad, FISTA, coordinate descent, upsample, polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 40.0)

    # Get starting point
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a = np.clip(a, 0, 1000)
        if a.sum() < 0.01:
            raise ValueError("sum too small")
    except Exception:
        a = make_init(8192, "cosine", rng)

    n0 = len(a)
    S0 = np.sqrt(2.0 * n0)
    a = a * (S0 / a.sum())
    obj0 = obj_val(a)
    print(f"Start: n={n0}, obj={obj0:.8f}")

    best_a = a.copy()
    best_obj = obj0

    # === Phase 1: Quick diverse screening (10% time) ===
    remaining = deadline - time.monotonic()
    p1_time = remaining * 0.10
    if p1_time > 30:
        print(f"\n--- Phase 1: Diverse screening ({p1_time:.0f}s) ---")
        methods = ["chirp", "gaussian", "random"]
        per_t = p1_time / len(methods)
        for m in methods:
            if time.monotonic() >= deadline:
                break
            a_div = make_init(n0, m, rng)
            a_opt, obj_opt = run_exp_grad(a_div, per_t * 0.7, rng,
                                           beta_init=30, beta_rate=0.01,
                                           beta_max=5000)
            print(f"  {m}: obj={obj_opt:.8f}")
            if obj_opt < best_obj:
                best_a, best_obj = a_opt, obj_opt
                print(f"    *** NEW BEST ***")

    # === Phase 2: Exponential gradient main optimization (25% time) ===
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.25
    print(f"\n--- Phase 2: Exp-gradient optimization ({p2_time:.0f}s) ---")
    for beta_i in [80, 300, 1000]:
        rem = deadline - time.monotonic()
        if rem < 10:
            break
        t_sub = min(p2_time / 3.0, rem * 0.95)
        a_opt, obj_opt = run_exp_grad(best_a, t_sub, rng,
                                       beta_init=float(beta_i),
                                       beta_rate=0.002,
                                       beta_max=100000.0)
        print(f"  beta_init={beta_i}: obj={obj_opt:.8f}")
        if obj_opt < best_obj:
            best_a, best_obj = a_opt, obj_opt

    # === Phase 3: FISTA with backtracking (20% time) ===
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.20
    print(f"\n--- Phase 3: FISTA backtrack ({p3_time:.0f}s) ---")
    for beta_f in [500, 2000]:
        rem = deadline - time.monotonic()
        if rem < 10:
            break
        t_sub = min(p3_time / 2.0, rem * 0.95)
        a_opt, obj_opt = run_fista_backtrack(best_a, t_sub, rng, beta=beta_f)
        print(f"  FISTA beta={beta_f}: obj={obj_opt:.8f}")
        if obj_opt < best_obj:
            best_a, best_obj = a_opt, obj_opt

    # === Phase 4: Coordinate descent polish (20% time) ===
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.20
    print(f"\n--- Phase 4: Coordinate descent ({p4_time:.0f}s) ---")
    a_cd, obj_cd = pair_cd(best_a, p4_time, rng, top_k=5, pairs_per_iter=10)
    print(f"  CD: obj={obj_cd:.8f}")
    if obj_cd < best_obj:
        best_a, best_obj = a_cd, obj_cd

    # Follow-up FISTA after CD
    remaining = deadline - time.monotonic()
    if remaining > 15:
        a_opt, obj_opt = run_fista_backtrack(best_a, remaining * 0.4, rng, beta=3000)
        print(f"  FISTA post-CD: obj={obj_opt:.8f}")
        if obj_opt < best_obj:
            best_a, best_obj = a_opt, obj_opt

    # === Phase 5: Upsample to larger n (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and len(best_a) <= 8192:
        n2 = len(best_a) * 2
        print(f"\n--- Phase 5: Upsample to n={n2} ({remaining:.0f}s) ---")
        a_up = upsample_seq(best_a, 2)
        S2 = np.sqrt(2.0 * n2)
        a_up = a_up * (S2 / a_up.sum())
        obj_up = obj_val(a_up)
        print(f"  Upsampled init: obj={obj_up:.8f}")
        if obj_up < best_obj:
            best_a, best_obj = a_up, obj_obj if False else a_opt, obj_opt
            best_a, best_obj = a_up, obj_up

        rem = deadline - time.monotonic()
        if rem > 20:
            a_opt, obj_opt = run_exp_grad(best_a, rem * 0.4, rng,
                                           beta_init=50, beta_rate=0.005,
                                           beta_max=30000)
            print(f"  Exp-grad upsampled: obj={obj_opt:.8f}")
            if obj_opt < best_obj:
                best_a, best_obj = a_opt, obj_opt

        rem = deadline - time.monotonic()
        if rem > 15:
            a_cd2, obj_cd2 = pair_cd(best_a, rem * 0.6, rng, top_k=4,
                                      pairs_per_iter=8)
            print(f"  CD upsampled: obj={obj_cd2:.8f}")
            if obj_cd2 < best_obj:
                best_a, best_obj = a_cd2, obj_cd2

    # === Phase 6: Final high-precision polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"\n--- Phase 6: Final polish ({remaining:.0f}s) ---")
        n_f = len(best_a)
        a_opt, obj_opt = run_exp_grad(best_a, remaining * 0.5, rng,
                                       beta_init=5000, beta_rate=0.001,
                                       beta_max=200000.0)
        print(f"  Final exp-grad: obj={obj_opt:.8f}")
        if obj_opt < best_obj:
            best_a, best_obj = a_opt, obj_opt

        rem = deadline - time.monotonic()
        if rem > 5:
            a_cd3, obj_cd3 = pair_cd(best_a, rem, rng, top_k=3,
                                      pairs_per_iter=6)
            print(f"  Final CD: obj={obj_cd3:.8f}")
            if obj_cd3 < best_obj:
                best_a, best_obj = a_cd3, obj_cd3

    # Final verification
    final = [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
    ver = evaluate_sequence(final)
    print(f"\nFinal verified: obj={ver:.8f}, n={len(final)}")

    if not np.isfinite(ver):
        print("WARNING: invalid result, using fallback")
        n_fb = 8192
        final = [float(np.sqrt(2.0 * n_fb) / n_fb)] * n_fb

    return final
