Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses multiplicative mirror descent, high-precision L-BFGS with escalating beta,
subgradient polishing, and multi-length exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute softmax weights for LSE gradient."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:

    return e / tot


def conv_gradient(y, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. y, where b=conv(y,y)."""
    n = len(y)
    y_rev = y[::-1]
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    conv_wy = np.real(ifft(fft(w, L) * fft(y_rev, L)))
    cross = conv_wy[n - 1: 2 * n - 1]
    return 2.0 * cross


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    return conv_gradient(y, w_full)

                grad = exact_subgradient(y, c, tol_frac=tol)
            else:
                w = lse_weights(c, beta)
                grad = conv_gradient(y, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Noise injection when stuck
        if stuck_count > 3000 and it % 100 == 0:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stuck_count / 10000.0)
            if noise_scale > 1e-8:
                noise = rng.normal(0, noise_scale, n)
                F = fft(noise)
                cutoff = max(5, n // 100)
                F[cutoff:-cutoff] *= 0.1
                noise_smooth = np.real(ifft(F))[:n]
                y = project_simplex(y + noise_smooth, S)
                t_mom = 1.0
                stuck_count = 0

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a


def run_mult_descent(a_init, max_time, rng, beta_start=50.0, beta_rate=0.01,
                     eta_start=0.1, eta_decay=0.9995):
    """
    Multiplicative mirror descent (natural gradient) for the LSE-smoothed objective.
    Update: a_i <- a_i * exp(-eta * g_i / S), then normalize.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    eta = eta_start

    while time.monotonic() - start < max_time:
        it += 1
        beta = beta_start + beta_rate * it
        beta = min(beta, 20000.0)

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_conv - 1e-15:
            best_conv = cur_max
            best_a = a.copy()
            last_improve = it

        stuck = it - last_improve

        # Compute weights
        if stuck > 2000:
            # Use exact subgradient with tolerance
            tol = min(0.02, 0.001 * (stuck - 2000) / 1000)
            grad = exact_subgradient(a, c, tol_frac=tol)
        else:
            w = lse_weights(c, beta)
            grad = conv_gradient(a, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Multiplicative update: a_i *= exp(-eta * g_i / S)
        log_update = -eta * grad / S
        log_update = np.clip(log_update, -5.0, 5.0)  # prevent overflow
        a = a * np.exp(log_update)
        a = np.maximum(a, 0.0)

        # Normalize
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s)

        eta *= eta_decay

        # Periodic restart with smoothing
        if stuck > 5000 and it % 200 == 0:
            # Mix with smoothed version
            F = fft(a)
            cutoff = max(3, n // 50)
            F[cutoff:-cutoff] *= 0.3
            smooth = np.real(ifft(F))[:n]
            smooth = np.maximum(smooth, 0.0)
            ss = smooth.sum()
            if ss > 1e-10:
                smooth = smooth * (S / ss)
                a = 0.7 * a + 0.3 * smooth
                a = a * (S / a.sum())

    return best_a


def run_coord_descent(a_init, max_time, rng, n_sweeps=50):
    """
    Coordinate descent: for each coordinate, find optimal value
    that minimizes the max of affected convolution lags.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best_conv = c.max()
    best_a = a.copy()

    sweep = 0
    while time.monotonic() - start < max_time and sweep < n_sweeps:
        sweep += 1
        improved = False

        # Random order for diversity
        order = rng.permutation(n)

        for i in order:
            # Compute base convolution (without a_i's contribution)
            # (a⊛a)_k = sum_j a_j * a_{k-j}
            # Terms involving a_i: for each k, a_i appears as a_i * a_{k-i} (j=i)
            # and a_{k-i} * a_i (j=k-i). So contribution is:
            #   if k-i in [0,n-1] and k-i != i: 2 * a_i * a_{k-i}
            #   if k-i == i (k=2i): a_i^2
            base = c - 2.0 * a[i] * np.roll(a, 0)  # placeholder, will compute properly

            # Actually, let's just do a simple line search
            # Current value of a[i]
            ai = a[i]

            # Compute max_conv as function of a[i] = x (keeping others fixed, then projecting)
            # For efficiency, compute affected lags
            # Lags affected: k in [i, i+n-1] intersect [0, 2n-2]
            k_start = max(0, i)
            k_end = min(2 * n - 2, i + n - 1)

            # base_k = c_k - (contribution from a_i)
            # contribution: for k in [i, i+n-1]:
            #   if k == 2*i: a[i]^2
            #   else: 2 * a[i] * a[k-i] (where a[k-i] is the other factor)

            # Let's just try a few candidate values and pick the best
            # Candidates: 0, current, and a few fractions
            if ai < 1e-15:
                candidates = [0.0, S / n * 0.5, S / n]
            else:
                candidates = [0.0, ai * 0.5, ai * 0.8, ai, ai * 1.2, ai * 1.5, S / n * 2]
                candidates = [x for x in candidates if 0 <= x <= 1000]

            best_x = ai
            best_max = c.max()

            for x in candidates:
                if abs(x - ai) < 1e-15 * max(1, ai):
                    continue
                a[i] = x
                # Recompute affected lags
                # For speed, just recompute the full convolution if n is small enough
                # Otherwise, update only affected lags
                if n <= 16384:
                    c_new = fft_convolve_self(a)
                    mx = c_new.max()
                    if mx < best_max:
                        best_max = mx
                        best_x = x
                else:
                    # Update only affected lags
                    k_lo = max(0, 2 * i - (n - 1))
                    k_hi = min(2 * n - 2, 2 * i + (n - 1))
                    # This is complex; just do full recompute for correctness
                    c_new = fft_convolve_self(a)
                    mx = c_new.max()
                    if mx < best_max:
                        best_max = mx
                        best_x = x

                a[i] = x

            a[i] = best_x

            # If we improved, update c
            if best_x != ai:
                c = fft_convolve_self(a)
                if c.max() < best_conv - 1e-15:
                    best_conv = c.max()
                    best_a = a.copy()
                    improved = True

                # Check time
                if time.monotonic() - start > max_time:
                    break

        # Project back onto simplex at end of sweep
        a = project_simplex(a, S)
        c = fft_convolve_self(a)
        if c.max() < best_conv - 1e-15:
            best_conv = c.max()
            best_a = a.copy()
            improved = True

        if not improved and sweep > 10:
            break

    return best_a


def run_lbfgs_polish(a_init, max_time, rng, beta_levels=None, n_iter_max=200):
    """
    L-BFGS-B polishing with escalating beta for LSE approximation.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())

    if beta_levels is None:
        beta_levels = [100.0, 300.0, 1000.0, 3000.0, 10000.0, 30000.0]

    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    for beta in beta_levels:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break

        time_budget_this = remaining * 0.7

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            b = fft_convolve_self(x)
            cm_local = b.max()
            shifted = beta * (b - cm_local)
            shifted = shifted - shifted.max()
            e = np.exp(shifted)
            e_sum = e.sum()
            w = e / e_sum
            obj = cm_local + np.log(e_sum) / beta
            # Add penalty for sum constraint
            sum_x = x.sum()
            pen = 10.0
            obj += pen * (sum_x - S) ** 2
            # Gradient
            grad = conv_gradient(x, w)
            grad += 2.0 * pen * (sum_x - S)
            return obj, grad

        t0 = time.monotonic()
        try:
            result = scipy_minimize(
                obj_and_grad, a.copy(), method='L-BFGS-B',
                jac=True, bounds=[(0, 1000)] * n,
                options={'maxiter': n_iter_max, 'maxls': 20,
                         'ftol': 1e-16, 'gtol': 1e-12, 'iprint': -1}
            )
        except Exception:
            break

        elapsed = time.monotonic() - t0
        a_new = np.clip(result.x, 0, 1000)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new * (S / s_new)
        obj_new = compute_obj(a_new)

        print("    L-BFGS beta=%.0f: obj=%.9f (t=%.1fs, iters=%d)" %
              (beta, obj_new, elapsed, result.nit))

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def run_subgrad_polish(a_init, max_time, rng):
    """
    Final polishing with exact subgradient and very small adaptive steps.
    Uses Nesterov momentum with periodic resets.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    a_prev = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step_base = S / (np.linalg.norm(a) * 2.0)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-16:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive tolerance for active set
        if stuck < 100:
            tol = 0.0
        elif stuck < 500:
            tol = 0.001
        elif stuck < 2000:
            tol = 0.005
        else:
            tol = min(0.02, 0.01 + 0.001 * (stuck - 2000) / 1000)

        grad = exact_subgradient(y, c, tol_frac=tol)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break

        # Decreasing step size
        step = step_base / (1.0 + 0.01 * it)
        step = min(step, step_base * 0.5)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 5000 and it % 500 == 0:
            # Small perturbation
            noise = rng.normal(0, 1e-5 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
            last_improve = it

        if it % 20000 == 0:
            t_mom = 1.0

    return best_a


def generate_inits(n, rng, S):
    """Generate diverse initial sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Cosine harmonics
    for freq in [1, 2, 3, 5]:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t)
        a = np.maximum(a, 0.001)
        a = a * (S / a.sum())
        inits.append(a.copy())

    # Gaussian-like
    center = n // 2
    sigma = n / 10
    a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    a = a * (S / a.sum())
    inits.append(a.copy())

    # Two peaks
    a = np.exp(-0.5 * ((np.arange(n) - n * 0.25) / (n / 15)) ** 2) + \
        np.exp(-0.5 * ((np.arange(n) - n * 0.75) / (n / 15)) ** 2)
    a = a * (S / a.sum())
    inits.append(a.copy())

    # Smooth random
    raw = rng.random(n)
    F = fft(raw)
    cutoff = max(3, n // 50)
    F[cutoff:-cutoff] *= 0.01
    smooth = np.real(ifft(F))[:n]
    smooth = np.maximum(smooth, 0.01)
    smooth = smooth * (S / smooth.sum())
    inits.append(smooth.copy())

    # Uniform
    inits.append(np.ones(n) * S / n)

    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA warmup, multiplicative descent, coordinate descent,
    high-precision L-BFGS, subgradient polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 60.0)

    global_best_obj = np.inf
    global_best_a = None

    # Load initial sequence if available
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is not None:
        obj0 = compute_obj(a_start)
        print("Init from height_sequence_1: n=%d, obj=%.9f" % (len(a_start), obj0))
        global_best_obj = obj0
        global_best_a = a_start.copy()
    else:
        n0 = 8192
        a_start = np.ones(n0) * np.sqrt(2.0 * n0) / n0
        global_best_obj = compute_obj(a_start)
        global_best_a = a_start.copy()
        print("Fallback init: n=%d, obj=%.9f" % (n0, global_best_obj))

    n0 = len(global_best_a)

    # === Phase 1: FISTA warm-up (20% time) ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.20
    print("\n=== Phase 1: FISTA warm-up (n=%d, budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    a1 = run_fista(global_best_a, p1_budget * 0.5, rng,
                   beta_start=20.0, mode="lse", beta_rate=0.005, step_scale=1.0)
    obj1 = compute_obj(a1)
    print("  FISTA-lse: obj=%.9f (t=%.0fs)" % (obj1, time.monotonic() - t1))
    if obj1 < global_best_obj:
        global_best_obj = obj1
        global_best_a = a1.copy()

    rem1 = p1_budget - (time.monotonic() - t1)
    if rem1 > 20:
        a1b = run_fista(global_best_a, rem1, rng,
                        beta_start=50.0, mode="hybrid", beta_rate=0.003, step_scale=0.7)
        obj1b = compute_obj(a1b)
        print("  FISTA-hybrid: obj=%.9f" % obj1b)
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: Multiplicative descent (20% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_budget = remaining * 0.20
        print("\n=== Phase 2: Multiplicative descent (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()

        a2 = run_mult_descent(global_best_a, p2_budget * 0.6, rng,
                              beta_start=100.0, beta_rate=0.05,
                              eta_start=0.05, eta_decay=0.999)
        obj2 = compute_obj(a2)
        print("  Multi-descent: obj=%.9f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

        rem2 = p2_budget - (time.monotonic() - t2)
        if rem2 > 20:
            a2b = run_mult_descent(global_best_a, rem2, rng,
                                   beta_start=200.0, beta_rate=0.1,
                                   eta_start=0.02, eta_decay=0.998)
            obj2b = compute_obj(a2b)
            print("  Multi-descent2: obj=%.9f" % obj2b)
            if obj2b < global_best_obj:
                global_best_obj = obj2b
                global_best_a = a2b.copy()

    # === Phase 3: Coordinate descent (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:

        t3 = time.monotonic()

            obj3b = compute_obj(a3b)

                global_best_obj = obj3b

    # === Phase 4: High-precision L-BFGS (30% time) ===
    remaining = deadline - time.monotonic()

        p4_budget = remaining * 0.30

        a4, obj4 = run_lbfgs_polish(global_best_a, p4_budget, rng,

        print("  L-BFGS done: obj=%.9f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

    # === Phase 5: Subgradient final polish (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:

        print("\n=== Phase 5: Subgradient polish (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()

        a5 = run_subgrad_polish(global_best_a, p5_budget, rng)
        obj5 = compute_obj(a5)
        print("  Subgrad polish: obj=%.9f (t=%.0fs)" % (obj5, time.monotonic() - t5))
        if obj5 < global_best_obj:
            global_best_obj = obj5
            global_best_a = a5.copy()

    # === Phase 6: Final L-BFGS squeeze ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 6: Final L-BFGS squeeze (budget %.0fs) ===" % remaining)
        t6 = time.monotonic()
        a6, obj6 = run_lbfgs_polish(global_best_a, remaining * 0.85, rng,
                                     beta_levels=[500.0, 5000.0, 20000.0],
                                     n_iter_max=200)
        print("  Final squeeze: obj=%.9f (t=%.0fs)" % (obj6, time.monotonic() - t6))
        if obj6 < global_best_obj:
            global_best_obj = obj6
            global_best_a = a6.copy()

    # === Phase 7: Try alternative n (if time remains) ===
    remaining = deadline - time.monotonic()
    if remaining > 120 and n0 != 16384:
        n_alt = 16384
        S_alt = np.sqrt(2.0 * n_alt)
        # Resample best sequence to new length
        old_n = len(global_best_a)
        old_x = np.linspace(0, 1, old_n, endpoint=False)
        new_x = np.linspace(0, 1, n_alt, endpoint=False)
        a_alt = np.interp(new_x, old_x, global_best_a)
        a_alt = np.maximum(a_alt, 0.0)
        a_alt = a_alt * (S_alt / a_alt.sum())

        print("\n=== Phase 7: Alt length n=%d (budget %.0fs) ===" % (n_alt, remaining * 0.6))
        t7 = time.monotonic()

        a7 = run_fista(a_alt, remaining * 0.3, rng,
                       beta_start=50.0, mode="lse", beta_rate=0.005, step_scale=1.0)
        obj7 = compute_obj(a7)
        print("  Alt-n FISTA: obj=%.9f (t=%.0fs)" % (obj7, time.monotonic() - t7))
        if obj7 < global_best_obj:

            global_best_a = a7.copy()

        rem7 = deadline - time.monotonic()
        if rem7 > 30:
            a7b, obj7b = run_lbfgs_polish(global_best_a, rem7 * 0.8, rng,
                                           beta_levels=[100.0, 500.0, 2000.0],
                                           n_iter_max=150)
            if obj7b < global_best_obj:
                global_best_obj = obj7b
                global_best_a = a7b.copy()
                print("  Alt-n L-BFGS: obj=%.9f" % obj7b)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):

            if ver < global_best_obj:

        final_list = [float(x) for x in global_best_a]

    print("\nBest objective found: %.9f" % global_best_obj)
    return final_list
