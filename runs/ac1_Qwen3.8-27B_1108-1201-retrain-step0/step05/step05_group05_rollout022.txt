Minimize 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B with LSE smoothing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

# Helper functions (all top-level)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def lse_objective_and_grad(x, beta, S_penalty=1.0):
    """
    x: flattened vector (the sequence a)
    Returns (objective, gradient)
    """
    a = np.maximum(x, 0)  # Enforce non-negativity (L-BFGS-B handles bounds, but this is safer)
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    
    c = fft_convolve_self(a)
    cmax = c.max()
    scaled = beta * (c - cmax)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 0)
    e = np.exp(scaled)
    lse_val = cmax + np.log(e.sum()) / beta
    
    # Gradient of LSE w.r.t. c
    w = e / e.sum()
    
    # Gradient of c = a*a w.r.t. a: grad_a[k] = 2 * (w * a)[k] (truncated to n)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    # Scale invariance correction: the objective is scale-invariant,
    # so we add a penalty to fix the scale
    # Actually, since L-BFGS-B handles bounds and the objective is scale-invariant,
    # the gradient along the scaling direction is 0 (in exact arithmetic).
    # We add a small quadratic penalty to help convergence.
    penalty_grad = 2 * S_penalty * (a.sum() - np.sqrt(2*n)) / n  # uniform distribution of penalty
    
    obj = lse_val + S_penalty * (a.sum() - np.sqrt(2*n))**2 / n
    grad = grad_a + penalty_grad
    
    return obj, grad

    """Run L-BFGS-B optimization."""
    start = time.monotonic()
    x0 = a_init.copy()

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B with LSE smoothing,
basin hopping with Fourier mutations, and active-set KKT polishing.

Key improvements over previous approach:
1. L-BFGS-B replaces FISTA for much faster local convergence
2. Smooth beta schedule from low to very high
3. Fourier-domain basin hopping for global exploration
4. Active-set subgradient method for final precision polish
5. Multi-length exploration (4096, 8192, 16384)

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_objective(a):
    """Compute the objective value directly."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = len(a)
    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (s * s))


def lse_func_and_grad(x, beta, target_sum):
    """
    LSE-smoothed max of self-convolution with scale penalty.
    x: the sequence (non-negative enforced by L-BFGS bounds)
    Returns (function_value, gradient).
    """
    a = x
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 1e10, np.zeros(n)

    c = fft_convolve_self(a)
    cmax = c.max()
    shifted = beta * (c - cmax)
    shifted = np.clip(shifted, -700.0, 0.0)
    e = np.exp(shifted)
    esum = e.sum()
    if esum < 1e-300:
        lse_val = cmax
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        lse_val = cmax + np.log(esum) / beta
        w = e / esum

    # Gradient: 2 * (w cross-correlate a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    # Small scale penalty to break homogeneity and help L-BFGS
    pen_coeff = 0.005
    pen = pen_coeff * (s - target_sum) ** 2
    pen_grad = 2.0 * pen_coeff * (s - target_sum) * np.ones(n) * 0.01

    return lse_val + pen, grad_a + pen_grad


def run_lbfgs(a_init, S, max_time, beta, max_iter=300):
    """
    Run L-BFGS-B on the LSE-smoothed objective for a given beta.
    Returns (optimized_a, max_b_value, elapsed_time).
    """
    start = time.monotonic()
    n = len(a_init)

    def obj_wrapper(x):
        val, grad = lse_func_and_grad(x, beta, S)
        return val, grad

    result = minimize(
        obj_wrapper,
        a_init.copy(),
        method='L-BFGS-B',
        bounds=[(0.0, 1000.0)] * n,
        options={
            'maxiter': max_iter,
            'ftol': 1e-15,
            'gtol': 1e-9,
            'maxfun': 10000,
            'maxls': 40,
            'factr': 1e12,
            'eps': 1e-7,
        }
    )

    a_opt = np.maximum(result.x, 0.0)
    c = fft_convolve_self(a_opt)
    elapsed = time.monotonic() - start
    return a_opt, float(c.max()), elapsed

def fourier_mutate(a, rng, scale=0.1):
    """
    Fourier-domain phase mutation: perturb DFT phases, take IFFT.
    Preserves the magnitude spectrum structure.
    """
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)

    # Frequency-dependent noise (more at high freqs)
    freqs = np.arange(fsize, dtype=np.float64)
    weight = 1.0 + 3.0 * freqs / fsize
    phase_noise = rng.normal(0, scale * weight, fsize)
    new_A = mag * np.exp(1j * (phase + phase_noise))
    new_A[0] = A[0]  # Preserve DC

    a_new = np.real(ifft(new_A, fsize))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new * (S / s)


def fourier_crossover(a1, a2, rng):
    """Fourier-domain crossover between two sequences."""
    n = len(a1)
    S = a1.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    A1 = fft(a1, fsize)
    A2 = fft(a2, fsize)

    r = rng.random(fsize)
    alpha = 0.3 + 0.4 * r
    new_mag = alpha * np.abs(A1) + (1.0 - alpha) * np.abs(A2)
    new_phase = np.where(r < 0.5, np.angle(A1), np.angle(A2))
    new_A = new_mag * np.exp(1j * new_phase)
    new_A[0] = A1[0]

    a_new = np.real(ifft(new_A, fsize))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a1.copy()
    return a_new * (S / s)


def run_basin_hopping(a_init, S, max_time, rng, n_local=100):

            a = a_opt.copy()

        # Update population
        pop.append((obj, a_opt.copy()))
        if len(pop) > 6:

def active_set_subgradient(a, c, tol=1e-5, max_active=30):
    """
    Compute the minimum-norm subgradient of max(c) using active set.
    Returns the combined subgradient direction.
    """
    n = len(a)
    cmax = c.max()
    active = np.where(c >= cmax - tol * max(cmax, 1e-10))[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    active = active[:max_active]

    # Compute subgradient for each active lag
    # grad_k[i] = 2 * a[k-i] for valid k-i
    fsize = 1 << (len(c) + n - 1).bit_length()
    A = fft(a, fsize)

    grads = np.zeros((len(active), n))
    for diag, k in enumerate(active):
        w_full = np.zeros(len(c))
        w_full[k] = 1.0
        W = fft(w_full, fsize)
        grads[diag] = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    # Find weights minimizing ||sum lambda_k * grad_k|| with lambda >= 0, sum = 1
    # Use a simple iterative method: start uniform, project gradient descent
    m = len(active)
    lam = np.ones(m) / m

    # Minimize ||G^T * lam||^2 subject to lam >= 0, sum(lam) = 1
    # where G is (n x m) matrix of subgradients
    # Gradient of objective w.r.t. lam: 2 * G * (G^T * lam)
    G transpose dot lam = G.T @ lam  shape (m,)
    for _ in range(50):
        obj_grad = 2.0 * (grads.T @ (grads @ lam))
        # Projected gradient step on simplex
        lr = 0.1 / (1.0 + np.abs(obj_grad).max())
        lam_new = lam - lr * obj_grad
        # Project onto simplex
        lam_new, _ = project_simplex_vec(lam_new, 1.0)
        if np.linalg.norm(lam_new - lam) < 1e-10:
            break
        lam = lam_new

    return grads @ lam


def project_simplex_vec(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    return result, None


def run_kkt_polish(a_init, S, max_time, rng):
    """
    Final polishing using active-set subgradient method.
    Very small steps for precision.
    """
    n = len(a_init)
    a = a_init.copy()
    start = time.monotonic()

    c = fft_convolve_self(a)
    best_obj = float(compute_objective(a))
    best_a = a.copy()
    best_maxb = c.max()

    it = 0
    while time.monotonic() - start < max_time - 2:
        it += 1
        progress = (time.monotonic() - start) / max_time

        c = fft_convolve_self(a)
        maxb = c.max()
        obj = float(compute_objective(a))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            best_maxb = maxb

        # Compute active-set subgradient
        tol = 1e-4 * (1.0 - 0.5 * progress) + 1e-6
        grad = active_set_subgradient(a, c, tol=tol, max_active=min(40, 2 * n // 10))
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 0.5 * S / (n * (1.0 + 0.005 * it) * gnorm / (S / n) + 1.0)
        step = min(step, 0.1 * S / n)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new * (S / s)

        # Accept if improvement or small step
        c_new = fft_convolve_self(a_new)
        if c_new.max() <= maxb + 1e-10 * maxb:
            a = a_new
        else:
            # Halve step
            a_new = a - 0.5 * step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new * (S / s)
                a = a_new

        # Periodic small perturbation
        if it % 200 == 0 and progress < 0.8:
            noise = 0.001 * S / np.sqrt(n) * (1.0 - progress)
            a = a + rng.normal(0, noise, n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-10:
                a = a * (S / s)

    return best_a, best_obj


def generate_inits(n, rng, given_seq=None):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    if given_seq is not None:
        gs = np.clip(given_seq, 0, 1000)
        if len(gs) == n and gs.sum() > 0.01:
            inits.append(gs.copy())

    # Various windows and shapes
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * tt) + 0.1 * np.cos(4 * np.pi * tt), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2 + 0.01)
    inits.append(np.maximum(1.0 + 0.4 * np.sin(6 * np.pi * tt) + 0.2 * np.cos(10 * np.pi * tt + 0.5), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.sin(4 * np.pi * tt + 1.0) + 0.25 * np.cos(9 * np.pi * tt), 0.01))
    inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * 3 * tt) * np.cos(2 * np.pi * 7 * tt), 0.01))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.abs(rng.normal(1.0, 0.5, n)) + 0.05)

    # Block structures
    for nb in [8, 16, 32]:
        block_len = max(1, n // nb)
        heights = rng.uniform(0.1, 2.0, nb)
        block = np.repeat(heights, block_len)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n] + 0.01)

    return inits


def fit_length(a, target_n):
    """Resample a sequence to a different length using interpolation."""
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 0:
        return np.ones(target_n)
    # Resample using linear interpolation
    old_x = np.linspace(0, 1, n)
    new_x = np.linspace(0, 1, target_n)
    return np.interp(new_x, old_x, a)

                best_obj = obj_ib
                best_a = a_ib
            if best_obj < global_best_obj:
                global_best_obj = best_obj
                global_best_a = best_a.copy()
            print("    After IB: obj=%.8f" % best_obj)

        # Phase 3: Basin hopping (40% of time)
        bh_time = min(800.0, time_alloc * 0.40)
        if time.monotonic() < deadline - 100 and bh_time > 30:
            print("  Phase 3: Basin hopping (%.0fs)" % bh_time)
            a_bh, obj_bh = run_basin_hopping(best_a, S, bh_time, rng,
                                             n_local=int(bh_time / 20))
            if obj_bh < best_obj:
                best_obj = obj_bh
                best_a = a_bh
            if best_obj < global_best_obj:
                global_best_obj = best_obj
                global_best_a = best_a.copy()
            print("    After BH: obj=%.8f" % best_obj)

        # Phase 4: KKT polish (remaining time, min 60s)
        polish_time = max(30.0, min(300.0, deadline - time.monotonic() - 20))
        if time.monotonic() < deadline - 30 and polish_time > 20:
            print("  Phase 4: KKT polish (%.0fs)" % polish_time)
            a_kkt, obj_kkt = run_kkt_polish(best_a, S, polish_time, rng)
            if obj_kkt < best_obj:
                best_obj = obj_kkt
                best_a = a_kkt
            if best_obj < global_best_obj:
                global_best_obj = best_obj
                global_best_a = best_a.copy()
            print("    After KKT: obj=%.8f" % best_obj)

    # Try primary length
    if 8192 in lengths_to_try:
        idx = lengths_to_try.index(8192)
        try_length(8192, budget * 0.6)
        lengths_to_try.remove(8192)

    # If time remains, try another length
    for n_try in lengths_to_try:
        remaining = deadline - time.monotonic() - 50
        if remaining > 200:
            try_length(n_try, remaining)
        else:
            break

    # Final verification
    if global_best_a is not None:
        global_best_a = np.clip(global_best_a, 0.0, 1000.0)
        if global_best_a.sum() < 0.01:
            global_best_a = np.ones(8192) * np.sqrt(2 * 8192) / 8192
        final_list = [float(x) for x in global_best_a]
        ver = evaluate_sequence(final_list)
        print("\n=== Verification: obj=%.8f (n=%d) ===" % (ver, len(final_list)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_best_a = np.ones(8192) * np.sqrt(2 * 8192) / 8192
            final_list = [float(x) for x in global_best_a]
    else:
        global_best_a = np.ones(8192) * np.sqrt(2 * 8192) / 8192
        final_list = [float(x) for x in global_best_a]

    elapsed = time.monotonic() - total_start
    print("Done in %.0fs. Final n=%d" % (elapsed, len(final_list)))
    return final_list
