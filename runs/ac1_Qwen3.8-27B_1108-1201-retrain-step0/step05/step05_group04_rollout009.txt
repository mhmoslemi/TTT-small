def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)

        max_c = c[top_idx[0]]
        w[top_idx] = np.exp(beta * (c[top_idx] - max_c))

    else:

        w = np.exp(beta * (c - max_c))
        w /= w.sum()
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def compute_grad_exact(a, c, threshold_frac=0.0):
    """
    Exact subgradient of max(c): average of gradients at active lags.
    threshold_frac: include lags within this fraction of max.
    """
    n = len(a)
    max_c = c.max()
    threshold = max_c * (1 - threshold_frac) if threshold_frac > 0 else max_c - 1e-10
    active = c >= threshold
    if not np.any(active):
        active[np.argmax(c)] = True
    
    # grad[i] = 2 * sum_{k active} a[k-i] * I(0 <= k-i < n)
    w = active.astype(np.float64)
    w /= w.sum()
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

Let me focus on a **KKT-based refinement**:

At the optimum, the KKT conditions are:
- For the active constraints (c[k] = max): the gradient of c[k] must be in the normal cone of the feasible set
- For inactive constraints: c[k] < max
- Complementary slackness: a[i] * (grad_f[i] - λ - μ[k]*grad_c[k][i]) = 0 for all i

Where μ[k] >= 0 are the Lagrange multipliers for the active constraints, λ is for the sum constraint.

The key insight: at the optimum, for each i with a[i] > 0:
sum_{k active} μ[k] * 2*a[k-i] = λ

This means the "weighted average" of a[k-i] over active lags must be constant for all i with a[i] > 0.

We can use this to check how close we are to the optimum and guide the optimization.

1. Identify the active set S (lags at or near the max)

3. If V[i] varies significantly among i with a[i] > 0, adjust a to reduce this variation
4. Update the active set and repeat

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences of length 8192.
Strategy: Multi-phase optimization combining Adam-LSE, FISTA with exact subgradient,
KKT-based refinement, spectral flattening, and coordinate-level peak shaving.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_grad_lse(a, c, beta, K=None):
    """
    Gradient of LSE (or top-K LSE) of convolution values.
    Returns gradient of length n.
    """
    n = len(a)
    if K is not None and K < len(c):
        top_idx = np.argsort(c)[::-1][:K]
        max_c = c[top_idx[0]]
        w = np.zeros(len(c))
        w[top_idx] = np.exp(beta * (c[top_idx] - max_c))
        w /= w.sum()
    else:
        max_c = c.max()
        w = np.exp(beta * (c - max_c))
        tot = w.sum()
        if tot < 1e-300:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

    """
    Compute KKT residual: how well the condition
    V[i] = sum_{k in S} a[k-i] is constant over {i: a[i] > 0}
    is satisfied. Returns the relative standard deviation.
    """
    n = len(a)
    max_c = c.max()
    top_idx = np.argsort(c)[::-1][:num_active]
    active = np.zeros(len(c), dtype=bool)
    active[top_idx] = True
    w = active.astype(np.float64)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)

    if mask.sum() < 2:
        return 0.0
    V_active = V[mask]
    if V_active.mean() < 1e-30:

        active[top_idx] = True
        w = active.astype(np.float64)
        w /= w.sum()

def spectral_flatten_step(a, n, S, strength=0.1):
    """
    One step of spectral flattening: modify the sequence to make
    its power spectrum more uniform, reducing autocorrelation peaks.
    """
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    P = np.abs(A) ** 2

    # Target: flat spectrum = total_power / fsize
    total_power = P.sum()
    target = total_power / fsize

    # Flattened amplitude
    A_flat = A * np.sqrt(np.maximum(target / np.maximum(P, 1e-30), 0.1))
    # Blend
    A_new = (1.0 - strength) * A + strength * A_flat
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new

    best_a = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Top peaks
        top_idx = np.argsort(c)[::-1][:8]

        # Compute contribution matrix for top peaks
        for k in top_idx[:4]:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo >= hi:
                continue
            js = np.arange(lo, hi + 1)
            partners = k - js
            contribs = a[js] * a[partners]
            total_c = c[k]
            if total_c < 1e-10:
                continue

            # Find the contributing pair with highest product
            i_best = js[np.argmax(contribs)]
            j_best = partners[np.argmax(contribs)]

            # Reduce a[i_best] and a[j_best] slightly, boost others
            reduction = 0.01 * min(a[i_best], a[j_best] if i_best != j_best else a[i_best])
            a[i_best] -= reduction
            if i_best != j_best:
                a[j_best] -= reduction
            else:
                a[i_best] -= reduction

            # Redistribute to maintain sum
            a = project_simplex(a, S)

    return best_a, best


def run_hybrid_phases(a_init, n, S, max_time, rng):
    """
    Run all optimization phases in sequence, keeping the best result.
    """
    a = normalize_sum(a_init, n)
    a = np.maximum(a, 0.0)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    def update_best(a_cand):
        nonlocal best, best_a
        c = fft_convolve_self(a_cand)
        v = c.max()
        if v < best:
            best = v
            best_a = a_cand.copy()
            return True
        return False

    # Phase A: Adam-LSE (40% of time)
    tA = max_time * 0.35
    rem = max_time - (time.monotonic() - start)
    tA = min(tA, rem)
    if tA > 10:
        a, _ = run_adam_lse(a, n, S, tA, rng, beta_start=30.0,
                            beta_end=30000.0, beta_steps=300, K=10,
                            lr=0.008, stagnation_limit=600)
        update_best(a)
        print("  After Adam-LSE: %.8f" % best)

    # Phase B: FISTA with exact subgradient (25%)
    rem = max_time - (time.monotonic() - start)
    tB = rem * 0.3
    if tB > 10:
        a, _ = run_fista_subgrad(best_a, n, S, tB, rng, num_active=3,
                                 step_scale=0.3, stagnant_limit=1000)
        update_best(a)
        print("  After FISTA: %.8f" % best)

    # Phase C: KKT refinement (15%)
    rem = max_time - (time.monotonic() - start)
    tC = rem * 0.2
    if tC > 10:
        a, _ = run_kkt_refinement(best_a, n, S, tC, rng, num_active=10,
                                  n_inner=100, lr=0.001)
        update_best(a)
        print("  After KKT: %.8f" % best)

    # Phase D: Spectral flattening (15%)
    rem = max_time - (time.monotonic() - start)
    tD = rem * 0.2
    if tD > 10:
        a, _ = run_spectral(best_a, n, S, tD, rng, strength=0.03, every=3)
        update_best(a)
        print("  After Spectral: %.8f" % best)

    # Phase E: High-beta LSE polish (remaining)
    rem = max_time - (time.monotonic() - start)
    if rem > 10:
        a, _ = run_adam_lse(best_a, n, S, rem, rng, beta_start=10000.0,
                            beta_end=100000.0, beta_steps=50, K=5,
                            lr=0.002, stagnation_limit=200)
        update_best(a)
        print("  After High-beta polish: %.8f" % best)

    return best_a, best


def generate_random_init(n, rng, style='smooth'):
    """Generate a random initialization with various styles."""
    t = np.linspace(0, 1, n)
    if style == 'gaussian':
        mu = rng.uniform(0.2, 0.8)
        sigma = rng.uniform(0.1, 0.3)
        a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    elif style == 'bump':
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-((t - center) / width) ** 4)
    elif style == 'cosine':
        k = rng.integers(1, 5)
        phase = rng.uniform(0, np.pi)
        a = 0.5 + 0.5 * np.cos(k * np.pi * t + phase)
    elif style == 'random_smooth':
        # Random low-pass filtered
        fsize = 1 << n.bit_length()
        A = rng.standard_normal(fsize) + 1j * rng.standard_normal(fsize)
        # Low-pass
        mask = np.zeros(fsize)
        bandwidth = int(fsize * rng.uniform(0.05, 0.2))
        mask[:bandwidth] = 1.0
        mask[-bandwidth:] = 1.0
        A *= mask
        a = np.real(ifft(A))[:n] + 5.0  # Make positive
    else:
        a = np.ones(n) * 0.5 + 0.5 * np.sin(np.pi * t)
    return np.maximum(a, 1e-10)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Multi-phase: Adam-LSE, FISTA subgradient, KKT refinement, spectral flattening.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)
    n = 8192
    S = np.sqrt(2.0 * n)

    print("Budget: %.0fs, safety: %.0fs, n=%d" % (budget_s, safety, n))

    # Load initial sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            # If different length, interpolate to n
            if len(given) != n:
                t_old = np.linspace(0, 1, len(given))
                t_new = np.linspace(0, 1, n)
                given = np.interp(t_new, t_old, given)
            print("Loaded initial: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception as e:
        print("Could not load initial: %s" % str(e))

    global_best = np.inf
    global_a = None

    def try_seq(a_cand):
        nonlocal global_best, global_a
        c = fft_convolve_self(a_cand)
        v = c.max()
        if v < global_best:
            global_best = v
            global_a = a_cand.copy()
            print("  * New best: %.8f" % v)

    # Quick evaluation of given
    if given is not None:
        a_g = normalize_sum(given, n)
        c_g = fft_convolve_self(a_g)
        print("Initial value: max_b=%.8f" % c_g.max())

    # === Main optimization loop ===
    # Phase 1: Quick Adam-LSE to warm up (60s)
    t1 = min(60.0, (deadline - time.monotonic()) * 0.1)
    print("=== Phase 1: Quick Adam-LSE (%.0fs) ===" % t1)
    a_work = normalize_sum(given if given is not None else np.ones(n), n)
    a_work, _ = run_adam_lse(a_work, n, S, t1, rng, beta_start=50.0,
                             beta_end=5000.0, beta_steps=50, K=10, lr=0.01)
    try_seq(a_work)

    # Phase 2: Full hybrid optimization on best (70% of remaining)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t2 = remaining * 0.55
        print("=== Phase 2: Full hybrid (%.0fs) ===" % t2)
        a_work, _ = run_hybrid_phases(global_a, n, S, t2, rng)
        try_seq(a_work)

    # Phase 3: Try a different initialization (random smooth)
    remaining = deadline - time.monotonic()
    if remaining > 120:
        t3 = remaining * 0.25
        print("=== Phase 3: Random restart (%.0fs) ===" % t3)
        for trial in range(3):
            if time.monotonic() > deadline - 60:
                break
            a_rand = generate_random_init(n, rng, style='random_smooth' if trial == 0
                                          else ('gaussian' if trial == 1 else 'bump'))
            a_rand = normalize_sum(a_rand, n)
            t3t = min(40.0, (deadline - time.monotonic()) * 0.3)
            a_r, _ = run_adam_lse(a_rand, n, S, t3t, rng, beta_start=20.0,
                                  beta_end=10000.0, beta_steps=100, K=10, lr=0.01)
            try_seq(a_r)

    # Phase 4: Final intensive polish on best (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("=== Phase 4: Final polish (%.0fs) ===" % remaining)

        # 4a: FISTA with very narrow active set
        t4a = remaining * 0.4
        if t4a > 10:
            a_f, _ = run_fista_subgrad(global_a, n, S, t4a, rng,
                                       num_active=2, step_scale=0.2,
                                       stagnant_limit=800)
            try_seq(a_f)

        # 4b: Very high beta LSE
        remaining2 = deadline - time.monotonic()
        t4b = remaining2 * 0.5
        if t4b > 10:
            a_h, _ = run_adam_lse(global_a, n, S, t4b, rng,
                                  beta_start=20000.0, beta_end=200000.0,
                                  beta_steps=30, K=3, lr=0.001,
                                  stagnation_limit=300)
            try_seq(a_h)

        # 4c: Peak shaving
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            a_p = peak_shaving_advanced(global_a.copy(), n, S, remaining3 * 0.5, rng)
            try_seq(a_p)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: eval=%.8f, internal=%.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  Rejected! Using fallback.")
            global_a = np.ones(n) * (S / n)
    else:
        print("WARNING: Using uniform fallback.")
        global_a = np.ones(n) * (S / n)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(n) * (S / n)

    print("RETURNING: length=%d, best=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
