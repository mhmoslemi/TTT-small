    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_gradient_active(a, c, n_active_threshold=1.0):
    """
    Compute gradient using active set (lags within n_active_threshold of max).
    Returns (gradient, active_lags, max_val).
    """
    n = len(a)
    c_max = c.max()
    # Active lags: those within threshold of max
    threshold = c_max - n_active_threshold
    active_mask = c >= threshold
    n_active = active_mask.sum()
    if n_active == 0:
        active_mask = np.zeros_like(c, dtype=bool)
        active_mask[np.argmax(c)] = True
        n_active = 1
    
    w = active_mask.astype(np.float64) / n_active
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, c_max, n_active


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    c_shifted = c - cm
    # Clip to avoid overflow
    scaled = np.clip(beta * c_shifted, -700, 700)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + np.log(tot) / beta
    return grad, lse_val


def normalize_to_sum(a, S):

    return a * (S / s)

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """
    Compute gradient of the Log-Sum-Exp approximation of max(c).
    Returns (gradient_array, lse_value).
    """
    n = len(a)
    cm = c.max()
    scaled = np.clip(beta * (c - cm), -700.0, 700.0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    # Gradient: 2 * cross_correlation(w, a)[:n]
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + np.log(tot) / beta
    return grad, lse_val


def active_set_gradient(a, c, top_k=10):
    """
    Compute gradient using the top-k lags of c (hard active set).
    Returns (gradient_array, max_val, number_active).
    """
    n = len(a)
    c_max = c.max()
    # Get top-k indices
    if top_k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    # Only include lags that are within a reasonable threshold
    threshold = c_max - 0.1 * max(1e-10, c_max)
    active = top_idx[c[top_idx] >= threshold]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, c_max, len(active)


def compute_kkt_residual(a, c, top_k=20):
    """
    Compute the KKT residual: at optimality, the subgradient should be
    constant (equal to the Lagrange multiplier) wherever a > 0.
    Returns the standard deviation of the subgradient over the support.
    """
    n = len(a)
    c_max = c.max()
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    threshold = c_max - 0.05 * max(1e-10, c_max)
    active = top_idx[c[top_idx] >= threshold]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    # KKT: grad[m] should be constant where a[m] > 0
    support = a > 1e-10 * a.max()
    if support.sum() < 2:
        return 0.0
    grad_support = grad[support]
    return np.std(grad_support)

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = y.copy()
            last_improve = it
            if verbose and (it % 1000 == 0 or best < 1.51):
                print("    LSE-FISTA it %d: max=%.10f, beta=%.0f" % (it, best, beta))

        stagnant = it - last_improve

        # Adaptive beta
        grad, lse_val = lse_gradient(y, c, beta)
        gap = lse_val - cur_max
        if gap < 0.005:
            beta = min(beta * 1.4, beta_max)
        elif gap > 0.5:
            beta = max(beta / 1.3, beta_min)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step = 1.0 / beta
        improved = False
        for _ in range(10):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max - 1e-12:
                improved = True
                break
            step *= 0.5

        if not improved:
            x_new = project_simplex(y - step * grad, S)

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y_new, 0.0)
        x = x_new
        t_mom = t_new

        # Momentum restart + noise on stagnation
        if stagnant > 500:
            t_mom = 1.0
            y = x.copy()
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stagnant / 8000.0)
            if noise_scale > 1e-12:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                x = y.copy()

    # Final evaluation
    for candidate in [x, y]:
        c_f = fft_convolve_self(candidate)
        if c_f.max() < best:
            best = c_f.max()
            best_a = candidate.copy()

    return best_a, best


def run_active_refinement(a_init, n, S, max_time, rng, verbose=True):
    """
    Active-set gradient descent with top-K lag focus.
    Uses smaller steps and no momentum for precision.
    """
    a = normalize_to_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            last_improve = it
            if verbose and (it % 500 == 0 or best < 1.51):
                print("    Active it %d: max=%.10f" % (it, best))

        stagnant = it - last_improve
        if stagnant > 1000:
            break

        # Use increasing top-K as we converge
        top_k = min(5 + stagnant // 100, 50)
        grad, _, n_act = active_set_gradient(a, c, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: adaptive based on gradient magnitude
        step = 0.005 * S / (gnorm * np.sqrt(n))
        step = min(step, 0.05)

        # Try a few step sizes
        improved = False
        for trial in range(5):
            a_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max - 1e-13:
                a = a_new
                improved = True
                break
            step *= 0.5

        if not improved:
            a = project_simplex(a - step * grad * 0.5, S)

        # Periodic noise
        if it % 300 == 0 and stagnant > 200:
            ns = 0.001 * S / np.sqrt(n) * np.exp(-stagnant / 3000.0)
            if ns > 1e-13:
                a = project_simplex(a + rng.normal(0, ns, n), S)

    c_f = fft_convolve_self(a)
    if c_f.max() < best:
        best = c_f.max()
        best_a = a.copy()

    return best_a, best


def run_kkt_polish(a_init, n, S, max_time, rng, verbose=True):
    """
    KKT-directed polishing: explicitly try to make the subgradient flat
    over the support of a. This is the key refinement step.
    """
    a = normalize_to_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-13:
            best = cur_max
            best_a = a.copy()
            last_improve = it
            if verbose and it % 200 == 0:
                print("    KKT it %d: max=%.10f, residual=%.2e" % (it, best, compute_kkt_residual(a, c)))

        stagnant = it - last_improve
        if stagnant > 500:
            break

        # Compute the "flattening" direction
        top_k = min(10 + stagnant // 50, 30)
        grad, _, _ = active_set_gradient(a, c, top_k=top_k)

        # We want grad to be constant over the support of a.
        # The "excess" at position m is grad[m] - mean(grad[support])
        support = a > 1e-10 * a.max()
        if support.sum() < 2:
            break
        grad_mean = np.mean(grad[support])
        direction = np.zeros(n)
        direction[support] = grad[support] - grad_mean
        # Also push up elements near zero (to maintain full support)
        near_zero = (a > 0) & (a < 0.01 * a.max())
        direction[near_zero] -= 0.1

        # Apply small step in the flattening direction
        step = 0.001 * S / (np.linalg.norm(direction) + 1e-15)
        a_new = project_simplex(a + step * direction, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur_max:
            a = a_new
            if c_new.max() < best:
                best = c_new.max()
                best_a = a.copy()
                last_improve = it
        else:
            # Try the opposite direction (reduce the max directly)
            a_new2 = project_simplex(a - 0.001 * grad / (np.linalg.norm(grad) + 1e-15), S)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < cur_max:
                a = a_new2
                if c_new2.max() < best:
                    best = c_new2.max()
                    best_a = a.copy()
                    last_improve = it

    return best_a, best


def run_peak_deflation(a_init, n, S, max_time, rng, verbose=True):
    """
    Targeted peak deflation: directly reduce contributions to the worst lags.
    Uses multiplicative updates on high-responsibility coordinates.
    """
    a = normalize_to_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-13:
            best = cur_max
            best_a = a.copy()
            if verbose and it % 50 == 0:
                print("    Deflate it %d: max=%.10f" % (it, best))

        # Find top peaks
        n_peaks = min(5, len(c))
        top_idx = np.argsort(c)[::-1][:n_peaks]

        # Compute responsibility for each coordinate
        resp = np.zeros(n)
        for k in top_idx:
            weight = max(0.0, c[k] - cur_max * 0.995)
            if weight < 1e-15:
                continue
            # Contribution of a[j] to c[k] is a[j] * a[k-j]
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo > hi:
                continue
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            resp[js] += contrib * weight

        resp_max = resp.max()
        if resp_max < 1e-30:
            break

        # Multiplicative decrease for high-responsibility coords
        lr = 0.005 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)

    return best_a, best


def spectral_perturbation(a, n, rng, strength=0.01):
    """
    Perturb in the frequency domain to explore different shapes.
    Add small random phase perturbations to the DFT.
    """
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    # Add small random real perturbations to the spectrum
    noise = rng.normal(0, strength * a.max(), n)
    noise_f = fft(noise, fsize)
    A += noise_f * 0.1
    a_new = np.real(ifft(A))[:n]
    return np.maximum(a_new, 0.0)


def try_sequence(a, global_state):
    """Evaluate a candidate and update global best."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a, n)
    c = fft_convolve_self(a)
    val = c.max()
    if val < global_state['best']:
        global_state['best'] = val
        global_state['a'] = a.copy()
        print("  ** New best: %.10f (n=%d) **" % (val, n))
    return val


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Multi-phase: LSE-FISTA -> Active-set -> KKT polish -> Peak deflation.
    Uses the previous best as starting point with perturbation strategies.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))

    global_state = {'best': np.inf, 'a': None}

    # Load previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception:
        pass

    if given is None:
        # Fallback: raised cosine initialization
        n = 8192
        t = np.linspace(0, 1, n)
        given = 0.5 * (1 + np.cos(2 * np.pi * (t - 0.3)))
        given = np.maximum(given, 0.0)
        given = np.clip(given, 0, 1000)
        print("Using raised cosine fallback: length=%d" % n)

    n = len(given)
    S = np.sqrt(2.0 * n)

    # Quick initial evaluation
    a_start = normalize_to_sum(given, n)
    c_start = fft_convolve_self(a_start)
    print("Initial: max=%.10f" % c_start.max())
    try_sequence(a_start, global_state)

    remaining = deadline - time.monotonic()
    if remaining < 30:
        print("Not enough time, returning initial.")
        return [float(x) for x in global_state['a']]

    # === Phase 1: LSE-FISTA (40% of time) ===
    t1 = remaining * 0.40
    print("\n=== Phase 1: LSE-FISTA (%.0fs) ===" % t1)
    a1, v1 = run_lse_fista(a_start, n, S, t1, rng, verbose=True)
    try_sequence(a1, global_state)

    # === Phase 2: Active-set refinement (25% of time) ===
    remaining = deadline - time.monotonic()
    t2 = remaining * 0.30
    print("\n=== Phase 2: Active-set refinement (%.0fs) ===" % t2)
    a2, v2 = run_active_refinement(global_state['a'], n, S, t2, rng, verbose=True)
    try_sequence(a2, global_state)

    # === Phase 3: KKT-directed polishing (20% of time) ===
    remaining = deadline - time.monotonic()
    t3 = remaining * 0.25
    print("\n=== Phase 3: KKT polish (%.0fs) ===" % t3)
    a3, v3 = run_kkt_polish(global_state['a'], n, S, t3, rng, verbose=True)
    try_sequence(a3, global_state)

    # === Phase 4: Peak deflation (10% of time) ===
    remaining = deadline - time.monotonic()
    t4 = remaining * 0.15
    print("\n=== Phase 4: Peak deflation (%.0fs) ===" % t4)
    a4, v4 = run_peak_deflation(global_state['a'], n, S, t4, rng, verbose=True)
    try_sequence(a4, global_state)

    # === Phase 5: Spectral perturbations + re-optimization (15% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t5 = remaining * 0.6
        print("\n=== Phase 5: Perturbation search (%.0fs) ===" % t5)
        n_perts = 3
        per_pert = t5 / n_perts

        for p in range(n_perts):
            if time.monotonic() > deadline - 30:
                break
            # Different perturbation strategies
            if p == 0:
                # Frequency domain noise
                a_pert = spectral_perturbation(global_state['a'], n, rng, strength=0.005)
            elif p == 1:
                # Asymmetric shift: emphasize one side
                shift = rng.integers(5, 50)
                a_pert = global_state['a'].copy()
                a_pert[shift:] *= 1.0 + 0.02
                a_pert[:-shift] *= 1.0 - 0.02
                a_pert = np.maximum(a_pert, 0.0)
            else:
                # Small random multiplicative perturbation
                mult = 1.0 + rng.normal(0, 0.002, n)
                a_pert = global_state['a'] * mult
                a_pert = np.maximum(a_pert, 0.0)

            a_pert = normalize_to_sum(a_pert, n)
            v_pert = try_sequence(a_pert, global_state)

            # Quick LSE-FISTA from perturbed start
            time_left_for_this = min(per_pert, deadline - time.monotonic() - 15)
            if time_left_for_this > 5:
                a_opt, v_opt = run_lse_fista(a_pert, n, S, time_left_for_this, rng, verbose=False)
                try_sequence(a_opt, global_state)

        # Final KKT polish on best
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print("\n=== Final KKT polish (%.0fs) ===" % remaining)
            a_f, v_f = run_kkt_polish(global_state['a'], n, S, remaining * 0.7, rng, verbose=True)
            try_sequence(a_f, global_state)

    # Final verification
    if global_state['a'] is not None:
        final_list = [float(x) for x in np.clip(global_state['a'], 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nFinal: evaluate_sequence=%.10f, internal=%.10f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_state['best'], n))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_state['a'] = np.ones(256) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result, using uniform fallback.")
        global_state['a'] = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_state['a'] = np.clip(global_state['a'], 0.0, 1000.0)
    if global_state['a'].sum() < 0.01:
        global_state['a'] = np.ones(256) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.10f" % (len(global_state['a']), global_state['best']))
    return [float(x) for x in global_state['a']]
