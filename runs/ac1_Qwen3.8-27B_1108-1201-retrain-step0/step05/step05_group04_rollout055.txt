def lp_dual_solve(a, active_set, S, n_iter=1000):
    """Solve the LP dual for the active set problem."""
    m = len(active_set)
    mu = np.ones(m) / m  # uniform initialization
    c_vals = np.array([c[k] for k in active_set])
    
    for it in range(n_iter):
        # Compute h[j] = sum_k mu[k] * a[active_set[k] - j] (valid indices)
        # This is like a correlation
        h = np.zeros(len(a))
        for i, k in enumerate(active_set):
            if mu[i] < 1e-15:
                continue
            # h[j] += mu[i] * a[k-j] for valid j
            # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
            lo = max(0, k - len(a) + 1)
            hi = min(k, len(a) - 1)
            h[lo:hi+1] += mu[i] * a[lo:hi+1][::-1]  # a[k-j] for j in [lo, hi]
        
        j_star = np.argmax(h)
        
        # Gradient
        grad = np.zeros(m)
        for i, k in enumerate(active_set):
            lo = max(0, k - len(a) + 1)
            hi = min(k, len(a) - 1)
            if lo <= j_star <= hi:
                # a[k - j_star]
                grad[i] = -2 * S * a[k - j_star]
            grad[i] -= c_vals[i]
        
        # Projected gradient step
        step = 1.0 / (it + 10)  # or use backtracking
        mu_new = project_simplex(mu - step * grad, 1.0)
        mu = mu_new

    top_idx = np.argsort(c)[::-1][:max_lags]

    return sorted(active.tolist())


def lp_dual_subgradient(a, c, active_set, S, n_dual_iters=100):
    """
    Compute optimal subgradient via LP dual.
    Returns gradient vector g where g[j] = 2*h[j], h[j] = sum_k mu[k]*a[k-j].
    """
    n = len(a)
    m = len(active_set)
    
    if m == 1:
        k = active_set[0]
        h = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        h[lo:hi+1] = a[lo:hi+1][::-1]
        return 2.0 * h
    
    # LP Dual: maximize -2*S*max_j(h[j]) - sum_k mu[k]*c[k]
    # subject to sum mu = 1, mu >= 0
    # h[j] = sum_k mu[k]*a[k-j]
    
    mu = np.ones(m) / m
    c_vals = np.array([c[k] for k in active_set])
    
    fsize = 1 << (2 * n).bit_length()
    A_fft = fft(a, fsize)
    
    for it in range(n_dual_iters):
        # Compute h via FFT
        V = np.zeros(fsize)
        for i, k in enumerate(active_set):
            V[k] = mu[i]
        H = np.real(ifft(fft(V) * np.conj(A_fft)))[:n]
        
        j_star = np.argmax(H)
        
        # Subgradient of dual objective w.r.t. mu
        grad = -c_vals.copy()
        for i, k in enumerate(active_set):
            diff = k - j_star
            if 0 <= diff < n:
                grad[i] -= 2.0 * S * a[diff]
        
        # Projected gradient ascent with adaptive step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-20:
            break
        step = 0.5 / (it + 5) / (grad_norm / m)
        mu_new = project_simplex(mu + step * grad, 1.0)
        
        if np.linalg.norm(mu_new - mu) < 1e-14:
            mu = mu_new
            break
        mu = mu_new
    
    # Compute final gradient
    V = np.zeros(fsize)
    for i, k in enumerate(active_set):
        V[k] = mu[i]
    H = np.real(ifft(fft(V) * np.conj(A_fft)))[:n]
    
    return 2.0 * H

def lse_fista_optimize(a_init, n, S, max_time, rng,

        stagnant = it - last_improve
        if stagnant > 300:

            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve = it

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def peak_shaving_refined(a, n, S, n_peaks=5, n_steps=100):
    """Surgical peak shaving with responsibility-based updates."""
    a = a.copy()
    for _ in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            weight = max(0, c[k] - cm * 0.995)
            np.add.at(resp, js, contrib * weight)
        
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        
        lr = 0.01 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)
    
    return a


def spectral_shaping(a, n, S, n_steps=50, rng=None):
    """
    Spectral shaping: reduce peaks by modifying the spectrum.
    In frequency domain, the autocorrelation peak relates to the 
    inverse DFT of |A(k)|^2. We try to flatten this.
    """
    a = a.copy()
    fsize = 1 << (2 * n).bit_length()
    
    for it in range(n_steps):
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        # The circular autocorrelation is ifft(power)
        circ_ac = np.real(ifft(power))[:n]
        cm = circ_ac.max()
        
        # Compute gradient: reduce power at frequencies that contribute to the peak
        # The peak is at lag 0 of circ_ac, which is sum(power)/(2*pi) = sum(a^2)
        # To reduce the peak, we want to redistribute power
        # Gradient of max(circ_ac) w.r.t. |A|^2 is concentrated at lag 0
        # So we want to reduce |A|^2 at frequencies where the phase alignment creates the peak
        
        # Simple approach: reduce the "coherence" by adding random phase
        if rng is not None:
            phase_noise = rng.normal(0, 0.1, fsize)
            A_new = A * np.exp(1j * phase_noise)
            a_new = np.real(ifft(A_new))[:n]
            a_new = np.maximum(a_new, 0)
            a_new = project_simplex(a_new, S)
        else:
            break
        
        c = fft_convolve_self(a_new)
        if c.max() < cm:
            a = a_new
    
    return a


def backtracking_subgradient(a, S, n, max_time, rng, 
                              active_size=10, rel_thresh=0.002):
    """
    Pure subgradient method with backtracking line search.
    No momentum - more stable for fine-tuning.
    """
    a = normalize_sum(a, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    step = 0.05
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            no_improve = 0

                print("  backtrack iter %d: max=%.8f" % (it, best))
        
        if no_improve > 1000:
            break
        
        active = compute_active_set(c, max_lags=active_size, rel_thresh=rel_thresh)
        g = lp_dual_subgradient(a, c, active, S, n_dual_iters=80)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        improved = False
        for attempt in range(15):
            step_try = step / (2 ** attempt)
            a_try = project_simplex(a - step_try * g, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < cur_max * (1 - 1e-8):
                a = a_try
                step = step_try * 1.1

            step *= 0.5
            if step < 1e-6:
                # Inject noise
                noise_scale = 0.0005 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                step = 0.01
                no_improve = 0
        
        no_improve += 1 if not improved else 0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

This is a weighted average of shifted copies of a, where the weights are determined by the LP dual. Intuitively, lags with higher c[k] values should get more weight (since they're "more active"), but the LP dual also accounts for the structure of a (through the a[k-j] terms in the dual constraints).
