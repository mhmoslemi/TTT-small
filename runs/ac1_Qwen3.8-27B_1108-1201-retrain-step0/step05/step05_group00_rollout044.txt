import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    """Self-convolution via FFT. Returns c[2n-1]."""
    n = len(a)
    Nf = int(N)
    if Nf < 2 * n - 1:
        Nf = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(Nf)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def lse_objective_and_grad(a, N, beta):
    """Compute LSE-smoothed peak and its gradient."""
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    if beta < 1e-6:
        weights = np.ones(len(c)) / len(c)
        lse_val = c_max
    else:
        shifted = c - c_max
        exp_bc = np.exp(beta * shifted)
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
            lse_val = c_max
        else:
            weights = exp_bc / total
            lse_val = c_max + np.log(total) / beta
    
    fsize = 1 << (3 * n - 2).bit_length()

def lbfgs_two_loop(s_hist, y_hist, g):
    """L-BFGS two-loop recursion."""
    m = len(s_hist)
    n = len(g)
    if m == 0:
        gn = np.linalg.norm(g)
        if gn < 1e-18:
            return np.zeros(n)
        return -g / gn
    q = g.copy()
    alphas = np.zeros(m)
    for i in range(m - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma_use = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma_use = 1.0
    r = gamma_use * q
    for i in range(m):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def freq_perturb(a, N, rng, strength=0.3):
    """Perturb in frequency domain to escape local minima."""
    n = len(a)
    S = a.sum()
    A = fft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    noise = strength * rng.standard_normal(N)
    phase_new = phase + noise
    A_new = mag * np.exp(1j * phase_new)
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def lse_lbfgs_optimize(a_init, n, S, N, max_time, rng, beta_init=2.0, beta_max=500.0, m_hist=15):
    """L-BFGS optimization on LSE objective with adaptive beta."""
    a = safe_normalize(a_init, n, S)
    beta = beta_init
    best_a = a.copy()
    best_cmax = np.inf
    s_hist = []
    y_hist = []
    start_time = time.monotonic()
    it = 0
    stall_count = 0
    
    _, g, c, c_max = lse_objective_and_grad(a, N, beta)
    if c_max < best_cmax:
        best_cmax = c_max
        best_a = a.copy()
    
    step_scale = S / (n * np.sqrt(beta + 1))
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        if it % 50 == 0:
            beta = min(beta_max, beta * 1.05)
            step_scale = S / (n * np.sqrt(beta + 1))
            _, g, c, c_max = lse_objective_and_grad(a, N, beta)
            if c_max < best_cmax - 1e-10:
                best_cmax = c_max
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1
        
        if stall_count > 150 and it % 30 == 0:
            a = freq_perturb(a, N, rng, strength=0.2 + 0.1 * min(stall_count / 100, 1.0))
            s_hist = []
            y_hist = []
            _, g, c, c_max = lse_objective_and_grad(a, N, beta)
            if c_max < best_cmax:
                best_cmax = c_max
                best_a = a.copy()
            stall_count = 0
        
        if it % 50 == 0 and len(s_hist) >= m_hist:
            s_hist = s_hist[-m_hist:]
            y_hist = y_hist[-m_hist:]
        
        direction = lbfgs_two_loop(s_hist, y_hist, g)
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-18:
            break
        
        alpha_ls = step_scale
        a_new = a
        cmax_new = c_max
        for _ in range(15):
            candidate = project_simplex(a + alpha_ls * direction, S)
            c_ls = compute_conv(candidate, N)
            cm_ls = c_ls.max()
            if cm_ls <= c_max or alpha_ls < 1e-10:
                a_new = candidate
                cmax_new = cm_ls
                break
            alpha_ls *= 0.4
        
        if cmax_new < best_cmax:
            best_cmax = cmax_new
            best_a = a_new.copy()
        
        _, g_new, c_nw, cmax_nw = lse_objective_and_grad(a_new, N, beta)
        
        s_k = a_new - a
        y_k = g_new - g
        sy = np.dot(s_k, y_k)
        if sy > 1e-12:
            s_hist.append(s_k)
            y_hist.append(y_k)
            if len(s_hist) > m_hist + 5:
                s_hist = s_hist[-m_hist:]
                y_hist = y_hist[-m_hist:]
        
        a = a_new
        g = g_new
        c_max = cmax_nw
    
    return best_a, best_cmax


def peak_reduction_polish(a_init, n, S, N, max_time, rng):
    """Local peak reduction heuristic for final polishing."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_cmax = np.inf
    start_time = time.monotonic()
    
    c = compute_conv(a, N)
    best_cmax = c.max()
    best_a = a.copy()
    
    while time.monotonic() - start_time < max_time:
        c = compute_conv(a, N)
        c_max = c.max()
        if c_max < best_cmax - 1e-10:
            best_cmax = c_max
            best_a = a.copy()
            c = compute_conv(a, N)
            c_max = c.max()
        
        m_star = int(np.argmax(c))
        lo = max(0, m_star - n + 1)
        hi = min(n - 1, m_star)
        pairs = []
        for i in range(lo, hi + 1):
            j = m_star - i
            if 0 <= j < n and i != j:
                pairs.append((i, j, a[i] * a[j]))
        
        if not pairs:
            break
        
        pairs.sort(key=lambda x: x[2], reverse=True)
        i_max, j_max, _ = pairs[0]
        
        max_transfer = 0.05 * min(a[i_max], a[j_max])
        if max_transfer < 1e-15:
            break
        
        a_try = a.copy()
        a_try[i_max] -= max_transfer
        a_try[j_max] += max_transfer
        a_try = safe_normalize(a_try, n, S)
        
        c_try = compute_conv(a_try, N)
        if c_try.max() < c_max:
            a = a_try
        else:
            a_try2 = a.copy()
            a_try2[i_max] -= max_transfer / 2
            a_try2[j_max] += max_transfer / 2
            a_try2 = safe_normalize(a_try2, n, S)
            c_try2 = compute_conv(a_try2, N)
            if c_try2.max() < c_max:
                a = a_try2
            else:
                break
    
    return best_a, best_cmax


def make_init_zadoff(n, rng):
    """Zadoff-Chu inspired: use cosine of quadratic phase."""
    i = np.arange(n, dtype=np.float64)
    sqrt_n = np.sqrt(n)
    phase = -np.pi * i * i / n
    a = 1.0 + 0.8 * np.cos(phase) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_chirp(n, rng):
    """Chirp sequence with multiple frequencies."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    a += 0.7 * np.sin(2 * np.pi * (n * t * t / 2))
    a += 0.3 * np.sin(2 * np.pi * 7 * t)
    a += 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.0, None)


def make_init_multi_sin(n, rng):
    """Sum of multiple sinusoids with random phases."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 1.0
    for k in range(1, 8):
        freq = rng.integers(2, n // 2)
        phase = rng.uniform(0, 2 * np.pi)
        a += 0.3 / k * np.sin(2 * np.pi * freq * t + phase)
    return np.clip(a, 0.0, None)


def make_init_from_parent(parent_seq, n, rng):
    """Interpolate parent sequence to new length n."""
    if parent_seq is None or len(parent_seq) < 16:
        return None
    parent = np.asarray(parent_seq, dtype=np.float64)
    parent = np.clip(parent, 0, 1000)
    if parent.sum() < 1e-10:
        return None
    nparent = len(parent)
    x_old = np.linspace(0, 1, nparent)
    x_new = np.linspace(0, 1, n)
    interpolated = np.interp(x_new, x_old, parent)
    interpolated += 0.01 * rng.standard_normal(n)
    return np.clip(interpolated, 0.0, None)

        return c_max
    
    build_start = time.monotonic()
    inits = []
    
    if parent is not None:

    inits.append(("zadoff", make_init_zadoff(n, rng)))
