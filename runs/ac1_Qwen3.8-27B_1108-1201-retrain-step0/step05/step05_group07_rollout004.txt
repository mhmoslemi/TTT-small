Uses Adam-optimized LSE-smoothed subgradient with simulated annealing,
multi-beta schedule, block restarts, and population diversity.

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_lse_weights(c, beta):
    """Compute softmax weights for LSE-smoothed max."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.minimum(scaled, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return e / tot


def compute_lse_grad(a, w, n):
    """Gradient of LSE max of conv(a,a) w.r.t. a.
    grad_j = 2 * sum_k w_k * a_{k-j} for valid indices.
    """
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    g = 2.0 * np.real(ifft(W * A))[:n]
    return g


def compute_topk_grad(a, c, k):
    """Weighted subgradient from top-k convolution lags."""
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    median_c = np.median(c)
    weights = np.maximum(c[top_idx] - median_c, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights = weights / wsum

    # Build full weight vector
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_idx):
        w_full[idx] = weights[i]
    
    return compute_lse_grad(a, w_full, n)


def adam_update(state, params, grad, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update step. Returns new params and updated state."""
    t = state['t'] + 1
    m = beta1 * state['m'] + (1 - beta1) * grad
    v = beta2 * state['v'] + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    new_params = params - lr * m_hat / (np.sqrt(v_hat) + eps)
    return new_params, {'t': t, 'm': m, 'v': v}

def block_restart(a, S, rng, block_frac=0.1):
    """Perturb a random block of elements."""
    n = len(a)
    block_size = max(16, int(n * block_frac))
    start = rng.integers(0, n - block_size)
    a_new = a.copy()
    noise = rng.normal(0, 0.1 * S / np.sqrt(n), block_size)
    a_new[start:start + block_size] += noise
    return project_simplex(a_new, S)

Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses Adam-optimized LSE-smoothed gradient with simulated annealing,
top-k subgradient, block restarts, multi-start diversity, and
adaptive beta scheduling. All helper functions are top-level.

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_lse_weights(c, beta):
    """Compute softmax weights for LSE-smoothed max."""
    cm = c.max()
    scaled = beta * (c - cm)
    scaled = np.minimum(scaled, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return e / tot


def compute_grad_from_weights(a, w, n):
    """Gradient of weighted sum of convolution: grad_j = 2*sum_k w_k * a_{k-j}."""
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    g = 2.0 * np.real(ifft(W * A))[:n]
    return g


def compute_topk_weights(c, k):
    """Build weight vector focused on top-k lags."""
    n_c = len(c)
    k = min(k, n_c)
    top_idx = np.argsort(c)[::-1][:k]
    median_c = np.median(c)
    raw_w = np.maximum(c[top_idx] - median_c, 0.0)
    wsum = raw_w.sum()
    if wsum < 1e-30:
        raw_w = np.ones(k) / k
    else:
        raw_w = raw_w / wsum
    w_full = np.zeros(n_c)
    for i, idx in enumerate(top_idx):
        w_full[idx] = raw_w[i]
    return w_full


def init_adam_state(n):
    """Initialize Adam optimizer state."""
    return {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}


def adam_step(state, params, grad, lr, beta1, beta2, eps):
    """One Adam step. Returns (new_params, new_state)."""
    t = state['t'] + 1
    m = beta1 * state['m'] + (1.0 - beta1) * grad
    v = beta2 * state['v'] + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    new_params = params - lr * m_hat / (np.sqrt(v_hat) + eps)
    return new_params, {'t': t, 'm': m, 'v': v}

    """Cosine-based initial sequence."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 + 0.4 * np.cos(2.0 * np.pi * tt) + 0.15 * np.cos(4.0 * np.pi * tt)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a


def gen_init_chirp(n, rng):
    """Chirp-inspired initial sequence (real non-negative)."""
    tt = np.linspace(0, 1, n, endpoint=False)
    a = np.abs(np.cos(np.pi * tt * tt * 8.0)) + 0.05
    return a


def gen_init_interp(base, n):
    """Interpolate base sequence to new length n."""
    if base is None or len(base) < 2:
        return None
    t_old = np.linspace(0, 1, len(base))
    t_new = np.linspace(0, 1, n)
    a = np.interp(t_new, t_old, base)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a

        if it % 1000 == 0:
            cur_obj = 2.0 * n * best / (S * S)
            print(f"      it={it}, t={elapsed:.0f}s, beta={beta:.0f}, obj={cur_obj:.6f}, stuck={stuck}")

    return best_a, best


def optimize_topk_adam(a0, S, max_time, rng, k_init, k_max):
    """
    Adam-optimized top-k subgradient descent.
    Returns (best_a, best_max).
    """
    n = len(a0)
    a = project_simplex(a0, S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    st = init_adam_state(n)
    beta1_o = 0.9
    beta2_o = 0.999
    eps = 1e-8

    start = time.monotonic()
    it = 0
    stuck = 0
    patience = 400

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        k = int(k_init + (k_max - k_init) * (progress ** 0.7))

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best - 1e-12:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        w = compute_topk_weights(c, k)
        grad = compute_grad_from_weights(a, w, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        grad_n = grad / (gnorm + 1e-30)

        a_try, st = adam_step(st, a, grad_n, 0.08, beta1_o, beta2_o, eps)
        a_try = project_simplex(a_try, S)

        c_try = fft_convolve_self(a_try)
        if c_try.max() <= cur_max * (1.0 + 1e-10):
            a = a_try
        else:
            a_half, st = adam_step(st, a, grad_n, 0.04, beta1_o, beta2_o, eps)
            a_half = project_simplex(a_half, S)
            c_half = fft_convolve_self(a_half)
            if c_half.max() <= cur_max * (1.0 + 1e-10):
                a = a_half

        if stuck > patience:
            noise_scale = 0.015 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0.0, noise_scale, n), S)
            stuck = 0
            patience = int(patience * 1.2)
            st = init_adam_state(n)

        if it % 1000 == 0:
            cur_obj = 2.0 * n * best / (S * S)
            print(f"      it={it}, t={elapsed:.0f}s, k={k}, obj={cur_obj:.6f}, stuck={stuck}")

    return best_a, best


def do_block_restart(a, S, rng, block_frac):
    """Perturb a random block of elements and reproject."""
    n = len(a)
    block_size = max(32, int(n * block_frac))
    start = int(rng.integers(0, max(1, n - block_size)))
    a_new = a.copy()
    noise = rng.normal(0.0, 0.08 * S / np.sqrt(n), block_size)
    a_new[start:start + block_size] += noise
    return project_simplex(a_new, S)


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)
