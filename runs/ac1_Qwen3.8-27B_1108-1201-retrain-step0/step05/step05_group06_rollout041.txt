"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses:
- Power-2 warmup (FISTA)
- LSE continuation with FISTA + backtracking
- L-BFGS polish on log-parameterization
- Multi-n exploration
- Diverse initializations

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_and_grad(a, beta):
    """Compute LSE value and gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    p = e / e.sum()
    lse_val = cm + np.log(e.sum()) / beta
    # Gradient: 2 * ifft(fft(p, N) * conj(fft(a, N)))[0:n]
    fsize = 1 << (len(c) + n - 1).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(P * np.conj(A)))[:n]
    return lse_val, grad, c

                  n_beta_steps=20, verbose=True):

    FISTA with LSE continuation.
    Beta increases from beta_start to beta_end over n_beta_steps.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    
    best_obj = np.inf
    best_a = a.copy()
    
    # Determine beta schedule
    if n_beta_steps > 1:
        log_beta_start = np.log(beta_start)
        log_beta_end = np.log(beta_end)
        betas = np.exp(np.linspace(log_beta_start, log_beta_end, n_beta_steps))
    else:
        betas = np.array([beta_start])
    
    time_per_beta = budget / n_beta_steps
    
    for bi, beta in enumerate(betas):
        t_start = time.monotonic()
        time_left = t_start + time_per_beta - time.start  # handle below
        
        # FISTA on LSE_beta
        y = a.copy()
        x = a.copy()
        t_mom = 1.0
        it = 0
        L = 1.0  # initial Lipschitz estimate
        last_improve = 0
        best_lse = np.inf

            lse_val, grad, c = lse_and_grad(y, beta)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
- Power-2 warmup (FISTA)
- LSE continuation with FISTA + backtracking line search
- L-BFGS polish on log-softmax parameterization

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_obj(a):
    """Compute the actual objective 2*n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def compute_max_conv(a):
    """Compute max of self-convolution."""
    c = fft_convolve_self(a)
    return c.max(), c


def power2_grad_and_val(a):
    """Compute sum(c^2) and its gradient for the power-2 warmup."""
    n = len(a)
    c = fft_convolve_self(a)
    val = np.sum(c * c)
    fsize = 1 << (len(c) + n - 1).bit_length()
    C = fft(c, fsize)
    A = fft(a, fsize)
    grad = 4.0 * np.real(ifft(C * np.conj(A)))[:n]
    return val, grad


def lse_grad_and_val(a, beta):
    """Compute LSE_beta(conv) value and gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    e_sum = e.sum()
    val = cm + np.log(e_sum) / beta
    p = e / e_sum
    fsize = 1 << (len(c) + n - 1).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(P * np.conj(A)))[:n]
    return val, grad, c, p


def subgradient_at_max(a, c):
    """Subgradient from the argmax position of c."""
    n = len(a)
    k_star = int(np.argmax(c))
    w = np.zeros(len(c))
    w[k_star] = 1.0
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def weighted_subgradient(a, c, tol_frac=0.05):
    """Weighted subgradient from lags within tolerance of max."""
    n = len(a)
    max_val = c.max()
    active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_power2_fista(a_init, budget, rng):
    """Phase 1: Minimize sum(c^2) using FISTA."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    x = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_val = np.inf
    best_a = a.copy()
    L = 1.0
    it = 0

    while time.monotonic() - start < budget:
        it += 1
        val, grad = power2_grad_and_val(y)
        if val < best_val:
            best_val = val
            best_a = y.copy()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (8.0 * a.max() + 1.0)
        x_new = project_simplex(y - step * grad, S)

        # Check if we should reduce step
        val_new, _ = power2_grad_and_val(x_new)
        bs = 0
        while val_new > val - 0.5 * step * gnorm**2 and bs < 10 and time.monotonic() - start < budget:
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
            val_new, _ = power2_grad_and_val(x_new)
            bs += 1

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0.0)
        x = x_new
        t_mom = t_new

        if it % 1000 == 0:
            t_mom = 1.0
            y = x.copy()

    return best_a

def _lbfgs_objective(b, n, S, beta):
    """Objective and gradient for L-BFGS in log-softmax parameterization."""
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted)
    Z = exp_b.sum()
    if Z < 1e-300:
        return 1e10, np.zeros(n)
    a = (S / Z) * exp_b

    c = fft_convolve_self(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    e_sum = e.sum()
    fun = cm + np.log(e_sum) / beta

    p = e / e_sum
    fsize = 1 << (len(c) + n - 1).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    ga = 2.0 * np.real(ifft(P * np.conj(A)))[:n]

    ga_dot_a = np.dot(ga, a)
    gb = a * (ga - ga_dot_a / S)

    return fun, gb


def run_lbfgs_polish(a_init, budget, rng, beta=5000.0):
    """Phase 3: L-BFGS polish using log-softmax parameterization."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)

    best_obj = compute_obj(a)
    best_a = a.copy()

    # Initialize b from a: b_i = log(a_i / S * Z) = log(a_i) + const
    # Use b_i = log(a_i + eps) as initial log-parameters
    eps = 1e-10
    b0 = np.log(a + eps)

    t_start = time.monotonic()
    result = minimize(
        lambda b: _lbfgs_objective(b, n, S, beta),
        b0,
        method='L-BFGS-B',
        options={
            'maxiter': 5000,
            'ftol': 1e-14,
            'gtol': 1e-10,
            'maxfun': 20000,
            'maxls': 50,
        }
    )

    # Recover a from optimized b
    b_opt = result.x
    b_shifted = b_opt - b_opt.max()
    exp_b = np.exp(b_shifted)
    Z = exp_b.sum()
    a_opt = (S / Z) * exp_b
    a_opt = np.clip(a_opt, 0, 1000)
    a_opt = normalize_seq(a_opt, S)

    obj_opt = compute_obj(a_opt)
    if obj_opt < best_obj:
        best_obj = obj_opt
        best_a = a_opt.copy()

    elapsed = time.monotonic() - t_start
    print("  L-BFGS (beta=%.0f): obj=%.6f (nfev=%d, t=%.1fs, success=%s)" %
          (beta, best_obj, result.nfev, elapsed, result.success))

    return best_a, best_obj

def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)


def make_harding_init(n):
    """Initialize using a cosine squared (Harding-like) shape."""
    tt = np.linspace(0, 1, n, endpoint=False)
    v = 0.5 + 0.5 * np.cos(2 * np.pi * tt)
    return v * v


def make_gaussian_init(n):
    tt = np.arange(n, dtype=np.float64)
    center = n / 2.0
    sigma = n / 4.0
    return np.exp(-0.5 * ((tt - center) / sigma)**2)


def make_dual_freq_init(n):
    """Two-frequency initialization for better spectral spreading."""
    tt = np.arange(n, dtype=np.float64)
    v = 0.4 + 0.3 * np.cos(2 * np.pi * tt / n) + 0.3 * np.cos(4 * np.pi * tt / n + 0.5)
    return np.maximum(v, 0.01)

        obj_p2 = update_best(a_p2)
        print("  After power-2: obj=%.6f" % obj_p2)
        best_init = a_p2

    # === Phase 2: LSE FISTA continuation ===
    remaining = deadline - time.monotonic()
    p2_budget = min(remaining * 0.45, 1000.0)
    if p2_budget > 20:
        print("\n=== Phase 2: LSE FISTA continuation (%.0fs) ===" % p2_budget)
        a_lse, obj_lse = run_lse_fista(best_init, p2_budget, rng,
                                        beta_start=1.0, beta_end=5000.0,
                                        n_steps=20)
        update_best(a_lse)
        print("  After LSE FISTA: obj=%.6f" % global_best_obj)
        best_init = global_best_a.copy()

    # === Phase 3: L-BFGS polish ===
    remaining = deadline - time.monotonic()
    p3_budget = min(remaining * 0.20, 300.0)
    if p3_budget > 10:
        print("\n=== Phase 3: L-BFGS polish (%.0fs) ===" % p3_budget)
        a_lbfgs, obj_lbfgs = run_lbfgs_polish(best_init, p3_budget, rng, beta=2000.0)
        update_best(a_lbfgs)
        print("  After L-BFGS: obj=%.6f" % global_best_obj)

    # === Phase 4: LSE FISTA high beta ===
    remaining = deadline - time.monotonic()
    p4_budget = min(remaining * 0.20, 400.0)
    if p4_budget > 20:
        print("\n=== Phase 4: LSE high beta (%.0fs) ===" % p4_budget)
        a_lse2, obj_lse2 = run_lse_fista(global_best_a, p4_budget, rng,
                                          beta_start=2000.0, beta_end=100000.0,
                                          n_steps=15)
        update_best(a_lse2)
        print("  After high-beta LSE: obj=%.6f" % global_best_obj)

    # === Phase 5: Subgradient polish ===
    remaining = deadline - time.monotonic()
    p5_budget = min(remaining * 0.15, 250.0)
    if p5_budget > 10:
        print("\n=== Phase 5: Subgradient polish (%.0fs) ===" % p5_budget)
        a_sg, obj_sg = run_subgradient_polish(global_best_a, p5_budget, rng)
        update_best(a_sg)
        print("  After subgradient: obj=%.6f" % global_best_obj)

    # === Phase 6: Final L-BFGS polish ===
    remaining = deadline - time.monotonic()
    p6_budget = min(remaining * 0.10, 200.0)
    if p6_budget > 10:
        print("\n=== Phase 6: Final L-BFGS (%.0fs) ===" % p6_budget)
        a_fin, obj_fin = run_lbfgs_polish(global_best_a, p6_budget, rng, beta=10000.0)
        update_best(a_fin)
        print("  After final L-BFGS: obj=%.6f" % global_best_obj)

    # === Phase 7: Quick subgradient + L-BFGS alternating ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 7: Alternating polish (%.0fs) ===" % remaining)
        for round_i in range(3):
            if time.monotonic() > deadline - 10:
                break
            half = (deadline - time.monotonic()) * 0.6
            a_sg2, _ = run_subgradient_polish(global_best_a, half, rng)
            update_best(a_sg2)
            rem = deadline - time.monotonic()
            if rem > 10:
                a_lbfgs2, _ = run_lbfgs_polish(global_best_a, rem * 0.7, rng,
                                                beta=50000.0)
                update_best(a_lbfgs2)
        print("  After alternating: obj=%.6f" % global_best_obj)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        total_time = time.monotonic() - total_start
        if np.isfinite(ver):
            print("\n=== FINAL RESULT ===")
            print("Objective: %.6f (n=%d)" % (ver, len(final_list)))
            print("Total time: %.1fs" % total_time)
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
