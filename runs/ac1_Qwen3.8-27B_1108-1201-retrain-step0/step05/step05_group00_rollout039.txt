        if new_max == c_unaffected_max:

            # (decreasing delta increases the affected lags... wait, no)
            # If the max is at an unaffected lag, the subgradient is 0 (or we should increase delta)
            # Actually, if the global max is at an unaffected lag, then the current delta is already

            # We should increase delta to push the affected lags up? No, we want to minimize the max.
            # If the peak is at an unaffected lag, we can't change it by changing delta.
            # So we should just stop or try a different pair.
            break
        else:
            k_star_local = int(np.argmax(new_c_aff))
            sg = 2.0 * dc_coeff[k_star_local]

                hi = mid
            elif sg < 0:

                lo = mid
            else:

Here's the key formula:
If a[i_hot] -= d and a[i_cold] += d, then:
Δc[k] = 2d * (a[k-i_hot] - a[k-i_cold]) + d^2 * ([k=2*i_hot] - 2*[k=i_hot+i_cold] + [k=2*i_cold])

The "no change" lags are those where both k-i_hot and k-i_cold are out of range [0, n-1]. The "affected" lags are the complement.

```python
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, N):
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))
    return c[:2*n-1]


def lse_grad_and_val(a, N, beta):
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    scale = beta * (c - c_max)
    # Clip to avoid overflow
    scale = np.clip(scale, -100, 100)
    exp_bc = np.exp(scale)
    total = exp_bc.sum()
    if total < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
        lse_val = c_max
    else:
        weights = exp_bc / total
        lse_val = c_max + np.log(total) / beta
    
    fsize = 1 << (3*n - 2).bit_length()
    W = fft(weights, fsize)
    Apad = fft(a, fsize)
    corr = np.real(ifft(np.conj(W) * Apad))[:n]
    grad = 2.0 * corr
    return grad, lse_val, c, c_max


def toppk_grad_and_val(a, N, K):
    n = len(a)
    c = compute_conv(a, N)
    c_max = c.max()
    top_idx = np.argsort(c)[::-1][:K]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-300:
        weights = np.zeros(len(c))
        weights[int(np.argmax(c))] = 1.0
    else:
        weights = np.zeros(len(c))
        for i, k in enumerate(top_idx):
            weights[k] = top_vals[i] / total
    
    fsize = 1 << (3*n - 2).bit_length()
    W = fft(weights, fsize)
    Apad = fft(a, fsize)
    corr = np.real(ifft(np.conj(W) * Apad))[:n]
    grad = 2.0 * corr
    return grad, c_max, c, c_max


def lbfgs_two_loop(s_hist, y_hist, g, m=10):
    m_used = min(len(s_hist), m)
    if m_used == 0:
        gn = np.linalg.norm(g)
        return -g / gn if gn > 1e-18 else np.zeros(len(g))
    
    n = len(g)
    q = g.copy()
    alphas = np.zeros(m_used)
    
    for i in range(m_used - 1, -1, -1):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            alphas[i] = 0.0
        else:
            alphas[i] = np.dot(s_i, q) / yy
            q = q - alphas[i] * y_i
    
    s_last = s_hist[-1]
    y_last = y_hist[-1]
    yy_last = np.dot(y_last, y_last)
    if yy_last > 1e-30:
        gamma = max(1e-8, np.dot(s_last, y_last) / yy_last)
    else:
        gamma = 1.0
    r = gamma * q
    
    for i in range(m_used):
        s_i = s_hist[i]
        y_i = y_hist[i]
        yy = np.dot(y_i, y_i)
        if yy < 1e-30:
            continue
        beta_i = np.dot(s_i, y_i) / yy
        r = r + (alphas[i] - beta_i) * y_i
    
    direction = -r
    if np.dot(g, direction) > 0:
        direction = -g
    return direction


def safe_normalize(a, n, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if
