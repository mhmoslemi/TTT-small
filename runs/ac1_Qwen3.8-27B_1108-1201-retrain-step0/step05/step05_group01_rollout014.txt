    c = np.real(ifft(A * A))

"""
Minimize 2*n*max_conv/sum^2 using log-domain FISTA with FFT-accelerated gradients,
combined with active-set polishing and multi-size exploration.
Key innovation: log-parameterization a=exp(x) improves conditioning and removes
the non-negativity constraint from the optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def cross_corr_w_a(w, a):
    """Compute c_j = sum_{k=0}^{n-1} w[j+k] * a_k for j=0,...,n-1.
    w has length 2n-1, a has length n."""
    n = len(a)
    L = 1 << (3 * n - 2).bit_length()
    w_padded = np.zeros(3 * n - 2)
    w_padded[:len(w)] = w
    w_flip = w_padded[::-1]
    conv = np.real(ifft(fft(w_flip, L) * fft(a, L)))
    # c_j = conv[3n-3-j]
    indices = [3 * n - 3 - j for j in range(n)]
    return conv[indices]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_weights(c, beta):
    """Compute LSE weights."""
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / tot


def run_log_fista(a_init, max_time, rng, beta_start=20.0, lambda_pen=1.0,
                  mode="adaptive"):
    """FISTA in log-domain with adaptive strategy."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    
    a = np.clip(np.asarray(a_init, dtype=np.float64), 1e-15, 1000.0).copy()
    a = a * (S / a.sum())
    x = np.log(a)
    
    y_x = x.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    stuck = 0
    
    print("    Log-FISTA start: obj=%.8f" % best_obj)
    
    while time.monotonic() - start < max_time:
        it += 1
        
        a_y = np.exp(np.clip(y_x, -50, 10))
        a_sum = a_y.sum()
        
        b = fft_convolve_self(a_y)
        b_max = b.max()
        obj = 2.0 * n * b_max / (a_sum * a_sum)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_y.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Adaptive beta and mode
        if mode == "adaptive":
            if stuck < 200:
                beta = beta_start * (1.0 + 0.005 * it)
                w = lse_weights(b, min(beta, 5000.0))
            elif stuck < 1000:
                beta = beta_start * (1.0 + 0.005 * it) * 2.0
                # Blend LSE and top-8
                w_lse = lse_weights(b, min(beta, 10000.0))
                k = 8
                idx = np.argsort(b)[::-1][:k]
                w_top = np.zeros(len(b))
                w_top[idx] = 1.0 / k
                blend = min(0.3, 0.0005 * stuck)
                w = (1 - blend) * w_lse + blend * w_top
            else:
                # Focus on top peaks
                k = min(32, 8 + stuck // 200)
                idx = np.argsort(b)[::-1][:k]
                vals = b[idx]
                pw = np.power(np.maximum(vals, 1e-30), 1.5)
                w = np.zeros(len(b))
                w[idx] = pw / pw.sum()
        elif mode == "lse":
            beta = beta_start * (1.0 + 0.002 * it)
            w = lse_weights(b, min(beta, 20000.0))
        else:  # topk
            k = min(16, max(4, n // 256))
            idx = np.argsort(b)[::-1][:k]
            w = np.zeros(len(b))
            w[idx] = 1.0 / k
        
        # Gradient in x-domain
        cc = cross_corr_w_a(w, a_y)
        g_x = 2.0 * a_y * cc
        
        # Penalty for sum constraint
        lambda_eff = lambda_pen * (1.0 + 0.005 * stuck)
        g_x += lambda_eff * (a_sum - S) * a_y
        
        gnorm = np.linalg.norm(g_x)
        if gnorm < 1e-15:
            break
        
        # Step size: use the Lipschitz constant estimate
        # The gradient of f in x-domain has norm related to ||a * cc||
        # Use a conservative step
        step = min(2.0 / (gnorm + 1e-10), 1.0 / (S * np.max(np.abs(cc)) + 1e-10))
        step = max(step, 1e-10)
        
        # Damped step when stuck
        if stuck > 1000:
            step *= 0.5
        
        # FISTA update
        x_new = y_x - step * g_x
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_x = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        y_x = np.clip(y_x, -50, 10)
        t_mom = t_new
        x = x_new
        x = np.clip(x, -50, 10)
        
        # Momentum reset
        if it % 3000 == 0:
            t_mom = 1.0
        
        # Restart noise when very stuck
        if stuck > 5000 and it % 1000 == 0:
            noise_scale = 0.005 * np.exp(-stuck / 10000.0)
            y_x += rng.normal(0, noise_scale, n)
            y_x = np.clip(y_x, -50, 10)
            t_mom = 1.0
            stuck = 0
        
        if it % 5000 == 0:
            print("      iter=%d, obj=%.8f, stuck=%d, step=%.2e" % (it, best_obj, stuck, step))
    
    # Final conversion
    a_final = np.exp(np.clip(np.log(np.maximum(best_a, 1e-30)), -50, 10))
    a_final = a_final * (S / a_final.sum())
    obj_final = compute_obj(a_final)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a_final
    
    print("    Log-FISTA done: obj=%.8f (iters=%d)" % (best_obj, it))
    return best_a, best_obj
