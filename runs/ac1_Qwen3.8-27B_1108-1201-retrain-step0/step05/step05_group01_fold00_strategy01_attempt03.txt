    result_val, best_val = run_fista_dynamic(a_val, max_time)
    S_val = (run_lbfgs_penalty(result_val, max_time)) or best_val  # choose if both are valid
    
    # Use fallback if either method fails to find better than parent.
    fallback = np.ones(8192) * S_val
    fallback_score = evaluate_safe(fallback)
    if fallback_score >= best_val and fallback_score >= favorites.best:
        # Fallback is not worse, use it for future restart.
        return fallback, fallback_score
        
    # Multi-start with variations
    variations = []
    for _ in range(3):  # generate diverse starting points
        a_var_scaled = a_val * (1 + 0.2*(np.random.rand(n_val)-0.5))
        a_var_scaled = project_simplex(a_var_scaled, S_val)
        variations.append(a_var_scaled)
    
    # Sample the best three variations and combine them
    # due to time constraints in implementation, avoid full retraining.
    best_variation = None
    best_score = float('inf')
    for v in variations:
        score = evaluate_safe(v)
        if score < best_score:
            best_score = score
            best_variation = v
            
    # Use best variation as candidate if better than fallback evolution.
    # Final validation and fallback protection.
    validate_candidate(best_variation)
    return best_variation, best_score

best_seq, best_score = main_search()
print(f"Best sequence found: length={len(best_seq)}, score={best_score}")
# Handle abrupt termination if time exceeds 2400 seconds
# Evaluate final best_seq against fallback
final_score = evaluate_safe(best_seq)
if np.isfinite(final_score) and final_score < best_score * 0.95:
    best_seq = best_seq
else:
    # Use fallback as default
    best_seq = np.ones(8192) * S_val  # already normalized S_val

# Return normalized sequence of 8192 tokens (ensuring finite and non-negative)
final_sequence = best_seq.tolist()
validate_candidate(final_sequence)  # double-check normalization
return final_sequence
