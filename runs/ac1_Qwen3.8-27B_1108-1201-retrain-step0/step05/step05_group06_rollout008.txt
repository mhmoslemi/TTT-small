"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase optimization with aggressive LSE smoothing, exact-max polishing,
and perturb-and-reoptimize for escaping local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, tol_frac=0.0):
    """Top-k gradient."""
    n = len(y)
    k = min(k, len(c))
    if tol_frac > 0:
        threshold = c.max() * (1.0 - tol_frac)
        idx = np.where(c >= threshold)[0]
        if len(idx) > k:
            top_vals = np.sort(c)[::-1][:k]
            threshold = top_vals[-1]
            idx = np.where(c >= threshold)[0]
    else:
        idx = np.argsort(c)[::-1][:k]
    w = np.ones(len(idx)) / len(idx)
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_lse_fista(a_init, max_time, rng, beta_start=10.0, beta_end=500.0,
                  momentum_restart=200, noise_every=3000):
    """LSE FISTA with multiplicative beta schedule and adaptive momentum restart."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_start
    # Compute rate for multiplicative schedule
    # We want beta to reach beta_end in roughly max_time seconds
    # Estimate iterations per second ~ 500 for n=8192 (rough)
    est_iters = int(max_time * 400)
    if est_iters < 100:
        est_iters = 100
    beta_mult = (beta_end / beta_start) ** (1.0 / est_iters)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Update beta (multiplicative)
        if it % 3 == 0:
            beta = min(beta_end, beta * beta_mult)

        # Extra speedup of beta when stuck
        if stuck > 1000:
            beta = min(beta_end, beta * 1.01)

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: conservative Lipschitz-based
        step = 0.9 / max(2.0 * beta * np.max(a) + 4.0 * np.max(a), 1.0)
        step = max(step, 1e-14)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Momentum restart
        if it % momentum_restart == 0:
            t_mom = 1.0

        # Noise injection when very stuck
        if stuck > noise_every and it % 100 == 0:
            noise_scale = max(1e-7, 0.003 * S / np.sqrt(n) * np.exp(-stuck / (noise_every * 3)))
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it

    c_final = fft_convolve_self(best_a)
    return best_a


def run_exact_max_polish(a_init, max_time, rng, initial_step_frac=0.05):
    """Exact max subgradient descent with Nesterov momentum and progressive step decay."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive: use top-1 when not stuck, top-few when stuck
        if stuck < 500:
            grad = exact_subgradient(y, c, tol_frac=0.0)
        elif stuck < 2000:
            grad = topk_gradient(y, c, k=3)
        else:
            grad = topk_gradient(y, c, k=8, tol_frac=0.005)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Progressive step decay
        progress = min(1.0, it / (max_time * 300))
        step_frac = initial_step_frac * (1.0 - 0.8 * progress)
        step = step_frac * S / max(gnorm, 1e-15)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Frequent momentum restart
        if it % 100 == 0:
            t_mom = 1.0

        # Noise to escape when stuck
        if stuck > 3000 and it % 200 == 0:
            noise_scale = max(1e-8, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 10000))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it

    c_final = fft_convolve_self(best_a)
    return best_a


def run_hybrid_polish(a_init, max_time, rng):
    """Alternates between LSE (moderate beta) and exact max for final polish."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    cycle = 0

    while time.monotonic() - start < max_time:
        it += 1
        cycle = (it // 50) % 3  # 0: LSE, 1: exact, 2: top-k

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        if cycle == 0:
            beta = 100.0 + 0.1 * it
            grad = lse_gradient(y, c, min(beta, 2000.0))
        elif cycle == 1:
            grad = exact_subgradient(y, c, tol_frac=0.0)
        else:
            k = min(5, 2 + stuck // 1000)
            grad = topk_gradient(y, c, k=k)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 0.03 * S / max(gnorm, 1e-15)
        step = min(step, 0.5 / (2.0 * 100.0 * np.max(a) + 4.0 * np.max(a) + 1.0))
        step = max(step, 1e-14)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 150 == 0:
            t_mom = 1.0

        if stuck > 5000 and it % 300 == 0:
            noise_scale = max(1e-9, 0.0005 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    return best_a


def make_diverse_init(n, rng):
    """Create diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    # Cosine
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * tt))
    # Raised cosine window
    inits.append(0.5 * (1 + np.cos(2 * np.pi * tt)))
    # Gaussian-like
    inits.append(np.exp(-8 * (tt - 0.5) ** 2))
    # Two-hump
    inits.append(0.5 + 0.5 * np.cos(4 * np.pi * tt))
    # Random
    r = rng.random(n)
    inits.append(r ** 2)
    # sawtooth-like
    inits.append(np.abs(2 * tt - 1))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: diverse init -> LSE FISTA (low beta) -> LSE FISTA (high beta)
    -> hybrid polish -> exact max polish -> perturb & reoptimize -> final.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    # Leave 90s buffer for final evaluation and safety
    deadline = time.monotonic() + max(1.0, budget_s - 90.0)

    global_best_obj = np.inf
    global_best_a = None

    # Try to get the provided sequence
    a_start = None
    try:
        a_start = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    # Determine base n
    if a_start is not None:
        n0 = len(a_start)
    else:
        n0 = 8192

    print("Base n = %d" % n0)

    n = n0
    S = np.sqrt(2.0 * n)

    # Generate and evaluate diverse initializations
    inits = []
    names = []

    if a_start is not None:
        a_norm = normalize_seq(a_start, S)
        inits.append(a_norm)
        names.append("provided")

    for i, init in enumerate(make_diverse_init(n, rng)):
        inits.append(normalize_seq(init, S))
        names.append("init_%d" % i)

    print("\n=== Initialization evaluation ===")
    best_init = None
    best_init_obj = np.inf
    for a_init, name in zip(inits, names):
        obj = compute_obj(a_init)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_init.copy()
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a_init.copy()
        print("  %s: obj=%.8f" % (name, obj))

    print("Best init: %s obj=%.8f" % (names[np.argbest_init_obj if False else [names.index(names[x]) for x in names if x == "provided"][0] if "provided" in names else 0]], best_init_obj))

    # === Phase 1: LSE FISTA, low beta range ===
    remaining = deadline - time.monotonic()
    p1_budget = remaining * 0.18
    if p1_budget > 10:
        print("\n=== Phase 1: LSE FISTA beta 10->200 (budget %.0fs) ===" % p1_budget)
        t1 = time.monotonic()
        a_p1 = run_lse_fista(best_init, p1_budget, rng,
                             beta_start=10.0, beta_end=200.0,
                             momentum_restart=150, noise_every=2000)
        obj_p1 = compute_obj(a_p1)
        print("  Phase 1: obj=%.8f (t=%.1fs)" % (obj_p1, time.monotonic() - t1))
        if obj_p1 < global_best_obj:
            global_best_obj = obj_p1
            global_best_a = a_p1.copy()

    # === Phase 2: LSE FISTA, high beta range ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.28
    if p2_budget > 10:
        print("\n=== Phase 2: LSE FISTA beta 200->3000 (budget %.0fs) ===" % p2_budget)
        t2 = time.monotonic()
        a_p2 = run_lse_fista(global_best_a, p2_budget, rng,
                             beta_start=200.0, beta_end=3000.0,
                             momentum_restart=100, noise_every=3000)
        obj_p2 = compute_obj(a_p2)
        print("  Phase 2: obj=%.8f (t=%.1fs)" % (obj_p2, time.monotonic() - t2))
        if obj_p2 < global_best_obj:
            global_best_obj = obj_p2
            global_best_a = a_p2.copy()

    # === Phase 3: Hybrid polish (alternating LSE/exact/topk) ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.22
    if p3_budget > 15:
        print("\n=== Phase 3: Hybrid polish (budget %.0fs) ===" % p3_budget)
        t3 = time.monotonic()
        a_p3 = run_hybrid_polish(global_best_a, p3_budget, rng)
        obj_p3 = compute_obj(a_p3)
        print("  Phase 3: obj=%.8f (t=%.1fs)" % (obj_p3, time.monotonic() - t3))
        if obj_p3 < global_best_obj:
            global_best_obj = obj_p3
            global_best_a = a_p3.copy()

    # === Phase 4: Perturb and re-optimize ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.18
    if p4_budget > 30:
        print("\n=== Phase 4: Perturb & reoptimize (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        n_attempts = 5
        per_attempt = p4_budget / n_attempts - 2
        for attempt in range(n_attempts):
            if time.monotonic() > deadline - 30:
                break
            # Decreasing noise scale
            noise_scale = (0.008 - attempt * 0.001) * S / np.sqrt(n)
            a_perturbed = project_simplex(global_best_a + rng.normal(0, max(noise_scale, 1e-7), n), S)
            # Re-optimize with moderate LSE
            a_re = run_lse_fista(a_perturbed, per_attempt, rng,
                                 beta_start=50.0, beta_end=1000.0,
                                 momentum_restart=100, noise_every=5000)
            obj_re = compute_obj(a_re)
            print("  Attempt %d: obj=%.8f (noise=%.4f)" % (attempt + 1, obj_re, noise_scale))
            if obj_re < global_best_obj:
                global_best_obj = obj_re
                global_best_a = a_re.copy()
        print("  Phase 4 total time: %.1fs" % (time.monotonic() - t4))

    # === Phase 5: Final exact-max polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p5_budget = remaining * 0.75
        print("\n=== Phase 5: Final exact-max polish (budget %.0fs) ===" % p5_budget)
        t5 = time.monotonic()
        a_p5 = run_exact_max_polish(global_best_a, p5_budget, rng,
                                    initial_step_frac=0.03)
        obj_p5 = compute_obj(a_p5)
        print("  Phase 5: obj=%.8f (t=%.1fs)" % (obj_p5, time.monotonic() - t5))
        if obj_p5 < global_best_obj:
            global_best_obj = obj_p5
            global_best_a = a_p5.copy()

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL RESULT ===")
            print("Objective: %.8f (n=%d)" % (ver, len(final_list)))
            print("Total time: %.0fs" % (time.monotonic() - (deadline - budget_s + 90.0)))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]
    else:
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
