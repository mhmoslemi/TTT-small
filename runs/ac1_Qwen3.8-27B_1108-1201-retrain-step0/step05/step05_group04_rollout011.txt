"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: LP-based active set method combined with LSE-FISTA refinement.
Key innovations:
1. LP-based direction finding (solve max sum(g) s.t. f*g <= M, g>=0)
2. Adaptive active set management
3. Multi-scale exploration (different n values)
4. High-precision final refinement with direct subgradient
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    e = np.minimum(e, 1e300)  # Prevent overflow
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def lp_direction(f, M, solver='MOSEK'):
    """
    Solve: maximize sum(g) subject to (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g.
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix: C[k, j] = f[j] if 0 <= k-j < n
    # This is a banded matrix
    # For efficiency, build it as a sparse matrix
    import scipy.sparse as sp
    
    # The convolution matrix: row k, col j has entry f[j] if 0 <= k-j < n, i.e., k-n+1 <= j <= k
    # Size: (2n-1) x n
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            data.extend(f[lo:hi+1].tolist())
    
    C = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    
    objective = cp.Maximize(cp.sum(g))
    constraints = [C @ g <= M]
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if g.value is not None:
            return np.asarray(g.value).flatten()
    except Exception as e:
        print("  LP solve failed: %s" % str(e))
    
    # Fallback to other solvers
    for sol in ['GUROBI', 'CLARABEL', 'SCS', 'ECOS']:
        try:
            prob.solve(solver=sol)
            if g.value is not None:
                return np.asarray(g.value).flatten()
        except:
            pass
    
    return None

def lp_direction_active_set(f, M, active_lags, solver='MOSEK'):
    """
    Solve LP with only active lag constraints (faster).
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    constraints = []
    for k in active_lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        constraints.append(f[lo:hi+1] @ g[js] <= M)
    
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if g.value is not None:
            return np.asarray(g.value).flatten()
    except:
        pass
    
    for sol in ['GUROBI', 'CLARABEL', 'SCS']:
        try:
            prob.solve(solver=sol)
            if g.value is not None:
                return np.asarray(g.value).flatten()
        except:
            pass
    
    return None

    c = fft_convolve_self(f)
    M = c.max()

    threshold = M * (1 - 1e-6)
    active = np.where(c >= threshold)[0]
    if len(active) > active_set_size:
        # Take top active_set_size lags
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:active_set_size]
    
    print("  Active set size: %d, M = %.8f" % (len(active), M))
    
    g = lp_direction_active_set(f, M, active.tolist())
    if g is None:
        return f, M
    
    g = np.clip(g, 0, None)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return f, M
    
    # Rescale g to have sum S
    g = g * (S / g_sum)
    
    # Check new objective
    c_new = fft_convolve_self(g)
    M_new = c_new.max()
    print("  After LP: M_new = %.8f (was %.8f)" % (M_new, M))
    
    if M_new < M:
        return g, M_new
    else:
        # LP didn't help, try a blend
        for t in [0.5, 0.3, 0.1]:
            f_blend = (1-t) * f + t * g
            f_blend = normalize_sum(f_blend, n)
            c_blend = fft_convolve_self(f_blend)
            M_blend = c_blend.max()
            if M_blend < M:

        return f, M

def run_lse_fista(a_init, n, S, max_time, rng,
                  beta_start=50.0, beta_end=20000.0, beta_steps=200,
                  momentum_restart_gap=500, noise_decay=80000.0):
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 50000.0)
        
        c = fft_convolve_self(y)
        cur_true_max = c.max()
        
        if cur_true_max < best:
            best = cur_true_max
            best_a = y.copy()
            last_improve_it = it
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / beta
        
        stagnant = it - last_improve_it
        if stagnant > momentum_restart_gap:
            t_mom = 1.0
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stagnant / (noise_decay / n))
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve_it = it
            step *= 0.3
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)

def direct_subgradient_refine(a, n, S, max_time, rng, n_top=10):
    """
    Direct subgradient method: use hard active set (top n_top lags) with
    very small steps and no momentum for precise final optimization.
    """
    a = normalize_sum(a, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Hard active set: top n_top lags
        top_idx = np.argsort(c)[::-1][:n_top]
        # Weight by how close to max
        weights = np.zeros(len(c))
        for k in top_idx:
            weights[k] = 1.0 / n_top
        weights /= weights.sum()
        
        # Gradient: 2 * (weights * a_convolution)
        fsize = 1 << (len(weights) + n - 1).bit_length()
        W = fft(weights, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Very small step with backtracking
        step = 0.001
        base = cur_max
        improved = False
        for _ in range(10):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < base * (1 - 1e-8):
                a = a_test
                improved = True
                break
            step *= 0.5
        
        if not improved and it % 500 == 0:
            # Add tiny noise to escape
            noise = 0.0001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)
        
        it += 1
    
    return best_a, best

def peak_shaving(a, n, S, n_peaks=5, n_steps=50, lr_scale=0.01):
    a = a.copy()
    for _ in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()
        top_idx = np.argsort(c)[::-1][:n_peaks]
        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            weight = max(0, c[k] - cm * 0.99)
            np.add.at(resp, js, contrib * weight)
        resp_max = resp.max()
        if resp_max < 1e-30:
            break
        lr = lr_scale * S / (n * resp_max)
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)
    return a

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: LP-based active-set direction finding combined with LSE-FISTA and
direct subgradient refinement. The LP solves max sum(g) s.t. (f*g)[k]<=M, g>=0
to find globally-optimal directions that the gradient method might miss.
Multi-scale exploration across different sequence lengths.
"""
import numpy as np
import time
import cvxpy as cp
import scipy.sparse as sp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    shifted = c - cm
    e = np.exp(np.minimum(beta * shifted, 50.0))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def hard_active_gradient(a, c, n_top=20):
    """Gradient using hard active set (top lags evenly weighted)."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:n_top]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / n_top
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def build_conv_matrix_row(f, k, n):
    """Get the row k of the convolution matrix (sparse)."""
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    js = np.arange(lo, hi + 1)
    return js, f[lo:hi + 1]


def solve_lp_direction(f, M, active_lags, timeout=30):
    """
    Solve LP: maximize sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0.
    Returns optimal g or None on failure.
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in active_lags:
        js, coeffs = build_conv_matrix_row(f, k, n)
        if len(js) > 0:
            constraints.append(coeffs @ g[js] <= M)

    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)

    solvers = ['MOSEK', 'GUROBI', 'CLARABEL', 'ECOS', 'SCS']
    start_t = time.monotonic()
    for solver in solvers:
        if time.monotonic() - start_t > timeout:
            break
        try:
            prob.solve(solver=solver)
            if g.value is not None and np.sum(g.value) > 0:
                return np.asarray(g.value).flatten()
        except Exception:
            continue
    return None


def lp_iteration(f, n, S, active_set_size=30, timeout=45):
    """
    One LP-based iteration. Returns (new_f, improved).
    """
    c = fft_convolve_self(f)
    M = c.max()

    # Determine active set
    top_idx = np.argsort(c)[::-1][:active_set_size]
    # Also include any lags within 1e-8 of max
    near_max = np.where(c >= M - 1e-10 * M)[0]
    active = np.unique(np.concatenate([top_idx, near_max]))
    if len(active) > 50:
        active = top_idx[:50]

    g = solve_lp_direction(f, M, active.tolist(), timeout=timeout)
    if g is None:
        return f, False

    g = np.clip(g, 0, None)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return f, False

    # Rescale g to sum S
    g_rescaled = g * (S / g_sum)
    c_new = fft_convolve_self(g_rescaled)
    M_new = c_new.max()

    if M_new < M * (1 - 1e-10):
        return g_rescaled, True
    else:
        # Try blends
        best_blend = f
        best_val = M
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            blend = (1 - t) * f + t * g_rescaled
            blend = normalize_sum(blend, n)
            c_blend = fft_convolve_self(blend)
            if c_blend.max() < best_val:
                best_val = c_blend.max()
                best_blend = blend.copy()
        return best_blend, best_val < M


def run_lse_fista(a_init, n, S, max_time, rng,
                  beta_start=50.0, beta_end=30000.0, beta_steps=200,
                  momentum_restart_gap=300):
    """FISTA with adaptive LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 80000.0)

        c = fft_convolve_self(y)
        cur_true_max = c.max()

        if cur_true_max < best:
            best = cur_true_max
            best_a = y.copy()
            last_improve_it = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta
        stagnant = it - last_improve_it
        if stagnant > momentum_restart_gap:
            t_mom = 1.0
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-stagnant / 5000.0)
            if noise_scale > 1e-9:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_improve_it = it
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def direct_subgradient_refine(a, n, S, max_time, rng, n_top=20):

    Direct subgradient with backtracking and periodic restarts.
    Uses hard active set for very precise final optimization.
    """
    a = normalize_sum(a, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best:
            best = cur_max
            best_a = a.copy()

        # Use weighted active set
        top_idx = np.argsort(c)[::-1][:n_top]
        # Weight by exponential of how close to max
        vals = c[top_idx]
        wts = np.exp(-10.0 * (cur_max - vals) / max(cur_max * 1e-6, 1e-15))
        wts = wts / wts.sum()

        w = np.zeros(len(c))
        w[top_idx] = wts
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Backtracking line search
        step = 0.01 / (1 + 0.001 * it)
        improved = False
        for _ in range(15):
            a_test = project_simplex(a - step * grad, S)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur_max * (1 - 1e-9):
                a = a_test
                improved = True
                break
            step *= 0.5

        # Periodic noise to escape local minima
        if it % 1000 == 0 and it > 0:
            noise_scale = 0.0005 * S / np.sqrt(n) * np.exp(-it / 20000.0)
            if noise_scale > 1e-12:
                a_noisy = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c_noisy = fft_convolve_self(a_noisy)
                if c_noisy.max() < best:
                    a = a_noisy

        it += 1

    return best_a, best


def peak_shaving_direct(a, n, S, n_steps=300, n_peaks=10):
    """
    Direct peak shaving: for each top peak, reduce the contributing pairs.
    Uses multiplicative updates with adaptive learning rate.
    """
    a = a.copy()
    for step_i in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()
        top_idx = np.argsort(c)[::-1][:n_peaks]

        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            # Weight by excess over 99.5% of max
            weight = max(0, c[k] - cm * 0.995)
            np.add.at(resp, js, contrib * weight)

        resp_max = resp.max()
        if resp_max < 1e-30:
            break

        # Adaptive learning rate
        lr = 0.005 * S / (n * resp_max) * (1.0 + 0.1 * np.sin(step_i * 0.01))
        a = a * np.exp(-lr * resp)
        a = project_simplex(a, S)

    return a


def pairwise_swap_refine(a, n, S, max_time, rng):
    """
    Local refinement: try swapping mass between pairs of elements to reduce peaks.
    If moving mass from i to j (or vice versa) reduces the max convolution, do it.
    """
    a = normalize_sum(a, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    c = fft_convolve_self(a)
    cm = c.max()
    if cm < best:
        best = cm
        best_a = a.copy()

    # Identify elements near peaks
    top_peaks = np.argsort(c)[::-1][:20]

    while time.monotonic() - start < max_time:
        improved = False
        # For each top peak, try to reduce it by adjusting contributing elements
        for k in top_peaks:
            if time.monotonic() - start > max_time:
                break
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]

            # Find the pair with maximum contribution
            if len(js) == 0:
                continue
            max_pair_idx = np.argmax(contrib)
            i = js[max_pair_idx]
            j = partners[max_pair_idx]

            if i == j:
                continue

            # Try moving small amount from larger to smaller
            if a[i] > a[j]:
                for frac in [0.01, 0.005, 0.001]:
                    delta = frac * (a[i] - a[j])
                    if delta < 1e-15:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    a_test = np.maximum(a_test, 0)
                    a_test = normalize_sum(a_test, n)
                    c_test = fft_convolve_self(a_test)
                    if c_test.max() < best * (1 - 1e-10):
                        a = a_test
                        best = c_test.max()
                        best_a = a.copy()
                        improved = True
                        c = c_test
                        top_peaks = np.argsort(c)[::-1][:20]
                        break
            else:
                for frac in [0.01, 0.005, 0.001]:
                    delta = frac * (a[j] - a[i])
                    if delta < 1e-15:
                        continue
                    a_test = a.copy()
                    a_test[j] -= delta
                    a_test[i] += delta
                    a_test = np.maximum(a_test, 0)
                    a_test = normalize_sum(a_test, n)
                    c_test = fft_convolve_self(a_test)
                    if c_test.max() < best * (1 - 1e-10):
                        a = a_test
                        best = c_test.max()
                        best_a = a.copy()
                        improved = True
                        c = c_test
                        top_peaks = np.argsort(c)[::-1][:20]
                        break

        # Periodic noise
        if not improved:
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-10 * (time.monotonic() - start) / max_time)
            if noise_scale > 1e-10:
                a_test = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < best:
                    a = a_test
                    best = c_test.max()
                    best_a = a.copy()
            else:
                break

    return best_a, best


def generate_bump_init(n, center=0.4, width=0.35, sharpness=1.0):
    """Generate a smooth bump initialization with asymmetry."""
    t = np.linspace(0, 1, n)
    x = (t - center) / width
    a = np.exp(-sharpness * x * x)
    # Add slight asymmetry
    a = a * (1.0 + 0.1 * np.sin(np.pi * t))
    return np.maximum(a, 1e-10)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(f*f)/sum(f)^2.
    Uses LP-based direction finding, LSE-FISTA, direct subgradient,
    peak shaving, and pairwise swap refinement.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    print("Budget: %.0fs, deadline in %.0fs" % (budget_s, budget_s - safety))

    # Load previous best
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded previous best: length=%d" % len(given))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    def try_sequence(a):
        nonlocal global_best, global_a
        n_local = len(a)
        S_local = np.sqrt(2.0 * n_local)
        a_norm = normalize_sum(a, n_local)
        c = fft_convolve_self(a_norm)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a_norm.copy()
            print("  ** New best: %.10f (n=%d) **" % (val, n_local))
            return True
        return False

    # === Phase 1: Quick eval + LSE on given ===
    if given is not None:
        try_sequence(given)
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        t1 = min(45.0, (deadline - time.monotonic()) * 0.08)
        print("=== Phase 1: LSE polish (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_lse_fista(given, n_g, S_g, t1, rng,
                                beta_start=200.0, beta_end=10000.0, beta_steps=50)
        try_sequence(a1)

    # === Phase 2: LP-based iterations ===
    if global_a is not None and (deadline - time.monotonic()) > 100:
        n_lp = len(global_a)
        S_lp = np.sqrt(2.0 * n_lp)
        remaining = deadline - time.monotonic()
        t2 = min(240.0, remaining * 0.2)
        print("=== Phase 2: LP iterations (n=%d, budget=%.0fs) ===" % (n_lp, t2))

        f = global_a.copy()
        start_lp = time.monotonic()
        for lp_iter in range(8):
            if time.monotonic() - start_lp > t2:
                break
            remaining_lp = t2 - (time.monotonic() - start_lp)
            if remaining_lp < 20:
                break
            timeout_lp = min(45.0, remaining_lp * 0.8)
            print("  LP iteration %d (timeout=%.0fs)..." % (lp_iter + 1, timeout_lp))
            f_new, improved = lp_iteration(f, n_lp, S_lp,
                                            active_set_size=25,
                                            timeout=timeout_lp)
            if improved:
                f = f_new
                try_sequence(f)
            else:
                # Try with smaller active set
                f_new2, improved2 = lp_iteration(f, n_lp, S_lp,
                                                  active_set_size=10,
                                                  timeout=timeout_lp * 0.7)
                if improved2:
                    f = f_new2
                    try_sequence(f)
                else:
                    # LP not helping, add noise and try again
                    noise = 0.002 * S_lp / np.sqrt(n_lp)
                    f = project_simplex(f + rng.normal(0, noise, n_lp), S_lp)
                    try_sequence(f)

    # === Phase 3: Main LSE-FISTA with aggressive schedule ===
    if global_a is not None:
        n_main = len(global_a)
        S_main = np.sqrt(2.0 * n_main)
        remaining = deadline - time.monotonic()
        t3 = remaining * 0.30
        print("=== Phase 3: Main LSE-FISTA (n=%d, %.0fs) ===" % (n_main, t3))
        a3, v3 = run_lse_fista(global_a, n_main, S_main, t3, rng,
                                beta_start=30.0, beta_end=50000.0, beta_steps=150,
                                momentum_restart_gap=200)
        try_sequence(a3)

    # === Phase 4: Try different lengths ===
    if global_a is not None and (deadline - time.monotonic()) > 80:
        for n_try in [4096, 16384]:
            if time.monotonic() > deadline - 60:
                break
            S_try = np.sqrt(2.0 * n_try)
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n_try)
            init_try = np.interp(t_new, t_old, global_a)
            init_try = normalize_sum(init_try, n_try)
            # Quick eval of interpolation
            try_sequence(init_try)

            remaining = deadline - time.monotonic()
            t4 = min(90.0, remaining * 0.12)
            if t4 < 15:
                continue
            print("=== Phase 4: n=%d (%.0fs) ===" % (n_try, t4))
            a4, v4 = run_lse_fista(init_try, n_try, S_try, t4, rng,
                                    beta_start=30.0, beta_end=20000.0, beta_steps=80)
            try_sequence(a4)

    # === Phase 5: Peak shaving ===
    if global_a is not None and (deadline - time.monotonic()) > 15:
        n_p = len(global_a)
        S_p = np.sqrt(2.0 * n_p)
        print("=== Phase 5: Peak shaving (n=%d) ===" % n_p)
        a_p = peak_shaving_direct(global_a, n_p, S_p, n_steps=200, n_peaks=12)
        try_sequence(a_p)
        # Quick LSE after shaving
        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_p2, _ = run_lse_fista(a_p, n_p, S_p, min(20.0, remaining * 0.4), rng,
                                     beta_start=2000.0, beta_end=50000.0, beta_steps=30)
            try_sequence(a_p2)

    # === Phase 6: Pairwise swap refinement ===
    if global_a is not None and (deadline - time.monotonic()) > 20:
        n_sw = len(global_a)
        S_sw = np.sqrt(2.0 * n_sw)
        t6 = min(60.0, (deadline - time.monotonic()) * 0.25)
        print("=== Phase 6: Pairwise swap (n=%d, %.0fs) ===" % (n_sw, t6))
        a_sw, v_sw = pairwise_swap_refine(global_a, n_sw, S_sw, t6, rng)
        try_sequence(a_sw)

    # === Phase 7: Final direct subgradient ===
    if global_a is not None and (deadline - time.monotonic()) > 10:
        n_f = len(global_a)
        S_f = np.sqrt(2.0 * n_f)
        t7 = max(5.0, (deadline - time.monotonic()) * 0.5)
        print("=== Phase 7: Direct subgradient (n=%d, %.0fs) ===" % (n_f, t7))
        a_f, v_f = direct_subgradient_refine(global_a, n_f, S_f, t7, rng, n_top=25)
        try_sequence(a_f)

    # === Phase 8: One more LP attempt on final best ===
    if global_a is not None and (deadline - time.monotonic()) > 30:
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * n_final)
        remaining = deadline - time.monotonic()
        print("=== Phase 8: Final LP attempt (remaining=%.0fs) ===" % remaining)
        f_final, improved = lp_iteration(global_a, n_final, S_final,
                                          active_set_size=15,
                                          timeout=min(40.0, remaining * 0.6))
        if improved:
            try_sequence(f_final)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.10f, internal = %.10f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result found, using uniform fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.10f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
