Strategy: Exact subgradient method with backtracking line search,
adaptive active set, periodic perturbations, and multi-scale exploration.
Key improvement over previous: uses exact (unsmoothed) subgradient with
proper line search for the "last mile" optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_full(a):
    """Compute full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    # Quick check
    if v.sum() == S and np.all(v >= -1e-15):
        return np.maximum(v, 0.0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, c, k_active, rng=None):
    """
    Compute subgradient of max(c) where c = conv(a,a).
    Uses top k_active lags (or random selection if rng provided).
    Returns gradient vector of length n.
    """
    n = len(a)
    if rng is not None and k_active > 1:
        # Random selection from top 3*k_active
        top_3k = np.argsort(c)[::-1][:min(3 * k_active, len(c))]
        selected = rng.choice(top_3k, size=min(k_active, len(top_3k)), replace=False)
    else:
        selected = np.argsort(c)[::-1][:k_active]
    
    # g[j] = (1/k) * sum_{k in selected} 2 * a[k-j] (with a[m]=0 outside [0,n-1])
    # = (2/k) * sum_{k in selected} a[k-j]
    # This is (2/k) * (w ⋆ a)[j] where w is indicator of selected / k
    
    w = np.zeros(2 * n - 1)
    w[selected] = 1.0 / k_active
    
    # Cross-correlation: (w ⋆ a)[j] = sum_k w[k] * a[k-j]
    # FFT: ifft(conj(FFT(w)) * FFT(a))
    L = 1 << (2 * n - 1 + n - 1).bit_length()  # 3n-2
    W = fft(w, L)
    A = fft(a, L)
    g = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    
    return g


def subgradient_optimize(a_init, n, S, max_time, rng, 
                          k_start=1, k_max=10, 
                          step_scales=(0.1, 0.05, 0.01, 0.005, 0.001, 0.0005, 0.0001),
                          perturb_every=100, perturb_scale=0.01):
    """
    Subgradient method with backtracking for minimizing max(conv(a,a)).
    """
    a = project_simplex(a_init.copy(), S)
    c = fft_convolve_full(a)
    best = c.max()
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    stagnant = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_full(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Adaptive k: grow k_active when stagnating
        k = min(k_max, k_start + stagnant // 50)
        # Occasionally use random active set
        if rng.random() < 0.1:
            k = rng.integers(1, k_max + 1)
        
        # Compute subgradient
        g = compute_subgradient(a, c, k, rng)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        g = g / gnorm  # Normalize for step size control
        
        # Backtracking line search
        improved = False
        for scale in step_scales:
            step = scale * S / np.sqrt(n)
            a_new = project_simplex(a - step * g, S)
            c_new = fft_convolve_full(a_new)
            if c_new.max() < cur_max * (1 - 1e-9):
                a = a_new
                improved = True
                break
        
        if not improved:
            # Try very small step
            step = 1e-6 * S / np.sqrt(n)
            a = project_simplex(a - step * g, S)
        
        # Periodic perturbation
        if it % perturb_every == 0 and stagnant > 20:
            noise_scale = perturb_scale * S / np.sqrt(n) * np.exp(-stagnant / 500.0)
            if noise_scale > 1e-10:
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_a, best


def lse_optimize(a_init, n, S, max_time, rng,
                 beta_start=50.0, beta_end=20000.0, n_beta_steps=100):
    """
    LSE-FISTA optimization (similar to previous but with improvements).
    """
    a = project_simplex(a_init.copy(), S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / n_beta_steps)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        c = fft_convolve_full(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
        
        # LSE gradient
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            g = np.zeros(n)
        else:
            w = e / tot
            L = 1 << (2 * n - 1 + n - 1).bit_length()
            W = fft(w, L)
            A = fft(a, L)
            g = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / beta
        
        # Momentum restart
        if it - last_improve > 300:
            t_mom = 1.0
            noise = 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 2000.0)
            if noise > 1e-10:
                y = project_simplex(y + rng.normal(0, noise, n), S)
            step *= 0.5
            last_improve = it
        
        x_new = project_simplex(y - step * g, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    return best_a, best


def lp_active_set_step(a, c, n, S, n_active=50):
    """
    One step of the LP-based active set method.
    Solve: min M s.t. (b*a)[k] <= M for k in top n_active, b >= 0, sum(b) = S.
    Uses scipy linprog.
    """
    from scipy.optimize import linprog
    
    top_k = np.argsort(c)[::-1][:n_active]
    
    # Build constraint matrix: for each k in top_k, row is a[k-i] for i=0..n-1
    # (b*a)[k] = sum_i b[i] * a[k-i]
    A_ub = np.zeros((n_active, n))
    for idx, k in enumerate(top_k):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[idx, i] = a[j]
    
    # Constraints: A_ub @ b <= M (but M is also a variable)
    # Rewrite: A_ub @ b - M <= 0
    # Variables: [b_0, ..., b_{n-1}, M]
    A_full = np.zeros((n_active, n + 1))
    A_full[:, :n] = A_ub
    A_full[:, n] = -1.0  # -M
    
    b_full = np.zeros(n_active)  # A_full @ x <= b_full
    
    # Objective: minimize M = x[n]
    c_full = np.zeros(n + 1)
    c_full[n] = 1.0
    
    # Equality constraint: sum(b) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [S]
    
    # Bounds: b_i >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c_full, A_ub=A_full, b_ub=b_full,
                     A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n].copy()
    return None
