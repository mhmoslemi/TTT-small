"""

Combines LSE-FISTA with improved schedules, power-smoothing, spectral flattening

Key innovations: top-k gradient stability at high concentration, spectral peak

import numpy as np
from scipy.fft import fft, ifft
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Rescale to sum = sqrt(2n), clip to [0, 1000]."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE(c; beta)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm + np.log(tot) / beta


def topk_gradient(a, c, k=5, temperature=10.0):
    """
    Gradient using soft top-k weighting. More stable than LSE at high concentration.
    Weights are softmax over the top-k lags with given temperature.
    """
    n = len(a)
    # Get top k lags
    top_idx = np.argsort(c)[::-1][:k]
    # Soft weighting within top-k
    top_vals = c[top_idx]
    top_cm = top_vals.max()
    e = np.exp((top_vals - top_cm) * temperature)
    w = np.zeros_like(c)
    w[top_idx] = e / e.sum()
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, c.max()


def power_gradient(a, c, p):
    """Gradient of (sum c^p)^(1/p) with respect to a."""
    n = len(a)
    # Clip c to avoid overflow
    c_pos = np.maximum(c, 1e-30)
    obj = np.sum(c_pos ** p) ** (1.0 / p)
    if obj < 1e-30:
        return np.zeros(n), obj
    scale = obj ** (1.0 - p)
    w = p * np.power(c_pos, p - 1) * scale
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, obj

    This reduces the peak of the auto-convolution.
    """
    a = normalize_sum(a.copy(), n)
    S = np.sqrt(2.0 * n)
    fsize = 1 << (2 * n - 1 - 1).bit_length()

    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)

    # Target: geometric mean of magnitude (flat spectrum)
    log_mag = np.log(mag + 1e-30)
    target_log_mag = np.full_like(log_mag, np.mean(log_mag))

    for _ in range(n_iters):
        # Interpolate towards flat
        target_log_mag = (1 - alpha) * log_mag + alpha * np.mean(log_mag)
        target_mag = np.exp(target_log_mag)

        # GS iteration: alternate between magnitude constraint and time-domain non-neg
        for inner in range(5):
            A = target_mag * np.exp(phase)
            a_time = np.real(ifft(A, fsize))[:n]
            a_time = np.maximum(a_time, 0.0)
            A = fft(a_time, fsize)
            mag = np.abs(A)
            phase = np.angle(A)
            log_mag = np.log(mag + 1e-30)
            # Don't fully flatten - keep some of the current structure
            target_log_mag = 0.9 * log_mag + 0.1 * np.mean(log_mag)
            target_mag = np.exp(target_log_mag)

    a = normalize_sum(a_time, n)
    return a


def peak_shaving_v2(a, n, S, n_steps=100, rng=None):
    """
    Improved peak shaving: identifies top lags and uses
    multiplicative updates weighted by responsibility.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    a = a.copy()
    for step_i in range(n_steps):
        c = fft_convolve_self(a)
        cm = c.max()

        # Get top few peaks
        n_peek = min(8, len(c))
        top_idx = np.argsort(c)[::-1][:n_peek]

        # Compute responsibility
        resp = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo > hi:
                continue
            js = np.arange(lo, hi + 1)
            partners = k - js
            valid = (partners >= 0) & (partners < n)
            js = js[valid]
            partners = partners[valid]
            if len(js) == 0:
                continue
            contrib = a[js] * a[partners]
            weight = max(0.0, c[k] - cm * 0.98)
            resp[js] += contrib * weight

        resp_max = resp.max()
        if resp_max < 1e-30:
            break

        lr = 0.005 * S / (n * resp_max)
        a = a * np.exp(-lr * resp)

        # Small random perturbation to avoid getting stuck
        noise = 0.0005 * S / np.sqrt(n) * np.exp(-step_i / 50.0)
        if noise > 1e-10:
            a = a + rng.normal(0, noise, n)

        a = project_simplex(a, S)

    return a


def away_step_fw(a, g, S, t=0.1):
    """Away-step Frank-Wolfe update."""
    n = len(a)
    # Add step: where gradient is most negative
    i_star = np.argmin(g)
    # Away step: where a is largest and gradient is most positive
    j_candidates = np.argsort(g)[::-1][:10]
    j_candidates = j_candidates[a[j_candidates] > 0.01 * S / n]
    if len(j_candidates) > 0:
        j_star = j_candidates[0]
    else:
        j_star = -1

    # Update
    a_new = a.copy()
    a_new[i_star] += t * S  # Add mass at i_star
    if j_star >= 0:
        a_new[j_star] -= t * a[j_star]  # Remove mass from j_star
    a_new = np.maximum(a_new, 0.0)
    a_new = project_simplex(a_new, S)
    return a_new


def run_fw_refinement(a_init, n, S, max_time, rng):
    """Away-step FW with LSE gradient."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(100.0 + it * 5.0, 10000.0)
        t = max(0.01, 2.0 / (it + 2))

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            last_improve = it

        grad, _ = lse_gradient(a, c, beta)
        a = away_step_fw(a, grad, S, t)

        if it - last_improve > 2000:
            noise = 0.001 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 3000.0)
            if noise > 1e-10:
                a = project_simplex(a + rng.normal(0, noise, n), S)
            last_improve = it

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_final_polish(a_init, n, S, max_time, rng):
    """
    Final polishing: very high beta LSE with small steps,
    interleaved with peak shaving.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        # Phase: LSE with very high beta
        if time.monotonic() - start < max_time * 0.6:
            beta = 50000.0
            c = fft_convolve_self(a)
            cur_max = c.max()
            if cur_max < best:
                best = cur_max
                best_a = a.copy()

            grad, _ = lse_gradient(a, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-14:
                step = 0.005 / max(gnorm / c.max(), 1e-10)
                step = min(step, 0.01)
                a_new = project_simplex(a - step * grad, S)
                c_new = fft_convolve_self(a_new)
                if c_new.max() <= cur_max * 1.0001:
                    a = a_new
            it += 1
        else:
            # Phase: peak shaving
            a = peak_shaving_v2(a, n, S, n_steps=10, rng=rng)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def generate_asymmetric_init(n, rng):
    """Generate an asymmetric initialization with controlled structure."""
    # Main bump offset from center
    center = rng.uniform(0.25, 0.45)
    width = rng.uniform(0.15, 0.35)
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    # Add a smaller secondary bump on the other side
    center2 = 1.0 - center + rng.uniform(-0.05, 0.05)
    width2 = width * rng.uniform(0.5, 0.8)
    a2 = 0.3 * np.exp(-0.5 * ((t - center2) / width2) ** 2)
    a = a + a2
    # Add small oscillation
    a += 0.01 * np.sin(2 * np.pi * rng.uniform(5, 20) * t) * a
    return np.maximum(a, 1e-10)


def propose_candidate(seed=42, budget_s=2300.0):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Multi-strategy: LSE-FISTA, power-FISTA, spectral flattening, FW, peak shaving.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    safety = 60.0
    deadline = time.monotonic() + max(1.0, budget_s - safety)

    print("Budget: %.0fs, safety: %.0fs" % (budget_s, safety))

    global_best = np.inf
    global_a = None

    def try_sequence(a):
        nonlocal global_best, global_a
        n = len(a)
        S = np.sqrt(2.0 * n)
        a = normalize_sum(a, n)
        c = fft_convolve_self(a)
        val = c.max()
        if val < global_best:
            global_best = val
            global_a = a.copy()
            print("  New best: %.8f (n=%d)" % (val, n))

    # Load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            print("Loaded given: length=%d, sum=%.2f" % (len(given), given.sum()))
    except Exception:
        pass

    # === Phase 1: Quick LSE-FISTA on given (or asymmetric init) ===
    if given is not None:
        n_g = len(given)
        S_g = np.sqrt(2.0 * n_g)
        t1 = min(300.0, (deadline - time.monotonic()) * 0.15)
        print("=== Phase 1: LSE-FISTA v2 (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_lse_fista_v2(given, n_g, S_g, t1, rng,
                                   beta_start=50.0, beta_end=20000.0,
                                   beta_steps=150, k_transition=60,
                                   topk_size=3, topk_temp=30.0)
        try_sequence(a1)
    else:
        n_g = 8192
        S_g = np.sqrt(2.0 * n_g)
        t1 = min(300.0, (deadline - time.monotonic()) * 0.15)
        init = generate_asymmetric_init(n_g, rng)
        print("=== Phase 1: LSE-FISTA from asymmetric init (n=%d, %.0fs) ===" % (n_g, t1))
        a1, v1 = run_lse_fista_v2(init, n_g, S_g, t1, rng)
        try_sequence(a1)

    # === Phase 2: Power-smoothing FISTA (different landscape) ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        t2 = remaining * 0.2
        n_p = len(global_a)
        S_p = np.sqrt(2.0 * n_p)
        print("=== Phase 2: Power-FISTA (n=%d, %.0fs) ===" % (n_p, t2))
        a2, v2 = run_power_fista(global_a, n_p, S_p, t2, rng,
                                  p_start=2.0, p_end=12.0, p_steps=80)
        try_sequence(a2)

    # === Phase 3: Spectral flattening ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t3 = min(60.0, remaining * 0.1)
            n_s = len(global_a)
            print("=== Phase 3: Spectral flattening (n=%d) ===" % n_s)
            # Try different alphas
            for alpha in [0.2, 0.4, 0.6]:
                a_s = spectral_flatten(global_a, n_s, n_iters=15, alpha=alpha)
                # Quick LSE polish on the spectrally-flattened result
                S_s = np.sqrt(2.0 * n_s)
                a_s2, _ = run_lse_fista_v2(a_s, n_s, S_s, t3 / 3.0, rng,
                                            beta_start=100.0, beta_end=10000.0,
                                            beta_steps=40)
                try_sequence(a_s2)

    # === Phase 4: Main long LSE-FISTA with different schedule ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        t4 = remaining * 0.25
        n_m = len(global_a)
        S_m = np.sqrt(2.0 * n_m)
        print("=== Phase 4: Long LSE-FISTA (n=%d, %.0fs) ===" % (n_m, t4))
        a4, v4 = run_lse_fista_v2(global_a, n_m, S_m, t4, rng,
                                   beta_start=200.0, beta_end=50000.0,
                                   beta_steps=300, k_transition=40,
                                   topk_size=3, topk_temp=50.0)
        try_sequence(a4)

    # === Phase 5: Away-step FW ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t5 = remaining * 0.1
            n_fw = len(global_a)
            S_fw = np.sqrt(2.0 * n_fw)
            print("=== Phase 5: Away-step FW (n=%d, %.0fs) ===" % (n_fw, t5))
            a5, v5 = run_fw_refinement(global_a, n_fw, S_fw, t5, rng)
            try_sequence(a5)

    # === Phase 6: Final polish ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t6 = min(remaining * 0.5, 200.0)
            n_f = len(global_a)
            S_f = np.sqrt(2.0 * n_f)
            print("=== Phase 6: Final polish (n=%d, %.0fs) ===" % (n_f, t6))
            a6, v6 = run_final_polish(global_a, n_f, S_f, t6, rng)
            try_sequence(a6)

    # === Phase 7: Try larger n ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 120:
            n_large = len(global_a) * 2
            if n_large <= 32768:
                S_large = np.sqrt(2.0 * n_large)
                t_old = np.linspace(0, 1, len(global_a))
                t_new = np.linspace(0, 1, n_large)
                init_large = np.interp(t_new, t_old, global_a)
                init_large = normalize_sum(init_large, n_large)
                t7 = min(remaining * 0.3, 300.0)
                print("=== Phase 7: Larger n=%d (%.0fs) ===" % (n_large, t7))
                a7, v7 = run_lse_fista_v2(init_large, n_large, S_large, t7, rng,
                                           beta_start=50.0, beta_end=15000.0,
                                           beta_steps=100)
                try_sequence(a7)

    # === Phase 8: Try different initialization ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 90:
            n_try = len(global_a)
            S_try = np.sqrt(2.0 * n_try)
            t8 = min(remaining * 0.3, 200.0)
            print("=== Phase 8: Different init (n=%d, %.0fs) ===" % (n_try, t8))

            # Generate several different asymmetric inits
            best_init_val = np.inf
            best_init = None
            for trial in range(5):
                init = generate_asymmetric_init(n_try, rng)
                init = normalize_sum(init, n_try)
                c = fft_convolve_self(init)
                if c.max() < best_init_val:
                    best_init_val = c.max()
                    best_init = init.copy()

            if best_init is not None:
                a8, v8 = run_lse_fista_v2(best_init, n_try, S_try, t8, rng,
                                           beta_start=30.0, beta_end=30000.0,
                                           beta_steps=200)
                try_sequence(a8)

                # Polish
                remaining2 = deadline - time.monotonic()
                if remaining2 > 30 and global_a is not None:
                    a8p, _ = run_final_polish(global_a, n_try, S_try, min(remaining2 * 0.4, 100.0), rng)
                    try_sequence(a8p)

    # === Final verification ===
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final: evaluate_sequence = %.8f, internal = %.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  WARNING: evaluate_sequence rejected! Fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        print("WARNING: No valid result found, using uniform fallback.")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
