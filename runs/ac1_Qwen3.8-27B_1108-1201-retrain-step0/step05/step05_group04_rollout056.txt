```python
"""
Minimize 2*n*max(f*f)/sum(f)^2 using multi-phase approach:
1. Spectral shaping (flat-spectrum targeting via iterative phase/magnitude optimization)
2. Subgradient descent with exact active set
3. Peak-targeted local search

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    return c.max()

def subgradient_step(a, c, S, active_thresh_frac=0.0, n_top=None):
    """
    Compute subgradient and take a step.
    active_thresh_frac: include lags within this fraction of max
    n_top: or include top n_top lags
    """
    n = len(a)
    cm = c.max()
    
    if n_top is not None:
        # Top n lags
        idx = np.argsort(c)[::-1][:n_top]
        w = np.zeros_like(c)
        w[idx] = 1.0
    elif active_thresh_frac > 0:
        thresh = cm - active_thresh_frac * cm
        w = (c >= thresh).astype(float)
    else:
        # Only exact max lags
        w = (c >= cm - 1e-10).astype(float)
    
    tot = w.sum()
    if tot < 1:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        tot = 1.0
    w /= tot
    
    # Gradient: grad_j = 2 * sum_k w[k] * a[k-j] (with valid range)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return grad

def run_subgradient(a_init, n, S, max_time, rng, 
                    top_k_start=5, top_k_end=1, 
                    lr_start=0.01, lr_end=0.0001,
                    noise_every=100, noise_scale=0.005):
    """Subgradient descent with decreasing active set and learning rate."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Adaptive parameters
        frac = min(1.0, it / 5000.0)
        top_k = max(top_k_end, int(top_k_start - frac * (top_k_start - top_k_end)))
        lr = lr_start * (lr_end / lr_start) ** frac
        
        grad = subgradient_step(a, c, S, n_top=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Normalize gradient to unit norm for consistent step
        grad = grad / gnorm * lr
        a = project_simplex(a - grad, S)

            ns = noise_scale * S / np.sqrt(n) * np.exp(-it / 5000.0)
            if ns > 1e-10:
                a = project_simplex(a + rng.normal(0, ns, n), S)
    
    return best_a, best

def spectral_shaping(a_init, n, S, max_time, rng, 
                     target_flatness=0.8, n_iters=None):
    """
    Iteratively flatten the spectrum while maintaining non-negativity.
    This is a Gerchberg-Saxton-like approach.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    fsize = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Current spectrum
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude (geometric mean of current magnitudes)
        # Use partial flattening
        target_mag = np.ones_like(mag) * np.sqrt(mag.max() * mag.min())
        
        # Blend between current and target
        blend = target_flatness
        new_mag = mag * (1 - blend) + target_mag * blend
        
        # Reconstruct
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(new_A))[:n]
        
        # Enforce non-negativity and normalize
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_sum(a_new, n)
        
        # Evaluate
        c = fft_convolve_self(a_new)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a_new.copy()
        
        # Accept or reject (simulated annealing style)
        a = a_new
    
    return best_a, best

def peak_targeting(a, n, S, max_time, rng, n_peaks=3):
    """
    Directly target the peak lags for reduction.
    For each peak lag k, find the j that maximizes a[j]*a[k-j] and
    move mass from that position.
    """
    a = a.copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
        
        # Get top peaks
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # For each peak, find the "worst" contributor
        moves = []
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo > hi:
                continue
            js = np.arange(lo, hi + 1)
            partners = k - js
            contrib = a[js] * a[partners]
            j_max = js[np.argmax(contrib)]
            # Move mass from j_max to a "quiet" position
            moves.append(j_max)
        
        if not moves:
            break
        
        # Find quiet positions (low contribution to all peaks)
        quiet_score = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo > hi:
                continue
            js = np.arange(lo, hi + 1)
            partners = k - js
            quiet_score[js] += a[js] * a[partners]
        
        # Choose destination as position with lowest quiet_score
        # (but must have some mass to not be zero)
        dest_candidates = np.where(a > 0.01 * S / n)[0]
        if len(dest_candidates) == 0:
            break
        dest = dest_candidates[np.argmin(quiet_score[dest_candidates])]
        
        # Move small amount of mass
        amount = 0.001 * S / n
        sources = set(moves)
        for src in sources:
            if a[src] > amount:
                a[src] -= amount
                a[dest] += amount
                break
        
        a = project_simplex(a, S)
    
    return best_a, best

def lse_fista(a_init, n, S, max_time, rng,
              beta_start=100.0, beta_end=20000.0, beta_steps=200):
    """LSE-FIST for smooth approximation."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    fsize = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        c = fft_convolve_self(y)
        cm = c.max()
        
        if cm < best:
            best = cm
            best_a = y.copy()
        
        # LSE gradient
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / beta
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))

    c_final = fft_convolve_self(a)
    if c_final.max() < best:

        best_a = a.copy()
```

- For each active lag k (where c_k = max): the "pressure" from that lag is balanced
- For each positive a_j: the gradient is zero (or the same Lagrange multiplier)
- For zero a_j: the gradient is non-negative

The gradient of c_k with respect to a_j is: `a_{k-j} + a_{k-j'} ` where j' is the "mirror" index... actually let me be more careful.

`dc_k / da_j = a_{k-j}` (if 0 <= k-j < n) plus `a_{k'-j}` for any other k' where j also appears... 

Wait, actually for the full sum of c_k over active lags:
`d(sum_{k in A} c_k) / da_j = sum_{k in A} a_{k-j}` (with valid bounds)

This is what the subgradient computes.

Now, at the optimum with sum(a) = S constraint:
`sum_{k in A} a_{k-j} = lambda` for all j with a_j > 0
`sum_{k in A} a_{k-j} >= lambda` for all j with a_j = 0

Where lambda is the Lagrange multiplier for the sum constraint, and A is the active set of lags.

```python
"""
Minimize 2*n*max(f*f)/sum(f)^2 using:
1. LSE-FISTA for initial convergence
2. Active-set subgradient with Barzilai-Borwein step
3. Multiplicative updates for non-negative refinement
4. Peak-targeted local search
5. Adaptive multi-phase with time budget management

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def compute_subgradient(a, c, top_k=5):
    """Compute subgradient using top_k lags."""
    n = len(a)
    cm = c.max()
    idx = np.argsort(c)[::-1][:top_k]
    w = np.zeros_like(c)
    w[idx] = 1.0 / top_k
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def lse_fista(a_init, n, S, max_time, rng,
              beta_start=200.0, beta_end=30000.0, beta_steps=200,
              restart_every=300):
    """LSE-FISTA with momentum restart."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    fsize = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = min(50000.0, beta_start * (beta_end / beta_start) ** frac)
        
        c = fft_convolve_self(y)
        cm = c.max()
        
        if cm < best:
            best = cm
            best_a = y.copy()
            last_improve = it
        
        # LSE weights
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot
        
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / beta
        
        # Momentum restart
        if it - last_improve > restart_every:
            t_mom = 1.0
            # Small noise
            ns = 0.002 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 1000.0)
            if ns > 1e-10:
                y = project_simplex(y + rng.normal(0, ns, n), S)
            last_improve = it
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

def bb_subgradient(a_init, n, S, max_time, rng,
                   top_k_start=10, top_k_end=1,
                   min_lr=1e-6, max_lr=0.1,
                   noise_every=200, noise_scale=0.003):
    """
    Subgradient descent with Barzilai-Borwein adaptive step size.
    Active set shrinks from top_k_start to top_k_end.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    # BB state
    a_prev = a.copy()
    grad_prev = None
    lr = 0.01
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()
        
        if cm < best:
            best = cm
            best_a = a.copy()
            last_improve = it
        
        # Adaptive top_k
        frac = min(1.0, it / 3000.0)
        top_k = max(top_k_end, int(round(top_k_start - frac * (top_k_start - top_k_end))))
        
        grad = compute_subgradient(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # BB step size
        if grad_prev is not None:
            s = a - a_prev
            y = grad - grad_prev
            sy = np.dot(s, y)
            yy = np.dot(y, y)
            ss = np.dot(s, s)
            
            if yy > 1e-14 and sy > 1e-14:
                lr = max(min_lr, min(max_lr, ss / sy))
            elif yy > 1e-14:
                lr = max(min_lr, min(max_lr, 0.5 / np.sqrt(yy)))
        
        a_prev = a.copy()
        grad_prev = grad.copy()
        
        a = project_simplex(a - lr * grad, S)
        
        # Noise for escape
        if it % noise_every == 0:
            stagnant = it - last_improve
            ns = noise_scale * S / np.sqrt(n) * np.exp(-stagnant / 2000.0)
            if ns > 1e-10:
                a = project_simplex(a + rng.normal(0, ns, n), S)
                last_improve = it
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

def multiplicative_update(a_init, n, S, max_time, rng,
                          lr=0.01, top_k=5):
    """
    Multiplicative update: a_j *= exp(-lr * grad_j)
    Naturally preserves non-negativity.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()
        
        if cm < best:
            best = cm
            best_a = a.copy()
        
        grad = compute_subgradient(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive lr
        lr_adapt = lr * (1.0 + 100.0 / it) * (S / (n * gnorm + 1e-10))
        lr_adapt = min(lr_adapt, 0.5)
        
        # Multiplicative update
        a = a * np.exp(-lr_adapt * grad)
        a = normalize_sum(a, n)
        
        # Periodic noise
        if it % 500 == 0:
            ns = 0.001 * S / np.sqrt(n) * np.exp(-it / 3000.0)
            if ns > 1e-10:
                a = normalize_sum(a + rng.normal(0, ns, n), n)
    
    return best_a, best

def active_set_qp_solve(a, n, S, active_lags, max_iters=100):
    """
    Try to solve the KKT conditions for a fixed active set.
    At optimum: D^T a = lambda * 1 for all j with a_j > 0
    where D is the matrix mapping a to the active lag values.
    
    We use iterative projection (POCS-like) to find a solution.
    """
    a = a.copy()
    cm = 0.0
    for _ in range(max_iters):
        c = fft_convolve_self(a)
        # Check if active lags are approximately equal
        active_vals = c[active_lags]
        spread = active_vals.max() - active_vals.min()
        cm = c.max()
        
        if spread < 1e-6 * cm:
            break
        
        # Gradient: try to equalize the active lags
        # Reduce lags that are above the mean of active lags
        mean_active = active_vals.mean()
        w = np.zeros_like(c)
        for k in active_lags:
            w[k] = (c[k] - mean_active) / max(spread, 1e-10)
        # Also penalize lags above max
        w += (c > cm * 0.99).astype(float) * 0.5
        
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = np.real(ifft(W * np.conj(A)))[:n]
        
        a = project_simplex(a - 0.001 * grad, S)
    
    c = fft_convolve_self(a)
    return a.copy(), c.max()

def spectral_flatten(a_init, n, S, max_time, rng, blend=0.3):
    """
    Flatten the power spectrum iteratively (Gerchberg-Saxton style).
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    fsize = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: geometric mean magnitude
        log_mag = np.log(mag + 1e-10)
        target_log = np.ones_like(log_mag) * (log_mag.max() + log_mag.min()) / 2
        target_mag = np.exp(target_log)
        
        # Blend
        new_mag = mag * (1 - blend) + target_mag * blend
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(new_A, fsize))[:n]
        
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_sum(a_new, n)
        
        c = fft_convolve_self(a_new)
        if c.max() < best:
            best = c.max()
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best
```

subject to: c_k <= t for k in A

This is a QP (quadratic objective, linear constraints). The quadratic part is `sum_k (c_k - t)^2` for k in A.

**Alternative: "Leveling" approach**

At the optimum, the active lags should all have the same value (by KKT conditions, if we're at a point where multiple lags are tied at the maximum). So:

1. Identify the current active set A
2. Compute the "leveling gradient": for each a_j, compute how it affects the spread of active lags
3. Take a step that reduces the spread

The leveling gradient for a_j is:
`g_j = sum_{k in A} (c_k - mean_A) * a_{k-j}`

```python
def water_filling_step(a, n, S, n_peaks=3, move_frac=0.01):
    """
    For each peak lag, move mass from the "bottleneck" position.
    """
    a = a.copy()
    c = fft_convolve_self(a)
    
    for k in np.argsort(c)[::-1][:n_peaks]:
        # Find the j that maximizes a[j]*a[k-j]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        js = np.arange(lo, hi + 1)
        partners = k - js
        contribs = a[js] * a[partners]
        
        j_max = js[np.argmax(contribs)]
        partner_max = k - j_max
        
        # Determine which to reduce (the larger one)
        if a[j_max] >= a[partner_max]:
            src = j_max
        else:
            src = partner_max
        
        # Find destination: position with minimum contribution to all top peaks
        top_lags = np.argsort(c)[::-1][:n_peaks]
        dest_score = np.zeros(n)
        for kl in top_lags:
            lo_l = max(0, kl - n + 1)
            hi_l = min(kl, n - 1)
            if lo_l > hi_l:
                continue
            js_l = np.arange(lo_l, hi_l + 1)
            partners_l = kl - js_l
            # Only positions that aren't the source
            contrib_l = a[js_l] * a[partners_l]
            np.add.at(dest_score, js_l, contrib_l)
        
        # Exclude source
        dest_score[src] = np.inf
        dest = np.argmin(dest_score)
        
        # Move mass
        amount = move_frac * a[src]
        if amount > 1e-15:
            a[src] -= amount
            a[dest] += amount
    
    a = project_simplex(a, S)
    return a
```

```python
"""
Minimize 2*n*max(f*f)/sum(f)^2 using multi-phase optimization:
1. LSE-FISTA with adaptive beta for initial convergence
2. Barzilai-Borwein subgradient with adaptive active set
3. Spectral flattening (Gerchberg-Saxton style)
4. Multiplicative updates preserving non-negativity
5. Water-filling peak targeting
6. Final high-precision LSE polish
Key innovation: BB step sizes + active set adaptation + spectral shaping
for the "last mile" improvement.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_subgradient(a, c, top_k):
    n = len(a)
    top_k = max(1, min(top_k, len(c)))
    idx = np.argsort(c)[::-1][:top_k]
    w = np.zeros_like(c)
    w[idx] = 1.0 / top_k
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def lse_fista(a_init, n, S, max_time, rng,
              beta_start=500.0, beta_end=50000.0, beta_steps=150,
              restart_every=250, noise_scale=0.002):
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, beta_steps))
        beta = min(80000.0, beta_start * (beta_end / beta_start) ** frac)

        c = fft_convolve_self(y)
        cm = c.max()

        if cm < best:
            best = cm
            best_a = y.copy()
            last_improve = it

        shift = beta * (c - cm)
        shift[np.minimum(shift, -50) < -50] = -50
        e = np.exp(shift)
        tot = e.sum()
        if tot < 1e-300:
            break
        w = e / tot

        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / beta

        stagnant = it - last_improve
        if stagnant > restart_every:
            t_mom = 1.0
            ns = noise_scale * S / np.sqrt(n) * np.exp(-stagnant / 1500.0)
            if ns > 1e-10:
                y = project_simplex(y + rng.normal(0, ns, n), S)
            last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def bb_subgradient(a_init, n, S, max_time, rng,
                   top_k_start=10, top_k_end=1,
                   min_lr=1e-8, max_lr=0.05,
                   noise_every=300, noise_scale=0.003):
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    a_prev = a.copy()
    grad_prev = None
    lr = 0.01
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()

        if cm < best:
            best = cm
            best_a = a.copy()
            last_improve = it

        frac = min(1.0, it / 5000.0)
        top_k = max(top_k_end, int(round(top_k_start - frac * (top_k_start - top_k_end))))

        grad = compute_subgradient(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        if grad_prev is not None:
            s_vec = a - a_prev
            y_vec = grad - grad_prev
            sy = np.dot(s_vec, y_vec)
            yy = np.dot(y_vec, y_vec)
            ss = np.dot(s_vec, s_vec)

            if yy > 1e-14:
                if sy > 1e-10:
                    lr = max(min_lr, min(max_lr, ss / sy))
                else:
                    lr = max(min_lr, min(max_lr, ss / (yy + 1e-14)))

        a_prev = a.copy()
        grad_prev = grad.copy()

        a = project_simplex(a - lr * grad, S)

        stagnant = it - last_improve
        if it % noise_every == 0 and stagnant > noise_every:
            ns = noise_scale * S / np.sqrt(n) * np.exp(-stagnant / 3000.0)
            if ns > 1e-10:
                a = project_simplex(a + rng.normal(0, ns, n), S)
                last_improve = it

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def multiplicative_update(a_init, n, S, max_time, rng,
                          base_lr=0.005, top_k=5,
                          noise_every=500, noise_scale=0.002):
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()

        if cm < best:
            best = cm
            best_a = a.copy()

        grad = compute_subgradient(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        lr_adapt = base_lr * (S / (n * gnorm + 1e-10))
        lr_adapt = min(lr_adapt, 0.3)

        a = a * np.exp(-lr_adapt * grad)
        a = normalize_sum(a, n)

        if it % noise_every == 0:
            ns = noise_scale * S / np.sqrt(n) * np.exp(-it / 5000.0)
            if ns > 1e-10:
                a = normalize_sum(a + rng.normal(0, ns, n), n)

    return best_a, best


def spectral_flatten(a_init, n, S, max_time, rng, blend=0.25,
                     target_spread=0.0):
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        A = fft(a, fsize)
        mag = np.abs(A) + 1e-15
        phase = np.angle(A)

        log_mag = np.log(mag)
        log_mean = (log_mag.max() + log_mag.min()) * 0.5
        target_mag = np.exp(np.ones_like(log_mag) * log_mean)

        new_mag = mag * (1.0 - blend) + target_mag * blend
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(new_A, fsize))[:n]

        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_sum(a_new, n)

        c = fft_convolve_self(a_new)
        if c.max() < best:
            best = c.max()
            best_a = a_new.copy()

        a = a_new

    return best_a, best


def water_filling(a_init, n, S, max_time, rng,
                  n_peaks=5, move_frac=0.005):
    a = a.copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cm = c.max()

        if cm < best:
            best = cm
            best_a = a.copy()

        top_lags = np.argsort(c)[::-1][:n_peaks]

        dest_score = np.zeros(n)
        sources = []

        for k in top_lags:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo > hi:
                continue
            js = np.arange(lo, hi + 1)
            partners = k - js
            contribs = a[js] * a[partners]

            j_max = js[np.argmax(contribs)]
            partner_max = k - j_max

            if a[j_max] >= a[partner_max]:
                src = j_max
            else:
                src = partner_max

            sources.append(src)

            np.add.at(dest_score, js, contribs)

        dest_score[sources] = np.inf
        dest = int(np.argmin(dest_score))

        for src in sources:
            amount = move_frac * a[src]
            if amount > 1e-15 and a[src] > amount:
                a[src] -= amount
                a[dest] += amount

        a = project_simplex(a, S)

    return best_a, best


def active_equalization(a_init, n, S, max_time, rng,
                        thresh_frac=0.01):
    """
    Equalize the active lags by computing a gradient that reduces
    the variance among lags close to the maximum.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    lr = 0.01

    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cm = c.max()

        if cm < best:
            best = cm
            best_a = a.copy()

        thresh = cm - thresh_frac * cm
        w = (c >= thresh).astype(np.float64)
        tot = w.sum()
        if tot < 1:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
            tot = 1.0

        # Weight by how much each lag exceeds the mean of active lags
        active_mean = c[w > 0].mean()
        w = w * np.maximum(c - active_mean, 0)
        w = w / max(w.sum(), 1e-10)

        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = lr / (1.0 + it * 0.0001)
        a = project_simplex(a - step * grad, S)

        if it % 500 == 0:
            ns = 0.001 * S / np.sqrt(n) * np.exp(-it / 5000.0)
            if ns > 1e-10:
                a = project_simplex(a + rng.normal(0, ns, n), S)

    return best_a, best

        for n_try in [n * 2, n // 2]:
            if n_try < 128 or n_try > 32768:
                continue
            if time.monotonic() > deadline - 40:
                break
            t9 = min(80.0, (deadline - time.monotonic()) * 0.4)
            print("\n=== Phase 9: Try n=%d (%.0fs) ===" % (n_try, t9))
            t_old = np.linspace(0, 1, n)
            t_new = np.linspace(0, 1, n_try)
            init = np.interp(t_new, t_old, global_a)
            S_try = np.sqrt(2.0 * n_try)
            init = normalize_sum(init, n_try)
            a9, _ = lse_fista(init, n_try, S_try, t9, rng,
                              beta_start=500.0, beta_end=30000.0, beta_steps=60)
            try_sequence(a9)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nFinal: evaluate_sequence = %.8f, internal = %.8f (n=%d)" %
              (ver if np.isfinite(ver) else 999.0, global_best, len(global_a)))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("RETURNING: length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
```
