```python
"""
Search for non-negative sequences minimizing
    2*n*max(conv(a,a))/(sum(a)**2)
using a corrected full-convolution subgradient.

Main ingredients:
1. Full linear self-convolution via zero-padded FFT (A*A, not A*conj(A)).
2. Exact adjoint gradient for weighted convolution coefficients.
3. LSE-smoothed FISTA on the nonnegative simplex.
4. Multiplicative mirror descent as a fast alternate.
5. Stochastic active-set line search for final polishing.
6. Multi-start with several structural/random initializations, including
   the previous state-of-the-art sequence when available.
"""

import numpy as np
import time
from scipy.fft import fft, ifft, irfft

FFT_SIZE_DICT = {}


def next_pow2_int(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def get_fft_sizes(n):
    key = int(n)
    if key in FFT_SIZE_DICT:
        return FFT_SIZE_DICT[key]
    csize = next_pow2_int(2 * n - 1)
    gsize = next_pow2_int(3 * n - 2)
    FFT_SIZE_DICT[key] = (csize, gsize)
    return csize, gsize


def clean_array(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    return a


def clip_scale_sum(v, S):
    """Fast valid feasibility: clip to >=0 and rescale to sum S."""
    v = clean_array(v)
    v = np.clip(v, 0.0, 1000.0)
    s = float(v.sum())
    n = v.size
    if s <= 1e-300 or n == 0:
        return np.full(n, S / float(n) if n else 0.0, dtype=np.float64)
    v = v * (S / s)
    v = np.clip(v, 0.0, 1000.0)
    resid = S - float(v.sum())
    if resid > 0 and n:
        v[int(np.argmax(v))] += resid
    elif resid < 0 and n:
        v[int(np.argmax(v))] += resid
        if v[int(np.argmax(v))] < 0.0:
            v[int(np.argmax(v))] = 0.0
    return v


def normalize_to_sum(a, S):
    a = clip_scale_sum(a, S)
    return a


def project_simplex(v, S):
    """Project onto {x >= 0, sum(x)=S}."""
    v = clean_array(v)
    v = np.clip(v, 0.0, None)
    n = v.size
    if n == 0:
        return v
    if S <= 0.0:
        return np.zeros(n, dtype=np.float64)

    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cs - float(S)) / idx
    valid = tc <= w
    if np.any(valid):
        r = int(np.max(np.where(valid)[0])) + 1
        theta = (cs[r - 1] - float(S)) / float(r)
    else:
        theta = float(S) / float(n)

    out = v - theta
    out = np.maximum(out, 0.0)
    resid = float(S - out.sum())
    if resid > 0 and n:
        out[int(np.argmax(out))] += resid
    elif resid < 0 and n:
        out[int(np.argmax(out))] += resid
        if out[int(np.argmax(out))] < 0.0:
            out[int(np.argmax(out))] = 0.0
    return out


def fft_conv_self(a):
    """Linear self-convolution a*a, matching np.convolve(a,a)."""
    a = clean_array(a)
    n = a.size
    csize, _ = get_fft_sizes(n)
    A = fft(a, csize)
    c = np.real(ifft(A * A))
    return c[: 2 * n - 1]


def local_obj_from_c(a, c):
    n = a.size
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        return np.inf
    mb = float(np.max(c))
    if not np.isfinite(mb):
        return np.inf
    return float(2.0 * n * mb / (s * s))


def local_obj(a, c=None):
    a = clean_array(a)
    n = a.size
    if c is None:
        c = fft_conv_self(a)
    return local_obj_from_c(a, c)


def softmax_weights(c, beta):
    """Weights proportional to exp(beta*c), with a large-beta active-set fallback."""
    c = clean_array(c)
    cmax = float(np.max(c))
    cmin = float(np.min(c))
    diff = cmax - cmin
    if beta * diff > 45.0:
        threshold = cmax - 45.0 / max(1e-300, beta)
        active = np.flatnonzero(c >= threshold)
        if active.size == 0:
            active = np.flatnonzero(c >= cmax - 1e-12)
        w = np.zeros(c.size, dtype=np.float64)
        w[active] = 1.0 / max(1, active.size)
        return w

    scaled = beta * (c - cmax)
    e = np.exp(scaled)
    se = float(e.sum())
    if se <= 1e-300 or not np.isfinite(se):
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / se


def gradient_from_weights(a, w):
    """
    Gradient of sum_t w_t (a*a)_t with respect to a.
    For c_t = sum_i a_i a_{t-i}, derivative is 2 sum_t w_t a_{t-j}.
    """
    a = clean_array(a)
    w = clean_array(w)
    n = a.size
    L = 2 * n - 1
    if w.shape[0] != L:
        w = w[:L] if w.size >= L else np.pad(w, (0, L - w.size))
    _, gsize = get_fft_sizes(n)

    A = fft(a, gsize)
    Wrev = fft(w[::-1], gsize)
    conv = np.real(ifft(A * Wrev))
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def thue_morse_seq(n):
    arr = np.array([0.0], dtype=np.float64)
    while arr.size < n:
        arr = np.concatenate((arr, 1.0 - arr))
    return arr[:n]


def make_inits(n, S, rng, given):
    out = []

    if given is not None:
        g = clean_array(given)
        g = np.clip(g, 0.0, None)
        if g.size == 0:
            g = np.ones(n, dtype=np.float64)
        elif g.size != n:
            x = np.linspace(0.0, 1.0, g.size, endpoint=False)
            y = np.linspace(0.0, 1.0, n, endpoint=False)
            g = np.interp(y, x, g)
        g = normalize_to_sum(g, S)
        out.append(g.copy())
        out.append(normalize_to_sum(0.5 * (g + g[::-1]), S))
        out.append(normalize_to_sum(g * (1.0 + 0.05 * rng.standard_normal(n)), S))
        out.append(normalize_to_sum(g * (1.0 - 0.05 * rng.standard_normal(n)), S))

    out.append(np.full(n, S / float(n), dtype=np.float64))

    tt = np.linspace(0.0, 1.0, n, endpoint=False)
    out.append(normalize_to_sum(1.0 + 0.5 * np.cos(2.0 * np.pi * tt), S))
    out.append(normalize_to_sum(1.0 + 0.5 * np.sin(2.0 * np.pi * tt) + 0.25 * np.cos(4.0 * np.pi * tt), S))
    out.append(normalize_to_sum(1.0 + 0.35 * np.cos(3.0 * np.pi * tt) + 0.25 * np.sin(7.0 * np.pi * tt), S))
    out.append(normalize_to_sum(np.sin(np.pi * tt) ** 2 + 0.1, S))
    out.append(normalize_to_sum(np.exp(-8.0 * (tt - 0.5) ** 2) + 0.2, S))
    out.append(normalize_to_sum(thue_morse_seq(n), S))
    out.append(normalize_to_sum(thue_morse_seq(n) * (1.0 + 0.1 * np.abs(rng.standard_normal(n))), S))

    csize, _ = get_fft_sizes(n)
    k = csize // 2 + 1
    freqs = np.arange(k, dtype=np.float64)
    tweens = [0.08, 0.25, 1.0, 3.0, 10.0]
    for gamma in tweens:
        rz = rng.normal(0.0, 1.0, k) + 1j * rng.normal(0.0, 1.0, k)
        rz[0] = 0.0
        if gamma > 0:
            rz = rz * np.exp(-freqs / max(1.0, gamma * (k / 10.0 + 1.0)))
        aa = np.real(irfft(rz, csize)[:n])
        aa = np.clip(aa, 0.0, None)
        if aa.sum() > 1e-12:
            out.append(normalize_to_sum(aa, S))

    for p in (0.35, 0.5, 0.65):
        aa = (rng.random(n) < p).astype(np.float64)
        out.append(normalize_to_sum(aa, S))

    for nb in (8, 16, 32, 64):
        blen = max(1, n // nb)
        nb_use = int(np.ceil(float(n) / float(blen)))
        vals = rng.uniform(0.2, 1.5, nb_use)
        block = np.repeat(vals, blen)
        block = block[:n]
        out.append(normalize_to_sum(block, S))

    return out


def fourier_phase_mutate(a, rng, scale=0.2):
    a = clean_array(a)
    n = a.size
    S = float(a.sum())
    csize, _ = get_fft_sizes(n)
    A = fft(a, csize)
    mag = np.abs(A)
    phase = np.angle(A)
    freqs = np.arange(csize, dtype=np.float64)
    weight = 1.0 + freqs / (0.05 * csize)
    phase_noise = rng.normal(0.0, scale * weight, csize)
    newA = mag * np.exp(1j * (phase + phase_noise))
    newA[0] = A[0]
    a_new = np.real(ifft(newA, csize)[:n])
    return project_simplex(a_new, S)


def run_fista(a_init, S, max_time, rng,
              beta_init=100.0, beta_rate=0.15, beta_max=30000.0,
              step_factor=0.5, stuck_limit=8000, start_time=None):
    if max_time <= 0.0:
        return a_init, local_obj(a_init)

    start = start_time if start_time is not None else time.monotonic()
    limit = start + float(max_time)

    n = len(a_init)
    y = normalize_to_sum(a_init, S)
    x = y.copy()
    t_mom = 1.0
    beta_iter = 0
    it = 0
    last_improve = 0
    sf = float(step_factor)

    c = fft_conv_self(y)
    best_val = local_obj_from_c(y, c)
    best = y.copy()

    while time.monotonic() < limit:
        it += 1
        beta_iter += 1
        beta = min(float(beta_max), float(beta_init) + float(beta_rate) * float(beta_iter))

        c = fft_conv_self(y)
        val = local_obj_from_c(y, c)
        if np.isfinite(val) and val < best_val:
            best_val = val
            best = y.copy()
            last_improve = it

        stuck = it - last_improve
        if stuck > stuck_limit:
            kind = (stuck // stuck_limit) % 3
            if kind == 2:
                y = fourier_phase_mutate(y, rng, scale=0.18 + 0.12 * rng.random())
            else:
                noise = 0.0012 * S / max(1.0, np.sqrt(float(n))) * (1.0 + rng.random())
                y = clip_scale_sum(y + rng.normal(0.0, float(noise), n), S)
            x = y.copy()
            t_mom = 1.0
            last_improve = it
            if sf < 0.7:
                sf = 0.7
            continue

        w = softmax_weights(c, beta)
        grad = gradient_from_weights(y, w)
        grad = grad - float(np.mean(grad))
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-14:
            break

        l2 = float(np.linalg.norm(y))
        curv_scale = max(1.0, l2 / np.sqrt(2.0))
        step = sf / (max(1e-9, beta) * curv_scale)
        step = min(step, 2.0 / max(1.0, gnorm))
        if step <= 0.0 or not np.isfinite(step):
            step = 1e-6

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        y_new = clip_scale_sum(y_new, S)

        x = x_new
        y = y_new
        t_mom = t_new

        if it % 4000 == 0:
            t_mom = 1.0
        if it % 10000 == 0 and sf < 0.8:
            t_mom = 1.0

    if time.monotonic() >= limit - 0.05:
        return best, best_val

    c = fft_conv_self(y)
    val = local_obj_from_c(y, c)
    if val < best_val:
        return y.copy(), val
    return best, best_val


def run_mirror(a_init, S, max_time, rng,

        val = local_obj_from_c(y, c)
        if np.isfinite(val) and val < best_val:
            best_val = val
            best = y.copy()
            last_improve = it
            eta = min(2.0, eta * 1.02)

        if it - last_improve > 8000:
            noise = 0.0015 * S / max(1.0, np.sqrt(float(n)))
            y = clip_scale_sum(y + rng.normal(0.0, float(noise), n), S)
            eta = max(0.08, eta * 0.5)
            last_improve = it

        w = softmax_weights(c, beta)
        grad = gradient_from_weights(y, w)
        grad = grad - float(np.mean(grad))
        rms = float(np.sqrt(np.mean(grad * grad)))
        if not np.isfinite(rms) or rms < 1e-14:
            break

        delta = -eta * grad / (rms + 1e-15)
        delta = np.clip(delta, -40.0, 40.0)
        y_new = clip_scale_sum(y * np.exp(delta), S)

        c2 = fft_conv_self(y_new)
        v2 = local_obj_from_c(y_new, c2)

        if np.isfinite(v2) and v2 < val * (1.0 - 1e-11):
            y = y_new
            eta = min(2.0, eta * 1.04)
        else:
            eta = max(0.02, eta * 0.65)

    c = fft_conv_self(y)
    val = local_obj_from_c(y, c)
    if val < best_val:
        return y.copy(), val
    return best, best_val


def run_active_polish(a_init, S, max_time, rng, k_min=4, k_max=128):
    if max_time <= 0.0:
        return a_init, local_obj(a_init)

    start = time.monotonic()
    limit = start + float(max_time)
    n = len(a_init)
    L = 2 * n - 1

    y = normalize_to_sum(a_init, S)
    best = y.copy()
    c = fft_conv_self(y)
    best_val = local_obj_from_c(y, c)

    it = 0
    no_improve = 0
    step_mode = 0.35

    while time.monotonic() < limit:
        it += 1
        progress = min(1.0, max(0.0, (time.monotonic() - start) / float(max_time)))
        k = int(k_min + (k_max - k_min) * (progress ** 0.65))
        k = max(1, min(L, k))

        c = fft_conv_self(y)
        val = local_obj_from_c(y, c)
        if np.isfinite(val) and val < best_val:
            best_val = val
            best = y.copy()
            no_improve = 0
            step_mode = min(1.0, step_mode * 1.05)

        if no_improve > 60:
            noise = 0.0011 * S / max(1.0, np.sqrt(float(n))) * (1.0 + rng.random())
            y = clip_scale_sum(y + rng.normal(0.0, float(noise), n), S)
            if no_improve > 180:
                y = fourier_phase_mutate(y, rng, scale=0.22 + 0.18 * rng.random())
            no_improve = 0
            step_mode = 0.4
            continue

        part = np.argpartition(c, -k)[-k:]
        part = part[np.argsort(c[part])[::-1]]

        if it % 5 == 0:
            top = min(8, part.size)
            idx = part[:top]
            weights = np.ones(idx.size, dtype=np.float64) / float(idx.size)
        else:
            csel = c[part]
            cspan = float(csel.max() - csel.min() + 1e-14)
            bias = (csel - float(csel.min())) / cspan
            weights = 0.25 + rng.random(part.size) + 1.25 * bias
            weights = weights / float(weights.sum())
            idx = part

        wfull = np.zeros(L, dtype=np.float64)
        wfull[idx] = weights

        grad = gradient_from_weights(y, wfull)
        grad = grad - float(np.mean(grad))
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-12:
            no_improve += 1
            continue

        base = min(0.45, 2.0 / max(1e-9, gnorm)) * step_mode
        if base <= 0.0 or not np.isfinite(base):
            base = 1e-4
        base = max(1e-9, base)

        lams = np.logspace(-7.0, np.log10(base), 9)
        best_lam = 0.0
        best_ls_val = val
        best_fast_x = None

        for lam in lams[::-1]:
            z = y - float(lam) * grad
            xn = clip_scale_sum(z, S)
            cn = fft_conv_self(xn)
            vn = local_obj_from_c(xn, cn)
            if np.isfinite(vn) and vn < best_ls_val:
                best_ls_val = vn
                best_lam = float(lam)
                best_fast_x = xn

        accepted = None
        if best_lam > 0.0:
            x_fast = None if best_fast_x is None else best_fast_x.copy()
            v_fast = best_ls_val

            x_exact = project_simplex(y - best_lam * grad, S)
            ce = fft_conv_self(x_exact)
            ve = local_obj_from_c(x_exact, ce)

            if np.isfinite(ve) and ve < val:
                accepted = x_exact
                vv = ve
            if x_fast is not None and np.isfinite(v_fast) and (accepted is None or v_fast < vv):
                accepted = x_fast
                vv = v_fast

        if accepted is not None:
            y = accepted
            no_improve = 0
        else:
            no_improve += 1
            step_mode = max(0.08, step_mode * 0.82)

    c = fft_conv_self(y)
    val = local_obj_from_c(y, c)
    if val < best_val:
        return y.copy(), val
    return best, best_val


def direct_objective(a):
    """Call the supplied evaluator if available; otherwise fall back to FFT objective."""
    try:
        a = clean_array(a)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s < 0.01:
            return np.inf
        lst = [float(x) for x in a]
        ev = globals().get("evaluate_sequence")
        if ev is None:
            return local_obj(a)
        val = ev(lst)
        return float(val)
    except Exception:
        return local_obj(clean_array(a))


def propose_candidate():
    total_start = time.monotonic()
    deadline = total_start + 2360.0

    given = None
    raw = globals().get("height_sequence_1", None)
    if raw is not None:
        try:
            given = clean_array(raw)
            given = np.clip(given, 0.0, None)
        except Exception:
            given = None

    if given is not None and 1024 <= given.size <= 16384:
        n = int(given.size)
    else:
        n = 8192

    S = float(np.sqrt(2.0 * float(n)))
    rng = np.random.default_rng(1417339)

    print("n=%d, S=%.3f" % (n, S))

    cands = make_inits(n, S, rng, given)
    scored = []
    for i, aa in enumerate(cands):
        aa = normalize_to_sum(aa, S)
        cc = fft_conv_self(aa)
        vv = local_obj_from_c(aa, cc)
        if np.isfinite(vv):
            scored.append((vv, i, aa))

    scored.sort()

    best_a = None
    best_val = np.inf
    if scored:
        best_val = float(scored[0][0])
        best_a = scored[0][2].copy()
        print("initial best local obj=%.8f" % best_val)

    # Quick screening of the best few inits.
    if time.monotonic() < deadline - 900.0:
        for rank in range(min(4, len(scored))):
            if time.monotonic() >= deadline - 900.0:
                break
            cand = scored[rank][2]
            alloc = min(20.0, max(0.0, deadline - time.monotonic() - 900.0))
            if alloc < 5.0:
                break
            aa, vv = run_fista(
                cand, S, alloc, rng,
                beta_init=50.0,
                beta_rate=0.06,
                beta_max=4000.0,
                step_factor=0.45,
                stuck_limit=1200
            )
            if np.isfinite(vv) and vv < best_val:
                best_val = vv
                best_a = aa.copy()
                print("screening obj=%.8f" % best_val)

    # Main accelerated FISTA / mirror loop.
    phase = 0
    while time.monotonic() < deadline - 680.0:
        rem = max(0.0, deadline - time.monotonic() - 680.0)
        alloc = min(110.0, rem)
        if alloc < 15.0:
            break

        phase += 1
        if phase % 4 == 1:
            beta_start = min(2500.0, 60.0 * (1.0 + 0.35 * phase))
            beta_rate = 0.10
        else:
            beta_start = min(8000.0, 120.0 * (1.0 + 0.8 * phase))
            beta_rate = 0.18

        aa, vv = run_fista(
            best_a, S, alloc, rng,
            beta_init=beta_start,
            beta_rate=beta_rate,
            beta_max=35000.0,
            step_factor=0.52,
            stuck_limit=9000
        )
        if np.isfinite(vv) and vv < best_val:
            best_val = vv
            best_a = aa.copy()
            print("FISTA phase %d obj=%.8f" % (phase, best_val))

        rem = max(0.0, deadline - time.monotonic() - 680.0)
        alloc2 = min(15.0, rem * 0.25)
        if alloc2 > 4.0:
            aa, vv = run_mirror(
                best_a, S, alloc2, rng,
                beta_init=min(2000.0, 1.5 * beta_start),
                beta_rate=0.08,
                beta_max=12000.0,
                eta_init=0.7
            )
            if np.isfinite(vv) and vv < best_val:
                best_val = vv
                best_a = aa.copy()
                print("mirror phase %d obj=%.8f" % (phase, best_val))

    # Active-set polish.
    rem = max(0.0, deadline - time.monotonic() - 260.0)
    if rem > 25.0:
        alloc = min(300.0, rem)
        aa, vv = run_active_polish(
            best_a, S, alloc, rng,
            k_min=4,
            k_max=150
        )
        if np.isfinite(vv) and vv < best_val:
            best_val = vv
            best_a = aa.copy()
            print("active polish obj=%.8f" % best_val)

    # Basin-hopping style mutations and short recovery.
    while time.monotonic() < deadline - 120.0:
        rem = max(0.0, deadline - time.monotonic() - 120.0)
        if rem < 8.0:
            break

        scale = 0.12 + 0.25 * rng.random()
        mut = fourier_phase_mutate(best_a, rng, scale)
        alloc = min(30.0, rem * 0.35)
        if alloc > 3.0:
            aa, vv = run_fista(
                mut, S, alloc, rng,
                beta_init=300.0,
                beta_rate=0.12,
                beta_max=15000.0,
                step_factor=0.42,
                stuck_limit=2500
            )
            if np.isfinite(vv) and vv < best_val:
                best_val = vv
                best_a = aa.copy()
                print("basin mutation obj=%.8f" % best_val)

        rem = max(0.0, deadline - time.monotonic() - 120.0)
        if rem > 8.0:
            aa, vv = run_mirror(
                best_a, S, min(20.0, rem * 0.25), rng,
                beta_init=800.0,
                beta_rate=0.09,
                beta_max=12000.0,
                eta_init=0.55
            )
            if np.isfinite(vv) and vv < best_val:
                best_val = vv
                best_a = aa.copy()
                print("basin mirror obj=%.8f" % best_val)

    # Final high-beta FISTA.
    rem = max(0.0, deadline - time.monotonic() - 25.0)
    if rem > 8.0:
        aa, vv = run_fista(
            best_a, S, rem, rng,
            beta_init=5000.0,
            beta_rate=0.20,
            beta_max=100000.0,
            step_factor=0.32,
            stuck_limit=3000
        )
        if np.isfinite(vv) and vv < best_val:
            best_val = vv
            best_a = aa.copy()
            print("final FISTA obj=%.8f" % best_val)

    # Final direct evaluation among a small pool; fallback to constant.
    pool = [best_a.copy()]
    if scored:
        pool.append(scored[0][2].copy())
    pool.append(np.full(n, S / float(n), dtype=np.float64))

    direct_best_a = None
    direct_best_val = np.inf

    for aa in pool:
        aa = normalize_to_sum(aa, S)
        lv = local_obj(aa)
        dv = direct_objective(aa)
        if np.isfinite(dv) and dv < direct_best_val:
            direct_best_val = dv
            direct_best_a = aa.copy()
        print("candidate local=%.8f direct=%.8f" % (lv, dv))

    if direct_best_a is None or not np.isfinite(direct_best_val):
        direct_best_a = np.full(n, S / float(n), dtype=np.float64)

    direct_best_a = clean_array(direct_best_a)
    direct_best_a = np.clip(direct_best_a, 0.0, 1000.0)

    s = float(direct_best_a.sum())
    if s < 0.01:
        direct_best_a = np.full(n, S / float(n), dtype=np.float64)
    else:
        direct_best_a = direct_best_a * (S / s)
        direct_best_a = np.clip(direct_best_a, 0.0, 1000.0)

    print("elapsed=%.1fs final local=%.8f" % (time.monotonic() - total_start, local_obj(direct_best_a)))
    return [float(x) for x in direct_best_a]
```
