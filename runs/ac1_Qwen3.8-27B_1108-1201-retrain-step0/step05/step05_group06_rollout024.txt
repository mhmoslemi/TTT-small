import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    M = 1 << (size - 1).bit_length()
    A = fft(a, M)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, S):
    """Normalize sequence to sum = S."""
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

    """
    Gerchberg-Saxton iterative projection to initialize a sequence
    whose self-convolution is approximately flat.
    
    Target: c[k] = S^2/(2n-1) for all k in [0, 2n-2]
    Target spectrum: |A[k]| = S/sqrt(2n-1) * |sinc(k*(2n-1)/M)|
    """
    rng = np.random.default_rng(seed)
    M = 1 << (2 * n - 2).bit_length()
    S = np.sqrt(2.0 * n)
    
    # Target magnitudes
    k = np.arange(M, dtype=np.float64)
    L = 2 * n - 1
    # sinc function: sin(pi*k*L/M) / sin(pi*k/M)
    # Handle k=0 separately (limit is L)
    target_mag = np.zeros(M)
    target_mag[0] = S  # DC component
    for idx in range(1, M):
        num = np.sin(np.pi * idx * L / M)
        den = np.sin(np.pi * idx / M)
        if abs(den) < 1e-15:
            target_mag[idx] = S / np.sqrt(L) * L
        else:
            target_mag[idx] = S / np.sqrt(L) * abs(num / den)
    
    # Initialize with random phases (Hadamard-like)
    A = np.zeros(M, dtype=np.complex128)
    A[0] = S
    for idx in range(1, M):
        A[idx] = target_mag[idx] * np.exp(1j * rng.uniform(0, 2 * np.pi))
    # Make it conjugate symmetric for real output
    for idx in range(1, M // 2 + 1):
        if idx < M - idx:
            A[M - idx] = np.conj(A[idx])
    
    a = np.zeros(n)
    best_a = a.copy()
    best_obj = np.inf
    
    for it in range(n_iter):
        a_new = np.real(ifft(A))[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-15:
            # Fallback
            a_new = np.ones(n)
        a_new = normalize_seq(a_new, S)
        
        if it % 100 == 0:
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
            if it % 500 == 0:
                print("    GS iter %d: obj=%.6f" % (it, obj))
        
        # Update spectrum
        Ap = fft(a_new, M)
        # Blend: keep some of the target magnitude
        phase_new = Ap / np.maximum(np.abs(Ap), 1e-30)
        # Use target magnitude
        A_new = target_mag * phase_new
        # Ensure conjugate symmetry
        for idx in range(1, M // 2 + 1):
            if idx < M - idx:
                A_new[M - idx] = np.conj(A_new[idx])
        A = A_new
    
    # Final check
    obj = compute_obj(a_new)
    if obj < best_obj:
        best_obj = obj
        best_a = a_new.copy()
    
    return best_a, best_obj

    return best_a, best_obj


def lse_gradient(a, c, beta):
    """Gradient of LSE(beta*c) w.r.t. a using FFT."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgrad(a, c, tol_frac=0.0):
    """Exact subgradient from active lags."""
    n = len(a)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_lse_fista(a_init, max_time, rng, beta_start=10.0, beta_end=5000.0):
    """LSE-FISTA with adaptive beta and periodic restarts."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_start

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cm = c.max()
        s = y.sum()
        cur_obj = 2.0 * n * cm / (s * s) if s > 1e-15 else np.inf

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive beta: increase over time
        progress = min(1.0, it / 50000.0)
        if stuck > 2000:
            beta = min(beta_end, beta * 1.02)
        else:
            beta = min(beta_end, beta_start * (1.0 + 0.01 * it))

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: conservative Lipschitz bound
        max_a = np.max(y)
        L_bound = 2.0 * beta * max_a + 2.0 * max_a + 1.0 / max(n, 1)
        step = 1.0 / max(L_bound, 0.1)
        step = min(step, S / (n * max(step, 1e-10) * 0.5))

        x_new = project_simplex_v(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Restart when stuck
        if stuck > 5000:
            t_mom = 1.0
            noise_scale = max(1e-6, 0.003 * S / np.sqrt(n) * np.exp(-stuck / 20000.0))
            if noise_scale > 1e-9:
                y = project_simplex_v(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it

        if it % 10000 == 0:
            t_mom = 1.0

    return best_a, best_obj


def run_subgradient_polish(a_init, max_time, rng):
    """Steepest descent with backtracking on the exact max objective."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    a = normalize_seq(a, S)
    start = time.monotonic()
    best_obj = compute_obj(a)
    best_a = a.copy()
    it = 0
    step_base = 0.5

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / (a.sum() ** 2)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()

        # Find active lags
        max_c = c.max()
        active = np.where(c >= max_c * (1.0 - 1e-8))[0]
        if len(active) > 8:
            # Keep only top 8
            top_idx = np.argsort(c[active])[::-1][:8]
            active = active[top_idx]

        # Subgradient (average over active lags)
        sg = np.zeros(n)
        for k in active:
            shift = k
            if shift < n:
                sg[:n - shift] += 2.0 * a[shift:]
            if shift > 0:
                sg[shift:] += 2.0 * a[:n - shift]
        sg /= len(active)

        sgnorm = np.linalg.norm(sg)
        if sgnorm < 1e-15:
            break

        # Direction: negative subgradient, projected to simplex tangent
        d = -sg
        # Remove component along a (tangent to level set of sum)
        a_sum = a.sum()
        d = d - (np.dot(d, a) / (a_sum * a_sum + 1e-30)) * a
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-15:
            d = -sg / sgnorm
            dnorm = 1.0
        else:
            d = d / dnorm

        # Backtracking line search
        step = step_base
        improved = False
        for _ in range(20):
            a_try = a + step * d
            a_try = np.maximum(a_try, 0.0)
            a_try = normalize_seq(a_try, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / (a_try.sum() ** 2)
            if obj_try < cur_obj - 1e-12:
                a = a_try
                step_base = min(step * 1.2, 2.0)
                improved = True
                break
            step *= 0.5
            if step < 1e-12:
                break

        if not improved:
            # Try smaller perturbation
            step = max(1e-8, step_base * 1e-3)
            a_try = a + step * d
            a_try = np.maximum(a_try, 0.0)
            a_try = normalize_seq(a_try, S)
            a = a_try

        # Periodic noise to escape
        if it % 200 == 0:
            noise = rng.normal(0, 1e-4 * S / np.sqrt(n), n)
            a = normalize_seq(np.maximum(a + noise, 0.0), S)

    return best_a, best_obj


def run_topk_polish(a_init, max_time, rng, k_start=4, k_end=64):
    """Polish using top-k weighted subgradient with FISTA momentum."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    a = normalize_seq(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cm = c.max()
        s = y.sum()
        cur_obj = 2.0 * n * cm / (s * s) if s > 1e-15 else np.inf
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve
        k = min(k_start + stuck // 100, k_end, len(c) // 2)

        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:k]
        w_full = np.zeros(len(c))
        vals = c[top_idx]
        # Weight by how close to max
        w_full[top_idx] = 1.0
        w_full[top_idx] = 1.0 / k

        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 0.5 / max(gnorm, 1.0)
        step = min(step, S / (n + 1.0))

        x_new = project_simplex_v(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stuck > 8000:
            t_mom = 1.0
            noise_scale = max(1e-7, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 20000.0))
            if noise_scale > 1e-10:
                y = project_simplex_v(y + rng.normal(0, noise_scale, n), S)
            stuck = 0
            last_improve = it

    return best_a, best_obj


def make_cosine_init(n):
    tt = np.linspace(0, 1, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt)


def make_chirp_init(n):
    tt = np.arange(n, dtype=np.float64)
    return 0.5 + 0.5 * np.cos(2 * np.pi * tt * tt / (2.0 * n))


def upsample(a, factor=2):
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: GS init -> LSE-FISTA -> subgradient polish -> multi-scale.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_best_a = None

    # Try to get provided sequence
    a_provided = None
    try:
        a_provided = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        a_provided = np.clip(a_provided, 0, 1000)
        if a_provided.sum() < 0.01:
            a_provided = None
    except Exception:
        a_provided = None

    # Determine base n
    if a_provided is not None:
        n0 = len(a_provided)
    else:
        n0 = 8192

    print("Base n = %d" % n0)

    # === Phase 1: Generate initializations ===
    n = n0
    S = np.sqrt(2.0 * n)
    candidates = []
    names = []

    if a_provided is not None:
        candidates.append(normalize_seq(a_provided, S))
        names.append("provided")

    # GS spectral init
    t_gs = time.monotonic()
    a_gs, obj_gs = gs_spectral_init(n, n_iter=5000, seed=seed, max_time=40.0)
    candidates.append(a_gs)
    names.append("gs_spectral")
    print("GS init: obj=%.6f (t=%.1fs)" % (obj_gs, time.monotonic() - t_gs))

    # Cosine
    a_cos = normalize_seq(make_cosine_init(n), S)
    candidates.append(a_cos)
    names.append("cosine")

    # Chirp
    a_chirp = normalize_seq(make_chirp_init(n), S)
    candidates.append(a_chirp)
    names.append("chirp")

    # Random
    a_rand = normalize_seq(rng.random(n) ** 0.5, S)
    candidates.append(a_rand)
    names.append("random")

    # Evaluate all
    print("\n=== Initialization evaluation (n=%d) ===" % n)
    best_init = None
    best_init_obj = np.inf
    for a_c, name in zip(candidates, names):
        obj = compute_obj(a_c)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_c.copy()
        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a_c.copy()
        print("  %s: obj=%.6f" % (name, obj))

    print("Best init: %s, obj=%.6f" % (names[candidates.index(best_init)], best_init_obj))

    # === Phase 2: LSE-FISTA main optimization ===
    remaining = deadline - time.monotonic()
    p2_budget = remaining * 0.40
    print("\n=== Phase 2: LSE-FISTA (budget %.0fs) ===" % p2_budget)
    t2 = time.monotonic()

    a_lse1, _ = run_lse_fista(best_init, p2_budget * 0.4, rng,
                                beta_start=5.0, beta_end=2000.0)
    obj_lse1 = compute_obj(a_lse1)
    print("  LSE1: obj=%.6f (t=%.1fs)" % (obj_lse1, time.monotonic() - t2))
    if obj_lse1 < global_best_obj:
        global_best_obj = obj_lse1
        global_best_a = a_lse1.copy()

    rem2 = p2_budget - (time.monotonic() - t2)
    if rem2 > 10:
        a_lse2, _ = run_lse_fista(global_best_a, rem2, rng,
                                    beta_start=50.0, beta_end=20000.0)
        obj_lse2 = compute_obj(a_lse2)
        print("  LSE2: obj=%.6f (t=%.1fs)" % (obj_lse2, time.monotonic() - t2))
        if obj_lse2 < global_best_obj:
            global_best_obj = obj_lse2
            global_best_a = a_lse2.copy()

    # === Phase 3: Subgradient polishing ===
    remaining = deadline - time.monotonic()
    p3_budget = remaining * 0.25
    print("\n=== Phase 3: Subgradient polish (budget %.0fs) ===" % p3_budget)
    t3 = time.monotonic()

    a_sg, _ = run_subgradient_polish(global_best_a, p3_budget * 0.6, rng)
    obj_sg = compute_obj(a_sg)
    print("  Subgrad: obj=%.6f (t=%.1fs)" % (obj_sg, time.monotonic() - t3))
    if obj_sg < global_best_obj:
        global_best_obj = obj_sg
        global_best_a = a_sg.copy()

    rem3 = p3_budget - (time.monotonic() - t3)
    if rem3 > 10:
        a_topk, _ = run_topk_polish(global_best_a, rem3, rng, k_start=2, k_end=32)
        obj_topk = compute_obj(a_topk)
        print("  TopK: obj=%.6f (t=%.1fs)" % (obj_topk, time.monotonic() - t3))
        if obj_topk < global_best_obj:
            global_best_obj = obj_topk
            global_best_a = a_topk.copy()

    # === Phase 4: LSE at high beta (final squeeze) ===
    remaining = deadline - time.monotonic()
    p4_budget = remaining * 0.20
    if p4_budget > 15:
        print("\n=== Phase 4: High-beta LSE (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()
        a_hb, _ = run_lse_fista(global_best_a, p4_budget, rng,
                                  beta_start=200.0, beta_end=100000.0)
        obj_hb = compute_obj(a_hb)
        print("  HighBeta: obj=%.6f (t=%.1fs)" % (obj_hb, time.monotonic() - t4))
        if obj_hb < global_best_obj:
            global_best_obj = obj_hb
            global_best_a = a_hb.copy()

    # === Phase 5: Multi-scale (upsample to 2n) ===
    remaining = deadline - time.monotonic()
    if remaining > 180 and len(global_best_a) <= 8192:
        n2 = len(global_best_a) * 2
        p5_budget = remaining * 0.75
        print("\n=== Phase 5: Upsample to n=%d (budget %.0fs) ===" % (n2, p5_budget))
        t5 = time.monotonic()

        a_up = upsample(global_best_a, factor=2)
        S2 = np.sqrt(2.0 * n2)
        a_up = normalize_seq(a_up, S2)

        # Quick LSE on upsampled
        a_up1, _ = run_lse_fista(a_up, p5_budget * 0.5, rng,
                                   beta_start=5.0, beta_end=3000.0)
        obj_up1 = compute_obj(a_up1)
        print("  Upsampled LSE: obj=%.6f (t=%.1fs)" % (obj_up1, time.monotonic() - t5))
        if obj_up1 < global_best_obj:
            global_best_obj = obj_up1
            global_best_a = a_up1.copy()

        rem5 = p5_budget - (time.monotonic() - t5)
        if rem5 > 15:
            a_up2, _ = run_subgradient_polish(global_best_a, rem5, rng)
            obj_up2 = compute_obj(a_up2)
            print("  Upsampled SG: obj=%.6f (t=%.1fs)" % (obj_up2, time.monotonic() - t5))
            if obj_up2 < global_best_obj:
                global_best_obj = obj_up2
                global_best_a = a_up2.copy()

    # === Phase 6: Final intensive polish ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p6_budget = remaining * 0.85
        print("\n=== Phase 6: Final polish (budget %.0fs) ===" % p6_budget)
        t6 = time.monotonic()

        a_fin, _ = run_subgradient_polish(global_best_a, p6_budget * 0.5, rng)
        obj_fin = compute_obj(a_fin)
        print("  Final SG: obj=%.6f (t=%.1fs)" % (obj_fin, time.monotonic() - t6))
        if obj_fin < global_best_obj:
            global_best_obj = obj_fin
            global_best_a = a_fin.copy()

        rem6 = p6_budget - (time.monotonic() - t6)
        if rem6 > 10:
            a_fin2, _ = run_lse_fista(global_best_a, rem6, rng,
                                        beta_start=500.0, beta_end=200000.0)
            obj_fin2 = compute_obj(a_fin2)
            print("  Final LSE: obj=%.6f (t=%.1fs)" % (obj_fin2, time.monotonic() - t6))
            if obj_fin2 < global_best_obj:
                global_best_obj = obj_fin2
                global_best_a = a_fin2.copy()

    # Final verification
    print("\n=== FINAL RESULT ===")
    print("Local best: obj=%.6f (n=%d)" % (global_best_obj, len(global_best_a)))
    final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):

    else:
        print("WARNING: evaluate_sequence rejected! Using fallback.")
        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]
        ver = evaluate_sequence(final_list)
        print("Fallback obj: %.6f" % ver)

    return final_list
