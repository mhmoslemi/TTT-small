        weights = np.ones(k) / k
    else:
        scaled = (c_top - c_top[-1]) / range_c * temperature
        weights = np.exp(scaled - scaled.max())
        weights /= weights.sum()
    
    a_flip = a[::-1]
    # Use FFT for efficiency when k is large
    if k > 20:
        w_vec = np.zeros(2 * n - 1)
        w_vec[top_idx] = weights
        fsize = 1 << (3 * n - 3).bit_length()
        WF = fft(w_vec, fsize)
        AF = fft(a_flip, fsize)
        conv = np.real(ifft(WF * AF))
        grad = 2.0 * conv[n - 1:n - 1 + n]
    else:
        grad = np.zeros(n)
        for ii in range(k):
            kidx = int(top_idx[ii])
            wi = weights[ii]
            i_start = max(0, kidx - n + 1)
            i_end = min(n - 1, kidx)
            if i_start <= i_end:
                indices = np.arange(kidx - i_end, kidx - i_start + 1)
                grad[i_start:i_end + 1] += 2.0 * wi * a[indices]
    return grad


def compute_lse_grad(a, c, beta):
    """Gradient of LSE-smoothed max convolution."""
    n = len(a)
    cm = c.max()
    delta = c - cm
    max_d = delta.max()
    if max_d > 1e-15:
        beta_eff = min(beta, 350.0 / max_d)
    else:
        beta_eff = 1.0
    scaled = beta_eff * delta
    e = np.exp(scaled - scaled.max())
    w = e / e.sum()
    
    a_flip = a[::-1]
    fsize = 1 << (3 * n - 3).bit_length()
    WF = fft(w, fsize)
    AF = fft(a_flip, fsize)
    conv = np.real(ifft(WF * AF))
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad


def obj(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def optimize_subgrad(a0, S, max_time, rng, momentum=0.9):
    """Subgradient descent with momentum and adaptive step size."""
    n = len(a0)
    a = project_simplex(a0.copy(), S)
    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()
    velocity = np.zeros(n)
    
    start = time.monotonic()
    it = 0
    stuck = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best - 1e-10:
            best = cur_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Adaptive k
        k = int(2 + 200 * progress ** 0.7)
        k = min(k, 2 * n - 1)
        
        grad = compute_grad_topk(a, c, k, temperature=3.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        gap = cur_max - c.mean()
        if gap < 1e-10:
            break
        
        # Step size with Polyak-like heuristic
        step = 2.0 * gap / (gnorm * gnorm + 1e-30)
        step = min(step, 3.0 / (gnorm + 1e-30))
        # Scale down with progress for more careful steps later
        step *= (1.0 - 0.5 * progress)
        
        # Momentum update
        velocity = momentum * velocity - step * grad
        a_try = project_simplex(a + velocity, S)
        
        # Line search
        accepted = False
        for ls in range(8):
            c_try = fft_convolve_self(a_try)
            if c_try.max() <= cur_max * (1.0 + 1e-8):
                a = a_try
                accepted = True
                break
            velocity *= 0.5
            a_try = project_simplex(a + velocity, S)
        
        if not accepted:
            a = project_simplex(a - 0.01 * step / gnorm * grad, S)
            velocity *= 0.1
        
        # Noise injection
        if stuck > 200:
            scale = min(0.05, 0.005 * stuck / 200) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, scale, n), S)
            velocity *= 0.1
            stuck = 0
    
    return best_a, best
