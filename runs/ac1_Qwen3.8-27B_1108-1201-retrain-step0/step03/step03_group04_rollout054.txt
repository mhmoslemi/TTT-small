Strategy: Multi-method approach with:

2. LP-based direction finding for refinement
3. Frequency-domain inspired smoothing

import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_gradient_fft(a, weights):
    """Gradient of weighted sum of convolution values w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def excess_weights(c, target, beta):
    """Weights proportional to excess over target, with softmax smoothing."""
    excess = np.maximum(c - target, 0)
    if excess.sum() < 1e-30:
        # Fall back to top-1
        top_idx = np.argmax(c)
        w = np.zeros(len(c))
        w[top_idx] = 1.0
        return w
    # Temperature-based weighting
    scaled = beta * excess
    scaled_max = scaled.max()
    exp_s = np.exp(scaled - scaled_max)
    total = exp_s.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return w
    return exp_s / total

    """FISTA with excess-based weighting and backtracking."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 0.1  # Initial step size
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = y.copy()
        
        # Excess-based weights: focus on lags above current best
        target = best_score * 0.999
        w = excess_weights(c, target, beta_init + 0.5 * it)
        grad = compute_gradient_fft(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        ls_step = step
        x_new = y - ls_step * grad
        x_new = project_simplex(x_new, S)
        
        # Check if it improves
        for _ in range(10):
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur:
                break
            ls_step *= 0.5
            x_new = project_simplex(y - ls_step * grad, S)
        
        step = ls_step
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic perturbation to escape local minima
        if it % 5000 == 0 and rng.random() < 0.3:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
    
    return best_local, best_score

    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)

    if s < 1e-15:
        a = np.ones(n) * S / n

        M = c.max()
        if M < best_score - 1e-14:
            best_score = M
            best_local = a.copy()
        
        # Solve LP: max sum(g) s.t. g >= 0, (a*g)(k) <= M for all k
        g = cp.Variable(n)
        constraints = [g >= 0]

        # (a*g)(k) = sum_i a(i) * g(k-i) for valid i
        # This is a convolution constraint

            # sum_{j=k-hi}^{k-lo} a[k-j] * g[j] <= M

                j = k - i
                coeffs[j] = a[i]
            constraints.append(coeffs @ g <= M)
        
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.SCS, max_iters=10000)
        
        if g.value is not None:

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    c = np.real(ifft(A * A))

        step = base_step / np.sqrt(it)

            t = 1.0
            base_step = min(base_step * 1.1, 2.0)

def balance_peak(a, n, S, step_fraction=1.0):
    """Balance the peak by moving mass from high-contribution to low-contribution positions."""
    c = fft_convolve_self(a)
    k_star = int(np.argmax(c))
    
    # For the peak lag k*, the contribution of a[i] is a[i] * a[k*-i]
    # We want to reduce a[i] where a[i]*a[k*-i] is large and increase it where it's small
    contributions = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            contributions[i] = a[i] * a[j]
        else:
            contributions[i] = 0

    to_idx = order[half:]

    total_move = max_from.sum()

    move_each = total_move / len(to_idx)
    
    a_new = a.copy()
    a_new[from_idx] -= max_from

    # Renormalize
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (S / s)
    
    return a_new

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase approach: Nesterov subgradient with adaptive step size,
peak-balancing heuristics, LP-based direction finding, and final fine-tuning.
Key innovations over previous approach:
- Proper subgradient step size schedule (1/sqrt(t))
- Peak-balancing coordinate moves
- LP-based direction for reducing specific peak lags
- Multiple diverse starting points
- Adaptive method switching based on progress
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def conv_grad_single_lag(a, k):
    """Gradient of (a*a)(k) w.r.t. a. Returns vector of length n."""
    n = len(a)
    result = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    js = np.arange(lo, hi + 1)
    result[js] = 2.0 * a[k - js]
    return result


def compute_grad_weighted(a, weights):
    """Compute weighted sum of convolution gradients via FFT."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def run_nesterov_subgrad(a_init, n, S, max_time, rng,
                         use_top1=True, perturb_every=5000):
    """Nesterov-accelerated subgradient with top-1 or top-k peak focus."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    best_step = 0.8
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()

        if cur < best_score - 1e-14:
            best_score = cur
            best_local = y.copy()
            no_improve = 0
            best_step = min(best_step * 1.005, 1.0)
        else:
            no_improve += 1
            if no_improve > 2000:
                best_step *= 0.95
                no_improve = 0

        # Gradient computation
        if use_top1:
            peak_lag = int(np.argmax(c))
            grad = conv_grad_single_lag(y, peak_lag)
        else:
            top_k = min(5, 2 * n - 1)
            top_lags = np.argpartition(c, -top_k)[-top_k:]
            excess = c[top_lags] - c[top_lags].min()
            excess = np.maximum(excess, 1e-6)
            w_local = excess / excess.sum()
            weights = np.zeros(2 * n - 1)
            weights[top_lags] = w_local
            grad = compute_grad_weighted(y, weights)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: G / sqrt(t) where G ~ S/n is typical gradient scale
        step = best_step * (S / n) / np.sqrt(1.0 + 0.001 * it)
        step = min(step, 2.0 * S / n)

        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        # Check improvement (every few iterations for speed)
        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic perturbation
        if it % perturb_every == 0 and rng.random() < 0.3:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            best_step = max(best_step * 0.9, 0.1)

    return best_local, best_score


def run_peak_balance(a_init, n, S, max_time, rng, step_frac=0.1):
    """Iterative peak balancing: move mass from high-contribution to low-contribution positions."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = a.copy()

        # Find top 3 peak lags
        top_lags = np.argpartition(c, -3)[-3:]
        top_lags = top_lags[np.argsort(c[top_lags])[::-1]]

        # Compute combined contribution at each position
        contributions = np.zeros(n)
        for k in top_lags:
            weight = c[k] - c[top_lags[-1]] + 1.0
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    contributions[i] += weight * a[j]

        # Sort by contribution
        order = np.argsort(contributions)
        half = n // 2
        low_idx = order[:half]
        high_idx = order[half:]

        # Move mass from high to low
        move_amount = step_frac * 0.01
        max_from = a[high_idx] * move_amount
        total = max_from.sum()
        if total < 1e-15:
            break
        per_to = total / len(low_idx)

        a_new = a.copy()
        a_new[high_idx] -= max_from
        a_new[low_idx] += per_to
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * (S / s_new)

        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur:
            a = a_new
            step_frac = min(step_frac * 1.01, 1.0)
        else:
            step_frac *= 0.7
            if step_frac < 0.001:
                break

    return best_local, best_score


def run_lp_direction(a_init, n, S, max_time, rng, num_lags=10):
    """LP-based direction finding: minimize (a*d)(k_peak) s.t. d >= -a, sum(d)=0."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = a.copy()

        # Find peak lags
        k_lags = np.argpartition(c, -num_lags)[-num_lags:]

        # For each peak lag, compute the "cost" of increasing each a[i]
        # (a*d)(k) = sum_i d[i] * a[k-i]
        # We want to minimize the max over peak lags
        # Approximate: minimize sum over peak lags of (a*d)(k)
        # = sum_i d[i] * sum_k a[k-i] (over peak lags k where k-i is valid)

        costs = np.zeros(n)
        for k in k_lags:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for i in range(lo, hi + 1):
                costs[i] += a[k - i]

        # LP: minimize costs @ d
        # s.t. d >= -a (non-negativity of a+d)
        #      sum(d) = 0
        #      d <= a (don't increase too much, optional)

        # Simplified: just move mass from high-cost to low-cost positions
        order = np.argsort(costs)
        n_move = max(1, n // 10)
        from_idx = order[n - n_move:]  # high cost
        to_idx = order[:n_move]  # low cost

        # Maximum move from "from" indices
        max_from = a[from_idx] * 0.05
        total = max_from.sum()
        if total < 1e-15:
            break
        per_to = total / len(to_idx)

        a_try = a.copy()
        a_try[from_idx] -= max_from
        a_try[to_idx] += per_to
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try > 1e-15:
            a_try = a_try * (S / s_try)

        c_try = fft_convolve_self(a_try)
        if c_try.max() < cur:
            a = a_try
        else:
            # Try smaller step
            a_try2 = a.copy()
            a_try2[from_idx] -= max_from * 0.1
            a_try2[to_idx] += per_to * 0.1
            a_try2 = np.maximum(a_try2, 0)
            s_try2 = a_try2.sum()
            if s_try2 > 1e-15:
                a_try2 = a_try2 * (S / s_try2)
            c_try2 = fft_convolve_self(a_try2)
            if c_try2.max() < cur:
                a = a_try2
            else:
                # Try gradient step as fallback
                peak_lag = int(np.argmax(c))
                grad = conv_grad_single_lag(a, peak_lag)
                gnorm = np.linalg.norm(grad)
                if gnorm > 1e-14:
                    for frac in [0.001, 0.0005, 0.0001]:

                        c_try3 = fft_convolve_self(a_try3)
                        if c_try3.max() < cur:
                            a = a_try3
                            break

    return best_local, best_score
