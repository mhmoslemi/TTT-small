"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization combining FISTA with adaptive LSE smoothing,
exact subgradient peak-targeting, and structured perturbations. Starts from the
given sequence and refines through progressively more aggressive peak reduction.
Key improvements over baseline: adaptive step-size with backtracking, hybrid
gradient (LSE + exact subgradient), sinusoidal escape moves, and multi-n exploration.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def topk_subgradient(a, c, K):
    """Subgradient using top-K largest lags with equal weight."""
    n = len(a)
    K = min(K, len(c))
    topk = np.argsort(c)[::-1][:K]
    grad = np.zeros(n)
    for k in topk:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    return grad / K


def soft_topk_gradient(a, c, K, margin=1e-6):
    """Gradient of soft top-K: sum max(0, c[i] - tau)."""
    n = len(a)
    K = min(K, len(c))
    sorted_c = np.sort(c)[::-1]
    tau = sorted_c[K - 1] + margin
    mask = c > tau
    n_active = mask.sum()
    if n_active == 0:
        return np.zeros(n)
    grad = np.zeros(n)
    active_idx = np.where(mask)[0]
    for k in active_idx:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    return grad / n_active


def hybrid_gradient(a, c, beta, use_exact_prob=0.3):
    """Hybrid: mix LSE gradient with exact subgradient."""
    n = len(a)
    g_lse = lse_gradient(a, c, beta)
    g_sub = exact_subgradient(a, c)
    # Normalize both to same scale
    nl = np.linalg.norm(g_lse)
    ns = np.linalg.norm(g_sub)
    if nl < 1e-30 and ns < 1e-30:
        return np.zeros(n)
    if nl < 1e-30:
        return g_sub
    if ns < 1e-30:
        return g_lse
    # Weight by relative magnitude, favor the subgradient
    w_sub = min(1.0, 0.5 + 0.5 * (ns / max(nl, 1e-30)))
    return (1 - w_sub) * (g_lse / nl) * ns + w_sub * g_sub


def run_fista_adaptive(a_init, n, S, max_time, rng, mode="adaptive"):

    FISTA with adaptive step size and hybrid gradients.
    mode: 'adaptive' (LSE with ramping beta), 'peak' (exact subgradient),

    y = a.copy()
    t = 1.0

    step_factor = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()

            best = cur
            best_a = y.copy()
            last_improve = it

            beta = min(2000.0, 8.0 * np.exp(0.006 * it))

                step = 1.0 / (2.0 * beta * max(0.1, cur / S))
            elif it < 1000:

                step = 1.0 / (2.0 * beta * 0.5 * max(0.1, cur / S))
            else:

                    grad = exact_subgradient(y, c)
                else:
                    grad = topk_subgradient(y, c, 3)
                gnorm = np.linalg.norm(grad)
                step = S / max(gnorm, 1e-15) * step_factor
        elif mode == "peak":
            if rng.random() < 0.5:
                grad = exact_subgradient(y, c)
            else:
                grad = topk_subgradient(y, c, 5)
            gnorm = np.linalg.norm(grad)

def run_peak_fighting(a_init, n, S, max_time, rng):
    """
    Dedicated peak-fighting: repeatedly target the current max lag
    with carefully sized steps.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()

    best_a = a.copy()
    it = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:

        c = fft_convolve_self(a)
        cur = c.max()

            best = cur
            best_a = a.copy()

        # Target the top few peaks
        K = min(10, 1 + it // 100)

        # Compute weighted subgradient
        vals = c[topk]
        weights = vals / vals.sum()

        grad = np.zeros(n)
        for wi, k in zip(weights, topk):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            grad[idx] += wi * 2.0 * a[k - idx]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step: start large, shrink when stuck
        base_step = S / (n * 2.0)
        step = base_step * step_scale
        x_new = project_simplex(a - step * grad / gnorm, S)

        # Check improvement
        c_new = fft_convolve_self(x_new)
        if c_new.max() < cur:
            a = x_new
            step_scale = min(2.0, step_scale * 1.05)
        else:
            step_scale = max(0.1, step_scale * 0.7)
            # Try half step
            step2 = step * 0.5
            x_half = project_simplex(a - step2 * grad / gnorm, S)
            c_half = fft_convolve_self(x_half)
            if c_half.max() < cur:
                a = x_half

        # Occasionally add noise

            tt = np.arange(n)
            noise = 0.001 * S / np.sqrt(n) * np.sin(2 * np.pi * freq * tt / n)
            a = project_simplex(a + noise, S)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_lse_refine(a_init, n, S, max_time, rng, beta_start=100.0, beta_end=5000.0):
    """Pure LSE refinement with exponentially increasing beta."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        it += 1
        # Exponential beta ramp
        progress = min(1.0, it / max(total_iters_est, it + 100))
        beta = beta_start * (beta_end / beta_start) ** progress

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size: 1/L where L ~ 2*beta*max(c) roughly
        step = 1.0 / (2.0 * beta * max(0.01, cur / S))
        step = min(step, 1.0 / gnorm * S * 0.1)

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def generate_structured_inits(n, rng):
    """Generate structured initializations for diversity."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)

    # Multi-sinusoidal
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * 3 * tt + 0.5))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.3, n)))

    return [np.maximum(x, 0.01) for x in inits]


def downsample_upsample(a, n_target):
    """Resample a to length n_target via averaging and interpolation."""
    n_orig = len(a)
    if n_target == n_orig:
        return a.copy()
    n_blocks = max(1, n_orig // n_target)
    a_small = a[:n_blocks * n_target].reshape(n_target, n_blocks).mean(axis=1)
    t_old = np.linspace(0, 1, n_target)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a_small)
