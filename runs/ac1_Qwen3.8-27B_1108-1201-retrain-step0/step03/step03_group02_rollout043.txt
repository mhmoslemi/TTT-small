def unit_convolution(a):
    """Compute self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho_idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / rho_idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Algorithm:
- Start from parent sequence (length 8192)
- Phase 1: Hybrid LP-guided + top-K subgradient optimization
  * Periodically solve reduced LP to get "spreading" direction
  * Use weighted top-K subgradient for fine direction
  * Backtracking line search with quadratic upper bound
  * Momentum with adaptive restart on non-improvement
- Phase 2: Fine-grained polish via coordinate descent on random blocks
- FFT-based O(n log n) convolution throughout
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def unit_convolution(a):
    """Compute full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho_idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / rho_idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, lags, weights):
    """
    Weighted sum of subgradients of (a*a)[k] w.r.t. a[j].
    subgrad[j] = 2 * sum_k w_k * a[k-j] (valid indices only)
    """
    n = len(a)
    w_arr = np.zeros(2 * n - 1)
    for i, k in enumerate(lags):
        w_arr[int(k)] = weights[i]
    fsize = 1 << (len(w_arr) + n - 1).bit_length()
    W = fft(w_arr, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return grad


def solve_lp_direction(f, M, S, K):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for top-K lags, g >= 0.
    Returns g (or None on failure).
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(f)
    c = unit_convolution(f)
    top_lags = np.argsort(c)[::-1][:K]

    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0]

    for k in top_lags:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min > j_max:
            continue
        length = j_max - j_min + 1
        vec = f[k - j_max:k - j_min + 1]
        if vec.sum() < 1e-30:
            continue
        constraints.append(vec @ g[j_min:j_max + 1] <= M)

    prob = cp.Problem(objective, constraints)
    for solver in [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if g_val.sum() < 1e-15:
        return None
    return g_val


def backtracking_step(f, grad, S, fsize_init, max_iter=50):
    """
    Backtracking line search for projected gradient step.
    Returns new point and step size used.
    """
    c_cur = unit_convolution(f)
    obj_cur = c_cur.max()

    step = fsize_init
    best_f = f.copy()
    best_obj = obj_cur

    for _ in range(max_iter):
        f_new = project_simplex(f - step * grad, S)
        if np.allclose(f_new, f):
            step *= 0.5
            continue
        c_new = unit_convolution(f_new)
        obj_new = c_new.max()
        if obj_new <= obj_cur * (1.0 - 1e-7):
            return f_new, step
        step *= 0.5
        if step < 1e-12:
            break

    return best_f, step


def run_hybrid_optim(a_init, n, S, max_time, rng, lp_interval=15, lp_K=50):
    """
    Main hybrid optimizer: LP-guided direction + top-K subgradient + momentum.
    """
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()  # momentum variable
    t_mom = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_obj = unit_convolution(a).max()
    it = 0
    step_init = S / (2.0 * np.sqrt(n))
    stale_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = unit_convolution(y)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = y.copy()
            stale_count = 0
        else:
            stale_count += 1

        # Determine top-K lags
        K = min(200, 20 + it // 5)
        top_lags = np.argsort(c)[::-1][:K]
        top_vals = c[top_lags]
        M = c.max()

        # Choose direction
        if it % lp_interval == 0 and it < max_time * 60:
            # Try LP direction
            g = solve_lp_direction(y, M, S, lp_K)
            if g is not None:
                g_sum = g.sum()
                if g_sum > 1e-15:
                    g = g * (S / g_sum)
                    # Blend direction
                    direction = g - y
                    dnorm = np.linalg.norm(direction)
                    if dnorm > 1e-15:
                        direction = direction / dnorm * dnorm  # keep as is
                        # Try a few step sizes
                        best_t = 0
                        best_f = y.copy()
                        best_o = cur_obj
                        for t_try in [0.05, 0.1, 0.2, 0.3, 0.5]:
                            f_try = project_simplex(y + t_try * direction, S)
                            c_try = unit_convolution(f_try)
                            if c_try.max() < best_o:
                                best_o = c_try.max()
                                best_f = f_try
                                best_t = t_try
                        if best_t > 0:
                            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
                            y = best_f + ((t_mom - 1.0) / t_new) * (best_f - a)
                            y = np.maximum(y, 0.0)
                            a = best_f
                            t_mom = t_new
                            continue
            # Fallback to subgradient

        # Top-K subgradient with proportional weights
        # Weight peaks by how much they exceed the median of top-K
        med = np.median(top_vals)
        weights = np.maximum(top_vals - med, 0.0) + 0.1 * M
        weights = weights / weights.sum()

        grad = compute_subgradient(y, top_lags, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = step_init / (1.0 + 0.001 * it)
        y_new, step_used = backtracking_step(y, grad, S, step, max_iter=30)

        # Momentum update with restart
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y_moment = y_new + ((t_mom - 1.0) / t_new) * (y_new - a)
        y_moment = np.maximum(y_moment, 0.0)

        # Check if momentum helps
        c_mom = unit_convolution(y_moment)
        if c_mom.max() < unit_convolution(y_new).max() + 1e-8:
            y = y_moment
            t_mom = t_new
        else:
            # Restart momentum
            y = y_new
            t_mom = 1.0

        a = y_new

        # Periodic reheat
        if stale_count > 50 and rng.random() < 0.3:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_mom = 1.0
            stale_count = 0

        # Gradually decrease initial step
        step_init *= 0.9995

    return best_a, best_obj


def run_coord_descent(a_init, n, S, max_time, rng):
    """Fine-grained coordinate descent on random blocks."""
    a = a_init.copy()
    start = time.monotonic()
    best_a = a.copy()
    best_obj = unit_convolution(a).max()
    block_size = 64

    while time.monotonic() - start < max_time:
        # Pick random block
        start_idx = rng.integers(0, n - block_size)
        block = a[start_idx:start_idx + block_size].copy()
        orig_sum = a.sum()

        # Try local optimization of this block via simple gradient steps
        c = unit_convolution(a)
        M = c.max()
        block_sum = block.sum()

        for inner in range(20):
            # Gradient of obj w.r.t. block elements
            grad_full = np.zeros(n)
            top_lags = np.argsort(c)[::-1][:20]
            weights = np.ones(len(top_lags))
            grad_full = compute_subgradient(a, top_lags, weights)

            block_grad = grad_full[start_idx:start_idx + block_size]
            step = block_sum / (50.0 + inner)
            block = np.maximum(block - step * block_grad, 0.0)

            # Rescale to maintain sum
            if block.sum() > 1e-15:
                block = block * (block_sum / block.sum())

        a_try = a.copy()
        a_try[start_idx:start_idx + block_size] = block
        a_try = project_simplex(a_try, S)

        c_try = unit_convolution(a_try)
        if c_try.max() < best_obj - 1e-12:
            best_obj = c_try.max()
            best_a = a_try.copy()
            a = a_try.copy()
        elif c_try.max() < best_obj:
            a = a_try.copy()

    return best_a, best_obj


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def get_initializations(n, rng):
    """Generate diverse asymmetric initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Exponential decays (asymmetric)
    for alpha in [2.0, 3.0, 4.0, 5.0, 7.0]:
        inits.append(np.exp(-alpha * tt))
        inits.append(np.exp(-alpha * (1.0 - tt)))

    # Power-law decays
    for p in [0.5, 1.0, 1.5, 2.0]:
        inits.append((tt + 0.02) ** (-p))
        inits.append((1.0 - tt + 0.02) ** (-p))

    # Bump + tail
    for center in [0.1, 0.2, 0.3]:
        width = 0.05
        bump = np.exp(-((tt - center) / width) ** 2)
        tail = np.exp(-3.0 * np.abs(tt - center)) * 0.3
        inits.append(bump + tail + 0.01)

    # Random with trend
    for trial in range(5):
        base = rng.random(n)
        trend = np.exp(-rng.uniform(1, 5) * tt)
        inits.append(base * trend + 0.01)

    return inits
