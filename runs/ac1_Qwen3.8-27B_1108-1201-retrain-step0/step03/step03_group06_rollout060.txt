1. Parametric shape search (Gaussian, raised cosine, polynomial windows)
2. Adaptive temperature that responds to objective plateau
3. Direct gradient computation via FFT-based cross-correlation
4. Multi-scale parallel optimization with different T values
5. Frequency-domain guided perturbation for restarts
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    fsize = 1 << (n - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:n]


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def compute_gradient(c, a, T):
    """Compute gradient of log-sum-exp smooth objective.
    Returns grad of (1/T)*log(sum exp(T*c)) w.r.t. a, plus p (softmax weights).
    """
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n), np.ones(len(c)) / len(c)

    scaled = T * c / (T * c_max + 1e-15)
    # Use stable softmax: T*c - max(T*c)
    Tc = T * c
    Tc_shifted = Tc - Tc.max()
    exp_Tc = np.exp(Tc_shifted)
    p = exp_Tc / exp_Tc.sum()

    # grad[m] = 2 * sum_k p[k] * a[k-m] (a[j]=0 outside [0,n-1])
    # = 2 * cross_corr(p, a)[m]
    fsize = 1 << (3 * n - 2).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(P * np.conj(A)))
    grad = 2.0 * corr[:n]
    return grad, p


def smooth_objective(c, T):
    """Compute smooth approximation of max(c)."""
    Tc = T * c
    return (Tc.max() + np.log(np.exp(Tc - Tc.max()).sum())) / T


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def make_gaussian(n, alpha):
    """Gaussian window: exp(-alpha * (j - center)^2)."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-alpha * (j - center) ** 2)
    return a


def make_raised_cosine(n):
    """Raised cosine window."""
    j = np.arange(n, dtype=np.float64)
    t = 2.0 * j / (n - 1) - 1.0
    a = 0.5 * (1.0 + np.cos(np.pi * t))
    return a


def make_parabolic(n):
    """Parabolic window: 1 - (2j/(n-1) - 1)^2."""
    j = np.arange(n, dtype=np.float64)
    t = 2.0 * j / (n - 1) - 1.0
    a = 1.0 - t ** 2
    return np.maximum(a, 0.0)


def make_polynomial(n, power):
    """Polynomial window: (1 - |2j/(n-1) - 1|^power)^(1/power)."""
    j = np.arange(n, dtype=np.float64)
    t = np.abs(2.0 * j / (n - 1) - 1.0)
    a = (1.0 - t ** power) ** (1.0 / power)
    return np.maximum(a, 0.0)


def make_super_gaussian(n, p):
    """Super-Gaussian: exp(-|j-center|^p)."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (j - center) / (n / 4.0)
    a = np.exp(-np.abs(x) ** p)
    return a


def make_sinc_window(n, width):
    """Truncated sinc window."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (j - center) / width
    a = np.sinc(x)  # sin(pi*x)/(pi*x)
    return np.maximum(a, 0.0)


def parametric_search(n, S, time_budget):
    """Search over parametric families for good initializations."""
    start = time.monotonic()
    best_obj = np.inf
    best_a = None

    candidates = []

    # Gaussian family
    for alpha in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        candidates.append(("gauss_alpha=%.4f" % alpha, make_gaussian(n, alpha)))

    # Raised cosine
    candidates.append(("raised_cosine", make_raised_cosine(n)))

    # Parabolic
    candidates.append(("parabolic", make_parabolic(n)))

    # Polynomial windows
    for p in [1.5, 2, 2.5, 3, 4, 6, 8]:
        candidates.append(("poly_p=%.1f" % p, make_polynomial(n, p)))

    # Super-Gaussian
    for p in [1, 1.5, 2, 3, 4, 6]:
        candidates.append(("super_gauss_p=%d" % p, make_super_gaussian(n, p)))

    # Sinc windows
    for w in [n/4, n/3, n/2, n/1.5]:
        candidates.append(("sinc_w=%.0f" % w, make_sinc_window(n, w)))

    # Asymmetric shapes (shifted Gaussians)
    for shift_frac in [0.3, 0.35, 0.4, 0.6, 0.65, 0.7]:
        for alpha in [0.01, 0.05, 0.1, 0.2]:
            j = np.arange(n, dtype=np.float64)
            center = shift_frac * (n - 1)
            a = np.exp(-alpha * (j - center) ** 2)
            candidates.append(("asym_gauss_s=%.2f_a=%.3f" % (shift_frac, alpha), a))

    # Product of two Gaussians (double-peak)
    for gap in [0.2, 0.3, 0.4]:
        for alpha in [0.01, 0.05, 0.1]:
            j = np.arange(n, dtype=np.float64)
            c1 = (1 - gap) / 2 * (n - 1)
            c2 = (1 + gap) / 2 * (n - 1)
            a = np.exp(-alpha * (j - c1) ** 2) + np.exp(-alpha * (j - c2) ** 2)
            candidates.append(("dual_gauss_g=%.1f_a=%.3f" % (gap, alpha), a))

    # Evaluate all candidates
    for name, a_raw in candidates:
        if time.monotonic() - start > time_budget:
            break
        a = normalize_nonneg(a_raw, S)
        c = fft_convolve_self(a)
        obj = c.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    print("  Parametric search: best_obj=%.7f (n=%d)" % (best_obj, n))
    return best_a, best_obj


def run_smooth_gradient_descent(a_init, n, S, time_budget, T_schedule="adaptive",
                                 T_init=10.0, T_max=5000.0, lr_init=0.1,
                                 verbose=False):
    """
    Gradient descent on log-parameters with log-sum-exp smooth objective.
    Adaptive temperature scheduling.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    a = np.maximum(a, 1e-15)
    theta = np.log(a / a.sum())

    n_par = n
    m_acc = np.zeros(n_par)
    v_acc = np.zeros(n_par)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    current_T = T_init

    start = time.monotonic()
    no_improve = 0
    plateau_threshold = 30

    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Adaptive T: increase when plateauing
        if T_schedule == "adaptive":
            if no_improve > plateau_threshold:
                current_T = min(current_T * 1.5, T_max)
                no_improve = 0
            elif t % 100 == 0 and progress > 0.3:
                current_T = min(current_T * 1.05, T_max)
        else:
            current_T = T_init * (T_max / T_init) ** (progress ** 0.5)

        # Convert theta to a
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n_par, S / n_par)
            theta = np.zeros(n_par)
        else:
            a = S * exp_theta / total

        # Compute autocorrelation
        c = fft_convolve_self(a)
        cur_obj = c.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute gradient
        grad_a, p = compute_gradient(c, a, current_T)
        dot = np.dot(grad_a, a)
        g_theta = grad_a * a - (a / S) * dot

        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break

        # Adam update
        m_acc = beta1 * m_acc + (1.0 - beta1) * g_theta
        v_acc = beta2 * v_acc + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m_acc / (1.0 - beta1 ** t)
        v_hat = v_acc / (1.0 - beta2 ** t)

        # Learning rate: decay with progress
        lr = lr_init * (1.0 - 0.6 * progress) * (1.0 / (1.0 + 0.1 * no_improve / 10))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if verbose and t % 200 == 0:
            print("    Step %d: obj=%.7f, T=%.1f, no_improve=%d" % (t, best_obj, current_T, no_improve))

    # Final: recompute from best_a
    return best_a, best_obj


def run_multi_scale(a_init, n, S, time_budget):
    """Run gradient descent at multiple T scales, keeping the best."""
    start = time.monotonic()
    best_a = a_init.copy()
    best_obj = np.inf

    T_configs = [
        (5.0, 100.0, 0.08),
        (10.0, 500.0, 0.06),
        (50.0, 2000.0, 0.04),
    ]

    per_time = time_budget / len(T_configs)

    for T_init, T_max, lr in T_configs:
        if time.monotonic() - start > time_budget - 2:
            break
        a_best, obj = run_smooth_gradient_descent(
            best_a, n, S, per_time,
            T_schedule="exponential", T_init=T_init, T_max=T_max, lr_init=lr
        )
        a_best = normalize_nonneg(a_best, S)
        c = fft_convolve_self(a_best)
        obj = c.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_best.copy()

    return best_a, best_obj


def run_anneal_restart(a_init, n, S, time_budget, n_restarts=5):
    """Annealing with periodic restarts using frequency-domain perturbations."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    per_restart = time_budget / (n_restarts + 1)
    rng = np.random.default_rng(777)

    for r in range(n_restarts):
        if time.monotonic() - start > time_budget - 2:
            break

        # Perturb in frequency domain
        A = fft(a, 2 * n - 1)
        noise_scale = 0.01 * (r + 1) * np.sqrt(n)
        phase_noise = noise_scale * rng.standard_normal(len(A))
        A_perturbed = A + 1j * phase_noise
        a_perturbed = np.real(ifft(A_perturbed))[:n]
        a_perturbed = normalize_nonneg(a_perturbed, S)

        # Optimize the perturbed version
        a_opt, obj = run_smooth_gradient_descent(
            a_perturbed, n, S, per_restart,
            T_schedule="adaptive", T_init=20.0 + r * 50, T_max=3000.0,
            lr_init=0.05
        )
        a_opt = normalize_nonneg(a_opt, S)
        c_opt = fft_convolve_self(a_opt)
        obj = c_opt.max()

        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            a = a_opt.copy()

    return best_a, best_obj


def coordinate_descent(a, n, S, time_budget, max_sweeps=20):
    """Coordinate descent: optimally adjust each coordinate."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(999)

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            original = a[i]
            if original < 1e-14:
                continue

            best_local = original
            best_local_obj = best_obj

            # Try multiplicative adjustments
            for factor in [0.3, 0.5, 0.7, 1.4, 1.7, 2.0, 0.1]:
                a_trial = a.copy()
                a_trial[i] = original * factor
                a_trial = normalize_nonneg(a_trial, S)
                c_check = fft_convolve_self(a_trial)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-12:
                    best_local_obj = obj_check
                    best_local = a_trial.copy()

            if best_local_obj < best_obj - 1e-12:
                a = best_local
                best_obj = best_local_obj
                best_a = a.copy()
                improved = True
            else:
                a[i] = original

        if not improved and sweep >= 2:
            break

    return best_a, best_obj


def try_zeroing(a, n, S, max_zeros=50):
    """Try zeroing small coordinates to potentially improve objective."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    sorted_idx = np.argsort(a)
    for idx in sorted_idx[:max_zeros * 5]:
        if a[idx] < 1e-12:
            continue
        saved = a[idx]
        a[idx] = 0.0
        a = normalize_nonneg(a, S)
        c_check = fft_convolve_self(a)
        obj_check = c_check.max()
        if obj_check < best_obj - 1e-12:
            best_obj = obj_check
            best_a = a.copy()
        else:
            a[idx] = saved
            a = normalize_nonneg(a, S)

    return best_a, best_obj


def propose_candidate(budget_s=2400):
    """Search for a non-negative sequence minimizing the evaluation function."""
    total_budget = max(1.0, float(budget_s) - 30.0)
    deadline = time.monotonic() + total_budget

    global_best = np.inf
    global_a = None

    # Load given sequence if available
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given sequence: " + str(e))

    # ============ Phase 1: Parametric search across lengths ============
    explore_budget = 0.15 * total_budget
    explore_deadline = time.monotonic() + explore_budget

    lengths_to_try = [4096]
    if given_a is not None:
        ng = len(given_a)
        if ng not in lengths_to_try:
            lengths_to_try.insert(0, ng)
    lengths_to_try.extend([2048, 8192, 16384])

    print("Phase 1: Parametric exploration (budget=%.0fs)" % explore_budget)
    per_length = max(3.0, explore_budget / len(lengths_to_try))

    for n in lengths_to_try:
        if time.monotonic() > explore_deadline - 3:
            break
        S = np.sqrt(2.0 * float(n))
        a_best, obj_best = parametric_search(n, S, per_length)

        if a_best is not None:
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()

    # ============ Phase 2: Gradient-based optimization ============
    exploit_budget = 0.40 * total_budget
    if global_a is not None and exploit_budget > 20:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a = normalize_nonneg(global_a, S)

        print("\nPhase 2: Gradient optimization n=%d (budget=%.0fs)" % (n, exploit_budget))

        # Multi-scale optimization
        ms_budget = 0.5 * exploit_budget
        a_best, _ = run_multi_scale(a, n, S, ms_budget)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  MultiScale: obj=%.7f" % ver)

        # Anneal + restart
        ar_budget = 0.3 * exploit_budget
        a_best, _ = run_anneal_restart(global_a, n, S, ar_budget, n_restarts=4)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  AnnealRestart: obj=%.7f" % ver)

        # High-T focused optimization
        ht_budget = 0.2 * exploit_budget
        a_best, _ = run_smooth_gradient_descent(
            global_a, n, S, ht_budget,
            T_schedule="adaptive", T_init=200.0, T_max=5000.0, lr_init=0.03
        )
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  HighT: obj=%.7f" % ver)

    # ============ Phase 3: Polish ============
    polish_budget = 0.25 * total_budget
    if global_a is not None and polish_budget > 10:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a = normalize_nonneg(global_a, S)

        print("\nPhase 3: Polish (budget=%.0fs)" % polish_budget)

        # Coordinate descent
        cd_budget = 0.4 * polish_budget
        a_best, _ = coordinate_descent(a, n, S, cd_budget, max_sweeps=25)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  CD: obj=%.7f" % ver)

        # Zeroing
        a_best, _ = try_zeroing(global_a, n, S, max_zeros=30)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  Zeroing: obj=%.7f" % ver)

        # Another CD pass
        remaining = polish_budget - 0.4 * polish_budget - 2
        if remaining > 3:
            a_best, _ = coordinate_descent(global_a, n, S, remaining, max_sweeps=15)
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  CD2: obj=%.7f" % ver)

    # ============ Phase 4: Final refinement ============
    final_budget = deadline - time.monotonic()
    if global_a is not None and final_budget > 10:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))

        print("\nPhase 4: Final refinement (budget=%.0fs)" % final_budget)

        # Adaptive T gradient descent with very high T
        a_best, _ = run_smooth_gradient_descent(
            global_a, n, S, 0.6 * final_budget,

        )
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  FinalGrad: obj=%.7f" % ver)

        # Final coordinate descent
        rem = final_budget * 0.4 - 2
        if rem > 2:
            a_best, _ = coordinate_descent(global_a, n, S, rem, max_sweeps=10)
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  FinalCD: obj=%.7f" % ver)

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:

    final_a = global_a.tolist()

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)

    return final_a
