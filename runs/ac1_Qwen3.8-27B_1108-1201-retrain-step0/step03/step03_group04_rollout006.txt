import numpy as np
import time
from scipy.fft import fft, ifft

def proj_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = np.where(valid)[0].max() + 1
    theta_val = (cumsum[rho-1] - S) / rho
    result = np.maximum(v - theta_val, 0.0)
    return result

def self_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1 << (2*n-1).bit_length()
    A = fft(a, size)
    return np.real(ifft(A*A))[:2*n-1]

def grad_smooth_max(a, c, beta):
    """Gradient of smooth_max(c) w.r.t. a, where c = self_conv(a)."""
    n = len(a)
    cmax = c.max()
    w = np.exp(beta * (c - cmax))
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.full_like(c, 1.0/(2*n-1))
    else:
        w /= wsum
    a_flip = a[::-1]
    # Compute conv(a_flip, w) via FFT
    size = 1 << (n + 2*n - 2).bit_length()  # n + (2n-1) - 1 = 3n-2
    Af = fft(a_flip, size)
    W = fft(w, size)
    conv = np.real(ifft(Af * W))[:3*n-2]
    return 2.0 * conv[n-1:2*n-1]

                            beta_schedule, iters_per_beta=5000):

    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    
    best_a = a.copy()
    best_val = self_conv(a).max()
    
    start = time.monotonic()
    
    for beta in beta_schedule:
        y = a.copy()
        t = 1.0
        # Lipschitz estimate (rough)
        L = 2.0 * beta * S  # rough upper bound, will refine
        
        for it in range(iters_per_beta):
            if time.monotonic() - start > time_budget:
                break
            
            c = self_conv(y)
            val = c.max()
            if val < best_val:
                best_val = val
                best_a = y.copy()
            
            g = grad_smooth_max(y, c, beta)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-12 * S:
                break

            step = 1.0 / L
            for _ in range(20):
                x_new = proj_simplex(y - step * g, S)
                # Check sufficient decrease
                c_new = self_conv(x_new)
                # Use the smooth approximation for the check
                w = np.exp(beta * (c - c.max()))
                w /= w.sum()
                old_smooth = (1.0/beta) * np.log(w.sum() * np.exp(beta * c.max()))  # just c.max() effectively
                # Actually for the line search, just check the actual max
                if c_new.max() <= val + 0.5 * step * np.dot(g, x_new - y):
                    break
                step *= 0.5
            
            # Nesterov update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0*t*t))
            y = x_new + ((t-1.0)/t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t = t_new
            
            # Update Lipschitz
            # L is related to the Hessian norm; for our problem it's roughly 2*beta*S
            # But we can adapt: if we needed many backtracks, increase L
            L = max(L * 1.0, 2.0 * beta * S * 0.5)
        
        # Reset momentum at each beta change
        t = 1.0
    
    return best_a, best_val

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using FISTA with β-continuation,
adaptive backtracking, and multi-phase peak-focused polishing.
Key improvements over previous: better β schedule, proper backtracking,
momentum reset, and targeted peak refinement.

import numpy as np
import time
from scipy.fft import fft, ifft


def proj_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = v.shape[0]
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    sv = np.sort(v)[::-1]
    css = np.cumsum(sv)
    rho = np.arange(1, n+1, dtype=np.float64)
    theta = (css - S) / rho
    valid = theta <= sv
    if not np.any(valid):
        k = n
    else:
        k = int(np.where(valid)[0].max()) + 1
    theta_k = (css[k-1] - S) / k
    return np.maximum(v - theta_k, 0.0)


def self_conv_fft(a):
    """Full self-convolution of a via FFT, length 2n-1."""
    n = a.shape[0]
    size = 1 << (2*n - 1).bit_length()
    A = fft(a, size)
    c = ifft(A * A)
    return np.real(c[:2*n-1])


def grad_weighted_conv(a, w):
    """Gradient of sum_k w[k]*c[k] where c = self_conv(a).
    = 2 * conv(a_reversed, w)[n-1 : 2n-1]
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # conv of a_rev (len n) and w (len 2n-1) → len 3n-2
    total_len = n + (2*n - 1) - 1  # = 3n-2
    size = 1 << total_len.bit_length()
    Ar = fft(a_rev, size)
    W = fft(w, size)
    conv = np.real(ifft(Ar * W))[:total_len]
    return 2.0 * conv[n-1 : 2*n-1]


def compute_objective(a):
    """Compute the evaluation objective (without calling evaluate_sequence)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = self_conv_fft(a)
    return 2.0 * n * c.max() / (s * s)


def normalize_to_sum(a, S):
    """Clip to [0, 1000] and rescale to sum = S."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, S / len(a))
    return a * (S / s)


def fista_beta_continuation(a_init, n, S, time_budget, rng,
                              betas=(10, 30, 100, 300, 1000, 3000, 10000)):
    """FISTA with continuation on beta (temperature for smooth max)."""
    a = normalize_to_sum(a_init.copy(), S)
    best_a = a.copy()
    best_c = self_conv_fft(a).max()
    start = time.monotonic()
    
    for beta in betas:
        if time.monotonic() - start >= time_budget:
            break
        phase_budget = (time_budget - (time.monotonic() - start)) / len(betas)
        
        y = a.copy()
        t_mom = 1.0
        step = 1.0  # Will be set by backtracking
        it = 0
        no_improve = 0
        
        while time.monotonic() - start < time_budget and it < int(1e7):
            it += 1
            
            c = self_conv_fft(y)
            cur = c.max()
            if cur < best_c:

                best_a = y.copy()
                no_improve = 0

            exp_bc = np.exp(beta * (c - cmax))
            wsum = exp_bc.sum()
            if wsum < 1e-300:

            else:
                w = exp_bc / wsum
            
            g = grad_weighted_conv(y, w)

            if gnorm < 1e-10 * S:
                break

            accepted = False
            for bt in range(15):
                x_new = proj_simplex(y - step * g, S)
                # Check: new max should be <= old max + small tolerance
                c_new = self_conv_fft(x_new)
                if c_new.max() <= cur + 1e-10 * S:
                    accepted = True
                    step = min(step * 1.5, 10.0 * S / (n * beta + 1))
                    break
                step *= 0.5
            if not accepted:
                # Too many backtracks, just take the step
                x_new = proj_simplex(y - step * g, S)
                step = max(step, 1e-10)
            
            # Nesterov momentum
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

                a = normalize_to_sum(a + noise, S)
                y = a.copy()
                t_mom = 1.0
                no_improve = 0

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using FISTA with β-continuation,
adaptive backtracking, peak-focused polishing, and diversified restarts.
Key improvements: better β schedule, proper gradient via FFT,
momentum management, targeted peak refinement, and multi-phase strategy.

import numpy as np
import time
from scipy.fft import fft, ifft


def proj_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = v.shape[0]
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    sv = np.sort(v)[::-1]
    css = np.cumsum(sv)
    rho = np.arange(1, n + 1, dtype=np.float64)
    theta = (css - S) / rho
    valid = theta <= sv
    if not np.any(valid):
        k = n
    else:
        k = int(np.where(valid)[0].max()) + 1
    theta_k = (css[k - 1] - S) / k
    return np.maximum(v - theta_k, 0.0)


def self_conv_fft(a):
    """Full self-convolution of a via FFT, returns array of length 2n-1."""
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()
    A = fft(a, size)
    c = ifft(A * A)
    return np.real(c[:2 * n - 1])


def grad_weighted_conv(a, w):
    """
    Gradient of sum_k w[k]*c[k] where c = self_conv(a).
    Formula: grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    Computed as: 2 * conv(a_reversed, w)[n-1 : 2n-1]
    """
    n = a.shape[0]
    a_rev = a[::-1]
    total_len = n + (2 * n - 1) - 1  # = 3n - 2
    size = 1 << (total_len - 1).bit_length()
    Ar = fft(a_rev, size)
    W = fft(w, size)
    conv = np.real(ifft(Ar * W))[:total_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def normalize_to_sum(a, S):
    """Clip to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, S / len(a))
    return a * (S / s)


def fista_continuation(a_init, n, S, time_budget, rng,
                       betas=(5, 15, 50, 150, 500, 1500, 5000, 15000)):
    """
    FISTA with continuation on beta (smooth max temperature).
    Uses backtracking line search and Nesterov momentum.
    """
    a = normalize_to_sum(a_init.copy(), S)
    best_a = a.copy()
    best_c = self_conv_fft(a).max()
    start = time.monotonic()

    for bi, beta in enumerate(betas):
        if time.monotonic() - start >= time_budget:
            break

        y = a.copy()
        t_mom = 1.0
        # Initial step size estimate
        step = S / (beta * n + 1.0)
        step = min(step, 10.0 * S / n)
        it = 0
        no_improve = 0

        while time.monotonic() - start < time_budget:
            it += 1

            c = self_conv_fft(y)
            cur = c.max()
            if cur < best_c - 1e-12:
                best_c = cur
                best_a = y.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Compute softmax weights
            cmax = c.max()
            exp_bc = np.exp(beta * (c - cmax))
            wsum = exp_bc.sum()
            if wsum < 1e-300:
                w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
            else:
                w = exp_bc / wsum

            # Gradient
            g = grad_weighted_conv(y, w)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-12 * S:
                break

            # Backtracking line search (limited)

            for bt in range(8):
                x_new = proj_simplex(y - step * g, S)
                c_new = self_conv_fft(x_new)
                if c_new.max() <= cur + 0.5 * step * np.dot(g, x_new - y) / max(gnorm, 1e-15) + 1e-10:
                    accepted = True
                    break
                step *= 0.5

                step = min(step * 1.3, 20.0 * S / n)

            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)

            t_mom = t_new

def peak_polish(a_init, n, S, time_budget, rng, top_k_max=4, beta_focus=100.0):

        c = self_conv_fft(a)
        cur = c.max()

            best_c = cur
            best_a = a.copy()
            step = min(step * 1.01, 2.0 * S / n)
        else:
            step *= 0.997

        tv_max = top_vals.max()
        weights = np.zeros(2 * n - 1)
        exp_w = np.exp(beta_focus * (top_vals - tv_max))
        wsum = exp_w.sum()
        if wsum < 1e-300:
            weights[top_idx[0]] = 1.0
        else:

                break

                a = normalize_to_sum(a + rng.normal(0, 0.0003 * S / np.sqrt(n), n), S)

        freq = 3 + restart_id * 2
        perturbed = a_base + 0.004 * S / np.sqrt(n) * np.sin(freq * t)
    else:
        # Mix of high and low frequency
        noise = rng.normal(0, 1, n)
        perturbed = a_base + 0.004 * S / np.sqrt(n) * noise

    perturbed = normalize_to_sum(perturbed, S)

        if c_p1 < global_best_c:
            global_best_c = c_p1
            global_best_a = a_p1.copy()
        print(f"  max_conv={c_p1:.8f}, obj={2*n*c_p1/S**2:.8f} [{elapsed:.0f}s]")

    # === Phase 2: Peak-focused polish (20% of time) ===
    remaining = deadline - time.monotonic()
    phase2_budget = 0.20 * remaining
    if phase2_budget > 5:
        print(f"\nPhase 2: Peak polish ({phase2_budget:.0f}s)")
        t0 = time.monotonic()
        a_p2, c_p2 = peak_polish(global_best_a, n, S, phase2_budget, rng,
                                  top_k_max=3, beta_focus=80.0)
        elapsed = time.monotonic() - t0
        if c_p2 < global_best_c:
            global_best_c = c_p2
            global_best_a = a_p2.copy()
        print(f"  max_conv={c_p2:.8f}, obj={2*n*c_p2/S**2:.8f} [{elapsed:.0f}s]")

    # === Phase 3: Second FISTA pass from improved point (15% of time) ===
    remaining = deadline - time.monotonic()
    phase3_budget = 0.15 * remaining
    if phase3_budget > 5:
        print(f"\nPhase 3: Second FISTA pass ({phase3_budget:.0f}s)")
        t0 = time.monotonic()
        a_p3, c_p3 = fista_continuation(
            global_best_a, n, S, phase3_budget, rng,
            betas=(100, 500, 2000, 10000, 30000)
        )
        elapsed = time.monotonic() - t0
        if c_p3 < global_best_c:
            global_best_c = c_p3
            global_best_a = a_p3.copy()
        print(f"  max_conv={c_p3:.8f}, obj={2*n*c_p3/S**2:.8f} [{elapsed:.0f}s]")

    # === Phase 4: Diversified restarts (10% of time) ===
    remaining = deadline - time.monotonic()
    phase4_budget = 0.10 * remaining
    if phase4_budget > 15:
        print(f"\nPhase 4: Diversified restarts ({phase4_budget:.0f}s)")
        n_restarts = 3
        per_restart = phase4_budget / n_restarts
        for r in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            t0 = time.monotonic()
            a_r, c_r = diversified_restart(
                global_best_a, n, S, per_restart, rng, r
            )
            elapsed = time.monotonic() - t0
            if c_r < global_best_c:
                global_best_c = c_r
                global_best_a = a_r.copy()
                print(f"  Restart {r}: max_conv={c_r:.8f}, obj={2*n*c_r/S**2:.8f} [{elapsed:.0f}s] ***")
            else:
                print(f"  Restart {r}: max_conv={c_r:.8f} [{elapsed:.0f}s] (no improvement)")

    # === Phase 5: Final ultra-fine polish (5% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 3:
        print(f"\nPhase 5: Final ultra-fine polish ({remaining:.0f}s)")
        t0 = time.monotonic()
        a_f, c_f = peak_polish(global_best_a, n, S, remaining, rng,
                                top_k_max=2, beta_focus=200.0)
        elapsed = time.monotonic() - t0
        if c_f < global_best_c:
            global_best_c = c_f
            global_best_a = a_f.copy()
        print(f"  max_conv={c_f:.8f}, obj={2*n*c_f/S**2:.8f} [{elapsed:.0f}s]")

    # === Final output ===
    global_best_a = normalize_to_sum(global_best_a, S)
    final_list = [float(x) for x in global_best_a]
    final_obj = 2.0 * n * global_best_c / (S * S)
    print(f"\nFINAL: n={n}, max_conv={global_best_c:.8f}, obj={final_obj:.8f}")

    # Verify with evaluator
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print(f"Verified objective: {ver:.8f}")
    else:
        print("Verification FAILED! Using fallback.")
        final_list = [float(x) for x in normalize_to_sum(given_a, S)]
        ver2 = evaluate_sequence(final_list)
        print(f"Fallback verified: {ver2:.8f}")

    return final_list
