def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z (softmax params)."""
    n = len(z)
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = np.max(c)
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_conv_grad(a, w)

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs(a_init, n, S, max_time, rng, beta_schedule=None, maxiter_per_beta=200):
    """L-BFGS-B with continuation on beta."""
    if beta_schedule is None:
        beta_schedule = [3, 5, 10, 20, 50, 100, 200, 400, 800, 1500, 3000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        for attempt in range(1 + (1 if beta >= 400 else 0)):
            if time.monotonic() - start > max_time - 3:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.01, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            try:
                result = scipy_minimize(
                    fun, z, jac=jac, method='L-BFGS-B',
                    options={'maxiter': maxiter_per_beta, 'maxfun': maxiter_per_beta * 2,
                             'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50, 'iprint': -1}
                )
                z = result.x
                z = z - z.mean()
            except Exception:
                break

            z_shifted = z - np.max(z)
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            c = fft_convolve_self(a)
            score = c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def run_repulsive_field(a_init, n, S, max_time, rng, 
                        tau_init=0.1, tau_min=0.01, n_steps=50000,
                        lr_init=0.01, momentum=0.9):
    """
    Smooth repulsive field method.
    Uses softmax weights on the autocorrelation to create a smooth gradient.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    v = np.zeros(n)  # momentum
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    
    tau = tau_init
    lr = lr_init
    
    for it in range(n_steps):
        if time.monotonic() - start > max_time:
            break
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            
        # Compute smooth weights (softmax of c/tau)
        c_shifted = c - c.max()
        exp_c = np.exp(c_shifted / tau)
        w = exp_c / exp_c.sum()
        
        # Compute gradient
        grad = compute_conv_grad(a, w)
        
        # Nesterov momentum
        v = momentum * v - lr * grad
        a_new = a + v
        
        # Project
        a_new = project_simplex(a_new, S)
        
        # Adaptive learning rate
        if it % 100 == 0:
            lr = lr_init / (1.0 + 0.01 * it)
            tau = max(tau_min, tau_init * np.exp(-0.005 * it))
        
        # Occasionally add noise for exploration
        if it % 5000 == 0 and it > 0:
            noise_scale = 0.001 * S / np.sqrt(n)
            idx = rng.choice(n, size=n//10, replace=False)
            a_new[idx] += rng.normal(0, noise_scale, len(idx))
            a_new = project_simplex(a_new, S)
            v = np.zeros(n)
        
        a = a_new
    
    return best_a, best_score


def run_simulated_annealing(a_init, n, S, max_time, rng, 
                            temp_init=0.1, temp_min=0.001,
                            n_perturb_range=(5, 50),
                            perturbation_scale=0.05):
    """
    Simulated annealing with structured moves.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    current_score = best_score
    
    total_time = max_time
    elapsed = 0.0
    
    it = 0
    while elapsed < max_time:
        it += 1
        # Cooling schedule (geometric)
        frac = it / 1000000  # Normalize
        temp = temp_init * np.exp(-3.0 * frac)
        if temp < temp_min:
            temp = temp_min
        
        # Choose number of coordinates to perturb
        n_perturb = rng.integers(n_perturb_range[0], n_perturb_range[1])
        
        # Propose move
        idx = rng.choice(n, size=n_perturb, replace=False)
        scale = perturbation_scale * S / np.sqrt(n) * (1.0 + 0.5 * np.sqrt(temp / temp_init))
        a_proposed = a.copy()
        a_proposed[idx] += rng.normal(0, scale, n_perturb)
        a_proposed = project_simplex(a_proposed, S)
        
        # Evaluate
        c = fft_convolve_self(a_proposed)
        proposed_score = c.max()
        
        # Accept/reject
        delta = proposed_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_proposed
            current_score = proposed_score
            if proposed_score < best_score:
                best_score = proposed_score
                best_a = a.copy()
        
        elapsed = time.monotonic() - start
    
    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""

    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using a multi-phase
optimization strategy:
1. L-BFGS with log-sum-exp continuation (adaptive beta schedule)
2. Smooth repulsive-field gradient method with Nesterov momentum
3. Simulated annealing with gradient-guided moves
4. Final L-BFGS polishing from multiple perturbed starts

Key insights:
- Score ≈ PARR of autocorrelation (scale-invariant, ~independent of n for large n)
- Optimal sequence is strongly asymmetric (symmetric gives score >= 2)
- Use antisymmetric component to suppress the middle autocorrelation peak
- Smooth softmax weighting on autocorrelation gives better gradient than hard top-k
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_conv_grad(a, w):
    """
    Gradient of sum_k w[k]*conv[k] w.r.t. a, where conv = a*a (full linear).
    d(conv[k])/d(a[j]) = 2*a[k-j] (with a[m]=0 outside [0,n-1]).
    So grad[j] = 2 * sum_k w[k]*a[k-j] = 2 * cross_correlation(w, a)[j].
    Computed via FFT: ifft(FFT(w) * conj(FFT(a))).
    """
    n = len(a)
    # Need N >= len(w) + n - 1 to avoid circular aliasing
    need = len(w) + n - 1
    fsize = 1 << (need - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * cross


def softmax_obj_grad(z, beta, S):
    """
    Compute smoothed objective (1/beta)*log(sum exp(beta*c_k)) and its gradient
    w.r.t. softmax parameters z, where a = S*softmax(z).
    """
    n = len(z)
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = np.max(c)
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_conv_grad(a, w)

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return float(obj), grad_z


def run_lbfgs(a_init, n, S, max_time, rng, beta_schedule=None, maxiter_per_beta=200):
    """L-BFGS-B with continuation on beta (smooth-to-sharp)."""
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        # At high beta, try 2 attempts with perturbation
        n_attempts = 2 if beta >= 500 and max_time > 30 else 1

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 3:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.008, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            try:
                result = scipy_minimize(
                    fun, z, jac=jac, method='L-BFGS-B',
                    options={'maxiter': maxiter_per_beta,
                             'maxfun': maxiter_per_beta * 3,
                             'ftol': 1e-16, 'gtol': 1e-11,
                             'maxls': 60, 'iprint': -1}
                )
                z = result.x
                z = z - z.mean()
            except Exception:
                break

            # Convert back
            z_shifted = z - np.max(z)
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, float(best_score)


def run_repulsive_field(a_init, n, S, max_time, rng,
                        tau_init=0.5, tau_decay=0.9995, tau_min=0.05,
                        lr_scale=0.1, momentum_init=0.9, momentum_decay=0.9999):
    """
    Smooth repulsive-field gradient method with adaptive step size.
    Uses softmax weighting on autocorrelation (smooth approximation to max).
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    v = np.zeros(n)
    start = time.monotonic()
    best_score = float(fft_convolve_self(a).max())
    best_a = a.copy()

    tau = tau_init
    momentum = momentum_init
    base_lr = lr_scale * S / np.sqrt(n)
    it = 0
    last_improve = 0
    stall_count = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur = float(c.max())
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Update temperature and momentum
        tau = max(tau_min, tau * tau_decay)
        momentum = max(0.8, momentum * momentum_decay)

        # Adaptive step size: reduce when stalling
        if stall_count > 2000:
            base_lr *= 0.7
            stall_count = 0
        elif it % 5000 == 0:
            base_lr = min(base_lr * 1.05, lr_scale * S / np.sqrt(n))

        # Compute smooth weights
        c_shifted = c - c.max()
        exp_c = np.exp(c_shifted / tau)
        w = exp_c / exp_c.sum()

        # Compute gradient
        grad = compute_conv_grad(a, w)

        # Nesterov momentum update
        v = momentum * v - base_lr * grad
        a_new = a + v

        # Project onto feasible set
        a_new = project_simplex(a_new, S)

        # Periodic exploration kicks
        if it % 10000 == 0:
            n_kick = max(100, n // 20)
            idx = rng.choice(n, size=n_kick, replace=False)
            kick_scale = 0.005 * S / np.sqrt(n)
            a_new[idx] += rng.normal(0, kick_scale, n_kick)
            a_new = np.maximum(a_new, 0.0)
            a_new = project_simplex(a_new, S)
            v = np.zeros(n)

        a = a_new

        if it % 20000 == 0:
            print("    RF iter %d: score=%.8f, tau=%.4f, lr=%.6f" % (it, best_score, tau, base_lr))

    return best_a, best_score


def run_simulated_annealing(a_init, n, S, max_time, rng,
                            temp_init=1.0, cool_rate=0.99995, temp_min=0.0005,
                            n_perturb_min=5, n_perturb_max=200,
                            base_scale=0.04):
    """
    Simulated annealing with gradient-informed move proposals.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = float(c.max())
    best_a = a.copy()
    current_score = best_score
    temp = temp_init
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        temp = max(temp_min, temp * cool_rate)

        # Move proposal: combine random perturbation with gradient direction
        n_perturb = int(rng.integers(n_perturb_min, n_perturb_max + 1))
        idx = rng.choice(n, size=n_perturb, replace=False)

        # Scale depends on temperature
        scale = base_scale * S / np.sqrt(n) * (1.0 + 2.0 * temp)
        a_proposed = a.copy()
        a_proposed[idx] += rng.normal(0, scale, n_perturb)

        # Occasionally do a "swap" move (transfer mass)
        if it % 7 == 0 and n >= 16:
            i1, i2 = rng.choice(n, size=2, replace=False)
            delta = rng.normal(0, 0.02 * S / np.sqrt(n))
            a_proposed[i1] += delta
            a_proposed[i2] -= delta

        a_proposed = np.maximum(a_proposed, 0.0)
        a_proposed = project_simplex(a_proposed, S)

        # Evaluate
        c_new = fft_convolve_self(a_proposed)
        proposed_score = float(c_new.max())

        # Metropolis acceptance
        delta = proposed_score - current_score
        if delta < 0 or (temp > temp_min * 2 and rng.random() < np.exp(-delta / temp)):
            a = a_proposed
            current_score = proposed_score
            if proposed_score < best_score:
                best_score = proposed_score
                best_a = a.copy()

        if it % 50000 == 0:
            print("    SA iter %d: best=%.8f, cur=%.8f, T=%.6f" % (it, best_score, current_score, temp))

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: L-BFGS continuation -> Repulsive field -> Simulated annealing -> Polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 12.0)

    global_best = np.inf
    global_a = None

    # --- Gather initializations ---
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0.0, 1000.0)))
    except Exception:
        pass

    if inits:
        n = len(inits[0][1])
    else:
        n = 8192
        inits.append(("ones", np.ones(n) * 0.5))

    S = np.sqrt(2.0 * float(n))
    tt = np.linspace(0, 1, n, endpoint=False)

    # Asymmetric initializations (key: optimal is asymmetric)
    inits.append(("rand", rng.random(n) + 0.1))

    # Strongly asymmetric: ramp shape
    ramp = 0.1 + 1.5 * tt + 0.3 * np.abs(np.sin(2 * np.pi * 4 * tt))
    inits.append(("ramp", ramp))

    # Two separated bumps (asymmetric positions)
    bump = (np.exp(-0.5 * ((tt - 0.15) * 12.0) ** 2) +
            0.6 * np.exp(-0.5 * ((tt - 0.75) * 6.0) ** 2) +
            0.5 * np.exp(-0.5 * ((tt - 0.5) * 4.0) ** 2) + 0.02)
    inits.append(("bump_asym", bump))

    # Frequency-modulated, abs to ensure positivity
    fm = (0.8 + 0.4 * np.sin(2 * np.pi * 3 * tt + 0.7 * np.sin(2 * np.pi * 11 * tt)) +
          0.2 * np.cos(2 * np.pi * 23 * tt) + 0.1 * np.sin(2 * np.pi * 7 * tt))
    fm = np.abs(fm) + 0.05
    inits.append(("fm", fm))

    # Concentrated at edges (minimizes middle autocorrelation)
    edge = np.exp(-8.0 * tt) + 0.7 * np.exp(-8.0 * (tt - 1.0)) + 0.03
    inits.append(("edge", edge))

    n_inits = len(inits)
    print("=== Opt: n=%d, %d inits, budget=%.0fs ===" % (n, n_inits, budget_s))
    print("Target: 1.5027, Previous SOTA: 1.508805")

    # Evaluate inits
    init_scores = []
    for name, arr in inits:
        a_norm = safe_normalize(arr, n)
        c = fft_convolve_self(a_norm)
        sc = float(c.max())
        init_scores.append(sc)
        if sc < global_best:
            global_best = sc
            global_a = a_norm.copy()

    for (name, _), sc in zip(inits, init_scores):
        print("  Init %-12s: %.8f" % (name, sc))
    print("Best init: %.8f" % global_best)

    # Sort inits by quality
    order = np.argsort(init_scores)
    sorted_inits = [inits[i] for i in order[:min(4, n_inits)]]

    # --- Phase 1: L-BFGS on top inits ---
    remaining = deadline - time.monotonic()
    n_p1 = len(sorted_inits)
    per_init = max(15.0, remaining * 0.30 / n_p1)

    print("\n=== Phase 1: L-BFGS Continuation (%.1fs each x%d) ===" % (per_init, n_p1))

    phase1_best = None
    phase1_score = np.inf

    for name, arr in sorted_inits:
        if time.monotonic() > deadline - 180:
            break
        t0 = time.monotonic()
        a_norm = safe_normalize(arr, n)
        a_new, score = run_lbfgs(
            a_norm, n, S, per_init, rng,
            beta_schedule=[5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000],
            maxiter_per_beta=180
        )
        a_new = safe_normalize(a_new, n)
        c = fft_convolve_self(a_new)
        sc = float(c.max())
        if sc < phase1_score:
            phase1_score = sc
            phase1_best = a_new.copy()
        if sc < global_best:
            global_best = sc
            global_a = a_new.copy()
            print("  [OK] %-12s: %.8f (%.1fs)" % (name, sc, time.monotonic() - t0))
        else:
            print("  %-12s: %.8f (best=%.6f, %.1fs)" % (name, sc, global_best, time.monotonic() - t0))

    print("After Phase 1: %.8f" % global_best)

    # --- Phase 2: Repulsive field method ---
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        seg_time = min(remaining * 0.25, 400.0)
        print("\n=== Phase 2: Repulsive Field (%.1fs) ===" % seg_time)

        # Run from best
        a_new, score = run_repulsive_field(
            global_a, n, S, seg_time, rng,
            tau_init=0.8, tau_decay=0.9997, tau_min=0.03,
            lr_scale=0.15, momentum_init=0.92, momentum_decay=0.99995
        )
        a_new = safe_normalize(a_new, n)
        c = fft_convolve_self(a_new)
        sc = float(c.max())
        if sc < global_best:
            global_best = sc
            global_a = a_new.copy()
            print("  RF (main): %.8f" % sc)
        else:
            print("  RF (main): %.8f (best=%.6f)" % (sc, global_best))

        # Run from perturbed version
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            seg2 = min(remaining2 * 0.3, 200.0)
            a_pert = global_a + rng.normal(0, 0.03 * S / np.sqrt(n), n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = safe_normalize(a_pert, n)
            a_new, score = run_repulsive_field(
                a_pert, n, S, seg2, rng,
                tau_init=1.5, tau_decay=0.9997, tau_min=0.1,
                lr_scale=0.08, momentum_init=0.85, momentum_decay=0.9999
            )
            a_new = safe_normalize(a_new, n)
            c = fft_convolve_self(a_new)
            sc = float(c.max())
            if sc < global_best:
                global_best = sc
                global_a = a_new.copy()
                print("  RF (perturbed): %.8f" % sc)
            else:
                print("  RF (perturbed): %.8f (best=%.6f)" % (sc, global_best))

    # --- Phase 3: Simulated Annealing ---
    remaining = deadline - time.monotonic()
    if remaining > 80 and global_a is not None:
        seg_time = min(remaining * 0.25, 400.0)
        print("\n=== Phase 3: Simulated Annealing (%.1fs) ===" % seg_time)

        a_new, score = run_simulated_annealing(
            global_a, n, S, seg_time, rng,
            temp_init=2.0, cool_rate=0.9999, temp_min=0.001,
            n_perturb_min=10, n_perturb_max=300,
            base_scale=0.03
        )
        a_new = safe_normalize(a_new, n)
        c = fft_convolve_self(a_new)
        sc = float(c.max())
        if sc < global_best:
            global_best = sc
            global_a = a_new.copy()
            print("  SA: %.8f" % sc)
        else:
            print("  SA: %.8f (best=%.6f)" % (sc, global_best))

        # Second SA round with different params
        remaining3 = deadline - time.monotonic()
        if remaining3 > 40:
            seg3 = min(remaining3 * 0.3, 200.0)
            a_new, score = run_simulated_annealing(
                global_a, n, S, seg3, rng,
                temp_init=0.5, cool_rate=0.99995, temp_min=0.0005,
                n_perturb_min=3, n_perturb_max=50,
                base_scale=0.015
            )
            a_new = safe_normalize(a_new, n)
            c = fft_convolve_self(a_new)
            sc = float(c.max())
            if sc < global_best:
                global_best = sc
                global_a = a_new.copy()
                print("  SA (fine): %.8f" % sc)
            else:
                print("  SA (fine): %.8f (best=%.6f)" % (sc, global_best))

    # --- Phase 4: Final L-BFGS polishing ---
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("\n=== Phase 4: Final L-BFGS Polish (%.1fs) ===" % remaining)

        n_polish = min(6, max(2, int(remaining / 25)))
        for p in range(n_polish):
            if time.monotonic() > deadline - 15:
                break
            noise_scale = 0.008 / (1.0 + 0.4 * p)
            a_pert = global_a + rng.normal(0, noise_scale * S / np.sqrt(n), n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = safe_normalize(a_pert, n)

            seg = min(25.0, (deadline - time.monotonic()) - 10)
            if seg < 8:
                break
            a_new, score = run_lbfgs(
                a_pert, n, S, seg, rng,
                beta_schedule=[200, 500, 1000, 3000, 10000],
                maxiter_per_beta=150
            )
            a_new = safe_normalize(a_new, n)
            c = fft_convolve_self(a_new)
            sc = float(c.max())
            if sc < global_best:
                global_best = sc
                global_a = a_new.copy()
            print("  Polish %d: %.8f" % (p, sc))

    # --- Phase 5: One more repulsive field pass (if time) ---
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        seg = min(remaining - 10, 100.0)
        if seg > 10:
            a_new, score = run_repulsive_field(
                global_a, n, S, seg, rng,
                tau_init=0.1, tau_decay=0.9998, tau_min=0.02,
                lr_scale=0.05, momentum_init=0.95, momentum_decay=0.99995
            )
            a_new = safe_normalize(a_new, n)
            c = fft_convolve_self(a_new)
            sc = float(c.max())
            if sc < global_best:
                global_best = sc
                global_a = a_new.copy()
            print("  Final RF: %.8f" % sc)

    # --- Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        try:
            ver = evaluate_sequence(final_list)
            print("\nVerification: %.6f" % ver if np.isfinite(ver) else "inf (rejected)")
            if not np.isfinite(ver):
                print("  evaluate_sequence rejected, using fallback")
                global_a = np.ones(256, dtype=np.float64) * 2.0
                global_best = float('inf')
        except Exception as e:
            print("  Eval error: %s" % str(e))
            global_a = np.ones(256, dtype=np.float64) * 2.0
    else:
        global_a = np.ones(256, dtype=np.float64) * 2.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * 2.0

    result = [float(x) for x in global_a]
    print("\nReturning length %d, best_score=%.8f" % (len(result), global_best))
    print("Time used: %.1fs" % (time.monotonic() - (deadline - budget_s + 12)))
    return result
