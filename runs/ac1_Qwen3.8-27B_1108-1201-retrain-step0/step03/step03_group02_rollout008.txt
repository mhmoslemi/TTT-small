def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):

    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def topk_subgradient(a, c, K):
    """
    Compute (scaled) subgradient of max(c) w.r.t. a using top-K peaks.
    Uses cross-correlation via FFT for efficiency.
    Returns gradient g such that moving a in -g direction reduces the top peaks.
    """
    n = len(a)
    top_indices = np.argsort(c)[::-1][:K]
    
    spike = np.zeros(2 * n - 1)
    for idx in top_indices:
        spike[int(idx)] = 1.0 / K
    
    fsize = 1 << (len(spike) + n - 1).bit_length()
    W = fft(spike, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_flatten_step(a, S, lam, rng):
    """
    Apply spectral compression to reduce peaks in autocorrelation.
    Modifies the power spectrum to be flatter, then projects back.
    """
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    A = fft(a, fsize)
    P = np.abs(A) ** 2
    phase = np.angle(A)
    
    # Compress the dynamic range of the power spectrum
    P_mean = P.mean()
    if P_mean < 1e-30:
        return a.copy()
    
    # Apply exponential compression to peaks above mean
    P_new = P.copy()
    excess = np.maximum(P_new - P_mean, 0)
    P_new = P_new - lam * excess
    P_new = np.maximum(P_new, 0.0)
    
    # Ensure energy is preserved (roughly)
    energy = P.sum()
    new_energy = P_new.sum()
    if new_energy < 1e-30:
        return a.copy()
    P_new = P_new * (energy / new_energy)
    
    # Reconstruct
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, fsize))[:n]
    
    # Project to non-negative and renormalize
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new = a_new * (S / s)
    
    return a_new

def backtrack_step(a, grad, S, c_current, c_max_current, alpha_init, max_backtracks=25):
    """
    Backtracking line search for subgradient step.
    Finds alpha such that max(conv(a_new, a_new)) < max(conv(a, a)).
    """
    alpha = alpha_init
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a, c_current, c_max_current, 0.0
    
    grad_normed = grad / gnorm
    
    for _ in range(max_backtracks):
        a_new = project_simplex(a - alpha * grad_normed * gnorm * 0.5, S)
        c_new = fft_convolve_self(a_new)
        c_max_new = c_new.max()
        
        if c_max_new < c_max_current:
            return a_new, c_new, c_max_new, alpha
        alpha *= 0.5
    
    return a, c_current, c_max_current, 0.0

def normalize_to_sqrt2n(a, n):
    """Normalize sequence to have sum = sqrt(2n)."""
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    a = normalize_to_sqrt2n(parent, n)
    
    # Verify starting point
    c_start = fft_convolve_self(a)
    start_c_max = c_start.max()
    print("Starting: n=%d, max_c=%.8f, sum=%.6f" % (n, start_c_max, a.sum()))
    
    global_best_a = a.copy()
    global_best_c_max = start_c_max
    
    # --- Phase 1: Subgradient + Spectral (35% of time) ---
    phase1_time = (deadline - time.monotonic()) * 0.35
    print("=== Phase 1: Subgradient+spectral (%.1f sec) ===" % phase1_time)
    
    if phase1_time > 10:
        a1, c1 = run_subgradient_optimize(
            a, S, phase1_time, rng,
            init_K=8, final_K=3, alpha_init=2.0,
            spectral_interval=30, spectral_lam=0.15, verbose=True)
        
        if c1 < global_best_c_max:
            global_best_c_max = c1
            global_best_a = a1.copy()
            print("  Phase 1 result: %.8f" % c1)
        else:
            print("  Phase 1 did not improve (still %.8f)" % global_best_c_max)
    
    # --- Phase 2: Fine polishing (40% of time) ---
    remaining = deadline - time.monotonic()
    phase2_time = remaining * 0.55
    print("=== Phase 2: Fine polish (%.1f sec) ===" % phase2_time)
    
    if phase2_time > 10:
        a2, c2 = run_fine_polish(global_best_a, S, phase2_time, rng)
        
        if c2 < global_best_c_max:
            global_best_c_max = c2
            global_best_a = a2.copy()
            print("  Phase 2 result: %.8f" % c2)
        else:
            print("  Phase 2 did not improve (still %.8f)" % global_best_c_max)
    
    # --- Phase 3: Multi-restart (remaining time) ---
    remaining = deadline - time.monotonic()
    phase3_time = remaining * 0.8
    print("=== Phase 3: Multi-restart (%.1f sec) ===" % phase3_time)
    
    if phase3_time > 10:
        a3, c3 = run_multi_restart(global_best_a, S, phase3_time, rng, n_restarts=8)
        
        if c3 < global_best_c_max:
            global_best_c_max = c3
            global_best_a = a3.copy()
            print("  Phase 3 result: %.8f" % c3)
        else:
            print("  Phase 3 did not improve (still %.8f)" % global_best_c_max)
    
    # --- Final micro-polish with whatever time remains ---
    remaining = deadline - time.monotonic()
    if remaining > 5:
        print("=== Final micro-polish (%.1f sec) ===" % remaining)
        a_f, c_f = run_fine_polish(global_best_a, S, remaining, rng)
        if c_f < global_best_c_max:
            global_best_c_max = c_f
            global_best_a = a_f.copy()
            print("  Final: %.8f" % c_f)
    
    # --- Verification ---
    final_a = np.clip(global_best_a, 0.0, 1000.0)
    final_a = normalize_to_sqrt2n(final_a, n)
    
    final_list = [float(x) for x in final_a]
    ver = evaluate_sequence(final_list)
    
    print("Internal best: %.8f" % global_best_c_max)
    print("Verified by evaluate_sequence: %s" % ("%.8f" % ver if np.isfinite(ver) else "inf"))
    
    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Using fallback.")
        final_a = np.ones(n, dtype=np.float64) * S / n
        final_list = [float(x) for x in final_a]
    
    # Double-check: also verify the internal score matches
    c_final = fft_convolve_self(final_a)
    internal = 2.0 * n * c_final.max() / (final_a.sum() ** 2)
    print("Final: len=%d, internal=%.8f, verified=%.8f" % (len(final_list), internal, ver))
    
    return final_list
