import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def exact_subgradient(a, k_star):
    """Exact subgradient of max at lag k_star: 2*a shifted by k_star."""
    n = len(a)
    g = np.zeros(n)
    # g[i] = 2*a[k_star - i] if 0 <= k_star - i < n
    i_min = max(0, k_star - (n - 1))
    i_max = min(k_star, n - 1)
    if i_min <= i_max:
        indices = np.arange(i_min, i_max + 1)
        g[indices] = 2.0 * a[k_star - indices]
    return g


def topk_subgradient(a, c, top_k):
    """Subgradient averaged over top-k lags."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    g = np.zeros(n)
    for idx in top_indices:
        idx = int(idx)
        i_min = max(0, idx - (n - 1))
        i_max = min(idx, n - 1)
        if i_min <= i_max:
            indices = np.arange(i_min, i_max + 1)
            g[indices] += 2.0 * a[idx - indices]
    g /= k
    return g


def softmax_subgradient(a, c, temperature):
    """Subgradient with softmax weighting over all lags."""
    n = len(a)
    c_max = c.max()
    weights = np.exp(c / max(temperature, 1e-10) - c_max / max(temperature, 1e-10))
    weights = weights / weights.sum()
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    return a


def make_periodic_init(n, rng, S, num_freqs=3):
    """Periodic init with multiple sinusoids."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 25)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.5
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_bimodal_init(n, rng, S):
    """Two Gaussian bumps at random positions."""
    a = np.zeros(n)
    for _ in range(2):
        center = rng.integers(n // 4, 3 * n // 4)
        width = rng.integers(n // 20, n // 5)
        tt = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((tt - center) / width) ** 2)
    a = a + 0.01
    a = a * (S / a.sum())
    return a


def make_cosine_window_init(n, rng, S):
    """Raised cosine window init."""
    tt = np.arange(n, dtype=np.float64) / n
    k = rng.integers(2, 8)
    a = 1.0 + 0.4 * np.cos(2 * np.pi * k * tt)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_sparse_init(n, rng, S, density=0.5):
    """Sparse init: only a fraction of positions are non-zero."""
    mask = rng.random(n) < density
    a = np.zeros(n)
    a[mask] = rng.random(mask.sum()) + 0.1
    a = a * (S / a.sum())
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    S = np.sqrt(2.0 * n)
    inits = []
    if parent is not None and len(parent) >= 64:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if p.sum() > 0:
            inits.append(p * (S / p.sum()))
    for i in range(2):
        inits.append(make_anti_palindromic_init(n, rng, S))
    for i in range(2):
        inits.append(make_periodic_init(n, rng, S))
    inits.append(make_bimodal_init(n, rng, S))
    inits.append(make_cosine_window_init(n, rng, S))
    inits.append(make_sparse_init(n, rng, S))
    # Random
    rand = rng.random(n) + 0.05
    inits.append(rand * (S / rand.sum()))
    return inits


def run_fista_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=200.0):
    """FISTA with log-sum-exp smoothing and adaptive beta."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = 4.0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stall = 0
        elif it % 100 == 0:
            stall += 1

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / L
        f_y = c.max()
        x_new = project_simplex(y - step * grad, S)

        # Backtracking every 10 iterations
        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() > f_y * 1.01 + 1e-10:
                L *= 1.3
            elif it % 100 == 0:
                L = max(L * 0.97, 2.0)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stall > 50 and rng.random() < 0.2:
            noise = 0.02 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0
            stall = 0

    return best_local, best_score


def run_exact_subgrad_nesterov(a_init, n, S, max_time, rng, top_k_init=1):
    """Exact subgradient with Nesterov acceleration and backtracking."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    base_step = 2.0 * S / (n * 5.0)

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(5, top_k_init + it // 1000)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = topk_subgradient(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step
        step = base_step / max(gnorm / (S / n), 0.1)
        step = min(step, 4.0 * S / n)

        # Backtracking line search
        alpha = 1.0
        f_y = c.max()
        x_new = None
        for bt in range(8):
            x_candidate = project_simplex(y - alpha * step * grad, S)
            if bt == 0 or it % 5 == 0:
                c_cand = fft_convolve_self(x_candidate)
                if c_cand.max() <= f_y + 1e-10:
                    x_new = x_candidate
                    break
            else:
                x_new = x_candidate
                break
            alpha *= 0.5

        if x_new is None:
            x_new = project_simplex(y - alpha * step * grad, S)

        # Nesterov momentum with restart
        t_prev = t_mom
        t_mom = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))

        # Check for restart: if gradient direction changed a lot
        if it > 10 and it % 5 == 0:
            c_new = fft_convolve_self(x_new)
            if c_new.max() > f_y * 1.005:
                t_mom = 1.0

        y = x_new + ((t_prev - 1.0) / t_mom) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new

        # Periodic perturbation
        if it % 2000 == 0 and rng.random() < 0.3:
            noise = 0.01 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0

    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, eta_init=0.1, top_k=3):
    """Mirror descent (multiplicative updates) with top-k subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    # Ensure no zeros for log
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        eta = eta_init / (1.0 + 0.0005 * it)
        grad = topk_subgradient(a, c, top_k)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        log_update = -eta * grad / gmax
        log_update = np.clip(log_update, -3.0, 3.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-10)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new * (S / s_new)
        else:
            a = np.ones(n) * S / n
            a = np.maximum(a, 1e-10)

    return best_local, best_score


def run_pairwise_swap_polish(a_init, n, S, max_time, rng):
    """Polish using pairwise mass swaps targeting the peak lag."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Find peak lag
        k_star = int(np.argmax(c))

        # Find contributing pairs for this lag
        i_min = max(0, k_star - (n - 1))
        i_max = min(k_star, n - 1)
        if i_min > i_max:
            break
        indices = np.arange(i_min, i_max + 1)
        partners = k_star - indices
        products = a[indices] * a[partners]

        # Sort by product (largest contributors first)
        order = np.argsort(products)[::-1]

        # Try to swap mass from high-product pairs to low-product pairs
        n_tries = min(20, len(order) // 2)
        improved = False

        for trial in range(n_tries):
            # Pick a high-product index
            hi_idx = order[trial] if trial < len(order) else order[0]
            # Pick a low-product index
            lo_idx = order[min(trial + len(order) // 2, len(order) - 1)]

            hi_i = indices[hi_idx]
            hi_p = partners[hi_idx]
            lo_i = indices[lo_idx]
            lo_p = partners[lo_idx]

            # Try swapping: reduce a[hi_i] (or hi_p), increase a[lo_i] (or lo_p)
            delta = 0.005 * S / n
            for (src, dst) in [(hi_i, lo_i), (hi_p, lo_p), (hi_i, lo_p), (hi_p, lo_i)]:
                if src == dst:
                    continue
                if a[src] < delta:
                    continue
                a_trial = a.copy()
                a_trial[src] -= delta
                a_trial[dst] += delta
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial * (S / a_trial.sum())
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur - 1e-12:
                    a = a_trial
                    cur = c_trial.max()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try random small perturbations
            noise = 0.003 * S / np.sqrt(n)
            a_trial = project_simplex(a + rng.normal(0, noise, n), S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < best_score:
                a = a_trial

        if it % 500 == 0:
            if best_score - cur < 1e-6:
                # Converged, try larger perturbation
                noise = 0.02 * S / np.sqrt(n)
                a_trial = project_simplex(a + rng.normal(0, noise, n), S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < best_score:
                    a = a_trial

    return best_local, best_score


def run_coordinate_descent(a_init, n, S, max_time, rng):
    """Coordinate descent: optimize pairs of elements at a time."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        k_star = int(np.argmax(c))

        # Find the pair with largest product at the peak lag
        i_min = max(0, k_star - (n - 1))
        i_max = min(k_star, n - 1)
        if i_min > i_max:
            break
        indices = np.arange(i_min, i_max + 1)
        partners = k_star - indices
        products = a[indices] * a[partners]

        # Find max and min product indices
        max_pos = np.argmax(products)
        min_pos = np.argmin(products)

        hi_i = indices[max_pos]
        hi_p = partners[max_pos]
        lo_i = indices[min_pos]
        lo_p = partners[min_pos]

        # Move mass from the larger element in the high-product pair
        # to the smaller element in the low-product pair
        if hi_i != hi_p:
            if a[hi_i] >= a[hi_p]:
                src = int(hi_i)
            else:
                src = int(hi_p)
        else:
            src = int(hi_i)

        if lo_i != lo_p:
            if a[lo_i] <= a[lo_p]:
                dst = int(lo_i)
            else:
                dst = int(lo_p)
        else:
            dst = int(lo_i)

        if src == dst:
            continue

        # Optimal delta: binary search for the amount that reduces the peak
        max_delta = a[src] * 0.5
        lo = 0.0
        hi = max_delta
        for _ in range(15):
            mid = (lo + hi) / 2.0
            a_trial = a.copy()
            a_trial[src] -= mid
            a_trial[dst] += mid
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial > 1e-15:
                a_trial = a_trial * (S / s_trial)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur:
                lo = mid
                a_best_trial = a_trial.copy()
            else:
                hi = mid

        if lo > 1e-15:
            a = a_best_trial

    return best_local, best_score


def evaluate_internal(a):
    """Internal evaluation: 2*n*max(conv)/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    primary_n = 8192
    S_primary = np.sqrt(2.0 * primary_n)

    if parent is not None and len(parent) >= 64:
        # Use parent length if reasonable
        if 2048 <= len(parent) <= 16384:
            primary_n = len(parent)
            S_primary = np.sqrt(2.0 * primary_n)

    print("Primary length: %d, parent: %s" % (primary_n, "yes" if parent is not None else "no"))
    print("Budget: %.0f s, deadline at t=%.1f" % (budget_s, deadline - time.monotonic()))

    # === Phase 1: FISTA with log-sum-exp (30% of time) ===
    t_phase1_end = time.monotonic() + (deadline - time.monotonic()) * 0.25
    inits = generate_inits(primary_n, rng, parent)
    n_inits = len(inits)
    per_init = max(5.0, (t_phase1_end - time.monotonic()) / n_inits)

    print("=== Phase 1: FISTA LSE, %d inits, %.1f s each ===" % (n_inits, per_init))

    phase1_best = None
    for name_idx, arr in enumerate(inits):
        if time.monotonic() > t_phase1_end:
            break
        n = len(arr)
        S = np.sqrt(2.0 * n)
        t0 = time.monotonic()

        a_f, s_f = run_fista_lse(arr, n, S, per_init, rng,
                                  beta_init=5.0, beta_max=150.0)
        a_f = safe_normalize(a_f, n)
        obj = evaluate_internal(a_f)
        if obj < global_best:
            global_best = obj
            global_a = a_f.copy()
            print("  FISTA[%d]: obj=%.6f (t=%.0fs)" % (name_idx, obj, time.monotonic() - t0))

    print("=== After Phase 1: best=%.6f ===" % global_best)

    # === Phase 2: Exact subgradient with Nesterov (35% of time) ===
    t_phase2_end = time.monotonic() + (deadline - time.monotonic()) * 0.35
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        print("=== Phase 2: Exact subgrad + Nesterov (remaining %d s) ===" %
              int(t_phase2_end - time.monotonic()))

        a_p2, s_p2 = run_exact_subgrad_nesterov(global_a, n, S,
                                                 t_phase2_end - time.monotonic(),
                                                 rng, top_k_init=1)
        a_p2 = safe_normalize(a_p2, n)
        obj = evaluate_internal(a_p2)
        if obj < global_best:
            global_best = obj
            global_a = a_p2.copy()
            print("  ExactSubgrad: obj=%.6f" % obj)

        # Also try from a fresh anti-palindromic start
        if time.monotonic() < t_phase2_end - 30:
            a_alt = make_anti_palindromic_init(n, rng, S)
            a_p2b, _ = run_exact_subgrad_nesterov(a_alt, n, S,
                                                   min(60.0, t_phase2_end - time.monotonic()),
                                                   rng, top_k_init=1)
            a_p2b = safe_normalize(a_p2b, n)
            obj_b = evaluate_internal(a_p2b)
            if obj_b < global_best:
                global_best = obj_b
                global_a = a_p2b.copy()
                print("  ExactSubgrad (alt): obj=%.6f" % obj_b)

    print("=== After Phase 2: best=%.6f ===" % global_best)

    # === Phase 3: Mirror descent (15% of time) ===
    t_phase3_end = time.monotonic() + (deadline - time.monotonic()) * 0.15
    if global_a is not None and time.monotonic() < t_phase3_end:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        print("=== Phase 3: Mirror descent (%d s) ===" % int(t_phase3_end - time.monotonic()))

        a_m, s_m = run_mirror_descent(global_a, n, S,
                                       min(120.0, t_phase3_end - time.monotonic()),
                                       rng, eta_init=0.05, top_k=3)
        a_m = safe_normalize(a_m, n)
        obj = evaluate_internal(a_m)
        if obj < global_best:
            global_best = obj
            global_a = a_m.copy()
            print("  Mirror: obj=%.6f" % obj)

    print("=== After Phase 3: best=%.6f ===" % global_best)

    # === Phase 4: Pairwise swap + coordinate descent polishing (20% of time) ===
    t_phase4_end = deadline - 30.0
    if global_a is not None and time.monotonic() < t_phase4_end:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        remaining4 = t_phase4_end - time.monotonic()
        print("=== Phase 4: Polishing (%d s total) ===" % int(remaining4))

        # 4a: Coordinate descent
        seg4a = remaining4 * 0.5
        a_cd, _ = run_coordinate_descent(global_a, n, S, seg4a, rng)
        a_cd = safe_normalize(a_cd, n)
        obj = evaluate_internal(a_cd)
        if obj < global_best:
            global_best = obj
            global_a = a_cd.copy()
            print("  CoordDescend: obj=%.6f" % obj)

        # 4b: Pairwise swap
        remaining4b = t_phase4_end - time.monotonic()
        if remaining4b > 10:
            a_ps, _ = run_pairwise_swap_polish(global_a, n, S, remaining4b, rng)
            a_ps = safe_normalize(a_ps, n)
            obj = evaluate_internal(a_ps)
            if obj < global_best:
                global_best = obj
                global_a = a_ps.copy()
                print("  PairwiseSwap: obj=%.6f" % obj)

    print("=== After Phase 4: best=%.6f ===" % global_best)

    # === Phase 5: Try longer sequence if time permits ===
    remaining = deadline - 30.0 - time.monotonic()
    if remaining > 120 and global_a is not None:
        n_long = max(16384, primary_n * 2)
        if n_long <= 32768:
            S_long = np.sqrt(2.0 * n_long)
            print("=== Phase 5: Trying n=%d (%d s) ===" % (n_long, int(remaining)))

            # Upsample or use fresh init
            if parent is not None and len(parent) >= 64:
                tt_old = np.linspace(0, 1, len(parent))
                tt_new = np.linspace(0, 1, n_long)
                a_long = np.interp(tt_new, tt_old, parent)
                a_long = np.maximum(a_long, 0.01)
            else:
                a_long = make_anti_palindromic_init(n_long, rng, S_long)

            seg5 = min(100.0, remaining - 20)
            a_l, _ = run_fista_lse(a_long, n_long, S_long, seg5, rng,
                                    beta_init=5.0, beta_max=100.0)
            a_l = safe_normalize(a_l, n_long)
            obj_l = evaluate_internal(a_l)
            if obj_l < global_best:
                global_best = obj_l
                global_a = a_l.copy()
                print("  Long n=%d: obj=%.6f" % (n_long, obj_l))

    # === Phase 6: Final FISTA polish with high beta ===
    remaining = deadline - 25.0 - time.monotonic()
    if remaining > 10 and global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        print("=== Phase 6: Final FISTA polish (%d s) ===" % int(remaining))

        a_fin, _ = run_fista_lse(global_a, n, S, remaining, rng,
                                  beta_init=50.0, beta_max=400.0)
        a_fin = safe_normalize(a_fin, n)
        obj = evaluate_internal(a_fin)
        if obj < global_best:
            global_best = obj
            global_a = a_fin.copy()
            print("  Final FISTA: obj=%.6f" % obj)

    # === Verification ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    try:
        ver = evaluate_sequence(final_list)
    except Exception as e:
        print("  Error in evaluate_sequence: %s" % str(e))
        ver = np.inf

    print("=== Final: internal=%.6f, verified=%.6f, len=%d ===" %
          (global_best, ver if np.isfinite(ver) else float('inf'), len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: using fallback")
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback verified: %.6f" % ver)

    return final_list
