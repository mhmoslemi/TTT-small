"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n < 2:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def topk_weighted_gradient(a, c, k_count, power=1.0):
    """Weighted gradient over top-k lags."""
    n = len(a)
    k_count = min(k_count, len(c))
    idx_sorted = np.argsort(c)[::-1][:k_count]
    vals = c[idx_sorted]
    w = np.power(np.maximum(vals - c.min(), 1e-30), power)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx_sorted] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_score(a):
    """Compute the actual objective value (without calling evaluate_sequence)."""
    n = len(a)
    c = fft_convolve_self(a)
    max_c = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a * sum_a)


def run_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_rate=0.005,
              beta_max=500.0, step_scale=1.0, mode="lse"):
    """FISTA with LSE smoothing. Returns (best_a, best_score)."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stall_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_score = 2.0 * n * cur_max / (S * S)

        if cur_score < best_score:
            best_score = cur_score
            best_a = y.copy()
            last_improve = it
            stall_count = 0
        else:
            stall_count += 1

        # Choose gradient
        if mode == "hybrid" and it > 200 and rng.random() < min(0.2, 0.0005 * it):
            grad = exact_subgradient(y, c)
            step = step_scale * S / (np.linalg.norm(grad) + 1e-15)
            step = min(step, 1.0 / (2.0 * max(beta, 1.0) * 0.05))
        elif mode == "topk":
            k = min(64, 8 + it // 200)
            grad = topk_weighted_gradient(y, c, k, power=1.5)
            step = step_scale / (2.0 * max(beta, beta_init))
        else:
            grad = lse_gradient(y, c, beta)
            step = step_scale / (2.0 * max(beta, 1.0))

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step reduction if stuck
        if stall_count > 1500:
            step *= 0.5
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
            stall_count = 0

        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15) * gnorm, S)
        # Simplified: x_new = project(y - step * grad)
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    c_final = fft_convolve_self(a)
    final_score = 2.0 * n * c_final.max() / (S * S)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


def peak_shave_polish(a_init, n, S, max_time, rng):
    """Targeted peak reduction using weighted subgradients."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()

    y = a.copy()
    t = 1.0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_score = 2.0 * n * cur_max / (S * S)

        if cur_score < best_score:
            best_score = cur_score
            best_a = y.copy()

        # Find top lags and compute weighted subgradient
        k_count = min(16, 4 + it // 100)
        top_idx = np.argsort(c)[::-1][:k_count]
        top_vals = c[top_idx]

        # Weight by how much each lag exceeds the mean
        mean_c = c.mean()
        excess = np.maximum(top_vals - mean_c, 0.0)
        w = excess ** 2
        wsum = w.sum()
        if wsum < 1e-30:
            break
        w = w / wsum

        # Build weight vector over all lags
        w_full = np.zeros(len(c))
        w_full[top_idx] = w

        # Compute gradient
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size with diminishing schedule
        base_step = S / (n * 0.5)
        step = base_step / (1.0 + 0.001 * it)

        x_new = project_simplex(y - step * grad / gnorm, S)

        # Momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic noise injection
        if it % 500 == 0 and rng.random() < 0.1:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0

    return best_a, best_score


def freq_domain_iter(a_init, n, S, max_time, rng):
    """Frequency-domain iteration: flatten |FFT(a)|^2 subject to a >= 0."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()

    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        A = fft(a, fsize)
        power = np.abs(A) ** 2

        # Target: flatten the power spectrum (geometric mean)
        log_power = np.log(np.maximum(power, 1e-30))
        mean_log = log_power.mean()
        # Move power towards flat
        target_power = np.exp(mean_log) * np.ones_like(power)
        # Partial update
        power_new = 0.7 * power + 0.3 * target_power

        # Reconstruct with same phases
        phase = np.angle(A)
        A_new = np.sqrt(power_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fsize))[:n]

        # Project to non-negative with correct sum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new * (S / s)

        # Check improvement
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        # Add small random perturbation for diversity
        if rng.random() < 0.05:
            a = normalize_sum(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), n)

    return best_a, best_score


def generate_inits(n, rng, parent=None):

        # Symmetrized

        inits.append(("parent_sym", np.maximum(sym, 0.01)))

    # Window functions
    inits.append(("hann", np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01)))
    inits.append(("hamming", 0.54 + 0.46 * np.cos(2 * np.pi * tt)))
    inits.append(("gauss", np.exp(-8.0 * (tt - 0.5) ** 2)))
    inits.append(("sinc2", np.maximum(np.sinc(tt - 0.5) ** 2, 0.01)))
    inits.append(("sin2", np.sin(np.pi * tt) ** 2 + 0.05))
    inits.append(("sin4", np.sin(np.pi * tt) ** 4 + 0.05))
    inits.append(("cos_plateau", np.maximum(np.cos(np.pi * tt), 0.01)))

    # Multi-tone
    for freq in [1, 2, 3, 5]:
        inits.append(("mt%d" % freq, 1.0 + 0.3 * np.sin(2 * np.pi * freq * tt) + 0.1 * np.cos(2 * np.pi * (2 * freq) * tt + 0.7)))

    # Random
    inits.append(("rand", rng.random(n) + 0.05))
    inits.append(("rand_exp", np.exp(rng.normal(0, 0.8, n))))

    # Block structures
    for nb in [4, 8, 16]:
        bl = n // nb
        h = rng.uniform(0.3, 1.5, nb)
        blk = np.repeat(h, bl)
        if len(blk) < n:
            blk = np.concatenate([blk, np.full(n - len(blk), h[-1])])
        inits.append(("blk%d" % nb, blk[:n]))

    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits.append(("alt", alt))

    # Triangle
    inits.append(("tri", 1.0 - 2.0 * np.abs(tt - 0.5) + 0.05))

    return inits

    print("\n=== Phase 1: Multi-start FISTA (%d inits, %.0fs) ===" % (n_inits, phase1_time))

    # Time allocation: give parent more time
    time_weights = np.ones(n_inits)
    for i, (nm, _) in enumerate(inits):
        if "parent" in nm:
            time_weights[i] = 2.5
        elif nm in ("hann", "gauss", "sinc2"):
            time_weights[i] = 1.5
    time_weights /= time_weights.sum()
    per_init_time = phase1_time * time_weights

    p1_start = time.monotonic()
    for i, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - (budget_s - 20.0) * 0.35:
            break
        t_alloc = max(2.0, per_init_time[i])
        if time.monotonic() - p1_start > phase1_time:
            break

        a_opt, score = run_fista(arr, n, S, t_alloc, rng,
                                  beta_init=5.0, beta_rate=0.004, beta_max=400.0,
                                  mode="lse")
        update_best(a_opt, "fista_%s" % name)

        # Also try hybrid mode for some
        if i % 3 == 0 and time.monotonic() < p1_start + phase1_time:
            a_opt2, score2 = run_fista(arr, n, S, t_alloc * 0.5, rng,
                                        beta_init=20.0, beta_rate=0.003, beta_max=300.0,
                                        mode="hybrid")
            update_best(a_opt2, "hybrid_%s" % name)

    print("After Phase 1: best=%.6f, elapsed=%.0fs" % (global_best, time.monotonic() - p1_start))

    # === Phase 2: Peak-shaving polish (25% of time) ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        phase2_time = max(10.0, remaining * 0.45)
        print("\n=== Phase 2: Peak-shaving polish (%.0fs) ===" % phase2_time)

        a_pol, s_pol = peak_shave_polish(global_a, n, S, phase2_time, rng)
        update_best(a_pol, "peak_shave")

    # === Phase 3: High-beta FISTA polish (10% of time) ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        phase3_time = max(5.0, remaining * 0.5)
        print("\n=== Phase 3: High-beta FISTA polish (%.0fs) ===" % phase3_time)

        # Try multiple beta starting points
        for beta_start in [50, 100, 200]:
            if time.monotonic() > deadline - 10:
                break
            t_seg = max(3.0, phase3_time / 3.0)
            a_hi, s_hi = run_fista(global_a, n, S, t_seg, rng,
                                    beta_init=float(beta_start), beta_rate=0.001,
                                    beta_max=2000.0, mode="hybrid")
            update_best(a_hi, "high_beta_%d" % beta_start)

    # === Phase 4: Frequency domain iteration (remaining time) ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print("\n=== Phase 4: Frequency-domain iteration (%.0fs) ===" % (remaining - 15))
            a_fd, s_fd = freq_domain_iter(global_a, n, S, remaining - 20, rng)
            update_best(a_fd, "freq_domain")

            # Polish the freq domain result
            if time.monotonic() < deadline - 10:
                a_fin, s_fin = run_fista(a_fd, n, S, max(3.0, deadline - time.monotonic() - 15),
                                          rng, beta_init=50.0, beta_rate=0.002, beta_max=1000.0,
                                          mode="topk")
                update_best(a_fin, "freq_polish")

    # === Phase 5: Final micro-polish ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            print("\n=== Phase 5: Final micro-polish (%.0fs) ===" % remaining)
            a_final, s_final = run_fista(global_a, n, S, remaining - 5, rng,
                                          beta_init=200.0, beta_rate=0.0005,
                                          beta_max=5000.0, mode="hybrid",
                                          step_scale=0.5)
            update_best(a_final, "micro_polish")

    # Final verification
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        global_best = compute_score(global_a)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\nFinal: internal=%.6f, verified=%.6f, length=%d" % (global_best, ver if np.isfinite(ver) else float('inf'), len(final_list)))

    if not np.isfinite(ver) or ver > global_best + 0.1:
        print("Warning: verification mismatch, falling back to parent")
        if parent is not None:
            parent = np.clip(parent, 0, 1000)
            if parent.sum() > 0.01:
                return [float(x) for x in parent]
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        return [float(x) for x in global_a]

    return final_list
