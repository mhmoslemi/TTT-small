def peak_directed_polish(a_init, n, S, max_time, rng, top_k=5):
    """Peak-directed subgradient method with adaptive step size."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Adaptive top-k: start small, grow over time
        k = min(50, top_k + it // 200)
        top_indices = np.argsort(c)[::-1][:k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break

        # Weight peaks by their excess over the mean
        mean_c = c.mean()
        weights = np.clip(top_vals - mean_c, 0, None)
        if weights.sum() < 1e-30:
            weights = top_vals
        weights = weights / weights.sum()

        w_full = np.zeros(2 * n - 1)
        for i, k_idx in enumerate(top_indices):
            w_full[int(k_idx)] = weights[i]

        grad = compute_conv_gradient(a, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Adaptive step size
        step = S * 0.01 / (1.0 + 0.01 * it)
        a_new = a - step * grad / max(gnorm, 1e-15)
        a_new = project_simplex(a_new, S)

        # Line search: check if improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() > cur * 1.01:

            a_new = a - step * grad / max(gnorm, 1e-15)
            a_new = project_simplex(a_new, S)

        # Occasional randomization

            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)

    return best_local, best_score

        grad = compute_conv_gradient(a, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 5000 == 0 and rng.random() < 0.05:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    total_budget = 2350.0
    deadline = time.monotonic() + total_budget
    t_start = time.monotonic()

    global_best = np.inf
    global_a = None

    # --- Get initial sequence from height_sequence_1 ---
    n = 8192
    S = np.sqrt(2.0 * float(n))

    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            n = len(given)
            S = np.sqrt(2.0 * float(n))
            inits.append(("given", np.clip(given, 0.0, 1000.0)))
            print("Loaded height_sequence_1: n=%d" % n)
    except Exception as e:
        print("Could not load height_sequence_1: %s" % str(e))

    if not inits:
        n = 8192
        S = np.sqrt(2.0 * float(n))
        tt = np.linspace(0, 1, n, endpoint=False)
        inits.append(("sin", 1.0 + 0.3 * np.sin(2*np.pi*3*tt) +
                     0.2 * np.cos(2*np.pi*7*tt) + 0.1 * np.sin(2*np.pi*13*tt)))
        inits.append(("rand", rng.random(n) + 0.1))

    # Evaluate initial scores
    for name, arr in inits:
        a_norm = safe_normalize(arr, n)
        c = fft_convolve_self(a_norm)
        score = c.max()
        print("Init '%s': score=%.8f" % (name, score))
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()

    print("\n=== Starting optimization: n=%d, budget=%.0fs ===" % (n, total_budget))
    print("Target: 1.5027, Starting: %.8f" % global_best)

    # === Phase 1: Main L-BFGS continuation from best init ===
    remaining = deadline - time.monotonic()
    p1_time = min(remaining * 0.3, 600)
    print("\n=== Phase 1: L-BFGS Continuation (%.0f sec) ===" % p1_time)

    a_main = global_a.copy()
    a_new, score = run_lbfgs_continuation(
        a_main, n, S, p1_time, rng,
        beta_schedule=[5, 20, 50, 100, 200, 500, 1000, 2000, 4000, 8000],
        maxiter_per_beta=250
    )
    a_new = safe_normalize(a_new, n)
    c_new = fft_convolve_self(a_new)
    if c_new.max() < global_best:
        global_best = c_new.max()
        global_a = a_new.copy()
        print("  Phase 1 result: %.10f" % global_best)
    else:
        print("  Phase 1: no improvement (best=%.10f)" % global_best)

    # === Phase 2: Peak-directed polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_time = min(remaining * 0.25, 300)
        print("\n=== Phase 2: Peak-Directed Polish (%.0f sec) ===" % p2_time)

        a_pol, score_pol = peak_directed_polish(global_a, n, S, p2_time, rng, top_k=5)
        a_pol = safe_normalize(a_pol, n)
        c_pol = fft_convolve_self(a_pol)
        if c_pol.max() < global_best:
            global_best = c_pol.max()
            global_a = a_pol.copy()
            print("  Phase 2 result: %.10f" % global_best)
        else:
            print("  Phase 2: no improvement (best=%.10f)" % global_best)

    # === Phase 3: Nesterov top-k subgradient ===
    remaining = deadline - time.monotonic()
    if remaining > 40:
        p3_time = min(remaining * 0.2, 250)
        print("\n=== Phase 3: Nesterov Top-K (%.0f sec) ===" % p3_time)

        a_nes, score_nes = run_topk_subgradient(global_a, n, S, p3_time, rng,
                                                 top_k_init=5, top_k_max=256)
        a_nes = safe_normalize(a_nes, n)
        c_nes = fft_convolve_self(a_nes)
        if c_nes.max() < global_best:
            global_best = c_nes.max()
            global_a = a_nes.copy()
            print("  Phase 3 result: %.10f" % global_best)
        else:
            print("  Phase 3: no improvement (best=%.10f)" % global_best)

    # === Phase 4: Restart with perturbations + L-BFGS ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 4: Perturbation + L-BFGS Restarts ===" % ())

        n_restarts = max(3, int(remaining / 80))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break

            # Different perturbation strategies
            strategy = r % 3
            if strategy == 0:
                # Gaussian noise
                noise_scale = 0.02 / (1.0 + 0.2 * r)
                a_pert = global_a + rng.normal(0, noise_scale * S / np.sqrt(n), n)
            elif strategy == 1:
                # Block shift
                block = n // 4
                a_pert = global_a.copy()
                shift = rng.integers(1, block // 2)
                a_pert[shift:block] = global_a[0:block-shift]
                a_pert[block:shift+block] = global_a[block-shift:block]
            else:
                # Segment swap
                seg = n // 8
                i1, i2 = rng.integers(0, n - 2*seg, 2)
                a_pert = global_a.copy()
                a_pert[i1:i1+seg], a_pert[i2:i2+seg] = a_pert[i2:i2+seg].copy(), a_pert[i1:i1+seg].copy()

            a_pert = np.maximum(a_pert, 0.0)
            a_pert = safe_normalize(a_pert, n)

            restart_time = min(60, (deadline - time.monotonic() - 30) / max(1, n_restarts - r))
            a_new, score = run_lbfgs_continuation(
                a_pert, n, S, restart_time, rng,
                beta_schedule=[50, 200, 500, 1000, 2000, 4000],
                maxiter_per_beta=150
            )
            a_new = safe_normalize(a_new, n)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < global_best:
                global_best = c_new.max()
                global_a = a_new.copy()
                print("  Restart %d (strat=%d): %.10f" % (r, strategy, global_best))

    # === Phase 5: Final high-beta L-BFGS polish ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 5: Final High-Beta Polish (%.0f sec) ===" % remaining)
        p5_time = min(remaining - 10, 200)

        a_final, score = run_lbfgs_continuation(
            global_a, n, S, p5_time, rng,
            beta_schedule=[500, 1000, 2000, 4000, 8000, 15000],
            maxiter_per_beta=300
        )
        a_final = safe_normalize(a_final, n)
        c_final = fft_convolve_self(a_final)
        if c_final.max() < global_best:
            global_best = c_final.max()
            global_a = a_final.copy()
            print("  Final polish: %.10f" % global_best)

    # === Phase 6: Ultra-fine peak polishing ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        p6_time = min(remaining - 5, 60)
        print("\n=== Phase 6: Ultra-Fine Polish (%.0f sec) ===" % p6_time)

        a_ultra, score_ultra = peak_directed_polish(global_a, n, S, p6_time, rng, top_k=3)
        a_ultra = safe_normalize(a_ultra, n)
        c_ultra = fft_convolve_self(a_ultra)
        if c_ultra.max() < global_best:
            global_best = c_ultra.max()
            global_a = a_ultra.copy()
            print("  Ultra-fine: %.10f" % global_best)

    # --- Verification and return ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.8f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    elapsed = time.monotonic() - t_start
    print("\nReturning sequence of length %d" % len(global_a))
    print("Internal score: %.10f" % global_best)
    print("Time used: %.1fs / %.0fs" % (elapsed, total_budget))
    return [float(x) for x in global_a]
