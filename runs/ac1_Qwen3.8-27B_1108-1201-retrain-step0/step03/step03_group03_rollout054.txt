"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Strategy: FISTA with adaptive LSE smoothing + active-set subgradient + local pair-swap search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_conv(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if v.sum() <= 0:
        return np.full(n, S / n)
    # Sort and find threshold
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    rho_candidates = (cumsum - S) / np.arange(1, n + 1, dtype=np.float64)
    valid = rho_candidates < w
    if not np.any(valid):
        return v * (S / v.sum())
    rho = (cumsum[int(np.max(np.where(valid)[0]))] - S) / (np.max(np.where(valid)[0]) + 1)
    return np.maximum(v - rho, 0.0)

def lse_obj_grad(a, beta):
    """Compute LSE objective value and gradient."""
    n = len(a)
    c = fft_conv(a)
    cm = c.max()
    bc = beta * (c - cm)
    exp_bc = np.exp(bc)
    Z = exp_bc.sum()
    obj = cm + np.log(Z) / beta
    if Z < 1e-300:
        return obj, np.zeros(n)
    w = exp_bc / Z
    fsize = 1 << (2 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = (2.0 / beta) * np.real(ifft(W * np.conj(A)))[:n]
    return obj, grad

def subgrad_max(a, top_k=10, temperature=1.0):
    """Subgradient of max(conv) using top-k lags with softmax weighting."""
    n = len(a)
    c = fft_conv(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    tv = top_vals / max(temperature, 1e-10)
    w = np.exp(tv - tv.max())
    w = w / w.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_idx):
        w_full[int(idx)] = w[i]
    fsize = 1 << (2 * n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return c.max(), grad

def fista_optimize(a_init, S, time_limit, rng, beta_start=5.0, beta_end=300.0, beta_rate=1.003):
    """FISTA with adaptive LSE beta."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) * S / len(a)
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    L = 1.0  # Lipschitz estimate
    start = time.monotonic()
    best_a = a.copy()
    best_obj = np.inf
    stall = 0
    it = 0
    
    while time.monotonic() - start < time_limit:
        it += 1
        beta = min(beta_end, beta_start * (beta_rate ** it))
        
        # Compute gradient at y
        obj_y, grad_y = lse_obj_grad(y, beta)
        gn = np.linalg.norm(grad_y)
        if gn < 1e-15:
            break
        
        # Backtracking line search
        step = 1.0 / L
        x = None
        for _ in range(8):
            x_cand = project_simplex(y - step * grad_y, S)
            if step < 1e-10:
                x = x_cand
                break
            obj_x, _ = lse_obj_grad(x_cand, beta)
            # Descent condition for smoothed function
            if obj_x <= obj_y - 0.5 * step * gn * gn:
                x = x_cand
                break
            step *= 0.5
        if x is None:
            x = project_simplex(y - step * grad_y, S)
        
        # Update best
        c_x = fft_conv(x)
        max_x = c_x.max()
        if max_x < best_obj:
            best_obj = max_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        a = x
        t = t_new
        
        # Adaptive L
        if obj_x > obj_y + 1e-8:
            L *= 1.5
        elif it % 50 == 0:
            L = max(L * 0.97, 1.0)
        
        # Perturbation to escape local minima
        if stall > 100 and rng.random() < 0.2:
            noise = 0.02 * S / np.sqrt(len(a))
            y = project_simplex(y + rng.normal(0, noise, len(a)), S)
            t = 1.0
            stall = 0
    
    return best_a, best_obj

def active_set_optimize(a_init, S, time_limit, rng, top_k_start=5, top_k_end=64):
    """Active-set subgradient method with adaptive step."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) * S / len(a)
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_obj = np.inf
    step_base = 1.0 / np.sqrt(len(a))
    it = 0
    
    while time.monotonic() - start < time_limit:
        it += 1
        top_k = min(top_k_end, top_k_start + it // 100)
        temp = max(0.1, 5.0 * (1.0 - it / max(it, 1)))
        
        obj, grad = subgrad_max(y, top_k=top_k, temperature=temp)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        # Adaptive step
        step = step_base / (1.0 + 0.01 * it) * S
        x = project_simplex(y - step * grad, S)
        
        # Check improvement
        c_x = fft_conv(x)
        if c_x.max() < best_obj:
            best_obj = c_x.max()
            best_a = x.copy()
        
        # Nesterov
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        a = x
        t = t_new
        
        # Periodic restart
        if it % 2000 == 0 and rng.random() < 0.3:
            noise = 0.01 * S / np.sqrt(len(a))
            y = project_simplex(y + rng.normal(0, noise, len(a)), S)
            t = 1.0
    
    return best_a, best_obj

def local_pair_swap(a_init, S, time_limit, rng, n_trials_per_iter=100):
    """Local search via pair swaps."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) * S / len(a)
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = fft_conv(a).max()
    n = len(a)
    no_improve = 0
    
    while time.monotonic() - start < time_limit:
        batch_start = time.monotonic()
        improved = False
        
        for _ in range(n_trials_per_iter):
            if time.monotonic() - start >= time_limit:
                break
            
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            
            # Try transferring mass from i to j
            for frac in [0.01, 0.05, 0.1, 0.5]:
                d = frac * a[i]
                if d < 1e-15:
                    continue
                
                a_try = a.copy()
                a_try[i] -= d
                a_try[j] += d
                
                if a_try[i] < 0:
                    continue
                
                c_try = fft_conv(a_try)
                if c_try.max() < best_obj - 1e-12:
                    a = a_try
                    best_obj = c_try.max()
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Perturb and restart local search
                noise = 0.01 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)
                no_improve = 0
        
        # Throttle if too slow
        batch_time = time.monotonic() - batch_start
        if batch_time > 5.0:
            n_trials_per_iter = max(10, n_trials_per_iter // 2)
    
    return best_a, best_obj

def make_anti_palindromic(n, rng, S):
    """Anti-palindromic initialization."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a

Actually, for the incremental update of the convolution, let me think more carefully.

If we change `a[i]` to `a[i] - d` and `a[j]` to `a[j] + d`, then:
`c_new[l] = sum_k a_new[k] * a_new[l-k]`

The difference `c_new[l] - c[l]`:
Let's denote `a_new[k] = a[k] + delta[k]` where `delta[i] = -d`, `delta[j] = d`, and 0 elsewhere.

`c_new[l] = sum_k (a[k]+delta[k]) * (a[l-k]+delta[l-k])`
`= sum_k a[k]*a[l-k] + sum_k a[k]*delta[l-k] + sum_k delta[k]*a[l-k] + sum_k delta[k]*delta[l-k]`
`= c[l] + sum_k a[k]*delta[l-k] + sum_k delta[k]*a[l-k] + sum_k delta[k]*delta[l-k]`

Now `sum_k a[k]*delta[l-k]`:
- `delta[l-k]` is nonzero only when `l-k = i` (i.e., `k = l-i`) or `l-k = j` (i.e., `k = l-j`)
- So = `a[l-i]*(-d)` (if `0 <= l-i < n`) + `a[l-j]*d` (if `0 <= l-j < n`)

Similarly `sum_k delta[k]*a[l-k]`:
- = `(-d)*a[l-i]` (if `0 <= l-i < n`) + `d*a[l-j]` (if `0 <= l-j < n`)

And `sum_k delta[k]*delta[l-k]`:
- `delta[k]` is nonzero at k=i and k=j
- `delta[l-k]` is nonzero at l-k=i (k=l-i) or l-k=j (k=l-j)
- So terms: delta[i]*delta[l-i] + delta[j]*delta[l-j]
- = (-d)*delta[l-i] + d*delta[l-j]
- delta[l-i] = -d if l-i=j, d if l-i=i (i.e., l=2i), 0 otherwise
- delta[l-j] = -d if l-j=i, d if l-j=j (i.e., l=2j), 0 otherwise

So:
- If l = i+j: delta[l-i] = delta[j] = d, delta[l-j] = delta[i] = -d
  → (-d)*d + d*(-d) = -2d^2
- If l = 2i: delta[l-i] = delta[i] = -d, delta[l-j] = 0 (assuming j≠i)
  → (-d)*(-d) + 0 = d^2
- If l = 2j: delta[l-i] = 0, delta[l-j] = delta[j] = d
  → 0 + d*d = d^2
- Otherwise: 0

- quadratic = 0 otherwise

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Strategy: FISTA with adaptive LSE smoothing, then fast local pair-swap search
with O(1) incremental lag updates for final polish.
Key insight: with sum(a)=sqrt(2n), the objective equals max(conv(a,a)) directly.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv(a):
    """Self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if v.sum() <= 0:
        return np.full(n, S / n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    rho_candidates = (cumsum - S) / idx
    valid = rho_candidates < w
    if not np.any(valid):
        return v * (S / v.sum())
    rho_idx = int(np.max(np.where(valid)[0]))
    rho = (cumsum[rho_idx] - S) / (rho_idx + 1)
    return np.maximum(v - rho, 0.0)


def lse_obj_grad(a, beta):
    """Compute LSE-smoothed max-conv objective and its gradient."""
    n = len(a)
    c = fft_conv(a)
    cm = c.max()
    bc = beta * (c - cm)
    exp_bc = np.exp(bc)
    Z = exp_bc.sum()
    obj = cm + np.log(Z) / beta
    if Z < 1e-300:
        return obj, np.zeros(n)
    w = exp_bc / Z
    fsize = 1 << (2 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = (2.0 / beta) * np.real(ifft(W * np.conj(A)))[:n]
    return obj, grad


def fista_optimize(a_init, S, time_limit, rng, beta_start=5.0, beta_end=300.0, beta_rate=1.002):
    """FISTA with adaptive LSE beta and backtracking line search."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) * S / len(a)
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    L = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_obj = np.inf
    stall = 0
    it = 0

    while time.monotonic() - start < time_limit:
        it += 1
        beta = min(beta_end, beta_start * (beta_rate ** it))

        obj_y, grad_y = lse_obj_grad(y, beta)
        gn = np.linalg.norm(grad_y)
        if gn < 1e-15:
            break

        step = 1.0 / L
        x = None
        for _ in range(10):
            x_cand = project_simplex(y - step * grad_y, S)
            if step < 1e-12:
                x = x_cand
                break
            obj_x, _ = lse_obj_grad(x_cand, beta)
            if obj_x <= obj_y - 0.4 * step * gn * gn:
                x = x_cand
                break
            step *= 0.5
        if x is None:
            x = project_simplex(y - step * grad_y, S)

        c_x = fft_conv(x)
        max_x = c_x.max()
        if max_x < best_obj:
            best_obj = max_x
            best_a = x.copy()
            stall = 0
        else:
            stall += 1

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        a = x
        t = t_new

        # Adaptive Lipschitz
        if obj_x > obj_y + gn * gn * step * 0.1:
            L *= 1.3
        elif it % 100 == 0:
            L = max(L * 0.98, 1.0)

        # Escape local minima
        if stall > 80 and rng.random() < 0.25:
            noise = 0.015 * S / np.sqrt(len(a))
            y = project_simplex(y + rng.normal(0, noise, len(a)), S)
            t = 1.0
            stall = 0

    return best_a, best_obj


def active_set_optimize(a_init, S, time_limit, rng, top_k_start=5, top_k_end=64):
    """Active-set subgradient with adaptive step and Nesterov momentum."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) * S / len(a)
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_obj = np.inf
    n = len(a)
    it = 0

    while time.monotonic() - start < time_limit:
        it += 1
        top_k = min(top_k_end, top_k_start + it // 200)

        c = fft_conv(y)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        temp = max(0.5, 3.0 - 2.0 * it / max(it + 1, 1))
        w = np.exp(top_vals / max(temp, 0.1) - max(top_vals) / max(temp, 0.1))
        w = w / w.sum()

        w_full = np.zeros(2 * n - 1)
        for i, idx in enumerate(top_idx):
            w_full[int(idx)] = w[i]

        fsize = 1 << (2 * n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        step = 2.0 * S / (n * (1.0 + 0.005 * it) * max(gn / (S / n), 0.1))
        x = project_simplex(y - step * grad, S)

        c_x = fft_conv(x)
        if c_x.max() < best_obj:
            best_obj = c_x.max()
            best_a = x.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        a = x
        t = t_new

        if it % 3000 == 0 and rng.random() < 0.2:
            noise = 0.01 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t = 1.0

    return best_a, best_obj


def local_pair_swap(a_init, S, time_limit, rng, top_k=30):
    """
    Local search via pair swaps with fast O(top_k) incremental check.
    When transferring mass d from index i to index j:
      dc[l] = -2d*a[l-i] + 2d*a[l-j] + quadratic_correction(l)
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) * S / len(a)
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_a = a.copy()
    c = fft_conv(a)
    best_obj = c.max()
    n = len(a)
    no_improve = 0
    trial_count = 0

    while time.monotonic() - start < time_limit:
        # Refresh top lags periodically
        if trial_count % 50 == 0:
            top_lags = np.argsort(c)[::-1][:top_k]
            top_max = c[top_lags].max()

        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue

        # Try transferring from i to j at various fractions
        best_frac = 0.0
        best_trial_peak = top_max

        for frac in (0.05, 0.1, 0.2, 0.5):
            d = frac * a[i]
            if d < 1e-15:
                continue

            peak = 0.0
            for l in top_lags:
                l = int(l)
                li = l - i
                lj = l - j
                ai_li = a[li] if 0 <= li < n else 0.0
                ai_lj = a[lj] if 0 <= lj < n else 0.0
                delta = -2.0 * d * ai_li + 2.0 * d * ai_lj
                # Quadratic correction
                if l == 2 * i:
                    delta += d * d
                if l == 2 * j:
                    delta += d * d
                if l == i + j and i != j:
                    delta -= 2.0 * d * d
                new_val = c[l] + delta
                if new_val > peak:
                    peak = new_val

            if peak < best_trial_peak:
                best_trial_peak = peak
                best_frac = frac

        if best_frac > 0.0 and best_trial_peak < top_max - 1e-12:
            d = best_frac * a[i]
            a[i] -= d
            a[j] += d
            trial_count += 1
            no_improve = 0

            # Periodic full verification
            if trial_count % 20 == 0:
                c = fft_conv(a)
                cur_max = c.max()
                if cur_max < best_obj:
                    best_obj = cur_max
                    best_a = a.copy()
                top_lags = np.argsort(c)[::-1][:top_k]
                top_max = c[top_lags].max()
        else:
            no_improve += 1

        # Escape stagnation
        if no_improve > 300:
            c = fft_conv(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()
            # Try targeted swaps: transfer from high to low within peak-lag pairs
            c = fft_conv(a)
            top_lags = np.argsort(c)[::-1][:5]
            for l in top_lags:
                l = int(l)
                i_min = max(0, l - n + 1)
                i_max = min(l, n - 1)
                if i_min > i_max:
                    continue
                for ii in range(i_min, i_max + 1, max(1, (i_max - i_min) // 10)):
                    jj = l - ii
                    if jj < 0 or jj >= n or jj == ii:
                        continue
                    if a[ii] > a[jj]:
                        d = 0.05 * a[ii]
                        a[ii] -= d
                        a[jj] += d
            no_improve = 0
            trial_count = 0

    # Final check
    c = fft_conv(a)
    if c.max() < best_obj:
        best_obj = c.max()
        best_a = a.copy()

    return best_a, best_obj


def make_anti_palindromic(n, rng, S):
    """Anti-palindromic initialization: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.05
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-15:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_chirp_init(n, rng, S):
    """Chirp-like initialization with quadratic phase."""
    tt = np.arange(n, dtype=np.float64)
    rate = rng.uniform(0.5, 3.0)
    a = 1.0 + 0.3 * np.cos(np.pi * rate * tt * tt / n)
    a = np.maximum(a, 0.05)
    a = a * (S / a.sum())
    return a

        print("  obj=%.6f" % (2.0 * n * global_best / S**2))

    # === High-beta FISTA ===
    remain = deadline - time.monotonic()
    t2 = min(500.0, max(40.0, remain * 0.25))
    print("=== Phase 2: FISTA beta 50->500 (%.0f sec) ===" % t2)
    a2, obj2 = fista_optimize(global_a, S, t2, rng,
                               beta_start=50.0, beta_end=500.0, beta_rate=1.003)
    if obj2 < global_best:
        global_best = obj2
        global_a = a2.copy()
        print("  obj=%.6f" % (2.0 * n * global_best / S**2))

    # === Very high-beta FISTA ===
    remain = deadline - time.monotonic()
    t3 = min(300.0, max(20.0, remain * 0.15))
    print("=== Phase 3: FISTA beta 100->1000 (%.0f sec) ===" % t3)
    a3, obj3 = fista_optimize(global_a, S, t3, rng,
                               beta_start=100.0, beta_end=1000.0, beta_rate=1.005)
    if obj3 < global_best:
        global_best = obj3
        global_a = a3.copy()
        print("  obj=%.6f" % (2.0 * n * global_best / S**2))

    # === Active-set subgradient ===
    remain = deadline - time.monotonic()
    t4 = min(300.0, max(20.0, remain * 0.15))
    print("=== Phase 4: Active-set (%.0f sec) ===" % t4)
    a4, obj4 = active_set_optimize(global_a, S, t4, rng,
                                    top_k_start=5, top_k_end=50)
    if obj4 < global_best:
        global_best = obj4
        global_a = a4.copy()
        print("  obj=%.6f" % (2.0 * n * global_best / S**2))

    # === Local pair-swap search (most important for final improvement) ===
    remain = deadline - time.monotonic()
    t5 = min(600.0, max(30.0, remain * 0.80))
    print("=== Phase 5: Local pair-swap (%.0f sec) ===" % t5)
    a5, obj5 = local_pair_swap(global_a, S, t5, rng, top_k=40)
    if obj5 < global_best:
        global_best = obj5
        global_a = a5.copy()
        print("  obj=%.6f" % (2.0 * n * global_best / S**2))

    # === Final micro-polish with local search ===
    remain = deadline - time.monotonic()
    if remain > 10:
        a6, obj6 = local_pair_swap(global_a, S, remain * 0.7, rng, top_k=20)
        if obj6 < global_best:
            global_best = obj6
            global_a = a6.copy()
            print("  Final polish: obj=%.6f" % (2.0 * n * global_best / S**2))

    # === Finalize and verify ===
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL: evaluate=%.6f, n=%d ===" % (ver if np.isfinite(ver) else float('inf'), len(global_a)))

    if not np.isfinite(ver):
        print("  Fallback to uniform")
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback eval: %.6f" % (ver if np.isfinite(ver) else float('inf')))

    return final_list
