def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)

"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
- Multi-phase optimization: L2 warmup -> soft-max FISTA -> top-K peak polishing
- Backtracking line search for robust step selection
- Adaptive peak weighting with power parameter for aggressive max reduction
- Multi-length search (n=4096, 8192) to avoid getting stuck at one length
- Periodic elastic restarts (stretch/compress) to escape local minima
- Use parent sequence as primary starting point

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax_grad(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) wrt a. O(n log n) via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def l2_grad(a, c):
    """
    Gradient of sum_k c[k]^2 wrt a.
    grad[j] = 4 * sum_m a[m] * c[j+m]
    Computed via FFT cross-correlation.
    """
    n = len(a)
    fsize = 1 << (len(c) + n - 1).bit_length()
    A = fft(a, fsize)
    C = fft(c, fsize)
    corr = np.real(ifft(np.conj(A) * C))[:n]
    return 4.0 * corr


def toppk_grad(a, c, top_k, power=1.0):
    """
    Gradient using top-K peaks with power weighting.
    Higher power focuses more on the largest peaks.
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx].copy()
    if power != 1.0:
        top_vals = np.maximum(top_vals, 0) ** power
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    w = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = w[i]
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_obj(a):
    """Compute 2n*max_conv/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def normalize_sum(a, target_sum):
    """Rescale a to have the given sum."""
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * target_sum / len(a)
    return a * (target_sum / s)


def phase_l2_warmup(a, n, S, max_time, rng):

        c = fft_convolve_self(a)
        l2_val = np.sum(c * c)
        if l2_val < best_val:
            best_val = l2_val
            best_a = a.copy()

        grad = l2_grad(a, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        ls_step = step / (1.0 + 0.05 * it)
        improved = False
        for _ in range(10):
            candidate = project_simplex(a - ls_step * grad / max(gnorm, 1e-15), S)
            c_new = fft_convolve_self(candidate)
            l2_new = np.sum(c_new * c_new)
            if l2_new < l2_val * 0.999:
                a = candidate
                improved = True
                break
            ls_step *= 0.5

        obj4 = compute_obj(a4)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase optimization with L2 warmup, soft-max FISTA, and top-K polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # --- Gather initializations ---
    inits = []

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 64 and np.sum(np.clip(parent, 0, 1000)) > 0.01:
            inits.append(("parent", np.clip(parent, 0, 1000)))
    except Exception:
        pass

    # Determine primary n
    if inits:
        n_primary = len(inits[0][1])
    else:
        n_primary = 8192

    # Alternative n values to try
    n_candidates = [n_primary]
    if n_primary != 4096:
        n_candidates.append(4096)
    if n_primary != 16384:
        n_candidates.append(16384)

    # Generate diverse initializations for the primary n
    tt = np.linspace(0, 1, n_primary, endpoint=False)
    diverse_inits = []

    # Exponential ramps
    for alpha in [1.0, 2.0, 3.0, 5.0, 8.0]:
        diverse_inits.append(("exp_f_%d" % int(alpha), np.exp(alpha * tt)))
        diverse_inits.append(("exp_b_%d" % int(alpha), np.exp(alpha * (1.0 - tt))))

    # Power-law
    for p in [0.3, 0.5, 1.0, 2.0, 3.0]:
        diverse_inits.append(("pow_f_%s" % p, (tt + 0.001) ** p))
        diverse_inits.append(("pow_b_%s" % p, (1.0 - tt + 0.001) ** p))

    # Two-sided (bump at different positions)
    for center in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        bump = np.exp(-8.0 * (tt - center) ** 2)
        diverse_inits.append(("bump_%s" % center, bump + 0.05))

    # Mixed exponential (fast rise, slow decay - asymmetric)
    for rise in [3, 5, 8]:
        for decay in [1, 2, 3]:
            arr = np.exp(-rise * np.minimum(tt, 0.3) ** 2) * np.exp(-decay * np.maximum(tt - 0.3, 0))
            diverse_inits.append(("mix_%d_%d" % (rise, decay), arr + 0.01))

    # Random with trend
    for trial in range(4):
        rand = rng.random(n_primary)
        trend = np.linspace(0.5, 2.0, n_primary)
        diverse_inits.append(("rand_%d" % trial, rand * trend))

    # Sinusoidal modulations
    for freq in [2, 3, 5]:
        div = 1.0 + 0.3 * np.sin(2 * np.pi * freq * tt) + 0.5 * tt
        diverse_inits.append(("sin_%d" % freq, div))

    # Cap the number of diverse inits
    max_diverse = 16
    if len(diverse_inits) > max_diverse:
        diverse_inits = diverse_inits[:max_diverse]

    # Combine all initializations
    all_inits = inits + [(n, arr) for n, arr in diverse_inits if len(arr) == n_primary]

    # For alternative n values, create simple initializations
    for n_alt in n_candidates:
        if n_alt == n_primary:
            continue
        tt_alt = np.linspace(0, 1, n_alt, endpoint=False)
        all_inits.append(("alt_exp_%d" % n_alt, np.exp(3.0 * tt_alt)))
        all_inits.append(("alt_pow_%d" % n_alt, (tt_alt + 0.01) ** 2))

    n_total = len(all_inits)
    usable_time = max(30.0, deadline - time.monotonic())

    # Time allocation: 60% exploration, 40% polish on best
    explore_time = usable_time * 0.60
    polish_time = usable_time * 0.40

    # Time per initialization (give more time to parent)
    time_per = explore_time / n_total

    print("=== Phase 1: Exploration (%d inits, total %.0f sec) ===" % (n_total, explore_time))
    t_start = time.monotonic()

    results = []
    for i, (name, arr) in enumerate(all_inits):
        if time.monotonic() > deadline - 60:
            break
        n_i = len(arr)
        if n_i < 32:
            continue
        S_i = np.sqrt(2.0 * float(n_i))

        # Give parent and first few more time
        if i == 0:
            t_alloc = time_per * 3.0
        elif i < 4:
            t_alloc = time_per * 1.5
        else:
            t_alloc = time_per

        t_alloc = min(t_alloc, explore_time - (time.monotonic() - t_start))
        if t_alloc < 3:
            continue

        a_opt, obj = run_full_optimization(arr, n_i, S_i, t_alloc, rng)
        obj = compute_obj(a_opt)

        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
            print("  [%s] n=%d: obj=%.6f" % (name, n_i, obj))

        results.append((obj, a_opt, n_i))

        elapsed = time.monotonic() - t_start
        frac = elapsed / explore_time
        print("  Progress: %d/%d inits, time used %.0f%%, best=%.6f" %
              (i + 1, n_total, frac * 100, global_best))

    print("=== After exploration: best=%.6f ===" % global_best)

    # --- Phase 2: Deep polish on the best candidate ---
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = deadline - time.monotonic()

        if remaining > 10:
            print("=== Phase 2: Deep Polish (remaining %.0f sec) ===" % remaining)

            # Polish round 1: Combined soft-max + top-K
            seg1 = remaining * 0.25
            if seg1 > 3:
                a_p, obj_p = run_full_optimization(global_a, n_best, S_best, seg1, rng)
                obj_p = compute_obj(a_p)
                if obj_p < global_best:
                    global_best = obj_p
                    global_a = a_p.copy()
                    print("  Polish 1: %.6f" % global_best)

            # Polish round 2: Focused top-K with high power
            remaining2 = deadline - time.monotonic()
            seg2 = remaining2 * 0.25
            if seg2 > 3:
                a_p2, _ = phase_toppk_polish(global_a, n_best, S_best, seg2, rng,
                                              k_start=16, k_end=512, power=3.0)
                obj_p2 = compute_obj(a_p2)
                if obj_p2 < global_best:
                    global_best = obj_p2
                    global_a = a_p2.copy()
                    print("  Polish 2 (topK power=3): %.6f" % global_best)

            # Polish round 3: Final fine-tuning
            remaining3 = deadline - time.monotonic()
            seg3 = remaining3 * 0.30
            if seg3 > 2:
                a_p3, _ = phase_final_polish(global_a, n_best, S_best, seg3, rng)
                obj_p3 = compute_obj(a_p3)
                if obj_p3 < global_best:
                    global_best = obj_p3
                    global_a = a_p3.copy()
                    print("  Polish 3 (final): %.6f" % global_best)

            # One more round if time allows
            remaining4 = deadline - time.monotonic()
            if remaining4 > 5:
                a_p4, _ = phase_toppk_polish(global_a, n_best, S_best, remaining4 * 0.8, rng,
                                              k_start=32, k_end=1024, power=2.0)
                obj_p4 = compute_obj(a_p4)
                if obj_p4 < global_best:
                    global_best = obj_p4
                    global_a = a_p4.copy()
                    print("  Polish 4: %.6f" % global_best)

    # --- Verification ---
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        global_a = np.ones(4096, dtype=np.float64)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, internal=%.6f, verified=%.6f" %
          (len(final_list), global_best, ver if np.isfinite(ver) else -1))
    return final_list
