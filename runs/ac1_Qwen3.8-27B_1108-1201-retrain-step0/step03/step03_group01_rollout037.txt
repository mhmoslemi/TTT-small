"""
Minimize 2*n*max(convolve(a,a)) / sum(a)^2 using OAA spectral flattening + LSE optimization.

Approach:
1. Multi-length search (try several sequence lengths)
2. OAA (Overlapping Additive) spectral flattening with adaptive target
3. Log-sum-exp smooth optimization (FISTA with increasing sharpness)
4. Subgradient polishing with adaptive top-k
5. Diverse initializations including structured patterns
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve(a, N):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    return c


def compute_score(a, N):
    """Compute the objective value (before the 2n/sum^2 factor)."""
    c = fft_convolve(a, N)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    return c.max() * 2 * len(a) / (S * S)


def oaa_flatten(a, N, target_P, lam, n_iter, rng, alpha_schedule=None):
    """
    OAA-based spectral flattening.
    
    a: input sequence
    N: FFT size (power of 2, >= 2n-1)
    target_P: target power spectrum
    lam: OAA blending parameter
    n_iter: number of iterations
    rng: random number generator
    """
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_cmax = np.inf
    
    for it in range(n_iter):
        # Compute current FFT
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        c = np.real(ifft(A * A))[:2 * n - 1]
        cmax = c.max()
        
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
        
        # Target spectrum (possibly adaptive)
        P = target_P.copy()
        
        # OAA step
        phase = np.angle(A)
        mag = np.sqrt(np.maximum(P, 0.0))
        A_new = mag * np.exp(1j * phase)
        a_spec = np.real(ifft(A_new))[:n]
        
        # Blend
        a_time = project_simplex(a, S)
        a_new = (1 - lam) * a_time + lam * a_spec
        a = project_simplex(a_new, S)
        
        # Occasionally perturb
        if it % 500 == 250 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
    
    return best_a, best_cmax

def subgradient_polish(a, n, S, N, max_time, rng, top_k_init=20, top_k_max=256):

    a = np.clip(a, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)
        
        c = fft_convolve(y, N)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        w = np.zeros(2 * n - 1)
        w[top_idx] = top_vals / total
        
        # Gradient via cross-correlation
        w_full = np.zeros(N)
        w_full[:len(w)] = w
        W = fft(w_full)
        a_pad = np.zeros(N)
        a_pad[:n] = y
        A_pad = fft(a_pad)
        grad = 2.0 * np.real(ifft(W * np.conj(A_pad)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 2.0 / (n * gnorm * (1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - y)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        t_mom = t_new
        
        if it % 1000 == 0 and rng.random() < 0.15:

def make_flat_target(N, S, n, low_freq_boost=0.0):
    """
    Create target power spectrum.
    DC = S^2, others flat with optional low-frequency boost.
    """
    # For a flat spectrum: P[k] = E/(N-1) for k != 0, P[0] = S^2
    # where E = N * sum(a_i^2) is the total spectral energy
    # We want to minimize the peak of IFFT(P), which means we want P to be as flat as possible
    # but P[0] = S^2 is fixed.
    # The "ideal" flat P would have P[k] = C for all k, but P[0] = S^2 forces C = S^2,
    # which gives IFFT(P) = constant = S^2/N for all lags (circular).
    # For linear (zero-padded), this gives a triangle-ish shape.
    # A better target: P[k] = S^2 for k=0, and P[k] = (N*S^2/n - S^2)/(N-1) for k!=0
    # (this corresponds to the minimum-energy signal with sum S, i.e., constant signal)
    # But we want to go BELOW the constant signal's peak, so we need a different target.

    # and slightly below flat at high frequencies. This compensates for the
    # non-negativity projection which tends to boost low frequencies.

    k = np.arange(N)
    # Normalized frequency (0 to 1, with 0 = DC, 0.5 = Nyquist)
    fnorm = k.astype(float) / N
    fnorm[norm > 0.5] = 1.0 - fnorm[fnorm > 0.5]  # Fold: 0 to 0.5
    
    # Slight boost at low frequencies
    if low_freq_boost > 0:
        P = 1.0 + low_freq_boost * np.exp(-fnorm * 5.0)
    
    # Normalize so that P[0] = S^2
    P[0] = S**2
    # Scale the rest to maintain reasonable total energy
    # Target: total energy = N * S^2 / n (corresponds to constant signal energy)
    # So sum(P) = S^2 + (N-1) * avg_rest = N * S^2 / n
    # avg_rest = (N*S^2/n - S^2) / (N-1) = S^2 * (N/n - 1) / (N-1)
    avg_rest = S**2 * (N / n - 1.0) / (N - 1)
    P[1:] = P[1:] * (avg_rest / P[1:].mean())

def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""

   C_new = C - alpha * FFT(c - c_mean)
   where c = IFFT(C)[0:2n-1] (or the full N-length IFFT) and c_mean = mean(c).
