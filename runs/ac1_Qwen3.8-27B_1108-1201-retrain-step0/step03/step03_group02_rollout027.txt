    # The sum is over i. For a fixed j, which terms in the sum involve a[j]?
    # - The term with i=j: a[j] * a[k-j] (if k-j is valid)
    # - The term with i=k-j: a[k-j] * a[j] = a[j] * a[k-j] (if k-j is valid and k-j != j)

    # d/d(a1) = 2*a1 = 2*a[k-1] = 2*a[1] ✓
    # d/d(a2) = 2*a0 = 2*a[k-2] = 2*a[0] ✓

    result = np.zeros(n)
    if 0 <= k < 2*n - 1:

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)

Wait, I realize the gradient computation can be done more efficiently using FFT (as the previous code does). Let me use the FFT approach.

For a single lag k, the gradient is simply 2*a[k-j] for valid j, which is just a shifted (and possibly truncated) version of a. So it's O(n) to compute.

For the softmax version, the gradient is sum_k w[k] * grad_k, where w[k] are softmax weights. This is a weighted sum of shifted versions of a, which can be computed via cross-correlation (FFT-based).

Let me now write the full solution:

"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Uses a combination of:
1. Mirror descent (exponential gradient) with adaptive softmax temperature
2. Asymmetric initializations
3. Backtracking line search
4. Multi-phase optimization (explore -> polish)
5. Periodic noise injection to escape local minima
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def grad_softmax(a, c, beta):
    """
    Gradient of soft-max approximation of max(c) w.r.t. a.
    Uses softmax weights and cross-correlation via FFT.
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # grad = 2 * cross-correlation of w and a
    # cross_corr[m] = sum_k w[k] * a[k-m] (with appropriate indexing)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def grad_toppk(a, c, k_list):
    """Gradient using specified top-k active peaks."""
    n = len(a)
    total = c[k_list].sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = c[k_list] / total
    
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(k_list):
        w_full[int(k)] = weights[i]
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def mirror_descent_step(a, grad, eta, S):
    """
    Mirror descent (exponential gradient) step on the simplex.
    a_new[j] = a[j] * exp(-eta * grad[j]) / Z
    where Z is chosen so sum(a_new) = S.
    """
    # To avoid overflow/underflow, center the gradient
    grad_centered = grad - grad.mean()
    log_a_new = np.log(a + 1e-300) - eta * grad_centered
    # Normalize
    log_a_new -= log_a_new.max()  # for numerical stability
    a_new = np.exp(log_a_new)
    s = a_new.sum()
    if s < 1e-300:
        a_new = np.ones(len(a)) * S / len(a)
    else:
        a_new = a_new * (S / s)
    return a_new


def objective_val(a):
    """Compute 2n*max_conv/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def run_optimize(a_init, n, S, max_time, rng, mode='hybrid'):

    Main optimization loop with adaptive step size and mirror descent.
    """
    a = np.clip(a_init.astype(np.float64), 1e-10, 1000.0).copy()
    s = a.sum()
    a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0  # Nesterov momentum
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    best_obj = 2.0 * n * fft_convolve_self(a).max() / (S * S)
    
    eta = S / (2.0 * np.sqrt(n))  # initial step size
    eta_min = 1e-8 * S
    beta = 5.0  # initial softmax temperature
    
    patience = 0
    patience_limit = 5000
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta: increase over time
        beta = min(500.0, 5.0 + 0.5 * it ** 0.7)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_local = y.copy()
            patience = 0
        else:
            patience += 1
        
        # Compute gradient
        if cur_max < 1e-15:
            break
        grad = grad_softmax(y, c, beta)
        
        # Nesterov momentum update
        x_candidate = mirror_descent_step(y, grad, eta, S)
        
        # Backtracking line search
        alpha = 1.0
        accepted = False
        for _ in range(20):
            x_try = (1 - alpha) * y + alpha * x_candidate
            x_try = np.maximum(x_try, 0)
            s_try = x_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            x_try = x_try * (S / s_try)
            
            c_try = fft_convolve_self(x_try)
            obj_try = 2.0 * n * c_try.max() / (S * S)
            
            if obj_try <= cur_obj + 1e-10:
                accepted = True
                break
            alpha *= 0.5
        
        if not accepted:
            x_new = y.copy()
            eta *= 0.5
        else:
            x_new = x_try
            if obj_try < cur_obj * 0.999:
                eta = min(eta * 1.1, S / (0.5 * np.sqrt(n)))
            else:
                eta *= 0.9
        
        # Update Nesterov
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 1e-10)
        s_y = y.sum()
        y = y * (S / s_y)
        a = x_new
        t_mom = t_new

            y = np.maximum(y + noise, 1e-10)
            y = y * (S / y.sum())
            t_mom = 1.0
        
        # If stuck, restart from best with perturbation
        if patience > patience_limit:
            y = best_local + rng.normal(0, 0.05 * S / np.sqrt(n), n)
            y = np.maximum(y, 1e-10)
            y = y * (S / y.sum())
            t_mom = 1.0
            patience = 0
            beta = 5.0
            eta = S / (2.0 * np.sqrt(n))
    
    return best_local, best_obj


def safe_normalize(a, n):
    """Clip to [0,1000] and rescale."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * np.sqrt(2.0 * n) / n
    return a * (np.sqrt(2.0 * n) / s)


def polish_exact(a_init, n, S, max_time, rng, top_k=16):
    """
    Polish using exact top-k peak gradient with aggressive step size.
    """
    a = np.clip(a_init.astype(np.float64), 1e-10, 1000.0).copy()
    a = a * (S / a.sum())
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_obj = 2.0 * n * fft_convolve_self(a).max() / (S * S)
    best_local = a.copy()
    it = 0
    step = S / (4.0 * np.sqrt(n))
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_local = y.copy()
        
        # Top-k gradient
        k = min(256, top_k + it // 100)
        top_indices = np.argsort(c)[::-1][:k]
        grad = grad_toppk(y, c, top_indices)
        
        # Projected gradient step with backtracking
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        alpha = 1.0
        accepted = False
        for _ in range(15):
            x_try = y - alpha * step * grad / grad_norm
            x_try = project_simplex(x_try, S)
            c_try = fft_convolve_self(x_try)
            obj_try = 2.0 * n * c_try.max() / (S * S)
            if obj_try <= cur_obj + 1e-10:
                accepted = True
                break
            alpha *= 0.5
        
        if not accepted:
            x_new = y.copy()
            step *= 0.5
        else:
            x_new = x_try
            if obj_try < cur_obj * 0.999:
                step = min(step * 1.15, S / (0.5 * np.sqrt(n)))
            else:
                step *= 0.95
        
        # Nesterov
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 1e-10)
        y = y * (S / y.sum())
        a = x_new
        t_mom = t_mom  # keep constant for stability
        
        # Occasional noise

            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            y = np.maximum(y + noise, 1e-10)
            y = y * (S / y.sum())
            t_mom = 1.0
    
    return best_local, best_obj

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def grad_from_weights(a, w):
    """
    Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Computed via cross-correlation of w and a using FFT.
    grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.array(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def spectral_flatten(a, n, S, mix_ratio=0.3):
    """
    Spectral magnitude flattening: partially equalize |FFT(a)|.
    Returns a mixed signal that's a blend of the original and the flattened version.
    """
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Target magnitude: geometric mean of current and uniform
    target_mag = np.sqrt(mag * (S / np.sqrt(fsize)))
    
    # Mix: partially move toward target
    new_mag = (1 - mix_ratio) * mag + mix_ratio * target_mag
    new_A = new_mag * np.exp(1j * phase)
    a_new = np.real(ifft(new_A, fsize))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (S / s)
    else:
        a_new = a.copy()
    return a_new
