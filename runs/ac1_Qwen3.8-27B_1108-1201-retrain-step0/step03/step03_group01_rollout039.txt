def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, c, top_k):
    """Compute subgradient of max(c) using top-k active lags."""
    n = len(a)
    if top_k > len(c):
        top_k = len(c)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    # Gradient of c[k] w.r.t. a[i] = a[k-i] + a[i-k] (with a out of range = 0)
    # For weighted sum: grad[i] = sum_k weights[k] * (a[k-i] + a[i-k])
    # This is a weighted cross-correlation of a with the weight kernel
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_step(a, N, top_k, alpha):
    """One spectral shaping step. Returns (new_a, c, c_max)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    m = np.arange(N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:

    return a_new, c, c_max

import numpy as np

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, c, top_k):
    """Compute subgradient of max(c) using top-k active lags via FFT cross-correlation."""
    n = len(a)
    if top_k > len(c):
        top_k = len(c)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

    total = top_vals.sum()
    if total < 1e-30:
        return a, c, c_max

    weights = top_vals / total

            break

        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 1e-10:
            y = y * (S / y_sum)
        else:
            y = project_simplex(x_new, S)
        a = x_new
        t_mom = t_new

        if stall >= momentum_reset_period:
            y = project_simplex(y + rng.normal(0, 0.004 * S / np.sqrt(n), n), S)
            t_mom = 1.0
            stall = 0

    return best_local, best_score


def fista_lse(a_init, n, S, max_time, rng, beta_init=20.0, beta_max=400.0):
    """FISTA with log-sum-exp smooth approximation of max(c)."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.001 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        c_max = c.max()
        scaled = beta * (c - c_max)
        exp_bc = np.exp(scaled - scaled.max())
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / (beta * (1.0 + 0.0002 * it))
        step = max(step, 1e-9)
        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 1e-10:
            y = y * (S / y_sum)
        a = x_new
        t_mom = t_new

        if it % 2000 == 1000 and rng.random() < 0.08:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t_mom = 1.0

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses two-scale spectral shaping + Nesterov subgradient + FISTA LSE with adaptive restarts.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None
    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d" % (n, S, N))
    print("Budget: %.0f s, deadline in %.0f s" % (budget_s, deadline - time.monotonic()))

    # ---- Build initializations ----
    init_list = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            given_clipped = np.clip(given, 0, 1000).astype(np.float64)
            init_list.append(("given", given_clipped))
    except Exception:
        pass

    tt = np.linspace(0, 1, n, endpoint=False)
    init_list.append(("rand", rng.random(n) + 0.1))
    init_list.append(("chirp", np.clip(0.5 + 0.3 * np.sin(2 * np.pi * (4 * tt + 0.3 * tt**2)) + 0.1 * rng.standard_normal(n), 0.01, None)))
    init_list.append(("multi_sin", np.clip(1.0 + 0.2 * np.sin(2*np.pi*5*tt) + 0.15*np.sin(2*np.pi*11*tt + 0.5) + 0.1*np.sin(2*np.pi*23*tt+1.0), 0.01, None)))
    init_list.append(("bump", np.clip(np.exp(-0.5*((tt-0.5)*3.5)**2) + 0.2 + 0.05*rng.standard_normal(n), 0.01, None)))

    n_inits = len(init_list)
    print("=== Phase 1: Exploration (%d inits) ===" % n_inits)

    t_phase1 = time.monotonic()
    phase1_budget = (deadline - t_phase1) * 0.25
    per_init = phase1_budget / n_inits

    for name, arr in init_list:
        if time.monotonic() > deadline - 120:
            break
        a_norm = safe_normalize(arr, n)
        c0 = fft_convolve_self(a_norm)
        s0 = c0.max()
        print("  [%s] init max_conv: %.6f" % (name, s0))

        if s0 < global_best:
            global_best = s0
            global_a = a_norm.copy()

        t0 = time.monotonic()
        a_work = a_norm.copy()

        # Sub-phase 1a: Coarse spectral (top 1-3, large alpha)
        ss_coarse_time = min(per_init * 0.15, 30.0)
        if ss_coarse_time > 1:
            a_work, _ = spectral_phase(a_work, N, n, S, ss_coarse_time, rng,
                                        alpha_init=0.6, alpha_decay=0.999,
                                        top_k_start=1, top_k_end=3, target=0.0,
                                        noise_period=200, noise_scale=0.002)
            a_work = safe_normalize(a_work, n)
            c_w = fft_convolve_self(a_work)
            if c_w.max() < global_best:
                global_best = c_w.max()
                global_a = a_work.copy()
                print("  [%s] coarse SS: %.6f" % (name, global_best))

        # Sub-phase 1b: Fine spectral (top 5-20, medium alpha)
        ss_fine_time = min(per_init * 0.25, 50.0)
        rem_init = per_init - (time.monotonic() - t0)
        if ss_fine_time > 1 and rem_init > 3:
            a_work, _ = spectral_phase(a_work, N, n, S, ss_fine_time, rng,
                                        alpha_init=0.15, alpha_decay=0.9995,
                                        top_k_start=3, top_k_end=15, target=0.0,
                                        noise_period=300, noise_scale=0.001)
            a_work = safe_normalize(a_work, n)
            c_w = fft_convolve_self(a_work)
            if c_w.max() < global_best:
                global_best = c_w.max()
                global_a = a_work.copy()
                print("  [%s] fine SS: %.6f" % (name, global_best))

        # Sub-phase 1c: Nesterov polish
        rem = per_init - (time.monotonic() - t0)
        if rem > 5:
            a_ng, _ = nesterov_polish(a_work, n, S, rem, rng,
                                       top_k_init=3, top_k_max=60, target=0.0)
            a_ng = safe_normalize(a_ng, n)
            c_ng = fft_convolve_self(a_ng)
            if c_ng.max() < global_best:
                global_best = c_ng.max()
                global_a = a_ng.copy()
                print("  [%s] NG: %.6f" % (name, global_best))

    print("=== Phase 1 done. Best max_conv: %.6f (score~%.4f) ===" % (global_best, 2*n*global_best/S**2))

    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = fft_convolve_self(global_a).max()

    # ---- Phase 2: Focused alternating optimization ----
    print("=== Phase 2: Focused optimization ===")
    t_phase2 = time.monotonic()
    phase2_budget = (deadline - t_phase2) * 0.40

    a_curr = global_a.copy()
    stall_count = 0
    it_outer = 0
    best_phase2 = global_best

    while time.monotonic() < t_phase2 + phase2_budget:
        it_outer += 1
        rem_total = (t_phase2 + phase2_budget) - time.monotonic()
        if rem_total < 10:
            break

        # Spectral fine-tuning
        ss_time = min(rem_total * 0.35, 40.0)
        if ss_time > 2:
            alpha_base = max(0.01, 0.12 / (1.0 + 0.15 * it_outer))
            a_curr, _ = spectral_phase(a_curr, N, n, S, ss_time, rng,
                                        alpha_init=alpha_base, alpha_decay=0.999,
                                        top_k_start=2, top_k_end=12, target=0.0,
                                        noise_period=400, noise_scale=0.0008)
            a_curr = safe_normalize(a_curr, n)
            c_curr = fft_convolve_self(a_curr)
            sc = c_curr.max()
            if sc < global_best:
                global_best = sc
                global_a = a_curr.copy()
                best_phase2 = sc
                stall_count = 0
            else:
                stall_count += 1

        # Nesterov with growing top-k
        rem = (t_phase2 + phase2_budget) - time.monotonic()
        if rem > 5:
            tk_base = 3 + (it_outer % 15)
            a_ng, _ = nesterov_polish(a_curr, n, S, min(rem, 50.0), rng,
                                       top_k_init=tk_base, top_k_max=tk_base + 40, target=0.0)
            a_ng = safe_normalize(a_ng, n)
            c_ng = fft_convolve_self(a_ng)
            if c_ng.max() < global_best:
                global_best = c_ng.max()
                global_a = a_ng.copy()
                a_curr = a_ng.copy()
                best_phase2 = global_best
                stall_count = 0
            else:
                stall_count += 1

        if it_outer % 4 == 0:
            print("  [P2-%d] best: %.6f, stall: %d, rem: %.0fs" % (it_outer, global_best, stall_count, rem_total))

        if stall_count >= 8:
            print("  [P2-%d] Stalled, perturbation" % it_outer)
            a_curr = project_simplex(a_curr + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            stall_count = 0

    print("=== Phase 2 done. Best max_conv: %.6f ===" % global_best)

    # ---- Phase 3: FISTA LSE (smooth optimization) ----
    print("=== Phase 3: FISTA LSE ===")
    t_phase3 = time.monotonic()
    remaining = deadline - t_phase3
    if global_a is not None and remaining > 15:
        fista_time = min(remaining * 0.30, 250.0)
        a_f, _ = fista_lse(global_a, n, S, fista_time, rng, beta_init=30.0, beta_max=400.0)
        a_f = safe_normalize(a_f, n)
        c_f = fft_convolve_self(a_f)
        if c_f.max() < global_best:
            global_best = c_f.max()
            global_a = a_f.copy()
            print("  FISTA improved: %.6f" % global_best)
        else:
            print("  FISTA no improvement (best stays %.6f)" % global_best)

    # ---- Phase 4: Final polishing (exact subgradient, very small steps) ----
    print("=== Phase 4: Final polish ===")
    t_phase4 = time.monotonic()
    remaining = deadline - t_phase4
    if global_a is not None and remaining > 5:
        # Top-1 subgradient (near-exact max reduction)
        polish_time = min(remaining * 0.4, 150.0)
        a_p, _ = nesterov_polish(global_a, n, S, polish_time, rng,
                                  top_k_init=1, top_k_max=5, target=0.0,
                                  momentum_reset_period=200)
        a_p = safe_normalize(a_p, n)
        c_p = fft_convolve_self(a_p)
        if c_p.max() < global_best:
            global_best = c_p.max()
            global_a = a_p.copy()
            print("  Polish top-1: %.6f" % global_best)

        # Very fine spectral
        rem2 = deadline - time.monotonic()
        if rem2 > 5:
            a_s, _ = spectral_phase(global_a, N, n, S, min(rem2 * 0.4, 80.0), rng,
                                     alpha_init=0.01, alpha_decay=0.999,
                                     top_k_start=1, top_k_end=3, target=0.0,
                                     noise_period=1000, noise_scale=0.0005)
            a_s = safe_normalize(a_s, n)
            c_s = fft_convolve_self(a_s)
            if c_s.max() < global_best:
                global_best = c_s.max()
                global_a = a_s.copy()
                print("  Polish SS ultra-fine: %.6f" % global_best)

        # Final Nesterov burst
        rem3 = deadline - time.monotonic()
        if rem3 > 5:
            a_n2, _ = nesterov_polish(global_a, n, S, rem3, rng,
                                       top_k_init=2, top_k_max=15, target=0.0,
                                       momentum_reset_period=150)
            a_n2 = safe_normalize(a_n2, n)
            c_n2 = fft_convolve_self(a_n2)
            if c_n2.max() < global_best:
                global_best = c_n2.max()
                global_a = a_n2.copy()
                print("  Final NG: %.6f" % global_best)

    # ---- Verification ----
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification score: %.6f" % ver)
        else:
            print("Verification FAILED (inf), using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("DONE. Returning length=%d" % len(global_a))
    return [float(x) for x in global_a]
