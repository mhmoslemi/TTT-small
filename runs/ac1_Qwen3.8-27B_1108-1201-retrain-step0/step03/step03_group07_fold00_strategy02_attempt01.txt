def evaluate_sequence(sequence: list[float]) -> float:
    # Security checks
    if not isinstance(sequence, list):
        return np.inf
    if not sequence:
        return np.inf
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
    sequence = [float(x) for x in sequence]
    sequence = [max(0, x) for x in sequence]
    sequence = [min(x, 1000) for x in sequence]

    n = len(sequence)
    if n == 0:
        return np.inf

    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf

    S = np.sqrt(2 * n)  # Normalization target

    seq_norm_factor = 1.0 / (sum_a / S)
    seq_normalized = [x * seq_norm_factor for x in sequence]

    # Calculate full convolution explicitly via FFT
    convolved = fft(seq_normalized, 2*n-1)**2 / (n)  # Scale by factor
    conv_full = ifft(convolved)[:2*n-1].real
    max_conv_val = np.max(conv_full)

    # Objective is 2*n * max_conv_value / sum_a^2
    return float(2 * n * max_conv_val / (sum_a**2))
