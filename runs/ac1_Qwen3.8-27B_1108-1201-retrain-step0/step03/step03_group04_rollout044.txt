def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_for_lag(a, k):
    """Gradient of c_k = sum_j a_j * a_{j+k} with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 2.0 * a
    elif k >= 0 and k < n:
        # c_k = sum_{j=0}^{n-1-k} a_j * a_{j+k}
        # d/d a_j: a_{j+k} (for j < n-k) + a_{j-k} (for j >= k)
        grad[:n - k] += a[k:]
        grad[k:] += a[:n - k]
    elif k < 0:
        # By symmetry, same as lag |k|
        grad[:] = compute_grad_for_lag(a, -k)
    return grad


def compute_grad_topk(a, top_indices, n):
    """Weighted sum of gradients for the given lag indices."""
    grad = np.zeros(n)
    for k in top_indices:
        if k == 0:
            grad += 2.0 * a
        elif 0 < k < n:
            grad[:n - k] += a[k:]
            grad[k:] += a[:n - k]
        elif k >= n:
            # By symmetry c_k = c_{2n-2-k}, so equivalent lag is 2n-2-k
            k_sym = 2 * n - 2 - k
            if 0 < k_sym < n:
                grad[:n - k_sym] += a[k_sym:]
                grad[k_sym:] += a[:n - k_sym]
    return grad / len(top_indices)


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def score_of(a):
    """Compute the normalized peak convolution (equals the objective when sum=sqrt(2n))."""
    c = fft_convolve_self(a)
    return c.max()


def run_sharp_descent(a_init, n, S, max_time, rng, top_k=1, step_init=None,
                      use_line_search=True, diffusion_period=0):
    """
    Sharp peak-directed subgradient descent with optional line search.
    Focuses only on the top_k lag(s).
    """
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    stall = 0
    step = step_init if step_init else 0.1 * S / n

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()

        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            break

        # Find top_k lags
        top_k_actual = min(top_k, 2 * n - 1)
        top_indices = np.argpartition(c, -top_k_actual)[-top_k_actual:]
        # Filter to unique lags (handle symmetry)
        top_indices = sorted(set(int(k) for k in top_indices))
        top_indices = top_indices[:top_k]  # re-limit

        # Compute gradient
        grad = compute_grad_topk(a, top_indices, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        grad = grad / gnorm  # normalize

        # Try step with line search
        if use_line_search:
            alpha = step
            best_alpha = 0.0
            best_val = cur
            for _ in range(5):
                a_trial = project_simplex(a - alpha * grad * gnorm, S)
                c_trial = fft_convolve_self(a_trial)
                val = c_trial.max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
                    best_a_trial = a_trial.copy()
                    # Try slightly larger
                    alpha *= 1.3
                else:
                    alpha *= 0.5
            if best_alpha > 0:
                a = best_a_trial
            else:
                # No improvement; try a tiny random step
                if rng.random() < 0.3:
                    noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                    a = project_simplex(a + noise, S)
                else:
                    break
        else:
            a_new = project_simplex(a - step * grad * gnorm, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur:
                a = a_new
                step = min(step * 1.05, 1.0 * S / n)
            else:
                step *= 0.7
                if step < 1e-10 * S / n:
                    break
                a = project_simplex(a - step * grad * gnorm, S)

        # Periodic diffusion (smooth)
        if diffusion_period > 0 and it % diffusion_period == 0 and rng.random() < 0.2:
            # Gaussian-like smoothing
            kernel_size = min(16, n // 4)
            kernel = np.exp(-0.5 * (np.arange(-kernel_size // 2, kernel_size // 2 + 1)) ** 2 / (kernel_size // 4 + 1))
            kernel /= kernel.sum()
            a_smooth = np.convolve(a, kernel, mode='same')
            a = safe_normalize(a_smooth, n)
            stall = min(stall, 500)

    return best_a, best_score


def run_multi_phase(a_init, n, S, max_time, rng):
    """Run multiple phases of optimization with increasing refinement."""
    a = safe_normalize(a_init, n)
    best_score = np.inf
    best_a = a.copy()

    # Phase 1: Aggressive top-1, no line search (fast exploration)
    t1 = 0.3 * max_time
    t0 = time.monotonic()
    a1, s1 = run_sharp_descent(a, n, S, t1, rng, top_k=1, use_line_search=False,
                                step_init=0.5 * S / n)
    a1 = safe_normalize(a1, n)
    if s1 < best_score:
        best_score = s1
        best_a = a1.copy()
    elapsed = time.monotonic() - t0
    print("    Phase1 (top1 fast): score=%.6f, time=%.1fs" % (s1, elapsed))

    # Phase 2: Top-3 with line search
    remaining = max_time - (time.monotonic() - t0)
    t2 = 0.25 * remaining
    if t2 > 5:
        t0 = time.monotonic()
        a2, s2 = run_sharp_descent(best_a, n, S, t2, rng, top_k=3, use_line_search=True,
                                    step_init=0.2 * S / n)
        a2 = safe_normalize(a2, n)
        if s2 < best_score:
            best_score = s2
            best_a = a2.copy()
        print("    Phase2 (top3 LS): score=%.6f, time=%.1fs" % (s2, time.monotonic() - t0))

    # Phase 3: Top-1 with line search, diffusion
    remaining = max_time - (time.monotonic() - t0 - t2)
    t3 = 0.3 * remaining
    if t3 > 5:
        t0 = time.monotonic()
        a3, s3 = run_sharp_descent(best_a, n, S, t3, rng, top_k=1, use_line_search=True,
                                    step_init=0.05 * S / n, diffusion_period=500)
        a3 = safe_normalize(a3, n)
        if s3 < best_score:
            best_score = s3
            best_a = a3.copy()
        print("    Phase3 (top1 LS+diff): score=%.6f, time=%.1fs" % (s3, time.monotonic() - t0))

    # Phase 4: Ultra-fine top-1
    remaining = max_time - (time.monotonic() - t0 - t2 - t3)
    t4 = remaining * 0.5
    if t4 > 3:
        t0 = time.monotonic()
        a4, s4 = run_sharp_descent(best_a, n, S, t4, rng, top_k=1, use_line_search=True,
                                    step_init=0.01 * S / n)
        a4 = safe_normalize(a4, n)
        if s4 < best_score:
            best_score = s4
            best_a = a4.copy()
        print("    Phase4 (ultra fine): score=%.6f, time=%.1fs" % (s4, time.monotonic() - t0))

    return best_a, best_score


def generate_initializations(n, parent_a, rng):
    """Generate diverse initializations for optimization."""
    inits = []

    # 1. Parent (if available)
    if parent_a is not None:
        inits.append(("parent", safe_normalize(parent_a[:n], n) if len(parent_a) >= n
                      else np.resize(safe_normalize(parent_a, len(parent_a)), n)))

    # 2. Uniform (baseline = 2.0)
    inits.append(("uniform", np.ones(n) * np.sqrt(2.0 * n) / n))

    # 3. Exponential random
    exp_a = rng.exponential(1.0, n)
    inits.append(("exponential", safe_normalize(exp_a, n)))

    # 4. Bimodal (U-shaped)
    t = np.linspace(0, 1, n)
    bimodal = 0.5 + 2.0 * (t - 0.5) ** 2  # U-shape
    inits.append(("bimodal", safe_normalize(bimodal, n)))

    # 5. Cosine window (low at edges, high in middle)
    cosine_w = np.cos(np.pi * t) ** 2 + 0.1
    inits.append(("cosine", safe_normalize(cosine_w, n)))

    # 6. Two blocks
    two_block = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
    # Add some to the zero part to keep it positive
    two_block[n // 2:] = 0.2
    inits.append(("twoblock", safe_normalize(two_block, n)))

    # 7. Three blocks (high-low-high)
    third = n // 3
    three_block = np.concatenate([np.ones(third), np.full(n - 2 * third, 0.3),
                                  np.ones(n - 2 * third if n - 2 * third > 0 else 1)])
    # Fix length
    if len(three_block) != n:
        three_block = np.resize(three_block, n)
    inits.append(("threeblock", safe_normalize(three_block, n)))

    # 8. Parent with multiplicative noise
    if parent_a is not None:
        p = safe_normalize(parent_a[:n], n) if len(parent_a) >= n else np.resize(safe_normalize(parent_a, len(parent_a)), n)
        noisy = p * (1.0 + 0.3 * rng.standard_normal(n))
        inits.append(("parent_noisy", safe_normalize(noisy, n)))

    return inits


def propose_candidate(seed=42, budget_s=2385):
    """
    Minimize 2*n*max_conv/sum^2 over non-negative sequences.
    Multi-phase sharp subgradient descent with diverse initializations
    and multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s - 30)

    global_best = np.inf
    global_a = None
    global_n = 0

    # Get parent sequence
    parent_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            parent_a = np.clip(given, 0, 1000)
            print("Parent sequence: n=%d, sum=%.4f" % (len(given), given.sum()))
    except Exception:
        pass

    # Determine which n values to try
    n_candidates = []
    if parent_a is not None and 64 <= len(parent_a) <= 32768:
        n_candidates.append(len(parent_a))
    # Always try 8192 and 4096
    for nv in [8192, 4096, 16384]:
        if nv not in n_candidates:
            n_candidates.append(nv)

    # Sort by likely priority (parent first, then 8192, then others)
    n_candidates.sort(key=lambda x: (x != (len(parent_a) if parent_a is not None else -1), x))

    print("Trying n values: %s" % n_candidates)

    for n in n_candidates:
        if time.monotonic() > deadline - 60:
            break

        S = np.sqrt(2.0 * float(n))
        print("\n=== n = %d ===" % n)

        # Generate initializations
        inits = generate_initializations(n, parent_a, rng)
        print("  %d initializations" % len(inits))

        # Time budget per initialization
        total_time = min(deadline - time.monotonic(), 2000.0)
        time_per_init = total_time / len(inits)

        for init_name, init_a in inits:
            if time.monotonic() > deadline - 30:
                break

            t0 = time.monotonic()
            print("  Init '%s': " % init_name, end="", flush=True)

            a_opt, score = run_multi_phase(init_a, n, S, time_per_init, rng)
            a_opt = safe_normalize(a_opt, n)

            # Final score
            c_final = fft_convolve_self(a_opt)
            final_score = c_final.max()
            obj = 2.0 * n * final_score / (S * S)

            print("score=%.6f, obj=%.6f, time=%.1fs" % (final_score, obj, time.monotonic() - t0))

            if obj < global_best:
                global_best = obj
                global_a = a_opt.copy()
                global_n = n
                print("    NEW BEST: %.6f (n=%d)" % (global_best, global_n))

    # Final verification and output
    print("\n=== FINAL ===")
    print("Best objective: %.6f (n=%d)" % (global_best, global_n))

    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified: %.6f" % ver)
            return final_list
        else:
            print("Verification failed!")

    # Fallback
    print("Using fallback")
    fallback = [0.5] * 1024
    ver = evaluate_sequence(fallback)
    print("Fallback verified: %.6f" % ver)
    return fallback
