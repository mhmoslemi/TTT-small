"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using smoothed-max (log-sum-exp) gradient
descent with temperature annealing, multiplicative updates, backtracking line
search, adaptive learning rate, and multi-pass coordinate polish.

Key improvements over previous approach:
- Exact gradient of a smooth surrogate (log-sum-exp of convolution) instead of
  approximate top-k subgradient
- Temperature annealing from smooth (broad) to sharp (peak-focused)
- Multiplicative (mirror) updates preserving non-negativity
- Multiple divergent initializations + adaptive time allocation
- Final surgical coordinate polish targeting the specific worst lag
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def cross_corr_fft(w, a):
    """
    Compute g[m] = sum_k w[k] * a[k-m] for m=0..n-1.
    This is the cross-correlation needed for the gradient.
    """
    n = len(a)
    N = 1 << (len(w) + n - 1 - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    return np.real(ifft(W * np.conj(A)))[:n]


def true_obj(a, R=None):
    """Compute the true objective: 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    if R is None:
        R = fft_convolve_self(a)
    return 2.0 * n * R.max() / (S * S)


def smoothed_gradient_descent(a0, time_budget, T_start_frac=0.05, T_end_frac=0.001,
                              lr_init=1.0, lr_max=5.0, lr_min=0.01,
                              backtracks_max=12, stall_limit=3000,
                              verbose=True, seed=0):
    """
    Smoothed-max gradient descent with temperature annealing.
    Minimize 2n*max(R) subject to a>=0, sum(a)=1.
    Uses log-sum-exp(R/T) as smooth surrogate for max(R).
    """
    rng = np.random.default_rng(seed)
    a = a0.copy().astype(np.float64)
    n = len(a)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a, np.inf
    a = a / S  # normalize to sum = 1

    R = fft_convolve_self(a)
    true_obj_val = 2.0 * n * R.max()
    best_obj = true_obj_val
    best_a = a.copy()
    best_iter = 0

    T0 = T_start_frac * R.max() if R.max() > 1e-15 else 1.0
    T = T0
    lr = lr_init
    iter_no = 0
    start_time = time.monotonic()

    while time.monotonic() - start_time < time_budget:
        R = fft_convolve_self(a)
        R_max = R.max()
        true_obj_val = 2.0 * n * R_max

        if true_obj_val < best_obj:
            best_obj = true_obj_val
            best_a = a.copy()
            best_iter = iter_no

        # Anneal temperature logarithmically
        progress = min(1.0, iter_no / max(1, 200000))
        T = T0 * (T_end_frac / T_start_frac) ** progress
        T = max(T, 1e-20 + 1e-10 * R_max)

        # Compute softmax weights over R
        R_scaled = R / T
        R_shifted = R_scaled - R_scaled.max()
        with np.errstate(over='ignore'):
            exp_R = np.exp(R_shifted)
        w_sum = exp_R.sum()
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(len(R))
            w[R.argmax()] = 1.0
        else:
            w = exp_R / w_sum

        # Compute gradient: 2 * cross_corr(w, a)
        grad = 2.0 * cross_corr_fft(w, a)

        # Compute mirror-descent step size
        # Use weighted variance: sum a[m] * (grad[m]/sqrt(a[m]))^2
        a_safe = np.maximum(a, 1e-30)
        weighted_var = np.sum(a_safe * (grad / np.sqrt(a_safe)) ** 2)
        if weighted_var < 1e-30:
            break
        step_scale = lr / np.sqrt(weighted_var)

        # Backtracking line search
        accepted = False
        current_obj = true_obj_val
        for bt in range(backtracks_max):
            s = step_scale / (bt + 1)
            a_new = a * np.exp(-s * grad)
            a_new = np.maximum(a_new, 1e-300)
            s_new = a_new.sum()
            if s_new < 1e-15:
                continue
            a_new = a_new / s_new

            R_new = fft_convolve_self(a_new)
            obj_new = 2.0 * n * R_new.max()

            if obj_new < current_obj * (1.0 - 1e-12):
                a = a_new
                R = R_new
                true_obj_val = obj_new
                current_obj = obj_new
                accepted = True
                break
            else:
                current_obj = min(current_obj, obj_new)

        if not accepted:
            # Accept a tiny step to avoid complete stagnation
            s = step_scale / (backtracks_max + 1)
            a = a * np.exp(-s * grad)
            a = np.maximum(a, 1e-300)
            a = a / a.sum()
            R = fft_convolve_self(a)
            true_obj_val = 2.0 * n * R.max()

        # Adaptive lr
        if accepted:
            lr = min(lr * 1.08, lr_max)
        else:
            lr = max(lr * 0.7, lr_min)

        # Stagnation handling: structured perturbation
        if iter_no - best_iter > stall_limit:
            # Geometric perturbation with structure
            noise_scale = 0.03 + 0.02 * (iter_no / max(1, 50000))
            noise = rng.standard_normal(n) * (noise_scale / np.sqrt(n))
            # Add some low-frequency structure
            freq = rng.integers(1, 20)
            noise += 0.01 * np.cos(2 * np.pi * freq * np.arange(n) / n)
            a = a * np.exp(noise)
            a = np.maximum(a, 1e-300)
            a = a / a.sum()
            R = fft_convolve_self(a)
            true_obj_val = 2.0 * n * R.max()
            best_iter = iter_no
            lr = lr_init
            if verbose:
                elapsed = time.monotonic() - start_time
                print(f"      [perturb] iter={iter_no}, obj={true_obj_val:.6f}, "
                      f"best={best_obj:.6f}, t={elapsed:.1f}s")

        iter_no += 1
        if verbose and iter_no % 10000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"      iter={iter_no}, obj={true_obj_val:.6f}, best={best_obj:.6f}, "
                  f"T={T:.2e}, lr={lr:.3f}, t={elapsed:.1f}s")

    return best_a, best_obj

        i_lo = max(0, k_max - n + 1)

        top_idx = np.argsort(contributions)[::-1][:30]

            if time.monotonic() - start_time > time_budget:
                break
            if contributions[idx] < 1e-15:
                continue
            orig = a[idx]
            for factor in [0.7, 0.8, 0.85, 0.9]:
                a[idx] = orig * factor
                a = a / a.sum()
                R_new = fft_convolve_self(a)
                obj_new = 2.0 * n * R_new.max()

def diverse_initializations(n, rng):

        a = a / a.sum()
        candidates.append(a)

    for freq in [3, 5, 7, 11, 13]:
        for phase in [0, 0.3, 0.7]:

            a = np.maximum(a, 0.01)
            a = a / a.sum()

    for pos1, pos2, w1, w2 in [(0.2, 0.7, 0.04, 0.04),

        a = np.maximum(a, 0.01)

            # Decreasing temperature range: broader to sharper
            t_frac_start = [0.15, 0.05, 0.01][sub]
            t_frac_end = [0.02, 0.005, 0.001][sub]
            lr = [1.0, 0.7, 0.4][sub]
            stall = [3000, 2000, 1500][sub]

            print(f"  Sub-phase {sub+1}: T_frac=[{t_frac_start},{t_frac_end}], "
                  f"lr={lr}, {sub_t:.0f}s")
            a_best, obj_best = smoothed_gradient_descent(
                global_a, sub_t,
                T_start_frac=t_frac_start, T_end_frac=t_frac_end,
                lr_init=lr, stall_limit=stall,
                verbose=True, seed=seed + sub * 100
            )
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver):
                obj_best = ver
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print(f"    -> improved to {obj_best:.6f}")

    print(f"\nAfter Phase 2: best={global_best:.6f}")

    # --- Phase 3: Multi-pass polish (20% of remaining) ---
    remaining = deadline - time.monotonic()
    polish_time = 0.20 * remaining
    print(f"\nPhase 3: Polish ({polish_time:.0f}s)")

    if global_a is not None and polish_time > 10:
        # Pass 1: Peak-targeted coordinate polish
        t_p1 = polish_time * 0.4
        print(f"  P3a: Peak-targeted polish ({t_p1:.0f}s)")
        a_best, obj_best = peak_targeted_polish(global_a, t_p1, verbose=True)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver):
            obj_best = ver
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> improved to {obj_best:.6f}")

        # Pass 2: Fine smoothed descent
        t_p2 = polish_time * 0.35
        print(f"  P3b: Fine smoothed descent ({t_p2:.0f}s)")
        a_best, obj_best = smoothed_gradient_descent(
            global_a, t_p2,
            T_start_frac=0.005, T_end_frac=0.0005,
            lr_init=0.3, stall_limit=2000,
            verbose=False, seed=seed + 999
        )
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver):
            obj_best = ver
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> improved to {obj_best:.6f}")

        # Pass 3: Another coordinate polish
        t_p3 = polish_time * 0.25
        print(f"  P3c: Final coordinate polish ({t_p3:.0f}s)")
        a_best, obj_best = peak_targeted_polish(global_a, t_p3, verbose=False)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver):
            obj_best = ver
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> improved to {obj_best:.6f}")

    # --- Phase 4: Final squeeze with remaining time ---
    remaining = deadline - time.monotonic()
    print(f"\nPhase 4: Final squeeze ({remaining:.0f}s)")
    if global_a is not None and remaining > 10:
        a_best, obj_best = smoothed_gradient_descent(
            global_a, remaining,
            T_start_frac=0.002, T_end_frac=0.0002,
            lr_init=0.2, stall_limit=1000,
            verbose=True, seed=seed + 2000
        )
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver):
            obj_best = ver
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> improved to {obj_best:.6f}")

    # --- Final verification and cleanup ---
    if global_a is None:
        # Fallback: use the given or a uniform sequence
        if given_a is not None:
            global_a = given_a.copy()
        else:
            global_a = np.ones(n_work)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_list = global_a.tolist()
    final_ver = evaluate_sequence(final_list)
    print(f"\nFINAL: best tracked={global_best:.6f}, verified={final_ver:.6f}")

    if not np.isfinite(final_ver):
        print("Fallback to uniform")
        final_list = [1.0] * 256
        final_ver = evaluate_sequence(final_list)
        print(f"  fallback obj={final_ver:.6f}")

    return final_list
