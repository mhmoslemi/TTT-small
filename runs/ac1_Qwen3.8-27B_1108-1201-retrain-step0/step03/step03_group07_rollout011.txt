"""
Minimize 2n*max(conv(a,a))/sum(a)^2 using adaptive multiplicative weight updates
with soft-max smoothing, spectral annealing, and multi-scale optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def soft_max_grad(c, temperature):
    """Compute softmax weights over the top peaks."""
    c_max = c.max()
    # Use softmax for smooth approximation of max
    w = np.exp((c - c_max) / temperature)
    w /= w.sum()
    return w

def compute_grad_from_weights(w, a):
    """Compute gradient: grad[m] = 2 * sum_k w[k] * a[k-m]"""
    n = len(a)
    a_rev = a[::-1]
    out_len = len(w) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad

def multiplicative_update(a, grad, temperature, S, lr=0.1):
    """
    Multiplicative weight update in log domain.
    a_new[m] = a[m] * exp(-lr * (grad[m]/max_grad - 2*a[m]/S))
    Then renormalize to sum S.
    """
    max_grad = grad.max()
    if max_grad < 1e-15:
        return a.copy()
    # Normalized gradient (subtract mean for stability)
    grad_norm = grad / max_grad
    # The "force" pushing down on each coordinate
    force = grad_norm - 2.0 * a / S
    # Clip to avoid extreme updates
    force = np.clip(force, -5.0, 5.0)
    a_new = a * np.exp(-lr * force)
    a_new = np.maximum(a_new, 1e-300)
    a_new = a_new * (S / a_new.sum())
    return a_new

    a = np.maximum(a, 0.0)

        S = n
    else:

    best_a = a.copy()
    best_iter = 0
    
    v = np.zeros(n)  # momentum
    lr = lr_init
    temp = temperature_init
    iter_no = 0
    start = time.monotonic()
    stagnant = 0
    
    while time.monotonic() - start < time_budget:
        # Compute gradient using soft max
        w = soft_max_grad(c, temp)
        grad = compute_grad_from_weights(w, a)
        
        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:

        v = v / v_norm

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_obj(a, c):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)

def soft_top_weights(c, temperature, top_k=100):
    """Get soft weights for the top peaks."""
    # Take top-k for efficiency
    if len(c) <= top_k:
        top_idx = np.arange(len(c))
        top_vals = c
    else:
        # Use partition for speed
        part = np.argpartition(c, -top_k)[-top_k:]
        top_idx = part
        top_vals = c[part]
        # Re-sort
        order = np.argsort(top_vals)[::-1]
        top_idx = top_idx[order]
        top_vals = top_vals[order]
    
    c_max = top_vals[0]
    # Softmax over top-k
    w = np.exp((top_vals - c_max) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(top_idx))
        w[0] = 1.0
    else:
        w = w / w_sum
    
    # Place in full array
    w_full = np.zeros(len(c))
    w_full[top_idx] = w
    return w_full

def grad_from_weights(w_full, a):
    """Compute gradient vector: grad[m] = 2 * sum_k w_full[k] * a[k-m]"""
    n = len(a)
    a_rev = a[::-1]
    out_len = len(w_full) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w_full, fsize)
    R = fft(a_rev, fsize)
    conv = np.real(ifft(W * R))
    # The gradient is conv[n-1 : 2n-1]
    grad = 2.0 * conv[n-1:2*n-1]
    return grad

def optimize(a0, time_budget, lr=0.1, temp_init=1.0, temp_decay=0.9999,
             momentum=0.9, top_k=100, verbose=True, tag=""):
    """Optimize using adaptive multiplicative updates with soft-max gradient."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    best_obj = compute_obj(a, c)
    best_a = a.copy()
    best_iter = 0
    
    v = np.zeros(n)
    temp = temp_init
    it = 0
    start = time.monotonic()
    stagnant = 0
    
    # Ideal flat level
    flat_level = S * S / (2 * n - 1)
    
    while time.monotonic() - start < time_budget:
        # Get soft weights
        w = soft_top_weights(c, temp, top_k)
        grad = grad_from_weights(w, a)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm
        
        # Adaptive step size: scale by how far we are from flat
        peak_ratio = c.max() / flat_level  # > 1 means peaked
        adaptive_lr = lr * min(peak_ratio, 10.0)
        
        # Multiplicative update in log domain
        # a_new = a * exp(-lr * (grad_dir - 2*a/S))
        # The -2*a/S term comes from the gradient of -2*log(sum(a))
        force = v - 2.0 * a / S
        force = np.clip(force, -3.0, 3.0)
        
        a_new = a * np.exp(-adaptive_lr * force)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new * (S / a_new.sum())
        
        c_new = fft_convolve_self(a_new)
        obj_new = compute_obj(a_new, c_new)
        
        if obj_new < best_obj * (1 - 1e-9):
            best_obj = obj_new
            best_a = a_new.copy()
            best_iter = it
            stagnant = 0
            a = a_new
            c = c_new
        else:

            a_half = a * np.exp(-adaptive_lr * 0.5 * force)
            a_half = np.maximum(a_half, 1e-300)
            a_half = a_half * (S / a_half.sum())
            c_half = fft_convolve_self(a_half)

            else:

def spectral_flatten(a, alpha=0.5):
    """Make the magnitude spectrum flatter."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    mag_mean = mag.mean()
    # Blend: new_mag = alpha * mag + (1-alpha) * mag_mean
    new_mag = alpha * mag + (1 - alpha) * mag_mean
    # Reconstruct with same phase
    phase = A / np.where(mag > 0, mag, 1)
    new_A = new_mag * phase
    a_new = np.real(ifft(new_A, fsize)[:n])
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-10:
        return a.copy()
    a_new = a_new * (S / a_new.sum())
    return a_new

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a, c):
    """Compute 2n * max(conv) / sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def soft_top_weights(c, temperature, top_k=200):
    """Get soft-max weights over the top-k peaks for smooth gradient."""
    m = len(c)
    if m <= top_k:
        top_idx = np.arange(m)
        top_vals = c.copy()
    else:
        part = np.argpartition(c, -top_k)[-top_k:]
        top_idx = part.copy()
        top_vals = c[part].copy()
        order = np.argsort(top_vals)[::-1]
        top_idx = top_idx[order]
        top_vals = top_vals[order]
    c_max = top_vals[0]
    exponents = (top_vals - c_max) / max(temperature, 1e-10)
    exponents = np.clip(exponents, -50.0, 0.0)
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(top_idx))
        w[0] = 1.0
    else:
        w = w / w_sum
    w_full = np.zeros(m)
    w_full[top_idx] = w
    return w_full


def grad_from_weights(w_full, a):
    """Compute gradient: grad[m] = 2 * sum_k w_full[k] * a[k-m] via FFT."""
    n = len(a)
    a_rev = a[::-1]
    out_len = len(w_full) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w_full, fsize)
    R = fft(a_rev, fsize)
    conv = np.real(ifft(W * R))
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def spectral_flatten(a, alpha=0.5):
    """Iteratively flatten the magnitude spectrum (move toward flat-top)."""
    n = len(a)
    S = a.sum()
    fsize = 1 << (n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    mag_mean = mag[1:n].mean() if n > 1 else mag[0]
    phase = np.where(mag > 1e-15, A / np.maximum(mag, 1e-15), 0.0)
    new_mag = alpha * mag + (1.0 - alpha) * mag_mean
    new_mag[0] = mag[0]  # Keep DC component
    new_A = new_mag * phase
    a_new = np.real(ifft(new_A, fsize)[:n])
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() < 1e-10:
        return a.copy()
    a_new = a_new * (S / a_new.sum())
    return a_new


def optimize_seq(a0, time_budget, lr=0.1, temp_init=1.0, temp_decay=0.99995,
                 momentum=0.9, top_k=200, verbose=True, tag="",
                 use_spectral=False, spectral_prob=0.05, spectral_alpha=0.3,
                 perturb_period=800):
    """Main optimization: adaptive multiplicative updates with soft-max gradient."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
        S = n
    else:
        a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = compute_obj(a, c)
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)
    temp = temp_init
    it = 0
    start = time.monotonic()
    stagnant = 0
    flat_level = S * S / (2 * n - 1)

    while time.monotonic() - start < time_budget:
        # Occasional spectral flatten move
        if use_spectral and np.random.random() < spectral_prob:
            a_spec = spectral_flatten(a, spectral_alpha)
            c_spec = fft_convolve_self(a_spec)
            obj_spec = compute_obj(a_spec, c_spec)
            if obj_spec < best_obj:
                best_obj = obj_spec
                best_a = a_spec.copy()
                a = a_spec
                c = c_spec
                stagnant = 0
            it += 1
            if it % 5000 == 0:
                continue
            temp = max(0.001, temp * temp_decay)
            continue

        # Compute soft-max weighted gradient
        w = soft_top_weights(c, temp, top_k)
        grad = grad_from_weights(w, a)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            v = grad
            vnorm = 1.0
        v = v / vnorm

        # Adaptive learning rate
        peak_ratio = c.max() / max(flat_level, 1e-10)
        adaptive_lr = lr * min(peak_ratio, 20.0)
        adaptive_lr = max(adaptive_lr, 0.001)

        # Multiplicative update (mirror descent in log domain)
        force = v - 2.0 * a / S
        force = np.clip(force, -4.0, 4.0)

        a_new = a * np.exp(-adaptive_lr * force)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new * (S / a_new.sum())

        c_new = fft_convolve_self(a_new)
        obj_new = compute_obj(a_new, c_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_iter = it
            stagnant = 0
            a = a_new
            c = c_new
            # Slightly increase lr on success
            lr = min(lr * 1.005, 1.0)
        else:
            # Try half step
            a_half = a * np.exp(-adaptive_lr * 0.4 * force)
            a_half = np.maximum(a_half, 1e-300)
            a_half = a_half * (S / a_half.sum())
            c_half = fft_convolve_self(a_half)
            obj_half = compute_obj(a_half, c_half)

            if obj_half < best_obj:
                best_obj = obj_half
                best_a = a_half.copy()
                best_iter = it
                stagnant = 0
                a = a_half
                c = c_half
            else:
                stagnant += 1
                lr = max(lr * 0.98, 0.005)

        # Anneal temperature
        temp = max(0.001, temp * temp_decay)

        # Perturbation when stagnant
        if stagnant > perturb_period:
            scale = 0.04 * S / np.sqrt(n)
            a = a + np.random.randn(n) * scale
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.1
            lr = max(lr, 0.08)
            temp = temp_init
            stagnant = 0
            if verbose:
                print(f"    {tag} [perturb] it={it} obj={compute_obj(a,c):.6f} "
                      f"best={best_obj:.6f} t={time.monotonic()-start:.0f}s")

        it += 1
        if verbose and it % 2000 == 0:
            print(f"  {tag} it={it} obj={compute_obj(a,c):.6f} best={best_obj:.6f} "
                  f"lr={lr:.4f} temp={temp:.4f} topk={top_k} t={time.monotonic()-start:.0f}s")

    return best_a, best_obj


def coordinate_polish(a0, time_budget, tag="coord"):
    """Coordinate descent: try adjusting individual elements to reduce max peak."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = compute_obj(a, c)
    c_max = c.max()
    k_max = np.argmax(c)
    best_a = a.copy()
    start = time.monotonic()

    # Focus on elements that contribute to the peak
    # Elements a[m] where m is in [k_max-n+1, k_max] contribute to c[k_max]
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    indices = np.arange(lo, hi + 1)
    # Sort by contribution (a[m] * a[k_max - m])
    contribs = np.array([a[m] * a[k_max - m] for m in indices])
    order = np.argsort(contribs)[::-1]
    top_indices = indices[order[:min(200, len(indices))]]

    sweep = 0
    while time.monotonic() - start < time_budget:
        improved_this_sweep = False
        for idx in np.random.permutation(top_indices):
            if time.monotonic() - start > time_budget:
                break
            orig = a[idx]
            if orig < 1e-15:
                continue
            # Try reducing this element
            for factor in [0.5, 0.7, 0.85]:
                a[idx] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[idx] = orig
                    break
                a_norm = a * (S / s)
                c_new = fft_convolve_self(a_norm)
                if c_new.max() < c_max - 1e-12:
                    a = a_norm
                    c = c_new
                    c_max = c.max()
                    k_max = np.argmax(c)
                    best_obj = compute_obj(a, c)
                    best_a = a.copy()
                    improved_this_sweep = True
                    break
                a[idx] = orig
            if not improved_this_sweep and factor == 0.5:
                a[idx] = orig

        # Update top_indices periodically
        if sweep % 5 == 0:
            lo = max(0, k_max - n + 1)
            hi = min(n - 1, k_max)
            indices = np.arange(lo, hi + 1)
            contribs = np.array([a[m] * a[k_max - m] for m in indices if 0 <= k_max - m < n])
            if len(contribs) > 0:
                valid_indices = np.array([m for m in indices if 0 <= k_max - m < n])
                order = np.argsort(contribs)[::-1]
                top_indices = valid_indices[order[:min(200, len(valid_indices))]]

        sweep += 1
        if not improved_this_sweep and sweep > 2:
            break

    return best_a, best_obj

    rng = np.random.default_rng(123)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    global_best = np.inf
    global_a = None

            n_given = len(given_raw)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_raw * (S_given / given_raw.sum())
            ver = evaluate_sequence(given_a.tolist())
            print(f"Given sequence: n={n_given}, obj={ver:.6f}")
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print(f"Could not load given sequence: {e}")

    if global_a is None:
        # Fallback initialization
        n_val = 4096
        S_val = np.sqrt(2.0 * n_val)
        global_a = np.ones(n_val)
        global_a = global_a * S_val / global_a.sum()
        c = fft_convolve_self(global_a)
        global_best = compute_obj(global_a, c)
        print(f"Using uniform init: n={n_val}, obj={global_best:.6f}")

    # --- Phase 1: Exploration (10% of time) ---
    remaining = deadline - time.monotonic()
    explore_time = 0.10 * remaining
    print(f"\nPhase 1: Exploration ({explore_time:.0f}s)")

    n_val = len(global_a)
    S_val = global_a.sum()

    candidates = [("given", global_a.copy())]

    # Additional initializations for n_val
    i = np.arange(n_val)
    for freq in [3, 7, 13]:
        a_cos = 1.0 + 0.7 * np.cos(2.0 * np.pi * freq * i / n_val)
        a_cos = np.maximum(a_cos, 0.01)
        a_cos *= S_val / a_cos.sum()
        candidates.append((f"cos{freq}", a_cos))

    a_rand = np.maximum(rng.random(n_val), 0.01)
    a_rand *= S_val / a_rand.sum()
    candidates.append(("rand", a_rand))

    # Triangle (peaked in middle)
    a_tri = 1.0 - 2.0 * np.abs(i - n_val / 2) / n_val
    a_tri = np.maximum(a_tri, 0.01)
    a_tri *= S_val / a_tri.sum()
    candidates.append(("tri", a_tri))

    # Inverted triangle (peaked at edges)
    a_inv_tri = 2.0 * np.abs(i - n_val / 2) / n_val + 0.1
    a_inv_tri *= S_val / a_inv_tri.sum()
    candidates.append(("invtri", a_inv_tri))

    per_time = explore_time / len(candidates)
    for name, a_cand in candidates:
        if time.monotonic() > deadline - 60:
            break
        a_best, obj_best = optimize_seq(
            a_cand, per_time,
            lr=0.15, temp_init=1.0, temp_decay=0.9999,
            momentum=0.85, top_k=100, verbose=False, tag=name
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  {name}: NEW BEST {obj_best:.6f}")
        else:
            print(f"  {name}: obj={obj_best:.6f}")

    # --- Phase 2: Main optimization (60% of time) ---
    remaining = deadline - time.monotonic()
    main_time = 0.60 * remaining
    print(f"\nPhase 2: Main optimization ({main_time:.0f}s), current best={global_best:.6f}")

    if global_a is not None and main_time > 30:
        # Sub-phase 2a: Broad (high temp, many peaks)
        sub_t = main_time * 0.35
        a_best, obj_best = optimize_seq(
            global_a, sub_t,
            lr=0.1, temp_init=3.0, temp_decay=0.99997,
            momentum=0.87, top_k=500, verbose=True, tag="2a_broad",
            use_spectral=True, spectral_prob=0.03, spectral_alpha=0.4
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  2a: {obj_best:.6f}")

        # Sub-phase 2b: Medium focus
        sub_t = main_time * 0.35
        a_best, obj_best = optimize_seq(
            global_a, sub_t,
            lr=0.08, temp_init=0.8, temp_decay=0.99997,
            momentum=0.90, top_k=100, verbose=True, tag="2b_med",
            use_spectral=True, spectral_prob=0.02, spectral_alpha=0.3
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  2b: {obj_best:.6f}")

        # Sub-phase 2c: Tight focus (low temp, few peaks)
        sub_t = main_time * 0.30
        a_best, obj_best = optimize_seq(
            global_a, sub_t,
            lr=0.06, temp_init=0.2, temp_decay=0.99998,
            momentum=0.92, top_k=30, verbose=True, tag="2c_tight",
            use_spectral=False
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  2c: {obj_best:.6f}")

    # --- Phase 3: Polish (30% of time) ---
    polish_time = deadline - time.monotonic() - 15.0
    print(f"\nPhase 3: Polish ({polish_time:.0f}s)")

    if global_a is not None and polish_time > 10:
        # 3a: Very fine mirror descent
        t_3a = polish_time * 0.45
        a_best, obj_best = optimize_seq(
            global_a, t_3a,
            lr=0.03, temp_init=0.05, temp_decay=0.99999,
            momentum=0.94, top_k=10, verbose=True, tag="3a_fine",
            use_spectral=False, perturb_period=500
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  3a: {obj_best:.6f}")

        # 3b: Coordinate polish
        t_3b = polish_time * 0.25
        a_best, obj_best = coordinate_polish(global_a, t_3b)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  3b: {obj_best:.6f}")

        # 3c: Final ultra-fine
        t_3c = polish_time * 0.30
        a_best, obj_best = optimize_seq(
            global_a, t_3c,
            lr=0.02, temp_init=0.03, temp_decay=0.99999,
            momentum=0.95, top_k=5, verbose=True, tag="3c_ultra",
            use_spectral=False, perturb_period=300
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  3c: {obj_best:.6f}")

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: local={global_best:.6f}, verified={ver:.6f}")

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.6f}")

    return final_a
