def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_ab(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def obj_value(a, c=None):

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection."""
    n = len(v)
    if n == 0:
        return v
    v_clipped = np.clip(v, 0, None)
    s = v_clipped.sum()
    if s <= S:
        # Already sum <= S, just scale up (uniform addition)
        return v_clipped + (S - s) / n
    # Need to find tau such that sum(max(v - tau, 0)) = S

        mid = (lo + hi) / 2.0

        if s_mid > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0

def compute_subgrad(c, a, top_k, soft_temperature=0.0):
    """
    Compute subgradient of max_k c[k] (approximated by top-k weighted average).
    grad[m] = 2 * sum_k w[k] * a[k-m] for valid indices.
    """
    n = len(a)
    top_k = max(1, min(top_k, len(c)))
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    c_max = c[top_idx[0]]

    if soft_temperature > 0:
        w = np.exp(soft_temperature * (c[top_idx] - c_max))
        w /= w.sum()
    else:
        w = np.full(top_k, 1.0 / top_k)

    # Build full weight vector of length 2n-1
    w_full = np.zeros(len(c))
    w_full[top_idx] = w

    # grad[m] = 2 * sum_k w_full[k] * a[k-m]
    # = 2 * conv(w_full, a_rev)[n-1 + m]
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1 - 1).bit_length()
    W = fft(w_full, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))

    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad

def coordinate_polish(a0, S, time_budget, max_sweeps=200):
    """Coordinate descent polishing: try reducing each a[i] to lower the max."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        # Process indices in order of their contribution to peaks
        c_max = c.max()
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            # Try a few multiplicative factors
            for factor in [0.7, 0.85, 1.15, 1.3]:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_norm = a * (S / s)
                c_new = fft_convolve_self(a_norm)
                obj_new = c_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_norm.copy()
                    a = a_norm
                    c = c_new
                    improved = True
                    break
            a[i] = a[i]  # keep current
            # Renormalize
            s = a.sum()
            if s > 0:
                a = a * (S / s)

        if not improved:
            break

    return best_a, best_obj


def rl_polish(a0, S, time_budget, lr=0.001, top_k=1):
    """Reinforcement-learning style polish: focus on the single worst peak."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    v = np.zeros(n)

    while time.monotonic() - start < time_budget:
        # Find the single worst peak
        k_max = np.argmax(c)
        c_max_val = c[k_max]

        # Gradient of c[k_max] w.r.t. a
        # grad[m] = 2 * a[k_max - m] for valid m (0 <= k_max - m < n)
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for m in range(lo, hi + 1):
            grad[m] += 2.0 * a[k_max - m]

        # Normalized gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Momentum
        v = 0.9 * v + 0.1 * grad

        # Multiplicative update
        a = a * np.exp(-lr * v)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())

        c = fft_convolve_self(a)
        cur_obj = obj_value(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

    return best_a, best_obj
