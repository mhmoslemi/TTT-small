"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy:
- Anti-palindromic structured optimization with LSE-smoothed gradients
- Adaptive beta scheduling for progressive sharpening of the max approximation
- Soft anti-palindromic attraction to exploit structural insight
- Coordinate-descent polish targeting high-convolution lags
- Multi-initialization with diverse structures and periodic restarts
- Two-phase approach: coarse optimization then fine polishing
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient_fft(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_gradient_fft(a, c, top_k, temperature=1.0):
    """Gradient using top-k lags with softmax weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices].copy()
    tv_max = top_vals.max()
    temp = max(temperature, 1e-6)
    weights = np.exp(top_vals / temp - tv_max / temp)
    weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    w_full[top_indices] = weights
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def envelope_gradient_fft(a, c, threshold_factor=1.0):
    """Gradient targeting all lags above threshold_factor * mean."""
    n = len(a)
    mean_c = c.mean()
    threshold = threshold_factor * mean_c
    mask = c > threshold
    if not np.any(mask):
        mask = c > c.max() * 0.9
    w = np.zeros(2 * n - 1)
    w[mask] = 1.0 / mask.sum()
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def anti_palindromic_attract(a, S, strength=0.1):
    """Softly pull a towards anti-palindromic structure while preserving sum."""
    n = len(a)
    half = n // 2
    c_target = 2.0 * S / n
    a_pull = a.copy()
    for i in range(half):
        j = n - 1 - i
        avg = (a[i] + a[j]) / 2.0
        a_pull[i] = a[i] + strength * (avg - a[i])
        a_pull[j] = a[j] + strength * (avg - a[j])
    if n % 2 == 1:
        a_pull[half] = a[half] + strength * (c_target / 2 - a[half])
    s = a_pull.sum()
    if s > 0:
        a_pull = a_pull * (S / s)
    return np.maximum(a_pull, 0.0)


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic initialization with random first half."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_sinusoidal_init(n, rng, S, num_components=3):
    """Sinusoidal mixture initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_components):
        freq = rng.integers(2, 30)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.5
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_paired_init(n, rng, S):
    """Initialize with strong anti-palindromic pairing."""
    half = n // 2
    c = 2.0 * S / n
    # Random deviations from the uniform anti-palindromic
    d = rng.normal(0, c * 0.3, half)
    d = np.clip(d, -c * 0.45, c * 0.45)
    a = np.zeros(n)
    for i in range(half):
        a[i] = c / 2 + d[i]
        a[n - 1 - i] = c / 2 - d[i]
    if n % 2 == 1:
        a[half] = c / 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def run_lse_optimization(a_init, n, S, max_time, rng, beta_start=1.0, beta_end=200.0,
                         use_antipal=True, antipal_strength=0.05):
    """LSE-smoothed gradient descent with adaptive parameters."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    stall = 0
    last_best = np.inf
    L = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: logarithmic increase
        frac = min(1.0, it / (2 * max(1.0, max_time / 0.5)))
        beta = beta_start * (beta_end / beta_start) ** frac

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = y.copy()
            stall = 0
        elif it % 100 == 0:
            if cur_max < last_best - 1e-10:
                last_best = cur_max
                stall = 0
            else:
                stall += 1
                if stall > 40:
                    # Restart with perturbation
                    noise = 0.02 * S / np.sqrt(n) * rng.standard_normal(n)
                    y = project_simplex(y + noise, S)
                    t_mom = 1.0
                    stall = 0

        # Compute gradient
        grad = lse_gradient_fft(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size with backtracking
        step = 1.0 / max(L, gnorm * beta / max(1.0, n))
        f_y = cur_max

        # Try step with backtracking
        x_new = None
        for bt in range(4):
            candidate = y - step * grad
            # Soft anti-palindromic attraction
            if use_antipal and it % 5 == 0:
                candidate = anti_palindromic_attract(candidate, S, antipal_strength * max(0, 1 - it / (2 * max(1, int(max_time / 0.5)))))
            candidate = project_simplex(candidate, S)
            if bt == 0:
                x_new = candidate
            else:
                c_cand = fft_convolve_self(candidate)
                if c_cand.max() <= f_y + 0.5 * step * gnorm:
                    x_new = candidate
                    break
                step *= 0.5
        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        # Update L adaptively
        c_new = fft_convolve_self(x_new)
        if c_new.max() > f_y * 1.01:
            L *= 1.5
        elif it % 50 == 0:
            L = max(L * 0.97, beta / max(1.0, n / 100.0))

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-project to maintain sum
        ys = y.sum()
        if ys > 0:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new

    return best_a, best_score


def run_coordinate_polish(a_init, n, S, max_time, rng, top_k=5):
    """Coordinate descent polishing: target top lags one coordinate at a time."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = a.sum() ** 2 / n  # initial lower bound
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    base_scale = S / n

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # Find top lags
        top_lags = np.argsort(c)[::-1][:top_k]

        # For each top lag, find contributing pairs and try to reduce them
        improved = False
        for lag in top_lags:
            lag = int(lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue
            indices = np.arange(i_min, i_max + 1)
            partner = lag - indices
            products = a[indices] * a[partner]

            # Target the largest contributing pairs
            top_prod_idx = np.argsort(products)[::-1][:min(20, len(indices))]
            for ii in top_prod_idx:
                idx = int(indices[ii])
                pidx = int(partner[ii])
                if idx == pidx:
                    continue
                # Try reducing the larger element and increasing a small one
                if a[idx] >= a[pidx]:
                    reduce_idx = idx
                    increase_pool_condition = np.array([a < base_scale * 0.8])
                else:
                    reduce_idx = pidx
                    increase_pool_condition = np.array([a < base_scale * 0.8])

                if not np.any(increase_pool_condition):
                    continue

                # Try a few step sizes
                for frac in [0.01, 0.005, 0.002]:
                    delta = frac * a[reduce_idx]
                    if delta < 1e-15:
                        continue
                    # Find where to add
                    candidates = np.where(increase_pool_condition)[0]
                    # Choose the one that helps most (smallest contribution to top lags)
                    add_idx = candidates[rng.integers(len(candidates))]
                    if add_idx == reduce_idx:
                        continue

                    a_trial = a.copy()
                    a_trial[reduce_idx] -= delta
                    a_trial[add_idx] += delta
                    if a_trial[reduce_idx] < 0:
                        a_trial[reduce_idx] = 0.0
                        a_trial[add_idx] += a_trial[add_idx] - (a[add_idx] + delta) + a[add_idx]
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial * (S / a_trial.sum()) if a_trial.sum() > 0 else a_trial

                    c_trial = fft_convolve_self(a_trial)
                    if c_trial.max() < cur - 1e-12:
                        a = a_trial
                        cur = c_trial.max()
                        improved = True
                        break

        if not improved and rng.random() < 0.1:
            # Small random perturbation
            noise = 0.005 * base_scale * rng.standard_normal(n)
            a = project_simplex(a + noise, S)

    return best_a, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, top_k=10):
    """Mirror descent (exponential updates) with top-k subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        eta = 2.0 / (it + 10)
        grad = topk_gradient_fft(a, c, top_k, temperature=max(1.0, S / 10))
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        log_update = -eta * grad / gmax
        log_update = np.clip(log_update, -3.0, 3.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new * (S / s_new)

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp and rescale."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

            a_long = safe_normalize(a_long, n_long)

            seg_long = min(200.0, max(30.0, (deadline - time.monotonic()) * 0.25))
            a_long_opt, _ = run_lse_optimization(a_long, n_long, S_long, seg_long, rng,
                                                  beta_start=0.5, beta_end=200.0)
            a_long_opt = safe_normalize(a_long_opt, n_long)
            c_long = fft_convolve_self(a_long_opt)
            obj_long = 2.0 * n_long * c_long.max() / S_long ** 2
            if obj_long < global_best:
                global_best = obj_long
                global_a = a_long_opt.copy()
                print("  Long n=%d: obj=%.6f" % (n_long, global_best))

    # Phase 3: Intensive polishing on best
    if global_a is not None:
        n = len(global_a)
        S_best = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        print("Phase 3: Polishing (remaining %.0f sec), n=%d" % (remaining, n))

        # 3a: High-beta LSE
        seg = max(5.0, remaining * 0.15)
        a_p, _ = run_lse_optimization(global_a, n, S_best, seg, rng,
                                       beta_start=50.0, beta_end=800.0,
                                       use_antipal=True, antipal_strength=0.02)
        a_p = safe_normalize(a_p, n)
        c_p = fft_convolve_self(a_p)
        obj_p = 2.0 * n * c_p.max() / S_best ** 2
        if obj_p < global_best:
            global_best = obj_p
            global_a = a_p.copy()
            print("  HiBeta LSE: obj=%.6f" % global_best)

        # 3b: Coordinate polish
        remaining2 = deadline - time.monotonic()
        seg2 = max(5.0, remaining2 * 0.20)
        a_cp, _ = run_coordinate_polish(global_a, n, S_best, seg2, rng, top_k=3)
        a_cp = safe_normalize(a_cp, n)
        c_cp = fft_convolve_self(a_cp)
        obj_cp = 2.0 * n * c_cp.max() / S_best ** 2
        if obj_cp < global_best:
            global_best = obj_cp
            global_a = a_cp.copy()
            print("  CoordPolish: obj=%.6f" % global_best)

        # 3c: Mirror descent
        remaining3 = deadline - time.monotonic()
        seg3 = max(5.0, remaining3 * 0.15)
        a_md, _ = run_mirror_descent(global_a, n, S_best, seg3, rng, top_k=10)
        a_md = safe_normalize(a_md, n)
        c_md = fft_convolve_self(a_md)
        obj_md = 2.0 * n * c_md.max() / S_best ** 2
        if obj_md < global_best:
            global_best = obj_md
            global_a = a_md.copy()
            print("  Mirror: obj=%.6f" % global_best)

        # 3d: Final ultra-high-beta LSE
        remaining4 = deadline - time.monotonic()
        if remaining4 > 10:
            a_fin, _ = run_lse_optimization(global_a, n, S_best, remaining4 * 0.7, rng,
                                             beta_start=200.0, beta_end=1500.0,
                                             use_antipal=False, antipal_strength=0.0)
            a_fin = safe_normalize(a_fin, n)
            c_fin = fft_convolve_self(a_fin)
            obj_fin = 2.0 * n * c_fin.max() / S_best ** 2
            if obj_fin < global_best:
                global_best = obj_fin
                global_a = a_fin.copy()
                print("  Final LSE: obj=%.6f" % global_best)

    # Fallback
    if global_a is None:
        n_fb = 8192
        S_fb = np.sqrt(2.0 * n_fb)
        global_a = make_paired_init(n_fb, rng, S_fb)
        global_a = safe_normalize(global_a, n_fb)

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Final verification: %.6f (internal obj: %.6f), n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: invalid, using fallback")
        global_a = np.ones(4096, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback eval: %.6f" % ver)

    return final_list
