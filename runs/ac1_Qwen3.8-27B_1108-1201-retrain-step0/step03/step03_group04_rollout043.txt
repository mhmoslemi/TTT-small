"""
Minimize 2*n*max_conv/sum^2 using L-BFGS-B with log-sum-exp approximation.
Uses softmax parameterization for non-negativity and sum constraint.
Multi-phase: low-beta L-BFGS-B → high-beta L-BFGS-B → subgradient polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

        c_max = c.max()

        total = exp_bc.sum()

                      options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-8})

    x_shifted = result.x - result.x.max()
    exp_x = np.exp(x_shifted)
    a_new = exp_x / exp_x.sum() * S
    
    return a_new, result

"""
Minimize 2*n*max_conv/sum^2 using multi-phase optimization:
1. FISTA with softmax weights for global exploration
2. L-BFGS-B with log-sum-exp for local refinement  
3. Active-set subgradient for final polish

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_grad_from_weights(a, p):
    """
    Compute gradient of sum_j p[j] * c[j] where c = a*a (linear conv).
    Returns grad of shape (n,).
    grad[i] = 2 * sum_k a[k] * p[k+i] (cross-correlation of a and p at lag i)
    """
    n = len(a)
    fsize = 1 << (3 * n - 2).bit_length()
    A = fft(a, fsize)
    P = fft(p, fsize)
    cross = np.real(ifft(np.conj(A) * P, fsize))[:n]
    return 2.0 * cross


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.full(len(c), 1.0 / len(c))
        return w
    return exp_bc / total


def toppk_weights(c, top_k):
    """Compute normalized top-k weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights
