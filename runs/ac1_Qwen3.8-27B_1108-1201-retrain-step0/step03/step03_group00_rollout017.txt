"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_value_and_grad(a, c, beta):
    """Compute LSE value and gradient scaled by beta."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    val = cm + np.log(tot) / beta
    if tot < 1e-300:
        return val, np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return val, grad


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_rate=0.003,
              beta_max=1000.0, use_hybrid=False, verbose=False):
    """FISTA with LSE surrogate and adaptive step."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    patience = 3000

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        val, grad = lse_value_and_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        step = 1.0 / (2.0 * beta)

        # Adaptive: if stuck, reduce step and inject noise
        if it - last_improve_it > patience:
            step *= 0.3
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve_it = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

                t = 1.0

    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

        # So t ranges from max(0, k-(n-1)) to min(k, n-1)

            best = cur

            top_k = 256
        elif it <= 8:
            top_k = 512
        else:
            top_k = 1024

            break

def simulated_annealing(a, n, S, max_time, rng, T_init=0.1, T_rate=0.999):
    """Simulated annealing on the simplex."""
    a = normalize_sum(a, n).copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    T = T_init
    epoch = 0

    while time.monotonic() - start < max_time:
        epoch += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()

        # Perturb: pick a random subset of coordinates and modify them
        n_perturb = max(10, n // 50)
        idx = rng.choice(n, n_perturb, replace=False)
        noise_scale = T * S / np.sqrt(n)
        a_perturbed = a.copy()
        a_perturbed[idx] += rng.normal(0, noise_scale, n_perturb)
        a_perturbed = project_simplex(a_perturbed, S)

        c_new = fft_convolve_self(a_perturbed)
        cur_new = c_new.max()

        # Accept if better or with probability exp(-(cur_new - cur)/T)
        if cur_new < cur:
            a = a_perturbed
        else:
            delta = cur_new - cur
            if delta < 1e-15:
                a = a_perturbed
            else:
                prob = np.exp(-delta / max(T, 1e-15))
                if rng.random() < prob:
                    a = a_perturbed

        T *= T_rate
        if T < 1e-6:
            T = 1e-6

        # Occasionally boost temperature to escape
        if epoch % 500 == 0 and rng.random() < 0.1:
            T = max(T, 0.05)

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def downsample_upsample(a, n_small):
    """Downsample by averaging, upsample by interpolation."""
    n = len(a)
    nblocks = n // n_small
    a_small = a[:nblocks * n_small].reshape(n_small, nblocks).mean(axis=1)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n)
    return np.interp(t_new, t_old, a_small)


def power_mean_fista(a_init, n, S, max_time, rng, p_init=4.0, p_rate=0.005, p_max=200.0):

        w = w / wsum

    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Get starting sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given = np.clip(given, 0, 1000)
            if given.sum() > 0.01:
                pass
            else:
                given = None
    except Exception:
        passed = True

    print("Given sequence length: %s" % (len(given) if given is not None else "None"))

    # Determine which lengths to try
    lengths_to_try = []
    if given is not None:
        lengths_to_try.append(len(given))
    # Add nearby lengths
    if given is not None:
        for factor in [0.5, 1.5, 2.0]:
            alt = int(len(given) * factor)
            alt = max(512, min(32768, alt))
            if alt not in lengths_to_try:
                lengths_to_try.append(alt)
    else:
        for n_try in [4096, 8192, 16384]:
            lengths_to_try.append(n_try)

    # Time allocation: give most time to the first (given) length
    total_available = deadline - time.monotonic()
    if total_available < 60:
        print("Not enough time, returning given or default")
        if given is not None:
            a_norm = normalize_sum(given, len(given))
            return [float(x) for x in a_norm]
        return [float(x) for x in np.ones(256) * (np.sqrt(512.0) / 256)]

    n_main = lengths_to_try[0]
    n_others = len(lengths_to_try) - 1

    # Allocate: 75% to main, 25% split among others (if any)
    if n_others > 0 and total_available > 120:
        main_time = total_available * 0.72
        other_time_each = (total_available * 0.28) / n_others
    else:
        main_time = total_available
        other_time_each = 0

    print("=== Main length: %d, time: %.0fs ===" % (n_main, main_time))
    print("=== Other lengths: %s ===" % (lengths_to_try[1:] if n_others > 0 else "[]"))

    # ===== MAIN LENGTH OPTIMIZATION =====
    def optimize_length(n, time_budget, start_seq=None):
        """Full optimization pipeline for a given length."""
        S = np.sqrt(2.0 * n)
        t0 = time.monotonic()
        local_best = np.inf
        local_best_a = None

        # Step 1: FISTA exploration (40% of time)
        t1 = time_budget * 0.35
        print("  [n=%d] FISTA exploration (%.0fs)" % (n, t1))

        inits = []
        if start_seq is not None and len(start_seq) == n:
            inits.append(start_seq)
        if start_seq is not None and len(start_seq) != n:
            # Resample
            t_old = np.linspace(0, 1, len(start_seq))
            t_new = np.linspace(0, 1, n)
            inits.append(np.interp(t_new, t_old, start_seq))

        if not inits:
            inits.append(np.ones(n) * S / n)
            inits.append(rng.random(n) + 0.1)

        # Also add structured inits
        tt = np.linspace(0, 1, n, endpoint=False)
        inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
        inits.append(np.sin(np.pi * tt) ** 2 + 0.01)

        t_per_init = t1 / len(inits)
        for i, init in enumerate(inits):
            if time.monotonic() - t0 > time_budget * 0.35:
                break
            a_opt, score = run_fista(init, n, S, t_per_init, rng,
                                     beta_init=5.0, beta_rate=0.004,
                                     beta_max=800.0)
            a_opt = normalize_sum(a_opt, n)
            c_chk = fft_convolve_self(a_opt)
            real_score = c_chk.max()
            if real_score < local_best:
                local_best = real_score
                local_best_a = a_opt.copy()
                print("    init %d: %.6f (t=%.0fs)" % (i, real_score, time.monotonic() - t0))

        if local_best_a is None:
            local_best_a = inits[0].copy()
            local_best = fft_convolve_self(local_best_a).max()

        # Step 2: Power mean FISTA (15% of time)
        remaining = time_budget - (time.monotonic() - t0)
        t2 = max(5.0, remaining * 0.15)
        print("  [n=%d] Power mean FISTA (%.0fs)" % (n, t2))
        a_pm, score_pm = power_mean_fista(local_best_a, n, S, t2, rng,
                                          p_init=4.0, p_rate=0.005, p_max=100.0)
        a_pm = normalize_sum(a_pm, n)
        c_pm = fft_convolve_self(a_pm)
        if c_pm.max() < local_best:
            local_best = c_pm.max()
            local_best_a = a_pm.copy()
            print("    Power mean: %.6f" % local_best)

        # Step 3: LP polishing (15% of time, limited to a few calls)
        remaining = time_budget - (time.monotonic() - t0)
        t3 = max(5.0, remaining * 0.15)
        print("  [n=%d] LP polish (%.0fs)" % (n, t3))
        a_lp, score_lp = run_lp_polish(local_best_a, n, S, t3, rng)
        a_lp = normalize_sum(a_lp, n)
        c_lp = fft_convolve_self(a_lp)
        if c_lp.max() < local_best:
            local_best = c_lp.max()
            local_best_a = a_lp.copy()
            print("    LP: %.6f" % local_best)

        # Step 4: Simulated annealing (15% of time)
        remaining = time_budget - (time.monotonic() - t0)
        t4 = max(5.0, remaining * 0.15)
        print("  [n=%d] Simulated annealing (%.0fs)" % (n, t4))
        a_sa, score_sa = simulated_annealing(local_best_a, n, S, t4, rng,
                                             T_init=0.05, T_rate=0.9995)
        a_sa = normalize_sum(a_sa, n)
        c_sa = fft_convolve_self(a_sa)
        if c_sa.max() < local_best:
            local_best = c_sa.max()
            local_best_a = a_sa.copy()
            print("    SA: %.6f" % local_best)

        # Step 5: Final high-beta FISTA (15% of time)
        remaining = time_budget - (time.monotonic() - t0)
        t5 = max(5.0, remaining * 0.20)
        print("  [n=%d] Final FISTA (%.0fs)" % (n, t5))
        a_fin, score_fin = run_fista(local_best_a, n, S, t5, rng,
                                     beta_init=50.0, beta_rate=0.002,
                                     beta_max=3000.0)
        a_fin = normalize_sum(a_fin, n)
        c_fin = fft_convolve_self(a_fin)
        if c_fin.max() < local_best:
            local_best = c_fin.max()
            local_best_a = a_fin.copy()
            print("    Final FISTA: %.6f" % local_best)

        # Step 6: Coordinate descent (remaining time)
        remaining = time_budget - (time.monotonic() - t0)
        if remaining > 10:
            print("  [n=%d] Coordinate descent (%.0fs)" % (n, remaining))
            a_cd, score_cd = coordinate_descent(local_best_a, n, S, remaining, rng)
            a_cd = normalize_sum(a_cd, n)
            c_cd = fft_convolve_self(a_cd)
            if c_cd.max() < local_best:
                local_best = c_cd.max()
                local_best_a = a_cd.copy()
                print("    CD: %.6f" % local_best)

        print("  [n=%d] Best after optimization: %.6f (total time %.0fs)" %
              (n, local_best, time.monotonic() - t0))
        return local_best_a, local_best

    # Run main length
    a_main, score_main = optimize_length(n_main, main_time, start_seq=given)
    if score_main < global_best:
        global_best = score_main
        global_a = a_main.copy()
        n_best = n_main

    # Run other lengths
    for n_alt in lengths_to_try[1:]:
        if time.monotonic() > deadline:
            break
        alt_time = min(other_time_each, deadline - time.monotonic())
        if alt_time < 30:
            break
        print("\n=== Alt length: %d, time: %.0fs ===" % (n_alt, alt_time))
        a_alt, score_alt = optimize_length(n_alt, alt_time, start_seq=given)
        # Compute actual objective
        S_alt = np.sqrt(2.0 * n_alt)
        obj_alt = 2.0 * n_alt * score_alt / (S_alt ** 2)
        # Compare with global (all normalized to sum=sqrt(2n), so obj = 2n*max/S^2 = 2n*max/(2n) = max)
        # Wait: if sum = sqrt(2n), then obj = 2n * max / (2n) = max. So the objective equals max_b!
        # That means all we need to compare is max_b directly.
        if score_alt < global_best:
            global_best = score_alt
            global_a = a_alt.copy()
            n_best = n_alt
            print("  New best: n=%d, max_b=%.6f (obj=%.6f)" % (n_alt, score_alt, obj_alt))

    # Final: verify and return
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    result = [float(x) for x in global_a]

    # Verify with evaluate_sequence
    ver = evaluate_sequence(result)
    print("\n=== FINAL: n=%d, internal score=%.6f, evaluate_sequence=%.6f ===" %
          (len(global_a), global_best, ver if np.isfinite(ver) else float('inf')))

    return result
