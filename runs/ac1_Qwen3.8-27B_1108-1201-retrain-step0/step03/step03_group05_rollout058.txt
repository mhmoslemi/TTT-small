import numpy as np

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = theta_candidates[-1]
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = theta_candidates[rho - 1]
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_subgradient_direct(a, active_indices, normalize=True):
    """
    Compute subgradient of max R(k) w.r.t. a, using direct computation.
    active_indices: array of k values where R(k) is in the top-k.
    Returns gradient vector of length n.
    """
    n = len(a)
    p = len(active_indices)
    g = np.zeros(n, dtype=np.float64)
    for k in active_indices:
        k = int(k)
        # dR(k)/da[m] = 2*a[k-m] for 0 <= k-m < n
        # i.e., for m in [max(0, k-n+1), min(k, n-1)]:
        #   g[m] += a[k-m]
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        if m_start <= m_end:
            # a[k-m] for m = m_start, ..., m_end
            # = a[k-m_end], ..., a[k-m_start]
            # = a[m_start_reversed], ..., a[end]
            # k-m for m=m_start: k-m_start
            # k-m for m=m_end: k-m_end
            # So a[k-m] ranges from a[k-m_start] down to a[k-m_end]
            # k-m_start = k - max(0, k-n+1)
            # k-m_end = k - min(k, n-1)
            idx_end = k - m_start  # = min(k, n-1)
            idx_start = k - m_end  # = max(0, k-n+1)
            g[m_start:m_end+1] += a[idx_start:idx_end+1][::-1]
    g *= 2.0 / p
    return g

def logsumexp_smoothed_convol_max(a, beta):
    """Compute (1/beta)*log(sum exp(beta*c_k)) for c = conv(a,a)."""
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    return (c_max + np.log(exp_bc.sum())) / beta

def softmax_update(a, z, S):
    """Convert softmax params z to sequence a = S * softmax(z)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return S * exp_z / exp_z.sum()

    else:

    t = 1.0
    start = time.monotonic()

    best_local = a.copy()

        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)

        c = fft_convolve_self(y)

            top_indices = part

        g = compute_subgradient_direct(y, top_indices)

        ys = y.sum()
        if ys > 1e-15:

        a = x_new
        t = t_new

                N_fft = 1 << (2*n-1).bit_length()
                A = fft(a, N_fft)
                target_mag = S / np.sqrt(N_fft)
                # Blend: reduce peaks, boost valleys
                mag = np.abs(A)
                phase = A / (mag + 1e-30)
                # Target: flat magnitude
                new_mag = 0.7 * mag + 0.3 * target_mag
                A_new = new_mag * phase
                a_try = np.real(ifft(A_new))[:n]
                a_try = np.maximum(a_try, 0.0)
                a_try_s = a_try.sum()
                if a_try_s > 1e-15:
                    a_try = a_try * (S / a_try_s)
                    c_try = fft_convolve_self(a_try)
                    if c_try.max() < best_score:
                        best_score = c_try.max()
                        best_local = a_try.copy()
                        a = a_try
                        y = a_try.copy()
                        t = 1.0
        
        # Random restart
        if it % 3000 == 0 and rng.random() < 0.1:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
        
        if it % 500 == 0:
            print("    SG iter %d: best=%.8f, top_k=%d" % (it, best_score, top_k))
    
    return best_local, best_score

def run_lbfgs_smoothed(a_init, n, S, max_time, rng,
                       beta_schedule=None, maxiter_per_beta=200):
    """L-BFGS-B on log-sum-exp smoothed objective with continuation."""
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
    
    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    
    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break
        
        def fun(z_val):
            c = fft_convolve_self(softmax_update(None, z_val, S))
            c_max = c.max()
            exp_bc = np.exp(beta * (c - c_max))
            return (c_max + np.log(exp_bc.sum()))
