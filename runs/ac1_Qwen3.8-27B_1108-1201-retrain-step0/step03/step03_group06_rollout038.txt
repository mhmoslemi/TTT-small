"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS on a log-sum-exp smoothed
objective with temperature continuation, multiple spectral initializations,
and targeted peak-shaving polish. Uses softmax parameterization for constraints.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as sp_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_a(theta, S):
    """Compute a = S * softmax(theta)."""
    t = theta - theta.max()
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * e / s


def smooth_obj_val(a, T):
    """Compute (1/T) log sum exp(T * c_k) for c = a*a."""
    c = fft_convolve_self(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    return c_max + np.log(np.exp(shifted).sum()) / T


def compute_smooth_grad(theta, T, S, n):
    """Compute objective value and gradient w.r.t. theta."""
    a = softmax_a(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    obj_val = c_max + np.log(exp_s.sum()) / T

    # Compute r[i] = sum_j w[i+j] * a[j] via flip-and-convolve
    # e[k] = w[2n-2-k], then r[i] = (e*b)[2n-2-i]
    fsize = 1 << (3 * n + 1).bit_length()
    e = np.zeros(fsize)
    e[:2 * n - 1] = w[::-1]  # e[k] = w[2n-2-k]
    b = np.zeros(fsize)
    b[:n] = a
    E = fft(e, fsize)
    B = fft(b, fsize)
    C = np.real(ifft(E * B))
    # r[i] = C[2n-2-i]
    idx = 2 * n - 2 - np.arange(n)
    r = C[idx]

    # Diagonal correction: a[i]*w[2i] for 2i < 2n-1
    di = 2 * np.arange(n)
    dm = di < (2 * n - 1)
    diag = np.zeros(n)
    diag[dm] = a[dm] * w[di[dm]]

    g_a = r + diag
    dot = np.dot(g_a, a)
    g_theta = g_a * a - (a / S) * dot

    return obj_val, g_theta


def run_lbfgs_continuation(theta_init, S, n, T_list, time_budget, max_iter_per_T=300):

    step = 0

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_list[min(step, len(T_list) - 1)] if len(T_list) < step else T_list[step % len(T_list)]
        # Actually use a schedule
        T = T_list[0] * (T_list[-1] / T_list[0]) ** (0.5 + 0.5 * progress)

        # Nesterov lookahead
        beta = momentum_init * (1.0 - 0.3 * progress)
        y = theta + (beta - 1.0) / (1.0 + beta) * (theta - a_prev)

        f_y, g_y = compute_smooth_grad(y, T, S, n)
        f_cur, g_cur = compute_smooth_grad(theta, T, S, n)

        gnorm = np.linalg.norm(g_cur)
        if gnorm < 1e-15:
            break

        # Backtracking line search from current point
        lr = lr_init / (1.0 + 0.5 * gnorm)
        theta_new = theta
        f_new = f_cur
        for i in range(20):
            candidate = theta - lr * g_cur
            f_trial, _ = compute_smooth_grad(candidate, T, S, n)
            if f_trial < f_cur - 1e-12 * abs(f_cur):
                theta_new = candidate
                f_new = f_trial
                break
            lr *= 0.5

        a_prev = theta.copy()
        theta = theta_new
        step += 1

        # Track best
        a = softmax_a(theta, S)
        c = fft_convolve_self(a)
        peak = c.max()
        if peak < best_peak:
            best_peak = peak
            best_theta = theta.copy()

    return best_theta, best_peak


def make_spectral_init(n, S, mode, rng):
    """Create initial theta vectors based on spectral shape."""
    i = np.arange(n)
    if mode == "uniform":
        a = np.ones(n)
    elif mode == "gaussian":
        a = np.exp(-0.5 * ((i - n / 2) / (n / 4)) ** 2)
    elif mode == "bilinear":
        t = i / (n - 1)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * t) + 0.2 * np.cos(2 * np.pi * 7 * t + rng.random() * 2)
        a = np.maximum(a, 0.1)
    elif mode == "tapered":
        t = i / (n - 1)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 0.05)
    elif mode == "random":
        a = rng.random(n)
        a = np.maximum(a, 0.01)
    elif mode == "chirp_like":
        # Non-negative chirp: modulate amplitude to create interference patterns
        freq = 0.02 + 0.03 * rng.random()
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * i + 0.5 * np.pi * freq * i ** 2 / n)
        a = np.maximum(a, 0.05)
    elif mode == "piecewise":
        # Piecewise constant with random levels
        n_seg = rng.integers(5, 20)
        seg_len = n // n_seg
        a = np.zeros(n)
        for s in range(n_seg):
            level = rng.random() * 0.8 + 0.2
            start = s * seg_len
            end = min(start + seg_len, n)
            a[start:end] = level
    else:
        a = np.ones(n)

    a = np.maximum(a, 1e-6)
    theta = np.log(a / a.sum()) + np.log(n)  # normalize so softmax gives roughly a/a.sum()*n
    # Actually just use log(a) since softmax will normalize
    theta = np.log(np.maximum(a, 1e-10))
    return theta


def targeted_peak_shave(a, S, n, time_budget, top_k=5):
    """Targeted optimization: minimize sum of top-k autocorrelation peaks."""
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a * (S / s)

    start = time.monotonic()
    best_peak = np.inf
    best_a = a.copy()
    rng = np.random.default_rng(99)

    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        cur_peak = c.max()
        if cur_peak < best_peak:
            best_peak = cur_peak
            best_a = a.copy()

        # Get top-k peak indices
        top_idx = np.argsort(c)[-top_k:]
        top_w = np.zeros(len(c))
        for k in top_idx:
            top_w[k] = 1.0 / top_k

        # Gradient of sum of top peaks: g_a[i] = sum_k top_w[k] * (a[k-i] + a[i]*[2i==k])
        fsize = 1 << (3 * n + 1).bit_length()
        e = np.zeros(fsize)
        e[:2 * n - 1] = top_w[::-1]
        b = np.zeros(fsize)
        b[:n] = a
        C = np.real(ifft(fft(e, fsize) * fft(b, fsize)))
        idx = 2 * n - 2 - np.arange(n)
        r = C[idx]
        di = 2 * np.arange(n)
        dm = di < (2 * n - 1)
        diag = np.zeros(n)
        diag[dm] = a[dm] * top_w[di[dm]]
        g_a = r + diag

        # Project gradient onto tangent space of the simplex
        g_proj = g_a - np.mean(g_a)

        # Line search
        gnorm = np.linalg.norm(g_proj)
        if gnorm < 1e-14:
            break
        g_dir = g_proj / gnorm

        lr_base = 0.01 * S / gnorm
        improved = False
        for lr_try in [lr_base, 0.5 * lr_base, 0.25 * lr_base, 0.1 * lr_base]:
            a_trial = a - lr_try * gnorm * g_dir
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial * (S / st)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_peak - 1e-10:
                a = a_trial
                improved = True
                break

        if not improved:
            # Add small random perturbation
            noise = 0.002 * S * rng.standard_normal(n)
            a = a + noise
            a = np.maximum(a, 0)
            st = a.sum()
            if st > 1e-15:
                a = a * (S / st)
            else:
                a = best_a.copy()

    return best_a


def coordinate_polish(a, S, n, time_budget):
    """Fast coordinate descent polish."""
    a = a.copy()
    start = time.monotonic()
    best_peak = fft_convolve_self(a).max()
    best_a = a.copy()
    rng = np.random.default_rng(777)

    n_factors = 8
    for sweep in range(200):
        if time.monotonic() - start > time_budget:
            break
        order = rng.permutation(n)
        any_improve = False
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-10:
                continue
            best_f = orig
            best_v = best_peak
            for frac in [0.5, 0.7, 1.3, 1.5, 0.3, 0.8, 1.2, 2.0]:
                a[i] = orig * frac
                st = a.sum()
                if st < 1e-15:
                    a[i] = orig
                    continue
                a_scaled = a * (S / st)
                c_ch = fft_convolve_self(a_scaled)
                v = c_ch.max()
                if v < best_v - 1e-11:
                    best_v = v
                    best_f = orig * frac
            a[i] = best_f
            st = a.sum()
            if st > 1e-15:
                a = a * (S / st)
            if best_v < best_peak - 1e-11:
                best_peak = best_v
                best_a = a.copy()
                any_improve = True
        if not any_improve and sweep > 5:
            break

    return best_a
