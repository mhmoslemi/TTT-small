"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Hybrid optimization combining exact top-k subgradient with adaptive
backtracking step size, mirror (multiplicative) descent as alternative metric,
structured perturbations, and LP-inspired direction finding. Multi-start from
diverse initializations including the previous best.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def mirror_descent_step(a, grad, eta, S):
    """Multiplicative (mirror) descent step maintaining sum = S and a >= 0."""
    log_a = np.log(np.maximum(a, 1e-300))
    new_log_a = log_a - eta * grad
    # Clamp to prevent underflow
    new_log_a = np.clip(new_log_a, -700, 700)
    new_a = np.exp(new_log_a - new_log_a.max())
    s = new_a.sum()
    if s < 1e-300:
        return a.copy()
    return new_a * (S / s)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def topk_subgradient(a, c, k):
    """Subgradient using top-k peaks of convolution."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for j in idx:
        lo = max(0, j - n + 1)
        hi = min(j, n - 1)
        i_vals = np.arange(lo, hi + 1)
        grad[i_vals] += 2.0 * a[j - i_vals]
    return grad / k


def lse_subgradient(a, c, beta):
    """LSE-smoothed gradient of max."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

            elif phase < 50:

            a_new = project_simplex(a - step * grad, S)

        # Check improvement
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / S**2

        if obj_new < cur_obj:
            a = a_new
            c = c_new
            step = min(step * 1.05, max_step)
        else:
            consecutive_fail += 1
            step *= 0.7
            if consecutive_fail > 20:
                # Restart with perturbation
                noise_scale = 0.002 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                step = 0.5
                consecutive_fail = 0

            if consecutive_fail > 50:
                # Larger perturbation
                noise_scale = 0.01 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c = fft_convolve_self(a)
                step = 1.0
                consecutive_fail = 0

    return best_a, best_obj


def run_fista_refined(a_init, n, S, max_time, rng):
    """FISTA with refined step size and hybrid gradient."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    base_step = 1.0

    while time.monotonic() - start < max_time:
        it += 1

        # Vary the gradient strategy
        phase = it % 300
        if phase < 10:
            c_y = fft_convolve_self(y)
            grad = topk_subgradient(y, c_y, 1)
            lip = np.linalg.norm(grad) * (n / S)
        elif phase < 100:
            c_y = fft_convolve_self(y)
            grad = topk_subgradient(y, c_y, 16)
            lip = np.linalg.norm(grad) * (n / S)
        else:
            beta = min(400.0, 10.0 + 0.03 * it)
            c_y = fft_convolve_self(y)
            grad = lse_subgradient(y, c_y, beta)
            lip = 2.0 * beta * S  # Lipschitz constant of LSE gradient

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / max(lip, 1e-15)
        step = min(step, 2.0 * S / (n * gnorm + 1e-15))

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        c_a = fft_convolve_self(a)
        cur_obj = 2.0 * n * c_a.max() / S**2
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = it

        # Periodic perturbations
        if it - last_improve > 3000 and rng.random() < 0.1:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-it / 50000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    return best_a, best_obj


def run_lp_direction(a, n, S, M, max_iter=20, tol=1e-6):
    """
    Approximate LP direction finding: find g >= 0 maximizing sum(g)
    subject to (a*g)[k] <= M for all k.
    Uses multiplicative weights on the dual.
    """
    # Dual: min M * sum(lambda) + sum(mu_j) * 0
    #   s.t. for each j: sum_k lambda_k * a[k-j] + mu_j >= 1
    #   lambda_k >= 0, mu_j >= 0
    # At optimum, complementary slackness: g_j > 0 => constraint tight.
    # Use multiplicative weights on lambda.

    size = 2 * n - 1
    # Initialize dual variables
    lam = np.ones(size)
    g = np.ones(n)  # primal variable

    for _ in range(max_iter):
        # Update primal: g_j = 1 if dual constraint is tight
        # Dual constraint for j: sum_k lam[k] * a[k-j] >= 1 (approximately)
        # (a * lam)[j] = sum_k lam[k] * a[k-j]... this is cross-correlation
        # Actually (a*g)[k] = sum_i a[i]*g[k-i], so the constraint for g_j
        # involves sum_k lam[k] * a[k-j] (treating a as fixed)

        # Compute (a cross-correlation lam) at each j
        # (a cross lam)[j] = sum_k a[k] * lam[k-j] ... this is getting complex
        # Let's use a simpler approach: projected gradient on the LP

        # For each constraint (a*g)[k] <= M:
        # (a*g)[k] = sum_{i: valid} a[i] * g[k-i]
        # Gradient w.r.t. g[j]: sum_k lambda[k] * a[k-j] (if 0 <= k-j < n)

        # Compute gradient of Lagrangian w.r.t. g
        # L = -sum(g) + sum_k lam[k] * ((a*g)[k] - M)
        # dL/dg[j] = -1 + sum_k lam[k] * a[k-j]

        # a[k-j] is non-zero when 0 <= k-j < n, i.e., j <= k < j+n
        # So dL/dg[j] = -1 + sum_{k=j}^{j+n-1} lam[k] * a[k-j] (with bounds)

        # This is a cross-correlation of lam and a
        fsize = 1 << (size + n - 1).bit_length()
        LAM = fft(lam, fsize)
        A = fft(a, fsize)
        # cross_corr[j] = sum_k lam[k] * a[k-j] = ifft(fft(lam) * conj(fft(a)))
        cross = np.real(ifft(LAM * np.conj(A)))[:size]
        # We need cross at specific indices... actually let me just compute (a*g)[k] directly

        # Simpler: use subgradient method on the primal LP
        # Primal: max 1^T g s.t. T g <= M * 1, g >= 0
        # where T is the convolution matrix (Toeplitz)

        # Current (a*g) values:
        ag = fft_convolve_cross(a, g)
        violated = np.maximum(ag - M, 0)

        if violated.max() < tol:
            break

        # Subgradient of constraint violation: for each violated k,
        # grad_k[j] = a[k-j] (if 0 <= k-j < n)
        viol_idx = np.where(ag > M + tol)[0]
        if len(viol_idx) == 0:
            break

        # Reduce g in the direction that violates most
        red = np.zeros(n)
        for k in viol_idx[:min(100, len(viol_idx))]:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            i_vals = np.arange(lo, hi + 1)
            red[i_vals] += a[k - i_vals] * (ag[k] - M)

        g = np.maximum(g - 0.5 * red, 0)

    # Normalize g
    gs = g.sum()
    if gs < 1e-15:
        return None
    g = g / gs
    return g


def fft_convolve_cross(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def structured_perturb(a, n, S, rng):
    """Apply a structured perturbation to escape local minima."""
    strategy = rng.integers(0, 5)
    a_new = a.copy()

    if strategy == 0:
        # Add a sinusoidal modulation
        freq = rng.uniform(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        mod = 1.0 + 0.3 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        a_new = a * mod
    elif strategy == 1:
        # Add a second harmonic
        t = np.arange(n) / n
        mod = 1.0 + 0.2 * np.cos(4 * np.pi * t) + 0.1 * np.cos(8 * np.pi * t)
        a_new = a * mod
    elif strategy == 2:
        # Scale edges vs center
        t = np.arange(n) / n
        edge_boost = 1.0 + 0.5 * np.cos(np.pi * t) ** 4
        a_new = a * edge_boost
    elif strategy == 3:
        # Random block modulation
        nblocks = rng.integers(4, 16)
        block_mod = rng.uniform(0.5, 1.5, nblocks)
        mod = np.repeat(block_mod, n // nblocks + 1)[:n]
        a_new = a * mod
    else:
        # Gaussian perturbation in frequency domain
        A = fft(a)
        noise = rng.normal(0, 0.05, len(A)) + 1j * rng.normal(0, 0.05, len(A))
        A = A * (1 + noise)
        a_new = np.real(ifft(A))

    a_new = np.maximum(a_new, 0)
    return project_simplex(a_new, S)


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(1.0 + 0.2 * np.cos(4 * np.pi * tt) + 0.1 * np.sin(6 * np.pi * tt))
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.25))
    inits.append(1.0 / (1.0 + np.exp(-10 * (tt - 0.3))))
    inits.append(1.0 / (1.0 + np.exp(-10 * (0.7 - tt))))

    # Edge-heavy (inverse window)
    inits.append(1.0 + 2.0 * (tt ** 4 + (1 - tt) ** 4))

    # Multi-periodic
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt + 0.5))

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    # Try to get the previous best sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose main length
    if given is not None and 1024 <= len(given) <= 16384:
        main_len = len(given)
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * n)

    print("Starting optimization with n=%d, budget=%.0fs" % (n, budget_s))
    t0 = time.monotonic()

    # Collect initializations
    inits = []
    if given is not None:
        inits.append(("given", given.copy()))

    gen_inits = generate_initializations(n, rng)
    for i, g in enumerate(gen_inits):
        inits.append(("init_%d" % i, g))

    # Structured perturbations of the given
    if given is not None:
        for i in range(3):
            pert = structured_perturb(given, n, S, rng)
            inits.append(("pert_%d" % i, pert))

    n_inits = len(inits)
    usable_time = max(120.0, deadline - time.monotonic())

    # Phase 1: Diverse exploration (55% of time)
    exploration_time = usable_time * 0.55
    per_init = exploration_time / n_inits

    print("=== Phase 1: Exploration (%d inits, total %.0fs) ===" % (n_inits, exploration_time))

    p1_start = time.monotonic()
    methods = ["hybrid", "topk", "lse", "mirror", "exact"]

    for i, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - usable_time * 0.40 - 20:
            break
        arr_n = len(arr)
        if arr_n < 64:
            continue
        S_i = np.sqrt(2.0 * arr_n)
        t_alloc = per_init
        if t_alloc < 5:
            continue

        method = methods[i % len(methods)]

        a_opt, score = run_adaptive_subgrad(arr, arr_n, S_i, t_alloc, rng, method=method)
        a_norm = normalize_sum(a_opt, arr_n)
        real_score = compute_objective(a_norm)

        if real_score < global_best:
            global_best = real_score
            global_a = a_norm.copy()
            print("  [%s] n=%d: obj=%.6f (elapsed %.0fs)" % (
                name, arr_n, global_best, time.monotonic() - p1_start))

    print("=== After Phase 1: best=%.6f, elapsed=%.0fs ===" % (
        global_best, time.monotonic() - p1_start))

    # Phase 2: Refined FISTA on best candidates (25% of time)
    remaining = deadline - time.monotonic()
    polish_time = remaining * 0.45
    print("=== Phase 2: Refined FISTA (remaining %.0fs) ===" % polish_time)

    if global_a is not None and polish_time > 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)

        # Run refined FISTA
        seg1 = polish_time * 0.5
        a_f, s_f = run_fista_refined(global_a, n_best, S_best, seg1, rng)
        a_f = normalize_sum(a_f, n_best)
        s_f = compute_objective(a_f)
        if s_f < global_best:
            global_best = s_f
            global_a = a_f.copy()
            print("  After FISTA: %.6f" % global_best)

        # Run adaptive subgradient with 'hybrid' method
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            seg2 = remaining2 * 0.5
            a_h, s_h = run_adaptive_subgrad(global_a, n_best, S_best, seg2, rng, method="hybrid")
            a_h = normalize_sum(a_h, n_best)
            s_h = compute_objective(a_h)
            if s_h < global_best:
                global_best = s_h
                global_a = a_h.copy()
                print("  After hybrid subgrad: %.6f" % global_best)

    # Phase 3: Intensive micro-optimization (20% of time)
    remaining = deadline - time.monotonic()
    micro_time = remaining * 0.40
    print("=== Phase 3: Micro-optimization (remaining %.0fs) ===" % micro_time)

    if global_a is not None and micro_time > 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)

        # Very small steps with exact subgradient
        a_micro = global_a.copy()
        micro_start = time.monotonic()
        step = 0.01
        it = 0
        best_micro = global_best

        while time.monotonic() - micro_start < micro_time:
            it += 1
            c = fft_convolve_self(a_micro)
            cur = 2.0 * n_best * c.max() / S_best**2

            if cur < best_micro - 1e-12:
                best_micro = cur

            # Top-1 subgradient
            grad = topk_subgradient(a_micro, c, 1)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            # Very small adaptive step
            eff_step = step / (1.0 + 0.001 * it)
            a_try = project_simplex(a_micro - eff_step * grad, S_best)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n_best * c_try.max() / S_best**2

            if obj_try < cur:
                a_micro = a_try
                step = min(step * 1.02, 0.5)
            else:
                step *= 0.9
                if step < 0.001:
                    # Try a perturbation
                    noise_scale = 0.001 * S_best / np.sqrt(n_best)
                    a_micro = project_simplex(a_micro + rng.normal(0, noise_scale, n_best), S_best)
                    step = 0.01

            if it % 5000 == 0:
                print("    micro it=%d, obj=%.6f" % (it, best_micro))

        a_micro = normalize_sum(a_micro, n_best)
        s_micro = compute_objective(a_micro)
        if s_micro < global_best:
            global_best = s_micro
            global_a = a_micro.copy()
            print("  After micro: %.6f" % global_best)

    # Phase 4: Try LP-inspired direction (if time remains)
    remaining = deadline - time.monotonic()
    print("=== Phase 4: LP direction (remaining %.0fs) ===" % remaining)

    if global_a is not None and remaining > 30:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_lp = global_a.copy()
        c_lp = fft_convolve_self(a_lp)
        M_lp = c_lp.max()

        # Run a simple LP-direction iteration
        lp_start = time.monotonic()
        for lp_it in range(50):
            if time.monotonic() - lp_start > remaining - 10:
                break

            # Find direction: reduce the peak lag
            k_star = int(np.argmax(c_lp))
            # Gradient at the peak
            grad_peak = topk_subgradient(a_lp, c_lp, 1)
            gnorm = np.linalg.norm(grad_peak)
            if gnorm < 1e-15:
                break

            # Take a step
            step_lp = S_best / (n_best * gnorm * (1.0 + 0.1 * lp_it))
            a_try = project_simplex(a_lp - step_lp * grad_peak, S_best)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n_best * c_try.max() / S_best**2
            obj_cur = 2.0 * n_best * c_lp.max() / S_best**2

            if obj_try < obj_cur:
                a_lp = a_try
                c_lp = c_try
            else:
                # Try top-k instead
                for k_try in [5, 20, 50]:
                    grad_k = topk_subgradient(a_lp, c_lp, k_try)
                    a_try2 = project_simplex(a_lp - step_lp * grad_k, S_best)
                    c_try2 = fft_convolve_self(a_try2)
                    obj_try2 = 2.0 * n_best * c_try2.max() / S_best**2
                    if obj_try2 < obj_cur:
                        a_lp = a_try2
                        c_lp = c_try2
                        break

        a_lp = normalize_sum(a_lp, n_best)
        s_lp = compute_objective(a_lp)
        if s_lp < global_best:
            global_best = s_lp
            global_a = a_lp.copy()
            print("  After LP direction: %.6f" % global_best)

    # Final phase: if we still have time, try one more full optimization
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 60:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("=== Final optimization (remaining %.0fs) ===" % remaining)

        # Run with a different random seed for diversity
        rng2 = np.random.default_rng(seed + 12345)
        a_fin, s_fin = run_fista_refined(global_a, n_best, S_best, remaining * 0.7, rng2)
        a_fin = normalize_sum(a_fin, n_best)
        s_fin = compute_objective(a_fin)
        if s_fin < global_best:
            global_best = s_fin
            global_a = a_fin.copy()
            print("  Final: %.6f" % global_best)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("Returning sequence of length %d, obj=%.6f" % (len(global_a), global_best))
    print("Total time: %.0fs" % (time.monotonic() - t0))
    return [float(x) for x in global_a]
