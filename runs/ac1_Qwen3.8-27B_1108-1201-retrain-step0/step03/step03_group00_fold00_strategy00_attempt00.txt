def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf for invalid input."""
    n = len(sequence)

    sequence = [min(1000.0, x) for x in sequence]  # Protect against too-large values
    S = sum(sequence)
    if S < 0.01:
        return np.inf
    convolved = np.convolve(sequence, sequence, mode='full')
    M = max(convolved)
    return float(2 * n * M / (S ** 2))

def fft_convolve_self(a):
    """Fast FFT-based self-convolution."""
    n = len(a)
    size = next power of 2 >= n+n-1
    n_pad = math.ceil(size / n) * n
    a_pad = np.zeros(n_pad)
    a_pad[:n] = a
    A = fft(a_pad, n_pad)
    conv = ifft(A * A)[:size]
    return np.abs(conv[:len(conv)])

    # Pattern 1: simple periodic (square wave)
    pattern1 = np.zeros(n_seq)
    pattern1[[i for i in range(n_seq) if i % 2 == 0]] = 0.54
    pattern1 = [pattern1[i] + 0.1*rng.random() for i in range(n_seq)]
    init_patterns.append(('trig_square', pattern1))
    # Pattern 2: sinusoidal with modulated amplitude
    pattern2 = 0.4 + 0.1 * np.sin(2 * np.pi * tt + 0.3)
    pattern2 = [max(0.01, max(0, x)) for x in pattern2]
    init_patterns.append(('trig_sin', pattern2))
    # Pattern 3: modulated amplitude with exponential decay
    decay = rng.uniform(0.5, 0.8, 1)
    pattern3 = 1.0 / (1 + np.exp(-4.0 * (tt - 0.5) * decay))
    pattern3 = [max(0.01, max(0, x)) for x in pattern3]
    init_patterns.append(('sigmoidal', pattern3))
    # If n is large enough for block structures, generate block patterns
    if n >= 256:
        n_blocks = min(32, n//32)
        block_size = rng.integers(2,10, size=n_blocks)
        block_vals = rng.uniform(0.5,1.5, size=n_blocks)
        pattern4 = np.sort(block_vals)[::-1]
        grab_size = min(n, pattern4.shape[0] * block_size[0])
        pattern4 = pattern4[:grab_size] * block_size[0]
        pattern4 = [pattern4[i] + 0.1*rng.random() for i in range(len(pattern4))]
        init_patterns.append(('block_peak', pattern4))
    # Random initialization (clamped)
    pattern5 = rng.random(n_seq) + 0.1
    pattern5 = [max(0, min(1000, x)) for x in pattern5]
    init_patterns.append(('random_base', pattern5))
    return init_patterns

def propose_candidate(seed=42, budget_s=2400, max_length=15000, **kwargs):
    """Propose a sequence of coefficients minimizing the evaluation function.

    This function searches over multiple sequence lengths using FISTA. The optimization
    is done for each length separately, then the best candidate across all lengths is chosen.

    Steps:
    1. Test various key lengths (powers of two) using the same optimization strategy.
    2. Generate initializations for each length with distinct patterns.
    3. For each candidate length, assign time proportional to length but capped at 800 seconds.
    4. Run FISTA until time runs out or improvement stalls.
    5. Maintain a global best candidate and objective value.
    6. If no candidate found, use fallback sinusoid.

    Returns: best candidate sequence from the smallest upper bound (closest to the target)

    Note: The previous program found 1.507018 for n=8192. This program aims for better.

    """
    rng = np.random.default_rng(seed)
    # Time window
    budget_s_total = budget_s
    try:
        deadline = time.time() + budget_s_total
    except RuntimeError:
        deadline = time.time() + budget_s_total
    
    # Define candidate lengths: powers of two around 2048, 8192
    lengths = [256, 512, 1024, 4096, 8192, 16384]
    # Filter lengths within maximum allowed
    valid_lengths = [int(l) for l in lengths if l <= max_length]
    # Adjust time budget proportionally, but don't exceed 8000 seconds
    time_budget = max(30.0, budget_s_total / len(valid_lengths))
    
    # List to track candidates and their objectives
    candidates = []
    objectives = {}
    best_obj_so_far = float('inf')
    
    # Run FISTA for each length
    for L in valid_lengths:
        # Check if we have time left to run at least one optimization
        remain_time = deadline - time.time()
        if remain_time <= 60.0:
            if best_obj_so_far is not None:
                candidates.append(('fallback', generate_initializations(L)[0]))
            break
            
        # Generate several initializations for this length
        init_list = generate_initializations(L, seed)
        n_inits = len(init_list)
        
        # Precompute expected run time for each
        time_est = [estimate_convolution_time(L, rng) for (_, _) in init_list]
        # Normalize total time per initialization
        time_slice = [max(120.0, (remain_time * t) / sum(time_est)) for t in time_est]
        
        # Run FISTA for each initialization
        for idx, (name, arr) in enumerate(init_list):
            try:
                # Get remaining time for this run
                current_time = time.time()
                time_left = deadline - current_time
                if time_left < time_slice[idx] * 0.2:
                    time_slice[idx] = min(time_left, time_slice[idx])
                    
                # Optimization Parameters per length and iteration
                beta_range = (5.0, 1000.0)  # Lower range for larger n (better to be sharper)

                evaluated_obj = evaluate_sequence(a_opt)
                if np.isfinite(evaluated_obj) and evaluated_obj < best_obj_so_far:
                    best_obj_so_far = evaluated_obj
                    candidates.append((name, a_opt))
                    
                # Backup regardless of improvement
                a_opt = normalize_sum(a_opt, L)
                # Save for fallback if necessary later
                a_opt = [max(0, min(1000, x)) for x in a_opt]
                objectives[name] = min(objectives.get(name, float('inf')), evaluated_obj)
                
            except KeyboardInterrupt:
                break
                
        # Update best candidate after each length
        best_candidate = candidates[np.argmin([x[1] for x in candidates])]
        break  # Once we have found a candidate, break
    
    # If we broke out due to time, but no candidate found, use fallback
    if len(candidates) == 0:
        # Generate fallback sequence for the smallest length tested
        L_fallback = valid_lengths[0]
        arr = [0.5 + 0.5 * math.cos(2 * math.pi * x) for x in np.linspace(0,1,L_fallback)]

            obj_fallback = evaluate_sequence(arr)
        except:
            obj_fallback = np.inf

        best_candidate = ('fallback', arr)
        candidates.append(best_candidate)
        objectives['fallback'] = obj_fallback
        
    # Return the best candidate sequence
    best_name, best_seq = candidates[0]
    return [float(x) for x in best_seq]
