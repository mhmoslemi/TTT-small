import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT (linear, not circular)."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def cross_corr_fft(w, a):
    """Compute cross-correlation (w * a)[i] = sum_k w[k]*a[k-i] using FFT."""
    n_w = len(w)
    n_a = len(a)
    fsize = 1 << (n_w + n_a - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return corr


def normalize_nonneg(a, S):
    """Project to non-negative and scale to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_weighted_grad(c, a, T, top_k=None):
    """Compute weighted subgradient with top-k peak focus.
    
    g[i] = 2 * sum_k w[k] * a[k-i] where w is based on exp(T*c/c_max)
    If top_k specified, only use weights for the top-k peaks.
    """
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)

    if top_k is not None and top_k < len(c):
        # Use top-k peaks
        c_sorted_idx = np.argpartition(c, -top_k)[-top_k:]
        w = np.zeros(len(c))
        c_top = c[c_sorted_idx]
        scaled = T * c_top / c_max
        scaled -= scaled.max()
        w_top = np.exp(scaled)
        w_top /= w_top.sum()
        w[c_sorted_idx] = w_top
    else:
        scaled = T * c / c_max
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

    # Cross-correlation: g[i] = 2 * sum_k w[k] * a[k-i]
    corr = cross_corr_fft(w, a)
    return 2.0 * corr[:n]


def run_log_adam_osc(a_init, n, S, time_budget, T_min=5.0, T_max=500.0,
                     lr_init=0.04, beta1=0.9, beta2=0.999, top_k=None,
                     oscillation_period=500, verbose=False):
    """Adam on log-parameters with oscillating temperature schedule."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    no_improve = 0
    max_no_improve = 200

        # Oscillating temperature: T = T_min + (T_max - T_min) * (0.5 + 0.5*sin) * progress_adj
        phase = 2 * np.pi * t / oscillation_period
        osc = 0.5 + 0.5 * np.sin(phase)
        # Overall trend upward with oscillation
        T = T_min + (T_max - T_min) * progress * (0.3 + 0.7 * osc)

        # Convert theta to a
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
        else:
            a = S * exp_theta / total

        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve > max_no_improve:
            noise_scale = 0.01 * (1.0 - progress)
            rng = np.random.default_rng(t)
            a = best_a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 1e-12)
            a = a * (S / a.sum())
            theta = np.log(a / a.sum())
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            continue

        # Compute weighted gradient
        g = compute_weighted_grad(c, a, T, top_k=top_k)

        # Chain rule for log-parameters

        g_theta = g * a - (a / S) * dot

        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-14:
            break

        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)

        v_hat = v / (1.0 - beta2 ** t)

        if verbose and t % 2000 == 0:
            print("    AdamOsc t=%d: obj=%.7f, T=%.1f" % (t, best_obj, T))

    return best_a, best_obj


def run_subgrad_nesterov(a_init, n, S, time_budget, T_init=50.0, T_final=2000.0,
                         lr_init=0.08, momentum=0.9, top_k=None,
                         noise_scale=0.01, verbose=False):
    """Direct subgradient with Nesterov momentum and backtracking."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    a_prev = a.copy()
    beta = momentum
    start = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 100

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.8)

        # Nesterov lookahead
        y = a + (beta - 1.0) / (1.0 + beta) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        else:
            y = a.copy()

        c_y = fft_convolve_self(y)
        obj_y = c_y.max()

        # Track best at current point
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart
        if no_improve > max_no_improve:
            scale = noise_scale * max(0.1, 1.0 - progress)
            rng = np.random.default_rng(step + 1000)
            a = best_a + scale * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            a_prev = a.copy()
            no_improve = 0
            step += 1
            continue

        # Gradient at lookahead
        g = compute_weighted_grad(c_y, y, T, top_k=top_k)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g_dir = g / gnorm

        # Backtracking line search
        lr = lr_init * (1.0 - 0.6 * progress)
        a_trial = a - lr * g_dir
        a_trial = normalize_nonneg(a_trial, S)
        c_trial = fft_convolve_self(a_trial)
        obj_trial = c_trial.max()

        for _ in range(10):
            if obj_trial < cur_obj - 1e-10 * max(cur_obj, 0.1):
                break
            lr *= 0.5
            if lr < 1e-10:
                break
            a_trial = a - lr * g_dir
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()

        a_prev = a.copy()
        a = a_trial
        step += 1

        if verbose and step % 500 == 0:
            print("    SG t=%d: obj=%.7f, T=%.1f" % (step, best_obj, T))

    return best_a, best_obj


def coordinate_descent(a, n, S, time_budget, max_sweeps=100, verbose=False):
    """Coordinate descent with multiplicative updates."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    rng = np.random.default_rng(42)

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            original = a[i]
            if original < 1e-14:
                continue

            best_local = original
            best_local_obj = c.max()

            # Try multiplicative changes
            for factor in [0.7, 1.3, 0.5, 1.5, 0.85, 1.15]:
                a[i] = original * factor
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-12:
                    best_local_obj = obj_check
                    best_local = original * factor

            if best_local != original:
                a[i] = best_local
                a = normalize_nonneg(a, S)
                c = fft_convolve_self(a)
                if c.max() < best_obj - 1e-12:
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
            else:
                a[i] = original

        if not improved and sweep >= 5:
            if verbose:
                print("    CD converged at sweep %d" % sweep)
            break

    return best_a, best_obj


def try_zero_masking(a, n, S, n_candidates=30):

        saved = a[idx]

    return best_a, best_obj


def get_initialization(name, n, S, rng):

    elif name == "cosine_mix":

        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "two_hump":
        a = np.exp(-0.5 * ((i - n * 0.3) / (n / 12)) ** 2) + \
            np.exp(-0.5 * ((i - n * 0.7) / (n / 12)) ** 2)
        return a * (S / a.sum())
    elif name == "gaussian":
        a = np.exp(-0.5 * ((i - n / 2) / (n / 5)) ** 2)

    elif name == "sinc_like":
        x = (i - n / 2) / (n / 4)
        a = np.sinc(x) ** 2
        a = np.maximum(a, 0.001)
        return a * (S / a.sum())
    else:

    # Round 1: Adam oscillating (25% time)
    r1_time = min(0.25 * total_budget, deadline - time.monotonic() - 5)
    if r1_time > 15:
        print("\n[Round 1] Adam Osc T=10-800, top_k=50 (%.0fs)" % r1_time)
        t0 = time.monotonic()
        a_best, obj_best = run_optimization_round(
            a_start, primary_n, S, r1_time, strategy="adam_osc", verbose=True)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  R1 done: obj=%.8f (time=%.0fs)" % (ver, time.monotonic() - t0))
        a_start = a_best

    # Round 2: Subgradient high-T (20% time)
    r2_time = min(0.20 * total_budget, deadline - time.monotonic() - 5)
    if r2_time > 15:
        print("\n[Round 2] Subgrad T=100-3000, top_k=30 (%.0fs)" % r2_time)
        t0 = time.monotonic()
        a_best, obj_best = run_optimization_round(
            a_start, primary_n, S, r2_time, strategy="sg_high", verbose=True)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  R2 done: obj=%.8f (time=%.0fs)" % (ver, time.monotonic() - t0))
        a_start = a_best if (np.isfinite(ver) and ver < global_best) else a_start

    # Round 3: Subgradient mid-T for smoothing (15% time)
    r3_time = min(0.15 * total_budget, deadline - time.monotonic() - 5)
    if r3_time > 15:
        print("\n[Round 3] Subgrad T=20-200, top_k=100 (%.0fs)" % r3_time)
        t0 = time.monotonic()
        a_best, obj_best = run_optimization_round(
            a_start, primary_n, S, r3_time, strategy="sg_mid", verbose=True)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  R3 done: obj=%.8f (time=%.0fs)" % (ver, time.monotonic() - t0))
        a_start = a_best

    # Round 4: Coordinate descent (10% time)
    r4_time = min(0.10 * total_budget, deadline - time.monotonic() - 5)
    if r4_time > 10 and global_a is not None:
        print("\n[Round 4] Coordinate Descent (%.0fs)" % r4_time)
        t0 = time.monotonic()
        a_best, _ = coordinate_descent(global_a, primary_n, S, r4_time, max_sweeps=80)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  R4 done: obj=%.8f (time=%.0fs)" % (ver, time.monotonic() - t0))

    # Round 5: Zero masking + re-optimize (5% time)
    r5_time = min(0.05 * total_budget, deadline - time.monotonic() - 5)
    if r5_time > 5 and global_a is not None:
        print("\n[Round 5] Zero Masking + Quick Opt (%.0fs)" % r5_time)
        a_z, _ = try_zero_masking(global_a, primary_n, S, n_candidates=40)
        a_z = normalize_nonneg(a_z, S)
        ver = evaluate_sequence(a_z.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_z.copy()
            print("  ZeroMask: obj=%.8f" % ver)
        # Quick subgrad on the result
        if r5_time > 15:
            a_best, _ = run_subgrad_nesterov(global_a, primary_n, S, r5_time * 0.7,
                                              T_init=200, T_final=2000,
                                              lr_init=0.03, top_k=20)
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  R5 opt: obj=%.8f" % ver)

    # Round 6: Random restarts (10% time)
    r6_time = min(0.10 * total_budget, deadline - time.monotonic() - 5)
    if r6_time > 20 and global_a is not None:
        print("\n[Round 6] Random Restarts (%.0fs)" % r6_time)
        n_restarts = max(2, int(r6_time / 30))
        per_restart = r6_time / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 5:
                break
            noise = 0.003 + 0.002 * r
            a_noisy = global_a + noise * rng.standard_normal(primary_n)
            a_noisy = normalize_nonneg(a_noisy, S)

            a_best, _ = run_subgrad_nesterov(
                a_noisy, primary_n, S, per_restart * 0.8,
                T_init=50, T_final=1500, lr_init=0.04, top_k=30)
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  Restart %d: obj=%.8f" % (r, ver))

    # Round 7: Final ultra-high-T polish (remaining time)
    r7_time = deadline - time.monotonic() - 3
    if r7_time > 10 and global_a is not None:
        print("\n[Round 7] Final Ultra-High-T Polish (%.0fs)" % r7_time)
        # Split between subgrad and CD
        sg_time = r7_time * 0.6
        cd_time = r7_time * 0.4

        a_best, _ = run_subgrad_nesterov(
            global_a, primary_n, S, sg_time,
            T_init=500, T_final=5000, lr_init=0.02,
            momentum=0.92, top_k=10)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  Final SG: obj=%.8f" % ver)

        if cd_time > 5:
            a_best, _ = coordinate_descent(global_a, primary_n, S, cd_time, max_sweeps=50)
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  Final CD: obj=%.8f" % ver)

    # Final verification
    if global_a is None:
        global_a = np.ones(256).tolist()

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.8f (verified=%.8f)" % (global_best, ver))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.8f" % ver)

    return final_a
