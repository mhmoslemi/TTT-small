```python
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def softmax_obj_grad(z, beta, S):
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp
    
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta
    
    w = exp_bc / sum_exp_bc
    # grad_a[j] = 2 * sum_k w[k] * a[j-k] (cross-correlation)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)
    
    return obj, grad_z

def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
```

```python
def adam_softmax_optimize(a_init, n, S, max_time, rng, beta_schedule=None):
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
    
    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()
    
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    
    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break
        
        # Adam state
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        lr = 0.01 * S / np.sqrt(n)  # initial learning rate
        
        n_iters = 500  # iterations per beta level
        
        for it in range(n_iters):
            if time.monotonic() - start > max_time - 5:
                break
            
            obj, grad = softmax_obj_grad(z, beta, S)
            t += 1
            
            # Adam update
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * grad**2
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            lr_t = lr / (1.0 + 0.01 * it)  # decay
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
            z = z - z.mean()  # re-center for stability
            
            # Convert to a and check
            if it % 20 == 0:
                z_shifted = z - z.max()
                exp_z = np.exp(z_shifted)
                a = S * exp_z / exp_z.sum()
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a * S / s
                score = fft_convolve_self(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        # Convert final z to a
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = S * exp_z / exp_z.sum()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * S / s
        score = fft_convolve_self(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score
```

Now for the Nesterov with Polyak step:

The Polyak step size is: `eta = (f(x) - f_lb) / ||g||^2`

where f_lb is a lower bound on the optimal value. A valid lower bound is the average autocorrelation: `S^2 / (2n-1)`.

```python
def nesterov_polyak(a_init, n, S, max_time, rng, n_steps=100000):
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    
    y = a.copy()
    t = 1.0
    f_lb = S**2 / (2*n - 1)  # lower bound
    f_lb_obj = 2.0 * n * f_lb / (S**2)  # in terms of the objective
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    
    for it in range(n_steps):
        if time.monotonic() - start > max_time:
            break
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = y.copy()
        
        # Subgradient: grad_a[j] = 2 * sum_{active k} a[j-k]
        # Use exact subgradient (all lags with weight proportional to their value)
        # Actually, the subgradient of max(c) at y is {2 * (y * e_k) : k in argmax(c)}
        # where e_k is the basis vector at position k
        # A valid subgradient is the average: 2 * (y * w) where w is uniform over argmax
        
        # For a better subgradient, use the softmax-weighted version
        beta_sub = 50.0
        c_shifted = c - c.max()
        w = np.exp(beta_sub * c_shifted)
        w = w / w.sum()
        
        # grad = 2 * cross_correlation(y, w)
        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        Y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Polyak step size
        obj_cur = 2.0 * n * cur_max / (S**2)
        obj_lb = 2.0 * n * f_lb / (S**2)
        gap = obj_cur - obj_lb
        if gap < 1e-12:
            break
        # In terms of the unnormalized objective (max of conv), the step is:
        # We want to reduce max(conv), so step = (cur_max - f_lb) / (||grad||^2 / (2*n))
        # Actually let's just use a scaled version
        step = (cur_max - f_lb) / (gnorm**2 / (2.0 * n))
        step = min(step, S / 10.0)  # cap the step
        
        x_new = project_simplex(y - step * grad / gnorm, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
```

```python
"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- Phase 1: Adam optimizer on softmax parameterization with log-sum-exp continuation
- Phase 2: Nesterov-accelerated subgradient with adaptive step size
- Phase 3: L-BFGS-B polishing with very high beta and perturbation restarts
- Phase 4: Cyclic coordinate descent with bisection on 1D subproblems
- Phase 5: Final high-precision L-BFGS
- Key differences from previous: Adam (per-coordinate adaptive LR), coordinate descent,
  more diverse initializations, better exploration schedule

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z (softmax params)."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def z_to_a(z, S):
    """Convert softmax params to normalized sequence."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = S * exp_z / exp_z.sum()
    return np.maximum(a, 0.0)


def a_to_z(a, S):
    """Convert sequence to softmax params."""
    a = np.clip(a, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        return np.zeros(len(a))
    a = a * S / s
    z = np.log(a / S + 1e-30)
    return z - z.mean()


def run_adam_continuation(a_init, n, S, max_time, rng,
                          beta_schedule=None, iters_per_beta=300):
    """Adam optimizer with log-sum-exp continuation on softmax params."""
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    z = a_to_z(a_init, S)
    start = time.monotonic()
    best_score = fft_convolve_self(a_init).max()
    best_a = a_init.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        m = np.zeros(n)
        v = np.zeros(n)
        t_count = 0
        base_lr = 0.05 / np.sqrt(n)

        for it in range(iters_per_beta):
            if time.monotonic() - start > max_time - 5:
                break

            obj, grad = softmax_obj_grad(z, beta, S)
            t_count += 1

            # Adam update
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * grad * grad
            m_hat = m / (1.0 - 0.9 ** t_count)
            v_hat = v / (1.0 - 0.999 ** t_count)
            lr_t = base_lr / (1.0 + 0.005 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
            z = z - z.mean()

            # Periodically check score
            if it % 30 == 0:
                a = z_to_a(z, S)
                c = fft_convolve_self(a)
                score = c.max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        # Convert final z to a
        a = z_to_a(z, S)
        c = fft_convolve_self(a)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def run_nesterov_subgrad(a_init, n, S, max_time, rng,
                         top_k_init=8, top_k_max=256, beta_soft=20.0):
    """Nesterov-accelerated subgradient with adaptive step size."""
    a = project_simplex(np.clip(a_init, 0.0, 1000.0), S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    f_lb = S * S / (2.0 * n - 1.0)

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Compute subgradient using top-k lags
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: Polyak-inspired
        gap = cur - f_lb
        if gap < 1e-10:
            break
        step_gap = gap / (gnorm * gnorm / (2.0 * n))
        step_dim = 2.0 * S / (np.sqrt(n) * np.sqrt(1.0 + 0.001 * it))
        step = min(step_gap, step_dim)
        step = max(step, 1e-6)

        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        y_new = y_new * (S / max(y_new.sum(), 1e-15))
        a = x_new
        y = y_new
        t_mom = t_new

        # Occasional random restart
        if it % 2000 == 0 and rng.random() < 0.05:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t_mom = 1.0

    return best_local, best_score


def run_lbfgs_polish(a_init, n, S, max_time, rng,
                     beta_schedule=None, maxiter_per_beta=200):
    """L-BFGS-B polishing with continuation."""
    if beta_schedule is None:
        beta_schedule = [200, 500, 1000, 2000, 5000]

    z = a_to_z(a_init, S)
    start = time.monotonic()
    best_score = fft_convolve_self(a_init).max()
    best_a = a_init.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        def fun(z_val):
            return softmax_obj_grad(z_val, beta, S)[0]

        def jac(z_val):
            return softmax_obj_grad(z_val, beta, S)[1]

        t0 = time.monotonic()
        try:
            result = scipy_minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_beta, 'maxfun': maxiter_per_beta * 2,
                         'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50, 'iprint': -1}
            )
        except Exception:
            break

        z = result.x - result.x.mean()
        a = z_to_a(z, S)
        c = fft_convolve_self(a)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if time.monotonic() - t0 > 1:
            print("    L-BFGS beta=%d: score=%.8f (%.1fs)" % (beta, best_score, time.monotonic() - t0))

    return best_a, best_score


def run_coordinate_descent(a_init, n, S, max_time, rng, n_passes_max=500):
    """Cyclic coordinate descent with bisection on 1D subproblems."""
    a = project_simplex(np.clip(a_init, 0.0, 1000.0), S).copy()
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    order = np.arange(n)

    for p in range(n_passes_max):
        if time.monotonic() - start > max_time - 10:
            break

        # Shuffle order for diversity
        if p % 5 == 0:
            order = rng.permutation(n)

        improved = False
        for j in order:
            if time.monotonic() - start > max_time - 10:
                break

            old_val = a[j]
            if old_val < 1e-15:
                continue

            # Compute c_m = a[m-j] for valid m (the "cross coefficients")
            # a[j] appears in (a*a)[m] as a[j] * a[m-j]
            # Valid m range: max(0, j) to min(2n-2, j+n-1)
            m_lo = j
            m_hi = min(2 * n - 1, j + n)
            c_m = np.zeros(2 * n - 1)
            c_m[m_lo:m_hi] = a[0:m_hi - m_lo]

            # R0[m] = r[m] - a[j] * c_m (autocorrelation without a[j]'s contribution)
            # We need current r; compute lazily
            # Actually, let's compute r once per pass for efficiency
            # For now, compute on the fly (costly but correct)
            r = fft_convolve_self(a)
            R0 = r - old_val * c_m

            # Bisection to find optimal x = a[j]
            # Minimize g(x) = max_m(R0[m] + x * c_m) for x >= 0
            x_lo = 0.0
            x_hi = old_val * 3.0 + S / n

            for _ in range(40):
                x_mid = (x_lo + x_hi) / 2.0
                vals = R0 + x_mid * c_m
                max_val = vals.max()
                active_mask = vals >= max_val - 1e-12
                c_active = c_m[active_mask]
                if c_active.size == 0:
                    break
                if c_active.min() > 1e-10:
                    x_hi = x_mid
                elif c_active.max() < -1e-10:
                    x_lo = x_mid
                else:
                    x_lo = x_mid
                    x_hi = x_mid
                    break

            x_opt = max(0.0, (x_lo + x_hi) / 2.0)

            # Compare with x=0
            g_0 = R0.max()
            g_opt = (R0 + x_opt * c_m).max()
            if g_0 <= g_opt:
                x_opt = 0.0

            # Apply update with re-normalization
            if abs(x_opt - old_val) > 1e-12:
                a[j] = x_opt
                a_sum = a.sum()
                if a_sum > 1e-15:
                    a = a * (S / a_sum)
                improved = True

        # Check score periodically
        if p % 10 == 0 or improved:
            c = fft_convolve_self(a)
            score = c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if score < best_score * 0.999:
                    print("    CD pass %d: score=%.8f" % (p, best_score))

    return best_a, best_score

        print("\n=== Phase 3: L-BFGS Polish + Perturb (%.1fs) ===" % phase3_time)

        n_perturb = max(3, int(phase3_time / 60))
        per_perturb = phase3_time / n_perturb

        # First: direct polish
        a_p3, score_p3 = run_lbfgs_polish(
            global_a, n, S, per_perturb, rng,
            beta_schedule=[500, 1000, 2000, 5000, 10000],
            maxiter_per_beta=300
        )
        a_p3 = safe_normalize(a_p3, n)
        c_p3 = fft_convolve_self(a_p3)
        if c_p3.max() < global_best:
            global_best = c_p3.max()
            global_a = a_p3.copy()
            print("  L-BFGS direct: %.8f" % global_best)

        # Perturbation restarts with varying scales
        for p in range(1, n_perturb):
            if time.monotonic() > deadline - 60:
                break

            noise_scale = 0.02 / (1.0 + 0.5 * p)
            a_pert = global_a + rng.normal(0, noise_scale * S / np.sqrt(n), n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = safe_normalize(a_pert, n)

            t_rem = min(per_perturb, deadline - time.monotonic() - 30)
            if t_rem < 10:
                break

            a_pp, score_pp = run_lbfgs_polish(
                a_pert, n, S, t_rem, rng,
                beta_schedule=[200, 500, 1000, 3000],
                maxiter_per_beta=200
            )
            a_pp = safe_normalize(a_pp, n)
            c_pp = fft_convolve_self(a_pp)
            if c_pp.max() < global_best:
                global_best = c_pp.max()
                global_a = a_pp.copy()
                print("  Perturb %d (scale=%.4f): %.8f" % (p, noise_scale, global_best))

    print("After Phase 3: %.8f" % global_best)

    # --- Phase 4: Coordinate Descent (20% of time) ---
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        phase4_time = min(remaining * 0.8, 300)
        print("\n=== Phase 4: Coordinate Descent (%.1fs) ===" % phase4_time)

        a_cd, score_cd = run_coordinate_descent(
            global_a, n, S, phase4_time, rng, n_passes_max=200
        )
        a_cd = safe_normalize(a_cd, n)
        c_cd = fft_convolve_self(a_cd)
        if c_cd.max() < global_best:
            global_best = c_cd.max()
            global_a = a_cd.copy()
            print("  CD: %.8f" % global_best)
        else:
            print("  CD: %.8f (no improvement)" % c_cd.max())

    print("After Phase 4: %.8f" % global_best)

    # --- Phase 5: Final polish ---
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        print("\n=== Phase 5: Final Polish (%.1fs) ===" % remaining)

        a_final, score_final = run_lbfgs_polish(
            global_a, n, S, remaining * 0.6, rng,
            beta_schedule=[2000, 5000, 10000, 20000],
            maxiter_per_beta=300
        )
        a_final = safe_normalize(a_final, n)
        c_final = fft_convolve_self(a_final)
        if c_final.max() < global_best:
            global_best = c_final.max()
            global_a = a_final.copy()
            print("  Final L-BFGS: %.8f" % global_best)

        # One more CD pass
        t_rem = deadline - time.monotonic() - 10
        if t_rem > 15:
            a_cd2, score_cd2 = run_coordinate_descent(
                global_a, n, S, t_rem, rng, n_passes_max=100
            )
            a_cd2 = safe_normalize(a_cd2, n)
            c_cd2 = fft_convolve_self(a_cd2)
            if c_cd2.max() < global_best:
                global_best = c_cd2.max()
                global_a = a_cd2.copy()
                print("  Final CD: %.8f" % global_best)

    # --- Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.6f" % ver if np.isfinite(ver) else "INF"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * np.sqrt(512.0) / 256
    else:
        global_a = np.ones(256, dtype=np.float64) * np.sqrt(512.0) / 256

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * np.sqrt(512.0) / 256

    elapsed = time.monotonic() - (deadline - budget_s + 20)
    print("Returning sequence of length %d, internal score=%.8f" % (len(global_a), global_best))
    print("Time used: %.1fs" % elapsed)
    return [float(x) for x in global_a]
```
