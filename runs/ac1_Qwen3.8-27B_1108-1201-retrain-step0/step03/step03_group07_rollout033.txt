"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive-temperature mirror descent
with spectral flattening, structured perturbations, and multi-candidate tracking.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def compute_smooth_grad(c, a, epsilon):
    """Compute gradient of logsumexp(c/epsilon) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / epsilon)
    weights = exp_c / exp_c.sum()
    
    # grad[m] = (2/epsilon) * sum_k weights[k] * a[k-m] (for valid k-m)
    # This is a convolution of weights with reversed a
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(weights, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = (2.0 / epsilon) * full_conv[n-1:2*n-1]
    return grad

def spectral_flatten(a, alpha=0.1):
    """Apply a gentle spectrum flattening step."""
    n = len(a)
    fsize = 1 << (2*n).bit_length()
    A = fft(a, fsize)
    P = A * np.conj(A)  # Power spectrum
    P = np.maximum(P, 1e-10)
    
    # Target: flat spectrum
    target = P.mean()
    
    # Gentle correction
    correction = np.power(target / P, alpha)
    A_new = A * correction
    
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.maximum(a_new, 0.0)
    S = a.sum()
    if a_new.sum() > 1e-10:
        a_new = a_new * (S / a_new.sum())
    else:
        a_new = a.copy()
    return a_new

def mirror_descent_v2(a0, time_budget, eta0=0.3, momentum=0.9,
                      eps_start=1.0, eps_end=0.05, eps_decay=None,
                      perturb_every=500, perturb_frac=0.1,
                      flat_every=200, flat_alpha=0.05,
                      stagnation_limit=1000, verbose=False):
    """Mirror descent with adaptive temperature and spectral flattening."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    best_iter = 0
    
    v = np.zeros(n)
    eps = eps_start
    if eps_decay is None:
        eps_decay = (eps_end / eps_start) ** (1.0 / max(1, int(time_budget * 50)))
    
    iter_no = 0
    start = time.monotonic()
    recent_objs = []
    
    while time.monotonic() - start < time_budget:
        # Compute smooth gradient
        grad = compute_smooth_grad(c, a, eps)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm
        
        # Mirror descent step
        eta = eta0 * (0.9995 ** (iter_no % 10000))
        a_trial = a * np.exp(-eta * v)
        a_trial = np.maximum(a_trial, 1e-300)
        a_trial = a_trial * (S / a_trial.sum())
        
        c_trial = fft_convolve_self(a_trial)
        obj_trial = obj_value(a_trial, c_trial)
        
        if obj_trial < best_obj * 1.001:
            a = a_trial
            c = c_trial
        else:
            # Try smaller step
            a_trial = a * np.exp(-eta * 0.5 * v)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)
            if obj_trial < obj_value(a, c):
                a = a_trial
                c = c_trial
        
        cur_obj = obj_value(a, c)
        recent_objs.append(cur_obj)
        if len(recent_objs) > 50:
            recent_objs.pop(0)
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
        
        # Spectral flattening periodically
        if flat_every > 0 and iter_no % flat_every == 0 and iter_no > 0:
            a_flat = spectral_flatten(a, flat_alpha)
            c_flat = fft_convolve_self(a_flat)
            obj_flat = obj_value(a_flat, c_flat)
            if obj_flat < cur_obj:
                a = a_flat
                c = c_flat
        
        # Decay temperature
        eps *= eps_decay
        
        # Stagnation detection
        if iter_no - best_iter > stagnation_limit:
            # Structured perturbation
            noise_scale = perturb_frac * S / np.sqrt(n)
            noise = np.random.randn(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.5
            best_iter = iter_no
            if verbose:
                print(f"    [perturb] iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}")
        
        iter_no += 1
        if verbose and iter_no % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, eps={eps:.4f}, t={elapsed:.1f}s")
    
    return best_a, best_obj

def coordinate_refine(a0, S, time_budget):
    """Coordinate descent: try reducing individual coordinates that contribute to peaks."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        c_max = c.max()
        k_max = np.argmax(c)
        
        # Find coordinates that contribute most to the peak
        # c[k_max] = sum_j a[j] * a[k_max - j]
        # grad_j of c[k_max] = 2 * a[k_max - j] (for valid indices)
        
        contributing = np.zeros(n)
        for j in range(n):
            k = k_max - j
            if 0 <= k < n:
                contributing[j] = a[j] * a[k]
        
        # Sort by contribution and try reducing the top contributors
        order = np.argsort(contributing)[::-1]
        improved = False
        
        for idx in order[:50]:
            if time.monotonic() - start > time_budget:
                break
            orig = a[idx]
            if orig < 1e-10:
                continue
            for factor in [0.5, 0.7, 0.85]:
                a[idx] = orig * factor
                c_new = fft_convolve_self(a)
                obj_new = c_new.max()  # Since S is fixed, this is proportional to objective
                if obj_new < c_max:
                    improved = True
                    c = c_new
                    # Check if overall improvement
                    S_new = a.sum()
                    a = a * (S / S_new)
                    c = fft_convolve_self(a)
                    obj_check = obj_value(a, c)
                    if obj_check < best_obj:
                        best_obj = obj_check
                        best_a = a.copy()
                    break
            else:
                a[idx] = orig  # Restore if no factor helped
        
        if not improved:
            # Try increasing some coordinates
            for idx in order[-20:]:
                if time.monotonic() - start > time_budget:
                    break
                orig = a[idx]
                for factor in [1.3, 1.5]:
                    a[idx] = orig * factor
                    c_new = fft_convolve_self(a)
                    obj_new = c_new.max()
                    if obj_new < c_max:
                        improved = True
                        c = c_new
                        S_new = a.sum()
                        a = a * (S / S_new)
                        c = fft_convolve_self(a)
                        obj_check = obj_value(a, c)
                        if obj_check < best_obj:
                            best_obj = obj_check
                            best_a = a.copy()
                        break
                else:
                    a[idx] = orig
            if not improved:
                break
    
    return best_a, best_obj

Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive-temperature mirror descent
with AdaGrad step sizes, spectral flattening, multi-perturbation restarts,
and coordinate descent polishing. Multi-restart strategy from diverse initializations.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_val(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_conv_self(a)
    return 2.0 * n * c.max() / (S * S)


def smooth_grad(c, a, eps):
    """Gradient of logsumexp(c/eps) w.r.t. a (times 2)."""
    n = len(a)
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / eps)
    w = exp_c / exp_c.sum()
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full = np.real(ifft(W * R))
    grad = (2.0 / eps) * full[n - 1:2 * n - 1]
    return grad

    n = len(a)

    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full = np.real(ifft(W * R))
    return 2.0 * full[n - 1:2 * n - 1]


def spectral_flatten(a, alpha=0.05):
    """Gentle spectrum flattening in frequency domain."""
    n = len(a)
    fsize = 1 << (2 * n).bit_length()
    A = fft(a, fsize)
    P = np.abs(A) ** 2
    P = np.maximum(P, 1e-15)
    target = P[P > 0].mean() if np.any(P > 0) else 1.0
    corr = np.power(target / P, alpha)
    A_new = A * corr
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.maximum(a_new, 0.0)
    S = a.sum()
    s_new = a_new.sum()
    if s_new > 1e-10:
        a_new = a_new * (S / s_new)
    else:
        a_new = a.copy()
    return a_new


def perturb_gaussian(a, S, frac=0.05):
    """Gaussian perturbation."""
    n = len(a)
    noise = np.random.randn(n) * frac * S / np.sqrt(n)
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new * (S / s)
    else:
        a_new = a.copy()
    return a_new


def perturb_shift(a, S):
    """Random circular shift (changes phase, preserves spectrum)."""
    n = len(a)
    shift = np.random.randint(1, n)
    return np.roll(a, shift)


def perturb_modulate(a, S):
    """Random slow amplitude modulation."""
    n = len(a)
    i = np.arange(n)
    freq = np.random.uniform(0.5, 3.0)
    phase = np.random.uniform(0, 2 * np.pi)
    mod = 1.0 + 0.3 * np.cos(2 * np.pi * freq * i / n + phase)
    a_new = a * mod
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new * (S / s)
    else:
        a_new = a.copy()
    return a_new


def mirror_descent_adv(a0, time_budget, eta0=0.3, momentum=0.9,
                       eps_start=2.0, eps_end=0.05,
                       perturb_every=600, perturb_frac=0.06,
                       flat_every=200, flat_alpha=0.04,
                       mode='auto', verbose=False):
    """
    Advanced mirror descent with AdaGrad, adaptive temperature,
    momentum, spectral flattening, and multi-type perturbations.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    a = a * (S / a.sum())

    c = fft_conv_self(a)
    best_obj = obj_val(a, c)
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)
    adagrad_state = np.ones(n) * 1e-8
    eps = eps_start
    total_iters = max(1, int(time_budget * 80))
    eps_decay = (eps_end / eps_start) ** (1.0 / total_iters)

    iter_no = 0
    start = time.monotonic()
    perturb_type = 0

    while time.monotonic() - start < time_budget:
        # Choose gradient type based on mode and iteration
        if mode == 'smooth':
            grad = smooth_grad(c, a, eps)
        elif mode == 'topk':
            k = max(1, int(50 * (1.0 - iter_no / total_iters)) + 1)
            grad = topk_grad(c, a, k)
        else:  # auto: switch from smooth to topk
            if iter_no < total_iters // 3:
                grad = smooth_grad(c, a, eps)
            elif iter_no < 2 * total_iters // 3:
                k = max(1, int(20 * (1.0 - iter_no / total_iters)) + 1)
                grad = topk_grad(c, a, k)
            else:
                grad = topk_grad(c, a, 1)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # AdaGrad update
        adagrad_state += grad ** 2
        step_scale = 1.0 / np.sqrt(adagrad_state)

        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        # Adaptive eta
        eta = eta0 * (0.9998 ** (iter_no % 20000))
        eta = min(eta, 2.0)

        # Multiplicative update with AdaGrad scaling
        a_trial = a * np.exp(-eta * v * step_scale)
        a_trial = np.maximum(a_trial, 1e-300)
        s_trial = a_trial.sum()
        if s_trial < 1e-10:
            a_trial = a.copy()
        else:
            a_trial = a_trial * (S / s_trial)

        c_trial = fft_conv_self(a_trial)
        obj_trial = obj_val(a_trial, c_trial)

        # Accept or backtrack
        if obj_trial <= obj_val(a, c) * 1.0005:
            a = a_trial
            c = c_trial
        else:
            # Try smaller step
            a_trial = a * np.exp(-eta * 0.3 * v * step_scale)
            a_trial = np.maximum(a_trial, 1e-300)
            s_trial = a_trial.sum()
            if s_trial > 1e-10:
                a_trial = a_trial * (S / s_trial)
                c_trial = fft_conv_self(a_trial)
                obj_trial = obj_val(a_trial, c_trial)
                if obj_trial < obj_val(a, c):
                    a = a_trial
                    c = c_trial

        cur_obj = obj_val(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no

        # Spectral flattening
        if flat_every > 0 and iter_no % flat_every == 0 and iter_no > 0:
            a_flat = spectral_flatten(a, flat_alpha)
            c_flat = fft_conv_self(a_flat)
            obj_flat = obj_val(a_flat, c_flat)
            if obj_flat < cur_obj:
                a = a_flat
                c = c_flat

        # Decay temperature
        eps *= eps_decay
        eps = max(eps, 0.01)

        # Perturbation when stuck
        if iter_no - best_iter > perturb_every:
            perturb_type = (perturb_type + 1) % 4
            if perturb_type == 0:
                a = perturb_gaussian(a, S, perturb_frac)
            elif perturb_type == 1:
                a = perturb_gaussian(a, S, perturb_frac * 0.5)
                a = spectral_flatten(a, 0.1)
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-10:
                    a = a * (S / s)
            elif perturb_type == 2:
                a = perturb_modulate(a, S)
            else:
                # Larger perturbation
                a = perturb_gaussian(a, S, perturb_frac * 2.0)

            c = fft_conv_self(a)
            v *= 0.3
            adagrad_state = np.ones(n) * 1e-8
            best_iter = iter_no
            if verbose:
                elapsed = time.monotonic() - start
                print(f"      [perturb #{perturb_type}] iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, t={elapsed:.1f}s")

        iter_no += 1
        if verbose and iter_no % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"      iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, eps={eps:.4f}, t={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_descent(a0, S, time_budget, max_rounds_per_coord=3):
    """
    Efficient coordinate descent using incremental convolution updates.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fft_conv_self(a)
    best_obj = obj_val(a, c)
    best_a = a.copy()
    start = time.monotonic()

    # Precompute which c indices are affected by each coordinate
    # Changing a[i] affects c[k] for k in [i, i+n-1] (clipped to [0, 2n-2])

    while time.monotonic() - start < time_budget:
        c_max = c.max()
        k_max = int(np.argmax(c))
        improved = False

        # Find the coordinate that most contributes to the peak
        j_lo = max(0, k_max - n + 1)
        j_hi = min(n - 1, k_max)

        # Compute contributions
        best_j = -1
        best_contrib = -1.0
        for j in range(j_lo, j_hi + 1):
            contrib = a[j] * a[k_max - j]
            if contrib > best_contrib:
                best_contrib = contrib
                best_j = j

        if best_j < 0:
            break

        orig = a[best_j]
        if orig < 1e-12:
            break

        # Try reducing this coordinate
        found_improvement = False
        for frac in [0.5, 0.65, 0.8, 0.9]:
            delta = -(1.0 - frac) * orig
            # Quick check: will c[k_max] decrease?
            new_ckmax = c[k_max] + 2.0 * delta * a[k_max - best_j]
            if k_max == 2 * best_j:
                new_ckmax += delta * delta
            if new_ckmax < c_max - 1e-10:
                # Apply update
                a[best_j] = orig + delta
                # Incrementally update c
                for k in range(max(0, best_j), min(2 * n - 1, best_j + n)):
                    koff = k - best_j
                    if 0 <= koff < n:
                        c[k] += 2.0 * delta * a[k - best_j]
                    if k == 2 * best_j:
                        c[k] += delta * delta
                found_improvement = True
                improved = True
                break

        if not found_improvement:
            # Try a few random coordinates
            for _ in range(5):
                j_rand = np.random.randint(0, n)
                orig_r = a[j_rand]
                if orig_r < 1e-12:
                    continue
                for frac in [0.6, 0.8]:
                    delta = -(1.0 - frac) * orig_r
                    new_ckmax = c[k_max] + 2.0 * delta * a[k_max - j_rand]
                    if k_max == 2 * j_rand:
                        new_ckmax += delta * delta
                    if new_ckmax < c_max - 1e-10:
                        a[j_rand] = orig_r + delta
                        for k in range(max(0, j_rand), min(2 * n - 1, j_rand + n)):
                            koff = k - j_rand
                            if 0 <= koff < n:
                                c[k] += 2.0 * delta * a[k - j_rand]
                            if k == 2 * j_rand:
                                c[k] += delta * delta
                        improved = True
                        break

        if improved:
            # Renormalize periodically (every 100 improvements changes sum significantly)
            S_now = a.sum()
            if abs(S_now - S) / S > 0.01:
                a = a * (S / S_now)
                c = fft_conv_self(a)

            cur_obj = 2.0 * n * c.max() / (S * S)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            # Stuck: apply a perturbation and break out
            if np.random.rand() < 0.3:
                j_p = np.random.randint(0, n)
                scale = 0.05 * S / np.sqrt(n)
                delta_p = np.random.randn() * scale
                old_val = a[j_p]
                new_val = max(0.0, old_val + delta_p)
                actual_delta = new_val - old_val
                a[j_p] = new_val
                for k in range(max(0, j_p), min(2 * n - 1, j_p + n)):
                    koff = k - j_p
                    if 0 <= koff < n:
                        c[k] += 2.0 * actual_delta * a[k - j_p]
                    if k == 2 * j_p:
                        c[k] += actual_delta * actual_delta
            else:
                break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses multi-phase strategy: exploration -> main optimization -> polishing.
    """
    np.random.seed(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    global_best = np.inf
    global_a = None

    # --- Load given sequence ---
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw.copy()
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print(f"Given sequence: n={n_given}, obj={ver:.6f}")
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print(f"Could not load given sequence: {e}")

    # --- Phase 1: Quick exploration (8% time) ---
    rem = deadline - time.monotonic()
    explore_time = 0.08 * rem
    print(f"\nPhase 1: Exploration ({explore_time:.0f}s)")

    candidates = []
    if given_a is not None:
        candidates.append(('given', given_a))

    n_use = 4096
    S_use = np.sqrt(2.0 * n_use)
    i = np.arange(n_use)

    # Various initializations
    for freq, amp in [(5, 0.3), (7, 0.4), (9, 0.35), (13, 0.25)]:
        a_c = 1.0 + amp * np.cos(2 * np.pi * freq * i / n_use)
        a_c = np.maximum(a_c, 0.01)
        a_c *= S_use / a_c.sum()
        candidates.append((f'cos{freq}_{amp}', a_c))

    # Two bumps
    a_2b = np.exp(-0.5 * ((i - n_use * 0.3) / (n_use * 0.12)) ** 2) + \
           np.exp(-0.5 * ((i - n_use * 0.7) / (n_use * 0.12)) ** 2)
    a_2b = np.maximum(a_2b, 0.01)
    a_2b *= S_use / a_2b.sum()
    candidates.append(('2bump', a_2b))

    # Four bumps
    a_4b = sum(np.exp(-0.5 * ((i - n_use * f) / (n_use * 0.07)) ** 2)
               for f in [0.1, 0.35, 0.6, 0.85])
    a_4b = np.maximum(a_4b, 0.01)
    a_4b *= S_use / a_4b.sum()
    candidates.append(('4bump', a_4b))

    # Smoothed random
    a_r = np.random.random(n_use)
    fsize = 1 << (2 * n_use).bit_length()
    Ar = fft(a_r, fsize)
    # Lowpass filter
    freqs = np.fft.fftfreq(fsize, d=1.0 / fsize)
    lowpass = np.exp(-freqs ** 2 / (2.0 * 50.0 ** 2))
    a_s = np.real(ifft(Ar * lowpass, fsize))[:n_use]
    a_s = np.maximum(a_s, 0.01)
    a_s *= S_use / a_s.sum()
    candidates.append(('sm_rand', a_s))

    per_time = explore_time / len(candidates)
    print(f"  Testing {len(candidates)} candidates, {per_time:.1f}s each")

    for name, a_cand in candidates:
        if time.monotonic() > deadline - 180:
            break
        t0 = time.monotonic()
        a_best, obj_best = mirror_descent_adv(
            a_cand, per_time,
            eta0=0.3, momentum=0.85,
            eps_start=3.0, eps_end=0.5,
            perturb_every=200, perturb_frac=0.05,
            flat_every=100, flat_alpha=0.03,
            mode='smooth', verbose=False
        )
        elapsed = time.monotonic() - t0
        print(f"  {name}: obj={obj_best:.6f} ({elapsed:.1f}s)")
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    ** New best: {obj_best:.6f}")

    # --- Phase 2: Main optimization (67% time) ---
    rem = deadline - time.monotonic()
    main_time = 0.67 * rem
    print(f"\nPhase 2: Main optimization ({main_time:.0f}s)")

    if global_a is not None and main_time > 60:
        n = len(global_a)
        S = global_a.sum()
        quarter = main_time / 4.0

        # 2a: Smooth objective, broad exploration
        print(f"  2a: Smooth/broad ({quarter:.0f}s)")
        a_best, obj_best = mirror_descent_adv(
            global_a, quarter,
            eta0=0.35, momentum=0.85,
            eps_start=5.0, eps_end=1.0,
            perturb_every=400, perturb_frac=0.08,
            flat_every=150, flat_alpha=0.05,
            mode='smooth', verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

        # 2b: Transition, medium temperature
        print(f"  2b: Medium ({quarter:.0f}s)")
        a_best, obj_best = mirror_descent_adv(
            global_a, quarter,
            eta0=0.25, momentum=0.9,
            eps_start=1.5, eps_end=0.3,
            perturb_every=600, perturb_frac=0.04,
            flat_every=200, flat_alpha=0.03,
            mode='auto', verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

        # 2c: Sharp, top-k=5
        print(f"  2c: Sharp/topk ({quarter:.0f}s)")
        a_best, obj_best = mirror_descent_adv(
            global_a, quarter,
            eta0=0.15, momentum=0.92,
            eps_start=0.5, eps_end=0.1,
            perturb_every=800, perturb_frac=0.03,
            flat_every=300, flat_alpha=0.02,
            mode='topk', verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

        # 2d: Very sharp, single peak focus
        print(f"  2d: Single peak ({quarter:.0f}s)")
        a_best, obj_best = mirror_descent_adv(
            global_a, quarter,
            eta0=0.1, momentum=0.95,
            eps_start=0.15, eps_end=0.03,
            perturb_every=1200, perturb_frac=0.02,
            flat_every=0, flat_alpha=0.0,
            mode='topk', verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

    # --- Phase 3: Polishing (18% time) ---
    rem = deadline - time.monotonic()
    polish_time = 0.18 * rem
    print(f"\nPhase 3: Polishing ({polish_time:.0f}s)")

    if global_a is not None and polish_time > 10:
        n = len(global_a)
        S = global_a.sum()

        # 3a: Coordinate descent
        half1 = polish_time * 0.55
        print(f"  3a: Coordinate descent ({half1:.0f}s)")
        a_best, obj_best = coordinate_descent(global_a, S, half1)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

        # 3b: Fine mirror descent
        half2 = polish_time * 0.45
        print(f"  3b: Fine mirror descent ({half2:.0f}s)")
        a_best, obj_best = mirror_descent_adv(
            global_a, half2,
            eta0=0.05, momentum=0.95,
            eps_start=0.08, eps_end=0.02,
            perturb_every=500, perturb_frac=0.015,
            flat_every=0, flat_alpha=0.0,
            mode='topk', verbose=False
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

    # --- Phase 4: Final micro-polish (7% time) ---
    rem = deadline - time.monotonic()
    final_time = 0.07 * rem
    print(f"\nPhase 4: Final micro-polish ({final_time:.0f}s)")

    if global_a is not None and final_time > 5:
        n = len(global_a)
        S = global_a.sum()

        # One more coordinate descent pass
        a_best, obj_best = coordinate_descent(global_a, S, final_time)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

    # --- Final verification ---
    print(f"\n=== FINAL RESULTS ===")
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"Local best: {global_best:.6f}")
    print(f"Verified:   {ver:.6f}")
    print(f"Length:     {len(final_a)}")

    if not np.isfinite(ver):
        print("WARNING: Verification failed, using fallback")
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback obj: {ver:.6f}")

    return final_a
