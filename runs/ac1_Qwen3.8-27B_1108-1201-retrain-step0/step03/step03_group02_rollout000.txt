import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):

    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_soft_grad(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_toppk_grad(a, c, top_k):

    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]

    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = fft_soft_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 1.0 / (beta * 2.0 * (1.0 + 0.001 * it))
        step = max(step, 1e-8)

        x_new = project_simplex(y - step * grad, S)

        # Momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic restart with noise to escape local minima
        if it % restart_interval == 0:
            if rng.random() < 0.3:
                noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
                y = project_simplex(y + noise, S)
                t = 1.0

    return best_local, best_score


def run_peak_polish(a_init, n, S, max_time, rng, top_k_start=32, top_k_max=512,
                    noise_scale=0.001, use_backtrack=True):
    """
    Peak-polishing method: iteratively reduce top-k peaks with adaptive step.
    Uses backtracking to ensure monotonic improvement.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Gradually increase number of tracked peaks
        k = min(top_k_max, top_k_start + it // 20)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Use weighted top-k gradient (weights proportional to peak values)
        grad = fft_toppk_grad(y, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: scale with S and decrease over time
        base_step = 2.0 * S / (n * (1.0 + 0.002 * it))
        step = base_step

        # Try the step with backtracking
        best_x = None
        best_val = cur
        for trial in range(5):
            trial_step = step * (0.7 ** trial)
            x_new = project_simplex(y - trial_step * grad / max(gnorm, 1e-15), S)
            c_new = fft_convolve_self(x_new)
            val_new = c_new.max()
            if val_new < best_val:
                best_val = val_new
                best_x = x_new
            if val_new < cur * 0.999:
                break
            step = trial_step

        if best_x is not None:
            x_new = best_x
        else:
            # Fallback: small step
            x_new = project_simplex(y - 0.1 * step * grad / max(gnorm, 1e-15), S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic noise injection
        if it % 3000 == 0 and rng.random() < 0.1:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0

    return best_local, best_score


def run_focused_polish(a_init, n, S, max_time, rng, top_k=256):
    """
    Very focused polishing: track many peaks, use uniform weighting,
    very small steps, no momentum (pure gradient descent).
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = a.sum()  # placeholder
    best_local = a.copy()
    it = 0

    # Compute initial
    c = fft_convolve_self(a)
    best_score = c.max()

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Use uniform top-k gradient
        grad = fft_toppk_grad_uniform(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Very small step
        step = S / (n * 50.0 * (1.0 + 0.0005 * it))
        x_new = project_simplex(a - step * grad / max(gnorm, 1e-15), S)

        # Verify improvement
        c_new = fft_convolve_self(x_new)
        if c_new.max() > cur:
            # Backtrack
            x_new = project_simplex(0.5 * a + 0.5 * x_new, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() > cur:
                a = 0.99 * a + 0.01 * x_new
                a = project_simplex(a, S)
                continue

        a = x_new

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def generate_initializations(n, rng):
    """Generate diverse asymmetric initializations of length n."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Try parent sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= n:
            inits.append(("parent", np.clip(given[:n], 0, 1000)))
        elif given.size >= 64:
            # Resample to length n
            idx = np.linspace(0, len(given) - 1, n).astype(int)
            inits.append(("parent_resample", np.clip(given[idx], 0, 1000)))
    except Exception:
        pass

    # Exponential ramps
    for alpha in [2.0, 3.0, 5.0, 8.0]:
        inits.append(("exp_fwd_%d" % alpha, np.exp(alpha * tt)))
        inits.append(("exp_bwd_%d" % alpha, np.exp(alpha * (1.0 - tt))))

    # Power-law ramps
    for p in [0.5, 1.0, 2.0, 3.0]:
        inits.append(("pow_fwd_%s" % p, (tt + 0.01) ** p))
        inits.append(("pow_bwd_%s" % p, (1.0 - tt + 0.01) ** p))

    # Smooth asymmetric (Gaussian offset from center)
    center = 0.3
    inits.append(("gauss_off", np.exp(-((tt - center) / 0.2) ** 2)))
    center = 0.7
    inits.append(("gauss_off2", np.exp(-((tt - center) / 0.2) ** 2)))

    # Sinusoidal with trend
    inits.append(("sin_trend", 1.0 + 0.5 * np.sin(2 * np.pi * 5 * tt) + tt))
    inits.append(("sin_trend2", 1.0 + 0.5 * np.cos(2 * np.pi * 3 * tt) + 0.5 * tt))

    # Linear + quadratic
    inits.append(("poly2", tt + tt * tt + 0.1))
    inits.append(("poly2_rev", (1 - tt) + (1 - tt) * (1 - tt) + 0.1))

    # Random with trend
    rand_a = rng.random(n)
    ramp = np.linspace(0.3, 2.0, n)
    inits.append(("rand_trend", rand_a * ramp))

    # Chirp-like (frequency increases)
    freq = np.linspace(1, 10, n)
    inits.append(("chirp", 1.0 + 0.3 * np.sin(2 * np.pi * np.cumsum(freq) / n)))

    return inits


def run_optimization_pipeline(a_init, n, S, total_time, rng):
    """Run the full optimization pipeline on a single initialization."""
    start = time.monotonic()
    best_a = a_init.copy()
    best_score = np.inf

    # Phase 1: FISTA (40% of time)
    t1 = total_time * 0.4
    a1, s1 = run_fista_adaptive(a_init, n, S, t1, rng,
                                 beta_init=5.0, beta_max=300.0,
                                 noise_scale=0.003, restart_interval=3000)
    a1 = safe_normalize(a1, n)
    c1 = fft_convolve_self(a1)
    score1 = c1.max()
    if score1 < best_score:
        best_score = score1
        best_a = a1.copy()

    elapsed = time.monotonic() - start
    remaining = total_time - elapsed

    # Phase 2: Peak polish (35% of time)
    if remaining > 5:
        t2 = remaining * 0.35
        a2, s2 = run_peak_polish(best_a, n, S, t2, rng,
                                  top_k_start=16, top_k_max=256,
                                  noise_scale=0.001)
        a2 = safe_normalize(a2, n)
        c2 = fft_convolve_self(a2)
        score2 = c2.max()
        if score2 < best_score:
            best_score = score2
            best_a = a2.copy()

    elapsed = time.monotonic() - start
    remaining = total_time - elapsed

    # Phase 3: Focused polish (25% of time)
    if remaining > 5:
        t3 = remaining * 0.75
        a3, s3 = run_focused_polish(best_a, n, S, t3, rng, top_k=256)
        a3 = safe_normalize(a3, n)
        c3 = fft_convolve_self(a3)
        score3 = c3.max()
        if score3 < best_score:
            best_score = score3
            best_a = a3.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses asymmetric initializations, adaptive FISTA, and peak-polishing optimization.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Determine which n values to try
    # Start with parent length if available, otherwise 8192
    n_parent = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            n_parent = len(given)
    except Exception:
        pass

    # Try n=8192 primarily, and n=16384 if time allows
    n_candidates = [8192]
    if n_parent != 8192 and n_parent >= 1024:
        n_candidates.insert(0, n_parent)
    if n_parent != 16384:
        n_candidates.append(16384)

    # Remove duplicates, keep unique
    n_candidates = list(dict.fromkeys(n_candidates))

    print("Trying n values: %s" % n_candidates)
    print("Parent n: %d" % n_parent)

    # Phase 1: Quick exploration with parent/best init on primary n
    n_primary = n_candidates[0]  # Most likely the parent length
    S_primary = np.sqrt(2.0 * float(n_primary))

    # Get the best starting point
    a_start = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if len(given) == n_primary:
            a_start = np.clip(given, 0, 1000).astype(np.float64)
        elif len(given) >= 1024:
            # Resample
            idx = np.linspace(0, len(given) - 1, n_primary).astype(int)
            a_start = np.clip(given[idx], 0, 1000).astype(np.float64)
    except Exception:
        pass

    if a_start is None:
        # Generate a good asymmetric start
        tt = np.linspace(0, 1, n_primary, endpoint=False)
        a_start = np.exp(3.0 * tt)

    a_start = safe_normalize(a_start, n_primary)
    c_start = fft_convolve_self(a_start)
    print("Starting score (n=%d): %.6f" % (n_primary, c_start.max()))
    global_best = c_start.max()
    global_a = a_start.copy()

    # Phase 1: Main optimization on primary n (60% of budget)
    t_main = (deadline - time.monotonic()) * 0.55
    print("=== Phase 1: Main optimization (n=%d, %.1f sec) ===" % (n_primary, t_main))

    a_opt, score_opt = run_optimization_pipeline(a_start, n_primary, S_primary, t_main, rng)
    a_opt = safe_normalize(a_opt, n_primary)
    c_opt = fft_convolve_self(a_opt)
    if c_opt.max() < global_best:
        global_best = c_opt.max()
        global_a = a_opt.copy()
        print("  Phase 1 best: %.6f" % global_best)

    # Phase 2: Try a few additional initializations (20% of budget)
    elapsed = time.monotonic() - (deadline - (deadline - time.monotonic()))
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_explore = remaining * 0.2
        print("=== Phase 2: Exploration (%.1f sec) ===" % t_explore)

        inits = generate_initializations(n_primary, rng)
        # Skip parent (already tried)
        inits = [x for x in inits if x[0] != "parent" and x[0] != "parent_resample"]

        if inits:
            per_init = t_explore / len(inits)
            for name, arr in inits:
                if time.monotonic() > deadline - 60:
                    break
                S_i = np.sqrt(2.0 * float(n_primary))
                arr = safe_normalize(arr, n_primary)

                a_i, s_i = run_optimization_pipeline(arr, n_primary, S_i, per_init, rng)
                a_i = safe_normalize(a_i, n_primary)
                c_i = fft_convolve_self(a_i)
                if c_i.max() < global_best:
                    global_best = c_i.max()
                    global_a = a_i.copy()
                    print("  [%s] n=%d: obj=%.6f" % (name, n_primary, c_i.max()))

    # Phase 3: Try different n if time allows (15% of budget)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_other = remaining * 0.2
        print("=== Phase 3: Alternative n (%.1f sec) ===" % t_other)

        for n_try in n_candidates:
            if n_try == n_primary:
                continue
            if time.monotonic() > deadline - 60:
                break
            S_try = np.sqrt(2.0 * float(n_try))
            # Resample global best to new length
            if global_a is not None:
                a_try = np.interp(np.linspace(0, 1, n_try),
                                   np.linspace(0, 1, len(global_a)),
                                   global_a)
                a_try = safe_normalize(a_try, n_try)
            else:
                tt = np.linspace(0, 1, n_try, endpoint=False)
                a_try = np.exp(3.0 * tt)
                a_try = safe_normalize(a_try, n_try)

            a_tr, s_tr = run_optimization_pipeline(a_try, n_try, S_try, t_other, rng)
            a_tr = safe_normalize(a_tr, n_try)
            c_tr = fft_convolve_self(a_tr)
            if c_tr.max() < global_best:
                global_best = c_tr.max()
                global_a = a_tr.copy()
                print("  [n=%d] obj=%.6f" % (n_try, c_tr.max()))

    # Phase 4: Final polishing (remaining time minus safety margin)
    remaining = deadline - time.monotonic()
    if remaining > 10 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        t_final = remaining * 0.85
        print("=== Phase 4: Final polish (n=%d, %.1f sec) ===" % (n_best, t_final))

        # Multiple rounds of polishing with different parameters
        a_cur = global_a.copy()

        # Round 1: Peak polish with high k
        seg1 = t_final * 0.3
        a_p1, _ = run_peak_polish(a_cur, n_best, S_best, seg1, rng,
                                   top_k_start=64, top_k_max=512, noise_scale=0.0005)
        a_p1 = safe_normalize(a_p1, n_best)
        c_p1 = fft_convolve_self(a_p1)
        if c_p1.max() < global_best:
            global_best = c_p1.max()
            global_a = a_p1.copy()
            a_cur = a_p1.copy()
            print("  Polish r1: %.6f" % global_best)

        # Round 2: Focused polish
        remaining2 = deadline - time.monotonic()
        if remaining2 > 5:
            seg2 = remaining2 * 0.3
            a_p2, _ = run_focused_polish(a_cur, n_best, S_best, seg2, rng, top_k=512)
            a_p2 = safe_normalize(a_p2, n_best)
            c_p2 = fft_convolve_self(a_p2)
            if c_p2.max() < global_best:
                global_best = c_p2.max()
                global_a = a_p2.copy()
                a_cur = a_p2.copy()
                print("  Polish r2: %.6f" % global_best)

        # Round 3: FISTA with very high beta
        remaining3 = deadline - time.monotonic()
        if remaining3 > 5:
            seg3 = remaining3 * 0.3
            a_p3, _ = run_fista_adaptive(a_cur, n_best, S_best, seg3, rng,
                                          beta_init=100.0, beta_max=1000.0,
                                          noise_scale=0.0003, restart_interval=5000)
            a_p3 = safe_normalize(a_p3, n_best)
            c_p3 = fft_convolve_self(a_p3)
            if c_p3.max() < global_best:
                global_best = c_p3.max()
                global_a = a_p3.copy()
                print("  Polish r3: %.6f" % global_best)

        # Round 4: Final focused
        remaining4 = deadline - time.monotonic()
        if remaining4 > 5 and global_a is not None:
            a_p4, _ = run_focused_polish(global_a, n_best, S_best, remaining4 * 0.8, rng, top_k=512)
            a_p4 = safe_normalize(a_p4, n_best)
            c_p4 = fft_convolve_self(a_p4)
            if c_p4.max() < global_best:
                global_best = c_p4.max()
                global_a = a_p4.copy()
                print("  Polish r4: %.6f" % global_best)

    # Final verification
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Final: n=%d, internal=%.6f, verified=%.6f" %
          (len(final_list), global_best, ver if np.isfinite(ver) else -1))

    if not np.isfinite(ver):
        # Fallback
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver2 = evaluate_sequence(final_list)
        print("Fallback: verified=%.6f" % (ver2 if np.isfinite(ver2) else -1))

    return final_list
