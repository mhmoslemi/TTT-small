"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using multi-strategy optimization:
1) Refined mirror descent with exact subgradient and backtracking line search
2) Peak-shaving: targeted pair adjustments to reduce the max convolution value
3) Simulated annealing with structured perturbations
4) Multi-start with different sequence lengths (4096, 8192)

Key insight: with sum(a)=sqrt(2n), the objective equals max(conv(a,a)).
We minimize the peak of the self-convolution of a non-negative sequence.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def normalize(a, S_target):
    s = a.sum()
    if s < 1e-30:
        return a
    return a * (S_target / s)


def peak_grad(c, a, top_k=1, soft_T=0.0):
    n = len(a)
    top_k = max(1, min(top_k, len(c)))
    order = np.argsort(c, kind='stable')[::-1][:top_k]
    c_max = c[order[0]]
    if soft_T > 0:
        w = np.exp(soft_T * (c[order] - c_max))
        w /= w.sum()
    else:
        w = np.full(top_k, 1.0 / top_k)
    grad = np.zeros(n)
    for idx, wt in zip(order, w):
        k = int(idx)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for m in range(lo, hi + 1):
            grad[m] += 2.0 * wt * a[k - m]
    return grad


def peak_grad_fast(c, a, top_k=1, soft_T=0.0):
    n = len(a)
    top_k = max(1, min(top_k, len(c)))
    order = np.argsort(c, kind='stable')[::-1][:top_k]
    c_max = c[order[0]]
    if soft_T > 0:
        w = np.exp(soft_T * (c[order] - c_max))
        w /= w.sum()
    else:
        w = np.full(top_k, 1.0 / top_k)
    w_full = np.zeros(len(c))
    w_full[order] = w
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 2).bit_length()
    W = fft(w_full, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    return 2.0 * full_conv[n - 1:2 * n - 1]


def mirror_descent_refined(a0, time_budget, eta0=0.5, momentum=0.85,
                           top_k=1, soft_T=0.0, verbose=True,
                           stagnation_limit=500, perturb_scale=0.03,
                           eta_min=0.01):
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    a = normalize(a, S)

    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)
    eta = eta0
    iter_no = 0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        if time_budget < 1.0 and time.monotonic() - start > 0.5:
            break

        grad = peak_grad_fast(c, a, top_k, soft_T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        cur_obj = obj_value(a, c)
        accepted = False
        for bt in range(10):
            eta_try = eta * (0.6 ** bt)
            a_trial = a * np.exp(-eta_try * v)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = normalize(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)
            if obj_trial <= cur_obj * (1.0 + 1e-12):
                a = a_trial
                c = c_trial
                accepted = True
                break
        if not accepted:
            eta *= 0.5
            if eta < eta_min:
                v *= 0.5
                eta = eta0 * 0.3
                continue

        cur_obj = obj_value(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
            eta = min(eta * 1.05, eta0 * 3.0)
        else:
            eta *= 0.92

        if iter_no - best_iter > stagnation_limit:
            noise = np.random.randn(n) * (perturb_scale * S / np.sqrt(n))
            a = a + noise
            a = np.maximum(a, 0.0)
            a = normalize(a, S)
            c = fft_convolve_self(a)
            v *= 0.3
            eta = eta0
            best_iter = iter_no
            if verbose:
                el = time.monotonic() - start
                print(f"    [perturb] iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, t={el:.1f}s")

        eta *= 0.99997
        iter_no += 1
        if verbose and iter_no % 3000 == 0:
            el = time.monotonic() - start
            print(f"    iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, eta={eta:.4f}, t={el:.1f}s")

    return best_a, best_obj


def peak_shave(a0, time_budget, verbose=True):
    """Targeted peak reduction: find the worst peak and adjust contributing pairs."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    a = normalize(a, S)
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    iter_no = 0

    while time.monotonic() - start < time_budget:
        c_max = c.max()
        k_star = int(np.argmax(c))

        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[k_star - m]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Find safe positions (where a[m] is small and contributes little to peaks)
        # Approximate: positions where a[m] is in the bottom 20% and
        # the corresponding convolution contribution is low
        a_sorted_idx = np.argsort(a)
        safe_count = max(1, n // 5)
        safe_positions = a_sorted_idx[:safe_count]

        # Build direction: decrease where grad is high, increase at safe positions
        direction = -grad
        safe_mean_grad = np.mean(direction[safe_positions])
        direction[safe_positions] = max(0, -safe_mean_grad) * 1.5
        # Ensure zero net change (for sum preservation)
        direction -= direction.mean()

        # Line search
        cur_obj = obj_value(a, c)
        accepted = False
        for alpha_try in [0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = a + alpha_try * S * direction / np.sqrt(n)
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = normalize(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)
            if obj_trial < cur_obj:
                a = a_trial
                c = c_trial
                accepted = True
                break

        if accepted:
            cur_obj = obj_value(a, c)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        else:
            # Try a pure multiplicative step
            a_trial = a * np.exp(-0.05 * grad)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = normalize(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)
            if obj_trial < cur_obj:
                a = a_trial
                c = c_trial
                cur_obj = obj_trial
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
            else:
                # Stagnation: perturb
                noise = np.random.randn(n) * (0.02 * S / np.sqrt(n))
                a = a + noise
                a = np.maximum(a, 0.0)
                a = normalize(a, S)
                c = fft_convolve_self(a)

        iter_no += 1
        if verbose and iter_no % 2000 == 0:
            el = time.monotonic() - start
            print(f"    [shave] iter={iter_no}, obj={obj_value(a,c):.6f}, best={best_obj:.6f}, t={el:.1f}s")

    return best_a, best_obj


def simulated_annealing(a0, time_budget, verbose=True):
    """SA with structured perturbations: swap, mix, noise."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    a = normalize(a, S)
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    cur_obj = best_obj
    start = time.monotonic()
    temp = 0.01
    iter_no = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - start < time_budget:
        batch = 100
        for _ in range(batch):
            # Choose perturbation type
            ptype = rng.random()
            a_trial = a.copy()

            if ptype < 0.4:
                # Gaussian noise on random subset
                k = int(rng.integers(max(1, n // 64), max(2, n // 8)))
                idx = rng.choice(n, size=k, replace=False)
                scale = 0.05 * S / np.sqrt(n)
                a_trial[idx] += rng.randn(k) * scale
            elif ptype < 0.7:
                # Swap: move mass from high to low
                hi_idx = np.argmax(a)
                lo_idx = np.argmin(a)
                delta = 0.02 * a[hi_idx]
                a_trial[hi_idx] -= delta
                a_trial[lo_idx] += delta
            else:
                # Local mix: blend two random segments
                seg_len = int(n // (rng.integers(4, 16)))
                p1 = int(rng.integers(0, n - seg_len))
                p2 = int(rng.integers(0, n - seg_len))
                p = rng.random()
                seg1 = a_trial[p1:p1+seg_len].copy()
                seg2 = a_trial[p2:p2+seg_len].copy()
                a_trial[p1:p1+seg_len] = (1-p)*seg1 + p*seg2
                a_trial[p2:p2+seg_len] = (1-p)*seg2 + p*seg1

            a_trial = np.maximum(a_trial, 0.0)
            a_trial = normalize(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)

            delta_obj = obj_trial - cur_obj
            if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(temp, 1e-10)):
                a = a_trial
                c = c_trial
                cur_obj = obj_trial
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
        temp *= 0.97
        iter_no += 1
        if verbose and iter_no % 50 == 0:
            el = time.monotonic() - start
            print(f"    [SA] iter={iter_no}, cur={cur_obj:.6f}, best={best_obj:.6f}, temp={temp:.5f}, t={el:.1f}s")

    return best_a, best_obj
