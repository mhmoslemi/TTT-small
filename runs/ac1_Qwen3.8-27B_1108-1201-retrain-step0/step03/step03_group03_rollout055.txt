"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
- Log-sum-exp gradient with adaptive beta
- Peak-valley balanced gradient
- FISTA with backtracking
- Coordinate descent polishing
- Multi-length exploration

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_grad(a, c, beta, negate=False):
    """Gradient of ±(1/beta)*log(sum(exp(±beta*c))) w.r.t. a."""
    n = len(a)
    if negate:
        c_work = -c
    else:
        c_work = c
    c_max = c_work.max()
    exp_bc = np.exp(beta * (c_work - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    sign = -1.0 if negate else 1.0
    grad = sign * 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def balanced_grad(a, c, beta, valley_weight=0.3):
    """Combined gradient: push down peaks, push up valleys."""
    g_peak = lse_grad(a, c, beta, negate=False)
    g_valley = lse_grad(a, c, beta, negate=True)
    return g_peak - valley_weight * g_valley


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=1000.0,
              valley_weight=0.3, use_balanced=True):
    """FISTA for minimizing max(conv(a,a)) subject to a>=0, sum(a)=S."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    L = 1.0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
        
        if use_balanced:
            grad = balanced_grad(y, c, beta, valley_weight)
        else:
            grad = lse_grad(y, c, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Step with backtracking
        step = 1.0 / L
        x_new = None
        for bt in range(4):
            x_cand = project_simplex(y - step * grad, S)
            if bt == 0:
                x_new = x_cand
            else:
                c_cand = fft_convolve_self(x_cand)
                if c_cand.max() <= cur_max + 1e-12:
                    x_new = x_cand
                    break
                step *= 0.5
        if x_new is None:
            x_new = project_simplex(y - step * grad, S)
        
        # Update L
        c_x = fft_convolve_self(x_new)
        if c_x.max() > cur_max * 1.01:
            L *= 1.5
        elif it % 50 == 0:
            L = max(L * 0.97, 0.1)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Restart momentum periodically
        if it % 200 == 0:
            t_mom = 1.0
            y = a.copy()
    
    return best_a, best


def run_coordinate_descent(a_init, n, S, max_time, rng):
    """Coordinate descent: update one coordinate at a time."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best = c.max()
    
    # Precompute the "background" convolution (conv without each coordinate)
    # This is expensive, so we'll update incrementally
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        # Random coordinate or structured sweep
        j = rng.integers(n)
        
        # Compute contribution of a[j] to convolution
        # conv[k] includes a[j]*a[k-j] and a[k-j]*a[j] (same term unless k=2j)
        # If we change a[j] to a[j]+delta, the change in conv[k] is:
        #   delta * (a[k-j] + a[k-j]) = 2*delta*a[k-j] for k != 2j
        #   delta * (2*a[j] + delta) for k = 2j (quadratic)
        
        # For simplicity, use a small step
        # Try increasing and decreasing a[j]
        for direction in [1, -1]:
            for scale in [0.01, 0.05, 0.1]:
                a_new = a.copy()
                a_new[j] = a[j] + direction * scale * S / n
                if a_new[j] < 0:
                    continue
                # Renormalize
                s_new = a_new.sum()
                a_new = a_new * (S / s_new)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < best:
                    best = c_new.max()
                    best_a = a_new.copy()
                    a = a_new.copy()
                    break
        
        # Periodic evaluation
        if it % 100 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() < best:
                best = c_check.max()
                best_a = a.copy()
    
    return best_a, best


def run_ponderos(a_init, n, S, max_time, rng):
    """
    'Ponderos' method: alternate between time and frequency domain.
    In frequency: flatten |A(ω)|^2 to make self-convolution flatter.
    In time: project onto a >= 0, sum(a) = S.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    fsize = 1 << (2 * n - 1).bit_length()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Time domain: current a
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Frequency domain step
        A = fft(a, fsize)
        A_sq = A * A  # This is the DFT of the circular self-conv (padded)
        
        # We want |A_sq| to be as flat as possible in the region [0, size-1]
        # (the linear conv region). The remaining bins are "free".
        size = 2 * n - 1
        
        # Strategy: make |A|^2 more uniform
        # Compute target: geometric mean of |A|^2 in the active region
        mag_sq = np.abs(A[:size]) ** 2
        # We want to flatten this
        target = np.geomsqrt(np.geomsqrt(mag_sq[mag_sq > 0]))  # ormean
        target = np.mean(mag_sq[mag_sq > 0])  # use arithmetic mean as target
        
        # Blend towards flat spectrum
        alpha = 0.1  # mixing parameter
        new_mag_sq = mag_sq * (1 - alpha) + target * alpha
        # Preserve phase
        phase = np.angle(A[:size])
        new_A_active = np.sqrt(new_mag_sq) * np.exp(1j * phase)
        
        # For the remaining bins, keep them (or set to 0 for minimum energy there)
        new_A = np.zeros(fsize, dtype=np.complex128)
        new_A[:size] = new_A_active
        # Copy the "tail" bins from original (they correspond to wraparound in linear conv)
        new_A[size:] = A[size:]
        
        # Back to time domain
        a_new = np.real(ifft(new_A))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s_new)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur:
            a = a_new
        else:
            # Try smaller alpha
            for alpha_try in [0.05, 0.01]:
                new_mag_sq2 = mag_sq * (1 - alpha_try) + target * alpha_try
                new_A2_active = np.sqrt(new_mag_sq2) * np.exp(1j * phase)
                new_A2 = np.zeros(fsize, dtype=np.complex128)
                new_A2[:size] = new_A2_active
                new_A2[size:] = A[size:]
                a_try = np.real(ifft(new_A2))[:n]
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try * (S / s_try)
                    c_try = fft_convolve_self(a_try)
                    if c_try.max() < cur:
                        a = a_try
                        break
    
    return best_a, best


def make_anti_palindromic_init(n, rng, S):
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_pectoral_init(n, rng, S):
    """High at edges, low in middle (pectoral/bird-wing shape)."""
    tt = np.linspace(0, 1, n)
    # U-shape: high at 0 and 1, low in middle
    a = (tt - 0.5) ** 2 + 0.1
    # Add some noise
    a += 0.05 * rng.random(n)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_gaussian_inv_init(n, rng, S):
    """Inverse Gaussian: high at edges, dip in center."""
    tt = np.linspace(-1, 1, n)
    sigma = rng.uniform(0.3, 0.6)
    a = 1.0 / (1.0 + (tt / sigma) ** 2)  # Lorentzian (wide peak in middle)
    # Actually we want the OPPOSITE: high at edges
    a = 1.0 - 0.7 * np.exp(-tt**2 / (2*sigma**2))
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_multisine_init(n, rng, S):
    """Multiple sinusoids with random frequencies and phases."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    n_freqs = rng.integers(2, 6)
    for _ in range(n_freqs):
        freq = rng.integers(2, 30)
        phase = rng.random() * 2 * np.pi
        amp = rng.uniform(0.1, 0.4)
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_weighted_anti_pal(n, rng, S):
    """Anti-palindromic with a power-law weighting."""
    half = n // 2
    idx = np.arange(half, dtype=np.float64) + 1
    p = rng.uniform(0.5, 2.0)
    h = idx ** p + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):

    if n % 2 == 1:

    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:

    return a


def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
- Log-sum-exp smoothed FISTA with adaptive beta (up to 10000)
- Peak-valley balanced gradient for faster convergence
- Frequency-domain spectral flattening (novel)
- Direct peak-reduction coordinate updates
- Multi-initialization: anti-palindromic, spectral, structured
- Primary length n=16384 with optional n=32768 exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def lse_gradient_neg(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(-beta*c))) w.r.t. a (valley gradient)."""
    n = len(a)
    c_min = c.min()
    exp_bc = np.exp(-beta * (c - c_min))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return -2.0 * np.real(ifft(W * np.conj(A)))[:n]


def balanced_lse_gradient(a, c, beta, valley_weight):
    """Combined gradient: push down peaks AND push up valleys."""
    g_peak = lse_gradient(a, c, beta)
    g_valley = lse_gradient_neg(a, c, beta)
    return g_peak + valley_weight * g_valley


def topk_soft_gradient(a, c, k, temperature):
    """Gradient using top-k lags with softmax weighting."""
    n = len(a)
    k_eff = min(k, len(c))
    top_indices = np.argsort(c)[::-1][:k_eff]
    top_vals = c[top_indices]
    tv_max = top_vals.max()
    weights = np.exp(top_vals / max(temperature, 1e-10) - tv_max / max(temperature, 1e-10))
    weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista_lse(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=10000.0,
                  valley_weight=0.25, use_topk=False, topk_k=20):
    """
    FISTA with log-sum-exp gradient and adaptive beta.
    Minimizes max(conv(a,a)) subject to a>=0, sum(a)=S.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    L = 1.0
    c_prev = None

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.02 * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()

        if use_topk:
            grad = topk_soft_gradient(y, c, topk_k, temperature=1.0)
        elif valley_weight > 0:
            grad = balanced_lse_gradient(y, c, beta, valley_weight)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = 1.0 / L
        x_new = None
        for bt in range(4):
            x_cand = project_simplex(y - step * grad, S)
            if bt == 0:
                x_new = x_cand
            else:
                c_cand = fft_convolve_self(x_cand)
                if c_cand.max() <= cur_max + 1e-12:
                    x_new = x_cand
                    break
                step *= 0.5
        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        c_x = fft_convolve_self(x_new)
        if c_x.max() > cur_max * 1.005:
            L *= 1.5
        elif it % 100 == 0:
            L = max(L * 0.95, 0.01)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 300 == 0:
            t_mom = 1.0
            y = a.copy()

    return best_a, best


def run_spectral_flatten(a_init, n, S, max_time, rng, alpha_init=0.15):
    """
    Frequency-domain spectral flattening.
    Iteratively flatten |A(w)|^2 in the active frequency region,
    then project back to a>=0, sum=S.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    fsize = 1 << (2 * n - 1).bit_length()
    size = 2 * n - 1
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()

        A = fft(a, fsize)
        mag_sq = np.abs(A[:size]) ** 2

        # Target: flatten the spectrum in the active region
        # Use geometric mean as target (preserves total energy better)
        nonzero_mask = mag_sq > 1e-20
        if np.sum(nonzero_mask) < 2:
            break
        log_mag = np.log(mag_sq[nonzero_mask])
        target_log = np.mean(log_mag)
        target = np.exp(target_log)

        # Decay alpha over iterations
        alpha = alpha_init / (1.0 + 0.005 * it)
        alpha = max(alpha, 0.01)

        # Blend: new_mag_sq = mag_sq^(1-alpha) * target^alpha (geometric blend)
        new_log_mag = (1 - alpha) * np.log(mag_sq + 1e-30) + alpha * target_log
        new_mag_sq = np.exp(new_log_mag)

        # Reconstruct A with new magnitude, preserve phase
        phase = np.angle(A[:size])
        new_A = np.zeros(fsize, dtype=np.complex128)
        new_A[:size] = np.sqrt(new_mag_sq) * np.exp(1j * phase)
        new_A[size:] = A[size:]

        # Back to time domain
        a_new = np.real(ifft(new_A))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s_new)

        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur:
            a = a_new
        else:
            # Try smaller alpha
            alpha2 = alpha * 0.3
            new_log_mag2 = (1 - alpha2) * np.log(mag_sq + 1e-30) + alpha2 * target_log
            new_mag_sq2 = np.exp(new_log_mag2)
            new_A2 = np.zeros(fsize, dtype=np.complex128)
            new_A2[:size] = np.sqrt(new_mag_sq2) * np.exp(1j * phase)
            new_A2[size:] = A[size:]
            a_try = np.real(ifft(new_A2))[:n]
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try * (S / s_try)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < cur:
                    a = a_try

    return best_a, best

        it += 1
        c = fft_convolve_self(a)
        cur = c.max()

            best = cur
            best_a = a.copy()

        top_lags_arr = np.argsort(c)[::-1][:top_lags]
