"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding.

Key approach:
- At each iteration, identify the "active" lags (near the peak of convolution)
- Solve an LP to find the best direction to move (maximizing sum while keeping 
  convolution with current sequence bounded at active lags)
- Update the sequence along this direction
- Polish with FISTA/LSE smoothing
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def solve_lp_direction(f, M, active_lags, n):
    """
    Solve: max sum(b) s.t. (f*b)_k <= M for k in active_lags, b >= 0.
    
    Returns the optimal b (or None if infeasible).
    """
    K = len(active_lags)
    if K == 0:
        return None
    
    # Build sparse constraint matrix A (K x n)
    # A[r, j] = f[active_lags[r] - j] if 0 <= active_lags[r] - j < n
    rows = []
    cols = []
    vals = []
    
    for r, k in enumerate(active_lags):
        k = int(k)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i
            if f[i] > 1e-15:  # Skip zeros
                rows.append(r)
                cols.append(j)
                vals.append(f[i])
    
    A_sparse = csr_matrix((vals, (rows, cols)), shape=(K, n))
    
    # LP: minimize -sum(b) s.t. A_sparse @ b <= M, b >= 0
    c_obj = -np.ones(n)
    res = linprog(
        c_obj,
        A_ub=A_sparse,
        b_ub=np.full(K, M),
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True, 'time_limit': 60}
    )
    
    if res.status == 0:
        return res.x
    return None

def run_fista(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=500.0):
    """FISTA with LSE smoothing for polishing."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    L = 4.0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)
        
        # Simple backtracking
        c_x = fft_convolve_self(x_new)
        retry = 0
        while c_x.max() > cur + 1e-10 and retry < 5:
            L *= 1.5
            step = 1.0 / L
            x_new = project_simplex(y - step * grad, S)
            c_x = fft_convolve_self(x_new)
            retry += 1
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic noise
        if it % 200 == 0 and rng.random() < 0.3:
            noise = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0
    
    return best_a, best_score


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic initialization."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def safe_normalize(a, n):
    """Clamp and normalize."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

**Cheaper approximation to the LP direction:**

The LP `max sum(b) s.t. A@b <= M, b >= 0` has the dual:

**Frank-Wolfe for this problem:**
The Frank-Wolfe algorithm for minimizing `f(a)` over the simplex:
1. Start with `a_0` (any point in the simplex)
2. At each iteration:
   a. Compute gradient `g = grad(f(a_t))`
   b. Find the vertex (or minimum of linear function) of the simplex: `v = argmin_{x in simplex} g^T x`
      For the simplex `{x >= 0, sum(x) = S}`, this is `v = S * e_j` where `j = argmin g_j`
   c. Update: `a_{t+1} = (1 - gamma_t) * a_t + gamma_t * v`
      where `gamma_t` is a step size (e.g., `2/(t+2)` for the standard FW, or found by line search)

The advantage of FW is that each step is O(n) (just find the min gradient component and do a convex combination). No projection needed!

However, FW is known to be slow (O(1/T) convergence for convex functions, same as subgradient but with a worse constant). It also has the issue of "slow convergence near the boundary" (it takes many iterations to get components exactly to zero).

A variant called "Away-Step Frank-Wolfe" (AFW) is faster in practice. It allows moving away from the current support, which helps with convergence near the boundary.

Let me consider implementing AFW:
1. At each iteration, compute gradient `g`
2. Find `v = S * e_{argmin g}` (towards step)
3. Find `w = argmin_{j in support(a)} g_j / a_j` (away step) - the element in the current support with the largest positive gradient relative to its value
4. Choose to move towards `v` or away from `w` (or both) based on descent
5. Update

This might be competitive with FISTA for this problem.

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding + FISTA polishing.

Key innovations over previous approach:
- LP-based direction finding: solves a restricted LP at each iteration to find the
  optimal direction within a feasible polytope defined by cross-convolution constraints
- Adaptive active-set selection (top-K lags with threshold)
- FISTA polishing with adaptive beta annealing
- Multiple restart strategies with structured noise
- Coordinate descent flash polish for final millimeter improvements

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, lil_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def solve_lp_direction(f, M, active_lags, n, time_limit=30):
    """
    Solve: max sum(b) s.t. (f*b)_k <= M for k in active_lags, b >= 0.
    Returns optimal b or None.
    """
    K = len(active_lags)
    if K == 0:
        return None

    # Build sparse constraint matrix
    rows_list = []
    cols_list = []
    vals_list = []

    for r, k in enumerate(active_lags):
        k = int(k)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i
            if f[i] > 1e-18:
                rows_list.append(r)
                cols_list.append(j)
                vals_list.append(f[i])

    if len(rows_list) == 0:
        return None

    A_sparse = csr_matrix((vals_list, (rows_list, cols_list)), shape=(K, n))

    c_obj = -np.ones(n)
    b_ub = np.full(K, M)

    try:
        res = linprog(
            c_obj,
            A_ub=A_sparse,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': time_limit}
        )
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def run_fista(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=500.0):
    """FISTA with adaptive beta and backtracking."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    L = 4.0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.02 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-13:
            best_score = cur
            best_a = y.copy()
            stall = 0
        elif it % 50 == 0:
            stall += 1

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)

        # Backtracking
        retry = 0
        while retry < 4:
            c_x = fft_convolve_self(x_new)
            if c_x.max() <= cur + 1e-12:
                break
            L *= 1.5
            step = 1.0 / L
            x_new = project_simplex(y - step * grad, S)
            retry += 1

        # Adaptive L decrease
        if it % 50 == 0 and stall > 3:
            L = max(L * 0.9, 4.0)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic perturbation
        if stall > 5 and rng.random() < 0.4:
            noise = 0.01 * S / np.sqrt(n) * (1.0 + 0.1 * stall)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0
            stall = 0

    return best_a, best_score


def run_coordinate_polish(a_init, n, S, max_time, rng, beta=50.0):
    """Coordinate gradient descent for fine polishing."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # Compute LSE gradient
        grad = lse_gradient(a, c, beta)

        # Process coordinates in random order
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - start > max_time - 2:
                break
            gj = grad[j]
            if abs(gj) < 1e-15:
                continue
            # Small step for this coordinate
            alpha = 0.01 * S / (n * max(np.linalg.norm(grad), 1e-15))
            old_val = a[j]
            a[j] = max(0.0, a[j] - alpha * gj)
            # Renormalize (cheap: adjust one other element)
            diff = S - a.sum()
            if abs(diff) > 1e-15:
                # Distribute to a random other element
                k = rng.integers(n)
                if k != j:
                    a[k] = max(0.0, a[k] + diff)
                else:
                    a[j] = old_val
            # Update gradient incrementally (approximate: just skip for simplicity)

        # Full renormalize every few iterations
        if it % 10 == 0:
            a = project_simplex(a, S)

    return best_a, best_score


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic initialization: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-15:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_periodic_init(n, rng, S, num_freqs=3):
    """Periodic sinusoidal initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 20)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.35
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_w_shape_init(n, rng, S):
    """W/M-shaped initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    k = rng.integers(2, 6)
    a = np.abs(np.sin(2 * np.pi * k * tt)) + 0.05
    a = a * (S / a.sum())
    return a


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for optimal non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
    1. LP-based direction finding (main optimizer)
    2. FISTA with adaptive beta (polishing)
    3. Coordinate descent (final fine-tuning)
    4. Multiple restarts with structured perturbations
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    # Get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    primary_n = 8192
    if parent is not None and len(parent) >= 64:
        primary_n = len(parent)

    S = np.sqrt(2.0 * primary_n)
    print("n=%d, parent=%s, budget=%.0fs" % (primary_n, parent is not None, budget_s))

    # Initial sequence
    if parent is not None:
        a0 = safe_normalize(parent, primary_n)
    else:
        a0 = make_anti_palindromic_init(primary_n, rng, S)

    # === Phase 1: LP-based optimization (60% of time) ===
    print("=== Phase 1: LP optimization ===")
    t0 = time.monotonic()
    t_phase1 = max(120.0, (deadline - t0) * 0.55)

    # First LP round with small K (fast iterations)
    t1 = min(t_phase1 * 0.4, max(60.0, (deadline - t0) * 0.25))
    a1, s1 = run_lp_optimization(a0, primary_n, S, t1, rng, K_init=150, K_max=500, verbose=True)
    a1 = safe_normalize(a1, primary_n)
    c1 = fft_convolve_self(a1)
    score1 = c1.max()
    if score1 < global_best:
        global_best = score1
        global_a = a1.copy()
        print("  After LP round 1: %.7f" % global_best)

    # Second LP round with larger K
    if time.monotonic() < deadline - t_phase1 * 0.2:
        rem = deadline - time.monotonic() - t_phase1 * 0.15
        if rem > 60:
            t2 = max(60.0, rem * 0.8)
            a2, s2 = run_lp_optimization(a1, primary_n, S, t2, rng, K_init=400, K_max=800, verbose=True)
            a2 = safe_normalize(a2, primary_n)
            c2 = fft_convolve_self(a2)
            score2 = c2.max()
            if score2 < global_best:
                global_best = score2
                global_a = a2.copy()
                print("  After LP round 2: %.7f" % global_best)
            a1 = a2

    print("=== Phase 1 complete: best=%.7f ===" % global_best)

    # === Phase 2: FISTA polishing (20% of time) ===
    print("=== Phase 2: FISTA polish ===")
    rem = deadline - time.monotonic()
    if rem > 60 and global_a is not None:
        t_fista = max(30.0, rem * 0.5)
        a_f, s_f = run_fista(global_a, primary_n, S, t_fista, rng,
                              beta_init=100.0, beta_max=600.0)
        a_f = safe_normalize(a_f, primary_n)
        c_f = fft_convolve_self(a_f)
        if c_f.max() < global_best:
            global_best = c_f.max()
            global_a = a_f.copy()
            print("  FISTA: %.7f" % global_best)

    # === Phase 3: Another LP round (15% of time) ===
    rem = deadline - time.monotonic()
    if rem > 60 and global_a is not None:
        print("=== Phase 3: LP round 3 ===")
        t3 = max(30.0, rem * 0.6)
        a3, s3 = run_lp_optimization(global_a, primary_n, S, t3, rng,
                                      K_init=500, K_max=1000, verbose=True)
        a3 = safe_normalize(a3, primary_n)
        c3 = fft_convolve_self(a3)
        if c3.max() < global_best:
            global_best = c3.max()
            global_a = a3.copy()
            print("  LP round 3: %.7f" % global_best)

    # === Phase 4: Final fine polish (remaining time) ===
    rem = deadline - time.monotonic()
    if rem > 30 and global_a is not None:
        print("=== Phase 4: Fine polish ===")

        # High-beta FISTA
        t_fin = max(10.0, rem * 0.5)
        a_fin, _ = run_fista(global_a, primary_n, S, t_fin, rng,
                              beta_init=300.0, beta_max=1000.0)
        a_fin = safe_normalize(a_fin, primary_n)
        c_fin = fft_convolve_self(a_fin)
        if c_fin.max() < global_best:
            global_best = c_fin.max()
            global_a = a_fin.copy()
            print("  Fine FISTA: %.7f" % global_best)

        # Coordinate polish
        rem2 = deadline - time.monotonic()
        if rem2 > 20:
            a_coord, _ = run_coordinate_polish(global_a, primary_n, S,
                                                max(5.0, rem2 * 0.7), rng, beta=200.0)
            a_coord = safe_normalize(a_coord, primary_n)
            c_coord = fft_convolve_self(a_coord)
            if c_coord.max() < global_best:
                global_best = c_coord.max()
                global_a = a_coord.copy()
                print("  Coord polish: %.7f" % global_best)

    # === Final verification ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL: evaluate=%.7f, internal=%.7f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: rejected, using fallback")
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback: %.7f" % ver)

    return final_list
