def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
        theta = (cumsum[n-1] - S) / n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

            res = sp_opt.linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                                 bounds=(0, None), method='highs',
                                 options={'time_limit': 30})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.maximum(res.x, 0.0)
        b_sum = b.sum()
        if b_sum < 1e-10:
            break

        c_ab = fft_convolve(a, b)
        violated = np.where(c_ab > M * (1.0 + 1e-7))[0]

        if len(violated) == 0:
            break

        viol_amt = c_ab[violated] - M
        top_v = violated[np.argsort(viol_amt)[::-1][:10]]
        for v in top_v:
            if int(v) not in active_set:
                if len(active_set) < max_active:
                    active_set.append(int(v))
                else:
                    break

    if b is None or b.sum() < 1e-10:
        return None
    return b


def spectral_shave(a, S, n, num_iters=50, rng=None):
    """
    Iterative spectral water-filling: reduce power at frequencies contributing most
    to the peak autocorrelation lag.
    """
    fsize = 1 << (2 * n - 1 - 1).bit_length()

    for it in range(num_iters):
        A = fft(a, fsize)
        mag = np.abs(A)
        ang = np.angle(A)

        c = fft_convolve_self(a)
        peak_lag = int(np.argmax(c))
        peak_val = c.max()

        # Phase factor for the peak lag
        freqs = np.arange(fsize)
        phase = np.exp(-2j * np.pi * freqs * peak_lag / fsize)
        # Frequencies where the real part of phase is positive contribute constructively
        contrib = np.real(mag**2 * phase)

        # Reduce power at contributing frequencies, increase at others
        alpha = 0.01
        mag_reduction = np.maximum(contrib, 0) - np.minimum(contrib, 0)
        mag_reduction = mag_reduction / (np.max(np.abs(mag_reduction)) + 1e-15)

        new_mag = mag * (1.0 - alpha * mag_reduction)
        new_mag = np.maximum(new_mag, 0.1 * mag.min() + 1e-10)

        # Preserve total energy approximately
        energy_ratio = np.sum(mag**2) / (np.sum(new_mag**2) + 1e-30)
        new_mag = new_mag * np.sqrt(energy_ratio)

        A_new = new_mag * np.exp(1j * ang)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_sum(a_new, n)

        if rng is not None and it % 10 == 0:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a_new = project_simplex(a_new + noise, S)

        a = a_new

    return a


def generate_initializations(n, rng):
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / 0.25))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.5, n)))
    inits.append(1.0 - 2.0 * np.abs(tt - 0.5))
    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA exploration, LP constraint-generation polishing,
    spectral shaving, and high-beta FISTA final polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Try to get the given parent sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose main length
    if given is not None and 4096 <= len(given) <= 32768:
        main_len = len(given)
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * n)
    print("Starting optimization: n=%d, budget=%.0fs" % (n, budget_s))

    # Collect initializations
    inits = []
    if given is not None:
        inits.append(("parent", np.clip(given, 0, 1000)))

    gen_inits = generate_initializations(n, rng)
    for i, g in enumerate(gen_inits):
        inits.append(("init_%d" % i, g))

    # ============ PHASE 1: Quick FISTA Exploration ============
    phase1_time = min(240.0, (deadline - time.monotonic()) * 0.12)
    print("=== Phase 1: FISTA Exploration (%.0fs) ===" % phase1_time)
    p1_start = time.monotonic()

    n_inits = len(inits)
    per_init_time = max(10.0, phase1_time / n_inits)

    for i, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - (deadline - p1_start) * 0.85:
            break
        arr_n = len(arr)
        if arr_n < 64:
            continue
        S_i = np.sqrt(2.0 * arr_n)
        t_alloc = min(per_init_time, deadline - time.monotonic() - 60)
        if t_alloc < 5:
            break

        modes = ["lse", "topk", "lse"]
        a_opt, score = run_fista(arr, arr_n, S_i, t_alloc, rng,
                                  mode=modes[i % 3],
                                  beta_init=5.0, beta_rate=0.005,
                                  beta_max=1000.0,
                                  topk_frac=0.05)
        a_norm = normalize_sum(a_opt, arr_n)
        c_chk = fft_convolve_self(a_norm)
        real_score = c_chk.max()

        if real_score < global_best:
            global_best = real_score
            global_a = a_norm.copy()
            print("  [%s] n=%d: peak=%.6f (elapsed %.0fs)" %
                  (name, arr_n, global_best, time.monotonic() - p1_start))

    print("Phase 1 best peak: %.6f" % global_best)

    # ============ PHASE 2: LP Constraint-Generation Polishing ============
    remaining = deadline - time.monotonic()
    phase2_time = remaining * 0.45
    print("=== Phase 2: LP Polishing (%.0fs) ===" % phase2_time)
    p2_start = time.monotonic()

    if global_a is not None:
        n_pol = len(global_a)
        S_pol = np.sqrt(2.0 * n_pol)
        a_work = global_a.copy()
        current_peak = fft_convolve_self(a_work).max()

        lp_step_count = 0
        while time.monotonic() < p2_start + phase2_time:
            it_start = time.monotonic()
            elapsed_it = time.monotonic() - it_start

            if elapsed_it > phase2_time * 0.3:
                print("  LP step too slow (%.1fs), skipping to next phase" % elapsed_it)
                break

            M = current_peak
            b_opt = lp_polish_step(a_work, M, max_cong_iters=20, max_active=200)

            if b_opt is None or b_opt.sum() <= S_pol * (1 + 1e-6):
                print("  LP: no improvement (sum_b <= S), stopping LP phase")
                break

            # Normalize and convex combine
            g = S_pol * b_opt / b_opt.sum()
            # Adaptive step size
            improvement_ratio = b_opt.sum() / S_pol - 1.0
            t_step = min(0.3, max(0.02, improvement_ratio))

            a_new = (1.0 - t_step) * a_work + t_step * g
            a_new = normalize_sum(a_new, n_pol)
            new_peak = fft_convolve_self(a_new).max()

            if new_peak < current_peak:
                a_work = a_new
                current_peak = new_peak
                if current_peak < global_best:
                    global_best = current_peak
                    global_a = a_work.copy()
                print("  LP step %d: peak=%.6f (t=%.3f, ratio=%.4f, elapsed=%.0fs)" %
                      (lp_step_count, current_peak, t_step, improvement_ratio,
                       time.monotonic() - p2_start))
            else:
                # Try smaller step
                t_step2 = t_step * 0.3
                a_new2 = (1.0 - t_step2) * a_work + t_step2 * g
                a_new2 = normalize_sum(a_new2, n_pol)
                new_peak2 = fft_convolve_self(a_new2).max()
                if new_peak2 < current_peak:
                    a_work = a_new2
                    current_peak = new_peak2
                    if current_peak < global_best:
                        global_best = current_peak
                        global_a = a_work.copy()
                    print("  LP step %d (small): peak=%.6f (elapsed=%.0fs)" %
                          (lp_step_count, current_peak, time.monotonic() - p2_start))
                else:
                    print("  LP step %d: no improvement, reducing active set" % lp_step_count)
                    # Perturb to escape
                    noise = rng.normal(0, 0.001 * S_pol / np.sqrt(n_pol), n_pol)
                    a_work = project_simplex(a_work + noise, S_pol)
                    current_peak = fft_convolve_self(a_work).max()

            lp_step_count += 1
            if lp_step_count > 50:
                break

    print("Phase 2 best peak: %.6f" % global_best)

    # ============ PHASE 3: Spectral Shaving ============
    remaining = deadline - time.monotonic()
    phase3_time = remaining * 0.25
    print("=== Phase 3: Spectral Shaving (%.0fs) ===" % phase3_time)

    if global_a is not None and phase3_time > 10:
        n_spec = len(global_a)
        S_spec = np.sqrt(2.0 * n_spec)
        a_spec = global_a.copy()

        # Run spectral shaving in chunks
        chunk_time = min(30.0, phase3_time / 3)
        for chunk in range(3):
            if time.monotonic() > deadline - 60:
                break
            t_chunk = min(chunk_time, deadline - time.monotonic() - 60)
            if t_chunk < 5:
                break
            a_spec = spectral_shave(a_spec, S_spec, n_spec,
                                    num_iters=200, rng=rng)
            # Actually spectral_shave doesn't take time, let me just run it
            # with limited iterations based on time
            c_spec = fft_convolve_self(a_spec)
            if c_spec.max() < global_best:
                global_best = c_spec.max()
                global_a = a_spec.copy()
                print("  Spectral chunk %d: peak=%.6f" % (chunk, global_best))

    # ============ PHASE 4: High-Beta FISTA Final Polish ============
    remaining = deadline - time.monotonic()
    phase4_time = max(30.0, remaining * 0.5)
    print("=== Phase 4: High-Beta FISTA Polish (%.0fs) ===" % phase4_time)

    if global_a is not None:
        n_fin = len(global_a)
        S_fin = np.sqrt(2.0 * n_fin)

        # Sub-phase 4a: Top-k FISTA
        t_4a = phase4_time * 0.4
        a_4a, s_4a = run_fista(global_a, n_fin, S_fin, t_4a, rng,
                                mode="topk",
                                beta_init=50.0, beta_rate=0.003,
                                beta_max=3000.0, topk_frac=0.02)
        a_4a = normalize_sum(a_4a, n_fin)
        c_4a = fft_convolve_self(a_4a)
        if c_4a.max() < global_best:
            global_best = c_4a.max()
            global_a = a_4a.copy()
            print("  After topk FISTA: %.6f" % global_best)

        # Sub-phase 4b: Very high beta LSE
        remaining_4b = deadline - time.monotonic() - 20
        if remaining_4b > 10:
            a_4b, s_4b = run_fista(global_a, n_fin, S_fin, remaining_4b, rng,
                                    mode="lse",
                                    beta_init=200.0, beta_rate=0.002,
                                    beta_max=8000.0)
            a_4b = normalize_sum(a_4b, n_fin)
            c_4b = fft_convolve_self(a_4b)
            if c_4b.max() < global_best:
                global_best = c_4b.max()
                global_a = a_4b.copy()
                print("  After high-beta FISTA: %.6f" % global_best)

    # ============ PHASE 5: Try different length if time remains ============
    remaining = deadline - time.monotonic()
    if remaining > 120:
        alt_n = n * 2 if n <= 16384 else n // 2
        if 4096 <= alt_n <= 32768 and alt_n != n:
            S_alt = np.sqrt(2.0 * alt_n)
            print("=== Phase 5: Trying n=%d (remaining %.0fs) ===" % (alt_n, remaining))
            alt_inits = generate_initializations(alt_n, rng)
            t_alt = remaining * 0.5
            a_alt, s_alt = run_fista(alt_inits[0], alt_n, S_alt, t_alt, rng,
                                      mode="lse",
                                      beta_init=10.0, beta_rate=0.005,
                                      beta_max=2000.0)
            a_alt = normalize_sum(a_alt, alt_n)
            c_alt = fft_convolve_self(a_alt)
            if c_alt.max() < global_best:
                global_best = c_alt.max()
                global_a = a_alt.copy()
                print("  Alt n=%d: %.6f" % (alt_n, global_best))

    # ============ Final Verification ============
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Final verification: %.6f (length=%d)" % (ver, len(final_list)))
    else:
        print("Verification failed, using fallback")
        final_list = [float(x) for x in np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)]
        ver = evaluate_sequence(final_list)
        print("Fallback verification: %.6f" % ver)

    print("Returning sequence of length %d" % len(final_list))
    return final_list
