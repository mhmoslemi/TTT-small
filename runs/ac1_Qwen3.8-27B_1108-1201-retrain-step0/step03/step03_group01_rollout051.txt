```python
"""
Minimize 2*n*max_conv/sum^2 using adaptive FISTA-LSE, block coordinate descent,
coarse-to-fine optimization, and spectral shaping with diversity restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_lse_grad(a, c, beta):
    """Compute gradient of LSE approximation of max(c) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    total = exp_c.sum()
    if total < 1e-300:
        return np.zeros(n)
    p = exp_c / total
    fsize = 1 << (2 * n - 1).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(P * np.conj(A)))[:n]
    return grad

def fista_lse_adaptive(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=500.0, 
                        restart_interval=200, target=1.5):
    """FISTA with LSE, adaptive beta, and restarts."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    last_restart = 0
    f_prev = np.inf
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        # Check for restart (Pei-Terrapon)
        if it - last_restart >= restart_interval:
            c_y = fft_convolve_self(y)
            c_a = fft_convolve_self(a)
            if c_y.max() > c_a.max() * 1.001:
                y = a.copy()
                t = 1.0
                last_restart = it
                continue
        
        grad = compute_lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = 1.0 / (beta * 2.0 * max(a.max(), 1e-6) * 2.0)
        step = max(step, 1e-8)
        x_new = project_simplex(y - step * grad, S)
        
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if cur <= target:
            break
    
    return best_local, best_score
```

```python
"""
Minimize 2*n*max_conv/sum^2 using adaptive FISTA-LSE, block coordinate descent,
coarse-to-fine optimization, spectral shaping, and diversity restarts.
Key innovations: block CD for local refinement, coarse-to-fine for global exploration,
and adaptive beta scheduling with intelligent restarts.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_lse_grad(a, c, beta):
    """Compute gradient of LSE approximation of max(c) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    total = exp_c.sum()
    if total < 1e-300:
        return np.zeros(n)
    p = exp_c / total
    fsize = 1 << (2 * n - 1).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(P * np.conj(A)))[:n]
    return grad


def fista_lse(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=1000.0,
              target=1.45, restart_check=50):
    """FISTA with LSE smooth approximation, adaptive beta, and smart restarts."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    last_restart = 0
    stagnation_count = 0
    prev_best = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.02 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stagnation_count = 0
        else:
            if best_score < prev_best - 1e-8:
                prev_best = best_score
            else:
                stagnation_count += 1

        if cur <= target:
            break

        # Smart restart: if stalled, reset momentum
        if stagnation_count > restart_check * 3:
            y = a.copy()
            t_momentum = 1.0
            stagnation_count = 0
            # Also try a small perturbation
            if rng.random() < 0.5:
                noise_scale = 0.005 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            last_restart = it

        grad = compute_lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        a_max = max(a.max(), 1e-6)
        step = 1.0 / (2.0 * beta * a_max + 4.0)
        step = max(step, 1e-10)

        x_new = project_simplex(y - step * grad, S)

        # Check improved
        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

    return best_local, best_score


def spectral_shaping(a, N, top_k, alpha, n_steps, rng, S, target=1.45):
    """Spectral shaping: modify power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    best_a = a.copy()
    best_score = np.inf

    for it in range(n_steps):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = c.max()

        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        if c_max <= target:
            break

        # Compute effective cosine for top-k peaks
        m = np.arange(N)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

        # Adaptive alpha
        excess = max(0.0, c_max - target) / max(c_max, 1e-10)
        alpha_curr = alpha * (0.2 + 0.8 * excess)

        P_new = np.maximum(P * (1.0 - alpha_curr * cos_eff), 0.0)
        # Ensure P_new has similar total energy
        if P_new.sum() > 1e-30 and P.sum() > 1e-30:
            P_new = P_new * (P.sum() / P_new.sum())

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)

        # Occasional noise for diversity
        if it % 100 == 50 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)

    return best_a, best_score


def block_coordinate_descent(a_init, n, S, max_time, rng, block_size=32,
                             n_inner=150, beta=30.0):
    """Block coordinate descent: optimize small blocks of variables at a time."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        # Choose random block
        bs = block_size if rng.random() < 0.7 else min(64, block_size * 2)
        if bs >= n:
            bs = n // 2
        start_idx = rng.integers(0, n - bs)
        end_idx = start_idx + bs

        block = a[start_idx:end_idx].copy()
        block_sum = block.sum()

        for it in range(n_inner):
            a[start_idx:end_idx] = block
            c = fft_convolve_self(a)
            c_max = c.max()
            if c_max < best_score:
                best_score = c_max
                best_local = a.copy()

            # LSE gradient restricted to block
            c_shifted = c - c.max()
            exp_c = np.exp(beta * c_shifted)
            p = exp_c / exp_c.sum()

            P = fft(p, fsize)
            A = fft(a, fsize)
            grad_full = 2.0 * np.real(ifft(P * np.conj(A)))[:n]
            grad_block = grad_full[start_idx:end_idx]

            # Projected gradient step on block
            step = 1.0 / (2.0 * beta * max(a.max(), 0.1) + 2.0)
            block = block - step * grad_block
            block = project_simplex(block, block_sum)

        a[start_idx:end_idx] = block

    return best_local, best_score


def coarse_to_fine(a_init, n, S, max_time, rng):
    """Downsample, optimize at coarse scale, upsample, and refine."""
    factor = 4
    n_coarse = n // factor

    # Downsample by averaging
    a_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        a_coarse[i] = a_init[i * factor:(i + 1) * factor].mean()
    a_coarse = project_simplex(a_coarse, S)

    # Optimize at coarse scale
    t_coarse = min(60.0, max_time * 0.3)
    a_coarse, _ = fista_lse(a_coarse, n_coarse, S, t_coarse, rng,
                            beta_init=10, beta_max=200)

    # Upsample by repeating
    a_up = np.zeros(n)
    for i in range(n_coarse):
        a_up[i * factor:(i + 1) * factor] = a_coarse[i]
    a_up = project_simplex(a_up, S)

    # Blend with original
    a_blend = 0.5 * a_up + 0.5 * a_init
    a_blend = project_simplex(a_blend, S)

    # Refine at full scale
    t_fine = max_time - t_coarse - (time.monotonic() - (time.monotonic() - max_time))
    t_fine = max(10.0, max_time * 0.6)
    a_refined, score = fista_lse(a_blend, n, S, min(t_fine, max_time * 0.6),
                                  rng, beta_init=20, beta_max=500)
    return a_refined, score


def multi_restart(global_a, n, S, max_time, rng, n_restarts=10):
    """Try multiple perturbed versions of the best solution."""
    best_a = global_a.copy()
    best_score = fft_convolve_self(global_a).max()
    start = time.monotonic()
    budget_per_restart = (max_time - 10.0) / max(n_restarts, 1)

    for r in range(n_restarts):
        if time.monotonic() - start > max_time - 10:
            break

        # Different perturbation strategies
        strategy = r % 4
        a_pert = global_a.copy()

        if strategy == 0:
            # Gaussian noise
            scale = 0.01 * S / np.sqrt(n)
            a_pert = project_simplex(a_pert + rng.normal(0, scale, n), S)
        elif strategy == 1:
            # Block swap
            bs = n // 8
            i1 = rng.integers(0, n - bs)
            i2 = rng.integers(0, n - bs)
            if abs(i1 - i2) > bs:
                a_pert[i1:i1 + bs], a_pert[i2:i2 + bs] = a_pert[i2:i2 + bs].copy(), a_pert[i1:i1 + bs].copy()
                a_pert = project_simplex(a_pert, S)
        elif strategy == 2:
            # Frequency domain perturbation
            N = 1 << (2 * n - 1).bit_length()
            a_pad = np.zeros(N)
            a_pad[:n] = a_pert
            A = fft(a_pad)
            phase = np.angle(A) + rng.normal(0, 0.1, N)
            A_new = np.abs(A) * np.exp(1j * phase)
            a_new = np.real(ifft(A_new))[:n]
            a_pert = project_simplex(a_new, S)
        else:
            # Random subset replacement
            n_replace = n // 5
            idx = rng.choice(n, size=n_replace, replace=False)
            a_pert[idx] = 0.1 * S / n * (1 + 0.5 * rng.random(n_replace))
            a_pert = project_simplex(a_pert, S)

        # Optimize the perturbed version
        t_avail = min(budget_per_restart, max_time - (time.monotonic() - start) - 5)
        if t_avail < 5:
            break
        a_opt, score = fista_lse(a_pert, n, S, t_avail, rng,
                                  beta_init=20, beta_max=300)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print("    Restart %d improved: %.6f" % (r, best_score))

    return best_a, best_score
```
