"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: FISTA exploration with LP-based polishing (cutting plane),
exact subgradient line search, and multi-scale warm starts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csc_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full cross-convolution a * b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def compute_objective(a):
    """Compute the objective value 2*n*max_conv/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def run_fista(a_init, n, S, max_time, rng, mode="hybrid",
              beta_init=10.0, beta_rate=0.005, beta_max=1000.0):
    """FISTA with adaptive smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Gradient selection
        if mode == "hybrid" and it > 200 and (it % 50 == 0 or rng.random() < 0.01):
            grad = exact_subgradient(y, c)
        elif mode == "exact":
            grad = exact_subgradient(y, c)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        if mode == "exact" or (mode == "hybrid" and np.array_equal(grad, exact_subgradient(y, c))):
            step = S / (gnorm * (1.0 + 0.01 * np.log1p(it / 100.0)))
        else:
            step = 1.0 / (2.0 * beta)

        # Adaptive: reduce step if stuck
        if it - last_improve > 3000:
            step *= 0.3
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 10000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
            last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def lp_direction(a, M, max_time, max_planes=500):
    """
    Find direction g maximizing sum(g) subject to:
      g >= 0, convolve(a, g)[k] <= M for all k.
    Uses cutting plane method.
    Returns g (the LP solution) or None if failed.
    """
    n = len(a)
    active = []  # list of active constraint indices
    
    # Start with the most active constraints
    c_cross = fft_cross_convolve(a, np.ones(n) / n)
    # Actually, let's start with the constraints most likely to be active
    # The peak of convolve(a, ones) is at the lag where a is most "concentrated"
    # Let's just start with a few constraints near the expected peak
    
    # Initial active set: constraints near the center
    center = n - 1
    for k in range(max(0, center - 10), min(2*n-1, center + 11)):
        active.append(k)
    
    active = list(set(active))
    
    g_solution = None
    for iteration in range(max_planes):
        if time.monotonic() > max_time:
            break
        
        # Build constraint matrix for active constraints
        # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
        n_active = len(active)
        A_rows = []
        A_cols = []
        A_vals = []
        
        for idx, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for j in range(lo, hi + 1):
                A_rows.append(idx)
                A_cols.append(j)
                A_vals.append(a[k - j])
        
        if not A_rows:
            g_solution = np.ones(n)
            break
        
        A_mat = lil_matrix((n_active, n))
        for r, c, v in zip(A_rows, A_cols, A_vals):
            A_mat[r, c] = v
        A_mat = A_mat.tocsr()
        
        # Solve LP: max sum(g) s.t. A_mat @ g <= M, g >= 0
        c_obj = -np.ones(n)  # minimize -sum(g)
        res = linprog(c_obj, A_ub=A_mat, b_ub=np.full(n_active, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max(1, (max_time - (time.monotonic()))) * 0.8,
                               'presolve': True})
        
        if not res.success:
            break
        
        g_solution = res.x
        
        # Check which constraints are most violated
        cross = fft_cross_convolve(a, g_solution)
        violations = cross - M
        if violations.max() < 1e-6 * max(1.0, M):
            # All constraints satisfied (approximately)
            break
        
        # Add the most violated constraint
        worst_k = int(np.argmax(violations))
        if worst_k not in active:
            active.append(worst_k)
            
        # Also add a few near the worst
        for dk in [-2, -1, 1, 2]:
            k2 = worst_k + dk
            if 0 <= k2 < 2*n-1 and k2 not in active:
                active.append(k2)
    
    return g_solution


def lp_polish(a_init, n, S, max_time, rng):
    """Polish using LP-based direction finding."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = compute_objective(a)
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < max_time - 5:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()
        
        # Find LP direction
        remaining_lp = min(30.0, (max_time - (time.monotonic() - start)) * 0.3)
        g = lp_direction(a, M, start + remaining_lp)
        
        if g is None or g.max() < 1e-15:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        
        # Rescale g to have sum S
        g = g * (S / g_sum)
        
        # Blend
        t = min(0.3, 0.5 / (1 + 0.1 * it))
        a_new = normalize_sum((1 - t) * a + t * g, n)
        
        # Check if it improved
        obj_new = compute_objective(a_new)
        if obj_new < best:
            best = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller t
            t = t * 0.3
            a_new = normalize_sum((1 - t) * a + t * g, n)
            obj_new = compute_objective(a_new)
            if obj_new < best:
                best = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                # LP direction didn't help, try a different approach
                if t < 0.01:
                    break
    
    return best_a, best


def subgradient_polish(a_init, n, S, max_time, rng):
    """Polish using exact subgradient with backtracking line search."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = compute_objective(a)
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < max_time - 2:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        
        grad = exact_subgradient(a, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = S / (gnorm * (1.0 + 0.05 * np.log1p(it / 50.0)))
        improved = False
        for _ in range(20):
            a_trial = project_simplex(a - step * grad, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur - 1e-12:
                a = a_trial
                cur = c_trial.max()
                improved = True
                if cur < best:
                    best = cur
                    best_a = a.copy()
                break
            step *= 0.7
        
        if not improved:
            # Inject small noise to escape
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            if it > 100:
                break
    
    return best_a, best

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: FISTA with hybrid LSE/exact-subgradient for exploration,
LP-based (cutting plane) direction finding for polishing, and
exact subgradient with line search for final refinement.
Uses the provided sequence as a warm start when available.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import lil_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def compute_objective(a):
    """Compute the objective value 2*n*max_conv/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def run_fista(a_init, n, S, max_time, rng, mode="hybrid",
              beta_init=10.0, beta_rate=0.006, beta_max=2000.0):
    """FISTA with adaptive smoothing and hybrid gradient."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Gradient selection with hybrid strategy
        use_exact = False
        if mode == "exact":
            use_exact = True
        elif mode == "hybrid":
            # Use exact subgradient periodically and when LSE might be too smooth
            if it > 100 and (it % 30 == 0 or beta > 200):
                use_exact = True
            elif it > 500 and rng.random() < min(0.15, 0.002 * it):
                use_exact = True

        if use_exact:
            grad = exact_subgradient(y, c)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = S / (gnorm * (1.0 + 0.02 * np.log1p(it / 200.0)))
        else:
            grad = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta)

        # Adaptive step reduction when stuck
        stall = it - last_improve
        if stall > 2000:
            step *= 0.5
            if stall > 5000:
                noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stall / 20000.0)
                if noise_scale > 1e-10:
                    y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                    t_mom = 1.0
            last_improve = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def solve_lp_direction(a, M, max_time):
    """
    Find g maximizing sum(g) subject to g>=0 and convolve(a,g)[k] <= M.
    Uses cutting plane method with sparse LP.
    Returns g or None.
    """
    n = len(a)
    active = []

    # Initialize active set near the center of convolution
    center = n - 1
    for k in range(max(0, center - 20), min(2 * n - 1, center + 21)):
        active.append(k)
    active = list(set(active))

    g_solution = None
    start = time.monotonic()

    for iteration in range(80):
        if time.monotonic() - start > max_time:
            break

        n_active = len(active)
        if n_active > 300:
            # Keep only the most active constraints
            if g_solution is not None:
                cross = fft_cross_convolve(a, g_solution)
                violations = cross[M] if isinstance(M, np.ndarray) else (cross.max() - M)
                active = list(np.argsort(cross)[::-1][:n_active])
            else:
                break

        # Build sparse constraint matrix
        row_list = []
        col_list = []
        val_list = []

        for idx, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            for j in range(lo, hi + 1):
                row_list.append(idx)
                col_list.append(j)
                val_list.append(a[k - j])

        if not row_list:
            g_solution = np.ones(n)
            break

        A_mat = lil_matrix((n_active, n), dtype=np.float64)
        for r, c, v in zip(row_list, col_list, val_list):
            A_mat[r, c] = v
        A_mat = A_mat.tocsr()

        # Solve LP: max sum(g) s.t. A_mat @ g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(n_active, M)

        lp_time_limit = max(2.0, (max_time - (time.monotonic() - start)) * 0.5)
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': lp_time_limit, 'presolve': True})

        if not res.success or res.x is None:
            break

        g_solution = res.x

        # Check constraint satisfaction
        cross = fft_cross_convolve(a, g_solution)
        max_violation = cross.max() - M
        if max_violation < 1e-8 * max(1.0, M):
            break

        # Add most violated constraints
        worst_indices = np.argsort(cross)[::-1]
        added = 0
        for k_idx in worst_indices:
            k_val = int(k_idx)
            if k_val not in active and added < 10:
                active.append(k_val)
                added += 1

    return g_solution


def lp_polish(a_init, n, S, max_time, rng):
    """Polish using LP-based direction finding with line search."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = compute_objective(a)
    best_a = a.copy()

    it = 0
    while time.monotonic() - start < max_time - 10:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()

        # Find LP direction
        lp_budget = min(20.0, (max_time - (time.monotonic() - start)) * 0.2)
        g = solve_lp_direction(a, M, lp_budget)

        if g is None or g.max() < 1e-15 or g.sum() < 1e-15:
            break

        # Rescale g to match normalization
        g = g * (S / g.sum())

        # Try blending with decreasing t
        found = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_trial = normalize_sum((1 - t) * a + t * g, n)
            obj_trial = compute_objective(a_trial)
            if obj_trial < best - 1e-12:
                a = a_trial
                best = obj_trial
                best_a = a.copy()
                found = True
                break

        if not found:
            # LP direction didn't help; try small perturbation
            noise = 0.001 * S / np.sqrt(n)
            a_trial = project_simplex(a + rng.normal(0, noise, n), S)
            obj_trial = compute_objective(a_trial)
            if obj_trial < best:
                a = a_trial
                best = obj_trial
                best_a = a.copy()
            elif it > 15:
                break

    return best_a, best


def subgradient_polish(a_init, n, S, max_time, rng):
    """Polish using exact subgradient with backtracking line search."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = compute_objective(a)
    best_a = a.copy()

    it = 0
    while time.monotonic() - start < max_time - 5:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()

        grad = exact_subgradient(a, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = S / (gnorm * (1.0 + 0.03 * np.log1p(it / 100.0)))
        improved = False
        for _ in range(25):
            a_trial = project_simplex(a - step * grad, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur * (1.0 - 1e-10):
                a = a_trial
                cur = c_trial.max()
                if cur < best:
                    best = cur
                    best_a = a.copy()
                improved = True
                break
            step *= 0.6

        if not improved:
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            if it > 200:
                break

    return best_a, best


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2))
    inits.append(np.maximum(1.0 - np.abs(2 * tt - 1), 0.01))

    # Multi-sinusoidal
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * 3 * tt + 0.5) + 0.15 * np.cos(2 * np.pi * 7 * tt))
    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * 1 * tt) + 0.2 * np.sin(2 * np.pi * 4 * tt + 1.0))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.5, n)))

    # Bump structures
    for pos in [0.3, 0.5, 0.7]:
        inits.append(np.exp(-30.0 * (tt - pos) ** 2) + 0.05)

    return [np.maximum(x, 0.001) for x in inits]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    
    Phases:
    1. FISTA exploration on given sequence (if available)
    2. FISTA on diverse initializations
    3. LP-based (cutting plane) direction polishing
    4. Exact subgradient with line search
    5. Final FISTA micro-polish
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + budget_s - 15.0

    global_best = np.inf
    global_a = None

    # Load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
            given_obj = compute_objective(given)
            print("Given sequence: n=%d, obj=%.6f" % (len(given), given_obj))
            if given_obj < global_best:
                global_best = given_obj
                global_a = given.copy()
    except Exception as e:
        print("Could not load given sequence: %s" % str(e))

    # Determine main length
    if given is not None and 2048 <= len(given) <= 32768:
        main_n = len(given)
    else:
        main_n = 8192

    n = main_n
    S = np.sqrt(2.0 * n)
    print("Main length n=%d, initial best=%.6f, budget=%.0fs" % (n, global_best, budget_s))

    # === Phase 1: FISTA on given sequence ===
    if global_a is not None:
        p1_time = min(500.0, (deadline - time.monotonic()) * 0.35)
        print("=== Phase 1: FISTA on given seq (%.0fs) ===" % p1_time)
        a1, s1 = run_fista(global_a, n, S, p1_time, rng,
                           mode="hybrid", beta_init=10.0, beta_rate=0.007, beta_max=3000.0)
        a1 = normalize_sum(a1, n)
        obj1 = compute_objective(a1)
        print("  Phase 1: obj=%.6f" % obj1)
        if obj1 < global_best:
            global_best = obj1
            global_a = a1.copy()

    # === Phase 2: FISTA on diverse inits ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p2_time = min(300.0, remaining * 0.2)
        inits = generate_initializations(n, rng)
        per_init = p2_time / len(inits)
        print("=== Phase 2: Diverse FISTA (%.0fs, %d inits) ===" % (p2_time, len(inits)))

        for i, init in enumerate(inits):
            if time.monotonic() > deadline - (deadline - time.monotonic()) * 0.5:
                break
            if per_init < 5:
                continue
            mode = "hybrid" if i % 2 == 0 else "lse"
            a_i, s_i = run_fista(init, n, S, per_init, rng, mode=mode,
                                 beta_init=5.0, beta_rate=0.01, beta_max=1500.0)
            a_i = normalize_sum(a_i, n)
            obj_i = compute_objective(a_i)
            if obj_i < global_best:
                global_best = obj_i
                global_a = a_i.copy()
                print("  Init %d: obj=%.6f" % (i, global_best))

    # === Phase 3: LP-based polishing ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_best = len(global_a)
            S_best = np.sqrt(2.0 * n_best)
            p3_time = min(500.0, remaining * 0.35)
            print("=== Phase 3: LP polishing (%.0fs) ===" % p3_time)
            a_lp, obj_lp = lp_polish(global_a, n_best, S_best, p3_time, rng)
            a_lp = normalize_sum(a_lp, n_best)
            obj_lp = compute_objective(a_lp)
            print("  LP polish: obj=%.6f" % obj_lp)
            if obj_lp < global_best:
                global_best = obj_lp
                global_a = a_lp.copy()

    # === Phase 4: Subgradient line search ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            n_best = len(global_a)
            S_best = np.sqrt(2.0 * n_best)
            p4_time = min(200.0, remaining * 0.3)
            print("=== Phase 4: Subgradient polish (%.0fs) ===" % p4_time)
            a_sg, obj_sg = subgradient_polish(global_a, n_best, S_best, p4_time, rng)
            a_sg = normalize_sum(a_sg, n_best)
            obj_sg = compute_objective(a_sg)
            print("  Subgrad: obj=%.6f" % obj_sg)
            if obj_sg < global_best:
                global_best = obj_sg
                global_a = a_sg.copy()

    # === Phase 5: Final FISTA micro-polish ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_best = len(global_a)
            S_best = np.sqrt(2.0 * n_best)
            p5_time = remaining * 0.85
            print("=== Phase 5: Final micro-polish (%.0fs) ===" % p5_time)
            a_fin, s_fin = run_fista(global_a, n_best, S_best, p5_time, rng,
                                     mode="hybrid", beta_init=100.0,
                                     beta_rate=0.004, beta_max=8000.0)
            a_fin = normalize_sum(a_fin, n_best)
            obj_fin = compute_objective(a_fin)
            print("  Final: obj=%.6f" % obj_fin)
            if obj_fin < global_best:
                global_best = obj_fin
                global_a = a_fin.copy()

    # Final verification and return
    if global_a is not None:
        final = np.clip(global_a, 0.0, 1000.0)
        final_list = [float(x) for x in final]
        assert all(x >= 0 for x in final_list)
        assert all(np.isfinite(x) for x in final_list)
        assert len(final_list) > 0
        assert sum(final_list) > 0.01
        print("Final: length=%d, score=%.6f" % (len(final_list), global_best))
        return final_list

    # Fallback
    n_fb = 256
    a_fb = np.ones(n_fb) * np.sqrt(2 * n_fb) / n_fb
    print("Fallback uniform sequence, n=%d" % n_fb)
    return [float(x) for x in a_fb]
