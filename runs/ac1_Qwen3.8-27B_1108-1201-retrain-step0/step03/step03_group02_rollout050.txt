"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Hybrid subgradient + local search, starting from parent solution.
- Phase 1: Projected subgradient descent with adaptive step size
- Phase 2: Random coordinate perturbations (local search)
- Phase 3: Fine-grained polishing
Uses FFT for efficient convolution. Explores multiple sequence lengths.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    # If sum is already S and all non-negative, return as is
    if v.min() >= 0 and abs(v.sum() - S) < 1e-12:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_toppk(a, c, top_k):
    """
    Subgradient of max(c) using top-k peaks.
    grad[i] = sum_{k in top_k} a[k-i] + a[i-k] (with zero padding)
    """
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    # Weight each peak by how much it exceeds the (top_k+1)th
    if top_k < len(c) - 1:
        threshold = c[np.argsort(c)[top_k]]
        excess = c[top_indices] - threshold
        excess = np.maximum(excess, 0)
        if excess.sum() < 1e-30:
            weights = np.ones(top_k) / top_k
        else:
            weights = excess / excess.sum()
    else:
        weights = np.ones(top_k) / top_k
    
    # Compute gradient via FFT cross-correlation
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * grad

def run_local_search(a_init, n, S, max_time, rng):
    """
    Local search: randomly perturb small groups of coordinates.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve(a)
        cur = c.max()
        if cur < best_score - 1e-15:
            best_score = cur
            best_local = a.copy()
        
        # Random perturbation: pick a random window and redistribute
        window_size = rng.integers(4, 64)
        start_idx = rng.integers(0, n - window_size)
        
        # Save original values
        orig = a[start_idx:start_idx + window_size].copy()
        orig_sum = orig.sum()
        
        if orig_sum < 1e-15:
            continue
        
        # Random redistribution
        mode = rng.random()
        if mode < 0.3:
            # Uniform redistribution
            new_vals = np.ones(window_size) * orig_sum / window_size
        elif mode < 0.6:
            # Exponential redistribution
            alpha = rng.random() * 3.0
            new_vals = np.exp(alpha * np.linspace(0, 1, window_size))
            new_vals = new_vals * (orig_sum / new_vals.sum())
        elif mode < 0.8:
            # Random
            new_vals = rng.random(window_size)
            new_vals = new_vals * (orig_sum / new_vals.sum())
        else:
            # Gaussian perturbation
            new_vals = orig + rng.normal(0, orig_sum / window_size, window_size)
            new_vals = np.clip(new_vals, 0, None)
            if new_vals.sum() < 1e-15:
                continue
            new_vals = new_vals * (orig_sum / new_vals.sum())
        
        a[start_idx:start_idx + window_size] = new_vals
        a = np.clip(a, 0, 1000)
        # Renormalize
        s = a.sum()
        a = a * (S / s)
    
    return best_local, best_score


def run_momentum_descent(a_init, n, S, max_time, rng, peak_k=3):
    """
    Nesterov-accelerated gradient descent (FISTA-like) with top-peak subgradient.

    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 1.0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve(y)
        cur = c.max()
        if cur < best_score - 1e-15:
            best_score = cur
            best_local = y.copy()
        
        grad = subgradient_toppk(y, c, peak_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        a_new = project_simplex(y - step * grad / gnorm, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        # Renormalize y to maintain sum = S
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = a_new
        t = t_new
        
        # Adapt step
        if it % 100 == 0:
            step *= 0.95
            if step < 1e-10:
                break
        
        # Occasional noise
        if it % 500 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
    
    return best_local, best_score

import numpy as np

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_soft_grad(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a.
    Uses softmax weights on the convolution values.
    """
    n = len(a)
    c_shifted = c - c.max()
    exp_bc = np.exp(beta * c_shifted)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

    Subgradient using top-k peaks with proportional weighting.

    if k < len(c):
        threshold = c[np.argsort(c)[k]]
    else:
        threshold = 0.0
    excess = c[top_indices] - threshold
    excess = np.maximum(excess, 1e-30)
    weights = excess / excess.sum()

    w_full = np.zeros(2 * n - 1)
    for i, k_idx in enumerate(top_indices):
        w_full[int(k_idx)] = weights[i]

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista_phase(a_init, n, S, max_time, rng, beta_start=5.0, beta_end=300.0):
    """
    Phase 1: FISTA with adaptive soft-max.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: linear interpolation from beta_start to beta_end
        frac = min(1.0, it / max(1, int(max_time * 80)))
        beta = beta_start + (beta_end - beta_start) * frac

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        grad = compute_soft_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: 1/(L*sqrt(t)) where L ~ 2*beta*S
        step = S / (beta * np.sqrt(1.0 + it * 0.01))
        step = max(step, S * 1e-8)

        a_new = project_simplex(y - step * grad, S)

        # Verify: if objective increased significantly, shrink step
        if it % 5 == 0:
            c_new = fft_convolve_self(a_new)
            if c_new.max() > cur * 1.1:
                a_new = project_simplex(y - 0.3 * step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = a_new
        t = t_new

        # Noise for escape (geometrically decreasing)
        if stuck_count > 500 and rng.random() < 0.1:
            noise_scale = 0.01 * S / np.sqrt(n) * (stuck_count / 1000.0) ** 0.5
            noise = rng.normal(0, min(noise_scale, 0.05 * S / np.sqrt(n)), n)
            y = project_simplex(y + noise, S)
            t = 1.0
            stuck_count = 0

        # Full restart if very stuck
        if stuck_count > 3000:
            noise = rng.normal(0, 0.02 * S / np.sqrt(n), n)
            y = project_simplex(best_local + noise, S)
            a = y.copy()
            t = 1.0
            stuck_count = 0

    return best_local, best_score


def run_momentum_phase(a_init, n, S, max_time, rng, peak_k=3):
    """
    Phase 2: Momentum (Nesterov) subgradient on top peaks with backtracking.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (4.0 * np.sqrt(n))
    max_step = S / (2.0 * np.sqrt(n))
    min_step = S * 1e-7
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        grad = compute_toppk_grad(y, c, peak_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Attempt step with backtracking
        step_try = step
        accepted = False
        for _ in range(5):
            a_new = project_simplex(y - step_try * grad / gnorm, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() <= cur * 1.001:
                accepted = True
                break
            step_try *= 0.5

        if not accepted:
            # Just do a half step
            a_new = project_simplex(y - 0.5 * step * grad / gnorm, S)
            step *= 0.7
        else:
            # Good step, maybe increase slightly
            if c_new.max() < cur * 0.999:
                step = min(max_step, step * 1.02)
            step = max(min_step, step)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = a_new
        t = t_new

        # Noisy restart
        if no_improve > 2000 and rng.random() < 0.05:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
            no_improve = 0
            step = min(max_step, step * 1.5)

        # Adaptive peak_k: increase when stuck
        if no_improve > 1000:
            peak_k = min(32, peak_k + 1)
        elif no_improve < 100:
            peak_k = max(2, peak_k - 1)

    return best_local, best_score


def run_fine_polish(a_init, n, S, max_time, rng):
    """
    Phase 3: Very fine subgradient on the single top peak.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (10.0 * np.sqrt(n))
    min_step = S * 1e-9

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()

        # Use top 1-2 peaks
        k_top = min(2, len(c))
        grad = compute_toppk_grad(y, c, k_top)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        a_new = project_simplex(y - step * grad / gnorm, S)
        c_new = fft_convolve_self(a_new)

        if c_new.max() > cur:
            step *= 0.5
            if step < min_step:
                # Try tiny noise
                noise = rng.normal(0, min_step * 0.1, n)
                a_new = project_simplex(y + noise, S)
                c_new = fft_convolve_self(a_new)
        else:
            step = min(step * 1.05, S / (4.0 * np.sqrt(n)))

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = a_new
        t = t_new

    return best_local, best_score


def run_local_perturb(a_init, n, S, max_time, rng):
    """
    Local search via random window perturbations with acceptance criterion.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    c = fft_convolve_self(a)
    best_score = c.max()
    base_score = best_score

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        cur = best_score

        # Pick a random window
        wsize = rng.integers(8, min(128, n // 4))
        wstart = rng.integers(0, n - wsize)

        orig = a[wstart:wstart + wsize].copy()
        orig_sum = orig.sum()
        if orig_sum < 1e-15:
            continue

        # Generate perturbation
        mode = rng.random()
        if mode < 0.25:
            # Smooth exponential
            alpha = rng.uniform(-3.0, 3.0)
            new_vals = np.exp(alpha * np.linspace(0, 1, wsize))
        elif mode < 0.5:
            # Linear ramp
            slope = rng.uniform(-2.0, 2.0)
            new_vals = 1.0 + slope * np.linspace(0, 1, wsize)
        elif mode < 0.75:
            # Gaussian bump
            center = rng.uniform(0, wsize - 1)
            width = rng.uniform(0.1, 0.5) * wsize
            x = np.linspace(0, wsize - 1, wsize)
            new_vals = 1.0 + rng.uniform(0.5, 2.0) * np.exp(-((x - center) ** 2) / (2 * width ** 2))
        else:
            # Pure random
            new_vals = rng.random(wsize)

        new_vals = np.maximum(new_vals, 0)
        if new_vals.sum() < 1e-15:
            continue
        new_vals = new_vals * (orig_sum / new_vals.sum())

        # Apply and check
        a[wstart:wstart + wsize] = new_vals
        a = np.clip(a, 0, 1000)
        a = a * (S / max(a.sum(), 1e-15))

        c_new = fft_convolve_self(a)
        if c_new.max() < cur - 1e-12:
            best_score = c_new.max()
            best_local = a.copy()
        else:
            # Revert
            a[wstart:wstart + wsize] = orig
            a = a * (S / max(a.sum(), 1e-15))

    return best_local, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: FISTA soft-max -> momentum subgradient -> fine polish.
    Starts from parent solution, explores alternative lengths if time permits.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Load parent
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            parent = np.clip(given, 0, 1000)
            print(f"Parent loaded: n={len(parent)}, sum={parent.sum():.4f}")
    except Exception:
        pass

    # Determine primary length
    if parent is not None:
        n_primary = len(parent)
    else:
        n_primary = 8192

    def normalize(a, n):
        """Clip and normalize to sum = sqrt(2n)."""
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        S = np.sqrt(2.0 * float(n))
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) * S / n
        return a * (S / s)

    def score_of(a):
        """Compute objective assuming sum(a) = sqrt(2n)."""
        c = fft_convolve_self(a)
        return c.max()

    # === OPTIMIZE PRIMARY LENGTH ===
    n = n_primary
    S = np.sqrt(2.0 * float(n))

    if parent is not None:
        a_start = normalize(parent, n)
    else:
        tt = np.linspace(0, 1, n, endpoint=False)
        a_start = normalize(np.exp(2.0 * tt) + 0.5 * rng.random(n), n)

    s0 = score_of(a_start)
    print(f"Initial score (n={n}): {s0:.6f}")
    global_best = s0
    global_a = a_start.copy()

    remaining = deadline - time.monotonic()
    if remaining < 30:
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            return final_list
        return [1.0] * 8192

    # Phase 1: FISTA soft-max (35% of time)
    phase1_time = remaining * 0.35
    print(f"\n--- Phase 1: FISTA soft-max ({phase1_time:.0f}s, n={n}) ---")
    t0 = time.monotonic()
    a1, s1 = run_fista_phase(a_start, n, S, phase1_time, rng,
                              beta_start=5.0, beta_end=400.0)
    a1 = normalize(a1, n)
    s1_actual = score_of(a1)
    elapsed = time.monotonic() - t0
    print(f"  Phase 1 done in {elapsed:.1f}s: score={s1_actual:.6f}")

    if s1_actual < global_best:
        global_best = s1_actual
        global_a = a1.copy()

    # Phase 2: Momentum subgradient on top peaks (35% of time)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        phase2_time = remaining * 0.35
        print(f"\n--- Phase 2: Momentum subgradient ({phase2_time:.0f}s, n={n}) ---")
        t0 = time.monotonic()
        a2, s2 = run_momentum_phase(a1, n, S, phase2_time, rng, peak_k=3)
        a2 = normalize(a2, n)
        s2_actual = score_of(a2)
        elapsed = time.monotonic() - t0
        print(f"  Phase 2 done in {elapsed:.1f}s: score={s2_actual:.6f}")

        if s2_actual < global_best:
            global_best = s2_actual
            global_a = a2.copy()

    # Phase 3: Fine polish (20% of time)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        phase3_time = remaining * 0.25
        print(f"\n--- Phase 3: Fine polish ({phase3_time:.0f}s, n={n}) ---")
        t0 = time.monotonic()
        a3, s3 = run_fine_polish(global_a, n, S, phase3_time, rng)
        a3 = normalize(a3, n)
        s3_actual = score_of(a3)
        elapsed = time.monotonic() - t0
        print(f"  Phase 3 done in {elapsed:.1f}s: score={s3_actual:.6f}")

        if s3_actual < global_best:
            global_best = s3_actual
            global_a = a3.copy()

    # Phase 4: Local perturbation search (15% of time)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        phase4_time = remaining * 0.3
        print(f"\n--- Phase 4: Local perturbation ({phase4_time:.0f}s, n={n}) ---")
        t0 = time.monotonic()
        a4, s4 = run_local_perturb(global_a, n, S, phase4_time, rng)
        a4 = normalize(a4, n)
        s4_actual = score_of(a4)
        elapsed = time.monotonic() - t0
        print(f"  Phase 4 done in {elapsed:.1f}s: score={s4_actual:.6f}")

        if s4_actual < global_best:
            global_best = s4_actual
            global_a = a4.copy()

    # Phase 5: Another round of fine polish on the best
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print(f"\n--- Phase 5: Final polish ({remaining:.0f}s, n={n}) ---")
        t0 = time.monotonic()
        a5, s5 = run_fine_polish(global_a, n, S, remaining * 0.6, rng)
        a5 = normalize(a5, n)
        s5_actual = score_of(a5)
        elapsed = time.monotonic() - t0
        print(f"  Phase 5 done in {elapsed:.1f}s: score={s5_actual:.6f}")

        if s5_actual < global_best:
            global_best = s5_actual
            global_a = a5.copy()

    # Phase 6: Try a different length if time remains
    remaining = deadline - time.monotonic()
    if remaining > 120:
        alt_n = n * 2 if n <= 16384 else n // 2
        if alt_n >= 1024 and alt_n <= 65536 and alt_n != n:
            # Interpolate parent to new length
            if parent is not None:
                x_old = np.linspace(0, 1, len(parent))
                x_new = np.linspace(0, 1, alt_n)
                # Simple linear interpolation
                a_alt = np.interp(x_new, x_old, parent)
            else:
                tt = np.linspace(0, 1, alt_n, endpoint=False)
                a_alt = np.exp(2.0 * tt) + 0.5 * rng.random(alt_n)

            a_alt = normalize(a_alt, alt_n)
            s_alt = score_of(a_alt)
            S_alt = np.sqrt(2.0 * float(alt_n))
            print(f"\n--- Phase 6: Try n={alt_n} (initial score={s_alt:.6f}) ---")

            alt_time = remaining * 0.7
            a6, s6 = run_momentum_phase(a_alt, alt_n, S_alt, alt_time * 0.5, rng, peak_k=3)
            a6 = normalize(a6, alt_n)
            s6_actual = score_of(a6)
            print(f"  n={alt_n} momentum: score={s6_actual:.6f}")

            remaining2 = deadline - time.monotonic()
            if remaining2 > 20 and s6_actual < s_alt:
                a7, s7 = run_fine_polish(a6, alt_n, S_alt, remaining2 * 0.5, rng)
                a7 = normalize(a7, alt_n)
                s7_actual = score_of(a7)
                print(f"  n={alt_n} polish: score={s7_actual:.6f}")
                if s7_actual < s6_actual:
                    s6_actual = s7_actual
                    a6 = a7

            if s6_actual < global_best:
                global_best = s6_actual
                global_a = a6.copy()
                print(f"  New global best from n={alt_n}: {global_best:.6f}")

    # Final verification
    print(f"\n=== FINAL: global_best={global_best:.6f}, n={len(global_a)} ===")

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)

    if np.isfinite(ver):
        print(f"Verified score: {ver:.6f}")
        return final_list
    else:
        print("Verification failed, using fallback")
        return [1.0] * 8192
