"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Bathtub initialization + multi-temperature FISTA + direct peak manipulation + symmetry exploitation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    return result


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad(a, weights):
    """Compute gradient of sum_k weights[k]*c[k] where c = convolve(a,a)."""
    n = len(a)
    # Need fsize >= len(weights) + n - 1
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(np.conj(W) * A))[:n]
    return 2.0 * cross


def softmax_weights(c, beta):
    """Softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Top-k normalized weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights


def bathtub_profile(n, alpha=0.5):
    """Create a bathtub-shaped profile (high at edges, low in center)."""
    j = np.arange(n)
    center = (n - 1) / 2.0
    # Distance from center, normalized
    d = np.abs(j - center) / center
    # Profile: higher at edges (d=1), lower at center (d=0)
    profile = 1.0 + alpha * (d ** 2)
    return profile


def init_bathtub(n, S, alpha=0.5):
    """Initialize with bathtub profile, normalized to sum = S."""
    profile = bathtub_profile(n, alpha)
    a = profile / profile.sum() * S
    return a


def make_symmetric(a):
    """Make a sequence symmetric: a[j] = a[n-1-j]."""
    n = len(a)
    half = n // 2
    b = (a[:half] + a[n-1-half:-half-1:-1]) / 2.0
    if n % 2 == 0:
        return np.concatenate([b, b[::-1]])
    else:
        mid = a[half]
        return np.concatenate([b, [mid], b[::-1]])


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def objective_value(a, n, S):
    """Compute the objective 2*n*max_conv/S^2."""
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def run_fista_multi_temp(a_init, n, S, max_time, rng, 
                          beta_start=20.0, beta_end=2000.0,
                          use_symmetry=True):
    """FISTA with adaptive temperature and optional symmetry."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    if use_symmetry:
        a = make_symmetric(a)
        a = safe_normalize(a, n)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    halt = False
    
    while time.monotonic() - start < max_time and not halt:
        it += 1
        # Schedule beta from beta_start to beta_end
        frac = min(1.0, it / max_time / 3.0)  # roughly 3 seconds per beta step
        beta = beta_start * (beta_end / beta_start) ** frac
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
        
        w = softmax_weights(c, beta)
        grad = compute_grad(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: 1/L where L is Lipschitz constant estimate
        # L ≈ 2 * max |c| * n (rough)
        step = 2.0 * S / (n * max(1.0, gnorm))

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences of length n.
Multi-start FISTA with bathtub initialization and adaptive temperature.
Exploits the insight that optimal sequences have a "bathtub" profile
(high at edges, low in center) to minimize the peak of self-convolution.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad(a, weights):
    """Gradient of sum_k weights[k]*c[k] where c=convolve(a,a)."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(np.conj(W) * A))[:n]
    return 2.0 * cross


def softmax_weights(c, beta):
    """Softmax weights."""
    c_max = c.max()
    exp_bc = np.exp(np.clip(beta * (c - c_max), -700, 0))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Top-k normalized weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights


def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_bathtub(n, alpha):
    """Bathtub profile: high at edges, low at center."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    d = np.abs(j - center) / center  # 0 at center, 1 at edges
    profile = 1.0 + alpha * d
    return profile


def make_bathtub2(n, alpha):
    """Alternative bathtub: inverse power law from edge."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    d = np.abs(j - center) / center
    # 1/d would be inf at center, so use 1/(1-d*epsilon) or similar
    # Actually, we want high at edges (d=1) and low at center (d=0)
    # So: profile = 1/d^alpha would give high at edges, but inf at d=0
    # Use: profile = (1 + alpha*d) or profile = d^(-alpha) with clipping
    profile = 1.0 + alpha * (1.0 / (0.1 + d))
    return profile / profile.max() * (1.0 + alpha)


def run_fista(a_init, n, S, max_time, rng,
              beta_start=10.0, beta_end=5000.0,
              step_init=None, perturb_interval=3000, perturb_prob=0.3):
    """FISTA with adaptive temperature."""
    a = safe_normalize(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    if step_init is None:
        step_init = 1.0 / (2.0 * beta_start)
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = y.copy()
        
        # Choose weight function based on beta
        if beta < 100:
            w = softmax_weights(c, beta)
        else:
            # For high beta, use top-k with k growing slowly
            k = min(len(c), max(1, int(100.0 / (beta / 100.0))))
            w = toppk_weights(c, max(1, k))
        
        grad = compute_grad(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        # L (Lipschitz constant) ≈ 2 * max(w) * 2 * max(a) * n (rough)
        # Simpler: use 1/(2*L) where L ~ max_conv * n / S^2 * 2
        step = 2.0 * S * S / (n * max(1.0, 2.0 * cur))
        step = min(step, S / max(gnorm, 1e-15))
        
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        x_new = np.maximum(x_new, 0.0)
        
        # Nesterov
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic perturbation
        if it % perturb_interval == 0 and rng.random() < perturb_prob:
            sigma = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, sigma, n), S)
            t = 1.0
    
    return best_local, best_score


def run_toppk_polish(a_init, n, S, max_time, rng, top_k=1):
    """Very focused top-k optimization for final polish."""
    a = safe_normalize(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

        c = fft_convolve_self(y)
