```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding.

Key innovation: At each iteration, solve a linear program to find the optimal
direction that maximizes total mass while respecting the current peak constraint.
The LP dual variables provide optimal weights for subgradient descent.
This combines global constraint awareness with local optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import scipy.sparse as sp
from scipy.optimize import linprog


def fft_conv(a, b=None):
    """Full linear convolution via FFT. If b is None, compute a*a."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def build_conv_pattern(n):
    """Build row/col indices for the convolution matrix sparse pattern.
    A[m, k] = a[m-k] if valid, so A @ g = convolve(a, g)."""
    n_rows = 2 * n - 1
    m_vals = np.arange(n_rows)
    k_starts = np.maximum(0, m_vals - n + 1).astype(np.int64)
    k_ends = np.minimum(n - 1, m_vals).astype(np.int64)
    lengths = k_ends - k_starts + 1
    
    total_nz = int(lengths.sum())
    rows = np.repeat(m_vals, lengths)
    
    # Vectorized construction of cols
    # For row m, cols are k_starts[m], k_starts[m]+1, ..., k_ends[m]
    # Offset within each row's block: 0, 1, ..., length-1
    offsets = np.zeros(total_nz, dtype=np.int64)
    pos = 0
    for i in range(n_rows):
        l = lengths[i]
        offsets[pos:pos+l] = np.arange(l)
        pos += l
    cols = np.repeat(k_starts, lengths) + offsets
    
    return rows, cols, lengths


def solve_direction_lp(a, M, rows, cols, n):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns (g, dual_values) or (None, None) on failure."""
    n_rows = 2 * n - 1
    # A[m, k] = a[m-k] for valid (m,k)
    m_idx = rows
    k_idx = cols
    vals = a[m_idx - k_idx]
    
    # Build sparse matrix
    A = sp.csc_matrix((vals, (m_idx, k_idx)), shape=(n_rows, n))
    
    # LP: max c^T x s.t. A x <= b, x >= 0
    # linprog minimizes, so: min -c^T x s.t. A x <= b, x >= 0
    c_obj = -np.ones(n)
    b = np.full(n_rows, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        g = res.x
        # Get dual variables (shadow prices) for the inequality constraints
        # In HiGHS, res.ineqlin.marginals gives the dual values
        dual = -res.ineqlin.marginals  # Negate because linprog minimizes
        dual = np.maximum(dual, 0)  # Dual variables for <= constraints are <= 0 in min form
        # Actually, for min c^T x s.t. Ax <= b, the dual variables are >= 0
        # and equal to the negative of the shadow prices from the primal
        # Let me just use res.ineqlin.marginals directly
        dual = np.abs(res.ineqlin.marginals)
        return g, dual
    else:
        return None, None


def line_search_conv(a, g, S, max_iter=30):
    """Find t in [0, 1] that minimizes max(conv((1-t)a + t*g, (1-t)a + t*g)).
    Returns (best_t, best_peak)."""
    n = len(a)
    c_a = fft_conv_self(a)
    c_ag = fft_conv(a, g)
    c_g = fft_conv_self(g)
    
    def peak_at_t(t):
        # conv((1-t)a + t*g, (1-t)a + t*g) 
        # = (1-t)^2 conv(a,a) + 2t(1-t) conv(a,g) + t^2 conv(g,g)
        return ((1-t)**2) * c_a + 2*t*(1-t) * c_ag + (t**2) * c_g
    
    # Golden section search on [0, 1]
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    lo = 0.0
    hi = 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = peak_at_t(c1).max()
    f2 = peak_at_t(c2).max()
    
    for _ in range(max_iter):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = peak_at_t(c1).max()
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = peak_at_t(c2).max()
    
    t = (lo + hi) / 2
    peak = peak_at_t(t).max()
    return t, peak


def coordinate_descent(a, n, S, time_budget, max_sweeps=20):
    """Fast coordinate descent for final polishing."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()
    
    c = fft_conv_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(42)
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            original = a[i]
            if original < 1e-12:
                continue
            
            best_local = original
            best_local_obj = best_obj
            
            for factor in [0.7, 1.3, 0.5, 1.5, 0.85, 1.15]:
                a[i] = original * factor
                a_norm = normalize_nonneg(a, S)
                c_check = fft_conv_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-10:
                    best_local_obj = obj_check
                    best_local = original * factor
            
            a[i] = best_local
            a = normalize_nonneg(a, S)
            c = fft_conv_self(a)
            
            if c.max() < best_obj - 1e-10:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def subgradient_step(a, c_weights, S, lr):
    """Take a subgradient step using given peak weights.
    c_weights: weights for each peak position (sum to 1).
    g[m] = 2 * sum_k c_weights[k] * a[k-m] (cross-correlation)
    """
    n = len(a)
    # g[m] = 2 * cross_corr(weights, a)[m]
    # Using FFT
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(c_weights, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    g = 2.0 * corr[:n]
    
    # Subgradient descent: a_new = a - lr * g, projected to {a >= 0, sum(a) = S}
    a_new = a - lr * g
    a_new = normalize_nonneg(a_new, S)
    return a_new
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding with
dual-guided subgradient descent, coordinate descent polishing, and adaptive
restarts. The LP provides globally-aware peak weights (dual variables) that
guide efficient descent, while subgradient steps provide fast local progress.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import scipy.sparse as sp
from scipy.optimize import linprog


def fft_conv(a, b):
    """Full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def build_conv_pattern(n):
    """Build row/col indices for convolution matrix. A[m,k]=a[m-k] if valid."""
    n_rows = 2 * n - 1
    m_vals = np.arange(n_rows)
    k_starts = np.maximum(0, m_vals - (n - 1)).astype(np.int64)
    k_ends = np.minimum(n - 1, m_vals).astype(np.int64)
    lengths = (k_ends - k_starts + 1).astype(np.int64)

    total_nz = int(lengths.sum())
    rows = np.repeat(m_vals, lengths)

    cum_lengths = np.concatenate([[0], np.cumsum(lengths)]).astype(np.int64)
    p = np.arange(total_nz)
    r = np.searchsorted(cum_lengths, p, side='right') - 1
    offsets = p - cum_lengths[r]
    cols = np.repeat(k_starts, lengths) + offsets

    return rows, cols


def solve_lp_direction(a, M, rows, cols, n, max_constraints=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns (g, dual) or (None, None)."""
    n_rows = 2 * n - 1

    if max_constraints is not None and max_constraints < n_rows:
        # Only use top constraints (rows with largest a values in that row)
        # For now, just use all
        pass

    m_idx = rows
    k_idx = cols
    vals = a[m_idx - k_idx]

    A = sp.csc_matrix((vals, (m_idx, k_idx)), shape=(n_rows, n))

    c_obj = -np.ones(n)
    b = np.full(n_rows, M)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            g = np.asarray(res.x)
            dual = np.abs(np.asarray(res.ineqlin.marginals))
            return g, dual
    except Exception:
        pass
    return None, None


def subgradient_direction(a, weights, n):
    """Compute subgradient: g[m] = 2 * cross_corr(weights, a)[m]."""
    fsize = 1 << (3 * n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def line_search_peak(a, direction, S, n, candidate_ts=None):

    best_t = 0.0
    best_peak = fft_conv_self(a).max()

    base = a

        a_trial = normalize_nonneg(a_trial, S)
        c_trial = fft_conv_self(a_trial)
        peak = c_trial.max()
        if peak < best_peak - 1e-12:

            best_t = t
    return best_t, best_peak

    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()

    c = fft_conv_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(777)

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break
            original = a[i]
            if original < 1e-12:
                continue

            best_local = original
            best_local_obj = best_obj

            for factor in [0.7, 1.3, 0.5, 1.5, 0.85, 1.15, 0.6, 1.4]:
                a[i] = original * factor
                a_norm = normalize_nonneg(a, S)
                c_check = fft_conv_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-11:
                    best_local_obj = obj_check
                    best_local = original * factor

            a[i] = best_local
            a = normalize_nonneg(a, S)
            c = fft_conv_self(a)

            if c.max() < best_obj - 1e-11:
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        if not improved:
            break

def run_exponential_subgradient(a, n, S, time_budget, T_init=30.0, T_final=500.0):
    """Subgradient with exponential peak weighting as backup."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_conv_self(a)
    best_obj = c.max()
    best_a = a.copy()

    step = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.6)

        c = fft_conv_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 60:
            noise = 0.004
            a = best_a + noise * np.random.default_rng(step + 1000).standard_normal(n)
            a = normalize_nonneg(a, S)
            no_improve = 0
            step += 1
            continue

        # Exponential weights
        c_max = cur_obj
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0

        a_trial = normalize_nonneg(a_trial, S)
        c_trial = fft_conv_self(a_trial)
        if c_trial.max() < cur_obj - 1e-12:
            a = a_trial

        step += 1

    return best_a, best_obj

    a = global_a.copy() if global_a is not None else np.full(n, S / n)
    a = normalize_nonneg(a, S)

    init_obj = evaluate_sequence(a.tolist())
    print("Starting: n=%d, S=%.4f, initial_evaluate=%s" % (
        n, S, "%.7f" % init_obj if np.isfinite(init_obj) else "inf"))

    # Pre-build convolution matrix pattern
    print("Building convolution pattern (%d non-zeros)..." % (n * (n - 1) + n))
    t0 = time.monotonic()
    rows, cols = build_conv_pattern(n)
    print("Pattern built in %.1fs" % (time.monotonic() - t0))

    # ============ PHASE 1: LP + Subgradient hybrid (55% time) ============
    p1_budget = 0.55 * total_budget
    print("\nPhase 1: LP+SG hybrid (budget=%.0fs)" % p1_budget)

    if p1_budget > 30:
        a_best, obj_best = run_lp_subgradient_hybrid(
            a, n, S, p1_budget, rows, cols,
            lp_every_s=6.0, verbose=True
        )
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("Phase 1 result: obj=%.7f" % ver)
        a = a_best
    else:
        a = normalize_nonneg(a, S)

    # ============ PHASE 2: Exponential subgradient (25% time) ============
    p2_budget = 0.25 * total_budget
    print("\nPhase 2: Exponential SG (budget=%.0fs)" % p2_budget)

    if global_a is not None and p2_budget > 15:
        a = global_a.copy()
        a = normalize_nonneg(a, S)
        a_best, obj_best = run_exponential_subgradient(
            a, n, S, p2_budget, T_init=50.0, T_final=800.0
        )
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("Phase 2 result: obj=%.7f" % ver)

    # ============ PHASE 3: Coordinate descent polish (15% time) ============
    p3_budget = 0.15 * total_budget
    print("\nPhase 3: Coordinate descent (budget=%.0fs)" % p3_budget)

    if global_a is not None and p3_budget > 10:
        a = global_a.copy()
        a = normalize_nonneg(a, S)
        a_best, obj_best = coordinate_descent(a, n, S, p3_budget, max_sweeps=25)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("Phase 3 result: obj=%.7f" % ver)

    # ============ PHASE 4: Remaining time - more LP + CD ============
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 20:
        n_cur = len(global_a)
        S_cur = np.sqrt(2.0 * float(n_cur))
        a = global_a.copy()
        a = normalize_nonneg(a, S_cur)

        # Alternate between LP-SG and CD
        half = remaining / 2.0

        if half > 10:
            print("\nPhase 4a: More LP+SG (%.0fs)" % half)
            a_best, _ = run_lp_subgradient_hybrid(
                a, n_cur, S_cur, half, rows, cols,
                lp_every_s=5.0, verbose=False
            )
            a_best = normalize_nonneg(a_best, S_cur)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  Phase 4a: obj=%.7f" % ver)

        remaining2 = deadline - time.monotonic()
        if remaining2 > 5:
            print("Phase 4b: Final CD (%.0fs)" % remaining2)
            a = global_a.copy()
            a = normalize_nonneg(a, S_cur)
            a_best, _ = coordinate_descent(a, n_cur, S_cur, remaining2, max_sweeps=15)
            a_best = normalize_nonneg(a_best, S_cur)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  Phase 4b: obj=%.7f" % ver)

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: internal_best=%.7f, verified=%s" % (
        global_best, "%.7f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)

    return final_a
```
