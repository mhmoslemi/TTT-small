def run_nesterov_pg(a_init, n, S, max_time, rng):
    """Nesterov-accelerated projected gradient descent with adaptive step."""
    a = np.clip(a_init, 1e-15, 10000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    beta = 50.0
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    step_base = 1.0 / (beta * S)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()

        grad = compute_grad_smoothed(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = step_base / (1.0 + 0.01 * beta)
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 10000 == 0:
            y = project_simplex(y + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            t = 1.0

    z = np.log(a / S + 1e-30)
    z = z - z.mean()

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        def fun(z_val):
            return softmax_obj_grad(z_val, beta, S)[0]

        def jac(z_val):
            return softmax_obj_grad(z_val, beta, S)[1]

        t0 = time.monotonic()
        try:
            result = scipy_minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_beta,
                         'maxfun': maxiter_per_beta * 3,
                         'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 50, 'iprint': -1}
            )
        except Exception:
            break

        z = result.x
        z = z - z.mean()

        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = S * exp_z / exp_z.sum()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * S / s

        c = fft_convolve_self(a)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if time.monotonic() - t0 > 0.5:
            pass  # continue

    return best_a, best_score


def run_coord_descent(a_init, n, S, max_time, rng):
    """Vectorized coordinate descent polish."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()

    order = rng.permutation(n)
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        j0 = int(order[(it - 1) % n])
        v = a[j0]

        if v < 1e-15:
            continue

        k_min = j0
        k_max = min(j0 + n - 1, 2 * n - 2)

        # Unaffected max
        unaffected_max = 0.0
        if k_min > 0:
            unaffected_max = float(c[:k_min].max())
        if k_max < 2 * n - 2:
            unaffected_max = max(unaffected_max, float(c[k_max + 1:].max()))

        # Affected lags
        c_aff = c[k_min:k_max + 1].copy()
        j_range = np.arange(k_min, k_max + 1, dtype=np.int64) - j0
        slopes = 2.0 * a[j_range].astype(np.float64)

        quad_idx = -1
        if 2 * j0 >= k_min and 2 * j0 <= k_max:
            quad_idx = 2 * j0 - k_min

        # Try several multiplicative factors
        factors = [0.05, 0.15, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.6, 2.0]
        best_peak = c.max()
        best_factor = 1.0

        for f in factors:
            if f == 1.0:
                continue
            v_new = v * f
            delta = v_new - v
            new_c_aff = c_aff + slopes * delta
            if quad_idx >= 0:
                new_c_aff[quad_idx] += delta * delta
            peak = max(unaffected_max, float(new_c_aff.max()))
            if peak < best_peak:
                best_peak = peak
                best_factor = f

        if best_factor != 1.0:
            delta = v * best_factor - v
            a[j0] = v * best_factor
            c[k_min:k_max + 1] = c_aff + slopes * delta
            if quad_idx >= 0:
                c[2 * j0] += delta * delta

            cur_peak = c.max()
            if cur_peak < best_score:
                best_score = cur_peak
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > n * 2:
            # Perturb to escape
            a = np.maximum(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), 0.0)
            c = fft_convolve_self(a)
            no_improve = 0
            order = rng.permutation(n)

    s = a.sum()
    if s > 1e-15:
        a = a * S / s
    c = fft_convolve_self(a)
    if c.max() < best_score:
        best_score = c.max()
        best_a = a.copy()

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def upscale_sequence(a_small, n_target):
    """Upscale a sequence to a larger length using cubic-like interpolation."""
    n_small = len(a_small)
    if n_small >= n_target:
        return a_small[:n_target].copy()
    x_small = np.linspace(0, 1, n_small, endpoint=False)
    x_target = np.linspace(0, 1, n_target, endpoint=False)
    # Use np.interp for linear interpolation (fast and stable)
    a_target = np.interp(x_target, x_small, a_small, left=a_small[0], right=a_small[-1])
    return a_target


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # --- Collect initializations ---
    inits = []

    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 128 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0.0, 1000.0)))
    except Exception:
        pass

    if inits:
        main_len = len(inits[0][1])
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * float(n))
    t = np.linspace(0, 1, n, endpoint=False)

    # Diverse asymmetric initializations
    inits.append(("rand", rng.random(n) + 0.05))

    # Asymmetric dual bump
    inits.append(("dual_bump",
                  np.exp(-0.5 * ((t - 0.25) / 0.12) ** 2) +
                  0.6 * np.exp(-0.5 * ((t - 0.75) / 0.15) ** 2) + 0.02))

    # Tapered oscillation (asymmetric)
    inits.append(("taper_osc",
                  (1.0 - 0.6 * t) * (1.0 + 0.3 * np.sin(2 * np.pi * 5 * t) +
                                      0.2 * np.cos(2 * np.pi * 11 * t)) + 0.05))

    # Chirp-like (quadratic phase amplitude)
    inits.append(("chirp_amp",
                  np.abs(1.0 + 0.4 * np.cos(2 * np.pi * (3 * t + 0.5 * t * t))) + 0.05))

    # Edge-concentrated asymmetric
    inits.append(("edge_left",
                  np.exp(-3.0 * t) * (1.0 + 0.3 * np.sin(2 * np.pi * 7 * t)) + 0.03))

    # High-pass structure
    inits.append(("highpass",
                  0.3 + 0.7 * t * (1 - t) * 4 * (1 + 0.2 * np.cos(2 * np.pi * 9 * t))))

    # Multi-frequency asymmetric
    inits.append(("multi_freq",
                  0.4 + 0.25 * np.sin(2 * np.pi * 2 * t + 0.5) +
                  0.2 * np.cos(2 * np.pi * 5 * t + 1.0) +
                  0.15 * np.sin(2 * np.pi * 13 * t + 2.0) +
                  0.1 * np.cos(2 * np.pi * 29 * t + 3.0) + 0.15))

    n_inits = len(inits)
    print("=== Optimization: n=%d, %d inits, budget=%.0fs ===" % (n, n_inits, budget_s))
    print("Target: 1.5027, Previous best: 1.508805")

    # --- Try smaller n first for quick pattern discovery ---
    small_n = n // 4
    if small_n >= 512 and time.monotonic() < deadline - 120:
        print("\n=== Small-n warmup (n=%d) ===" % small_n)
        S_small = np.sqrt(2.0 * float(small_n))
        t_small = np.linspace(0, 1, small_n, endpoint=False)
        small_inits = [
            np.exp(-0.5 * ((t_small - 0.3) / 0.1) ** 2) + 0.5 * np.exp(-0.5 * ((t_small - 0.7) / 0.12) ** 2) + 0.02,
            (1.0 - 0.5 * t_small) * (1.0 + 0.3 * np.sin(2 * np.pi * 4 * t_small)) + 0.05,
            0.4 + 0.3 * np.sin(2 * np.pi * 3 * t_small + 0.7) + 0.2 * np.cos(2 * np.pi * 8 * t_small) + 0.2,
        ]
        small_best_a = None
        small_best_score = np.inf
        per_small = min(8.0, (deadline - time.monotonic()) * 0.05 / len(small_inits))

        for i, si in enumerate(small_inits):
            if time.monotonic() > deadline - 100:
                break
            sa = safe_normalize(si, small_n)
            sc = fft_convolve_self(sa).max()
            if sc < small_best_score:
                small_best_score = sc
                small_best_a = sa.copy()
            sa, sc = run_lbfgs_continuation(
                sa, small_n, S_small, per_small, rng,
                beta_schedule=[10, 30, 100, 300, 1000], maxiter_per_beta=50)
            sa = safe_normalize(sa, small_n)
            sc = fft_convolve_self(sa).max()
            if sc < small_best_score:
                small_best_score = sc
                small_best_a = sa.copy()

        if small_best_a is not None:
            upsc = upscale_sequence(small_best_a, n)
            upsc_norm = safe_normalize(upsc, n)
            inits.append(("upsc_small", upsc_norm))
            print("  Small-n best score: %.6f" % small_best_score)

    print("\n=== Phase 1: Nesterov PG + L-BFGS (%d inits) ===" % n_inits)

    usable = deadline - time.monotonic()
    per_init = max(10.0, usable * 0.45 / n_inits)

    results = {}
    for name, arr in inits:
        if time.monotonic() > deadline - 180:
            break

        t0 = time.monotonic()
        a_norm = safe_normalize(arr, n)
        c = fft_convolve_self(a_norm)
        init_score = c.max()

        if init_score < global_best:
            global_best = init_score
            global_a = a_norm.copy()

        # Quick Nesterov PG
        pg_time = min(per_init * 0.3, 8.0)
        a_pg, pg_score = run_nesterov_pg(a_norm, n, S, pg_time, rng)
        a_pg = safe_normalize(a_pg, n)

        # L-BFGS continuation
        lbfgs_time = max(5.0, per_init - pg_time - 2)
        a_new, score = run_lbfgs_continuation(
            a_pg, n, S, lbfgs_time, rng,
            beta_schedule=[10, 30, 100, 300, 1000, 3000, 10000],
            maxiter_per_beta=150)
        a_new = safe_normalize(a_new, n)
        c_new = fft_convolve_self(a_new)
        final_score = c_new.max()

        results[name] = (a_new, final_score)

        if final_score < global_best:
            global_best = final_score
            global_a = a_new.copy()
            print("  [OK] %s: %.8f (init=%.4f, %.1fs)" %
                  (name, global_best, init_score, time.monotonic() - t0))
        else:
            print("  [%s: %.8f, best=%.6f, %.1fs]" %
                  (name, final_score, global_best, time.monotonic() - t0))

    print("\n=== After Phase 1: global_best=%.8f ===" % global_best)

    # --- Phase 2: Coordinate descent polish on best result ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            cd_time = min(remaining * 0.25, 120.0)
            print("\n=== Phase 2: Coordinate Descent Polish (%.0fs) ===" % cd_time)
            a_cd, cd_score = run_coord_descent(global_a, n, S, cd_time, rng)
            a_cd = safe_normalize(a_cd, n)
            c_cd = fft_convolve_self(a_cd)
            if c_cd.max() < global_best:
                global_best = c_cd.max()
                global_a = a_cd.copy()
                print("  CD: %.8f" % global_best)
            else:
                print("  CD: %.8f (no improvement)" % c_cd.max())

    print("\n=== After Phase 2: global_best=%.8f ===" % global_best)

    # --- Phase 3: Perturbation + L-BFGS restart ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print("\n=== Phase 3: Perturbation + L-BFGS (%.0fs) ===" % remaining)
            n_perturb = max(3, int(remaining / 50))
            perturb_budget = max(10.0, remaining * 0.6 / n_perturb)

            for p in range(n_perturb):
                if time.monotonic() > deadline - 60:
                    break
                noise_scales = [0.05, 0.03, 0.02, 0.01, 0.08]
                ns = noise_scales[p % len(noise_scales)]
                a_pert = global_a + rng.normal(0, ns * S / np.sqrt(n), n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = safe_normalize(a_pert, n)

                a_new, score = run_lbfgs_continuation(
                    a_pert, n, S, perturb_budget, rng,
                    beta_schedule=[30, 100, 300, 1000, 3000, 10000],
                    maxiter_per_beta=150)
                a_new = safe_normalize(a_new, n)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < global_best:
                    global_best = c_new.max()
                    global_a = a_new.copy()
                    print("  Perturb %d (ns=%.3f): %.8f" % (p, ns, global_best))
                else:
                    print("  Perturb %d (ns=%.3f): %.8f" % (p, ns, c_new.max()))

    print("\n=== After Phase 3: global_best=%.8f ===" % global_best)

    # --- Phase 4: Cross-pollination ---
    if len(results) >= 2 and global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print("\n=== Phase 4: Cross-pollination (%.0fs) ===" % remaining)
            sorted_res = sorted(results.items(), key=lambda x: x[1][1])
            n_xp = max(2, int(remaining / 60))
            xp_budget = max(10.0, remaining * 0.5 / n_xp)

            for xp in range(n_xp):
                if time.monotonic() > deadline - 45:
                    break
                i1 = xp % len(sorted_res)
                i2 = (xp + 1) % len(sorted_res)
                if i2 == i1:
                    i2 = (i2 + 1) % len(sorted_res)
                a1 = sorted_res[i1][1][0]
                a2 = sorted_res[i2][1][0]

                for blend in [0.4, 0.6]:
                    if time.monotonic() > deadline - 30:
                        break
                    a_bl = blend * a1 + (1 - blend) * a2
                    a_bl = safe_normalize(a_bl, n)
                    a_new, score = run_lbfgs_continuation(
                        a_bl, n, S, xp_budget / 2, rng,
                        beta_schedule=[50, 200, 1000, 5000],
                        maxiter_per_beta=100)
                    a_new = safe_normalize(a_new, n)
                    c_new = fft_convolve_self(a_new)
                    if c_new.max() < global_best:
                        global_best = c_new.max()
                        global_a = a_new.copy()
                        print("  XP %d (%s+%s, b=%.1f): %.8f" %
                              (xp, sorted_res[i1][0], sorted_res[i2][0], blend, global_best))

    print("\n=== After Phase 4: global_best=%.8f ===" % global_best)

    # --- Phase 5: Final high-beta L-BFGS + CD polish ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        print("\n=== Phase 5: Final Polish (%.0fs) ===" % remaining)

        if remaining > 30:
            lb_time = min(remaining * 0.5, 90.0)
            a_fin, score = run_lbfgs_continuation(
                global_a, n, S, lb_time, rng,
                beta_schedule=[500, 2000, 10000, 50000],
                maxiter_per_beta=200)
            a_fin = safe_normalize(a_fin, n)
            c_fin = fft_convolve_self(a_fin)
            if c_fin.max() < global_best:
                global_best = c_fin.max()
                global_a = a_fin.copy()
                print("  Final LBFGS: %.8f" % global_best)

        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            cd_time2 = min(remaining2 * 0.8, 60.0)
            a_fin2, score2 = run_coord_descent(global_a, n, S, cd_time2, rng)
            a_fin2 = safe_normalize(a_fin2, n)
            c_fin2 = fft_convolve_self(a_fin2)
            if c_fin2.max() < global_best:
                global_best = c_fin2.max()
                global_a = a_fin2.copy()
                print("  Final CD: %.8f" % global_best)

    # --- Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.8f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
