Let me think more carefully about the coordinate descent polish. In the simplex, a coordinate descent step is:
1. Pick a pair (i, j)
2. Transfer mass from i to j (or vice versa): a[i] -= t, a[j] += t for some t in [0, a[i]] (or a[j] to i)

"""
Minimize 2*n*max_conv / sum^2 using multi-beta spectral optimization 
with adaptive peak targeting and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_lse_grad(a, c, beta, top_k=None):
    """Compute gradient of log-sum-exp approximation of max(c)."""
    n = len(a)
    c_max = c.max()
    if top_k is not None:
        # Only use top-k lags
        sort_idx = np.argsort(c)[::-1]
        if top_k < len(c):
            mask = np.zeros(len(c), dtype=bool)
            mask[sort_idx[:top_k]] = True
            c_work = c.copy()
            c_work[~mask] = -np.inf
        else:
            c_work = c
    else:
        c_work = c
    
    exp_bc = np.exp(beta * (c_work - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def multi_beta_grad(a, c, betas, top_k=None):
    """Weighted average of gradients from multiple beta values."""
    n = len(a)
    total_grad = np.zeros(n)
    total_weight = 0.0
    for beta in betas:
        grad = compute_lse_grad(a, c, beta, top_k)
        # Weight higher betas more (they focus on the peak)
        weight = beta
        total_grad += weight * grad
        total_weight += weight
    if total_weight < 1e-30:
        return np.zeros(n)
    return total_grad / total_weight


def nesterov_multibeta(a_init, n, S, max_time, rng, 
                        betas=(50.0, 100.0, 200.0, 500.0),
                        top_k_init=20, top_k_max=200,
                        step_scale=1.0):
    """Nesterov-accelerated with multi-beta gradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    no_improve = 0
    last_best = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive top_k
        top_k = min(top_k_max, top_k_init + it // 20)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Multi-beta gradient
        grad = multi_beta_grad(y, c, betas, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 2.0 * S / (n * (1.0 + 0.002 * it)) * step_scale
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        # Check improvement
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic perturbation
        if no_improve > 2000 and rng.random() < 0.1:
            # Structured perturbation: low-frequency sinusoid
            freq = rng.integers(2, 10)
            t_arr = np.arange(n) / n
            pert = 0.005 * S * np.sin(2 * np.pi * freq * t_arr + rng.random() * 2 * np.pi)
            y = project_simplex(y + pert, S)
            t = 1.0
            no_improve = 0

    return best_local, best_score


def spectral_step(a, N, n, top_k, alpha, c):
    """Single spectral shaping step."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    
    m = np.arange(N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total

    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    return a_new_full


def spectral_burst(a, N, n, S, max_time, rng, top_k=20, alpha=0.15, target=1.45):
    """Spectral shaping burst with adaptive parameters."""
    start = time.monotonic()
    best_a = a.copy()
    best_score = np.inf
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        if c_max <= target:
            break
        
        excess = max(0.0, c_max - target) / max(c_max, 1e-10)
        alpha_curr = alpha * (0.5 + 0.5 * excess)
        
        a_raw = spectral_step(a, N, n, top_k, alpha_curr, c)
        a = project_simplex(a_raw, S)
        
        # Occasional noise
        if rng.random() < 0.01:
            a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
    
    return best_a, best_score


def coordinate_descent_polish(a, n, S, max_time, rng, n_pairs_per_iter=50):
    """
    Coordinate descent on the simplex: transfer mass between pairs of coordinates.
    For each pair (i,j), find the optimal transfer amount that minimizes the max convolution.
    """
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    a = a * (S / s)
    
    start = time.monotonic()
    best_a = a.copy()
    best_score = np.inf
    
    c = fft_convolve_self(a)
    best_score = c.max()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        improved = False
        
        # Sample random pairs
        pairs = rng.integers(0, n, size=(n_pairs_per_iter, 2))
        
        for idx in range(n_pairs_per_iter):
            i, j = int(pairs[idx, 0]), int(pairs[idx, 1])
            if i == j:
                continue
            if a[i] < 1e-15:
                continue
            
            # Try transferring from i to j (and vice versa)
            # Use a small perturbation and check if it helps
            best_t = 0.0
            best_val = best_score
            
            # Try a few transfer amounts
            for t_frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                t = t_frac * a[i]
                if t > a[i]:
                    continue
                # Transfer from i to j
                a[i] -= t
                a[j] += t
                c_new = fft_convolve_self(a)
                val = c_new.max()
                if val < best_val:
                    best_val = val
                    best_t = t
                # Undo
                a[i] += t
                a[j] -= t
                
                # Try reverse: transfer from j to i
                if a[j] > 0:
                    t2 = t_frac * a[j]
                    a[j] -= t2
                    a[i] += t2
                    c_new2 = fft_convolve_self(a)
                    val2 = c_new2.max()
                    if val2 < best_val:
                        best_val = val2
                        best_t = -t2  # negative means transfer from j to i
                    a[j] += t2
                    a[i] -= t2
            
            # Apply best transfer
            if best_t != 0.0:
                if best_t > 0:
                    a[i] -= best_t
                    a[j] += best_t
                else:
                    a[j] -= (-best_t)
                    a[i] += (-best_t)
                if best_val < best_score:
                    best_score = best_val
                    best_a = a.copy()
                    improved = True
        
        # Full FFT update periodically
        if it % 10 == 0:
            c = fft_convolve_self(a)
            if c.max() < best_score:
                best_score = c.max()
                best_a = a.copy()
            if not improved:
                break
    
    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_structured_init(n, rng, style):
    """Create various structured initializations."""
    t = np.arange(n) / n
    if style == 0:
        return 0.5 + 0.4 * np.sin(2 * np.pi * (3 + rng.random() * 5) * t) + 0.1 * rng.standard_normal(n)
    elif style == 1:
        return np.exp(-0.5 * ((t - rng.random()) * rng.uniform(3, 8)) ** 2) + 0.05 * rng.random(n)
    elif style == 2:
        return 0.3 + 0.3 * np.cos(2 * np.pi * rng.uniform(2, 10) * t) * np.cos(2 * np.pi * rng.uniform(3, 15) * t)
    elif style == 3:
        # Tent-like
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.2, 0.5)
        return np.maximum(0, 1 - np.abs(t - center) / width) + 0.05 * rng.random(n)
    elif style == 4:
        # Multi-peak
        n_peaks = rng.integers(2, 6)
        centers = np.sort(rng.random(n_peaks))
        result = np.zeros(n)
        for c in centers:
            w = rng.uniform(0.05, 0.2)
            result += np.exp(-0.5 * ((t - c) / w) ** 2)
        return result + 0.02 * rng.random(n)
    else:
        return 0.3 + 0.2 * rng.random(n)

For n=8192, each FFT is about 8192 * log2(16383) ≈ 8192 * 14 ≈ 115K operations. With 50 pairs per iteration and 2 evaluations per pair (forward and reverse), that's 100 FFTs per CD iteration, so about 11.5M operations per CD iteration. That's probably too slow.

Let me optimize: instead of computing the full convolution for each transfer, I can compute the change analytically.

When we transfer t from a[i] to a[j]:
- The change in `(a*a)[k]` is:
  - For the i-related terms: -t * a[k-i] (if k-i valid) - t * a[k-i] (if k-i valid) = -2t*a[k-i] (if k-i valid and k != 2i)
  - For the j-related terms: +t * a[k-j] (if k-j valid) + t * a[k-j] (if k-j valid) = +2t*a[k-j] (if k-j valid and k != 2j)
  - Special terms for k=2i and k=2j

So the new convolution at lag k is:
`c_new[k] = c[k] - 2t*a[k-i] + 2t*a[k-j]` (approximately, ignoring the special cases)

This means we can update the convolution in O(1) per lag (just two additions), or O(n) for all lags (vectorized). Much faster than an FFT!

Let me implement this:

def convolution_change(a, c, i, j, t):

    n = len(a)

    # For each lag k, the term involving a[i] changes:
    #   If k-i is valid (0 <= k-i < n): change -= t * a[k-i] (from j=i term)
    #   If k-i is valid: change -= t * a[k-i] (from j=k-i term, i.e., a[k-i]*a[i])
    #   Special: if k = 2i, change -= 2*t*a[i] + t^2 (the a[i]^2 term)
    
    # Actually let me think about this differently.
    # (a*a)[k] = sum_l a[l] * a[k-l]
    # After change: a[i] -> a[i]-t, a[j] -> a[j]+t
    # Change in (a*a)[k]:
    #   Terms involving a[i]: 
    #     l=i: (a[i]-t)*a[k-i] - a[i]*a[k-i] = -t*a[k-i] (if 0<=k-i<n)
    #     l=k-i: a[k-i]*(a[i]-t) - a[k-i]*a[i] = -t*a[k-i] (if 0<=k-i<n and k-i != i)

    #   Terms involving a[j]:

    delta = np.zeros(2 * n - 1)

    lo_i = i
    hi_i = i + n - 1

    # minus t * a[k-i] (from l=k-i term) if k-i != i
    # = -2*t*a[k-i] if k != 2i
    # = -2*t*a[i] + t^2 if k = 2i
    
    # Let's build it:
    # The "pattern" for a[i] is: it contributes to lags i through i+n-1
    # with values a[0], a[1], ..., a[n-1] (from the l=i perspective)
    # and the same from the l=k-i perspective
    
    # Actually, let's just compute it directly:
    # delta_i[k] = -t * a[k-i] if 0 <= k-i < n else 0 (from l=i term)
    #             -t * a[k-i] if 0 <= k-i < n and k-i != i else 0 (from l=k-i term)
    #             + t^2 if k = 2i (correction for the quadratic term)

    # (a*a)[k] with a[i] replaced by a[i]-t:
    #   The terms that change are those where l=i or k-l=i
    #   l=i: contributes (a[i]-t)*a[k-i] instead of a[i]*a[k-i], diff = -t*a[k-i]
    #   k-l=i => l=k-i: contributes a[k-i]*(a[i]-t) instead of a[k-i]*a[i], diff = -t*a[k-i]
    #   If l=i and k-l=i (i.e., k=2i): both are the same term, so:
    #     (a[i]-t)^2 instead of a[i]^2, diff = -2t*a[i] + t^2
    #     (not -t*a[k-i] + -t*a[k-i] = -2t*a[i], we need +t^2 correction)

    #   For k in [i, i+n-1]: delta[k] -= t * a[k-i]
    #   For k in [i, i+n-1]: if k-i < n, delta[k] -= t * a[k-i]  (this is the same range)

    # (a*a)[k] = sum_{l: 0<=l<n, 0<=k-l<n} a[l] * a[k-l]

    # Terms where l = i: a[i]*a[k-i] -> (a[i]-t)*a[k-i], change = -t*a[k-i]
    #   Valid when 0<=i<n (always) and 0<=k-i<n, i.e., i<=k<=i+n-1
    # Terms where k-l = i, i.e., l = k-i: a[k-i]*a[i] -> a[k-i]*(a[i]-t), change = -t*a[k-i]

    # For simplicity, ignore t^2 terms:

Actually, the key insight is: the change in the convolution when we change a[i] by -t is:
`-t * (a convolved with delta_i)` where delta_i is a unit vector at position i... no, that's not quite right because it's self-convolution.

Let me think again. If a' = a - t*e_i + t*e_j (where e_i is the unit vector at position i), then:
a'*a' = (a + t*e_j - t*e_i)(a + t*e_j - t*e_i)
      = a*a + t*a*e_j + t*e_j*a - t*a*e_i - t*e_i*a + t^2*(e_j*e_j - e_j*e_i - e_i*e_j + e_i*e_i)

Ignoring t^2:
change = t*(a*e_j + e_j*a - a*e_i - e_i*a)

Now, (a*e_j)[k] = a[k-j] (with a[m]=0 for m outside [0,n-1]). So (a*e_j) is just a shifted version of a.
Similarly, (e_j*a)[k] = a[k-j].
So (a*e_j + e_j*a)[k] = 2*a[k-j] for k in [j, j+n-1].

Therefore:
change[k] = 2t * (a[k-j] - a[k-i]) for k in the union of [i, i+n-1] and [j, j+n-1]
           = 0 elsewhere

This is very simple! The change in the convolution is just `2t * (shifted_a_j - shifted_a_i)` where the shifts are by j and i respectively.

So to compute the new max:
`new_c[k] = c[k] + 2t * (a[k-j] - a[k-i])` (with a[m]=0 for m outside [0,n-1])

And `new_max = max(new_c)`

This can be computed in O(n) time (vectorized numpy), which is much faster than an FFT (O(n log n)).

```python
def conv_after_transfer(c, a, i, j, t, n):
    """Compute new convolution after transferring t from a[i] to a[j]."""
    # Change: 2t * (a shifted by -j) - 2t * (a shifted by -i)
    # a shifted by -d: the d-th element of a is placed at position 0
    # In the convolution of length 2n-1:
    #   2t * a[k-j] for k in [j, j+n-1] (where 0 <= k-j < n)
    #   -2t * a[k-i] for k in [i, i+n-1] (where 0 <= k-i < n)
    
    new_c = c.copy()  # We could avoid the copy by working in-place, but for clarity

    # Add contribution from j
    lo = j
    hi = min(j + n, 2 * n - 1)
    if lo < hi:
        new_c[lo:hi] += vec * a[:hi - lo]
    
    # Subtract contribution from i
    lo = i
    hi = min(i + n, 2 * n - 1)
    if lo < hi:
        new_c[lo:hi] -= vec * a[:hi - lo]
    
    return new_c

```python
def conv_after_transfer(c, a, i, j, t, n):
    new_c = c.copy()
    vec = 2.0 * t
    
    # Add: 2t * a[k-j] for k in [j, j+n-1]
    hi_j = min(j + n, 2 * n - 1)
    if j < hi_j:
        new_c[j:hi_j] += vec * a[:hi_j - j]
    
    # Subtract: 2t * a[k-i] for k in [i, i+n-1]
    hi_i = min(i + n, 2 * n - 1)
    if i < hi_i:
        new_c[i:hi_i] -= vec * a[:hi_i - i]
    
    return new_c
```
