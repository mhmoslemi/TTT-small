Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm combining:

- L-BFGS-B continuation (refined schedule, more iterations at high beta)
- Peak-directed Nesterov with exponential weighting of all peaks
- Block coordinate descent with per-block L-BFGS
- Diversified perturbations and restarts

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_a(a, w):
    """Compute grad_a[j] = 2 * sum_k w[k] * a[j-k] (cross-correlation via FFT)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_from_z(z):
    """Convert softmax params to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def smooth_obj_and_grad(a, beta, S):
    """
    Compute peak-weighted log-sum-exp smoothed objective and gradient w.r.t. a.
    Uses exponential weighting: peaks in the convolution get higher weight.
    """
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    
    # Compute weights: exponential of beta * (c - c_max), normalized
    # This gives more weight to peaks
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    w = exp_bc / sum_exp_bc
    
    obj = (c_max + np.log(sum_exp_bc)) / beta
    
    # Gradient w.r.t. a
    grad_a = compute_grad_a(a, w)
    
    return obj, grad_a, c

                     beta_schedule=None, lr_base=0.1, lr_decay=0.999):
    """
    Adam optimizer on softmax parameters with log-sum-exp continuation.
    Returns best (a, score) found.
    """
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
    
    # Convert to softmax params
    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    
    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break
        
        elapsed_beta = 0.0
        n_steps = 0
        lr = lr_base / (np.sqrt(n) + 1.0)
        
        while time.monotonic() - start < max_time - 5:
            t += 1
            n_steps += 1
            
            # Step size with Adam decay
            lr_t = lr * np.sqrt(1 - lr_decay**t) / (1 - lr_decay**t)
            
            # Compute a from z
            z = z - z.mean()  # recenter for stability
            a = S * softmax_from_z(z)
            
            # Compute smoothed objective gradient w.r.t. a
            _, grad_a, _ = smooth_obj_and_grad(a, beta, S)
            
            # Convert gradient to z-space: grad_z = a * (grad_a - mean(grad_a * a) / S)
            dot_ga_a = np.dot(grad_a, a)
            grad_z = a * (grad_a - dot_ga_a / S)
            
            # Adam update
            m = 0.9 * m + 0.1 * grad_z
            v = 0.999 * v + 0.001 * grad_z**2
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
            
            # Check score periodically
            if n_steps % 50 == 0:
                a = S * softmax_from_z(z - z.mean())
                a = np.maximum(a, 0.0)
                c = fft_convolve_self(a)
                score = c.max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
            
            # Early stop if converged
            if n_steps > 200 and np.max(np.abs(grad_z)) < 1e-10:
                break
        
        z = z - z.mean()
        a = S * softmax_from_z(z)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * S / s
        c = fft_convolve_self(a)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if n_steps > 0:
            print("    Adam beta=%5d: score=%.8f (%d steps, %.1fs)" %
                  (beta, best_score, n_steps, time.monotonic() - start - (max_time - (max_time - elapsed_beta))))
    
    return best_a, best_score


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=300):
    """
    Run L-BFGS-B with continuation on beta (smooth-to-sharp).
    Returns best (a, score) found.
    """
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
    
    # Convert to softmax params
    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()
    
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    
    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break
        
        # At high beta, try multiple runs with perturbation
        n_attempts = 1
        if beta >= 1000 and max_time > 60:
            n_attempts = 2
        
        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 5:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.01, n)
                z = z - z.mean()
            
            def fun(z_val):
                z_val = np.asarray(z_val)
                n_local = len(z_val)
                a_local = S * softmax_from_z(z_val - z_val.mean())
                a_local = np.maximum(a_local, 0.0)
                s_local = a_local.sum()
                if s_local < 1e-15:
                    return 1e10
                a_local = a_local * S / s_local
                c = fft_convolve_self(a_local)
                c_max = c.max()
                exp_bc = np.exp(beta * (c - c_max))
                return (c_max + np.log(exp_bc.sum())) / beta
            
            def jac(z_val):
                z_val = np.asarray(z_val)
                a_local = S * softmax_from_z(z_val - z_val.mean())
                a_local = np.maximum(a_local, 0.0)
                s_local = a_local.sum()
                if s_local < 1e-15:
                    return np.zeros(len(z_val))
                a_local = a_local * S / s_local
                _, grad_a, _ = smooth_obj_and_grad(a_local, beta, S)
                dot_ga_a = np.dot(grad_a, a_local)
                grad_z = a_local * (grad_a - dot_ga_a / S)
                return grad_z
            
            t0 = time.monotonic()
            try:
                result = scipy_minimize(
                    fun, z.copy(),
                    jac=jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_beta,
                        'maxfun': maxiter_per_beta * 3,
                        'ftol': 1e-15,
                        'gtol': 1e-10,
                        'maxls': 50,
                        'iprint': -1
                    }
                )
            except Exception as e:
                print("      L-BFGS error: %s" % str(e))
                break
            
            elapsed_beta = time.monotonic() - t0
            z = result.x
            z = z - z.mean()
            
            # Convert back to a
            a = S * softmax_from_z(z)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a * S / s
            
            c = fft_convolve_self(a)
            score = c.max()
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        print("    L-BFGS beta=%5d: score=%.8f (%d attempts, total %.1fs)" %
              (beta, best_score, n_attempts, time.monotonic() - start))
    
    return best_a, best_score


def run_nesterov_exponential(a_init, n, S, max_time, rng,
                              beta=50.0, top_k_init=20, top_k_max=2048):
    """
    Nesterov-accelerated subgradient with exponential peak weighting.
    Uses all peaks weighted by exp(beta * (c_k - c_max)) for smooth gradient.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    # Adaptive beta: start moderate, increase over time
    beta_schedule = lambda step: min(5000.0, beta * (1.0 + step / 2000.0))
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)
        cur_beta = beta_schedule(it)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        # Compute weights with exponential peak focusing
        c_max = cur
        exp_bc = np.exp(cur_beta * (c - c_max))
        
        # Only use top-k for efficiency
        if top_k < len(c):
            top_indices = np.argsort(c)[::-1][:top_k]
            top_vals = exp_bc[top_indices]
            total = top_vals.sum()
            if total < 1e-30:
                break
            weights = top_vals / total
            w_full = np.zeros(len(c))
            for i, k in enumerate(top_indices):
                w_full[int(k)] = weights[i]
        else:
            total = exp_bc.sum()
            if total < 1e-30:
                break
            w_full = exp_bc / total
        
        # Gradient via cross-correlation
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size with Nesterov acceleration
        step = 2.0 * S / (n * (1.0 + 0.005 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Occasional random restart to escape local minima
        if it % 3000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score


def run_block_cd(a_init, n, S, max_time, rng, n_blocks=16):
    """
    Block coordinate descent: split sequence into blocks, optimize each
    with L-BFGS while keeping others fixed. Cycles through blocks.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    
    block_size = n // n_blocks
    blocks = []
    for i in range(n_blocks):
        start_idx = i * block_size
        end_idx = start_idx + block_size if i < n_blocks - 1 else n
        blocks.append((start_idx, end_idx))
    
    beta = 50.0
    iteration = 0
    
    while time.monotonic() - start < max_time:
        iteration += 1
        # Increase beta over time for sharper peak focus
        beta = min(5000.0, 50.0 * (1.0 + iteration * 0.5))
        
        for (bs, be) in blocks:
            if time.monotonic() - start > max_time - 2:
                break
            
            # Get current block values
            a_fixed = a.copy()
            a_block = a[bs:be]
            block_sum_others = a_fixed.sum() - a_block.sum()
            
            # Optimize this block using L-BFGS
            def block_obj(x_block):
                x_block = np.maximum(x_block, 0.0)
                x_block = x_block * max(0.001, 1.0 - x_block.sum() / (block_sum_others + x_block.sum() + 1e-15))
                # Renormalize to maintain total sum
                total = block_sum_others + x_block.sum()
                if total < 1e-15:
                    return 1e10
                a_trial = a_fixed.copy()
                a_trial[bs:be] = x_block * (S / total)
                a_fixed[bs:be] = 0  # we'll set them from x_block
                
                # Reconstruct
                a_trial = np.zeros(n)
                a_trial[bs:be] = x_block
                a_trial[bs:be] = 0
                a_trial[bs:be] = np.maximum(x_block, 0)
                # Keep others fixed
                for i in range(n):
                    if i < bs or i >= be:
                        a_trial[i] = a_fixed[i]
                
                total_sum = a_trial.sum()
                if total_sum < 1e-15:
                    return 1e10
                a_trial = a_trial * S / total_sum
                
                c = fft_convolve_self(a_trial)
                c_max = c.max()
                exp_bc = np.exp(beta * (c - c_max))
                return (c_max + np.log(exp_bc.sum())) / beta
            
            # Initial point for block
            x0 = a[bs:be].copy()
            x0 = np.maximum(x0, 1e-10)
            
            try:
                res = scipy_minimize(
                    block_obj, x0,
                    method='L-BFGS-B',
                    options={'maxiter': 30, 'ftol': 1e-12, 'gtol': 1e-8, 'iprint': -1}
                )
                a[bs:be] = np.maximum(res.x, 0.0)
            except Exception:
                pass
        
        # Renormalize
        s = a.sum()
        if s > 1e-15:
            a = a * S / s
        
        c = fft_convolve_self(a)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if iteration % 5 == 0:
            print("    BlockCD iter %d: score=%.8f (beta=%.0f, %.1fs)" %
                  (iteration, best_score, beta, time.monotonic() - start))
    
    return best_a, best_score


def run_direct_subgradient(a_init, n, S, max_time, rng):
    """
    Direct projected subgradient on the simplex with backtracking line search.
    More aggressive peak-shaving than Nesterov.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    
    # Step size parameters
    gamma_0 = S / np.sqrt(n)
    gamma = gamma_0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        gamma = gamma_0 / (1.0 + 0.01 * it)
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
        
        # Find top peak positions
        top_k = min(64, 8 + it // 100)
        top_indices = np.argsort(c)[::-1][:top_k]
        
        # Build gradient: contribution of each a[j] to the top peaks
        w_full = np.zeros(len(c))
        # Weight peaks by their value (normalized)
        peak_vals = c[top_indices]
        peak_max = peak_vals[0]
        if peak_max < 1e-15:
            break
        # Exponential weighting for sharper focus
        weights = np.exp(30.0 * (peak_vals - peak_max))
        weights = weights / weights.sum()
        
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]
        
        # Cross-correlation for gradient
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Only push down elements that contribute positively to peaks
        grad = np.maximum(grad, 0.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Projected gradient step
        a_new = project_simplex(a - gamma * grad / max(gnorm, 1e-15), S)
        
        # Backtracking: check if we improved
        c_new = fft_convolve_self(a_new)
        if c_new.max() > cur * 1.01:
            # Step was too large, reduce
            gamma *= 0.5
            a_new = project_simplex(a - gamma * grad / max(gnorm, 1e-15), S)
        
        a = a_new
        
        # Occasional random perturbation
        if it % 5000 == 0 and rng.random() < 0.05:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 10.0)
    
    global_best = np.inf
    global_a = None
    
    def update_best(a_candidate):
        nonlocal global_best, global_a
        a_candidate = safe_normalize(a_candidate, len(a_candidate))
        c = fft_convolve_self(a_candidate)
        score = c.max()
        if score < global_best:
            global_best = score
            global_a = a_candidate.copy()
            return True
        return False
    
    # --- Get initial sequence ---
    inits = []
    primary = None
    
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            primary = np.clip(given, 0.0, 1000.0).copy()
    except Exception:
        pass
    
    if primary is not None:
        n = len(primary)
    else:
        n = 8192
        primary = np.ones(n) + 0.1 * rng.standard_normal(n)
        primary = np.maximum(primary, 0.01)
    
    S = np.sqrt(2.0 * float(n))
    
    # Normalize primary
    a_primary = safe_normalize(primary, n)
    c_prim = fft_convolve_self(a_primary)
    print("=== Starting: n=%d, initial score=%.8f ===" % (n, c_prim.max()))
    print("Target: 1.5027, Previous best: 1.508805")
    
    update_best(a_primary)
    
    # Also prepare diverse initializations
    tt = np.linspace(0, 1, n, endpoint=False)

    alt_inits.append(("rand", np.maximum(rng.random(n) + 0.05, 0.01)))
    alt_inits.append(("multi_sin", 1.0 + 0.3*np.sin(2*np.pi*3*tt) +

                      0.1*np.cos(2*np.pi*29*tt) + 0.08*np.sin(2*np.pi*47*tt)))

                      np.cos(2*np.pi*5*tt) + 0.3*np.exp(-2.0*(tt-0.7)**2) *
                      np.cos(2*np.pi*8*tt) + 0.05))
    
    total_starts = 1 + len(alt_inits)
    print("Total starting points: %d" % total_starts)

            )
            if update_best(a_new):
                print("  Final Nesterov: %.8f" % global_best)
    
    # --- Phase 8: Try alternative initializations if time remains ---
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Phase 8: Alternative initializations (%.0fs) ===" % remaining)
        per_alt = remaining / len(alt_inits)
        
        for name, arr in alt_inits:
            if time.monotonic() > deadline - 20:
                break
            
            a_alt = safe_normalize(arr, n)
            a_new, score = run_lbfgs_continuation(
                a_alt, n, S, per_alt, rng,
                beta_schedule=[10, 50, 200, 1000, 5000],
                maxiter_per_beta=150
            )
            if update_best(a_new):
                print("  Alt %s: %.8f" % (name, global_best))
    
    # --- Final verification ---
    print("\n=== Final Results ===")
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Verification score: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
    
    print("Returning sequence of length %d, best score=%.8f" % (len(global_a), global_best))
    elapsed = time.monotonic() - (deadline - budget_s + 10)
    print("Total time: %.1fs" % elapsed)
    return [float(x) for x in global_a]
