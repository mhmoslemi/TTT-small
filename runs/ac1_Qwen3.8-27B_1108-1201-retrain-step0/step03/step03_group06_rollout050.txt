**Alternative: Use the "probe" / LP approach from the LaTeX description**

The idea is:
1. At the current point `f`, compute `M = max(f*f)`.
2. Solve the LP: maximize `sum(b)` subject to `b >= 0` and `||f * b||_inf <= M`.
3. The optimal `b` gives a direction. Rescale and blend.

The LP in step 2 has `n` variables and `2n-1` constraints. For n=4096, that's a 4096-variable LP with 8191 constraints. This is solvable with a modern LP solver in a fraction of a second!

This is actually a very promising approach because:
- It solves an exact subproblem (not an approximation)
- It's related to the "probe" method in combinatorial optimization
- It guarantees progress (non-increasing objective)
- It naturally handles the non-negativity constraint

The constraints are: `(f * b)[k] <= M` for all `k`, which is a linear constraint in `b`:
`sum_i f[i] * b[k-i] <= M` (for each k)

This is a system of linear inequalities. The objective is `max sum(b)`.

Using cvxpy with a fast LP solver (like GLPK or GUROBI), this should be very fast.

Let me estimate: The constraint matrix is a (2n-1) x n Toeplitz-like matrix. For n=4096, that's about 8191 * 4096 = 33M entries. But since it's structured (each row is a shifted version of f), we can express it more compactly.

Actually, the constraint for lag k is:
`sum_{i=0}^{n-1} f[i] * b[k-i] <= M`

where `b[j] = 0` for `j < 0` or `j >= n`.

This can be written as `(T_k) @ b <= M` where `T_k` is a row vector with `T_k[i] = f[i]` if `k-i` is in `[0, n-1]`, else 0.

Actually, `T_k[i] = f[i]` if `0 <= k-i <= n-1`, i.e., `k-(n-1) <= i <= k`. So each row of the constraint matrix has at most `n` non-zero entries, and they're just shifts of `f`.

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS quasi-Newton optimization
with log-sum-exp smoothing, active-set peak focusing, LP-based directed search,
and coordinate descent polish.

Key improvements over previous approach:
1. L-BFGS quasi-Newton method (exploits curvature, much faster convergence)
2. Adaptive temperature with "re-focusing" (periodically reset to high T)
3. Peak-focused gradient (exact top-K lag selection instead of soft weighting)
4. LP probe for directed line searches (cutting plane version)
5. Exact 1D coordinate descent with golden section search
6. Multi-scale optimization (coarse to fine)
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import cvxpy as cp
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def theta_to_a(theta, S):
    """Convert log-proportion parameters to sequence."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def a_to_theta(a, S):
    """Convert sequence to log-proportion parameters."""
    a = np.maximum(a, 1e-30)
    return np.log(a / S)


def compute_weighted_grad(c, a, T):
    """Compute gradient of log-sum-exp smoothed max(c) w.r.t. a.
    Returns grad_a (not yet transformed to theta-space)."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    scaled = T * (c - c_max)
    scaled = np.maximum(scaled, -700)  # prevent underflow
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (cross-correlation)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def topk_weighted_grad(c, a, K):
    """Compute gradient focusing on top-K lags (hard weighting)."""
    n = len(a)
    # Get top K indices
    if K >= len(c):
        K = len(c)
    top_idx = np.argsort(c)[-K:]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / K
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def lbfgs_optimize(a_init, n, S, time_budget, T_init=20.0, T_final=2000.0,
                   max_iter=None, verbose=False):
    """Run L-BFGS on the smoothed objective in log-parameter space."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    theta = a_to_theta(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    total_iters = 0
    
    def objective_and_grad(theta_t):
        a_t = theta_to_a(theta_t, S)
        c_t = fft_convolve_self(a_t)
        obj = c_t.max()
        
        if obj < best_obj - 1e-12:
            nonlocal best_obj, best_a
            best_obj = obj
            best_a = a_t.copy()
        
        # Compute smoothed gradient
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.5)
        
        c_max = c_t.max()
        if c_max < 1e-30:
            return 0.0, np.zeros(n)
        
        scaled = T * (c_t - c_max)
        scaled = np.maximum(scaled, -700)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c_t))
            w[np.argmax(c_t)] = 1.0
        else:
            w /= w_sum
        
        grad_a = compute_weighted_grad_from_w(w, a_t, n)
        
        # Transform to theta-space
        dot = np.dot(grad_a, a_t)
        grad_theta = grad_a * a_t - (a_t / S) * dot
        
        # Return the smoothed objective (not the exact max)
        smooth_obj = c_max + (1.0 / T) * np.log(np.maximum(w_sum, 1e-300))
        return smooth_obj, grad_theta
    
    def compute_weighted_grad_from_w(w, a_t, n):
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a_t, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        return 2.0 * corr[:n]
    
    # Run L-BFGS in chunks (to allow temperature updates)
    chunk_iters = min(200, max_iter or 500)
    
    while time.monotonic() - start < time_budget:
        res = minimize(
            objective_and_grad, theta, method='L-BFGS-B',
            jac=True,
            options={
                'maxiter': chunk_iters,
                'maxfun': chunk_iters * 3,
                'gtol': 1e-6,
                'ftol': 1e-12,
                'maxls': 50,
            }
        )
        theta = res.x
        total_iters += res.nit if hasattr(res, 'nit') else chunk_iters
        
        if verbose:
            print("    L-BFGS chunk: status=%s, niter=%d, obj=%.8f" % (
                res.status, res.nit, best_obj))
        
        # Check for convergence
        if res.status not in ['Optimality condition achieved', 'Convergence to desired accuracy achieved']:
            # Add small perturbation to escape
            theta = theta + 0.01 * np.random.default_rng(total_iters).standard_normal(n)
    
    a_final = theta_to_a(theta, S)
    a_final = normalize_nonneg(a_final, S)
    c_final = fft_convolve_self(a_final)
    final_obj = c_final.max()
    
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a_final.copy()
    
    return best_a, best_obj

def lp_probe_step(f, M, max_constraints=500):
    """
    LP probe: find direction g that maximizes sum(g) subject to
    f*g[k] <= M for (a subset of) k, and g >= 0.
    Returns the optimal g (or None if infeasible).
    """
    n = len(f)
    
    # Use only the most important constraints (top violations + marginals)
    conv_fg = fft_convolve_self(f)  # This is f*f, not f*g...
    # Actually, we need f * g, which depends on g. So we can't pre-compute it.
    # Instead, we use the current f*f to identify which constraints are "active".
    
    # At the current point f, the active constraints are those where (f*f)[k] is close to M.
    # But the LP is about finding g such that f*g[k] <= M.
    # The "most likely to be violated" constraints for a new g are those where f has large values.
    
    # Simple approach: use constraints for k where (f*f)[k] > 0.5 * M
    active_k = np.where(conv_fg > 0.3 * M)[0]
    
    # Limit to max_constraints
    if len(active_k) > max_constraints:
        # Keep the worst ones
        vals = conv_fg[active_k]
        top_idx = np.argsort(vals)[-max_constraints:]
        active_k = active_k[top_idx]
    
    # Build constraint matrix A where A[k, j] = f[j] if 0 <= k-j <= n-1
    # Use sparse matrix
    rows = []
    cols = []
    vals = []
    
    for k in active_k:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    if len(rows) == 0:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_k), n))
    
    # Solve: max sum(g) s.t. A @ g <= M, g >= 0
    try:
        # Use scipy linprog with HiGHS
        c_obj = -np.ones(n)
        b_ub = np.full(len(active_k), M)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'maxiter': 10000})
        if res.success:
            return res.x
    except:
        pass
    
    return None


def directed_search(a, n, S, time_budget, verbose=False):
    """Directed search using LP probe."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    M = c.max()
    
    start = time.monotonic()
    rng = np.random.default_rng(777)
    
    iteration = 0
    while time.monotonic() - start < time_budget - 5:
        iteration += 1
        M = best_obj
        
        g = lp_probe_step(a, M, max_constraints=300)
        if g is None:
            break
        
        # Normalize g to have same sum as a
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g * (S / g_sum)
        
        # Line search: find best t in [0, 1]
        best_t = 0.0
        best_t_obj = best_obj
        
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_trial = (1 - t) * a + t * g
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial < best_t_obj - 1e-12:
                best_t_obj = obj_trial
                best_t = t
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.01, 0.05]:
                a_trial = (1 - t) * a + t * g
                a_trial = normalize_nonneg(a_trial, S)
                c_trial = fft_convolve_self(a_trial)
                obj_trial = c_trial.max()
                if obj_trial < best_t_obj - 1e-12:
                    best_t_obj = obj_trial
                    best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = normalize_nonneg(a, S)
            best_obj = best_t_obj
            best_a = a.copy()
            if verbose:
                print("    LP iter %d: t=%.2f, obj=%.8f" % (iteration, best_t, best_obj))
        else:
            break
    
    return best_a, best_obj


def get_initialization(name, n, S, rng):

        return a * (S / a.sum())
    elif name == "cosine_mix":
        a = 1.0 + 0.25 * np.cos(2 * np.pi * 5 * i / n + rng.random() * 2) + \
            0.15 * np.cos(2 * np.pi * 11 * i / n + rng.random() * 2) + \

        a = np.maximum(a, 0.01)

    elif name == "flat_spectrum":
        # Try to create near-flat power spectrum

        a = np.maximum(a, 0.01)
