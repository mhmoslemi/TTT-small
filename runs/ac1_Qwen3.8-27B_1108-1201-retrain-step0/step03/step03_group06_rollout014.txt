    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(W) * A))
    return grad[:n]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def run_smooth_adam(a_init, n, S, time_budget, T_init=10.0, T_final=2000.0,
                    lr_init=0.05, beta1=0.9, beta2=0.999, verbose=False):
    """Adam on log-parameters with log-sum-exp smooth objective."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    fsize = 1 << (3 * n - 2).bit_length()

    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.5)

        # Convert theta to a
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
        else:
            a = S * exp_theta / total

        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()

        # Compute weighted direction
        c_max = c.max()
        if c_max < 1e-30:
            break
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

        # grad_a[j] = 2 * sum_k w[k] * a[k-j]
        W = fft(w, fsize)
        A = fft(a, fsize)
        g_a = 2.0 * np.real(ifft(np.conj(W) * A))[:n]

        # Chain rule for log-parameters
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot

        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break

        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_init * (0.2 + 0.8 * (1.0 - progress))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if verbose and t % 5000 == 0:
            print("      Adam t=%d: obj=%.8f, T=%.1f" % (t, best_obj, T))

    return best_a, best_obj


def solve_lp_direction(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b)[k] <= M for all k, b >= 0.
    Returns b or None if infeasible/too slow.
    """
    n = len(f)
    # Constraint: conv(f, b)[k] = sum_i f[i]*b[k-i] <= M for k=0,...,2n-2
    # This is a linear constraint in b: for each k, sum_{j: 0<=k-j<n} f[k-j] * b[j] <= M
    # The constraint matrix A_ub has row k: A_ub[k,j] = f[k-j] if 0<=k-j<n else 0
    
    # For n=4096, this is 8191 x 4096 matrix - too large for direct construction
    # Use a truncated version with only the top constraints
    
    # Actually, let's use a smarter approach: only include constraints that are "active"
    # near the current solution or that are most likely to bind
    
    # For practical purposes, include all constraints but use sparse representation
    from scipy.sparse import csr_matrix
    
    # Build the constraint matrix sparsely
    # Row k, col j: f[k-j] if 0 <= k-j < n
    # For each j, it appears in rows k=j, j+1, ..., j+n-1 (i.e., n rows)
    # Total non-zeros: n*n
    
    nnz = n * n
    rows = np.zeros(nnz, dtype=np.int32)
    cols = np.zeros(nnz, dtype=np.int32)
    vals = np.zeros(nnz)
    
    idx = 0
    for j in range(n):
        start_k = j
        end_k = min(j + n, 2 * n - 1)
        for k in range(start_k, end_k):
            i = k - j
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[i]
            idx += 1
    
    A_ub = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c_obj = np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            return res.x
        return None
    except Exception:
        return None


def run_lp_guided(a, n, S, time_budget, max_lp_steps=5, verbose=False):
    """Iteratively apply LP-guided blending."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    for step in range(max_lp_steps):
        if time.monotonic() - start > time_budget - 10:
            break
        
        M = c.max()
        if M < 1e-30:
            break
        
        if verbose:
            print("      LP step %d: M=%.6f" % (step, M))
        
        b = solve_lp_direction(a, M)
        if b is None or b.sum() < 1e-10:
            break
        
        # Normalize b to same sum as a
        b = b * (S / b.sum())
        
        # Try different blending ratios
        for t in [0.1, 0.2, 0.3, 0.15, 0.05]:
            a_trial = (1 - t) * best_a + t * b
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial < best_obj - 1e-12:
                best_obj = obj_trial
                best_a = a_trial.copy()
                c = c_trial
                break
    
    return best_a, best_obj


def run_direct_subgrad(a_init, n, S, time_budget, T_init=50.0, T_final=3000.0,
                       lr_init=0.03, momentum=0.8, verbose=False):
    """Direct subgradient with momentum and backtracking (no log parameterization)."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    a_prev = a.copy()
    beta = momentum
    
    start = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 200
    fsize = 1 << (3 * n - 2).bit_length()
    
    while time.monotonic() - start < time_budget:
        t = time.monotonic()
        progress = min(1.0, (t - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.6)
        
        # Momentum step
        y = a + (beta - 1.0) / (1.0 + beta) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        else:
            y = a.copy()
        
        c_y = fft_convolve_self(y)
        cur_obj = c_y.max()
        
        c = fft_convolve_self(a)
        a_obj = c.max()
        if a_obj < best_obj - 1e-14:
            best_obj = a_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Restart with noise
            scale = 0.01 * max(0.1, 1.0 - progress)
            a = best_a + scale * np.random.default_rng(step * 7919).standard_normal(n)
            a = normalize_nonneg(a, S)
            a_prev = a.copy()
            no_improve = 0
            step += 1
            continue
        
        # Compute direction
        g = compute_weighted_direction(c, a, T)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g_dir = g / gnorm
        
        # Backtracking line search
        lr = lr_init * (1.0 - 0.5 * progress)
        accepted = False
        for i in range(15):
            a_trial = a - lr * g_dir
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < a_obj - 1e-10 * max(1.0, a_obj):
                a_prev = a.copy()
                a = a_trial
                accepted = True
                break
            lr *= 0.5
        
        if not accepted:
            a_prev = a.copy()
            # Very small step
            a = a - lr * g_dir
            a = normalize_nonneg(a, S)
        
        step += 1
        if verbose and step % 2000 == 0:
            print("      SG step %d: obj=%.8f, T=%.1f" % (step, best_obj, T))
    
    return best_a, best_obj


def coordinate_descent(a_init, n, S, time_budget, max_sweeps=100, verbose=False):
    """Coordinate descent with golden ratio search on each coordinate."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(42)
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            original = a[i]
            if original < S / n * 1e-3:
                # Try setting to zero
                a[i] = 0.0
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                if c_check.max() < best_obj - 1e-14:
                    best_obj = c_check.max()
                    best_a = a.copy()
                    improved = True
                    c = c_check
                else:
                    a[i] = original
                    a = normalize_nonneg(a, S)
                    c = fft_convolve_self(a)
                continue
            
            # Try a few multiplicative factors
            best_local = original
            best_local_obj = best_obj
            
            for factor in [0.3, 0.5, 0.7, 1.0/phi, phi, 1.3, 1.5, 2.0, 0.1]:
                a[i] = original * factor
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-14:
                    best_local_obj = obj_check
                    best_local = original * factor
            
            if best_local != original:
                a[i] = best_local
                a = normalize_nonneg(a, S)
                c = fft_convolve_self(a)
                if c.max() < best_obj - 1e-14:
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
            else:
                a[i] = original
                a = normalize_nonneg(a, S)
                c = fft_convolve_self(a)
        
        if not improved and sweep >= 5:
            if verbose:
                print("      CD converged at sweep %d" % sweep)
            break
    
    return best_a, best_obj


def peak_shaving(a, n, S, time_budget, top_k=5, verbose=False):
    """Focus on reducing the top K peaks using targeted coordinate adjustments."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    iris = (np.sqrt(5) - 1) / 2
    
    iteration = 0
    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Find top K peaks
        top_idx = np.argsort(c)[-top_k:][::-1]
        
        for k in top_idx:
            if time.monotonic() - start > time_budget:
                break
            
            # Compute gradient of c_k
            # dc_k/da_j = 2*a[k-j] (with zero extension)
            g = np.zeros(n)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    g[j] = 2.0 * a[idx]
            
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-15:
                continue
            
            # Try to reduce this peak by moving in -g direction
            # Use line search with a few trial steps
            best_step = 0.0
            best_step_obj = cur_obj
            
            for alpha in [0.01, 0.02, 0.05, 0.01, 0.005]:
                a_trial = a - alpha * g / gnorm * (S / n)
                a_trial = normalize_nonneg(a_trial, S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < best_step_obj - 1e-14:
                    best_step_obj = c_trial.max()
                    best_step = alpha
            
            if best_step > 0:
                a = a - best_step * g / gnorm * (S / n)
                a = normalize_nonneg(a, S)
        
        iteration += 1
    
    # Also track best
    c = fft_convolve_self(a)
    if c.max() < best_obj:
        best_obj = c.max()
        best_a = a.copy()
    
    return best_a, best_obj


def get_asymmetric_init(n, S, rng):
    """Generate asymmetric initializations to exploit the non-symmetric structure."""
    i = np.arange(n)
    choice = rng.integers(0, 5)
    
    if choice == 0:
        # Linear ramp
        a = i.astype(float) + 1
    elif choice == 1:
        # Two-sided with different widths
        left_width = n * rng.uniform(0.2, 0.5)
        right_width = n * rng.uniform(0.2, 0.5)
        a = np.exp(-0.5 * (i / left_width) ** 2) + np.exp(-0.5 * ((n - 1 - i) / right_width) ** 2)
    elif choice == 2:
        # Cosine with phase shift (breaks symmetry)
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * i / n + phase)
    elif choice == 3:
        # Random but with smooth correlation
        a = rng.random(n)
        # Smooth with a running average
        kernel_size = max(3, n // 100)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    else:
        # Exponential decay from random center
        center = rng.integers(0, n)
        a = np.exp(-np.abs(i - center) / (n * rng.uniform(0.1, 0.4)))
    
    a = np.maximum(a, 0.01)
    return a * (S / a.sum())

    except Exception as e:

        per_round = restart_budget / (n_rounds + 1)
        
        for r in range(n_rounds):
            if time.monotonic() > deadline - 5:
                break
            
            # Add structured noise
            noise_scale = 0.003 * (1 + 0.5 * r)
            a_noisy = best_a + noise_scale * rng.standard_normal(n)
            a_noisy = normalize_nonneg(a_noisy, S)
            
            # Quick Adam
            a_best, _ = run_smooth_adam(a_noisy, n, S, per_round,
                                         T_init=200, T_final=3000, lr_init=0.02)
            a_best = normalize_nonneg(a_best, S)
            obj = evaluate_sequence(a_best.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_best.copy()
                best_a = a_best.copy()
                print("  Restart %d: obj=%.8f" % (r, obj))
            
            # Quick peak shaving on the result
            a_shaved, _ = peak_shaving(a_best, n, S, min(5, per_round * 0.3), top_k=3)
            a_shaved = normalize_nonneg(a_shaved, S)
            obj = evaluate_sequence(a_shaved.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_shaved.copy()
                best_a = a_shaved.copy()
                print("  Restart %d shaved: obj=%.8f" % (r, obj))
    
    # Final verification
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.8f (verified=%.8f)" % (global_best, ver))
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.8f" % ver)
    
    return final_a
