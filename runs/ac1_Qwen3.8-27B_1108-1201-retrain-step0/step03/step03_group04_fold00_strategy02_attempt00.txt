def compute_convolution_peak(sequence):
    """Compute the peak of the self-convolution using FFT (surrogate version)."""
    n = len(sequence)
    if n == 0:
        return 0.0
    
    # Take the FFT of the sequence (with proper padding for symmetry)
    padded_length = 2 * n - 1
    n_fft = 1 << (padded_length - 1).bit_length()
    wave = np.fft.fft(sequence, n_fft)
    
    # Autocorrelation is the inverse FT of |FT|^2
    power_spectrum = np.power(np.abs(wave), 2)  # Alternatively, power_spectrum = np.power(wave.real, 2) + np.power(wave.imag, 2)
    autocorr = np.fft.ifft(power_spectrum, n_fft)
    
    # Return the peak in the autocorrelation (with the unshifted positive convolution)
    peak = np.max(np.abs(autocorr)) 
    return peak

def run_toppk_refined_s(sequence_init, n, S_val, max_eval_time=0.1):
    """Perform a number of gradient descent steps (using FFT) to refine the sequence."""
    sequence = np.clip(np.array(sequence_init), 0.0001, 1000.0).astype(np.float64)
    s_val = sum(sequence)
    sequence = sequence * (S_val / s_val) if s_val > 0 else np.ones(n) * 0.01 * S_val
    
    # Initial peak for early termination
    initial_peak = compute_convolution_peak(sequence)
    if initial_peak < 1.5:
        max_time = min(0.2, max_eval_time)
    elif initial_peak < 3.0:
        max_time = min(0.5, max_eval_time)
    else:
        max_time = max_eval_time
    
    # Ensure we have a time budget for expensive evaluations
    start_time = time.time()
    best_sequence = sequence
    best_peak = initial_peak
    
    # Simple gradient descent (approximate by multiple iterations)
    n_steps = int(max_time * 2 + n)  # Few steps: relate to time and problem size
    topk_thresholds = [1.5, 2.0, 2.5, 3.0]
    
    try:
        for step_num in range(n_steps):
            if time.time() - start_time > max_time:
                break
                
            # Compute gradient (using implicit FFT-based method)
            gradient = np.gradient(np.convolve(sequence, sequence))
            
            # Prevent negative updates
            step_size = 0.5 / (step_num + 1)
            candidate_sequence = sequence - step_size * gradient
            candidate_sequence = np.maximum(candidate_sequence, 0.0001)
            candidate_sum = sum(candidate_sequence)
            if candidate_sum > 0.1:  # Normalization to simplex requires positive sum
                candidate_seq_normed = candidate_sequence * (S_val / candidate_sum)
            else:
                candidate_seq_normed = np.ones(n) * 0.01 * S_val
                
            new_peak = compute_convolution_peak(candidate_seq_normed)
            if new_peak > best_peak - 0.001:  # Very small improvement threshold
                break
                
            sequence = candidate_seq_normed
            if new_peak < best_peak:
                best_sequence = sequence.copy()
                best_peak = new_peak
                
            # Occasionally output for checkpointing
            if step_num % 100 == 0:
                verify = evaluate_sequence(best_sequence)
                if verify < best_peak:
                    best_peak = verify
                    print("Verified lower P-value found in FISTA-like:", best_peak)
                    
    except Exception as e:
        print("Fallback in FISTA-like due to intermediate computation:", str(e))
        # But keep the best_sequence anyway
    
    return best_sequence, best_peak
