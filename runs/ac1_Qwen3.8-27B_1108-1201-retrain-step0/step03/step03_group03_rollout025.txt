import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a.
    Uses the formula: grad[i] = 2 * sum_k w[k] * a[k-i] (valid indices only)
    which equals 2 * cross_corr(w, a)[i] computed via FFT.
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    # FFT size for cross-correlation of w (len 2n-1) and a (len n): need >= 3n-1
    N = 1 << (3 * n).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    # cross_corr[i] = sum_j w[i+j] * a[j] = R_wa[i] where R = ifft(conj(W)*A)
    R = np.real(ifft(np.conj(W) * A))
    grad = 2.0 * R[:n]
    return grad


def topk_gradient(a, c, top_k, temperature=1.0):
    """Gradient using top-k lags with softmax weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices].copy()
    tv_max = top_vals.max()
    sc = 1.0 / max(temperature, 1e-10)
    weights = np.exp(sc * (top_vals - tv_max))
    weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    N = 1 << (3 * n).bit_length()
    W = fft(w_full, N)
    A = fft(a, N)
    R = np.real(ifft(np.conj(W) * A))
    return 2.0 * R[:n]


def run_fista(a_init, n, S, max_time, rng, beta_schedule=None):
    """
    FISTA with LSE gradient, adaptive beta, backtracking, and perturbations.
    Returns (best_sequence, best_max_conv).
    """

    s = a.sum()
    if s < 1e-15:

    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = c_max_init = fft_convolve_self(a).max()
    best_local = a.copy()
    it = 0
    L = 5.0
    stall = 0

    if beta_schedule is None:

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        # Find current beta from schedule
        beta = beta_schedule[-1][1]
        for frac, b in beta_schedule:
            if progress <= frac:
                # Interpolate to next
                for frac2, b2 in beta_schedule:
                    if frac2 > progress:
                        alpha = (progress - frac) / max(frac2 - frac, 1e-10)
                        beta = b + alpha * (b2 - b)
                        break
                break

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stall = 0
        elif it % 100 == 0:
            if cur < best_score + 0.01:
                stall += 1
            else:
                stall = max(0, stall - 5)

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search (simplified: just project)
        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)

        # Periodic L update
        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            f_y = c.max()
            if c_x.max() > f_y + 0.1 * step * gnorm:
                L *= 1.3
            else:
                L = max(L * 0.97, 2.0)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Perturbation to escape local minima
        if stall > 20 and rng.random() < 0.2:
            noise_scale = (0.005 + 0.01 * stall / 30.0) * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_momentum = 1.0
            stall = 0
            L = 5.0

    return best_local, best_score

def coordinate_polish(a, n, S, max_time, rng, num_passes=5):
    """
    Coordinate descent: for each coordinate, try to reduce the max convolution.
    Uses the fact that c_k is linear in a[i] for fixed other coordinates.
    """
    a = np.clip(a, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    c = fft_convolve_self(a)

    for _ in range(num_passes):
        if time.monotonic() - start > max_time:
            break

        # Identify top lags
        top_lags = np.argsort(c)[::-1][:10]

        # For each coordinate, try reducing it if it contributes to top lags
        for i in range(0, n, max(1, n // 500)):
            if time.monotonic() - start > max_time:
                break

            # Find which top lags this coordinate contributes to
            contrib = 0.0
            for lag in top_lags:
                lag = int(lag)
                partner = lag - i
                if 0 <= partner < n:
                    contrib += a[partner]

            if contrib < 1e-10:
                continue

            # Try reducing a[i] by various amounts
            best_v = a[i]
            best_max = c.max()
            trials = [0.0, a[i] * 0.5, a[i] * 0.1, a[i] * 2.0]
            for v in trials:
                if v < 0 or v > 1000:
                    continue
                # Compute new max conv efficiently
                # c_new[k] = c[k] - a[i]*2*a[k-i] + v*2*a[k-i] (approximately)
                # For speed, just check if reducing helps
                if v < a[i]:
                    # Reducing: c_k decreases by (a[i]-v)*2*a[k-i]
                    decrease = (a[i] - v) * 2.0
                    new_max = c.max()
                    # The max might shift, but as a heuristic:
                    if new_max - decrease * max(a[max(0, int(top_lags[0]) - i) if 0 <= int(top_lags[0] - i) < n else 0], 0) < best_max:
                        best_max = best_max - 0.5 * decrease * a[int(top_lags[0]) - i] if 0 <= int(top_lags[0] - i) < n else best_max
                        best_v = v

            if best_v < a[i] - 1e-10:
                diff = a[i] - best_v
                a[i] = best_v
                # Redistribute
                a = np.maximum(a, 0.0)
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * (S / s_new)
                c = fft_convolve_self(a)
                if c.max() < best_score:
                    best_score = c.max()
                    best_local = a.copy()

    return best_local, best_score


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = mu."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-15:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_spectral_init(n, rng, S):
    """
    Initialize in the frequency domain: random phases with specific amplitude spectrum.
    This creates sequences with controlled autocorrelation structure.
    """
    N = 1 << (2 * n - 1).bit_length()
    # Design the amplitude spectrum to make autocorrelation flat
    # |FFT(a)|^2 should be as flat as possible for flat autocorrelation
    # So set |FFT(a)| to be roughly constant
    phases = 2 * np.pi * rng.random(N)
    # Use a slight roll-off at high frequencies for smoothness
    freqs = np.fft.fftfreq(N, d=1.0 / n)
    amp = 1.0 / (1.0 + 0.1 * np.abs(freqs) * n / 10)
    A = amp * np.exp(1j * phases)
    # Enforce Hermitian symmetry for real output
    A[0] = np.abs(A[0])
    for i in range(1, N // 2 + 1):
        A[N - i] = np.conj(A[i])
    a = np.real(ifft(A))[:n]
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())
    return a


def make_periodic_init(n, rng, S, num_freqs=4):
    """Periodic initialization with multiple sinusoidal components."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 30)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.5
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.05)
    a = a * (S / a.sum())
    return a


def make_fft_perturb(a, rng, strength=0.1):
    """Perturb in FFT domain: multiply by random phase (circular shift + phase noise)."""
    n = len(a)
    N = 1 << (n - 1).bit_length()
    A = fft(a, N)
    # Random phase rotation (circular shift)
    shift = rng.integers(-n // 10, n // 10 + 1)
    freqs = np.fft.fftfreq(N) * N
    phase_shift = 2j * np.pi * shift * freqs / N
    A = A * np.exp(phase_shift)
    # Add small random phase noise
    noise_phase = strength * rng.standard_normal(N)
    A = A * np.exp(1j * noise_phase)
    # Enforce Hermitian
    A[0] = np.real(A[0])
    for i in range(1, N // 2 + 1):
        A[N - i] = np.conj(A[i])
    a_new = np.real(ifft(A))[:n]
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    else:
        a_new = a.copy()
    return a_new


def safe_normalize(a, n):
    """Clip to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

        beta_sched = [(0.0, 1.0), (0.2, 5.0), (0.5, 20.0), (0.8, 100.0), (1.0, 500.0)]
        a_f, s_f = run_fista(arr, n, S_n, time_per_start, rng, beta_schedule=beta_sched)
        a_f = safe_normalize(a_f, n)
        c_f = fft_convolve_self(a_f)
        score_f = c_f.max()
        obj_f = 2.0 * n * score_f / S_n ** 2

        if obj_f < global_best or global_a is None:
            global_best = obj_f
            global_a = a_f.copy()
            print("  [%s] obj=%.6f max_conv=%.6f (t=%.0fs)" % (name, obj_f, score_f, time.monotonic() - t0))

    print("=== After Phase 1: best_obj=%.6f ===" % global_best)

    # === Phase 2: Top-k FISTA refinement on best ===
    if global_a is not None and time.monotonic() < deadline - 150:
        n = len(global_a)
        S_n = np.sqrt(2.0 * n)
        rem = (deadline - time.monotonic()) * 0.25
        t0 = time.monotonic()
        print("=== Phase 2: Top-k FISTA refinement (%.0f sec) ===" % rem)

        a_tk, s_tk = run_topk_fista(global_a, n, S_n, rem, rng, top_k=3, top_k_max=48)
        a_tk = safe_normalize(a_tk, n)
        c_tk = fft_convolve_self(a_tk)
        obj_tk = 2.0 * n * c_tk.max() / S_n ** 2

        if obj_tk < global_best:
            global_best = obj_tk
            global_a = a_tk.copy()
            print("  Top-k: obj=%.6f" % obj_tk)
        else:
            print("  Top-k: obj=%.6f (no improvement)" % obj_tk)

    # === Phase 3: High-beta polish ===
    if global_a is not None and time.monotonic() < deadline - 100:
        n = len(global_a)
        S_n = np.sqrt(2.0 * n)
        rem = (deadline - time.monotonic()) * 0.20
        t0 = time.monotonic()
        print("=== Phase 3: High-beta polish (%.0f sec) ===" % rem)

        beta_sched_hi = [(0.0, 50.0), (0.5, 200.0), (1.0, 800.0)]
        a_p, s_p = run_fista(global_a, n, S_n, rem, rng, beta_schedule=beta_sched_hi)
        a_p = safe_normalize(a_p, n)
        c_p = fft_convolve_self(a_p)
        obj_p = 2.0 * n * c_p.max() / S_n ** 2

        if obj_p < global_best:
            global_best = obj_p
            global_a = a_p.copy()
            print("  High-beta: obj=%.6f" % obj_p)
        else:
            print("  High-beta: obj=%.6f (no improvement)" % obj_p)

    # === Phase 4: FFT perturbation + re-optimization ===
    if global_a is not None and time.monotonic() < deadline - 80:
        n = len(global_a)
        S_n = np.sqrt(2.0 * n)
        rem = (deadline - time.monotonic()) * 0.20
        t0 = time.monotonic()
        print("=== Phase 4: FFT perturb + re-opt (%.0f sec) ===" % rem)

        best_perturb = global_a.copy()
        best_perturb_score = global_best

        for trial in range(5):
            if time.monotonic() > deadline - 40:
                break
            strength = 0.03 + 0.02 * trial
            a_pert = make_fft_perturb(global_a, rng, strength)
            a_pert = safe_normalize(a_pert, n)
            # Quick FISTA on perturbed
            seg = rem / 5.0
            beta_sched = [(0.0, 10.0), (0.5, 50.0), (1.0, 300.0)]
            a_opt, s_opt = run_fista(a_pert, n, S_n, seg, rng, beta_schedule=beta_sched)
            a_opt = safe_normalize(a_opt, n)
            c_opt = fft_convolve_self(a_opt)
            obj_opt = 2.0 * n * c_opt.max() / S_n ** 2

            if obj_opt < best_perturb_score:
                best_perturb_score = obj_opt
                best_perturb = a_opt.copy()
                print("  FFT pert %d: obj=%.6f" % (trial, obj_opt))

        if best_perturb_score < global_best:
            global_best = best_perturb_score
            global_a = best_perturb.copy()

    # === Phase 5: Coordinate polish ===
    if global_a is not None and time.monotonic() < deadline - 30:
        n = len(global_a)
        S_n = np.sqrt(2.0 * n)
        rem = max(10.0, (deadline - time.monotonic()) * 0.15)
        print("=== Phase 5: Coordinate polish (%.0f sec) ===" % rem)
        a_cp, s_cp = coordinate_polish(global_a, n, S_n, rem, rng, num_passes=3)
        a_cp = safe_normalize(a_cp, n)
        c_cp = fft_convolve_self(a_cp)
        obj_cp = 2.0 * n * c_cp.max() / S_n ** 2
        if obj_cp < global_best:
            global_best = obj_cp
            global_a = a_cp.copy()
            print("  Coord: obj=%.6f" % obj_cp)

    # === Phase 6: Try longer sequence if time permits ===
    if time.monotonic() < deadline - 120:
        n_long = 16384
        S_long = np.sqrt(2.0 * n_long)
        rem = min(100.0, (deadline - time.monotonic()) * 0.3)
        if rem > 30:
            t0 = time.monotonic()
            print("=== Phase 6: Long sequence n=%d (%.0f sec) ===" % (n_long, rem))

            if parent is not None:
                from numpy import interp
                tt_old = np.linspace(0, 1, len(parent))
                tt_new = np.linspace(0, 1, n_long)
                a_long = interp(tt_new, tt_old, parent)
                a_long = np.maximum(a_long, 0.0)
            else:
                a_long = make_anti_palindromic_init(n_long, rng, S_long)

            a_long = safe_normalize(a_long, n_long)
            beta_sched = [(0.0, 2.0), (0.5, 30.0), (1.0, 200.0)]
            a_l, s_l = run_fista(a_long, n_long, S_long, rem * 0.7, rng, beta_schedule=beta_sched)
            a_l = safe_normalize(a_l, n_long)
            c_l = fft_convolve_self(a_l)
            obj_l = 2.0 * n_long * c_l.max() / S_long ** 2

            if obj_l < global_best:
                global_best = obj_l
                global_a = a_l.copy()
                print("  Long: obj=%.6f" % obj_l)
            else:
                print("  Long: obj=%.6f (no improvement)" % obj_l)

    # === Final verification ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL: eval=%.6f, internal=%.6f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        print("  WARN: rejected, using fallback")
        final_list = [1.0] * 1024
        ver = evaluate_sequence(final_list)

    return final_list
