"""

Strategy: Multi-n exploration with peak-peeling exact subgradient,
Barzilai-Borwein adaptive steps, structured perturbations for escaping
local optima, and diversity via multiple initializations and sequence lengths.
Key innovations: single-peak focus, BB step size, block-shifting perturbations,

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient of sum_k weights[k]*c[k] w.r.t. a, where c=convolve(a,a)."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def exact_peak_weight(c, top_k=1):
    """Uniform weight on top-k peaks (exact subgradient approximation)."""
    top_k = min(top_k, len(c))
    idx = np.argpartition(c, -top_k)[-top_k:]
    w = np.zeros(len(c))
    w[idx] = 1.0 / top_k
    return w


def bb_step_size(g_diff, x_diff, prev_step):
    """Barzilai-Borwein step size computation."""
    num = np.dot(x_diff, g_diff)
    den = np.dot(g_diff, g_diff)
    if den < 1e-30:
        return prev_step
    step = num / den
    if step < 1e-10 or not np.isfinite(step) or step > 100.0:
        return prev_step
    return np.clip(step, 1e-8, 100.0)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def init_random(n, rng):
    """Random uniform initialization."""
    return rng.uniform(0.1, 1.0, n)


def init_sine(n, rng):
    """Sine-based initialization with random frequency and phase."""
    freq = rng.uniform(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    t = np.arange(n) / n
    a = np.abs(np.sin(2 * np.pi * freq * t + phase))
    return a + 0.01


def init_binary_sparse(n, rng, density=0.5):
    """Binary sequence with given density."""
    a = (rng.random(n) < density).astype(np.float64)
    if a.sum() == 0:
        a[0] = 1.0
    return a


def init_gaussian_envelope(n, rng):
    """Gaussian envelope times random."""
    center = n // 2
    sigma = n * rng.uniform(0.2, 0.5)
    t = np.arange(n)
    env = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    return env + rng.uniform(0, 0.3, n)


def structured_perturb(a, rng, scale=0.01):
    """Apply structured perturbations: block shifts, flips, scalings."""
    n = len(a)
    S = a.sum()
    method = rng.integers(4)
    if method == 0:
        # Block circular shift
        block = rng.integers(max(1, n // 64), max(2, n // 8))
        a_new = np.roll(a, rng.integers(-block, block + 1))
    elif method == 1:
        # Flip a random middle segment
        seg_len = rng.integers(max(2, n // 16), max(4, n // 4))
        start = rng.integers(0, max(1, n - seg_len))
        a_new = a.copy()
        a_new[start:start + seg_len] = a[start:start + seg_len][::-1]
    elif method == 2:
        # Multiplicative block scaling
        block = n // rng.integers(4, 16)
        a_new = a.copy()
        idx = rng.integers(0, n - block)
        a_new[idx:idx + block] *= rng.uniform(0.5, 1.5)
    else:
        # Smooth sinusoidal perturbation
        freq = rng.uniform(2, 20)
        t = np.linspace(0, 1, n)
        a_new = a + scale * S * np.sin(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
    return np.maximum(a_new, 0.0) * (S / max(a_new.sum(), 1e-15))


def run_peak_peel(a_init, n, S, max_time, rng, beta_start=10.0, beta_end=500.0):

    Peak-peeling optimization: alternates between focusing on a single peak
    and using smoothed gradient. Uses BB step size and periodic structured
    perturbations to escape local optima.
    """
    a = safe_normalize(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (n * 0.5)
    prev_a = a.copy()
    prev_g = None
    stall = 0
    best_stall_reset = best_score

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: start smooth, end sharp
        progress = min(1.0, it / max(1, int(max_time * 10)))
        beta = beta_start * (beta_end / beta_start) ** progress

        c = fft_convolve_self(y)
        cur = c.max()

        if cur < best_score - 1e-10:
            best_score = cur
            best_local = y.copy()
            stall = 0
        else:
            stall += 1

        # Choose weighting strategy based on stall
        if stall < 50:
            # Smooth gradient
            w = softmax_weights(c, beta)
        elif stall < 200:
            # Focus on top-3 peaks
            w = exact_peak_weight(c, top_k=min(3, it // 50 + 1))
        else:
            # Focus on single peak (exact subgradient)
            w = exact_peak_weight(c, top_k=1)

        grad = compute_gradient_fft(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # BB step size update
        if prev_g is not None:
            x_diff = y - prev_a
            g_diff = grad - prev_g
            step_bb = bb_step_size(g_diff, x_diff, step)
            step = 0.5 * step + 0.5 * step_bb

        # Normalized gradient step
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        prev_a = a.copy()
        prev_g = grad.copy()

        # Adaptive step reduction on stall
        if stall > 300:
            step *= 0.5
            stall = 150

        # Structured perturbation on longer stalls
        if stall > 500 and rng.random() < 0.3:
            y = structured_perturb(y, rng, scale=0.02)
            y = safe_normalize(y, n)
            t = 1.0
            step = S / (n * 0.5)
            prev_g = None
            stall = 0

        # Random restart on very long stall
        if stall > 2000 and rng.random() < 0.1:
            noise = rng.normal(0, 0.05 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
            step = S / (n * 0.5)
            prev_g = None
            stall = 0

    return best_local, best_score

def run_mirror_descent(a_init, n, S, max_time, rng):

        else:

    total_weight = sum(weights)

    time_cursor = 0.0
    results = []

    for (n, inits) in configs:
        if time.monotonic() > deadline - 60:
            break

        config_time = (budget_s - 40.0) * weights[configs.index((n, inits))] / total_weight
        # Cap each config
        config_time = min(config_time, 600.0)
        # Ensure at least 20s
        config_time = max(config_time, 20.0)
        if time.monotonic() + config_time > deadline - 30:
            config_time = deadline - 30 - time.monotonic()
        if config_time < 15:
            continue

        print("=== n=%d, time=%.0fs, inits=%s ===" % (n, config_time, inits))
        S = np.sqrt(2.0 * float(n))

        for init_name in inits:
            if time.monotonic() > deadline - 40:
                break

            # Create initialization
            if init_name == 'parent' and given_a is not None:
                if len(given_a) == n:
                    a_init = given_a.copy()
                else:
                    # Interpolate or resize
                    from numpy import interp
                    old_n = len(given_a)
                    new_t = np.linspace(0, 1, n)
                    old_t = np.linspace(0, 1, old_n)
                    a_init = interp(new_t, old_t, given_a)
            elif init_name == 'random':
                a_init = init_random(n, rng)
            elif init_name == 'sine':
                a_init = init_sine(n, rng)
            elif init_name == 'binary':
                a_init = init_binary_sparse(n, rng, density=rng.uniform(0.3, 0.7))
            elif init_name == 'gaussian':
                a_init = init_gaussian_envelope(n, rng)
            else:
                a_init = np.ones(n)

            a_init = safe_normalize(a_init, n)
            c_init = fft_convolve_self(a_init)
            init_score = c_init.max()
            obj_init = 2.0 * n * init_score / (S * S)
            print("  Init '%s': score=%.6f, obj=%.6f" % (init_name, init_score, obj_init))

            if init_score < global_best:
                global_best = init_score
                global_a = a_init.copy()

            # Allocate time between algorithms
            per_algo = config_time / 4.0
            used_time = 0.0

            # Algorithm 1: Peak peeling (most promising)
            if time.monotonic() < deadline - 50 and per_algo > 5:
                t0 = time.monotonic()
                a_res, score_res = run_peak_peel(a_init, n, S, per_algo, rng,
                                                  beta_start=5.0, beta_end=300.0)
                elapsed = time.monotonic() - t0
                used_time += elapsed
                a_res = safe_normalize(a_res, n)
                c_res = fft_convolve_self(a_res)
                s_res = c_res.max()
                if s_res < global_best:
                    global_best = s_res
                    global_a = a_res.copy()
                    print("  PeakPeel '%s': %.6f (obj %.6f) [%.0fs]" % (
                        init_name, s_res, 2*n*s_res/S**2, elapsed))

            # Algorithm 2: Adaptive smooth
            if time.monotonic() < deadline - 50 and (config_time - used_time) > 5:
                t0 = time.monotonic()
                remaining_algo = min(per_algo, config_time - used_time - 5)
                if remaining_algo > 5:
                    a_res, score_res = run_adaptive_smooth(a_init, n, S, remaining_algo, rng)
                    elapsed = time.monotonic() - t0
                    used_time += elapsed
                    a_res = safe_normalize(a_res, n)
                    c_res = fft_convolve_self(a_res)
                    s_res = c_res.max()
                    if s_res < global_best:
                        global_best = s_res
                        global_a = a_res.copy()
                        print("  AdaSmooth '%s': %.6f (obj %.6f) [%.0fs]" % (
                            init_name, s_res, 2*n*s_res/S**2, elapsed))

            # Algorithm 3: Two-phase
            if time.monotonic() < deadline - 50 and (config_time - used_time) > 10:
                t0 = time.monotonic()
                remaining_algo = min(per_algo, config_time - used_time - 5)
                if remaining_algo > 5:
                    a_res, score_res = run_two_phase(a_init, n, S, remaining_algo, rng)
                    elapsed = time.monotonic() - t0
                    used_time += elapsed
                    a_res = safe_normalize(a_res, n)
                    c_res = fft_convolve_self(a_res)
                    s_res = c_res.max()
                    if s_res < global_best:
                        global_best = s_res
                        global_a = a_res.copy()
                        print("  TwoPhase '%s': %.6f (obj %.6f) [%.0fs]" % (
                            init_name, s_res, 2*n*s_res/S**2, elapsed))

            # Algorithm 4: Mirror descent (diversity)
            if time.monotonic() < deadline - 50 and (config_time - used_time) > 5:
                t0 = time.monotonic()
                remaining_algo = min(per_algo * 0.5, config_time - used_time - 3)
                if remaining_algo > 3:
                    a_res, score_res = run_mirror_descent(a_init, n, S, remaining_algo, rng)
                    elapsed = time.monotonic() - t0
                    used_time += elapsed
                    a_res = safe_normalize(a_res, n)
                    c_res = fft_convolve_self(a_res)
                    s_res = c_res.max()
                    if s_res < global_best:
                        global_best = s_res
                        global_a = a_res.copy()
                        print("  Mirror '%s': %.6f (obj %.6f) [%.0fs]" % (
                            init_name, s_res, 2*n*s_res/S**2, elapsed))

    # === Final polishing phase ===
    remaining = deadline - time.monotonic()
    print("\n=== Final polishing: %.0fs remaining ===" % remaining)

    if global_a is not None and remaining > 10:
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * float(n_final))
        a_polish = global_a.copy()

        # Very focused peak shaving with backtracking
        best_polish = global_best
        step = S_final / n_final * 0.3

        while time.monotonic() < deadline - 10:
            c = fft_convolve_self(a_polish)
            cur = c.max()
            if cur < best_polish - 1e-12:
                best_polish = cur

            w = exact_peak_weight(c, top_k=1)
            grad = compute_gradient_fft(a_polish, w)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break

            # Backtracking
            alpha = step
            improved = False
            for _ in range(4):
                x_trial = project_simplex(a_polish - alpha * grad / max(gnorm, 1e-15), S_final)
                c_trial = fft_convolve_self(x_trial)
                if c_trial.max() < cur * (1 - 1e-7):
                    a_polish = x_trial
                    improved = True
                    break
                alpha *= 0.4

            if improved:
                step = min(step * 1.05, S_final / n_final)
            else:
                step *= 0.6
                if step < 1e-10:
                    break
                # Tiny perturbation
                noise = rng.normal(0, 0.0005 * S_final / np.sqrt(n_final), n_final)
                a_polish = project_simplex(a_polish + noise, S_final)

            if best_polish < global_best:
                global_best = best_polish
                global_a = a_polish.copy()

        if best_polish < global_best + 1e-12:
            global_best = best_polish
            global_a = a_polish.copy()
            n_final_check = len(global_a)
            S_check = np.sqrt(2.0 * float(n_final_check))
            print("  Polished: %.6f (obj %.6f)" % (global_best, 2*n_final_check*global_best/S_check**2))

    # === Final output ===
    if global_a is None:
        final_list = [0.5] * 256
    else:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]

    n_out = len(final_list)
    S_out = np.sqrt(2.0 * float(n_out))
    print("\nFINAL: length=%d, obj=%.6f" % (n_out, 2.0 * n_out * global_best / (S_out * S_out)))

    # Verify with evaluate_sequence
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: %.6f" % ver)
    else:
        print("Verification FAILED, using fallback")
        final_list = [0.5] * 256
        ver2 = evaluate_sequence(final_list)
        print("Fallback verified: %.6f" % ver2)

    return final_list
