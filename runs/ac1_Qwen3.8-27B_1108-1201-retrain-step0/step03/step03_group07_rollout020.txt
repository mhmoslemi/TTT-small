import numpy as np

from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgrad_fft(c, a, W):
    """
    Compute subgradient g[m] = sum_j W[j] * (a[j-m] + a[m-j]) for m=0..n-1.
    W is a weight vector of length 2n-1.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    L = 1 << (3 * n - 2 - 1).bit_length()
    A = fft(a, L)
    Wf = fft(W, L)
    # first[m] = sum_i a[i] * W[i+m] = cross-correlation
    first = np.real(ifft(A * np.conj(Wf)))
    # second[m] = sum_i W[i] * a[m-i] = linear convolution
    second = np.real(ifft(Wf * A))
    g = first[:n] + second[:n]
    return g


def ideal_shape(n):
    """
    Compute the 'ideal' shape for flat autocorrelation using the recursion:

    c = np.zeros(n)
    c[0] = 1.0

        for i in range(1, k):
            s += c[i] * c[k - i]

    c = np.maximum(c, 0.0)

Uses ideal-shape initialization (from flat autocorrelation recursion),

import numpy as np

def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgrad_fft(a, W):
    """
    Compute subgradient g[m] = sum_j W[j]*(a[j-m]+a[m-j]) for m=0..n-1.
    W is a weight vector of length 2n-1.
    """
    n = len(a)
    L = 1 << (3 * n - 2 - 1).bit_length()
    A = fft(a, L)
    Wf = fft(W, L)
    first = np.real(ifft(A * np.conj(Wf)))
    second = np.real(ifft(Wf * A))
    return first[:n] + second[:n]


def ideal_shape(n):
    """Compute the 'ideal' shape for flat autocorrelation."""
    c = np.zeros(n)
    c[0] = 1.0
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            s += c[i] * c[k - i]
        c[k] = (1.0 - s) / 2.0
    return np.maximum(c, 0.0)


def normalize(a, target_sum):
    """Normalize a to have the given sum, keeping non-negativity."""
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) * target_sum / len(a)
    return a * (target_sum / s)


def obj_from_conv(c, n, S):
    """Compute objective from convolution and sum."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= S + 1e-15:
        return v + (S - s) / n
    lo, hi = 0.0, v.max()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s_mid = np.maximum(v - mid, 0.0).sum()
        if s_mid > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.maximum(v - tau, 0.0)
