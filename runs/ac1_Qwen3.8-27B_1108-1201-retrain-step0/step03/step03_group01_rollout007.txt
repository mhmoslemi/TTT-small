        k_star = int(c.argmax())
        # Compute sensitivity for each element
        s = np.zeros(n)
        # a[i] contributes to c[k_star] as a[i]*a[i+k_star] (if i+k_star < n)
        # and a[i-k_star]*a[i] (if i-k_star >= 0)

        for i in range(n):

            if i + k_star < n:
                s[i] += a[i + k_star]
            if i - k_star >= 0:
                s[i] += a[i - k_star]
        
        # Transfer from high-s to low-s
        s_sorted = np.sort(s)
        median_s = s_sorted[n // 2]
        donors = a[s > median_s]
        receivers = a[s < median_s]
        
        if len(donors) == 0 or len(receivers) == 0:
            continue

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def soft_clip_normalize(v, S):
    """Soft non-negativity: clip negatives to 0, then normalize to sum S."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.zeros(len(v))
    return v * (S / s)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the score 2*n*max(c)/sum^2 for a sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def safe_normalize(a, S):

    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def fourier_init(n, S, rng, flatness=0.9):
    """
    Initialize by targeting a near-flat power spectrum.
    Start with flat spectrum, add controlled perturbation, inverse FFT, project.
    """
    N = 1 << (2 * n - 1).bit_length()
    m = np.arange(N)
    # Target: mostly flat with slight DC peak
    P_target = np.ones(N)
    P_target[0] = 1.0 + (1.0 / flatness - 1.0)
    # Add small random perturbation to break symmetry
    P_target = P_target * (1.0 + 0.05 * (rng.random(N) - 0.5))
    P_target = np.maximum(P_target, 0.1)
    # Random phase with conjugate symmetry for real output
    phase = rng.uniform(0, 2 * np.pi, N)
    # Enforce Hermitian symmetry
    phase_sym = np.zeros(N)
    phase_sym[0] = 0.0
    if N % 2 == 0:
        phase_sym[N // 2] = 0.0
    for i in range(1, N // 2 + 1):
        if i != N - i:
            phase_sym[i] = phase[i]
            phase_sym[N - i] = -phase[i]
    A = np.sqrt(P_target) * np.exp(1j * phase_sym)
    a_raw = np.real(ifft(A))[:n]
    a = project_simplex(a_raw, S)
    return a

def spectral_shaping(a, N, n, S, top_k, alpha, use_soft_clip=False):
    """
    One spectral shaping step.
    Returns (a_new, c, c_max).
    """
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    if c_max < 1e-30:
        return a, c, c_max

    m = np.arange(N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a, c, c_max
    weights = top_vals / total

    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    # Ensure we only reduce power (don't amplify)
    P_new = P * (1.0 - alpha * np.maximum(cos_eff, 0.0))
    P_new = np.maximum(P_new, P * 0.01)  # Floor to avoid killing power

    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]

    if use_soft_clip:
        a_new = soft_clip_normalize(a_new_full, S)
    else:
        a_new = project_simplex(a_new_full, S)

    return a_new, c, c_max


def spectral_shaping_adaptive(a, N, n, S, max_time, rng,

                               alpha_start=0.5, alpha_end=0.05):

    Multi-scale spectral shaping: top_k grows, alpha shrinks.
    Returns best a found.
    """

def peak_peel(a, n, S, max_time, rng, n_steps_target=50000):
    """
    Direct peak peeling: use top-1 gradient with small steps.
    This directly attacks the maximum lag.
    """
    a_curr = a.copy()

        # and a[i-k_star]*a[i] if i-k_star >= 0
        lo = max(0, k_star)
        hi = min(n, n - k_star)
        # For i in [0, n-k_star): contribution is a[i+k_star]

        if n - k_star > 0:
            grad[:n - k_star] += a_curr[k_star:]
        if k_star < n:

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

def lse_gradient_step(a, n, S, beta, step_size):
    """One FISTA-like step using LSE-smoothed objective."""
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return a, c_max
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a, c_max
    a_new = project_simplex(a - step_size * grad, S)
    return a_new, c_max


def fista_lse(a_init, n, S, max_time, rng, beta_init=20.0, beta_max=500.0):

    a = safe_normalize(a_init.copy(), S)
    y = a.copy()

            a_new = peak_peel(a_best, n, S, chunk, rng)
        elif tech == "fista":
            a_new = fista_lse(a_best, n, S, chunk, rng, beta_init=100.0, beta_max=800.0)
        else:  # ng_high
            a_new = nesterov_toppk(a_best, n, S, chunk, rng, top_k_init=50, top_k_max=512)

        c_new = fft_convolve_self(a_new)
        if c_new.max() < score_best:
            score_best = c_new.max()
            a_best = a_new.copy()
            print("      [polish-%s/%d] %.6f" % (tech, n, score_best))

    return a_best, score_best


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses multi-length exploration with spectral shaping, peak peeling, and FISTA.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    budget = float(budget_s)
    deadline = t_start + max(0.0, budget - 20.0)  # Leave 20s safety margin

    global_best = np.inf
    global_a = None
    global_n = 0

    # Try to use given sequence
    given_seq = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given_seq = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Decide which lengths to try
    if given_seq is not None and 64 <= len(given_seq) <= 32768:
        lengths = [len(given_seq)]
    else:
        lengths = [8192]

    # Add some alternatives if time permits
    if budget > 1200:
        for alt_n in [4096, 16384]:
            if alt_n not in lengths:
                lengths.append(alt_n)

    # Sort by priority: given length first, then 8192, then others
    def length_priority(n):
        if given_seq is not None and n == len(given_seq):
            return 0
        if n == 8192:
            return 1
        if n == 4096:
            return 2
        if n == 16384:
            return 3
        return 4

    lengths.sort(key=length_priority)

    print("Trying lengths: %s" % lengths)
    print("Total budget: %.0fs" % budget)

    # Allocate time to each length
    time_alloc = {}
    remaining_lengths = lengths[:]
    total_weight = 0
    for n in remaining_lengths:
        w = 1.0 / (1 + length_priority(n) * 0.5)
        total_weight += w
    for n in remaining_lengths:
        w = 1.0 / (1 + length_priority(n) * 0.5)
        time_alloc[n] = minimum_time = max(60.0, (budget - 20.0) * w / total_weight * 0.8)

    # Final polish gets remaining time
    polish_budget = (budget - 20.0) * 0.15

    for n in lengths:
        S = np.sqrt(2.0 * float(n))
        N = 1 << (2 * n - 1).bit_length()

        if time.monotonic() > deadline - polish_budget - 30:
            break

        alloc = min(time_alloc[n], deadline - time.monotonic() - polish_budget - 10)
        if alloc < 30:
            break

        print("=== n=%d, budget=%.0fs ===" % (n, alloc))
        a_try, score_try = try_length(n, S, N, alloc, rng, given_seq)

        final_score = compute_score(safe_normalize(a_try, S))
        print("  Best for n=%d: %.6f (direct: %.6f)" % (n, final_score, score_try))

        if final_score < global_best:
            global_best = final_score
            global_a = safe_normalize(a_try, S).copy()
            global_n = n
            print("  NEW GLOBAL BEST: %.6f at n=%d" % (global_best, global_n))

    # Final polish on the global best
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print("=== Final polish (budget=%.0fs) ===" % remaining)
            S = np.sqrt(2.0 * float(global_n))
            N = 1 << (2 * global_n - 1).bit_length()
            a_pol, sc_pol = final_polish(global_a, global_n, S, N, remaining, rng)
            final_sc = compute_score(safe_normalize(a_pol, S))
            if final_sc < global_best:
                global_best = final_sc
                global_a = safe_normalize(a_pol, S).copy()
                print("  POLISHED: %.6f" % global_best)

    # Safety fallback
    if global_a is None:
        print("  FALLBACK: using uniform sequence")
        global_a = np.ones(256, dtype=np.float64)
        global_n = 256

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification score: %.6f" % ver if np.isfinite(ver) else "Verification: INF")

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback verification: %.6f" % ver)

    elapsed = time.monotonic() - t_start
    print("Done in %.1fs. Best internal: %.6f, n=%d" % (elapsed, global_best, global_n))
    return final_list
