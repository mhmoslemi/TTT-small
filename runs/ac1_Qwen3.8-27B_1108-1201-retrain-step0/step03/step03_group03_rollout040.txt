"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Anti-palindromic structured initialization
2. Exact top-k subgradient with Nesterov momentum
3. Barzilai-Borwein adaptive step size
4. Periodic coordinate polish and pairing heuristics
5. Multi-length exploration (8192, 16384)
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_max_conv(a, c, top_k=10, temperature=1.0):
    """Compute subgradient of max(conv(a,a)) using top-k lags with softmax weights."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices]
    
    # Softmax weights (sharper = more focus on the max)
    tv_max = top_vals.max()
    weights = np.exp((top_vals - tv_max) / max(temperature, 1e-10))
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for i, lag in enumerate(top_indices):
        lag = int(lag)
        w = weights[i]
        # For lag k, subgradient at position j is:
        # 2*a[k-j] if 0 <= k-j < n and j != k-j
        # a[j] if j == k/2
        # 0 otherwise
        # Efficient computation:
        # The subgradient vector for lag k is 2 * reverse(a padded) shifted
        # Specifically, sg_k = 2 * flip(a) but only the part that overlaps
        
        # For lag k: valid j are those with 0 <= j < n and 0 <= k-j < n
        # i.e., max(0, k-(n-1)) <= j <= min(k, n-1)
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        if j_min > j_max:
            continue
        js = np.arange(j_min, j_max + 1)
        partners = k - js
        vals = 2.0 * a[partners]
        # Handle the case where j == k/2 (counted twice above, should count once)
        if k % 2 == 0:
            mid = k // 2
            if j_min <= mid <= j_max:
                idx_in = mid - j_min
                vals[idx_in] = a[mid]
        grad[js] += w * vals
    
    return grad


def objective(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def run_optimizer(a_init, n, S, max_time, rng, top_k=10, beta_temp=1.0):
    """Main optimization loop with Nesterov momentum and BB step size."""
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()  # Nesterov iterate
    t_mom = 1.0
    start = time.monotonic()
    best_score = objective(a)
    best_a = a.copy()
    it = 0
    
    # BB step size memory
    a_prev = a.copy()
    g_prev = None
    step_bb = S / (n * 5.0)  # initial step
    
    # Momentum restart counter
    stall = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_score:
            best_score = cur_obj
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute subgradient
        temp = max(0.1, beta_temp * np.exp(-0.001 * it))
        grad = subgradient_max_conv(y, c, top_k=top_k, temperature=temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # BB step size
        if g_prev is not None:
            s_vec = y - a_prev
            y_vec = grad - g_prev
            sy = np.dot(s_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            if sy > 1e-15 and ss > 1e-15:
                step_bb = ss / sy
                step_bb = np.clip(step_bb, 1e-6 * S / n, 10.0 * S / n)
        
        a_prev = y.copy()
        g_prev = grad.copy()
        
        # Backtracking line search
        step = step_bb
        f_y = cur_max
        for bt in range(8):
            x_new = project_simplex(y - step * grad, S)
            if bt > 0:
                c_new = fft_convolve_self(x_new)
                if c_new.max() <= f_y + 1e-8:
                    break
            step *= 0.5
        
        a = x_new
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = a + ((t_mom - 1.0) / t_new) * (a - a_prev)
        y = np.maximum(y, 0.0)
        # Renormalize y to maintain sum = S (Nesterov can drift)
        s_y = y.sum()
        if s_y > 0:
            y = y * (S / s_y)
        else:
            y = a.copy()
        t_mom = t_new
        
        # Periodic restart
        if stall > 50 and rng.random() < 0.2:
            noise = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0
            stall = 0
            a_prev = y.copy()
            g_prev = None
            step_bb = S / (n * 5.0)
    
    return best_a, best_score


def coordinate_polish(a, n, S, max_time, rng, top_k=20):
    """Polish by trying coordinate-wise moves."""
    a = np.clip(a, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    a = a * (S / s)
    
    start = time.monotonic()
    best_score = objective(a)
    best_a = a.copy()
    
    # Find the most "active" coordinates (high gradient)
    c = fft_convolve_self(a)
    grad = subgradient_max_conv(a, c, top_k=top_k, temperature=0.5)
    
    # Sort coordinates by |gradient|
    active_coords = np.argsort(np.abs(grad))[::-1][:min(500, n)]
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        # Cycle through active coordinates
        for idx in active_coords:
            if time.monotonic() - start > max_time:
                    break
            idx = int(idx)
            # Try moving mass from idx to a random other coordinate
            for trial in range(3):
                if rng.random() < 0.5:
                    # Decrease idx, increase random j
                    j = rng.integers(n)
                    if j == idx:
                        continue
                    delta = a[idx] * rng.uniform(0.01, 0.1)
                    a_trial = a.copy()
                    a_trial[idx] -= delta
                    a_trial[j] += delta
                    if a_trial[idx] < 0:
                        continue
                else:
                    # Increase idx, decrease random j
                    j = rng.integers(n)
                    if j == idx:
                        continue
                    delta = a[j] * rng.uniform(0.01, 0.1)
                    a_trial = a.copy()
                    a_trial[j] -= delta
                    a_trial[idx] += delta
                    if a_trial[j] < 0:
                        continue
                
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < c.max() * 0.99999:  # Require significant improvement
                    a = a_trial
                    c = c_trial
                    best_score = 2.0 * n * c.max() / (S * S)
                    best_a = a.copy()
                    break
    
    return best_a, best_score


def make_anti_palindromic(n, rng, S, smoothness=0.3):
    """Create anti-palindromic initialization with some smoothness."""
    half = n // 2
    # Generate smooth random first half
    h = rng.random(half)
    # Apply smoothing
    from scipy.ndimage import uniform_filter1d
    h = uniform_filter1d(h, size=max(2, half // 50))
    h = h + 0.1  # Ensure positive
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_cosine_anti_palindromic(n, rng, S, freq):
    """Anti-palindromic with cosine structure."""
    half = n // 2
    t = np.arange(half) / half
    h = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t + rng.random() * 2 * np.pi)
    h = np.maximum(h, 0.1)
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, c, top_k, temperature):
    """Compute subgradient of max(conv(a,a)) using top-k lags with softmax weighting.
    
    For lag k, the subgradient at position j is 2*a[k-j] if 0<=k-j<n, else 0.
    """
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices]

    tv_max = top_vals.max()
    weights = np.exp((top_vals - tv_max) / max(temperature, 1e-10))
    weights = weights / weights.sum()

    grad = np.zeros(n)
    for i in range(k):
        lag = int(top_indices[i])
        w = weights[i]
        if w < 1e-12:
            continue
        j_min = max(0, lag - (n - 1))
        j_max = min(lag, n - 1)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            grad[js] += w * 2.0 * a[lag - js]

    return grad

    n = len(a)

    if S < 1e-15:

    if c is None:
        c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (S * S))

        for bt in range(10):
            candidate = y - step * grad
            x_new = project_simplex(candidate, S)
            if bt > 0:
                c_new = fft_convolve_self(x_new)
                if c_new.max() <= f_y + 1e-10 * S:
                    break
            step *= 0.5

        a = x_new

        # Heavy-ball momentum update
        y_new = (1.0 + momentum) * a - momentum * a_prev
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y = y_new * (S / s_y)
        else:
            y = a.copy()

        # Periodic restart to escape local minima
        if stall > patience and rng.random() < 0.15:
            noise_scale = rng.uniform(0.002, 0.01) * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a_prev = y.copy()
            g_prev = None
            step_bb = S / (n * 5.0)
            stall = 0
            if verbose:
                print("    Restart at it=%d, stall=%d" % (it, stall))

    return best_a, best_score


def pair_move_polish(a, n, S, max_time, rng, n_pairs=100):
    """Local search: try moving mass between pairs of coordinates.
    Focus on coordinates that contribute most to the peak autocorrelation lags.
    """
    a = np.clip(a, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_max = c.max()
    best_a = a.copy()
    it = 0

    # Identify important coordinates: those in the top lags
    top_lags = np.argsort(c)[::-1][:10]
    important = set()
    for lag in top_lags:
        lag = int(lag)
        j_min = max(0, lag - (n - 1))
        j_max = min(lag, n - 1)
        for j in range(j_min, j_max + 1):
            if a[j] > 0 or a[lag - j] > 0:
                important.add(j)
                important.add(lag - j)
    important_list = list(important)[:min(2000, len(important))]

    # Also include highest-value coordinates
    top_coords = np.argsort(a)[::-1][:min(500, n)]
    for tc in top_coords:
        if int(tc) not in important:
            important.add(int(tc))
    important_list = list(important)[:min(3000, len(important))]

    while time.monotonic() - start < max_time:
        it += 1
        improved = False

        # Shuffle which pairs to try
        rng.shuffle(important_list)
        for idx in important_list:
            if time.monotonic() - start > max_time:
                break
            idx = int(idx)
            if a[idx] < 1e-12:
                continue

            # Try moving a fraction of a[idx] to a few random other positions
            n_tries = 3
            for _ in range(n_tries):
                j = rng.integers(n)
                if j == idx:
                    continue

                # Try decreasing idx and increasing j
                frac = rng.uniform(0.02, 0.15)
                delta = a[idx] * frac
                a[idx] -= delta
                a[j] += delta

                # Quick check: recompute only affected lags
                # For efficiency, just check the global max periodically
                if it % 5 == 0 or delta > 0.01 * S:
                    c_new = fft_convolve_self(a)
                    if c_new.max() > best_max * 1.001:
                        # Revert
                        a[idx] += delta
                        a[j] -= delta
                    else:
                        c = c_new
                        if c.max() < best_max:
                            best_max = c.max()
                            best_a = a.copy()
                            improved = True
                        else:
                            # Slight increase, revert
                            a[idx] += delta
                            a[j] -= delta
                else:
                    # Small change, keep it (will verify later)
                    pass

        # Periodic full verification
        if it % 10 == 0:
            c = fft_convolve_self(a)
            if c.max() >= best_max * 1.01:
                # Too much drift, revert to best
                a = best_a.copy()
                c = fft_convolve_self(a)
            elif c.max() < best_max:
                best_max = c.max()
                best_a = a.copy()

    return best_a, 2.0 * n * best_max / (S * S)


def make_anti_palindromic_smooth(n, rng, S, n_freqs=3):
    """Anti-palindromic init with smooth multi-frequency structure."""
    half = n // 2
    t = np.arange(half, dtype=np.float64) / half
    h = np.ones(half)
    for _ in range(n_freqs):
        freq = rng.integers(1, 15)
        phase = rng.random() * 2 * np.pi
        amp = rng.uniform(0.1, 0.5)
        h += amp * np.cos(2 * np.pi * freq * t + phase)
    h = np.maximum(h, 0.05)
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-15:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_anti_palindromic_planar(n, rng, S):
    """Anti-palindromic with a linear ramp in the first half."""
    half = n // 2
    t = np.arange(half, dtype=np.float64) / half
    slope = rng.uniform(-0.5, 0.5)
    h = 1.0 + slope * (2 * t - 1) + rng.normal(0, 0.05, half)
    h = np.maximum(h, 0.05)
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-15:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_two_block(n, rng, S):
    """Two-block initialization: first half high, second half low."""
    ratio = rng.uniform(0.3, 0.7)
    a = np.ones(n)
    a[:n // 2] = 1.0 + ratio
    a[n // 2:] = 1.0 - ratio * 0.5
    a = np.maximum(a, 0.05)
    # Make it anti-palindromic
    half = n // 2
    mu = S / half
    h = a[:half].copy()
    a_new = np.zeros(n)
    a_new[:half] = h
    for i in range(half):
        a_new[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a_new[half] = mu / 2
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() > 1e-15:
        a_new = a_new * (S / a_new.sum())
    return a_new


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Parent: %s (len=%s)" % ("yes" if parent is not None else "no",
                                    len(parent) if parent is not None else 0))

    # === Phase 1: Optimize from parent (highest priority) ===
    if parent is not None:
        n = len(parent)
        S = np.sqrt(2.0 * n)
        a0 = parent * (S / parent.sum())

        t0 = time.monotonic()
        time_budget = min(700.0, max(120.0, (deadline - time.monotonic()) * 0.35))
        print("Phase 1: Parent optimization, n=%d, budget=%.0fs" % (n, time_budget))

        a1, s1 = run_subgradient_optimizer(a0, n, S, time_budget, rng,
                                            top_k_init=20, top_k_min=2,
                                            temp_init=5.0, temp_min=0.02,
                                            momentum=0.9)
        c1 = fft_convolve_self(a1)
        obj1 = obj_value(a1, c1)
        print("  Result: obj=%.8f, time=%.0fs" % (obj1, time.monotonic() - t0))

        if obj1 < global_best:
            global_best = obj1
            global_a = a1.copy()

        # Pair-move polish
        if time.monotonic() < deadline - 60:
            t0 = time.monotonic()
            time_budget2 = min(300.0, max(60.0, (deadline - time.monotonic()) * 0.15))
            print("  Pair-move polish, budget=%.0fs" % time_budget2)
            a2, obj2 = pair_move_polish(a1, n, S, time_budget2, rng)
            c2 = fft_convolve_self(a2)
            obj2 = obj_value(a2, c2)
            print("  Result: obj=%.8f, time=%.0fs" % (obj2, time.monotonic() - t0))

            if obj2 < global_best:
                global_best = obj2
                global_a = a2.copy()

    # === Phase 2: Try structured initializations at n=8192 ===
    n2 = 8192
    S2 = np.sqrt(2.0 * n2)

    if time.monotonic() < deadline - 150:
        inits = []
        inits.append(("ap_smooth3", make_anti_palindromic_smooth(n2, rng, S2, n_freqs=3)))
        inits.append(("ap_smooth5", make_anti_palindromic_smooth(n2, rng, S2, n_freqs=5)))
        inits.append(("ap_planar", make_anti_palindromic_planar(n2, rng, S2)))
        inits.append(("two_block", make_two_block(n2, rng, S2)))

        time_per = max(30.0, (deadline - time.monotonic()) * 0.20 / len(inits))
        print("Phase 2: %d structured inits at n=%d, %.0fs each" % (len(inits), n2, time_per))

        for name, a_init in inits:
            if time.monotonic() > deadline - 120:
                break
            t0 = time.monotonic()
            a_opt, s_opt = run_subgradient_optimizer(a_init, n2, S2, time_per, rng,
                                                      top_k_init=15, top_k_min=3,
                                                      temp_init=3.0, temp_min=0.05,
                                                      momentum=0.9)
            c_opt = fft_convolve_self(a_opt)
            obj_opt = obj_value(a_opt, c_opt)
            print("  %s: obj=%.8f, time=%.0fs" % (name, obj_opt, time.monotonic() - t0))

            if obj_opt < global_best:
                global_best = obj_opt
                global_a = a_opt.copy()
                # Also do a quick polish
                if time.monotonic() < deadline - 60:
                    a_pol, obj_pol = pair_move_polish(a_opt, n2, S2, min(60.0, (deadline - time.monotonic()) * 0.05), rng)
                    c_pol = fft_convolve_self(a_pol)
                    obj_pol = obj_value(a_pol, c_pol)
                    if obj_pol < global_best:
                        global_best = obj_pol
                        global_a = a_pol.copy()
                        print("    After polish: obj=%.8f" % obj_pol)

    # === Phase 3: Try longer sequence n=16384 ===
    if time.monotonic() < deadline - 200:
        n3 = 16384
        S3 = np.sqrt(2.0 * n3)
        t0 = time.monotonic()
        print("Phase 3: n=%d" % n3)

        if parent is not None:
            tt_old = np.linspace(0, 1, len(parent))
            tt_new = np.linspace(0, 1, n3)
            a3_init = np.interp(tt_new, tt_old, parent)
            a3_init = np.maximum(a3_init, 0.0)
        else:
            a3_init = make_anti_palindromic_smooth(n3, rng, S3, n_freqs=4)

        time_budget3 = min(300.0, max(60.0, (deadline - time.monotonic()) * 0.25))
        a3, s3 = run_subgradient_optimizer(a3_init, n3, S3, time_budget3, rng,
                                            top_k_init=20, top_k_min=3,
                                            temp_init=4.0, temp_min=0.05,
                                            momentum=0.9)
        c3 = fft_convolve_self(a3)
        obj3 = obj_value(a3, c3)
        print("  Result: obj=%.8f, time=%.0fs" % (obj3, time.monotonic() - t0))

        if obj3 < global_best:
            global_best = obj3
            global_a = a3.copy()

    # === Phase 4: Final intensive polish on best solution ===
    if global_a is not None and time.monotonic() < deadline - 30:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        print("Phase 4: Final polish, n=%d, remaining=%.0fs" % (n, remaining))

        # 4a: Sharp subgradient (low temp, few top-k)
        seg = remaining * 0.35
        a_f1, s_f1 = run_subgradient_optimizer(global_a, n, S, seg, rng,
                                                top_k_init=5, top_k_min=1,
                                                temp_init=0.5, temp_min=0.01,
                                                momentum=0.85)
        c_f1 = fft_convolve_self(a_f1)
        obj_f1 = obj_value(a_f1, c_f1)
        print("  Sharp subgrad: obj=%.8f" % obj_f1)
        if obj_f1 < global_best:
            global_best = obj_f1
            global_a = a_f1.copy()

        # 4b: Pair-move polish
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            seg2 = remaining2 * 0.4
            a_f2, obj_f2 = pair_move_polish(global_a, n, S, seg2, rng)
            c_f2 = fft_convolve_self(a_f2)
            obj_f2 = obj_value(a_f2, c_f2)
            print("  Pair polish: obj=%.8f" % obj_f2)
            if obj_f2 < global_best:
                global_best = obj_f2
                global_a = a_f2.copy()

        # 4c: One more round of subgradient from the best
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            a_f3, s_f3 = run_subgradient_optimizer(global_a, n, S, remaining3, rng,
                                                    top_k_init=3, top_k_min=1,
                                                    temp_init=0.2, temp_min=0.01,
                                                    momentum=0.8)
            c_f3 = fft_convolve_self(a_f3)
            obj_f3 = obj_value(a_f3, c_f3)
            print("  Final subgrad: obj=%.8f" % obj_f3)
            if obj_f3 < global_best:
                global_best = obj_f3
                global_a = a_f3.copy()

    # === Final verification ===
    if global_a is None:
        global_a = np.ones(1024) * 1.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL: eval=%.8f, internal=%.8f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        print("  Fallback to uniform!")
        global_a = np.ones(1024) * 1.0
        final_list = [1.0] * 1024

    return final_list
