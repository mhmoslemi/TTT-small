    n = len(a)

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n

        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax_weights(c, beta):
    """Softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Normalized top-k weights."""

    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))

                # Use smallest step that doesn't worsen too much
                t_small = t_fw * 0.05
                a_new = (1.0 - t_small) * a + t_small * g
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new * (S / a_new.sum())
                a = a_new
        else:
            # Fallback: subgradient of max lag
            k_max = int(c.argmax())
            grad = compute_grad_single_lag(a, k_max)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-15:
                step = 0.5 * S / (n * gnorm)
                a_new = project_simplex(a - step * grad, S)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < cur:
                    a = a_new
                else:
                    a = project_simplex(a - 0.5 * step * grad, S)
        
        # Periodic perturbation to escape local optima
        if it % 500 == 0 and rng.random() < 0.25:
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            stall_count = 0
            t_fw = t_init
    
    return best_local, best_score


def run_toppk_fista(a_init, n, S, max_time, rng, top_k_init=1, top_k_max=100, top_k_rate=100):
    """Top-k subgradient FISTA (fallback method)."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    fsize = 1 << (2 * n - 2).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // top_k_rate)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
        
        w = toppk_weights(c, top_k)
        W = fft(w, fsize)
        Y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = S / (n * 0.5) / max(gnorm, 1e-15)
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        a = x_new
        t_mom = t_new
        
        if it % 3000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
    
    return best_local, best_score


def run_coordinate_polish(a_init, n, S, max_time, rng):
    """
    Fine polish: for each coordinate, try small adjustments
    to reduce the max convolution value.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    
    # Identify the peak lag
    k_max = int(c.argmax())
    
    # For coordinate descent: try moving mass from position i to position j
    # The change in c_k for k = k_max is:
    # delta = delta_i * a[k_max - i] + delta_j * a[k_max - j] + cross terms
    # We want to reduce c[k_max], so move mass from high-contributing to low-contributing positions
    
    # Compute contributions to peak
    contrib = np.zeros(n)
    j_start = max(0, k_max - n + 1)
    j_end = min(n - 1, k_max)
    if j_start <= j_end:
        indices = np.arange(j_start, j_end + 1)
        contrib[indices] = a[k_max - indices]
    
    # Sort by contribution (descending) - we want to reduce high contributors
    order = np.argsort(contrib)[::-1]
    
    attempts = 0
    while time.monotonic() - start < max_time and attempts < 50000:
        attempts += 1
        # Pick a high-contributor to reduce
        i = order[rng.integers(0, min(len(order), n))]
        contrib_i = contrib[i]
        if contrib_i <= 0 or a[i] <= 0:
            # Refresh
            c = fft_convolve_self(a)
            if c.max() < best_score:
                best_score = c.max()
                best_local = a.copy()
            k_max = int(c.argmax())
            contrib = np.zeros(n)
            j_start = max(0, k_max - n + 1)
            j_end = min(n - 1, k_max)
            if j_start <= j_end:
                indices = np.arange(j_start, j_end + 1)
                contrib[indices] = a[k_max - indices]
            order = np.argsort(contrib)[::-1]
            continue
        
        # Pick a low-contributor to increase
        j_candidates = np.where(contrib < contrib_i * 0.5)[0]
        j_candidates = j_candidates[j_candidates != i]
        if len(j_candidates) == 0:
            continue
        j = j_candidates[rng.integers(0, len(j_candidates))]
        
        # Try moving a small amount from i to j
        delta = min(a[i] * 0.1, 0.01 * S / n)
        if delta <= 0:
            continue
        
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        
        c_new = fft_convolve_self(a_new)
        if c_new.max() < best_score - 1e-13:
            a = a_new
            best_score = c_new.max()
            best_local = a_new.copy()
            # Update contribution
            contrib[i] -= delta  # approximate
            contrib[j] += a[k_max - j] if 0 <= k_max - j < n else 0
    
    return best_local, best_score


def propose_candidate(seed=42, budget_s=2385):
    """
    Multi-phase optimization:
    Phase 1: FISTA with softmax smoothing (fast convergence)
    Phase 2: LP-based Frank-Wolfe (directional improvement)
    Phase 3: Top-k FISTA polish
    Phase 4: Coordinate polish
    Multiple initializations for diversity.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s - 15.0)
    
    # Get starting sequence
    given_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given_a = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    if given_a is not None:
        n = len(given_a)
    else:
        n = 8192
    
    S = np.sqrt(2.0 * float(n))
    
    # Generate multiple diverse initializations
    inits = []
    
    # Init 1: Given sequence (or uniform if not available)
    if given_a is not None:
        inits.append(safe_normalize(given_a, n))
    else:
        inits.append(np.ones(n) * S / n)
    
    # Init 2: Cosine window
    jj = np.arange(n)
    cos_init = 0.5 + 0.5 * np.cos(2 * np.pi * (jj + 0.5) / n)
    cos_init = np.maximum(cos_init, 0.01)
    inits.append(safe_normalize(cos_init, n))
    
    # Init 3: Raised cosine (sharper)
    rc_init = 0.5 * (1 + np.cos(np.pi * (jj / (n - 1) * 2 - 1)))
    rc_init = np.maximum(rc_init, 0.01)
    inits.append(safe_normalize(rc_init, n))
    
    # Init 4: Random structured
    # Use random phases -> IDFT of (phases)^2 -> take real part
    phases = np.exp(2j * np.pi * rng.random(n))
    spec = phases * np.sqrt(n)  # flat magnitude
    real_part = np.real(ifft(spec * spec, n))
    real_part = np.maximum(real_part, 0.0)
    if real_part.sum() > 1e-10:
        inits.append(safe_normalize(real_part, n))
    else:
        inits.append(np.ones(n) * S / n)
    
    # Init 5: Random (smoothed)
    raw = rng.random(n)
    smooth = np.convolve(raw, np.ones(32) / 32, mode='same')
    smooth = np.maximum(smooth, 0.0)
    inits.append(safe_normalize(smooth, n))
    
    # Evaluate initializations
    global_best = np.inf
    global_a = None
    for i, init in enumerate(inits):
        c = fft_convolve_self(init)
        score = c.max()
        if score < global_best:
            global_best = score
            global_a = init.copy()
        print("Init %d: score=%.6f, obj=%.6f" % (i, score, 2 * n * score / S**2))
    
    print("Best init: obj=%.6f" % (2.0 * n * global_best / S**2))
    
    # === Phase 1: FISTA with softmax (30% of time) ===
    remaining = deadline - time.monotonic()
    phase1_time = 0.30 * remaining
    if phase1_time > 10:
        print("\nPhase 1: FISTA softmax (%.0f sec)" % phase1_time)
        t0 = time.monotonic()
        a_p1, _ = run_fista_smoothed(global_a, n, S, phase1_time, rng,
                                      beta_start=20.0, beta_end=300.0)
        a_p1 = safe_normalize(a_p1, n)
        c_p1 = fft_convolve_self(a_p1)
        if c_p1.max() < global_best:
            global_best = c_p1.max()
            global_a = a_p1.copy()
            print("  FISTA: obj=%.6f [%.0fs]" % (2 * n * global_best / S**2, time.monotonic() - t0))
    
    # === Phase 2: LP-based Frank-Wolfe (35% of time) ===
    remaining = deadline - time.monotonic()
    phase2_time = 0.35 * remaining
    if phase2_time > 30:
        print("\nPhase 2: LP Frank-Wolfe (%.0f sec)" % phase2_time)
        t0 = time.monotonic()
        a_p2, _ = run_fw_lp(global_a, n, S, phase2_time, rng,
                            K_init=4, K_max=20, K_rate=200,
                            t_init=0.15, t_min=0.03)
        a_p2 = safe_normalize(a_p2, n)
        c_p2 = fft_convolve_self(a_p2)
        if c_p2.max() < global_best:
            global_best = c_p2.max()
            global_a = a_p2.copy()
            print("  FW-LP: obj=%.6f [%.0fs]" % (2 * n * global_best / S**2, time.monotonic() - t0))
    
    # === Phase 3: Second LP-FW pass with different params (15%) ===
    remaining = deadline - time.monotonic()
    phase3_time = 0.15 * remaining
    if phase3_time > 20:
        print("\nPhase 3: LP-FW pass 2 (%.0f sec)" % phase3_time)
        t0 = time.monotonic()
        a_p3, _ = run_fw_lp(global_a, n, S, phase3_time, rng,
                            K_init=8, K_max=32, K_rate=100,
                            t_init=0.10, t_min=0.02)
        a_p3 = safe_normalize(a_p3, n)
        c_p3 = fft_convolve_self(a_p3)
        if c_p3.max() < global_best:
            global_best = c_p3.max()
            global_a = a_p3.copy()
            print("  FW-LP2: obj=%.6f [%.0fs]" % (2 * n * global_best / S**2, time.monotonic() - t0))
    
    # === Phase 4: Top-k FISTA polish (10%) ===
    remaining = deadline - time.monotonic()
    phase4_time = 0.10 * remaining
    if phase4_time > 10:
        print("\nPhase 4: Top-k FISTA polish (%.0f sec)" % phase4_time)
        t0 = time.monotonic()
        a_p4, _ = run_toppk_fista(global_a, n, S, phase4_time, rng,
                                   top_k_init=1, top_k_max=32, top_k_rate=200)
        a_p4 = safe_normalize(a_p4, n)
        c_p4 = fft_convolve_self(a_p4)
        if c_p4.max() < global_best:
            global_best = c_p4.max()
            global_a = a_p4.copy()
            print("  TopK: obj=%.6f [%.0fs]" % (2 * n * global_best / S**2, time.monotonic() - t0))
    
    # === Phase 5: Coordinate polish (5%) ===
    remaining = deadline - time.monotonic()
    phase5_time = 0.05 * remaining
    if phase5_time > 5:
        print("\nPhase 5: Coordinate polish (%.0f sec)" % phase5_time)
        t0 = time.monotonic()
        a_p5, _ = run_coordinate_polish(global_a, n, S, phase5_time, rng)
        a_p5 = safe_normalize(a_p5, n)
        c_p5 = fft_convolve_self(a_p5)
        if c_p5.max() < global_best:
            global_best = c_p5.max()
            global_a = a_p5.copy()
            print("  Coord: obj=%.6f [%.0fs]" % (2 * n * global_best / S**2, time.monotonic() - t0))
    
    # === Phase 6: Multi-restart exploration (5%) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\nPhase 6: Multi-restart (%.0f sec)" % remaining)
        per_restart = remaining * 0.8 / 3
        for r in range(3):
            if time.monotonic() > deadline - 10:
                break
            # Different perturbation for each restart
            if r == 0:
                pert = global_a + rng.normal(0, 0.004 * S / np.sqrt(n), n)
            elif r == 1:
                # Swap pairs of values
                pert = global_a.copy()
                idx1 = rng.integers(0, n, n // 10)
                idx2 = rng.integers(0, n, n // 10)
                for ix, jx in zip(idx1, idx2):
                    pert[ix], pert[jx] = pert[jx], pert[ix]
            else:
                # Spectral perturbation
                A = fft(pert if 'pert' in dir() else global_a, 1 << (n - 1).bit_length())
                phase_pert = np.exp(2j * np.pi * rng.random(1 << (n - 1).bit_length()))
                A_pert = A * phase_pert[:len(A)]
                pert = np.real(ifft(A_pert, 1 << (n - 1).bit_length()))[:n]
            
            pert = safe_normalize(pert, n)
            # Quick FW-LP
            a_r, _ = run_fw_lp(pert, n, S, per_restart, rng,
                               K_init=4, K_max=16, K_rate=200,
                               t_init=0.15, t_min=0.03)
            a_r = safe_normalize(a_r, n)
            c_r = fft_convolve_self(a_r)
            if c_r.max() < global_best:
                global_best = c_r.max()
                global_a = a_r.copy()
                print("  Restart %d: obj=%.6f" % (r, 2 * n * global_best / S**2))
    
    # === Final output ===
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    final_obj = 2.0 * n * global_best / (S * S)
    print("\nFINAL: length=%d, score=%.6f, objective=%.6f" % (len(final_list), global_best, final_obj))
    
    # Verify with evaluate_sequence
    try:
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified: %.6f" % ver)
        else:
            print("Verification returned inf, using fallback")
            final_list = [0.5] * 256
    except Exception as e:
        print("Verification error: %s" % str(e))
    
    return final_list
