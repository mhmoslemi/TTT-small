import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    # Use the standard algorithm
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = np.max(np.where(valid)[0]) + 1
    theta_val = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta_val, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_gradient_fft(a, weights):
    """Gradient of weighted sum of convolution lags."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def softmax_weights(c, beta):
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0/len(c))
    return exp_bc / total

def spectral_flatten(a, eta, fsize=None):
    """Flatten the power spectrum while keeping DC fixed."""
    n = len(a)
    if fsize is None:
        fsize = 1 << (2*n - 1 - 1).bit_length()
    A = fft(a, fsize)
    P = np.abs(A)**2
    # Mean of non-DC bins
    P_non_dc = P[1:].mean()
    # Pull non-DC bins toward their mean
    P_new = P.copy()
    P_new[1:] = P[1:] + eta * (P_non_dc - P[1:])
    P_new = np.maximum(P_new, 0)  # Ensure non-negative
    # Reconstruct with same phase
    A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    a_new = np.real(ifft(A_new))[:n]
    return a_new

    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S*S)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Novel approach: Combines time-domain FISTA with frequency-domain spectral shaping,
multiple parametric initializations, adaptive dual-domain optimization,
and population-based diversification. The spectral step directly flattens the
power spectrum to reduce convolution peaks, providing a different optimization
landscape than pure time-domain gradient methods.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient of weighted sum of convolution lags using FFT."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Compute normalized top-k weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights


def spectral_flatten(a, eta, fsize=None):
    """
    Flatten the power spectrum of a while keeping DC fixed.
    This reduces concentration in the spectrum, which helps flatten the convolution.
    """
    n = len(a)
    if fsize is None:
        fsize = 1 << (2 * n - 2).bit_length()
    A = fft(a, fsize)
    P = np.abs(A) ** 2
    # Mean of non-DC bins
    if fsize > 1:
        P_non_dc_mean = P[1:].mean()
        # Pull non-DC bins toward their mean (flatten)
        P_new = P.copy()
        P_new[1:] = P[1:] + eta * (P_non_dc_mean - P[1:])
        # Also add a small floor to prevent zeros
        P_new[1:] = np.maximum(P_new[1:], 1e-10 * P[0])
        P_new = np.maximum(P_new, 0)
        # Reconstruct with same phase
        phase = np.angle(A)
        amplitude = np.sqrt(P_new)
        A_new = amplitude * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
    else:
        a_new = a.copy()
    return a_new


def spectral_sharp(a, eta, fsize=None):
    """
    Opposite of spectral_flatten: concentrate the power spectrum at DC.
    This is useful as an exploration step - sometimes concentrating the spectrum
    followed by time-domain optimization finds better solutions.
    """
    n = len(a)
    if fsize is None:
        fsize = 1 << (2 * n - 2).bit_length()
    A = fft(a, fsize)
    P = np.abs(A) ** 2
    if fsize > 1:
        P_dc = P[0]
        P_new = P.copy()
        # Reduce non-DC bins (pull toward 0, keeping a fraction)
        P_new[1:] = P[1:] * (1.0 - eta)
        # Add small noise to avoid stagnation
        P_new[1:] += 1e-10 * P_dc
        P_new = np.maximum(P_new, 0)
        phase = np.angle(A)
        amplitude = np.sqrt(P_new)
        A_new = amplitude * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
    else:
        a_new = a.copy()
    return a_new


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def init_parametric(n, family, rng):
    """Generate a parametric initialization of various shapes."""
    t = np.linspace(0, 1, n)
    center = (n - 1) / 2.0
    if family == 'uniform':
        a = np.ones(n)
    elif family == 'gaussian':
        sigma = n / 5.0
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    elif family == 'triangular':
        a = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float) + 1
    elif family == 'raised_cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a[0] = max(a[0], 0.01)
        a[-1] = max(a[-1], 0.01)
    elif family == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 0.01)
    elif family == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 0.001)
    elif family == 'quartic':
        x = 2 * t - 1
        a = (1 - x**4)
        a = np.maximum(a, 0.001)
    elif family == 'waterfall':
        # Asymmetric: high on left, tapering right
        a = np.exp(-2.0 * t) + 0.1
    elif family == 'bimodal':
        s1 = n / 4
        s2 = n / 4
        a = np.exp(-0.5 * ((np.arange(n) - center * 0.5) / s1) ** 2) + \
            np.exp(-0.5 * ((np.arange(n) - center * 1.5) / s2) ** 2)
    elif family == 'random_smooth':
        raw = rng.normal(0, 1, n)
        # Smooth with a long kernel
        ksize = max(3, n // 64)
        kernel = np.ones(ksize) / ksize
        a = np.convolve(raw, kernel, mode='same')
        a = a - a.min() + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    return a


def run_combined_opt(a_init, n, S, max_time, rng, fsize,
                     beta_init=30.0, beta_max=2000.0,
                     spectral_every=100, spectral_eta=0.15,
                     mode='alternating'):
    """
    Combined time-domain FISTA + frequency-domain spectral shaping.
    Alternates between gradient steps and spectral flattening steps.
    """
    a = safe_normalize(a_init, n, S)
    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    it = 0
    step = 1.0 / (beta_init * 3.0)
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            stall = 0
        else:
            stall += 1

        # Time-domain FISTA step
        if it % spectral_every != 0:
            w = softmax_weights(c, beta)
            grad = compute_gradient_fft(y, w)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum ** 2))
            y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            # Renormalize to maintain sum constraint
            s_y = y.sum()
            if s_y > 1e-15:
                y = y * (S / s_y)
            a = x_new
            t_momentum = t_new

            # Adaptive step
            if stall > 300:
                step *= 0.7
                stall = 0
            elif stall == 0:
                step = min(step * 1.02, 2.0 * S / n)
        else:
            # Spectral step: flatten the power spectrum
            a_spectral = spectral_flatten(y, spectral_eta, fsize)
            a_spectral = safe_normalize(a_spectral, n, S)
            # Also try a small time-domain step from the spectral result
            c_s = fft_convolve_self(a_spectral)
            if c_s.max() < cur:
                y = a_spectral
                a = a_spectral
                t_momentum = 1.0
                # Reset momentum after spectral step
                if c_s.max() < best_score:
                    best_score = c_s.max()
                    best_local = a_spectral.copy()
                    stall = 0
            else:
                # Spectral step didn't help; do a small perturbation instead
                noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
                y = project_simplex(y + noise, S)
                t_momentum = 1.0

    return best_local, best_score

        print("  Phase 1 done: %.6f (obj %.6f) [%.0fs]" %
              (global_best, objective_from_conv(fft_convolve_self(global_a), n, S),
               time.monotonic() - t0))

    # === Phase 2: Pure FISTA with high beta (25% of time) ===
    remaining = deadline - time.monotonic()
    phase2_time = 0.25 * remaining
    if phase2_time > 15:
        print("\nPhase 2: Pure FISTA high-beta (%.0f sec)" % phase2_time)
        t0 = time.monotonic()
        a_p2, score_p2 = run_fista_pure(global_a, n, S, phase2_time, rng, fsize,
                                         beta_init=100.0, beta_max=5000.0)
        a_p2 = safe_normalize(a_p2, n, S)
        c_p2 = fft_convolve_self(a_p2)
        if c_p2.max() < global_best:
            global_best = c_p2.max()
            global_a = a_p2.copy()
        print("  Phase 2 done: %.6f (obj %.6f) [%.0fs]" %
              (global_best, objective_from_conv(fft_convolve_self(global_a), n, S),
               time.monotonic() - t0))

    # === Phase 3: Top-k fine polish (15% of time) ===
    remaining = deadline - time.monotonic()
    phase3_time = 0.15 * remaining
    if phase3_time > 10:
        print("\nPhase 3: Top-k polish (%.0f sec)" % phase3_time)
        t0 = time.monotonic()
        a_p3, score_p3 = run_toppk_polish(global_a, n, S, phase3_time, rng, fsize,
                                           top_k_start=1, top_k_max=8)
        a_p3 = safe_normalize(a_p3, n, S)
        c_p3 = fft_convolve_self(a_p3)
        if c_p3.max() < global_best:
            global_best = c_p3.max()
            global_a = a_p3.copy()
        print("  Phase 3 done: %.6f (obj %.6f) [%.0fs]" %
              (global_best, objective_from_conv(fft_convolve_self(global_a), n, S),
               time.monotonic() - t0))

    # === Phase 4: Diversification (10% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\nPhase 4: Diversification (%.0f sec)" % remaining)
        n_variants = 4
        per_variant = remaining * 0.8 / n_variants

        for v in range(n_variants):
            if time.monotonic() > deadline - 30:
                break

            if v == 0:
                # Gaussian noise perturbation
                perturbed = global_a + rng.normal(0, 0.004 * S / np.sqrt(n), n)
            elif v == 1:
                # Smooth low-frequency perturbation
                freq = np.arange(fsize)
                low_freq = freq < fsize // 32
                noise_freq = rng.normal(0, 1, fsize)
                noise_freq[~low_freq] = 0
                smooth_noise = np.real(ifft(noise_freq))[:n]
                perturbed = global_a + 0.003 * S / np.sqrt(n) * smooth_noise / max(smooth_noise.std(), 1e-10)
            elif v == 2:
                # Multiplicative perturbation
                perturbed = global_a * (1.0 + rng.normal(0, 0.015, n))
            else:
                # Spectral perturbation: modify some frequency bins
                A = fft(global_a, fsize)
                P = np.abs(A) ** 2
                phase = np.angle(A)
                # Perturb mid-frequency bins
                mid_start = fsize // 8
                mid_end = fsize // 2
                P[mid_start:mid_end] *= (1.0 + rng.normal(0, 0.1, mid_end - mid_start))
                P = np.maximum(P, 0)
                A_new = np.sqrt(P) * np.exp(1j * phase)
                perturbed = np.real(ifft(A_new))[:n]

            perturbed = safe_normalize(perturbed, n, S)

            # Short combined optimization
            a_v, _ = run_combined_opt(perturbed, n, S, per_variant, rng, fsize,
                                       beta_init=80.0, beta_max=3000.0,
                                       spectral_every=60, spectral_eta=0.1)
            a_v = safe_normalize(a_v, n, S)
            c_v = fft_convolve_self(a_v)
            if c_v.max() < global_best:
                global_best = c_v.max()
                global_a = a_v.copy()
                print("  Variant %d: %.6f (obj %.6f)" %
                      (v, global_best, objective_from_conv(c_v, n, S)))

    # === Phase 5: Final ultra-fine polish (5% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 5:
        print("\nPhase 5: Final ultra-fine polish (%.0f sec)" % remaining)
        t0 = time.monotonic()
        a_final, _ = run_toppk_polish(global_a, n, S, remaining * 0.6, rng, fsize,
                                       top_k_start=1, top_k_max=4)
        a_final = safe_normalize(a_final, n, S)
        c_final = fft_convolve_self(a_final)
        if c_final.max() < global_best:
            global_best = c_final.max()
            global_a = a_final.copy()

        remaining2 = deadline - time.monotonic()
        if remaining2 > 3:
            a_f2, _ = run_fista_pure(global_a, n, S, remaining2, rng, fsize,
                                      beta_init=500.0, beta_max=10000.0)
            a_f2 = safe_normalize(a_f2, n, S)
            c_f2 = fft_convolve_self(a_f2)
            if c_f2.max() < global_best:
                global_best = c_f2.max()
                global_a = a_f2.copy()

        print("  Final: %.6f (obj %.6f) [%.0fs]" %
              (global_best, objective_from_conv(fft_convolve_self(global_a), n, S),
               time.monotonic() - t0))

    # === Final output ===
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    final_obj = objective_from_conv(fft_convolve_self(global_a), n, S)
    print("\nFINAL: length=%d, max_conv=%.6f, objective=%.6f" % (len(final_list), global_best, final_obj))

    # Verify with the actual evaluator
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: %.6f" % ver)
    else:
        print("WARNING: Verification failed! Using fallback.")
        # Fallback: use a simple uniform sequence
        final_list = [S / n] * n
        ver = evaluate_sequence(final_list)
        print("Fallback verified: %.6f" % ver)

    return final_list
