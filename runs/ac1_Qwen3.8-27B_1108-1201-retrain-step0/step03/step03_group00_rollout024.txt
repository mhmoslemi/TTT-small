```python
"""
Minimize 2n*max_conv/sum^2 over non-negative sequences using multi-scale
FISTA with adaptive LSE smoothing, top-k subgradient hybridization,
Nesterov restarts, and coordinate-descent local refinement.
Uses progressively larger lengths (4096 -> 8192 -> 16384) for better
attack surface, with warm-starting between scales.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_gradient(a, w_full):
    """
    Compute gradient of sum_k w_full[k] * c[k] where c = a*a (self-conv).
    grad[j] = 2 * sum_k w_full[k] * a[k-j] = 2 * (w_full cross-corr a)[j]
    """
    n = len(a)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def lse_weights(c, beta):
    """Compute LSE soft-max weights on convolution values."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
        return w
    return e / tot


def topk_weights(c, k):
    """One-hot weights on top-k lags, normalized."""
    k = min(k, len(c))
    idx = np.argpartition(-c, k)[:k]
    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    return w


def blended_weights(c, beta, topk_frac, rng):
    """Blend LSE and top-k weights for robustness."""
    if rng.random() < topk_frac:
        k = int(4 + 16 * rng.random())
        return topk_weights(c, k)
    return lse_weights(c, beta)


def fista_optimize(a_init, n, S, max_time, rng, beta_init=10.0,
                   beta_rate=0.003, beta_max=2000.0, topk_frac=0.15,
                   verbose=False):
    """
    FISTA optimization with adaptive LSE/top-k hybrid gradients.
    Returns (best_a, best_conv_max).
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
            if verbose and it % 10000 == 0:
                print("    it=%d: conv_max=%.8f, beta=%.1f" % (it, cur, beta))

        # Adaptive beta increase
        if it - last_improve > 3000:
            beta = min(beta_max, beta * 1.5)
        else:
            beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        # Choose gradient type
        beta_eff = max(beta, 1.0)
        w_full = blended_weights(c, beta_eff, topk_frac, rng)
        grad = compute_gradient(y, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: for LSE, L = 2*beta*max|a| roughly; use conservative
        # For exact subgradient (top-1), use larger step
        if topk_frac > 0 and rng.random() < topk_frac:
            # Exact subgradient step
            step = S / (gnorm * (1.0 + 0.01 * np.log1p(it / 200.0)))
            step = min(step, 5.0 / gnorm * S / n)
        else:
            step = 1.0 / (2.0 * beta_eff)

        # Backtracking: ensure descent
        # (skip for speed, rely on projection)
        x_new = project_simplex(y - step * grad, S)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

        # Periodic restart with noise
        if it - last_improve > 5000 and it > 1000:
            decay = np.exp(-(it - last_improve) / 5000.0)
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            if noise_scale > 1e-7:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_k = 1.0
                last_improve = it
                # Also try a bigger kick
                if rng.random() < 0.3:
                    noise_scale = 0.02 * S / np.sqrt(n)
                    y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                    t_k = 1.0

        # Hard restart if completely stuck
        if it - last_improve > 20000:
            noise_scale = 0.05 * S / np.sqrt(n)
            y = project_simplex(best_a + rng.normal(0, noise_scale, n), S)
            t_k = 1.0
            last_improve = it
            beta = beta_init

    # Final evaluation
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def local_refine(a_init, n, S, max_time, rng, verbose=False):
    """
    Coordinate-descent style local refinement targeting the peak lag.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()

        # Find top lags
        k_star = int(np.argmax(c))
        # Gradient at peak lag
        w = np.zeros(len(c))
        w[k_star] = 1.0
        grad = compute_gradient(a, w)

        # Also compute gradient for top-3 lags
        top3_idx = np.argpartition(-c, 3)[:4]  # top 4 (argpartition is 0-indexed)
        w3 = np.zeros(len(c))
        w3[top3_idx] = 1.0 / 4.0
        grad3 = compute_gradient(a, w3)

        # Blend: 70% peak, 30% top-4
        grad_blend = 0.7 * grad + 0.3 * grad3
        gnorm = np.linalg.norm(grad_blend)
        if gnorm < 1e-14:
            break

        # Small step
        step = 2.0 * S / (n * (1.0 + 0.005 * it))
        a_new = project_simplex(a - step * grad_blend / max(gnorm, 1e-15) * S / np.sqrt(n), S)

        # Accept if better
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur:
            a = a_new
        else:
            # Try smaller step
            step *= 0.25
            a_new = project_simplex(a - step * grad_blend / max(gnorm, 1e-15) * S / np.sqrt(n), S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur:
                a = a_new
            else:
                # Inject small noise
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)

    return best_a, best


def upsample(a, n_new):
    """Upsample a to length n_new via linear interpolation."""
    n_old = len(a)
    if n_new == n_old:
        return a.copy()
    if n_new < n_old:
        # Downsample by averaging
        blocks = n_old // n_new
        a_small = a[:blocks * n_new].reshape(n_new, blocks).mean(axis=1)
        return a_small
    t_old = np.linspace(0, 1, n_old)
    t_new = np.linspace(0, 1, n_new)
    return np.interp(t_new, t_old, a)


def make_initializations(n, rng, given_seq=None):
    """Generate diverse initializations of length n."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-3.0 * (tt - 0.5) ** 2 / 0.1))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)

    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(4 * np.pi * tt) + 0.2 * np.cos(10 * np.pi * tt))
    inits.append(1.0 + 0.25 * np.sin(6 * np.pi * tt + 0.7) + 0.15 * np.cos(14 * np.pi * tt))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.4, n)))

    # Block structure
    for nb in [4, 8, 16]:
        blen = max(1, n // nb)
        heights = rng.uniform(0.3, 1.5, nb)
        block = np.repeat(heights, blen)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n])

    # Given sequence (upsampled/downsampled)
    if given_seq is not None and len(given_seq) > 10:
        inits.append(np.clip(given_seq, 0.0, 1000.0))

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max_conv/sum^2.
    Multi-scale approach: 4096 -> 8192 -> 16384 with FISTA + local refinement.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)
    total_start = time.monotonic()

    # Try to load previous solution
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
        else:
            given = None
    except Exception:
        given = None

    print("Starting multi-scale optimization, budget=%.0fs" % budget_s)
    if given is not None:
        print("  Using warm start from given sequence, length=%d" % len(given))

    global_best = np.inf
    global_a = None

    # Multi-scale lengths to try
    scales = [4096, 8192, 16384]

    # Filter scales based on available time
    # Each scale needs roughly: exploration + polishing
    # Give more time to larger scales
    time_alloc = {}
    remaining_total = deadline - time.monotonic()
    # Allocate: 25% to smallest, 30% to medium, 40% to largest, 5% buffer
    for i, n in enumerate(scales):
        time_alloc[n] = remaining_total * [0.20, 0.30, 0.45][i]

    # If we have less total time, skip smaller scales
    if remaining_total < 600:
        # Only do the largest scale
        scales = [16384]
        time_alloc = {16384: remaining_total * 0.95}
    elif remaining_total < 1200:
        scales = [8192, 16384]
        time_alloc = {8192: remaining_total * 0.35, 16384: remaining_total * 0.60}

    for scale_idx, n in enumerate(scales):
        if time.monotonic() > deadline - 30:
            break

        S = np.sqrt(2.0 * n)
        t_scale = min(time_alloc.get(n, 300.0), deadline - time.monotonic() - 30)
        if t_scale < 30:
            break

        print("\n=== Scale %d: n=%d, time=%.0fs ===" % (scale_idx, n, t_scale))

        # Generate initializations
        # Use previous scale's best as warm start
        warm_start = None
        if global_a is not None:
            warm_start = upsample(global_a, n)
            warm_start = normalize_sum(warm_start, n)
        if given is not None and warm_start is None:
            warm_start = normalize_sum(upsample(given, n), n)

        inits = make_initializations(n, rng, warm_start)
        if warm_start is not None:
            inits.insert(0, warm_start)

        # Phase A: Parallel exploration (try multiple inits)
        n_inits = min(len(inits), 6)  # Limit to 6 for time management
        explore_time = t_scale * 0.65
        per_init_time = explore_time / n_inits
        # Give warm start more time
        if warm_start is not None:
            per_init_time = explore_time * 0.45  # 45% to warm start
            other_time = explore_time * 0.55 / max(1, n_inits - 1)
        else:
            per_init_time = explore_time / n_inits
            other_time = per_init_time

        best_at_scale = np.inf
        best_a_scale = None
        p_start = time.monotonic()

        for i in range(n_inits):
            if time.monotonic() > deadline - 30:
                break
            t_init = min(per_init_time, deadline - time.monotonic() - 30)
            if t_init < 5:
                break

            # Vary parameters for diversity
            beta_init = 5.0 + 5.0 * (i % 3)
            beta_rate = 0.002 + 0.001 * (i % 2)
            topk_frac = 0.1 + 0.05 * (i % 3)

            a_opt, score = fista_optimize(
                inits[i], n, S, t_init, rng,
                beta_init=beta_init, beta_rate=beta_rate,
                beta_max=1500.0, topk_frac=topk_frac,
                verbose=(i == 0 and scale_idx > 0)
            )
            a_opt = normalize_sum(a_opt, n)
            c_chk = fft_convolve_self(a_opt)
            real_score = c_chk.max()

            if real_score < best_at_scale:
                best_at_scale = real_score
                best_a_scale = a_opt.copy()
                if real_score < global_best:
                    global_best = real_score
                    global_a = a_opt.copy()
                    obj = 2.0 * n * global_best / (S ** 2)
                    print("  [init %d] n=%d: obj=%.6f (elapsed %.0fs)" % (i, n, obj, time.monotonic() - p_start))

        # Phase B: Polish the best at this scale
        if best_a_scale is not None:
            remaining = min(t_scale * 0.35, deadline - time.monotonic() - 15)
            if remaining > 10:
                print("  Polishing (time=%.0fs)..." % remaining)
                # FISTA polish with high beta
                a_pol, s_pol = fista_optimize(
                    best_a_scale, n, S, remaining * 0.6, rng,
                    beta_init=100.0, beta_rate=0.002,
                    beta_max=5000.0, topk_frac=0.3,
                    verbose=False
                )
                a_pol = normalize_sum(a_pol, n)
                c_pol = fft_convolve_self(a_pol)
                if c_pol.max() < best_at_scale:
                    best_at_scale = c_pol.max()
                    best_a_scale = a_pol.copy()

                # Local refinement
                remaining2 = deadline - time.monotonic() - 15
                if remaining2 > 10:
                    a_loc, s_loc = local_refine(
                        best_a_scale, n, S, remaining2 * 0.5, rng
                    )
                    a_loc = normalize_sum(a_loc, n)
                    c_loc = fft_convolve_self(a_loc)
                    if c_loc.max() < best_at_scale:
                        best_at_scale = c_loc.max()
                        best_a_scale = a_loc.copy()

                # Final FISTA micro-polish
                remaining3 = deadline - time.monotonic() - 15
                if remaining3 > 10 and best_a_scale is not None:
                    a_fin, s_fin = fista_optimize(
                        best_a_scale, n, S, remaining3 * 0.8, rng,
                        beta_init=500.0, beta_rate=0.001,
                        beta_max=10000.0, topk_frac=0.5,
                        verbose=False
                    )
                    a_fin = normalize_sum(a_fin, n)
                    c_fin = fft_convolve_self(a_fin)
                    if c_fin.max() < best_at_scale:
                        best_at_scale = c_fin.max()
                        best_a_scale = a_fin.copy()

                if best_a_scale is not None:
                    if best_at_scale < global_best:
                        global_best = best_at_scale
                        global_a = best_a_scale.copy()
                    obj = 2.0 * n * best_at_scale / (S ** 2)
                    print("  Scale n=%d final: obj=%.6f" % (n, obj))

    # If we still have time, try one more polish at the best length
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("\n=== Final polish (remaining=%.0fs, n=%d) ===" % (remaining, n_best))

        # Multiple short FISTA runs with different parameters
        for trial in range(min(3, int(remaining // 30))):
            t_run = min(remaining * 0.5 / (min(3, int(remaining // 30)) - trial),
                        deadline - time.monotonic() - 10)
            if t_run < 5:
                break
            beta_i = 200.0 * (trial + 1)
            tk = 0.2 + 0.1 * trial
            a_t, s_t = fista_optimize(
                global_a, n_best, S_best, t_run, rng,
                beta_init=beta_i, beta_rate=0.001,
                beta_max=10000.0, topk_frac=tk
            )
            a_t = normalize_sum(a_t, n_best)
            c_t = fft_convolve_self(a_t)
            if c_t.max() < global_best:
                global_best = c_t.max()
                global_a = a_t.copy()
                obj = 2.0 * n_best * global_best / (S_best ** 2)
                print("  Trial %d: obj=%.6f" % (trial, obj))

            remaining2 = deadline - time.monotonic()
            if remaining2 < 10:
                break

            # Local refinement
            if remaining2 > 10 and global_a is not None:
                a_l, s_l = local_refine(global_a, n_best, S_best, min(remaining2 * 0.4, 60), rng)
                a_l = normalize_sum(a_l, n_best)
                c_l = fft_convolve_self(a_l)
                if c_l.max() < global_best:
                    global_best = c_l.max()
                    global_a = a_l.copy()
                    obj = 2.0 * n_best * global_best / (S_best ** 2)
                    print("  After local: obj=%.6f" % obj)

    # Final verification
    if global_a is None:
        # Fallback: uniform sequence
        n_fb = 256
        global_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)

    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("\nFinal verification: obj=%.6f, length=%d" % (ver, len(final_list)))
    else:
        print("\nVerification FAILED, using fallback")
        n_fb = 256
        final_list = [float(np.sqrt(2 * n_fb) / n_fb)] * n_fb
        ver = evaluate_sequence(final_list)
        print("Fallback verification: obj=%.6f" % ver)

    total_time = time.monotonic() - total_start
    print("Total time: %.1fs" % total_time)
    return final_list
```
