    import numpy as np
    import time
    import cvxpy as cp
    from scipy.fft import fft, ifft

    def fft_convolve_self(a):
        n = len(a)
        size = 2 * n - 1
        fsize = 1 << (size - 1).bit_length()
        A = fft(a, fsize)
        c = np.real(ifft(A * A))
        return c[:size]

    def project_simplex(v, S):
        n = len(v)
        v = np.asarray(v, dtype=np.float64)
        w = np.sort(v)[::-1]
        cumsum = np.cumsum(w)
        idx = np.arange(1, n + 1, dtype=np.float64)
        theta_candidates = (cumsum - S) / idx
        valid = theta_candidates <= w
        if not np.any(valid):
            rho = n
        else:
            rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
        return np.maximum(v - theta, 0.0)

    def bundle_method(a_init, n, S, max_time, K=30, mu=0.5):
        a = a_init.copy()
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s)
        
        start = time.monotonic()
        best_score = np.inf
        best_a = a.copy()
        
        # Bundle: list of (k, c_k, g) where g = grad at the point where we added this cut
        bundle = []
        
        while time.monotonic() - start < max_time:
            c = fft_convolve_self(a)
            cur_max = c.max()
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()
            
            # Find top K lags
            top_k = np.argsort(c)[::-1][:K]
            
            # Build bundle: for each top lag, compute gradient at current point
            # g_k[i] = a[i+k] + a[i-k] (with zero-padding)
            G = np.zeros((K, n))
            for j, k in enumerate(top_k):
                k = int(k)
                # g[i] = a[i+k] + a[i-k] with zero-padding
                # a[i+k]: shift a left by k (valid for i = 0, ..., n-1-k)
                # a[i-k]: shift a right by k (valid for i = k, ..., n-1)
                if k == 0:
                    G[j] = 2.0 * a
                else:
                    G[j, :n-k] += a[k:]  # a[i+k] for i = 0,...,n-1-k
                    G[j, k:] += a[:n-k]  # a[i-k] for i = k,...,n-1
            
            # Build the affine model: b_j = c[k_j] - G_j @ a
            b = c[top_k] - G @ a
            
            # Solve QP: min 0.5*mu*||z-a||^2 + max_j(G_j z + b_j)
            #   s.t. z >= 0, sum(z) = S
            z_var = cp.Variable(n)
            obj = cp.Minimize(mu * cp.sum_squares(z_var - a) + cp.max(G @ z_var + b))
            constraints = [z_var >= 0, cp.sum(z_var) == S]
            prob = cp.Problem(obj, constraints)
            
            # Solve with appropriate solver
            try:
                prob.solve(solver=cp.ECOS, max_iters=20000, rel_tol=1e-8, abs_tol=1e-8)
            except:
                try:
                    prob.solve(solver=cp.SCS, max_iters=10000)
                except:
                    break
            
            if z_var.value is None:
                break
            
            z_new = np.clip(z_var.value, 0, 1000)
            s_new = z_new.sum()

import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_gradient_for_lag(a, k):
    """Compute gradient of c[k] = sum_i a[i]*a[k-i] w.r.t. a."""
    n = len(a)
    k = int(k)
    g = np.zeros(n)
    if k == 0:
        g = 2.0 * a
    else:
        if k <= n - 1:
            g[:n - k] += a[k:]
            g[k:] += a[:n - k]
        else:
            j = 2 * (n - 1) - k
            if j >= 0:
                g[:n - j] += a[j:]
                g[j:] += a[:n - j]
    return g


def quadratic_subgradient_step(a, S, n, top_lags, mu=1.0):
    """
    K=1 quadratic subgradient: solve min mu||z-a||^2 + g_top^T z s.t. simplex.
    This is equivalent to projecting (a - g_top/(2*mu)) onto the simplex.
    """
    k_star = top_lags[0]
    g = compute_gradient_for_lag(a, k_star)
    step = g / (2.0 * mu)
    z = project_simplex(a - step, S)
    return z


def run_fista_curriculum(a_init, n, S, max_time, rng, beta_start=5.0, beta_end=300.0):
    """FISTA with curriculum: beta increases over time for sharper peak targeting."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = 2.0 * beta_start

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / 5000.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)

        # Occasional backtracking
        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() > cur * 1.02:
                L *= 1.5
                x_new = project_simplex(y - step * grad, S)
            elif it % 50 == 0:
                L = max(L * 0.97, 2.0 * beta)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)

        a = x_new
        t_momentum = t_new

        # Periodic perturbation

            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_momentum = 1.0

    return best_local, best_score

        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Periodic perturbation to escape local minima
        if stall_count > 50 and rng.random() < 0.3:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_momentum = 1.0
            stall_count = 0

    return best_local, best_score


def run_multi_lag_qsubgradient(a_init, n, S, max_time, rng, K=3):
    """
    Multi-lag quadratic subgradient: average gradients from top K lags.
    Better than K=1 as it accounts for multiple near-peak lags.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Find top K lags with softmax weighting
        K_eff = min(K, 2 * n - 1)
        top_lags = np.argsort(c)[::-1][:K_eff]
        top_vals = c[top_lags]

        # Softmax weights (sharper = more weight on the actual peak)
        tv_max = top_vals.max()
        weights = np.exp(5.0 * (top_vals - tv_max))
        weights = weights / weights.sum()

        # Weighted gradient
        g = np.zeros(n)
        for j in range(K_eff):
            g += weights[j] * compute_gradient_for_lag(y, top_lags[j])

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        mu_eff = max(0.5, gnorm / (S / n))
        x_new = project_simplex(y - g / (2.0 * mu_eff), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        if it % 5000 == 0 and rng.random() < 0.1:
            noise = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_momentum = 1.0

    return best_local, best_score


def run_pair_targeting(a_init, n, S, max_time, rng):
    """
    Directly target high-contributing pairs at the peak lag.
    For the peak lag k*, find pairs (i, k*-i) with large product and
    redistribute mass to reduce the peak.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Find top 2 lags
        top_lags = np.argsort(c)[::-1][:2]

        improved = False
        for lag in top_lags:
            lag = int(lag)
            if lag < 0 or lag >= 2 * n - 1:
                continue

            # Find contributing indices
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue

            indices = np.arange(i_min, i_max + 1)
            partners = lag - indices
            products = a[indices] * a[partners]

            # Target the top contributing pairs
            n_pairs = len(indices)
            n_target = min(20, n_pairs)
            top_pair_idx = np.argsort(products)[::-1][:n_target]

            # Redistribute: reduce large elements, add to small ones
            scale = 0.01 * S / n
            perturbation = np.zeros(n)

            for ii in top_pair_idx:
                idx = indices[ii]
                pidx = partners[ii]
                # Reduce the larger element
                if a[idx] >= a[pidx]:
                    perturbation[idx] -= 1.0
                    # Add to a small element far from both
                    candidates = np.where((a < np.percentile(a, 25)) &
                                          (np.abs(np.arange(n) - idx) > n // 4) &
                                          (np.abs(np.arange(n) - pidx) > n // 4))[0]
                else:
                    perturbation[pidx] -= 1.0
                    candidates = np.where((a < np.percentile(a, 25)) &
                                          (np.abs(np.arange(n) - pidx) > n // 4) &
                                          (np.abs(np.arange(n) - idx) > n // 4))[0]

                if len(candidates) > 0:
                    add_idx = candidates[rng.integers(len(candidates))]
                    perturbation[add_idx] += 1.0

            # Try the perturbation
            a_new = project_simplex(a + scale * perturbation, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur:
                a = a_new
                improved = True
                break

        if not improved:
            # Try small random perturbation
            if it % 50 == 0 and rng.random() < 0.3:
                noise = 0.002 * S / np.sqrt(n)
                a_new = project_simplex(a + rng.normal(0, noise, n), S)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < cur:
                    a = a_new

    return best_local, best_score


def make_init_from_parent(parent, n, S, rng):
    """Create initialization from parent (possibly different length)."""
    p = np.clip(parent, 0, 1000).astype(np.float64)
    if p.sum() < 1e-15:
        return np.ones(n) * S / n

    if len(p) == n:
        a = p * (S / p.sum())
    elif len(p) > n:
        # Downsample
        step = len(p) / n
        indices = (np.arange(n) * step + step / 2).astype(int)
        indices = np.clip(indices, 0, len(p) - 1)
        a = p[indices]
        a = a * (S / a.sum())
    else:
        # Upsample using interpolation
        tt_old = np.linspace(0, 1, len(p))
        tt_new = np.linspace(0, 1, n)
        a = np.interp(tt_new, tt_old, p)
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / a.sum())
    return a


def safe_normalize(a, n):
    """Clamp and normalize."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses bundle method (CVXPY QP) + fast quadratic subgradient + FISTA polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    # Get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Determine working length
    if parent is not None and len(parent) >= 64:
        primary_n = len(parent)
    else:
        primary_n = 4096

    print("Primary length: %d, parent available: %s" % (primary_n, parent is not None))

    # === Phase 0: Quick exploration at reduced scale ===
    # Use a smaller n to quickly find a good structure
    if parent is not None and len(parent) > 2048:
        n_small = min(2048, len(parent) // 2)
        if n_small >= 512:
            S_small = np.sqrt(2.0 * n_small)
            print("=== Phase 0: Quick exploration at n=%d ===" % n_small)
            t0 = time.monotonic()
            a_small = make_init_from_parent(parent, n_small, S_small, rng)

            # Quick FISTA at small scale
            seg0 = 60.0
            a_s, s_s = run_fista_curriculum(a_small, n_small, S_small, seg0, rng,
                                             beta_start=5.0, beta_end=100.0)
            a_s = safe_normalize(a_s, n_small)
            obj_s = 2.0 * n_small * a_s.max() / a_s.sum() ** 2  # rough
            c_s = fft_convolve_self(a_s)
            obj_s = 2.0 * n_small * c_s.max() / (a_s.sum() ** 2)
            print("  Small n FISTA: obj=%.6f" % obj_s)

            # Use as initial guess for full scale
            a_full_init = make_init_from_parent(a_s, primary_n, np.sqrt(2.0 * primary_n), rng)
            elapsed0 = time.monotonic() - t0
            print("  Phase 0 done in %.1fs" % elapsed0)
        else:
            a_full_init = make_init_from_parent(parent, primary_n, np.sqrt(2.0 * primary_n), rng)
    else:
        a_full_init = make_init_from_parent(parent, primary_n, np.sqrt(2.0 * primary_n), rng)

    n = primary_n
    S = np.sqrt(2.0 * n)
    a_full_init = safe_normalize(a_full_init, n)

    # === Phase 1: Fast quadratic subgradient (K=1 with momentum) ===
    remaining = deadline - time.monotonic()
    seg1 = remaining * 0.25
    print("=== Phase 1: Fast Q-Subgradient (%.0f sec, n=%d) ===" % (seg1, n))
    t0 = time.monotonic()

    a1, s1 = run_fast_qsubgradient(a_full_init, n, S, seg1, rng)
    a1 = safe_normalize(a1, n)
    c1 = fft_convolve_self(a1)
    obj1 = 2.0 * n * c1.max() / (a1.sum() ** 2)
    print("  Phase 1: obj=%.6f (elapsed %.0fs)" % (obj1, time.monotonic() - t0))

    if c1.max() / (a1.sum() ** 2 / (2.0 * n)) < global_best / (2.0 * n):
        global_best = c1.max()
        global_a = a1.copy()

    # === Phase 2: Multi-lag quadratic subgradient ===
    remaining = deadline - time.monotonic()
    seg2 = remaining * 0.20
    print("=== Phase 2: Multi-lag Q-Subgradient K=3 (%.0f sec) ===" % seg2)
    t0 = time.monotonic()

    a2, s2 = run_multi_lag_qsubgradient(a1, n, S, seg2, rng, K=3)
    a2 = safe_normalize(a2, n)
    c2 = fft_convolve_self(a2)
    obj2 = 2.0 * n * c2.max() / (a2.sum() ** 2)
    print("  Phase 2: obj=%.6f (elapsed %.0fs)" % (obj2, time.monotonic() - t0))

    if c2.max() < global_best or global_a is None:
        global_best = c2.max()
        global_a = a2.copy()

    # === Phase 3: Bundle method with CVXPY ===
    remaining = deadline - time.monotonic()
    seg3 = remaining * 0.30
    print("=== Phase 3: Bundle CVXPY (%.0f sec, n=%d) ===" % (seg3, n))
    t0 = time.monotonic()

    # For CVXPY, use a potentially reduced dimension for speed
    n_cvx = min(n, 4096)
    if n_cvx < n:
        a_cvx_init = make_init_from_parent(global_a if global_a is not None else a2,
                                           n_cvx, np.sqrt(2.0 * n_cvx), rng)
    else:
        a_cvx_init = safe_normalize(global_a if global_a is not None else a2, n)

    a3, s3 = run_bundle_cvxpy(a_cvx_init, n_cvx, np.sqrt(2.0 * n_cvx), seg3, K=15, mu=0.5)
    a3 = safe_normalize(a3, n_cvx)
    c3 = fft_convolve_self(a3)
    obj3 = 2.0 * n_cvx * c3.max() / (a3.sum() ** 2)
    print("  Phase 3: obj=%.6f, n=%d (elapsed %.0fs)" % (obj3, n_cvx, time.monotonic() - t0))

    # Compare with global best (accounting for different n)
    if obj3 < (2.0 * len(global_a) * global_best / (np.sqrt(2.0 * len(global_a)) ** 2)
               if global_a is not None else np.inf):
        # Upsample to full n
        a3_full = make_init_from_parent(a3, n, S, rng) if n_cvx < n else a3
        global_a = safe_normalize(a3_full, n)
        c_full = fft_convolve_self(global_a)
        global_best = c_full.max()
        print("  Updated global best with bundle result: %.6f" % global_best)

    # === Phase 4: FISTA polish from best ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        seg4 = remaining * 0.15
        print("=== Phase 4: FISTA polish (%.0f sec) ===" % seg4)
        t0 = time.monotonic()

        a4, s4 = run_fista_curriculum(global_a, n, S, seg4, rng,
                                       beta_start=20.0, beta_end=200.0)
        a4 = safe_normalize(a4, n)
        c4 = fft_convolve_self(a4)
        if c4.max() < global_best:
            global_best = c4.max()
            global_a = a4.copy()
            print("  FISTA polish: obj=%.6f" % global_best)
        else:
            print("  FISTA polish: no improvement (best=%.6f)" % global_best)

    # === Phase 5: Pair targeting ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        seg5 = remaining * 0.10
        print("=== Phase 5: Pair targeting (%.0f sec) ===" % seg5)
        t0 = time.monotonic()

        a5, s5 = run_pair_targeting(global_a, n, S, seg5, rng)
        a5 = safe_normalize(a5, n)
        c5 = fft_convolve_self(a5)
        if c5.max() < global_best:
            global_best = c5.max()
            global_a = a5.copy()
            print("  Pair targeting: obj=%.6f" % global_best)
        else:
            print("  Pair targeting: no improvement (best=%.6f)" % global_best)

    # === Phase 6: Final bundle + multi-lag on best ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            seg6 = remaining * 0.7
            print("=== Phase 6: Final optimization (%.0f sec) ===" % seg6)
            t0 = time.monotonic()

            # Try multi-lag Q-Subgradient with K=5
            a6, s6 = run_multi_lag_qsubgradient(global_a, n, S, seg6 * 0.5, rng, K=5)
            a6 = safe_normalize(a6, n)
            c6 = fft_convolve_self(a6)
            if c6.max() < global_best:
                global_best = c6.max()
                global_a = a6.copy()
                print("  Final multi-lag: obj=%.6f" % global_best)

            # Try another FISTA pass with different params
            rem6 = deadline - time.monotonic()
            if rem6 > 10:
                a7, s7 = run_fista_curriculum(global_a, n, S, rem6 * 0.7, rng,
                                               beta_start=50.0, beta_end=500.0)
                a7 = safe_normalize(a7, n)
                c7 = fft_convolve_self(a7)
                if c7.max() < global_best:
                    global_best = c7.max()
                    global_a = a7.copy()
                    print("  Final FISTA: obj=%.6f" % global_best)

    # === Phase 7: Try different length ===
    if global_a is not None and time.monotonic() < deadline - 60:
        # Try a slightly different length for potential improvement
        n_alt = n + 1024 if n <= 8192 else n - 1024
        n_alt = max(1024, min(n_alt, 16384))
        if n_alt != n:
            remaining = deadline - time.monotonic()
            if remaining > 60:
                S_alt = np.sqrt(2.0 * n_alt)
                a_alt_init = make_init_from_parent(global_a, n_alt, S_alt, rng)
                seg_alt = min(60.0, remaining * 0.5)
                print("=== Phase 7: Try n=%d (%.0f sec) ===" % (n_alt, seg_alt))
                a_alt, _ = run_fast_qsubgradient(a_alt_init, n_alt, S_alt, seg_alt, rng)
                a_alt = safe_normalize(a_alt, n_alt)
                c_alt = fft_convolve_self(a_alt)
                obj_alt = 2.0 * n_alt * c_alt.max() / (a_alt.sum() ** 2)
                obj_cur = 2.0 * len(global_a) * global_best / (np.sqrt(2.0 * len(global_a)) ** 2)
                if obj_alt < obj_cur:
                    global_a = a_alt
                    global_best = c_alt.max()
                    print("  n=%d improved! obj=%.6f" % (n_alt, obj_alt))
                else:
                    print("  n=%d no improvement (obj=%.6f vs %.6f)" % (n_alt, obj_alt, obj_cur))

    # === Final verification ===
    if global_a is None:
        global_a = np.ones(4096, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(2048, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Verification: %.6f (n=%d) ===" % (ver if np.isfinite(ver) else float('inf'), len(final_list)))

    if not np.isfinite(ver):
        print("  WARNING: rejected, using fallback")
        global_a = np.ones(4096, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback: %.6f" % ver)

    return final_list
