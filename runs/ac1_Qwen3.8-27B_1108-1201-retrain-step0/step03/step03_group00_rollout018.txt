"""

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def run_fista(a_init, n, S, max_time, rng, mode="lse",

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        if mode == "hybrid" and it > 500 and rng.random() < min(0.3, 0.0005 * it):
            grad = exact_subgradient(y, c)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)

        a = x_new
        t = t_new

def build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix C where C[k,j] = f[k-j] (0 if out of range)."""
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            C[k, lo:hi+1] = f[k - hi:k - lo + 1][::-1]
    return C


def solve_lp_for_direction(f, M, n):
    """
    Solve: max sum(b) s.t. conv(f,b) <= M, b >= 0.
    Returns optimal b and its sum.
    """
    # Build constraint matrix: A_ub @ b <= M * ones
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n
    # Vectorized construction
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # For fixed j, A_ub[k, j] = f[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        lo_k = j
        hi_k = min(2 * n - 2, j + n - 1)
        if lo_k <= hi_k:
            A_ub[lo_k:hi_k+1, j] = f[0:hi_k-lo_k+1]
    
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    else:
        # Fallback: return f itself (which is feasible)
        return f.copy(), f.sum()

    best = np.inf
    best_a = a.copy()

        it += 1
        c = fft_convolve_self(a)
        M = c.max()
        
        if M < best:
            best = M
            best_a = a.copy()
            print("    LP iter %d: M = %.8f" % (it, M))
        
        # Solve LP
        b, b_sum = solve_lp_for_direction(a, M, n)
        
        if b_sum <= 0:
            break
        
        # Rescale b to have sum S
        g = b * (S / b_sum)
        
        # Adaptive step size
        # Start with t=0.3, reduce if oscillating
        t_step = 0.3
        
        # Update
        a_new = (1.0 - t_step) * a + t_step * g
        a_new = np.maximum(a_new, 0.0)
        # Renormalize

        if s_new > 1e-15:

            if s_new > 1e-15:

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    
    Strategy:
    Phase 1: FISTA exploration with multiple initializations (30% of time)
    Phase 2: LP-based refinement of the best candidate (50% of time)  
    Phase 3: Final FISTA polish (20% of time)
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)
    
    global_best = np.inf
    global_a = None
    
    # Try to get the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    # Choose main length
    if given is not None and 1024 <= len(given) <= 16384:
        main_len = len(given)
    else:
        main_len = 8192
    
    n = main_len
    S = np.sqrt(2.0 * n)
    
    print("Starting with n=%d, budget=%.0fs" % (n, budget_s))
    if given is not None:
        c_giv = fft_convolve_self(given)
        score_giv = 2 * n * c_giv.max() / (given.sum() ** 2)
        print("Given sequence score: %.6f" % score_giv)
    
    # === Phase 1: FISTA Exploration ===
    t1 = time.monotonic()
    phase1_time = (deadline - t1) * 0.30
    print("Phase 1: FISTA exploration (%.0fs)" % phase1_time)
    
    # Build list of initializations
    inits = []
    if given is not None:
        inits.append(("given", given.copy()))
    
    gen_inits = generate_initializations(n, rng)
    for i, g in enumerate(gen_inits):
        inits.append(("init_%d" % i, g))
    
    # Allocate time: give "given" more time
    n_inits = len(inits)
    weights = np.ones(n_inits)
    if given is not None:
        weights[0] = 3.0  # Give 3x time to the given sequence
    weights /= weights.sum()
    
    for i, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - (deadline - t1) * 0.70:
            break
        
        arr_n = len(arr)
        S_i = np.sqrt(2.0 * arr_n)
        t_alloc = phase1_time * weights[i]
        if t_alloc < 5:
            continue
        
        modes = ["lse", "hybrid", "lse"]
        mode = modes[i % len(modes)]
        
        a_opt, score = run_fista(arr, arr_n, S_i, t_alloc, rng, mode=mode,
                                   beta_init=5.0, beta_rate=0.003, beta_max=500.0)
        a_norm = normalize_sum(a_opt, arr_n)
        c_chk = fft_convolve_self(a_norm)
        real_score = 2 * arr_n * c_chk.max() / (a_norm.sum() ** 2)
        
        if real_score < global_best:
            global_best = real_score
            global_a = a_norm.copy()
            print("  [%s] n=%d: score=%.6f (elapsed %.0fs)" % (name, arr_n, global_best, time.monotonic() - t1))
    
    print("Phase 1 done. Best: %.6f (elapsed %.0fs)" % (global_best, time.monotonic() - t1))
    
    # === Phase 2: LP-based Refinement ===
    # Use a smaller n for LP (since LP is expensive)
    # We'll downsample to n_lp and refine there, then upsample back
    t2 = time.monotonic()
    remaining = deadline - t2
    phase2_time = remaining * 0.55
    
    if global_a is not None:
        # Choose LP length: use the actual length if small enough, else downsample
        n_lp = len(global_a)
        if n_lp > 4096:
            # Downsample
            n_lp = 4096
            n_orig = len(global_a)
            # Simple downsampling by averaging
            nblocks = n_orig // n_lp
            global_a_lp = global_a[:nblocks * n_lp].reshape(n_lp, nblocks).mean(axis=1)
            global_a_lp = normalize_sum(global_a_lp, n_lp)
        else:
            global_a_lp = global_a.copy()
        
        S_lp = np.sqrt(2.0 * n_lp)
        print("Phase 2: LP refinement (n=%d, %.0fs)" % (n_lp, phase2_time))
        
        a_lp, score_lp = run_lp_refinement_v2(global_a_lp, n_lp, S_lp, phase2_time, rng)
        score_lp_full = 2 * n_lp * fft_convolve_self(normalize_sum(a_lp, n_lp)).max() / (normalize_sum(a_lp, n_lp).sum() ** 2)
        
        print("  LP result: score=%.6f (n=%d)" % (score_lp_full, n_lp))
        
        if score_lp_full < global_best:
            global_best = score_lp_full
            # Upsample back to original length if we downsampled
            if len(a_lp) < len(global_a) if global_a is not None else False:
                pass  # Keep at LP resolution for now
            global_a = normalize_sum(a_lp, n_lp).copy()
    
    print("Phase 2 done. Best: %.6f (elapsed %.0fs)" % (global_best, time.monotonic() - t2))
    
    # === Phase 3: Final Polish ===
    t3 = time.monotonic()
    remaining = deadline - t3
    phase3_time = remaining * 0.85
    
    if global_a is not None:
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * n_final)
        print("Phase 3: Final polish (n=%d, %.0fs)" % (n_final, phase3_time))
        
        # Split into multiple FISTA runs with different hyperparameters
        sub_time = phase3_time / 3.0
        
        # 3a: High-beta FISTA
        a3a, s3a = run_fista(global_a, n_final, S_final, sub_time, rng,
                              mode="hybrid", beta_init=100.0, beta_rate=0.002, beta_max=3000.0)
        a3a = normalize_sum(a3a, n_final)
        score3a = 2 * n_final * fft_convolve_self(a3a).max() / (a3a.sum() ** 2)
        if score3a < global_best:
            global_best = score3a
            global_a = a3a.copy()
            print("  3a: %.6f" % global_best)
        
        # 3b: Continue LP if we have time and n is small enough
        remaining3 = phase3_time - (time.monotonic() - t3)
        if n_final <= 4096 and remaining3 > 30:
            a3b, s3b = run_lp_refinement_v2(global_a, n_final, S_final, remaining3 * 0.5, rng)
            a3b = normalize_sum(a3b, n_final)
            score3b = 2 * n_final * fft_convolve_self(a3b).max() / (a3b.sum() ** 2)
            if score3b < global_best:
                global_best = score3b
                global_a = a3b.copy()
                print("  3b (LP): %.6f" % global_best)
        
        # 3c: Final FISTA
        remaining3 = phase3_time - (time.monotonic() - t3)
        if remaining3 > 5:
            a3c, s3c = run_fista(global_a, n_final, S_final, remaining3, rng,
                                  mode="lse", beta_init=200.0, beta_rate=0.001, beta_max=5000.0)
            a3c = normalize_sum(a3c, n_final)
            score3c = 2 * n_final * fft_convolve_self(a3c).max() / (a3c.sum() ** 2)
            if score3c < global_best:
                global_best = score3c
                global_a = a3c.copy()
                print("  3c: %.6f" % global_best)
