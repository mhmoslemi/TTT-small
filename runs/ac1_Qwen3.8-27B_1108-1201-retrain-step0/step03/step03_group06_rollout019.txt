import time
import cvxpy as cp

from scipy.linalg import toeplitz

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def build_toeplitz_sparse(a, n):

    first_col = np.zeros(2 * n - 1)
    first_col[:n] = a
    first_row = np.zeros(n)
    first_row[0] = a[0]

    A = sptoeplitz(first_col, first_row)
    return A

    A = build_toeplitz_sparse(a, n)

    objective = cp.Maximize(cp.sum(b))

    try:

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based Frank-Wolfe (conditional gradient)
method with adaptive step size, combined with gradient polishing and coordinate descent.

Key innovation: Use a linear program at each Frank-Wolfe step to find the optimal
direction, which is more principled than subgradient methods. The LP maximizes sum(b)
subject to (a*b)[k] <= M for all k and b >= 0, finding the "best" non-negative
sequence that doesn't increase the peak when convolved with the current sequence.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csc_matrix
from scipy.linalg import toeplitz
from scipy.fft import fft, ifft


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def build_toeplitz_sparse(a, n):
    """Build sparse Toeplitz matrix T such that T @ b = a * b (full convolution)."""
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = a
    first_row = np.zeros(n)
    first_row[0] = a[0]
    from scipy.sparse import toeplitz as sptoeplitz
    T = sptoeplitz(first_col, first_row)
    return T.tocsc()


def solve_lp_direction(a, M, n, solver_list=None):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    if solver_list is None:
        solver_list = ['PDLP', 'ECOS', 'SCS']
    
    T = build_toeplitz_sparse(a, n)
    
    b = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b))
    constraints = [T @ b <= M]
    
    prob = cp.Problem(objective, constraints)
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if b.value is not None:
                return np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
        except Exception:
            continue
    
    return None


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def frank_wolfe_lp(a_init, n, S, time_budget, max_iters=300, verbose=True):
    """
    Frank-Wolfe with LP oracle.
    At each step: solve LP to get direction g, then line search for step size.
    """
    a = normalize_nonneg(a_init.copy(), S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    no_improve = 0
    max_no_improve = 30
    
    while time.monotonic() - start < time_budget and it < max_iters:
        if no_improve > max_no_improve:
            break
        
        t0 = time.monotonic()
        
        M = c.max()
        
        # Solve LP
        b = solve_lp_direction(a, M, n)
        if b is None:
            break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        # Rescale to match sum
        g = b * (S / b_sum)
        
        # Line search
        best_t = 0.0
        best_t_obj = best_obj
        
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial < best_t_obj - 1e-14:
                best_t_obj = obj_trial
                best_t = t
            if t == 1.0:
                break
        
        if best_t == 0.0:
            no_improve += 1
            it += 1
            continue
        
        a = (1.0 - best_t) * a + best_t * g
        a = normalize_nonneg(a, S)
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        it += 1
        elapsed = time.monotonic() - start
        if verbose and (it % 5 == 0 or it < 5):
            lp_time = time.monotonic() - t0
            print("  FW-LP iter %d: obj=%.7f, t=%.2f, LP_time=%.1fs, elapsed=%.1fs" % 
                  (it, best_obj, best_t, lp_time, elapsed))
    
    return best_a, best_obj


def weighted_subgrad_step(a, S, T, lr):
    """One weighted subgradient step (for polishing between LP iterations)."""
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    if c_max < 1e-30:
        return a.copy()
    
    scaled = T * c / c_max
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Gradient: g[m] = 2 * sum_k w[k] * a[k-m]
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    g = 2.0 * corr[:n]
    
    a_new = a - lr * g
    return normalize_nonneg(a_new, S)


def gradient_polish(a, n, S, time_budget, T_init=50, T_final=500, lr_init=0.01, verbose=False):
    """Polish with weighted subgradient descent."""
    a = normalize_nonneg(a.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    step = 0
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** progress
        lr = lr_init * (1.0 - 0.5 * progress)
        
        a_new = weighted_subgrad_step(a, S, T, lr)
        
        c = fft_convolve_self(a_new)
        obj = c.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        step += 1
    
    return best_a, best_obj


def coordinate_descent(a, n, S, time_budget, max_sweeps=20, verbose=False):
    """Coordinate descent polishing."""
    a = normalize_nonneg(a.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(999)
    start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            orig = a[i]
            if orig < 1e-12:
                continue
            
            best_val = orig
            best_obj_i = c.max()
            
            for factor in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]:
                a[i] = orig * factor
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_obj_i - 1e-12:
                    best_obj_i = obj_check
                    best_val = orig * factor
            
            a[i] = best_val
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            
            if c.max() < best_obj - 1e-12:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        if not improved and sweep >= 2:
            break
    
    return best_a, best_obj

    """Project v onto {x : x >= 0, sum(x) = S}."""
    n = len(v)

    cumsum = np.cumsum(v_sorted)

    theta = (cumsum[rho-1] - S) / rho

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based Frank-Wolfe (conditional gradient)
with line search, reduced-constraint LP acceleration, gradient polishing, and coordinate descent.

Core idea: At each Frank-Wolfe step, solve an LP to find the best non-negative direction
g that keeps the cross-convolution (a*g) bounded by the current peak M. This gives a
principled, globally-informed direction (vs. local subgradient), leading to faster convergence.
The LP is: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
We use reduced constraint sets (middle K + violated) for speed after warm-up iterations.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csc_matrix
from scipy.fft import fft, ifft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def build_toeplitz_sparse(a, n):
    """Build sparse Toeplitz matrix T (2n-1 x n) such that T @ b = a * b."""
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = a
    first_row = np.zeros(n)
    first_row[0] = a[0]
    from scipy.sparse import toeplitz as sptoeplitz
    T = sptoeplitz(first_col, first_row)
    return T.tocsc()


def normalize_nonneg(a, S):
    """Clip to non-negative and rescale to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    ar = np.arange(1, n + 1)
    mask = v_sorted * ar > cumsum - S
    if not mask.any():
        return np.full(n, S / n)
    rho = np.where(mask)[0][-1] + 1
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(a, M, n, constraint_idx=None, solvers=None):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in constraint_idx, b >= 0."""
    if solvers is None:
        solvers = ['PDLP', 'ECOS', 'SCS']

    T_full = build_toeplitz_sparse(a, n)

    if constraint_idx is None:
        T = T_full
    else:
        T = T_full[constraint_idx, :].tocsc()

    b_var = cp.Variable(n, nonneg=True)
    objective = cp.Maximize(cp.sum(b_var))
    constraints = [T @ b_var <= M]

    prob = cp.Problem(objective, constraints)

    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if b_var.value is not None:
                return np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
        except Exception:
            continue

    return None


def frank_wolfe_lp(a_init, n, S, time_budget, max_iters=300, verbose=True,
                   full_iters=3, reduced_K=400, alpha=1.0):
    """
    LP-based Frank-Wolfe with adaptive line search.
    alpha scales the constraint M (alpha > 1 makes LP looser, allowing more progress).
    """
    a = normalize_nonneg(a_init.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    no_improve = 0
    max_no_improve = 50

    while time.monotonic() - start < time_budget and it < max_iters:
        if no_improve > max_no_improve:
            break

        t0 = time.monotonic()
        M_cur = c.max() * alpha

        # Determine constraint set
        if it < full_iters:
            cidx = np.arange(2 * n - 1)
        else:
            center = n - 1
            half = reduced_K // 2
            lo = max(0, center - half)
            hi = min(2 * n - 2, center + half)
            cidx = np.arange(lo, hi + 1)

        # Solve LP
        b = solve_lp_direction(a, M_cur, n, constraint_idx=cidx)

        if b is None:
            if verbose:
                print("    LP failed at iter %d, stopping." % it)
            break

        # Check if solution violates excluded constraints; if so, re-solve with them
        if it >= full_iters:
            c_ab = fft_convolve(a, b)
            violated = np.where((c_ab > M_cur * 1.001) & (~np.isin(np.arange(2 * n - 1), cidx)))[0]
            if len(violated) > 0:
                cidx2 = np.unique(np.concatenate([cidx, violated]))
                b2 = solve_lp_direction(a, M_cur, n, constraint_idx=cidx2)
                if b2 is not None and b2.sum() >= b.sum() - 1e-10:
                    b = b2

        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        # Rescale to match sum
        g = b * (S / b_sum)

        # Line search
        best_t = 0.0
        best_t_obj = best_obj
        t_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]

        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial < best_t_obj - 1e-14:
                best_t_obj = obj_trial
                best_t = t

        if best_t == 0.0:
            # Try a few smaller steps
            for t in [0.01, 0.02, 0.03]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = normalize_nonneg(a_trial, S)
                c_trial = fft_convolve_self(a_trial)
                obj_trial = c_trial.max()
                if obj_trial < best_t_obj - 1e-14:
                    best_t_obj = obj_trial
                    best_t = t
                    break

        if best_t == 0.0:
            no_improve += 1
            it += 1
            continue

        # Update
        a = (1.0 - best_t) * a + best_t * g
        a = normalize_nonneg(a, S)
        c = fft_convolve_self(a)
        cur_obj = c.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        it += 1
        if verbose and (it % 5 == 0 or it <= 3):
            elapsed = time.monotonic() - start
            print("  FW-LP it %d: obj=%.8f, t=%.3f, LP+ls=%.1fs, elapsed=%.0fs/%.0fs" %
                  (it, best_obj, best_t, time.monotonic() - t0, elapsed, time_budget))

    return best_a, best_obj


def subgrad_polish(a, n, S, time_budget, T_init=100.0, T_final=3000.0,
                   lr_init=0.008, verbose=False):
    """Weighted subgradient descent for polishing."""
    a = normalize_nonneg(a.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    step = 0

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** progress
        lr = lr_init * (0.3 + 0.7 * (1.0 - progress))

        c_max = c.max()
        if c_max < 1e-30:
            break
        scaled = T * c / c_max
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum

        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g = 2.0 * corr[:n]

        # Backtracking
        a_new = a
        obj_new = best_obj
        lr_try = lr
        for _ in range(6):
            a_try = a - lr_try * g
            a_try = normalize_nonneg(a_try, S)
            c_try = fft_convolve_self(a_try)
            obj_try = c_try.max()
            if obj_try < c.max() - 1e-12:
                a_new = a_try
                obj_new = obj_try
                break
            lr_try *= 0.5

        a = a_new
        c = fft_convolve_self(a)
        if c.max() < best_obj - 1e-12:
            best_obj = c.max()
            best_a = a.copy()

        step += 1

    return best_a, best_obj


def coordinate_descent(a, n, S, time_budget, max_sweeps=25, verbose=False):
    """Coordinate descent with multiplicative updates."""
    a = normalize_nonneg(a.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(777)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            orig = a[i]
            if orig < 1e-14:
                continue

            best_val = orig
            best_obj_i = c.max()

            for factor in [0.75, 0.85, 1.15, 1.25, 0.6, 1.5, 0.9, 1.1]:
                a[i] = orig * factor
                a = normalize_nonneg(a, S)
                c_chk = fft_convolve_self(a)
                obj_chk = c_chk.max()
                if obj_chk < best_obj_i - 1e-13:
                    best_obj_i = obj_chk
                    best_val = orig * factor

            a[i] = best_val
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)

            if c.max() < best_obj - 1e-13:
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        if not improved and sweep >= 3:
            if verbose:
                print("    CD converged at sweep %d" % sweep)
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses LP-based Frank-Wolfe as primary optimizer,
    with gradient polishing, coordinate descent, and multi-phase strategy.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 25.0)
    deadline = time.monotonic() + total_budget

    global_best = np.inf
    global_a = None

    # Load given sequence
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            n_given = len(given_raw)
            S_given = np.sqrt(2.0 * n_given)
            given_raw = given_raw * (S_given / given_raw.sum())
            ver = evaluate_sequence(given_raw.tolist())
            print("Given: n=%d, obj=%.8f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_raw.copy()
    except Exception as e:
        print("Could not load given: " + str(e))

    if global_a is None:
        n = 4096
        S = np.sqrt(2.0 * n)
        global_a = np.full(n, S / n)
        global_best = evaluate_sequence(global_a.tolist())
        print("Uniform init: n=%d, obj=%.8f" % (n, global_best))

    n = len(global_a)
    S = np.sqrt(2.0 * n)
    a = global_a.copy()
    a = normalize_nonneg(a, S)

    print("\nStarting search: n=%d, S=%.2f, initial obj=%.8f, budget=%.0fs" %
          (n, S, global_best, total_budget))

    # ============ PHASE 1: LP-based Frank-Wolfe (main) ============
    p1_budget = 0.50 * total_budget
    print("\n=== Phase 1: FW-LP (budget=%.0fs) ===" % p1_budget)

    a_best, _ = frank_wolfe_lp(a, n, S, p1_budget, max_iters=500, verbose=True,
                                full_iters=3, reduced_K=400, alpha=1.0)
    a_best = normalize_nonneg(a_best, S)
    obj_chk = evaluate_sequence(a_best.tolist())
    if np.isfinite(obj_chk) and obj_chk < global_best:
        global_best = obj_chk
        global_a = a_best.copy()
        print("  Phase 1 done: obj=%.8f" % obj_chk)

    # ============ PHASE 2: FW-LP with looser constraint ============
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p2_budget = min(0.20 * total_budget, remaining * 0.5)
        print("\n=== Phase 2: FW-LP alpha=1.3 (budget=%.0fs) ===" % p2_budget)
        a = global_a.copy()
        a = normalize_nonneg(a, S)
        a_best, _ = frank_wolfe_lp(a, n, S, p2_budget, max_iters=200, verbose=True,
                                    full_iters=0, reduced_K=500, alpha=1.3)
        a_best = normalize_nonneg(a_best, S)
        obj_chk = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj_chk) and obj_chk < global_best:
            global_best = obj_chk
            global_a = a_best.copy()
            print("  Phase 2 done: obj=%.8f" % obj_chk)

    # ============ PHASE 3: Gradient polishing ============
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p3_budget = min(0.15 * total_budget, remaining * 0.5)
        print("\n=== Phase 3: Gradient polish (budget=%.0fs) ===" % p3_budget)
        a = global_a.copy()
        a = normalize_nonneg(a, S)
        a_best, _ = subgrad_polish(a, n, S, p3_budget, T_init=200, T_final=5000, lr_init=0.005)
        a_best = normalize_nonneg(a_best, S)
        obj_chk = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj_chk) and obj_chk < global_best:
            global_best = obj_chk
            global_a = a_best.copy()
            print("  Phase 3 done: obj=%.8f" % obj_chk)

    # ============ PHASE 4: Coordinate descent ============
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p4_budget = min(0.12 * total_budget, remaining * 0.6)
        print("\n=== Phase 4: Coordinate descent (budget=%.0fs) ===" % p4_budget)
        a = global_a.copy()
        a = normalize_nonneg(a, S)
        a_best, _ = coordinate_descent(a, n, S, p4_budget, max_sweeps=20, verbose=True)
        a_best = normalize_nonneg(a_best, S)
        obj_chk = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj_chk) and obj_chk < global_best:
            global_best = obj_chk
            global_a = a_best.copy()
            print("  Phase 4 done: obj=%.8f" % obj_chk)

    # ============ PHASE 5: More FW-LP if time remains ============
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 5: Additional FW-LP (budget=%.0fs) ===" % remaining)
        a = global_a.copy()
        a = normalize_nonneg(a, S)
        a_best, _ = frank_wolfe_lp(a, n, S, remaining * 0.7, max_iters=200, verbose=True,
                                    full_iters=0, reduced_K=300, alpha=1.0)
        a_best = normalize_nonneg(a_best, S)
        obj_chk = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj_chk) and obj_chk < global_best:
            global_best = obj_chk
            global_a = a_best.copy()
            print("  Phase 5 done: obj=%.8f" % obj_chk)

        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a = global_a.copy()
            a = normalize_nonneg(a, S)
            a_best, _ = coordinate_descent(a, n, S, remaining2, max_sweeps=10)
            a_best = normalize_nonneg(a_best, S)
            obj_chk = evaluate_sequence(a_best.tolist())
            if np.isfinite(obj_chk) and obj_chk < global_best:
                global_best = obj_chk
                global_a = a_best.copy()
                print("  Final CD: obj=%.8f" % obj_chk)

    # Final safety
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.8f (verified: %s)" % (
        global_best, "%.8f" % ver if np.isfinite(ver) else "INF"))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        print("Fallback to uniform: obj=%.8f" % evaluate_sequence(final_a))

    return final_a
