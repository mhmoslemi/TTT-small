The problem can be written as:
Minimize `t`
Subject to: `a^T M_k a <= t` for all `k = 0, ..., 2n-2`
            `sum(a) = S`
            `a >= 0`

This is a quadratic optimization problem (not convex in general, since we're minimizing the maximum of quadratic forms). However, if we fix the sum constraint and the non-negativity, it becomes a problem of minimizing the maximum eigenvalue of a certain matrix... no, that's not quite right either.

Actually, this is a **quatically constrained quadratic program (QCQP)**, which is NP-hard in general. So SDP relaxation is the best we can do for an exact solution, but it would be too slow for n=4096.

**Alternative: Alternating Direction Method of Multipliers (ADMM)**

We can write the problem as:
Minimize `max_k a^T M_k a`
Subject to: `a >= 0`, `sum(a) = S`

Using ADMM, we can split this into:
- Subproblem 1: Minimize `max_k a^T M_k a + (rho/2)||a - z + u||^2` (the quadratic part)
- Subproblem 2: Project `z - u` onto `{z >= 0, sum(z) = S}` (the constraint part)

The first subproblem is still hard (max of quadratics), but we can approximate it.

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses soft-max subgradient descent, spectral equalization, water-filling peak
reduction, and coordinate descent polishing with analytic 1D updates.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_from_conv(a, c):
    """Compute objective from precomputed convolution."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def compute_subgrad_soft(c, a, temperature):
    """Subgradient with soft-max weighting over all peaks."""
    n = len(a)
    c_max = c.max()
    # Shifted exponential weights
    shifted = c - c_max
    # Avoid overflow/underflow
    w = np.exp(temperature * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        top_len = max(1, n // 50)
        top_idx = np.argsort(c)[::-1][:top_len]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / top_len
    else:
        w /= w_sum

    # grad[m] = 2 * sum_k w[k] * a[k-m]
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 2).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def spectral_equalize(a, aggressiveness=0.05):
    """Compress dynamic range of |FFT(a)| to flatten time-domain peaks."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)

def water_fill_step(a, c, S, num_steps=3):
    """Iteratively reduce the peak by adjusting contributing coordinates."""
    n = len(a)
    a = a.copy()
    for _ in range(num_steps):
        k_max = np.argmax(c)
        c_peak = c[k_max]
        # Gradient of c[k_max] w.r.t. a[i] = 2 * a[k_max - i]
        # We want to reduce c_peak while keeping sum = S
        # Reduce a[i] proportionally to a[k_max-i] for the largest contributors
        contrib = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for i in range(lo, hi + 1):
            contrib[i] = 2.0 * a[k_max - i]

        # Pick top contributors and reduce them
        top_count = max(1, n // 32)
        top_idx = np.argsort(contrib)[::-1][:top_count]

        # Reduce top contributors, redistribute to others proportionally
        total_reduce = 0.0
        for i in top_idx:
            reduce_amt = min(a[i] * 0.1, contrib[i] * 0.01)
            a[i] -= reduce_amt
            total_reduce += reduce_amt

        # Redistribute to the rest proportionally to current value
        others = np.setdiff1d(np.arange(n), top_idx)
        if len(others) > 0:
            others_sum = a[others].sum()
            if others_sum > 1e-10:
                a[others] += total_reduce * a[others] / others_sum

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n)
            s = n
        a = a * (S / s)
        c = fft_convolve_self(a)
    return a, c

Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses soft-max subgradient descent, spectral equalization, water-filling peak
reduction, and coordinate descent polishing with analytic 1D updates.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_from_conv(a, c):
    """Compute objective from precomputed convolution."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def compute_subgrad_soft(c, a, temperature):
    """Subgradient with soft-max weighting over all peaks."""
    n = len(a)
    c_max = c.max()
    shifted = c - c_max
    w = np.exp(temperature * shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        top_len = max(1, n // 50)
        top_idx = np.argsort(c)[::-1][:top_len]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / top_len
    else:
        w /= w_sum
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 2).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def spectral_equalize(a, aggressiveness=0.05):
    """Compress dynamic range of |FFT(a)| to flatten time-domain peaks."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_max = mag.max()
    if mag_max < 1e-30:
        return a.copy()
    alpha = 1.0 + aggressiveness
    safe_mag = np.maximum(mag, 1e-30)
    mag_new = mag_max * (safe_mag / mag_max) ** alpha
    mag_new[mag < 1e-30] = 0.0
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:size]
    a_new = np.maximum(a_new, 0.0)
    S = a.sum()
    s_new = a_new.sum()
    if s_new < 1e-10:
        return a.copy()
    a_new = a_new * (S / s_new)
    return a_new


def water_fill_step(a, c, S, num_rounds=2):
    """Iteratively reduce the peak by adjusting top-contributing coordinates."""
    n = len(a)
    a = a.copy()
    for _ in range(num_rounds):
        k_max = np.argmax(c)
        contrib = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for i in range(lo, hi + 1):
            contrib[i] = 2.0 * a[k_max - i]
        top_count = max(1, n // 32)
        top_idx = np.argsort(contrib)[::-1][:top_count]
        total_reduce = 0.0
        for i in top_idx:
            reduce_amt = min(a[i] * 0.08, contrib[i] * 0.005)
            a[i] -= reduce_amt
            total_reduce += reduce_amt
        others_mask = np.ones(n, dtype=bool)
        others_mask[top_idx] = False
        others_idx = np.where(others_mask)[0]
        if len(others_idx) > 0:
            others_sum = a[others_idx].sum()
            if others_sum > 1e-10:
                a[others_idx] += total_reduce * a[others_idx] / others_sum
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n)
            s = n
        a = a * (S / s)
        c = fft_convolve_self(a)
    return a, c
