"""
Annealed log-sum-exp subgradient optimization with multiplicative updates,
spectral perturbations, and multi-phase strategy to minimize
2n * max(conv(a,a)) / sum(a)^2 for non-negative sequences.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]


def compute_lse_gradient(a, c, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*c)) w.r.t. a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # Numerically stable softmax
    c_shifted = T * (c - c.max())
    c_shifted = np.clip(c_shifted, -50.0, 0.0)
    w = np.exp(c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: uniform
        w = np.ones(len(c)) / len(c)
    else:
        w = w / w_sum

    # grad[i] = 2 * sum_j a[j] * w[i+j] for valid j
    # Using FFT: grad = 2 * (a_flipped * w_flipped)[n-1 : 2n-1][::-1]
    a_flipped = a[::-1]
    w_flipped = w[::-1]
    fsize = 1 << (n + len(w) - 2).bit_length()
    A = fft(a_flipped, fsize)
    W = fft(w_flipped, fsize)
    conv_result = np.real(ifft(A * W))
    grad = 2.0 * conv_result[n - 1:2 * n - 1][::-1]
    return grad, w


def compute_obj(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_conv_self(a)
    return 2.0 * n * c.max() / (S * S)


def annealed_optimize(a0, time_budget, T_start=100.0, T_end=5000.0,
                      cooling_rate=0.99997, eta0=0.3, eta_min=0.001,
                      perturb_every=3000, momentum=0.85,
                      stagnation_limit=2000, verbose=True):
    """
    Main annealed subgradient optimization with multiplicative updates.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 1e-300)
    a = a * (S / a.sum())

    c = fft_conv_self(a)
    cur_obj = compute_obj(a, c)
    best_obj = cur_obj
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)
    T = T_start
    eta = eta0
    iter_no = 0
    start = time.monotonic()
    stagnation_count = 0

    while time.monotonic() - start < time_budget:
        # Compute gradient
        grad, w = compute_lse_gradient(a, c, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        # Multiplicative update with backtracking
        accepted = False
        for bt in range(6):
            eta_try = eta * (0.7 ** bt)
            a_trial = a * np.exp(-eta_try * v)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_conv_self(a_trial)
            obj_trial = compute_obj(a_trial, c_trial)
            if obj_trial <= cur_obj * (1.0 + 1e-9):
                a = a_trial
                c = c_trial
                cur_obj = obj_trial
                accepted = True
                break

        if not accepted:
            # Very small step to avoid getting stuck
            a = a * np.exp(-eta * 0.05 * v)
            a = np.maximum(a, 1e-300)
            a = a * (S / a.sum())
            c = fft_conv_self(a)
            cur_obj = compute_obj(a, c)

        # Track best
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
            stagnation_count = 0
            eta = min(eta * 1.01, eta0 * 3.0)
        else:
            stagnation_count += 1
            eta = max(eta * 0.97, eta_min)

        # Perturbation when stuck
        if stagnation_count >= stagnation_limit:
            # Spectral perturbation: randomize phases in frequency domain
            fsize = 1 << (n - 1).bit_length()
            A = fft(a, fsize)
            phase = np.exp(1j * np.random.uniform(-0.3, 0.3, fsize))
            A *= phase
            a_new = np.real(ifft(A))[:n]
            a_new = np.maximum(a_new, 1e-300)
            a_new = a_new * (S / a_new.sum())
            c = fft_conv_self(a_new)
            obj_perturb = compute_obj(a_new, c)
            if obj_perturb < best_obj * 1.05:
                a = a_new
                cur_obj = obj_perturb
            else:
                # Try additive perturbation
                noise = np.random.randn(n) * 0.05 * S / np.sqrt(n)
                a = a + noise
                a = np.maximum(a, 1e-300)
                a = a * (S / a.sum())
                c = fft_conv_self(a)
                cur_obj = compute_obj(a, c)
            v *= 0.2
            eta = eta0
            stagnation_count = 0
            T = max(T * 0.5, T_start)  # Reheat
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    [perturb] iter={iter_no}, obj={cur_obj:.6f}, "
                      f"best={best_obj:.6f}, T={T:.1f}, t={elapsed:.1f}s")

        # Cool down temperature (increase T = sharper peak focus)
        T = min(T * (1.0 - cooling_rate + 1.0) * cooling_rate, T_end)
        # Actually: T should increase for sharper focus
        T = min(T * (2.0 - cooling_rate), T_end)
        # Simpler: geometric increase
        T = min(T * 1.0001, T_end)

        iter_no += 1
        if verbose and iter_no % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"T={T:.1f}, eta={eta:.4f}, t={elapsed:.1f}s")

    return best_a, best_obj, best_iter


def peak_directed_optimize(a0, time_budget, lr=0.002, momentum=0.9,
                           verbose=True):
    """
    Focus exclusively on reducing the single largest peak.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_conv_self(a)
    cur_obj = compute_obj(a, c)
    best_obj = cur_obj
    best_a = a.copy()
    v = np.zeros(n)
    start = time.monotonic()
    iter_no = 0

    while time.monotonic() - start < time_budget:
        # Find the worst peak
        k_max = np.argmax(c)
        c_max_val = c[k_max]

        # Gradient of c[k_max] w.r.t. a:
        # grad[i] = 2 * a[k_max - i] if 0 <= k_max - i < n, else 0
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[k_max - m]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        # Multiplicative update
        a = a * np.exp(-lr * v)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())

        c = fft_conv_self(a)
        cur_obj = compute_obj(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        iter_no += 1
        if verbose and iter_no % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    [peak] iter={iter_no}, obj={cur_obj:.6f}, "
                  f"best={best_obj:.6f}, t={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_descent(a0, time_budget, n_iterations_max=50000,
                       verbose=True):
    """
    Coordinate descent: try adjusting individual elements to reduce the max peak.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_conv_self(a)
    cur_obj = compute_obj(a, c)
    best_obj = cur_obj
    best_a = a.copy()
    start = time.monotonic()
    iter_no = 0

    # Determine which indices contribute to the max peak
    k_max = np.argmax(c)

    # Random scan order
    rng = np.random.default_rng(123)

    while time.monotonic() - start < time_budget and iter_no < n_iterations_max:
        # Pick a random index to try modifying
        i = rng.integers(0, n)
        orig = a[i]
        if orig < 1e-15:
            a[i] = S * 1e-6  # Small perturbation for zero elements
            s = a.sum()
            a = a * (S / s)
            c = fft_conv_self(a)
            obj_new = compute_obj(a, c)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
            else:
                a[i] = orig
            iter_no += 1
            continue

        # Try several multiplicative factors
        best_local = orig
        best_local_obj = compute_obj(a, c)
        for factor in [0.5, 0.7, 0.85, 1.1, 1.3, 1.5]:
            a[i] = orig * factor
            s = a.sum()
            if s < 1e-10:
                a[i] = orig
                break
            a_norm = a * (S / s)
            c_new = fft_conv_self(a_norm)
            obj_new = compute_obj(a_norm, c_new)
            if obj_new < best_local_obj:
                best_local_obj = obj_new
                best_local = orig * factor

        a[i] = best_local
        s = a.sum()
        a = a * (S / s)
        c = fft_conv_self(a)
        cur_obj = compute_obj(a, c)

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            # Update k_max
            k_max = np.argmax(c)

        iter_no += 1
        if verbose and iter_no % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"    [coord] iter={iter_no}, obj={cur_obj:.6f}, "
                  f"best={best_obj:.6f}, t={elapsed:.1f}s")

    return best_a, best_obj


def hybrid_direction_step(a, c, S, T=100.0):
    """
    Take a few steps using a hybrid direction: blend LSE gradient
    with direct peak gradient.
    """
    n = len(a)
    grad_lse, w = compute_lse_gradient(a, c, T)
    gl_norm = np.linalg.norm(grad_lse)
    if gl_norm < 1e-15:
        return a, c

    # Peak gradient
    k_max = np.argmax(c)
    grad_peak = np.zeros(n)
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    for m in range(lo, hi + 1):
        grad_peak[m] = 2.0 * a[k_max - m]
    gp_norm = np.linalg.norm(grad_peak)
    if gp_norm < 1e-15:
        gp_norm = 1.0
    grad_peak = grad_peak / gp_norm

    # Blend
    blend = 0.5
    grad = blend * (grad_lse / gl_norm) + (1 - blend) * grad_peak
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a, c
    grad = grad / gnorm

    # Update
    eta = 0.1
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-300)
    a_new = a_new * (S / a_new.sum())
    c_new = fft_conv_self(a_new)
    return a_new, c_new
