"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
1. Parametric search over families like exp(-alpha*(i/n)^beta) to find good initial shapes.
2. Local optimization using adaptive peak-pincer (top-k subgradient) with line search.
3. Multi-restart: perturb the best solution and re-optimize to escape local minima.
4. Final polish with high-precision peak targeting.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv(a):
    """Full convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_conv(a)
    return 2.0 * n * c.max() / (S * S)


def grad_max_conv(a, k_list):
    """
    Compute subgradient of max(conv) at the specified lag positions.
    Returns a vector of length n.
    For each k in k_list, the gradient of conv[k] w.r.t. a is the
    cross-correlation of a with a delta at position k.
    """
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    # Build weight vector: 1 at each k in k_list
    w = np.zeros(size)
    for k in k_list:
        w[int(k)] = 1.0
    
    W = fft(w, fsize)
    A = fft(a, fsize)
    # grad[m] = 2 * sum_k_in_list a[k-m] (cross-correlation)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def grad_logsumexp(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a.
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    
    size = 2 * n - 1
    fsize = 1 << (size + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def optimize_fista(a_init, max_time, rng, beta_start=5.0, beta_end=200.0):
    """
    FISTA optimization with soft-max objective and simplex constraint.
    Normalizes to sum = sqrt(2n).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = a.copy()
    best_score = fft_conv(y).max()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_time * 50))
        
        c = fft_conv(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best = y.copy()
        
        grad = grad_logsumexp(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        step = S / (beta * (1.0 + 0.01 * it))
        step = min(step, S / (2.0 * gnorm))
        
        x_new = project_simplex(y - step * grad, S)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        a = x_new
        t = t_new
        
        # Occasional noise to escape local minima
        if it % 500 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
    
    return best, best_score

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

(solving e^alpha = 1+2*alpha), but better shapes exist via generalized exponentials.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def grad_cross(a, w):
    """
    Compute cross-correlation: result[i] = sum_k w[k] * a[k-i].
    w is defined on [0, 2n-2], a on [0, n-1].
    Uses FFT for efficiency.
    """
    n = len(a)
    fsize = 1 << (2 * n - 1 - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return np.real(ifft(W * np.conj(A)))[:n]


def local_optimize(a_init, max_time, rng):
    """
    Combined local optimizer alternating soft-max FISTA and peak-pincer.
    Works with sum-normalized input (sum = sqrt(2n)).
    Returns (best_solution, best_maxconv).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)

    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = a.copy()
    best_score = fft_conv(y).max()
    it = 0
    stall_count = 0
    last_improve = best_score

    while time.monotonic() - start < max_time:
        it += 1
        # Switching: use soft-max early (smooth), peak-pincer later (precise)
        progress = min(1.0, it / (max_time * 30))
        use_fista = (it % 300 < 150) or (progress < 0.3)

        c = fft_conv(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best = y.copy()
            last_improve = best_score
            stall_count = 0
        else:
            stall_count += 1

        if use_fista:
            beta = min(500.0, 5.0 * (1.0 + 0.1 * it))
            c_max = c.max()
            exp_bc = np.exp(beta * (c - c_max))
            total = exp_bc.sum()
            if total < 1e-300:
                break
            w = exp_bc / total
            cross = grad_cross(y, w)
            grad = 2.0 * cross
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            # Step size: inverse of Lipschitz-like constant
            step = S / (beta * (1.0 + 0.01 * it))
        else:
            # Peak-pincer with adaptive k
            k = min(512, 8 + it // 5)
            top = np.argsort(c)[::-1][:k]
            w = np.zeros(2 * n - 1)
            for idx in top:
                w[int(idx)] = 1.0
            cross = grad_cross(y, w)
            grad = 2.0 * cross
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            step = S / (50.0 * (1.0 + 0.002 * it))

        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        # Line search: verify improvement, backtrack if needed
        c_new = fft_conv(x_new)
        new_max = c_new.max()
        if new_max > cur * 1.02:
            # Backtrack
            for _ in range(3):
                x_new = 0.5 * y + 0.5 * x_new
                x_new = project_simplex(x_new, S)
                c_new = fft_conv(x_new)
                if c_new.max() <= cur * 1.01:
                    break

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        a = x_new
        t = t_new

        # Noise injection to escape local minima
        if stall_count > 200 and it % 100 == 0:
            noise_scale = 0.01 * S / np.sqrt(n) * max(0.1, 1.0 - stall_count / 500.0)
            noise = rng.normal(0, noise_scale, n)
            y = project_simplex(y + noise, S)
            t = 1.0
            stall_count = 0

    return best, best_score


def make_parametric_inits(n):
    """Generate diverse parametric initializations."""
    tt = np.arange(n, dtype=np.float64) / n
    inits = []

    # Optimal exponential (alpha* = 1.256 from e^a = 1+2a)
    inits.append(("exp_1.256", np.exp(-1.256 * tt)))
    inits.append(("exp_1.0", np.exp(-1.0 * tt)))
    inits.append(("exp_1.5", np.exp(-1.5 * tt)))
    inits.append(("exp_2.0", np.exp(-2.0 * tt)))
    inits.append(("exp_0.8", np.exp(-0.8 * tt)))

    # Generalized exponential: exp(-alpha * (i/n)^beta)
    inits.append(("ge_1.5_1.2", np.exp(-1.5 * tt**1.2)))
    inits.append(("ge_1.5_1.5", np.exp(-1.5 * tt**1.5)))
    inits.append(("ge_2.0_1.3", np.exp(-2.0 * tt**1.3)))
    inits.append(("ge_2.0_1.5", np.exp(-2.0 * tt**1.5)))
    inits.append(("ge_3.0_1.5", np.exp(-3.0 * tt**1.5)))
    inits.append(("ge_1.2_0.8", np.exp(-1.2 * tt**0.8)))
    inits.append(("ge_1.0_0.7", np.exp(-1.0 * tt**0.7)))

    # Gaussian exponent: exp(-alpha * (i/n)^2)
    inits.append(("gauss_2", np.exp(-2.0 * tt**2)))
    inits.append(("gauss_4", np.exp(-4.0 * tt**2)))
    inits.append(("gauss_6", np.exp(-6.0 * tt**2)))
    inits.append(("gauss_10", np.exp(-10.0 * tt**2)))

    # Linear modulation: exp(-alpha*i/n) * (1 + gamma*i/n)
    inits.append(("mod_1.0_0.5", np.maximum(0, np.exp(-1.0 * tt) * (1 + 0.5 * tt))))
    inits.append(("mod_1.0_1.0", np.maximum(0, np.exp(-1.0 * tt) * (1 + 1.0 * tt))))
    inits.append(("mod_1.5_0.5", np.maximum(0, np.exp(-1.5 * tt) * (1 + 0.5 * tt))))

    # Power law (back-loaded): (1 - i/n + eps)^p
    inits.append(("pow_0.5", (1.0 - tt + 0.01)**0.5))
    inits.append(("pow_1.0", (1.0 - tt + 0.01)**1.0))
    inits.append(("pow_1.5", (1.0 - tt + 0.01)**1.5))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):

    Returns the best list found within the time budget.

    deadline = time.monotonic() + max(10.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    print("n = %d, S = %.4f, budget = %.0f sec" % (n, S, budget_s))

    # Collect initializations
    inits = []

    # Parent sequence (if available)
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("parent", np.clip(given, 0, 1000)))
    except Exception:
        pass

    # Parametric initializations
    param_inits = make_parametric_inits(n)
    inits.extend(param_inits)

    n_inits = len(inits)
    print("Total initializations: %d" % n_inits)

    # Phase 1: Quick evaluation of all initializations
    print("=== Phase 1: Evaluate inits ===")
    t0 = time.monotonic()
    init_scores = []
    for name, arr in inits:
        a = np.clip(arr.astype(np.float64), 0, 1000)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * (S / s)
        c = fft_conv(a)
        score = c.max()
        init_scores.append((score, name, a.copy()))

    init_scores.sort(key=lambda x: x[0])
    for i, (sc, nm, _) in enumerate(init_scores[:10]):
        print("  %d. %-20s obj=%.6f" % (i + 1, nm, sc))
    print("Phase 1 time: %.1f sec" % (time.monotonic() - t0))

    # Phase 2: Deep optimization of top candidates (55% of time)
    print("=== Phase 2: Deep optimization ===")
    top_n = min(4, len(init_scores))
    remaining = deadline - time.monotonic()
    time_per = remaining * 0.55 / top_n

    for rank in range(top_n):
        if time.monotonic() > deadline - 90:
            break
        score, name, a = init_scores[rank]
        seg = min(time_per, deadline - time.monotonic() - 60)
        if seg < 5:
            break
        print("  [%d/%d] Optimizing '%s' (init=%.6f) for %.0f sec" % (rank + 1, top_n, name, score, seg))
        a_opt, opt_score = local_optimize(a, seg, rng)
        if opt_score < global_best:
            global_best = opt_score
            global_a = a_opt.copy()
            print("    -> best = %.6f" % global_best)

    # Phase 3: Multi-restart from best (25% of time)
    print("=== Phase 3: Multi-restart ===")
    if global_a is not None:
        remaining = deadline - time.monotonic()
        n_restarts = 8
        time_per_r = remaining * 0.25 / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break
            # Decreasing noise
            noise_scale = (0.03 - 0.02 * r / n_restarts) * S / np.sqrt(n)
            perturbed = global_a + rng.normal(0, noise_scale, n)
            perturbed = project_simplex(perturbed, S)

            seg = min(time_per_r, deadline - time.monotonic() - 30)
            if seg < 3:

            a_opt, opt_score = local_optimize(perturbed, seg, rng)
            if opt_score < global_best:
                global_best = opt_score
                global_a = a_opt.copy()
                print("  Restart %d: improved to %.6f" % (r, global_best))

    # Phase 4: Final polish with remaining time
    print("=== Phase 4: Final polish ===")
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            print("  Polishing for %.0f sec" % remaining)
            a_opt, opt_score = local_optimize(global_a, remaining, rng)
            if opt_score < global_best:
                global_best = opt_score
                global_a = a_opt.copy()
                print("  -> final best = %.6f" % global_best)

    # Fallback
    if global_a is None:
        print("WARNING: No valid solution found, using uniform fallback")
        global_a = np.ones(n, dtype=np.float64) * S / n

        print("  evaluate_sequence rejected, using safe fallback")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
