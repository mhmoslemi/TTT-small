def fft_convolve_self(a):

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

    c = np.real(ifft(A * A))

def spectral_shaping(a, n, S, time_budget, beta=0.3, lambda_or=1.0, T_target=None, verbose=False):
    """Spectral shaping: iteratively flatten the power spectrum to reduce autocorrelation peak."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S/n)
    else:
        a = a * (S/s)
    
    fsize = 1 << (2*n-1).bit_length()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    step = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Compute spectrum
        A = fft(a, fsize)
        power = np.abs(A)**2
        
        # Target power: flat spectrum (except DC)
        # Total power = sum(power) = L * sum(a^2) by Parseval... 
        # Actually sum(power[k]) = L * sum(a_pad[j]^2) = L * sum(a[j]^2)
        # We want the autocorrelation to be flat: c[k] ≈ S^2/(2n-1) for all k
        # This means |A[k]|^2 should be approximately constant (except DC)
        
        # DC component
        dc = A[0]
        
        # For non-DC components, compute target
        # The "excess" power in non-DC components causes non-flat autocorrelation
        # We want to reduce peaks in the power spectrum
        
        # Method: compress the dynamic range of the power spectrum
        # power_new[k] = power[k]^(1-beta) * (mean of power)^(beta) for k != 0
        # This makes the spectrum more flat
        
        non_dc = np.ones(fsize)
        non_dc[0] = 0
        
        p_non_dc = power.copy()
        p_non_dc[0] = 0
        
        # Compute mean power of non-DC components
        mean_p = p_non_dc.sum() / (fsize - 1)
        
        if mean_p < 1e-30:
            break
        
        # Compress: new_power = (power * mean_p)^beta / mean_p^(2beta-1)... 
        # Simpler: new_power[k] = mean_p * (power[k]/mean_p)^(1-beta) for k != 0
        # This reduces the dynamic range by factor (power/mean)^(beta)
        
        ratio = p_non_dc / mean_p
        ratio = np.maximum(ratio, 1e-10)
        new_power = mean_p * ratio**(1.0 - beta)
        new_power[0] = power[0]  # Keep DC fixed
        
        # Take square root to get amplitude
        new_amp = np.sqrt(new_power)
        
        # Preserve phase
        phase = np.angle(A)
        A_new = new_amp * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.real(ifft(A_new))[:n]
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.full(n, S/n)
        else:
            a_new = a_new * (S / s_new)
        
        # Over-relaxation
        a_new = (1.0 - lambda_or) * a + lambda_or * a_new
        
        # Check objective
        c_new = fft_convolve_self(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        step += 1
        
        # Adapt beta: increase it slowly to push harder
        # (or decrease for stability)
        if no_improve > 50:
            # Try a different beta
            beta = beta * 0.9 if beta > 0.05 else beta * 1.1
            no_improve = 0
        
        if verbose and step % 100 == 0:
            print("  Spectral step %d: obj=%.7f, beta=%.3f, no_improve=%d" % (step, best_obj, beta, no_improve))
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based active set method with
line search, followed by log-Adam polishing and coordinate descent.

Key innovation: Use linear programming to find the optimal direction that
simultaneously reduces the top-K active autocorrelation lags, rather than
using a softmax-weighted subgradient. This gives a "smarter" descent direction
that accounts for the combinatorial structure of the active constraints.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def solve_direction_lp(a, c, active_lags, S, n):
    """Solve LP to find direction that reduces active lag autocorrelations."""
    # Variables: x = [a[0], ..., a[n-1], t]
    # Minimize t
    # s.t. 2*sum_j a_0[k-j]*a[j] - t <= c[k] for k in active_lags
    #      sum(a) = S
    #      a >= 0
    
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0
    
    # Build constraint matrix for active lags
    n_act = len(active_lags)
    A_act = np.zeros((n_act, n_vars))
    b_act = np.zeros(n_act)
    
    for idx, k in enumerate(active_lags):
        # Row: [2*a[k], 2*a[k-1], ..., 2*a[k-(n-1)], -1]
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A_act[idx, j] = 2.0 * a[kj]
        A_act[idx, -1] = -1.0
        b_act[idx] = c[k]
    
    # Non-negativity
    A_nn = -np.eye(n)
    b_nn = np.zeros(n)
    
    A_ub = np.vstack([A_act, A_nn])
    b_ub = np.concatenate([b_act, b_nn])
    
    # Equality: sum(a) = S
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0, 1000)] * n + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n]
    return None


def lp_active_set_method(a_init, S, n, time_budget, max_active=30,
                         restart_every=200, verbose=False):
    """LP-based active set method for minimizing max autocorrelation."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - start < time_budget:
        it += 1
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic restart
        if no_improve > restart_every:
            noise = 0.005 * (1.0 + 0.5 * np.random.random())
            a = best_a + noise * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            no_improve = 0
            it = 0
            continue
        
        # Identify active lags (top max_active)
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[min(max_active, len(c_sorted)) - 1]
        active = np.where(c >= threshold)[0]
        
        # Also include lags that are within 1% of max
        thresh2 = cur_obj * 0.99
        active2 = np.where(c >= thresh2)[0]
        active = np.unique(np.concatenate([active, active2]))
        
        # Limit active set size
        if len(active) > 2 * max_active:
            active = np.argsort(c[active])[::-1][:max_active]
            active = np.sort(active)
        
        # Solve LP
        a_new = solve_direction_lp(a, c, active, S, n)
        
        if a_new is None:
            no_improve += 1
            continue
        
        # Line search
        direction = a_new - a
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            no_improve += 1
            continue
        
        found = False
        for alpha in [min(1.0, 1.0/dir_norm*0.1), 0.5, 0.3, 0.1, 0.05, 0.02]:
            a_trial = a + alpha * direction
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            
            if obj_trial < cur_obj * (1 - 1e-10):
                a = a_trial
                found = True
                break
        
        if not found:
            no_improve += 1
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print("  LP iter %d: obj=%.7f, active=%d, no_improve=%d, t=%.0fs" % (
                it, best_obj, len(active), no_improve, elapsed))
    
    return best_a, best_obj
