```python
"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.

The solver uses three complementary ideas:
1. Structured initializations based on truncated generalized binomial polynomials,
   which have almost flat low-order self-convolution.
2. Smooth gradient descent in log-probability space using a log-sum-exp
   approximation to the max, with increasing temperature, Adam updates, and
   periodic noise restarts.
3. Exact one-coordinate descent on the objective with global renormalization,
   implemented with O(n) convolution updates per candidate value.

The function propose_candidate() manages a 2400-second budget, explores a few
sequence lengths, and returns the best valid sequence found.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def next_p2(m):
    """Smallest power of two >= m."""
    m = int(max(1, m))
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_nonneg(a, S):
    """Make a nonnegative and set its sum to S."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.empty(0, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(a.size, max(1e-12, S / a.size), dtype=np.float64)
    return a * (S / s)


def fft_conv_self(a):
    """Linear self-convolution using a zero-padded FFT."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = next_p2(2 * n - 1)
    A = fft(a, size)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def fast_obj(a):
    """
    Evaluate the objective directly.  This is the same as evaluate_sequence()
    except for the evaluator's security/clipping, which we avoid by returning
    valid non-negative values below the clip bound.
    """
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = fft_conv_self(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def generalized_binomial(n, alpha):
    """
    Coefficients of (1 - z)^(-alpha):
    a_0 = 1, a_k = a_{k-1} * (k + alpha - 1) / k.
    """
    a = np.empty(n, dtype=np.float64)
    if n == 0:
        return a
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((float(k) + alpha - 1.0) / float(k))
    return a


def make_initial(n, alpha=0.5, lam=0.0, S=1.0):
    """
    Create an initial non-negative sequence by combining a generalized binomial
    sequence with a reversed copy, then normalize to sum S.
    """
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    a = generalized_binomial(n, alpha)
    if abs(lam) > 1e-14:
        lam = float(min(1.0, max(0.0, lam)))
        ar = a[::-1]
        a = (1.0 - lam) * a + lam * ar
    a = np.maximum(a, 1e-30)
    return normalize_nonneg(a, S)


def best_initial_for_length(n, variants=None):
    """Return the best of several structured initializations for length n."""
    S = float(np.sqrt(2.0 * float(n)))
    if variants is None:
        variants = [
            (0.50, 0.00),
            (0.55, 0.00),
            (0.60, 0.00),
            (0.50, 0.15),
            (0.50, 0.30),
        ]
    best_a = None
    best_obj = np.inf
    for alpha, lam in variants:
        a = make_initial(n, alpha=alpha, lam=lam, S=S)
        obj = float(fft_conv_self(a).max())
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    if best_a is None:
        best_a = make_initial(n, 0.5, 0.0, S)
        best_obj = float(fft_conv_self(best_a).max())
    return best_a, best_obj, S


def correlation_grad(p, a):
    """
    For weighted self-convolution F(a) = sum_k p_k (a*a)_k,
    returns dF/da_i = 2 * sum_j p_{i+j} a_j.
    """
    n = len(a)
    L = 2 * n - 1
    pr = p[::-1]
    fsize = next_p2(n + L - 1)
    A = fft(a, fsize)
    P = fft(pr, fsize)
    conv = np.real(ifft(A * P))
    v = conv[L - 1::-1]
    return 2.0 * v


def consider_candidate(best_a, best_obj, a, S):
    """Normalize a to sum S and update best if objective improves."""
    if a is None:
        return best_a, best_obj
    a = normalize_nonneg(a, S)
    c = fft_conv_self(a)
    obj = float(c.max())
    if np.isfinite(obj) and obj < best_obj:
        return a, obj
    return best_a, best_obj


def run_adam(a0, S, time_budget, T_start=10.0, T_end=10000.0, lr=0.06,
             beta1=0.9, beta2=0.999, seed=1, max_no_improve=350,
             noise_scale=0.02, verbose=False):
    """
    Adam in log-probability coordinates for the log-sum-exp approximated
    max self-convolution, with fixed sum S.  Tracks and returns the objective
    maximum (which equals the true objective when S = sqrt(2n)).
    """
    a = normalize_nonneg(a0, S)
    n = a.size
    if n <= 1:
        obj = float(fft_conv_self(a).max()) if n == 1 else 0.0
        return a, obj

    start = time.monotonic()
    if time_budget <= 0.0:
        c = fft_conv_self(a)
        return a, float(c.max())

    c = fft_conv_self(a)
    best_obj = float(c.max())
    best_a = a.copy()

    theta = np.log(np.maximum(a / S, 1e-30))
    theta -= theta.max()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    no_improve = 0
    rng = np.random.default_rng(seed)

    while True:
        elapsed = time.monotonic() - start
        if elapsed >= time_budget:
            break

        progress = min(1.0, elapsed / time_budget)
        if T_end > T_start:
            T = T_start * (T_end / T_start) ** (progress ** 0.7)
        else:
            T = float(T_start)

        c = fft_conv_self(a)
        cur_obj = float(c.max())
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            noise = noise_scale * (0.5 + progress)
            noisy = best_a + noise * S * np.abs(rng.standard_normal(n))
            noisy = noisy * np.exp(0.5 * noise * rng.standard_normal(n))
            a = normalize_nonneg(noisy, S)
            theta = np.log(np.maximum(a / S, 1e-30))
            theta -= theta.max()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0
            if verbose:
                print("      Adam restart, best=%.8f" % best_obj)
            continue

        if T <= 1e-6:
            p = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)
        else:
            x = T * c
            xm = float(x.max())
            ex = np.exp(x - xm)
            ss = float(ex.sum())
            if ss < 1e-300:
                p = np.zeros(2 * n - 1, dtype=np.float64)
                p[int(np.argmax(c))] = 1.0
            else:
                p = ex / ss

        g = correlation_grad(p, a)
        q = a / S
        dot = float(np.dot(q, g))
        gt = q * g - dot * q

        tnorm = float(np.linalg.norm(gt))
        if not np.isfinite(tnorm) or tnorm < 1e-16:
            break

        t += 1
        m *= beta1
        m += (1.0 - beta1) * gt
        gt2 = gt * gt
        v *= beta2
        v += (1.0 - beta2) * gt2

        b1 = 1.0 - beta1 ** t
        b2 = 1.0 - beta2 ** t
        mh = m / b1
        vh = v / b2
        denom = np.sqrt(vh) + 1e-8

        lr_t = lr * (0.25 + 0.75 * (1.0 - progress))
        theta = theta - lr_t * (mh / denom)

        if not np.all(np.isfinite(theta)):
            a = best_a.copy()
            theta = np.log(np.maximum(a / S, 1e-30))
            theta -= theta.max()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = max(0, no_improve - max_no_improve // 2)
            continue

        theta -= theta.max()
        exp_t = np.exp(theta)
        tot = float(exp_t.sum())
        if not np.isfinite(tot) or tot <= 1e-300:
            a = best_a.copy()
            theta = np.log(np.maximum(a / S, 1e-30))
            theta -= theta.max()
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = max(0, no_improve - max_no_improve // 2)
            continue

        a = S * exp_t / tot

        if verbose and t % 5000 == 0:
            print("      Adam t=%d obj=%.8f T=%.1f" % (t, best_obj, T))

    return best_a, best_obj


def run_coordinate_descent(a0, S, time_budget, seed=0, max_sweeps=1000,
                           verbose=False):
    """
    Exact one-coordinate descent with global renormalization.

    For changing a_i from x to y while keeping other coordinates fixed and
    then scaling the whole vector back to sum S, the raw convolution update is
    linear in y and can be updated in O(n).  The objective after renormalization
    is exactly
        max(raw_new_b) * (S / (S + y - x))^2.
    """
    a = normalize_nonneg(a0, S)
    n = a.size
    if n <= 1:
        obj = float(fft_conv_self(a).max()) if n == 1 else 0.0
        return a, obj

    L = 2 * n - 1
    S2 = S * S
    mean = S / float(n)
    b = fft_conv_self(a)
    cur_obj = float(b.max())
    best_obj = cur_obj
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(seed)
    tmp = np.empty(n, dtype=np.float64)
    mult = np.empty(n, dtype=np.float64)

    rel_factors = (0.0, 0.25, 0.5, 0.75, 1.25, 1.7, 2.2, 3.0)
    abs_factors = (0.0, 0.5, 1.0, 3.0, 10.0)
    max_y = min(900.0, 4.0 * S)
    tiny = 1e-14 * max(1.0, S)

    sweep = 0
    no_improve_sweeps = 0

    while True:
        if time.monotonic() - start >= time_budget:
            break
        if sweep >= max_sweeps:
            break

        order = rng.permutation(n)
        improved_sweep = False

        for ii, i in enumerate(order):
            if time.monotonic() - start >= time_budget:
                break

            i = int(i)
            x = float(a[i])

            best_y = x
            best_cand_obj = cur_obj

            seg_st = i
            seg_en = i + n

            # Fast outside-max: if the current global maximum is outside the
            # affected interval, it is unchanged for this coordinate move.
            idx_max = int(np.argmax(b))
            if idx_max < seg_st or idx_max >= seg_en:
                outside = cur_obj
            else:
                outside = -np.inf
                if i > 0:
                    ml = float(b[:i].max())
                    if ml > outside:
                        outside = ml
                if seg_en < L:
                    mr = float(b[seg_en:].max())
                    if mr > outside:
                        outside = mr
                if outside == -np.inf:
                    outside = 0.0

            base_seg = b[seg_st:seg_en]

            # Relative coordinate candidates.
            if x > 1e-14 * S:
                for f in rel_factors:
                    y = x * f
                    if y < 0.0 or y > max_y:
                        continue
                    d = y - x
                    if abs(d) < tiny:
                        continue
                    Snew = S + d
                    if Snew <= 1e-14:
                        continue
                    if d != 0.0:
                        np.multiply(a, 2.0 * d, out=mult)
                        np.add(base_seg, mult, out=tmp)
                        tmp[i] += d * d
                        max_seg = float(tmp.max())
                    else:
                        max_seg = float(base_seg.max())
                    raw_max = outside if outside > max_seg else max_seg
                    cand_obj = raw_max * (S2 / (Snew * Snew))
                    if cand_obj < best_cand_obj - 1e-14:
                        best_cand_obj = cand_obj
                        best_y = y

            # Absolute coordinate candidates, useful for small or zero entries.
            for f in abs_factors:
                y = mean * f
                if y < 0.0 or y > max_y:
                    continue
                d = y - x
                if abs(d) < tiny:
                    continue
                Snew = S + d
                if Snew <= 1e-14:
                    continue
                if d != 0.0:
                    np.multiply(a, 2.0 * d, out=mult)
                    np.add(base_seg, mult, out=tmp)
                    tmp[i] += d * d
                    max_seg = float(tmp.max())
                else:
                    max_seg = float(base_seg.max())
                raw_max = outside if outside > max_seg else max_seg
                cand_obj = raw_max * (S2 / (Snew * Snew))
                if cand_obj < best_cand_obj - 1e-14:
                    best_cand_obj = cand_obj
                    best_y = y

            if best_y != x:
                d = best_y - x
                Snew = S + d
                seg_view = b[seg_st:seg_en]
                np.multiply(a, 2.0 * d, out=mult)
                seg_view += mult
                seg_view[i] += d * d
                a[i] = best_y
                if Snew > 1e-14:
                    scale = S / Snew
                    if abs(scale - 1.0) > 1e-14:
                        a *= scale
                        b *= (scale * scale)
                cur_obj = best_cand_obj
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a.copy()
                    improved_sweep = True

        sweep += 1

        # Recompute from a to remove drift.
        b = fft_conv_self(a)
        cur_obj = float(b.max())
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            improved_sweep = True

        if improved_sweep:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:
                break

        if verbose and sweep % 10 == 0:
            print("      CD sweep=%d obj=%.8f" % (sweep, best_obj))

    return normalize_nonneg(best_a, S), best_obj


def optimize_candidate(a0, S, time_budget, seed=0, verbose=False):
    """Run Adam -> coordinate descent -> high-T Adam -> coordinate polish."""
    a = normalize_nonneg(a0, S)
    if a.size <= 1:
        obj = float(fft_conv_self(a).max()) if a.size == 1 else 0.0
        return a, obj

    start = time.monotonic()
    best_a = a.copy()
    c = fft_conv_self(best_a)
    best_obj = float(c.max())

    def remaining():
        return max(0.0, time_budget - (time.monotonic() - start))

    n = a.size
    if n > 32768:
        ad1 = 0.75
        cd1 = 0.0
        ad2 = 0.25
        cd2 = 0.0
    elif n > 16384:
        ad1 = 0.55
        cd1 = 0.15
        ad2 = 0.25
        cd2 = 0.05
    elif n > 8192:
        ad1 = 0.48
        cd1 = 0.22
        ad2 = 0.22
        cd2 = 0.08
    else:
        ad1 = 0.42
        cd1 = 0.30
        ad2 = 0.18
        cd2 = 0.10

    # Phase 1: mid-temperature Adam.
    t1 = ad1 * time_budget
    if t1 > 5.0:
        t1 = min(t1, remaining())
        a1, obj1 = run_adam(
            best_a, S, t1,
            T_start=10.0, T_end=6000.0,
            lr=0.06, seed=seed + 1,
            max_no_improve=350, noise_scale=0.018,
            verbose=False
        )
        best_a, best_obj = consider_candidate(best_a, best_obj, a1, S)
        if verbose:
            print("    Phase 1 Adam: %.8f" % best_obj)

    # Phase 2: coordinate descent.
    t2 = cd1 * time_budget
    if t2 > 20.0 and n <= 32768:
        t2 = min(t2, remaining())
        a2, obj2 = run_coordinate_descent(
            best_a, S, t2,
            seed=seed + 2,
            max_sweeps=500,
            verbose=False
        )
        best_a, best_obj = consider_candidate(best_a, best_obj, a2, S)
        if verbose:
            print("    Phase 2 CD: %.8f" % best_obj)

    # Phase 3: high-temperature Adam to focus on active peaks.
    t3 = ad2 * time_budget
    if t3 > 5.0:
        t3 = min(t3, remaining())
        a3, obj3 = run_adam(
            best_a, S, t3,
            T_start=500.0, T_end=25000.0,
            lr=0.025, seed=seed + 3,
            max_no_improve=250, noise_scale=0.008,
            verbose=False
        )
        best_a, best_obj = consider_candidate(best_a, best_obj, a3, S)
        if verbose:
            print("    Phase 3 HighT Adam: %.8f" % best_obj)

    # Phase 4: final coordinate polish.
    t4 = min(cd2 * time_budget, remaining())
    if t4 > 20.0 and n <= 32768:
        a4, obj4 = run_coordinate_descent(
            best_a, S, t4,
            seed=seed + 4,
            max_sweeps=250,
            verbose=False
        )
        best_a, best_obj = consider_candidate(best_a, best_obj, a4, S)
        if verbose:
            print("    Phase 4 Final CD: %.8f" % best_obj)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=None):
    """
    Main entry point.  Searches for the best non-negative coefficient sequence.
    """
    t0 = time.monotonic()

    if budget_s is None:
        total_time = 2360.0
    else:
        try:
            total_time = float(budget_s)
        except Exception:
            total_time = 2360.0
    total_time = max(120.0, min(total_time, 2370.0))

    # Reserve time for final verification and a short final polish.
    reserve = 90.0 if total_time > 400 else 30.0
    main_budget = max(120.0, total_time - reserve)
    deadline = t0 + total_time - 8.0

    rng = np.random.default_rng(seed)
    _ = rng

    global_a = None
    global_S = 1.0
    global_obj = np.inf

    # ------------------------------------------------------------------
    # Use the previously provided sequence if available.
    # ------------------------------------------------------------------
    given_a = None
    given_S = None
    given_n = None
    try:
        raw = height_sequence_1
        arr = np.asarray(raw, dtype=np.float64).reshape(-1)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size > 0 and float(arr.sum()) > 0.01:
            given_n = int(arr.size)
            given_S = float(np.sqrt(2.0 * float(given_n)))
            given_a = normalize_nonneg(arr, given_S)
            c = fft_conv_self(given_a)
            gob = float(c.max())
            print("Given sequence: n=%d internal_obj=%.8f" % (given_n, gob))
            global_a, global_obj = consider_candidate(global_a, global_obj, given_a, given_S)
            global_S = given_S
    except Exception as e:
        print("Could not load height_sequence_1: %s" % str(e))

    # ------------------------------------------------------------------
    # Decide which lengths to optimize.
    # ------------------------------------------------------------------
    lengths = []
    if given_n is not None:
        lengths.append(given_n)
    if 4096 not in lengths:
        lengths.append(4096)
    if 8192 not in lengths:
        lengths.append(8192)
    lengths = lengths[:3]

    candidates = []

    if given_a is not None:
        candidates.append((given_a, given_S, 0.45, "given"))

    for L in lengths:
        if L <= 0:
            continue
        a_init, obj_init, S_init = best_initial_for_length(L)
        if L == given_n and given_a is not None:
            # A same-length structured start can still provide a different basin.
            weight = 0.20
        else:
            if L == 4096:
                weight = 0.40
            elif L == 8192:
                weight = 0.32
            else:
                weight = 0.25
        candidates.append((a_init, S_init, weight, "base_%d" % L))
        print("Init n=%d: internal_obj=%.8f" % (L, obj_init))

    if not candidates:
        a_init, obj_init, S_init = best_initial_for_length(4096)
        candidates.append((a_init, S_init, 1.0, "base_4096"))
        print("Fallback init n=4096: internal_obj=%.8f" % obj_init)

    # Normalize weights.
    total_w = 0.0
    for _, _, w, _ in candidates:
        total_w += max(0.0, float(w))
    if total_w <= 0.0:
        total_w = 1.0
        for i in range(len(candidates)):
            candidates[i] = (candidates[i][0], candidates[i][1], 1.0, candidates[i][3])

    print("Main search budget: %.1fs, candidates: %s" % (
        main_budget, [c[3] + "(n=%d,w=%.2f)" % (len(c[0]), c[2] / total_w) for c in candidates]
    ))

    # ------------------------------------------------------------------
    # Main optimization over candidate starts.
    # ------------------------------------------------------------------
    cand_idx = 0
    for a_cand, S_cand, w_cand, label in candidates:
        cand_idx += 1
        now = time.monotonic()
        if now > deadline - 45:
            break

        time_cand = (w_cand / total_w) * main_budget
        time_cand = min(time_cand, deadline - now - 30.0)
        if time_cand < 30.0:
            break

        print("Optimizing %s: n=%d for %.1fs" % (
            label, len(a_cand), time_cand
        ))

        a_opt, obj_opt = optimize_candidate(a_cand, S_cand, time_cand,
                                            seed=1000 * cand_idx + int(seed),
                                            verbose=True)
        global_a, global_obj = consider_candidate(global_a, global_obj, a_opt, S_cand)
        global_S = S_cand
        print("After %s: best internal_obj=%.8f" % (label, global_obj))

    # ------------------------------------------------------------------
    # Final polish on the global best.
    # ------------------------------------------------------------------
    if global_a is not None:
        now = time.monotonic()
        rem = max(0.0, deadline - now - 12.0)

        if rem > 40 and len(global_a) <= 32768:
            cd_time = min(rem * 0.8, 240.0)
            a_pol, obj_pol = run_coordinate_descent(
                global_a, global_S, cd_time,
                seed=9991, max_sweeps=300, verbose=False
            )
            global_a, global_obj = consider_candidate(global_a, global_obj, a_pol, global_S)
            print("Final CD: %.8f" % global_obj)

        now = time.monotonic()
        rem = max(0.0, deadline - now - 12.0)
        if rem > 40 and len(global_a) <= 65536:
            a_pol, obj_pol = run_adam(
                global_a, global_S, min(rem, 180.0),
                T_start=500.0, T_end=25000.0,
                lr=0.02,
                seed=9992,
                max_no_improve=250,
                noise_scale=0.006,
                verbose=False
            )
            global_a, global_obj = consider_candidate(global_a, global_obj, a_pol, global_S)
            print("Final residual Adam: %.8f" % global_obj)

    # ------------------------------------------------------------------
    # Final validation and fallback.
    # ------------------------------------------------------------------
    if global_a is None:
        global_a, global_obj, global_S = best_initial_for_length(4096)

    global_a = np.asarray(global_a, dtype=np.float64)
    global_a = np.clip(global_a, 0.0, 1000.0)
    global_a = normalize_nonneg(global_a, global_S)

    final_list = [float(x) for x in global_a]

    try:
        verified = evaluate_sequence(final_list)
    except Exception as e:
        verified = np.inf
        print("Final evaluate_sequence error: %s" % str(e))

    if not np.isfinite(verified) or verified > 2.5:
        # Very robust fallback: 256 ones gives objective 2.
        final_list = [1.0] * 256
        try:
            verified = evaluate_sequence(final_list)
        except Exception:
            verified = 2.0

    print("FINAL internal_obj=%.8f verified_obj=%s" % (
        global_obj, "%.8f" % verified if np.isfinite(verified) else "inf"
    ))

    return final_list
```
