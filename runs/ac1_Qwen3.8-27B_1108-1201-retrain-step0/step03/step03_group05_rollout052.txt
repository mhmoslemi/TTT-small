def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_given_weights(a, w):
    """grad_a[i] = 2 * sum_k w[k] * (a[i-k] + a[k-i]) via FFT."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) * S / n

def score_of(a):
    """Compute the score 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()

    c = fft_convolve_self(a)

    """
    Adaptive peak-shaving with exponential weighting and backtracking.

    n = len(a)

def lbfgs_smooth_obj(z, beta, S):
    """LSE-smoothed objective and gradient w.r.t. softmax params z."""
    n = len(z)
    z_s = z - z.max()
    exp_z = np.exp(z_s)
    a = S * exp_z / exp_z.sum()
    
    c = fft_convolve_self(a)
    c_max = c.max()
    
    # Log-sum-exp of beta * c
    beta_c = beta * (c - c_max)
    beta_c = np.clip(beta_c, -700, 700)
    sum_exp = np.exp(beta_c).sum()
    obj = (c_max + np.log(sum_exp) / beta)
    
    w = np.exp(beta_c) / sum_exp
    grad_a = compute_grad_given_weights(a, w)
    
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)
    
    return obj, grad_z
