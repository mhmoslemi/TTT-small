"""
Minimize 2*n*max_conv/sum^2 using adaptive FISTA with LSE smoothing,
structured perturbations, and top-k subgradient polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_score(a):
    """Compute the evaluation score (same as evaluate_sequence but faster)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    # Clip to prevent overflow
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def topk_weighted_subgradient(a, c, k, power=2.0):
    """Weighted subgradient from top-k lags."""
    n = len(a)
    k = min(k, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    vals = c[top_idx]
    # Weight by value^power
    w = np.power(np.maximum(vals, 1e-30), power)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    grad = np.zeros(n)
    for j in range(k):
        lag = int(top_idx[j])
        lo = max(0, lag - n + 1)
        hi = min(lag, n - 1)
        idx = np.arange(lo, hi + 1)
        grad[idx] += w[j] * 2.0 * a[lag - idx]
    return grad


def structured_perturb(a, n, S, rng, strength=0.02):
    """Add a random sinusoidal perturbation."""
    freq = rng.uniform(1, n // 4)
    phase = rng.uniform(0, 2 * np.pi)
    tt = np.arange(n)
    perturbation = strength * S / np.sqrt(n) * np.sin(2 * np.pi * freq * tt / n + phase)
    a_new = a + perturbation
    return project_simplex(a_new, S)


def run_fista_adaptive(a_init, n, S, max_time, rng, 
                       beta_init=10.0, beta_rate=0.005, beta_max=5000.0,
                       use_topk=False, topk_init=4, topk_rate=2,
                       perturb_every=3000, perturb_strength=0.02):
    """FISTA with adaptive beta and structured perturbations."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    patience = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it
            patience = 0
        else:
            patience += 1

        # Choose gradient
        if use_topk and patience > 2000:
            topk = min(64, topk_init + topk_rate * (patience // 2000))
            grad = topk_weighted_subgradient(y, c, topk, power=2.0)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        a_norm_sq = np.dot(a, a)
        L = max(beta * a_norm_sq, gnorm * gnorm / max(np.dot(grad, y - a) + 1e-10, 1e-10))
        # Actually, let's use a simpler approach

        # Damped step if we're not making progress

            step *= max(0.5, 1.0 - patience / 10000.0)

        x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

            a = y.copy()
            t = 1.0
            patience = 0
            last_improve_it = it

    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

def run_exact_polish(a_init, n, S, max_time, rng):
    """Polish using exact subgradient with backtracking line search."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    patience = 0
    y = a.copy()
    t_mom = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            patience = 0
        else:
            patience += 1

        if patience > 5000:
            break

        # Exact subgradient
        grad = exact_subgradient(y, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Line search (backtracking)
        step = S / (gnorm * 2.0)
        for _ in range(10):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur * (1 - 1e-6) or step < 1e-10:
                break
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)

        # Momentum (milder than FISTA for non-smooth)
        t_mom_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_mom_new) * (x_new - y)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_mom_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.arange(n) / n

    # Window functions
    inits.append(0.5 * (1 + np.cos(2 * np.pi * tt)))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(0.42 - 0.5 * np.cos(2 * np.pi * tt) + 0.08 * np.cos(4 * np.pi * tt))
    inits.append(np.exp(-8.0 * (tt - 0.5) ** 2))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(np.sin(1.5 * np.pi * tt) ** 2)

    # Multi-sinusoidal (various frequencies)
    for k in [2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.3 * np.sin(2 * np.pi * k * tt) + 0.1 * np.cos(2 * np.pi * (k+1) * tt))
    
    # Chirp (linear frequency modulation)
    inits.append(np.abs(np.exp(1j * np.pi * tt ** 2)))  # constant magnitude
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * (tt + tt**2)))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.3, n)))

    # Block structures
    for nblock in [3, 5, 7]:
        block_len = n // nblock
        heights = rng.uniform(0.5, 1.2, nblock)
        block_seq = np.repeat(heights, block_len)
        if len(block_seq) < n:
            block_seq = np.concatenate([block_seq, np.full(n - len(block_seq), heights[-1])])
        inits.append(block_seq[:n])

    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.3
    alt[1::2] = 0.7
    inits.append(alt)

    return [np.maximum(x, 0.01) for x in inits]


def downsample_upsample(a, n_small):
    """Downsample by averaging, then upsample by interpolation."""
    n = len(a)
    if n_small >= n:
        return a.copy()
    nblocks = n // n_small
    a_small = a[:nblocks * n_small].reshape(n_small, nblocks).mean(axis=1)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n)
    return np.interp(t_new, t_old, a_small)
