"""
Minimize 2*n*max_conv/sum^2 using:
1. L-BFGS with hot-set log-sum-exp smoothing (focused on top-k lags)
2. Adaptive Nesterov with top-k subgradient
3. Equiripple refinement
4. Multiple initializations and restarts

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize

# Helper functions (top-level, no closures)

def fft_convolve_self(a):
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A*A))[:size]

def compute_grad_hot(a, w_hot, hot_indices):
    """Compute gradient contribution from hot lags only."""
    n = len(a)
    w_full = np.zeros(len(w_hot) + n - 1)  # need to be careful about size
    # Actually, w_hot corresponds to specific lags in the full convolution
    # The gradient of b[k] w.r.t. a[m] is a[k-m] (+ a[k-m] from the other side)
    # = 2*a[k-m] (or a[m] if k=2m)
    # So grad_a[m] = 2 * sum_k w[k] * a[k-m] (with appropriate boundary handling)
    # This is a cross-correlation of a with w
    fsize = 1 << (n + len(w_hot) + 1).bit_length()
    # w_full is the weight vector placed at the hot indices in the full convolution
    # But for the gradient, we need the cross-correlation
    # grad_a[m] = sum_k w_k * (a[k-m] + a[k-m]) for valid indices
    # = 2 * sum_k w_k * a[k-m]
    # This is 2 * cross_correlation(a, w_full)[m] where w_full has w_k at position k
    w_full = np.zeros(2*n - 1)
    for i, k in enumerate(hot_indices):

"""
Minimize 2*n*max_conv/sum^2 using:
1. Hot-set log-sum-exp smoothing with L-BFGS (focused on top-k convolution lags)
2. Adaptive Nesterov with top-k subgradient
3. Equiripple refinement via iterative hot-set updates
4. Multiple diverse initializations (ramp, cosine, step) with restarts
Key improvement: focusing optimization on the "hot" lags (where peak occurs)
converges faster than full log-sum-exp smoothing.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]


def compute_grad_full(a, w_full):
    """Compute grad_a = 2 * cross_correlation(a, w_full) via FFT."""
    n = len(a)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def softmax_a(z, S):
    """Convert softmax params to sequence."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = S * exp_z / exp_z.sum()
    return np.maximum(a, 0.0)


def hot_set_obj_grad(z, beta, S, hot_indices, full_mode=False):
    """
    Compute smoothed objective and gradient using hot-set or full log-sum-exp.
    Returns (obj, grad_z, c_max, hot_indices_updated).
    """
    n = len(z)
    a = softmax_a(z, S)
    c = fft_convolve_self(a)

    if full_mode:
        idx = np.arange(len(c))
    else:
        idx = hot_indices

    c_sel = c[idx]
    c_max = c_sel.max()
    exp_bc = np.exp(beta * (c_sel - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w_sel = exp_bc / sum_exp_bc
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(idx):
        w_full[int(k)] = w_sel[i]

    grad_a = compute_grad_full(a, w_full)
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z, c_max


def run_lbfgs_hots(a_init, n, S, max_time, rng, n_hots=20,
                   beta_schedule=None, maxiter_per_beta=300):
    """
    L-BFGS continuation with hot-set focus.
    """
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 400, 800, 1500, 3000, 5000, 10000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()

    # Initial hot set
    hot_indices = np.argsort(c)[::-1][:max(n_hots, 10)]

    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() - start > max_time - 5:
            break

        # Update hot set based on current a
        c_curr = fft_convolve_self(a)
        hot_indices = np.argsort(c_curr)[::-1][:max(n_hots, 10)]

        n_attempts = 1
        if beta >= 500 and max_time > 120:

            if attempt > 0:

def run_nesterov_adaptive(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=256):
    """Nesterov-accelerated top-k with adaptive step size."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 2.0 * S / (n * 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 200)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step: decrease if score increased
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        # Simple adaptive: shrink step if we're oscillating
        if it % 10 == 0:
            step *= 0.995

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 10000 == 0 and rng.random() < 0.05:
            y = project_simplex(y + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            t = 1.0
            step = 2.0 * S / (n * 1.0)

    return best_local, best_score


def equiripple_refine(a_init, n, S, max_time, rng, n_hots=15):
    """
    Equiripple refinement: iteratively equalize the peak convolution values
    at the hot lags using L-BFGS with increasing beta.
    """
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    betas = [50, 100, 200, 400, 800, 1600, 3200, 6400, 12800, 25600]

    for beta in betas:
        if time.monotonic() - start > max_time - 3:
            break

        c = fft_convolve_self(a)
        hot_indices = np.argsort(c)[::-1][:n_hots]

        z = np.log(np.maximum(a, 1e-30) / S)
        z = z - z.mean()

        def fun(z_val):
            a_t = softmax_a(z_val, S)
            c_t = fft_convolve_self(a_t)
            c_hot = c_t[hot_indices]
            c_max = c_hot.max()
            exp_bc = np.exp(beta * (c_hot - c_max))
            return (c_max + np.log(exp_bc.sum())) / beta

        def jac(z_val):
            a_t = softmax_a(z_val, S)
            c_t = fft_convolve_self(a_t)
            c_hot = c_t[hot_indices]
            c_max = c_hot.max()
            exp_bc = np.exp(beta * (c_hot - c_max))
            w_hot = exp_bc / exp_bc.sum()
            w_full = np.zeros(2 * n - 1)
            for i, k in enumerate(hot_indices):
                w_full[int(k)] = w_hot[i]
            grad_a = compute_grad_full(a_t, w_full)
            dot_ga_a = np.dot(grad_a, a_t)
            return a_t * (grad_a - dot_ga_a / S)

        try:
            result = scipy_minimize(fun, z, jac=jac, method='L-BFGS-B',
                                     options={'maxiter': 200, 'maxfun': 600,
                                              'ftol': 1e-14, 'gtol': 1e-10,
                                              'maxls': 30, 'iprint': -1})
        except Exception:
            continue

        z = result.x - result.x.mean()
        a = softmax_a(z, S)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * S / s

        c = fft_convolve_self(a)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score

                top_k_init=15, top_k_max=512
            )
            a_pol = safe_normalize(a_pol, n)
            c_pol = fft_convolve_self(a_pol)
            if c_pol.max() < global_best:
                global_best = c_pol.max()
                global_a = a_pol.copy()
                print("  Nesterov: %.8f" % global_best)

    # --- Phase 6: Final high-beta L-BFGS ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print("\n=== Phase 6: Final Polish (%.1f sec) ===" % remaining)
            a_fin, score_fin = run_lbfgs_hots(
                global_a, n, S, remaining, rng,
                n_hots=10,
                beta_schedule=[500, 1000, 2000, 5000, 10000, 20000],
                maxiter_per_beta=300
            )
            a_fin = safe_normalize(a_fin, n)
            c_fin = fft_convolve_self(a_fin)
            if c_fin.max() < global_best:
                global_best = c_fin.max()
                global_a = a_fin.copy()
                print("  Final: %.8f" % global_best)

    # --- Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.8f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
