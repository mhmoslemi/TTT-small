Minimize 2*n*max(conv(a,a))/sum(a)^2 using enhanced mirror descent with

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def softmax_subgrad(c, a, temperature):
    """
    Compute subgradient using softmax over all peaks.
    Temperature controls how "peaky" the subgradient is.
    """
    n = len(a)
    # Softmax weights
    c_max = c.max()
    # Shift for numerical stability
    exp_c = np.exp((c - c_max) / max(temperature, 1e-10))
    w = exp_c / exp_c.sum()

    # = 2 * sum_k w[k] * a[k-m] for valid (k, m)

    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1 - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def topk_soft_subgrad(c, a, top_k, temperature):

    Hybrid: use top-k peaks with softmax weights among them.

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
