    """Build the sparse convolution matrix where (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    size = 2 * n - 1
    
    # For row k, the nonzeros are at columns max(0, k-n+1) to min(n-1, k)

    # Let's use the COO format
    
    # For efficiency, build row by row
    rows = np.empty(n*n + size, dtype=np.int32)  # approximate size
    cols = np.empty(n*n + size, dtype=np.int32)
    vals = np.empty(n*n + size, dtype=np.float64)

        m_start = max(0, k - (n-1))
        m_end = min(n-1, k)
        length = m_end - m_start + 1
        rows[pos:pos+length] = k
        cols[pos:pos+length] = np.arange(m_start, m_end+1)
        vals[pos:pos+length] = a[k-m_end:k-m_start+1][::-1]

        rows[pos] = k
        cols[pos] = n
        vals[pos] = -1.0
        pos += 1
    
    A = sp.csc_matrix((vals[:pos], (rows[:pos], cols[:pos])), shape=(size, n+1))
    return A
