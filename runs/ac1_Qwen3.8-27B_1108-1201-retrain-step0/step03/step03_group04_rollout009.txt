"""
Minimize peak of self-convolution over non-negative sequences with fixed sum.
Objective: 2*n*max(conv(a,a))/sum(a)^2, equivalently min max(conv(a,a)) with sum(a)=sqrt(2n).
Strategy: Multi-phase FISTA with backtracking line search, top-k subgradient
(k=1 focus), adaptive momentum, and fine polishing with random perturbations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.asarray(v, dtype=np.float64)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]


def compute_grad_weighted(a, weights):
    """
    Compute directional gradient: sum_k weights[k] * d(conv(a,a)[k])/da
    Using FFT cross-correlation.
    """
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def obj_value(a):
    """Compute the objective 2*n*max_conv/sum^2 (for reporting)."""
    c = fft_conv_self(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * len(a) * c.max() / (s * s)


def run_fista_bt(a_init, n, S, max_time, rng, top_k_init=1, top_k_max=50,
                 top_k_rate=100, beta_max=5000.0):
    """
    FISTA with backtracking line search on the actual objective.
    Uses top-k subgradient with adaptive k.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_c = fft_conv_self(a).max()
    best_a = a.copy()
    it = 0
    step = 2.0 * S / n  # Initial step size
    
    # Backtracking params
    armijo_c = 1e-4
    armijo_rho = 0.7
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // top_k_rate)
        
        c = fft_conv_self(y)
        cur_max = c.max()
        
        if cur_max < best_c - 1e-15:
            best_c = cur_max
            best_a = y.copy()
        
        # Compute top-k weights
        if top_k >= len(c):
            weights = np.ones(len(c)) / len(c)
        else:
            top_idx = np.argpartition(c, -top_k)[-top_k:]
            top_vals = c[top_idx]
            total = top_vals.sum()
            if total < 1e-30:
                weights = np.ones(len(c)) / len(c)
            else:
                weights = np.zeros(len(c))
                weights[top_idx] = top_vals / total
        
        grad = compute_grad_weighted(y, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Backtracking line search
        alpha = step
        improved = False
        for _ in range(10):
            x_trial = project_simplex(y - alpha * grad, S)
            c_trial = fft_conv_self(x_trial)
            if c_trial.max() < cur_max - armijo_c * alpha * gnorm + 1e-15:
                x_new = x_trial
                improved = True
                break
            alpha *= armijo_rho
        
        if not improved:
            # Use the last trial anyway (small step)
            x_new = project_simplex(y - alpha * grad, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        # y = x_new + ((t-1)/t_new) * (x_new - a)
        coef = (t - 1.0) / t_new
        y = x_new + coef * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Update step size
        if improved:
            step = min(alpha * 1.5, 4.0 * S / n)
        else:
            step = alpha * 0.5
        
        # Periodic perturbation
        if it % 2000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
    
    return best_a, best_c
