"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
Strategy: Multi-start FISTA warmup + exact subgradient optimization with
backtracking, adaptive top-k, and turbulent perturbations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv(a):
    """Full self-convolution via FFT. Returns length 2n-1 array."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def subgradient_toppk(a, c, top_k):
    """
    Exact subgradient of max_k (a*a)[k] using top-k active peaks.
    grad[m] = (2/k) * sum_{j in top_k} a[k_j - m] (valid indices only).
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    # For each peak k_j, the contribution to grad[m] is 2*a[k_j - m] if 0 <= k_j - m < n
    # grad[m] = (2/top_k) * sum_j a[k_j - m] (for valid k_j - m)
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        # a[k-m] is valid when 0 <= k-m < n, i.e., max(0, k-n+1) <= m <= min(k, n-1)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            # grad[m] += 2 * a[k-m] for m in [lo, hi]
            # k-m ranges from k-lo to k-hi
            # a[k-m] for m in [lo,hi] = a[k-lo : k-hi+1] (reversed)
            grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    grad /= top_k
    return grad


def subgradient_soft(a, c, beta):
    """Soft-max subgradient via FFT cross-correlation."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv(a):
    """Full self-convolution via FFT. Returns length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def subgrad_toppk(a, c, top_k):
    """Exact subgradient using top-k active peaks."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]

    for k in top_idx:

        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= float(top_k)
    return grad

        inits.append(("exp_b%d" % int(alpha), np.exp(alpha * (1.0 - tt))))

    for p in [0.5, 1.0, 2.0, 3.0]:
        inits.append(("pow_f%s" % p, (tt + 0.01) ** p))
        inits.append(("pow_b%s" % p, (1.0 - tt + 0.01) ** p))

    inits.append(("bimodal", np.exp(-6 * (tt - 0.3) ** 2) + np.exp(-6 * (tt - 0.7) ** 2) + 0.1))
    inits.append(("sin_t", 1.0 + 0.7 * np.sin(2 * np.pi * 3 * tt) + tt))
    inits.append(("sin_t2", 1.0 + 0.5 * np.cos(2 * np.pi * 7 * tt) + 0.5 * tt))
    inits.append(("lin", tt + 0.05))
    inits.append(("lin_rev", 1.0 - tt + 0.05))

    rand_a = rng.random(n)
    inits.append(("rand_ramp", rand_a * np.linspace(0.3, 2.0, n)))

    # Step functions
    for frac in [0.3, 0.5, 0.7]:
        step_arr = np.ones(n)
        step_arr[:int(n * frac)] = 3.0
        inits.append(("step%s" % frac, step_arr))

    n_inits = len(inits)
    usable_time = max(10.0, deadline - time.monotonic())

    # Phase 1: Quick FISTA exploration (25% of time)
    explore_time = usable_time * 0.25
    per_init = explore_time / n_inits
    print("Phase 1: %d inits, %.1f sec each, n=%d" % (n_inits, per_init, n))

    for name, arr in inits:
        if time.monotonic() > deadline - 60:
            break
        n_i = len(arr)
        if n_i < 64:
            continue
        S = np.sqrt(2.0 * n_i)
        a_opt, score = run_fista(arr, n_i, S, per_init, rng,
                                  beta_init=5.0, beta_max=150.0)
        a_norm = safe_normalize(a_opt, n_i)
        c = fft_conv(a_norm)
        obj = c.max()
        if obj < global_best:
            global_best = obj
            global_a = a_norm.copy()
            print("  [%s] obj=%.6f" % (name, obj))

    print("After Phase 1: best=%.6f" % global_best)

    # Phase 2: Intensive subgradient optimization (50% of time)
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        remaining = deadline - time.monotonic()

        if remaining > 10:
            # Subgradient with backtracking
            seg1 = remaining * 0.45
            print("Phase 2a: Subgradient optimization (%.1f sec)" % seg1)
            a_sg, score_sg = run_subgrad_opt(
                global_a, n_best, S_best, seg1, rng,
                top_k_init=1, top_k_max=10,
                noise_every=80, noise_scale_init=0.002)
            a_sg = safe_normalize(a_sg, n_best)
            c_sg = fft_conv(a_sg)
            if c_sg.max() < global_best:
                global_best = c_sg.max()
                global_a = a_sg.copy()
                print("  SubGrad: %.6f" % global_best)

            # FISTA high-beta polish
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                seg2 = remaining2 * 0.3
                print("Phase 2b: FISTA high-beta (%.1f sec)" % seg2)
                a_f, score_f = run_fista(global_a, n_best, S_best, seg2, rng,
                                          beta_init=30.0, beta_max=400.0,
                                          noise_scale=0.001)
                a_f = safe_normalize(a_f, n_best)
                c_f = fft_conv(a_f)
                if c_f.max() < global_best:
                    global_best = c_f.max()
                    global_a = a_f.copy()
                    print("  FISTA hb: %.6f" % global_best)

            # Another subgradient round (different params)
            remaining3 = deadline - time.monotonic()
            if remaining3 > 10:
                seg3 = remaining3 * 0.4
                print("Phase 2c: SubGrad round 2 (%.1f sec)" % seg3)
                a_s2, score_s2 = run_subgrad_opt(
                    global_a, n_best, S_best, seg3, rng,
                    top_k_init=2, top_k_max=8,
                    noise_every=120, noise_scale_init=0.001)
                a_s2 = safe_normalize(a_s2, n_best)
                c_s2 = fft_conv(a_s2)
                if c_s2.max() < global_best:
                    global_best = c_s2.max()
                    global_a = a_s2.copy()
                    print("  SubGrad2: %.6f" % global_best)

    # Phase 3: Final precision polish (25% of time)
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        remaining = deadline - time.monotonic()

        if remaining > 5:
            seg = remaining
            print("Phase 3: Final polish (%.1f sec)" % seg)

            # Very careful single-peak subgradient
            a_fin, score_fin = run_subgrad_opt(
                global_a, n_best, S_best, seg * 0.6, rng,
                top_k_init=1, top_k_max=3,
                noise_every=200, noise_scale_init=0.0005)
            a_fin = safe_normalize(a_fin, n_best)
            c_fin = fft_conv(a_fin)
            if c_fin.max() < global_best:
                global_best = c_fin.max()
                global_a = a_fin.copy()
                print("  Final subgrad: %.6f" % global_best)

            # Tiny FISTA polish
            remaining2 = deadline - time.monotonic()
            if remaining2 > 3:
                a_p, score_p = run_fista(global_a, n_best, S_best, remaining2, rng,
                                          beta_init=100.0, beta_max=800.0,
                                          noise_scale=0.0003)
                a_p = safe_normalize(a_p, n_best)
                c_p = fft_conv(a_p)
                if c_p.max() < global_best:
                    global_best = c_p.max()
                    global_a = a_p.copy()
                    print("  Final FISTA: %.6f" % global_best)

    # Verification
    if global_a is None:
        global_a = np.ones(8192, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Final: score=%.6f, length=%d" % (ver, len(final_list)))
    else:
        print("Verification FAILED, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]

    return final_list
