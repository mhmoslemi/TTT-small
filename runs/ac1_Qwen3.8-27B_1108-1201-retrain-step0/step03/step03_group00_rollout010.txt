import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), cm
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, cm


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=2000.0,
              beta_ramp_iters=2000, use_hybrid=True):
    """FISTA with adaptive LSE smoothing and optional hybrid subgradient."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: ramp up linearly then hold
        if it < beta_ramp_iters:
            beta = beta_init + (beta_max - beta_init) * (it / beta_ramp_iters)
        else:
            beta = beta_max

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Choose gradient
        if use_hybrid and it > 1000 and rng.random() < min(0.3, 0.0005 * it):
            grad = exact_subgradient(y, c)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = S / (gnorm * (1.0 + 0.01 * np.log1p(it / 200.0)))
            step = min(step, 0.5 / (beta * 0.05))
        else:
            grad, _ = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta)
            # Adaptive: reduce step if no improvement
            if it - last_improve > 1500:
                step *= 0.5

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic noise injection to escape local minima
        if it % 3000 == 0:
            decay = np.exp(-it / 30000.0)
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-7:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def spectral_flatten(a_init, n, S, max_time, rng, alpha=0.5, n_inner=50):
    """Gerchberg-Saxton style spectral flattening."""
    a = normalize_sum(a_init, n)
    fsize = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        a_pad = np.zeros(fsize)
        a_pad[:n] = a
        F = fft(a_pad)
        mag = np.abs(F)
        phase = F / np.maximum(mag, 1e-30)
        P = mag ** 2

        # Flatten the power spectrum
        P_mean = P.mean()
        if P_mean < 1e-30:
            break
        # Soft flattening: P_new = P^alpha / mean(P^alpha)
        P_flat = np.power(np.maximum(P, 1e-30), alpha)
        P_flat = P_flat / P_flat.mean()
        # Blend with original for stability
        P_new = (1.0 - alpha) * P / P_mean + alpha * P_flat

        # Reconstruct with new magnitude, same phase
        mag_new = np.sqrt(P_new)
        # Ensure total power is conserved (approximately)
        mag_new = mag_new * np.sqrt(P.sum() / P_new.sum())
        F_new = phase * mag_new
        a_new = np.real(ifft(F_new))[:n]

        # Project to non-negative with fixed sum
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a_new.copy()

        a = a_new

        # Adaptive alpha: increase flattening over time
        alpha = min(0.95, alpha + 0.002)

    return best_a, best


def coordinate_descent(a_init, n, S, max_time, rng):
    """Paired coordinate descent on the simplex."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    c = fft_convolve_self(a)
    if c.max() < best:
        best = c.max()
        best_a = a.copy()

    # Precompute indices for efficiency
    while time.monotonic() - start < max_time:
        it += 1
        improved = False

        # Try a set of (j, m) pairs: local neighbors and random
        n_try = min(50 + it // 10, 200)
        for _ in range(n_try):
            if time.monotonic() - start > max_time - 1:
                break

            # Choose pair: mix of local and random
            if rng.random() < 0.5:
                j = rng.integers(0, n)
                m = rng.integers(0, n)
            else:
                j = rng.integers(0, n - 2)
                m = j + rng.integers(1, min(5, n - 1))
                m = m % n

            if j == m or a[j] < 1e-10 or a[m] < 1e-10:
                continue

            # Compute d_k = change in c_k per unit delta (transfer from m to j)
            # c_k = sum_i a_i * a_{i+k}
            # dc_k/ddelta = a_{j+k} + a_{j-k} - a_{m+k} - a_{m-k} (with bounds)

            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                # Contribution from j
                jk = j + k
                jmk = j - k
                val_j = 0.0
                if 0 <= jk < n:
                    val_j += a[jk]
                if 0 <= jmk < n:
                    val_j += a[jmk]
                # Contribution from m
                mk = m + k
                mmk = m - k
                val_m = 0.0
                if 0 <= mk < n:
                    val_m += a[mk]
                if 0 <= mmk < n:
                    val_m += a[mmk]
                d[k] = val_j - val_m

            # c_k(delta) = c[k] + delta * d[k] (ignoring k=0 quadratic for now)
            # For k=0: c_0(delta) = c_0 + delta*(a_j + a_m - a_j - a_m) + ... 
            # Actually c_0 = sum a_i^2, change = (a_j+d)^2 - a_j^2 + (a_m-d)^2 - a_m^2
            # = 2a_j*d + d^2 - 2a_m*d + d^2 = 2(a_j-a_m)*d + 2d^2
            # So d[0] should be 2*(a_j - a_m) and we have a quadratic term 2*d^2

            # Find delta* minimizing max_k (c[k] + delta * d[k])
            # with delta in [-a[j], a[m]]
            lo = -a[j]
            hi = a[m]

            # Ternary search for 1D convex minimization
            def f_of_delta(dlt):
                vals = c + dlt * d
                vals[0] += 2.0 * dlt ** 2  # quadratic correction for k=0
                return vals.max()

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo
            b_g = hi
            c_g = b_g - gr * (b_g - a_g)
            d_g = a_g + gr * (b_g - a_g)
            fc = f_of_delta(c_g)
            fd = f_of_delta(d_g)

            for _ in range(40):
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - gr * (b_g - a_g)
                    fc = f_of_delta(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + gr * (b_g - a_g)
                    fd = f_of_delta(d_g)

            delta = (a_g + b_g) / 2.0
            delta = np.maximum(lo, np.minimum(hi, delta))

            # Apply if improvement
            a_trial = a.copy()
            a_trial[j] += delta
            a_trial[m] -= delta
            # Ensure non-negative
            if a_trial[j] < 0 or a_trial[m] < 0:
                continue

            c_trial = c + delta * d
            c_trial[0] += 2.0 * a[j] * delta + 2.0 * delta ** 2
            # More precisely recompute c_0
            c_trial[0] = c[0] + 2 * (a[j] - a[m]) * delta + 2 * delta ** 2

            cur_trial = c_trial.max()
            if cur_trial < best - 1e-12:
                a = a_trial
                c = c_trial
                best = cur_trial
                best_a = a.copy()
                improved = True

        # Full recompute periodically to avoid drift
        if it % 20 == 0:
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

        if not improved and it > 5:
            # No improvement in this pass, add small noise
            noise_scale = 0.001 * S / np.sqrt(n) * np.exp(-it / 100.0)
            if noise_scale < 1e-8:
                break
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c = fft_convolve_self(a)
            if c.max() < best:
                best = c.max()
                best_a = a.copy()

    return best_a, best

def make_palindromic(a, n):
    """Make a sequence palindromic."""
    half = a[:n // 2]
    if n % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        return np.concatenate([half, a[n // 2:n // 2 + 1], half[::-1]])


def generate_variants(a, n, rng):

    variants.append(a[::-1].copy())

    # Small random perturbations
    for scale in [0.01, 0.05, 0.1]:
        noise = rng.normal(0, scale * S_global / np.sqrt(n), n)

    a_hp = np.maximum(a_hp, 0)
    s = a_hp.sum()
    if s > 1e-15:

                                  beta_ramp_iters=1500, use_hybrid=True)
        update_best(a_opt, "fista_" + name)

    print("After Phase 1: best=%.6f" % global_best)

    # Phase 2: Spectral flattening on best
    print("\n=== Phase 2: Spectral Flattening ===")
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        S_global = S_best
        p2_budget = (deadline - time.monotonic()) * 0.20
        if p2_budget > 10:
            t_sf1 = p2_budget * 0.6
            t_sf2 = p2_budget * 0.4
            a_sf1, _ = spectral_flatten(global_a, n_best, S_best, t_sf1, rng, alpha=0.4)
            update_best(a_sf1, "spectral_v1")
            a_sf2, _ = spectral_flatten_v2(global_a, n_best, S_best, t_sf2, rng)
            update_best(a_sf2, "spectral_v2")

    print("After Phase 2: best=%.6f" % global_best)

    # Phase 3: Coordinate descent polish
    print("\n=== Phase 3: Coordinate Descent ===")
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        S_global = S_best
        p3_budget = (deadline - time.monotonic()) * 0.25
        if p3_budget > 10:
            a_cd, _ = coordinate_descent(global_a, n_best, S_best, p3_budget, rng)
            update_best(a_cd, "coord_descent")

    print("After Phase 3: best=%.6f" % global_best)

    # Phase 4: Final FISTA polish from best
    print("\n=== Phase 4: Final Polish ===")
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        S_global = S_best
        remaining = deadline - time.monotonic()
        if remaining > 15:
            t_final = remaining * 0.7
            a_fin, _ = run_fista(global_a, n_best, S_best, t_final, rng,
                                  beta_init=100.0, beta_max=5000.0,
                                  beta_ramp_iters=1000, use_hybrid=True)
            update_best(a_fin, "final_fista")

            # Another coordinate descent pass
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_cd2, _ = coordinate_descent(global_a, n_best, S_best, remaining2 * 0.8, rng)
                update_best(a_cd2, "final_cd")

    # Phase 5: Try a different length if time permits
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        print("\n=== Phase 5: Alternate length ===")
        # Try a slightly different length
        alt_n = n * 2 if n <= 8192 else n // 2
        if 2048 <= alt_n <= 65536:
            S_alt = np.sqrt(2.0 * alt_n)
            S_global = S_alt
            # Upsample or downsample the best sequence
            if alt_n > n:
                t_old = np.linspace(0, 1, n)
                t_new = np.linspace(0, 1, alt_n)
                alt_start = np.interp(t_new, t_old, global_a)
            else:
                nblocks = n // alt_n
                alt_start = global_a[:nblocks * alt_n].reshape(alt_n, nblocks).mean(axis=1)
            alt_start = normalize_sum(alt_start, alt_n)

            t_alt = remaining * 0.4
            a_alt, _ = run_fista(alt_start, alt_n, S_alt, t_alt, rng,
                                  beta_init=10.0, beta_max=1000.0)
            update_best(a_alt, "alt_n")

    # Final verification
    print("\n=== Final Verification ===")
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification score: %.6f" % ver if np.isfinite(ver) else "Verification: inf")

    if not np.isfinite(ver):
        print("  Fallback to uniform")
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning length=%d, final_score=%.6f, total_time=%.0fs" % (
        len(global_a), ver, time.monotonic() - t_start))
    return final_list
