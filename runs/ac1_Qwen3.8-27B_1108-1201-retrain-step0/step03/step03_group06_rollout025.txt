def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    # Solve: find tau such that sum(max(0, v[i] - tau)) = S
    # Use bisection on tau
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v)

    cond = sorted_v - cssv / rho > 0

        i_max = idx[-1] + 1  # 1-indexed
        tau = cssv[i_max - 1] / i_max
    return np.maximum(v - tau, 0.0)

def softmax_obj(c, T):

    c_shifted = T * c - T * c.max()

```python
def compute_gradient(c, a, T):
    n = len(a)
    # Compute weights
    Tc = T * c
    Tc_max = Tc.max()
    w = np.exp(Tc - Tc_max)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient using FFT
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    g = 2.0 * corr[:n]
    
    return g
```
