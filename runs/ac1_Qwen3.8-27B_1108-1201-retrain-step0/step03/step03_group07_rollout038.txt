```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_convolve_ab(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]

def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def build_conv_matrix(a, n):
    """
    Build the matrix C such that C @ g = f * g (convolution).
    C has shape (2n-1, n).
    C[k][j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                C[k, j] = a[idx]
    return C
```

```python
    n = len(a)
    size = 2 * n - 1

    c = np.real(ifft(A * A))
```

```python
def lp_optimize(f0, time_budget, verbose=True):
    """
    LP-based iterative optimization.
    """
    f = f0.copy().astype(np.float64)
    n = len(f)
    S = f.sum()
    f = f * (S / f.sum())  # ensure sum is S
    
    c = fft_convolve_self(f)
    M = c.max()
    best_obj = obj_value(f, c)
    best_f = f.copy()
    
    # Pre-compute sparsity structure
    rows, cols, f_idx = build_conv_structure(n)
    
    start = time.monotonic()
    iter_no = 0
    
    while time.monotonic() - start < time_budget:
        # Build the convolution matrix
        vals = f[f_idx]
        A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        
        # Solve LP: maximize sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                      bounds=[(0, None)]*n, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        
        if not res.success:
            if verbose:
                print(f"  LP failed at iter {iter_no}: {res.message}")
            break
        
        g0 = res.x
        g0_sum = g0.sum()
        
        if g0_sum < 1e-10:
            break
        
        # Normalize g0 to have sum S
        g = g0 * (S / g0_sum)
        
        # Backtracking line search
        best_t = 0.0
        best_new_obj = best_obj
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            f_trial = (1 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0)
            c_trial = fft_convolve_self(f_trial)
            obj_trial = obj_value(f_trial, c_trial)
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based convex combination method
combined with adaptive subgradient mirror descent and coordinate polishing.

Key innovation: At each LP iteration, solve maximize sum(g) s.t. (f*g)[k] <= M, g>=0
to find the optimal direction that respects the current peak budget. This is more
principled than subgradient methods as it directly characterizes the feasible
improvement region via linear programming.
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def build_conv_structure(n):
    """Pre-compute sparsity structure of convolution matrix (2n-1 x n)."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        count = j_end - j_start + 1
        rows_list.extend([k] * count)
        cols_list.extend(range(j_start, j_end + 1))
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    f_idx = (rows - cols).astype(np.int32)
    return rows, cols, f_idx


def make_conv_sparse(f, rows, cols, f_idx):
    """Create sparse convolution matrix for current f."""
    vals = f[f_idx]
    n = len(f)
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve_step(f, M, rows, cols, f_idx, time_limit_lp=30):
    """
    Solve LP: maximize sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None if LP fails.
    """
    n = len(f)
    A = make_conv_sparse(f, rows, cols, f_idx)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': time_limit_lp})

    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def mirror_descent_fast(a0, time_budget, eta0=0.3, momentum=0.85,
                        top_k=50, stagnation_limit=1000, verbose=False):
    """Fast mirror descent with subgradient for peak chasing."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)
    eta = eta0
    iter_no = 0

    while time.monotonic() - start < time_budget:
        # Find top peaks

        # Compute subgradient
        grad = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1):
                grad[m] += a[k - m]
        grad *= 2.0 / top_k_cur

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_n = grad / gnorm

        v = momentum * v + (1.0 - momentum) * grad_n

def coordinate_polish(a0, time_budget):
    """Coordinate descent: try reducing elements that contribute most to peaks."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()

    for sweep in range(2000):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        c_max = c.max()
        k_max = np.argmax(c)

        # Find elements contributing most to the peak
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        contrib = np.zeros(n)
        for m in range(lo, hi + 1):
            contrib[m] = a[k_max - m]

        # Sort by contribution (descending)
        order = np.argsort(contrib)[::-1]
        n_try = min(100, n)

        for i in order[:n_try]:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            for factor in [0.8, 0.9, 1.1, 1.2]:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_norm = a * (S / s)
                c_new = fft_convolve_self(a_norm)
                obj_new = obj_value(a_norm, c_new)
                if obj_new < best_obj - 1e-12:
                    best_obj = obj_new
                    best_a = a_norm.copy()
                    a = a_norm
                    c = c_new
                    improved = True
                    break
            a[i] = a[i]

        if not improved and sweep > 50:
            break

    return best_a, best_obj

        no_imp5 = 0

        while time.monotonic() - start5 < remaining_time - 3:
            rem5 = remaining_time - (time.monotonic() - start5)
            if rem5 < 5:
                break
            g0 = lp_solve_step(f, M * (1 + 1e-8), rows, cols, f_idx,
                               time_limit_lp=min(20, rem5 - 2))
            if g0 is None or g0.sum() < 1e-10:
                no_imp5 += 1
                if no_imp5 > 10:
                    break
                continue

            g = g0 * (S / g0.sum())
            best_t = 0.0
            for t in [0.1, 0.05, 0.02, 0.01, 0.005]:
                f_trial = (1.0 - t) * f + t * g
                f_trial = np.maximum(f_trial, 0.0)
                f_trial = f_trial * (S / f_trial.sum())
                c_trial = fft_convolve_self(f_trial)
                obj_trial = obj_value(f_trial, c_trial)
                if obj_trial < best_obj5 - 1e-13:
                    best_t = t
                    break

            if best_t > 0:
                f = (1.0 - best_t) * f + best_t * g
                f = np.maximum(f, 0.0)
                f = f * (S / f.sum())
                c = fft_convolve_self(f)
                M = c.max()
                no_imp5 = 0
            else:
                no_imp5 += 1
                if no_imp5 > 10:
                    break

            cur_obj = obj_value(f, c)
            if cur_obj < best_obj5:
                best_obj5 = cur_obj
                best_f5 = f.copy()

        if best_obj5 < global_best:
            global_best = best_obj5
            global_a = best_f5.copy()
            print(f"  Phase 5 result: {global_best:.6f}")

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: local_obj={global_best:.6f}, verified={ver:.6f}, n={len(final_a)}")

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.6f}")

    return final_a
```
