"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses multi-start FISTA with log-sum-exp smoothing, spectral initialization,
and pair-wise local search for fine-tuning.
"""

import numpy as np
import time
from scipy.fft import fft, ifft

# --- Core utilities ---

def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Binary search for the threshold
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        return v * (S / v.sum()) if v.sum() > 0 else np.ones(n) * S/n
    rho = np.max(np.where(valid)) + 1
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_conv(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2*n - 1
    M = 1 << (size-1).bit_length()
    A = fft(a, M)
    c = np.real(ifft(A*A))
    return c[:size]

def lse_grad(a, c, beta):
    """Gradient of smooth approximation of max(c) w.r.t. a."""
    n = len(a)
    c_shifted = c - c.max()
    exp_bc = np.exp(beta * c_shifted)

"""

Strategy: Multi-start FISTA with log-sum-exp smoothing, spectral and
anti-palindromic initializations, adaptive beta scheduling, and

Key innovations over previous: spectral initialization via controlled
magnitude spectrum, two-stage beta annealing, targeted pair-wise
local refinement, and adaptive time allocation.

import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    s = v.sum()
    if s <= 1e-30:
        return np.ones(n) * S / n
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cand = (cumsum - S) / idx
    valid = theta_cand <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    M = 1 << (size - 1).bit_length()
    A = fft(a, M)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c_k)) w.r.t. a.
    Uses the formula: grad = 2 * ifft(W * conj(A))[:n]
    where W = fft(weights), A = fft(a).
    """
    n = len(a)
    c_shifted = c - c.max()
    exp_bc = np.exp(beta * c_shifted)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    M = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, M)
    A = fft(a, M)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_subgradient(a, c, top_k):
    """Subgradient using uniform weights on top-k lags."""
    n = len(a)
    k = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / k
    M = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, M)
    A = fft(a, M)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_start=1.0, beta_end=500.0):
    """FISTA with adaptive beta (annealing from beta_start to beta_end)."""
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_max = np.inf
    best_a = a.copy()
    it = 0
    L = 10.0 / S  # Initial Lipschitz estimate
    stall = 0
    last_best = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        # Anneal beta logarithmically
        frac = min(1.0, it / (max_time * 20.0))
        beta = beta_start * (beta_end / beta_start) ** frac

        c = fft_conv(y)
        cur_max = c.max()
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = y.copy()
            stall = 0
        elif it % 100 == 0:
            if cur_max < last_best - 1e-12:
                last_best = cur_max
                stall = 0
            else:
                stall += 1

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Backtracking line search
        step = 1.0 / L
        x_new = None
        for bt in range(6):
            x_cand = project_simplex(y - step * grad, S)
            if bt == 0:
                x_new = x_cand
            else:
                c_cand = fft_conv(x_cand)
                if c_cand.max() <= cur_max + 1e-4:
                    x_new = x_cand
                    break
                step *= 0.5
        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        # Adaptive L
        c_x = fft_conv(x_new)
        if c_x.max() > cur_max * 1.01:
            L *= 1.3
        elif it % 50 == 0:
            L = max(L * 0.92, 2.0 / S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Random perturbation to escape local minima

            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

                for frac in [0.02, 0.05, 0.1]:
                    delta = frac * a[src]
                    if delta < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= delta
                    a_trial[dst] += delta
                    a_trial = np.maximum(a_trial, 0.0)
                    s_trial = a_trial.sum()
                    if s_trial < 1e-15:
                        continue
                    a_trial = a_trial * (S / s_trial)
                    c_trial = fft_conv(a_trial)
                    if c_trial.max() < cur_max - 1e-10:
                        a = a_trial
                        cur_max = c_trial.max()
                        improved = True
                        break

        if not improved and it % 100 == 0:
            # Add small noise to escape
            noise = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)

    return best_a, best_max


def make_anti_palinit(n, S, rng, n_variants=5):
    """Generate anti-palindromic initializations."""
    inits = []
    half = n // 2
    for v in range(n_variants):
        # Different distributions for the first half
        if v == 0:
            h = rng.random(half) * 1.5 + 0.1
        elif v == 1:
            h = np.sin(np.linspace(0, np.pi, half)) ** rng.uniform(1, 3) + 0.1
        elif v == 2:
            h = rng.exponential(1.0, half) + 0.1
        elif v == 3:
            t = np.linspace(0, 1, half)
            h = (t ** rng.uniform(0.5, 2.0)) * 2.0 + 0.1
        else:
            h = rng.normal(1.0, 0.5, half)
            h = np.maximum(h, 0.05)

        mu = S / half
        a = np.zeros(n)
        a[:half] = h
        for i in range(half):
            a[n - 1 - i] = mu - h[i]
        if n % 2 == 1:
            a[half] = mu / 2
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            a = a * (S / a.sum())
        else:
            a = np.ones(n) * S / n
        inits.append(a)
    return inits


def make_spectral_init(n, S, rng, n_variants=3):
    """Spectral initialization: controlled magnitude spectrum with random phases."""
    inits = []
    M = 1 << (2 * n - 1 - 1).bit_length()
    freqs = np.arange(M, dtype=np.float64)

    for v in range(n_variants):
        # Choose magnitude profile
        if v == 0:
            # Slow decay
            mag = 1.0 / (1.0 + freqs * 0.1)
        elif v == 1:
            # Faster decay (more concentrated at DC)
            mag = 1.0 / (1.0 + freqs * 0.5)
        else:
            # Gaussian in frequency
            sigma = n * 0.5
            mag = np.exp(-freqs ** 2 / (2 * sigma ** 2))

        mag[0] = 0  # Handle DC separately

        # Random phases
        phases = rng.uniform(0, 2 * np.pi, M)
        A = (mag * np.exp(1j * phases)).astype(np.complex128)
        A[0] = S  # DC component

        # Compute time-domain sequence
        a_full = np.real(ifft(A))[:n]
        a = np.maximum(a_full, 0.0)
        if a.sum() > 1e-10:
            a = a * (S / a.sum())
        else:
            a = np.ones(n) * S / n
        inits.append(a)
    return inits


def make_window_init(n, S, rng, window_type='hann'):
    """Window function initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    if window_type == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif window_type == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    elif window_type == 'flat_top':
        a = (0.21557895 - 0.41663158 * np.cos(2 * np.pi * t)
             + 0.277263158 * np.cos(4 * np.pi * t)
             - 0.083578947 * np.cos(6 * np.pi * t)
             + 0.006947368 * np.cos(8 * np.pi * t))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-6)
    # Add small random perturbation
    a += rng.normal(0, 0.01 * a.mean(), n)
    a = np.maximum(a, 1e-6)
    a = a * (S / a.sum())
    return a


def make_gaussian_init(n, S, rng, width=0.3):
    """Gaussian bump initialization."""
    center = n // 2
    t = np.arange(n, dtype=np.float64) - center
    a = np.exp(-t ** 2 / (2 * (width * n) ** 2))
    # Add slight asymmetry
    if rng.random() < 0.5:
        a *= (1.0 + 0.2 * t / (n / 2))
    a = np.maximum(a, 1e-8)
    a = a * (S / a.sum())
    return a


def make_upsampled_init(n, S, parent):
    """Upsample parent sequence to length n."""
    from numpy import interp
    n_parent = len(parent)
    t_old = np.linspace(0, 1, n_parent)
    t_new = np.linspace(0, 1, n)
    a = interp(t_new, t_old, parent)
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-10:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def propose_candidate(seed=42, budget_s=1800):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best_obj = np.inf
    global_a = None
    global_n = 0

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Parent available: %s (len=%d)" % (parent is not None, len(parent) if parent is not None else 0))

    def update_best(a, n):
        nonlocal global_best_obj, global_a, global_n
        S = np.sqrt(2.0 * n)
        c = fft_conv(a)
        max_b = c.max()
        obj = 2.0 * n * max_b / (S ** 2)
        if obj < global_best_obj:
            global_best_obj = obj
            global_a = a.copy()
            global_n = n
            print("  NEW BEST: obj=%.6f (n=%d, max_b=%.6f)" % (obj, n, max_b))

    # === Phase 1: Exploration with n=8192 ===
    n1 = 8192
    S1 = np.sqrt(2.0 * n1)
    print("=== Phase 1: Exploration n=%d ===" % n1)

    # Generate diverse initializations
    inits = []

    # Parent (upsampled if needed)
    if parent is not None and len(parent) >= 64:
        if len(parent) == n1:
            p = parent.copy()
            if p.sum() > 0:
                inits.append(("parent", p * (S1 / p.sum())))
        else:
            inits.append(("parent_interp", make_upsampled_init(n1, S1, parent)))

    # Anti-palindromic variants
    for i, a in enumerate(make_anti_palinit(n1, S1, rng, 5)):
        inits.append(("anti_pal_%d" % i, a))

    # Spectral variants
    for i, a in enumerate(make_spectral_init(n1, S1, rng, 3)):
        inits.append(("spectral_%d" % i, a))

    # Window functions
    for wt in ['hann', 'blackman', 'flat_top']:
        inits.append((("win_%s" % wt), make_window_init(n1, S1, rng, wt)))

    # Gaussian bumps
    for w in [0.2, 0.4, 0.6]:
        inits.append(("gauss_%.1f" % w, make_gaussian_init(n1, S1, rng, w)))

    # Random
    rand_a = rng.random(n1) + 0.05
    inits.append(("random", rand_a * (S1 / rand_a.sum())))

    # Symmetric random (a[i] = a[n-1-i])
    half = n1 // 2
    h = rng.random(half) + 0.1
    sym_a = np.zeros(n1)
    sym_a[:half] = h
    sym_a[half:] = h[::-1]
    inits.append(("sym_random", sym_a * (S1 / sym_a.sum())))

    n_inits = len(inits)
    usable_time = max(20.0, deadline - time.monotonic())
    per_init_phase1 = usable_time * 0.45 / n_inits

    print("Phase 1: %d inits, ~%.1f sec each" % (n_inits, per_init_phase1))

    results = []
    for name, arr in inits:
        if time.monotonic() > deadline - 180:
            break
        t0 = time.monotonic()
        # Short FISTA
        a_f, s_f = run_fista(arr, n1, S1, per_init_phase1, rng,
                             beta_start=1.0, beta_end=100.0)
        a_f = np.clip(a_f, 0, 1000)
        sf = a_f.sum()
        if sf > 0:
            a_f = a_f * (S1 / sf)
        c_f = fft_conv(a_f)
        obj_f = 2.0 * n1 * c_f.max() / (S1 ** 2)
        results.append((obj_f, name, a_f.copy()))
        update_best(a_f, n1)

        rem = per_init_phase1 - (time.monotonic() - t0)
        if rem > 3:
            a_sg, s_sg = run_subgradient(a_f, n1, S1, rem * 0.6, rng,
                                         topk_start=3, topk_end=32)
            a_sg = np.clip(a_sg, 0, 1000)
            sfg = a_sg.sum()
            if sfg > 0:
                a_sg = a_sg * (S1 / sfg)
            update_best(a_sg, n1)

    # Sort by objective
    results.sort(key=lambda x: x[0])
    print("Phase 1 done. Top 3: %s" % [(r[1], "%.6f" % r[0]) for r in results[:3]])

    # === Phase 2: Focused optimization on best candidates ===
    print("=== Phase 2: Focused optimization ===")
    remaining = deadline - time.monotonic()
    n_focus = min(3, len(results))
    per_focus = remaining * 0.30 / n_focus

    for i in range(n_focus):
        if time.monotonic() > deadline - 120:
            break
        _, name, a_start = results[i]
        print("  Focusing on %s (obj=%.6f)" % (name, results[i][0]))

        # Longer FISTA with higher beta range
        seg = per_focus * 0.6
        a_f2, _ = run_fista(a_start, n1, S1, seg, rng,
                           beta_start=10.0, beta_end=500.0)
        a_f2 = np.clip(a_f2, 0, 1000)
        sf2 = a_f2.sum()
        if sf2 > 0:
            a_f2 = a_f2 * (S1 / sf2)
        update_best(a_f2, n1)

        # Subgradient polish
        rem2 = per_focus - seg
        if rem2 > 3:
            a_sg2, _ = run_subgradient(a_f2, n1, S1, rem2 * 0.7, rng,
                                       topk_start=5, topk_end=50)
            a_sg2 = np.clip(a_sg2, 0, 1000)
            sfg = a_sg2.sum()
            if sfg > 0:
                a_sg2 = a_sg2 * (S1 / sfg)
            update_best(a_sg2, n1)

    # === Phase 3: Pair-wise local search on best ===
    if global_a is not None and time.monotonic() < deadline - 60:
        print("=== Phase 3: Pair-wise local search ===")
        n_best = global_n
        S_best = np.sqrt(2.0 * n_best)
        seg_pw = min(200.0, (deadline - time.monotonic()) * 0.25)
        a_pw, _ = pairwise_local_search(global_a, n_best, S_best, seg_pw, rng)
        a_pw = np.clip(a_pw, 0, 1000)
        spw = a_pw.sum()
        if spw > 0:
            a_pw = a_pw * (S_best / spw)
        update_best(a_pw, n_best)

    # === Phase 4: Try larger n ===
    if global_a is not None and time.monotonic() < deadline - 120:
        print("=== Phase 4: Try n=%d ===" % (n1 * 2))
        n2 = n1 * 2  # 16384
        S2 = np.sqrt(2.0 * n2)
        a_long = make_upsampled_init(n2, S2, global_a)
        seg_long = min(300.0, (deadline - time.monotonic()) * 0.30)

        a_l, _ = run_fista(a_long, n2, S2, seg_long * 0.6, rng,
                          beta_start=5.0, beta_end=200.0)
        a_l = np.clip(a_l, 0, 1000)
        sl = a_l.sum()
        if sl > 0:
            a_l = a_l * (S2 / sl)
        update_best(a_l, n2)

        rem_l = seg_long * 0.4
        if rem_l > 5:
            a_l2, _ = run_subgradient(a_l, n2, S2, rem_l, rng,
                                      topk_start=5, topk_end=64)
            a_l2 = np.clip(a_l2, 0, 1000)
            sl2 = a_l2.sum()
            if sl2 > 0:
                a_l2 = a_l2 * (S2 / sl2)
            update_best(a_l2, n2)

    # === Phase 5: Final polish ===
    if global_a is not None and time.monotonic() < deadline - 20:
        print("=== Phase 5: Final polish ===")
        n_f = global_n
        S_f = np.sqrt(2.0 * n_f)
        rem_f = max(5.0, (deadline - time.monotonic()) * 0.4)

        # High-beta FISTA
        seg5a = rem_f * 0.5
        a_p, _ = run_fista(global_a, n_f, S_f, seg5a, rng,
                          beta_start=100.0, beta_end=1000.0)
        a_p = np.clip(a_p, 0, 1000)
        sp = a_p.sum()
        if sp > 0:
            a_p = a_p * (S_f / sp)
        update_best(a_p, n_f)

        # Quick local search
        rem5b = rem_f * 0.4
        if rem5b > 3 and global_a is not None:
            a_ps, _ = pairwise_local_search(global_a, n_f, S_f, rem5b, rng)
            a_ps = np.clip(a_ps, 0, 1000)
            sps = a_ps.sum()
            if sps > 0:
                a_ps = a_ps * (S_f / sps)
            update_best(a_ps, n_f)

    # === Verification ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Final verification: %.6f (n=%d) ===" %
          (ver if np.isfinite(ver) else float('inf'), len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: rejected, using fallback")
        final_list = [1.0] * 1024
        ver = evaluate_sequence(final_list)
        print("  Fallback verification: %.6f" % ver)

    return final_list
