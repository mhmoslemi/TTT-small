import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def objective_direct(a):
    """Compute 2n*sum(a^2)/sum(a)^2 directly."""
    S = a.sum()
    if S < 1e-15:
        return np.inf
    return 2.0 * len(a) * np.sum(a ** 2) / (S * S)


def gradient_direct(a):
    """Gradient of 2n*Q/S^2 where Q=sum(a^2), S=sum(a)."""
    S = a.sum()
    Q = np.sum(a ** 2)
    n = len(a)
    # d(2nQ/S^2)/da_i = 2n*(2a_i*S^2 - 2S*Q)/S^4 = 4n*(a_i*S - Q)/S^3
    return 4.0 * n * (a * S - Q) / (S ** 3)


def optimize_direct(a_init, max_time, rng, lr_init=1.0, verbose=True):

    Minimize 2n*sum(a^2)/sum(a)^2 via projected gradient descent.
    Since objective is scale-invariant, fix sum=1 and minimize sum(a^2).
    """

        a = np.ones(len(a)) / len(a)

        a = a / S  # normalize to sum = 1

    best_obj = objective_direct(a)
    best_a = a.copy()

    stall = 0

        grad = 2.0 * a  # gradient of sum(a^2) on the simplex (fixed sum=1)

    # Normalize given
    S = given.sum()
    if S > 0.01:
        given_norm = given / S
    else:
        given_norm = np.ones(len(given)) / len(given)
    try_sequence(given_norm, "given_norm")

    n_given = len(given)

    print(f"\n=== Phase 1: Direct gradient optimization ({(deadline - time.monotonic()):.0f}s budget) ===")
    remaining = deadline - time.monotonic()

    # 1a: Direct optimization from given
    if remaining > 30:
        t1 = min(remaining * 0.3, 300)
        print(f"  Direct GD from given: {t1:.0f}s")
        a_opt, obj_opt = optimize_direct(given, t1, rng, lr_init=2.0)
        try_sequence(a_opt, "direct_gd_given")

    remaining = deadline - time.monotonic()

    # 1b: Direct optimization from uniform (should stay uniform)
    if remaining > 20:
        t1 = min(remaining * 0.1, 60)
        print(f"  Direct GD from uniform: {t1:.0f}s")
        a_opt, obj_opt = optimize_direct(np.ones(n_given) / n_given, t1, rng)
        try_sequence(a_opt, "direct_gd_unif")

    remaining = deadline - time.monotonic()

    # 1c: Mirror descent from given
    if remaining > 20:
        t1 = min(remaining * 0.15, 180)
        print(f"  Mirror descent from given: {t1:.0f}s")
        a_opt, obj_opt = optimize_mirror_descent(given, t1, rng)
        try_sequence(a_opt, "mirror_given")

    remaining = deadline - time.monotonic()

    # 1d: LP-inspired (convergence to uniform)
    if remaining > 10:
        t1 = min(remaining * 0.05, 30)
        print(f"  LP-inspired: {t1:.0f}s")
        a_opt, obj_opt = optimize_lp_inspired(given, t1, rng)
        try_sequence(a_opt, "lp_inspired")

    remaining = deadline - time.monotonic()

    print(f"\n=== Phase 2: FFT subgradient verification ({remaining:.0f}s) ===")
    if remaining > 30:
        t2 = min(remaining * 0.4, 300)
        print(f"  FFT subgradient from given: {t2:.0f}s")
        a_opt, obj_opt = optimize_fft_subgradient(given, t2, rng)
        try_sequence(a_opt, "fft_subgrad_given")

    remaining = deadline - time.monotonic()

    print(f"\n=== Phase 3: Exploration ({remaining:.0f}s) ===")
    if remaining > 30:
        # Try different structured patterns
        patterns = []

        # Triangular
        tri = np.abs(np.linspace(-1, 1, n_given))
        patterns.append((tri, "triangular"))

        # Gaussian
        x = np.linspace(-3, 3, n_given)
        gauss = np.exp(-x ** 2 / 2)
        patterns.append((gauss, "gaussian"))

        # Sinc-like
        sinc_x = np.linspace(-5, 5, n_given)
        sinc_pat = np.sinc(sinc_x / np.pi)
        sinc_pat = np.abs(sinc_pat)
        patterns.append((sinc_pat, "sinc"))

        # Random
        patterns.append((rng.dirichlet(np.ones(n_given)), "random_dirichlet"))

        # Bimodal
        bimodal = np.exp(-((np.linspace(0, 1, n_given) - 0.25) ** 2) / 0.01) + \
                  np.exp(-((np.linspace(0, 1, n_given) - 0.75) ** 2) / 0.01)
        patterns.append((bimodal, "bimodal"))

        # Exponential decay
        exp_pat = np.exp(-np.linspace(0, 5, n_given))
        patterns.append((exp_pat, "exp_decay"))

        for pat, name in patterns:
            try_sequence(pat, name)

        # Optimize from best pattern
        remaining = deadline - time.monotonic()
        if remaining > 30:
            per_pat = remaining * 0.5 / len(patterns)
            for pat, name in patterns:
                if time.monotonic() > deadline - 20:
                    break
                t_pat = min(per_pat, 60)
                a_opt, obj_opt = optimize_direct(pat, t_pat, rng)
                try_sequence(a_opt, f"opt_{name}")

    remaining = deadline - time.monotonic()

    print(f"\n=== Phase 4: Multistart refinement ({remaining:.0f}s) ===")
    if remaining > 20 and best_seq is not None:
        best_arr = np.asarray(best_seq, dtype=np.float64)
        n_best = len(best_arr)
        S_best = best_arr.sum()

        # Multiple perturbed restarts
        n_restarts = min(8, int(remaining / 30))
        per_restart = remaining * 0.6 / max(n_restarts, 1)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            # Different perturbation strategies
            r = rng.random()
            if r < 0.25:
                perturbed = best_arr * (1.0 + rng.normal(0, 0.05, n_best))
            elif r < 0.5:
                noise = rng.normal(0, 0.02 / np.sqrt(n_best), n_best)
                perturbed = best_arr + noise
            elif r < 0.75:
                # Smooth perturbation
                smooth_noise = np.convolve(
                    rng.normal(0, 1, n_best), np.ones(128) / 128, mode='same')
                perturbed = best_arr + 0.01 * smooth_noise
            else:
                # Mix with uniform
                mix = rng.uniform(0.05, 0.3)
                perturbed = (1 - mix) * best_arr + mix * np.ones(n_best)

            perturbed = np.maximum(perturbed, 0)
            sp = perturbed.sum()
            if sp > 0.01:
                perturbed = perturbed / sp

            a_opt, obj_opt = optimize_direct(perturbed, per_restart, rng)
            try_sequence(a_opt, f"restart_{i}")

    remaining = deadline - time.monotonic()

    print(f"\n=== Phase 5: Final polish ({remaining:.0f}s) ===")
    if remaining > 10 and best_seq is not None:
        best_arr = np.asarray(best_seq, dtype=np.float64)
        # Very fine optimization
        a_final, _ = optimize_direct(best_arr, remaining * 0.7, rng, lr_init=0.5)
        try_sequence(a_final, "final_polish")

        remaining2 = deadline - time.monotonic()
        if remaining2 > 5:
            a_final2, _ = optimize_mirror_descent(best_arr, remaining2, rng, eta_init=0.05)
            try_sequence(a_final2, "final_mirror")

    # Final verification
    print(f"\n=== FINAL RESULT ===")
    if best_seq is not None:
        print(f"Best score: {best_score:.6f}")
        print(f"Sequence length: {len(best_seq)}")
        final_check = evaluate_sequence(best_seq)
        print(f"Verifier confirms: {final_check:.6f}")
        if not np.isfinite(final_check) or final_check != best_score:
            print("Warning: verifier mismatch! Using uniform fallback.")
            best_seq = [1.0] * 8192
            best_score = evaluate_sequence(best_seq)
            print(f"Fallback uniform score: {best_score:.6f}")
    else:
        best_seq = [1.0] * 8192
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback uniform score: {best_score:.6f}")

    return best_seq
