import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_toppk(a, c, top_k):
    """Gradient using top-k active lags, computed via FFT cross-correlation."""
    n = len(a)
    if top_k > len(c):
        top_k = len(c)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_shaping_excess(a, N, top_k, alpha, target_max, rng=None):
    """
    One step of excess-based spectral shaping.
    Damps lags that exceed target_max more aggressively.
    Returns (new_a, c, c_max).
    """
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    if c_max <= target_max:
        return a, c, c_max

    # Compute excess: how much each lag exceeds the target
    excess = np.maximum(c - target_max, 0.0)
    # Weight by excess (lags further above target get more damping)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_excess = excess[top_idx]
    total_excess = top_excess.sum()
    if total_excess < 1e-30:
        return a, c, c_max

    weights = top_excess / total_excess

    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    # Adaptive alpha based on how far above target we are
    ratio = c_max / max(target_max, 1e-10)
    alpha_eff = alpha * min(2.0, ratio)

    P_new = np.maximum(P * (1.0 - alpha_eff * cos_eff), 0.0)

    # Preserve DC component
    P_new[0] = P[0]
    # Preserve Hermitian symmetry
    P_new[1:] = 0.5 * (P_new[1:] + P_new[1:][::-1].copy())

    phase = np.angle(A)
    A_new = np.sqrt(np.maximum(P_new, 0.0)) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]

    a_new = project_simplex(a_new_full, S)
    return a_new, c, c_max

def nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=512):
    """Nesterov-accelerated top-k subgradient for polishing."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    last_improve = time.monotonic()

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 40)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            last_improve = time.monotonic()

        grad = fft_grad_toppk(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
                last_improve = time.monotonic()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Restart if stalled
        if time.monotonic() - last_improve > 30 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t = 1.0
            last_improve = time.monotonic()

    return best_local, best_score


def swap_random_walk(a, S, n_steps, rng, n=0):
    """Random walk using swap and split moves to explore nearby solutions."""
    n = len(a)
    best_a = a.copy()
    best_score = np.inf

    for _ in range(n_steps):
        temp = a.copy()
        move_type = rng.random()
        if move_type < 0.4:
            # Swap: move mass from i to j
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                amt = rng.uniform(0.01, 0.1) * S * 0.01
                temp[i] -= amt
                temp[j] += amt
        elif move_type < 0.7:
            # Split: take from i, distribute to j and k
            i = rng.integers(n)
            j = rng.integers(n)
            k = rng.integers(n)
            if i != j and i != k:
                amt = rng.uniform(0.005, 0.05) * S * 0.01
                temp[i] -= amt
                temp[j] += amt * 0.5
                temp[k] += amt * 0.5
        else:
            # Small random perturbation
            temp += rng.normal(0, 0.002 * S / np.sqrt(n), n)

        temp = np.maximum(temp, 0.0)
        ts = temp.sum()
        if ts < 1e-15:
            continue
        temp = temp * (S / ts)

        c = fft_convolve_self(temp)
        if c.max() < best_score:
            best_score = c.max()
            best_a = temp.copy()

        a = temp

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_diverse_init(n, rng, init_type="random"):
    """Create diverse initializations."""
    tt = np.linspace(0, 1, n, endpoint=False)
    if init_type == "random":
        return rng.random(n) + 0.05
    elif init_type == "chirp":
        return 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)
    elif init_type == "sin_multi":
        return 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt) + 0.1 * np.sin(2 * np.pi * 13 * tt)
    elif init_type == "bump":
        return np.exp(-0.5 * ((tt - 0.5) * 4.0) ** 2) + 0.1
    elif init_type == "triband":
        # Three bands of different heights
        a = np.ones(n)
        a[:n//3] = 1.0
        a[n//3:2*n//3] = 0.5
        a[2*n//3:] = 0.8
        return a + 0.05 * rng.standard_normal(n)
    elif init_type == "alternating":
        a = np.ones(n)
        a[::2] = 1.0
        a[1::2] = 0.6
        return a + 0.05 * rng.standard_normal(n)
    elif init_type == "log_chirp":
        return 0.5 + 0.3 * np.sin(2 * np.pi * tt * np.exp(3 * tt)) + 0.1 * rng.standard_normal(n)
    else:
        return rng.random(n) + 0.05


def optimize_single_length(n, S, N, initial_a, time_budget, rng, target=1.45):
    """Optimize a single length. Returns (best_a, best_score)."""
    start = time.monotonic()
    a = safe_normalize(initial_a, n)
    best_a = a.copy()
    best_score = np.inf

    c_initial = fft_convolve_self(a)
    best_score = c_initial.max()
    best_a = a.copy()

    # Stage 1: Spectral shaping (exploration)
    t1 = time.monotonic()
    ss_budget = min(time_budget * 0.3, 600.0)
    if time_budget - (t1 - start) > ss_budget + 10:
        n_iters = min(8000, int(ss_budget / 0.003))
        a_ss, score_ss = spectral_shaping_run(a, N, n_iters, rng, target=target,
                                              alpha_init=0.35, top_k_init=5,
                                              top_k_max=64, alpha_decay=0.9995)
        a_ss = safe_normalize(a_ss, n)
        c_ss = fft_convolve_self(a_ss)
        if c_ss.max() < best_score:
            best_score = c_ss.max()
            best_a = a_ss.copy()
            a = a_ss.copy()

    # Stage 2: Nesterov polish
    t2 = time.monotonic()
    ng_budget = min(time_budget * 0.35, 600.0)
    elapsed = t2 - start
    if time_budget - elapsed > ng_budget + 10:
        a_ng, score_ng = nesterov_toppk(a, n, S, ng_budget, rng,
                                        top_k_init=10, top_k_max=256)
        a_ng = safe_normalize(a_ng, n)
        c_ng = fft_convolve_self(a_ng)
        if c_ng.max() < best_score:
            best_score = c_ng.max()
            best_a = a_ng.copy()

    # Stage 3: Swap random walk for diversity
    t3 = time.monotonic()
    elapsed = t3 - start
    if time_budget - elapsed > 30:
        walk_budget = min(time_budget - elapsed - 10, 60.0)
        n_walk_steps = int(walk_budget / 0.005)  # ~5ms per step
        n_walk_steps = min(n_walk_steps, 20000)
        a_walk, score_walk = swap_random_walk(best_a.copy(), S, n_walk_steps, rng)
        a_walk = safe_normalize(a_walk, n)
        c_walk = fft_convolve_self(a_walk)
        if c_walk.max() < best_score:
            best_score = c_walk.max()
            best_a = a_walk.copy()

    # Stage 4: Final fine polish
    t4 = time.monotonic()
    remaining = time_budget - (t4 - start)
    if remaining > 10:
        a_final, score_final = nesterov_toppk(best_a, n, S, remaining - 5, rng,
                                              top_k_init=50, top_k_max=512)
        a_final = safe_normalize(a_final, n)
        c_final = fft_convolve_self(a_final)
        if c_final.max() < best_score:
            best_score = c_final.max()
            best_a = a_final.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses multi-length exploration with adaptive spectral shaping and
    Nesterov-accelerated projected gradient with diverse restarts.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Try to use the provided sequence
    provided_seq = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            provided_seq = np.clip(given, 0, 1000).astype(np.float64)
    except Exception:
        pass

    # List of lengths to try, prioritizing those close to provided length
    lengths_to_try = [8192, 4096, 16384]
    if provided_seq is not None:
        prop_len = len(provided_seq)
        # Put the provided length first
        lengths_to_try = [prop_len] + [l for l in lengths_to_try if l != prop_len]

    print("Trying lengths: %s" % lengths_to_try)
    if provided_seq is not None:
        print("Provided sequence length: %d" % len(provided_seq))

    # Allocate time: give more time to the first (provided) length
    n_lengths = len(lengths_to_try)
    # Time allocation: 50% to first, 30% to second, 20% to third
    if n_lengths >= 3:
        time_fracs = [0.5, 0.3, 0.2]
    elif n_lengths == 2:
        time_fracs = [0.6, 0.4]
    else:
        time_fracs = [1.0]

    for li, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - 30:
            break

        S = np.sqrt(2.0 * float(n))
        N = 1 << (2 * n - 1).bit_length()
        time_budget = (deadline - time.monotonic()) * time_fracs[li]
        time_budget = min(time_budget, (deadline - time.monotonic()))

        print("\n=== Length n=%d, S=%.4f, FFT_N=%d, time_budget=%.1fs ===" % (n, S, N, time_budget))

        if time_budget < 30:
            break

        # Multiple initializations for this length
        n_inits = 4 if time_budget > 100 else 2
        init_types = ["random", "chirp", "sin_multi", "bump", "triband", "alternating", "log_chirp"]
        chosen_inits = init_types[:n_inits]

        # If we have the provided sequence and it matches this length, use it as first init
        if provided_seq is not None and len(provided_seq) == n:
            chosen_inits = ["provided"] + chosen_inits[:n_inits-1]

        per_init_time = time_budget / (n_inits + 1)  # +1 for final combination

        for init_type in chosen_inits:
            if time.monotonic() > deadline - 60:
                break

            if init_type == "provided":
                initial_a = provided_seq.copy()
            else:
                initial_a = make_diverse_init(n, rng, init_type)

            print("  Init: %s" % init_type)
            a_opt, score_opt = optimize_single_length(n, S, N, initial_a, per_init_time, rng)

            # Convert to actual objective value
            c = fft_convolve_self(a_opt)
            actual_obj = 2.0 * n * c.max() / (a_opt.sum() ** 2)
            print("  Score (max_conv): %.6f, Objective: %.6f" % (score_opt, actual_obj))

            if score_opt < global_best:
                global_best = score_opt
                global_a = a_opt.copy()
                print("  New global best: %.6f (obj=%.6f)" % (score_opt, actual_obj))

        # Final combined polish on the best for this length
        if global_a is not None and time.monotonic() < deadline - 60:
            remaining = min(deadline - time.monotonic(), time_budget * 0.15)
            if remaining > 15:
                print("  Final polish for n=%d..." % n)
                a_p, score_p = nesterov_toppk(global_a, n, S, remaining, rng,
                                              top_k_init=20, top_k_max=200)
                a_p = safe_normalize(a_p, n)
                c_p = fft_convolve_self(a_p)
                if c_p.max() < global_best:
                    global_best = c_p.max()
                    global_a = a_p.copy()
                    print("  Polished: %.6f" % global_best)

    # Final verification and safety checks
    if global_a is None:
        # Fallback: uniform sequence
        n_fb = 256
        global_a = np.ones(n_fb, dtype=np.float64)
        global_best = 2.0
        print("Using fallback uniform sequence of length %d" % n_fb)

    # Ensure validity
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    # Final objective check
    n_final = len(global_a)
    c_final = fft_convolve_self(global_a)
    final_obj = 2.0 * n_final * c_final.max() / (global_a.sum() ** 2)
    print("\n=== Final: n=%d, max_conv=%.6f, objective=%.6f ===" % (n_final, c_final.max(), final_obj))

    # Verify with evaluate_sequence
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("evaluate_sequence verification: %.6f" % ver if np.isfinite(ver) else "evaluate_sequence: inf (INVALID!)")

    if not np.isfinite(ver):
        print("  Fallback to simple sequence")
        return [1.0] * 256

    return final_list
