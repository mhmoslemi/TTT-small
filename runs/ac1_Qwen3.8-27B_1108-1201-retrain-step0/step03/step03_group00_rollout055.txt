    size = 2 * n - 1

    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def convolve_gradient(a, w):
    """Gradient of sum_k w[k] * (a*a)[k] with respect to a."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

"""
Minimize 2*n*max(self-convolution)/sum^2 using adaptive subgradient
descent with frequency-domain operations, spectral flattening,
Polyak step sizes, and multi-start initialization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def gradient_weighted_conv(a, w, fsize=None):
    """Gradient of sum_k w[k]*(a*a)[k] w.r.t. a."""
    n = len(a)
    size = 2 * n - 1
    if fsize is None:
        fsize = 1 << (size - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def projection_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.array(v, dtype=np.float64, copy=True)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_conv_self(a)
    return float(2.0 * n * c.max() / (S * S))

import numpy as np

from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def gradient_weighted(a, w, A_cached, fsize):
    """Gradient of sum_k w[k]*(a*a)[k] w.r.t. a, using cached A=FFT(a)."""
    W = fft(w, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A_cached)))[:len(a)]
    return grad


def proj_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection-free method."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - S)
    if not np.any(cond):
        return np.full(n, S / n)
    rho = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_max_conv(a, fsize):
    """Compute max of self-convolution using FFT."""
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    n = len(a)
    return c[:2 * n - 1].max(), A


def spectral_flatten(a, S, fsize, alpha=0.3):
    """Reduce spectral peaks of A to flatten convolution."""
    A = fft(a, fsize)
    mag = np.abs(A)
    target = S / np.sqrt(fsize)
    ratio = mag / (target + 1e-30)
    excess = np.maximum(ratio - 1.0, 0.0)
    if excess.max() < 0.05:
        return a
    scale = 1.0 / np.sqrt(1.0 + alpha * excess * excess)
    A = A * scale
    A[0] = S
    a_new = np.real(ifft(A))[:len(a)]
    return proj_simplex(a_new, S)

def coord_descent_polish(a_init, n, S, max_time, rng):
    """Coordinate descent: swap pairs of elements to reduce max convolution."""
    a = np.maximum(a_init, 0.0).copy()
    a = proj_simplex(a, S)
    fsize = 1 << (2 * n - 2).bit_length()
    best_score = 2.0 * n * fft_conv_self(a).max() / (S * S)
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Pick random pair of indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Try moving delta from j to i (or vice versa)
        # a_i' = a_i + d, a_j' = a_j - d (or swap)
        # Try several delta values
        ai, aj = a[i], a[j]
        candidates = [
            0.1 * (ai - aj), 0.5 * (ai - aj), ai - aj,
            0.1 * (aj - ai), 0.5 * (aj - ai), aj - ai,
            0.01 * (ai + aj), -0.01 * (ai + aj),
            0.05 * (ai + aj), -0.05 * (ai + aj),
        ]

        best_delta = 0.0
        best_M = None
        A_cur = fft(a, fsize)
        M_cur = np.real(ifft(A_cur * A_cur))[:2 * n - 1].max()

        for d in candidates:
            if ai + d < 0 or aj - d < 0:
                continue
            a[i] = ai + d
            a[j] = aj - d
            A_new = fft(a, fsize)
            M_new = np.real(ifft(A_new * A_new))[:2 * n - 1].max()
            if best_M is None or M_new < best_M:
                best_M = M_new
                best_delta = d
            a[i] = ai
            a[j] = aj

        if best_M is not None and best_M < M_cur:
            a[i] = ai + best_delta
            a[j] = aj - best_delta
            A_cur = fft(a, fsize)
            score = 2.0 * n * best_M / (S * S)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 5000:
            break

    return best_a, best_score

def downsample_upsample(a, n_small):
    """Downsample by averaging, then upsample by interpolation."""
    n = len(a)
    if n_small >= n:
        return a.copy()
    nblocks = n // n_small
    a_small = a[:nblocks * n_small].reshape(n_small, nblocks).mean(axis=1)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n)
    return np.interp(t_new, t_old, a_small)

        arr_n = len(arr)
        S_i = np.sqrt(2.0 * arr_n)
        arr = proj_simplex(np.maximum(arr, 0.0), S_i)

        a_opt, sc = run_subgradient(arr, arr_n, S_i, time_per, rng,
                                     K_init=50, K_min=3,
                                     flatten_every=200, restart_every=3000)
        sc = 2.0 * arr_n * fft_conv_self(a_opt).max() / (S_i * S_i)
        if sc < global_best:
            global_best = sc
            global_a = a_opt.copy()
            print("  [%s] n=%d: %.6f" % (name, arr_n, global_best))

    print("After Phase 1: best=%.6f (%.0fs elapsed)" % (global_best, time.monotonic() - p1_start))

    # Phase 2: Intensive optimization on best (50% of time)
    remaining = deadline - time.monotonic()
    phase2_time = remaining * 0.50
    print("=== Phase 2: Intensive optimization (%.0fs) ===" % phase2_time)

    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)

        # Run with different parameter settings for diversity
        settings = [
            dict(K_init=100, K_min=2, flatten_every=200, restart_every=3000),
            dict(K_init=30, K_min=2, flatten_every=500, restart_every=5000),
            dict(K_init=60, K_min=4, flatten_every=300, restart_every=4000),
        ]
        time_per_setting = phase2_time / len(settings)

        for si, st in enumerate(settings):
            if time.monotonic() > deadline - remaining * 0.30:
                break
            a_opt, sc = run_subgradient(global_a, n_best, S_best, time_per_setting,
                                        rng, **st)
            sc = 2.0 * n_best * fft_conv_self(a_opt).max() / (S_best * S_best)
            if sc < global_best:
                global_best = sc
                global_a = a_opt.copy()
                print("  Setting %d: %.6f" % (si, global_best))

    # Phase 3: Coordinate descent polish + final FISTA (25% of time)
    remaining = deadline - time.monotonic()
    phase3_time = remaining * 0.40
    print("=== Phase 3: Polishing (%.0fs) ===" % phase3_time)

    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)

        # Coordinate descent
        cd_time = phase3_time * 0.5
        if cd_time > 10:
            a_cd, sc_cd = coord_descent_polish(global_a, n_best, S_best, cd_time, rng)
            sc_cd = 2.0 * n_best * fft_conv_self(a_cd).max() / (S_best * S_best)
            if sc_cd < global_best:
                global_best = sc_cd
                global_a = a_cd.copy()
                print("  After coord descent: %.6f" % global_best)

        # Final intensive subgradient
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a_fin, sc_fin = run_subgradient(global_a, n_best, S_best, remaining2 * 0.8,
                                            rng, K_init=20, K_min=1,
                                            flatten_every=100, restart_every=2000)
            sc_fin = 2.0 * n_best * fft_conv_self(a_fin).max() / (S_best * S_best)
            if sc_fin < global_best:
                global_best = sc_fin
                global_a = a_fin.copy()
                print("  After final subgradient: %.6f" % global_best)

    # Phase 4: Try different n if time remains
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n_try = main_n * 2 if main_n <= 8192 else main_n // 2
        if 1024 <= n_try <= 65536:
            S_try = np.sqrt(2.0 * n_try)
            print("=== Phase 4: Trying n=%d (%.0fs) ===" % (n_try, remaining))
            # Upsample or downsample the best solution
            if n_try > len(global_a):
                t_old = np.linspace(0, 1, len(global_a))
                t_new = np.linspace(0, 1, n_try)
                a_try = np.interp(t_new, t_old, global_a)
            else:
                a_try = downsample_upsample(global_a, n_try)
            a_try = proj_simplex(a_try, S_try)
            a_opt, sc = run_subgradient(a_try, n_try, S_try, remaining * 0.7,
                                         rng, K_init=50, K_min=3,
                                         flatten_every=200, restart_every=3000)
            sc = 2.0 * n_try * fft_conv_self(a_opt).max() / (S_try * S_try)
            if sc < global_best:
                global_best = sc
                global_a = a_opt.copy()
                print("  n=%d: %.6f" % (n_try, global_best))

    # Final verification and cleanup
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)

    # Verify with the actual evaluator
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Final verification: %.6f (target: 1.5027)" % ver if np.isfinite(ver)
          else "Final verification: INVALID")

    if not np.isfinite(ver):
        # Fallback
        global_a = np.ones(256) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, score=%.6f" % (len(final_list), ver))
    return final_list
