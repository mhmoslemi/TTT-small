import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, beta, fsize):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a via FFT."""
    n = len(a)
    c_max = c.max()

    scaled = np.clip(scaled, -500, 500)

    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

                new_mag = (1 - rho) * abs_A[k] + rho * target
                A_new[k] = new_mag * phase
        
        # Keep DC
        A_new[0] = A[0]
        
        # Enforce Hermitian symmetry for real signal
        # A[fsize-k] should be conj(A[k])
        # Actually scipy.fft handles this naturally, but let's enforce it
        for k in range(1, fsize // 2 + 1):
            A_new[fsize - k] = np.conj(A_new[k])
        
        # Inverse FFT
        a_new = np.real(ifft(A_new))[:n]
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_new * (S / s_new)
        
        # Evaluate
        c = fft_convolve_self(a)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Adaptive relaxation: increase rho over time for faster convergence
        # Then decrease near the end for fine-tuning
        progress = it / max(1, int(max_time / 0.005))  # rough
        rho = 0.3 + 0.4 * min(1.0, it / 200.0)
        if it > 500:
            rho = max(0.1, 0.7 - 0.01 * (it - 500) / 100.0)
        
        # Periodic break if stuck
        if it % 200 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() > best_score * 1.001:
                # Oscillating - restart with current best
                a = best_a.copy()
                rho = 0.3
    
    return best_a, best_score
