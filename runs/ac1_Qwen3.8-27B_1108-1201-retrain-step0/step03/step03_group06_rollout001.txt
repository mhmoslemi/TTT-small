```python
def coordinate_line_search(a, j, c):
    """Find the best delta for coordinate j given current convolution c."""
    n = len(a)
    # s[k] = d(conv(a,a)[k])/d(a[j]) = 2*a[k-j] if 0 <= k-j < n
    k_offset = j
    # s[k] = 2*a[k-j] for k in [j, j+n-1] (capped at 2n-2)
    k_start = max(0, j)
    k_end = min(2*n-1, j + n)
    s = np.zeros(2*n-1)
    idx = np.arange(k_start, k_end)
    s[idx] = 2.0 * a[idx - j]
    
    # We want to minimize max(c + delta * s) by choosing delta.
    # But we also need a[j] + delta >= 0, so delta >= -a[j].
    # We renormalize after, so the actual change is slightly different.
    # For now, ignore normalization and just find the best delta.
    
    # The function f(delta) = max_k (c[k] + s[k]*delta) is convex in delta.
    # Minimum is where the "active" constraints cross.
    
    # Simple approach: try a range of deltas and pick the best.
    # Or: binary search on the subgradient.
    
    # At the optimum, there exist k1, k2 such that:
    # c[k1] + s[k1]*delta = c[k2] + s[k2]*delta = max
    # and s[k1] < 0 < s[k2] (one going up, one going down)
    
    # For simplicity, let's just do a golden section search on f(delta).
    
    lo = -a[j]  # a[j] + delta >= 0
    hi = a[j]  # don't increase too much (heuristic)
    
    # Actually, let's be smarter. The maximum change in c is:
    # max_k |s[k]| * |delta|. So if we want c to change by at most c.max(),
    # then |delta| <= c.max() / max(|s|).
    max_s = np.abs(s).max()
    if max_s < 1e-15:
        return 0.0
    hi = min(a[j], c.max() / max_s)  # conservative upper bound
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a_gs = lo
    b_gs = hi
    c_gs = b_gs - gr * (b_gs - a_gs)
    d_gs = a_gs + gr * (b_gs - a_gs)
    f_c = (c + c_gs * s).max()
    f_d = (c + d_gs * s).max()
    
    for _ in range(50):
        if f_c < f_d:
            b_gs = d_gs
            d_gs = c_gs
            f_d = f_c
            c_gs = b_gs - gr * (b_gs - a_gs)
            f_c = (c + c_gs * s).max()
        else:
            a_gs = c_gs
            c_gs = d_gs
            f_c = f_d
            d_gs = a_gs + gr * (b_gs - a_gs)
            f_d = (c + d_gs * s).max()
    
    best_delta = (a_gs + b_gs) / 2
    return best_delta
```

```python
    """One sweep of coordinate descent with approximate line search."""
    a = a.copy()
    best_obj = c.max()
    rng = np.random.default_rng()
    order = rng.permutation(n)
    
    for j in order:
        if a[j] < 1e-12:
            continue
        
        # Compute sensitivity: s[k] = 2*a[k-j] for k in [j, j+n-1]
        k_start = j
        k_end = min(2*n-1, j + n)
        s = np.zeros(2*n-1)
        s[k_start:k_end] = 2.0 * a[:k_end-k_start]
        
        # Find candidate deltas
        # The function f(delta) = max(c + delta*s) is piecewise linear convex.
        # Try a few strategic deltas.
        candidates = [0.0]
        
        # Delta that would zero out the max (if s at max is negative)
        k_max = np.argmax(c)
        if s[k_max] != 0:
            # c[k_max] + delta * s[k_max] = c.mean() → delta = (c.mean() - c[k_max]) / s[k_max]
            target = c.mean() * 1.01  # slightly above mean
            d = (target - c[k_max]) / s[k_max]
            if d < 0 and d > -a[j]:
                candidates.append(d)
        
        # Delta to set a[j] to a fraction of its current value
        for frac in [0.1, 0.3, 0.5, 0.7, 1.5, 2.0]:
            d = (frac - 1.0) * a[j]
            candidates.append(d)
        
        # Clip to valid range
        candidates = [max(d, -a[j] * 0.99) for d in candidates]
        
        # Evaluate and pick best
        best_delta = 0.0
        best_val = c.max()
        for d in candidates:
            c_approx = c + d * s
            # Renormalization effect (approximate)
            new_sum = S + d
            if new_sum < 1e-10:
                continue
            scale = S / new_sum
            c_approx *= scale * scale  # Both a's are scaled
            val = c_approx.max()
            if val < best_val - 1e-12:
                best_val = val
                best_delta = d
        
        if best_delta != 0.0:
            a[j] += best_delta
            a[j] = max(a[j], 0.0)
            # Recompute convolution (expensive but necessary for accuracy)
            c = fft_convolve_self(a * (S / a.sum()))
```
