import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def safe_fft_size(n):
    """Next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = safe_fft_size(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_xcorr_grad(a, w, conv_L):
    """
    Compute 2 * (a cross-correlated with w) using FFT.
    a: length n, w: length 2n-1
    Returns gradient of length n.
    """
    n = len(a)
    A = fft(a, conv_L)
    W = fft(w, conv_L)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def nesterov_optimize(a_init, n, S, max_time, rng,
                      beta_start=50.0, beta_end=5000.0,
                      conv_L=None, corr_L=None):
    """
    Nesterov-accelerated projected gradient on simplex.
    Minimizes smooth max of self-convolution.
    """
    if conv_L is None:
        conv_L = safe_fft_size(2 * n - 1)
    if corr_L is None:
        corr_L = safe_fft_size(3 * n - 2)

    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = (time.monotonic() - start) / max_time
        # Quadratic ramp on beta
        beta = beta_start + (beta_end - beta_start) * frac * frac

        # Compute convolution of y (for tracking best)
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()

        # Compute weights: softmax with temperature 1/beta
        c_shifted = c - c.max()
        exp_bc = np.exp(beta * c_shifted)
        sum_exp = exp_bc.sum()
        if sum_exp < 1e-300:
            # Fallback: uniform on top-1
            w = np.zeros(2 * n - 1)
            w[c.argmax()] = 1.0
        else:
            w = exp_bc / sum_exp

        # Compute gradient: (2/beta) * cross_corr(a, w)
        A = fft(a, corr_L)
        W = fft(w, corr_L)
        grad = np.real(ifft(np.conj(A) * W))[:n]
        grad = (2.0 / beta) * grad

            break
        step = 2.0 * S / (n * gnorm * (1.0 + 0.005 * t))
        # Cap step to avoid overshoot
        step = min(step, 0.5 * S / n)

        x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Ensure y is on simplex (numerical safety)
        ys = y.sum()
        if ys > 0:
            y = y * (S / ys)
        else:
            y = x_new.copy()

        a = x_new
        t = t_new

        # Periodic perturbation to escape local minima
        if it % 2000 == 0 and rng.random() < 0.15:
            scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, scale, n), S)
            t = 1.0

        # Check convergence via KKT condition
        if it % 5000 == 0:
            # At optimum: (a cross-corr w) should be ~constant on support
            A = fft(a, corr_L)
            W = fft(w, corr_L)
            cc = np.real(ifft(np.conj(A) * W))[:n]
            support = a > 0.01 * S / n
            if support.sum() > n // 4:
                cc_active = cc[support]
                cv = cc_active.std() / (cc_active.mean() + 1e-30)
                if cv < 0.005:
                    # Near KKT optimum, break to L-BFGS polish
                    break

    return best_a, best_score


def softmax_obj_and_grad(z, beta, S, conv_L, corr_L):
    """Compute smooth max objective and gradient w.r.t. softmax params z."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    A = fft(a, corr_L)
    W = fft(w, corr_L)
    grad_a = 2.0 * np.real(ifft(np.conj(A) * W))[:n]

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def lbfgs_polish(a_init, n, S, max_time, rng,
                 beta_schedule=None, maxiter_per_beta=200,
                 conv_L=None, corr_L=None):

        conv_L = safe_fft_size(2 * n - 1)

    best_score = fft_convolve_self(a).max()

            return softmax_obj_and_grad(z_val, beta, S, conv_L, corr_L)[1]

def block_swap(a, n, rng):
    """Swap two random blocks of the sequence."""
    n1 = rng.integers(64, n // 3)
    n2 = rng.integers(64, (n - n1) // 2)
    start1 = rng.integers(0, n - n1)
    start2 = start1 + n1 + rng.integers(0, n - n1 - n2)
    if start2 + n2 > n:
        start2 = rng.integers(0, n - n2)
        if abs(start2 - start1) < n1:
            return a.copy()
    a_new = a.copy()
    temp = a_new[start1:start1+n1].copy()
    a_new[start1:start1+n1] = a_new[start2:start2+n1]
    a_new[start2:start2+n1] = temp
    return a_new


def block_mirror(a, n, rng):
    """Replace a block with its mirror image."""
    block_len = rng.integers(128, n // 3)
    start = rng.integers(0, n - block_len)
    a_new = a.copy()
    a_new[start:start+block_len] = a[start:start+block_len][::-1]
    return a_new


def smooth_blend(a, n, rng, sigma):
    """Gaussian-smoothed random perturbation."""
    noise = rng.normal(0, sigma, n)
    a_new = a + noise
    return np.maximum(a_new, 0.0)

            # Frequency-domain perturbation
            A = fft(global_a, conv_L)
            phase_noise = rng.normal(0, 0.05, conv_L)
            A_pert = A * np.exp(1j * phase_noise)
            a_mut = np.real(ifft(A_pert))[:n]
            a_mut = np.maximum(a_mut, 0.0)

        a_mut = normalize_to_sum(a_mut, S)

        # Short Nesterov + L-BFGS
        a_opt, score_opt = nesterov_optimize(
            a_mut, n, S, min(mut_budget * 0.5, 25), rng,
            beta_start=200.0, beta_end=5000.0,
            conv_L=conv_L, corr_L=corr_L
        )
        a_opt2, score_opt2 = lbfgs_polish(
            a_opt, n, S, min(mut_budget * 0.4, 20), rng,
            beta_schedule=[500, 2000, 5000],
            maxiter_per_beta=100,
            conv_L=conv_L, corr_L=corr_L
        )
        final_score = min(score_opt, score_opt2)
        final_a = a_opt if score_opt <= score_opt2 else a_opt2
        update_best(final_a, final_score)
        print("  Mutation %d (type=%d): obj=%.6f (%.1fs)" %
              (m, mut_type, 2 * n * final_score / S**2, time.monotonic() - t0))

    # === Phase 4: Extended Nesterov on best (15% time) ===
    remaining = deadline - time.monotonic()
    phase4_time = remaining * 0.15
    print("\n=== Phase 4: Extended Nesterov (%.0fs) ===" % phase4_time)
    t0 = time.monotonic()

    a_p4, score_p4 = nesterov_optimize(
        global_a, n, S, min(phase4_time, 200), rng,
        beta_start=500.0, beta_end=20000.0,
        conv_L=conv_L, corr_L=corr_L
    )
    score_obj = 2 * n * score_p4 / S**2
    update_best(a_p4, score_p4)
    print("  Phase 4 done: obj=%.6f (%.1fs)" % (score_obj, time.monotonic() - t0))

    # === Phase 5: Final L-BFGS with very high beta (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("\n=== Phase 5: Final high-beta L-BFGS (%.0fs) ===" % remaining)
        t0 = time.monotonic()

        a_p5, score_p5 = lbfgs_polish(
            global_a, n, S, min(remaining - 10, 200), rng,
            beta_schedule=[2000, 5000, 10000, 20000],
            maxiter_per_beta=200,
            conv_L=conv_L, corr_L=corr_L
        )
        score_obj = 2 * n * score_p5 / S**2
        update_best(a_p5, score_p5)
        print("  Phase 5 done: obj=%.6f (%.1fs)" % (score_obj, time.monotonic() - t0))

    # === Phase 6: Tiny random perturbations + quick check ===
    remaining = deadline - time.monotonic()
    if remaining > 15 and global_a is not None:
        print("\n=== Phase 6: Micro-perturbations (%.0fs) ===" % remaining)
        n_micro = max(5, int((remaining - 5) / 3))
        for mi in range(n_micro):
            if time.monotonic() > deadline - 5:
                break
            scale = 0.0005 * S / np.sqrt(n) * (1.0 + mi * 0.5)
            a_micro = global_a + rng.normal(0, scale, n)
            a_micro = normalize_to_sum(a_micro, S)
            c = fft_convolve_self(a_micro)
            if c.max() < global_best:
                global_best = c.max()
                global_a = a_micro.copy()
                print("  Micro %d: obj=%.8f" %
                      (mi, 2 * n * global_best / S**2))

    # === Verification ===
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_obj = 2 * len(global_a) * fft_convolve_self(global_a).max() / global_a.sum()**2
    print("\nFinal: length=%d, obj=%.8f" % (len(global_a), final_obj))
    return [float(x) for x in global_a]
