```python
"""
Minimize 2*n*max_conv/sum^2 for non-negative sequences of length n.
Strategy: Multi-start FISTA with adaptive beta and backtracking line search,
followed by top-k focused Nesterov, then coordinate descent polish.
Uses diverse initializations and periodic perturbations to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Projection onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def fft_convolve_self(a):
    """Self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_weights(c, beta):
    """Softmax weights emphasizing large values."""
    c_max = c.max()
    shifted = beta * (c - c_max)
    # Clip to avoid underflow
    shifted = np.maximum(shifted, -700.0)
    exp_c = np.exp(shifted)
    total = exp_c.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_c / total


def toppk_weights(c, k):
    """Normalized weights on top-k values."""
    k = min(k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx].sum()
    if top_vals < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_idx] = c[top_idx] / top_vals
    return weights


def compute_grad(a, weights):
    """Gradient of weighted sum of convolution values w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fista_with_backtrack(a_init, n, S, max_time, rng,
                          beta_init=50.0, beta_growth=1.005,
                          beta_max=10000.0, step_init=None):
    """FISTA with backtracking and geometric beta growth."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(y).max()
    best_a = y.copy()
    it = 0
    
    if step_init is None:
        # Estimate Lipschitz constant
        c = fft_convolve_self(y)
        w = softmax_weights(c, beta_init)
        grad = compute_grad(y, w)
        L = np.linalg.norm(grad) ** 2 / max(np.linalg.norm(y) ** 2, 1e-15)
        step = 1.0 / max(L * 1.1, 1e-10)
    else:
        step = step_init
    
    stall = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (beta_growth ** it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-13:
            best_score = cur
            best_a = y.copy()
            stall = 0
        else:
            stall += 1
        
        w = softmax_weights(c, beta)
        grad = compute_grad(y, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Backtracking line search
        x_prev = y - step * grad
        x_new = project_simplex(x_prev, S)
        c_new = fft_convolve_self(x_new)
        
        if c_new.max() > cur + 0.01 * step * grad_norm ** 2:
            # Step too large, reduce
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-project to maintain sum constraint approximately
        s_y = y.sum()
        if s_y > 0:
            y = y * (S / s_y)
        a = x_new
        t = t_new
        
        # Perturb if stalled
        if stall > 500 and rng.random() < 0.1:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
            stall = 0
        
        # This is expensive, so check time frequently
        if it % 10 == 0 and time.monotonic() - start > max_time:
            break
    
    return best_a, best_score


def toppk_nesterov(a_init, n, S, max_time, rng,
                   k_init=1, k_max=16, k_incr_every=200):
    """Top-k focused Nesterov accelerated gradient."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
```

```python
    best_peak = fft_convolve_self(a).max()

        c = fft_convolve_self(a)
        peak = c.max()
        if peak < best_peak - 1e-14:
            best_peak = peak

        peak_lags = np.where(c >= peak_thresh)[0]

            # Find contributing pairs
            # c[k] = sum_i a[i]*a[k-i] for valid i
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            valid_i = np.arange(lo, hi + 1)
            products = a[valid_i] * a[k - valid_i]
            
            # Find the pair with the largest product
            max_idx = valid_i[np.argmax(products)]
            other_idx = k - max_idx
            
            if max_idx == other_idx:
                continue  # Can't swap with self
            
            # Swap: move mass from max_idx to other_idx (if a[max_idx] > a[other_idx])
            if a[max_idx] > a[other_idx]:
                donor, receiver = max_idx, other_idx
            else:
                donor, receiver = other_idx, max_idx
            
            # Try different swap amounts
            for frac in [0.01, 0.005, 0.001]:
                delta = frac * (a[donor] - a[receiver])
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[donor] -= delta
                a_try[receiver] += delta
                c_try = fft_convolve_self(a_try)
                if c_try.max() < peak:
                    a = a_try
                    peak = c_try.max()
                    improved = True
                    break

            for _ in range(10):

                    improved = True
```

```python
Minimize 2*n*max_conv/sum^2 for non-negative sequences.
Multi-phase: Adaptive FISTA -> Top-k Nesterov -> Peak shaving -> Exploration.
Key improvements over previous: backtracking line search, geometric beta growth,

import numpy as np

from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Projection onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def fft_convolve_self(a):
    """Self-convolution via FFT, length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_weights(c, beta):
    """Softmax weights emphasizing large values."""
    c_max = c.max()
    shifted = np.maximum(beta * (c - c_max), -700.0)
    exp_c = np.exp(shifted)
    total = exp_c.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_c / total


def toppk_weights(c, k):
    """Normalized weights on top-k largest values."""
    k = min(k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx].sum()
    if top_vals < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_idx] = c[top_idx] / top_vals
    return weights

    """Gradient of sum_k weights[k]*conv_k w.r.t. a."""
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    W = fft(weights[:fsize], fsize)  # Pad or truncate weights
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad
```

```python
"""
Minimize 2*n*max_conv/sum^2 for non-negative sequences.
Multi-phase strategy: Adaptive FISTA with backtracking -> Top-k Nesterov ->
Peak shaving via pairwise swaps -> Multi-start exploration.
Key improvements: geometric beta growth, backtracking line search,
targeted swap-based refinement, and diverse initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Projection onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def fft_convolve_self(a):
    """Self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_weights(c, beta):
    """Softmax weights emphasizing large values."""
    c_max = c.max()
    shifted = np.maximum(beta * (c - c_max), -700.0)
    exp_c = np.exp(shifted)
    total = exp_c.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_c / total


def toppk_weights(c, k):
    """Normalized weights on top-k largest values."""
    k = min(k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx].sum()
    if top_vals < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_idx] = c[top_idx] / top_vals
    return weights


def compute_grad(a, weights):
    """Gradient of sum_k weights[k]*conv_k w.r.t. a via FFT cross-correlation."""
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def fista_backtrack(a_init, n, S, max_time, rng,
                    beta_init=40.0, beta_growth=1.008,
                    beta_max=8000.0):
    """FISTA with backtracking line search and geometric beta growth."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(y)
    best_score = c0.max()
    best_a = y.copy()
    it = 0

    # Initialize step size
    w0 = softmax_weights(c0, beta_init)
    g0 = compute_grad(y, w0)
    g0_norm = np.linalg.norm(g0)
    step = S / (n * 1.5) if g0_norm < 1e-15 else 2.0 * S / (n * g0_norm)

    stall = 0
    no_improve_best = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (beta_growth ** it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-13:
            best_score = cur
            best_a = y.copy()
            stall = 0
            no_improve_best = 0
        else:
            stall += 1
            no_improve_best += 1

        w = softmax_weights(c, beta)
        grad = compute_grad(y, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Gradient step with backtracking
        x_trial = project_simplex(y - step * grad, S)
        c_trial = fft_convolve_self(x_trial)
        # Backtrack if not improving
        if c_trial.max() > cur + 0.1 * step * grad_norm:
            step *= 0.5
            x_trial = project_simplex(y - step * grad, S)
        elif it % 100 == 0 and c_trial.max() < cur * 0.999:
            step *= 1.05  # Slightly increase if making good progress

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_trial + ((t - 1.0) / t_new) * (x_trial - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 0:
            y = y * (S / s_y)
        a = x_trial
        t = t_new

        # Perturb if stalled
        if stall > 300 and rng.random() < 0.15:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            step *= 0.8
            stall = 0

        if it % 50 == 0 and time.monotonic() - start > max_time:
            break

    return best_a, best_score


def toppk_refined(a_init, n, S, max_time, rng,
                  k_init=1, k_max=16, k_rate=300):
    """Top-k Nesterov with adaptive k and normalized steps."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(y).max()
    best_a = y.copy()
    it = 0
    base_step = 2.0 * S / (n * 1.0)

    while time.monotonic() - start < max_time:
        it += 1
        k = min(k_max, k_init + it // k_rate)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-13:
            best_score = cur
            best_a = y.copy()

        weights = toppk_weights(c, k)
        grad = compute_grad(y, weights)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = base_step / (1.0 + 0.003 * it)
        x_new = project_simplex(y - step * grad / max(grad_norm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 0:
            y = y * (S / s_y)
        a = x_new
        t = t_new

        # Periodic perturbation to escape local minima
        if it % 1500 == 0 and rng.random() < 0.12:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0

    return best_a, best_score


def peak_shave_swaps(a_init, n, S, max_time, rng):
    """Reduce peak via targeted pairwise mass transfers at peak-contributing lags."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_peak = c.max()
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        peak = c.max()
        if peak < best_peak - 1e-14:
            best_peak = peak
            best_a = a.copy()

        # Find top peak lags
        peak_threshold = peak * 0.995
        peak_lags = np.where(c >= peak_threshold)[0]
        if len(peak_lags) == 0:
            peak_lags = np.array([np.argmax(c)])

        improved = False
        # For each near-peak lag, try to reduce it via swaps
        for k in peak_lags[:8]:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            if lo > hi:
                continue
            valid_i = np.arange(lo, hi + 1)
            products = a[valid_i] * a[k - valid_i]
            # Find the pair with largest product
            max_pos = np.argmax(products)
            i1 = valid_i[max_pos]
            i2 = k - i1
            if i1 == i2 or i1 < 0 or i1 >= n or i2 < 0 or i2 >= n:
                continue

            # Transfer from larger to smaller
            if a[i1] > a[i2]:
                donor, receiver = i1, i2
            else:
                donor, receiver = i2, i1

            # Try different transfer fractions
            base_delta = a[donor] - a[receiver]
            if base_delta <= 0:
                continue
            for frac in [0.02, 0.01, 0.005, 0.001]:
                delta = frac * base_delta
                a_try = a.copy()
                a_try[donor] -= delta
                a_try[receiver] += delta
                c_try = fft_convolve_self(a_try)
                if c_try.max() < peak - 1e-15:
                    a = a_try
                    peak = c_try.max()
                    improved = True
                    break

        if not improved:
            # Try gradient-based micro-steps
            weights = softmax_weights(c, 500.0)
            grad = compute_grad(a, weights)
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 1e-15:
                alpha = 0.01 * S / (n * grad_norm)
                a_try = project_simplex(a - alpha * grad, S)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < peak:
                    a = a_try
                    peak = c_try.max()
                else:
                    # Random perturbation

                                a_try = a.copy()
                                a_try[j] = max(0.0, a_try[j] + sign * 0.001 * S / n)
                                a_try = project_simplex(a_try, S)
                                c_try = fft_convolve_self(a_try)
                                if c_try.max() < peak:
                                    a = a_try
                                    peak = c_try.max()
                                    break

    return best_a, best_peak


def multi_start_explore(a_parent, n, S, max_time, rng):
    """Try diverse perturbations of the parent and quick-optimize each."""
    best_a = a_parent.copy()
    best_score = fft_convolve_self(a_parent).max()

    strategies = [
        ("gaussian_noise", 0.005),
        ("smooth_noise", 0.003),
        ("multiplicative", 0.01),
        ("sinusoidal", 0.008),
    ]

    per_strategy = max_time / len(strategies)

    for name, scale in strategies:
        if time.monotonic() > start + max_time:
            break
        tt = np.linspace(0, 1, n, endpoint=False)
        if name == "gaussian_noise":
            a_try = a_parent + scale * S / np.sqrt(n) * rng.standard_normal(n)
        elif name == "smooth_noise":
            raw = rng.standard_normal(n)
            smooth = np.convolve(raw, np.ones(32) / 32, mode='same')
            a_try = a_parent + scale * S / np.sqrt(n) * smooth
        elif name == "multiplicative":
            a_try = a_parent * (1.0 + scale * rng.standard_normal(n))
        else:  # sinusoidal
            freq = rng.integers(3, 20)
            a_try = a_parent + scale * S / np.sqrt(n) * np.sin(2 * np.pi * freq * tt)

        a_try = safe_normalize(a_try, n)
        # Quick FISTA
        a_opt, score = fista_backtrack(a_try, n, S, per_strategy * 0.8, rng,
                                       beta_init=100.0, beta_growth=1.005,
                                       beta_max=5000.0)
        a_opt = safe_normalize(a_opt, n)
        score = fft_convolve_self(a_opt).max()
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print("    %s: improved to %.6f" % (name, score))

    return best_a, best_score
```
