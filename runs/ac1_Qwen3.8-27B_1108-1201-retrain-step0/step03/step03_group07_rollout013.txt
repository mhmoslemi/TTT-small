    """Build the convolution matrix C where (C @ b)[k] = (a*b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    C = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return C

def solve_lp_direction(a, M):
    """Solve the LP to find the best direction b."""
    try:
        import cvxpy as cp
        n = len(a)
        C = build_conv_matrix(a)
        b = cp.Variable(n)
        constraints = [C @ b <= M, b >= 0]
        objective = cp.Maximize(cp.sum(b))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CLARABEL)
        if b.value is not None:
            return np.maximum(b.value, 0.0)
    except Exception as e:
        print(f"    LP solve failed: {e}")
    return None

def lp_optimization(a0, time_budget, verbose=True):
    """LP-based optimization using CVXPY."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    M = c.max()
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    iter_no = 0
    
    # Pre-build the constraint matrix (it changes when a changes, so we can't pre-build)
    # Actually, C depends on a, so we need to rebuild it each iteration.
    # That's O(n^2) per iteration, which for n=4096 is about 16M operations.
    
    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        M = c.max()
        
        b = solve_lp_direction(a, M)
        if b is None or b.sum() < 1e-10:
            break
        
        # Normalize b to have the same sum as a
        b = b * (S / b.sum())
        
        # Line search
        accepted = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            a_trial = (1 - t) * a + t * b
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial * (S / s)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < M * (1 + 1e-10):
                a = a_trial
                c = c_trial
                M = c_trial.max()
                accepted = True
                break
        
        if not accepted:
            # Try a very small step
            t = 0.01
            a_trial = (1 - t) * a + t * b
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial * (S / a_trial.sum())
            a = a_trial
            c = fft_convolve_self(a)
        
        cur_M = c.max()
        cur_obj = 2.0 * n * cur_M / (S * S)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        iter_no += 1
        if verbose and iter_no % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP: iter={iter_no}, M={cur_M:.4f}, obj={cur_obj:.6f}, "
                  f"best={best_obj:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_obj

def spectral_flatten(a0, time_budget, verbose=True):
    """Spectral flattening: try to make the power spectrum flat."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    iter_no = 0
    
    fsize = 1 << (2 * n - 1 - 1).bit_length()
    
    while time.monotonic() - start < time_budget:
        A = fft(a, fsize)
        P = np.abs(A) ** 2
        
        # Flatten the spectrum (reduce peaks in P)
        P_mean = P.mean()
        P_flat = np.minimum(P, P_mean * 2.0)  # Cap peaks at 2x mean
        
        # Take the square root to get the new amplitude
        A_new = np.sqrt(P_flat) * np.exp(1j * np.angle(A))
        
        # Inverse FFT
        a_new = np.real(ifft(A_new, fsize))[:n]
        
        # Project onto {a >= 0, sum = S}
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new * (S / s)
        
        c_new = fft_convolve_self(a_new)
        obj_new = obj_value(a_new, c_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        # else: keep old a (reject the step)
        
        iter_no += 1
        if verbose and iter_no % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    Spec: iter={iter_no}, obj={obj_new:.6f}, best={best_obj:.6f}, "
                  f"t={elapsed:.1f}s")
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a multi-strategy optimization:
1. Mirror descent with adaptive subgradient for broad peak reduction
2. LP-based direction finding (maximizing sum(b) s.t. (a*b) <= M, b>=0) for precise optimization
3. Fine polishing with exact subgradient and spectral methods
Key innovations: LP direction for simultaneous multi-peak control, adaptive
step sizes, and careful time management across phases.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import coo_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build sparse matrix C where (C @ b)[k] = (a*b)[k]."""
    n = len(a)
    col_indices = np.tile(np.arange(n), n)
    row_parts = [np.arange(j, j + n) for j in range(n)]
    row_indices = np.concatenate(row_parts)
    values = np.tile(a, n)
    C = coo_matrix((values, (row_indices, col_indices)), shape=(2 * n - 1, n)).tocsr()
    return C


def solve_lp_direction(a, M, solver_pool=None):
    """Solve LP: max sum(b) s.t. C@b <= M, b >= 0. Returns b or None."""
    try:
        import cvxpy as cp
        n = len(a)
        C = build_conv_matrix(a)
        b = cp.Variable(n)
        constraints = [C @ b <= M, b >= 0]
        objective = cp.Maximize(cp.sum(b))
        prob = cp.Problem(objective, constraints)
        if solver_pool is not None:
            prob.solve(solver=solver_pool)
        else:
            prob.solve(solver=cp.CLARABEL)
        if b.value is not None:
            b_val = np.maximum(b.value, 0.0)
            if b_val.sum() > 1e-10:
                return b_val
    except Exception as e:
        print(f"    LP error: {e}")
    return None


def compute_subgrad_exact(c, a):
    """Compute subgradient at the exact argmax of c."""
    n = len(a)
    k_max = np.argmax(c)
    grad = np.zeros(n)
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[k_max - j]
    return grad


def compute_subgrad_topk(c, a, top_k):
    """Compute subgradient as average over top-k peaks."""
    n = len(a)
    top_k = max(1, min(top_k, len(c)))
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / top_k
    a_rev = a[::-1]
    conv = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def mirror_descent_phase(a0, time_budget, eta0=0.3, top_k=1, momentum=0.85,
                         perturb_every=400, perturb_scale=0.03,
                         verbose=True, decay=0.99995):
    """Mirror descent with subgradient and backtracking."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)
    eta = eta0
    iter_no = 0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        if top_k == 1:
            grad = compute_subgrad_exact(c, a)
        else:
            grad = compute_subgrad_topk(c, a, top_k)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        accepted = False
        current_obj = obj_value(a, c)
        for bt in range(8):
            eta_try = eta * (0.7 ** bt)
            a_trial = a * np.exp(-eta_try * v)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)
            if obj_trial < current_obj * (1.0 + 1e-10):
                a = a_trial
                c = c_trial
                accepted = True
                break

        if not accepted:
            a = a * np.exp(-eta * 0.05 * v)
            a = np.maximum(a, 1e-300)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)

        cur_obj = obj_value(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
            eta = min(eta * 1.015, eta0 * 3.0)
        else:
            eta = max(eta * 0.92, eta0 * 0.05)

        if iter_no - best_iter > perturb_every:
            noise = np.random.randn(n) * (perturb_scale * S / np.sqrt(n))
            a = np.maximum(a + noise, 0.0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.4
            eta = eta0
            best_iter = iter_no
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    [perturb] iter={iter_no}, obj={cur_obj:.6f}, "
                      f"best={best_obj:.6f}, t={elapsed:.1f}s")

        eta *= decay
        iter_no += 1
        if verbose and iter_no % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    MD: iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"eta={eta:.4f}, top_k={top_k}, t={elapsed:.1f}s")

    return best_a, best_obj


def lp_optimization_phase(a0, time_budget, verbose=True):
    """LP-based optimization: solve max sum(b) s.t. (a*b)<=M, b>=0 each iter."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    M = c.max()
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    iter_no = 0
    no_improve = 0

    # Try to find a working solver
    solver_to_use = None
    try:
        import cvxpy as cp
        for s in [cp.CLARABEL, cp.ECOS, cp.SCS]:
            solver_to_use = s
            break
    except:
        print("    CVXPY not available, skipping LP phase")
        return a, best_obj

    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        M = c.max()

        t0 = time.monotonic()
        b = solve_lp_direction(a, M, solver_to_use)
        lp_time = time.monotonic() - t0

        if b is None:
            break

        b = b * (S / b.sum())

        accepted = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - t) * a + t * b
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial * (S / s)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() <= M * (1.0 + 1e-9):
                a = a_trial
                c = c_trial
                M = c_trial.max()
                accepted = True
                break

        if not accepted:
            t = 0.01
            a = (1.0 - t) * a + t * b
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)

        cur_M = c.max()
        cur_obj = 2.0 * n * cur_M / (S * S)

        if cur_obj < best_obj * (1.0 - 1e-10):
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            # Perturb and reset
            noise = np.random.randn(n) * (0.02 * S / np.sqrt(n))
            a = np.maximum(a + noise, 0.0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            no_improve = 0

        iter_no += 1
        if verbose and (iter_no % 5 == 0 or iter_no < 3):
            elapsed = time.monotonic() - start
            print(f"    LP: iter={iter_no}, M={cur_M:.4f}, obj={cur_obj:.6f}, "
                  f"best={best_obj:.6f}, lp_time={lp_time:.2f}s, t={elapsed:.1f}s")

        # If LP is too slow, stop
        if lp_time > time_budget * 0.1:
            if verbose:
                print(f"    LP too slow, stopping")
            break

    return best_a, best_obj


def spectral_phase(a0, time_budget, verbose=True):
    """Spectral flattening: alternate time-domain projection with frequency flattening."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    iter_no = 0

    fsize = 1 << (2 * n - 1 - 1).bit_length()

    while time.monotonic() - start < time_budget:
        A = fft(a, fsize)
        P = np.abs(A) ** 2

        # Try different flattening strengths
        found = False
        for cap_factor in [1.5, 2.0, 3.0, 5.0]:
            P_flat = np.minimum(P, cap_factor * P[n // 2] if n % 2 == 0 else cap_factor * P.mean())
            P_flat[0] = P[0]  # Keep DC component
            # Also keep the last bin (Nyquist)
            if fsize % 2 == 0:
                P_flat[-1] = P[-1]

            A_new = np.sqrt(np.maximum(P_flat, 0.0)) * np.exp(1j * np.angle(A))
            a_new = np.real(ifft(A_new, fsize))[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new * (S / s)
            c_new = fft_convolve_self(a_new)
            obj_new = obj_value(a_new, c_new)

            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                found = True
                break

        if not found:
            # Try a gentle version
            P_smooth = P * 0.5 + P.mean() * 0.5
            A_new = np.sqrt(np.maximum(P_smooth, 0.0)) * np.exp(1j * np.angle(A))
            a_new = np.real(ifft(A_new, fsize))[:n]
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new * (S / s)
                a = a_new

        iter_no += 1
        if verbose and iter_no % 200 == 0:
            elapsed = time.monotonic() - start
            c_cur = fft_convolve_self(a)
            print(f"    Spec: iter={iter_no}, obj={obj_value(a, c_cur):.6f}, "
                  f"best={best_obj:.6f}, t={elapsed:.1f}s")

    return best_a, best_obj
