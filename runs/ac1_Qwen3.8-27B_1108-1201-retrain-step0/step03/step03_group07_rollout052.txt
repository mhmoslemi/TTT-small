"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using multi-phase optimization with:
- Multi-start exploration (given seq + constructed inits at n=4096, 8192)
- Smoothed Adam descent with adaptive log-sum-exp temperature

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_from_conv(a, c):
    """Compute objective from precomputed convolution."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def compute_smooth_gradient(c, a, temperature):
    """
    Compute gradient of T*logsumexp(c/T) w.r.t. a.
    grad[m] = 2 * sum_k w[k] * a[k-m] where w = softmax(c/T).
    """
    n = len(a)
    c_max = c.max()
    shifted = (c - c_max) / temperature
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()

    # grad[m] = 2 * sum_k w[k] * a[k-m] for valid k-m in [0, n-1]
    # This is 2 * (w cross-correlate a)[m]
    # = 2 * conv(w, a_rev)[n-1+m]
    a_rev = a[::-1]
    out_len = len(w) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def compute_topk_gradient(c, a, top_k, weights=None):
    """Compute subgradient using top-k peaks with optional custom weights."""
    n = len(a)
    top_k = max(1, min(top_k, len(c)))
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    if weights is None:
        weights = np.full(top_k, 1.0 / top_k)
    else:
        weights = np.array(weights, dtype=np.float64)
        weights = weights / weights.sum()

    c_max = c[top_idx[0]]

    # Build weight vector
    w_full = np.zeros(len(c))
    w_full[top_idx] = weights

    a_rev = a[::-1]
    out_len = len(w_full) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w_full, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad, top_idx


def softmax_weighted_gradient(c, a, temperature):
    """Gradient using exponential softmax over ALL peaks."""
    c_max = c.max()
    shifted = (c - c_max) / temperature
    shifted = np.clip(shifted, -50, 50)
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum

    n = len(a)
    a_rev = a[::-1]
    out_len = 2 * n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def adam_step(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8, S=None):
    """Adam update with simplex projection (multiplicative)."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    # Multiplicative update
    a_new = a * np.exp(-update)
    a_new = np.maximum(a_new, 1e-300)
    if S is not None:
        a_new = a_new * (S / a_new.sum())
    return a_new, m, v


def mirror_adam(a, time_budget, S, init_temp=1.0, final_temp=0.01,
                lr_init=0.5, lr_final=0.01, beta1=0.9, beta2=0.999,
                top_k_start=200, top_k_end=3, verbose=True,
                stagnation_limit=2000, perturb_scale=0.03):
    """
    Mirror descent with Adam momentum and adaptive temperature.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-300)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = obj_from_conv(a, c)
    best_a = a.copy()
    best_iter = 0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    iter_no = 0
    start = time.monotonic()
    progress = 0.0

    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, t / max(1, int(time_budget * 30)))

        # Linear interpolation of temperature and learning rate
        temp = init_temp * (final_temp / init_temp) ** progress
        lr = lr_init * (lr_final / lr_init) ** progress

        # Choose between smooth gradient and top-k gradient
        if progress < 0.3:
            # Early: smooth gradient (broad)
            grad = compute_smooth_gradient(c, a, temp)
        elif progress < 0.7:
            # Middle: top-k with many peaks
            k_cur = max(10, int(top_k_start * (1 - progress / 0.7)))
            grad, _ = compute_topk_gradient(c, a, k_cur)
        else:
            # Late: focus on few peaks
            k_cur = max(1, int(10 * (1 - (progress - 0.7) / 0.3)))
            grad, _ = compute_topk_gradient(c, a, k_cur)

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Adam step
        a, m, v = adam_step(a, grad, m, v, t, lr, beta1, beta2, S=S)

        c = fft_convolve_self(a)
        cur_obj = obj_from_conv(a, c)

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no

        # Stagnation detection
        if iter_no - best_iter > stagnation_limit:
            noise_scale = perturb_scale * S / np.sqrt(n)
            noise = np.random.randn(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 1e-300)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            m *= 0.1
            v *= 0.1
            best_iter = iter_no
            if verbose:
                elapsed = time.monotonic() - start
                print(f"      [perturb] iter={iter_no}, obj={cur_obj:.6f}, "
                      f"best={best_obj:.6f}, t={elapsed:.1f}s")

        iter_no += 1
        if verbose and iter_no % 3000 == 0:
            elapsed = time.monotonic() - start
            print(f"      iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"temp={temp:.4f}, lr={lr:.4f}, t={elapsed:.1f}s")

    return best_a, best_obj


def peak_chase(a, time_budget, S, n_peaks=5, verbose=True):
    """
    Chase the hardest peaks using convex combination subgradients.
    Weight peaks by their 'hardness' (resistance to reduction).
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-300)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = obj_from_conv(a, c)
    best_a = a.copy()
    start = time.monotonic()

    # Track peak values over iterations to estimate hardness
    peak_history = {}
    iter_no = 0

    while time.monotonic() - start < time_budget:
        # Find top peaks
        top_idx = np.argsort(c, kind='stable')[::-1][:n_peaks]
        top_vals = c[top_idx]
        c_max = top_vals[0]

        # Estimate hardness: peaks that haven't decreased much are "hard"
        # Use relative weight: harder peaks get more attention
        vals_normalized = top_vals / top_vals.sum()
        
        # Boost weights for peaks that are close to the max (they matter most)
        weights = np.exp(5.0 * (top_vals - c_max) / max(c_max - top_vals[-1], 1e-10))
        weights = weights / weights.sum()

        # Compute weighted subgradient
        w_full = np.zeros(len(c))
        w_full[top_idx] = weights
        a_rev = a[::-1]
        out_len = 2 * n - 1
        fsize = 1 << (out_len - 1).bit_length()
        W = fft(w_full, fsize)
        R = fft(a_rev, fsize)

        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())

        c = fft_convolve_self(a)
        cur_obj = obj_from_conv(a, c)

def coordinate_polish(a, time_budget, S, verbose=True):
    """
    Coordinate descent: try adjusting groups of coordinates to reduce peaks.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 1e-300)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = obj_from_conv(a, c)
    best_a = a.copy()
    start = time.monotonic()

    # Find which indices contribute most to the top peaks
    top_peak = np.argmax(c)
    # Indices contributing to peak at position top_peak:
    # j and top_peak-j both in [0, n-1]
    contrib_lo = max(0, top_peak - n + 1)
    contrib_hi = min(n - 1, top_peak)

    for sweep in range(200):
        if time.monotonic() - start > time_budget:
            break
        improved = False

        # Try adjusting contributing indices
        idx = np.random.choice(n, size=min(n, 512), replace=False)
        for i in idx:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            for factor in [0.8, 0.9, 1.1, 1.2]:
                a[i] = orig * factor
                a = np.maximum(a, 1e-300)
                a = a * (S / a.sum())
                c_new = fft_convolve_self(a)
                obj_new = c_new.max()  # proportional to objective since S is fixed
                if obj_new < c.max() * 0.999:
                    c = c_new
                    cur_obj = obj_from_conv(a, c)
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                    improved = True
                    break
                a[i] = orig
                a = a * (S / a.sum())
            else:
                a[i] = orig
                a = a * (S / a.sum())

        if not improved and sweep > 5:
            break

    return best_a, best_obj


def generate_construction(n, freq_factor=7.0, seed=42):
    """Generate a structured initial sequence."""
    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * n)
    i = np.arange(n)

    # Multi-frequency cosine with random phases
    phases = rng.uniform(0, 2 * np.pi, size=5)
    freqs = np.array([1, 3, 7, 13, 23]) * freq_factor / 7.0
    a = np.ones(n)
    for f, ph in zip(freqs, phases):
        a += 0.3 * np.cos(2 * np.pi * f * i / n + ph)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def upsample(a, factor=2):
    """Upsample a sequence by inserting interpolated values."""
    n = len(a)
    new_n = n * factor
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

        a_c = generate_construction(n4096, freq_factor=freq, seed=seed + int(freq))
        candidates.append((f'cos4096_f{freq}', a_c))

    # n=8192 constructions
    n8192 = 8192
    S8192 = np.sqrt(2.0 * n8192)
    a_8192 = generate_construction(n8192, freq_factor=7.0, seed=seed + 100)
    candidates.append(('cos8192', a_8192))

    # Upsample given to 8192
    if given_a is not None:
        a_up = upsample(given_a, factor=2)
        a_up = a_up * (S8192 / a_up.sum())
        candidates.append(('up8192', a_up))

    per_cand_time = explore_time / len(candidates)
    for name, a_cand in candidates:
        if time.monotonic() > deadline - 120:
            break
        t0 = time.monotonic()
        S_c = a_cand.sum()
        a_best, obj_best = mirror_adam(
            a_cand, per_cand_time, S_c,
            init_temp=2.0, final_temp=0.1,
            lr_init=0.3, lr_final=0.05,
            top_k_start=100, top_k_end=10,
            verbose=False
        )
        elapsed = time.monotonic() - t0
        print(f"  {name} (n={len(a_cand)}): obj={obj_best:.6f} ({elapsed:.1f}s)")
        update_global(a_best, obj_best)

    # --- Phase 2: Main optimization on best candidate (55% of time) ---
    remain = deadline - time.monotonic()
    main_time = 0.55 * remain
    print(f"\nPhase 2: Main optimization ({main_time:.0f}s)")

    if global_a is not None and main_time > 20:
        n = len(global_a)
        S = global_a.sum()

        # Split into sub-phases with progressively tighter focus
        sub_phases = [
            dict(init_temp=1.0, final_temp=0.05, lr_init=0.3, lr_final=0.02,
                 top_k_start=300, top_k_end=20, frac=0.35, label="broad"),
            dict(init_temp=0.5, final_temp=0.02, lr_init=0.2, lr_final=0.01,
                 top_k_start=50, top_k_end=5, frac=0.35, label="medium"),
            dict(init_temp=0.2, final_temp=0.005, lr_init=0.1, lr_final=0.003,
                 top_k_start=10, top_k_end=1, frac=0.30, label="tight"),
        ]

        for sp in sub_phases:
            t_budget = main_time * sp['frac']
            if time.monotonic() > deadline - 60:
                break
            print(f"  2-{sp['label']} ({t_budget:.0f}s)")
            a_best, obj_best = mirror_adam(
                global_a, t_budget, S,
                init_temp=sp['init_temp'], final_temp=sp['final_temp'],
                lr_init=sp['lr_init'], lr_final=sp['lr_final'],
                top_k_start=sp['top_k_start'], top_k_end=sp['top_k_end'],
                verbose=True
            )
            update_global(a_best, obj_best)

    # --- Phase 3: Peak chasing (20% of time) ---
    remain = deadline - time.monotonic()
    chase_time = 0.20 * remain
    print(f"\nPhase 3: Peak chasing ({chase_time:.0f}s)")

    if global_a is not None and chase_time > 10:
        S = global_a.sum()
        a_best, obj_best = peak_chase(global_a, chase_time, S, n_peaks=5, verbose=True)
        update_global(a_best, obj_best)

    # --- Phase 4: Coordinate polish (15% of time) ---
    remain = deadline - time.monotonic()
    polish_time = remain - 15.0
    print(f"\nPhase 4: Coordinate polish ({polish_time:.0f}s)")

    if global_a is not None and polish_time > 5:
        S = global_a.sum()
        a_best, obj_best = coordinate_polish(global_a, polish_time, S, verbose=True)
        update_global(a_best, obj_best)

    # --- Try n=8192 if time remains ---
    remain = deadline - time.monotonic()
    if remain > 200 and global_a is not None:
        print(f"\nBonus: Try n=8192 ({remain - 20:.0f}s)")
        n8192 = 8192
        S8192 = np.sqrt(2.0 * n8192)
        a_8k = upsample(global_a, factor=2) if len(global_a) == 4096 else generate_construction(n8192, seed=seed + 999)
        a_8k = a_8k * (S8192 / a_8k.sum())
        
        t_budget = remain - 30
        a_best, obj_best = mirror_adam(
            a_8k, t_budget * 0.6, S8192,
            init_temp=0.5, final_temp=0.02,
            lr_init=0.2, lr_final=0.02,
            verbose=False
        )
        obj_check = evaluate_sequence(a_best.tolist())
        print(f"  n=8192 result: local={obj_best:.6f}, verified={obj_check:.6f}")
        if np.isfinite(obj_check) and obj_check < global_best:
            global_best = obj_check
            global_a = a_best.copy()
            print(f"    ** NEW BEST from n=8192: {obj_check:.6f}")

    # --- Final verification ---
    if global_a is None:
        print("WARNING: No valid candidate found, using fallback")
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: local_obj={global_best:.6f}, verified={ver:.6f}, n={len(final_a)}")

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.6f}")

    return final_a
