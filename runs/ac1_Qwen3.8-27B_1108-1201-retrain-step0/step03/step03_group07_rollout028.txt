    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    c_max = c[top_idx[0]]
    if c_max < 1e-15:
        return np.zeros(n)
    if soft_T > 1e-10:
        w = np.exp(soft_T * (c[top_idx] - c_max) / max(c_max, 1e-15))
        w /= w.sum()
    else:
        w = np.full(top_k, 1.0 / top_k)
    w_full = np.zeros(len(c))
    w_full[top_idx] = w
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 2).bit_length()
    W = fft(w_full, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def solve_lp_direction(a, c, M, top_k_constr=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for top-k peaks, g >= 0."""
    n = len(a)
    top_k = min(top_k_constr, len(c))
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    A_ub = np.zeros((top_k, n))
    for r, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A_ub[r, i] = a[k - i]
    b_ub = M * np.ones(top_k)
    c_obj = -np.ones(n)
    bounds_list = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds_list,
                      method='highs', options={'time_limit': 5})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def mirror_descent_v2(a0, time_budget, eta0=0.3, top_k_start=200,
                      top_k_end=1, momentum=0.85, soft_T=0.0,
                      lp_every=0, lp_top_k=20,
                      stagnation_limit=1000, perturb_frac=0.05,
                      verbose=False, seed=42):
    """Mirror descent with adaptive peak chasing and optional LP direction."""
    rng = np.random.default_rng(seed)
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n)
        S = n
    a = np.maximum(a, 1e-15)
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    best_iter = 0
    v = np.zeros(n)
    eta = eta0
    iter_no = 0
    start = time.monotonic()
    last_lp_time = 0.0

    while time.monotonic() - start < time_budget:
        progress = min(1.0, iter_no / 8000.0)
        top_k_cur = max(1, int(top_k_start + (top_k_end - top_k_start) * progress))
        grad = compute_subgrad(c, a, top_k_cur, soft_T)

        # Periodic LP direction
        if lp_every > 0 and iter_no > 0 and iter_no % lp_every == 0:
            if time.monotonic() - last_lp_time > 2.0:
                M_cur = c.max()
                g_lp = solve_lp_direction(a, c, M_cur, lp_top_k)
                last_lp_time = time.monotonic()
                if g_lp is not None:
                    g_lp_centered = g_lp - g_lp.mean()
                    glp_n = np.linalg.norm(g_lp_centered)
                    gn = np.linalg.norm(grad)
                    if glp_n > 1e-15 and gn > 1e-15:
                        mix = 0.4
                        grad = (1.0 - mix) * (grad / gn) + mix * (g_lp_centered / glp_n)
                        grad /= np.linalg.norm(grad)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        # Backtracking line search
        cur_base = obj_value(a, c)
        accepted = False
        for bt in range(12):
            eta_try = eta * (0.7 ** bt)
            a_trial = a * np.exp(-eta_try * v)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)
            if obj_trial < cur_base * (1.0 - 1e-10):
                a = a_trial
                c = c_trial
                accepted = True
                break
        if not accepted:
            a = a * np.exp(-0.03 * eta * v)
            a = np.maximum(a, 1e-300)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)

        cur_obj = obj_value(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
            eta = min(eta * 1.015, eta0 * 4.0)
        else:
            eta = max(eta * 0.96, eta0 * 0.05)

        # Stagnation detection with targeted perturbation
        if iter_no - best_iter > stagnation_limit:
            # Focus perturbation on coordinates contributing to the peak
            k_peak = np.argmax(c)
            lo = max(0, k_peak - n + 1)
            hi = min(n - 1, k_peak)
            noise_scale = perturb_frac * S / np.sqrt(n)
            noise = rng.standard_normal(n) * noise_scale
            # Amplify noise at peak-contributing coordinates
            noise[lo:hi+1] *= 2.0
            a = a + noise
            a = np.maximum(a, 1e-15)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.1
            eta = eta0
            best_iter = iter_no
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    [perturb] iter={iter_no}, obj={cur_obj:.6f}, "
                      f"best={best_obj:.6f}, t={elapsed:.1f}s")

        eta *= 0.99998
        iter_no += 1
        if verbose and iter_no % 3000 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"eta={eta:.4f}, top_k={top_k_cur}, t={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_polish(a0, time_budget, max_random_trials=200):
    """Coordinate descent focusing on peak-contributing coordinates."""
    rng = np.random.default_rng(123)
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    no_improve = 0

    while time.monotonic() - start < time_budget and no_improve < 50:
        improved = False
        k_max = np.argmax(c)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        # Get peak-contributing coordinates sorted by contribution
        contrib = np.abs(a[lo:hi + 1])
        n_top = min(30, hi - lo + 1)
        top_local = np.argsort(contrib)[::-1][:n_top]
        top_idx = top_local + lo

        for i in top_idx:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            for factor in [0.6, 0.75, 0.85, 1.15, 1.25, 1.4]:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_norm = a * (S / s)
                c_new = fft_convolve_self(a_norm)
                obj_new = obj_value(a_norm, c_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_norm.copy()
                    a = a_norm
                    c = c_new
                    improved = True
                    no_improve = 0
                    break
                a[i] = orig
            # Renormalize
            s = a.sum()
            if s > 1e-10:
                a = a * (S / s)

        # Also try some random coordinates
        for _ in range(10):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            orig = a[i]
            if orig < 1e-12:
                continue
            for factor in [0.6, 0.8, 1.2, 1.5]:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_norm = a * (S / s)
                c_new = fft_convolve_self(a_norm)
                obj_new = obj_value(a_norm, c_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_norm.copy()
                    a = a_norm
                    c = c_new
                    improved = True
                    no_improve = 0
                    break
                a[i] = orig
            s = a.sum()
            if s > 1e-10:
                a = a * (S / s)

        if not improved:
            no_improve += 1

    return best_a, best_obj


def single_peak_chase(a0, time_budget, lr=0.01, momentum=0.9):
    """Aggressively chase the single worst peak."""
    rng = np.random.default_rng(99)
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 1e-15)
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    v = np.zeros(n)
    start = time.monotonic()
    stagn = 0

    while time.monotonic() - start < time_budget:
        k_max = np.argmax(c)
        c_max = c[k_max]

        # Gradient of c[k_max] w.r.t. a[i] = 2*a[k_max - i] for valid i
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[k_max - m]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v /= vnorm

        a = a * np.exp(-lr * v)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())
        c = fft_convolve_self(a)
        cur_obj = obj_value(a, c)

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            lr = min(lr * 1.02, 0.1)
            stagn = 0
        else:
            lr = max(lr * 0.95, 0.001)
            stagn += 1
            if stagn > 200:
                noise = rng.standard_normal(n) * 0.01 * S / np.sqrt(n)
                a = a + noise
                a = np.maximum(a, 1e-15)
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                v *= 0.3
                lr = 0.01
                stagn = 0

    return best_a, best_obj
