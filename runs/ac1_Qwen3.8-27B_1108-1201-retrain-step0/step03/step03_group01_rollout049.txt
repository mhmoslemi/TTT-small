Minimize 2*n*max_conv/sum^2 using:

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def mirror_descent_step(a, grad, step, S):
    """Multiplicative weights update: a_i * exp(-step * grad_i), renormalized."""
    # Clip grad to prevent overflow
    grad_clipped = np.clip(grad, -100.0, 100.0)
    log_ratios = -step * grad_clipped
    # Subtract max for numerical stability
    log_ratios -= log_ratios.max()
    weights = np.exp(log_ratios) * a
    total = weights.sum()
    if total < 1e-30:
        return np.ones(len(a)) * S / len(a)
    return weights * (S / total)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, c


def fista_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=500.0,
               use_mirror=False, noise_interval=300, noise_prob=0.15, noise_scale=0.003):
    """FISTA with LSE smoothing."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    last_print = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.02 * it))
        
        grad, c = lse_grad(y, beta)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()
        
        if it - last_print > 2000:
            print("    FISTA it=%d, beta=%.1f, score=%.6f" % (it, beta, best_score))
            last_print = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: LSE is L-smooth where L ~ 2*beta*||a||_inf roughly
        # Use conservative step
        a_max = a.max()
        L = 2.0 * beta * max(a_max, S / n) * 2.0
        step = 1.0 / (L + 1e-10)
        step = min(step, S / (n * 0.1))  # Don't make elements too negative before projection
        
        if use_mirror:
            x_new = mirror_descent_step(y, grad, step * gnorm, S)
        else:
            x_new = project_simplex(y - step * grad, S)
        
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_a = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Noise for escaping local minima
        if it % noise_interval == 0 and rng.random() < noise_prob:
            scale = noise_scale * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, scale, n), S)
            t = 1.0
    
    return best_a, best_score


def spectral_shaping_v2(a, N, top_k, alpha, n_iter, rng, S):
    """
    Enhanced spectral shaping: iterative multi-lag suppression.
    Each step identifies the top lags and applies a targeted spectral reduction.
    """
    n = len(a)
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    
    for it in range(n_iter):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = c.max()
        
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        
        # Find top k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        
        # For each top lag, compute the "interference" pattern in frequency domain
        m = np.arange(N)
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            w = top_vals[i] / total
            cos_eff += w * np.cos(2.0 * np.pi * m * int(k) / N)
        
        # Reduce power where cos > 0 (reinforces peak), increase where cos < 0
        # This is a "gradient descent" in the spectral domain on the peak
        P_new = P * (1.0 - alpha * cos_eff)
        P_new = np.maximum(P_new, 0.0)
        
        # Preserve phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        
        # Project
        a = project_simplex(a_new_full, S)
        
        # Occasional noise
        if it % 500 == 250 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
    
    return best_a, best_score


def basin_hopping(a_init, n, S, max_time, rng, inner_time=10.0):
    """Basin hopping: random perturbation followed by local optimization."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    start = time.monotonic()
    n_hops = 0
    
    while time.monotonic() - start < max_time:
        n_hops += 1
        # Perturb
        scale = (0.01 + 0.02 * np.exp(-n_hops / 50)) * S / np.sqrt(n)
        a_perturbed = project_simplex(a + rng.normal(0, scale, n), S)
        
        # Local optimization (short FISTA)
        a_opt, score = fista_lse(a_perturbed, n, S, min(inner_time, max_time / 20),
                                  rng, beta_init=20.0, beta_max=200.0,
                                  noise_interval=100, noise_prob=0.1, noise_scale=0.002)
        
        c_opt = fft_convolve_self(a_opt)
        if c_opt.max() < best_score:
            best_score = c_opt.max()
            best_a = a_opt.copy()
            a = a_opt.copy()
            print("    Basin hop %d: %.6f" % (n_hops, best_score))
    
    return best_a, best_score

def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def project_simplex(v, S):

    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def mirror_step(a, grad, step, S):
    """Multiplicative weights update keeping non-negativity and sum."""
    grad_clipped = np.clip(grad, -50.0, 50.0)
    log_ratios = -step * grad_clipped
    log_ratios -= log_ratios.max()
    weights = np.exp(log_ratios) * a
    total = weights.sum()
    if total < 1e-30:
        return np.ones(len(a)) * S / len(a)
    return weights * (S / total)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, beta):
    """Gradient of LSE(convolve(a,a), beta). Returns (grad, c)."""
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, c


def fista_lse_opt(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=500.0,
                  use_mirror=False, noise_interval=400, noise_prob=0.15,
                  noise_scale=0.003, step_factor=2.0):
    """FISTA with LSE smoothing and adaptive step."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    last_print = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.015 * it))

        grad, c = lse_gradient(y, beta)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()

        if it - last_print > 3000:
            print("      FISTA it=%d beta=%.1f best=%.6f" % (it, beta, best_score))
            last_print = it

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: normalize gradient, scale by S/n
        step = step_factor * S / (n * (1.0 + 0.005 * it))
        grad_normed = grad / max(gnorm, 1e-15)

        if use_mirror:
            x_new = mirror_step(y, grad, step * 0.5, S)
        else:
            x_new = project_simplex(y - step * grad_normed, S)

        if it % 4 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_a = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % noise_interval == 0 and rng.random() < noise_prob:
            scale = noise_scale * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, scale, n), S)
            t = 1.0

    return best_a, best_score


def spectral_shaping_enhanced(a, N, top_k, alpha, n_iter, rng, S):
    """
    Enhanced spectral shaping with adaptive multi-lag targeting.
    Uses a "water-filling" approach in the spectral domain.
    """
    n = len(a)
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()

    for it in range(n_iter):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = c.max()

        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        # Target top lags with weighted sum
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        # Use exponential weighting (heavier on the peak)
        top_vals_norm = top_vals / top_vals[0]
        weights = np.exp(-0.5 * (top_vals_norm - 1.0))
        weights = weights / weights.sum()

        m = np.arange(N)
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

        # Adaptive alpha: larger when far from flat
        target = S * S / (2 * n)
        excess = max(0.0, c_max - target) / max(c_max, 1e-10)
        alpha_curr = alpha * (0.2 + 0.8 * excess)

        P_new = P * (1.0 - alpha_curr * cos_eff)
        P_new = np.maximum(P_new, 1e-30)

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)

        # Periodic noise injection
        if it % 300 == 150 and rng.random() < 0.25:
            scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, scale, n), S)

    return best_a, best_score


def basin_hopping_opt(a_init, n, S, max_time, rng, inner_time=5.0):
    """Basin hopping with adaptive perturbation scale."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    start = time.monotonic()
    n_hops = 0

    while time.monotonic() - start < max_time:
        n_hops += 1
        scale = (0.005 + 0.03 * np.exp(-n_hops / 30.0)) * S / np.sqrt(n)
        a_pert = project_simplex(a + rng.normal(0, scale, n), S)

        a_opt, _ = fista_lse_opt(a_pert, n, S, min(inner_time, max_time / 15),
                                  rng, beta_init=30.0, beta_max=300.0,
                                  noise_interval=200, noise_prob=0.1,
                                  noise_scale=0.002)
        c_opt = fft_convolve_self(a_opt)
        if c_opt.max() < best_score:
            best_score = c_opt.max()
            best_a = a_opt.copy()
            a = a_opt.copy()
            print("      Basin hop %d: %.6f" % (n_hops, best_score))

        # Also try a larger jump every few hops
        if n_hops % 5 == 0:
            scale_big = 0.1 * S / np.sqrt(n)
            a_big = project_simplex(a + rng.normal(0, scale_big, n), S)
            a_big_opt, _ = fista_lse_opt(a_big, n, S, min(inner_time, max_time / 15),
                                          rng, beta_init=20.0, beta_max=200.0,
                                          noise_interval=300, noise_prob=0.1,
                                          noise_scale=0.002)
            c_big = fft_convolve_self(a_big_opt)
            if c_big.max() < best_score:
                best_score = c_big.max()
                best_a = a_big_opt.copy()
                a = a_big_opt.copy()
                print("      Big hop %d: %.6f" % (n_hops, best_score))

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def get_context_length():
    """Get sequence length from context if available."""
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            return len(given)
    except Exception:
        pass
    return 8192


def diversified_inits(n, rng):
    """Create a set of diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    inits.append(("uniform", np.ones(n)))
    inits.append(("rand", rng.random(n) + 0.1))
    inits.append(("chirp", 0.5 + 0.4 * np.sin(2 * np.pi * (4 * tt + 0.5 * tt * tt))))
    inits.append(("multi_sin", 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) +
                  0.2 * np.sin(2 * np.pi * 7 * tt) + 0.1 * np.sin(2 * np.pi * 11 * tt)))
    inits.append(("bump", np.exp(-2.0 * (tt - 0.3) ** 2) + 0.3 * np.exp(-2.0 * (tt - 0.7) ** 2) + 0.05))

    # Segment-based
    a_seg = np.zeros(n)
    a_seg[:n // 3] = 1.0
    a_seg[n // 3:2 * n // 3] = 0.2
    a_seg[2 * n // 3:] = 0.6
    inits.append(("seg3", a_seg))

    # Alternating
    a_alt = np.ones(n)
    a_alt[1::2] = 0.4
    inits.append(("alt", a_alt))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Multi-phase: FISTA-LSE + spectral shaping + basin hopping with adaptive switching.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    n = get_context_length()
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d S=%.4f FFTsize=%d budget=%.0fs" % (n, S, N, budget_s))

    # Try to use previous sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0.01:
            a_given = safe_normalize(given, n)
            c_given = fft_convolve_self(a_given)
            global_best = c_given.max()
            global_a = a_given.copy()
            print("  Context sequence: score=%.6f" % global_best)
    except Exception:
        pass

    if global_a is None:
        global_a = safe_normalize(np.ones(n), n)
        global_best = fft_convolve_self(global_a).max()
        print("  Default uniform: score=%.6f" % global_best)

    # === Phase 1: Aggressive FISTA-LSE ===
    print("=== Phase 1: FISTA-LSE (600s max) ===")
    t0 = time.monotonic()
    p1_time = min(600.0, (deadline - t0) * 0.25)

    # First run: simplex FISTA with moderate beta
    a1, _ = fista_lse_opt(global_a, n, S, p1_time * 0.5, rng,
                           beta_init=5.0, beta_max=150.0, use_mirror=False,
                           noise_interval=500, noise_prob=0.2, noise_scale=0.003)
    a1 = safe_normalize(a1, n)
    c1 = fft_convolve_self(a1)
    if c1.max() < global_best:
        global_best = c1.max()
        global_a = a1.copy()
        print("  P1a simplex: %.6f" % global_best)

    # Second run: mirror descent from current best
    rem = p1_time - (time.monotonic() - t0)
    if rem > 10:
        a2, _ = fista_lse_opt(global_a, n, S, rem, rng,
                               beta_init=20.0, beta_max=300.0, use_mirror=True,
                               noise_interval=300, noise_prob=0.15, noise_scale=0.002)
        a2 = safe_normalize(a2, n)
        c2 = fft_convolve_self(a2)
        if c2.max() < global_best:
            global_best = c2.max()
            global_a = a2.copy()
            print("  P1b mirror: %.6f" % global_best)

    # === Phase 2: Alternating SS + FISTA ===
    print("=== Phase 2: Alternating (600s max) ===")
    t1 = time.monotonic()
    p2_time = min(600.0, (deadline - t1) * 0.3)

    a_curr = global_a.copy()
    outer = 0
    stall = 0

    while time.monotonic() < t1 + p2_time:
        outer += 1
        prev_best = global_best

        # Spectral shaping with growing top_k
        top_k_ss = min(10 + 5 * outer, 60)
        alpha_ss = 0.25 / (1.0 + 0.08 * outer)
        n_ss = min(4000, max(1000, int(45.0 / 0.001)))
        a_ss, _ = spectral_shaping_enhanced(a_curr, N, top_k=top_k_ss,
                                             alpha=alpha_ss, n_iter=n_ss,
                                             rng=rng, S=S)
        a_ss = safe_normalize(a_ss, n)
        c_ss = fft_convolve_self(a_ss)
        if c_ss.max() < global_best:
            global_best = c_ss.max()
            global_a = a_ss.copy()
            a_curr = a_ss.copy()
            stall = 0
            if outer % 3 == 0:
                print("  [P2-%d] SS k=%d: %.6f" % (outer, top_k_ss, global_best))
        else:
            stall += 1

        # FISTA with higher beta
        rem2 = t1 + p2_time - time.monotonic()
        if rem2 > 15:
            fista_dur = min(rem2 * 0.4, 80.0)
            beta_start = 30.0 + 15.0 * outer
            use_mir = (outer % 3 == 0)
            a_f, _ = fista_lse_opt(a_curr, n, S, fista_dur, rng,
                                    beta_init=beta_start, beta_max=800.0,
                                    use_mirror=use_mir,
                                    noise_interval=200, noise_prob=0.12,
                                    noise_scale=0.002)
            a_f = safe_normalize(a_f, n)
            c_f = fft_convolve_self(a_f)
            if c_f.max() < global_best:
                global_best = c_f.max()
                global_a = a_f.copy()
                a_curr = a_f.copy()
                stall = 0
                print("  [P2-%d] FITA: %.6f" % (outer, global_best))

        # Basin hop when stalled
        if stall >= 3:
            rem3 = t1 + p2_time - time.monotonic()
            if rem3 > 40:
                print("  [P2-%d] Stalled -> basin hop" % outer)
                a_bh, _ = basin_hopping_opt(a_curr, n, S, min(rem3 * 0.35, 120.0),
                                             rng, inner_time=5.0)
                a_bh = safe_normalize(a_bh, n)
                c_bh = fft_convolve_self(a_bh)
                if c_bh.max() < global_best:
                    global_best = c_bh.max()
                    global_a = a_bh.copy()
                    a_curr = a_bh.copy()
                    print("  [P2-%d] BH: %.6f" % (outer, global_best))
                else:
                    # Try one more: large perturbation + quick optimize
                    a_big = project_simplex(a_curr + rng.normal(0, 0.05 * S / np.sqrt(n), n), S)
                    a_big_opt, _ = fista_lse_opt(a_big, n, S, 20.0, rng,
                                                  beta_init=50.0, beta_max=400.0,
                                                  noise_interval=500, noise_prob=0.1,
                                                  noise_scale=0.002)
                    a_big_opt = safe_normalize(a_big_opt, n)
                    c_big = fft_convolve_self(a_big_opt)
                    if c_big.max() < global_best:
                        global_best = c_big.max()
                        global_a = a_big_opt.copy()
                        a_curr = a_big_opt.copy()
                        print("  [P2-%d] BigBH: %.6f" % (outer, global_best))
            stall = 0

    # === Phase 3: Fine polishing ===
    print("=== Phase 3: Polish ===")
    t2 = time.monotonic()
    remaining = deadline - t2
    print("  Remaining: %.0fs" % remaining)

    if global_a is not None and remaining > 10:
        # Very high beta FISTA (peak-finding mode)
        seg = min(remaining * 0.3, 150.0)
        a_p1, _ = fista_lse_opt(global_a, n, S, seg, rng,
                                 beta_init=200.0, beta_max=2000.0,
                                 use_mirror=False,
                                 noise_interval=1500, noise_prob=0.08,
                                 noise_scale=0.001)
        a_p1 = safe_normalize(a_p1, n)
        c_p1 = fft_convolve_self(a_p1)
        if c_p1.max() < global_best:
            global_best = c_p1.max()
            global_a = a_p1.copy()
            print("  P3 high-beta: %.6f" % global_best)

        # Fine spectral shaping (very small alpha)
        rem3 = deadline - time.monotonic()
        if rem3 > 5:
            n_iters = min(int(rem3 * 0.3 / 0.001), 15000)
            a_p2, _ = spectral_shaping_enhanced(global_a, N, top_k=40,
                                                 alpha=0.03, n_iter=max(n_iters, 1000),
                                                 rng=rng, S=S)
            a_p2 = safe_normalize(a_p2, n)
            c_p2 = fft_convolve_self(a_p2)
            if c_p2.max() < global_best:
                global_best = c_p2.max()
                global_a = a_p2.copy()
                print("  P3 fine SS: %.6f" % global_best)

        # Final mirror descent polish
        rem4 = deadline - time.monotonic()
        if rem4 > 10:
            a_p3, _ = fista_lse_opt(global_a, n, S, rem4, rng,
                                     beta_init=500.0, beta_max=5000.0,
                                     use_mirror=True,
                                     noise_interval=3000, noise_prob=0.05,
                                     noise_scale=0.0005)
            a_p3 = safe_normalize(a_p3, n)
            c_p3 = fft_convolve_self(a_p3)
            if c_p3.max() < global_best:
                global_best = c_p3.max()
                global_a = a_p3.copy()
                print("  P3 mirror polish: %.6f" % global_best)

    # === Verification ===
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            if np.isfinite(ver):
                print("Verification score: %.6f" % ver)
            else:
                print("Verification FAILED, using fallback")
                global_a = np.ones(256, dtype=np.float64)
        except Exception:
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Final: length=%d internal=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
