"""
Minimize 2*n*max_conv/sum^2 using frequency-domain flattening (power compression)
with adaptive annealing, multi-start, and gradient polishing.

Key novelty: Instead of spectral shaping (which modifies spectrum based on
convolution peaks), we directly flatten the power spectrum using power
compression |A| -> |A|^p (p<1), which is equivalent to the error reduction
algorithm from phase retrieval adapted to the non-negative constraint.
"""

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def flatten_step(a, p, S, N):
    """One step of spectrum flattening via power compression."""
    A = fft(a, N)
    mag = np.abs(A)
    # Avoid log(0)
    mag_safe = np.maximum(mag, 1e-30)
    # Power compression: p < 1 flattens
    mag_new = mag_safe ** p
    # Preserve phase
    A_new = mag_new * np.exp(1j * np.angle(A))
    # Inverse FFT
    a_raw = np.real(ifft(A_new))[:len(a)]
    # Project to non-negative with sum S
    a_new = project_simplex(a_raw, S)
    return a_new

"""
Minimize 2*n*max_conv/sum^2 using adaptive frequency-domain power compression
(spectrum flattening) with over-relaxation, combined with FISTA-LSE and 
Nesterov top-k subgradient polishing. Multi-start with diverse initializations.

Key novelty: Directly flatten |FFT(a)| via power compression |A|^p (p<1) with
adaptive p based on current spectral flatness, plus over-relaxation to accelerate
convergence. This is simpler and more principled than the previous spectral
shaping approach.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def safe_normalize(a, n, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def flatten_step_raw(a, p, N):
    """Flattening step without projection: |A|^p with phase preserved."""
    A = fft(a, N)
    mag = np.abs(A)
    mag_safe = np.maximum(mag, 1e-30)
    mag_new = mag_safe ** p
    A_new = mag_new * np.exp(1j * np.angle(A))
    a_raw = np.real(ifft(A_new))[:len(a)]
    return a_raw


def optimize_flatten(a_init, S, N, max_time, rng, p_min=0.15, p_max=0.95):
    """
    Main spectrum-flattening optimizer with adaptive p and over-relaxation.
    Returns (best_a, best_score) where score = max of self-convolution.
    """
    a = safe_normalize(a_init, len(a_init), S)
    n = len(a)
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()

    c = fft_convolve_self(a)
    best_score = c.max()

    it = 0
    while time.monotonic() - start < max_time:
        it += 1

        # Compute spectral flatness
        A = fft(a, N)
        mag = np.abs(A)
        mag_mean = mag.mean()
        mag_max = mag.max()
        if mag_max < 1e-30:
            flatness = 1.0
        else:
            flatness = mag_mean / mag_max

        # Adaptive p: more aggressive when spectrum is less flat
        # Map flatness in [0,1] to p in [p_min, p_max]
        # When flatness is low (spectrum peaked), use small p (aggressive)
        p = p_min + (p_max - p_min) * (1.0 - (1.0 - flatness) ** 0.5)
        p = np.clip(p, p_min, p_max)

        # Over-relaxation: stronger when not converged
        omega = 1.0 + 0.5 * (1.0 - flatness) ** 1.5

        # Flattening step
        a_flat_raw = flatten_step_raw(a, p, N)
        a_proj = project_simplex(a_flat_raw, S)

        # Over-relaxation
        if omega > 1.0:
            a_new = (1.0 + omega) * a_proj - omega * a
            a = project_simplex(a_new, S)
        else:
            a = a_proj

        # Random perturbation to escape local minima
        if it % 300 == 150 and rng.random() < 0.25:
            noise_level = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_level, n), S)

        # Evaluate periodically
        if it % 3 == 0:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_a = a.copy()

        if time.monotonic() - start > max_time - 0.05:
            break

    c_final = fft_convolve_self(a)
    if c_final.max() < best_score:
        best_score = c_final.max()
        best_a = a.copy()

    return best_a, best_score


def fista_lse(a_init, S, N, max_time, rng, beta_start=5.0, beta_max=500.0):
    """FISTA with log-sum-exp smooth approximation of max."""
    a = safe_normalize(a_init, len(a_init), S)
    n = len(a)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    beta = beta_start
    it = 0

    c = fft_convolve_self(y)
    best_score = c.max()

    while time.monotonic() - start < max_time:
        it += 1

        # Grow beta slowly
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.005)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        c_max = c.max()
        # LSE: (1/beta) * log(sum(exp(beta * (c - c_max)))) + c_max
        # Gradient: w * (correlation with y) where w = softmax(beta*(c-c_max))
        scaled = beta * (c - c_max) / max(n, 1)
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        total = exp_scaled.sum()
        if total < 1e-300:
            break
        w = exp_scaled / total

        # Gradient of LSE w.r.t. a: 2 * (w correlated with y)
        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: L-smoothness of LSE is related to max |w| and ||y||
        step = n / (beta * max(np.max(w) * (2.0 * S), 1e-10) * (1.0 + 0.002 * it))
        step = np.clip(step, 1e-8, 2.0 * S / n)
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Occasional restart
        if it % 1500 == 0 and rng.random() < 0.12:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best_score:
        best_score = c_final.max()
        best_local = a.copy()

    return best_local, best_score


def nesterov_subgrad(a_init, S, N, max_time, rng, top_k_init=20, top_k_max=500):
    """Nesterov-accelerated top-k subgradient method."""
    a = safe_normalize(a_init, len(a_init), S)
    n = len(a)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    c = fft_convolve_self(y)
    best_score = c.max()

    while time.monotonic() - start < max_time:
        it += 1
        cur_k = min(2 * n - 1, top_k_init + it // 30)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:cur_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_val = t_new

        # Restart
        if it % 400 == 0 and rng.random() < 0.15:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t_val = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best_score:
        best_score = c_final.max()
        best_local = a.copy()

    return best_local, best_score


def power_mean_smooth_obj(a, S, N, p_power):
    """
    Compute the p-power-mean of the convolution as a smooth objective.
    Returns (objective_value, gradient).
    """
    n = len(a)
    c = fft_convolve_self(a)
    c_pos = np.maximum(c, 0.0)
    pm = (np.sum(c_pos ** p_power) / len(c)) ** (1.0 / p_power)

    # Gradient: d/d a_j sum_k c[k]^p / len(c) = (p/len(c)) * sum_k c[k]^(p-1) * dc[k]/da[j]
    # dc[k]/da[j] = a[k-j] + a[j-k] (with bounds) = 2 * (w * a)[j] where w[k]=c[k]^(p-1)
    w = c_pos ** (p_power - 1.0)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = (2.0 * p_power / len(c)) * np.real(ifft(W * np.conj(A)))[:n]

    return pm, grad


def power_mean_optimize(a_init, S, N, max_time, rng, p_start=8.0, p_end=200.0):
    """
    Optimize using power-mean smooth objective with FISTA.
    As p increases, the objective approaches the max.
    """
    a = safe_normalize(a_init, len(a_init), S)
    n = len(a)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    c = fft_convolve_self(y)
    best_score = c.max()

    while time.monotonic() - start < max_time:
        it += 1

        # Grow p over time
        frac = min(1.0, it / 2000.0)
        p_power = p_start + (p_end - p_start) * frac

        pm, grad = power_mean_smooth_obj(y, S, N, p_power)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step
        step = 1.0 / (max(p_power * 2.0 * S / n, 1e-6) * (1.0 + 0.003 * it))
        step = np.clip(step, 1e-9, 5.0 * S / n)
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 2000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best_score:
        best_score = c_final.max()
        best_local = a.copy()

    return best_local, best_score


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Multi-start with spectrum flattening + gradient polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.5, budget_s - 15.0)

    global_best = np.inf
    global_a = None

    # Determine length
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            n = len(given)
        else:
            n = 8192
    except Exception:
        n = 8192

    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    # --- Build initializations ---
    inits = []

    try:
        given_arr = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given_arr.size == n and np.sum(np.clip(given_arr, 0, 1000)) > 1e-10:
            inits.append(("given", np.clip(given_arr, 0, 1000).astype(np.float64)))
    except Exception:
        pass

    tt = np.linspace(0, 1, n, endpoint=False)
    inits.append(("rand", rng.random(n) + 0.05))
    inits.append(("chirp3", 0.5 + 0.4 * np.sin(2 * np.pi * (3 * tt + 0.5 * tt * tt)) + 0.1 * rng.standard_normal(n)))
    inits.append(("chirp7", 0.5 + 0.3 * np.sin(2 * np.pi * (7 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)))
    inits.append(("multi_sin", 1.0 + 0.3 * np.sin(2 * np.pi * 5 * tt) + 0.2 * np.sin(2 * np.pi * 11 * tt) + 0.1 * np.sin(2 * np.pi * 23 * tt)))
    inits.append(("bump", np.exp(-0.5 * ((tt - 0.5) * 4.0) ** 2) + 0.08))
    inits.append(("cosine", 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.08))
    inits.append(("triangle", 1.0 - 2.0 * np.abs(tt - 0.5) + 0.08))

    # Binary-ish (m-sequence inspired)
    bin_seq = (rng.random(n) > 0.5).astype(float)
    bin_seq = 0.1 + 0.7 * bin_seq
    inits.append(("binary", bin_seq))

    # Smoothed random (Gaussian-kernel smoothed)
    a_smooth = rng.random(n)
    # Simple smoothing via FFT lowpass
    A_s = fft(a_smooth, N)
    f = np.fft.fftfreq(N)
    lowpass = np.exp(-20.0 * (np.abs(f) * 2 * n / N) ** 2)
    A_s *= lowpass
    a_smooth = np.abs(np.real(ifft(A_s))[:n]) + 0.05
    inits.append(("smoothed_rand", a_smooth))

    n_inits = len(inits)
    print("  %d initializations" % n_inits)

    # --- Phase 1: Multi-start exploration ---
    # Allocate time: 50% exploration, 35% focused, 15% final polish
    t_start = time.monotonic()
    phase1_budget = (deadline - t_start) * 0.45
    per_init = phase1_budget / n_inits

    print("=== Phase 1: Exploration (%.0f sec total) ===" % phase1_budget)

    best_for_phase1 = np.inf
    best_a_phase1 = None

    for name, arr in inits:
        if time.monotonic() > deadline - 120:
            break

        a_norm = safe_normalize(arr, n, S)

        # Quick initial evaluation
        c_init = fft_convolve_self(a_norm)
        init_score = c_init.max()

        t0 = time.monotonic()

        # Flattening optimization
        flat_time = min(per_init * 0.55, 80.0)
        a_flat, score_flat = optimize_flatten(a_norm, S, N, flat_time, rng,
                                               p_min=0.15, p_max=0.92)
        a_flat = safe_normalize(a_flat, n, S)
        c_flat = fft_convolve_self(a_flat)
        sc = c_flat.max()

        if sc < best_for_phase1:
            best_for_phase1 = sc
            best_a_phase1 = a_flat.copy()
        if sc < global_best:
            global_best = sc
            global_a = a_flat.copy()
            print("  [%s] Flatten: %.6f (t=%.0fs)" % (name, global_best, time.monotonic() - t0))

        # Quick FISTA polish
        rem_fista = max(2.0, per_init * 0.25)
        rem_fista = min(rem_fista, deadline - time.monotonic() - 60)
        if rem_fista > 2:
            a_f, score_f = fista_lse(a_flat, S, N, rem_fista, rng,
                                      beta_start=5.0, beta_max=300.0)
            a_f = safe_normalize(a_f, n, S)
            c_f = fft_convolve_self(a_f)
            if c_f.max() < global_best:
                global_best = c_f.max()
                global_a = a_f.copy()
                print("  [%s] FISTA: %.6f" % (name, global_best))
            if c_f.max() < best_for_phase1:
                best_for_phase1 = c_f.max()
                best_a_phase1 = a_f.copy()

        # Quick Nesterov
        rem_ns = max(2.0, per_init * 0.15)
        rem_ns = min(rem_ns, deadline - time.monotonic() - 60)
        if rem_ns > 2 and best_a_phase1 is not None:
            a_ns, score_ns = nesterov_subgrad(best_a_phase1, S, N, rem_ns, rng,
                                               top_k_init=20, top_k_max=200)
            a_ns = safe_normalize(a_ns, n, S)
            c_ns = fft_convolve_self(a_ns)
            if c_ns.max() < global_best:
                global_best = c_ns.max()
                global_a = a_ns.copy()
                print("  [%s] Nesterov: %.6f" % (name, global_best))

    print("Phase 1 best: %.6f" % global_best)

    # --- Phase 2: Focused optimization on best candidate ---
    print("=== Phase 2: Focused optimization ===")
    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = 2.0

    t_phase2 = time.monotonic()
    phase2_budget = (deadline - t_phase2) * 0.40

    a_curr = global_a.copy()
    stall_count = 0
    last_best = global_best
    it_outer = 0

    while time.monotonic() < t_phase2 + phase2_budget:
        it_outer += 1

        # Flattening with decreasing aggressiveness
        rem = (t_phase2 + phase2_budget) - time.monotonic()
        if rem < 5:
            break

        flat_budget = min(rem * 0.4, 60.0)
        p_min_curr = 0.3 + 0.1 * min(it_outer, 5)
        p_max_curr = 0.95
        a_new, sc = optimize_flatten(a_curr, S, N, flat_budget, rng,
                                      p_min=p_min_curr, p_max=p_max_curr)
        a_new = safe_normalize(a_new, n, S)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < global_best:
            global_best = c_new.max()
            global_a = a_new.copy()
            a_curr = a_new.copy()
            last_best = global_best
            stall_count = 0
            if it_outer % 3 == 0:
                print("  [P2:%d] Flatten: %.6f" % (it_outer, global_best))
        else:
            stall_count += 1

        # FISTA with increasing beta
        rem2 = (t_phase2 + phase2_budget) - time.monotonic()
        if rem2 > 5:
            fista_budget = min(rem2 * 0.3, 45.0)
            beta_start = 10.0 + 5.0 * it_outer
            a_f2, _ = fista_lse(a_curr, S, N, fista_budget, rng,
                                 beta_start=min(beta_start, 200.0), beta_max=1000.0)
            a_f2 = safe_normalize(a_f2, n, S)
            c_f2 = fft_convolve_self(a_f2)
            if c_f2.max() < global_best:
                global_best = c_f2.max()
                global_a = a_f2.copy()
                a_curr = a_f2.copy()
                last_best = global_best
                stall_count = 0
                print("  [P2:%d] FISTA: %.6f" % (it_outer, global_best))
            else:
                stall_count += 1

        # Power-mean optimization
        rem3 = (t_phase2 + phase2_budget) - time.monotonic()
        if rem3 > 5:
            pm_budget = min(rem3 * 0.2, 30.0)
            p_start_pm = 16.0 + 8.0 * it_outer
            a_pm, _ = power_mean_optimize(a_curr, S, N, pm_budget, rng,
                                           p_start=min(p_start_pm, 100.0), p_end=300.0)
            a_pm = safe_normalize(a_pm, n, S)
            c_pm = fft_convolve_self(a_pm)
            if c_pm.max() < global_best:
                global_best = c_pm.max()
                global_a = a_pm.copy()
                a_curr = a_pm.copy()
                last_best = global_best
                stall_count = 0
                print("  [P2:%d] PowerMean: %.6f" % (it_outer, global_best))
            else:
                stall_count += 1

        # Nesterov
        rem4 = (t_phase2 + phase2_budget) - time.monotonic()
        if rem4 > 5:
            ns_budget = min(rem4 * 0.15, 20.0)
            tk = 30 + 20 * it_outer
            a_ns2, _ = nesterov_subgrad(a_curr, S, N, ns_budget, rng,
                                         top_k_init=tk, top_k_max=tk + 200)
            a_ns2 = safe_normalize(a_ns2, n, S)
            c_ns2 = fft_convolve_self(a_ns2)
            if c_ns2.max() < global_best:
                global_best = c_ns2.max()
                global_a = a_ns2.copy()
                a_curr = a_ns2.copy()
                last_best = global_best
                stall_count = 0
                print("  [P2:%d] Nesterov: %.6f" % (it_outer, global_best))
            else:
                stall_count += 1

        # Restart if stalled
        if stall_count >= 4:
            print("  [P2:%d] Stalled (last best=%.6f), perturbing" % (it_outer, last_best))
            a_curr = project_simplex(a_curr + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            stall_count = 0

    print("Phase 2 best: %.6f" % global_best)

    # --- Phase 3: Final polish ---
    print("=== Phase 3: Final polish ===")
    t_phase3 = time.monotonic()
    remaining = deadline - t_phase3

    if global_a is not None and remaining > 5:
        # High-beta FISTA
        seg1 = min(remaining * 0.3, 200.0)
        if seg1 > 5:
            a_p1, _ = fista_lse(global_a, S, N, seg1, rng,
                                 beta_start=100.0, beta_max=2000.0)
            a_p1 = safe_normalize(a_p1, n, S)
            c1 = fft_convolve_self(a_p1)
            if c1.max() < global_best:
                global_best = c1.max()
                global_a = a_p1.copy()
                print("  Final FISTA high-beta: %.6f" % global_best)

        # Long flattening with very gentle p
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            a_p2, _ = optimize_flatten(global_a, S, N, min(rem2 * 0.3, 150.0), rng,
                                        p_min=0.7, p_max=0.99)
            a_p2 = safe_normalize(a_p2, n, S)
            c2 = fft_convolve_self(a_p2)
            if c2.max() < global_best:
                global_best = c2.max()
                global_a = a_p2.copy()
                print("  Final gentle flatten: %.6f" % global_best)

        # High-k Nesterov
        rem3 = deadline - time.monotonic()
        if rem3 > 5:
            a_p3, _ = nesterov_subgrad(global_a, S, N, min(rem3 * 0.3, 100.0), rng,
                                        top_k_init=100, top_k_max=1024)
            a_p3 = safe_normalize(a_p3, n, S)
            c3 = fft_convolve_self(a_p3)
            if c3.max() < global_best:
                global_best = c3.max()
                global_a = a_p3.copy()
                print("  Final Nesterov high-k: %.6f" % global_best)

        # Very high p power-mean
        rem4 = deadline - time.monotonic()
        if rem4 > 5:
            a_p4, _ = power_mean_optimize(global_a, S, N, min(rem4, 60.0), rng,
                                           p_start=100.0, p_end=500.0)
            a_p4 = safe_normalize(a_p4, n, S)
            c4 = fft_convolve_self(a_p4)
            if c4.max() < global_best:
                global_best = c4.max()
                global_a = a_p4.copy()
                print("  Final power-mean: %.6f" % global_best)

    # --- Verification ---
    print("=== Verification ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified score: %.6f" % ver)
        else:
            print("Verification FAILED (inf), using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)
        print("No valid candidate, using fallback ones(256)")

    print("Returning sequence of length %d" % len(global_a))
    return [float(x) for x in global_a]
