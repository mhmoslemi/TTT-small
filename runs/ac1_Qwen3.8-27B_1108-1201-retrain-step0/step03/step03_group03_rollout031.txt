"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using spectral shaping + FISTA + active-set polishing.
Key novel ideas: spectral flattening via DFT magnitude control, adaptive restart, and 
multi-scale exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def lse_gradient(a, c, beta):
    """Gradient of log-sum-exp proxy for max(convolutions)."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def hard_topk_gradient(a, c, top_k):
    """Gradient using top-k lags with uniform weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    w_full = np.zeros(2 * n - 1)
    w_full[top_indices] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def spectral_shaping(a, n, S, max_time, rng, target_flatness=0.3):
    """
    Iteratively flatten the DFT magnitude while maintaining a >= 0 and sum(a) = S.
    This is inspired by the Griffin-Lim algorithm / spectral equalization.
    """
    a = np.clip(a, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    fsize = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        A = fft(a, fsize)
        
        # Target: make |A| as flat as possible
        # Use a geometric mean to smooth
        mag = np.abs(A)
        # Exclude DC (index 0) from flattening since it's fixed
        mag_target = np.ones_like(mag)
        mag_nonzero = mag[1:]
        # Target magnitude for non-DC bins
        target = np.sqrt(np.mean(mag_nonzero ** 2))  # RMS
        # Interpolate between current and target
        mag_new = mag.copy()
        mag_new[1:] = (1 - target_flatness) * mag[1:] + target_flatness * target
        mag_new[0] = mag[0]  # Keep DC
        
        # Reconstruct with new magnitude, same phase
        phase = A / np.where(np.abs(A) > 1e-30, np.abs(A), 1.0)
        A_new = mag_new * phase
        a_new = np.real(ifft(A_new, fsize))[:n]
        
        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / sn)
        
        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a_new.copy()
        
        a = a_new
    
    return best_local, best_score

def coordinate_descent_polish(a_init, n, S, max_time, rng):
    """
    Coordinate descent: for each coordinate, try to optimally adjust it
    while keeping the sum fixed (by adjusting another coordinate).
    Focus on coordinates that participate in the peak lags.
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        # Find the peak lag
        peak_lag = int(np.argmax(c))
        
        # Find indices that contribute to the peak
        i_min = max(0, peak_lag - (n - 1))
        i_max = min(peak_lag, n - 1)
        contributor_indices = np.arange(i_min, i_max + 1)
        
        # Try adjusting the largest contributor
        if len(contributor_indices) > 0:
            # Pick a random contributor
            idx = contributor_indices[rng.integers(len(contributor_indices))]
            partner = peak_lag - idx
            
            # Try reducing a[idx] (and a[partner]) by moving mass elsewhere
            amount = 0.01 * a[idx]
            if amount > 1e-10:
                # Move from idx to a random non-contributor
                non_contrib = np.setdiff1d(np.arange(n), 
                                           np.arange(max(0, peak_lag - (n-1)), min(peak_lag, n-1) + 1))
                if len(non_contrib) > 0:
                    target = non_contrib[rng.integers(len(non_contrib))]
                    a[idx] -= amount
                    a[target] += amount
                    a = np.maximum(a, 0)
                    sn = a.sum()
                    if sn > 0:
                        a *= S / sn
                    c_new = fft_convolve_self(a)
                    if c_new.max() < best_score:
                        best_score = c_new.max()
                        best_local = a.copy()
                        c = c_new
                    else:
                        # Revert
                        a[idx] += amount
                        a[target] -= amount
    
    return best_local, best_local  # Return best

def make_symmetric_init(n, rng, S):
    """Create a symmetric (palindromic) initialization."""
    half = (n + 1) // 2
    h = rng.random(half) + 0.05
    a = np.zeros(n)
    a[:half] = h
    a[n-1:(n-1-half):-1] = h[:n-1-half+1]  # Mirror
    # Actually let me be more careful
    a = np.zeros(n)
    for i in range(n):
        a[i] = h[min(i, n-1-i)]
    a = a * (S / a.sum())
    return a

def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.05
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    a = a * (S / a.sum())
    return a

def make_periodic_init(n, rng, S, num_freqs=4):
    """Periodic init with multiple frequencies."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 30)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.35
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.05)
    a = a * (S / a.sum())
    return a

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def weighted_topk_gradient(a, c, top_k, temperature=1.0):
    """Gradient using top-k lags with softmax temperature weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices].copy()
    tv_max = top_vals.max()
    tv_range = tv_max - top_vals.min()
    if tv_range > 1e-12:
        weights = np.exp((top_vals - tv_max) / (temperature * tv_range))
    else:
        weights = np.ones(k)
    weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fista_adaptive(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=500.0,
                   restart_period=500, restart_prob=0.4):
    """FISTA with adaptive beta, backtracking, and simulated-annealing restarts."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = beta_init * 4.0
    stall_count = 0
    last_best = np.inf
    annealing_temp = 0.05

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.02 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = y.copy()
            stall_count = 0
            last_best = cur
        elif it % 10 == 0:
            stall_count += 1

        # Annealing restart
        if stall_count > restart_period and rng.random() < restart_prob:
            noise_scale = annealing_temp * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_momentum = 1.0
            stall_count = 0
            annealing_temp = max(0.01, annealing_temp * 0.8)
            a = y.copy()

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)

        # Backtracking (every 20th iter to save time)
        if it % 20 == 0:
            c_cand = fft_convolve_self(x_new)
            if c_cand.max() > cur + 1e-6:
                L *= 2.0
                step = 1.0 / L
                x_new = project_simplex(y - step * grad, S)
            elif c_cand.max() < cur * 0.999:
                L = max(L * 0.85, beta * 2.0)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

    return best_local, best_score

    while time.monotonic() - start < max_time:

        if cur < best_score - 1e-14:

            best_local = a.copy()
            stall = 0

        step = max(step, S / (n * 100.0))

            a = a_new

def spectral_shaping(a_init, n, S, max_time, rng, alpha=0.2):
    """
    Iteratively flatten the DFT magnitude to reduce autocorrelation peaks.
    Uses soft magnitude constraints with Griffin-Lim style projection.
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    fsize = 1 << (2 * n - 1).bit_length()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)

        # Target: reduce the variance of non-DC magnitudes
        mag_non_dc = mag[1:]
        target_level = np.sqrt(np.mean(mag_non_dc ** 2))
        # Soft constraint: pull magnitudes toward target
        mag_new = mag.copy()
        mag_new[1:] = (1.0 - alpha) * mag[1:] + alpha * target_level
        # Slightly boost the DC to maintain sum
        mag_new[0] = mag[0]

        # Also try: reduce high-magnitude bins more aggressively
        threshold = np.percentile(mag_non_dc, 90)
        excess = np.maximum(mag_new[1:] - threshold, 0)
        mag_new[1:] = mag_new[1:] - 0.5 * excess

        A_new = mag_new * phase
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / sn)

        c = fft_convolve_self(a_new)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a_new.copy()
        a = a_new

    return best_local, best_score

        # (A*x)[j] = sum_m x[m] * f[j+m] for j = 0,...,n-1, m = 0,...,n-1
        # Pad x to length 2n-1: x_pad = [x, 0, ..., 0]
        # Then (A*x)[j] = sum_m x_pad[m] * f[j+m] = (x_pad correlated with f)[j]
        # In FFT: ifft(fft(x_pad) * conj(fft(f))) gives the correlation

        fsize_cv = 1 << (4 * n - 2).bit_length()
        X = fft(x_curr, fsize_cv)
        F = fft(f, fsize_cv)
        # Correlation: (x corr f)[j] = sum_m x[m+j] * conj(f[m]) ... 
        # Actually circular correlation: r[j] = ifft(conj(X) * F)[j]
        # r[j] = sum_m x[(m-j) mod N] * f[m]
        # We want: sum_m x[m] * f[j+m] = sum_m x[m] * f[j+m]
        # Let p = j+m: sum_p x[p-j] * f[p] = sum_p f[p] * x[p-j] = (f corr x)[j]
        # = ifft(conj(F) * X)[j]... let me verify with a small example.
        # f = [1,1,0,0,0], x = [1,2,3], n=3
        # (A*x)[0] = x[0]*f[0] + x[1]*f[1] + x[2]*f[2] = 1+2+0 = 3
        # (A*x)[1] = x[0]*f[1] + x[1]*f[2] + x[2]*f[3] = 1+0+0 = 1
        # (A*x)[2] = x[0]*f[2] + x[1]*f[3] + x[2]*f[4] = 0+0+0 = 0
        # ifft(conj(F)*X)[j] where F=fft(f,5), X=fft(x,5):
        # This gives circular correlation. For small enough fsize, it should match.
        corr_result = np.real(ifft(np.conj(F) * X, fsize_cv))[:n]
        # Check: corr_result should be [3, 1, 0] for our example. Let's trust it.

        # Solve A*v = 1 using iterative method
        v = np.ones(n)  # Initial guess
        r = rhs - corr_result  # Residual
        p_vec = r.copy()
        rr = np.dot(r, r)

        for cg_iter in range(200):
            Ap = np.real(ifft(np.conj(F) * fft(p_vec, fsize_cv), fsize_cv))[:n]
            pAp = np.dot(p_vec, Ap)
            if abs(pAp) < 1e-30:
                break
            alpha_cg = rr / pAp
            v += alpha_cg * p_vec
            r_new = r - alpha_cg * Ap
            rr_new = np.dot(r_new, r_new)
            if rr_new < 1e-20:
                r = r_new
                break
            p_vec = r_new + (rr_new / rr) * p_vec
            r = r_new
            rr = rr_new

        # Check if v >= 0
        if np.min(v) >= -0.01 * S / n:
            v = np.maximum(v, 0)
            sv = v.sum()
            if sv > 1e-15:
                a_candidate = S * v / sv
                c_candidate = fft_convolve_self(a_candidate)
                if c_candidate.max() < best_score:
                    best_score = c_candidate.max()
                    best_local = a_candidate.copy()
                    a = a_candidate
                    # Update active set
                    c = c_candidate
                    active_lags = np.argsort(c)[::-1][:max(n_active, 20)]
                    K_set = set(int(k) for k in active_lags)
                    f = np.zeros(2 * n - 1)
                    for k in K_set:
                        f[int(k)] = 1.0
                    F = fft(f, fsize_cv)
        else:
            # v has negative components; perturb the active set
            # Remove the lag corresponding to the most negative v component
            worst_j = int(np.argmin(v))
            # Remove a random active lag
            lag_list = list(K_set)
            if len(lag_list) > 10:
                remove_idx = rng.integers(len(lag_list))
                K_set.discard(lag_list[remove_idx])
                f = np.zeros(2 * n - 1)
                for k in K_set:
                    f[int(k)] = 1.0
                F = fft(f, fsize_cv)

    return best_local, best_score


def pairwise_swap(a_init, n, S, max_time, rng):
    """
    Local search via pairwise mass transfers. Focus on reducing peak lags.
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    it = 0
    no_improve = 0
    peak_c = c.copy()

    while time.monotonic() - start < max_time:
        it += 1
        # Focus on the current peak lag
        peak_lag = int(np.argmax(peak_c))
        i_min = max(0, peak_lag - (n - 1))
        i_max = min(peak_lag, n - 1)
        contributors = np.arange(i_min, i_max + 1)

        if len(contributors) < 2:
            break

        # Pick a random contributor (biased toward larger values)
        idx = contributors[rng.integers(len(contributors))]
        partner = peak_lag - idx

        # Try moving mass from the larger of (a[idx], a[partner]) to a small element
        if a[idx] >= a[partner]:
            src = idx
        else:
            src = partner

        # Find a small element to move to (not a contributor to the peak)
        all_idx = np.arange(n)
        non_contrib = np.setdiff1d(all_idx, contributors)
        if len(non_contrib) == 0:
            non_contrib = all_idx

        # Pick from elements with relatively small values
        small_threshold = np.percentile(a[non_contrib], 20)
        small_candidates = non_contrib[a[non_contrib] <= small_threshold]
        if len(small_candidates) == 0:
            small_candidates = non_contrib

        dst = small_candidates[rng.integers(len(small_candidates))]

        # Transfer a small amount
        frac = rng.uniform(0.05, 0.3)
        amount = frac * a[src]
        if amount < 1e-10:
            continue

        a[src] -= amount
        a[dst] += amount

        c_new = fft_convolve_self(a)
        if c_new.max() < best_score:
            best_score = c_new.max()
            best_local = a.copy()
            peak_c = c_new.copy()
            no_improve = 0
        else:
            # Revert
            a[src] += amount
            a[dst] -= amount
            no_improve += 1

        if no_improve > 5000:
            # Try a different peak or reset
            peak_c = fft_convolve_self(a)
            no_improve = 0
            if no_improve > 10000:
                break

    return best_local, best_score


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic initialization."""
    half = n // 2
    h = rng.random(half) + 0.05
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    a = a * (S / a.sum())
    return a


def make_periodic_init(n, rng, S, num_freqs=3):
    """Periodic initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 25)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.35
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.03)
    a = a * (S / a.sum())
    return a


def make_symmetric_init(n, rng, S):
    """Symmetric (palindromic) initialization."""
    half = (n + 1) // 2
    h = rng.random(half) + 0.05
    a = np.zeros(n)
    for i in range(n):
        a[i] = h[min(i, n - 1 - i)]
    a = a * (S / a.sum())
    return a


def make_ramp_init(n, rng, S):
    """Power-law ramp initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    p = rng.random() * 2.0 + 0.3
    a = tt ** p + 0.03
    if rng.random() < 0.5:
        a = (1.0 - tt) ** p + 0.03
    a = a * (S / a.sum())
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for best non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Multi-phase: diverse init -> FISTA -> active set -> KKT linear solve ->
    spectral shaping -> pairwise swap. Multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(5.0, float(budget_s) - 20.0)

    global_best = np.inf
    global_a = None

    # Get parent
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    primary_n = len(parent) if (parent is not None and len(parent) >= 64) else 8192
    print("Primary n=%d, parent=%s, budget=%.0fs" % (primary_n, parent is not None, budget_s))

    # === Phase 1: Quick FISTA from parent ===
    if parent is not None and time.monotonic() < deadline - 300:
        n = primary_n
        S = np.sqrt(2.0 * n)
        a_init = parent * (S / parent.sum()) if parent.sum() > 0 else np.ones(n) * S / n
        t0 = time.monotonic()
        seg = min(300.0, (deadline - time.monotonic()) * 0.20)
        a1, s1 = fista_adaptive(a_init, n, S, seg, rng, beta_init=5.0, beta_max=400.0)
        c1 = fft_convolve_self(a1)
        score1 = c1.max()
        print("Phase1 FISTA parent: max_b=%.6f (%.0fs)" % (score1, time.monotonic() - t0))
        if score1 < global_best:
            global_best = score1
            global_a = a1.copy()

    # === Phase 2: Diverse initializations + FISTA ===
    n = primary_n
    S = np.sqrt(2.0 * n)
    inits = []
    if parent is not None:
        inits.append(parent * (S / parent.sum()))
    for _ in range(3):
        inits.append(make_anti_palindromic_init(n, rng, S))
    for _ in range(2):
        inits.append(make_periodic_init(n, rng, S))
    inits.append(make_symmetric_init(n, rng, S))
    inits.append(make_ramp_init(n, rng, S))
    rand_a = rng.random(n) + 0.02
    inits.append(rand_a * (S / rand_a.sum()))

    remaining = deadline - time.monotonic()
    # Reserve time for later phases
    phase2_budget = min(500.0, remaining * 0.30)
    per_init = phase2_budget / len(inits)
    print("=== Phase 2: %d inits, %.1f sec each (total %.0f sec) ===" % (len(inits), per_init, phase2_budget))

    for i, arr in enumerate(inits):
        if time.monotonic() > deadline - 600:
            break
        t0 = time.monotonic()
        seg_f = per_init * 0.55
        a_f, _ = fista_adaptive(arr, n, S, seg_f, rng, beta_init=3.0, beta_max=300.0)
        c_f = fft_convolve_self(a_f)
        if c_f.max() < global_best:
            global_best = c_f.max()
            global_a = a_f.copy()
            print("  FISTA init[%d]: %.6f" % (i, global_best))

        rem = per_init - (time.monotonic() - t0)
        if rem > 5:
            a_as, _ = active_set_optimize(a_f, n, S, rem, rng, top_k_init=3, top_k_max=200)
            c_as = fft_convolve_self(a_as)
            if c_as.max() < global_best:
                global_best = c_as.max()
                global_a = a_as.copy()
                print("  ActiveSet init[%d]: %.6f" % (i, global_best))

    # === Phase 3: Intensive polishing ===
    if global_a is not None and time.monotonic() < deadline - 100:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        print("=== Phase 3: Polishing (best=%.6f, remaining=%.0f sec) ===" % (global_best, remaining))

        # 3a: High-beta FISTA
        seg = max(10.0, remaining * 0.18)
        a_p1, _ = fista_adaptive(global_a, n, S, seg, rng, beta_init=50.0, beta_max=800.0)
        c_p1 = fft_convolve_self(a_p1)
        if c_p1.max() < global_best:
            global_best = c_p1.max()
            global_a = a_p1.copy()
            print("  3a FISTA hi-beta: %.6f" % global_best)

        # 3b: Active set with high k
        remaining = deadline - time.monotonic()
        seg = max(10.0, remaining * 0.15)
        a_p2, _ = active_set_optimize(global_a, n, S, seg, rng, top_k_init=10, top_k_max=400)
        c_p2 = fft_convolve_self(a_p2)
        if c_p2.max() < global_best:
            global_best = c_p2.max()
            global_a = a_p2.copy()
            print("  3b ActiveSet hi-k: %.6f" % global_best)

        # 3c: KKT linear solve
        remaining = deadline - time.monotonic()
        seg = max(10.0, remaining * 0.10)
        a_p3, _ = kkt_linear_solve(global_a, n, S, seg, rng, n_active=30)
        c_p3 = fft_convolve_self(a_p3)
        if c_p3.max() < global_best:
            global_best = c_p3.max()
            global_a = a_p3.copy()
            print("  3c KKT: %.6f" % global_best)

        # 3d: Spectral shaping
        remaining = deadline - time.monotonic()
        seg = max(10.0, remaining * 0.08)
        a_p4, _ = spectral_shaping(global_a, n, S, seg, rng, alpha=0.15)
        c_p4 = fft_convolve_self(a_p4)
        if c_p4.max() < global_best:
            global_best = c_p4.max()
            global_a = a_p4.copy()
            print("  3d Spectral: %.6f" % global_best)

        # 3e: Pairwise swap
        remaining = deadline - time.monotonic()
        seg = max(5.0, remaining * 0.08)
        a_p5, _ = pairwise_swap(global_a, n, S, seg, rng)
        c_p5 = fft_convolve_self(a_p5)
        if c_p5.max() < global_best:
            global_best = c_p5.max()
            global_a = a_p5.copy()
            print("  3e Pairwise: %.6f" % global_best)

        # 3f: Final high-beta FISTA
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_p6, _ = fista_adaptive(global_a, n, S, remaining * 0.6, rng,
                                       beta_init=200.0, beta_max=2000.0)
            c_p6 = fft_convolve_self(a_p6)
            if c_p6.max() < global_best:
                global_best = c_p6.max()
                global_a = a_p6.copy()
                print("  3f Final FISTA: %.6f" % global_best)

    # === Phase 4: Try longer sequence ===
    if time.monotonic() < deadline - 80:
        for alt_n in [16384, 4096, 32768]:
            if time.monotonic() > deadline - 40:
                break
            S_alt = np.sqrt(2.0 * alt_n)
            t0 = time.monotonic()
            if parent is not None and len(parent) >= 10:
                from numpy import interp
                tt_old = np.linspace(0, 1, len(parent))
                tt_new = np.linspace(0, 1, alt_n)
                a_alt = np.maximum(interp(tt_new, tt_old, parent), 0)
            else:
                a_alt = make_anti_palindromic_init(alt_n, rng, S_alt)
            if a_alt.sum() < 1e-10:
                a_alt = np.ones(alt_n) * S_alt / alt_n
            a_alt = a_alt * (S_alt / a_alt.sum())

            seg = min(80.0, max(10.0, (deadline - time.monotonic()) * 0.12))
            a_alt_o, _ = fista_adaptive(a_alt, alt_n, S_alt, seg, rng, beta_init=5.0, beta_max=300.0)
            c_alt = fft_convolve_self(a_alt_o)
            obj_alt = c_alt.max()
            if obj_alt < global_best:
                global_best = obj_alt
                global_a = a_alt_o.copy()
                print("  Alt n=%d: %.6f" % (alt_n, obj_alt))
            else:
                print("  Alt n=%d: %.6f (no improvement)" % (alt_n, obj_alt))

    # === Final ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(512, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL: evaluate=%.6f, internal_max_b=%.6f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        print("  Fallback triggered")
        global_a = np.ones(512, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback score: %.6f" % ver)

    return final_list
