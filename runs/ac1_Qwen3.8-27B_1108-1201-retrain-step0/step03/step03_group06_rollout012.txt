def soft_max_smoothed_obj(theta, T, n, counts):
    """Objective and gradient for L-BFGS, parameterized by theta."""
    S = np.sqrt(2.0 * n)
    # Compute a from theta (softmax parameterization)
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    a = S * exp_theta / total
    
    # Compute convolution
    c = fft_convolve_self(a)
    
    # Log-sum-exp
    c_max = c.max()
    scaled = T * (c - c_max)
    exp_scaled = np.exp(scaled)
    f = c_max + np.log(exp_scaled.sum()) / T
    w = exp_scaled / exp_scaled.sum()
    
    # Gradient with respect to a (in the normalized space)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    grad_a = (2.0 / T) * corr[:n]
    
    # Chain rule: theta -> a
    # a[j] = S * exp(theta[j]) / sum(exp(theta))
    # da[j]/dtheta[k] = a[j] * (delta(j,k) - a[k]/S)
    # d f / d theta[k] = sum_j (df/da[j]) * da[j]/dtheta[k]
    # = sum_j grad_a[j] * a[j] * (delta(j,k) - a[k]/S)
    # = grad_a[k] * a[k] - (a[k]/S) * sum_j grad_a[j] * a[j]
    
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    
    counts[0] += 1
    return f, grad_theta

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS with log-sum-exp smoothing
and T-annealing, followed by coordinate descent polish and restarts.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global for counting function evaluations
_eval_count = [0]

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def obj_and_grad_normalized(a, T):
    """
    Compute log-sum-exp smoothed objective and gradient with respect to a,
    assuming a is already normalized (sum(a) = sqrt(2n)).
    """
    n = len(a)
    _eval_count[0] += 1
    c = fft_convolve_self(a)
    c_max = c.max()
    scaled = T * (c - c_max)
    exp_scaled = np.exp(scaled)
    f = c_max + np.log(exp_scaled.sum()) / T
    w = exp_scaled / exp_scaled.sum()
    
    # Gradient: 2/T * cross_corr(w, a)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    grad = (2.0 / T) * corr[:n]
    
    return f, grad

def make_obj_func(T, n):
    """Create objective function that works with theta parameterization."""
    S = np.sqrt(2.0 * n)
    
    def fun_grad(theta):
        # Convert theta to normalized a
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        a = S * exp_theta / total
        
        f, grad_a = obj_and_grad_normalized(a, T)
        
        # Chain rule
        dot = np.dot(grad_a, a)
        grad_theta = grad_a * a - (a / S) * dot
        
        return f, grad_theta
    
    return fun_grad

def run_lbfgs_trajectory(a_start, n, T_values, max_iter_per_T=500, time_per_T=30.0):
    """Run L-BFGS with increasing T values."""
    S = np.sqrt(2.0 * n)
    theta = np.log(a_start / a_start.sum())
    theta = theta - theta.min()  # shift for numerical stability
    
    best_obj = np.inf
    best_theta = theta.copy()
    
    for T in T_values:
        start_t = time.monotonic()
        fun = make_obj_func(T, n)
        
        result = minimize(
            fun, theta, jac=True, method='L-BFGS-B',
            options={
                'maxiter': max_iter_per_T,
                'maxfun': 5000,
                'ftol': 1e-14,
                'gtol': 1e-8,
                'maxls': 30,
                'iprint': -1,
            }
        )
        
        theta = result.x
        
        # Evaluate true objective
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        a = S * exp_theta / exp_theta.sum()
        c = fft_convolve_self(a)
        true_obj = c.max()
        
        if true_obj < best_obj:
            best_obj = true_obj
            best_theta = theta.copy()
        
        elapsed = time.monotonic() - start_t
        print(f"  T={T}: obj={true_obj:.8f}, L-BFGS iters={result.nit}, time={elapsed:.1f}s")
        
        # Stop if we've exceeded time for this T value
        if elapsed > time_per_T * 2:
            break
    
    # Convert back
    theta_shifted = best_theta - best_theta.max()
    exp_theta = np.exp(theta_shifted)
    a = S * exp_theta / exp_theta.sum()
    return a, best_obj

def coordinate_descent(a, n, S, time_budget, n_factors=5):
    """Coordinate descent to polish the solution."""
    a = a.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(42)
    start = time.monotonic()
    elapsed_since_improve = 0.0
    
    factors = np.array([0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 3.0])
    
    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            orig = a[i]
            if orig < 1e-14:
                continue
            
            best_local_obj = best_obj
            best_local_val = orig
            
            for f in factors:
                a[i] = orig * f
                s = a.sum()
                if s < 1e-15:
                    a[i] = orig
                    break
                a_norm = a * (S / s)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-13:
                    best_local_obj = obj_check
                    best_local_val = orig * f
            
            a[i] = best_local_val
            s = a.sum()
            a = a * (S / s) if s > 1e-15 else a
            
            c = fft_convolve_self(a)
            if c.max() < best_obj - 1e-13:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
                elapsed_since_improve = 0.0
            else:
                elapsed_since_improve += 1e-4  # rough per-coordinate time
        
        if not improved and sweep >= 3:
            break
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS with log-sum-exp smoothing
and T-annealing, coordinate descent polish, peak-focused subgradient, and restarts.
Key improvements over previous: L-BFGS for fast convergence near optimum,
structured T-annealing, and focused peak-reduction strategies.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def true_obj(a):
    """Compute true objective (max of convolution) for normalized a."""
    c = fft_convolve_self(a)
    return c.max()


def make_smooth_obj_grad(T, S, n):
    """Factory for smooth objective+gradient in softmax parameterization."""

    def fun_grad(theta):
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
        else:
            a = S * exp_theta / total

        c = fft_convolve_self(a)
        c_max = c.max()
        if c_max < 1e-30:
            return c_max, np.zeros(n)
        scaled = T * (c - c_max)
        exp_scaled = np.exp(scaled)
        f = c_max + np.log(exp_scaled.sum()) / T
        w = exp_scaled / exp_scaled.sum()

        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        grad_a = (2.0 / T) * corr[:n]

        dot = np.dot(grad_a, a)
        grad_theta = grad_a * a - (a / S) * dot

        return f, grad_theta

    return fun_grad


def lbfgs_run(theta, T, S, n, max_iter=400, maxfun=4000, ftol=1e-15, gtol=1e-10):
    """Run L-BFGS-B with given T."""
    fun_grad = make_smooth_obj_grad(T, S, n)
    t0 = time.monotonic()
    result = minimize(
        fun_grad, theta, jac=True, method='L-BFGS-B',
        options={
            'maxiter': max_iter,
            'maxfun': maxfun,
            'ftol': ftol,
            'gtol': gtol,
            'maxls': 50,
            'iprint': -1,
        }
    )
    elapsed = time.monotonic() - t0
    return result.x, result.nit, result.nfev, elapsed


def theta_to_a(theta, S, n):
    """Convert theta parameters to normalized non-negative sequence."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(n, S / n)
    a = S * exp_theta / total
    return np.maximum(a, 0.0) * (S / max(a.sum(), 1e-300))


def a_to_theta(a):
    """Convert sequence to log parameters (for softmax parameterization)."""
    a_safe = np.maximum(a, 1e-300)
    theta = np.log(a_safe)
    return theta - theta.min()


def coordinate_descent(a, n, S, time_budget, verbose=False):
    """Efficient coordinate descent for polishing."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(777)
    start = time.monotonic()
    factors = np.array([0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.7, 3.0, 0.1])
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            orig = a[i]
            if orig < 1e-14:
                continue

            best_local_obj = best_obj
            best_local_val = orig

            for f in factors:
                new_val = orig * f
                if new_val < 0:
                    continue
                a[i] = new_val
                s = a.sum()
                if s < 1e-15:
                    a[i] = orig
                    break
                a_temp = a * (S / s)
                c_check = fft_convolve_self(a_temp)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-14:
                    best_local_obj = obj_check
                    best_local_val = new_val

            a[i] = best_local_val
            s = a.sum()
            if s > 1e-15:
                a = a * (S / s)

            if best_local_val != orig:
                c = fft_convolve_self(a)
                if c.max() < best_obj - 1e-14:
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True

        if not improved and sweep >= 4:
            break

        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.9f}")

    return best_a, best_obj


def peak_focused_subgradient(a, n, S, time_budget, T=500, lr_init=0.05, top_k=10):
    """Subgradient descent focusing on top-k peak lags."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    step = 0

    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # Identify top-k lags
        top_indices = np.argsort(c)[-top_k:]
        top_vals = c[top_indices]
        spread = max(top_vals.max() - top_vals.min(), 1e-10)
        w = np.zeros(len(c))
        w[top_indices] = (top_vals - top_vals.min()) / spread + 0.1
        w = w / w.sum()

        # Gradient
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g = 2.0 * corr[:n]

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        progress = (time.monotonic() - start) / time_budget
        lr = lr_init * (1.0 - 0.5 * progress) / gnorm

        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * (S / s)

        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_obj:
            a = a_new
        else:
            lr *= 0.5
            a_new = a - lr * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new * (S / s)
                a = a_new
            else:
                break

        step += 1

    return best_a, best_obj
