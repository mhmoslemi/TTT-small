def project_simplex(v, S):
    """Project v onto {x>=0, sum(x)=S} using bisection."""
    n = len(v)
    v_clipped = np.clip(v, 0, None)
    s = v_clipped.sum()
    if s <= S:
        return v_clipped + (S - s) / n, s
    lo, hi = 0.0, v.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_mid = np.maximum(v - mid, 0.0).sum()
        if s_mid > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    v_proj = np.maximum(v - tau, 0.0)
    s_proj = v_proj.sum()
    return v_proj, s_proj
