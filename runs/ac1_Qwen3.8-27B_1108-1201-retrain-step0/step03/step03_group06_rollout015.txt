"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive log-sum-exp smoothed
gradient descent with phase-based temperature scheduling, Nesterov momentum,
peak-equalization post-processing, and multi-length exploration.

Key improvements over previous:
1. Phase-based T scheduling: low T first to equalize peaks, then high T to polish
2. Proper gradient projection and adaptive step sizing

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_smoothed_gradient(c, a, T):
    """Compute smoothed gradient of max(c) w.r.t. a using softmax weighting.
    Returns gradient projected onto the sum=0 tangent space.
    """
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)

    scaled = T * (c - c_max) / c_max
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # g_j = sum_k w[k] * a[k-j] (cross-correlation)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    g = corr[:n]

    # Project to tangent space of sum constraint
    g = g - np.mean(g)
    return g


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def main_optimize(a_init, n, S, time_budget, T_start=3.0, T_end=300.0,

    a = normalize_nonneg(a_init.copy(), S)

    best_obj = c.max()

    beta = momentum
    start = time.monotonic()

    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start

        T = T_start + (T_end - T_start) * (progress ** 1.5)

        c = fft_convolve_self(a)
        cur_obj = c.max()

def peak_equalization(a, n, S, time_budget, K=50, verbose=True):
    """
    Try to equalize the top K peaks by direct adjustment.
    At each step, find the highest and lowest among top K peaks,
    and take a small step to reduce the gap.
    """
    a = normalize_nonneg(a.copy(), S)
    best_a = a.copy()
    c = fft_convolve_self(a)
    best_obj = c.max()
    start = time.monotonic()
    step = 0
    
    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Get top K peak values
        sorted_idx = np.argsort(c)[::-1]
        top_K = sorted_idx[:K]
        top_vals = c[top_K]
        max_val = top_vals[0]
        min_val = top_vals[-1]
        
        if max_val - min_val < 1e-6 * max_val:
            # Peaks are well equalized, switch to pure max reduction
            break
        
        # Compute gradient that pushes max down and min up
        # w[k] = +1 for near-max peaks, -1 for near-min peaks (within top K)
        w = np.zeros(len(c))
        threshold_high = max_val - 0.1 * (max_val - min_val)
        threshold_low = min_val + 0.1 * (max_val - min_val)
        w[np.where((c >= threshold_high) & (c <= max_val + 1e-10))[0]] = 1.0
        w[np.where((c >= min_val - 1e-10) & (c <= threshold_low))[0]] = -1.0
        
        w_sum = w.sum()
        if abs(w_sum) < 1e-10:
            # Fall back to standard gradient
            g = compute_smoothed_gradient(c, a, T=50.0)
        else:
            # Compute gradient with this weight
            fsize = 1 << (3 * n - 2).bit_length()
            W = fft(w, fsize)
            A = fft(a, fsize)
            corr = np.real(ifft(W * np.conj(A)))
            g = corr[:n]
            g = g - np.mean(g)
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        g_dir = g / gnorm
        
        # Small step
        lr = 0.02 * max(1.0, n) / len(c)
        a_trial = a - lr * g_dir
        a_trial = normalize_nonneg(a_trial, S)
        c_trial = fft_convolve_self(a_trial)
        obj_trial = c_trial.max()
        
        if obj_trial < cur_obj:
            a = a_trial
        else:
            # Try smaller step
            lr *= 0.1
            a_trial = a - lr * g_dir
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial < cur_obj:
                a = a_trial
            else:
                break
        
        step += 1
        if verbose and step % 200 == 0:
            print(f"    PE step {step}: obj={best_obj:.8f}, gap={max_val-min_val:.2e}")
    
    return best_a, best_obj

        a = normalize_nonneg(global_a, S_main)
        a_best, _ = main_optimize(
            a, n_main, S_main, p5a,
            T_start=100.0, T_end=1000.0,
            lr_init=0.02, momentum=0.92, verbose=False
        )
        a_best = normalize_nonneg(a_best, S_main)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print(f"  Phase 5a result: obj={global_best:.8f}")
        
        if p5b > 3:
            a = normalize_nonneg(global_a, S_main)
            a_best, _ = coordinate_descent(a, n_main, S_main, p5b, verbose=False)
            a_best = normalize_nonneg(a_best, S_main)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print(f"  Phase 5b result: obj={global_best:.8f}")
    
    # Final verification
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 0.1
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: obj={global_best:.8f} (verified={ver:.8f})")
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        print("Fallback: uniform 256")
    
    return final_a
