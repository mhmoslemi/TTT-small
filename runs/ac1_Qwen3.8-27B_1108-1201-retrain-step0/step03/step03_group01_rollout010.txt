"""

Key improvements over previous:

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    S = float(S)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix sum to be exactly S (numerical roundoff)
    diff = S - result.sum()
    if abs(diff) > 1e-12:
        # Distribute difference to non-zero elements
        mask = result > 0
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.maximum(result, 0.0)
    return result


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_value_and_weights(c, beta):
    """Compute LSE_beta(c) and softmax weights."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        total = 1e-300
    w = exp_bc / total
    val = c_max + np.log(total) / beta
    return val, w


def compute_grad(a, w, n):
    """Compute gradient of LSE_beta(conv(a,a)) w.r.t. a.
    grad[i] = 2 * sum_k w[k] * a[k-i]
    """
    fsize = 1 << (2 * len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    R = np.real(ifft(A * np.conj(W)))
    return 2.0 * R[:n]


def objective_value(a):
    """Compute the actual objective 2*n*max_conv/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def nesterov_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_end=500.0,
                  beta_growth=1.15, iters_per_level=200, tol=1e-10):
    """
    Nesterov-accelerated gradient with LSE smoothing and continuation.
    Uses backtracking line search for step size.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_param = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    beta = beta_init
    step = S / (n * beta)  # initial step size
    
    it_total = 0
    plateau_count = 0
    last_best = np.inf
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(y)
        val, w = lse_value_and_weights(c, beta)
        
        # Track best based on actual objective (not smoothed)
        actual = c.max()
        if actual < best_score:
            best_score = actual
            best_local = y.copy()
        
        # Plateau detection
        if actual < last_best - 1e-10:
            last_best = actual
            plateau_count = 0
        else:
            plateau_count += 1
        
        # If plateaued at this beta level, increase beta
        if plateau_count >= 5 * iters_per_level or beta >= beta_end:
            if beta >= beta_end:
                break
            beta *= beta_growth
            plateau_count = 0
            t_param = 1.0
            y = a.copy()
            step = S / (n * beta)
            continue
        
        # Compute gradient
        grad = compute_grad(y, w, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        c_y = c  # already computed
        
        best_x = y.copy()
        best_f = val
        step_try = step
        for bt in range(30):
            x_try = project_simplex(y - step_try * grad / max(gnorm, 1e-15), S)
            c_x = fft_convolve_self(x_try)
            f_x, _ = lse_value_and_weights(c_x, beta)
            c_x_max = c_x.max()
            
            # Armijo condition (simplified: just check improvement)
            if c_x_max < c_y.max() - 1e-12 or f_x < best_f:
                best_x = x_try
                best_f = f_x
                break
            step_try *= 0.5
        
        if step_try > step:
            step = min(step * 1.5, S / (n * 0.5))
        elif step_try < step * 0.5:
            step = step_try
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        y = best_x + ((t_param - 1.0) / t_new) * (best_x - a)
        y = np.maximum(y, 0.0)
        # Re-project to maintain sum constraint
        y_s = y.sum()
        if y_s < S * 0.9 or y_s > S * 1.1:
            y = project_simplex(y, S)
        a = best_x
        t_param = t_new
        it_total += 1
        
        # Periodic perturbation
        if it_total % 500 == 250 and rng.random() < 0.2:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_param = 1.0
            a = y.copy()
    
    # Final evaluation
    c_final = fft_convolve_self(best_local)
    return best_local, c_final.max()


def coordinate_descent_polish(a_init, n, S, max_time, rng, block_size=8):
    """
    Coordinate descent on the exact (non-smooth) objective.
    Updates blocks of `block_size` consecutive elements.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Try random blocks
        n_blocks = max(1, int(max_time * 2))  # target ~2 blocks per second initially
        for _ in range(n_blocks):
            if time.monotonic() - start >= max_time:
                break
            # Random block position and size
            bs = rng.integers(2, min(block_size + 1, n // 2))
            start_idx = rng.integers(0, n - bs)
            
            # Save original
            orig = a[start_idx:start_idx + bs].copy()
            orig_sum = orig.sum()
            
            # Try a few random modifications
            best_block = orig
            best_c = cur
            
            for trial in range(5):
                # Random rescaling of the block
                scale = rng.uniform(0.5, 1.5)
                a[start_idx:start_idx + bs] = orig * scale
                # Re-normalize to maintain sum = S
                a_sum = a.sum()
                if a_sum > 0:
                    a *= S / a_sum
                c_new = fft_convolve_self(a)
                if c_new.max() < best_c:
                    best_c = c_new.max()
                    best_block = a[start_idx:start_idx + bs].copy()
                
                # Add small noise
                a[start_idx:start_idx + bs] = orig + rng.normal(0, 0.01 * S / np.sqrt(n), bs)
                a[start_idx:start_idx + bs] = np.maximum(a[start_idx:start_idx + bs], 0)
                a_sum = a.sum()
                if a_sum > 0:
                    a *= S / a_sum
                c_new = fft_convolve_self(a)
                if c_new.max() < best_c:
                    best_c = c_new.max()
                    best_block = a[start_idx:start_idx + bs].copy()
            
            a[start_idx:start_idx + bs] = best_block
            # Re-normalize
            a_sum = a.sum()
            if a_sum > 0:
                a *= S / a_sum
            else:
                a = best_local.copy()
    
    c_final = fft_convolve_self(best_local)
    return best_local, c_final.max()


def spectral_shaping_fine(a, N, n, S, top_k, alpha, n_iters, rng, target):
    """Fine spectral shaping with small alpha for polishing."""
    best_a = a.copy()
    best_score = np.inf
    
    for it in range(n_iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = c.max()
        
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        
        if c_max <= target:
            break
        
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        m = np.arange(N)
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
        
        P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)
    
    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

"""
Minimize 2*n*max_conv/sum^2 using continuation LSE + adaptive Nesterov + coordinate polish.

Key improvements over previous:
1. Backtracking line search for adaptive step size (no manual tuning)
2. Multi-n strategy: try both n=8192 and n=16384
3. Coordinate descent polish on exact (non-smooth) objective
4. Structured perturbation based on gradient analysis
5. Better continuation schedule with plateau detection
6. More diverse initializations including U-shape and bimodal patterns
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    S = float(S)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    diff = S - result.sum()
    if abs(diff) > 1e-10:
        mask = result > 1e-12
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.maximum(result, 0.0)
    return result


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_value_and_weights(c, beta):
    """Compute LSE_beta(c) and softmax weights."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        total = 1e-300
    w = exp_bc / total
    val = c_max + np.log(total) / beta
    return val, w


def compute_grad(a, w, n):
    """Compute gradient of LSE_beta(conv(a,a)) w.r.t. a. grad[i] = 2*sum_k w[k]*a[k-i]."""
    fsize = 1 << (2 * len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    R = np.real(ifft(A * np.conj(W)))
    return 2.0 * R[:n]


def objective_value(a):
    """Compute the actual objective 2*n*max_conv/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def nesterov_lse_continuation(a_init, n, S, max_time, rng,
                               beta_init=3.0, beta_end=800.0,
                               beta_growth=1.18, plateau_patience=80):
    """
    Nesterov-accelerated gradient with LSE smoothing and continuation.
    Uses backtracking line search for adaptive step size.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_param = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    beta = beta_init
    step = S / (n * beta * 2.0)
    plateau_count = 0
    last_best = np.inf
    it = 0
    n_levels = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        c_max = c.max()

        if c_max < best_score:
            best_score = c_max
            best_local = y.copy()

        if c_max < last_best - 1e-8 * max(1.0, last_best):
            last_best = c_max
            plateau_count = 0
        else:
            plateau_count += 1

        if plateau_count >= plateau_patience or beta >= beta_end:
            if beta >= beta_end:
                break
            beta *= beta_growth
            n_levels += 1
            plateau_count = 0
            t_param = 1.0
            y = best_local.copy()
            step = S / (n * beta * 2.0)
            if n_levels % 10 == 0:
                c_check = fft_convolve_self(y)
                print("    Level %d: beta=%.1f, best=%.6f" % (n_levels, beta, c_check.max()))
            continue

        _, w = lse_value_and_weights(c, beta)
        grad = compute_grad(y, w, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        grad_dir = grad / gnorm

        # Backtracking line search
        best_x = y.copy()
        best_c = c_max
        h = step
        for _bt in range(25):
            x_try = project_simplex(y - h * grad_dir, S)
            c_try = fft_convolve_self(x_try)
            if c_try.max() < best_c - 1e-14:
                best_x = x_try
                best_c = c_try.max()
                break
            h *= 0.6

        if h > step * 1.2:
            step = min(step * 1.3, S / (n * 0.5))
        elif h < step * 0.4:
            step = max(step * 0.7, S / (n * 1000.0))

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        y = best_x + ((t_param - 1.0) / t_new) * (best_x - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum < S * 0.85 or y_sum > S * 1.15:
            y = project_simplex(y, S)
        a = best_x
        t_param = t_param if t_param <= 50.0 else 50.0

        if it % 400 == 200 and rng.random() < 0.15:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_param = 1.0

    c_final = fft_convolve_self(best_local)
    return best_local, c_final.max()


def coordinate_descent_polish(a_init, n, S, max_time, rng, block_size=16, trials_per_block=8):
    """
    Coordinate descent on the exact (non-smooth) objective.
    Updates blocks of consecutive elements.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    c = fft_convolve_self(a)
    best_score = c.max()

    while time.monotonic() - start < max_time:
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        bs = int(rng.integers(4, block_size + 1))
        if bs >= n:
            continue
        start_idx = int(rng.integers(0, n - bs + 1))
        orig = a[start_idx:start_idx + bs].copy()
        orig_sum = orig.sum()
        base_c = c.max()

        for _ in range(trials_per_block):
            if time.monotonic() - start >= max_time:
                break
            a[start_idx:start_idx + bs] = orig

            u = rng.random(bs)
            u = u / max(u.sum(), 1e-30)
            a[start_idx:start_idx + bs] = orig_sum * u
            a_sum = a.sum()
            if a_sum > 1e-15:
                a *= S / a_sum
            c = fft_convolve_self(a)
            if c.max() < base_c:
                base_c = c.max()

            a[start_idx:start_idx + bs] = orig * rng.uniform(0.3, 2.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a *= S / a_sum
            c = fft_convolve_self(a)
            if c.max() < base_c:
                base_c = c.max()

            a[start_idx:start_idx + bs] = np.maximum(orig + rng.normal(0, 0.01 * S / np.sqrt(n), bs), 0.0)
            a_sum = a.sum()
            if a_sum > 1e-15:
                a *= S / a_sum
            c = fft_convolve_self(a)
            if c.max() < base_c:
                base_c = c.max()

            a[start_idx:start_idx + bs] = np.roll(orig, int(rng.integers(1, bs)))
            a_sum = a.sum()
            if a_sum > 1e-15:
                a *= S / a_sum
            c = fft_convolve_self(a)
            if c.max() < base_c:
                base_c = c.max()

            if rng.random() < 0.5:
                a[start_idx:start_idx + bs] = orig[::-1]
                a_sum = a.sum()
                if a_sum > 1e-15:
                    a *= S / a_sum
                c = fft_convolve_self(a)
                if c.max() < base_c:
                    base_c = c.max()
            else:
                scale = 1.0 + 0.5 * (2.0 * rng.random() - 1.0)
                a[start_idx:start_idx + bs] = orig * scale
                a_sum = a.sum()
                if a_sum > 1e-15:
                    a *= S / a_sum
                c = fft_convolve_self(a)
                if c.max() < base_c:
                    base_c = c.max()

    c_final = fft_convolve_self(best_local)
    return best_local, c_final.max()


def spectral_shaping(a, n, S, top_k, alpha, n_iters, rng, target):
    """Spectral shaping: modify |FFT(a)|^2 to reduce peak convolution."""
    N = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = np.inf

    for it in range(n_iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = c.max()

        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        if c_max <= target:
            break

        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        m = np.arange(N)
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

        P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new, S)

        if it % 500 == 250 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n)
