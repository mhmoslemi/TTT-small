import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def topk_subgradient(a, c, K):
    """Compute subgradient focusing on top K lags."""
    n = len(a)
    idx = np.argsort(c)[::-1][:K]
    grad = np.zeros(n)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        grad[lo:hi+1] += 2.0 * a[k - np.arange(lo, hi+1)]
    return grad

    a = normalize_sum(a_init, n)

    it = 0

        cur = c.max()
        if cur < best:
            best = cur

        grad = topk_subgradient(a, c, K)

import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.ones(n) * S / n
    return a * (S / s)

def get_subgradient_lag(a, k):
    """Subgradient contribution from lag k."""
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad

def run_mirror_descent(a_init, max_time, rng, name=""):

    a = normalize_sum(a_init.copy(), S)

    best_conv = np.inf

        c = fft_convolve_self(a)
        cur = c.max()

            best_a = a.copy()
            last_improve = it
            stall_count = 0

            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * a[k - idx]

def run_frank_wolfe(a_init, max_time, rng, name=""):
    """Frank-Wolfe (conditional gradient) method."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, S)

    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    lb = S * S / (2.0 * n - 1.0)

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur = c.max()

        if cur < best_conv - 1e-13:
            best_conv = cur
            best_a = a.copy()
            last_improve = it

        # Subgradient (use top-4 for stability)
        K = min(4, max(1, it // 5000))
        grad = compute_subgradient(a, c, K, weight_mode="value")
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Frank-Wolfe: find minimizer of linear oracle
        # min_{x in simplex} grad @ x = S * min_i grad[i]
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = S  # vertex of simplex

        # Line search: minimize f((1-t)*a + t*v) for t in [0,1]
        # Use backtracking
        t = 2.0 / (it + 2)  # standard FW step
        a_new = (1.0 - t) * a + t * v
        a_new = normalize_sum(a_new, S)

        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur:
            a = a_new
        else:
            # Halve step
            t /= 2.0
            a_new = (1.0 - t) * a + t * v
            a_new = normalize_sum(a_new, S)
            a = a_new

        # Stagnation recovery
        if it - last_improve > 5000:
            noise_scale = 0.005 * S / np.sqrt(n)
            a = normalize_sum(a + rng.normal(0, noise_scale, n), S)
            last_improve = it

    c_final = fft_convolve_self(a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
        best_a = a.copy()

def downsample_upsample(a, n_small):
    """Downsample by averaging, then upsample by interpolation."""
    n = len(a)
    if n_small >= n:
        return a.copy()
    nblocks = n // n_small
    a_small = a[:nblocks * n_small].reshape(n_small, nblocks).mean(axis=1)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n)
    return np.interp(t_new, t_old, a_small)

        obj = 2.0 * len(a_opt) * conv_val / (S_chk * S_chk)
        if obj < global_best_obj:
            global_best_obj = obj
            global_a = a_opt.copy()
            print(f"  Init {i}: obj={obj:.6f} (improvement!)")

    print(f"Phase 1 done. Best: {global_best_obj:.6f}, elapsed={time.monotonic()-phase1_start:.0f}s")

    # Phase 2: Intensive polishing of best candidate
    phase2_time = max(30.0, (deadline - time.monotonic()) * 0.65)
    phase2_start = time.monotonic()
    print(f"=== Phase 2: Polishing ({phase2_time:.0f}s) ===")

    if global_a is not None:
        n_best = len(global_a)
        # Split remaining time across multiple methods
        seg1 = phase2_time * 0.40
        seg2 = phase2_time * 0.30
        seg3 = phase2_time * 0.30

        # Method 1: Mirror hybrid (main workhorse)
        a1, c1 = run_mirror_hybrid(global_a, seg1, rng, name="polish1")
        S1 = np.sqrt(2.0 * len(a1))
        obj1 = 2.0 * len(a1) * c1 / (S1 * S1)
        if obj1 < global_best_obj:
            global_best_obj = obj1
            global_a = a1.copy()
        print(f"  After mirror polish: {global_best_obj:.6f}")

        # Method 2: Frank-Wolfe
        remaining2 = deadline - time.monotonic()
        seg2 = min(seg2, max(5.0, remaining2 - 30))
        if seg2 > 10:
            a2, c2 = run_frank_wolfe(global_a, seg2, rng, name="fw")
            S2 = np.sqrt(2.0 * len(a2))
            obj2 = 2.0 * len(a2) * c2 / (S2 * S2)
            if obj2 < global_best_obj:
                global_best_obj = obj2
                global_a = a2.copy()
            print(f"  After Frank-Wolfe: {global_best_obj:.6f}")

        # Method 3: Another mirror run with different params (restart)
        remaining3 = deadline - time.monotonic()
        if remaining3 > 10:
            # Perturb and restart
            n_r = len(global_a)
            S_r = np.sqrt(2.0 * n_r)
            perturbed = normalize_sum(global_a + rng.normal(0, 0.003 * S_r / np.sqrt(n_r), n_r), S_r)
            a3, c3 = run_mirror_hybrid(perturbed, min(seg3, remaining3 - 5), rng, name="restart")
            S3 = np.sqrt(2.0 * len(a3))
            obj3 = 2.0 * len(a3) * c3 / (S3 * S3)
            if obj3 < global_best_obj:
                global_best_obj = obj3
                global_a = a3.copy()
            print(f"  After restart: {global_best_obj:.6f}")

    # Phase 3: Try different length if time remains
    remaining = deadline - time.monotonic()
    if remaining > 60:
        for alt_n in lengths_to_try:
            if alt_n == (len(global_a) if global_a is not None else 0):
                continue
            if 2048 <= alt_n <= 32768:
                S_alt = np.sqrt(2.0 * alt_n)
                print(f"=== Phase 3: Trying n={alt_n} ({remaining:.0f}s) ===")
                # Initialize from current best (resample)
                if global_a is not None:
                    t_old = np.linspace(0, 1, len(global_a))
                    t_new = np.linspace(0, 1, alt_n)
                    alt_init = np.interp(t_new, t_old, global_a)
                else:
                    alt_init = np.ones(alt_n)
                t_alloc = min(remaining * 0.4, 300.0)
                a_alt, c_alt = run_mirror_hybrid(alt_init, t_alloc, rng, name=f"alt_{alt_n}")
                S_alt2 = np.sqrt(2.0 * len(a_alt))
                obj_alt = 2.0 * len(a_alt) * c_alt / (S_alt2 * S_alt2)
                if obj_alt < global_best_obj:
                    global_best_obj = obj_alt
                    global_a = a_alt.copy()
                    print(f"  Alt n={alt_n}: obj={obj_alt:.6f} (improvement!)")
                remaining = deadline - time.monotonic()
                if remaining < 30:
                    break

    # Final tiny polish
    remaining = deadline - time.monotonic()
    if remaining > 5 and global_a is not None:
        a_fin, c_fin = run_mirror_hybrid(global_a, remaining - 2, rng, name="final")
        S_fin = np.sqrt(2.0 * len(a_fin))
        obj_fin = 2.0 * len(a_fin) * c_fin / (S_fin * S_fin)
        if obj_fin < global_best_obj:
            global_best_obj = obj_fin
            global_a = a_fin.copy()
        print(f"  Final polish: {global_best_obj:.6f}")

    # Verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print(f"Verified objective: {ver:.6f}")
            if ver < global_best_obj:
                global_best_obj = ver
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            global_a = None

    # Fallback
    if global_a is None:
        n_fb = 256
        global_a = np.ones(n_fb) * (np.sqrt(2.0 * n_fb) / n_fb)
        print(f"Fallback used. Objective: {evaluate_sequence([float(x) for x in global_a]):.6f}")

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(512.0) / 256)

    print(f"Returning sequence of length {len(global_a)}, best obj={global_best_obj:.6f}")
    return [float(x) for x in global_a]
