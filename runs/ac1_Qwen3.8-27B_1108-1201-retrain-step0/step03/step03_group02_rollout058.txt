"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Uses LP-based alternating optimization (active set method) combined with
gradient-based polishing. The LP finds a direction g that keeps the
cross-convolution f*g bounded, then we update f toward g.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def solve_lp_for_g(f, M, active_indices, solver=None):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and (f*g)[k] <= M for k in active_indices.
    Returns g or None if infeasible.
    """
    n = len(f)
    if solver is None:
        solver = cp.CLARABEL
    
    g = cp.Variable(n, nonneg=True)
    constraints = []
    
    for k in active_indices:
        k = int(k)
        if k < 0 or k >= 2 * n - 1:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        # (f*g)[k] = sum_{j=j_min}^{j_max} f[j] * g[k-j]
        # g indices: k - j_max to k - j_min
        g_start = k - j_max
        g_end = k - j_min
        f_slice = f[j_min:j_max + 1]
        g_slice = g[g_start:g_end + 1]
        constraints.append(f_slice @ g_slice <= M)
    
    if not constraints:
        return np.ones(n)
    
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None
    
    if g.value is not None:
        result = np.array(g.value).flatten()
        result = np.clip(result, 0, 1000)
        if result.sum() > 1e-10:
            return result
    return None


def lp_alternation(f_init, S, max_time, rng, K_start=10, K_max=200, base_step=0.5):
    """
    LP-based alternating optimization.
    At each iteration:
      1. Compute M = max(conv(f,f))
      2. Find active set (top K lags of conv(f,f))
      3. Solve LP for g
      4. Normalize g to sum S
      5. Line search for step size t
      6. Update f = (1-t)*f + t*g
    """
    n = len(f_init)
    f = np.clip(f_init.copy(), 0, 1000)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) * S / n
    else:
        f = f * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_f = f.copy()
    it = 0
    no_improve_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        t0 = time.monotonic()
        
        c = fft_convolve_self(f)
        M = float(c.max())
        
        if M < best_score:
            best_score = M
            best_f = f.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Adaptive K
        K = min(K_max, K_start + it // 3)
        
        # Find active set: top K lags
        top_idx = np.argsort(c)[::-1][:K]
        active_indices = [int(k) for k in top_idx]
        
        # Also include lags near the center (n-1) since they often become active
        center = n - 1
        for offset in range(-2, 3):
            ci = center + offset
            if 0 <= ci < 2 * n - 1 and ci not in active_indices:
                active_indices.append(ci)
        
        # Solve LP
        g = solve_lp_for_g(f, M, active_indices)
        
        if g is None:
            # Fallback: use subgradient of max
            max_k = int(np.argmax(c))
            j_min = max(0, max_k - n + 1)
            j_max = min(n - 1, max_k)
            g = np.zeros(n)
            for j in range(j_min, j_max + 1):
                g[max_k - j] += f[j]
            g = g.max() - g + 1  # shift to be positive
            g = np.clip(g, 0, 1000)
        
        # Normalize
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g * (S / g_sum)
        g = np.clip(g, 0, 1000)
        
        # Line search
        t = base_step / (1 + 0.05 * np.log1p(it))
        f_new = None
        for _ in range(8):
            candidate = (1 - t) * f + t * g
            candidate = np.clip(candidate, 0, 1000)
            cand_sum = candidate.sum()
            if cand_sum < 1e-10:
                t *= 0.5
                continue
            candidate = candidate * (S / cand_sum)
            c_new = fft_convolve_self(candidate)
            if c_new.max() <= M * 1.001:
                f_new = candidate
                break
            t *= 0.5
            if t < 1e-6:
                break
        
        if f_new is not None:
            f = f_new
        else:
            # Try even smaller step
            t = 1e-4
            candidate = (1 - t) * f + t * g
            candidate = np.clip(candidate, 0, 1000)
            cand_sum = candidate.sum()
            if cand_sum > 1e-10:
                candidate = candidate * (S / cand_sum)
                f = candidate
        
        if it % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP iter {it}: M={M:.6f}, best={best_score:.6f}, K={K}, t={t:.4f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_score


def gradient_polish(f_init, S, max_time, rng, beta_start=10.0, beta_end=500.0):
    """FISTA with log-sum-exp smoothing for final polish."""
    n = len(f_init)
    f = np.clip(f_init.copy(), 0, 1000)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) * S / n
    else:
        f = f * (S / s)
    
    y = f.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_f = f.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = beta_start * (beta_end / beta_start) ** (it / max(1, int(max_time * 10)))
        beta = min(beta, beta_end)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_f = y.copy()
        
        # Gradient of log-sum-exp
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total
        
        # Gradient: d/dy[0] (1/beta)*log(sum exp(beta*c[k])) = 2 * sum_k w[k] * y_shift[k]
        # where the factor 2 comes from the self-convolution
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        Y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]

        if y_sum > 1e-10:

        t_mom = t_new

"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Uses LP-based alternating optimization (active set) combined with FISTA polishing.
The LP finds direction g maximizing sum(g) s.t. (f*g)[k] <= M at active lags,
then updates f toward g with line search. Multiple diverse initializations
are screened, then the best candidates receive deep LP + gradient optimization.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT, length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def solve_lp_for_g(f, M, active_indices, n):
    """
    Solve: maximize sum(g) s.t. g in [0, 1000]^n and (f*g)[k] <= M for k in active_indices.
    Returns g array or None.
    """
    g = cp.Variable(n, nonneg=True, upper_bound=1000.0)
    constraints = []

    for k in active_indices:
        k = int(k)
        if k < 0 or k >= 2 * n - 1:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        g_start = k - j_max
        g_end = k - j_min
        f_slice = f[j_min:j_max + 1]
        g_slice = g[g_start:g_end + 1]
        # Skip if all f coefficients are zero
        if np.all(f_slice < 1e-15):
            continue
        constraints.append(f_slice @ g_slice <= M)

    if len(constraints) < 1:
        return None

    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)

    solver_list = [cp.CLARABEL, cp.MOSEK, cp.SCS, cp.ECOS]
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                result = np.array(g.value).flatten()
                result = np.clip(result, 0, 1000)
                if result.sum() > 1e-10:
                    return result
            break
        except Exception:
            continue
    return None


def lp_alternation(f_init, S, max_time, rng, K_start=10, K_max=200, base_step=0.5):
    """
    LP-based alternating optimization with active set and line search.
    """
    n = len(f_init)
    f = np.clip(f_init.copy(), 0, 1000)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) * S / n
    else:
        f = f * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_f = f.copy()
    it = 0
    stagnation = 0

    while time.monotonic() - start < max_time:
        it += 1
        t0 = time.monotonic()

        c = fft_convolve_self(f)
        M = float(c.max())

        if M < best_score:
            best_score = M
            best_f = f.copy()
            stagnation = 0
        else:
            stagnation += 1

        # If stuck too long, perturb
        if stagnation > 30:
            noise = rng.normal(0, 0.02 * S / np.sqrt(n), n)
            f = project_simplex(f + noise, S)
            stagnation = 0
            continue

        # Adaptive active set size
        K = min(K_max, K_start + it // 4)

        # Find active set: top K lags + center neighborhood
        top_idx = np.argsort(c)[::-1][:K]
        active_indices = [int(k) for k in top_idx]
        center = n - 1
        for offset in range(-3, 4):
            ci = center + offset
            if 0 <= ci < 2 * n - 1:
                if ci not in active_indices:
                    active_indices.append(ci)

        # Solve LP
        g = solve_lp_for_g(f, M, active_indices, n)

        if g is None:
            # Fallback: subgradient direction from the single peak
            max_k = int(np.argmax(c))
            j_min = max(0, max_k - n + 1)
            j_max = min(n - 1, max_k)
            g = np.zeros(n)
            for j in range(j_min, j_max + 1):
                g[max_k - j] = f[j]
            g = np.maximum(g, 1e-10)
            g = g / g.max() * 100
            g = np.clip(g, 0, 1000)

        # Normalize g to sum S
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g * (S / g_sum)
        g = np.clip(g, 0, 1000)

        # Line search
        t = base_step / (1.0 + 0.03 * np.log1p(it))
        f_new = None
        for _ in range(10):
            candidate = (1.0 - t) * f + t * g
            candidate = np.clip(candidate, 0, 1000)
            cand_sum = candidate.sum()
            if cand_sum < 1e-10:
                t *= 0.5
                continue
            candidate = candidate * (S / cand_sum)
            c_new = fft_convolve_self(candidate)
            if c_new.max() <= M * 1.0001:
                f_new = candidate
                break
            t *= 0.5
            if t < 1e-8:
                break

        if f_new is not None:
            f = f_new
        else:
            # Minimal step
            t = 1e-5
            candidate = (1.0 - t) * f + t * g
            candidate = np.clip(candidate, 0, 1000)
            cand_sum = candidate.sum()
            if cand_sum > 1e-10:
                f = candidate * (S / cand_sum)

        if it % 50 == 1:
            elapsed = time.monotonic() - start
            print(f"    LP iter {it}: M={M:.6f}, best={best_score:.6f}, K={K}, "
                  f"t={t:.5f}, elapsed={elapsed:.1f}s")

    return best_f, best_score


def gradient_polish(f_init, S, max_time, rng, beta_start=10.0, beta_end=500.0, noise_scale=0.003):
    """FISTA with log-sum-exp smoothing."""
    n = len(f_init)
    f = np.clip(f_init.copy(), 0, 1000)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) * S / n
    else:
        f = f * (S / s)

    y = f.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_f = f.copy()
    it = 0
    expected_iters = max(100, int(max_time * 50))

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / expected_iters)
        beta = beta_start * (beta_end / beta_start) ** frac

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_f = y.copy()

        # Log-sum-exp gradient
        c_max = c.max()
        shifted = beta * (c - c_max)
        # Clip to avoid overflow
        shifted = np.clip(shifted, -500, 500)
        exp_bc = np.exp(shifted)
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        # Gradient via FFT cross-correlation
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        Y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta * (1.0 + 0.005 * it))
        step = max(step, 1e-8)
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - f)
        y = np.maximum(y, 0.0)
        if y.sum() > 1e-10:
            y = project_simplex(y, S)
        f = x_new
        t_mom = t_new

        # Occasional noise to escape
        if it % 500 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0

    return best_f, best_score

    # Exponential ramps
    for alpha in [1.0, 2.0, 3.0, 5.0, 8.0, 12.0]:
        inits.append((f"exp_f{alpha}", np.exp(alpha * tt)))
        inits.append((f"exp_b{alpha}", np.exp(alpha * (1.0 - tt))))

    # Power laws
    for p in [0.2, 0.5, 1.0, 2.0, 4.0]:
        inits.append((f"pow_f{p}", (tt + 0.001) ** p))
        inits.append((f"pow_b{p}", (1.0 - tt + 0.001) ** p))

    # U-shapes (bimodal)
    for w in [0.05, 0.1, 0.15, 0.25]:
        u = np.exp(-(tt - 0.5) ** 2 / (w * w))
        inits.append((f"u{w}", u))
        inits.append((f"u_inv{w}", 2.0 - u))

    # Linear
    inits.append(("lin_f", tt + 0.05))
    inits.append(("lin_b", 1.0 - tt + 0.05))

    # Sinusoidal + trend
    for freq in [2, 3, 5, 8]:
        inits.append((f"sin{freq}_f", 1.0 + 0.4 * np.sin(2 * np.pi * freq * tt) + 0.3 * tt))
        inits.append((f"sin{freq}_b", 1.0 + 0.4 * np.cos(2 * np.pi * freq * tt) + 0.3 * (1 - tt)))

    # Random asymmetric
    for i in range(4):
        bias = rng.random() * 4
        r = rng.random(n) * np.exp(bias * tt)
        inits.append((f"rand{i}", r))

    # Triangular
    tri = 1.0 - np.abs(2 * tt - 1)
    inits.append(("tri", tri))
    inits.append(("tri_skew", tri * (0.5 + tt)))

    # Geometric
    ratio = 0.98
    geom = ratio ** np.arange(n)
    inits.append(("geom", geom.astype(np.float64)))
    inits.append(("geom_r", (geom[::-1]).astype(np.float64)))

    print(f"Total initializations: {len(inits)}")

    # --- Phase 1: Quick screening (gradient polish only) ---
    print("=== Phase 1: Quick screening ===")
    t_start = time.monotonic()
    screen_time = 2.5
    screened = []

    for name, arr in inits:
        if time.monotonic() > deadline - 600:
            break
        arr_n = safe_normalize(arr, n)
        a_q, _ = gradient_polish(arr_n, S, screen_time, rng,
                                  beta_start=5.0, beta_end=50.0, noise_scale=0.005)
        a_q = safe_normalize(a_q, n)
        obj = get_objective(a_q)
        screened.append((obj, name, a_q))

    screened.sort(key=lambda x: x[0])
    print(f"  Screening done in {time.monotonic()-t_start:.1f}s")
    print(f"  Top 5:")
    for obj, name, _ in screened[:5]:
        print(f"    {name}: {obj:.6f}")

    if screened[0][0] < global_best:
        global_best = screened[0][0]
        global_a = screened[0][2].copy()

    # --- Phase 2: LP optimization on top candidates ---
    print("=== Phase 2: LP optimization ===")
    t_start2 = time.monotonic()

    # Select top candidates for LP
    top_k_cands = min(6, len(screened))
    candidates = [s[2].copy() for s in screened[:top_k_cands]]

    # Always include parent if available
    if parent is not None:
        parent_n = safe_normalize(parent, n)
        # Check if already in candidates
        candidates.append(parent_n)

    remaining2 = max(60, deadline - time.monotonic())
    time_per_cand = remaining2 * 0.55 / max(1, len(candidates))
    print(f"  {len(candidates)} candidates, ~{time_per_cand:.0f}s each")

    for i, cand in enumerate(candidates):
        if time.monotonic() > deadline - 120:
            break
        print(f"  Cand {i+1}/{len(candidates)}:")
        a_opt, score = lp_alternation(cand, S, time_per_cand, rng,
                                       K_start=15, K_max=150, base_step=0.5)
        a_opt = safe_normalize(a_opt, n)
        obj = get_objective(a_opt)
        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
            print(f"    *** New best: {obj:.6f} ***")
        else:
            print(f"    Score: {obj:.6f}")

    print(f"  Phase 2 done in {time.monotonic()-t_start2:.1f}s, best={global_best:.6f}")

    # --- Phase 3: Deep polish ---
    print("=== Phase 3: Deep polish ===")
    if global_a is not None:
        remaining3 = max(10, deadline - time.monotonic())

        # LP with larger active set
        seg1 = remaining3 * 0.35
        if seg1 > 15:
            a_lp, _ = lp_alternation(global_a, S, seg1, rng,
                                      K_start=50, K_max=300, base_step=0.3)
            a_lp = safe_normalize(a_lp, n)
            obj_lp = get_objective(a_lp)
            if obj_lp < global_best:
                global_best = obj_lp
                global_a = a_lp.copy()
                print(f"  LP polish (large K): {obj_lp:.6f}")

        # High-beta FISTA
        remaining3b = max(10, deadline - time.monotonic())
        seg2 = remaining3b * 0.3
        if seg2 > 10:
            a_g, _ = gradient_polish(global_a, S, seg2, rng,
                                      beta_start=30.0, beta_end=800.0, noise_scale=0.001)
            a_g = safe_normalize(a_g, n)
            obj_g = get_objective(a_g)
            if obj_g < global_best:
                global_best = obj_g
                global_a = a_g.copy()
                print(f"  FISTA high-beta: {obj_g:.6f}")

        # Final LP
        remaining3c = max(5, deadline - time.monotonic())
        if remaining3c > 10:
            a_f, _ = lp_alternation(global_a, S, remaining3c, rng,
                                     K_start=80, K_max=400, base_step=0.2)
            a_f = safe_normalize(a_f, n)
            obj_f = get_objective(a_f)
            if obj_f < global_best:
                global_best = obj_f
                global_a = a_f.copy()
                print(f"  Final LP: {obj_f:.6f}")

        # Final tiny FISTA
        remaining3d = max(3, deadline - time.monotonic())
        if remaining3d > 5:
            a_t, _ = gradient_polish(global_a, S, remaining3d, rng,
                                      beta_start=100.0, beta_end=2000.0, noise_scale=0.0001)
            a_t = safe_normalize(a_t, n)
            obj_t = get_objective(a_t)
            if obj_t < global_best:
                global_best = obj_t
                global_a = a_t.copy()
                print(f"  Final FISTA: {obj_t:.6f}")

    # --- Verification and return ---
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print(f"\nInternal best: {global_best:.6f}")
    print(f"Verified by evaluate_sequence: {ver:.6f}" if np.isfinite(ver)
          else f"Verified: INF (fallback needed)")
    print(f"Sequence length: {len(final_list)}")

    if not np.isfinite(ver):
        print("  Using fallback uniform sequence")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print(f"  Fallback score: {ver:.6f}")

    return final_list
