        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        k_star = int(np.argmax(c))

        valid_lo = max(0, k_star - (n - 1))
        valid_hi = min(n - 1, k_star)
        valid_i = np.arange(valid_lo, valid_hi + 1)
        if len(valid_i) == 0:
            break
        products = a[valid_i] * a[k_star - valid_i]
        i_max = valid_i[np.argmax(products)]
        j_max = k_star - i_max

        if i_max == j_max:
            add_idx = int(rng.integers(n))
            if add_idx == i_max:
                add_idx = (add_idx + 1) % n
            delta = a[i_max] * 0.1
            a[i_max] -= 2 * delta
            a[add_idx] += delta
            a = normalize_nonneg(a, S)
            c_new = fft_convolve_self(a)
            if c_new.max() >= cur_obj - 1e-12:
                a = best_a.copy()
            continue

        # Strategy 1: reduce the larger of the pair, add to random position
        if a[i_max] >= a[j_max]:
            red = i_max
            keep = j_max
        else:
            red = j_max
            keep = i_max

        moved = False
        for frac in [0.05, 0.1, 0.15, 0.03, 0.08]:
            delta = a[red] * frac
            add_idx = int(rng.integers(n))
            a[red] -= delta
            a[add_idx] += delta
            a = normalize_nonneg(a, S)
            c_new = fft_convolve_self(a)
            if c_new.max() < cur_obj - 1e-12:
                moved = True
                break
            a = best_a.copy()

        if not moved:
            # Strategy 2: reduce both, add to two random positions
            for frac in [0.05, 0.1, 0.15]:
                d1 = a[i_max] * frac
                d2 = a[j_max] * frac
                a[i_max] -= d1
                a[j_max] -= d2
                a1 = int(rng.integers(n))
                a2 = int(rng.integers(n))
                a[a1] += d1
                a[a2] += d2
                a = normalize_nonneg(a, S)
                c_new = fft_convolve_self(a)
                if c_new.max() < cur_obj - 1e-12:
                    moved = True
                    break
                a = best_a.copy()

        if not moved:
            # Strategy 3: swap mass within the sequence
            red_idx = i_max if a[i_max] > a[j_max] else j_max
            # Add to position far from peak contributions
            candidates = np.random.default_rng(no_improve).integers(0, n, size=5)
            for add_idx in candidates:
                if abs(add_idx - red_idx) < 5:
                    continue
                delta = a[red_idx] * 0.08
                a[red_idx] -= delta
                a[add_idx] += delta
                a = normalize_nonneg(a, S)
                c_new = fft_convolve_self(a)
                if c_new.max() < cur_obj - 1e-12:
                    moved = True
                    break
                a = best_a.copy()

        if not moved:
            # Random perturbation to escape
            a = best_a + 0.002 * rng.standard_normal(n)
            a = normalize_nonneg(a, S)

    return best_a, best_obj


def coord_descent(a_init, n, S, time_budget, max_sweeps=100):
    """Coordinate descent with many multiplicative factors."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    rng = np.random.default_rng(123)

    factors = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.4, 1.7]

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-10:
                continue

            orig = a[i]
            best_local = orig
            best_local_obj = best_obj

            for f in factors:
                a[i] = orig * f
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj = c_check.max()
                if obj < best_local_obj - 1e-12:
                    best_local_obj = obj
                    best_local = orig * f

            a[i] = best_local
            a = normalize_nonneg(a, S)

            if best_local_obj < best_obj - 1e-12:
                best_obj = best_local_obj
                best_a = a.copy()
                improved = True

        if not improved and sweep >= 3:
            break

    return best_a, best_obj


def geometric_init(n, S, decay):
    """Create a geometric decay initialization."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-decay * i)
    return a * (S / a.sum())


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)

    deadline = time.monotonic() + total_budget

    global_best = np.inf
    global_a = None
    n = 4096
    S = np.sqrt(2.0 * float(n))

    # Load given sequence
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            n = len(given_raw)
            S = np.sqrt(2.0 * float(n))
            a_given = given_raw * (S / given_raw.sum())
            ver = evaluate_sequence(a_given.tolist())
            print("Given: n=%d, obj=%.7f" % (n, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_given.copy()
    except Exception as e:
        print("Could not load given: " + str(e))

    if global_a is None:
        global_a = np.full(n, S / n)
        global_best = evaluate_sequence(global_a.tolist())
        print("Fallback uniform: n=%d, obj=%.7f" % (n, global_best))

    n = len(global_a)
    S = np.sqrt(2.0 * float(n))
    print("Optimization start: n=%d, best=%.7f, budget=%.0fs" % (n, global_best, total_budget))

    # Phase 0: Quick exploration with geometric inits (10% time)
    p0_budget = 0.10 * total_budget

    decays = [0.0005, 0.001, 0.002, 0.005]
    per_try = p0_budget / len(decays)

    for d in decays:
        if time.monotonic() > deadline - 10:

        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_best.copy()
            print("  Geom decay=%.4f: obj=%.7f" % (d, obj))

    # Phase 1: Main subgradient on best (35% time)
    p1_budget = 0.35 * total_budget
    print("\nPhase 1: Subgradient T50-5000 (budget=%.0fs)" % p1_budget)

    a_work = normalize_nonneg(global_a.copy(), S)

    a_best = normalize_nonneg(a_best, S)
    obj = evaluate_sequence(a_best.tolist())
    if np.isfinite(obj) and obj < global_best:
        global_best = obj
        global_a = a_best.copy()
        print("  P1: obj=%.7f" % obj)

    # Phase 2: Peak-pair local search (25% time)
    p2_budget = 0.25 * total_budget

        global_a = a_best.copy()
        print("  P2: obj=%.7f" % obj)

        global_a = a_best.copy()
        print("  P3: obj=%.7f" % obj)

        final_a = [1.0] * 256
        print("Fallback: obj=%.7f" % evaluate_sequence(final_a))

    return final_a
