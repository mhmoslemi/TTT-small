"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Gerchberg-Saxton style frequency-domain flattening
2. Exponential peak-weighted gradient descent (Adam)
3. Nesterov accelerated subgradient with backtracking
4. Coordinate descent polish
5. Multiple initialization strategies and random restarts

Key innovation: Alternating between frequency-domain spectral flattening
(iteratively compressing the dynamic range of the convolution spectrum)
and time-domain gradient steps with exponential peak weighting.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def gs_flatten(a, S, alpha=0.5, n_iter=1):
    """
    Gerchberg-Saxton style spectral flattening.
    Iteratively compresses the dynamic range of the convolution spectrum.
    """
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    a = a.copy()
    
    for _ in range(n_iter):
        A = fft(a, fsize)
        C = A * A  # spectrum of convolution
        
        # Target: compress |C| toward its geometric mean (or mean)
        abs_C = np.abs(C)
        abs_C_safe = np.maximum(abs_C, 1e-30)
        
        # Use geometric mean as target (preserves total "energy" better)
        log_mean = np.mean(np.log(abs_C_safe))
        target = np.exp(log_mean)
        
        # Compress: C_new = C * (target / |C|)^alpha
        ratio = (target / abs_C_safe) ** alpha
        C_new = C * ratio
        
        # Enforce DC: C_new[0] should be S^2
        C_new[0] = S * S
        
        # Recover A from C: A = sqrt(C) (element-wise)
        # sqrt of complex: sqrt(|z|) * exp(i*angle(z)/2)
        A_new = np.sqrt(C_new)  # numpy handles complex sqrt correctly
        
        # Ensure A_new[0] is real and positive (= S)
        A_new[0] = S
        
        a = np.real(ifft(A_new, fsize))[:n]
        a = normalize_nonneg(a, S)
    
    return a


def compute_peak_weighted_subgrad(c, a, T):
    """Compute subgradient with exponential peak weighting."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    
    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # g[m] = 2 * sum_k w[k] * a[k-m] (cross-correlation)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def run_adam_optimize(a_init, n, S, time_budget, T_init=10.0, T_final=500.0,
                      lr_init=0.05, gs_every=100, gs_alpha=0.3):
    """
    Adam optimization with exponential peak weighting and periodic GS flattening.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Log-parameterization for non-negativity
    a_safe = np.maximum(a, 1e-10)
    theta = np.log(a_safe / a_safe.sum())
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    no_improve = 0
    max_no_improve = 200
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        
        # Temperature schedule
        T = T_init * (T_final / T_init) ** (progress ** 0.6)
        
        # Convert theta to a
        a = np.exp(theta - theta.max())
        a = a * (S / a.sum())
        a = np.maximum(a, 0.0)
        
        # Compute objective
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # GS flattening every gs_every steps
        if t % gs_every == 0:
            a_gs = gs_flatten(a, S, alpha=gs_alpha * (1 - 0.5 * progress), n_iter=1)
            c_gs = fft_convolve_self(a_gs)
            if c_gs.max() < best_obj - 1e-12:
                best_obj = c_gs.max()
                best_a = a_gs.copy()
                a = a_gs.copy()
                a_safe = np.maximum(a, 1e-10)
                theta = np.log(a_safe / a_safe.sum())
                no_improve = 0
                continue
        
        # Restart if stuck
        if no_improve > max_no_improve:
            noise_scale = 0.01 * (1.0 - progress) + 0.001
            a = best_a + noise_scale * np.random.default_rng(t).standard_normal(n)
            a = normalize_nonneg(a, S)
            a_safe = np.maximum(a, 1e-10)
            theta = np.log(a_safe / a_safe.sum())
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            continue
        
        # Compute gradient
        g_a = compute_peak_weighted_subgrad(c, a, T)
        
        # Chain rule for log-parameters
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot
        
        # Adam update
        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break
        
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr = lr_init * (0.3 + 0.7 * (1.0 - progress))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
    
    return best_a, best_obj


def run_nesterov_sg(a_init, n, S, time_budget, T_init=50.0, T_final=2000.0,
                    lr_init=0.05, momentum=0.85, gs_every=200, gs_alpha=0.2):
    """Nesterov accelerated subgradient with periodic GS flattening."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    a_prev = a.copy()
    beta = momentum
    step = 0
    no_improve = 0
    max_no_improve = 150
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.7)
        
        # Nesterov lookahead
        y = a + (beta - 1.0) / (1.0 + beta) * (a - a_prev)
        y = normalize_nonneg(y, S)
        
        c_y = fft_convolve_self(y)
        
        # Track best at current point
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # GS flattening periodically
        if step % gs_every == 0 and step > 0:
            a_gs = gs_flatten(a, S, alpha=gs_alpha, n_iter=1)
            c_gs = fft_convolve_self(a_gs)
            if c_gs.max() < best_obj - 1e-12:
                best_obj = c_gs.max()
                best_a = a_gs.copy()
                a = a_gs.copy()
                a_prev = a.copy()
                no_improve = 0
                step += 1
                continue
        
        # Restart
        if no_improve > max_no_improve:
            scale = 0.01 * max(0.1, 1.0 - progress)
            a = best_a + scale * np.random.default_rng(step).standard_normal(n)
            a = normalize_nonneg(a, S)
            a_prev = a.copy()
            no_improve = 0
            step += 1
            continue
        
        # Subgradient
        g = compute_peak_weighted_subgrad(c, a, T)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g_dir = g / gnorm
        
        # Backtracking line search
        lr = lr_init * (1.0 - 0.7 * progress)
        best_step_a = a.copy()
        
        for i in range(10):
            a_trial = a - lr * g_dir
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_obj - 1e-10 * cur_obj:
                best_step_a = a_trial
                break
            lr *= 0.5
        
        a_prev = a.copy()
        a = best_step_a
        step += 1
    
    return best_a, best_obj


def coordinate_descent(a, n, S, time_budget, max_sweeps=50):
    """Coordinate descent with multiplicative updates."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(42)
    factors = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            original = a[i]
            if original < 1e-12:
                continue
            
            best_local = original
            best_local_obj = best_obj
            
            for factor in factors:
                a[i] = original * factor
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-12:
                    best_local_obj = obj_check
                    best_local = original * factor
            
            a[i] = best_local
            a = normalize_nonneg(a, S)
            
            if best_local_obj < best_obj - 1e-12:
                best_obj = best_local_obj
                best_a = a.copy()
                improved = True
            else:
                c = fft_convolve_self(a)
        
        if not improved:
            break
    
    return best_a, best_obj


def get_initialization(name, n, S, rng):
    """Generate various initializations."""
    i = np.arange(n)
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "cosine":
        # Multi-frequency cosine modulation
        a = 1.0
        for k in range(1, 6):
            freq = rng.integers(1, n // 2)
            phase = rng.random() * 2 * np.pi
            amp = 0.15 / k
            a += amp * np.cos(2 * np.pi * freq * i / n + phase)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "gs_init":
        # GS-style initialization: start with random phases in frequency domain
        fsize = 1 << (2 * n - 2).bit_length()
        phases = rng.random(fsize) * 2 * np.pi
        A = np.exp(1j * phases)
        A[0] = 1.0  # DC positive
        a = np.real(ifft(A, fsize))[:n]
        a = normalize_nonneg(a, S)
        # Run a few GS iterations
        a = gs_flatten(a, S, alpha=0.5, n_iter=5)
        return a
    elif name == "chirp":
        # Chirp-like sequence (quadratic phase)
        t = i / n
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * t + 0.5 * 2 * np.pi * 7 * t**2)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "alternating":
        # Alternating high/low
        a = np.ones(n)
        a[::2] = 1.3
        a[1::2] = 0.7
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def run_gs_only(a_init, n, S, time_budget, alpha_init=0.5, alpha_final=0.1):
    """Pure GS-style optimization with decreasing alpha."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    iters_done = 0
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        alpha = alpha_init * (alpha_final / alpha_init) ** progress
        
        # GS flatten with 1 iteration
        a_new = gs_flatten(a, S, alpha=alpha, n_iter=1)
        
        # Also try a "peak-shaving" variant
        # Find the peak and try to specifically reduce it
        c_new = fft_convolve_self(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller alpha
            a_new2 = gs_flatten(a, S, alpha=alpha * 0.3, n_iter=1)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < best_obj - 1e-14:
                best_obj = c_new2.max()
                best_a = a_new2.copy()
                a = a_new2
            else:
                # Stuck, add small perturbation
                a = a + 0.001 * np.random.default_rng(iters_done).standard_normal(n)
                a = normalize_nonneg(a, S)
        
        iters_done += 1
    
    return best_a, best_obj

Wait, I need to think about whether my GS approach is actually correct and effective. Let me trace through it:

1. We have `a` with `sum(a) = S`.
2. `A = FFT(a, fsize)` where fsize >= 2n-1.
3. `C = A * A` - this is the DFT of the circular convolution of the zero-padded `a` with itself. Since fsize >= 2n-1, this equals the linear convolution. So `ifft(C)` gives the full linear convolution of `a` with itself.
4. We want to flatten `C` (make its time-domain version, i.e., the convolution, as flat as possible).
5. `C_new = C * (target / |C|)^alpha` - this compresses the magnitude of C toward `target`.

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Gerchberg-Saxton style spectral flattening (compress convolution spectrum dynamic range)
2. Peak-shaving in frequency domain (targeted reduction of dominant autocorrelation peaks)
3. Exponential peak-weighted Adam optimization (smooth max approximation)
4. Nesterov accelerated subgradient with backtracking line search
5. Coordinate descent polish
6. Multiple initialization strategies and hybrid scheduling

Key innovation: Alternating between frequency-domain spectral operations
(GS flattening, peak shaving) and time-domain gradient steps.
The GS step provides global moves that escape local minima, while gradient

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def gs_flatten(a, S, alpha=0.5):
    """
    One step of Gerchberg-Saxton style spectral flattening.
    Compresses the dynamic range of the convolution spectrum toward its geometric mean.
    """
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    a = a.copy().astype(np.float64)

    A = fft(a, fsize)
    C = A * A  # spectrum of linear convolution

    abs_C = np.abs(C)
    abs_C_safe = np.maximum(abs_C, 1e-30)

    # Geometric mean as target
    log_mean = np.mean(np.log(abs_C_safe))
    target = np.exp(log_mean)

    # Compress magnitude: C_new = C * (target / |C|)^alpha
    ratio = (target / abs_C_safe) ** alpha
    C_new = C * ratio

    # Enforce DC
    C_new[0] = S * S

    # Recover A from C: A_new^2 = C_new
    A_new = np.sqrt(C_new)
    A_new[0] = S  # enforce positive real DC

    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = normalize_nonneg(a_new, S)
    return a_new


def gs_peak_shave(a, S, n_peaks=5, strength=0.3):
    """
    Targeted peak shaving: identify the top peaks in the convolution and
    reduce them by modifying the spectrum.
    """
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    a = a.copy().astype(np.float64)

    c = fft_convolve_self(a)
    c_max = c.max()
    if c_max < 1e-30:
        return a

    # Find top peak positions
    peak_threshold = 0.8 * c_max
    peak_positions = np.where(c >= peak_threshold)[0]
    if len(peak_positions) == 0:
        peak_positions = np.array([np.argmax(c)])

    A = fft(a, fsize)
    C = A * A

    # For each peak position, subtract a complex exponential from C
    # c[p] = (1/fsize) * sum_k C[k] * exp(2*pi*i*k*p/fsize)
    # To reduce c[p], subtract delta * exp(-2*pi*i*k*p/fsize) from C[k]
    for p in peak_positions[:n_peaks]:
        k = np.arange(fsize)
        exp_term = np.exp(-2j * np.pi * k * p / fsize)
        # Weight by how much this peak contributes
        weight = strength * (c[p] / c_max) ** 2
        C -= weight * c_max * exp_term

    # Enforce DC
    C[0] = S * S

    A_new = np.sqrt(C)
    A_new[0] = S

    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = normalize_nonneg(a_new, S)
    return a_new


def compute_peak_weighted_subgrad(c, a, T):
    """Compute subgradient with exponential peak weighting."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)

    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # g[m] = 2 * sum_k w[k] * a[k-m]
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def run_adam_gs(a_init, n, S, time_budget, T_init=10.0, T_final=500.0,
                lr_init=0.05, gs_every=40, gs_alpha=0.3, shave_every=80):
    """
    Adam optimization with peak weighting, periodic GS flattening and peak shaving.
    Uses log-parameterization to maintain non-negativity.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    a_safe = np.maximum(a, 1e-12)
    theta = np.log(a_safe / a_safe.sum())

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    no_improve = 0
    max_no_improve = 150

    start = time.monotonic()
    rng_local = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.6)

        # Convert theta to a
        a = np.exp(theta - theta.max())
        a = a * (S / a.sum())
        a = np.maximum(a, 0.0)

        # Compute objective
        c = fft_convolve_self(a)
        cur_obj = c.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # GS flattening
        if t % gs_every == 0:
            alpha_eff = gs_alpha * (1.0 - 0.5 * progress)
            a_gs = gs_flatten(a, S, alpha=alpha_eff)
            c_gs = fft_convolve_self(a_gs)
            if c_gs.max() < best_obj - 1e-13:
                best_obj = c_gs.max()
                best_a = a_gs.copy()
                a = a_gs.copy()
                a_safe = np.maximum(a, 1e-12)
                theta = np.log(a_safe / a_safe.sum())
                no_improve = 0
                continue

            # Peak shaving
            a_sh = gs_peak_shave(a, S, n_peaks=5, strength=0.2 * (1 - progress))
            c_sh = fft_convolve_self(a_sh)
            if c_sh.max() < best_obj - 1e-13:
                best_obj = c_sh.max()
                best_a = a_sh.copy()
                a = a_sh.copy()
                a_safe = np.maximum(a, 1e-12)
                theta = np.log(a_safe / a_safe.sum())
                no_improve = 0
                continue

        # Restart if stuck
        if no_improve > max_no_improve:
            noise_scale = 0.005 * (1.0 - 0.5 * progress) + 0.001
            a = best_a + noise_scale * rng_local.standard_normal(n)
            a = normalize_nonneg(a, S)
            a_safe = np.maximum(a, 1e-12)
            theta = np.log(a_safe / a_safe.sum())
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            continue

        # Compute gradient
        g_a = compute_peak_weighted_subgrad(c, a, T)
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot

        # Adam update
        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break

        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_init * (0.3 + 0.7 * (1.0 - progress))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_obj


def run_nesterov_gs(a_init, n, S, time_budget, T_init=50.0, T_final=2000.0,
                    lr_init=0.05, momentum=0.85, gs_every=100):
    """Nesterov accelerated subgradient with periodic GS operations."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    a_prev = a.copy()
    beta = momentum
    step = 0
    no_improve = 0
    max_no_improve = 120

    start = time.monotonic()
    rng_local = np.random.default_rng(999)

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.7)

        # Nesterov lookahead
        y = a + (beta - 1.0) / (1.0 + beta) * (a - a_prev)
        y = normalize_nonneg(y, S)

        # Track best at current point
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # GS operations periodically
        if step % gs_every == 0 and step > 0:
            a_gs = gs_flatten(a, S, alpha=0.25 * (1 - 0.5 * progress))
            c_gs = fft_convolve_self(a_gs)
            if c_gs.max() < best_obj - 1e-13:
                best_obj = c_gs.max()
                best_a = a_gs.copy()
                a = a_gs.copy()
                a_prev = a.copy()
                no_improve = 0
                step += 1
                continue

            a_sh = gs_peak_shave(a, S, n_peaks=3, strength=0.15)
            c_sh = fft_convolve_self(a_sh)
            if c_sh.max() < best_obj - 1e-13:
                best_obj = c_sh.max()
                best_a = a_sh.copy()
                a = a_sh.copy()
                a_prev = a.copy()
                no_improve = 0
                step += 1
                continue

        # Restart
        if no_improve > max_no_improve:
            scale = 0.008 * max(0.1, 1.0 - progress)
            a = best_a + scale * rng_local.standard_normal(n)
            a = normalize_nonneg(a, S)
            a_prev = a.copy()
            no_improve = 0
            step += 1
            continue

        # Subgradient
        g = compute_peak_weighted_subgrad(c, a, T)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g_dir = g / gnorm

        # Backtracking
        lr = lr_init * (1.0 - 0.7 * progress)
        best_step_a = a.copy()

        for i in range(10):
            a_trial = a - lr * g_dir
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_obj - 1e-10 * cur_obj:
                best_step_a = a_trial
                break
            lr *= 0.5

        a_prev = a.copy()
        a = best_step_a
        step += 1

    return best_a, best_obj


def run_gs_primary(a_init, n, S, time_budget, alpha_start=0.6, alpha_end=0.05):
    """
    GS-primary optimization: alternate between flattening and peak shaving,
    with occasional small gradient steps when stuck.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    iters = 0
    stagnant = 0
    rng_local = np.random.default_rng(555)

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        # GS flatten
        a_new = gs_flatten(a, S, alpha=alpha)
        c_new = fft_convolve_self(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
        else:
            # Try peak shaving
            a_sh = gs_peak_shave(a, S, n_peaks=5, strength=0.2 * (1 - progress))
            c_sh = fft_convolve_self(a_sh)
            if c_sh.max() < best_obj - 1e-14:
                best_obj = c_sh.max()
                best_a = a_sh.copy()
                a = a_sh
                stagnant = 0
            else:
                stagnant += 1
                if stagnant > 5:
                    # Try a gradient step
                    T = 50.0 + 500.0 * progress
                    g = compute_peak_weighted_subgrad(fft_convolve_self(a), a, T)
                    gnorm = np.linalg.norm(g)
                    if gnorm > 1e-15:
                        lr = 0.02 * (1 - 0.5 * progress)
                        a_trial = a - lr * g / gnorm
                        a_trial = normalize_nonneg(a_trial, S)
                        c_trial = fft_convolve_self(a_trial)
                        if c_trial.max() < best_obj - 1e-14:
                            best_obj = c_trial.max()
                            best_a = a_trial.copy()
                            a = a_trial
                            stagnant = 0
                        else:
                            # Perturb and try again
                            a = a + 0.002 * rng_local.standard_normal(n)
                            a = normalize_nonneg(a, S)
                    else:
                        a = a + 0.002 * rng_local.standard_normal(n)
                        a = normalize_nonneg(a, S)

        iters += 1

    return best_a, best_obj


def coordinate_descent(a, n, S, time_budget, max_sweeps=30):
    """Coordinate descent with multiplicative updates."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(42)
    factors = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            original = a[i]
            if original < 1e-12:
                continue

            best_local = original
            best_local_obj = best_obj

            for factor in factors:
                a[i] = original * factor
                a_norm = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-13:
                    best_local_obj = obj_check
                    best_local = original * factor

            a[i] = best_local
            a = normalize_nonneg(a, S)

            if best_local_obj < best_obj - 1e-13:
                best_obj = best_local_obj
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_obj


def get_initialization(name, n, S, rng):
    """Generate various initializations."""
    i = np.arange(n)
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "cosine":
        a = 1.0
        for k in range(1, 8):
            freq = rng.integers(1, n // 3)
            phase = rng.random() * 2 * np.pi
            amp = 0.12 / np.sqrt(k)
            a += amp * np.cos(2 * np.pi * freq * i / n + phase)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "gs_init":
        fsize = 1 << (2 * n - 2).bit_length()
        phases = rng.random(fsize) * 2 * np.pi
        A = np.exp(1j * phases)
        A[0] = 1.0
        a = np.real(ifft(A, fsize))[:n]
        a = normalize_nonneg(a, S)
        for _ in range(10):
            a = gs_flatten(a, S, alpha=0.5)
        return a
    elif name == "chirp":
        t = i / n
        a = 1.0 + 0.15 * np.cos(2 * np.pi * 5 * t) + 0.1 * np.cos(2 * np.pi * 13 * t ** 2)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "alternating":
        a = np.ones(n)
        a[::3] = 1.4
        a[1::3] = 0.6
        a[2::3] = 1.0
        return a * (S / a.sum())
    elif name == "gaussian_mod":
        a = np.exp(-0.3 * np.sin(2 * np.pi * 7 * i / n) ** 2)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses hybrid GS + gradient optimization with multiple phases.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(10.0, float(budget_s) - 30.0)
    deadline = time.monotonic() + total_budget

    global_best = np.inf
    global_a = None

    # Load given sequence
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: " + str(e))

    n = 4096
    S = np.sqrt(2.0 * n)

    # ============ PHASE 1: Multi-init GS exploration (12% time) ============
    p1_budget = 0.12 * total_budget
    p1_deadline = time.monotonic() + p1_budget

    print("=== Phase 1: GS Exploration (%.0fs, n=%d) ===" % (p1_budget, n))

    inits = ["gs_init", "cosine", "random", "chirp", "alternating", "gaussian_mod"]
    per_init = p1_budget / len(inits)

    for init_name in inits:
        if time.monotonic() > p1_deadline - 3:
            break
        a_init = get_initialization(init_name, n, S, rng)

        # Quick GS run
        a_best, obj_best = run_gs_primary(a_init, n, S, per_init,
                                           alpha_start=0.7, alpha_end=0.15)
        a_check = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_check.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_check.copy()
            print("  %s -> %.7f" % (init_name, ver))

    # ============ PHASE 2: Adam + GS Hybrid (35% time) ============
    p2_budget = 0.35 * total_budget
    p2_deadline = time.monotonic() + p2_budget

    if global_a is not None and p2_budget > 20:
        a = global_a.copy()
        a = normalize_nonneg(a, S)
        print("\n=== Phase 2: Adam+GS (%.0fs) ===" % p2_budget)

        # Round 1: Moderate temperature
        r1 = 0.4 * p2_budget
        a_best, _ = run_adam_gs(a, n, S, r1,
                                 T_init=5.0, T_final=200.0, lr_init=0.04,
                                 gs_every=40, gs_alpha=0.4)
        a = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            print("  A1: %.7f" % ver)

        # Round 2: Higher temperature
        r2 = 0.35 * p2_budget
        a_best, _ = run_adam_gs(a, n, S, r2,
                                 T_init=50.0, T_final=800.0, lr_init=0.03,
                                 gs_every=50, gs_alpha=0.25)
        a = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            print("  A2: %.7f" % ver)

        # Round 3: Very high T
        remaining = p2_deadline - time.monotonic()
        if remaining > 10:
            a_best, _ = run_adam_gs(a, n, S, remaining,
                                     T_init=300.0, T_final=2000.0, lr_init=0.02,
                                     gs_every=30, gs_alpha=0.15)
            a = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a.copy()
                print("  A3: %.7f" % ver)

    # ============ PHASE 3: Nesterov SG + GS (25% time) ============
    p3_budget = 0.25 * total_budget
    p3_deadline = time.monotonic() + p3_budget

    if global_a is not None and p3_budget > 20:
        a = global_a.copy()
        a = normalize_nonneg(a, S)
        print("\n=== Phase 3: Nesterov SG (%.0fs) ===" % p3_budget)

        r1 = 0.45 * p3_budget
        a_best, _ = run_nesterov_gs(a, n, S, r1,
                                     T_init=30.0, T_final=400.0, lr_init=0.05,
                                     momentum=0.85, gs_every=80)
        a = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            print("  S1: %.7f" % ver)

        r2 = 0.3 * p3_budget
        a_best, _ = run_nesterov_gs(a, n, S, r2,
                                     T_init=200.0, T_final=1500.0, lr_init=0.03,
                                     momentum=0.9, gs_every=60)
        a = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            print("  S2: %.7f" % ver)

        remaining = p3_deadline - time.monotonic()
        if remaining > 10:
            a_best, _ = run_gs_primary(a, n, S, remaining,
                                        alpha_start=0.4, alpha_end=0.05)
            a = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a.copy()
                print("  GS: %.7f" % ver)

    # ============ PHASE 4: Coordinate Descent (15% time) ============
    p4_budget = 0.15 * total_budget
    p4_deadline = time.monotonic() + p4_budget

    if global_a is not None and p4_budget > 10:
        a = global_a.copy()
        a = normalize_nonneg(a, S)
        print("\n=== Phase 4: CD Polish (%.0fs) ===" % p4_budget)

        # Split CD into two passes with a GS in between
        cd1 = p4_budget * 0.4
        a_best, _ = coordinate_descent(a, n, S, cd1, max_sweeps=20)
        a = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            print("  CD1: %.7f" % ver)

        # GS between CD passes
        gs_time = min(5.0, p4_budget * 0.15)
        a_best, _ = run_gs_primary(global_a, n, S, gs_time,
                                    alpha_start=0.3, alpha_end=0.05)
        a = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            print("  GS-mid: %.7f" % ver)

        remaining = p4_deadline - time.monotonic()
        if remaining > 5:
            a_best, _ = coordinate_descent(global_a, n, S, remaining, max_sweeps=15)
            a = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a.copy()
                print("  CD2: %.7f" % ver)

    # ============ PHASE 5: Final restarts + polish (13% time) ============
    p5_budget = deadline - time.monotonic()
    if global_a is not None and p5_budget > 10:
        a = global_a.copy()
        a = normalize_nonneg(a, S)
        print("\n=== Phase 5: Final (%.0fs) ===" % p5_budget)

        n_rounds = max(3, int(p5_budget / 20))
        per_round = p5_budget / (n_rounds + 2)

        for r in range(n_rounds):
            if time.monotonic() > deadline - 5:
                break
            noise = 0.003 * (1 + 0.3 * r)
            a_noisy = global_a + noise * rng.standard_normal(n)
            a_noisy = normalize_nonneg(a_noisy, S)

            # Alternate between GS-primary and Adam
            if r % 2 == 0:
                a_best, _ = run_gs_primary(a_noisy, n, S, per_round,
                                            alpha_start=0.5, alpha_end=0.1)
            else:
                a_best, _ = run_adam_gs(a_noisy, n, S, per_round,
                                         T_init=100.0, T_final=1000.0,
                                         lr_init=0.03, gs_every=50, gs_alpha=0.2)
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  R%d: %.7f" % (r, ver))

        # Final GS on best
        remaining = deadline - time.monotonic()
        if remaining > 3:
            a_best, _ = run_gs_primary(global_a, n, S, remaining,
                                        alpha_start=0.2, alpha_end=0.03)
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  FinalGS: %.7f" % ver)

            # Final CD
            remaining2 = deadline - time.monotonic()
            if remaining2 > 5:
                a_best, _ = coordinate_descent(global_a, n, S, remaining2, max_sweeps=10)
                a_best = normalize_nonneg(a_best, S)
                ver = evaluate_sequence(a_best.tolist())
                if np.isfinite(ver) and ver < global_best:
                    global_best = ver
                    global_a = a_best.copy()
                    print("  FinalCD: %.7f" % ver)

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: tracked=%.7f verified=%.7f" % (global_best, ver))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)

    return final_a
