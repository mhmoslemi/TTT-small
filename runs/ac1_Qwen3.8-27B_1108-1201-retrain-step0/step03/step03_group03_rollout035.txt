```python
def project_simplex(v, S):

        return v
    v = np.asarray(v, dtype=np.float64)
```

```python
def smooth_objective_and_grad(z, S, beta):
    """Returns (objective_value, gradient) for L-BFGS."""
    a = softmax_params_to_a(z, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    w = exp_bc / exp_bc.sum()

    obj = c_max + np.log(exp_bc.sum()) / beta
```

Actually, one more important consideration: the peak-shaving approach. Let me think about how to implement it efficiently.

Given `a` and `c = conv(a,a)`, the peak is at some lag `k*`. To reduce `c[k*]`, I should reduce the products `a[i]*a[k*-i]` for the pairs with the largest product.

The simplest approach: find the pair `(i*, k*-i*)` with the largest product `a[i*]*a[k*-i*]`. Then:
- Reduce `a[i*]` (or `a[k*-i*]`) by a small amount
- Add that amount to some other element that doesn't contribute much to any of the top lags

For the "add" step, I should add to an element `a[m]` where:
- `a[m]` is small (so adding to it doesn't create a new peak at `c[2m]` or nearby)
- `a[m]*a[k*-m]` is small (so it doesn't increase `c[k*]` much)
- More generally, `a[m]` should be in a "quiet" region

A simple heuristic: add to the element with the smallest value that is far from the peak-contributing pairs.

Let me implement a simple version:

```python
def peak_shave(a, S, max_iters, rng):
    n = len(a)
    best_c = fft_convolve_self(a).max()
    
    for it in range(max_iters):
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_c * (1 - 1e-8):
            # Keep going
            pass
        elif it > 100:
            break
        
        # Find the peak lag
        k_star = int(np.argmax(c))
        
        # Find the pair with largest contribution to c[k_star]
        i_min = max(0, k_star - n + 1)
        i_max = min(k_star, n - 1)
        indices = np.arange(i_min, i_max + 1)
        partners = k_star - indices
        products = a[indices] * a[partners]
        max_product_idx = int(np.argmax(products))
        
        ii = indices[max_product_idx]
        pi = partners[max_product_idx]
        
        # Decide which to reduce
        if a[ii] >= a[pi]:
            reduce_idx = ii
        else:
            reduce_idx = pi
        
        # Find where to add: smallest element not contributing to top lags
        top_lags = np.argsort(c)[::-1][:10]
        # Elements contributing to top lags: for each top lag k, elements i where a[i]*a[k-i] is large
        # Simplification: just find the smallest element
        add_idx = int(np.argmin(a))
        
        # If add_idx is the same as reduce_idx, find next smallest
        if add_idx == reduce_idx:
            a_sorted = np.argsort(a)
            for candidate in a_sorted:
                if candidate != reduce_idx:
                    add_idx = int(candidate)
                    break
        
        # Determine step size
        delta = 0.1 * a[reduce_idx]
        a[reduce_idx] -= delta
        a[add_idx] += delta
        a = np.maximum(a, 0)
        a = a * (S / a.sum())  # renormalize
        
        c_new = fft_convolve_self(a)
        if c_new.max() > cur_max * 1.01:
            # Revert
            a[reduce_idx] += delta
            a[add_idx] -= delta
            a = np.maximum(a, 0)
            a = a * (S / a.sum())
            break
    
    c_final = fft_convolve_self(a)
    return a, c_final.max()
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. L-BFGS on softmax parameterization (fundamentally different from FISTA)
2. Coarse-to-fine beta scheduling for the log-sum-exp smoothing
3. Multi-start with structured Fourier-mode perturbations
4. Greedy peak-shaving coordinate descent as final polish
5. Optional longer-sequence exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_params_to_a(z, S):
    """Convert log-space parameters z to non-negative sequence a with sum S."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    total = exp_z.sum()
    if total < 1e-300:
        return np.ones(len(z)) * S / len(z)
    a = exp_z / total * S
    return a


def smooth_obj_and_grad(z, S, beta):
    """
    Compute smoothed objective value and gradient w.r.t. z (softmax params).
    Objective: (1/beta)*log(sum_k exp(beta * c_k)) ~= max_k c_k
    """
    a = softmax_params_to_a(z, S)
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    w = exp_bc / total

    # Smoothed objective
    obj = c_max + np.log(total) / beta

    # Gradient of conv(a,a) w.r.t. a: grad_a[j] = sum_k w_k * (a[k-j] + a[j-k])
    fsize = 1 << (2 * (2 * n - 1) - 1).bit_length()
    A_full = fft(a, fsize)
    W_full = fft(w, fsize)
    conv_wa = np.real(ifft(W_full * A_full))[:n]
    corr_wa = np.real(ifft(W_full * np.conj(A_full)))[:n]
    grad_a = conv_wa + corr_wa

    # Chain rule through softmax: grad_z[l] = a_l * (grad_a[l] - 2*E_w[c]/S)
    E_wc = np.dot(w, c)
    grad_z = a * (grad_a - 2.0 * E_wc / S)

    return obj, grad_z


def run_lbgfs_coarse_to_fine(a_init, n, S, max_time, rng):
    """L-BFGS with coarse-to-fine beta schedule."""
    # Initialize z from a_init
    a_start = np.clip(a_init, 1e-10, None)
    z = np.log(a_start)  # log of the initial values (up to a constant)
    # Center z for numerical stability
    z = z - z.mean()

    start = time.monotonic()
    best_score = np.inf
    best_a = a_start.copy()

    # Coarse-to-fine beta schedule
    betas = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1000.0]

    for beta in betas:
        if time.monotonic() - start > max_time * 0.85:
            break

        # Local L-BFGS optimization for this beta
        def fun_and_grad(z_local):
            return smooth_obj_and_grad(z_local, S, beta)

        t0 = time.monotonic()
        try:
            result = minimize(
                fun_and_grad, z,
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': 300, 'maxfun': 500, 'ftol': 1e-10, 'gtol': 1e-8}
            )
            z = result.x
        except Exception:
            pass

        a = softmax_params_to_a(z, S)
        a = np.clip(a, 0, 1000)
        c = fft_convolve_self(a)
        score = c.max()

        if score < best_score:
            best_score = score
            best_a = a.copy()

        elapsed_seg = time.monotonic() - t0
        print("    L-BFGS beta=%d: score=%.6f (time=%.1fs)" % (beta, score, elapsed_seg))
        z = z - z.mean()  # re-center

    return best_a, best_score


def run_fista_smoothed(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=500.0):
    """FISTA with log-sum-exp smoothing and adaptive step."""
    a = np.clip(a_init, 1e-10, 1000).copy().astype(np.float64)
    a = a * (S / a.sum())

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    # Simplex projection
    def proj_simplex(v, S_val):
        nn = len(v)
        w = np.sort(v)[::-1]
        cumsum = np.cumsum(w)
        idx = np.arange(1, nn + 1, dtype=np.float64)
        theta_cands = (cumsum - S_val) / idx
        valid = theta_cands <= w
        if not np.any(valid):
            rho = nn
        else:
            rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S_val) / rho
        return np.maximum(v - theta, 0.0)

    L = 2.0 / S  # initial step size estimate

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))

        c_y = fft_convolve_self(y)
        cur = c_y.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Compute gradient via log-sum-exp weights
        c_max = c_y.max()
        exp_bc = np.exp(beta * (c_y - c_max))
        w = exp_bc / exp_bc.sum()

        fsize = 1 << (2 * (2 * n - 1) - 1).bit_length()
        A_full = fft(y, fsize)
        W_full = fft(w, fsize)
        conv_wa = np.real(ifft(W_full * A_full))[:n]
        corr_wa = np.real(ifft(W_full * np.conj(A_full)))[:n]
        grad = conv_wa + corr_wa

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step with backtracking
        step = 1.0 / L
        x_new = proj_simplex(y - step * grad, S)

        # Backtracking check every few iterations
        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() > c_y.max() * 1.001:
                L *= 1.3
            elif it % 100 == 0:
                L = max(L * 0.9, 1.0 / S)
        else:
            # Quick check
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic perturbation
        if it % 2000 == 0 and rng.random() < 0.3:
            noise = 0.005 * S / np.sqrt(n)
            y = proj_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0

    return best_local, best_score


def run_peak_shave(a_init, n, S, max_time, rng, batch_size=30):
    """Greedy peak-shaving: identify peak-contributing pairs and redistribute mass."""
    a = np.clip(a_init, 0, 1000).copy().astype(np.float64)
    a = a * (S / a.sum())

    def proj_simplex(v, S_val):
        nn = len(v)
        w = np.sort(v)[::-1]
        cumsum = np.cumsum(w)
        idx = np.arange(1, nn + 1, dtype=np.float64)
        theta_cands = (cumsum - S_val) / idx
        valid = theta_cands <= w
        if not np.any(valid):
            rho = nn
        else:
            rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S_val) / rho
        return np.maximum(v - theta, 0.0)

    start = time.monotonic()
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    no_improve = 0

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

        # Find the top lags
        n_top = 3
        top_lags = np.argsort(c)[::-1][:n_top]

        # Build reduction and addition masks
        reduce_vec = np.zeros(n)
        add_vec = np.zeros(n)

        for k in top_lags:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_min > i_max:
                continue
            indices = np.arange(i_min, i_max + 1)
            partners = k - indices
            products = a[indices] * a[partners]
            # Reduce the top contributing elements
            top_p = np.argsort(products)[::-1][:batch_size]
            for p in top_p:
                ii = int(indices[p])
                pi = int(partners[p])
                if a[ii] >= a[pi]:
                    reduce_vec[ii] += 1.0
                else:
                    reduce_vec[pi] += 1.0

        # Find quiet elements to add mass to
        # Quiet = low value, not heavily contributing to peak lags
        peak_contrib = np.zeros(n)
        for k in top_lags:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_min <= i_max:
                indices = np.arange(i_min, i_max + 1)
                partners = k - indices
                if a[indices] @ a[partners] > c.mean() * 0.5:
                    peak_contrib[indices] += a[partners]
                    peak_contrib[partners] += a[indices]

        # Add to elements with low (value + peak_contribution)
        add_score = a + 0.5 * peak_contrib
        add_score[reduce_vec > 0] = np.inf
        n_add = min(n // 10, 200)
        top_add = np.argsort(add_score)[:n_add]

        # Compute update
        total_reduce = reduce_vec.sum()
        if total_reduce == 0:
            break
        reduce_amount = 0.02 * S / n * (reduce_vec / total_reduce)
        add_per = reduce_amount.sum() / max(n_add, 1)

        a_new = a - reduce_amount
        for idx in top_add:
            a_new[int(idx)] += add_per

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new * (S / s_new)

        c_new = fft_convolve_self(a_new)
        if c_new.max() > cur_max * 1.05:
            # Too aggressive, try smaller step
            reduce_amount *= 0.3
            a_new = a - reduce_amount
            for idx in top_add:
                a_new[int(idx)] += reduce_amount.sum() / max(n_add, 1)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new * (S / s_new)
            c_new = fft_convolve_self(a_new)
            if c_new.max() > cur_max * 1.02:
                break

        a = a_new

    return best_a, best_score


def make_fourier_perturbation(a_base, n, S, rng, num_modes=5, amp=0.1):
    """Add random Fourier modes to create a diverse starting point."""
    tt = np.arange(n)
    perturbation = np.zeros(n)
    for _ in range(num_modes):
        freq = rng.integers(2, max(3, n // 4))
        phase = rng.random() * 2 * np.pi
        coeff = rng.random() * amp
        perturbation += coeff * np.sin(2.0 * np.pi * freq * tt / n + phase)
    a_new = a_base + perturbation * (S / n)
    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new * (S / a_new.sum())
    return a_new


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = mu."""
    half = n // 2
    mu = S / half
    h = rng.random(half) * mu * 0.8 + mu * 0.1
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    return a


def make_periodic_init(n, rng, S, num_freqs=4):
    """Multi-frequency sinusoidal initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 30)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.5
        a += amp * np.cos(2.0 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a

            global_best = score2
            global_a = a2.copy()

    # === Phase 3: Multi-start with perturbations ===
    if time.monotonic() < deadline - 120:
        print("=== Phase 3: Multi-start with perturbations ===")
        n_starts = 6
        per_start_time = max(5.0, (deadline - time.monotonic() - 60) / n_starts * 0.8)

        for start_i in range(n_starts):
            if time.monotonic() > deadline - 30:
                break

            # Create perturbed starting point
            if start_i == 0:
                # Anti-palindromic fresh
                a_pert = make_anti_palindromic_init(primary_n, rng, S)
            elif start_i == 1:
                # Periodic fresh
                a_pert = make_periodic_init(primary_n, rng, S)
            elif start_i == 2:
                # Fourier perturbation of current best
                a_pert = make_fourier_perturbation(global_a, primary_n, S, rng,
                                                    num_modes=3, amp=0.15)
            elif start_i == 3:
                # Larger Fourier perturbation
                a_pert = make_fourier_perturbation(global_a, primary_n, S, rng,
                                                    num_modes=8, amp=0.3)
            elif start_i == 4:
                # Random-ish perturbation
                a_pert = global_a + rng.normal(0, 0.02 * S / primary_n, primary_n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert * (S / a_pert.sum())
            else:
                # Sparse perturbation: zero out random quarter, redistribute
                a_pert = global_a.copy()
                zero_mask = rng.random(primary_n) < 0.25
                a_pert[zero_mask] = 0.0
                a_pert = a_pert * (S / a_pert.sum())

            # Optimize with L-BFGS (shorter run)
            t0 = time.monotonic()
            z = np.log(np.maximum(a_pert, 1e-10))
            z = z - z.mean()

            best_local_score = np.inf
            best_local_a = a_pert.copy()

            # Two-stage: medium then high beta
            for beta in [5.0, 50.0, 200.0]:
                if time.monotonic() - t0 > per_start_time * 0.9:
                    break

                def fun_local(z_local):
                    return smooth_obj_and_grad(z_local, S, beta)

                try:
                    res = minimize(fun_local, z, method='L-BFGS-B', jac=True,
                                   options={'maxiter': 200, 'maxfun': 300,
                                            'ftol': 1e-9, 'gtol': 1e-7})
                    z = res.x
                except Exception:
                    pass

                a_test = softmax_params_to_a(z, S)
                a_test = np.clip(a_test, 0, 1000)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < best_local_score:
                    best_local_score = c_test.max()
                    best_local_a = a_test.copy()
                z = z - z.mean()

            if best_local_score < global_best:
                global_best = best_local_score
                global_a = best_local_a.copy()
                print("  Start %d: improved to %.6f" % (start_i, global_best))
            else:
                print("  Start %d: best=%.6f (no improvement)" % (start_i, best_local_score))

    # === Phase 4: Peak shaving ===
    if time.monotonic() < deadline - 20:
        print("=== Phase 4: Peak shaving ===")
        t0 = time.monotonic()
        seg4 = min(60.0, deadline - time.monotonic() - 10)
        a4, s4 = run_peak_shave(global_a, primary_n, S, seg4, rng, batch_size=20)
        c4 = fft_convolve_self(a4)
        score4 = c4.max()
        print("  Peak shave: max_b=%.8f (time=%.1fs)" % (score4, time.monotonic() - t0))
        if score4 < global_best:
            global_best = score4
            global_a = a4.copy()

    # === Phase 5: L-BFGS on the peak-shaved result ===
    if time.monotonic() < deadline - 20:
        print("=== Phase 5: L-BFGS on shaved result ===")
        t0 = time.monotonic()
        seg5 = min(60.0, deadline - time.monotonic() - 10)
        a5, s5 = run_lbgfs_coarse_to_fine(global_a, primary_n, S, seg5, rng)
        c5 = fft_convolve_self(a5)
        score5 = c5.max()
        print("  L-BFGS shave: max_b=%.8f (time=%.1fs)" % (score5, time.monotonic() - t0))
        if score5 < global_best:
            global_best = score5
            global_a = a5.copy()

    # === Phase 6: Try longer sequence ===
    if time.monotonic() < deadline - 120:
        n_long = primary_n * 2
        if n_long <= 32768:
            print("=== Phase 6: Longer sequence n=%d ===" % n_long)
            S_long = np.sqrt(2.0 * n_long)
            t0 = time.monotonic()

            # Upsample
            tt_old = np.linspace(0, 1, len(global_a))
            tt_new = np.linspace(0, 1, n_long)
            a_long = np.interp(tt_new, tt_old, global_a)
            a_long = np.maximum(a_long, 1e-10)
            a_long = a_long * (S_long / a_long.sum())

            seg6 = min(90.0, deadline - time.monotonic() - 20)
            a6, s6 = run_lbgfs_coarse_to_fine(a_long, n_long, S_long, seg6, rng)
            c6 = fft_convolve_self(a6)
            score6 = c6.max()
            obj6 = 2.0 * n_long * score6 / S_long**2

            # Compare objectives (not raw max_b)
            obj_cur = 2.0 * primary_n * global_best / S**2
            if obj6 < obj_cur:
                global_best = score6
                global_a = a6.copy()
                print("  Long seq improved: obj=%.6f" % obj6)
            else:
                print("  Long seq: obj=%.6f (no improvement vs %.6f)" % (obj6, obj_cur))
            print("  Phase 6 time: %.1fs" % (time.monotonic() - t0))

    # === Phase 7: Final peak shave on best ===
    if time.monotonic() < deadline - 15:
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * n_final)
        t0 = time.monotonic()
        seg7 = max(5.0, deadline - time.monotonic() - 10)
        a7, s7 = run_peak_shave(global_a, n_final, S_final, seg7, rng, batch_size=30)
        c7 = fft_convolve_self(a7)
        score7 = c7.max()
        if score7 < global_best:
            global_best = score7
            global_a = a7.copy()
        print("=== Final shave: max_b=%.8f (time=%.1fs) ===" % (score7, time.monotonic() - t0))

    # === Final verification ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    obj_final = 2.0 * len(global_a) * global_best / (np.sqrt(2.0 * len(global_a)))**2
    print("=== FINAL: evaluate=%.6f, internal_obj=%.6f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), obj_final, len(global_a)))

    if not np.isfinite(ver):
        print("WARNING: evaluator rejected, using fallback")
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver2 = evaluate_sequence(final_list)
        print("Fallback: evaluate=%.6f" % (ver2 if np.isfinite(ver2) else float('inf')))

    return final_list
```
