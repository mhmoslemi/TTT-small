```python
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Binary search for threshold
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.argmax(1 - u/n * (css - S) > 0)
    theta = (css[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fft_conv(a):
    """Compute linear self-convolution via FFT."""
    n = len(a)
    L = 1 << (2*n - 1 - 1).bit_length()  # Next power of 2 >= 2n-1
    A = fft(a, L)
    c = np.real(ifft(A * A))
    return c[:2*n-1]

def grad_l2(a, L):
    """Gradient of sum(conv(a,a)^2) w.r.t. a."""
    A = fft(a, L)
    B = np.abs(A)**2 * np.conj(A)
    return 4.0 * np.real(ifft(B))[:len(a)]

def grad_lse(a, c, beta, L):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    w = exp_bc / exp_bc.sum()
    # grad_j = 2 * sum_k w_k * a_{k-j}
    # = 2 * cross_correlation(w, a)[j]
    W = fft(w, L)
    A = fft(a, L)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:len(a)]
    return grad

def run_l2_optim(a_init, S, time_budget, rng):
    """Phase 1: Minimize sum(c^2) using gradient descent with BB step."""
    n = len(a_init)
    L = 1 << (2*n - 1 - 1).bit_length()
    a = project_simplex(a_init.copy(), S)
    
    start = time.monotonic()
    best_a = a.copy()
    best_val = np.sum(fft_conv(a)**2)
    
    s_prev = None
    y_prev = None
    step = 1.0 / (4.0 * np.linalg.fft(a, L).max()**2 * L)  # Initial step estimate
    
    while time.monotonic() - start < time_budget:
        c = fft_conv(a)
        val = np.sum(c**2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        
        g = grad_l2(a, L)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        # BB step
        if s_prev is not None and y_prev is not None:
            sy = np.dot(s_prev, y_prev)
            ss = np.dot(s_prev, s_prev)
            if ss > 1e-30:
                bb_step = max(sy / ss, 1e-10)
                step = 0.5 * (step + bb_step)
        
        # Backtracking
        a_new = project_simplex(a - step * g, S)
        c_new = fft_conv(a_new)
        val_new = np.sum(c_new**2)
        
        bt = 0
        while val_new > val - 1e-10 * step * gn**2 and bt < 10:
            step *= 0.5
            a_new = project_simplex(a - step * g, S)
            c_new = fft_conv(a_new)
            val_new = np.sum(c_new**2)
            bt += 1
        
        s_prev = a_new - a
        y_prev = grad_l2(a_new, L) - g
        a = a_new
    
    return best_a, best_val

def run_lse_optim(a_init, S, time_budget, rng, beta_start=10.0, beta_end=500.0):
    """Phase 2: Minimize log-sum-exp with increasing beta."""
    n = len(a_init)
    L = 1 << (2*n - 1 - 1).bit_length()
    a = project_simplex(a_init.copy(), S)
    
    start = time.monotonic()
    best_a = a.copy()
    best_maxc = fft_conv(a).max()
    
    y = a.copy()
    t_mom = 1.0
    it = 0
    step = S / (4.0 * n)
    
    while time.monotonic() - start < time_budget:
        it += 1
        # Animate beta
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        c = fft_conv(y)
        cur_max = c.max()
        if cur_max < best_maxc:
            best_maxc = cur_max
            best_a = y.copy()
        
        g = grad_lse(y, c, beta, L)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        # Backtracking line search
        for bt in range(10):
            x_new = project_simplex(y - step * g, S)
            c_new = fft_conv(x_new)
            # AIFE: Armijo condition
            if c_new.max() <= cur_max - 1e-4 * step * gn**2:
                break
            step *= 0.5
        
        else:
            x_new = project_simplex(y - step * g, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        a = x_new
        t_mom = t_new
        
        # Periodic perturbation
        if it % 200 == 0 and rng.random() < 0.2:
            noise = 0.01 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0
    
    return best_a, best_maxc

def run_coord_descent(a_init, S, time_budget, rng, top_k=10):
    """Phase 3: Coordinate descent targeting top lags."""
    n = len(a_init)
    L = 1 << (2*n - 1 - 1).bit_length()
    a = project_simplex(a_init.copy(), S)
    
    start = time.monotonic()
    best_a = a.copy()
    best_maxc = fft_conv(a).max()
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        c = fft_conv(a)
        cur_max = c.max()
        if cur_max < best_maxc:
            best_maxc = cur_max
            best_a = a.copy()
        
        # Find top-k lags
        top_lags = np.argsort(c)[::-1][:top_k]
        
        improved = False
        for lag in top_lags:
            lag = int(lag)
            i_min = max(0, lag - n + 1)
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue
            
            # For this lag, c[lag] = sum_{i=i_min}^{i_max} a[i] * a[lag-i]
            # To reduce c[lag], reduce products a[i]*a[lag-i]
            # Strategy: move mass from large a values to small ones
            indices = np.arange(i_min, i_max + 1)
            partners = lag - indices
            products = a[indices] * a[partners]
            
            # Find the pair with the largest product where both are "large"
            # and move mass from the larger to a smaller element
            prod_sorted = np.argsort(products)[::-1]
            
            made_change = False
            for pi in prod_sorted[:5]:
                i = indices[pi]
                p = partners[pi]
                if i == p:
                    continue
                # Move mass from the larger of a[i], a[p] to a small element
                if a[i] > a[p]:
                    src = i
                    dst_val = a[p]
                else:
                    src = p
                    dst_val = a[i]
                
                # Find destination: a small element not involved in top lags
                small_mask = a < np.percentile(a, 20)
                small_mask[src] = False
                if np.any(small_mask):
                    candidates = np.where(small_mask)[0]
                    dst = candidates[rng.integers(len(candidates))]
                    if dst == src:
                        continue
                    
                    # Transfer: reduce a[src], increase a[dst]
                    amount = 0.01 * a[src]
                    a_new = a.copy()
                    a_new[src] -= amount
                    a_new[dst] += amount
                    a_new = np.maximum(a_new, 0)
                    a_new = project_simplex(a_new, S)
                    
                    c_new = fft_conv(a_new)
                    if c_new.max() < cur_max:
                        a = a_new
                        cur_max = c_new.max()
                        made_change = True
                        improved = True
                        break
            
            if made_change:
                break
        
        if not improved and it % 100 == 0:
            # Add small noise to escape
            noise = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)
    
    return best_a, best_maxc

def upsample(a, n_new):
    """Upsample sequence to new length using interpolation."""
    from numpy import interp as np_interp
    tt_old = np.linspace(0, 1, len(a))
    tt_new = np.linspace(0, 1, n_new)
    a_new = np_interp(tt_new, tt_old, a)
    return np.maximum(a_new, 0.0)

def make_init(n, rng, S, mode="uniform"):
    """Create initial sequence."""
    if mode == "uniform":
        a = np.ones(n)
    elif mode == "cosine":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    elif mode == "raised_cosine":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = a * a  # Squared raised cosine
    elif mode == "gaussian":
        t = np.linspace(0, 1, n, endpoint=False)
        center = 0.5
        sigma = 0.15
        a = np.exp(-0.5 * ((t - center) / sigma)**2)
    elif mode == "bimodal":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.exp(-0.5 * ((t - 0.25) / 0.1)**2) + np.exp(-0.5 * ((t - 0.75) / 0.1)**2)
    elif mode == "random":
        a = rng.random(n) + 0.1
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a * (S / s)
```

Hmm, the issue with using scipy.optimize is that the projection onto the simplex is not smooth, which makes the gradient incorrect. I need a smooth parameterization.

One approach: use `a_i = exp(x_i) / sum(exp(x_j)) * S` (softmax parameterization). This automatically satisfies `a >= 0` and `sum(a) = S`. The gradient needs to account for this.

If `a_i = S * exp(x_i) / Z` where `Z = sum(exp(x_j))`:
`da_i/dx_j = S * (exp(x_i) * δ_{ij} * Z - exp(x_i) * exp(x_j)) / Z^2`
`= a_i * (δ_{ij} - a_j/S)`

So `grad_x_j = sum_i grad_a_i * a_i * (δ_{ij} - a_j/S)`
`= grad_a_j * a_j - a_j/S * sum_i grad_a_i * a_i`

This is the gradient for the softmax parameterization. It's smooth and well-defined.

But the softmax parameterization has a problem: it makes all a_i > 0 (never exactly zero), which might prevent the optimizer from finding solutions with zeros. For this problem, I think all positive is fine (the optimal solution likely has no zeros for large n).

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Key improvements over previous approach:
1. Softmax parameterization (smooth constraint handling, no projection kinks)
2. scipy L-BFGS-B with curvature information for faster convergence
3. Power-mean objective as alternative to log-sum-exp
4. Aggressive multigrid (128 -> 512 -> 2048 -> 8192)

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv(a):
    """Compute linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def next_pow2(n):
    return 1 << (n - 1).bit_length()


def lse_value_and_grad_a(c, beta):
    """Compute LSE value and gradient w.r.t. a (given c = conv(a,a))."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, np.zeros(len(c))
    w = exp_bc / total
    val = c_max + (1.0 / beta) * np.log(total / len(c))
    return val, w


def grad_from_weights(w, a, fsize):
    """Compute gradient w.r.t. a given softmax weights w on convolution."""
    n = len(a)
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def power_mean_value_and_grad_a(c, p):
    """Compute power mean value and weights for gradient."""
    c_pos = np.maximum(c, 1e-30)
    cp = c_pos ** p
    total = cp.sum()
    if total < 1e-300:
        return c_pos.max(), np.zeros(len(c))
    w = cp / total
    val = (total / len(c)) ** (1.0 / p)
    return val, w


def softmax_from_x(x, S):
    """Convert unconstrained x to a on the probability simplex scaled by S."""
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    Z = exp_x.sum()
    if Z < 1e-300:
        return np.full(len(x), S / len(x))
    return S * exp_x / Z


def grad_x_from_grad_a(grad_a, a, S):
    """Chain rule: gradient w.r.t. x given gradient w.r.t. a (softmax parametrization)."""
    # da_i/dx_j = a_i*(delta_ij - a_j/S)
    # dx_j = sum_i da_i/dx_j * grad_a_i = a_j * (grad_a_j - sum(a_i*grad_a_i)/S)
    sum_ag = np.dot(a, grad_a)
    return a * (grad_a - sum_ag / S)


def run_lse_fista(a_init, S, time_budget, rng, beta_start=5.0, beta_max=500.0):
    """FISTA on LSE objective with simplex projection (no parametrization)."""
    n = len(a_init)
    fsize = next_pow2(2 * n - 1)
    a = project_simplex(np.maximum(a_init, 0.0), S)

    start = time.monotonic()
    best_a = a.copy()
    best_maxc = fft_conv(a).max()

    y = a.copy()
    t_mom = 1.0
    it = 0
    step = 2.0 / (n * 1.0)  # Initial step

    while time.monotonic() - start < time_budget:
        it += 1
        frac = min(1.0, (time.monotonic() - start) / time_budget)
        beta = beta_start * (beta_max / beta_start) ** (frac * frac)  # Quadratic ramp

        c = fft_conv(y)
        cur_max = c.max()
        if cur_max < best_maxc:
            best_maxc = cur_max
            best_a = y.copy()

        _, w = lse_value_and_grad_a(c, beta)
        g = grad_from_weights(w, y, fsize)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        # Backtracking
        x_new = None
        L = 1.0 / step
        for bt in range(8):
            cand = project_simplex(y - step * g, S)
            c_cand = fft_conv(cand)
            if c_cand.max() <= cur_max - 1e-4 * step * gn ** 2:
                x_new = cand
                break
            step *= 0.5
        if x_new is None:
            x_new = project_simplex(y - step * g, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        a = x_new
        t_mom = t_new

        # Periodic perturbation

            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0

    return best_a, best_maxc
