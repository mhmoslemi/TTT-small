- LP-based refinement using constraint generation (block coordinate method)

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_a(a, w):
    """Compute grad_a[j] = 2 * sum_k w[k] * a[j-k] (cross-correlation via FFT)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta
    w = exp_bc / sum_exp_bc
    grad_a = compute_grad_a(a, w)
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)
    return obj, grad_z


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

                           beta_schedule=None, maxiter_per_beta=300):
    """Run L-BFGS-B with continuation on beta."""
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break
        n_attempts = 1
        if beta >= 500 and max_time > 30:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 3:

            if attempt > 0:
                z = z + rng.normal(0, 0.015, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            t0 = time.monotonic()
            try:

                    fun, z, jac=jac, method='L-BFGS-B',

                        'maxfun': maxiter_per_beta * 2,
                        'ftol': 1e-15,
                        'gtol': 1e-10,
                        'maxls': 50,
                        'iprint': -1
                    }

            except Exception:
                break
            z = result.x
            z = z - z.mean()
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)

            if s < 1e-15:

    Uses constraint generation.
    Returns the optimal b vector.
    """
    # Convolution matrix construction for a given set of constraint rows
    def get_constraint_row(k):
        """Get row k of the convolution matrix: coefficient of b[i] in (a*b)[k]."""
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[i] = a[j]
        return row

    # Start with top constraints
    c_ab = fft_convolve_self(a)  # This is a*a, not a*b, but use as init
    # Actually we need (a*b)[k] for b = a initially
    init_b = a.copy()
    c_init = fft_convolve_self(init_b)
    # Find top-k constraint violations
    top_k = min(max_constraints, 2 * n - 1)
    active_indices = np.argsort(c_init)[::-1][:top_k]

    b_solution = None
    for iteration in range(max_iters):
        # Build constraint matrix
        A_ub = np.zeros((len(active_indices), n))
        for row_idx, k in enumerate(active_indices):
            A_ub[row_idx] = get_constraint_row(k)

        # Solve LP: max sum(b) s.t. A_ub @ b <= M, b >= 0
        try:
            res = linprog(
                c=-np.ones(n),
                A_ub=A_ub,
                b_ub=np.full(len(active_indices), M),
                bounds=[(0, None)] * n,
                method='highs',
                options={'presolve': True, 'dual_feasibility_tolerance': 1e-9}
            )
            if not res.success:
                return None
            b_solution = res.x
        except Exception:
            return None

        # Check for violated constraints
        if b_solution is None:
            return None

        # Compute (a * b_solution)[k] for all k
        # Use FFT for efficiency
        fsize = 1 << (2 * n - 2).bit_length()
        A_fft = fft(a, fsize)
        B_fft = fft(b_solution, fsize)
        c_full = np.real(ifft(A_fft * B_fft))[:2 * n - 1]

        max_violation = c_full.max()
        if max_violation <= M * 1.001:
            return b_solution

        # Add most violated constraints
        if len(active_indices) < 2 * n - 1:
            all_indices = np.array(range(2 * n - 1))
            active_set = set(active_indices.tolist())
            inactive = np.array([i for i in all_indices if i not in active_set])
            if len(inactive) > 0:
                violated = inactive[np.argsort(c_full[inactive])[::-1]]
                # Add top 10 most violated
                n_add = min(10, len(violated))
                for idx in violated[:n_add]:
                    if idx not in active_set:
                        active_indices = np.append(active_indices, idx)
                        active_set.add(idx)
            # Sort by violation magnitude
            active_indices = np.array(sorted(active_indices,
                                             key=lambda k: -c_full[int(k)]))
            # Keep only top max_constraints
            if len(active_indices) > max_constraints:
                active_indices = active_indices[:max_constraints]

    return b_solution


def lp_block_coordinate(a_init, n, S, max_time, rng, t_init=0.3):
    """
    Block coordinate method using LP refinement.
    At each step: solve LP for best g given current a, then blend.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    t = t_init
    stall_count = 0

    while time.monotonic() - start < max_time:
        M = best_score
        b_sol = solve_lp_refinement(a, M, n, max_constraints=200, max_iters=30)

        if b_sol is None:
            break

        b_sol = np.maximum(b_sol, 0.0)
        s_b = b_sol.sum()
        if s_b < 1e-15:
            break

        # Rescale b to match sum = S
        g = b_sol * S / s_b

        # Blend
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * S / s_new

        c_new = fft_convolve_self(a_new)
        score_new = c_new.max()

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            # Try to increase t slightly
            t = min(0.8, t * 1.05)
        else:
            a = a_new  # Still update for exploration
            stall_count += 1
            t = max(0.05, t * 0.9)
            if stall_count > 10:
                # Perturb and reset
                a = a + rng.normal(0, 0.01 * S / np.sqrt(n), n)
                a = np.maximum(a, 0.0)
                a = safe_normalize(a, n)
                t = t_init
                stall_count = 0

    return best_a, best_score


def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=512):
    """Nesterov-accelerated top-k subgradient with FFT gradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        if it % 3000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def symmetric_optimize(a_init, n, S, max_time, rng):
    """Optimize in the symmetric subspace a[j] = a[n-1-j]."""
    if n % 2 != 0:
        return a_init, fft_convolve_self(a_init).max()

    half = n // 2
    # Extract symmetric part
    a_sym = (a_init[:half] + a_init[half::-1]) / 2.0
    a_sym = np.maximum(a_sym, 1e-15)
    # Normalize
    # Full sum = 2 * sum(a_sym)
    s_full = 2.0 * a_sym.sum()
    a_sym = a_sym * (S / s_full) if s_full > 1e-15 else np.full(half, S / (2.0 * half))

    def expand(a_half):
        return np.concatenate([a_half, a_half[::-1]])

    start = time.monotonic()
    a_full = expand(a_sym)
    best_score = fft_convolve_self(a_full).max()
    best_a = a_full.copy()

    # Use L-BFGS on the half-sequence
    z = np.log(a_sym / (S / half) + 1e-30)

    beta_schedule = [10, 50, 100, 500, 1000, 3000]
    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break

        def fun_half(z_h):
            z_h_shifted = z_h - z_h.max()
            exp_z = np.exp(z_h_shifted)
            a_h = (S / 2.0) * exp_z / exp_z.sum()
            a_f = expand(a_h)
            c = fft_convolve_self(a_f)
            c_max = c.max()
            exp_bc = np.exp(beta * (c - c_max))
            sum_exp_bc = exp_bc.sum()
            obj = (c_max + np.log(sum_exp_bc)) / beta
            return obj

        def jac_half(z_h):
            z_h_shifted = z_h - z_h.max()
            exp_z = np.exp(z_h_shifted)
            sum_exp = exp_z.sum()
            a_h = (S / 2.0) * exp_z / sum_exp
            a_f = expand(a_h)
            c = fft_convolve_self(a_f)
            c_max = c.max()
            exp_bc = np.exp(beta * (c - c_max))
            sum_exp_bc = exp_bc.sum()
            w = exp_bc / sum_exp_bc
            grad_f = compute_grad_a(a_f, w)
            # Gradient w.r.t. a_h: each a_h[j] appears in a_f[j] and a_f[n-1-j]
            grad_h = grad_f[:half] + grad_f[half - 1::-1]
            # Chain rule for softmax
            dot_g = np.dot(grad_h, a_h)
            grad_z = a_h * (grad_h - dot_g / (S / 2.0))
            return grad_z

        try:
            result = scipy_minimize(
                fun_half, z, jac=jac_half, method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1}
            )
            z = result.x
            z = z - z.mean()
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a_sym = (S / 2.0) * exp_z / exp_z.sum()
            a_sym = np.maximum(a_sym, 0.0)
            a_full = expand(a_sym)
            s = a_full.sum()
            if s > 1e-15:
                a_full = a_full * S / s
            c = fft_convolve_self(a_full)
            if c.max() < best_score:
                best_score = c.max()
                best_a = a_full.copy()
        except Exception:
            pass

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # --- Get initial sequence ---
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0.0, 1000.0)))
    except Exception:
        pass

    if inits:
        main_len = len(inits[0][1])
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * float(n))
    tt = np.linspace(0, 1, n, endpoint=False)

    # Add diverse initializations
    inits.append(("rand", rng.random(n) * 0.5 + 0.1))
    inits.append(("multi_sin", 1.0 + 0.3 * np.sin(2 * np.pi * 5 * tt) +
                  0.25 * np.cos(2 * np.pi * 11 * tt) +
                  0.2 * np.sin(2 * np.pi * 17 * tt) +
                  0.15 * np.cos(2 * np.pi * 23 * tt) +
                  0.1 * np.sin(2 * np.pi * 31 * tt)))
    inits.append(("gauss_bumps", np.exp(-((tt - 0.2) * 8.0)**2) +
                 0.8 * np.exp(-((tt - 0.5) * 10.0)**2) +
                 0.6 * np.exp(-((tt - 0.8) * 8.0)**2) + 0.05))
    inits.append(("chirp_like", 1.0 + 0.3 * np.sin(2 * np.pi * (3 * tt + 0.5 * tt**2)) +
                  0.05 * rng.standard_normal(n)))

    n_inits = len(inits)
    print("=== n=%d, %d inits, budget=%.0fs ===" % (n, n_inits, budget_s))
    print("Target: 1.5027")

    # --- Phase 1: L-BFGS on given sequence (most important) ---
    if inits:
        print("\n=== Phase 1: L-BFGS on given sequence ===")
        t0 = time.monotonic()
        a_given = safe_normalize(inits[0][1], n)
        c = fft_convolve_self(a_given)
        init_score = c.max()
        print("  Initial score: %.8f" % init_score)

        if init_score < global_best:
            global_best = init_score
            global_a = a_given.copy()

        # Run L-BFGS with aggressive schedule
        remaining = deadline - time.monotonic()
        lbfgs_budget = min(remaining * 0.35, 400.0)
        a_new, score = run_lbfgs_continuation(
            a_given, n, S, lbfgs_budget, rng,
            beta_schedule=[5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000],
            maxiter_per_beta=250
        )
        a_new = safe_normalize(a_new, n)
        c_new = fft_convolve_self(a_new)
        final_score = c_new.max()
        print("  L-BFGS score: %.8f (%.1fs)" % (final_score, time.monotonic() - t0))

        if final_score < global_best:
            global_best = final_score
            global_a = a_new.copy()
            print("  [NEW BEST: %.8f]" % global_best)

    # --- Phase 2: LP-based block coordinate refinement ---
    if global_a is not None and time.monotonic() < deadline - 60:
        print("\n=== Phase 2: LP Block Coordinate ===")
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        lp_budget = min(remaining * 0.25, 300.0)
        a_lp, score_lp = lp_block_coordinate(
            global_a, n, S, lp_budget, rng, t_init=0.3
        )
        a_lp = safe_normalize(a_lp, n)
        c_lp = fft_convolve_self(a_lp)
        final_lp = c_lp.max()
        print("  LP score: %.8f (%.1fs)" % (final_lp, time.monotonic() - t0))

        if final_lp < global_best:
            global_best = final_lp
            global_a = a_lp.copy()
            print("  [NEW BEST: %.8f]" % global_best)

    # --- Phase 3: Try other initializations ---
    for name, arr in inits[1:]:
        if time.monotonic() > deadline - 30:
            break
        t0 = time.monotonic()
        a_norm = safe_normalize(arr, n)
        c = fft_convolve_self(a_norm)
        init_score = c.max()

        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        per_init = min(remaining * 0.15, 120.0)

        a_new, score = run_lbfgs_continuation(
            a_norm, n, S, per_init, rng,
            beta_schedule=[10, 50, 100, 300, 1000, 3000],
            maxiter_per_beta=200
        )
        a_new = safe_normalize(a_new, n)
        c_new = fft_convolve_self(a_new)
        final_score = c_new.max()
        print("  %s: %.8f (init=%.4f, %.1fs)" % (name, final_score, init_score, time.monotonic() - t0))

        if final_score < global_best:
            global_best = final_score
            global_a = a_new.copy()
            print("  [NEW BEST: %.8f]" % global_best)

    # --- Phase 4: Symmetric optimization ---
    if global_a is not None and time.monotonic() < deadline - 30:
        print("\n=== Phase 4: Symmetric Optimization ===")
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        sym_budget = min(remaining * 0.15, 150.0)
        a_sym, score_sym = symmetric_optimize(global_a, n, S, sym_budget, rng)
        a_sym = safe_normalize(a_sym, n)
        c_sym = fft_convolve_self(a_sym)
        final_sym = c_sym.max()
        print("  Symmetric score: %.8f (%.1fs)" % (final_sym, time.monotonic() - t0))

        if final_sym < global_best:
            global_best = final_sym
            global_a = a_sym.copy()
            print("  [NEW BEST: %.8f]" % global_best)

    # --- Phase 5: LP refinement on best ---
    if global_a is not None and time.monotonic() < deadline - 20:
        print("\n=== Phase 5: Final LP Refinement ===")
        remaining = deadline - time.monotonic()
        lp_budget = min(remaining * 0.5, 200.0)
        a_ref, score_ref = lp_block_coordinate(
            global_a, n, S, lp_budget, rng, t_init=0.2
        )
        a_ref = safe_normalize(a_ref, n)
        c_ref = fft_convolve_self(a_ref)
        final_ref = c_ref.max()
        print("  Final LP: %.8f" % final_ref)
        if final_ref < global_best:
            global_best = final_ref
            global_a = a_ref.copy()
            print("  [NEW BEST: %.8f]" % global_best)

    # --- Phase 6: Nesterov polish ---
    if global_a is not None and time.monotonic() < deadline - 10:
        print("\n=== Phase 6: Nesterov Polish ===")
        remaining = deadline - time.monotonic()
        nest_budget = min(remaining * 0.7, 120.0)
        a_pol, score_pol = run_nesterov_toppk(
            global_a, n, S, nest_budget, rng, top_k_init=20, top_k_max=1024
        )
        a_pol = safe_normalize(a_pol, n)
        c_pol = fft_convolve_self(a_pol)
        final_pol = c_pol.max()
        print("  Nesterov: %.8f" % final_pol)
        if final_pol < global_best:
            global_best = final_pol
            global_a = a_pol.copy()
            print("  [NEW BEST: %.8f]" % global_best)

    # --- Phase 7: Final L-BFGS with very high beta ---
    if global_a is not None and time.monotonic() < deadline - 5:
        print("\n=== Phase 7: Final High-Beta L-BFGS ===")
        remaining = deadline - time.monotonic()
        a_fin, score_fin = run_lbfgs_continuation(
            global_a, n, S, remaining, rng,
            beta_schedule=[1000, 3000, 5000, 10000, 20000],
            maxiter_per_beta=300
        )
        a_fin = safe_normalize(a_fin, n)
        c_fin = fft_convolve_self(a_fin)
        final_fin = c_fin.max()
        print("  Final L-BFGS: %.8f" % final_fin)
        if final_fin < global_best:
            global_best = final_fin
            global_a = a_fin.copy()
            print("  [NEW BEST: %.8f]" % global_best)

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning length=%d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
