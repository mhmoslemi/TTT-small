def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_lp_gradient(c, a, p, F_val=None):
    """
    Compute gradient of F_p = (sum c[k]^p)^(1/p) with respect to a.
    grad[j] = 2 * F_p^(1-p) * sum_m w[m] * a[m] where w[m] = c[m+j]^(p-1)

    if F_val is None:
        F_val = (np.sum(c ** p)) ** (1.0 / p)
    if F_val < 1e-30:
        return np.zeros(n)

    # w[k] = c[k]^(p-1)
    # For numerical stability with large p:
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)

    # w = c^(p-1) / c_max^(p-1) (scale doesn't matter, we normalize at end)
    log_w = (p - 1) * np.log(np.maximum(c, 1e-300) / c_max)
    log_w = log_w - log_w.max()  # shift for stability
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # cross_corr(w, a)[j] = sum_m w[m+j] * a[m] for valid indices
    # = sum_k w[k] * a[k-j] for valid k-j
    # Use FFT: cross_corr = IFFT(FFT(w) * conj(FFT(a))) with appropriate padding
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    grad = 2.0 * F_val ** (1.0 - p) * c_max ** (p - 1) * corr[:n]

    return grad


def compute_peak_gradient(c, a, top_k=5, decay=0.7):
    """
    Compute gradient targeting the top-k peaks with exponential decay.
    """
    n = len(a)
    c = np.maximum(c, 0.0)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)

    # Get top-k peak indices
    # Use partial sort for efficiency
    if len(c) > top_k * 4:
        part_idx = np.argpartition(c, -top_k)[-top_k:]
    else:
        part_idx = np.argsort(c)[-top_k:]

    peak_vals = c[part_idx]
    weights = decay ** np.arange(top_k)
    weights /= weights.sum()

    # Build weight vector
    w = np.zeros(len(c))
    for i, idx in enumerate(part_idx):
        w[idx] = weights[i] * peak_vals[i]

    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # cross_correlation via FFT
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def theta_to_a(theta, S):
    """Convert log-parameters to non-negative sequence with sum S."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def a_to_theta(a):
    """Convert non-negative sequence to log-parameters (proportions)."""
    a = np.maximum(a, 1e-30)
    return np.log(a / a.sum())


def simplex_gradient(grad_a, a, S):
    """
    Convert gradient w.r.t. a to gradient w.r.t. log-parameters (simplex).
    dF/dtheta[j] = a[j] * (grad_a[j] - dot(grad_a, a) / S)
    """
    dot = np.dot(grad_a, a)
    return a * (grad_a - dot / S)


def run_lp_adam(a_init, n, S, time_budget, p_init=2.0, p_final=5000.0,
                lr_init=0.1, beta1=0.9, beta2=0.999, top_k_switch=0.5, verbose=False):
    """
    Adam optimizer using L_p norm with increasing p.
    Switches to peak-targeting gradient after top_k_switch fraction of time.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    theta = a_to_theta(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    no_improve = 0

    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # p schedule: exponential ramp
        p = p_init * (p_final / p_init) ** (progress ** 0.8)

        # Convert theta to a
        a = theta_to_a(theta, S)

        # Compute autocorrelation
        c = fft_convolve_self(a)
        cur_obj = c.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve > 200:
            noise = 0.01 * (1.0 - progress)
            theta = a_to_theta(normalize_nonneg(best_a + noise * np.random.standard_normal(n), S))
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

        # Compute gradient
        use_peaks = progress > top_k_switch
        if use_peaks:
            # Blend L_p gradient with peak gradient
            k = min(10, 2 + int(20 * (progress - top_k_switch) / (1.0 - top_k_switch)))
            grad_peaks = compute_peak_gradient(c, a, top_k=k, decay=0.8)
            grad_p = compute_lp_gradient(c, a, p)
            blend = (progress - top_k_switch) / (1.0 - top_k_switch)
            grad_a = (1 - 0.5 * blend) * grad_p + 0.5 * blend * grad_peaks
        else:
            grad_a = compute_lp_gradient(c, a, p)

        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-15:
            break

        # Convert to simplex gradient
        g_theta = simplex_gradient(grad_a, a, S)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Learning rate schedule: start high, decay
        lr = lr_init * (1.0 - 0.5 * progress) * (1.0 + 0.5 * np.cos(np.pi * progress))

        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if verbose and t % 5000 == 0:
            print("    LpAdam step %d: obj=%.7f, p=%.0f, T=%.0f%%" % (t, best_obj, p, progress * 100))

    return best_a, best_obj


def run_peak_shaving(a_init, n, S, time_budget, verbose=False):
    """
    Directly target and reduce the single maximum peak using
    exact subgradient + backtracking line search.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    step = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        cur_obj = c.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            break

        # Find top peaks (up to 5)
        k = min(5, len(c))
        part_idx = np.argpartition(c, -k)[-k:]
        peak_vals = c[part_idx]

        # Weighted: emphasize the max
        w = np.zeros(len(c))
        for i, idx in enumerate(part_idx):
            w[idx] = peak_vals[i] ** 2
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum

        # Gradient via cross-correlation
        fsize = 1 << (len(c) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        grad_a = 2.0 * corr[:n]

        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-15:
            break
        grad_dir = grad_a / gnorm

        # Backtracking line search
        lr = 0.05
        found = False
        for i in range(15):
            a_trial = a - lr * grad_dir
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()

            if obj_trial < cur_obj - 1e-10 * cur_obj:
                a = a_trial
                found = True
                break
            lr *= 0.5

        if not found:
            # Try smaller step or skip
            if lr < 1e-8:
                break
            a = a - lr * grad_dir
            a = normalize_nonneg(a, S)

        step += 1
        if verbose and step % 100 == 0:
            print("    PeakShave step %d: obj=%.7f" % (step, best_obj))

    return best_a, best_obj


def coordinate_descent(a_init, n, S, time_budget, max_sweeps=100, verbose=False):
    """Coordinate descent: try multiplicative changes to individual coordinates."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(777)
    start = time.monotonic()

    factors = [0.3, 0.5, 0.7, 1.3, 1.5, 2.0]

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            original = a[i]
            if original < 1e-10:
                continue

            current_c = fft_convolve_self(a)
            cur_obj = current_c.max()
            best_factor = 1.0
            best_local = cur_obj

            for f in factors:
                a[i] = original * f
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local - 1e-12:
                    best_local = obj_check
                    best_factor = f

            a[i] = original * best_factor
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)

            if c.max() < best_obj - 1e-12:
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        if not improved and sweep >= 5:
            break

    return best_a, best_obj


def try_zero_masking(a, n, S, n_zero=20):
    """Try setting smallest coordinates to zero."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    sorted_idx = np.argsort(a)
    for idx in sorted_idx[:n_zero * 2]:
        if a[idx] < 1e-10:
            continue
        saved = a[idx]
        a[idx] = 0.0
        a = normalize_nonneg(a, S)
        c_check = fft_convolve_self(a)
        if c_check.max() < best_obj - 1e-12:
            best_obj = c_check.max()
            best_a = a.copy()
        else:
            a[idx] = saved
            a = normalize_nonneg(a, S)

    return best_a, best_obj


def get_initialization(name, n, S, rng):
    """Generate various initializations."""
    i = np.arange(n)
    t = i / n
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n) + 0.01
        return a * (S / a.sum())
    elif name == "gaussian":
        a = np.exp(-2.0 * ((t - 0.5) / 0.15) ** 2)
        return a * (S / a.sum())
    elif name == "cosine_rich":
        phases = rng.random(8) * 2 * np.pi
        a = np.ones(n)
        for k in range(8):
            freq = (2 * k + 3)  # odd frequencies
            a += 0.1 * np.cos(2 * np.pi * freq * t + phases[k])
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "sine_modulated":
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 7 * t + rng.random() * 2) + \
            0.2 * np.sin(2 * np.pi * 13 * t + rng.random() * 2) + \
            0.1 * np.sin(2 * np.pi * 29 * t + rng.random() * 2)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "bimodal":
        a = np.exp(-50 * (t - 0.3) ** 2) + np.exp(-50 * (t - 0.7) ** 2)
        return a * (S / a.sum())
    elif name == "ramp":
        a = np.linspace(0.1, 1.0, n)
        return a * (S / a.sum())
    elif name == "chirp":
        a = 1.0 + 0.2 * np.sin(2 * np.pi * (t * 50 + 5 * t * t))
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def run_peak_adaptive(a_init, n, S, time_budget, verbose=False):
    """
    Adaptive peak targeting: maintain a set of active peaks and
    recompute the gradient as a weighted combination that adapts
    based on which peaks are currently dominant.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    theta = a_to_theta(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    no_improve = 0

    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)

        a = theta_to_a(theta, S)
        c = fft_convolve_self(a)
        cur_obj = c.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            noise = 0.005
            a_noisy = best_a + noise * np.random.standard_normal(n)
            theta = a_to_theta(normalize_nonneg(a_noisy, S))
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

        # Adaptive top-k: k increases as we converge
        base_k = 3
        extra_k = int(10 * progress)
        k = min(base_k + extra_k, 20, len(c))

        grad_a = compute_peak_gradient(c, a, top_k=k, decay=0.75)

        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-15:
            break

        g_theta = simplex_gradient(grad_a, a, S)

        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = 0.08 * (1.0 - 0.4 * progress)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if verbose and t % 5000 == 0:
            print("    PeakAdapt step %d: obj=%.7f, k=%d" % (t, best_obj, k))

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 30.0)
    deadline = time.monotonic() + total_budget

    global_best = np.inf
    global_a = None

    # Load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: " + str(e))

    # Decide primary length
    if given_a is not None:
        n_primary = len(given_a)
    else:
        n_primary = 4096

    S = np.sqrt(2.0 * float(n_primary))
    n = n_primary

    print("Primary: n=%d, S=%.4f" % (n, S))

    # ============ PHASE 1: L_p norm optimization (40% time) ============
    p1_budget = 0.40 * total_budget
    print("\nPhase 1: L_p Adam (budget=%.0fs)" % p1_budget)

    if given_a is not None:
        a_start = given_a.copy()
    else:
        a_start = get_initialization("cosine_rich", n, S, rng)

    a_best, obj_best = run_lp_adam(
        a_start, n, S, p1_budget,
        p_init=2.0, p_final=3000.0,
        lr_init=0.08, top_k_switch=0.4,
        verbose=True
    )
    a_best = normalize_nonneg(a_best, S)
    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  Phase 1 done: obj=%.7f" % ver)

    # ============ PHASE 2: Peak adaptive refinement (25% time) ============
    p2_budget = 0.25 * total_budget
    print("\nPhase 2: Peak Adaptive (budget=%.0fs)" % p2_budget)

    if global_a is not None:
        a_best, _ = run_peak_adaptive(
            global_a, n, S, p2_budget, verbose=True
        )
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  Phase 2 done: obj=%.7f" % ver)

    # ============ PHASE 3: Peak shaving with line search (20% time) ============
    p3_budget = 0.20 * total_budget
    print("\nPhase 3: Peak Shaving (budget=%.0fs)" % p3_budget)

    if global_a is not None:
        a_best, _ = run_peak_shaving(
            global_a, n, S, p3_budget, verbose=True
        )
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  Phase 3 done: obj=%.7f" % ver)

    # ============ PHASE 4: Coordinate descent polish (10% time) ============
    p4_budget = 0.10 * total_budget
    print("\nPhase 4: Coordinate Descent (budget=%.0fs)" % p4_budget)

    if global_a is not None:
        a_best, _ = coordinate_descent(
            global_a, n, S, p4_budget, max_sweeps=50
        )
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  Phase 4 done: obj=%.7f" % ver)

        # Zero masking
        a_best, _ = try_zero_masking(global_a, n, S, n_zero=10)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  ZeroMask: obj=%.7f" % ver)

    # ============ PHASE 5: Additional L_p with different params (5% time) ============
    p5_budget = deadline - time.monotonic()
    if global_a is not None and p5_budget > 10:
        print("\nPhase 5: Final L_p refinement (budget=%.0fs)" % p5_budget)

        # Run with higher p immediately
        a_best, _ = run_lp_adam(
            global_a, n, S, p5_budget,
            p_init=100.0, p_final=10000.0,
            lr_init=0.03, top_k_switch=0.0,
            verbose=False
        )
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  Phase 5: obj=%.7f" % ver)

        # Final coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_best, _ = coordinate_descent(global_a, n, S, remaining * 0.8, max_sweeps=20)
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  Final CD: obj=%.7f" % ver)

    # Final verification
    if global_a is None:
        global_a = np.ones(256)
        S = np.sqrt(2.0 * 256)
        global_a = global_a * (S / global_a.sum())

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f (verified=%s)" % (
        global_best, "%.7f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        print("Fallback used")

    return final_a
