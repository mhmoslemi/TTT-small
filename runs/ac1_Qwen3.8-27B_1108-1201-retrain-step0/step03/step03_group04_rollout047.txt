Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Spectral (Fienup/Gerchberg-Saxton) alternating projection to flatten

The spectral method alternates between:
  - Setting |FFT(a_pad)| to a constant target (flat spectrum constraint)
  - Projecting onto non-negative, compact support (time-domain constraint)
This is followed by subgradient polishing on the residual peaks.

import numpy as np

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute weighted gradient of convolution via FFT."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def gradient_polish(a_init, n, S, max_time, rng):
    """Subgradient polishing with adaptive step size."""
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best = a.copy()
    best_obj = objective_value(a)
    start = time.monotonic()
    it = 0
    step = S / n  # Initial step size
    best_ever = best_obj
    stall = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best = a.copy()
            stall = 0
            step = min(step * 1.05, S)
        else:
            stall += 1
            if stall > 100:
                step *= 0.7
        
        # Find the top few lags
        top_k = min(10, 1 + it // 200)
        top_indices = np.argpartition(c, -top_k)[-top_k:]
        top_vals = c[top_indices]
        
        # Weighted subgradient
        w_sum = top_vals.sum()
        if w_sum < 1e-30:
            break
        weights = np.zeros(len(c))
        weights[top_indices] = top_vals / w_sum
        
        grad = compute_gradient_fft(a, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step
        x_new = project_simplex(a - step * grad / gnorm * np.sqrt(gnorm), S)
        
        # Check improvement
        c_new = fft_convolve_self(x_new)
        if c_new.max() < cur_max:
            a = x_new
        else:
            # Try smaller step
            x_half = project_simplex(a - 0.3 * step * grad / gnorm * np.sqrt(gnorm), S)
            c_half = fft_convolve_self(x_half)
            if c_half.max() < cur_max:
                a = x_half
                step *= 0.8
            else:
                # Random perturbation
                if rng.random() < 0.3:
                    noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                    a = project_simplex(a + noise, S)
        
    return best, best_obj


def propose_candidate(seed=42, budget_s=2385, **kwargs):
    """
    Minimize 2*n*max_conv/sum^2 over non-negative sequences.
    Uses Fienup spectral method + gradient polishing with multiple restarts.
    """
    global start_global
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s - 25.0)
    start_global = deadline
    
    # Determine n
    n = 8192
    S = np.sqrt(2.0 * n)
    
    # Get parent sequence as one initialization
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if len(p) == n and p.sum() > 0:
            parent = np.clip(p, 0, 1000)
    except Exception:
        pass
    
    global_best_obj = np.inf
    global_best_a = None
    
    # === Phase 1: Fienup spectral method (50% of time) ===
    remaining = deadline - time.monotonic()
    phase1_time = 0.5
    if remaining > 20:
        print("Phase 1: Fienup spectral method (%.0f sec)" % (0.5 * remaining))
        t0 = time.monotonic()
        
        # Run with multiple initializations
        inits = []
        
        # Uniform
        inits.append(('uniform', np.ones(n) * S / n))
        
        # Parent
        if parent is not None:
            inits.append(('parent', parent * (S / parent.sum())))
        
        # Random
        r = rng.uniform(0, 1, n)
        inits.append(('random', r * (S / r.sum())))
        
        # Sine-based
        t = np.arange(n) / n
        for freq in [1, 5]:
            s = 1.0 + 0.2 * np.sin(2 * np.pi * freq * t)
            inits.append(('sine_%d' % freq, s * (S / s.sum())))
        
        # Sparse
        mask = (rng.random(n) < 0.5).astype(float)
        inits.append(('sparse', mask * (S / mask.sum())))
        
        per_init = 0.5 * (deadline - time.monotonic()) / len(inits)
        for name, a_init in inits:
            if time.monotonic() > deadline - 60:
                break
            a, obj = fienup_flat(a_init, n, S, per_init, rng,
                                over_relax=1.0 + 0.1 * rng.random())

                global_best_a = a.copy()
                print("  Fienup [%s]: obj=%.6f" % (name, global_best_obj))
        
        print("Phase 1 done: best obj=%.6f [%.0fs]" % (global_best_obj, time.monotonic() - t0))
    
    # === Phase 2: Gradient polishing (30% of time) ===
    remaining = deadline - time.monotonic()

        if obj < global_best_obj:
            global_best_obj = obj
            global_best_a = a.copy()
            print("  Gradient polish: obj=%.6f" % global_best_obj)
        print("Phase 2 done [%.0fs]" % (time.monotonic() - t0))
