def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, c, top_k=50):
    """
    Compute the subgradient of max_k c[k] using top-k peaks.
    Returns gradient vector of length n.
    """
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    # Simple average of gradients at top peaks
    # grad[j] = (1/K) * sum_{k in top} a[k-j] (for valid indices)
    grad = np.zeros(n)
    for k in top_indices:
        k = int(k)
        # (a*a)[k] = sum_i a[i]*a[k-i], d/d a[j] = a[k-j] (if valid)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += a[k - j]
    grad /= top_k
    return grad


def compute_subgradient_fft(a, c, top_k=50):

    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(W * np.conj(A)))[:n]
    return grad


def solve_dual_lp(a, peak_indices, M):
    """
    Solve the dual LP:
    Min M * sum lambda_k
    s.t. sum_k lambda_k * a[peak_k - j] >= 1 for all j
         lambda_k >= 0
    Returns optimal lambda and the objective value.
    """
    n = len(a)
    K = len(peak_indices)
    
    # Build constraint matrix C (n x K): C[j, k] = a[peak_k - j] if valid, else 0
    # We need: C @ lambda >= 1 (all ones vector)
    # And: lambda >= 0
    # Objective: M * sum(lambda) = M * 1^T @ lambda
    
    # For efficiency, build C as a sparse-like structure
    # C[j, k] = a[peak_indices[k] - j] if 0 <= peak_indices[k] - j <= n-1, else 0
    
    c_obj = M * np.ones(K)  # minimize M * sum(lambda_k)
    
    # Inequality constraints: -C @ lambda <= -1
    # i.e., C @ lambda >= 1
    # linprog form: A_ub @ x <= b_ub
    # So: -C @ lambda <= -1
    
    # Build A_ub = -C (n x K)
    # This is large, so let's build it efficiently
    A_ub = np.zeros((n, K))
    for k_idx, pk in enumerate(peak_indices):
        pk = int(pk)
        lo = max(0, pk - n + 1)
        hi = min(n - 1, pk)
        A_ub[lo:hi+1, k_idx] = -a[pk - np.arange(lo, hi + 1)]
    
    b_ub = -np.ones(n)
    
    bounds = [(0, None)] * K
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
        else:
            return np.ones(K) / K  # Fallback: uniform weights
    except Exception:
        return np.ones(K) / K

    """

"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 using accelerated projected subgradient.

Key improvements over previous:
1. Focus 75% of time on the best-known sequence for deeper convergence.
2. Nesterov acceleration with adaptive step and periodic full-evaluation checks.
3. Adaptive top-k subgradient (start small, grow over time for finer resolution).
4. LP-guided direction for initial iterations (dual LP gives optimal peak weights).
5. Exploration phase with diverse asymmetric initializations and different n.
6. Final fine-grained polish with very large top-k and tiny noise.

import numpy as np
import time
import sys
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def subgradient_fft(a, c, top_k=10):
    """Compute subgradient using top-k peaks, via FFT cross-correlation."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    w = np.zeros(2 * n - 1)
    for k in top_indices:
        w[int(k)] = 1.0 / top_k
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(W * np.conj(A)))[:n]
    return grad


def solve_dual_lp_for_weights(a, peak_indices):
    """
    Solve the dual LP to get optimal weights on peaks.
    Min sum(lambda_k) s.t. sum_k lambda_k * a[peak_k - j] >= 1 for all j, lambda >= 0.
    Returns (lambdas, success).
    """
    n = len(a)
    K = len(peak_indices)
    if K == 0:
        return np.array([]), False

    c_obj = np.ones(K)
    A_ub = np.zeros((n, K))
    for k_idx, pk in enumerate(peak_indices):
        pk = int(pk)
        lo = max(0, pk - n + 1)
        hi = min(n - 1, pk)
        if lo <= hi:
            A_ub[lo:hi + 1, k_idx] = -a[pk - np.arange(lo, hi + 1)]
    b_ub = -np.ones(n)
    bounds = [(0.0, None)] * K

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10.0})
        if result.success:
            return result.x, True
    except Exception:
        pass
    return np.ones(K) / K, False


def lp_guided_subgradient(a, c, top_k=30):
    """Compute LP-guided subgradient using optimal peak weights from dual LP."""
    n = len(a)
    peak_indices = np.argsort(c)[::-1][:top_k]
    lambdas, ok = solve_dual_lp_for_weights(a, peak_indices)
    if not ok:
        return subgradient_fft(a, c, top_k)
    w = np.zeros(2 * n - 1)
    for k_idx, pk in enumerate(peak_indices):
        w[int(pk)] = lambdas[k_idx]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(W * np.conj(A)))[:n]
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def objective_val(a):
    """Compute the objective value 2n*max_conv/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def run_nesterov(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=512,
                 use_lp=True, lp_iters=30, noise_scale=0.003):
    """
    Nesterov-accelerated projected subgradient method.
    Uses LP-guided gradients for initial iterations, then FFT top-k.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    c = fft_convolve_self(a)
    best_score = c.max()

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive top-k: grow from top_k_init to top_k_max
        frac = min(1.0, it / max(1, int(max_time * 20)))
        k = int(top_k_init + (top_k_max - top_k_init) * frac)
        k = max(k, 3)

        # Compute gradient
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()

        if use_lp and it <= lp_iters:
            try:
                grad = lp_guided_subgradient(y, c, min(k, 20))
            except Exception:
                grad = subgradient_fft(y, c, k)
        else:
            grad = subgradient_fft(y, c, k)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: starts large, decays
        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        step = min(step, 0.5 * S / max(gnorm, 1e-15) * 2.0)

        x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a_prev)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        a_prev = a
        a = x_new
        t = t_new

        # Periodic full evaluation and noise injection
        if it % 200 == 0:
            c_check = fft_convolve_self(x_new)
            if c_check.max() < best_score:
                best_score = c_check.max()
                best_local = x_new.copy()

        if it % 800 == 0 and rng.random() < 0.25:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0

    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best_score:
        best_score = c_final.max()
        best_local = a.copy()

    return best_local, best_score

def generate_asymmetric_init(n, rng, method="exp"):
    """Generate diverse asymmetric initializations."""
    tt = np.linspace(0, 1, n, endpoint=False)
    if method == "exp_fwd":
        a = np.exp(3.0 * tt)
    elif method == "exp_bwd":
        a = np.exp(3.0 * (1.0 - tt))
    elif method == "pow_fwd":
        a = (tt + 0.01) ** 2.0
    elif method == "pow_bwd":
        a = (1.0 - tt + 0.01) ** 2.0
    elif method == "sin_trend":
        a = 1.0 + 0.5 * np.sin(2 * np.pi * 5.0 * tt) + 0.8 * tt
        a = np.maximum(a, 0.01)
    elif method == "bimodal":
        a = np.exp(-20.0 * (tt - 0.2) ** 2) + 0.5 * np.exp(-20.0 * (tt - 0.8) ** 2)
    elif method == "rand_ramp":
        a = rng.random(n) * (0.5 + tt)
    else:
        a = np.exp(2.0 * tt)
    return a
