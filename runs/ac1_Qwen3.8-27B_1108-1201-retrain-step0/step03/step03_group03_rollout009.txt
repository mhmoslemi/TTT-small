"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Uses FISTA with log-sum-exp smoothing, adaptive beta annealing,
backtracking line search, and a novel pair-redistribution local search
for final polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
        theta = (cumsum[rho - 1] - S) / rho
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_smoothed_gradient(a, c, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    a_ref = a[::-1]
    # conv = a_ref * w (linear convolution), length 3n-2
    fsize = 1 << (3 * n - 3).bit_length()
    A = fft(a_ref, fsize)
    W = fft(w, fsize)
    conv = np.real(ifft(A * W))[:3*n-2]
    grad = 2.0 * conv[n-1:2*n-1]
    return grad

def run_fista(a_init, n, S, max_time, rng, beta_start=1.0, beta_end=200.0, 
              beta_power=1.5, L_init=None, perturb_interval=200, perturb_prob=0.2):
    """
    FISTA with log-sum-exp smoothing and adaptive beta.
    Minimizes (1/beta)*log(sum(exp(beta*c[k]))) which approximates max(c) for large beta.
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    L = L_init if L_init else 4.0  # Initial Lipschitz estimate
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: increases over time
        frac = it / max(it, 1)  # We'll use time-based instead
        elapsed = time.monotonic() - start
        time_frac = min(1.0, elapsed / max_time)
        beta = beta_start * (beta_end / beta_start) ** (beta_power * time_frac)
        
        # Compute convolution and gradient
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_a = y.copy()
        
        grad = compute_smoothed_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        f_y = (1.0/beta) * np.log(np.sum(np.exp(beta * (c - c.max())))) + c.max()
        step = 1.0 / L
        x_new = None
        for bt in range(20):
            x_cand = project_simplex(y - step * grad, S)
            if bt == 0:
                x_new = x_cand
                # Only do line search every 10 iterations to save time
                if it % 10 == 0:
                    c_cand = fft_convolve_self(x_cand)
                    f_cand = (1.0/beta) * np.log(np.sum(np.exp(beta * (c_cand - c_cand.max())))) + c_cand.max()
                    if f_cand > f_y + 1e-8:
                        step *= 0.5
                        continue
                break
            else:
                c_cand = fft_convolve_self(x_cand)
                f_cand = (1.0/beta) * np.log(np.sum(np.exp(beta * (c_cand - c_cand.max())))) + c_cand.max()
                if f_cand <= f_y - 0.5 * step * gnorm * gnorm:
                    x_new = x_cand
                    break
                step *= 0.5
        
        if x_new is None:
            x_new = project_simplex(y - step * grad, S)
        
        # Update L
        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() > cur_max * 1.01:
                L *= 1.3
            elif it % 50 == 0:
                L = max(L * 0.9, 2.0)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2))
        alpha = (t_mom - 1.0) / t_new
        y = x_new + alpha * (x_new - a)
        y = np.maximum(y, 0.0)
        # Ensure sum is approximately S (projection handles it, but momentum might break it)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        a = x_new
        t_mom = t_new
        
        # Periodic perturbation
        if it % perturb_interval == 0 and rng.random() < perturb_prob:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
    
    return best_a, best_score

def run_pair_redistribution(a_init, n, S, max_time, rng, top_k=10, max_pairs=20):
    """
    Local search that directly targets the top convolution lags.
    For each top lag, find the pairs with largest product and try to reduce them
    by redistributing mass to "safe" positions.
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    no_improve = 0
    
    while time.monotonic() - start < max_time and no_improve < 500:
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find top lags
        top_lags = np.argsort(c)[::-1][:top_k]
        
        # For each top lag, try to reduce
        improved = False
        for lag in top_lags:
            lag = int(lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue
            indices = np.arange(i_min, i_max + 1)
            partner = lag - indices
            products = a[indices] * a[partner]
            
            # Find the pair with largest product
            max_pair_idx = np.argmax(products)
            i = indices[max_pair_idx]
            j = partner[max_pair_idx]
            
            if products[max_pair_idx] < 1e-15:
                continue
            
            # Try to reduce a[i] and a[j] and add to "safe" positions
            # Safe positions: those that don't contribute to any top lag
            top_lags_set = set(int(l) for l in top_lags)
            
            # Find safe positions (low contribution to all top lags)
            safe_scores = np.zeros(n)
            for tl in top_lags:
                tl = int(tl)
                for pos in range(n):
                    partner_pos = tl - pos
                    if 0 <= partner_pos < n:
                        safe_scores[pos] += a[partner_pos]
            
            # We want to add mass to positions with low safe_scores
            # But we need to be careful not to create new peaks
            
            # Simple strategy: reduce a[i] by delta, add delta to a random "low" position
            delta = 0.01 * a[i]
            if delta < 1e-15:
                continue
            
            # Try reducing the larger of a[i], a[j]
            if a[i] >= a[j]:
                reduce_idx = i
                add_amount = delta
            else:
                reduce_idx = j
                add_amount = delta
            
            # Find good positions to add mass
            # Positions where adding mass won't significantly increase any top lag
            candidate_positions = np.where(a < np.median(a))[0]
            if len(candidate_positions) == 0:
                continue
            
            # Score each candidate by how much it would increase the top lags
            best_add_score = np.inf
            best_add_pos = None
            for pos in candidate_positions[:min(100, len(candidate_positions))]:
                increase = 0
                for tl in top_lags:
                    tl = int(tl)
                    partner_pos = tl - pos
                    if 0 <= partner_pos < n:
                        increase += a[partner_pos] * add_amount
                score = increase + 0.1 * a[pos]  # prefer positions with small a
                if score < best_add_score:
                    best_add_score = score
                    best_add_pos = pos
            
            if best_add_pos is None:
                continue
            
            # Apply the change
            a_new = a.copy()
            a_new[reduce_idx] -= add_amount
            a_new[best_add_pos] += add_amount
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new * (S / s_new)
            
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max - 1e-10:
                a = a_new
                improved = True
        
        if not improved:
            # Try a random perturbation
            if rng.random() < 0.3:
                mask = rng.random(n) < 0.05
                a_pert = a.copy()
                a_pert[mask] *= (1 + rng.normal(0, 0.1, mask.sum()))
                a_pert = np.maximum(a_pert, 0.0)
                s_pert = a_pert.sum()
                if s_pert > 1e-15:
                    a_pert = a_pert * (S / s_pert)
                c_pert = fft_convolve_self(a_pert)
                if c_pert.max() < best_score:
                    a = a_pert
    
    return best_a, best_score

def run_subgradient_momentum(a_init, n, S, max_time, rng, top_k=5, step_init=0.1):
    """
    Subgradient method with momentum, targeting top-k lags with uniform weights.
    Simpler than FISTA but can be effective with the right step size.
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    v = np.zeros(n)  # momentum
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    step = step_init * S / n
    gamma = 0.9  # momentum decay
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        
        # Compute subgradient from top-k lags
        k = min(top_k, len(c))
        top_indices = np.argsort(c)[::-1][:k]
        grad = np.zeros(n)
        for idx in top_indices:
            idx = int(idx)
            for j in range(max(0, idx - n + 1), min(idx, n - 1) + 1):
                partner = idx - j
                if 0 <= partner < n:
                    grad[j] += 2.0 * a[partner] / k
        
        # Subgradient descent with momentum
        v = gamma * v + (1 - gamma) * grad
        step = 2.0 * S / (n * (np.linalg.norm(v) + 1e-10)) * 0.5
        
        a_new = project_simplex(a - step * v, S)
        a = a_new
    
    return best_a, best_score

def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) * 0.5 + 0.25
    mu = S / half
    a = np.zeros(n)
    a[:half] = h * mu  # scale h to be in reasonable range
    for i in range(half):
        a[n - 1 - i] = mu - a[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total > 0:
        a = a * (S / total)
    else:
        a = np.ones(n) * S / n
    return a

def make_periodic_init(n, rng, S, num_freqs=4):
    """Periodic initialization with multiple sinusoidal components."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, max(3, n // 100))
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.5
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.05)
    a = a * (S / a.sum())
    return a

def make_edge_concentrated_init(n, rng, S):
    """Concentrate mass at edges (motivated by the analysis that c[n-1] is the middle)."""
    tt = np.arange(n, dtype=np.float64)
    # Distance from center
    center = (n - 1) / 2.0
    dist = np.abs(tt - center) / center  # 0 at center, 1 at edges
    # Raise to a power to concentrate more at edges
    p = rng.random() * 2 + 1
    a = (dist ** p + 0.05)
    a = a * (S / a.sum())
    return a

def make_triang_inv_init(n, rng, S):
    """Inverse triangle: large at edges, small in middle."""
    tt = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 - np.abs(tt - center) / center * 0.8
    a = np.maximum(a, 0.05)
    a = a * (S / a.sum())
    return a

def safe_score(a):
    """Compute the objective value 2*n*max_b/sum_a^2."""
    n = len(a)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)
