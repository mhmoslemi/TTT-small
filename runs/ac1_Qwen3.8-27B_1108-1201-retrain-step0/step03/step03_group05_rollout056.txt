import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize

def fft_convolve_self(a):
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size-1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A*A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def evaluate_local(a):
    """Compute the score for a sequence (without calling evaluate_sequence)."""
    n = len(a)
    c = fft_convolve_self(a)
    max_c = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a**2)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Approach:
1. Directional line search with structured (sinusoidal, polynomial) and gradient-based directions
   using exact convex line search (objective is convex in alpha for fixed direction)
2. Nesterov-accelerated subgradient with FFT-accelerated gradient computation
3. L-BFGS with log-sum-exp smoothing for final polishing
4. Multi-length search and cross-initialization
Key innovation: Using the fact that max(conv(a+a*d, a+a*d)) is convex in alpha,
enabling exact line searches along structured directions that the previous
softmax-based L-BFGS approach may miss.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    if fsize < size:
        fsize = size
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def score_of(a):
    """Compute 2*n*max(c)/sum(a)^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_c = c.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


def neg_gradient_zero_mean(a, c, w):
    """
    Compute -2*conv(w, a) projected to zero-mean.
    w: weights on convolution bins (sum to 1, support on active bins).
    Returns a zero-mean descent direction.
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    if fsize < len(w) + n - 1:
        fsize = 1 << (len(w) + n - 1 - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    # Project to zero-mean
    grad = grad - grad.mean()
    return -grad  # descent direction


def line_search_direction(a, d, S, max_evals=60):
    """
    Exact line search: minimize score_of(a + alpha*d) over alpha.
    d is zero-mean (so sum is preserved). Project to a >= 0.
    Objective is convex in alpha, so use golden section search.
    """
    n = len(a)
    d = np.asarray(d, dtype=np.float64)
    d_norm = np.linalg.norm(d)
    if d_norm < 1e-15:
        return a, 0.0

    d = d / d_norm  # normalize direction

    # Determine valid alpha range (non-negativity)
    # a + alpha*d >= 0
    # For d[i] > 0: alpha >= -a[i]/d[i]
    # For d[i] < 0: alpha <= -a[i]/d[i]
    pos_mask = d > 1e-15
    neg_mask = d < -1e-15
    alpha_min = 0.0
    alpha_max = 0.0

    if np.any(pos_mask):
        alpha_min = max(alpha_min, -a[pos_mask].min() / d[pos_mask].max() * 0.9)
    if np.any(neg_mask):
        alpha_max = min(alpha_max, a[neg_mask].min() / (-d[neg_mask].max()) * 0.9)

    # Also bound by a reasonable range
    scale = S / n
    alpha_min = max(alpha_min, -3.0 * scale / d_norm * 10)
    alpha_max = min(alpha_max, 3.0 * scale / d_norm * 10)

    if alpha_max - alpha_min < 1e-15:
        return a, 0.0

    # Coarse grid search first
    alphas = np.linspace(alpha_min, alpha_max, 40)
    best_score = np.inf
    best_alpha = 0.0
    best_a = a.copy()

    for alpha in alphas:
        a_new = a + alpha * d
        if a_new.min() < -1e-10:
            a_new = project_simplex(a_new, S)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new * S / s
        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_alpha = alpha
            best_a = a_new.copy()

    # Fine golden section search around best
    step = (alpha_max - alpha_min) / 40.0
    lo = max(alpha_min, best_alpha - 3 * step)
    hi = min(alpha_max, best_alpha + 3 * step)

    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)

    def eval_at(alpha):
        a_new = a + alpha * d
        if a_new.min() < -1e-10:
            a_new = project_simplex(a_new, S)
        s = a_new.sum()
        if s < 1e-15:
            return np.inf
        a_new = a_new * S / s
        return score_of(a_new)

    fc = eval_at(c_pt)
    fd = eval_at(d_pt)
    evals = 2

    while hi - lo > 1e-12 and evals < max_evals:
        evals += 1
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = eval_at(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = eval_at(d_pt)

    opt_alpha = (lo + hi) / 2.0
    a_opt = a + opt_alpha * d
    if a_opt.min() < -1e-10:
        a_opt = project_simplex(a_opt, S)
    s = a_opt.sum()
    if s > 1e-15:
        a_opt = a_opt * S / s
    else:
        a_opt = best_a

    return a_opt, opt_alpha


def gen_cos_sin_dir(n, m, antisym=True):
    """Generate a zero-mean sinusoidal direction."""
    tt = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    if antisym:
        d = np.sin(2.0 * np.pi * m * (tt - center) / n)
    else:
        d = np.cos(2.0 * np.pi * m * (tt - center) / n)
    d = d - d.mean()
    dn = np.linalg.norm(d)
    if dn > 1e-15:
        d = d / dn
    return d


def gen_poly_dir(n, p, antisym=True):
    """Generate a zero-mean polynomial direction."""
    tt = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (tt - center) / center  # in [-1, 1]
    d = x ** p
    d = d - d.mean()
    dn = np.linalg.norm(d)
    if dn > 1e-15:
        d = d / dn
    return d

                best_score = sc

        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:

        for i, k in enumerate(top_indices):

        W = fft(w_full, fsize)

        # Nesterov update: y_new = x_new + (t-1)/t_new * (x_new - x_old)

def run_lbfgs_smoothed(a, S, max_time, rng):
    """L-BFGS on softmax parameterization with log-sum-exp smoothing."""
    n = len(a)
    a = np.clip(a, 1e-15, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:

    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = score_of(a)
    best_a = a.copy()

    beta_schedule = [10, 50, 200, 800, 2000]

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 10:
            break

        def fun_grad(z_val):
            z_s = z_val - z_val.max()
            exp_z = np.exp(z_s)
            a_val = S * exp_z / exp_z.sum()
            c = fft_convolve_self(a_val)
            c_max = c.max()
            exp_bc = np.exp(beta * (c - c_max))
            sum_ebc = exp_bc.sum()

            w = exp_bc / sum_ebc

            fsize = 1 << (len(w) + n - 2 - 1).bit_length()

            a1 = a2

    # ============================================================
    # PHASE 3: Nesterov subgradient
    # ============================================================
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n--- Phase 3: Nesterov Subgradient ---")
        t0 = time.monotonic()
        a3, _ = run_toppk_subgradient(a1, S, min(180, remaining * 0.4), rng, top_k=64)
        update_global(a3)
        if score_of(a3) < score_of(a1):
            a1 = a3

    # ============================================================
    # PHASE 4: L-BFGS smoothing
    # ============================================================
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n--- Phase 4: L-BFGS Polishing ---")
        t0 = time.monotonic()
        a4, _ = run_lbfgs_smoothed(a1, S, min(120, remaining * 0.4), rng)
        update_global(a4)
        if score_of(a4) < score_of(a1):
            a1 = a4

    # ============================================================
    # PHASE 5: More directional search (fine)
    # ============================================================
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n--- Phase 5: Fine Directional Search ---")
        t0 = time.monotonic()
        # Use gradient-based directions too
        c = fft_convolve_self(a1)
        top_k = min(128, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        w = top_vals / top_vals.sum()
        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = w
        fsize = 1 << (2 * n - 1 + n - 2 - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a1, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        d_grad = -(grad - grad.mean())
        gn = np.linalg.norm(d_grad)
        if gn > 1e-14:
            a1_g, _ = line_search_direction(a1, d_grad / gn, S, max_evals=50)
            update_global(a1_g)
            if score_of(a1_g) < score_of(a1):
                a1 = a1_g

        a5, _ = run_directional_search(a1, S, min(180, remaining * 0.5), rng, n_dirs=300)
        update_global(a5)
        if score_of(a5) < score_of(a1):
            a1 = a5

    # ============================================================
    # PHASE 6: Try longer sequence
    # ============================================================
    remaining = deadline - time.monotonic()
    if remaining > 120 and n_long > n:
        print("\n--- Phase 6: Longer Sequence (n=%d) ---" % n_long)
        n2 = n_long
        S2 = np.sqrt(2.0 * float(n2))

        # Upsample the best sequence to longer length
        from numpy import interp
        tt_old = np.linspace(0, 1, n, endpoint=False)
        tt_new = np.linspace(0, 1, n2, endpoint=False)
        a_long = interp(tt_new, tt_old, a1)
        a_long = np.maximum(a_long, 0.0)
        a_long = a_long * S2 / max(a_long.sum(), 1e-15)

        # Also try uniform + modulation for longer
        tt2 = np.linspace(0, 1, n2, endpoint=False)
        a_uniform = np.ones(n2) + 0.3 * np.sin(2 * np.pi * 5 * (tt2 - 0.5))
        a_uniform = np.maximum(a_uniform, 0.0)
        a_uniform = a_uniform * S2 / a_uniform.sum()

        best_long = a_long
        if score_of(a_uniform) < score_of(a_long):
            best_long = a_uniform

        update_global(best_long)

        # Directional search on longer
        t0 = time.monotonic()
        a_long, _ = run_directional_search(best_long, S2, min(200, remaining * 0.4), rng)
        update_global(a_long)

        # Gradient line search
        remaining2 = deadline - time.monotonic()
        if remaining2 > 60:
            a_long, _ = run_gradient_linesearch(a_long, S2, min(100, remaining2 * 0.4), rng)
            update_global(a_long)

    # ============================================================
    # PHASE 7: Final polishing
    # ============================================================
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("\n--- Phase 7: Final Polish ---")
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * float(n_final))
        a_final = global_a.copy() * S_final / max(global_a.sum(), 1e-15)

        # Quick directional search
        a_final, _ = run_directional_search(a_final, S_final, min(60, remaining * 0.4), rng, n_dirs=100)
        update_global(a_final)

        # L-BFGS
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_final, _ = run_lbfgs_smoothed(a_final, S_final, min(60, remaining2 * 0.5), rng)
            update_global(a_final)

    # ============================================================
    # Phase 8: Additional gradient iterations if time remains
    # ============================================================
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("\n--- Phase 8: Extra Gradient Steps ---")
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * float(n_final))
        a_final = global_a.copy() * S_final / max(global_a.sum(), 1e-15)
        a_final, _ = run_gradient_linesearch(a_final, S_final, min(120, remaining * 0.6), rng)
        update_global(a_final)

    # ============================================================
    # Verification
    # ============================================================
    print("\n" + "=" * 60)
    if global_a is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification score: %.8f" % (ver if np.isfinite(ver) else float('inf')))
        if not np.isfinite(ver):
            print("  WARNING: evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64) * np.sqrt(2.0 * 256) / 256
            final_list = [float(x) for x in global_a]
        else:
            final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    else:
        global_a = np.ones(256, dtype=np.float64) * np.sqrt(2.0 * 256) / 256
        final_list = [float(x) for x in global_a]

    total_time = time.monotonic() - t_global_start
    print("Final: n=%d, score=%.8f, time=%.1fs" % (len(final_list), global_best, total_time))
    print("Target was 1.5027. Result: %s" % ("ACHIEVED!" if global_best <= 1.5027 else "not yet"))
    print("=" * 60)

    return final_list
