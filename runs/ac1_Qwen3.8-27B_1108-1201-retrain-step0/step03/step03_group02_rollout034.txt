"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using smooth peak-reduction optimization.

Key innovations over previous approach:
1. Smooth "excess above threshold" objective: sum(max(0, C[k]-T)^2) - differentiable, no surrogate needed
2. Backtracking line search for robust step selection
3. Threshold annealing to progressively focus on peaks
4. Warm start from parent solution
5. Multiple restarts with perturbation to escape local minima
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def smooth_optimize(a_init, S, max_time, rng, T_init_frac=0.85, T_decay=0.9995, 
                    lr_init=None, n_backtrack=20, patience=200, perturb_interval=500):
    """
    Smooth peak-reduction optimization with threshold annealing and backtracking.
    
    Minimizes f(a) = sum_k max(0, C[k] - T)^2 by projected gradient descent
    with backtracking line search.
    """
    n = len(a)
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    # Initial convolution
    C = fft_convolve_self(a)
    max_C = C.max()
    T = T_init_frac * max_C
    
    if lr_init is None:
        lr_init = S / (2.0 * np.sqrt(n))
    lr = lr_init
    
    start = time.monotonic()
    best_maxC = max_C
    best_a = a.copy()
    no_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Update threshold (annealing)
        T = max(0.0, T * T_decay + 1e-10 * max_C)
        
        # Compute excess
        e = np.maximum(0.0, C - T)
        
        # Compute gradient
        g = fft_gradient_excess(a, e)
        gnorm = np.linalg.norm(g)
        
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        accepted = False
        for j in range(n_backtrack):
            step = lr / (gnorm + 1e-15)
            a_trial = project_simplex(a - step * g, S)
            C_trial = fft_convolve_self(a_trial)
            max_trial = C_trial.max()
            
            if max_trial < max_C * (1.0 - 1e-10):  # strict improvement
                a = a_trial
                C = C_trial
                max_C = max_trial
                accepted = True
                break
            else:
                lr *= 0.5
        
        if not accepted:
            # No improvement, try a smaller step or different direction
            lr *= 0.5
            if lr < 1e-10:
                # Reset learning rate and perturb
                lr = lr_init
                noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
                C = fft_convolve_self(a)
                max_C = C.max()
        
        if max_C < best_maxC:
            best_maxC = max_C
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic perturbation
        if no_improve > patience:
            noise = rng.normal(0, 0.02 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            C = fft_convolve_self(a)
            max_C = C.max()
            no_improve = 0
            lr = lr_init
        
        if it % 500 == 0:
            print("  iter %d: maxC=%.8f, T=%.6f, lr=%.4e" % (it, max_C, T, lr))
    
    return best_a, best_maxC
