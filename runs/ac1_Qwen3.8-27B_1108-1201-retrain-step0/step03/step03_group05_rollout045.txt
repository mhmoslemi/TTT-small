def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):

    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_weighted_gradient(a, w):
    """Compute sum_k w[k] * (db[k]/da) using FFT.
    Returns array of length n."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def smoothed_obj_and_grad(a, beta):
    """Compute log-sum-exp smoothed max of b and its gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp = exp_bc.sum()
    obj = c_max + np.log(sum_exp) / beta
    w = exp_bc / sum_exp
    grad = compute_weighted_gradient(a, w)
    return obj, grad, c


def nesterov_subgradient(a_init, n, S, max_time, rng,

    while time.monotonic() - start < max_time:

        total_iters += 1

            best_a = y.copy()

        else:

        top_vals = c[top_indices]
        # Use softmax weights over top-k for smoother gradient
        tk_max = top_vals.max()
        exp_tk = np.exp(50.0 * (top_vals - tk_max))
        weights = exp_tk / exp_tk.sum()

        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        lr = base_lr / (1.0 + 0.01 * it)

        success = False
        for bt in range(backtracking_max):

            c_new = fft_convolve_self(x_new)
            score_new = c_new.max()
            # Accept if score improved or is comparable

                success = True
                break

def lbfgs_smoothed(a_init, n, S, max_time, rng,

                jac=jac,

def softmax_from_z(z, n):
    """Convert z to softmax probabilities of length n."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

    # --- Phase 1: Nesterov subgradient on best available init ---
    usable = deadline - time.monotonic()
    if useful_init := (given_seq if given_seq is not None else None):
        n1 = len(given_seq)
    else:
        n1 = n_candidates[0]

    S1 = np.sqrt(2.0 * float(n1))
    a1 = safe_normalize(useful_init if useful_init is not None else rng.random(n1) + 0.1, n1)
    update_best(a1, n1)
    print("\n=== Phase 1: Nesterov (n=%d, %.0fs) ===" % (n1, usable * 0.35))

    nest_time = usable * 0.35
    t0 = time.monotonic()
    a1, sc1 = nesterov_subgradient(a1, n1, S1, nest_time, rng,
                                    top_k_start=3, top_k_end=60,
                                    lr_base=1.5, backtracking_max=15)
    a1 = safe_normalize(a1, n1)
    update_best(a1, n1)
    print("  Nesterov done: score=%.8f (%.1fs)" % (score_of(a1), time.monotonic() - t0))

    # --- Phase 2: L-BFGS refinement ---
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("\n=== Phase 2: L-BFGS refinement (%.0fs) ===" % (remaining * 0.30))
        lbfgs_time = remaining * 0.30
        t0 = time.monotonic()
        a2, sc2 = lbfgs_smoothed(global_a, global_n, np.sqrt(2.0 * float(global_n)),
                                  lbfgs_time, rng,
                                  beta_schedule=[10, 30, 100, 300, 1000, 3000],
                                  maxiter_per_beta=120)
        a2 = safe_normalize(a2, global_n)
        update_best(a2, global_n)
        print("  L-BFGS done: score=%.8f (%.1fs)" % (score_of(a2), time.monotonic() - t0))

    # --- Phase 3: Nesterov again (longer, with different params) ---
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("\n=== Phase 3: Nesterov round 2 (%.0fs) ===" % (remaining * 0.30))
        n3 = global_n
        S3 = np.sqrt(2.0 * float(n3))
        nest_time3 = remaining * 0.30
        t0 = time.monotonic()
        a3, sc3 = nesterov_subgradient(global_a, n3, S3, nest_time3, rng,
                                        top_k_start=5, top_k_end=100,
                                        lr_base=0.8, backtracking_max=15)
        a3 = safe_normalize(a3, n3)
        update_best(a3, n3)
        print("  Nesterov 2 done: score=%.8f (%.1fs)" % (score_of(a3), time.monotonic() - t0))

    # --- Phase 4: Perturbation + short L-BFGS ---
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("\n=== Phase 4: Perturbation + L-BFGS (%.0fs) ===" % remaining)
        n4 = global_n
        S4 = np.sqrt(2.0 * float(n4))
        n_perturb = max(3, int(remaining / 60))
        per_perturb = (remaining * 0.6) / n_perturb

        for p in range(n_perturb):
            if time.monotonic() > deadline - 30:
                break

            # Different perturbation strategies
            strategy = p % 3
            if strategy == 0:
                # Gaussian noise
                a_pert = global_a + rng.normal(0, 0.05 * S4 / np.sqrt(n4), n4)
            elif strategy == 1:
                # Structured: add sinusoid
                tt = np.linspace(0, 1, n4, endpoint=False)
                freq = rng.integers(2, 50)
                phase = rng.uniform(0, 2 * np.pi)
                a_pert = global_a + 0.08 * S4 / np.sqrt(n4) * np.sin(2 * np.pi * freq * tt + phase)
            else:
                # Random subset perturbation
                mask = rng.random(n4) < 0.1
                a_pert = global_a.copy()
                a_pert[mask] += rng.normal(0, 0.2 * S4 / np.sqrt(n4), mask.sum())

            a_pert = np.maximum(a_pert, 0.0)
            a_pert = safe_normalize(a_pert, n4)

            # Short L-BFGS
            a_new, _ = lbfgs_smoothed(a_pert, n4, S4, per_perturb, rng,
                                       beta_schedule=[20, 100, 500, 2000],
                                       maxiter_per_beta=80)
            a_new = safe_normalize(a_new, n4)
            update_best(a_new, n4)
            if p % 2 == 0:
                print("  Perturb %d: score=%.8f" % (p, score_of(a_new)))

    # --- Phase 5: Try different n if time allows ---
    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Try a different n
        alt_n = 16384 if global_n != 16384 else 4096
        print("\n=== Phase 5: Alt n=%d (%.0fs) ===" % (alt_n, remaining))
        S_alt = np.sqrt(2.0 * float(alt_n))

        # Initialize by interpolating/repeating the best sequence
        if global_a is not None:
            # Resample global_a to new length
            from numpy import interp
            old_n = len(global_a)
            new_indices = np.linspace(0, old_n - 1, alt_n)
            a_alt_init = interp(new_indices, np.arange(old_n), global_a)
        else:
            a_alt_init = rng.random(alt_n) + 0.1

        a_alt_init = safe_normalize(a_alt_init, alt_n)
        update_best(a_alt_init, alt_n)

        t0 = time.monotonic()
        # Quick Nesterov
        alt_time = (deadline - time.monotonic()) - 20
        a_alt, _ = nesterov_subgradient(a_alt_init, alt_n, S_alt, alt_time * 0.5, rng,
                                         top_k_start=5, top_k_end=50,
                                         lr_base=1.0, backtracking_max=12)
        a_alt = safe_normalize(a_alt, alt_n)
        update_best(a_alt, alt_n)

        # Short L-BFGS
        remaining_alt = deadline - time.monotonic() - 10
        if remaining_alt > 20:
            a_alt2, _ = lbfgs_smoothed(a_alt, alt_n, S_alt, remaining_alt, rng,
                                        beta_schedule=[20, 100, 500, 2000],
                                        maxiter_per_beta=100)
            a_alt2 = safe_normalize(a_alt2, alt_n)
            update_best(a_alt2, alt_n)

        print("  Alt n=%d: score=%.8f (%.1fs)" % (alt_n, score_of(a_alt), time.monotonic() - t0))

    # --- Phase 6: Final polishing ---
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        print("\n=== Phase 6: Final polish (%.0fs) ===" % remaining)
        n6 = global_n
        S6 = np.sqrt(2.0 * float(n6))

        # Very short Nesterov with small top-k
        a_polished, _ = nesterov_subgradient(global_a, n6, S6, remaining * 0.5, rng,
                                              top_k_start=2, top_k_end=20,
                                              lr_base=0.5, backtracking_max=10)
        a_polished = safe_normalize(a_polished, n6)
        update_best(a_polished, n6)

        # Final L-BFGS with high beta
        rem2 = deadline - time.monotonic() - 5
        if rem2 > 15:
            a_final, _ = lbfgs_smoothed(global_a, n6, S6, rem2, rng,
                                         beta_schedule=[100, 500, 2000, 5000, 10000],
                                         maxiter_per_beta=100)
            a_final = safe_normalize(a_final, n6)
            update_best(a_final, n6)

        print("  Final: score=%.8f" % global_best)

    # --- Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\n=== Verification ===")
        print("Computed score: %.8f" % global_best)
        print("evaluate_sequence: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  Fallback to uniform sequence")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("\nReturning sequence of length %d" % len(global_a))
    return [float(x) for x in global_a]
