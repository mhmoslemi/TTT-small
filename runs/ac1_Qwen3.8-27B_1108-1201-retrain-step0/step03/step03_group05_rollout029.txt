"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- Spectral shaping (Gerchberg-Saxton) for fast coarse optimization
- L-BFGS with log-sum-exp continuation for fine polishing
- Palindromic constraint option to reduce effective dimension
- Nesterov-accelerated subgradient for final touches
- Multiple initializations and restarts
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
        theta = (cumsum[rho-1] - S) / rho
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def score_of(a):
    """Compute the objective value for a normalized sequence (sum=sqrt(2n))."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def spectral_shaping(a_init, n, S, max_iters, gamma=0.5, rng=None):
    """
    Gerchberg-Saxton style spectral shaping.
    Alternates between:
    - Time domain: project to {x >= 0, sum = S}
    - Frequency domain: flatten |A(k)|^2 toward a target level
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    
    fsize = 1 << (2*n - 2).bit_length()
    best_score = score_of(a)
    best_a = a.copy()
    
    for it in range(max_iters):
        # Frequency domain step: flatten power spectrum
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        target = power.mean()
        
        # Reshape: scale each frequency to reduce peaks in power spectrum
        # This pushes energy from high-power frequencies to low-power ones
        scale = np.sqrt(target / (power + 1e-30))
        # Limit the dynamic range of scaling
        scale = np.clip(scale, 0.1, 10.0)
        # Blend with identity (relaxation)
        scale = 1.0 + gamma * (scale - 1.0)
        
        A_new = A * scale
        a_time = np.real(ifft(A_new, fsize))[:n]
        
        # Time domain step: project to non-negative simplex
        a_time = np.maximum(a_time, 0.0)
        s2 = a_time.sum()
        if s2 < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_time * S / s2
        
        # Track best
        if it % 100 == 0:
            sc = score_of(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = score_of(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    # grad_a[j] = 2 * sum_k w[k] * a[|k-j|] (cross-correlation)
    # Using FFT: grad_a = 2 * IFFT(FFT(w) * conj(FFT(a)))
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A_fft = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A_fft)))[:n]

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs(a_init, n, S, max_time, beta_schedule, maxiter_per_beta=200):
    """L-BFGS-B with continuation on beta."""
    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30) - np.log(S)  # log(a) - log(S) so that sum(softmax(z))*S = sum(a)

    start = time.monotonic()
    best_score = score_of(a)
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        def fun(z_val):
            return softmax_obj_grad(z_val, beta, S)[0]

        def jac(z_val):
            return softmax_obj_grad(z_val, beta, S)[1]

        result = scipy_minimize(
            fun, z, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter_per_beta, 'maxfun': maxiter_per_beta * 2,
                     'ftol': 1e-14, 'gtol': 1e-9, 'iprint': -1}
        )
        z = result.x - result.x.mean()

        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = S * exp_z / exp_z.sum()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * S / s

        c = fft_convolve_self(a)
        sc = c.max() * 2 * n / (a.sum() ** 2)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def nesterov_toppk(a_init, n, S, max_time, top_k=64):
    """Nesterov-accelerated top-k subgradient."""
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = score_of(a)
    best_local = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(y)
        cur_score = 2 * n * c.max() / (y.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_local = y.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A_fft = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A_fft)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.001 * (int((time.monotonic()-start)*100))))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    sc = score_of(y)
    if sc < best_score:
        best_score = sc
        best_local = y.copy()

    return best_local, best_score

def proximal_gradient_smoothed(a, n, S, max_time, beta, initial_alpha=1.0):
    a = a.copy()
    alpha = initial_alpha
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        w = exp_bc / exp_bc.sum()
        
        # Gradient
        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Backtracking line search
        for _ in range(20):
            a_trial = project_simplex(a - alpha * grad, S)
            # Check sufficient decrease (Armijo condition)
            # f(a_trial) < f(a) - 0.5 * alpha * ||grad||^2
            obj_current = (c_max + np.log(exp_bc.sum())) / beta
            c_trial = fft_convolve_self(a_trial)
            obj_trial = (c_trial.max() + np.log(np.exp(beta*(c_trial - c_trial.max())).sum())) / beta

            alpha *= 0.5
        
        a = a_trial

        if a[m_dec] < delta:
            delta = 0.5 * a[m_dec]
        a[m_dec] -= delta
        a[m_inc] += delta
    return a

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy:
1. Fast spectral shaping (Gerchberg-Saxton) for coarse optimization
2. L-BFGS with log-sum-exp continuation (high beta) for fine optimization
3. Nesterov-accelerated subgradient with cooling schedule for final polish
4. Palindromic constraint variant to explore different basins
5. Multiple sequence lengths and perturbation restarts
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def proj_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_c = (cs - S) / idx
    valid = theta_c <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cs[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def norm_seq(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * n)
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def get_score(a):
    n = len(a)
    c = fft_conv_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def spec_shape(a0, n, S, n_iter, gamma=0.7):
    a = a0.copy()
    fsize = 1 << (2 * n - 2).bit_length()
    best_sc = get_score(a)
    best_a = a.copy()
    for it in range(n_iter):
        A = fft(a, fsize)
        pw = np.abs(A) ** 2
        tgt = pw.mean()
        ratio = np.sqrt(tgt / (pw + 1e-30))
        ratio = np.clip(ratio, 0.05, 20.0)
        ratio = 1.0 + gamma * (ratio - 1.0)
        A2 = A * ratio
        at = np.real(ifft(A2, fsize))[:n]
        at = np.maximum(at, 0.0)
        s2 = at.sum()
        if s2 < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = at * S / s2
        if it % 500 == 0:
            sc = get_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    sc = get_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def sft_obj(z, beta, S):
    n = len(z)
    zs = z - z.max()
    ez = np.exp(zs)
    a = S * ez / ez.sum()
    c = fft_conv_self(a)
    cm = c.max()
    ebc = np.exp(beta * (c - cm))
    sebc = ebc.sum()
    obj = (cm + np.log(sebc)) / beta
    return obj, a, c, ebc, sebc


def sft_grad(z, beta, S):
    obj, a, c, ebc, sebc = sft_obj(z, beta, S)
    w = ebc / sebc
    fsize = 1 << (len(w) + len(a) - 2).bit_length()
    W = fft(w, fsize)
    Af = fft(a, fsize)
    ga = 2.0 * np.real(ifft(W * np.conj(Af)))[:len(a)]
    dz = a * (ga - np.dot(ga, a) / S)
    return obj, dz


def run_lbfgs(a0, n, S, max_time, betas, maxit=200):
    a = np.clip(a0, 1e-15, 1e6)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    t0 = time.monotonic()
    best_sc = get_score(a)
    best_a = a.copy()

    for beta in betas:
        if time.monotonic() - t0 > max_time - 5:
            break

        def fun(zv):
            return sft_obj(zv, beta, S)[0]

        def jac(zv):
            return sft_grad(zv, beta, S)[1]

        res = scipy_minimize(fun, z, jac=jac, method='L-BFGS-B',
                             options={'maxiter': maxit, 'maxfun': maxit * 3,
                                      'ftol': 1e-15, 'gtol': 1e-10, 'iprint': -1})
        z = res.x
        z = z - z.mean()

        zs = z - z.max()
        ez = np.exp(zs)
        a = S * ez / ez.sum()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * S / s
        sc = get_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def run_nesterov(a0, n, S, max_time, k_start=16, k_end=512, ramp_frac=0.3):
    a = a0.copy()
    y = a.copy()
    t = 1.0
    t0 = time.monotonic()
    best_sc = get_score(a)
    best_a = a.copy()
    it = 0

    while time.monotonic() - t0 < max_time:
        it += 1
        # Cooling schedule: start at k_start, ramp up to k_end, then cool back down
        frac = min(1.0, it / max(1, int(max_time * 100)))  # rough iteration fraction
        if it < max(10, int(100)):  # warmup
            k = k_start
        else:
            # Use time-based cooling
            elapsed = time.monotonic() - t0
            tf = elapsed / max_time
            if tf < 0.3:
                k = int(k_start + (k_end - k_start) * tf / 0.3)
            elif tf < 0.7:
                k = k_end
            else:
                k = max(k_start, int(k_end - (k_end - k_start) * (tf - 0.7) / 0.3))

        c = fft_conv_self(y)
        sc = 2 * n * c.max() / (y.sum() ** 2)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        top_idx = np.argsort(c)[::-1][:k]
        top_val = c[top_idx]
        tot = top_val.sum()
        if tot < 1e-30:
            break
        wts = top_val / tot

        wf = np.zeros(2 * n - 1)
        for i in range(k):
            wf[int(top_idx[i])] = wts[i]

        fsize = 1 << (len(wf) + n - 2).bit_length()
        W = fft(wf, fsize)
        Af = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Af)))[:n]

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        xn = proj_simplex(y - step * grad / max(gn, 1e-15), S)

        tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = xn + ((t - 1.0) / tn) * (xn - a)
        y = np.maximum(y, 0.0)
        a = xn
        t = tn

    sc = get_score(y)
    if sc < best_sc:
        best_sc = sc
        best_a = y.copy()
    return best_a, best_sc


def run_palindromic(a0, n, S, max_time, betas, maxit=200):
    """Optimize with palindromic constraint: a[i] = a[n-1-i]."""
    half = (n + 1) // 2
    # Extract the "free" half
    if n % 2 == 0:
        h = a0[:half].copy()
    else:
        h = a0[:half].copy()

    # Build full from half
    def build_full(h_val):
        full = np.zeros(n)
        full[:half] = h_val
        full[half:] = h_val[:n - half][::-1]
        return full

    a_full = build_full(h)
    s = a_full.sum()
    if s < 1e-15:
        h = np.ones(half) * S / n
    else:
        h = h * S / s

    t0 = time.monotonic()
    best_sc = get_score(build_full(h))
    best_h = h.copy()
    z = np.log(h / (S / n) + 1e-30)

    beta_S = S  # Total sum for full sequence

    for beta in betas:
        if time.monotonic() - t0 > max_time - 5:
            break

        def fun(zv):
            h_val = np.exp(zv - zv.max())
            h_val = h_val / h_val.sum() * S / n  # normalize so full sum = S
            full = build_full(h_val)
            c = fft_conv_self(full)
            cm = c.max()
            ebc = np.exp(beta * (c - cm))
            return (cm + np.log(ebc.sum())) / beta

        def jac(zv):
            h_val = np.exp(zv - zv.max())
            h_sum = h_val.sum()
            h_val = h_val / h_sum * S / n
            full = build_full(h_val)
            c = fft_conv_self(full)
            cm = c.max()
            ebc = np.exp(beta * (c - cm))
            sebc = ebc.sum()
            w = ebc / sebc

            # Gradient w.r.t. full
            fsize = 1 << (len(w) + n - 2).bit_length()
            Wf = fft(w, fsize)
            Af = fft(full, fsize)
            grad_full = 2.0 * np.real(ifft(Wf * np.conj(Af)))[:n]

            # Map back to half
            grad_h = grad_full[:half].copy()
            if n % 2 == 0:
                grad_h += grad_full[half:][:half][::-1]
            else:
                grad_h[half - 1] += 0  # center element only appears once
                grad_h += grad_full[half:][:half - 1][::-1]

            # Chain rule through softmax
            obj = (cm + np.log(sebc)) / beta
            dot = np.dot(grad_h, h_val)
            grad_z = h_val * (grad_h - dot / (S / n))

            return grad_z

        res = scipy_minimize(fun, z, jac=jac, method='L-BFGS-B',
                             options={'maxiter': maxit, 'maxfun': maxit * 3,
                                      'ftol': 1e-15, 'gtol': 1e-10, 'iprint': -1})
        z = res.x - res.x.mean()

        h_val = np.exp(z - z.max())
        h_val = np.maximum(h_val, 0)
        hs = h_val.sum()
        if hs < 1e-15:
            h_val = np.ones(half) * S / n
        else:
            h_val = h_val * S / (n * hs)  # so that full sum = S
        # Actually, full sum = sum(h) + sum(h reversed) = 2*sum(h) (for even n)
        # or 2*sum(h) - h[center] (for odd n)
        # Let me just build and normalize
        full = build_full(h_val)
        fs = full.sum()
        if fs > 1e-15:
            full = full * S / fs
        h_val = full[:half].copy()

        sc = get_score(full)
        if sc < best_sc:
            best_sc = sc
            best_h = h_val.copy()

    full_best = build_full(best_h)
    full_best = full_best * S / full_best.sum()
    return full_best, best_sc
