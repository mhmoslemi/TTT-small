"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using
hybrid FISTA + coordinate descent optimization.

Strategy: 
1. FISTA with LSE smoothing for initial optimization
2. Coordinate descent for precise polishing (exploiting the 1D structure)
3. Multi-scale and multi-length exploration

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize_scalar


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag (one-hot)."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_rate=0.004, beta_max=1000.0):
    """FISTA with LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)

        if it - last_improve_it > 3000:
            step *= 0.5
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve_it = it

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best

    best_obj = 2.0 * n * conv.max() / (sum_a ** 2)
    best_a = a.copy()
    
    start = time.monotonic()
    pass_count = 0
    no_improve_passes = 0

        pass_count += 1
        pass_start_improve = best_obj
        n_updated = 0
        
        for i in range(n):
            new_x = coord_descent_STEP(a, i, conv, sum_a, n)
            if abs(new_x - a[i]) > 1e-12:
                # Update a[i]
                old = a[i]
                a[i] = new_x
                sum_a = sum_a - old + new_x

                # conv[k] changes by:

                # Actually: the total change at k is:

                # But wait, a[k-i] might have changed... no, we only changed a[i].
                # And a[k-i] is the current value (which hasn't changed since we're doing CD).
                
                lo = max(0, i)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using
hybrid FISTA + coordinate descent optimization.

Key innovation: Cyclic coordinate descent exploiting the 1D structure
of the problem (each coordinate affects the convolution via a simple
linear/quadratic update), combined with FISTA-LSE for global exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize_scalar


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_rate=0.004, beta_max=1500.0):
    """FISTA with LSE smoothing and periodic exact-subgradient mixing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        # Mix LSE and exact subgradient
        if it > 200 and rng.random() < min(0.3, 0.0005 * it):
            k = int(np.argmax(c))
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            grad = np.zeros(n)
            idx = np.arange(lo, hi + 1)
            grad[idx] = 2.0 * y[k - idx]
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            step = S / (gnorm * (1.0 + 0.01 * np.log1p(it / 100.0)))
        else:
            grad = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            step = 1.0 / (2.0 * beta)

        if it - last_improve_it > 2000:
            step *= 0.5
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve_it = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def eval_1d_objective(x, old, conv_shifted, a_full, quad_val, rest_max, sum_a, n):
    """Evaluate the 1D objective for coordinate descent.
    
    x: candidate value for a[i]
    old: current value of a[i]
    conv_shifted: conv[i:n+i] (the affected region)
    a_full: the full a array (length n)
    quad_val: conv[2*i] - old^2 (the constant part of the quadratic term)
    rest_max: max of conv outside [i, n+i-1]
    sum_a: current sum of a
    n: sequence length
    """
    new_sum = sum_a - old + x
    if new_sum < 1e-15:
        return 1e30

    dx = x - old
    # Linear contribution to the shifted region
    shifted_new = conv_shifted + 2.0 * dx * a_full
    shifted_max = np.max(shifted_new)

    # Quadratic correction at position 2i (which is index i in the shifted array)
    quad_term = quad_val + x * x
    base_max = max(shifted_max, quad_term, rest_max)

    return 2.0 * n * base_max / (new_sum * new_sum)


def coord_descent_pass(a, conv, sum_a, n, max_time_step):
    """One full cyclic coordinate descent pass. Returns updated a, conv, sum_a."""
    orig = a.copy()
    start = time.monotonic()

    for i in range(n):
        # Check time
        if time.monotonic() - start > max_time_step:
            break

        old = a[i]
        if old < 1e-15 and a[i] < 1e-15:
            # Skip very small coordinates (but still try)
            pass

        conv_shifted = conv[i:n + i]
        i_in_shifted = i  # position 2i in conv is at index 2i-i = i in conv_shifted
        quad_val = conv[2 * i] - old * old  # constant part: conv[2i] - old^2, then we add x^2

        # Rest max: max of conv outside [i, n+i-1]
        if i > 0 and (2 * n - 1 - (n + i)) > 0:
            rest_max = max(np.max(conv[:i]), np.max(conv[n + i:2 * n - 1]))
        elif i > 0:
            rest_max = np.max(conv[:i])
        elif 2 * n - 1 - (n + i) > 0:
            rest_max = np.max(conv[n + i:2 * n - 1])
        else:
            rest_max = 0.0

        # Upper bound for x
        upper = max(2.0 * old + 0.1, sum_a / n * 3.0)

        # Try a few candidate points first
        candidates = [0.0, old * 0.3, old * 0.7, old, old * 1.3, old * 2.0, sum_a / n]
        best_f = 2.0 * n * conv.max() / (sum_a * sum_a)  # current objective
        best_x = old

        for cx in candidates:
            cx = max(0.0, min(upper, cx))
            f = eval_1d_objective(cx, old, conv_shifted, orig, quad_val, rest_max, sum_a, n)
            if f < best_f:
                best_f = f
                best_x = cx

        # Refine with golden section if we found an improvement
        if best_x != old:
            # Search in a neighborhood
            lo = max(0.0, best_x * 0.5)
            hi = upper
            try:
                res = minimize_scalar(
                    lambda x: eval_1d_objective(x, old, conv_shifted, orig, quad_val, rest_max, sum_a, n),
                    bounds=(lo, hi), method='bounded',
                    options={'xatol': max(old * 1e-10, 1e-14), 'maxiter': 60}
                )
                if res.success and res.fun < best_f:
                    best_x = res.x
                    best_f = res.fun
            except Exception:
                pass

        # Update a[i] and conv
        new_x = best_x
        dx = new_x - old
        if abs(dx) > 1e-14:
            a[i] = new_x
            sum_a = sum_a + dx
            # Update conv
            conv[i:n + i] += 2.0 * dx * a
            # Correction at 2i: we added 2*dx*a[i]=2*dx*new_x, but need new_x^2 - old^2
            # Extra: (new_x^2 - old^2) - 2*dx*new_x = -dx^2
            conv[2 * i] -= dx * dx

    return a, conv, sum_a


def coordinate_descent(a_init, n, S, max_time, rng):
    """Cyclic coordinate descent with restarts."""
    a = normalize_sum(a_init, n).copy()
    sum_a = S
    conv = fft_convolve_self(a)
    best_obj = 2.0 * n * conv.max() / (sum_a * sum_a)
    best_a = a.copy()

    start = time.monotonic()
    pass_count = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        pass_count += 1
        remaining = max_time - (time.monotonic() - start)
        # Estimate time per pass (first pass is slower due to no warm cache)
        if pass_count <= 3:
            time_step = remaining * 0.5
        else:
            time_step = remaining * 0.4

        a, conv, sum_a = coord_descent_pass(a, conv, sum_a, n, time_step)

        cur_obj = 2.0 * n * conv.max() / (sum_a * sum_a)
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if pass_count % 3 == 0:
            print("  CD pass %d: obj=%.8f, best=%.8f, time=%.0fs" % (
                pass_count, cur_obj, best_obj, time.monotonic() - start))

        if no_improve >= 4:
            # Perturb to escape local minimum
            noise_scale = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            sum_a = S
            conv = fft_convolve_self(a)
            no_improve = 0

        # If we've done many passes without improvement, reduce perturbation
        if pass_count > 30 and no_improve >= 3:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            sum_a = S
            conv = fft_convolve_self(a)

    return best_a, best_obj

    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * 3 * tt + 0.5))

    # Sinc-like
    center = (n - 1) / 2.0
    sigma = n / 4.0
    inits.append(np.exp(-(tt * n - center) ** 2 / (2 * sigma * sigma)))

    # U-shaped (high at edges)
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * tt))
    inits.append((1.0 - tt * 2.0) ** 2)

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.3, n)))

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Try to get the provided sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose main length
    if given is not None and 1024 <= len(given) <= 32768:
        main_len = len(given)
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * n)

    print("Starting with n=%d, budget=%.0fs" % (n, budget_s))

    # Phase 1: FISTA exploration (40% of time)
    fista_time = max(30.0, (deadline - time.monotonic()) * 0.40)
    print("=== Phase 1: FISTA exploration (%.0fs) ===" % fista_time)

    inits = []
    if given is not None:
        inits.append(given)

    gen_inits = generate_diverse_inits(n, rng)
    inits.extend(gen_inits[:6])  # Add a subset of generated inits

    # Distribute time among initializations (give 'given' more time)
    n_inits = len(inits)
    weights = np.ones(n_inits)
    if given is not None:
        weights[0] = 2.5
    weights /= weights.sum()

    for i, init in enumerate(inits):
        if time.monotonic() > deadline - (deadline - time.monotonic()) * 0.55:
            break
        t_alloc = fista_time * weights[i] * 0.9
        if t_alloc < 5:
            continue

        a_opt, score = run_fista(init, n, S, t_alloc, rng,
                                  beta_init=5.0, beta_rate=0.004, beta_max=1500.0)
        a_opt = normalize_sum(a_opt, n)
        c = fft_convolve_self(a_opt)
        obj = 2.0 * n * c.max() / (a_opt.sum() ** 2)

        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
            print("  FISTA init %d: obj=%.8f (time %.0fs)" % (i, global_best, time.monotonic() - (deadline - fista_time)))

    print("After FISTA: best=%.8f" % global_best)

    # Phase 2: Coordinate descent polishing (45% of time)
    cd_time = max(30.0, (deadline - time.monotonic()) * 0.45)
    print("=== Phase 2: Coordinate descent (%.0fs) ===" % cd_time)

    if global_a is not None:
        a_cd, obj_cd = coordinate_descent(global_a, n, S, cd_time, rng)
        a_cd = normalize_sum(a_cd, n)
        c = fft_convolve_self(a_cd)
        obj = 2.0 * n * c.max() / (a_cd.sum() ** 2)
        if obj < global_best:
            global_best = obj
            global_a = a_cd.copy()
            print("  After CD: obj=%.8f" % global_best)

    # Phase 3: Final FISTA polish + another CD (15% of time)
    remaining = deadline - time.monotonic()
    if remaining > 10 and global_a is not None:
        print("=== Phase 3: Final polish (%.0fs) ===" % remaining)

        # High-beta FISTA
        t_fista2 = remaining * 0.4
        a_f2, _ = run_fista(global_a, n, S, t_fista2, rng,
                             beta_init=50.0, beta_rate=0.002, beta_max=5000.0)
        a_f2 = normalize_sum(a_f2, n)
        c = fft_convolve_self(a_f2)
        obj = 2.0 * n * c.max() / (a_f2.sum() ** 2)
        if obj < global_best:
            global_best = obj
            global_a = a_f2.copy()

        # Final CD
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a_cd2, obj_cd2 = coordinate_descent(global_a, n, S, remaining2 * 0.8, rng)
            a_cd2 = normalize_sum(a_cd2, n)
            c = fft_convolve_self(a_cd2)
            obj = 2.0 * n * c.max() / (a_cd2.sum() ** 2)
            if obj < global_best:
                global_best = obj
                global_a = a_cd2.copy()
                print("  After final CD: obj=%.8f" % global_best)

    # Phase 4: Try different length if time permits
    remaining = deadline - time.monotonic()
    if remaining > 60:
        alt_n = n * 2 if n < 16384 else n // 2
        if 1024 <= alt_n <= 32768 and alt_n != n:
            S_alt = np.sqrt(2.0 * alt_n)
            print("=== Phase 4: Trying n=%d (%.0fs) ===" % (alt_n, remaining))

            # Use downsampled version of best as init
            if global_a is not None:
                t_old = np.linspace(0, 1, len(global_a))
                t_new = np.linspace(0, 1, alt_n)
                alt_init = np.interp(t_new, t_old, global_a)
            else:
                alt_init = np.ones(alt_n)

            a_alt, _ = run_fista(alt_init, alt_n, S_alt, remaining * 0.4, rng,
                                   beta_init=10.0, beta_rate=0.005, beta_max=2000.0)
            a_alt = normalize_sum(a_alt, alt_n)
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_alt2, _ = coordinate_descent(a_alt, alt_n, S_alt, remaining2 * 0.7, rng)
                a_alt2 = normalize_sum(a_alt2, alt_n)
                c = fft_convolve_self(a_alt2)
                obj = 2.0 * alt_n * c.max() / (a_alt2.sum() ** 2)
                if obj < global_best:
                    global_best = obj
                    global_a = a_alt2.copy()
                    print("  Alt n=%d: obj=%.8f" % (alt_n, global_best))

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        assert len(final_list) > 0
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.8f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            global_best = evaluate_sequence([float(x) for x in global_a])

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("Returning sequence of length %d, score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
