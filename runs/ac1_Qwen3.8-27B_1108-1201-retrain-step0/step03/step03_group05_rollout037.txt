def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_cross_corr(a, w):
    """Compute 2 * cross-correlation of a with w using FFT."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """
    Compute smoothed objective and gradient w.r.t. z (unconstrained softmax params).
    a = S * softmax(z), objective = (1/beta)*log(sum exp(beta*c_k))

                break
            if attempt > 0:
                z = z + rng.normal(0, 0.03 / (1 + 0.5 * i_beta), n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            t0 = time.monotonic()
            try:
                result = scipy_minimize(
                    fun, z,
                    jac=jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_beta,
                        'maxfun': maxiter_per_beta * 3,
                        'ftol': 1e-15,
                        'gtol': 1e-10,
                        'maxls': 50,
                        'iprint': -1
                    }
                )
            except Exception:
                break

            elapsed_beta += time.monotonic() - t0
            z = result.x
            z = z - z.mean()

            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

        if elapsed_beta > 0:
            print("    beta=%5d: score=%.8f (%d att, %.1fs)" %
                  (beta, best_score, n_attempts, elapsed_beta))

    return best_a, best_score


def spectral_rebalance(a, n, S, rng, n_iter=20, strength=0.3):
    """
    Spectral domain rebalancing: compress dynamic range of power spectrum,
    project back to non-negative. Acts as a global escape mechanism.
    """
    a = np.clip(a, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a * (S / s)

    best_a = a.copy()
    best_score = fft_convolve_self(a).max()

    for it in range(n_iter):
        A = fft(a)
        power = np.abs(A) ** 2

        # Compress dynamic range: power_new = power^alpha / mean(power^alpha) * S^2/n
        # alpha < 1 compresses, alpha > 1 expands
        alpha = 1.0 - strength * 0.5  # slight compression
        power_clipped = np.maximum(power, 1e-30)
        power_new = power_clipped ** alpha
        # Normalize to maintain A[0] = S
        power_new = power_new * (S ** 2 / power_new[0])

        # Reconstruct with original phase
        phase = np.angle(A)
        A_new = np.sqrt(power_new) * np.exp(1j * phase)
        A_new[0] = S  # enforce DC component

        a_new = np.real(ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * (S / s_new)

        c = fft_convolve_self(a_new)
        score = c.max()

        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        # Blend back towards original to avoid losing too much
        a = 0.7 * a_new + 0.3 * a
        a = np.maximum(a, 0.0)
        a = a * (S / a.sum())

    return best_a


def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=512):
    """Nesterov-accelerated top-k subgradient with FFT gradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(y).max()
    best_local = y.copy()
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            # Restart with perturbation
            y = project_simplex(y + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            t = 1.0
            no_improve = 0

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        grad = compute_cross_corr(a, w_full)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 2000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def pairwise_cd_polish(a, n, S, max_time, rng, n_pairs_per_pass=200):
    """
    Pairwise coordinate descent: pick pairs (j,k), optimize the split
    a[j]+a[k]=s to minimize the peak autocorrelation.
    """
    a = np.clip(a, 0.0, 1000.0).copy().astype(np.float64)
    s_total = a.sum()
    if s_total < 1e-15:
        return a
    a = a * (S / s_total)

    start = time.monotonic()
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()

    def peak_with_change(a, j, k, delta):
        """Compute peak of autocorrelation after changing a[j]+=delta, a[k]-=delta."""
        # Change in c[m] = delta * (a[m-j] - a[m-k]) for valid m
        # We need max over m of (c[m] + delta * d[m]) where d[m] = a[m-j]-a[m-k]
        # This is a convex piecewise-linear function of delta.
        # We want to find delta in [-a[j], a[k]] that minimizes the peak.
        # Use binary search on the subgradient.
        c = fft_convolve_self(a)
        # d[m] for m in [0, 2n-2]
        d = np.zeros(2 * n - 1)
        # a[m-j] is valid when 0 <= m-j < n, i.e., j <= m < j+n
        for m in range(j, min(j + n, 2 * n - 1)):
            d[m] += a[m - j]
        for m in range(k, min(k + n, 2 * n - 1)):
            d[m] -= a[m - k]

        lo = -a[j] + 1e-15
        hi = a[k] - 1e-15
        if lo >= hi:
            return delta, c.max()

        # Binary search: find where subgradient crosses 0
        # Subgradient at delta = max of {d[m] + (c[m]+delta*d[m] - f(delta))/f(delta) ... }
        # Simpler: just do a fine grid search (fast enough for polishing)
        for _ in range(30):
            mid = (lo + hi) / 2
            # Compute f(mid) and subgradient
            c_new = c + mid * d
            f_val = c_new.max()
            active = np.where(c_new >= f_val - 1e-12)[0]
            subgrad = d[active].mean()
            if subgrad > 0:
                hi = mid
            else:
                lo = mid

        delta_opt = (lo + hi) / 2
        return delta_opt, (c + delta_opt * d).max()

    # Precompute c once per pass
    while time.monotonic() - start < max_time:
        n_pairs = min(n_pairs_per_pass, max(50, n // 10))
        pairs = []
        for _ in range(n_pairs):
            j = rng.integers(0, n)
            k = rng.integers(0, n)
            if j != k:
                pairs.append((int(j), int(k)))

        for j, k in pairs:
            if time.monotonic() - start > max_time - 1:
                break
            delta, new_peak = peak_with_change(a, j, k, 0.0)
            if new_peak < fft_convolve_self(a).max() - 1e-10:
                a[j] += delta
                a[k] -= delta
                a[j] = max(a[j], 0.0)
                a[k] = max(a[k], 0.0)

        # Renormalize
        a = safe_normalize(a, n)
        c_check = fft_convolve_self(a)
        if c_check.max() < best_score:
            best_score = c_check.max()
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: L-BFGS continuation, spectral rebalancing, Nesterov subgradient,
    pairwise CD, and aggressive high-beta polishing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # --- Get initial sequence ---
    inits = []
    given_arr = None

    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given_arr = np.clip(given, 0.0, 1000.0)
            inits.append(("given", given_arr.copy()))
    except Exception:
        pass

    if inits:
        n = len(inits[0][1])
    else:
        n = 8192

    S = np.sqrt(2.0 * float(n))
    tt = np.linspace(0, 1, n, endpoint=False)

    # Add diverse initializations
    inits.append(("rand", rng.random(n) + 0.1))
    inits.append(("sin_multi", 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) +
                  0.25 * np.cos(2 * np.pi * 7 * tt) +
                  0.2 * np.sin(2 * np.pi * 13 * tt) +
                  0.15 * np.cos(2 * np.pi * 29 * tt) +
                  0.1 * np.sin(2 * np.pi * 41 * tt)))
    inits.append(("gauss_bumps", np.exp(-20.0 * (tt - 0.2) ** 2) +
                  np.exp(-20.0 * (tt - 0.5) ** 2) +
                  np.exp(-20.0 * (tt - 0.8) ** 2) + 0.05))
    inits.append(("cos_envelope", np.abs(np.cos(2 * np.pi * 5 * tt)) ** 1.5 + 0.1))
    inits.append(("pow_decay", np.exp(-3.0 * tt) + 0.3 * np.exp(-3.0 * (1 - tt)) + 0.05))

    n_inits = len(inits)
    print("=== Optimization: n=%d, %d inits, budget=%.0fs ===" % (n, n_inits, budget_s))
    print("Target: 1.5027, Previous best: 1.508805")

    # --- Phase 1: L-BFGS continuation (main workhorse, ~50% budget) ---
    usable = deadline - time.monotonic()
    phase1_time = usable * 0.50

    # Allocate more time to the "given" init if available
    if inits[0][0] == "given":
        per_init_main = phase1_time * 0.5
        per_init_other = (phase1_time - per_init_main) / max(1, n_inits - 1)
    else:
        per_init_main = 0
        per_init_other = phase1_time / n_inits

    print("\n=== Phase 1: L-BFGS Continuation (total %.1f sec) ===" % phase1_time)
    phase1_results = []

    for idx, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - 120:
            break

        t0 = time.monotonic()
        a_norm = safe_normalize(arr, n)
        c0 = fft_convolve_self(a_norm)
        init_score = c0.max()

        if init_score < global_best:
            global_best = init_score
            global_a = a_norm.copy()

        if name == "given" and per_init_main > 0:
            alloc = per_init_main
            beta_sched = [3, 5, 10, 20, 50, 100, 200, 400, 800, 1500, 3000, 6000, 10000]
            maxiter = 250
        else:
            alloc = max(per_init_other, 5.0)
            beta_sched = [5, 10, 20, 50, 100, 200, 500, 1000]
            maxiter = 150

        a_new, score = run_lbfgs_continuation(
            a_norm, n, S, alloc, rng,
            beta_schedule=beta_sched, maxiter_per_beta=maxiter
        )
        a_new = safe_normalize(a_new, n)
        c_new = fft_convolve_self(a_new)
        final_score = c_new.max()

        phase1_results.append((name, a_new, final_score))

        if final_score < global_best:
            global_best = final_score
            global_a = a_new.copy()
            print("  [IMPROVE] %s: %.8f (init=%.4f, %.1fs)" %
                  (name, global_best, init_score, time.monotonic() - t0))
        else:
            print("  %-12s: %.8f (best=%.6f, %.1fs)" %
                  (name, final_score, global_best, time.monotonic() - t0))

    print("\n=== After Phase 1: global_best=%.8f ===" % global_best)

    # --- Phase 2: Spectral rebalancing (~15% budget) ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            phase2_time = min(remaining * 0.15, 180)
            print("\n=== Phase 2: Spectral Rebalancing (%.1f sec) ===" % phase2_time)

            n_spec_iters = max(10, int(phase2_time / 5))
            for strength in [0.5, 0.3, 0.2]:
                if time.monotonic() > deadline - 60:
                    break
                a_spec = spectral_rebalance(
                    global_a, n, S, rng,
                    n_iter=n_spec_iters // 3, strength=strength
                )
                a_spec = safe_normalize(a_spec, n)
                c_spec = fft_convolve_self(a_spec)
                if c_spec.max() < global_best:
                    global_best = c_spec.max()
                    global_a = a_spec.copy()
                    print("  Spectral (strength=%.1f): %.8f" % (strength, global_best))

    print("\n=== After Phase 2: global_best=%.8f ===" % global_best)

    # --- Phase 3: Cross-pollination + refined L-BFGS (~20% budget) ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            phase3_time = remaining * 0.20
            print("\n=== Phase 3: Cross-pollination (%.1f sec) ===" % phase3_time)

            # Sort phase 1 results by score
            phase1_results.sort(key=lambda x: x[2])
            n_top = min(3, len(phase1_results))

            # Try blending top solutions
            blend_budget = phase3_time / max(1, 2 * n_top)
            for i in range(n_top):
                if time.monotonic() > deadline - 30:
                    break
                for j in range(i + 1, n_top):
                    if time.monotonic() > deadline - 30:
                        break
                    a1 = phase1_results[i][1]
                    a2 = phase1_results[j][1]
                    for blend in [0.3, 0.5, 0.7]:
                        if time.monotonic() > deadline - 20:
                            break
                        a_blend = blend * a1 + (1 - blend) * a2
                        a_blend = safe_normalize(a_blend, n)
                        c_b = fft_convolve_self(a_blend)
                        if c_b.max() < global_best:
                            global_best = c_b.max()
                            global_a = a_blend.copy()
                            print("  Blend(%.1f) [%s,%s]: %.8f" %
                                  (blend, phase1_results[i][0], phase1_results[j][0], global_best))

            # Refined L-BFGS on best
            if time.monotonic() < deadline - 30:
                rem = deadline - time.monotonic()
                a_ref, score = run_lbfgs_continuation(
                    global_a, n, S, min(rem, phase3_time * 0.5), rng,
                    beta_schedule=[100, 200, 500, 1000, 2000, 5000],
                    maxiter_per_beta=200
                )
                a_ref = safe_normalize(a_ref, n)
                c_ref = fft_convolve_self(a_ref)
                if c_ref.max() < global_best:
                    global_best = c_ref.max()
                    global_a = a_ref.copy()
                    print("  Refined L-BFGS: %.8f" % global_best)

    print("\n=== After Phase 3: global_best=%.8f ===" % global_best)

    # --- Phase 4: Nesterov subgradient polishing (~15% budget) ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            phase4_time = min(remaining * 0.20, 240)
            print("\n=== Phase 4: Nesterov Subgradient (%.1f sec) ===" % phase4_time)

            a_nest, score = run_nesterov_toppk(
                global_a, n, S, phase4_time, rng,
                top_k_init=20, top_k_max=1024
            )
            a_nest = safe_normalize(a_nest, n)
            c_nest = fft_convolve_self(a_nest)
            if c_nest.max() < global_best:
                global_best = c_nest.max()
                global_a = a_nest.copy()
                print("  Nesterov: %.8f" % global_best)

            # Quick L-BFGS polish after Nesterov
            if time.monotonic() < deadline - 15:
                rem2 = deadline - time.monotonic()
                a_pol, sc = run_lbfgs_continuation(
                    global_a, n, S, min(rem2, 60), rng,
                    beta_schedule=[500, 1000, 2000, 5000],
                    maxiter_per_beta=150
                )
                a_pol = safe_normalize(a_pol, n)
                c_pol = fft_convolve_self(a_pol)
                if c_pol.max() < global_best:
                    global_best = c_pol.max()
                    global_a = a_pol.copy()
                    print("  Post-Nesterov L-BFGS: %.8f" % global_best)

    print("\n=== After Phase 4: global_best=%.8f ===" % global_best)

    # --- Phase 5: Pairwise CD + final high-beta L-BFGS (remaining budget) ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print("\n=== Phase 5: Final Polishing (%.1f sec) ===" % remaining)

            # Pairwise CD (short)
            if remaining > 30:
                cd_time = min(remaining * 0.3, 60)
                a_cd = pairwise_cd_polish(
                    global_a, n, S, cd_time, rng,
                    n_pairs_per_pass=300
                )
                a_cd = safe_normalize(a_cd, n)
                c_cd = fft_convolve_self(a_cd)
                if c_cd.max() < global_best:
                    global_best = c_cd.max()
                    global_a = a_cd.copy()
                    print("  Pairwise CD: %.8f" % global_best)

            # Final aggressive L-BFGS
            rem_final = deadline - time.monotonic()
            if rem_final > 10:
                a_fin, sc = run_lbfgs_continuation(
                    global_a, n, S, rem_final, rng,
                    beta_schedule=[1000, 2000, 5000, 10000, 20000],
                    maxiter_per_beta=300
                )
                a_fin = safe_normalize(a_fin, n)
                c_fin = fft_convolve_self(a_fin)
                if c_fin.max() < global_best:
                    global_best = c_fin.max()
                    global_a = a_fin.copy()
                    print("  Final L-BFGS: %.8f" % global_best)

    # --- Phase 6: Random restart perturbation if time remains ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("\n=== Phase 6: Random Restarts (%.1f sec) ===" % remaining)
            n_restarts = max(2, int(remaining / 40))
            for r in range(n_restarts):
                if time.monotonic() > deadline - 10:
                    break
                noise_scale = 0.02 / (1.0 + 0.5 * r)
                a_pert = global_a + rng.normal(0, noise_scale * S / np.sqrt(n), n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = safe_normalize(a_pert, n)

                a_new, sc = run_lbfgs_continuation(
                    a_pert, n, S, 25, rng,
                    beta_schedule=[50, 100, 300, 800, 2000],
                    maxiter_per_beta=120
                )
                a_new = safe_normalize(a_new, n)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < global_best:
                    global_best = c_new.max()
                    global_a = a_new.copy()
                    print("  Restart %d: %.8f" % (r, global_best))

    # --- Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
