    n = len(a)
    N = 1 << (2*n - 1).bit_length()  # next power of 2 >= 2n-1
    a_padded = np.zeros(N)
    a_padded[:n] = a
    
    for it in range(n_iters):
        A = fft(a_padded)
        mag = np.abs(A)
        phase = np.angle(A)
        # Compress magnitude
        mag_new = mag ** alpha  # alpha < 1 compresses
        A_new = mag_new * np.exp(1j * phase)
        a_padded = np.real(ifft(A_new))
        # Enforce constraints: non-negative, zero-padded
        a_padded[:n] = np.maximum(a_padded[:n], 0)
        a_padded[n:] = 0
        # Renormalize
        s = a_padded[:n].sum()
        if s > 0:
            a_padded[:n] = a_padded[:n] * (np.sqrt(2*n) / s)
    
    return a_padded[:n]

    n = len(a)
    N = 1 << (2*n - 1).bit_length()
    
    for it in range(n_iters):

        a_pad = np.zeros(N)
        a_pad[:n] = a
        
        # FFT
        A = fft(a_pad)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)  # unit phase (or 0 where mag=0)
        
        # Flatten magnitude: new_mag = mag^alpha (alpha < 1 compresses)
        mag_new = mag ** alpha
        A_new = mag_new * phase
        
        # IFFT
        a_new = np.real(ifft(A_new))
        
        # Enforce constraints
        a_new = np.maximum(a_new[:n], 0)
        # Renormalize
        S = np.sqrt(2*n)
        s = a_new.sum()
        if s > 0:
            a_new = a_new * S / s
        
        a = a_new
        if it % 100 == 0:
            c = fft_convolve_self(a)
            print("  Fourier flatten iter %d: max_c = %.6f" % (it, c.max()))
    
    return a

import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_value_and_grad(a, beta):
    """
    Compute LSE(beta * conv(a,a)) and its gradient w.r.t. a.
    Returns (lse_val, grad_array).
    """
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    lse_val = c_max + np.log(total) / beta
    w = exp_bc / total
    # Gradient: dLSE/da[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)
    N = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad


def objective_and_grad(a, beta, S_target, lam=10.0):
    """
    Objective: LSE(beta * conv(a,a)) + lam * 0.5 * (sum(a) - S_target)^2
    Returns (obj, grad).
    """
    n = len(a)
    lse_val, grad_lse = lse_value_and_grad(a, beta)
    S = a.sum()
    penalty = lam * 0.5 * (S - S_target) ** 2
    grad_penalty = lam * (S - S_target) * np.ones(n)
    return lse_val + penalty, grad_lse + grad_penalty


def run_lbfgs(a_init, beta, S_target, max_time, maxiter=5000):
    """Run L-BFGS-B for a fixed beta. Returns (best_a, best_obj)."""
    a0 = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    n = len(a0)
    
    best_a = a0.copy()
    S = a0.sum()
    if S < 1e-10:
        a0 = np.ones(n) * S_target / n
    c0 = fft_convolve_self(a0)
    max_c0 = c0.max() / (S * S) if S > 0 else np.inf
    best_max_c = max_c0
    
    start = time.monotonic()
    
    # Callback to check time and track best
    class TimeCatcher(Exception):
        pass
    
    def func_and_grad(x):
        if time.monotonic() - start > max_time:
            raise TimeCatcher()
        obj, grad = objective_and_grad(x, beta, S_target)
        if not np.isfinite(obj):
            return 1e10, np.zeros_like(x)
        return obj, grad
    
    try:
        res = minimize(
            func_and_grad, a0, method='L-BFGS-B',
            jac=True, bounds=[(0.0, 1000.0)] * n,
            options={'maxiter': maxiter, 'maxls': 50, 'ftol': 1e-15, 'gtol': 1e-10, 'maxfun': 50000}
        )
        a_result = res.x
    except TimeCatcher:
        a_result = a0
    except Exception:
        a_result = a0
    
    # Evaluate result

            best_a = a_result.copy()
            best_max_c = max_c_r
    
    return best_a, best_max_c

def fourier_flatten(a_init, max_time, alpha_start=0.85, alpha_end=0.5, n_iters=2000):
    """
    Gerchberg-Saxton style Fourier flattening.
    Alternates between:
    - Time domain: clip to non-negative, zero-pad
    - Fourier domain: compress magnitude spectrum
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)
    N = 1 << (2 * n - 1).bit_length()
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    
    best_a = a.copy()
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S) if S > 0 else np.inf
    
    start = time.monotonic()
    it = 0
    
    while it < n_iters and (time.monotonic() - start) < max_time:
        # Animate alpha from alpha_start to alpha_end
        frac = it / max(1, n_iters)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        
        # Zero-pad
        a_pad = np.zeros(N)
        a_pad[:n] = a
        
        # FFT
        A = fft(a_pad)
        mag = np.abs(A)
        # Avoid division by zero
        mag_safe = np.maximum(mag, 1e-30)
        phase = A / mag_safe
        
        # Compress magnitude
        mag_new = mag ** alpha
        A_new = mag_new * phase
        
        # IFFT
        a_new = np.real(ifft(A_new))
        
        # Enforce non-negativity and zero-padding
        a_new = np.maximum(a_new[:n], 0.0)
        
        # Renormalize
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new * S_target / s
        else:
            a_new = a
        
        a = a_new
        it += 1
        
        # Periodic evaluation
        if it % 50 == 0:
            c = fft_convolve_self(a)
            s = a.sum()
            if s > 1e-10:
                obj = 2.0 * n * c.max() / (s * s)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    print("    Fourier flatten iter %d: obj = %.6f (alpha=%.3f)" % (it, obj, alpha))
    
    return best_a, best_obj


def gen_inits(n, rng):
    """Generate diverse asymmetric initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Exponential ramps
    for alpha in [1.0, 2.0, 3.0, 5.0, 8.0]:
        inits.append(np.exp(alpha * tt))
        inits.append(np.exp(alpha * (1.0 - tt)))
    
    # Power-law
    for p in [0.3, 0.5, 1.0, 2.0, 4.0]:
        inits.append((tt + 0.001) ** p)
        inits.append((1.0 - tt + 0.001) ** p)
    
    # Step functions (asymmetric)
    for frac in [0.2, 0.3, 0.4, 0.6, 0.7]:
        arr = np.ones(n)
        arr[:int(n * frac)] = 3.0
        inits.append(arr)
        arr2 = np.ones(n)
        arr2[int(n * (1 - frac)):] = 3.0
        inits.append(arr2)
    
    # Sinusoidal with trend
    for freq in [2, 3, 5, 7]:
        inits.append(1.0 + 0.5 * np.sin(2 * np.pi * freq * tt) + 0.5 * tt)
        inits.append(1.0 + 0.5 * np.cos(2 * np.pi * freq * tt) + 0.5 * (1 - tt))
    
    # Random Fourier phase (flat spectrum)
    for _ in range(4):
        N = 1 << (2 * n - 1).bit_length()
        A = rng.normal(size=N) + 1j * rng.normal(size=N)
        # Enforce conjugate symmetry for real output
        A[0] = np.abs(A[0])
        if N % 2 == 0:
            A[N // 2] = np.abs(A[N // 2])
        for k in range(1, N // 2 + 1):
            A[N - k] = np.conj(A[k])
        a_f = np.real(ifft(A))[:n]
        a_f = np.maximum(a_f, 0.0)
        s = a_f.sum()
        if s > 0:
            inits.append(a_f)
    
    # Random with various smoothness
    for smooth in [0.01, 0.1, 0.5]:
        a_r = rng.random(n)
        # Simple smoothing via cumulative sum differences
        kernel_size = max(2, int(n * smooth))
        kernel = np.ones(kernel_size) / kernel_size
        a_s = np.convolve(a_r, kernel, mode='same')
        a_s = np.maximum(a_s, 0.0)
        inits.append(a_s * np.linspace(0.5, 1.5, n))  # add asymmetry
    
    return inits


def peak_refine(a_init, max_time, top_k=20):
    """
    Peak-focused refinement: iteratively identify top peaks in convolution,
    compute combined subgradient, take projected steps.
    """
    n = len(a_init)
    S_target = np.sqrt(2.0 * n)

    if S < 1e-10:
        a = np.ones(n) * S_target / n
    else:
        a = a * S_target / S
    
    best_a = a.copy()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S_target * S_target)
    
    start = time.monotonic()
    it = 0
    step_size = S_target / np.sqrt(n)
    
    while (time.monotonic() - start) < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S_target * S_target)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Get top peaks
        k = min(2*n-1, top_k + it // 50)
        top_idx = np.argsort(c)[::-1][:k]
        top_vals = c[top_idx]
        
        # Weight: emphasize the largest peaks
        w = top_vals ** 2
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum
        
        # Build full weight array
        w_full = np.zeros(2 * n - 1)
        for i, idx in enumerate(top_idx):
            w_full[int(idx)] = w[i]
        
        # Gradient: 2 * (w_full cross-correlate a)
        N = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, N)
        A = fft(a, N)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Normalized step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        eff_step = step_size / (1.0 + 0.01 * it)
        a_new = a - eff_step * grad / gnorm
        a_new = project_simplex(a_new, S_target)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max() / (S_target * S_target)
        
        if obj_new > cur_obj * 1.005:
            # Backtrack
            a_new = 0.5 * a + 0.5 * a_new
            a_new = project_simplex(a_new, S_target)
        
        a = a_new
        
        # Periodic noise to escape local minima
        if it % 500 == 0 and rng_global.random() < 0.2:
            noise = rng_global.normal(0, 0.002 * S_target / np.sqrt(n), n)
            a = project_simplex(a + noise, S_target)
    
    return best_a, best_obj

            global_best_a = np.ones(n, dtype=np.float64) * S_target / n
            final_list = [float(x) for x in global_best_a]
            ver = evaluate_sequence(final_list)
            print("  Fallback verification: %.6f" % ver)
    except Exception as e:
        print("  Verification error: %s" % str(e))
    
    print("Returning sequence of length %d, internal best=%.6f, verified=%.6f" %
          (len(final_list), global_best_obj, ver if np.isfinite(ver) else -1))
    return final_list
