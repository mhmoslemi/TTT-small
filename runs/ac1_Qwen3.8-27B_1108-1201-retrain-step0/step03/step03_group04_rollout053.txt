    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_gradient_fft(a, weights):
    """Compute gradient of weighted sum of convolution values w.r.t. a."""
    n = len(a)
    # grad_i = sum_k weights[k] * d(a*a)_k / da_i
    # d(a*a)_k / da_i = a[k-i] + a[k-i] (if k-i != i) or 2*a[i] (if k-i == i)
    # More precisely: d(a*a)_k / da_i = a[k-i] (from term j=i) + a[i'] (from term j=i'=k-i, if i' != i)
    # = a[k-i] + (a[k-i] if k-i != i, a[i] if k-i == i)
    # Actually: (a*a)_k = sum_j a_j * a_{k-j}
    # d/d a_i = a_{k-i} (from j=i) + a_{k-i'} where i' = k-i (from j=k-i, if k-i != i)
    # If k-i == i (i.e., i = k/2): d/d a_i = 2 * a_i
    # If k-i != i: d/d a_i = a_{k-i} + a_{k-i} = 2 * a_{k-i}
    # Wait no: d/d a_i of sum_j a_j * a_{k-j}:
    #   = (k-i >= 0 and k-i < n) * a_{k-i}  [from j=i]
    #   + (i >= 0 and i < n) * a_{k-i}  [from j=k-i, if k-i != i]
    # Hmm, let me just use the FFT approach:
    # The gradient of sum_k w_k * (a*a)_k w.r.t. a_i is:
    # sum_k w_k * (a[k-i] + a[k-i]) ... this is getting confusing.
    # 
    # Let me use the FFT approach directly:
    # (a*a)_k = sum_j a_j * a_{k-j}
    # The "cross-correlation" interpretation: (a*a)_k = (a * a)_{k}
    # 
    # Actually, the simplest way: the gradient of (a*b) w.r.t. a is the cross-correlation of b with a (reversed).
    # d(a*a)_k / da_i = sum over all k of weights[k] * (contribution of a_i to (a*a)_k)
    # 
    # Using FFT: if we define the "effective weight" signal w_eff where w_eff[k] = weights[k],
    # then the gradient is the cross-correlation: grad = ifft(fft(w_eff) * conj(fft(a)))
    # But we need to be careful about the exact definition.
    
    # Let me derive it properly:
    # F(a) = sum_k weights[k] * (a*a)[k] = sum_k weights[k] * sum_j a[j] * a[k-j]
    # dF/da[i] = sum_k weights[k] * (d(a*a)[k]/da[i])
    # d(a*a)[k] = sum_j a[j]*a[k-j]
    # d/da[i] = a[k-i] (from j=i, if 0<=k-i<n) + a[k-i] (from j=k-i, if k-i != i and 0<=k-i<n)
    #          = a[k-i] * (1 + [k-i != i and 0<=k-i<n])
    #          = a[k-i] + [k-i != i] * a[k-i]  (with appropriate bounds)
    # 
    # This is essentially: grad[i] = sum_k weights[k] * a[k-i] + weights[k] * a[k-i] * [k-i != i]
    # = sum_k weights[k] * a[k-i] + sum_k weights[k] * a[k-i] - sum_k weights[k] * a[k-i] * [k-i == i]
    # = 2 * sum_k weights[k] * a[k-i] - sum_k weights[k] * a[k/2] * [k = 2i and 0<=k<n... ]
    # 
    # The "2 * cross-correlation minus diagonal correction" is complex.
    # 
    # Simpler approach: use the fact that for the FULL convolution (with zero padding),
    # the gradient of sum_k w_k * (a*a)_k w.r.t. a is:
    # grad = 2 * (w * a) restricted to [0, n-1]
    # where (w * a) is the full convolution of w and a, BUT with a specific indexing.
    #
    # Actually, let me just verify numerically for a small case and use the FFT trick.
    # 
    # The key identity: if c = a * a (full conv), then
    # d(sum_k w_k * c_k) / da_i = sum_k w_k * (a_{k-i} + a_{k-i} * (1 - delta_{k-i,i}))
    # Hmm this is getting too complicated analytically. Let me just use the standard result:
    #
    # For c = convolve(a, a), the Jacobian J[i,k] = dc_k/da_i = a_{k-i} + a_{k-i} (with bounds)
    # This is the same as the full convolution of [w reversed] with [a], evaluated at appropriate indices.
    #

    # But we need to be careful about the "full" vs "linear" convolution.

    # The gradient of sum_k w[k] * (a*a)[k] with respect to a[i] is:
    # = sum_k w[k] * (a[k-i] (if valid) + a[k-i] (if k-i != i and valid))

    # cross[i] = sum_k w[k] * a[k-i] = (w * a_reverse)[i + n - 1] ... 
    #
    # Actually, let me just use the FFT approach that the previous code uses and trust it:
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    # The gradient of sum_k w_k * (a*a)_k w.r.t. a_i:
    # Using the fact that if c = a*a, then dc/da = a (in the conv sense)
    # sum_k w_k * c_k = <w, a*a> = <w * a, a> (by adjointness of convolution)
    # d/da <w*a, a> = 2 * w*a - correction for self-term
    # Actually: <w, a*a> = sum_k w_k * sum_j a_j * a_{k-j}
    # = sum_j a_j * sum_k w_k * a_{k-j}
    # = sum_j a_j * (w_conj * a)[j]  where w_conj is w reversed
    # d/d a_i = (w_conj * a)[i] + sum_j a_j * d/d a_i (w_conj * a)[j]
    # = (w_conj * a)[i] + sum_j a_j * w_conj[j-i]
    # = (w_conv a)[i] + (a * w_conj)[i]
    # = 2 * (w_conv a)[i]  (by symmetry of convolution)
    # minus the diagonal correction...
    
    # I'll just use the empirically verified formula from the previous code:
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Compute normalized top-k weights (uniform on top k)."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    weights = np.zeros(len(c))
    weights[top_indices] = 1.0 / top_k
    return weights


def smooth_toppk_weights(c, top_k, temperature=1.0):
    """Smooth top-k: weight proportional to value for top-k, zero otherwise."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()

            A_ub = np.zeros((2 * n - 1, n))
            for k in range(2 * n - 1):
                j_start = max(0, k - (n - 1))
                j_end = min(n, k + 1)
                A_ub[k, j_start:j_end] = a[k - j_start:k - j_end + 1]
            b_ub = np.full(2 * n - 1, M)
        else:
            # Use only top 2000 most violated constraints
            top_idx = np.argpartition(c, -2000)[-2000:]
            n_constraints = len(top_idx)
            A_ub = np.zeros((n_constraints, n))
            for ii, k in enumerate(top_idx):
                j_start = max(0, k - (n - 1))
                j_end = min(n, k + 1)
                A_ub[ii, j_start:j_end] = a[k - j_start:k - j_end + 1]
            b_ub = np.full(n_constraints, M)

        c_obj = -np.ones(n)
        bounds = [(0, None)] * n

        t0 = time.monotonic()
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': 30, 'presolve': True})
        lp_time = time.monotonic() - t0

        if res.status != 0:
            # LP failed, do a subgradient step instead
            w = toppk_weights(c, 3)
            grad = compute_gradient_fft(a, w)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-15:
                step = S / (n * 2.0)
                a_new = project_simplex(a - step * grad / gnorm, S)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < M:
                    a = a_new
            lp_count += 1
            continue

        b_opt = res.x
        b_opt = np.maximum(b_opt, 0)
        s_b = b_opt.sum()
        if s_b < 1e-10:
            break
        b_opt = b_opt * (S / s_b)

        # Take a step toward b_opt
        # Try several step sizes
        best_a = a
        best_obj = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_candidate = safe_normalize((1 - alpha) * a + alpha * b_opt, n, S)
            c_candidate = fft_convolve_self(a_candidate)
            if c_candidate.max() < best_obj:
                best_obj = c_candidate.max()
                best_a = a_candidate

        a = best_a
        lp_count += 1

        if lp_count % 5 == 0:
            print("    LP iter %d: score=%.6f, lp_time=%.1fs" % (lp_count, fft_convolve_self(a).max(), lp_time))

        if stall > 10:
            # Perturb and try different constraints
            scale = 0.005 * S / np.sqrt(n)
            a = safe_normalize(a + rng.normal(0, scale, n), n, S)
            stall = 0

    return best_local, best_score


def run_coordinate_refine(a_init, n, S, max_time, rng):
    """Coordinate descent: optimize pairs of elements to reduce peak."""
    a = safe_normalize(a_init, n, S)
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    improvement = 0

    # Find the peak lag
    c = fft_convolve_self(a)
    peak_lag = int(np.argmax(c))

    while time.monotonic() - start < max_time:
        improved = False
        # Try random pairs
        for _ in range(200):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            # Optimize (a[i], a[j]) with a[i] + a[j] = const
            s_ij = a[i] + a[j]
            if s_ij < 1e-15:
                continue

            # The peak at lag k is affected by a[i]*a[k-i] and a[j]*a[k-j]
            # We want to minimize max over k of the convolution
            # For the specific peak: try to find optimal split
            # Binary search on x = a[i], a[j] = s_ij - x
            best_x = a[i]
            best_max = c.max()

            c_curr = fft_convolve_self(a)
            peak = c_curr.max()

            # Try a few splits
            for frac in [0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
                new_a = a.copy()
                new_a[i] = s_ij * frac
                new_a[j] = s_ij * (1 - frac)
                c_new = fft_convolve_self(new_a)
                if c_new.max() < best_max - 1e-12:
                    best_max = c_new.max()
                    best_x = s_ij * frac

            if best_max < peak - 1e-12:
                a[i] = best_x
                a[j] = s_ij - best_x
                improved = True
                improvement += 1

        if not improved:
            # Try structured moves: move mass from peak-contributing positions
            c = fft_convolve_self(a)
            peak_lag = int(np.argmax(c))
            # Find positions that contribute most to the peak
            contrib = np.zeros(n)
            for i in range(n):
                k = peak_lag
                partner = k - i
                if 0 <= partner < n:
                    contrib[i] = a[i] * a[partner]
                else:
                    contrib[i] = 0.0

            # Move mass from high-contrib to low-contrib
            top_ids = np.argsort(contrib)[-10:]
            bot_ids = np.argsort(contrib)[:10]

            for i in top_ids:
                for j in bot_ids:
                    if i == j:
                        continue
                    move = 0.1 * a[i]
                    if move > a[i]:
                        continue
                    new_a = a.copy()
                    new_a[i] -= move
                    new_a[j] += move
                    c_new = fft_convolve_self(new_a)
                    if c_new.max() < c.max() - 1e-12:
                        a = new_a.copy()
                        improved = True
                        improvement += 1
                        break
                if improved:
                    break

        c = fft_convolve_self(a)
        if c.max() < best_score - 1e-13:
            best_score = c.max()
            best_local = a.copy()

        if time.monotonic() - start > max_time - 2:
            break

    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, eta_init=0.05, top_k=3):
    """Mirror descent (multiplicative weights update)."""
    a = safe_normalize(a_init, n, S)
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-13:
            best_score = cur
            best_local = a.copy()

        eta = eta_init / (1.0 + 0.01 * it)
        w = toppk_weights(c, top_k)
        grad = compute_gradient_fft(a, w)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new * (S / s_new)

    return best_local, best_score


def propose_candidate(seed=42, budget_s=2385):
    """
    Multi-phase optimizer for minimizing 2*n*max_conv/sum^2.
    Combines subgradient, LP-based directions, coordinate descent,
    mirror descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s - 30.0)

    global_best = np.inf
    global_a = None
    global_n = 0

    # Get starting sequence
    given_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given_a = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Determine which lengths to try
    lengths_to_try = []
    if given_a is not None:
        lengths_to_try.append(len(given_a))
    # Add other lengths
    for n_candidate in [4096, 8192, 16384]:
        if n_candidate not in lengths_to_try:
            lengths_to_try.append(n_candidate)

    # Sort by priority (given length first, then by proximity)
    if given_a is not None:
        lengths_to_try.sort(key=lambda x: abs(x - len(given_a)))

    print("Lengths to try: %s" % lengths_to_try)

    for n in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break

        S = np.sqrt(2.0 * float(n))

        # Get starting point for this length
        if n == (len(given_a) if given_a is not None else -1):
            a_start = safe_normalize(given_a, n, S)
        else:
            # Resample or create new
            if given_a is not None and len(given_a) > 0:
                # Interpolate to new length
                x_old = np.linspace(0, 1, len(given_a))
                x_new = np.linspace(0, 1, n)
                a_resampled = np.interp(x_new, x_old, given_a)
                a_start = safe_normalize(a_resampled, n, S)
            else:
                a_start = np.ones(n) * S / n

        c_start = fft_convolve_self(a_start)
        initial_score = c_start.max()
        initial_obj = 2.0 * n * initial_score / (S * S)
        print("n=%d: initial objective=%.6f" % (n, initial_obj))

        local_best = a_start.copy()
        local_score = initial_score
        phase_budget = min(600, (deadline - time.monotonic()) * 0.3)

        # Phase A: Subgradient (main workhorse)
        t0 = time.monotonic()
        a_sa, score_sa = run_subgradient(a_start, n, S, phase_budget * 0.6, rng,
                                         beta_init=20.0, beta_max=500.0,
                                         top_k_start=1, top_k_end=30)
        a_sa = safe_normalize(a_sa, n, S)
        s_a = fft_convolve_self(a_sa).max()
        if s_a < local_score:
            local_score = s_a
            local_best = a_sa.copy()
        print("  n=%d subgrad: %.6f (obj=%.6f) [%.0fs]" % (n, local_score, 2*n*local_score/S**2, time.monotonic()-t0))

        # Phase B: LP-based direction (if time and n allow)
        remaining = deadline - time.monotonic()
        if remaining > 100 and n <= 8192:
            t0 = time.monotonic()
            lp_budget = min(200, remaining * 0.25)
            a_lp, score_lp = run_lp_direction(local_best, n, S, lp_budget, rng,
                                               max_lp_solves=20)
            a_lp = safe_normalize(a_lp, n, S)
            s_lp = fft_convolve_self(a_lp).max()
            if s_lp < local_score:
                local_score = s_lp
                local_best = a_lp.copy()
            print("  n=%d LP: %.6f (obj=%.6f) [%.0fs]" % (n, local_score, 2*n*local_score/S**2, time.monotonic()-t0))

        # Phase C: Coordinate refinement
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t0 = time.monotonic()
            coord_budget = min(120, remaining * 0.2)
            a_coord, score_coord = run_coordinate_refine(local_best, n, S, coord_budget, rng)
            a_coord = safe_normalize(a_coord, n, S)
            s_coord = fft_convolve_self(a_coord).max()
            if s_coord < local_score:
                local_score = s_coord
                local_best = a_coord.copy()
            print("  n=%d coord: %.6f (obj=%.6f) [%.0fs]" % (n, local_score, 2*n*local_score/S**2, time.monotonic()-t0))

        # Phase D: Mirror descent
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t0 = time.monotonic()
            mirror_budget = min(60, remaining * 0.15)
            a_md, score_md = run_mirror_descent(local_best, n, S, mirror_budget, rng,
                                                eta_init=0.03, top_k=2)
            a_md = safe_normalize(a_md, n, S)
            s_md = fft_convolve_self(a_md).max()
            if s_md < local_score:
                local_score = s_md
                local_best = a_md.copy()
            print("  n=%d mirror: %.6f (obj=%.6f) [%.0fs]" % (n, local_score, 2*n*local_score/S**2, time.monotonic()-t0))

        # Phase E: Final subgradient polish
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t0 = time.monotonic()
            polish_budget = min(120, remaining * 0.3)
            a_pol, score_pol = run_subgradient(local_best, n, S, polish_budget, rng,
                                               beta_init=50.0, beta_max=1000.0,
                                               top_k_start=1, top_k_end=10,
                                               perturb_rate=500)
            a_pol = safe_normalize(a_pol, n, S)
            s_pol = fft_convolve_self(a_pol).max()
            if s_pol < local_score:
                local_score = s_pol
                local_best = a_pol.copy()
            print("  n=%d polish: %.6f (obj=%.6f) [%.0fs]" % (n, local_score, 2*n*local_score/S**2, time.monotonic()-t0))

        # Update global best
        local_obj = 2.0 * n * local_score / (S * S)
        if local_obj < global_best:
            global_best = local_obj
            global_a = local_best.copy()
            global_n = n
            print("  NEW GLOBAL BEST: n=%d, obj=%.6f" % (n, global_best))

    # === Final exploration: perturb best and re-optimize ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        print("Final exploration (%.0fs remaining)" % remaining)
        n = global_n
        S = np.sqrt(2.0 * float(n))

        # Try several perturbations
        for variant in range(4):
            if time.monotonic() > deadline - 30:
                break
            if variant == 0:
                perturbed = global_a + rng.normal(0, 0.003 * S / np.sqrt(n), n)
            elif variant == 1:
                # Smooth perturbation
                kernel_size = min(128, n // 8)
                kernel = np.ones(kernel_size) / kernel_size
                smooth = np.convolve(rng.normal(0, 1, n), kernel, mode='same')
                perturbed = global_a + 0.005 * S / np.sqrt(n) * smooth
            elif variant == 2:
                # Multiplicative
                perturbed = global_a * (1.0 + rng.normal(0, 0.008, n))
            else:
                # Sinusoidal
                tt = np.linspace(0, 1, n, endpoint=False)
                freq = rng.integers(3, 20)
                perturbed = global_a + 0.004 * S / np.sqrt(n) * np.sin(2 * np.pi * freq * tt)

            perturbed = safe_normalize(perturbed, n, S)
            per_var_time = (deadline - time.monotonic() - 20) / max(1, (4 - variant))
            a_v, _ = run_subgradient(perturbed, n, S, min(per_var_time, 120), rng,
                                     beta_init=40.0, beta_max=800.0,
                                     top_k_start=1, top_k_end=15)
            a_v = safe_normalize(a_v, n, S)
            c_v = fft_convolve_self(a_v)
            obj_v = 2.0 * n * c_v.max() / (S * S)
            if obj_v < global_best:
                global_best = obj_v
                global_a = a_v.copy()
                print("  Variant %d: obj=%.6f" % (variant, global_best))

    # === Final polish on the absolute best ===
    remaining = deadline - time.monotonic()
    if remaining > 20 and global_a is not None:
        n = global_n
        S = np.sqrt(2.0 * float(n))
        a_final, _ = run_subgradient(global_a, n, S, remaining * 0.5, rng,
                                     beta_init=100.0, beta_max=2000.0,
                                     top_k_start=1, top_k_end=5,
                                     perturb_rate=300)
        a_final = safe_normalize(a_final, n, S)
        c_final = fft_convolve_self(a_final)
        obj_final = 2.0 * n * c_final.max() / (S * S)
        if obj_final < global_best:
            global_best = obj_final
            global_a = a_final.copy()
            print("  Final polish: obj=%.6f" % global_best)

    # Build final answer
    final_list = [float(max(0.0, x)) for x in global_a]
    print("FINAL: length=%d, objective=%.6f" % (len(final_list), global_best))

    # Verify
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: %.6f" % ver)
    else:
        print("Verification failed, using fallback")
        final_list = [0.5] * 256
        ver = evaluate_sequence(final_list)
        print("Fallback verified: %.6f" % ver)

    return final_list
