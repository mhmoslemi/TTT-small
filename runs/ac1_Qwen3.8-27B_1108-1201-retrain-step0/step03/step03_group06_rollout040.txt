"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using multi-phase Adam optimization with
log-sum-exp smoothing, top-K peak focusing, mirror descent alternation,
and coordinate descent polish. Key innovations: hybrid gradient (top-K + LSE),
entropy-based mirror steps for simplex constraints, and aggressive temperature
annealing with periodic perturbations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_lse(c, a, T):
    """Gradient of logsumexp(T*c)/T w.r.t. a."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    scaled = np.clip(scaled, -60, 60)
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    fsize = 1 << (3 * n - 2).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    X = np.real(ifft(P * np.conj(A)))
    Y = np.real(ifft(P * A))
    return (X + Y)[:n]


def compute_gradient_topk(c, a, K, power=2.0):
    """Gradient using top-K peaks with power weighting."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    # Select top K indices
    K_eff = min(K, len(c))
    top_idx = np.argpartition(c, -K_eff)[-K_eff:]
    top_c = c[top_idx]
    top_c = top_c ** power
    w = top_c / top_c.sum()
    # Build full weight vector
    p = np.zeros(len(c))
    p[top_idx] = w
    fsize = 1 << (3 * n - 2).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    X = np.real(ifft(P * np.conj(A)))
    Y = np.real(ifft(P * A))
    return (X + Y)[:n]


def compute_gradient_hybrid(c, a, T, K, alpha=0.5):
    """Hybrid gradient: alpha * top-K + (1-alpha) * LSE."""
    g1 = compute_gradient_topk(c, a, K)
    g2 = compute_gradient_lse(c, a, T)
    return alpha * g1 + (1.0 - alpha) * g2


def project_simplex(a, S):
    """Project to non-negative simplex with sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def mirror_step(a, grad, lr):
    """Multiplicative (mirror) update: a[i] *= exp(-lr*grad[i]), then normalize."""
    log_update = -lr * grad
    log_update = np.clip(log_update, -10, 10)
    a_new = a * np.exp(log_update)
    a_new = project_simplex(a_new, a.sum())
    return a_new

        mirror_frac=0.1, restart_interval=5.0, noise_scale=0.005,
        verbose=True
    )
    a_best2 = project_simplex(a_best2, S)
    ver = evaluate_sequence(a_best2.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best2.copy()
        print("  P1 Run2: obj=%.8f" % ver)
    
    # Run 3: Low temperature (broader smoothing)
    a_best3, obj3 = run_adam_opt(
        global_a, n, S, 0.3 * p1_budget,
        T_init=5.0, T_final=100.0, lr_init=0.06,
        use_hybrid=False, topk=50, alpha=0.5,
        mirror_frac=0.15, restart_interval=7.0, noise_scale=0.012,
        verbose=True
    )
    a_best3 = project_simplex(a_best3, S)
    ver = evaluate_sequence(a_best3.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best3.copy()
        print("  P1 Run3: obj=%.8f" % ver)
    
    # ============ PHASE 2: Diverse short runs (25% time) ============
    p2_budget = 0.25 * total_budget
    print("\n--- Phase 2: Diverse runs (budget=%.0fs) ---" % p2_budget)
    
    n_runs_p2 = 5
    per_run = p2_budget / n_runs_p2
    
    configs = [
        # (T_init, T_final, lr, topk, alpha, mirror_frac, noise)
        (20.0, 1000.0, 0.035, 40, 0.6, 0.05, 0.006),
        (100.0, 3000.0, 0.02, 20, 0.8, 0.0, 0.004),
        (5.0, 200.0, 0.05, 80, 0.4, 0.2, 0.01),
        (30.0, 500.0, 0.04, 60, 0.5, 0.1, 0.007),
        (500.0, 5000.0, 0.015, 10, 0.9, 0.0, 0.002),
    ]
    
    for r in range(n_runs_p2):
        if time.monotonic() > total_budget - 60:
            break
        cfg = configs[r % len(configs)]
        T_i, T_f, lr_r, topk_r, alpha_r, mir_r, noise_r = cfg
        
        # Add small perturbation for diversity
        a_pert = global_a + noise_r * 0.5 * rng.standard_normal(n)
        a_pert = project_simplex(a_pert, S)
        
        a_rr, _ = run_adam_opt(
            a_pert, n, S, per_run * 0.9,
            T_init=T_i, T_final=T_f, lr_init=lr_r,
            use_hybrid=True, topk=topk_r, alpha=alpha_r,
            mirror_frac=mir_r, restart_interval=per_run * 0.4,
            noise_scale=noise_r
        )
        a_rr = project_simplex(a_rr, S)
        ver = evaluate_sequence(a_rr.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_rr.copy()
            print("  P2 Run%d: obj=%.8f" % (r, ver))
    
    # ============ PHASE 3: Coordinate descent polish (15% time) ============
    p3_budget = 0.15 * total_budget
    print("\n--- Phase 3: Polish (budget=%.0fs) ---" % p3_budget)
    
    a = global_a.copy()
    a = project_simplex(a, S)
    
    # CD polish
    cd_budget = 0.5 * p3_budget
    a_cd, _ = coordinate_descent(a, n, S, cd_budget, max_sweeps=30)
    a_cd = project_simplex(a_cd, S)
    ver = evaluate_sequence(a_cd.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_cd.copy()
        print("  CD: obj=%.8f" % ver)
    
    # Zero masking
    a_z, _ = try_zero_masking(a_cd, n, S, n_candidates=30)
    a_z = project_simplex(a_z, S)
    ver = evaluate_sequence(a_z.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_z.copy()
        print("  ZeroMask: obj=%.8f" % ver)
    
    # Final high-T refinement
    rem = p3_budget - cd_budget - 5
    if rem > 10:
        a_fin, _ = run_adam_opt(
            global_a, n, S, rem,
            T_init=500.0, T_final=5000.0, lr_init=0.01,
            use_hybrid=True, topk=15, alpha=0.9,
            mirror_frac=0.0, restart_interval=rem * 0.5,
            noise_scale=0.001, verbose=True
        )
        a_fin = project_simplex(a_fin, S)
        ver = evaluate_sequence(a_fin.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_fin.copy()
            print("  Final HighT: obj=%.8f" % ver)
    
    # ============ PHASE 4: Remaining time - final polish ============
    remaining = total_budget - (time.monotonic() - (time.monotonic() - total_budget + (total_budget - (p1_budget + p2_budget + p3_budget))))
    # Compute remaining more directly
    elapsed_total = time.monotonic() - (time.monotonic() - total_budget)
    # Just check how much time is left of the total budget window
    # (We've been running within total_budget, so check against a safety margin)
    
    # Safety: ensure we return well before the hard deadline
    print("\n=== Final result: n=%d, obj=%.8f ===" % (n, global_best))
    
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("Verified final: obj=%.8f" % ver)
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        print("Fallback: obj=%.8f" % evaluate_sequence(final_a))
    
    return final_a
