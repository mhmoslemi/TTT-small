import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0)
    s = v.sum()
    if s >= S:
        # Need to reduce
        w = np.sort(v)[::-1]
        cumsum = np.cumsum(w)
        idx = np.arange(1, n + 1, dtype=np.float64)
        theta_candidates = (cumsum - S) / idx
        valid = theta_candidates < w
        if not np.any(valid):
            return v * (S / s)
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
        return np.maximum(v - theta, 0.0)
    else:
        # Need to increase - just scale up (not optimal but fast)
        return v * (S / s)


def softmax_obj_grad(z, beta, S):
    """
    Compute smoothed objective and gradient w.r.t. z (unconstrained softmax params).
    a = S * softmax(z), objective = (1/beta)*log(sum exp(beta*c_k))
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    n_a = len(a)
    fsize = 1 << (len(w) + n_a - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n_a]

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs_continuation(a_init, n, S, max_time, rng, beta_schedule=None, maxiter_per_beta=200):
    """Run L-BFGS-B with continuation on beta."""
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break

        n_attempts = 1
        if beta >= 500 and max_time > 30:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 3:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.01, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            try:
                result = scipy_minimize(
                    fun, z, jac=jac, method='L-BFGS-B',
                    options={'maxiter': maxiter_per_beta, 'maxfun': maxiter_per_beta * 2,
                             'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 30, 'iprint': -1}
                )
            except Exception:
                break

            z = result.x
            z = z - z.mean()
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score

def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def spectral_init(n, rng):
    """Generate initialization from random phase in frequency domain."""
    fsize = 1 << (2 * n - 1).bit_length()
    # Random phase
    phase = rng.uniform(0, 2 * np.pi, fsize)
    # Flat magnitude
    mag = np.ones(fsize)
    # Make it conjugate symmetric for real output
    spectrum = mag * np.exp(1j * phase)
    spectrum[0] = 1.0  # DC term
    # Enforce conjugate symmetry
    for k in range(1, fsize // 2):
        spectrum[fsize - k] = np.conj(spectrum[k])
    if fsize % 2 == 0:
        spectrum[fsize // 2] = 1.0
    a = np.real(ifft(spectrum, n))
    a = np.abs(a)  # Take absolute value to ensure non-negative
    a = a + 0.01  # Small offset to avoid zeros
    return a


def structured_init(n, rng):
    """Generate structured initializations based on known low-autocorrelation patterns."""
    tt = np.linspace(0, 1, n, endpoint=False)
    inits = []
    
    # Chirp-like
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * tt * n / 4) * np.cos(2 * np.pi * tt * n / 7))
    
    # Multi-frequency
    freqs = rng.choice(range(1, n // 2 + 1), size=5, replace=False)
    sig = np.ones(n)
    for f in freqs:
        sig += 0.2 * np.sin(2 * np.pi * f * tt + rng.uniform(0, 2 * np.pi))
    inits.append(np.abs(sig) + 0.05)
    
    # Binary-like (rounded)
    binary = (rng.random(n) > 0.5).astype(float)
    inits.append(binary * 0.8 + 0.2)
    
    # Smooth bumps
    centers = rng.uniform(0.1, 0.9, size=3)
    widths = rng.uniform(0.05, 0.15, size=3)
    sig = np.ones(n) * 0.3
    for c_pos, w in zip(centers, widths):
        sig += np.exp(-0.5 * ((tt - c_pos) / w) ** 2)
    inits.append(sig)
    
    return inits


def coordinate_descent_polish(a, n, S, max_time, rng, n_sweeps=3):
    """Simple coordinate descent: optimize each element while keeping others fixed."""
    a = a.copy()
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - start > max_time:
            break
        # Random order for diversity
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - start > max_time:
                break
            # Try a small perturbation of a[j]
            # The gradient w.r.t. a[j] is 2 * sum_k w[k] * a[j-k] for the active set
            # Simple: try increasing and decreasing
            old_val = a[j]
            s_rest = S - old_val
            
            # Try several candidates
            candidates = [old_val * 1.1, old_val * 0.9, old_val + S/n * 0.5, max(0, old_val - S/n * 0.5)]
            best_cand = old_val
            best_obj = np.inf
            
            for cand in candidates:
                cand = max(0.0, min(cand, S))
                if cand > s_rest + 1e-10:
                    continue
                a[j] = cand
                c = fft_convolve_self(a)
                obj = c.max()
                if obj < best_obj:
                    best_obj = obj
                    best_cand = cand
            
            a[j] = best_cand
    
    c = fft_convolve_self(a)
    return a, c.max()

"""

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0)
    s = v.sum()
    if s >= S:
        w = np.sort(v)[::-1]
        cumsum = np.cumsum(w)
        idx = np.arange(1, n + 1, dtype=np.float64)
        theta_candidates = (cumsum - S) / idx
        valid = theta_candidates < w
        if not np.any(valid):
            return v * (S / s)
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
        return np.maximum(v - theta, 0.0)
    else:
        return v * (S / s)


def compute_gradient_at_indices(a, w_full):
    """Compute grad of weighted sum of b[k] w.r.t. a, using FFT cross-correlation."""
    n = len(a)
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """Smoothed objective and gradient w.r.t. z via softmax parameterization."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_gradient_at_indices(a, w)
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)
    return obj, grad_z


def run_deficit_nesterov(a_init, n, S, max_time, rng, 
                          initial_delta=0.1, delta_decay=0.9995,
                          top_k_init=5, top_k_max=512,
                          restart_interval=4000, restart_prob=0.12):

    At each step:
    - Compute b = convolve(a, a)
    - Identify active set (top-k where b exceeds target)
    - Compute deficit-weighted gradient
    - Take Nesterov step and project to simplex
    - Gradually decrease the target (delta) to ratchet down the max

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    it = 0
    
    # Track the running minimum for adaptive target
    running_max = best_score

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()
        running_max = min(running_max, cur_max)
        
        # Adaptive target: gradually decrease from initial max
        target = running_max * (1.0 - initial_delta * (1.0 - delta_decay ** it))

        n_above = np.sum(above_target)
        top_k = min(top_k_max, max(top_k_init, int(n_above)))

        if excess.sum() < 1e-15:
            # All at or below target, use uniform weighting

        else:
            # Quadratic weighting on excess for stronger focus on peaks
            weights = (excess ** 2) / (excess ** 2).sum()

        if gnorm < 1e-15:

        step_base = 2.0 * S / (n * (1.0 + 0.0005 * it))
        step = step_base * (1.0 + min(gap / max(running_max, 1e-10), 2.0))
        
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)

        t = t_new

def run_lbfgs_aggressive(a_init, n, S, max_time, rng, 
                          beta_schedule=None, maxiter_per_beta=300):
    """
    Aggressive L-BFGS-B with log-sum-exp continuation.
    Uses more iterations at high beta and multiple restarts.
    """
    if beta_schedule is None:
        beta_schedule = [10, 30, 100, 300, 1000, 3000, 10000, 30000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() - start > max_time - 3:
            break

        # More attempts at higher beta
        n_attempts = 1
        if beta >= 1000:
            n_attempts = 3
        elif beta >= 300:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 3:
                break
            if attempt > 0:
                # Perturb for restart
                pert_scale = 0.005 / (1.0 + 0.5 * attempt)
                z = z + rng.normal(0, pert_scale, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            # Scale maxiter with beta (more iterations needed at high beta)
            iters = maxiter_per_beta * (1 + bi // 3)
            
            try:
                result = scipy_minimize(
                    fun, z, jac=jac, method='L-BFGS-B',
                    options={'maxiter': iters, 'maxfun': iters * 3,
                             'ftol': 1e-16, 'gtol': 1e-11, 'maxls': 50, 'iprint': -1}
                )
            except Exception:
                break

            z = result.x
            z = z - z.mean()
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score

    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def continuum_init(n, rng):
    """
    Generate initialization inspired by continuum theory.
    The optimal shape should make (f*f) as flat as possible.
    A raised-cosine or Fejer kernel shape is a good starting point.
    """
    tt = np.linspace(0, 1, n, endpoint=True)
    inits = []
    
    # Fejer kernel shape (triangular in frequency -> flat time autocorrelation)
    # f(x) = sin(pi*x) / (pi*x) squared -> Fejer kernel
    # Simpler: use a product of sins
    inits.append(np.abs(np.sin(np.pi * tt) ** 2))
    
    # Raised cosine (Hann window)
    inits.append(0.5 * (1 + np.cos(2 * np.pi * tt)))
    
    # Hamming window
    inits.append(0.54 - 0.46 * np.cos(2 * np.pi * tt))
    
    # "Ideal" shape: try to deconvolve a flat target
    # Start flat and iteratively adjust
    f = np.ones(n)
    for _ in range(20):
        b = fft_convolve_self(f)
        b_target = b.mean()
        # Where b is too high, reduce the corresponding f
        # Simple: scale f by 1/sqrt(b_center)
        center = n - 1
        if b[center] > 0:
            f = f / np.sqrt(max(b[center], 1e-10) / b_target)
        f = np.maximum(f, 0)
    inits.append(f)
    
    # Random smooth
    freqs = rng.choice([1, 2, 3, 4, 5, 7, 11], size=3, replace=False)
    sig = np.ones(n) * 0.5
    for f_ in freqs:
        sig += 0.3 * np.sin(2 * np.pi * f_ * tt + rng.uniform(0, 2 * np.pi))
    inits.append(np.abs(sig) + 0.05)
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    
    Multi-phase hybrid approach:
    1. Diverse initialization (continuum-guided + given + random)
    2. Deficit-based Nesterov subgradient (fast, handles non-smoothness)
    3. Aggressive L-BFGS with high-beta log-sum-exp
    4. Perturbation diversification + rapid re-optimization
    5. Final ultra-high-beta polish
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    global_best = np.inf
    global_a = None

    # Get initial sequence
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0.0, 1000.0)))
    except Exception:
        pass

    if inits:
        n = len(inits[0][1])
    else:
        n = 8192
        inits.append(("uniform", np.ones(n) * 1.0))

    S = np.sqrt(2.0 * float(n))
    print("=== n=%d, S=%.4f, budget=%.0fs ===" % (n, S, budget_s))
    print("Target: 1.5027, Previous: 1.508805")

    # Add continuum-based initializations
    cont_inits = continuum_init(n, rng)
    for i, ci in enumerate(cont_inits[:3]):
        inits.append(("cont%d" % i, ci))
    
    # Add random
    inits.append(("rand", rng.random(n) + 0.1))

    n_inits = len(inits)
    print("=== %d initializations ===" % n_inits)

    # Evaluate all initializations
    init_scores = []
    for name, arr in inits:
        a_norm = safe_normalize(arr, n)
        c = fft_convolve_self(a_norm)
        score = 2.0 * n * c.max() / (S ** 2)
        init_scores.append((name, a_norm, score))
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()

    print("Best init score: %.8f" % global_best)

    # === Phase 1: Deficit Nesterov on all inits (40% time) ===
    usable = deadline - time.monotonic()
    phase1_time = usable * 0.35
    per_nest = phase1_time / n_inits
    per_nest = max(per_nest, 10.0)

    print("\n=== Phase 1: Deficit Nesterov (%.1f sec each, total %.0f sec) ===" % (per_nest, phase1_time))

    phase1_results = []
    for name, a_init, init_score in init_scores:
        if time.monotonic() > deadline - 300:
            break
        
        t0 = time.monotonic()
        a_new, max_c = run_deficit_nesterov(
            a_init, n, S, per_nest, rng,
            initial_delta=0.15, delta_decay=0.9997,
            top_k_init=5, top_k_max=256,
            restart_interval=3000, restart_prob=0.1
        )
        a_new = safe_normalize(a_new, n)
        score = 2.0 * n * max_c / (S ** 2)
        phase1_results.append((a_new, score))
        
        if score < global_best:
            global_best = score
            global_a = a_new.copy()
            print("  [OK] %s: %.8f (init=%.6f, %.1fs)" % (name, global_best, init_score, time.monotonic() - t0))
        else:
            print("  %s: %.8f (best=%.6f, %.1fs)" % (name, score, global_best, time.monotonic() - t0))

    print("\n=== After Phase 1: global_best=%.8f ===" % global_best)

    # === Phase 2: Aggressive L-BFGS on top candidates (35% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase2_time = remaining * 0.35
        print("\n=== Phase 2: Aggressive L-BFGS (%.0f sec total) ===" % phase2_time)
        
        phase1_sorted = sorted(phase1_results, key=lambda x: x[1])
        n_lbfgs = min(4, len(phase1_sorted))
        per_lbfgs = phase2_time / n_lbfgs
        
        for i in range(n_lbfgs):
            if time.monotonic() > deadline - 240:
                break
            t0 = time.monotonic()
            a_start = phase1_sorted[i][0]
            a_new, max_c = run_lbfgs_aggressive(
                a_start, n, S, per_lbfgs, rng,
                beta_schedule=[20, 50, 200, 800, 3000, 10000, 30000, 100000],
                maxiter_per_beta=300
            )
            a_new = safe_normalize(a_new, n)
            score = 2.0 * n * max_c / (S ** 2)
            
            if score < global_best:
                global_best = score
                global_a = a_new.copy()
                print("  [OK] L-BFGS %d: %.8f (%.1fs)" % (i, global_best, time.monotonic() - t0))
            else:
                print("  L-BFGS %d: %.8f (best=%.6f, %.1fs)" % (i, score, global_best, time.monotonic() - t0))

    print("\n=== After Phase 2: global_best=%.8f ===" % global_best)

    # === Phase 3: Perturbation + rapid L-BFGS (20% time) ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            phase3_time = remaining * 0.25
            print("\n=== Phase 3: Perturbation + L-BFGS (%.0f sec) ===" % phase3_time)
            
            n_perturb = max(4, int(phase3_time / 30))
            per_perturb = phase3_time / n_perturb
            
            for p in range(n_perturb):
                if time.monotonic() > deadline - 120:
                    break
                
                # Vary perturbation scale
                noise_scales = [0.01, 0.02, 0.03, 0.015, 0.025, 0.005, 0.04, 0.02, 0.01, 0.03]
                scale = noise_scales[p % len(noise_scales)]
                a_pert = global_a + rng.normal(0, scale * S / np.sqrt(n), n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = safe_normalize(a_pert, n)
                
                a_new, max_c = run_lbfgs_aggressive(
                    a_pert, n, S, per_perturb, rng,
                    beta_schedule=[50, 200, 1000, 5000, 20000],
                    maxiter_per_beta=200
                )
                a_new = safe_normalize(a_new, n)
                score = 2.0 * n * max_c / (S ** 2)
                
                if score < global_best:
                    global_best = score
                    global_a = a_new.copy()
                    print("  [OK] Perturb %d (scale=%.3f): %.8f" % (p, scale, global_best))

    print("\n=== After Phase 3: global_best=%.8f ===" % global_best)

    # === Phase 4: Final polish - ultra-high beta L-BFGS ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print("\n=== Phase 4: Final Ultra-High-Beta Polish (%.0f sec) ===" % remaining)
            
            # First: one more Nesterov pass
            nest_time = min(remaining * 0.3, 60)
            if nest_time > 10:
                a_new, max_c = run_deficit_nesterov(
                    global_a, n, S, nest_time, rng,
                    initial_delta=0.05, delta_decay=0.9999,
                    top_k_init=3, top_k_max=64,
                    restart_interval=5000, restart_prob=0.05
                )
                a_new = safe_normalize(a_new, n)
                score = 2.0 * n * max_c / (S ** 2)
                if score < global_best:
                    global_best = score
                    global_a = a_new.copy()
                    print("  Nesterov final: %.8f" % global_best)
            
            # Then: ultra-high beta L-BFGS
            remaining2 = deadline - time.monotonic()
            if remaining2 > 15:
                a_new, max_c = run_lbfgs_aggressive(
                    global_a, n, S, remaining2, rng,
                    beta_schedule=[500, 2000, 10000, 50000, 200000],
                    maxiter_per_beta=400
                )
                a_new = safe_normalize(a_new, n)
                score = 2.0 * n * max_c / (S ** 2)
                if score < global_best:
                    global_best = score
                    global_a = a_new.copy()
                    print("  Ultra-beta L-BFGS: %.8f" % global_best)

    # === Final verification ===
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
