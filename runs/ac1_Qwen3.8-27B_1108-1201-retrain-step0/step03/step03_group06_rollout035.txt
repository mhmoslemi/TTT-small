"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using Nesterov accelerated gradient
descent on LogSumExp-smoothed objective, with active-set refinement and
coordinate descent polish. Exploits the convexity of the problem.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_smooth_gradient(c, a, T):
    """Compute gradient of T*log(sum(exp(c[k]/T))) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    scaled = (c - c_max) / T
    w = np.exp(scaled)
    w_sum = w.sum()
    w = w / w_sum
    
    # grad[m] = 2 * sum_k w[k] * a[k-m]
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]

def compute_subgradient(c, a, active_frac=0.01):
    """Compute subgradient using active set (k where c[k] is close to max)."""
    n = len(a)
    c_max = c.max()
    threshold = c_max - active_frac * c_max  # or c_max * (1 - active_frac)
    active = c >= threshold
    if not active.any():
        active[np.argmax(c)] = True
    w = np.zeros(len(c))
    w[active] = 1.0
    w /= w.sum()
    return compute_smooth_gradient_from_w(w, a)

def compute_smooth_gradient_from_w(w, a):
    """Compute gradient given weights w."""
    n = len(a)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]

def proj_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def nesterov_smoothed(a_init, n, S, time_budget, T_start=10.0, T_end=1.0, 
                       lr_0=0.1, beta=0.9, backtracking_max=10, verbose=False):
    """
    Nesterov accelerated gradient descent on LogSumExp-smoothed objective.
    """
    a = a_init.copy().astype(np.float64)
    a = proj_simplex(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    a_prev = a.copy()
    start = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        # T schedule: start high (smooth), end low (sharp)
        T = T_start * (T_end / T_start) ** progress
        
        # Nesterov lookahead
        y = a + (beta - 1.0) / (1.0 + beta) * (a - a_prev)
        y = proj_simplex(y, S)
        
        c_y = fft_convolve_self(y)
        cur_obj = c_y.max()
        
        # Update best
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Restart with noise
            noise_scale = 0.01 * (1.0 - progress)
            rng = np.random.default_rng(step * 7919)
            a = best_a + noise_scale * rng.standard_normal(n)
            a = proj_simplex(a, S)
            a_prev = a.copy()
            no_improve = 0
            step += 1
            continue
        
        # Compute gradient
        g = compute_smooth_gradient(c_y, y, T)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        lr = lr_0 * (1.0 + step * 0.01) ** (-0.5)  # decreasing LR
        found = False
        for i in range(backtracking_max):
            a_trial = y - lr * g
            a_trial = proj_simplex(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            # Armijo condition
            if obj_trial < cur_obj - 1e-4 * lr * gnorm * np.max(np.abs(g)):
                found = True
                break
            lr *= 0.5
        
        if not found:
            a_trial = y  # no progress
        
        a_prev = a.copy()
        a = a_trial
        step += 1
        
        if verbose and step % 500 == 0:
            print("    Nesterov step %d: obj=%.7f, T=%.2f" % (step, best_obj, T))
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using FISTA (Nesterov accelerated gradient)
on LogSumExp-smoothed objective, with exact subgradient refinement and
coordinate descent polish. Exploits the convexity of the problem for guaranteed
convergence to the global optimum. Key improvements: proper Nesterov schedule,
active-set tracking, multi-n exploration, and structured CAZAC-based init.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_smooth_grad(c, a, T):
    """Gradient of T*log(sum(exp(c/T))) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    scaled = (c - c_max) / T
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    return _grad_from_weights(w, a)


def _grad_from_weights(w, a):
    """Compute 2 * cross_corr(w, a)[:n] using FFT."""
    n = len(a)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def compute_active_subgrad(c, a, eps=0.01):
    """Exact subgradient using active set: average over k where c[k] >= (1-eps)*max."""
    n = len(a)
    c_max = c.max()
    threshold = c_max * (1.0 - eps)
    active = c >= threshold
    if not active.any():
        active[np.argmax(c)] = True
    w = np.zeros(len(c))
    w[active] = 1.0
    w /= w.sum()
    return _grad_from_weights(w, a)


def proj_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def fista_smoothed(a_init, n, S, time_budget, T_start=5.0, T_end=0.5,
                   lr_init=0.5, max_backtrack=10, verbose=False):
    """
    FISTA (Nesterov accelerated gradient) on LogSumExp-smoothed objective.
    Uses proper t_k schedule and backtracking line search.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = proj_simplex(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    a_prev = a.copy()
    t_k = 1.0
    start = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 300

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        # Geometric T schedule: smooth -> sharp
        T = T_start * (T_end / T_start) ** progress

        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        t_k = t_new

        # Lookahead
        y = a + momentum * (a - a_prev)
        y = proj_simplex(y, S)

        # Compute objective and gradient at y
        c_y = fft_convolve_self(y)
        cur_obj = c_y.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic restart
        if no_improve > max_no_improve:
            scale = 0.005 * (1.0 - 0.5 * progress)
            rng = np.random.default_rng(12345 + step)
            a = best_a + scale * rng.standard_normal(n)
            a = proj_simplex(a, S)
            a_prev = a.copy()
            t_k = 1.0
            no_improve = 0
            step += 1
            continue

        # Gradient
        g = compute_smooth_grad(c_y, y, T)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        # Backtracking line search (Armijo)
        lr = lr_init
        c_L = 1e-4
        found = False
        for i in range(max_backtrack):
            a_trial = y - lr * g
            a_trial = proj_simplex(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial <= cur_obj - c_L * lr * gnorm * np.max(np.abs(g)):
                found = True
                break
            lr *= 0.5

        if not found:
            a_trial = y

        a_prev = a.copy()
        a = a_trial
        step += 1

        if verbose and step % 1000 == 0:
            print("      FISTA step %d: obj=%.7f, T=%.3f, lr=%.6f" % (step, best_obj, T, lr))

    return best_a, best_obj


def subgradient_diminishing(a_init, n, S, time_budget, lr_scale=1.0, eps=0.005,
                            verbose=False):
    """
    Subgradient method with diminishing step size for the exact (non-smooth) objective.
    Uses active set subgradient.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = proj_simplex(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    a_prev = a.copy()
    t_k = 1.0
    start = time.monotonic()
    step = 0

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)

        # FISTA-style momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        t_k = t_new

        y = a + momentum * (a - a_prev)
        y = proj_simplex(y, S)

        c_y = fft_convolve_self(y)
        cur_obj = c_y.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = y.copy()

        # Exact active set subgradient
        g = compute_active_subgrad(c_y, y, eps=eps)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        # Diminishing step size
        lr = lr_scale / np.sqrt(step + 1.0)

        a_trial = y - lr * g
        a_trial = proj_simplex(a_trial, S)

        a_prev = a.copy()
        a = a_trial
        step += 1

        if verbose and step % 1000 == 0:
            print("      SG step %d: obj=%.7f, lr=%.6f" % (step, best_obj, lr))

    return best_a, best_obj


def coordinate_descent(a_init, n, S, time_budget, max_sweeps=20, verbose=False):
    """Coordinate descent: try scaling individual coordinates and re-normalizing."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = proj_simplex(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(42)
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            original = a[i]
            if original < 1e-12:
                continue

            best_local = original
            best_local_obj = c.max()

            # Try reducing (main direction for improvement)
            for factor in [0.5, 0.7, 0.85, 0.3, 0.9, 0.15, 1.3, 1.5]:
                a[i] = original * factor
                a = proj_simplex(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-13:
                    best_local_obj = obj_check
                    best_local = original * factor

            a[i] = best_local
            a = proj_simplex(a, S)
            c = fft_convolve_self(a)

            if c.max() < best_obj - 1e-13:
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        if not improved:
            if verbose:
                print("      CD converged at sweep %d" % sweep)
            break

    return best_a, best_obj


def zero_masking(a_init, n, S, max_mask=50):
    """Greedily zero out coordinates that reduce the objective."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = proj_simplex(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    sorted_idx = np.argsort(a)

    for idx in sorted_idx[:max_mask]:
        if a[idx] < 1e-12:
            continue
        saved = a[idx]
        a[idx] = 0.0
        a = proj_simplex(a, S)
        c_check = fft_convolve_self(a)
        obj_check = c_check.max()
        if obj_check < best_obj - 1e-13:
            best_obj = obj_check
            best_a = a.copy()
        else:
            a[idx] = saved
            a = proj_simplex(a, S)

    return best_a, best_obj


def structured_init(n, S, rng):
    """Generate structured initialization using chirp/CAZAC-like patterns."""
    i = np.arange(n)
    init_type = rng.integers(0, 4)

    if init_type == 0:
        # Chirp-based
        u = rng.integers(1, max(2, n // 4))
        phase = np.pi * u * i.astype(np.float64) ** 2 / n
        a = 1.0 + 0.4 * np.cos(phase) + 0.2 * np.cos(2.0 * phase + rng.random() * 2)
    elif init_type == 1:
        # Multi-frequency cosine
        n_freqs = rng.integers(3, 8)
        freqs = rng.integers(1, n // 2, size=n_freqs)
        phases = rng.random(n_freqs) * 2 * np.pi
        amplitudes = rng.random(n_freqs) * 0.3
        a = np.ones(n)
        for f, ph, amp in zip(freqs, phases, amplitudes):
            a += amp * np.cos(2 * np.pi * f * i / n + ph)
    elif init_type == 2:
        # Gaussian window mix
        centers = rng.random(3) * n
        widths = rng.random(3) * n * 0.3 + n * 0.05
        a = np.zeros(n)
        for ctr, w in zip(centers, widths):
            a += np.exp(-0.5 * ((i - ctr) / w) ** 2)
    else:
        # Random with smoothing
        a = rng.random(n)
        # Simple exponential smoothing
        kernel_size = rng.integers(3, 11)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')

    a = np.maximum(a, 0.01)
    return a * (S / a.sum())

        a = proj_simplex(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  CD: obj=%.7f" % obj)

        # More subgradient
        sg_budget = 0.3 * polish_budget
        a_best, _ = subgradient_diminishing(
            a, n, S, sg_budget,
            lr_scale=np.linalg.norm(a) * 0.3, eps=0.001
        )
        a = proj_simplex(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  SG polish: obj=%.7f" % obj)

        # Final FISTA
        rem = polish_deadline - time.monotonic()
        if rem > 5:
            a_best, _ = fista_smoothed(
                a, n, S, rem,
                T_start=1.0, T_end=0.05, lr_init=0.1
            )
            a = proj_simplex(a_best, S)
            obj = evaluate_sequence(a.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a.copy()
                print("  Final FISTA: obj=%.7f" % obj)

    # ============ PHASE 4: Restart exploration (5% time) ============
    restart_budget = deadline - time.monotonic()
    if global_a is not None and restart_budget > 20:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        best_a = global_a.copy()

        print("\nPhase 4: Restarts (budget=%.0fs)" % restart_budget)
        n_rounds = max(2, int(restart_budget / 30))
        per_round = restart_budget / (n_rounds + 1)

        for r in range(n_rounds):
            if time.monotonic() > deadline - 5:
                break
            noise = 0.003 * (1 + 0.5 * r)
            a_noisy = best_a + noise * rng.standard_normal(n)
            a_noisy = proj_simplex(a_noisy, S)

            a_best, _ = fista_smoothed(
                a_noisy, n, S, per_round,
                T_start=2.0, T_end=0.2, lr_init=0.2
            )
            a_best = proj_simplex(a_best, S)
            obj = evaluate_sequence(a_best.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_best.copy()
                best_a = a_best.copy()
                print("  Restart %d: obj=%.7f" % (r, obj))

    # ============ Final verification ============
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f (verified=%.7f)" % (global_best, ver))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)

    return final_a
