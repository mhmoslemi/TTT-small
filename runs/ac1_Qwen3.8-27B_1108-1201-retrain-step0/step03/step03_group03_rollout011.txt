"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using spectral flattening (GS algorithm)
combined with gradient-based refinement and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def gs_flat_spectrum(a, n_iter, S):
    """Gerchberg-Saxton style algorithm to flatten the power spectrum."""
    a = np.asarray(a, dtype=np.float64).copy()
    size = 2 * len(a) - 1
    fsize = 1 << (size - 1).bit_length()
    
    for i in range(n_iter):
        A = fft(a, fsize)
        mag = np.abs(A)
        # Target: flat magnitude
        # Sum of |A|^2 = fsize * sum(a^2) ... by Parseval
        # We want |A|^2 to be constant = (sum a)^2 / fsize * (fsize / (2n-1)) ... 
        # Actually, sum of conv(a,a) = (sum a)^2
        # And sum of |A|^2 = fsize * sum(a^2) by Parseval
        # We want max |A|^2 to be small. Target: mean |A|^2 = fsize * sum(a^2) / fsize = sum(a^2)
        # But we want to minimize max, so target = mean
        # Actually for flatness, we want |A[k]|^2 = constant for all k
        # The sum of |A[k]|^2 is fixed (= fsize * sum(a^2)), so the flat target is:
        # |A[k]|^2 = sum(a^2) for all k, i.e., |A[k]| = sqrt(sum(a^2))
        target_mag = np.sqrt(np.sum(a**2)) * np.ones(fsize)
        # But we also need to preserve the DC component (A[0] = sum(a))
        # A[0] must equal sum(a) = S
        target_mag[0] = S
        
        # Update phase: keep phase, change magnitude
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:len(a)]
        # Project to non-negative, sum = S
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() > 0:
            a_new = a_new * (S / a_new.sum())
        else:
            a_new = np.ones(len(a)) * S / len(a)
        a = a_new
    
    return a


def gs_adaptive(a, n_iter, S, warmup=50):
    """GS with adaptive target magnitude: water-filling approach."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    for i in range(n_iter):
        A = fft(a, fsize)
        mag = np.abs(A)
        
        # Water-filling: reduce large magnitudes, increase small ones
        if i < warmup:
            # Simple flat target
            target_sq = np.sum(A * np.conj(A)).real / fsize * fsize  # = sum(a^2) * fsize / fsize = sum(a^2)
            # Actually sum |A|^2 = fsize * sum|a|^2 by Parseval (with the normalization used by numpy)
            # numpy: ifft(fft(x)) = x, and sum|X|^2 = N * sum|x|^2 where N is FFT size
            total_pow = np.sum(mag**2)
            target_mag = np.sqrt(total_pow / fsize) * np.ones(fsize)
            target_mag[0] = S  # Preserve DC
        else:
            # Adaptive: target is geometric mean of current magnitude and flat target
            total_pow = np.sum(mag**2)
            flat_mag = np.sqrt(total_pow / fsize)
            # Blend: new_target = (mag * flat_dtype)^(1/2) ... 
            # Or: new_target = flat_mag + alpha * (mag - flat_mag) / (max_mag - flat_mag + eps)
            # Simpler: interpolate between current mag and flat target
            alpha = 0.5 + 0.5 * np.sin(np.pi * i / n_iter)  # Start with more change
            target_mag = alpha * flat_mag + (1 - alpha) * mag
            target_mag[0] = S
        
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() > 0:
            a_new = a_new * (S / a_new.sum())
        else:
            a_new = np.ones(n) * S / n
        a = a_new
    
    return a


def lse_objective_and_grad(a, beta):
    """Compute LSE-smoothed max of conv(a,a) and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        val = c_max
        grad = np.zeros(n)
        return val, grad
    
    # LSE value
    val = c_max + np.log(total) / beta
    
    # Gradient: d/d a [log(sum exp(beta*c))] / beta
    # c[k] = sum_i a[i]*a[k-i] for valid i
    # dc[k]/da[j] = a[k-j] + a[j-k] (both terms if both indices valid)
    # Actually: conv[j] = sum_i a[i] * a[conv_index - i]
    # d(conv[k])/d(a[j]) = a[k-j] + a[j-k] if both k-j and j-k are in [0,n-1]
    # Simplified: d(conv[k])/d(a[j]) = a[k-j] + a[(j-k) mod ...] 
    # Let's use the FFT-based approach:
    # If w[k] = exp(beta*c[k]) / sum(exp(beta*c)), then
    # grad = 2 * ifft(fft(w) * conj(fft(a)))[:n]
    
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = (2.0 / beta) * np.real(ifft(W * np.conj(A)))[:n]
    
    return val, grad


def optimize_lse(a, S, max_time, rng, beta_init=5.0, beta_max=200.0, fsize=None):
    """FISTA-style optimization of LSE-smoothed max convolution."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_c = fft_convolve_self(a).max()
    it = 0
    L = 10.0  # Initial Lipschitz estimate
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init + 0.5 * it)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_c:
            best_c = cur_max
            best_a = y.copy()
        
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total
        
        # Gradient of LSE
        fsize_g = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize_g)
        A = fft(y, fsize_g)
        grad = (2.0 / beta) * np.real(ifft(W * np.conj(A)))[:n]
        
        # FISTA step with backtracking
        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)
        
        # Simple backtracking (check every few iterations)
        if it % 10 == 0:
            c_new = fft_convolve_self(x_new)
            if c_new.max() > cur_max * 1.05:
                L *= 2.0
                step = 1.0 / L
                x_new = project_simplex(y - step * grad, S)
            else:
                L = max(L * 0.95, beta * 0.5)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    return best_a, best_c


def coordinate_descent(a, S, max_time, rng, n_passes=50):
    """Coordinate descent: pick highest-impact coordinate and adjust."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    if a.sum() > 0:
        a = a * (S / a.sum())
    
    start = time.monotonic()
    best_a = a.copy()
    c = fft_convolve_self(a)
    best_c = c.max()
    
    fsize = 1 << (2 * n - 2).bit_length()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_c:
            best_c = cur_max
            best_a = a.copy()
        
        # Find the lag with maximum value
        peak_lag = int(np.argmax(c))
        
        # Find coordinates that contribute most to this peak
        i_min = max(0, peak_lag - (n - 1))
        i_max = min(peak_lag, n - 1)
        
        if i_min > i_max:
            # Try random coordinate
            j = rng.integers(n)
        else:
            # Contribution of a[j] to c[peak_lag] is a[peak_lag - j] (if valid)
            indices = np.arange(i_min, i_max + 1)
            partners = peak_lag - indices
            contrib = a[indices] * a[partners]
            # Pick the one with largest contribution
            j = int(indices[np.argmax(contrib)])
        
        # Try to reduce a[j] (and increase another coordinate to maintain sum)
        # Reduce a[j] by delta, increase a[random_small] by delta
        delta = 0.01 * S / n
        a[j] -= delta
        if a[j] < 0:
            delta = a[j] * 0.5
            a[j] = 0
        
        # Find a small coordinate to increase
        small_candidates = np.where(a < np.median(a))[0]
        if len(small_candidates) > 0:
            k = small_candidates[rng.integers(len(small_candidates))]
            a[k] += delta
        
        # Check if improved
        c_new = fft_convolve_self(a)
        if c_new.max() >= cur_max:
            # Revert
            a[j] += delta
            a[k] -= delta
            # Try opposite: increase a[j], decrease a[k]
            a[j] += delta
            a[k] -= delta
            if a[k] < 0:
                a[j] -= delta
                a[k] += delta
            else:
                c_new2 = fft_convolve_self(a)
                if c_new2.max() >= cur_max:
                    # Revert both
                    a[j] -= delta
                    a[k] += delta
    
    return best_a, best_c


def spectral_project(a, S, n_gauss=5):
    """Apply spectral filtering to reduce peaks in the power spectrum."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    A = fft(a, fsize)
    mag = np.abs(A)
    
    # Identify frequency bins with high power
    mean_mag = np.mean(mag[1:])  # Skip DC
    threshold = mean_mag * 1.5
    
    for _ in range(n_gauss):
        # Reduce high-power bins
        high_mask = mag > threshold
        A[high_mask] *= 0.95
        # Restore DC
        A[0] = S
        
        a_new = np.real(ifft(A))[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() > 0:
            a_new = a_new * (S / a_new.sum())
        else:
            break
        a = a_new
        mag = np.abs(fft(a, fsize))
    
    return a

def make_initializations(n, S, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.arange(n, dtype=np.float64) / n
    
    # Uniform
    inits.append(np.ones(n) * S / n)
    
    # Anti-palindromic
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    inits.append(a)
    
    # Sinusoidal
    for freq in [3, 5, 7, 11]:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * tt)
        a = np.maximum(a, 0.01)
        a = a * (S / a.sum())
        inits.append(a)
    
    # Multi-frequency
    a = 1.0
    for freq in [3, 7, 13]:
        a = a + 0.2 * np.sin(2 * np.pi * freq * tt + rng.random() * np.pi)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    inits.append(a)
    
    # Random
    a = rng.random(n) + 0.05
    a = a * (S / a.sum())
    inits.append(a)
    
    # Exponential
    a = np.exp(-3 * tt) + 0.05
    a = a * (S / a.sum())
    inits.append(a)
    
    return inits

3. The constraint c[k*] <= t can be written as: sum_i a[i]*a[k*-i] <= t

Let me try a different approach: **alternating between time and frequency domain**.

Algorithm:
1. Start with a non-negative sequence a (sum = S)
2. Compute A = FFT(a, N)
3. Compute c = IFFT(|A|^2)[:2n-1]
4. Find the peak: c[k*] = max(c)
5. We want to reduce c[k*]. c[k*] = sum_i a[i]*a[k*-i]
6. The "gradient" of c[k*] w.r.t. a is: grad[j] = a[k*-j] + a[j-k*] (where valid)
7. Take a step in the -grad direction, project to {a >= 0, sum = S}
8. Repeat

This is essentially the subgradient method for the max function (using only the active constraint). The previous approach uses LSE smoothing which considers all lags, which is better for global optimization but might be slower to converge on the peak.

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using:
- GS-inspired spectral initialization
- Adaptive-beta LSE FISTA with backtracking
- Active-set subgradient polishing (top-k lags)
- Frequency-domain peak damping
- Multiple diverse restarts with perturbation
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def proj_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_c = (cs - S) / idx
    valid = theta_c <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cs[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def gs_init(a, n_iter, S, damp=0.98):
    """
    GS-style initialization: damp non-DC spectral components to create
    sequences with reduced peak-to-mean ratio in self-convolution.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    for i in range(n_iter):
        A = fft(a, fsize)
        # Damp non-DC components (sqrt of power for amplitude)
        if i < n_iter // 2:
            # Aggressive damping
            A[1:] *= damp
        else:
            # Gentle: only damp the peak non-DC components
            mag = np.abs(A)
            threshold = np.median(mag[1:])
            A[mag[1:] > threshold * 2] *= damp
        A[0] = S  # Preserve DC
        a_new = np.real(ifft(A))[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() > 1e-15:
            a = a_new * (S / a_new.sum())
        else:
            a = np.ones(n) * S / n
    
    return a


def lse_grad(a, c, beta):
    """
    Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a.
    c = conv(a,a), so dc/d a = 2 * circ_conv(d, a) where d = softmax(beta*c).
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fista_lse(a_init, S, max_time, rng, beta_start=5.0, beta_end=500.0):
    """
    FISTA for LSE-smoothed max self-convolution.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-15:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_c = convolve_self(a).max()
    it = 0
    L = 20.0  # Lipschitz estimate
    last_best_it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: increase slowly
        frac = min(1.0, it / (it + 100))
        beta = beta_start + (beta_end - beta_start) * frac
        
        c = convolve_self(y)
        cur = c.max()
        if cur < best_c - 1e-12:
            best_c = cur
            best_a = y.copy()
            last_best_it = it
        
        grad = lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step with backtracking
        step = 1.0 / L
        x_new = proj_simplex(y - step * grad, S)
        
        # Backtrack every 20 iterations
        if it % 20 == 0:
            c_new = convolve_self(x_new)
            if c_new.max() > cur * 1.02:
                L *= 2.0
                step = 1.0 / L
                x_new = proj_simplex(y - step * grad, S)
            else:
                L = max(L * 0.9, beta)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Stall detection and restart
        if it - last_best_it > 2000 and rng.random() < 0.1:
            noise = 0.01 * S / np.sqrt(n)
            y = proj_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0
            last_best_it = it
    
    return best_a, best_c


def active_set_polish(a_init, S, max_time, rng, top_k=10):
    """
    Active-set subgradient: only use top-k lags for the gradient.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-15:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) / n * S
    
    start = time.monotonic()
    best_a = a.copy()
    c = convolve_self(a)
    best_c = c.max()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = convolve_self(a)
        cur = c.max()
        if cur < best_c - 1e-14:
            best_c = cur
            best_a = a.copy()
        
        # Top-k lags with softmax weighting
        k = min(top_k + it // 500, 64, len(c))
        top_idx = np.argsort(c)[::-1][:k]
        top_vals = c[top_idx]
        tv_max = top_vals.max()
        scale = max(1.0, tv_max * 0.1)
        w = np.exp(top_vals / scale - tv_max / scale)
        w = w / w.sum()
        
        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = w
        
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: Polyak-like
        step = 2.0 * (cur - S**2 / (2*n-1) + 1e-10) / (gnorm**2 + 1e-15)
        step = min(step, S / (n * 0.1))  # Cap step size
        
        x_new = proj_simplex(a - step * grad, S)
        c_new = convolve_self(x_new)
        
        if c_new.max() < cur:
            a = x_new
        else:
            # Halve step and retry
            step *= 0.5
            x_new = proj_simplex(a - step * grad, S)
            c_new = convolve_self(x_new)
            if c_new.max() < cur:
                a = x_new
            else:
                step *= 0.5
    
    return best_a, best_c


def freq_domain_damp(a, S, n_iter=100, strength=0.02):
    """
    Directly damp the frequency components that contribute to the peak convolution values.
    For each peak lag k, the contribution in frequency domain is related to 
    the product A[k1]*conj(A[k2]) where k1-k2 = k.
    Simplified approach: reduce the variance of |A| (make it more flat).
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    if a.sum() > 0:
        a = a * (S / a.sum())
    
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    for i in range(n_iter):
        A = fft(a, fsize)
        mag = np.abs(A).copy()
        
        # Target: geometric mean of mag (preserves total power)
        log_mag = np.log(np.maximum(mag, 1e-30))
        target_log = np.mean(log_mag)
        target_mag = np.exp(target_log)
        
        # Blend: new_mag = mag * (target_mag/mag)^strength
        # = mag^(1-strength) * target_mag^strength
        new_mag = np.power(np.maximum(mag, 1e-30), 1.0 - strength) * (target_mag ** strength)
        new_mag[0] = mag[0]  # Preserve DC
        
        # Reconstruct with new magnitude, same phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        A_new[0] = A[0]
        
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() > 1e-15:
            a = a_new * (S / a_new.sum())
        else:
            break
    
    return a


def make_initializations(n, S, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.arange(n, dtype=np.float64) / n
    
    # Uniform
    inits.append(np.ones(n) * S / n)
    
    # Anti-palindromic (a[i] + a[n-1-i] = const)
    for _ in range(2):
        half = n // 2
        h = rng.random(half) * 0.8 + 0.1
        mu = S / half
        a = np.zeros(n)
        a[:half] = h
        for i in range(half):
            a[n - 1 - i] = mu - h[i]
        if n % 2 == 1:
            a[half] = mu / 2
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            a = a * (S / a.sum())
        inits.append(a)
    
    # Low-frequency sinusoidal
    for freq in [2, 3, 5]:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * tt)
        a = np.maximum(a, 0.01)
        a = a * (S / a.sum())
        inits.append(a)
    
    # Multi-frequency
    a = 1.0
    for freq in [2, 5, 9]:
        a = a + 0.15 * np.cos(2 * np.pi * freq * tt + rng.random() * np.pi)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    inits.append(a)
    
    # Random
    a = rng.random(n) + 0.1
    a = a * (S / a.sum())
    inits.append(a)
    
    # Exponential decay
    a = np.exp(-2.0 * tt) + 0.1
    a = a * (S / a.sum())
    inits.append(a)
    
    # Bump (concentrated in middle)
    center = n / 2
    sigma = n / 6
    a = np.exp(-0.5 * ((np.arange(n) - center) / sigma)**2) + 0.01
    a = a * (S / a.sum())
    inits.append(a)
    
    return inits
