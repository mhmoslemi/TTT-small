"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using spectral shaping,
FISTA, and active-set polishing with diverse asymmetric initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, S):
    """Rescale to sum = S, clamping to [0, 1000]."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def lse_smooth_max(c, beta):
    """LSE approximation of max."""
    cm = c.max()
    if beta * (c.max() - c.min()) < 1e-10:
        return c.max(), np.full(len(c), 1.0/len(c))
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return c.max(), np.full(len(c), 1.0/len(c))
    w = e / tot
    return cm + np.log(tot) / beta, w


def spectral_shaping(a, S, n_iters, alpha_start=0.5, alpha_end=0.9, rng=None):
    """
    Gerchberg-Saxton variant: iteratively flatten the power spectrum.
    """
    n = len(a)
    a = normalize_sum(a, S)
    fsize = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        alpha = alpha_start + (alpha_end - alpha_start) * it / max(1, n_iters - 1)
        
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: constant magnitude (flat spectrum)
        # Blend current magnitude with constant
        if mag.max() > 0:
            target_mag = np.full(fsize, mag.mean())
            new_mag = mag ** alpha * target_mag ** (1 - alpha)
        else:
            new_mag = mag
        
        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_sum(a_new, S)
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

    """

    The LSE is: L(a) = (1/beta) * log(sum_k exp(beta * c_k))
    where c_k = (a*a)_k.

Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses FISTA with verified gradient, spectral shaping, and diverse asymmetric initializations.

import numpy as np
import time
from scipy.fft import fft, ifft

# Find next power of 2
def next_pow2(m):
    return 1 << (m - 1).bit_length()

# FFT-based self-convolution
def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = next_pow2(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

# Project onto simplex {x >= 0, sum(x) = S}
def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

# Normalize to sum = S
def normalize_sum(a, S):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) * S / len(a)
    return a * (S / s)

# Compute LSE gradient
def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot  # length 2n-1
    
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    # = 2 * ifft(fft(w, L) * conj(fft(a, L)))[i]
    L = next_pow2(3 * n - 2)
    W = fft(w, L)
    A = fft(a, L)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]

# Exact subgradient at the argmax lag
def exact_subgrad(a, c):
    n = len(a)
    k = int(np.argmax(c))
    # grad[i] = 2 * a[k-i] if 0 <= k-i < n, else 0
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad

# Top-k subgradient
def topk_subgrad(a, c, k_count):
    n = len(a)
    k_count = min(k_count, len(c))
    top_k = np.argsort(c)[::-1][:k_count]
    # Weighted combination
    vals = c[top_k]
    w = vals / vals.sum()
    grad = np.zeros(n)
    for ki, wk in zip(top_k, w):
        lo = max(0, ki - n + 1)
        hi = min(ki, n - 1)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * wk * a[ki - idx]
    return grad

            step = 1.0 / (2.0 * beta)
        
        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Restart if stuck
        if it - last_improve > 5000:
            noise = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t = 1.0
            a = y.copy()
            last_improve = it
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best
