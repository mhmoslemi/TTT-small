   - `∇_a c[k][m] = a[k-m] + a[m-k]` (with zero-padding)
   
   Actually this is the same as the weighted cross-correlation that the previous code computes! So the main difference is in the optimization dynamics (Adam vs subgradient, annealing schedule, etc.)

2. **Multi-T**: Use several T values in sequence, from low (broad) to high (peak-focused)

3. **Peak surgery**: After optimization, find the top-k peaks and try to reduce them by redistributing mass

Let me write the code:

Minimize 2*n*max(conv(a,a))/sum(a)^2 using smooth-max gradient descent
with Adam on log-parameters, multi-T annealing, peak surgery, and
multi-scale exploration (n=4096 and n=8192).
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    # b[k] = sum_j a[j]*a[k-j] (valid j: max(0,k-n+1) <= j <= min(k, n-1))

def normalize_nonneg(a, S):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def upsample(a, new_n):
    """Upsample sequence a to new_n using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return np.interp(new_x, old_x, a)


def run_smooth_max_adam(a_init, n, S, time_budget, T_schedule, lr_init=0.03,
                        beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    """Adam optimization of smooth max with log-parameterization.
    T_schedule: list of (T_start, T_end, fraction_of_time) tuples."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a * (S / a.sum())
    theta = np.log(a)  # log-parameters (absolute, not relative)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    no_improve = 0
    max_no_improve = 500
    
    # Determine T at each step based on schedule
    total_time = sum(frac for _, _, frac in T_schedule)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Determine current T from schedule
        T = T_schedule[-1][1]  # default to last
        cum = 0
        for T_start, T_end, frac in T_schedule:
            cum_start = cum
            cum_end = cum + frac
            cum = cum_end
            p = (progress - cum_start) / max(frac, 1e-10)
            if 0 <= p <= 1:
                T = T_start + (T_end - T_start) * p
                break
        
        # Convert theta to a
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
            theta = np.full(n, np.log(S / n))
        else:
            a = S * exp_theta / total
        
        # Compute autocorrelation
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / (S * S)
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Restart with small perturbation
            theta = np.log(best_a + 1e-15) + 0.01 * np.random.standard_normal(n)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            continue
        
        # Compute smooth max gradient
        c_max = c.max()
        if c_max < 1e-30:
            break
        
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        w = w / w_sum
        
        # n is len(a), w has length 2n-1
        fsize = 1 << (3 * n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        
        xcorr1 = np.real(ifft(np.conj(W) * A))[:n]
        xcorr2 = np.real(ifft(W * A))[:n]
        g_a = (1.0 / T) * (xcorr1 + xcorr2)
        
        # Chain rule for log-parameters (theta[j] = log(a[j]))
        # dL/dtheta[j] = g_a[j] * a[j] - (a[j]/S) * sum_i g_a[i]*a[i]
        # (The second term ensures the constraint sum(a) = S is maintained)
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot
        
        # Adam update
        t += 1
        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break
        
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr = lr_init * (0.3 + 0.7 * (1.0 - progress))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if verbose and t % 2000 == 0:
            print("    Step %d: obj=%.7f, T=%.1f" % (t, best_obj, T))
    
    return best_a, best_obj


def peak_surgery(a, n, S, n_peaks=5, n_trials=20, verbose=False):
    """Try to reduce the top peaks by redistributing mass."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Find top peak positions
    top_k = np.argsort(c)[-n_peaks:][::-1]
    
    rng = np.random.default_rng(42)
    
    for _ in range(n_trials):
        # Pick a random peak to attack
        k = top_k[rng.integers(len(top_k))]
        
        # The peak c[k] = sum_j a[j]*a[k-j]
        # Find the pairs contributing most to this peak
        # For position k, the contributing pairs are (j, k-j) for valid j
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        
        if j_min >= j_max:
            continue
        
        # Find the pair with the largest product
        js = np.arange(j_min, j_max + 1)
        products = a[js] * a[k - js]
        pair_idx = js[np.argmax(products)]
        
        # Try to reduce the product by moving mass
        # Reduce a[pair_idx] and increase a[other] where other is far from the peak
        p = pair_idx
        q = k - pair_idx
        if p == q:
            # Diagonal element: reduce a[p]
            # Move mass to a position that doesn't contribute to this peak
            r = rng.integers(n)
            while r == p or (abs(r - k) < n // 4 and r != p):
                r = rng.integers(n)
            
            # Binary search for optimal transfer
            best_transfer = 0
            best_c = best_obj
            for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                transfer = a[p] * frac
                a_trial = a.copy()
                a_trial[p] -= transfer
                a_trial[r] += transfer
                a_trial = normalize_nonneg(a_trial, S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < best_c:
                    best_c = c_trial.max()
                    best_transfer = transfer
            if best_transfer > 0:
                a[p] -= best_transfer
                a[r] += best_transfer
                a = normalize_nonneg(a, S)
                c = fft_convolve_self(a)
                if c.max() < best_obj:
                    best_obj = c.max()
                    best_a = a.copy()
        else:
            # Off-diagonal: reduce min(a[p], a[q]) 
            # Move mass from the smaller to somewhere else
            if a[p] < a[q]:
                reduce_idx = p
            else:
                reduce_idx = q
            r = rng.integers(n)
            
            best_c = best_obj
            for frac in [0.1, 0.2, 0.3, 0.5]:
                transfer = a[reduce_idx] * frac
                a_trial = a.copy()
                a_trial[reduce_idx] -= transfer
                a_trial[r] += transfer
                a_trial = normalize_nonneg(a_trial, S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < best_c:
                    best_c = c_trial.max()
                    a = a_trial.copy()
                    c = c_trial.copy()
            
            if best_obj > c.max() if hasattr(c, 'max') else False:
                pass  # already updated
            # Actually let me redo this more carefully
            pass
    
    return best_a, best_obj


def antipodal_rebalance(a, n, S, n_iter=100, verbose=False):
    """Try to reduce the center peak by rebalancing antipodal pairs.
    b[n-1] = sum_j a[j]*a[n-1-j]. By making pairs unbalanced, we can reduce this."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(123)
    
    for _ in range(n_iter):
        # The center is at k = n-1
        # a[j] * a[n-1-j] contributes to b[n-1]
        # If we make a[j] and a[n-1-j] very different, the product is smaller
        # But we need to maintain the sum
        
        # Pick a random antipodal pair
        j = rng.integers(n // 2)
        k = n - 1 - j
        
        # Current: a[j]*a[k]
        # If we change to a[j]+d, a[k]-d: new product = (a[j]+d)(a[k]-d)
        # = a[j]*a[k] + d(a[k]-a[j]) - d^2
        # To minimize, we want d = (a[k]-a[j])/2, giving product = (a[j]+a[k])^2/4 - (a[k]-a[j])^2/4
        # = a[j]*a[k] - (a[k]-a[j])^2/4 ... that's just the minimum of the product for fixed sum
        # Which is 0 (by making one of them 0)
        
        # But we need to check if this helps the overall max (not just b[n-1])
        d_options = [(a[k] - a[j]) * f for f in [0.3, 0.5, 0.7, 0.9]]
        best_d = 0
        best_obj_local = best_obj
        
        for d in d_options:
            a_trial = a.copy()
            a_trial[j] += d
            a_trial[k] -= d
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < best_obj_local:
                best_obj_local = c_trial.max()
                best_d = d
        
        if best_d != 0:
            a[j] += best_d
            a[k] -= best_d
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()
    
    return best_a, best_obj


def coordinate_descent_peaks(a, n, S, time_budget, verbose=False):
    """Coordinate descent focusing on reducing peaks."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using smooth-max gradient descent with
Adam on log-parameters, multi-T annealing, multi-scale exploration (n=4096
and n=8192), peak-specific coordinate descent, and antipodal rebalancing.

Key improvements over previous approach:
1. Multi-scale: optimize at n=4096 then upsample to n=8192 for better bounds
2. Exact smooth-max gradient (both cross-correlation terms)
3. Multi-T annealing with coarse-to-fine peak focus
4. Peak-specific coordinate descent targeting the actual peak locations
5. Antipodal pair rebalancing to reduce center peak
6. Fresh random starts at larger n for diverse basins
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_obj_normalized(a, n, S):
    """Compute objective assuming sum(a)=S."""
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def upsample_linear(a, new_n):
    """Upsample sequence using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    if old_n > new_n:
        old_x = np.linspace(0, 1, old_n)
        new_x = np.linspace(0, 1, new_n)
        return np.interp(new_x, old_x, a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    return np.interp(new_x, old_x, a)


def smooth_max_gradient(c, a, T):
    """Compute exact gradient of smooth max L_T = (1/T)log(sum exp(T*c/cmax)) w.r.t. a.
    Uses both cross-correlation terms for the full convolution gradient."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)

    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # Gradient: g[m] = (1/T) * [sum_j w[j+m]*a[j] + sum_j w[m-j]*a[j]]
    # First term: cross-correlation of w and a at lag m
    # Second term: convolution of w and a at lag m
    # Both need zero-padding to size >= (2n-1) + n - 1 = 3n - 2
    fsize = 1 << (3 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)

    # xcorr1[m] = sum_j w[j+m]*a[j] = ifft(conj(W)*A)[m]
    xcorr1 = np.real(ifft(np.conj(W) * A))[:n]
    # xcorr2[m] = sum_j w[m-j]*a[j] = (w*a)[m] = ifft(W*A)[m]
    xcorr2 = np.real(ifft(W * A))[:n]

    return (1.0 / T) * (xcorr1 + xcorr2)


def run_smooth_max_adam(a_init, n, S, time_budget, T_schedule,
                        lr_init=0.03, beta1=0.9, beta2=0.999,
                        eps=1e-8, verbose=False):
    """Adam optimization of smooth max with log-parameterization.
    T_schedule: list of (T_start, T_end, time_fraction) tuples."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a * (S / a.sum())
    theta = np.log(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best_obj = compute_obj_normalized(a, n, S)
    best_a = a.copy()
    no_improve = 0
    max_no_improve = 400

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Determine current T
        T = T_schedule[-1][1]
        cum = 0.0
        for T_s, T_e, frac in T_schedule:
            cum_start = cum
            cum_end = cum + frac
            cum = cum_end
            if frac > 0:
                p = (progress - cum_start) / frac
                if 0.0 <= p <= 1.0:
                    T = T_s + (T_e - T_s) * min(1.0, max(0.0, p))
                    break

        # Convert theta to a
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
            theta = np.full(n, np.log(S / n))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
        else:
            a = S * exp_theta / total

        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / (S * S)

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            theta = np.log(best_a + 1e-15) + 0.005 * np.random.standard_normal(n)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            continue

        c_max = c.max()
        if c_max < 1e-30:
            break

        g_a = smooth_max_gradient(c, a, T)

        # Chain rule: dL/dtheta[j] = g_a[j]*a[j] - (a[j]/S)*sum_i g_a[i]*a[i]
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot

        t += 1
        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break

        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_init * (0.2 + 0.8 * (1.0 - progress))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if verbose and t % 3000 == 0:
            print("      Adam step %d: obj=%.7f, T=%.1f" % (t, best_obj, T))

    return best_a, best_obj


def peak_coordinate_descent(a, n, S, time_budget, verbose=False):

    a = a.copy().astype(np.float64)

def antipodal_optimize(a, n, S, n_iter=200, verbose=False):
    """Reduce center peak by rebalancing antipodal pairs a[j], a[n-1-j]."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    rng = np.random.default_rng(777)

    for _ in range(n_iter):
        j = rng.integers(max(1, n // 2))
        k = n - 1 - j
        if j == k:
            continue
        total = a[j] + a[k]
        if total < 1e-14:
            continue

        best_d = 0.0
        best_local = best_obj
        # Try moving mass from one to the other
        for frac in (0.3, 0.5, 0.7, 0.9, -0.3, -0.5, -0.7, -0.9):
            d = total * frac
            a_trial = a.copy()
            a_trial[j] = a[j] + d
            a_trial[k] = a[k] - d
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < best_local - 1e-13:
                best_local = c_trial.max()
                best_d = d

        if best_d != 0.0:
            a[j] += best_d
            a[k] -= best_d
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()

    return best_a, best_obj

    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.8f (verified=%.8f)" % (global_best, ver))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.8f" % ver)

    return final_a
