Minimize 2*n*max(conv(a,a))/sum(a)^2 using hybrid optimization:
- Soft gradient over all peaks (not just top-k)
- Multi-start with diverse initializations
- Simulated annealing for escape from local minima
- Multi-scale refinement

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_objective(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def soft_subgrad(c, a, temperature):
    """
    Compute subgradient using softmax over all peaks.
    grad[m] = 2 * sum_k w[k] * a[k-m]
    where w[k] = exp(c[k]/temperature) / sum(exp(c[k]/temperature))
    """
    n = len(a)
    # Softmax weights
    c_shifted = c - c.max()  # for numerical stability
    w = np.exp(c_shifted / temperature)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Flat softmax
        w = np.ones_like(c) / len(c)
    else:
        w /= w_sum
    
    # grad[m] = 2 * sum_k w[k] * a[k-m]
    # This is 2 * conv(w, a_rev)[n-1+m] where a_rev = a[::-1]
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1 - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n-1:2*n-1]
    return grad

def topk_subgrad(c, a, top_k):
    """Subgradient from top-k peaks only."""
    n = len(a)
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    c_max = c[top_idx[0]]
    # Soft weights among top-k
    w_top = np.exp((c[top_idx] - c_max) / max(c_max * 0.1, 1e-10))
    w_top /= w_top.sum()
    
    w_full = np.zeros(len(c))
    w_full[top_idx] = w_top
    
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1 - 1).bit_length()
    W = fft(w_full, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n-1:2*n-1]
    return grad

import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def gen_soft_grad(c, a, temperature):
    """Subgradient using softmax over ALL peaks with given temperature."""
    n = len(a)
    c_shifted = c - c.max()
    w = np.exp(c_shifted / temperature)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w /= w_sum
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def gen_topk_grad(c, a, top_k, soft_T=0.0):
    """Subgradient from top-k peaks with optional soft weighting."""
    n = len(a)
    top_k = max(1, min(top_k, len(c)))
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    if soft_T > 1e-10:
        c_max = c[top_idx[0]]
        w_top = np.exp((c[top_idx] - c_max) / soft_T)
        w_top /= w_top.sum()
    else:
        w_top = np.full(top_k, 1.0 / top_k)

    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]

    gamma < 1 compresses peaks; gamma > 1 expands them.
    Returns new a (non-negative, same sum).

    S = a.sum()
    fsize = 1 << (2 * n - 2).bit_length()
    a_padded = np.zeros(fsize)
    a_padded[:n] = a
    A = fft(a_padded)
    P = np.abs(A) ** 2
    # Guard against zero power
    P = np.maximum(P, 1e-300)
    # Gamma compression
    P_new = P ** gamma
    # Preserve DC component (sum)
    P_new[0] = P[0]
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_padded = np.real(ifft(A_new))
    a_new = a_new_padded[:n]
    a_new = np.clip(a_new, 0, None)
    s_new = a_new.sum()
    if s_new < 1e-10:
        return None
    a_new *= S / s_new
    return a_new


def mirror_step(a, grad, lr, S):
    """Multiplicative (mirror) descent step."""
    a_try = a * np.exp(-lr * grad)
    a_try = np.maximum(a_try, 1e-300)
    s = a_try.sum()
    if s < 1e-10:
        return None
    a_try *= S / s
    return a_try


def additive_step(a, grad, lr, S):
    """Additive descent step with projection."""
    a_try = a - lr * grad
    a_try = np.maximum(a_try, 0.0)
    s = a_try.sum()
    if s < 1e-10:
        return None
    a_try *= S / s
    return a_try


def optimize_hybrid(a0, time_budget, verbose=True, verbose_interval=3000,
                    gamma_init=0.95, lr_init=0.2, momentum=0.88,
                    restart_after=600, noise_frac=0.08,
                    temp_init=10.0, temp_decay=0.9997):
    """
    Hybrid optimization: alternates between spectral flattening and
    gradient-based mirror descent, with adaptive switching and restarts.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.clip(a, 0, None)
    if a.sum() > 0:
        a *= S / a.sum()
    else:
        a = np.full(n, S / n)

    c = fft_convolve_self(a)
    best_obj = compute_obj(a, c)
    best_a = a.copy()
    best_iter = 0
    iter_no = 0

    v = np.zeros(n)
    lr = lr_init
    gamma = gamma_init
    temperature = temp_init
    start = time.monotonic()
    last_improve_time = time.monotonic()
    use_spectral = True  # Alternate modes

    while time.monotonic() - start < time_budget:
        # Choose step type based on recent performance
        # Use spectral when gradient steps are stagnating
        obj_cur = compute_obj(a, c)

        if use_spectral:
            # Spectral flattening
            a_try = spectral_step(a, gamma)
            if a_try is None:
                gamma = max(gamma - 0.05, 0.3)
                a_try = a.copy()
            c_try = fft_convolve_self(a_try)
            obj_try = compute_obj(a_try, c_try)

            if obj_try < best_obj * (1 + 1e-9):
                best_obj = obj_try
                best_a = a_try.copy()
                best_iter = iter_no
                last_improve_time = time.monotonic()
                # Slightly more aggressive
                gamma = max(gamma - 0.002, 0.3)
            else:
                gamma = min(gamma + 0.02, 0.99)

            a = a_try
            c = c_try
        else:
            # Gradient-based step
            if temperature > 50.0:
                grad = gen_soft_grad(c, a, temperature)
            else:
                top_k = max(10, int(200 * temperature / max(temperature, 1)))
                grad = gen_topk_grad(c, a, min(top_k, 500), soft_T=max(temperature * 0.1, 0.1))

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                # No gradient, try spectral
                use_spectral = True
                continue
            grad /= gnorm

            v = momentum * v + (1.0 - momentum) * grad
            vn = np.linalg.norm(v)
            if vn < 1e-15:
                use_spectral = True
                continue
            v /= vn

            a_try = mirror_step(a, v, lr, S)
            if a_try is None:
                a_try = a.copy()
            c_try = fft_convolve_self(a_try)
            obj_try = compute_obj(a_try, c_try)

            if obj_try < best_obj * (1 + 1e-9):
                best_obj = obj_try
                best_a = a_try.copy()
                best_iter = iter_no
                last_improve_time = time.monotonic()
                lr = min(lr * 1.03, 2.0)
            else:
                lr = max(lr * 0.94, 0.01)

            a = a_try
            c = c_try

        # Switch mode every few iterations
        iter_no += 2
        if iter_no % 2 == 0:
            use_spectral = not use_spectral

        # Anneal temperature
        temperature *= temp_decay

        # Stagnation detection and restart
        if iter_no - best_iter > restart_after:
            noise = np.random.randn(n) * noise_frac * S / np.sqrt(n)
            a = a + noise
            a = np.clip(a, 0, None)
            a *= S / a.sum()
            c = fft_convolve_self(a)
            v *= 0.3
            lr = lr_init
            gamma = gamma_init
            temperature = max(temperature, temp_init * 0.5)
            if verbose:
                elapsed = time.monotonic() - start
                print(f"      [restart] iter={iter_no}, best={best_obj:.8f}, "
                      f"noise={noise_frac:.3f}, t={elapsed:.1f}s")

        # Adaptive noise increase if stuck for too long
        time_since_improve = time.monotonic() - last_improve_time
        if time_since_improve > 60.0:
            noise_frac = min(noise_frac * 1.05, 0.5)
            temperature = max(temperature, 5.0)

        if verbose and iter_no % verbose_interval == 0:
            elapsed = time.monotonic() - start
            cur = compute_obj(a, c)
            print(f"      iter={iter_no}, cur={cur:.8f}, best={best_obj:.8f}, "
                  f"lr={lr:.4f}, gamma={gamma:.3f}, T={temperature:.1f}, t={elapsed:.1f}s")

    return best_a, best_obj


def polish_coordinate(a0, time_budget, verbose=True):
    """Coordinate descent polish: try reducing individual elements at peak-contributing positions."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = compute_obj(a, c)
    best_a = a.copy()
    start = time.monotonic()
    sweep = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        improved = False
        # Focus on indices that contribute most to the top peaks
        top_peaks = np.argsort(c, kind='stable')[::-1][:20]
        # Find which indices of a contribute most to these peaks
        for k in top_peaks[:10]:
            if time.monotonic() - start > time_budget:
                break
            # a[j] * a[k-j] contributes to c[k]
            # Try reducing a[j] for the j with largest product
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo > hi:
                continue
            js = np.arange(lo, hi + 1)
            kjs = k - js
            valid = (kjs >= 0) & (kjs < n)
            if not np.any(valid):
                continue
            js = js[valid]
            kjs = kjs[valid]
            products = a[js] * a[kjs]
            # Try reducing the largest contributor
            for idx_j in js[np.argsort(products)[::-1][:5]]:
                for factor in [0.8, 0.9, 1.1, 1.2]:
                    orig = a[idx_j]
                    a[idx_j] = orig * factor
                    s = a.sum()
                    if s < 1e-10:
                        a[idx_j] = orig
                        break
                    a_norm = a * (S / s)
                    c_new = fft_convolve_self(a_norm)
                    obj_new = compute_obj(a_norm, c_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_norm.copy()
                        a = a_norm.copy()
                        c = c_new
                        improved = True
                        break
                    else:
                        a[idx_j] = orig
                if a[idx_j] != orig:
                    break
        if not improved:
            # Try increasing small elements
            for i in np.random.choice(n, min(50, n), replace=False):
                if time.monotonic() - start > time_budget:
                    break
                if a[i] < S / n * 0.5:
                    for factor in [2.0, 3.0, 5.0]:
                        orig = a[i]
                        a[i] = orig * factor
                        s = a.sum()
                        if s < 1e-10:
                            a[i] = orig
                            break
                        a_norm = a * (S / s)
                        c_new = fft_convolve_self(a_norm)
                        obj_new = compute_obj(a_norm, c_new)
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_a = a_norm.copy()
                            a = a_norm.copy()
                            c = c_new
                            improved = True
                            break
                        else:
                            a[i] = orig
                    a[i] = best_a[i]  # sync
                    if a[i] != orig and improved:
                        break
        if not improved and sweep > 20:
            break
        # Sync a with best_a periodically
        if sweep % 10 == 0:
            a = best_a.copy()
            c = fft_convolve_self(a)

    return best_a, best_obj


def generate_diverse_initializations(rng, n, S):
    """Generate multiple diverse non-negative sequences with sum S."""
    candidates = []
    i = np.arange(n)

    # Cosine with various frequencies
    for freq in [3, 7, 13, 21, 50]:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * i / n)
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
        candidates.append(a)

    # Two-hump
    a = (np.exp(-0.5 * ((i - n * 0.2) / (n * 0.1)) ** 2) +
         np.exp(-0.5 * ((i - n * 0.8) / (n * 0.1)) ** 2))
    a = np.maximum(a, 0.01)
    a *= S / a.sum()
    candidates.append(a)

    # V-shape (large at edges, small in middle)
    a = np.abs(i - n / 2) / (n / 2) + 0.1
    a *= S / a.sum()
    candidates.append(a)

    # Inverted V (small at edges, large in middle)
    a = 1.0 - 0.7 * np.abs(i - n / 2) / (n / 2)
    a = np.maximum(a, 0.01)
    a *= S / a.sum()
    candidates.append(a)

    # Random
    a = rng.random(n) + 0.01
    a *= S / a.sum()
    candidates.append(a)

    # Sinusoidal
    a = 1.0 + 0.3 * np.sin(2 * np.pi * 29 * i / n) + 0.2 * np.cos(2 * np.pi * 47 * i / n)
    a = np.maximum(a, 0.01)
    a *= S / a.sum()
    candidates.append(a)

    return candidates


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses hybrid spectral-gradient optimization with multi-start and adaptive switching.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    global_best = np.inf
    global_a = None

    # --- Load parent sequence ---
    parent_a = None
    try:
        if 'height_sequence_1' in globals():
            parent_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if parent_raw.size >= 16 and parent_raw.sum() > 0.01:
                parent_a = parent_raw.copy()
                ver = evaluate_sequence(parent_a.tolist())
                print(f"Parent: n={parent_a.size}, obj={ver:.8f}")
                if np.isfinite(ver) and ver < global_best:
                    global_best = ver
                    global_a = parent_a.copy()
    except Exception as e:
        print(f"Parent load error: {e}")

    if parent_a is None:
        # Generate a starting point
        n = 4096
        S = np.sqrt(2.0 * n)
        i = np.arange(n)
        parent_a = 1.0 + 0.4 * np.cos(2 * np.pi * 7 * i / n)
        parent_a = np.maximum(parent_a, 0.01)
        parent_a *= S / parent_a.sum()
        ver = evaluate_sequence(parent_a.tolist())
        print(f"Generated start: n={n}, obj={ver:.8f}")
        global_best = ver
        global_a = parent_a.copy()

    # --- Phase 1: Quick multi-start exploration (15% of time) ---
    phase1_time = 0.15 * (deadline - time.monotonic())
    print(f"\nPhase 1: Multi-start exploration ({phase1_time:.0f}s)")

    n_parent = len(global_a)
    S_parent = global_a.sum()

    # Try optimizing the parent and a few diverse inits in parallel (sequentially)
    test_candidates = [(parent_a, 'parent')]
    for a_div in generate_diverse_initializations(rng, n_parent, S_parent)[:4]:
        test_candidates.append((a_div, 'diverse'))

    per_candidate = phase1_time / len(test_candidates)
    for a_cand, name in test_candidates:
        if time.monotonic() > deadline - 120:
            break
        t0 = time.monotonic()
        a_best, obj_best = optimize_hybrid(
            a_cand, per_candidate, verbose=False,
            gamma_init=0.93, lr_init=0.25, momentum=0.85,
            restart_after=400, noise_frac=0.06,
            temp_init=20.0, temp_decay=0.9995
        )
        elapsed = time.monotonic() - t0
        print(f"  {name} (n={len(a_cand)}): obj={obj_best:.8f} ({elapsed:.1f}s)")
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

    # --- Phase 2: Main intensive optimization (55% of time) ---
    remaining = deadline - time.monotonic()
    phase2_time = 0.55 * remaining
    print(f"\nPhase 2: Main optimization ({phase2_time:.0f}s)")

    if global_a is not None and phase2_time > 30:
        # Split into sub-phases with different strategies
        sub_phases = [
            # (fraction, gamma_init, lr_init, momentum, restart, noise, temp_init, desc)
            (0.35, 0.90, 0.3, 0.85, 500, 0.06, 30.0, "spectral-heavy"),
            (0.35, 0.95, 0.15, 0.90, 800, 0.04, 10.0, "gradient-heavy"),
            (0.30, 0.85, 0.2, 0.88, 400, 0.10, 50.0, "aggressive"),
        ]

        for frac, g_init, lr_init, mom, rest, noise, t_init, desc in sub_phases:
            sub_time = frac * phase2_time
            if time.monotonic() > deadline - 60:
                break
            if sub_time < 10:
                break
            print(f"  2-{desc} ({sub_time:.0f}s)")
            a_best, obj_best = optimize_hybrid(
                global_a, sub_time, verbose=True, verbose_interval=5000,
                gamma_init=g_init, lr_init=lr_init, momentum=mom,
                restart_after=rest, noise_frac=noise,
                temp_init=t_init, temp_decay=0.9996
            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print(f"    -> improved to {obj_best:.8f}")
            else:
                print(f"    -> no improvement (best stays {global_best:.8f})")

    # --- Phase 3: Polish (25% of time) ---
    remaining = deadline - time.monotonic()
    phase3_time = 0.25 * remaining
    print(f"\nPhase 3: Polish ({phase3_time:.0f}s)")

    if global_a is not None and phase3_time > 10:
        # 3a: Coordinate descent polish
        polish_3a = phase3_time * 0.4
        print(f"  3a: Coordinate polish ({polish_3a:.0f}s)")
        a_best, obj_best = polish_coordinate(global_a, polish_3a, verbose=True)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.8f}")

        # 3b: Fine mirror descent (small steps, focus on top peaks)
        polish_3b = phase3_time * 0.35
        print(f"  3b: Fine gradient ({polish_3b:.0f}s)")
        a_best, obj_best = optimize_hybrid(
            global_a, polish_3b, verbose=True, verbose_interval=10000,
            gamma_init=0.97, lr_init=0.05, momentum=0.95,
            restart_after=1000, noise_frac=0.02,
            temp_init=3.0, temp_decay=0.9999
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.8f}")

        # 3c: Ultra-fine spectral
        polish_3c = phase3_time * 0.25
        print(f"  3c: Ultra-fine spectral ({polish_3c:.0f}s)")
        a_best, obj_best = optimize_hybrid(
            global_a, polish_3c, verbose=True, verbose_interval=10000,
            gamma_init=0.98, lr_init=0.03, momentum=0.92,
            restart_after=2000, noise_frac=0.01,
            temp_init=2.0, temp_decay=0.9999
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.8f}")

    # --- Try a larger sequence if time remains ---
    remaining = deadline - time.monotonic()
    if remaining > 300 and global_best > 1.505:
        print(f"\nBonus: Trying n=8192 ({remaining:.0f}s)")
        n_big = 8192
        S_big = np.sqrt(2.0 * n_big)
        # Initialize from parent by upsampling or using structure
        i = np.arange(n_big)
        a_big = 1.0 + 0.4 * np.cos(2 * np.pi * 7 * i / n_big) + \
                0.2 * np.cos(2 * np.pi * 23 * i / n_big)
        a_big = np.maximum(a_big, 0.01)
        a_big *= S_big / a_big.sum()
        ver_big = evaluate_sequence(a_big.tolist())
        print(f"  n=8192 init: obj={ver_big:.8f}")
        if ver_big < global_best:
            a_best, obj_best = optimize_hybrid(
                a_big, remaining - 60, verbose=True, verbose_interval=5000,
                gamma_init=0.92, lr_init=0.2, momentum=0.87,
                restart_after=500, noise_frac=0.05,
                temp_init=15.0, temp_decay=0.9996
            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print(f"  n=8192 final: obj={obj_best:.8f}")

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    # Ensure all values are valid floats
    final_a = [float(x) for x in global_a]

    # Verify
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: local_obj={global_best:.8f}, verified={ver:.8f}, n={len(final_a)}")

    if not np.isfinite(ver):
        # Fallback
        fallback = [1.0] * 256
        ver = evaluate_sequence(fallback)
        print(f"Fallback: obj={ver:.8f}")
        return fallback

    return final_a
