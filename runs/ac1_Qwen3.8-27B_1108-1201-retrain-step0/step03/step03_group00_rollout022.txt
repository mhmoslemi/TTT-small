"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization combining LSE-smoothed FISTA,
top-K adaptive subgradient with line search, active-set QP refinement
via scipy, and multi-scale exploration. Exploits reversal symmetry
and uses aggressive beta annealing for final polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max of convolution."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        k = int(np.argmax(c))
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        grad = np.zeros(n)
        grad[lo:hi+1] = 2.0 * a[k - np.arange(lo, hi + 1)]
        return grad
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(a, c, K, temperature=1.0):
    """Top-K weighted subgradient with softmax weighting."""
    n = len(a)
    K = min(K, len(c))
    top_idx = np.argsort(c)[::-1][:K]
    top_vals = c[top_idx]
    # Softmax weights over top-K
    w = np.exp((top_vals - top_vals.max()) / temperature)
    w = w / w.sum()
    # Build full weight vector
    w_full = np.zeros(len(c))
    for i, idx in enumerate(top_idx):
        w_full[idx] = w[i]
    # Compute gradient: sum_k w_k * 2*(Q_k a)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    grad[lo:hi+1] = 2.0 * a[k - np.arange(lo, hi + 1)]
    return grad


def objective_value(a):
    """Compute the actual objective 2*n*max_b/sum_a^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def run_fista_lse(a_init, n, S, max_time, rng, beta_init=10.0,
                  beta_rate=0.005, beta_max=5000.0, mode="lse"):
    """FISTA with LSE smoothing and adaptive step."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it

        if mode == "topk":
            K = min(1 + it // 100, 64)
            temp = max(0.01, 1.0 / (1.0 + it * 0.01))
            grad = topk_gradient(y, c, K, temp)
        elif mode == "exact":
            grad = exact_subgradient(y, c)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: for LSE, L = 2*beta*||c||_inf roughly
        if mode == "lse":
            step = 1.0 / (2.0 * beta * max(c.max(), 1e-10) * 0.5 + 1e-10)
        else:
            step = S / (gnorm * (1.0 + 0.01 * np.log1p(it)))
            step = min(step, 0.5)

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Stuck detection: inject noise
        if it - last_improve > 3000:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it

        # Periodic restart
        if it % 5000 == 0:
            decay = np.exp(-it / 30000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    return best_a, best


def run_topk_polish(a_init, n, S, max_time, rng):
    """Polishing with top-K subgradient and backtracking line search."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best_obj = objective_value(a)
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (a.sum() ** 2)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # Adaptive K: start with many, narrow to few
        K = max(1, min(128, 64 - it // 200))
        # Temperature: start warm, cool down
        temp = max(0.001, 2.0 * np.exp(-it * 0.005))
        grad = topk_gradient(a, c, K, temp)

        # Project gradient to tangent space (sum = 0)
        grad_tan = grad - np.mean(grad)

        gnorm = np.linalg.norm(grad_tan)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0
        a_old_obj = cur_obj
        feasible = True
        for _ in range(20):
            a_try = project_simplex(a - step * grad_tan, S)
            c_try = fft_convolve_self(a_try)
            obj_try = 2.0 * n * c_try.max() / (a_try.sum() ** 2)
            if obj_try < a_old_obj * (1.0 - 1e-6):
                a = a_try
                feasible = True
                break
            step *= 0.7
        else:
            # No improvement found; try smaller step or perturb
            if step < 1e-8:
                # Add small noise
                noise = rng.normal(0, 1e-5 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
            else:
                a = project_simplex(a - step * grad_tan, S)

        # Periodic symmetry averaging
        if it % 500 == 0:
            a_rev = a[::-1]
            a_avg = 0.5 * (a + a_rev)
            a_avg = normalize_sum(a_avg, n)
            obj_avg = objective_value(a_avg)
            if obj_avg < best_obj:
                best_obj = obj_avg
                best_a = a_avg.copy()
                a = a_avg.copy()

    return best_a, best_obj

        # Instead: compute the weighted gradient and use it directly
        w = np.zeros(len(c))
        active_vals = c[active]
        w_active = np.exp(active_vals - active_vals.max())
        w_active = w_active / w_active.sum()
        for i, idx in enumerate(active):
            w[idx] = w_active[i]

        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        grad_tan = grad - np.mean(grad)

        gnorm = np.linalg.norm(grad_tan)
        if gnorm < 1e-14:
            break

        # Line search
        step = 1.0 / (conductivity(gnorm, S, n))
        improved = False
        for _ in range(15):
            a_try = project_simplex(a - step * grad_tan, S)
            obj_try = objective_value(a_try)
            if obj_try < cur_obj * (1.0 - 1e-7):
                a = a_try
                improved = True
                break
            step *= 0.6
        if not improved:
            # Perturb
            a = project_simplex(a + rng.normal(0, 1e-4 * S / np.sqrt(n), n), S)

        if it % 200 == 0:
            a_rev = a[::-1]
            a_avg = 0.5 * (a + a_rev)
            a_avg = normalize_sum(a_avg, n)
            if objective_value(a_avg) < best_obj:
                best_obj = objective_value(a_avg)
                best_a = a_avg.copy()
                a = a_avg.copy()

    return best_a, best_obj


def conductivity(gnorm, S, n):
    """Heuristic Lipschitz constant estimate."""
    return max(gnorm * 2.0, 1.0)


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # U-shaped (good for flat convolution)
    inits.append(np.abs(tt - 0.5) * 2 + 0.1)
    inits.append(np.cos(np.pi * tt) ** 2 + 0.05)
    inits.append(1.0 / (0.1 + np.sin(np.pi * tt) ** 2))

    # Cosine family
    inits.append(1.0 + 0.5 * np.cos(2 * np.pi * tt))
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * tt) + 0.2 * np.cos(4 * np.pi * tt))
    inits.append(1.0 + 0.4 * np.cos(2 * np.pi * tt) - 0.2 * np.cos(4 * np.pi * tt) + 0.1 * np.cos(6 * np.pi * tt))

    # Gaussian centered
    inits.append(np.exp(-50 * (tt - 0.5) ** 2) + 0.05)

    # Bimodal (two peaks)
    inits.append(np.exp(-200 * (tt - 0.25) ** 2) + np.exp(-200 * (tt - 0.75) ** 2) + 0.05)

    # Random
    inits.append(rng.random(n) + 0.1)

    # Power law
    inits.append(tt ** 0.5 + (1 - tt) ** 0.5 + 0.1)

    return [np.maximum(x, 0.01) for x in inits]


def try_length(a_given, n_target, budget, rng):
    """Run optimization at a specific length, using a_given as warm start if compatible."""
    S = np.sqrt(2.0 * n_target)

    # Prepare initial sequence
    if a_given is not None and len(a_given) == n_target:
        a_init = normalize_sum(a_given, n_target)
    elif a_given is not None:
        # Interpolate to target length
        a_old = np.asarray(a_given, dtype=np.float64)
        t_old = np.linspace(0, 1, len(a_old))
        t_new = np.linspace(0, 1, n_target)
        a_interp = np.interp(t_new, t_old, a_old)
        a_init = normalize_sum(a_interp, n_target)
    else:
        a_init = generate_structured_inits(n_target, rng)[0]
        a_init = normalize_sum(a_init, n_target)

    # Phase 1: LSE FISTA
    t1 = budget * 0.4
    a1, _ = run_fista_lse(a_init, n_target, S, t1, rng,
                           beta_init=10.0, beta_rate=0.004, beta_max=3000.0, mode="lse")

    # Phase 2: Top-K polish
    t2 = budget * 0.3
    a2, obj2 = run_topk_polish(a1, n_target, S, t2, rng)

    # Phase 3: Active-set refinement
    remaining = budget - (time.monotonic() - (time.monotonic() - budget))
    # Just use remaining time
    elapsed = 0  # We'll track within the function
    a3, obj3 = run_active_set_qp(a2, n_target, S, budget * 0.25, rng)

    # Final: try high-beta LSE for a bit
    a4, _ = run_fista_lse(a3, n_target, S, budget * 0.05, rng,
                           beta_init=100.0, beta_rate=0.01, beta_max=10000.0, mode="lse")

    # Pick best
    candidates = [a1, a2, a3, a4]
    best_a = None
    best_obj = np.inf
    for ca in candidates:
        ca = normalize_sum(ca, n_target)
        obj = objective_value(ca)
        if obj < best_obj:
            best_obj = obj
            best_a = ca.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(10.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Get given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Starting optimization, budget=%.0fs" % budget_s)
    if given is not None:
        print("Given sequence length: %d" % len(given))
        given_obj = objective_value(given)
        print("Given sequence objective: %.6f" % given_obj)
        if given_obj < global_best:
            global_best = given_obj
            global_a = given.copy()

    # Strategy: try multiple lengths, allocate time wisely
    # Primary length: same as given or 8192
    if given is not None:
        primary_n = len(given)
    else:
        primary_n = 8192

    # Allocate time across phases
    # 50% on primary length, 20% on secondary, 30% on polishing best
    t_primary = max(60.0, (deadline - time.monotonic()) * 0.50)
    t_secondary = max(30.0, (deadline - time.monotonic() - t_primary) * 0.40)
    t_polish = max(30.0, (deadline - time.monotonic() - t_primary - t_secondary))

    print("Phase allocation: primary=%.0fs, secondary=%.0fs, polish=%.0fs" %
          (t_primary, t_secondary, t_polish))

    # === Phase 1: Primary length optimization ===
    print("=== Phase 1: Primary n=%d ===" % primary_n)
    p1_start = time.monotonic()

    # Run 1: Full pipeline
    a1, obj1 = try_length(given, primary_n, t_primary * 0.6, rng)
    if a1 is not None and obj1 < global_best:
        global_best = obj1
        global_a = a1.copy()
        print("  Run 1: obj=%.6f (elapsed %.0fs)" % (obj1, time.monotonic() - p1_start))

    # Run 2: Different mode (topk from start)
    remaining_p1 = max(10.0, p1_start + t_primary - time.monotonic())
    if remaining_p1 > 20:
        S1 = np.sqrt(2.0 * primary_n)
        if global_a is not None and len(global_a) == primary_n:
            a_init2 = normalize_sum(global_a, primary_n)
        elif given is not None:
            a_init2 = normalize_sum(given[:primary_n] if len(given) >= primary_n else given, primary_n)
            while len(a_init2) < primary_n:
                a_init2 = np.concatenate([a_init2, a_init2])
                a_init2 = a_init2[:primary_n]
        else:
            a_init2 = normalize_sum(generate_structured_inits(primary_n, rng)[1], primary_n)

        a2, _ = run_fista_lse(a_init2, primary_n, S1, remaining_p1 * 0.5, rng,
                               beta_init=5.0, beta_rate=0.006, beta_max=5000.0, mode="topk")
        obj2 = objective_value(a2)
        if obj2 < global_best:
            global_best = obj2
            global_a = a2.copy()
            print("  Run 2 (topk): obj=%.6f" % obj2)

    print("  After Phase 1: best=%.6f, elapsed=%.0fs" % (global_best, time.monotonic() - p1_start))

    # === Phase 2: Try secondary length ===
    if global_a is not None:
        print("=== Phase 2: Secondary length ===")
        p2_start = time.monotonic()
        # Try a different length
        if primary_n >= 8192:
            alt_n = primary_n // 2
        else:
            alt_n = primary_n * 2
        alt_n = max(1024, min(alt_n, 32768))
        if alt_n != primary_n and t_secondary > 30:
            a_alt, obj_alt = try_length(global_a, alt_n, t_secondary, rng)
            if a_alt is not None and obj_alt < global_best:
                global_best = obj_alt
                global_a = a_alt.copy()
                print("  Alt n=%d: obj=%.6f" % (alt_n, obj_alt))
        print("  After Phase 2: best=%.6f, elapsed=%.0fs" % (global_best, time.monotonic() - p2_start))

    # === Phase 3: Intensive polishing ===
    if global_a is not None and t_polish > 15:
        print("=== Phase 3: Polishing (%.0fs) ===" % t_polish)
        p3_start = time.monotonic()
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_polish = normalize_sum(global_a, n_best)

        # Sub-phase 3a: Top-K with line search
        t_3a = t_polish * 0.4
        a_3a, obj_3a = run_topk_polish(a_polish, n_best, S_best, t_3a, rng)
        if obj_3a < global_best:
            global_best = obj_3a
            global_a = a_3a.copy()
            a_polish = a_3a.copy()
            print("  After 3a: obj=%.6f" % global_best)

        # Sub-phase 3b: Active set QP
        remaining_3 = max(10.0, p3_start + t_polish - time.monotonic())
        t_3b = remaining_3 * 0.4
        a_3b, obj_3b = run_active_set_qp(a_polish, n_best, S_best, t_3b, rng)
        if obj_3b < global_best:
            global_best = obj_3b
            global_a = a_3b.copy()
            a_polish = a_3b.copy()
            print("  After 3b: obj=%.6f" % global_best)

        # Sub-phase 3c: Very high beta LSE (ultra-fine)
        remaining_3b = max(5.0, p3_start + t_polish - time.monotonic())
        t_3c = remaining_3b * 0.5
        if t_3c > 5:
            a_3c, _ = run_fista_lse(a_polish, n_best, S_best, t_3c, rng,
                                     beta_init=200.0, beta_rate=0.02, beta_max=20000.0, mode="lse")
            obj_3c = objective_value(a_3c)
            if obj_3c < global_best:
                global_best = obj_3c
                global_a = a_3c.copy()
                print("  After 3c (ultra LSE): obj=%.6f" % global_best)

        # Sub-phase 3d: Symmetry averaging + final check
        if global_a is not None:
            a_sym = 0.5 * (global_a + global_a[::-1])
            a_sym = normalize_sum(a_sym, n_best)
            obj_sym = objective_value(a_sym)
            if obj_sym < global_best:
                global_best = obj_sym
                global_a = a_sym.copy()
                print("  After symmetry: obj=%.6f" % global_best)

        print("  After Phase 3: best=%.6f, elapsed=%.0fs" % (global_best, time.monotonic() - p3_start))

    # Final verification
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)
        global_a *= np.sqrt(2 * 256) / 256

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    # Verify with evaluate_sequence
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Final verification: %.6f" % ver if np.isfinite(ver) else "Verification: INF")
    print("Sequence length: %d" % len(final_list))

    if not np.isfinite(ver):
        print("evaluate_sequence rejected! Using fallback.")
        fallback = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in fallback]

    return final_list
