For a sequence a of length n, the convolution a*a has length 2n-1. The sum of all convolution values is (sum a)^2. If the convolution were perfectly flat, each value would be (sum a)^2/(2n-1), and the objective would be 2n/(2n-1) ≈ 1.

The "flatness" of the convolution is measured by the ratio max(c)/mean(c) = max(c)*(2n-1)/(sum a)^2. The objective is 2n * max(c)/(sum a)^2 = (2n/(2n-1)) * max(c)/mean(c).

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w

Minimize 2*n*max(self_convolution) / sum^2 over non-negative sequences.

Algorithm: Multi-phase optimization with log-sum-exp smoothing and annealing.
Phase 1: FISTA on smoothed objective (log-sum-exp of convolution) with temperature annealing.
Phase 2: Exact subgradient on peak lags with Nesterov acceleration.
Phase 3: Ultra-fine polishing on the exact max.
Multiple initializations: parent sequence, raised cosine, binomial, and random.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def logsumexp_conv(a, T):
    """Compute log-sum-exp of self-convolution / T. Returns (value, weights)."""
    c = fft_convolve_self(a)
    c_max = c.max()
    if T < 1e-10:
        return c_max, None
    exp_vals = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_vals.sum())
    weights = exp_vals / exp_vals.sum()
    return lse, weights


def compute_grad_from_weights(a, weights):
    """Compute gradient of sum_k weights[k] * c[k] w.r.t. a.
    Uses FFT cross-correlation: grad[i] = sum_k weights[k] * a[k-i]
    """
    n = len(a)
    size_c = 2 * n - 1
    N = 1 << (size_c + n - 1).bit_length()
    W = fft(weights, N)
    A = fft(a, N)
    # grad[i] = sum_k W[k] * A[k-i] = cross-correlation of W and A at lag i
    # = ifft(conj(fft(W)) * fft(A))[i]  ... wait, let me re-derive.
    # Actually: sum_k W[k] * A[k-i]
    # Let's define this as (W reversed) * A convolution... 
    # Or: it's the cross-correlation (W ⋆ A)[i] where (x ⋆ y)[i] = sum_k x[k] * y[k-i]
    # DFT of cross-correlation: DFT(x ⋆ y) = conj(X) * Y (with appropriate zero-padding)
    # So: cross_corr = ifft(conj(fft(W, N)) * fft(A, N))
    grad = np.real(ifft(np.conj(W) * A))[:n]
    return grad
