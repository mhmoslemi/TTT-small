"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines LSE-FISTA, spectral shaping (Gerchberg-Saxton inspired),
and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def lse_objective_and_grad(a, beta):
    """Compute LSE approximation of max(conv(a,a)) and its gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    lse_val = c_max + np.log(total) / beta
    if total < 1e-300:
        grad = np.zeros(n)
    else:
        w = exp_bc / total
        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad

def run_fista_spectral(a_init, n, S, max_time, rng, beta_init=1.0, beta_max=500.0):
    """FISTA with LSE + periodic spectral shaping."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = 2.0
    stall = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))
        
        # Compute gradient at y
        lse_val, grad = lse_objective_and_grad(y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # FISTA update with backtracking
        step = 1.0 / L
        for bt in range(4):
            x_new = project_simplex(y - step * grad, S)
            if bt > 0:
                lse_new, _ = lse_objective_and_grad(x_new, beta)
                if lse_new <= lse_val - 0.5 * step * gnorm**2:
                    break
                step *= 0.5
            else:
                x_new = project_simplex(y - step * grad, S)
        
        # Update L
        lse_x, _ = lse_objective_and_grad(x_new, beta)
        if lse_x > lse_val + 1e-6:
            L *= 1.5
        elif it % 50 == 0:
            L = max(L * 0.9, beta * 1.5)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum**2))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new
        
        # Periodic spectral shaping
        if it % 100 == 0:
            A = fft(a, n)
            mag = np.abs(A)
            phase = np.angle(A)
            # Flatten: replace magnitude with interpolated smooth version
            mag_target = np.ones(n) * (mag.sum() / n)
            # Blend
            alpha = 0.1
            mag_new = (1 - alpha) * mag + alpha * mag_target
            a_spec = np.real(ifft(mag_new * np.exp(1j * phase)))
            a_spec = np.maximum(a_spec, 0.0)
            if a_spec.sum() > 1e-10:
                a_spec = a_spec * (S / a_spec.sum())
                # Check if spectral step improves
                c_spec = fft_convolve_self(a_spec)
                if c_spec.max() < c.max():  # where c is current conv
                    a = a_spec
                    y = a_spec.copy()
                    t_momentum = 1.0
        
        # Track best
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Perturbation if stalled
        if stall > 50 and rng.random() < 0.2:
            noise = 0.02 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_momentum = 1.0
            stall = 0
    
    return best_local, best_score

def run_coordinate_descent(a_init, n, S, max_time, rng, top_k=20):

    # a[i] affects conv[k] for k where a[i] appears, i.e., k = i+j for all j (as a[i]*a[j])
    # and k = j-i for all j > i (as a[j]*a[i])
    # So a[i] appears in conv[k] for k = 0 to 2(n-1), specifically:
    # conv[k] contains a[i]*a[k-i] (if 0<=k-i<n) and a[k-i]*a[i] (same thing for self-conv)
    # Actually conv[k] = sum_j a[j]*a[k-j], so a[i] appears when j=i or k-j=i
    # i.e., when k-i is in [0,n-1] (j=i, a[k-i]) or j=k-i is in [0,n-1] (a[i]*a[k-i])
    # These are the same condition: 0 <= k-i < n, i.e., i <= k < i+n
    
    # For coordinate descent, fixing all a[j] for j != i:
    # conv[k] = terms_not_involving_i + a[i] * (sum of relevant a's)
    # Specifically, conv[k] = B[k] + a[i] * C[k] where B and C don't depend on a[i]
    # B[k] = conv[k] - a[i]*(a[k-i] if valid + a[i-k] if valid and i-k != k-i)
    # C[k] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n, and i-k != k-i)
    # For self-conv, a[k-i] and a[i-k] are the same when k-i = i-k, i.e., k=2i.

        c = fft_convolve_self(a)

            best_local = a.copy()

        worst_k = int(c.argmax())

        # For lag k, conv[k] = sum_j a[j]*a[k-j]
        # To reduce conv[k], we want to reduce a[j]*a[k-j] for large products
        # Strategy: reduce the larger of each pair, increase the smaller

            # Reduce largest, increase smallest

                # Reduce a[max_j + k], increase a[max_j]

def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic initialization."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())
    return a

def make_sinusoidal_init(n, rng, S, freq=None):
    """Sinusoidal initialization."""
    if freq is None:
        freq = rng.integers(3, 15)
    tt = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    a = np.maximum(a, 0.05)
    return a * (S / a.sum())

def make_random_init(n, rng, S):
    """Random initialization."""
    a = rng.random(n) + 0.05
    return a * (S / a.sum())

def spectral_flatten(a, S, alpha=0.3):
    """Spectral flattening: blend magnitude with flat spectrum."""
    n = len(a)
    A = fft(a, n)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_flat = np.ones(n) * (mag.sum() / n)
    # Use a softer flattening: blend in log domain
    log_mag = np.log(mag + 1e-10)
    log_flat = np.log(mag_flat + 1e-10)
    log_new = (1 - alpha) * log_mag + alpha * log_flat
    mag_new = np.exp(log_new)
    a_new = np.real(ifft(mag_new * np.exp(1j * phase)))
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (S / a_new.sum())
    return a_new

4. Spectral shaping (flat spectrum target) for diversity
5. Multi-start with diverse initializations including anti-palindromic structures

Key insight: The objective is scale-invariant, so we normalize to sum=sqrt(2n)
making the objective equal to max(convolution). We minimize this peak.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_val_and_grad(a, c, beta):
    """Compute LSE approximation of max(c) and gradient w.r.t. a via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    lse_val = c_max + np.log(total) / beta
    if total < 1e-300:
        grad = np.zeros(n)
    else:
        w = exp_bc / total
        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad


def subgrad_topk(a, c, k_list, weights=None):
    """Subgradient using weighted sum of top-k lag contributions."""
    n = len(a)
    w_full = np.zeros(2 * n - 1)
    if weights is None:
        weights = np.ones(len(k_list)) / len(k_list)
    for idx, k in enumerate(k_list):
        w_full[int(k)] = weights[idx]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista_spectral(a_init, n, S, max_time, rng, beta_init=2.0, beta_max=500.0,
                       spectral_interval=80, spectral_alpha=0.15):
    """FISTA with LSE smoothing and periodic spectral flattening."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    stall = 0
    it = 0
    L = max(beta_init * 2.0, 4.0)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.008 * it))

        c_y = fft_convolve_self(y)
        lse_y, grad = lse_val_and_grad(y, c_y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step = 1.0 / L
        x_new = None
        for bt in range(4):
            candidate = project_simplex(y - step * grad, S)
            if bt == 0:
                x_new = candidate
            else:
                c_cand = fft_convolve_self(candidate)
                lse_cand, _ = lse_val_and_grad(candidate, c_cand, beta)
                if lse_cand <= lse_y - 1e-4 * step * gnorm * gnorm:
                    x_new = candidate
                    break
                step *= 0.5

        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        # Adaptive L
        c_x = fft_convolve_self(x_new)
        lse_x, _ = lse_val_and_grad(x_new, c_x, beta)
        if lse_x > lse_y + 1e-8:
            L *= 1.3
        elif it % 30 == 0:
            L = max(L * 0.92, beta * 1.5)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic spectral flattening
        if it % spectral_interval == 0 and rng.random() < 0.4:
            A_fft = fft(a, n)
            mag = np.abs(A_fft) + 1e-12
            phase = np.angle(A_fft)
            log_mag = np.log(mag)
            log_target = np.full(n, np.log(mag.sum() / n) + 1e-12)
            log_blend = (1.0 - spectral_alpha) * log_mag + spectral_alpha * log_target
            mag_new = np.exp(log_blend)
            a_spec = np.real(ifft(mag_new * np.exp(1j * phase)))
            a_spec = np.maximum(a_spec, 0.0)
            if a_spec.sum() > 1e-10:
                a_spec = a_spec * (S / a_spec.sum())
                c_spec = fft_convolve_self(a_spec)
                if c_spec.max() < c_x.max():
                    a = a_spec
                    y = a_spec.copy()
                    t_mom = 1.0

        # Track best
        c_a = fft_convolve_self(a)
        cur = c_a.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()
            stall = 0
        else:
            stall += 1

        # Perturbation to escape local minima
        if stall > 60 and rng.random() < 0.25:
            scale = 0.01 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, scale, n), S)
            t_mom = 1.0
            stall = 0

    return best_local, best_score


def run_barankan(a_init, n, S, max_time, rng, top_k_init=3, top_k_max=64):
    """Barankan-style method: subgradient over active lags with backtracking."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step_size = 2.0 * S / (n * 10.0)

    while time.monotonic() - start < max_time:
        it += 1
        k_use = min(top_k_max, top_k_init + it // 300)

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Select active lags (top k and those near peak)
        sorted_idx = np.argsort(c)[::-1]
        top_idx = sorted_idx[:k_use]

        # Weighted subgradient: weight by how close each lag is to max
        vals = c[top_idx]
        v_min = vals.min()
        v_max = vals.max()
        if v_max - v_min < 1e-12:
            wts = np.ones(k_use) / k_use
        else:
            wts = (vals - v_min) / (v_max - v_min)
            wts = wts ** 2  # Emphasize peaks more
            wts = wts / wts.sum()

        grad = subgrad_topk(a, c, top_idx, weights=wts)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking
        a_new = None
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            candidate = project_simplex(a - factor * step_size * grad / gnorm * S / n, S)
            c_cand = fft_convolve_self(candidate)
            if c_cand.max() < cur - 1e-10:
                a_new = candidate
                break

        if a_new is None:
            # Try positive direction too
            for factor in [0.5, 0.25, 0.1]:
                candidate = project_simplex(a + factor * step_size * grad / gnorm * S / n, S)
                c_cand = fft_convolve_self(candidate)
                if c_cand.max() < cur - 1e-10:
                    a_new = candidate
                    break

        if a_new is None:
            # Stuck: perturb
            if rng.random() < 0.3:
                noise = 0.008 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)
            continue

        a = a_new

        # Occasionally increase step size
        if it % 500 == 0:
            step_size *= 1.2
        # Decrease if no improvement recently
        if it % 100 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() > best_score * 1.01:
                step_size *= 0.7

    return best_local, best_score


def run_coordinate_descent(a_init, n, S, max_time, rng):
    """
    Coordinate descent via pair-flip: target the worst lag,
    reduce the largest product pair, compensate elsewhere.
    O(n) per step, no FFT needed for the decision.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    no_improve = 0

    c = fft_convolve_self(a)
    best_score = c.max()

    while time.monotonic() - start < max_time:
        it += 1

        # Find worst lag
        worst_k = int(c.argmax())

        # Determine pairs contributing to this lag
        if worst_k == 0:
            # conv[0] = sum a[i]^2 -> make more uniform
            idx_max = int(a.argmax())
            idx_min = int(a.argmin())
            if a[idx_max] > a[idx_min] * 1.01:
                delta = 0.03 * (a[idx_max] - a[idx_min])
                a[idx_max] -= delta
                a[idx_min] += delta
                a = np.maximum(a, 0.0)
                a = a * (S / a.sum())
        elif worst_k < n:
            # conv[k] = sum_{j=0}^{n-1-k} a[j+k] * a[j]
            j_range = np.arange(n - worst_k)
            products = a[j_range + worst_k] * a[j_range]
            max_j = int(j_range[np.argmax(products)])
            i1 = max_j
            i2 = max_j + worst_k
            # Reduce the larger, increase the smaller
            if a[i1] >= a[i2]:
                frac = 0.02 + 0.03 * (1.0 - it / max(1, it + 1000))
                delta = frac * a[i1]
                a[i1] -= delta
                a[i2] += delta
            else:
                frac = 0.02 + 0.03 * (1.0 - it / max(1, it + 1000))
                delta = frac * a[i2]
                a[i2] -= delta
                a[i1] += delta
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
        else:
            # By symmetry: conv[k] = conv[2(n-1)-k]
            k_sym = 2 * (n - 1) - worst_k
            if k_sym < n:
                j_range = np.arange(n - k_sym)
                products = a[j_range + k_sym] * a[j_range]
                max_j = int(j_range[np.argmax(products)])
                i1 = max_j
                i2 = max_j + k_sym
                if a[i1] >= a[i2]:
                    delta = 0.03 * a[i1]
                    a[i1] -= delta
                    a[i2] += delta
                else:
                    delta = 0.03 * a[i2]
                    a[i2] -= delta
                    a[i1] += delta
                a = np.maximum(a, 0.0)
                a = a * (S / a.sum())

        # Periodic full evaluation (expensive but necessary)
        if it % 200 == 0 or it < 10:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # If stuck, try a different strategy
            if no_improve > 5 and it % 200 == 0:
                # Try random pair flip
                for _ in range(10):
                    i1 = rng.integers(n)
                    i2 = rng.integers(n)
                    if i1 == i2:
                        continue
                    delta = 0.01 * S / n
                    old_i1, old_i2 = a[i1], a[i2]
                    a[i1] = old_i1 + delta
                    a[i2] = old_i2 - delta
                    if a[i1] < 0 or a[i2] < 0:
                        a[i1], a[i2] = old_i1, old_i2
                        continue
                    a = a * (S / a.sum())
                    c_test = fft_convolve_self(a)
                    if c_test.max() < best_score:
                        best_score = c_test.max()
                        best_local = a.copy()
                        c = c_test
                        break
                    else:
                        a[i1], a[i2] = old_i1, old_i2
                        a = a * (S / a.sum())

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.05
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())
    return a


def make_multi_sin_init(n, rng, S, n_freqs=3):
    """Multi-frequency sinusoidal initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(n_freqs):
        freq = rng.integers(2, 20)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.35
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.05)
    return a * (S / a.sum())


def make_w_shape_init(n, rng, S):
    """W/M-shaped initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    k = rng.integers(2, 6)
    a = np.abs(np.sin(2 * np.pi * k * tt)) + 0.05
    return a * (S / a.sum())


def make_random_init(n, rng, S):
    """Random positive initialization."""
    a = rng.random(n) + 0.05
    return a * (S / a.sum())


def spectral_flatten(a, S, alpha=0.4):
    """Blend magnitude spectrum toward flat."""
    n = len(a)
    A = fft(a, n)
    mag = np.abs(A) + 1e-12
    phase = np.angle(A)
    log_mag = np.log(mag)
    target_log = np.full(n, np.log(mag.sum() / n))
    log_new = (1.0 - alpha) * log_mag + alpha * target_log
    a_new = np.real(ifft(np.exp(log_new) * np.exp(1j * phase)))
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (S / a_new.sum())
    return a_new


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

    2*n*max(conv(a,a))/sum(a)^2.
