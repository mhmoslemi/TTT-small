```python
def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    # Binary search for threshold
    lo = v.min()
    hi = v.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(v - mid, 0))
        if s > S:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    return np.maximum(v - theta, 0.0)
```

Actually, the binary search approach for simplex projection is O(n * iterations) which is fine. But the sort-based approach is O(n log n) and more standard. Let me use the sort-based one.

```python
def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1)
    theta = (cumsum - S) / idx
    # Find largest rho such that theta[rho-1] <= w[rho-1]
    valid = theta <= w
    rho = np.max(np.where(valid)[0]) + 1 if np.any(valid) else n
    theta_val = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta_val, 0.0)
```

```python
def cross_correlate(a, U, n):
```

def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        theta_val = 0.0
    else:
        rho = int(np.max(np.where(valid)[0])) + 1
        theta_val = (cumsum[rho-1] - S) / rho
    result = np.maximum(v - theta_val, 0.0)
    # Ensure exact sum (handle floating point)
    ss = result.sum()
    if ss > 0:
        result = result * (S / ss)
    return result

    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]
