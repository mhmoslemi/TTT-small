def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    # If sum is already S, just clip
    # Standard simplex projection
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def project_symmetric(v, S, n):
    """Project onto {a >= 0, a symmetric, sum(a) = S}.
    v is the full sequence. Returns the projected full sequence.
    For symmetric a: a[i] = a[n-1-i].
    We project b = a[0:n_half] onto the appropriate simplex.
    """
    if n % 2 == 0:
        n_half = n // 2
        # sum(a) = 2 * sum(b) = S, so sum(b) = S/2
        b = v[:n_half].copy()
        b = project_simplex(b, S / 2.0)
        a = np.empty(n)
        a[:n_half] = b
        a[n_half:] = b[::-1]
        return a
    else:
        n_half = n // 2
        # sum(a) = 2*sum(b[0:n_half]) + b[n_half] = S
        # This is more complex. For simplicity, just project the full sequence
        # and then symmetrize: a[i] = (a[i] + a[n-1-i]) / 2
        a = project_simplex(v, S)
        a = (a + a[::-1]) / 2.0
        a = a * (S / a.sum())
        return a


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_peak_grad(a, k_peak, n):
    """Gradient of c[k_peak] w.r.t. a, where c = a*a (full convolution)."""
    j = np.arange(n)
    idx = k_peak - j
    valid = (idx >= 0) & (idx < n)
    grad = np.zeros(n)
    grad[valid] = 2.0 * a[idx[valid]]
    # Fix diagonal
    if k_peak % 2 == 0:
        jmid = k_peak // 2
        if 0 <= jmid < n:
            grad[jmid] = a[jmid]
    return grad


def run_nesterov(a_init, n, S, max_time, rng, use_symmetry=True, top_k=1):
    """Nesterov accelerated gradient descent for peak minimization."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    a = a * (S / max(a.sum(), 1e-15))
    if use_symmetry:
        a = (a + a[::-1]) / 2.0
        a = a * (S / a.sum())
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = a.copy()
    best_score = np.inf
    it = 0
    stall = 0
    momentum = 0.0  # will be set by Nesterov
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        
        if cur < best_score - 1e-10:
            best_score = cur
            best = y.copy()
            stall = 0
        else:
            stall += 1
        
        if it % 500 == 0:
            remaining = max_time - (time.monotonic() - start)
            print(f"    iter={it}, max_c={cur:.8f}, best={best_score:.8f}, stall={stall}, remain={remaining:.0f}s")
        
        # Compute gradient
        if top_k == 1:
            k_peak = c.argmax()
            grad = compute_peak_grad(y, k_peak, n)
        else:
            top_idx = np.argpartition(c, -top_k)[-top_k:]
            w = c[top_idx]
            w = w / w.sum()
            grad = np.zeros(n)
            for idx, wi in zip(top_idx, w):
                grad += wi * compute_peak_grad(y, int(idx), n)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Gradient step with backtracking
        step = 1.0 / (gnorm * gnorm + 1e-10) * S
        # Actually, let's use a simpler step size
        step = 2.0 * S / (n * gnorm)
        
        x_new = y
        c_new_max = cur
        for _ in range(20):
            x_try = y - step * grad
            if use_symmetry:
                x_try = (x_try + x_try[::-1]) / 2.0
            x_try = project_simplex(x_try, S)
            c_try = fft_convolve_self(x_try)
            if c_try.max() < c_new_max - 1e-12:
                c_new_max = c_try.max()
                x_new = x_try
                break
            step *= 0.5
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        if use_symmetry:
            y = (y + y[::-1]) / 2.0
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic momentum reset
        if stall > 300:
            t = 1.0
            y = a.copy()
        
        # Random perturbation for diversity
        if stall > 500 and rng.random() < 0.2:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = a + noise
            if use_symmetry:
                y = (y + y[::-1]) / 2.0
            y = project_simplex(y, S)
            t = 1.0
            stall = 0
    
    return best, best_score
