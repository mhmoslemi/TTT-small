"""

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_weighted_subgrad(c, a, T):
    """Compute weighted subgradient using exponential peak weighting."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n), 0.0

    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # g[m] = 2 * cross_corr(w, a)[m]
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n], w


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def peak_flatness(c):
    """Measure how flat the top peaks are. Lower = more concentrated."""
    c_sorted = np.sort(c)[::-1]
    top_k = c_sorted[:max(10, len(c) // 100)]
    return top_k.mean() / (c_sorted[0] + 1e-30)


def run_weighted_subgrad(a_init, n, S, time_budget, T_init=20.0, T_final=1000.0,
                         lr_init=0.08, momentum=0.85, use_nesterov=True):

    Weighted subgradient descent with Nesterov momentum and adaptive temperature.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    a_prev = a.copy()
    beta = momentum
    gamma = 1.0  # Nesterov parameter
    
    start = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)

        else:

        cur_obj = c.max()

            best_a = a.copy()
            no_improve = 0
        else:

def run_mirror_descent(a_init, n, S, time_budget, T_init=20.0, T_final=500.0,
                       lr_init=0.05):
    """
    Multiplicative update (mirror descent) for non-negative optimization.
    Update: a[i] <- a[i] * exp(-lr * (g[i] - mean(g[a>0]))) / (S / sum(a))
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    a = np.maximum(a, 1e-10)  # Avoid log(0)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    step = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_final - 0.5 * (T_final - T_init) * (1 + np.cos(np.pi * progress))

        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            break

        g, w = compute_weighted_subgrad(c, a, T)
        
        # Center the gradient (remove the component along the all-ones vector)
        # This ensures the sum is preserved in the multiplicative update
        g_centered = g - g.mean()
        
        lr = lr_init * (1.0 - 0.3 * progress)
        
        # Multiplicative update
        # a_new[i] = a[i] * exp(-lr * g_centered[i])
        # This automatically preserves non-negativity
        log_update = -lr * g_centered
        # Clip to avoid overflow
        log_update = np.clip(log_update, -50, 50)
        a = a * np.exp(log_update)
        a = normalize_nonneg(a, S)
        a = np.maximum(a, 1e-10)

        step += 1

    return best_a, best_obj

    start = time.monotonic()

        sweep_improved = False

        for i in order:

            if original < 1e-12:
                continue

            # Golden section search around the current value
            best_val = original
            best_local_obj = best_obj

            # Try a range of values
            for factor in [0.5, 0.7, 1.3, 1.5, 0.3, 2.0, 0.8, 1.2]:
                trial = original * factor
                a[i] = trial
                a_norm = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-13:
                    best_local_obj = obj_check
                    best_val = trial

            a[i] = best_val
            a = normalize_nonneg(a, S)

            if best_obj - a[i] * 0 > 0:  # placeholder
                pass
            
            c = fft_convolve_self(a)
            if c.max() < best_obj - 1e-12:
                best_obj = c.max()
                best_a = a.copy()
                sweep_improved = True

        if not sweep_improved:
            converged_sweeps += 1
            if converged_sweeps >= 3:

        else:

    elif name == "wavy":
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * i / n) + 0.1 * np.sin(2 * np.pi * 17 * i / n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def try_sparsification(a, n, S, n_zero_target=50):
    """Try setting small coordinates to zero."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    sorted_idx = np.argsort(a)
    for idx in sorted_idx[:min(n_zero_target * 5, n)]:
        if a[idx] < 1e-10:
            continue
        saved = a[idx]
        a[idx] = 0.0
        a = normalize_nonneg(a, S)
        c_check = fft_convolve_self(a)
        obj_check = c_check.max()
        if obj_check < best_obj - 1e-13:
            best_obj = obj_check
            best_a = a.copy()
        else:
            a[idx] = saved
            a = normalize_nonneg(a, S)

    return best_a, best_obj


# Global holder for deadline (to avoid lambda/closure)
deadline_holder = [0.0]


def propose_candidate(seed=42, budget_s=2200, **kwargs):
    """
    Multi-phase search for a non-negative sequence minimizing the evaluation function.
    
    Phases:
    1. Quick multi-start exploration (15% time)
    2. Deep optimization with alternating strategies (50% time)
    3. Coordinate descent polishing (20% time)
    4. Peak-focused refinement and sparsification (15% time)
    """
    global deadline_holder
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 40.0)
    overall_deadline = time.monotonic() + total_budget
    deadline_holder[0] = overall_deadline

    global_best = np.inf
    global_a = None

    # Load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: " + str(e))

    n = 4096
    S = np.sqrt(2.0 * n)

    # ============ PHASE 1: Exploration (15% time) ============
    explore_budget = 0.15 * total_budget
    explore_deadline = time.monotonic() + explore_budget
    print("Phase 1: Exploration (%.0fs)" % explore_budget)

    init_names = ["uniform", "random", "cosine_mix", "gaussian", "wavy", "broad_gaussian"]
    candidates_to_explore = []
    if given_a is not None:
        candidates_to_explore.append(("given", given_a.copy()))
    for name in init_names:
        candidates_to_explore.append((name, get_initialization(name, n, S, rng)))

    per_attempt = max(3.0, explore_budget / len(candidates_to_explore) * 0.85)

    for name, a_init in candidates_to_explore:
        if time.monotonic() > explore_deadline - 3:
            break

        a_init = normalize_nonneg(a_init, S)
        half_time = per_attempt / 2

        # Quick: subgradient then mirror descent
        a1, _ = run_weighted_subgrad(a_init, n, S, half_time,
                                      T_init=10, T_final=200, lr_init=0.1)
        a2, _ = run_mirror_descent(a1, n, S, half_time,
                                    T_init=10, T_final=200, lr_init=0.05)

        c1 = fft_convolve_self(a1)
        c2 = fft_convolve_self(a2)
        a_best = a2 if c2.max() < c1.max() else a1
        a_best = normalize_nonneg(a_best, S)

        final_obj = evaluate_sequence(a_best.tolist())
        if np.isfinite(final_obj) and final_obj < global_best:
            global_best = final_obj
            global_a = a_best.copy()
            print("  Explore [%s]: obj=%.7f" % (name, final_obj))
        else:
            obj_disp = final_obj if np.isfinite(final_obj) else 99.0
            print("  Explore [%s]: obj=%.7f (not better)" % (name, obj_disp))

    # ============ PHASE 2: Deep optimization (50% time) ============
    exploit_budget = 0.50 * total_budget
    exploit_deadline = time.monotonic() + exploit_budget

    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = global_a.copy()
        a = normalize_nonneg(a, S)
        print("\nPhase 2: Deep optimization n=%d (budget=%.0fs)" % (n, exploit_budget))

        # Strategy 1: High-T subgradient (focus on dominant peak)
        t1 = exploit_budget * 0.22
        a1, _ = run_weighted_subgrad(a, n, S, t1,
                                      T_init=100, T_final=3000, lr_init=0.04, momentum=0.9)
        a1 = normalize_nonneg(a1, S)
        obj1 = evaluate_sequence(a1.tolist())
        if np.isfinite(obj1) and obj1 < global_best:
            global_best = obj1
            global_a = a1.copy()
            a = a1.copy()
            print("  P2a Subgrad T100-3000: obj=%.7f" % obj1)
        else:
            print("  P2a Subgrad T100-3000: no improvement")

        # Strategy 2: Mirror descent medium T
        t2 = exploit_budget * 0.18
        a2, _ = run_mirror_descent(a, n, S, t2,
                                    T_init=20, T_final=500, lr_init=0.04)
        a2 = normalize_nonneg(a2, S)
        obj2 = evaluate_sequence(a2.tolist())
        if np.isfinite(obj2) and obj2 < global_best:
            global_best = obj2
            global_a = a2.copy()
            a = a2.copy()
            print("  P2b Mirror T20-500: obj=%.7f" % obj2)
        else:
            print("  P2b Mirror T20-500: no improvement")

        # Strategy 3: Low-T subgradient (broader smoothing)
        t3 = exploit_budget * 0.18
        a3, _ = run_weighted_subgrad(a, n, S, t3,
                                      T_init=5, T_final=100, lr_init=0.1, momentum=0.7)
        a3 = normalize_nonneg(a3, S)
        obj3 = evaluate_sequence(a3.tolist())
        if np.isfinite(obj3) and obj3 < global_best:
            global_best = obj3
            global_a = a3.copy()
            a = a3.copy()
            print("  P2c Subgrad T5-100: obj=%.7f" % obj3)
        else:
            print("  P2c Subgrad T5-100: no improvement")

        # Strategy 4: Mirror descent high T
        t4 = exploit_budget * 0.17
        a4, _ = run_mirror_descent(a, n, S, t4,
                                    T_init=100, T_final=2000, lr_init=0.03)
        a4 = normalize_nonneg(a4, S)
        obj4 = evaluate_sequence(a4.tolist())
        if np.isfinite(obj4) and obj4 < global_best:
            global_best = obj4
            global_a = a4.copy()
            a = a4.copy()
            print("  P2d Mirror T100-2000: obj=%.7f" % obj4)
        else:
            print("  P2d Mirror T100-2000: no improvement")

        # Strategy 5: Very high T subgradient
        t5 = exploit_budget * 0.18
        a5, _ = run_weighted_subgrad(a, n, S, t5,
                                      T_init=500, T_final=10000, lr_init=0.02, momentum=0.9)
        a5 = normalize_nonneg(a5, S)
        obj5 = evaluate_sequence(a5.tolist())
        if np.isfinite(obj5) and obj5 < global_best:
            global_best = obj5
            global_a = a5.copy()
            a = a5.copy()
            print("  P2e Subgrad T500-10000: obj=%.7f" % obj5)
        else:
            print("  P2e Subgrad T500-10000: no improvement")

    # ============ PHASE 3: Coordinate descent (20% time) ============
    cd_budget = 0.20 * total_budget
    cd_deadline = time.monotonic() + cd_budget

    if global_a is not None and cd_budget > 10:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = global_a.copy()
        a = normalize_nonneg(a, S)

        print("\nPhase 3: Coordinate descent (budget=%.0fs)" % cd_budget)

        # First CD pass
        t_cd1 = cd_budget * 0.5
        a_cd1, _ = run_coordinate_descent(a, n, S, t_cd1, max_sweeps=40)
        a_cd1 = normalize_nonneg(a_cd1, S)
        obj_cd1 = evaluate_sequence(a_cd1.tolist())
        if np.isfinite(obj_cd1) and obj_cd1 < global_best:
            global_best = obj_cd1
            global_a = a_cd1.copy()
            print("  CD pass 1: obj=%.7f" % obj_cd1)
        else:
            print("  CD pass 1: no improvement")

        # Interleaved: short subgradient then CD
        remaining_cd = cd_deadline - time.monotonic() - 2
        if remaining_cd > 10:
            t_sg = remaining_cd * 0.4
            t_cd2 = remaining_cd * 0.6
            a_tmp = global_a.copy()
            a_tmp, _ = run_weighted_subgrad(a_tmp, n, S, t_sg,
                                             T_init=200, T_final=2000, lr_init=0.03)
            a_tmp = normalize_nonneg(a_tmp, S)
            a_cd2, _ = run_coordinate_descent(a_tmp, n, S, t_cd2, max_sweeps=30)
            a_cd2 = normalize_nonneg(a_cd2, S)
            obj_cd2 = evaluate_sequence(a_cd2.tolist())
            if np.isfinite(obj_cd2) and obj_cd2 < global_best:
                global_best = obj_cd2
                global_a = a_cd2.copy()
                print("  CD pass 2: obj=%.7f" % obj_cd2)
            else:
                print("  CD pass 2: no improvement")

    # ============ PHASE 4: Final refinement (15% time) ============
    refine_deadline = overall_deadline - 5
    refine_budget = max(0, refine_deadline - time.monotonic())

    if global_a is not None and refine_budget > 10:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = global_a.copy()
        a = normalize_nonneg(a, S)

        print("\nPhase 4: Final refinement (budget=%.0fs)" % refine_budget)

        # Mirror descent at moderate T for final polish
        t_mirror = refine_budget * 0.3
        a_m, _ = run_mirror_descent(a, n, S, t_mirror,
                                     T_init=50, T_final=500, lr_init=0.03)
        a_m = normalize_nonneg(a_m, S)
        obj_m = evaluate_sequence(a_m.tolist())
        if np.isfinite(obj_m) and obj_m < global_best:
            global_best = obj_m
            global_a = a_m.copy()
            print("  Final mirror: obj=%.7f" % obj_m)

        # Try sparsification
        a_sp, _ = try_sparsification(global_a, n, S, n_zero_target=30)
        obj_sp = evaluate_sequence(a_sp.tolist())
        if np.isfinite(obj_sp) and obj_sp < global_best:
            global_best = obj_sp
            global_a = a_sp.copy()
            print("  Sparsify: obj=%.7f" % obj_sp)

        # Final short CD
        remaining = refine_deadline - time.monotonic() - 2
        if remaining > 8:
            a_f, _ = run_coordinate_descent(global_a, n, S, remaining, max_sweeps=20)
            a_f = normalize_nonneg(a_f, S)
            obj_f = evaluate_sequence(a_f.tolist())
            if np.isfinite(obj_f) and obj_f < global_best:
                global_best = obj_f
                global_a = a_f.copy()
                print("  Final CD: obj=%.7f" % obj_f)

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f (verified=%s)" % (
        global_best, "%.7f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)

    return final_a
