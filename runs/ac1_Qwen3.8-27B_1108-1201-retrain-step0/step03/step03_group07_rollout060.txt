from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.maximum(v, 0)
    s = v.sum()
    if s == 0:
        return np.ones(n) * S / n
    if s == S:
        return v
    # Bisection for tau: sum(max(v - tau, 0)) = S
    lo, hi = 0.0, v.max()
    for _ in range(80):
        mid = (lo + hi) / 2
        s_mid = np.maximum(v - mid, 0).sum()
        if s_mid > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.maximum(v - tau, 0)

def compute_smooth_grad(c, a, tau):
    n = len(a)
    # p = softmax(c / tau)
    c_scaled = c / tau
    c_max = c_scaled.max()
    p = np.exp(c_scaled - c_max)

    if p_norm < 1e-300:

    A = fft(a_rev, fsize)
    P = fft(p, fsize)
    full_conv = np.real(ifft(A * P))
    grad = 2.0 * full_conv[n-1 : 2*n-1]
    return grad, p

    best_max = cur_max

    t = 1.0
    a_prev = a.copy()
    y = a.copy()

    no_improve = 0
    
    while time.monotonic() - start < time_budget:

        tau = tau_init * (tau_min / tau_init) ** progress

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using FISTA with log-sum-exp smoothing,

import numpy as np

from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]


def proj_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} via bisection."""
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-300:
        return np.full(n, S / n)
    if abs(s - S) < 1e-13:
        return v
    lo, hi = 0.0, v.max()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s_mid = np.maximum(v - mid, 0.0).sum()
        if s_mid > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.maximum(v - tau, 0.0)


def soft_grad(c, a, tau):
    """Compute gradient of tau*logsumexp(c/tau) w.r.t. a.
    Returns grad of length n where grad[j] = 2 * sum_u a[u]*p[j+u].
    """
    n = len(a)
    cs = c / tau
    cs = cs - cs.max()
    e = np.exp(cs)
    p = e / e.sum()
    # grad[j] = 2 * conv(a_rev, p)[n-1+j]
    a_rev = a[::-1]
    L = 1 << (3 * n - 2 - 1).bit_length()
    F = fft(a_rev, L) * fft(p, L)
    full = np.real(ifft(F))
    return 2.0 * full[n - 1:2 * n - 1]

def gen_init(n, kind, rng):

        a = 1.0 + 0.6 * np.cos(2.0 * np.pi * 3.0 * i / n)
