    # This is 2 * (w cross-correlate a) evaluated at lags 0..n-1
    # In FFT: corr = ifft(conj(fft(w)) * fft(a))
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return grad


def topk_gradient(a, c, k, power=1.0):
    """
    Gradient of sum of top-k convolution values (with power weighting).
    Approximates the subgradient of the max function.
    """
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return grad


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    # grad_j = 2 * a[k-j] (if 0<=k-j<n) + 2 * a[j-k] (if 0<=j-k<n and j-k != k-j)
    # Simplified: grad_j = 2 * (a shifted) contribution
    # From our formula: grad_j = 2 * sum_l w[j+l] * a[l] where w is one-hot at k
    # = 2 * a[k-j] (if 0 <= k-j < n)
    grad = np.zeros(n)
    # w is one-hot at position k, so w[j+l] = 1 iff j+l = k, i.e., l = k-j
    # grad_j = 2 * a[k-j] if 0 <= k-j < n, else 0
    lo = max(0, k - n + 1)  # j >= k-n+1 for k-j <= n-1
    hi = min(k, n - 1)      # j <= k for k-j >= 0
    if hi >= lo:
        j_idx = np.arange(lo, hi + 1)
        l_idx = k - j_idx  # = k - j for each j
        grad[j_idx] = 2.0 * a[l_idx]
    return grad


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def run_adaptive_fista(a_init, n, S, max_time, rng, 
                       beta_init=10.0, beta_rate=0.002, beta_max=5000.0,
                       topk_init=32, topk_min=1, topk_decay=0.999,
                       restart_patience=500):
    """
    FISTA with adaptive LSE/top-k gradient, Nesterov restart, and adaptive step.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    c_prev = np.inf
    k = topk_init
    use_exact = False
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        # Update top-k (decrease over time)
        if not use_exact:
            k = max(topk_min, int(topk_init * (topk_decay ** it)))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        # Check for Nesterov restart
        if cur > c_prev * (1 + 1e-10) and it > 100:
            # Objective increased - restart momentum
            t = 1.0
            y = best_a.copy()  # Reset to best
            c = fft_convolve_self(y)
            cur = c.max()
        
        c_prev = cur
        
        # Compute gradient
        if use_exact or (beta > 100 and it > 5000):
            # Switch to exact subgradient for final polishing
            grad = exact_subgradient(y, c)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            # Step size for subgradient: use 2*f/(t*||g||^2) style
            step = 2.0 * S / (n * (1.0 + 0.001 * it))
            # Scale gradient to unit norm for consistent step
            grad = grad / gnorm
        else:
            # LSE or top-k gradient
            if k <= 4:
                grad = topk_gradient(y, c, k, power=1.0)
            else:
                grad = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta) if k > 4 else 2.0 * S / (n * (1.0 + 0.001 * it))
        
        # Adaptive step: reduce if not improving
        if it - last_improve > restart_patience:
            step *= 0.5
            # Inject small noise to escape flat regions
            noise_scale = 0.0005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it
            c_prev = np.inf
        
        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Switch to exact mode when beta is very high
        if beta > 500 and it > 10000:
            use_exact = True
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_polish_exact(a_init, n, S, max_time, rng):
    """
    Final polishing using exact subgradient with careful step size scheduling.
    Uses decreasing step size and periodic noise injection.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time - 1:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        grad = exact_subgradient(y, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Decreasing step size
        step = S / (np.sqrt(it + 1) * np.sqrt(n))
        
        x_new = project_simplex(y - step * grad, S)
        
        # Check if improvement
        c_new = fft_convolve_self(x_new)
        if c_new.max() > cur * 1.001:
            # Rejected - try smaller step or restart
            if it - last_improve > 1000:
                # Inject noise
                noise_scale = 0.001 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                last_improve = it
            step *= 0.1
            x_new = project_simplex(y - step * grad, S)
        
        y = x_new
    
    c_final = fft_convolve_self(y)
    if c_final.max() < best:
        best = c_final.max()
        best_a = y.copy()
    
    return best_a, best


def downsample_upsample(a, n_target):
    """Downsample to n_target/4 then upsample to n_target via interpolation."""
    n = len(a)
    n_coarse = max(64, n_target // 4)
    if n_coarse >= n:
        # Upsample
        t_old = np.linspace(0, 1, n)
        t_new = np.linspace(0, 1, n_target)
        return np.interp(t_new, t_old, a)
    else:
        # Downsample then upsample
        nblocks = n // n_coarse
        a_small = a[:nblocks * n_coarse].reshape(n_coarse, nblocks).mean(axis=1)
        t_old = np.linspace(0, 1, n_coarse)
        t_new = np.linspace(0, 1, n_target)
        return np.interp(t_new, t_old, a_small)
