"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses improved FISTA with LSE, coordinate descent polishing,
multiplicative weights, and multi-scale exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_soft_gradient(a, c, k, temperature):
    """Weighted gradient using top-k lags with softmax weighting."""
    n = len(a)
    k = min(k, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    w = np.exp((top_vals - top_vals[0]) / temperature)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones(k) / k
    else:
        w = w / wsum
    w_full = np.zeros(len(c))
    w_full[top_idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista_improved(a_init, n, S, max_time, rng, mode="adaptive"):
    """FISTA with adaptive LSE and multiple gradient strategies."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = 10.0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta: increase slowly, reset on restart
        beta = min(5000.0, 10.0 * np.exp(0.003 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Choose gradient
        if it < 500:
            grad = lse_gradient(y, c, beta)
        elif it < 3000:
            grad = lse_gradient(y, c, beta)
            if rng.random() < 0.1:
                grad = topk_soft_gradient(y, c, 64, 0.01)
        else:
            # Late stage: use harder gradients

                grad = topk_soft_gradient(y, c, 128, 0.005)
            else:
                grad = lse_gradient(y, c, min(beta * 2, 10000.0))

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        step = 1.0 / (2.0 * beta * max(gnorm / S, 1e-6))
        step = min(step, 2.0 / beta)

        # Stuck detection: inject noise and restart momentum
        if it - last_improve > 3000:
            noise_scale = 0.005 * S / np.sqrt(n) * np.exp(-it / 50000.0)
            y = project_simplex(y + rng.normal(0, max(noise_scale, 1e-8), n), S)
            t_mom = 1.0
            last_improve = it
            step *= 0.5

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

def coordinate_descent(a_init, n, S, max_time, rng, max_sweeps=50):
    """
    Coordinate descent polishing.
    For each coordinate j, find delta that minimizes max(b + delta * g_j).
    """
    a = normalize_sum(a_init, n)
    b = fft_convolve_self(a)
    start = time.monotonic()
    best_score = b.max()
    best_a = a.copy()

    # Pre-compute the "gradient directions" for each j
    # g_j[k] = a[k-j] for 0 <= k-j <= n-1 (shifted a)
    # Change in b when a[j] increases by delta: delta * g_j (first order, for conv(e_j, a))
    # Full change (both factors): 2*delta*a_shifted + delta^2 * delta_fn
    # We'll use the first-order approximation for the search, then verify.

    n_b = 2 * n - 1
    a_shifted = np.zeros((n, n_b))
    for j in range(n):
        a_shifted[j, j:j+n] = a

    # We'll do greedy coordinate descent
    order = np.arange(n)
    rng.shuffle(order)

    for sweep in range(max_sweeps):
        if time.monotonic() - start > max_time - 5:
            break

        improved = 0
        for j in order:
            if time.monotonic() - start > max_time - 5:
                break

            # Current a[j] value
            aj = a[j]
            if aj < 1e-10:
                continue

            # Gradient direction in b-space
            gj = a_shifted[j]  # Shape (n_b,)

            # Find optimal delta via ternary search
            # f(delta) = max(b + 2*delta*gj) subject to aj + delta >= 0
            # Try reducing and increasing
            lo = -aj * 0.5  # Can reduce by at most half
            hi = aj * 1.0   # Can increase by at most 100%

            # Quick check: which direction helps at the current max?
            kmax = np.argmax(b)
            if gj[kmax] > 0:
                # Reducing a[j] helps
                search_lo, search_hi = lo, 0
            else:
                # Increasing a[j] helps
                search_lo, search_hi = 0, hi

            # Ternary search
            for _ in range(25):
                m1 = search_lo + (search_hi - search_lo) / 3
                m2 = search_hi - (search_hi - search_lo) / 3
                f1 = (b + 2.0 * m1 * gj).max()
                f2 = (b + 2.0 * m2 * gj).max()
                if f1 < f2:
                    search_hi = m2
                else:
                    search_lo = m1

            delta = (search_lo + search_hi) / 2.0

            # Verify with exact convolution
            new_b = b + 2.0 * delta * gj + delta * delta * np.zeros(n_b)
            # Add the delta^2 term: conv(delta*e_j, delta*e_j) = delta^2 at position 2j
            if 2 * j < n_b:
                new_b[2 * j] += delta * delta

            new_max = new_b.max()
            if new_max < best_score - 1e-10:
                a[j] += delta
                b = new_b
                best_score = new_max
                best_a = a.copy()
                a_shifted[j, j:j+n] = a  # Update
                # Update all other shifted vectors that depend on a[j]
                # a_shifted[i, i+n] corresponds to a[0] at position i+n... 
                # Actually a_shifted[i, k] = a[k-i] for valid k-i
                # So a_shifted[i, i+j] = a[j] changed
                # This is expensive to update fully, so we'll recompute lazily
                improved += 1

        # Renormalize after each sweep
        a = normalize_sum(a, n)
        b = fft_convolve_self(a)
        if b.max() < best_score:
            best_score = b.max()
            best_a = a.copy()

        # Rebuild a_shifted (expensive but correct)
        if sweep % 5 == 0:
            for j2 in range(n):
                a_shifted[j2, j2:j2+n] = a

        if improved == 0 and sweep > 5:
            break

    return best_a, best_score


def multiplicative_weights(a_init, n, S, max_time, rng, k_top=32, lr_init=0.05):
    """Multiplicative weights (mirror descent) on the simplex."""
    a = np.maximum(a_init.copy(), 1e-10)
    a = a / a.sum() * S
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    lr = lr_init

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # Top-k weighted gradient
        k = min(k_top, len(c))
        top_idx = np.argsort(c)[::-1][:k]
        top_vals = c[top_idx]
        diff = top_vals - top_vals[0]
        w = np.exp(20.0 * diff)
        wsum = w.sum()
        if wsum < 1e-300:
            w = np.ones(k) / k
        else:
            w = w / wsum

        # Compute gradient
        grad = np.zeros(n)
        for idx, wi in zip(top_idx, w):
            ks = int(idx)
            lo = max(0, ks - n + 1)
            hi = min(ks, n - 1)
            grad[lo:hi+1] += wi * 2.0 * a[ks - hi:ks - lo + 1]

        # Mirror descent step (multiplicative weights)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= log_a.max()
        log_a -= lr * grad / max(np.linalg.norm(grad), 1e-10)
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-10)
        a_new = a_new / a_new.sum() * S
        a = np.clip(a_new, 0, 1000)

        # Decay learning rate
        lr *= 0.999

    return best_a, best_score


def generate_structured_inits(n, rng):

    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)

    # Asymmetric (to break symmetry and potentially lower center convolution)
    inits.append(np.maximum(0.3 + 0.7 * np.sin(np.pi * tt * 0.7 + 0.3), 0.01))
    inits.append(np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * tt - 1.0), 0.01))
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2 * np.pi * 1.5 * tt + 0.7), 0.01))

    # Multi-frequency
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.15 * np.cos(2 * np.pi * 5 * tt + 0.5))
    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * 3 * tt + 0.3) + 0.1 * np.cos(2 * np.pi * 7 * tt))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.4, n)))

    if n_small >= n:
        return a.copy()
    nblocks = n // n_small
    a_small = a[:nblocks * n_small].reshape(n_small, nblocks).mean(axis=1)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n)
    return np.interp(t_new, t_old, a_small)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max_conv/sum^2.
    Uses FISTA + coordinate descent + multiplicative weights with multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    total_deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Try to load given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Starting optimization, budget=%.0fs" % budget_s)
    if given is not None:
        print("Loaded given sequence of length %d" % len(given))
        g_score = evaluate_sequence([float(x) for x in given])
        print("Given sequence score: %.6f" % g_score)
        if np.isfinite(g_score):
            global_best = g_score
            global_a = given.copy()

    # Plan: try multiple n values, allocate time proportionally
    # Focus on n values around the given length and nearby
    n_candidates = []
    if given is not None and 1024 <= len(given) <= 32768:
        n_candidates.append(len(given))
        n_candidates.append(max(1024, len(given) // 2))
        n_candidates.append(min(32768, len(given) * 2))
    else:
        n_candidates = [8192, 16384, 4096]

    # Remove duplicates, sort by priority (given length first)
    n_candidates = list(dict.fromkeys(n_candidates))
    n_candidates = [n for n in n_candidates if 512 <= n <= 65536]

    # Allocate time: 60% to primary n, 20% to secondary, 20% to tertiary + polish
    usable = max(60.0, total_deadline - time.monotonic())
    weights = np.array([0.5, 0.2, 0.15, 0.15])
    while len(weights) < len(n_candidates):
        weights = np.append(weights, 0.1)
    weights = weights[:len(n_candidates)]
    weights /= weights.sum()
    n_times = usable * weights * 0.75  # 75% exploration, 25% polish

    print("n candidates: %s, times: %s" % (n_candidates, [round(t) for t in n_times]))

    for ni, n in enumerate(n_candidates):
        if time.monotonic() > total_deadline - 60:
            break
        S = np.sqrt(2.0 * n)
        t_budget = n_times[ni]
        if t_budget < 10:
            continue

        print("\n=== n=%d (time=%.0fs) ===" % (n, t_budget))

        # Collect initializations for this n
        inits = []
        if given is not None and len(given) == n:
            inits.append(("given", given.copy()))
        elif given is not None:
            # Resample given to this n
            t_old = np.linspace(0, 1, len(given))
            t_new = np.linspace(0, 1, n)
            inits.append(("resampled", np.interp(t_new, t_old, given)))

        # Downsample + upsample for warm start
        if given is not None and n > 2048:
            n_coarse = min(n // 4, 2048)
            coarse = downsample_upsample(given, n_coarse)
            inits.append(("coarse", coarse))

        # Structured inits
        struct_inits = generate_structured_inits(n, rng)
        for i, si in enumerate(struct_inits[:4]):
            inits.append(("struct_%d" % i, si))

        # Run FISTA on each init
        t_per_init = max(5.0, t_budget * 0.6 / len(inits))
        local_best = np.inf
        local_best_a = None

        for name, arr in inits:
            if time.monotonic() > total_deadline - 60:
                break
            arr_n = len(arr)
            if arr_n < 64:
                continue
            S_i = np.sqrt(2.0 * arr_n)
            t_init = min(t_per_init, total_deadline - time.monotonic() - 60)
            if t_init < 5:
                break

            # Alternate between FISTA and MW for diversity
            if name in ("given", "resampled", "coarse"):
                a_opt, score = run_fista_improved(arr, arr_n, S_i, t_init, rng)
            else:
                a_opt, score = run_fista_improved(arr, arr_n, S_i, t_init, rng)

            a_opt = normalize_sum(a_opt, arr_n)
            c_chk = fft_convolve_self(a_opt)
            real_score = c_chk.max()

            if real_score < local_best:
                local_best = real_score
                local_best_a = a_opt.copy()
                print("  [%s] n=%d: %.6f" % (name, arr_n, local_best))

        # Coordinate descent polish on best for this n
        if local_best_a is not None:
            n_cd = len(local_best_a)
            S_cd = np.sqrt(2.0 * n_cd)
            t_cd = min(60.0, max(10.0, t_budget * 0.2))
            print("  CD polish (time=%.0fs)..." % t_cd)
            a_cd, score_cd = coordinate_descent(local_best_a, n_cd, S_cd, t_cd, rng)
            a_cd = normalize_sum(a_cd, n_cd)
            c_cd = fft_convolve_self(a_cd)
            if c_cd.max() < local_best:
                local_best = c_cd.max()
                local_best_a = a_cd.copy()
                print("  After CD: %.6f" % local_best)

        # MW polish
        if local_best_a is not None:
            n_mw = len(local_best_a)
            S_mw = np.sqrt(2.0 * n_mw)
            remaining_n = total_deadline - time.monotonic() - 60
            t_mw = min(30.0, max(5.0, 0.1 * remaining_n))
            if t_mw > 5:
                a_mw, score_mw = multiplicative_weights(local_best_a, n_mw, S_mw, t_mw, rng)
                a_mw = normalize_sum(a_mw, n_mw)
                c_mw = fft_convolve_self(a_mw)
                if c_mw.max() < local_best:
                    local_best = c_mw.max()
                    local_best_a = a_mw.copy()
                    print("  After MW: %.6f" % local_best)

        # Update global
        obj_val = 2.0 * len(local_best_a) * local_best / (np.sqrt(2.0 * len(local_best_a)) ** 2)
        # Actually obj = 2*n*max_b/S^2 where S=sqrt(2n), so obj = max_b
        # Wait: S = sqrt(2n), S^2 = 2n, so obj = 2*n*max_b/(2n) = max_b
        # So the objective IS max_b when normalized to sum=sqrt(2n)
        if local_best < global_best:
            global_best = local_best
            global_a = local_best_a.copy()
            print("  NEW GLOBAL BEST: %.6f (n=%d)" % (global_best, len(global_best_a) if global_a else 0))

    # Final polish phase
    if global_a is not None:
        n_fin = len(global_a)
        S_fin = np.sqrt(2.0 * n_fin)
        remaining = total_deadline - time.monotonic()
        if remaining > 20:
            print("\n=== Final polish (remaining=%.0fs) ===" % remaining)

            # High-beta FISTA
            t_fh = min(remaining * 0.4, 120.0)
            if t_fh > 5:
                a_fh, s_fh = run_fista_improved(global_a, n_fin, S_fin, t_fh, rng)
                a_fh = normalize_sum(a_fh, n_fin)
                c_fh = fft_convolve_self(a_fh)
                if c_fh.max() < global_best:
                    global_best = c_fh.max()
                    global_a = a_fh.copy()
                    print("  After final FISTA: %.6f" % global_best)

            # Coordinate descent
            remaining2 = total_deadline - time.monotonic()
            if remaining2 > 10:
                t_cd2 = min(remaining2 * 0.4, 60.0)
                a_cd2, s_cd2 = coordinate_descent(global_a, n_fin, S_fin, t_cd2, rng)
                a_cd2 = normalize_sum(a_cd2, n_fin)
                c_cd2 = fft_convolve_self(a_cd2)
                if c_cd2.max() < global_best:
                    global_best = c_cd2.max()
                    global_a = a_cd2.copy()
                    print("  After final CD: %.6f" % global_best)

            # One more FISTA
            remaining3 = total_deadline - time.monotonic()
            if remaining3 > 10:
                a_f3, s_f3 = run_fista_improved(global_a, n_fin, S_fin, remaining3 * 0.8, rng)
                a_f3 = normalize_sum(a_f3, n_fin)
                c_f3 = fft_convolve_self(a_f3)
                if c_f3.max() < global_best:
                    global_best = c_f3.max()
                    global_a = a_f3.copy()
                    print("  After final FISTA2: %.6f" % global_best)

    # Verify with evaluate_sequence
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256) * (np.sqrt(512) / 256)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification score: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Fallback to simple sequence")
            global_a = np.ones(256) * (np.sqrt(512) / 256)
            final_list = [float(x) for x in global_a]

    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(512) / 256)

    result = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    print("\nReturning sequence of length %d, objective=%.6f" % (len(result), global_best))
    return result
