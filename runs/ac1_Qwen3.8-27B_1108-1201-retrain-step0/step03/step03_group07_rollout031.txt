    Perform one LP-based step.
    a0: current point
    S: target sum
    delta: max fraction of step to take
    Returns new point.
    """

    c0 = fft_convolve_self(a0)

    # Variables: a[0], ..., a[n-1], t

    cols = []

    if a_lp.sum() < 1e-10:

    a_lp = a_lp * (S / a_lp.sum())
    
    return a_lp, t_lp

def lp_optimize(a0, S, time_budget, K_start=100, K_end=500, 
                k_ramp=50, verbose=True):
    """
    Main LP-based optimization loop.
    """
    a = a0.copy()
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    iter_no = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Select active set
        progress = min(1.0, iter_no / k_ramp)
        K_cur = int(K_start + (K_end - K_start) * progress)
        K_cur = min(K_cur, 2 * n - 1)
        
        top_idx = np.argsort(c)[::-1][:K_cur]
        
        # Solve LP
        a_lp, t_lp = solve_lp_step(a, S, top_idx)
        
        # Line search
        best_alpha = 0.0
        best_trial_obj = best_obj
        for alpha in [1.0, 0.75, 0.5, 0.25, 0.1]:
            a_trial = a + alpha * (a_lp - a)
            a_trial = np.maximum(a_trial, 0.0)
            if a_trial.sum() < 1e-10:
                continue
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)
            if obj_trial < best_trial_obj:
                best_trial_obj = obj_trial
                best_alpha = alpha
        
        if best_alpha > 0:
            a = a + best_alpha * (a_lp - a)
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            cur_obj = obj_value(a, c)
            
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Stagnation: perturb
        if no_improve > 30:
            # Try different types of perturbations
            pert_type = (iter_no // 30) % 3
            if pert_type == 0:
                # Gaussian noise
                noise = np.random.randn(n) * 0.1 * S / np.sqrt(n)
                a = a + noise
            elif pert_type == 1:
                # Multiplicative noise
                mult = np.random.randn(n) * 0.2
                a = a * np.exp(mult)
            else:
                # Block swap
                block_size = max(1, n // 10)
                i = np.random.randint(0, n - block_size)
                j = np.random.randint(0, n - block_size)
                if i != j:
                    a[i:i+block_size], a[j:j+block_size] = a[j:j+block_size].copy(), a[i:i+block_size].copy()
            
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            no_improve = 0
        
        iter_no += 1
        if verbose and iter_no % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"  LP iter={iter_no}, obj={obj_value(a, c):.6f}, "
                  f"best={best_obj:.6f}, K={K_cur}, t={elapsed:.1f}s")
    
    return best_a, best_obj

def mirror_descent_polish(a0, S, time_budget, eta0=0.1, momentum=0.9):
    """Mirror descent polishing step."""
    a = a0.copy()
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    v = np.zeros(n)
    eta = eta0
    start = time.monotonic()
    iter_no = 0
    
    while time.monotonic() - start < time_budget:
        # Compute gradient at top peak
        k_max = np.argmax(c)
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[k_max - m]
        
        # Also add contributions from top-5 peaks
        top5 = np.argsort(c)[::-1][:5]
        for k in top5[1:]:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1):
                grad[m] += 0.5 * a[k - m]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        v = momentum * v + (1 - momentum) * grad
        v = v / np.linalg.norm(v)
        
        # Multiplicative update
        a = a * np.exp(-eta * v)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())
        
        c = fft_convolve_self(a)
        cur_obj = obj_value(a, c)
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            eta = min(eta * 1.01, eta0 * 3)
        else:
            eta = max(eta * 0.95, eta0 * 0.1)
        
        eta *= 0.99998
        iter_no += 1
    
    return best_a, best_obj
