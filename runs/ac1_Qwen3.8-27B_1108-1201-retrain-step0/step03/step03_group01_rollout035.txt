import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    # If all values are already >= 0 and sum is S, we're done
    if v.min() >= 0 and abs(v.sum() - S) < 1e-10:
        return v
    # Standard simplex projection
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_conv(a):
    """Compute full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def lse_gradient(a, c, beta):
    """Compute gradient of LSE(beta, c) with respect to a.
    c = conv(a, a), so grad[i] = 2 * sum_k w[k] * a[k-i] (valid range).
    """
    n = len(a)
    c_max = c.max()
    # Numerical stability: subtract max before exponentiating
    exp_c = np.exp(beta * (c - c_max))
    w = exp_c / exp_c.sum()
    
    # grad[i] = 2 * sum_d a[d] * w[i+d] for d=0..n-1, i=0..n-1
    # This is a cross-correlation: (a ⊛ w)[i]
    N_corr = 1 << (n + len(w) - 1).bit_length()
    A = fft(a, N_corr)
    W = fft(w, N_corr)
    grad = 2.0 * np.real(ifft(A * np.conj(W)))[:n]
    return grad

def fista_lse_optimize(a_init, n, S, max_time, rng, beta_start=3.0, beta_end=300.0, 
                        n_beta_steps=50, jump_prob=0.05, jump_every=200):
    """FISTA with LSE smooth approximation and spectral jumps."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    fsize = 1 << (2 * n - 2).bit_length()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta
        progress = min(1.0, it / n_beta_steps)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute convolution and objective
        c = compute_conv(y)
        cur_max = c.max()
        cur_score = 2.0 * n * cur_max / (y.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = y.copy()
        
        # Compute gradient
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: use 1/L where L ≈ 2 * max(|a|) * n (rough Lipschitz bound)
        # Actually, let's use backtracking
        step = 1.0 / (max(1.0, 4.0 * np.max(np.abs(y)) * n))
        
        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        
        # Check if x_new is better
        if it % 10 == 0:

            score_x = 2.0 * n * c_x.max() / (x_new.sum() ** 2)
            if score_x < best_score:
                best_score = score_x
                best_a = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Spectral jump to escape local minima
        if it % jump_every == 0 and rng.random() < jump_prob:
            a_jump = a.copy()

            for pi in phase_indices:

                y = a_jump.copy()

The Hessian is: ∇²f[a][i,j] = sum_k w_k * ∂²c_k/∂a[i]∂a[j] + (1/β) * sum_k (∂(w_k)/∂a[i]) * ∂c_k/∂a[j] + ...
