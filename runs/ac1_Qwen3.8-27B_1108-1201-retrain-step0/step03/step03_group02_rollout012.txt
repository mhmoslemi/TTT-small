def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def project_nonneg_sum1(v):
    """Projection onto {x >= 0, sum(x) = 1}."""
    return project_simplex(v, 1.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def objective_from_conv(a, c):

def subgradient_toppk(a, c, k):

    # 2 * sum_{j in top_lags} a[j - m] (where valid)

    w = np.zeros(2 * n - 1)
    for j in top_idx:

    # = 2 * (w cross-correlate a)[m]

    W = fft(w, fsize)
    A = fft(a, fsize)

def soft_gradient(a, c, beta):

"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Algorithm: Hybrid accelerated subgradient descent with adaptive top-k peak
targeting, Polyak step sizes, chirp-inspired initializations, and final

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv(self_a):
    """Full linear self-convolution via FFT."""
    n = len(self_a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(self_a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def subgrad_toppk(a, c, top_idx):
    """
    Subgradient of max(c) using specified peak indices.
    Returns gradient vector of length n.
    """
    n = len(a)
    w = np.zeros(2 * n - 1)
    for k in top_idx:
        w[int(k)] = 1.0
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def subgrad_softmax(a, c, beta):
    """Smooth subgradient via softmax weighting of peaks."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def obj_val(n, c):
    """Objective value given n and convolution (assumes sum(a)=1)."""
    return 2.0 * n * c.max()


def run_accelerated_subgrad(a_init, n, time_budget, rng,

    """
    Accelerated projected subgradient with adaptive top-k and Polyak step.
    Normalizes to sum(a) = 1 internally.
    """
    a = np.clip(a_init.copy(), 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s  # normalize to sum=1

    y = a.copy()
    t_nest = 1.0
    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    it = 0

    # Lower bound on max(c) when sum(a)=1: 1/(2n-1)

    obj_lower = 2.0 * n * lower_bound

    while time.monotonic() - start < time_budget:
        it += 1

        # Adaptive k
        frac = min(1.0, it * k_ramp)
        k = max(1, int(round(k_start + frac * (k_end - k_start))))

        cur_obj = 2.0 * n * cur_max
        if cur_obj < best_score:
            best_score = cur_obj
            best_a = y.copy()

        # Get gradient
        top_idx = np.argsort(c)[::-1][:k]
        grad = subgrad_toppk(y, c, top_idx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak-style step size
        gap = max(cur_obj - obj_lower, 1e-6)
        step = gap / (gnorm * gnorm)
        # Cap step to avoid instability
        step = min(step, 5.0 / (np.sqrt(n) * gnorm))
        step = max(step, 1e-10)

        # Gradient step
        x_new = project_simplex(y - step * grad, 1.0)

        # Nesterov momentum

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            y = x_new + ((t_nest - 1.0) / t_new) * (x_new - y)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            t_nest = t_new
        else:
            y = x_new

        # Periodic noise for exploration

            noise = rng.normal(0, 0.001 / np.sqrt(n), n)

def make_chirp_init(n, rng):
    """Chirp-based initialization (real non-negative)."""
    k = np.arange(n)
    # Real part of quadratic phase chirp, shifted to be non-negative
    alpha = rng.uniform(0.1, 0.5)
    phase = 2.0 * np.pi * alpha * k * k / n
    a = 1.0 + 0.8 * np.cos(phase)
    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        a = np.ones(n)
    return a / a.sum()


def make_exponentials_init(n, rng):
    """Exponential ramp initialization."""
    tt = np.arange(n) / n
    direction = rng.random()
    if direction < 0.5:
        a = np.exp(rng.uniform(1.0, 4.0) * tt)
    else:
        a = np.exp(rng.uniform(1.0, 4.0) * (1.0 - tt))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def make_random_init(n, rng):
    """Random initialization with mild trend."""
    a = rng.random(n)
    trend = np.linspace(0.5, 1.5, n)
    if rng.random() < 0.5:
        trend = trend[::-1]
    a = a * trend
    a = np.maximum(a, 0.0)
    return a / a.sum()


def convex_combine(a1, a2, alpha):
    """Convex combination maintaining sum=1."""
    return alpha * a1 + (1.0 - alpha) * a2

                parent = p / p.sum()
    except Exception:
        pass

    # Determine n values to explore
    n_list = [4096, 8192, 16384]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_list:
            n_list.insert(0, np_)

    print("Exploring n values: %s" % str(n_list))
    print("Parent available: %s" % (parent is not None))

    # --- Build initializations ---
    candidates = []  # list of (name, array, n)

    if parent is not None:
        candidates.append(("parent", parent.copy(), len(parent)))

    for n in n_list:
        candidates.append(("chirp_%d" % n, make_chirp_init(n, rng), n))
        candidates.append(("chirp_%d_b" % n, make_chirp_init(n, rng), n))
        candidates.append(("exp_%d" % n, make_exponentials_init(n, rng), n))
        candidates.append(("exp_%d_b" % n, make_exponentials_init(n, rng), n))
        candidates.append(("rand_%d" % n, make_random_init(n, rng), n))

    n_cands = len(candidates)
    print("Total initial candidates: %d" % n_cands)

    # --- Phase 1: Fast screening ---
    screen_total = min(120.0, (deadline - time.monotonic()) * 0.15)
    per_screen = max(2.0, screen_total / n_cands)
    print("Phase 1: Screening (%.1f sec total, %.1f sec each)" % (screen_total, per_screen))

    screened = []
    for name, arr, n in candidates:
        if time.monotonic() > deadline - 60:
            break
        arr = np.clip(arr, 0, 1000)
        s = arr.sum()
        if s < 0.01:
            continue
        arr = arr / s

        t0 = time.monotonic()
        a_opt, score = run_accelerated_subgrad(
            arr, n, per_screen, rng, k_start=1, k_end=32, k_ramp=0.05)
        screened.append((score, a_opt, n, name))
        print("  [%-12s] n=%5d:  obj = %.8f" % (name, n, score))

    screened.sort(key=lambda x: x[0])

    # --- Phase 2: Deep optimization on top candidates ---
    top_n = max(3, min(6, len(screened) // 2))
    top = screened[:top_n]
    remaining = deadline - time.monotonic()
    per_deep = max(20.0, remaining * 0.45 / top_n)
    print("Phase 2: Deep optimization on top %d (remaining %.0f sec)" % (top_n, remaining))

    results = []
    for score_i, a_i, n_i, name in top:
        if time.monotonic() > deadline - 60:
            break

        # Run phased optimization
        a_curr = a_i.copy()
        best_local = score_i
        best_arr = a_i.copy()

        # Sub-phase A: Aggressive subgradient (small k)
        seg_a = per_deep * 0.4
        a_r, s_r = run_accelerated_subgrad(
            a_curr, n_i, seg_a, rng, k_start=1, k_end=16, k_ramp=0.03)
        if s_r < best_local:
            best_local = s_r
            best_arr = a_r.copy()

        # Sub-phase B: Softmax smoothing
        remaining_b = per_deep * 0.25
        if time.monotonic() < deadline - 30:
            a_r2, s_r2 = run_softmax_descent(
                best_arr, n_i, remaining_b, rng, beta_start=10.0, beta_end=40.0)
            if s_r2 < best_local:
                best_local = s_r2
                best_arr = a_r2.copy()

        # Sub-phase C: Large-k subgradient
        remaining_c = per_deep * 0.35
        if time.monotonic() < deadline - 30:
            a_r3, s_r3 = run_accelerated_subgrad(
                best_arr, n_i, remaining_c, rng,
                k_start=8, k_end=256, k_ramp=0.01)
            if s_r3 < best_local:
                best_local = s_r3
                best_arr = a_r3.copy()

        results.append((best_local, best_arr, n_i, name))
        print("  Deep [%-12s] n=%5d:  obj = %.8f" % (name, n_i, best_local))

        if best_local < global_best:
            global_best = best_local
            global_a = best_arr.copy()

    # --- Phase 3: Local polish on global best ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        n_best = len(global_a)
        print("Phase 3: Local polish (remaining %.0f sec, n=%d)" % (remaining, n_best))

        if remaining > 15:
            # Multiple rounds of local polish with different random seeds
            a_pol = global_a.copy()
            seg = min(remaining * 0.4, 300.0)

            for round_i in range(3):
                if time.monotonic() > deadline - 20:
                    break
                a_p, s_p = local_coordinate_polish(a_pol, n_best, seg / 3.0, rng)
                if s_p < global_best:
                    global_best = s_p
                    global_a = a_p.copy()
                    a_pol = a_p.copy()
                    print("  Polish round %d: %.8f" % (round_i + 1, global_best))

        # Final subgradient sweep with very large k
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_fin, s_fin = run_accelerated_subgrad(
                global_a, n_best, remaining * 0.6, rng,
                k_start=32, k_end=512, k_ramp=0.005,
                use_momentum=False, noise_period=800)
            if s_fin < global_best:
                global_best = s_fin
                global_a = a_fin.copy()
                print("  Final subgrad: %.8f" % global_best)

        # One more local polish
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_l, s_l = local_coordinate_polish(global_a, n_best, remaining * 0.8, rng)
            if s_l < global_best:
                global_best = s_l
                global_a = a_l.copy()
                print("  Final local: %.8f" % global_best)

    # --- Phase 4: Try convex combinations of best results ---
    if len(results) >= 2:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("Phase 4: Convex combinations")
            # Try combinations of top-2 results if they have the same n
            for i in range(min(3, len(results))):
                for j in range(i + 1, min(3, len(results))):
                    if results[i][2] != results[j][2]:
                        continue
                    n_c = results[i][2]
                    for alpha in [0.3, 0.5, 0.7]:
                        a_comb = convex_combine(
                            results[i][1], results[j][1], alpha)
                        s_comb = a_comb.sum()
                        if s_comb < 0.01:
                            continue
                        a_comb = a_comb / s_comb
                        c_comb = fft_conv(a_comb)
                        obj_comb = 2.0 * n_c * c_comb.max()
                        if obj_comb < global_best:
                            global_best = obj_comb
                            global_a = a_comb.copy()
                            print("  Comb %d+%d a=%.1f: %.8f" %
                                  (i, j, alpha, global_best))

    # --- Final verification and return ---
    if global_a is None:
        global_a = np.ones(8192) / 8192

    global_a = np.clip(global_a, 0.0, 1000.0)
    sa = global_a.sum()
    if sa < 0.01:
        global_a = np.ones(8192) / 8192
        sa = 1.0

    # Scale to a reasonable magnitude (doesn't change objective but ensures safety)
    scale_target = np.sqrt(2.0 * len(global_a))
    global_a = global_a * (scale_target / sa)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=" * 60)
    print("FINAL: n=%d, internal=%.8f, verified=%.8f" %
          (len(final_list), global_best,
           ver if np.isfinite(ver) else -1.0))
    print("=" * 60)

    if not np.isfinite(ver):
        print("WARNING: evaluate_sequence rejected output, using fallback")
        final_list = [1.0] * 8192

    return final_list
