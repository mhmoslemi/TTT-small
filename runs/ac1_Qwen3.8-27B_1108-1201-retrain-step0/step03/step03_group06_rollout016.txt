import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_gradient_p_power(c, a, p):
    """Gradient of sum(c^p) with respect to a, using softmax parameterization."""
    n = len(a)
    # dc_k / da_j = 2 * a[k-j] (with a outside [0,n-1) being 0)
    # grad_a[j] = sum_k p * c[k]^(p-1) * 2 * a[k-j]
    # = 2p * cross_correlation(c^(p-1), a)[j]
    w = c ** (p - 1) if p > 1 else np.ones_like(c)
    fsize = 1 << (3*n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    grad_a = 2.0 * p * corr[:n]
    return grad_a

def softmax_normalize(theta, S):
    """Convert theta to a via softmax, normalized to sum S."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total

def run_adam_ppower(a_init, n, S, time_budget, p_init=4, p_final=32, lr=0.01):
    """Adam optimization of sum(conv(a,a)^p) with softmax parameterization."""
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        p = p_init * (p_final / p_init) ** progress
        
        # Update a from theta
        a = softmax_normalize(theta, S)
        
        # Compute objective
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Compute gradient
        grad_a = compute_gradient_p_power(c, a, p)
        
        # Chain rule for theta
        dot = np.dot(grad_a, a)
        g_theta = a * (grad_a - dot / S)
        
        # Adam update
        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break
        
        g_theta = g_theta / gnorm  # normalize for stable Adam
        
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * g_theta**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        lr_t = lr * (1.0 - 0.5 * progress)
        theta -= lr_t * m_hat / (np.sqrt(v_hat) + eps)
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive p-norm objective with
Adam optimization in softmax parameterization, diverse initializations,
aggressive coordinate descent with pairwise swaps, and simulated annealing
restarts. Key innovations: (1) p-norm objective with adaptive p schedule,
(2) softmax parameterization ensuring non-negativity and fixed sum,
(3) pairwise swap moves in coordinate descent, (4) diverse structured
initializations including spectral-shaping based starts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_normalize(theta, S):
    """Convert unconstrained theta to non-negative a with sum S via softmax."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def compute_grad_pnorm(c, a, p):
    """
    Gradient of (sum(c^p)/len(c))^(1/p) with respect to a.
    Returns grad_a (shape n). Chain rule to theta done separately.
    """
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    # Normalize to avoid overflow: w = (c/c_max)^(p-1)
    c_norm = c / c_max
    if p > 2:
        w = c_norm ** (p - 1)
    else:
        w = np.ones_like(c)
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] = 2 * cross_corr(w, a)[j]
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def run_adam_pnorm(a_init, n, S, time_budget, p_init=4.0, p_final=64.0,
                   lr=0.015, beta1=0.9, beta2=0.999, seed=0, verbose=False):
    """
    Adam optimizer on softmax parameters, minimizing p-norm of convolution.
    p is scheduled from p_init to p_final over the time budget.
    """
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
        s = S
    a = a * (S / s)
    theta = np.log(a / S)  # log of proportions (a_i/S)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8

    c = fft_convolve_self(a)
    c_max = c.max()
    best_obj = c_max
    best_a = a.copy()
    no_improve = 0
    max_no_improve = 500

    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Adaptive p schedule: smooth exponentiation
        p = p_init * (p_final / p_init) ** (progress ** 0.8)

        # Convert theta to a
        a = softmax_normalize(theta, S)

        # Compute objective
        c = fft_convolve_self(a)
        c_max = c.max()
        if c_max < best_obj - 1e-12:
            best_obj = c_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic restart within optimizer if stuck

            noise_scale = 0.05 * (1.0 - progress) + 0.01
            theta = np.log(np.maximum(a_best_local(a, best_a, n, S), 1e-15) / S)
            theta += noise_scale * rng.standard_normal(n)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            continue

        # Compute gradient
        grad_a = compute_grad_pnorm(c, a, p)

        dot = np.dot(grad_a, a)

        v_hat = v / (1.0 - beta2 ** t)

        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if verbose and t % 2000 == 0:
            print("      step=%d, p=%.1f, obj=%.7f, no_imp=%d" % (t, p, best_obj, no_improve))

def coordinate_descent_swaps(a, n, S, time_budget, max_sweeps=200, seed=42):
    """
    Coordinate descent with single-coordinate changes and pairwise swaps.
    More aggressive than basic CD: tries both multiplicative changes
    and value swaps between pairs.
    """
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    stagnant = 0
    max_stagnant = 80

    factors = [0.3, 0.5, 0.7, 1.3, 1.5, 2.0]

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-12:
                continue

            orig = a[i]
            best_local = orig
            best_local_obj = best_obj

            # Try multiplicative changes
            for factor in factors:
                a[i] = orig * factor
                a = np.maximum(a, 0.0)
                sa = a.sum()
                if sa < 1e-15:
                    a[i] = orig
                    continue
                a = a * (S / sa)
                c = fft_convolve_self(a)
                obj = c.max()
                if obj < best_local_obj - 1e-13:
                    best_local_obj = obj
                    best_local = orig * factor

            a[i] = best_local
            a = np.maximum(a, 0.0)
            sa = a.sum()
            if sa > 1e-15:
                a = a * (S / sa)
            c = fft_convolve_self(a)
            if c.max() < best_obj - 1e-13:
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        if not improved:
            stagnant += 1
        else:
            stagnant = 0

        if stagnant > max_stagnant:
            break

    return best_a, best_obj


def coordinate_swaps(a, n, S, time_budget, n_trials=5000, seed=123):
    """Try random pairwise swaps to find improvements."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    trial = 0

    while time.monotonic() - start < time_budget and trial < n_trials:
        trial += 1
        if time.monotonic() - start > time_budget:
            break

        # Pick two random indices
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        if a[i] < 1e-12 or a[j] < 1e-12:
            continue

        saved_i = a[i]
        saved_j = a[j]
        a[i] = saved_j
        a[j] = saved_i

        c = fft_convolve_self(a)
        obj = c.max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
        else:
            a[i] = saved_i
            a[j] = saved_j

    return best_a, best_obj


def get_initializations(n, S, rng):
    """Generate diverse initializations."""
    i = np.arange(n)
    inits = {}

    # 1. Uniform
    inits["uniform"] = np.full(n, S / n)

    # 2. Random
    a = rng.random(n) + 0.05
    inits["random"] = a * S / a.sum()

    # 3. Exponential decay
    a = np.exp(-i / (n / 3))
    inits["exp_decay"] = a * S / a.sum()

    # 4. Exponential rise
    a = np.exp(i / (n / 3))
    inits["exp_rise"] = a * S / a.sum()

    # 5. Damped oscillation
    a = np.exp(-i / (n / 4)) * (1.0 + 0.4 * np.sin(2 * np.pi * 11 * i / n))
    a = np.maximum(a, 0.01)
    inits["damped_osc"] = a * S / a.sum()

    # 6. Smoothed random (Gaussian kernel)
    a = rng.random(n)
    ks = 32
    kernel = np.exp(-0.5 * (np.arange(-ks // 2, ks // 2 + 1) / (ks // 4)) ** 2)
    kernel /= kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    inits["smoothed"] = a * S / a.sum()

    # 7. "Chirp" structure (quadratic phase, take magnitude-like)
    a = np.abs(np.cos(np.pi * (i / n) ** 2 * 8)) + 0.1
    inits["chirp"] = a * S / a.sum()

    # 8. Two-scale random
    a = 0.5 + 0.5 * rng.random(n // 8).repeat(8)[:n]
    a = np.maximum(a, 0.05)
    inits["two_scale"] = a * S / a.sum()

    # 9. Linear ramp with noise
    a = np.linspace(1.0, 0.2, n) + 0.15 * rng.standard_normal(n)
    a = np.maximum(a, 0.05)
    inits["ramp"] = a * S / a.sum()

    # 10. Power law
    a = (i + 1) ** (-0.3)
    inits["power_law"] = a * S / a.sum()

    return inits


def try_zero_masking(a, n, S, n_zero=30):

            p_init=4.0, p_final=32.0, lr=0.02, seed=hash(name) % 10000
        )
        a_best = np.maximum(a_best, 0.0)
        s = a_best.sum()
        if s > 0.01:
            a_best = a_best * (S / s)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver):
                print("  %-14s: obj=%.7f" % (name, ver))
                if ver < global_best:
                    global_best = ver
                    global_a = a_best.copy()
                if obj_best < best_phase1_obj:
                    best_phase1_obj = obj_best
                    best_phase1_a = a_best.copy()

    # ============ PHASE 2: Deep exploitation (45% time) ============
    phase2_time = 0.45 * total_budget
    phase2_deadline = time.monotonic() + phase2_time

    if global_a is not None and phase2_time > 30:
        a = global_a.copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.full(n, S / n)
            s = S
        a = a * (S / s)

        print("\nPhase 2: Deep optimization, budget=%.0fs, current obj=%.7f"
              % (phase2_time, global_best))

        # Multiple rounds with different p ranges
        rounds = [
            (0.25, 8.0, 64.0, 0.012, "mid_p"),
            (0.20, 32.0, 128.0, 0.008, "high_p"),
            (0.20, 2.0, 16.0, 0.025, "low_p"),
            (0.15, 16.0, 96.0, 0.01, "high_p2"),
            (0.20, 4.0, 48.0, 0.018, "mid_p2"),
        ]

        for frac, p_lo, p_hi, lr_val, label in rounds:
            r_time = frac * phase2_time
            if time.monotonic() > phase2_deadline - 10:
                break
            if r_time < 5:
                continue

            a_best, _ = run_adam_pnorm(
                a, n, S, r_time,
                p_init=p_lo, p_final=p_hi, lr=lr_val,
                seed=int(time.monotonic() * 1000) % 100000
            )
            a_best = np.maximum(a_best, 0.0)
            s = a_best.sum()
            if s > 0.01:
                a_best = a_best * (S / s)
                ver = evaluate_sequence(a_best.tolist())
                if np.isfinite(ver) and ver < global_best:
                    global_best = ver
                    global_a = a_best.copy()
                    a = a_best.copy()
                    print("  %s (p=%g-%g): obj=%.7f" % (label, p_lo, p_hi, ver))
                else:
                    a = a_best.copy()

    # ============ PHASE 3: Coordinate descent + swaps (15% time) ============
    phase3_time = 0.15 * total_budget
    if global_a is not None and phase3_time > 10:
        a = global_a.copy()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.01:
            a = a * (S / s)

        print("\nPhase 3: CD + Swaps, budget=%.0fs" % phase3_time)

        # Single-coordinate CD
        cd_time = 0.5 * phase3_time
        a_best, _ = coordinate_descent_swaps(a, n, S, cd_time, max_sweeps=300)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            a = a_best.copy()
            print("  CD: obj=%.7f" % ver)

        # Pairwise swaps
        swap_time = 0.3 * phase3_time
        if time.monotonic() < deadline - 10 and swap_time > 3:
            a_best, _ = coordinate_swaps(a, n, S, swap_time, n_trials=10000)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                a = a_best.copy()
                print("  Swaps: obj=%.7f" % ver)

        # Zero masking
        a_best = try_zero_masking(a, n, S, n_zero=30)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  ZeroMask: obj=%.7f" % ver)

        # Final short CD
        rem = min(0.2 * phase3_time, deadline - time.monotonic() - 5)
        if rem > 2:
            a_best, _ = coordinate_descent_swaps(a, n, S, rem, max_sweeps=100, seed=99)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  Final CD: obj=%.7f" % ver)

    # ============ PHASE 4: Annealing restarts (10% time) ============
    phase4_time = deadline - time.monotonic()
    if global_a is not None and phase4_time > 20:
        print("\nPhase 4: Annealing restarts, budget=%.0fs" % phase4_time)
        a = global_a.copy()
        n_rounds = max(3, int(phase4_time / 30))
        per_round = phase4_time / (n_rounds + 2)

        for r in range(n_rounds):
            if time.monotonic() > deadline - 15:
                break
            if per_round < 3:
                break

            # Geometrically decreasing noise
            noise_scale = 0.03 * (0.7 ** r)
            a_noisy = a + noise_scale * rng.standard_normal(n)
            a_noisy = np.maximum(a_noisy, 1e-8)
            s = a_noisy.sum()
            if s < 0.01:
                continue
            a_noisy = a_noisy * (S / s)

            a_best, _ = run_adam_pnorm(
                a_noisy, n, S, per_round,
                p_init=8.0, p_final=64.0, lr=0.012,
                seed=1000 + r * 7
            )
            a_best = np.maximum(a_best, 0.0)
            s = a_best.sum()
            if s > 0.01:
                a_best = a_best * (S / s)
                ver = evaluate_sequence(a_best.tolist())
                if np.isfinite(ver) and ver < global_best:
                    global_best = ver
                    global_a = a_best.copy()
                    a = a_best.copy()
                    print("  Restart %d (noise=%.4f): obj=%.7f" % (r, noise_scale, ver))

    # Final verification
    if global_a is None:
        global_a = np.ones(n) * (S / n)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(n) * (S / n)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f (verified=%.7f)" % (global_best, ver))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)

    return final_a
