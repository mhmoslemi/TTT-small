    # Let i = k-j: g[j] = sum_{i=0}^{n-1} W[i+j] * a[i]
    
    # This is the cross-correlation of W and a at shift j:
    # (W ⊛ a)[j] = sum_i W[i+j] * a[i]
    
    # In FFT terms: ifft(fft(W) * conj(fft(a)))... let me verify.
    # Actually, the cross-correlation (x ⊛ y)[m] = sum_i x[i+m]*y[i] = sum_i x[n-i-m]*y[i] (for circular)
    # For linear: (x ⊛ y)[m] = sum_i x[i+m]*y[i]
    
    # Using the property: x ⊛ y = conj(x * flip(y)) where * is convolution
    # Or in frequency: FFT(x ⊛ y) = conj(FFT(x)) * FFT(y) ... no that's not right either.
    
    # Let me just use the direct formula:
    # g[j] = sum_i W[i+j] * a[i] for i = 0,...,n-1
    # This is the cross-correlation of W (length 2n-1) with a (length n) at lag j.
    
    # For linear cross-correlation: (W ⊛ a)[j] = sum_i W[i+j] * a[i]
    # Length of result: (2n-1) + n - 1 = 3n-2, but we only need j = 0,...,n-1.
    
    # Using FFT: zero-pad both to length >= (2n-1)+n-1 = 3n-2, take next power of 2.
    fsize = 1 << (3 * n - 2 - 1).bit_length()
    # Hmm, that might be wasteful. Let me think of a better way.
    
    # Actually, I realize that g[j] = sum_k W[k] * a[k-j] is just a "sliding dot product" 
    # of W (window) with a. For each j, we're taking W[j:j+n] dot a[0:n].
    
    # Alternatively: g = ifft(conj(fft(W, fsize)) * fft(a, fsize))... 
    # No, let me use the standard formula.
    
    # Cross-correlation: (x ⋆ y)[m] = sum_n x*[n] * y[n+m]  (for complex)
    # For real: (x ⋆ y)[m] = sum_n x[n] * y[n+m]
    
    # We want g[j] = sum_i W[i+j] * a[i]
    # = (a ⋆ W)[j]  (cross-correlation of a and W at lag j)
    
    # In FFT: FFT(a ⋆ W) = conj(FFT(a)) * FFT(W)
    # So g = real(ifft(conj(fft(a, fsize)) * fft(W, fsize)))[:n]
    
    W_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        W_full[int(k)] = weights[i]
    
    fsize = 1 << (len(W_full) + n - 1 - 1).bit_length()
    W_fft = fft(W_full, fsize)
    A_fft = fft(a, fsize)
    g = 2.0 * np.real(ifft(np.conj(A_fft) * W_fft))[:n]
    
    return g


def objective_val(a):
    """Compute objective 2n*max_conv/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def run_adaptive_sg(a_init, n, S, max_time, rng, mode="combined"):
    """
    Adaptive subgradient descent with multi-peak tracking.
    
    Parameters:
    - a_init: initial sequence (will be normalized to sum=S)
    - n: length
    - S: target sum (sqrt(2n))
    - max_time: time budget in seconds
    - rng: random number generator
    - mode: "explore", "exploit", or "combined"
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    # Parameters
    base_step = S / (2.0 * np.sqrt(n))
    top_k_init = 8
    top_k_max = 64
    noise_base = 0.01 * S / np.sqrt(n)
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Compute convolution
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        
        # Adaptive top-k: start small, grow over time
        progress = min(1.0, it / 5000.0)
        k = top_k_init + int(progress * (top_k_max - top_k_init))
        
        # Compute subgradient
        quad = progress > 0.3  # Use quadratic weighting after some progress
        g = compute_subgradient(a, c, k, quad_weight=quad)
        gnorm = np.linalg.norm(g)
        
        if gnorm < 1e-14:
            break
        
        # Normalize gradient direction
        g = g / gnorm
        
        # Adaptive step size with decay
        step = base_step * (1.0 + 0.5 * (1.0 - progress))
        
        # Candidate step
        x_new = project_simplex(a - step * g, S)
        
        # Armijo-like check: if it made things worse, halve the step
        c_new = fft_convolve_self(x_new)
        if c_new.max() > cur_max * 1.02:
            # Backtrack
            for _ in range(5):
                step *= 0.5
                x_new = project_simplex(a - step * g, S)
                c_new = fft_convolve_self(x_new)
                if c_new.max() <= cur_max * 1.02:
                    break
        
        a = x_new
        
        # Periodic noise injection (simulated annealing style)
        temperature = max(0.001, 1.0 - progress)
        if it % 500 == 0 and rng.random() < 0.3 * temperature:
            noise_scale = noise_base * (1.0 + 2.0 * temperature)
            noise = rng.normal(0, noise_scale, n)
            a = project_simplex(a + noise, S)
        
        # Periodic restart from best (to avoid drift)
        if it % 2000 == 0 and rng.random() < 0.15:
            a = best_local.copy()
            noise = rng.normal(0, noise_base * 0.5, n)
            a = project_simplex(a + noise, S)
    
    return best_local, best_score


def run_conjugate_subgradient(a_init, n, S, max_time, rng):
    """
    Conjugate-gradient-like subgradient method.
    Maintains a search direction that is conjugate to the previous direction
    (in the simplex null space).
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    d = np.zeros(n)  # search direction
    base_step = S / (2.0 * np.sqrt(n))
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        
        # Compute subgradient
        k = min(32, 8 + it // 200)
        g = compute_subgradient(a, c, k, quad_weight=True)
        
        # Project gradient onto null space of sum constraint
        g_mean = g.mean()
        g_null = g - g_mean  # sum(g_null) = 0
        
        # Polak-Ribiere beta
        if it > 1 and np.linalg.norm(d) > 1e-14:
            g_prev = compute_subgradient(a + d, c, k)  # approximate
            # Actually this is expensive. Let's use a simpler momentum.
            beta = 0.9
        else:
            beta = 0.0
        
        d_new = -g_null + beta * d
        d_norm = np.linalg.norm(d_new)
        if d_norm > 1e-14:
            d_new = d_new / d_norm
        
        # Step with line search
        step = base_step * max(0.3, 1.0 - 0.5 * min(1.0, it / 3000.0))
        x_new = project_simplex(a + step * d_new, S)
        
        # Check improvement
        c_new = fft_convolve_self(x_new)
        if c_new.max() < cur_max:
            a = x_new
            d = d_new
        else:
            # Try smaller step
            step *= 0.3
            x_new = project_simplex(a + step * d_new, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max:
                a = x_new
                d = d_new
            else:
                # Reset direction
                d = -g_null / max(np.linalg.norm(g_null), 1e-14)
                step = base_step * 0.3
                x_new = project_simplex(a + step * d, S)
                a = x_new
        
        # Noise
        if it % 1000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            d = np.zeros(n)
    
    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng):
    """
    Multiplicative weights / mirror descent approach.
    Instead of Euclidean projection, use KL projection (which is just normalization
    after exponentiation). This keeps values positive automatically.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    # Ensure strictly positive for log
    a = np.maximum(a, 1e-10)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    eta = 0.1  # learning rate
    base_eta = eta
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        
        # Compute subgradient
        k = min(32, 8 + it // 300)
        g = compute_subgradient(a, c, k, quad_weight=True)
        
        # Mirror descent update: a_new[j] = a[j] * exp(-eta * g[j])
        # Then normalize to sum S
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * g
        # Clip to prevent overflow
        log_a_new = np.clip(log_a_new, -50, 50)
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / a_new_sum)
        
        a = a_new
        
        # Decay learning rate
        eta = base_eta * (1.0 + 0.001 * it) ** (-0.7)
        
        # Noise
        if it % 500 == 0 and rng.random() < 0.2:
            log_a = np.log(np.maximum(a, 1e-300))
            log_a += rng.normal(0, 0.05, n)
            log_a = np.clip(log_a, -50, 50)
            a = np.exp(log_a)
            a = a * (S / a.sum())
    
    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Algorithm: Multi-method adaptive optimization combining:
1. Efficient FFT-based multi-peak subgradient with quadratic weighting
2. Armijo backtracking line search for robust step selection
3. Mirror descent (multiplicative weights) as an alternative geometry
4. Coordinate-wise refinement on top-k coordinates for final precision
5. Diverse asymmetric initializations with adaptive time allocation
6. Periodic simulated annealing restarts to escape local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, c, top_k, quad_weight):
    """
    Weighted subgradient of max(c) w.r.t. a using top-k peaks.
    g[j] = 2 * sum_k w_k * a[k-j] (with boundary handling).
    Computed via FFT cross-correlation.
    """
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]

    if quad_weight:
        threshold = top_vals.min()
        excess = top_vals - threshold
        total_excess = excess.sum()
        if total_excess < 1e-30:
            weights = np.ones(top_k) / top_k
        else:
            weights = (excess / total_excess) ** 2
            weights = weights / weights.sum()
    else:
        total = top_vals.sum()
        if total < 1e-30:
            weights = np.ones(top_k) / top_k
        else:
            weights = top_vals / total

    W_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        W_full[int(k)] = weights[i]

    fsize = 1 << (3 * n - 3).bit_length()
    W_fft = fft(W_full, fsize)
    A_fft = fft(a, fsize)
    g = 2.0 * np.real(ifft(W_fft * np.conj(A_fft)))[:n]
    return g


def objective_val(a):
    """Compute 2n*max_conv/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def run_subgradient(a_init, n, S, max_time, rng):
    """
    Adaptive multi-peak subgradient descent with backtracking.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    base_step = S / (2.0 * np.sqrt(n))

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / 8000.0)

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        k = 8 + int(progress * 56)
        quad = progress > 0.2
        g = compute_subgradient(a, c, k, quad)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g = g / gnorm

        # Step with decay
        step = base_step * (0.8 + 0.6 * (1.0 - progress))

        # Backtracking line search
        for attempt in range(6):
            x_new = project_simplex(a - step * g, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur * (1.0 + 0.5 * (1.0 - progress)):
                break
            step *= 0.5

        a = x_new

        # Annealing kicks
        temp = max(0.005, 1.0 - progress)
        if it % 400 == 0 and rng.random() < 0.4 * temp:
            scale = S / np.sqrt(n) * (0.01 + 0.04 * temp)
            a = project_simplex(a + rng.normal(0, scale, n), S)
        if it % 3000 == 0 and rng.random() < 0.1:
            a = best_local.copy()
            a = project_simplex(a + rng.normal(0, S / np.sqrt(n) * 0.02, n), S)

    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng):
    """
    Mirror descent with KL projection (multiplicative weights update).
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    a = np.maximum(a, 1e-10)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    eta0 = 0.15

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / 6000.0)

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        k = 8 + int(progress * 48)
        g = compute_subgradient(a, c, k, quad_weight=(progress > 0.1))
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        # Mirror descent: a <- a * exp(-eta * g / gnorm), then normalize
        eta = eta0 * (1.0 + 0.002 * it) ** (-0.6) / gnorm
        log_a = np.log(a) - eta * g
        log_a = np.clip(log_a, -80, 80)
        a_new = np.exp(log_a)
        a_sum = a_new.sum()
        if a_sum < 1e-300:

        else:

            a = np.exp(log_a)

def coordinate_refine(a, n, S, max_time, rng, top_coords=200):

    # Also include coordinates near the peak lags

            if 0 <= k - j < n and j not in top_idx:

    top_idx = top_idx[:top_coords]

    iteration = 0
    while time.monotonic() - start < max_time:
        iteration += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Try adjusting each top coordinate
        for j in top_idx:
            orig = a[j]
            # Try several multipliers

                c_check = fft_convolve_self(a)
                if c_check.max() < cur:
                    cur = c_check.max()
                    # Accept this change
                else:
                    a[j] = orig

        # Renormalize
        a_sum = a.sum()
        if a_sum > 1e-15:
            a = a * (S / a_sum)

    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
