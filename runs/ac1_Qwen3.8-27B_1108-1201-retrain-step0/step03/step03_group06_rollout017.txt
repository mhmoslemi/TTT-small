from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

        if s < 1e-15:
            return np.full(len(a), S / len(a))
        return a * (S / s)
    else:
        # Need to reduce sum: find threshold t such that sum(max(a-t, 0)) = S
        # Bisection
        lo = 0.0
        hi = a.max()
        if hi - lo < 1e-15:
            return np.full(len(a), S / len(a))
        for _ in range(60):
            mid = (lo + hi) / 2
            s_mid = np.maximum(a - mid, 0).sum()
            if s_mid > S:
                lo = mid
            else:
                hi = mid
        t = (lo + hi) / 2
        result = np.maximum(a - t, 0.0)
        # Fix rounding
        s_result = result.sum()
        if s_result > 0:
            result = result * (S / s_result)
        else:
            result = np.full(len(a), S / len(a))
        return result


def subgradient_max(a, c, m_star):
    """Compute subgradient of max(c) = c[m_star] with respect to a.
    g[i] = 2 * a[m_star - i] (with a[j]=0 outside [0,n-1])
    """
    n = len(a)
    g = np.zeros(n)
    # g[i] = 2 * a[m_star - i] for i where 0 <= m_star - i <= n-1
    # i.e., m_star - (n-1) <= i <= m_star
    i_lo = max(0, m_star - (n - 1))
    i_hi = min(n - 1, m_star)
    j_lo = m_star - i_hi  # = max(0, m_star - (n-1)) ... wait
    # g[i] = 2 * a[m_star - i], so for i in [i_lo, i_hi]:
    # m_star - i ranges from m_star - i_hi to m_star - i_lo
    # m_star - i_hi = m_star - min(n-1, m_star) = max(0, m_star - (n-1))
    # m_star - i_lo = m_star - max(0, m_star - (n-1)) = min(n-1, m_star)
    j_start = m_star - i_hi
    j_end = m_star - i_lo
    g[i_lo:i_hi+1] = 2.0 * a[j_start:j_end+1]
    return g


def subgradient_smooth(a, c, T):
    """Compute weighted subgradient using softmax weights.
    g[i] = 2 * sum_m w[m] * a[m-i]
    """
    n = len(a)
    c_max = c.max()
    scaled = T * (c - c_max)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # g[i] = 2 * sum_m w[m] * a[m-i]
    # This is 2 * (w * a)[i] where (w * a) is the cross-correlation
    # (w * a)[i] = sum_m w[m] a[m-i]
    # Using FFT: this is the linear cross-correlation of w and a
    # corr[i] = sum_m w[m] * a[m-i], which is (w cross-correlate a)[i]
    # In terms of convolution: corr = ifft(fft(w) * conj(fft(a))) circularly
    # For linear: zero-pad
    
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # Linear cross-correlation: corr[k] = sum_m w[m] a[m-k]
    # This is ifft(conj(fft(w)) * fft(a))[k] for circular
    # Or: corr = ifft(fft(w) * conj(fft(a))) gives sum_m w[m-shift] * a[m]
    # Let me be careful.
    
    # np.convolve(w, a[::-1]) reversed = cross-correlation
    # Actually, let's use: corr[i] = sum_m w[m] * a[m-i]
    # Let b = a reversed: b[j] = a[n-1-j]
    # Then corr = convolve(w, b) ... hmm this gets confusing.
    
    # Simple approach: corr[i] = sum_m w[m] a[m-i]
    # = sum_j w[j+i] a[j] (substituting m = j+i)
    # This is the (i-th element of) the cross-correlation of w and a.
    
    # Using FFT: cross_corr(w, a)[i] = ifft(conj(FFT(w)) * FFT(a))[i] (circular)
    # For linear, zero-pad to size >= len(w) + len(a) - 1
    fsize2 = 1 << (2 * size - 1).bit_length()
    W2 = fft(w, fsize2)
    A2 = fft(a, fsize2)
    corr = np.real(ifft(np.conj(W2) * A2))
    # corr[i] = sum_m w[m] a[m-i] for i = -(n-1), ..., n-1
    # We want i = 0, ..., n-1
    # The circular corr of length fsize2 has the linear corr at positions 0 and fsize2 - (n-1), ..., fsize2-1
    # Actually for the linear cross-correlation with zero padding:
    # corr_linear[i] for i=0..n-1 is at positions i in the circular result
    # (since we padded enough)
    
    return 2.0 * corr[:n]


def fista_smooth(a_init, n, S, time_budget, T_init=1.0, T_final=1000.0, 
                  alpha_init=0.4, verbose=False):
    """FISTA with smooth approximation of max(conv(a,a))."""
    a = normalize_nonneg(a_init, S).copy()
    
    # FISTA variables
    y = a.copy()
    t_k = 1.0
    t_prev = 1.0
    
    # Compute initial
    c_y = fft_convolve_self(y)
    best_obj = c_y.max()
    best_a = a.copy()
    
    start = time.monotonic()
    step = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.8)
        
        # FISTA update of y
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_prev ** 2)) / 2.0
        y_new = a + ((t_prev - 1.0) / t_k) * (a - y)
        y = normalize_nonneg(y_new, S)
        
        # Compute convolution and subgradient at y
        c_y = fft_convolve_self(y)
        
        # Smooth subgradient
        c_max = c_y.max()
        if c_max < 1e-30:
            break
        scaled = T * (c_y - c_max)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(c_y))
            w[np.argmax(c_y)] = 1.0
        else:
            w /= w_sum
        
        # g[i] = 2 * sum_m w[m] * a_y[m-i]
        # Using FFT for linear cross-correlation
        size = 2 * n - 1
        fsize = 1 << (2 * size - 1).bit_length()
        W = fft(w, fsize)
        Ay = fft(y, fsize)
        corr = np.real(ifft(np.conj(W) * Ay))
        g = 2.0 * corr[:n]
        
        # Step size: for smooth approximation, L ≈ 2 + T * (spread)
        # Be conservative
        alpha = alpha_init / (1.0 + 0.1 * T)
        
        # Gradient step
        a_new = y - alpha * g
        a = normalize_nonneg(a_new, S)
        
        # Check improvement
        # (Don't recompute c at a every step for speed; do it periodically)
        if step % 10 == 0:
            c_a = fft_convolve_self(a)
            obj_a = c_a.max()
            if obj_a < best_obj - 1e-12:
                best_obj = obj_a
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        t_prev = t_k
        step += 1
        
        if verbose and step % 1000 == 0:
            print("    FISTA step %d: best_obj=%.7f, T=%.1f" % (step, best_obj, T))
    
    return best_a, best_obj


def fista_nonsmooth(a_init, n, S, time_budget, verbose=False):
    """FISTA with exact subgradient (max of conv)."""
    a = normalize_nonneg(a_init, S).copy()
    y = a.copy()
    t_k = 1.0
    t_prev = 1.0
    
    c_y = fft_convolve_self(y)
    best_obj = c_y.max()
    best_a = a.copy()
    
    start = time.monotonic()
    step = 0
    no_improve = 0
    alpha = 0.35  # conservative step size (L ≈ 2, use 1/L * safety)
    
    while time.monotonic() - start < time_budget:
        # FISTA y update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_prev ** 2)) / 2.0
        y_new = a + ((t_prev - 1.0) / t_k) * (a - y)
        y = np.maximum(y_new, 0.0)
        s_y = y.sum()
        if s_y > 0:
            y = y * (S / s_y)
        
        # Compute convolution at y
        c_y = fft_convolve_self(y)
        m_star = int(np.argmax(c_y))
        obj_y = c_y[m_star]
        
        # Exact subgradient
        g = subgradient_max(y, c_y, m_star)
        
        # Gradient step
        a_new = y - alpha * g
        a = normalize_nonneg(a_new, S)
        
        # Check improvement periodically
        if step % 5 == 0:
            c_a = fft_convolve_self(a)
            obj_a = c_a.max()
            if obj_a < best_obj - 1e-12:
                best_obj = obj_a
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 200:
                    # Reduce step size
                    alpha *= 0.9
                    no_improve = 0
        
        t_prev = t_k
        step += 1
        
        if verbose and step % 1000 == 0:
            print("    FISTA-NS step %d: best_obj=%.7f, alpha=%.4f" % (step, best_obj, alpha))
    
    return best_a, best_obj


def gs_alternating(a_init, n, S, time_budget, n_iters=10000, verbose=False):
    """Gerchberg-Saxton style alternating projections.
    Time domain: enforce a >= 0, sum(a) = S
    Frequency domain: flatten |FFT(a_padded)|²
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    L = 1 << (2 * n).bit_length()  # power of 2 >= 2n
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    step = 0
    
    while time.monotonic() - start < time_budget and step < n_iters:
        # Step 1: Compute FFT of zero-padded a
        A = fft(a, L)
        
        # Step 2: Flatten the spectrum
        # Target: |A[k]|² = C for k >= 1, A[0] = S (DC fixed)
        # By Parseval: S² + (L-1)*C = L * sum(a²)
        # We want to minimize max(conv), which is related to making spectrum flat
        # Simple approach: set |A[k]| = sqrt(mean of |A[k]|² for k>=1) for k>=1
        mag = np.abs(A)
        mag[0] = S  # Fix DC
        # Make the rest as flat as possible
        mag_rest = mag[1:]
        target_mag = np.sqrt(np.sum(mag_rest ** 2) / (L - 1))
        mag[1:] = target_mag
        
        # Reconstruct with original phases
        phase = np.angle(A)
        A_new = mag * np.exp(1j * phase)
        
        # Step 3: Inverse FFT
        a_new = np.real(ifft(A_new))[:n]
        
        # Step 4: Time domain projection
        a = normalize_nonneg(a_new, S)
        
        # Check improvement periodically
        if step % 100 == 0:
            c = fft_convolve_self(a)
            obj = c.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
        
        step += 1
    
    c = fft_convolve_self(a)
    obj = c.max()
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a, best_obj


def gs_with_relaxation(a_init, n, S, time_budget, tau=0.5, verbose=False):
    """GS with relaxation factor to avoid oscillation."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    L = 1 << (2 * n).bit_length()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    step = 0
    
    while time.monotonic() - start < time_budget:
        A = fft(a, L)
        
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Flatten with DC constraint
        mag_dc = S
        mag_rest = mag[1:]
        energy_rest = np.sum(mag_rest ** 2)
        target_mag = np.sqrt(energy_rest / (L - 1))
        
        mag_new = np.full(L, target_mag)
        mag_new[0] = mag_dc
        
        A_new = mag_new * np.exp(1j * phase)
        a_gs = np.real(ifft(A_new))[:n]
        
        # Relaxation
        a = tau * a_gs + (1 - tau) * a
        a = normalize_nonneg(a, S)
        
        if step % 200 == 0:
            c = fft_convolve_self(a)
            obj = c.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
        
        step += 1
    
    c = fft_convolve_self(a)
    obj = c.max()
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a, best_obj


def coordinate_descent(a, n, S, time_budget, verbose=False):
    """Coordinate descent polish."""
    a = normalize_nonneg(a, S).copy()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(99)
    start = time.monotonic()
    factors = [0.5, 0.7, 1.3, 1.5, 0.3, 2.0, 0.8, 1.2]
    
    improved = True
    n_sweeps = 0
    while improved and time.monotonic() - start < time_budget:
        improved = False
        n_sweeps += 1
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                return best_a, best_obj
            orig = a[i]
            if orig < 1e-14:
                continue
            
            best_f = orig
            best_o = best_obj
            
            for f in factors:
                a[i] = orig * f
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                o = c_check.max()
                if o < best_o - 1e-14:
                    best_o = o
                    best_f = orig * f
            
            a[i] = best_f
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            
            if c.max() < best_obj - 1e-14:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
    
    return best_a, best_obj


def fista_backtracking(a_init, n, S, time_budget, T_init=1.0, T_final=500.0,
                       verbose=False):
    """FISTA with backtracking line search and temperature annealing."""
    a = normalize_nonneg(a_init, S).copy()
    y = a.copy()
    t_prev = 1.0
    
    c_y = fft_convolve_self(y)
    best_obj = c_y.max()
    best_a = a.copy()
    
    start = time.monotonic()
    step = 0
    alpha = 0.5  # initial step size
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.8)
        
        # FISTA y update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_prev ** 2)) / 2.0
        y_new = a + ((t_prev - 1.0) / t_k) * (a - y)
        y = normalize_nonneg(y_new, S)
        
        # Compute subgradient at y
        c_y = fft_convolve_self(y)
        
        if T > 100:
            # Near-nonsmooth: use exact subgradient
            m_star = int(np.argmax(c_y))
            g = subgradient_max(y, c_y, m_star)
        else:
            # Smooth: use weighted subgradient
            c_max = c_y.max()
            scaled = T * (c_y - c_max)
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros(len(c_y))
                w[np.argmax(c_y)] = 1.0
            else:
                w /= w_sum
            
            size = 2 * n - 1
            fsize = 1 << (2 * size - 1).bit_length()
            W = fft(w, fsize)
            Ay = fft(y, fsize)
            corr = np.real(ifft(np.conj(W) * Ay))
            g = 2.0 * corr[:n]
        
        # Backtracking: try to improve
        a_candidate = y - alpha * g
        a_candidate = normalize_nonneg(a_candidate, S)
        
        # Evaluate (expensive, do every step for now)
        c_cand = fft_convolve_self(a_candidate)
        obj_cand = c_cand.max()
        
        if obj_cand < c_y.max() - 1e-14:
            a = a_candidate
            # Can try increasing alpha
            if alpha < 0.6:
                alpha = min(0.6, alpha * 1.05)
            no_improve = 0
        else:
            # Reduce step size
            alpha *= 0.7
            a = y  # stay at y
            no_improve += 1
        
        # Track best
        if obj_cand < best_obj - 1e-12:
            best_obj = obj_cand
            best_a = a_candidate.copy()
        
        t_prev = t_k
        step += 1
        
        if verbose and step % 500 == 0:
            print("    FISTA-BT step %d: obj=%.7f, T=%.1f, alpha=%.4f" % (step, best_obj, T, alpha))
    
    return best_a, best_obj


def get_initialization(name, n, S, rng):
    """Generate various initializations."""
    i = np.arange(n, dtype=np.float64)
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "gaussian_peak":
        a = np.exp(-0.5 * ((i - n / 2) / (n / 6)) ** 2)
        return a * (S / a.sum())
    elif name == "cosine_mix":
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 7 * i / n + rng.random() * 2) + \
            0.2 * np.cos(2 * np.pi * 13 * i / n + rng.random() * 2) + \
            0.15 * np.cos(2 * np.pi * 29 * i / n + rng.random() * 2)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "sparse_random":
        a = rng.random(n)
        mask = rng.random(n) < 0.5
        a[~mask] = 0
        if a.sum() < 1e-10:
            a = rng.random(n)
        return a * (S / a.sum())
    elif name == "fourier_random":
        # Random phases in frequency domain
        L = 1 << (2 * n).bit_length()
        phases = rng.uniform(0, 2 * np.pi, L)
        mags = np.ones(L)
        A = mags * np.exp(1j * phases)
        A[0] = S
        a = np.real(ifft(A))[:n]
        return normalize_nonneg(a, S)
    else:
        return np.full(n, S / n)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function.
    
    Strategy:
    1. Start from given sequence or good initializations
    2. Use GS alternating projections for quick improvement
    3. Use FISTA with temperature annealing for convergence
    4. Polish with coordinate descent
    5. Multiple restarts to escape local minima
    """
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 30.0)
    
    global_best = np.inf
    global_a = None
    
    # Try to load given sequence
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            ver = evaluate_sequence(given_raw.tolist())
            n_given = len(given_raw)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_raw * (S_given / given_raw.sum())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given sequence: " + str(e))
    
    # Choose working length
    if global_a is not None:
        n = len(global_a)
    else:
        n = 4096
    
    S = np.sqrt(2.0 * float(n))
    print("Working with n=%d, S=%.4f" % (n, S))
    
    if global_a is None:
        # Generate initial best
        a_init = np.full(n, S / n)
        global_a = a_init
        global_best = evaluate_sequence(a_init.tolist())
    
    a = global_a.copy()
    a = normalize_nonneg(a, S)
    
    # ============ PHASE 1: GS initialization (10% time) ============
    gs_budget = 0.08 * total_budget
    print("\nPhase 1: GS Alternating (%.0fs)" % gs_budget)
    
    a_gs, obj_gs = gs_with_relaxation(a, n, S, gs_budget, tau=0.6)
    a_gs = normalize_nonneg(a_gs, S)
    ver = evaluate_sequence(a_gs.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_gs.copy()
        print("  GS: obj=%.7f" % ver)
    
    a = global_a.copy()
    
    # ============ PHASE 2: FISTA with annealing (45% time) ============
    fista_budget = 0.45 * total_budget
    print("\nPhase 2: FISTA Annealing (%.0fs)" % fista_budget)
    
    # Round 1: Low to medium T
    r1_budget = 0.35 * fista_budget
    a_best, _ = fista_backtracking(a, n, S, r1_budget, T_init=1.0, T_final=100.0)
    a_best = normalize_nonneg(a_best, S)
    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  FISTA T1-100: obj=%.7f" % ver)
    
    a = global_a.copy()
    
    # Round 2: Medium to high T
    r2_budget = 0.35 * fista_budget
    a_best, _ = fista_backtracking(a, n, S, r2_budget, T_init=50.0, T_final=500.0)
    a_best = normalize_nonneg(a_best, S)
    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  FISTA T50-500: obj=%.7f" % ver)
    
    a = global_a.copy()
    
    # Round 3: High T (near-nonsmooth)
    r3_budget = 0.30 * fista_budget
    a_best, _ = fista_nonsmooth(a, n, S, r3_budget)
    a_best = normalize_nonneg(a_best, S)
    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  FISTA-NS: obj=%.7f" % ver)
    
    # ============ PHASE 3: GS + FISTA alternating (25% time) ============
    alt_budget = 0.25 * total_budget
    print("\nPhase 3: GS+FISTA Alternating (%.0fs)" % alt_budget)
    
    a = global_a.copy()
    n_rounds = max(3, int(alt_budget / 30))

"""

2. Smooth approximation with adaptive temperature

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def simplex_project(v, S):
    """Project v onto {x >= 0, sum(x) = S}. O(n log n) via sorting."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Sort in descending order
    sorted_v = np.sort(v)[::-1]
    # Find threshold using cumulative sums
    cumsum = np.cumsum(sorted_v)
    # Look for rho: largest index where sorted_v[rho] > (cumsum[rho] - S) / (rho+1)
    rho_candidates = np.arange(1, n + 1)
    vals = sorted_v[:n] > (cumsum - S) / rho_candidates
    if not vals.any():
        return np.full(n, S / n)
    rho = int(np.max(rho_candidates[vals]))
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s > 0 and abs(s - S) > 1e-10:
        result = result * (S / s)
    elif s == 0:
        result = np.full(n, S / n)
    return result


def subgradient_shift(a, m_star):
    """Subgradient of max(conv(a,a)) at maximizer m_star.
    g[i] = 2 * a[m_star - i] with a[j]=0 outside [0,n-1].
    """
    n = len(a)
    g = np.zeros(n)
    # g[i] = 2*a[m_star-i] for valid i
    i_lo = max(0, m_star - (n - 1))
    i_hi = min(n - 1, m_star)
    if i_hi >= i_lo:
        j_start = m_star - i_hi
        j_end = m_star - i_lo
        g[i_lo:i_hi + 1] = 2.0 * a[j_start:j_end + 1]
    return g


def smooth_subgradient(a, c, T):
    """Weighted subgradient: g[i] = 2 * sum_m w[m] * a[m-i]."""
    n = len(a)
    c_max = c.max()
    scaled = T * (c - c_max)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # Linear cross-correlation via FFT
    size = 2 * n - 1
    fsize = 1 << (2 * size - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(np.conj(W) * A))
    return 2.0 * corr[:n]


def fista_core(a_init, n, S, time_budget, T_init=1.0, T_final=1000.0,

    The key insight: the subgradient of max(conv(a,a)) at maximizer m*
    is simply 2*shift(a, m*), computable in O(n).
    """
    a = simplex_project(a_init, S).copy()
    y = a.copy()
    t_prev = 1.0
    alpha = alpha_init

    c_y = fft_convolve_self(y)
    best_obj = c_y.max()
    best_a = a.copy()

    start = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 500

        progress = min(1.0, elapsed / time_budget)

        # Temperature schedule (exponential ramp)
        T = T_init * (T_final / T_init) ** (progress ** 0.7)

        # FISTA momentum update

        y = simplex_project(y_new, S)

        c_y = fft_convolve_self(y)

def gs_alternating(a_init, n, S, time_budget, tau=0.6, verbose=False):
    """Gerchberg-Saxton: alternate between time-domain projection
    and frequency-domain flattening."""
    a = simplex_project(a_init, S).copy()
    L = 1 << (2 * n).bit_length()

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    step = 0

    while time.monotonic() - start < time_budget:
        # Frequency domain: flatten spectrum
        A = fft(a, L)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: flat magnitude for k>=1, DC = S
        mag_rest = mag[1:]
        energy = np.sum(mag_rest ** 2)
        target = np.sqrt(energy / (L - 1))

        mag_new = np.full(L, target)
        mag_new[0] = S  # DC component

        # Relaxation in frequency domain
        mag_mix = tau * mag_new + (1 - tau) * mag
        A_new = mag_mix * np.exp(1j * phase)

        # Time domain
        a_gs = np.real(ifft(A_new))[:n]
        a = simplex_project(a_gs, S)

        # Periodic check
        if step % 100 == 0:
            c = fft_convolve_self(a)
            obj = c.max()
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a.copy()

        step += 1

    c = fft_convolve_self(a)
    if c.max() < best_obj:
        best_obj = c.max()
        best_a = a.copy()

    return best_a, best_obj


def sparsify(a, n, S, time_budget, verbose=False):
    """Greedily zero out coordinates that reduce the objective."""
    a = simplex_project(a, S).copy()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    n_zeroed = 0

    # Sort by value (smallest first)
    while time.monotonic() - start < time_budget:
        sorted_idx = np.argsort(a)
        improved = False
        for idx in sorted_idx:
            if a[idx] < 1e-12:
                continue
            if time.monotonic() - start > time_budget:
                return best_a, best_obj

            saved = a[idx]
            a[idx] = 0.0
            a = simplex_project(a, S)
            c_check = fft_convolve_self(a)
            obj_check = c_check.max()

            if obj_check < best_obj - 1e-10:
                best_obj = obj_check
                best_a = a.copy()
                n_zeroed += 1
                improved = True
            else:
                a[idx] = saved
                a = simplex_project(a, S)

        if not improved:
            break

    if verbose:
        print("      Sparsified %d coordinates, obj=%.7f" % (n_zeroed, best_obj))

    return best_a, best_obj


def coordinate_descent(a, n, S, time_budget, verbose=False):
    """Coordinate descent with multiplicative updates."""
    a = simplex_project(a, S).copy()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(77)
    start = time.monotonic()
    factors = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0])

    n_sweeps = 0
    while time.monotonic() - start < time_budget:
        improved = False
        n_sweeps += 1
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                return best_a, best_obj
            orig = a[i]
            if orig < 1e-14:
                continue

            best_local = orig
            best_local_obj = best_obj

            for f in factors:
                a[i] = orig * f
                a = simplex_project(a, S)
                c_check = fft_convolve_self(a)
                obj = c_check.max()
                if obj < best_local_obj - 1e-12:
                    best_local_obj = obj
                    best_local = orig * f

            a[i] = best_local
            a = simplex_project(a, S)
            c = fft_convolve_self(a)

            if c.max() < best_obj - 1e-12:
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        if not improved and n_sweeps > 2:
            break

    return best_a, best_obj

    elif name == "cosine_mix":
        a = (1.0 + 0.3 * np.cos(2 * np.pi * 7 * i / n + rng.random() * 2) +
             0.2 * np.cos(2 * np.pi * 13 * i / n + rng.random() * 2) +
             0.15 * np.cos(2 * np.pi * 29 * i / n + rng.random() * 2))
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "fourier_flat":
        L = 1 << (2 * n).bit_length()
        phases = rng.uniform(0, 2 * np.pi, L)
        A = np.exp(1j * phases)
        A[0] = S
        a = np.real(ifft(A))[:n]
        return simplex_project(a, S)
    elif name == "two_hump":
        a = (np.exp(-0.5 * ((i - n * 0.25) / (n / 8)) ** 2) +
             np.exp(-0.5 * ((i - n * 0.75) / (n / 8)) ** 2))
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function.
    
    Multi-phase strategy:
    1. Load given sequence, quick GS initialization
    2. FISTA with adaptive temperature (main optimization)
    3. Sparsification
    4. FISTA on sparsified
    5. Coordinate descent polish
    6. Final restarts
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(1.0, float(budget_s) - 30.0)

    global_best = np.inf
    global_a = None

    # Load given sequence
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            ver = evaluate_sequence(given_raw.tolist())
            n_given = len(given_raw)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_raw * (S_given / given_raw.sum())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: " + str(e))

    # Determine working length
    if global_a is not None:
        n = len(global_a)
    else:
        n = 4096
    S = np.sqrt(2.0 * float(n))
    print("Working: n=%d, S=%.4f, budget=%.0fs" % (n, S, total_budget))

    if global_a is None:
        a_init = get_initialization("fourier_flat", n, S, rng)
        global_a = a_init
        global_best = evaluate_sequence(global_a.tolist())
        print("Initial: obj=%.7f" % global_best)

    a = global_a.copy()
    a = simplex_project(a, S)

    def remaining_time():
        return max(0.0, total_budget - (time.monotonic() - t_start))

    # ============ PHASE 1: GS quick init (5%) ============
    p1_budget = min(0.05 * total_budget, remaining_time() * 0.5)
    if p1_budget > 5:
        print("\nPhase 1: GS (%.0fs)" % p1_budget)
        a_gs, _ = gs_alternating(a, n, S, p1_budget, tau=0.7)
        a_gs = simplex_project(a_gs, S)
        ver = evaluate_sequence(a_gs.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_gs.copy()
            print("  GS: obj=%.7f" % ver)
        a = global_a.copy()

    # ============ PHASE 2: Main FISTA (50%) ============
    p2_budget = 0.50 * total_budget
    if remaining_time() > p2_budget + 30:
        print("\nPhase 2: FISTA Main (%.0fs)" % p2_budget)

        # Round A: Low-to-medium T (smooth)
        rA_budget = 0.40 * p2_budget
        a_best, _ = fista_core(a, n, S, rA_budget,
                               T_init=0.5, T_final=50.0,
                               alpha_init=0.35, use_smooth=True)
        a_best = simplex_project(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  FISTA T0.5-50: obj=%.7f" % ver)

        a = global_a.copy()

        # Round B: Medium-to-high T
        rB_budget = 0.35 * p2_budget
        a_best, _ = fista_core(a, n, S, rB_budget,
                               T_init=20.0, T_final=300.0,
                               alpha_init=0.25, use_smooth=True,
                               adaptive_temp=True)
        a_best = simplex_project(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  FISTA T20-300: obj=%.7f" % ver)

        a = global_a.copy()

        # Round C: Nonsmooth (exact subgradient)
        rC_budget = 0.25 * p2_budget
        a_best, _ = fista_core(a, n, S, rC_budget,
                               T_init=100.0, T_final=1000.0,
                               alpha_init=0.2, use_smooth=False)
        a_best = simplex_project(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  FISTA NS: obj=%.7f" % ver)

    # ============ PHASE 3: Sparsification (15%) ============
    p3_budget = 0.15 * total_budget
    if remaining_time() > p3_budget + 20:
        print("\nPhase 3: Sparsify (%.0fs)" % p3_budget)
        a_sp, _ = sparsify(global_a, n, S, p3_budget * 0.5)
        a_sp = simplex_project(a_sp, S)
        ver = evaluate_sequence(a_sp.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_sp.copy()
            print("  Sparsified: obj=%.7f" % ver)

        # FISTA on sparsified
        a = global_a.copy()
        a_best, _ = fista_core(a, n, S, p3_budget * 0.5,
                               T_init=50.0, T_final=500.0,
                               alpha_init=0.25, use_smooth=False)
        a_best = simplex_project(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  FISTA post-sparsify: obj=%.7f" % ver)

    # ============ PHASE 4: CD Polish (10%) ============
    p4_budget = 0.10 * total_budget
    if remaining_time() > p4_budget + 10:
        print("\nPhase 4: CD Polish (%.0fs)" % p4_budget)
        a_cd, _ = coordinate_descent(global_a, n, S, p4_budget)
        a_cd = simplex_project(a_cd, S)
        ver = evaluate_sequence(a_cd.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_cd.copy()
            print("  CD: obj=%.7f" % ver)

    # ============ PHASE 5: Final FISTA + CD (20%) ============
    p5_budget = 0.20 * total_budget
    if remaining_time() > p5_budget * 0.5:
        print("\nPhase 5: Final Polish (%.0fs)" % p5_budget)
        a = global_a.copy()

        # High-T FISTA
        r1 = p5_budget * 0.5
        a_best, _ = fista_core(a, n, S, r1,
                               T_init=100.0, T_final=2000.0,
                               alpha_init=0.15, use_smooth=False)
        a_best = simplex_project(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  Final FISTA: obj=%.7f" % ver)

        # CD
        a = global_a.copy()
        r2 = max(0.0, remaining_time() - 10)
        if r2 > 10:
            a_cd, _ = coordinate_descent(a, n, S, r2)
            a_cd = simplex_project(a_cd, S)
            ver = evaluate_sequence(a_cd.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_cd.copy()
                print("  Final CD: obj=%.7f" % ver)

    # Final safety
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f (verified=%.7f)" % (global_best, ver if np.isfinite(ver) else float('inf')))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        print("FALLBACK used")

    return final_a
