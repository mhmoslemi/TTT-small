    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def log_space_objective_and_grad(w, beta, S, n):
    """
    Compute objective and gradient in log-space.
    a = softmax(w) * S, so sum(a) = S automatically.
    """
    w_max = w.max()
    exp_w = np.exp(w - w_max)
    p = exp_w / exp_w.sum()  # softmax
    a = p * S
    
    c = fft_convolve_self(a)
    val, w_peaks = smooth_max_val_and_weights(c, beta)
    
    # Objective: 2n * val / S^2 (but since S = sqrt(2n), this equals val)
    obj = val  # Since S^2 = 2n, objective = val
    
    # Gradient w.r.t. a
    grad_a = grad_a_from_weights(a, w_peaks, beta)
    
    # Chain rule: da[m]/dw[i] = S * p[i] * (delta_{mi} - p[m])
    # grad_w[i] = sum_m grad_a[m] * S * p[i] * (delta_{mi} - p[m])
    # = S * p[i] * (grad_a[i] - sum_m grad_a[m] * p[m])
    dot = np.dot(grad_a, p)
    grad_w = S * p * (grad_a - dot)
    
    return obj, grad_w

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def peak_targeting_optimize(a_init, n, S, max_time, rng, k_init=8, k_max=256):
    """
    Peak-targeting projected gradient with backtracking line search.
    """

    s = a.sum()

    no_improve = 0
    max_no_improve = 500

        k = min(k_max, k_init + it // 20)
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            # Restart with perturbation
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            no_improve = 0
            max_no_improve = max(100, max_no_improve // 2)
            continue
        
        # Get top-k peaks
        top_idx = np.argsort(c)[::-1][:k]
        top_c = c[top_idx]

        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)

            break

        alpha = step_size
        improved = False

            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max - 1e-10:

                improved = True
                break

                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
        
    return best_a, best_score

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines L-BFGS in log-space with adaptive smoothness, peak-targeting 
projected gradient with backtracking line search, alternating peak 
subsets, and multi-restart exploration. Starts from parent sequence.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_grad_from_weights(a, w_full):
    """grad_a[m] = 2 * sum_k w_full[k] * a[k-m] via FFT cross-correlation."""
    n = len(a)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lbfgs_logspace(a_init, n, S, max_time, beta):
    """
    L-BFGS optimization in log-space on the smooth-max objective.
    Returns optimized a.
    """
    state = {'start': time.monotonic(), 'max_time': max_time, 'exceeded': False}

    def fun_and_grad(w):
        if state['exceeded']:
            return np.inf, np.zeros_like(w)
        if time.monotonic() - state['start'] > max_time:
            state['exceeded'] = True
            return np.inf, np.zeros_like(w)

        w_max = w.max()
        exp_w = np.exp(w - w_max)
        p = exp_w / exp_w.sum()
        a = p * S

        c = fft_convolve_self(a)
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            return 0.0, np.zeros_like(w)
        w_peaks = exp_bc / total
        val = c_max + np.log(total) / beta

        grad_a = compute_grad_from_weights(a, w_peaks)
        dot = np.dot(grad_a, p)
        grad_w = S * p * (grad_a - dot)

        return val, grad_w

    p_init = np.clip(a_init, 1e-15, None)
    w0 = np.log(p_init / p_init.sum())

    maxiter = max(200, int(max_time * 30))
    maxfun = max(maxiter * 5, 1000)

    try:
        result = minimize(
            fun_and_grad, w0, method='L-BFGS-B',
            options={'maxiter': maxiter, 'maxfun': maxfun, 'ftol': 1e-14, 'gtol': 1e-9}
        )
        w_opt = result.x
    except Exception:
        w_opt = w0

    w_max = w_opt.max()
    exp_w = np.exp(w_opt - w_max)
    p_opt = exp_w / exp_w.sum()
    a_opt = p_opt * S
    return a_opt


def peak_targeting(a_init, n, S, max_time, rng, k_min=4, k_max=256, alternating=True):
    """
    Peak-targeting projected gradient with backtracking line search.
    Alternates between different subsets of top peaks to avoid over-fitting.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    start = time.monotonic()
    it = 0
    step_size = S * 0.5 / np.sqrt(n)
    no_improve = 0
    offset = 0

    while time.monotonic() - start < max_time:
        it += 1
        k = min(k_max, k_min + it // 15)

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score - 1e-13:
            best_score = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200 + it // 10:
            noise_scale = 0.005 * S / np.sqrt(n) * (1.0 + no_improve / 500.0)
            noise = rng.normal(0, noise_scale, n)
            a = project_simplex(a + noise, S)
            no_improve = 0
            continue

        # Select active peaks with alternating offset
        if alternating and k > 4:
            top_all = np.argsort(c)[::-1][:min(2 * k, 2 * n - 1)]
            offset = offset % max(1, len(top_all) - k)
            top_idx = top_all[offset:offset + k]
        else:
            top_idx = np.argsort(c)[::-1][:k]

        top_c = c[top_idx]
        # Weight by proximity to max (sharper focus on peaks)
        w = top_c ** 4
        w_total = w.sum()
        if w_total < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / w_total

        w_full = np.zeros(2 * n - 1)
        for i in range(len(top_idx)):
            w_full[int(top_idx[i])] = w[i]

        grad = compute_grad_from_weights(a, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        # Backtracking line search
        alpha = step_size
        improved = False
        for _ in range(12):
            a_new = project_simplex(a - alpha * grad, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max - 1e-12:
                a = a_new
                improved = True
                step_size = min(step_size * 1.05, S / np.sqrt(n))
                break
            alpha *= 0.5

        if not improved:
            step_size = max(step_size * 0.6, 1e-8 * S)
            if it % 50 == 0:
                noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)

        offset += 1

    return best_a, best_score


def alternating_peak_optimize(a_init, n, S, max_time, rng):
    """
    More aggressive peak optimization: cycles through different peak groupings,
    using weighted gradients with adaptive focusing.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    start = time.monotonic()
    it = 0
    base_step = S * 0.3 / np.sqrt(n)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score - 1e-13:
            best_score = cur_max
            best_a = a.copy()

        # Use top peaks with emphasis on the very top
        n_peaks = min(64, 8 + it // 10)
        top_idx = np.argsort(c)[::-1][:n_peaks]
        top_c = c[top_idx]

        # Exponential weighting: emphasize peaks closest to max
        c_range = top_c[0] - top_c[-1] + 1e-10
        w = np.exp(10.0 * (top_c - top_c[-1]) / c_range)
        w = w / w.sum()

        w_full = np.zeros(2 * n - 1)
        for i in range(len(top_idx)):
            w_full[int(top_idx[i])] = w[i]

        grad = compute_grad_from_weights(a, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        # Adaptive step: larger when making progress, smaller when stuck
        progress = (best_score - c.max()) / (best_score + 1e-10) if best_score > c.max() else 0.0
        alpha = base_step * (1.0 + 2.0 * progress)

        a_new = project_simplex(a - alpha * grad, S)
        c_new = fft_convolve_self(a_new)
        if c_new.max() > cur_max:
            # Backtrack
            for factor in [0.5, 0.25, 0.1, 0.05]:
                a_try = project_simplex(a - alpha * factor * grad, S)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < cur_max:
                    a_new = a_try
                    break
            else:
                a_new = a  # No improvement, keep current

        a = a_new

    return best_a, best_score
