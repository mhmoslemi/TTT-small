"""
Minimize 2*n*max_conv/sum^2 using Adam optimizer with softmax parameterization
and log-sum-exp continuation, plus spectral initialization and Nesterov polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

# Helper functions at top level

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_smoothing_grad(a, c, beta):
    """Compute smoothed objective value and gradient w.r.t. a."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp = exp_bc.sum()
    obj = (c_max + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp
    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (cross-correlation)
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return obj, grad_a

def softmax_from_z(z, S):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return S * exp_z / exp_z.sum()

def run_adam_continuation(a_init, n, S, max_time, rng):
    """Adam optimizer with log-sum-exp continuation."""
    # Initialize z from a
    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()
    
    beta_schedule = [2, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
    
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break
        
        # Learning rate schedule for this beta
        base_lr = 0.5 / (1 + 0.1 * np.log1p(beta))
        n_iters = 2000
        
        for it in range(n_iters):
            if time.monotonic() - start > max_time - 3:
                break
            
            t += 1
            a = softmax_from_z(z, S)
            c = fft_convolve_self(a)
            obj, grad_a = compute_smoothing_grad(a, c, beta)
            
            # Convert to z-gradient
            dot_ga_a = np.dot(grad_a, a)
            grad_z = a * (grad_a - dot_ga_a / S)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            z = z - base_lr * m_hat / (np.sqrt(v_hat) + eps)
            z = z - z.mean()  # Re-center for stability
            
            # Check improvement
            if it % 100 == 99:
                a_check = softmax_from_z(z, S)
                c_check = fft_convolve_self(a_check)
                score = c_check.max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
    
    # Final check
    a_final = softmax_from_z(z, S)
    c_final = fft_convolve_self(a_final)
    if c_final.max() < best_score:
        best_score = c_final.max()
        best_a = a_final.copy()
    
    return best_a, best_score

def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=256):
    """Nesterov-accelerated top-k subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        w_full = np.zeros(len(c))
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]
        
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new
        
        if it % 10000 == 0 and rng.random() < 0.05:
            y = project_simplex(y + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            t_k = 1.0
    
    return best_local, best_score

def spectral_init(n, S, rng, n_gs_iter=50):
    """Spectral initialization via Gerchberg-Saxton style iteration."""
    # Start with random non-negative sequence
    a = rng.random(n) + 0.1
    
    # Zero-pad to 2n for FFT
    fsize = 1 << (2 * n - 1).bit_length()
    
    for _ in range(n_gs_iter):
        A = fft(a, fsize)
        mag = np.abs(A)
        # Target: constant magnitude (geometric mean)
        target_mag = np.geomsort(mag)[n_gs_iter] if n_gs_iter < len(mag) else np.sqrt(np.mean(mag**2))
        # Use a slightly tapered target
        target = np.full(fsize, np.sqrt(np.mean(mag**2)))
        target[0] = mag[0]  # Keep DC
        # Phase from current
        phase = np.angle(A)
        A_new = target * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, None))[:n]
        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * S / s
    
    return a

def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

"""
Minimize 2*n*max_conv/sum^2 using Adam optimizer with softmax parameterization,
log-sum-exp continuation, spectral-inspired initialization, and Nesterov polishing.
Key improvements: Adam (better for non-quadratic), larger n option, LR warmup+decay,
gradient clipping, and adaptive time allocation.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax_from_z(z, S):
    """Convert unconstrained z to a = S * softmax(z)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return S * exp_z / exp_z.sum()


def compute_smooth_obj_grad(a, c, beta):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. a."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp = exp_bc.sum()
    obj = (c_max + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return obj, grad_a


def z_to_grad_a_to_grad_z(a, grad_a, S):
    """Convert gradient w.r.t. a to gradient w.r.t. z (softmax params)."""
    dot_val = np.dot(grad_a, a)
    return a * (grad_a - dot_val / S)


def run_adam_continuation(a_init, n, S, max_time, rng):
    """Adam optimizer with log-sum-exp continuation on softmax params."""
    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()

    beta_schedule = [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096]
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break

        # Adaptive iterations per beta level
        remaining_frac = 1.0 - (beta / 4096.0)
        n_iters = max(200, int(4000 * remaining_frac))

        # Learning rate: base / sqrt(beta) with warmup
        base_lr = 0.8 / np.sqrt(1.0 + 0.05 * beta)
        warmup_iters = min(50, n_iters // 10)

        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.9, 0.999, 1e-8

        for it in range(n_iters):
            if time.monotonic() - start > max_time - 3:
                break

            t_adam += 1
            a = softmax_from_z(z, S)
            c = fft_convolve_self(a)
            obj, grad_a = compute_smooth_obj_grad(a, c, beta)
            grad_z = z_to_grad_a_to_grad_z(a, grad_a, S)

            # Gradient clipping
            gnorm = np.linalg.norm(grad_z)
            if gnorm > 10.0:
                grad_z = grad_z * (10.0 / gnorm)

            # Learning rate with warmup and cosine decay
            if it < warmup_iters:
                lr_mult = (it + 1) / warmup_iters
            else:
                progress = (it - warmup_iters) / max(1, n_iters - warmup_iters)
                lr_mult = 0.5 * (1.0 + np.cos(np.pi * progress))
            lr = base_lr * lr_mult

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * grad_z * grad_z
            m_hat = m / (1.0 - beta1 ** t_adam)
            v_hat = v / (1.0 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z = z - z.mean()

            # Periodic check
            if it % 200 == 199:
                a_check = softmax_from_z(z, S)
                c_check = fft_convolve_self(a_check)
                score = c_check.max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

    # Final evaluation
    a_final = softmax_from_z(z, S)
    c_final = fft_convolve_self(a_final)
    if c_final.max() < best_score:
        best_score = c_final.max()
        best_a = a_final.copy()

    return best_a, best_score


def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=512):
    """Nesterov-accelerated top-k subgradient with adaptive step."""
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    best_step = 2.0 * S / n

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 80)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(len(c))
        w_full[top_indices] = weights

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.0015 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

        # Random restart
        if it % 15000 == 0 and rng.random() < 0.06:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t_k = 1.0

    return best_local, best_score


def spectral_init(n, S, rng, n_gs_iter=40):
    """Initialize via spectral flattening (Gerchberg-Saxton style)."""
    a = rng.random(n) + 0.1
    fsize = 1 << (2 * n - 1).bit_length()

    for i in range(n_gs_iter):
        A = fft(a, fsize)
        mag = np.abs(A)
        log_mag = np.log(mag + 1e-30)
        target_mag = np.exp(np.mean(log_mag))
        # Slight taper: reduce edges
        target = np.full(fsize, target_mag)
        target[0] = 0.7 * mag[0] + 0.3 * target_mag
        target[fsize // 2] = 0.5 * target_mag
        # Taper last few bins
        for j in range(1, 6):
            target[j] = target_mag * (1.0 - 0.05 * j)
            target[fsize - j] = target_mag * (1.0 - 0.05 * j)
        phase = np.angle(A)
        A_new = target * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, None))[:n]
        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * S / s
    return a


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    from scipy.optimize import minimize as scipy_minimize

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break

        def fun(z_val):
            a_v = softmax_from_z(z_val, S)
            c_v = fft_convolve_self(a_v)

            a_v = softmax_from_z(z_val, S)
            c_v = fft_convolve_self(a_v)

        t0 = time.monotonic()

            t0 = time.monotonic()
            a_norm = safe_normalize(arr, n)
            a_new, score = run_adam_continuation(a_norm, n, S, per_init_adam, rng)
            a_new = safe_normalize(a_new, n)
            c_new = fft_convolve_self(a_new)
            final_score = c_new.max()

            if final_score < global_best:
                global_best = final_score
                global_a = a_new.copy()
            if final_score < best_after_adam_score:
                best_after_adam_score = final_score
                best_after
                best_after_adam = a_new.copy()

            elapsed = time.monotonic() - t0
            tag = "OK" if final_score < best_after_adam_score else "  "
            print("  [Adam] %-12s: %.8f (%ss)" % (name, final_score, "%.1f" % elapsed))

        # Phase B: L-BFGS polish on best Adam result
        if best_after_adam is not None and time.monotonic() < deadline - 30:
            t0 = time.monotonic()
            a_new, score = run_lbfgs_polish(best_after_adam, n, S, t_lbfgs, rng)
            a_new = safe_normalize(a_new, n)
            c_new = fft_convolve_self(a_new)
            final_score = c_new.max()
            if final_score < global_best:
                global_best = final_score
                global_a = a_new.copy()
            print("  [L-BFGS] polish: %.8f (%.1fs)" % (final_score, time.monotonic() - t0))
            best_after_adam = a_new
            best_after_adam_score = final_score

        # Phase C: Nesterov polishing
        if best_after_adam is not None and time.monotonic() < deadline - 20:
            t0 = time.monotonic()
            a_new, score = run_nesterov_toppk(best_after_adam, n, S, t_nesterov, rng,
                                              top_k_init=10, top_k_max=512)
            a_new = safe_normalize(a_new, n)
            c_new = fft_convolve_self(a_new)
            final_score = c_new.max()
            if final_score < global_best:
                global_best = final_score
                global_a = a_new.copy()
            print("  [Nesterov] n=%d: %.8f (%.1fs)" % (n, final_score, time.monotonic() - t0))

    # Final phase: perturbation restarts on global best
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            pert_n = len(global_a)
            pert_S = np.sqrt(2.0 * float(pert_n))
            n_perturb = max(2, int(remaining / 50))
            perturb_time = remaining * 0.7 / n_perturb

            print("\n=== Final: Perturbation restarts (%d attempts) ===" % n_perturb)
            for p in range(n_perturb):
                if time.monotonic() > deadline - 15:
                    break

                noise_scale = 0.025 / (1.0 + 0.4 * p)
                a_pert = global_a + rng.normal(0, noise_scale * pert_S / np.sqrt(pert_n), pert_n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = safe_normalize(a_pert, pert_n)

                # Short Adam run
                a_new, score = run_adam_continuation(a_pert, pert_n, pert_S, perturb_time, rng)
                a_new = safe_normalize(a_new, pert_n)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < global_best:
                    global_best = c_new.max()
                    global_a = a_new.copy()
                    print("  Perturb %d (scale=%.4f): %.8f" % (p, noise_scale, global_best))

    # Verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\n=== FINAL: evaluate_sequence = %.6f ===" % ver)
        else:
            print("\n=== WARNING: evaluate_sequence rejected, using fallback ===")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal best=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
