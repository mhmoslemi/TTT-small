The "natural" approach is convex optimization, but the problem is non-convex due to the quadratic nature of the self-convolution.

However, there's a useful trick: **fix the "shape" and optimize the "scale"**. Or better: use the fact that the problem is homogeneous (scaling a doesn't change the objective).

Another idea: **Frank-Wolfe (conditional gradient) method**. The constraint set is the simplex {a >= 0, sum(a) = S}, which is a polytope. The Frank-Wolfe method works as:
1. Compute gradient g of the objective at current point a
2. Solve: min_v <g, v> subject to v in simplex → this is just v = S * e_{argmin(g)}
3. Update: a <- a + t*(v - a) for some step size t

The advantage is that each "LP" is trivial (just pick the minimum gradient coordinate). The disadvantage is that Frank-Wolfe can be slow to converge (1/k rate).

But the gradient of max(c) (or its soft-max approximation) is:
- If using top-k: grad[i] proportional to sum over top-k lags of (a[j] where j relates to i)
- This is a cross-correlation, computable via FFT

The Frank-Wolfe step would be: put all mass on the coordinate with the smallest gradient. This is very aggressive and might not work well in practice.

A modified Frank-Wolfe with "lazy" updates (only update a few coordinates per step) might work better.

4. Project back onto the simplex
5. Repeat

The projection onto {a: c_k(a) <= M} is: minimize ||a - a_current||^2 subject to c_k(a) <= M, a >= 0.

"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
Uses dumbbell-shaped initializations and FISTA with soft-max smoothing,
backtracking line search, and peak-balancing heuristics.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta_cands = (cumsum - S) / idx
    valid = theta_cands <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_conv(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2*n - 1
    fsz = 1 << (size - 1).bit_length()
    A = fft(a, fsz)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_grad_toppk(a, c, k):
    """Subgradient using top-k peaks."""
    n = len(a)
    top = np.argsort(c)[::-1][:k]
    w = np.zeros(2*n-1)
    for idx in top:
        w[idx] = 1.0
    / k  # normalize
    
    fsz = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsz)
    A = fft(a, fsz)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def fft_grad_softmax(a, c, beta):
    """Gradient of soft-max smoothing."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    w = exp_bc / total
    
    fsz = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsz)
    A = fft(a, fsz)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def obj_val(a):
    """Compute objective 2n*max(c)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_conv(a)
    return 2.0 * n * c.max() / (S * S)

def run_fista_backtrack(a0, n, S, max_time, rng, beta0=5.0, beta_max=500.0):
    """FISTA with soft-max and backtracking line search."""
    a = np.clip(a0, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    step = S / (4.0 * np.sqrt(n))
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta0 * (1.0 + 0.01 * it))
        
        c = fft_conv(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best:
            best = cur_obj
            best_a = y.copy()
        
        grad = fft_grad_softmax(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        x_try = project_simplex(y - step * grad / gnorm, S)
        # Check if objective improved
        if it % 5 == 0:
            c_try = fft_conv(x_try)
            obj_try = 2.0 * n * c_try.max() / (S * S)
            if obj_try > cur_obj * 1.05:
                step *= 0.5
                x_try = project_simplex(y - step * grad / gnorm, S)
            else:
                step = min(step * 1.1, S / (np.sqrt(n) * 10))
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_try + ((t - 1.0) / t_new) * (x_try - a)
        y = np.maximum(y, 0.0)
        a = x_try
        t = t_new
        
        # Periodic noise for exploration
        if it % 5000 == 0 and rng.random() < 0.1:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
    
    return best_a, best

    a = np.clip(a0, 0, 1000).astype(np.float64)

    if s < 1e-15:

        y = x_new + ((t - 1.0) / t_new) * (x_new - a)

        a = x_new
        t = t_new

For the coordinate descent: at each step, pick a coordinate i, and optimize a[i] while keeping others fixed. The objective as a function of a[i] is:

This is complex but can be optimized by golden section search or by finding the critical points.

Actually, for coordinate descent, the problem is that changing a[i] affects many c[k] values. Specifically, c[k] changes for all k where i or k-i is a valid index. That's O(n) values of k.

A simpler approach: **pairwise swap**. Pick two coordinates i and j, and transfer mass from one to the other: a[i] += delta, a[j] -= delta (keeping sum constant). Optimize delta.

For a small delta, the change in c[k] is:

plus higher order terms.

The peak c[k_max] changes by approximately:
dc[k_max] ≈ 2*delta*(a[k_max-i] - a[k_max-j])

If a[k_max-i] > a[k_max-j], increasing a[i] (and decreasing a[j]) will increase the peak. So we want to decrease a[i] and increase a[j].

This gives a natural direction: move mass from positions that contribute to the peak to positions that don't.

"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
Strategy: Dumbbell-shaped asymmetric initializations + FISTA with adaptive
soft-max smoothing + backtracking line search + multi-scale search + final

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} via bisection on threshold."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cands = (cumsum - S) / idx
    valid = theta_cands <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsz = 1 << (size - 1).bit_length()
    A = fft(a, fsz)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_softmax(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsz = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsz)
    A = fft(a, fsz)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_toppk(a, c, top_k):
    """Subgradient using top-k peak lags with linear weighting."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    # Use linear weights (larger peaks get more weight)
    max_v = top_vals[0]
    min_v = top_vals[-1]
    if max_v - min_v < 1e-15:
        weights = np.ones(top_k) / top_k
    else:
        weights = (top_vals - min_v) / (max_v - min_v)
        weights = weights + 0.1  # floor
        weights = weights / weights.sum()
    
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]
    
    fsz = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsz)
    A = fft(a, fsz)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_obj(a):
    """Compute 2n*max(c)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_conv(a)
    return 2.0 * n * c.max() / (S * S)


def run_fista(a_init, n, S, max_time, rng, beta0=10.0, beta_rate=0.02, beta_max=1000.0):
    """FISTA with adaptive soft-max and backtracking."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    step = S / (4.0 * np.sqrt(n))
    best_step = step
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta0 * (1.0 + beta_rate * it))
        
        c = fft_conv(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best:
            best = cur_obj
            best_a = y.copy()
        
        grad = fft_grad_softmax(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        x_new = project_simplex(y - step * grad / gnorm, S)
        
        # Backtracking every 10 iterations
        if it % 10 == 0:
            c_new = fft_conv(x_new)
            obj_new = 2.0 * n * c_new.max() / (S * S)
            if obj_new > cur_obj * 1.02:
                step *= 0.6
                x_new = project_simplex(y - step * grad / gnorm, S)
            elif obj_new < best:
                best_step = step
                step = min(step * 1.05, S / (2.0 * np.sqrt(n)))
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Noise injection
        if it % 8000 == 0 and rng.random() < 0.15:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
    
    return best_a, best


def run_toppk_fista(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=256):
    """FISTA with top-k subgradient (more aggressive peak targeting)."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)

    y = a.copy()
    t = 1.0

    best_a = a.copy()
    it = 0
    step = S / (3.0 * n)
    
    while time.monotonic() - start < max_time:
        it += 1
        k = min(top_k_max, top_k_init + it // 30)
        
        c = fft_conv(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best:
            best = cur_obj
            best_a = y.copy()
        
        grad = fft_grad_toppk(y, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        # Backtrack
        if it % 5 == 0:
            c_new = fft_conv(x_new)
            if c_new.max() > cur_max * 1.03:
                step *= 0.5
                x_new = 0.5 * y + 0.5 * project_simplex(y - step * grad / max(gnorm, 1e-15), S)
            else:
                step = min(step * 1.02, S / (n * 0.5))

def make_init_dumbbell(n, rng, width_frac=0.15, center_dip=0.3):
    """Dumbbell: high at both ends, dip in middle."""
    t = np.linspace(0, 1, n, endpoint=False)
    # Two rectangular-ish bumps at each end
    w = int(n * width_frac)
    a = np.full(n, center_dip)
    # Left bump with smooth falloff
    for i in range(w):
        frac = i / w
        a[i] = 1.0 * (1.0 - frac) + center_dip * frac
    # Right bump
    for i in range(w):
        frac = i / w
        a[n - 1 - i] = 1.0 * (1.0 - frac) + center_dip * frac
    a += rng.random(n) * 0.05
    return np.maximum(a, 0.0)


def make_init_ramp(n, rng, direction='left', steepness=3.0):
    """Asymmetric ramp."""
    t = np.linspace(0, 1, n, endpoint=False)
    if direction == 'left':
        a = np.exp(steepness * t)
    else:
        a = np.exp(steepness * (1.0 - t))
    a += rng.random(n) * 0.05
    return np.maximum(a, 0.0)


def make_init_power(n, rng, exponent=2.0, direction='left'):
    """Power-law ramp."""
    t = np.linspace(0, 1, n, endpoint=False)
    if direction == 'left':
        a = (t + 0.01) ** exponent
    else:
        a = (1.0 - t + 0.01) ** exponent
    a += rng.random(n) * 0.05
    return np.maximum(a, 0.0)


def make_init_two_pulse(n, rng, pos_frac=0.25, width=0.05):
    """Two Gaussian pulses at asymmetric positions."""
    t = np.linspace(0, 1, n, endpoint=False)
    sigma = width
    pos1 = pos_frac
    pos2 = 1.0 - pos_frac * 0.3  # asymmetric
    a = 0.1 + np.exp(-((t - pos1) / sigma) ** 2) + 0.8 * np.exp(-((t - pos2) / sigma) ** 2)
    a += rng.random(n) * 0.03
    return np.maximum(a, 0.0)


def make_init_uniform(n, rng):
    """Uniform with noise."""
    a = np.ones(n) * (1.0 + 0.1 * rng.standard_normal(n))
    return np.maximum(a, 0.1)
