"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
- L-BFGS with power smoothing continuation on softmax parameterization
- Nesterov-accelerated projected subgradient on the probability simplex
- Frank-Wolfe hybrid steps for peak reduction
- Coordinate descent / greedy peak shaving
- Multi-start with diverse initializations
- FFT-accelerated gradient computation throughout

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full linear self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_cross_corr(w, a):
    """
    Compute sum_k w[k] * a[k-j] for j=0..n-1 where w has length 2n-1 and a has length n.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    result = np.real(ifft(W * np.conj(A)))[:n]
    return result


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def power_obj_grad_on_simplex(p, power):
    """
    Compute f_p = (sum b_k^power)^(1/power) and its gradient w.r.t. p
    where b = p*p (linear self-convolution).
    p is a probability vector (sum=1, p>=0).
    Returns (f_p, grad) where grad has same shape as p.
    """
    n = len(p)
    b = fft_convolve_self(p)
    b_clipped = np.maximum(b, 0.0)  # ensure non-negative for power
    sum_bp = np.sum(b_clipped ** power)
    if sum_bp < 1e-300:
        return 0.0, np.zeros(n)
    f_p = sum_bp ** (1.0 / power)
    # gradient: df_p/dp_j = 2 * (sum b^p)^(1/p - 1) * sum_k b_k^(p-1) * p_{k-j}
    coeff = 2.0 * sum_bp ** (1.0 / power - 1.0)
    w = b_clipped ** (power - 1.0)
    grad = coeff * compute_cross_corr(w, p)
    return f_p, grad


def logsumexp_obj_grad(p, beta):
    """
    Compute smoothed max via log-sum-exp: (1/beta)*log(sum exp(beta*b_k))
    and its gradient w.r.t. p.
    """
    n = len(p)
    b = fft_convolve_self(p)
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    sum_exp = exp_b.sum()
    obj = (b_max + np.log(sum_exp)) / beta
    w = exp_b / sum_exp
    grad = 2.0 * w * 1.0 * compute_cross_corr(w, p)
    # Actually: d(obj)/dp_j = (1/beta) * sum_k (beta * exp(beta*b_k)/sum_exp) * db_k/dp_j
    # = sum_k w_k * 2 * p_{k-j}
    # = 2 * sum_k w_k * p_{k-j}
    grad = 2.0 * compute_cross_corr(w, p)
    return obj, grad


def softmax_params_to_a(z, S):
    """Convert softmax params z to sequence a with sum S."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = S * exp_z / exp_z.sum()
    return np.maximum(a, 0.0)


def a_to_softmax_params(a, S):
    """Convert sequence a (sum=S) to softmax params."""
    a = np.maximum(a, 1e-15)
    return np.log(a / S)


def run_lbfgs_power(a_init, n, S, max_time, rng,
                    power_schedule=None, maxiter_per_power=200):
    """
    L-BFGS with continuation on power (smooth-to-sharp) using softmax parameterization.
    """
    if power_schedule is None:
        power_schedule = [2, 3, 4, 6, 8, 12, 16, 24, 32, 48, 64, 96, 128, 192, 256, 384, 512]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = a_to_softmax_params(a, S)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for power in power_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        t0_beta = time.monotonic()
        n_attempts = 1
        if power >= 64 and max_time > 60:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 5:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.015, n)
                z = z - z.mean()

            def fun(z_val):
                a_cur = softmax_params_to_a(z_val, S)
                b = fft_convolve_self(a_cur)
                b_clipped = np.maximum(b, 0.0)
                sp = np.sum(b_clipped ** power)
                if sp < 1e-300:
                    return 0.0
                return sp ** (1.0 / power)

            def jac(z_val):
                a_cur = softmax_params_to_a(z_val, S)
                b = fft_convolve_self(a_cur)
                b_clipped = np.maximum(b, 0.0)
                sp = np.sum(b_clipped ** power)
                if sp < 1e-300:
                    return np.zeros(len(z_val))
                f_p = sp ** (1.0 / power)
                coeff = 2.0 * sp ** (1.0 / power - 1.0)
                w = b_clipped ** (power - 1.0)
                grad_a = coeff * compute_cross_corr(w, a_cur)
                dot_ga_a = np.dot(grad_a, a_cur)
                grad_z = a_cur * (grad_a - dot_ga_a / S)
                return grad_z

            try:
                result = scipy_minimize(
                    fun, z,
                    jac=jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_power,
                        'maxfun': maxiter_per_power * 2,
                        'ftol': 1e-15,
                        'gtol': 1e-10,
                        'maxls': 40,
                        'iprint': -1
                    }
                )
            except Exception as e:
                print("      L-BFGS error at power=%d: %s" % (power, str(e)))
                break

            z = result.x
            z = z - z.mean()
            a = softmax_params_to_a(z, S)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

        elapsed_beta = time.monotonic() - t0_beta
        if power % 4 == 0 or power <= 4:
            print("    power=%4d: max_b=%.8f (%d att, %.1fs, total=%.1fs)" %
                  (power, best_score, n_attempts, elapsed_beta, time.monotonic() - start))

    return best_a, best_score


def run_nesterov_toppk(p_init, n, max_time, rng, top_k_init=8, top_k_max=1024,
                       use_power=True, base_power=4, power_growth=1.3):
    """
    Nesterov-accelerated top-k subgradient on the probability simplex.
    Works with p (sum=1, p>=0) directly.
    Returns best p and score (max of p*p).
    """
    p = np.clip(p_init, 0.0, 1.0).copy().astype(np.float64)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    y = p.copy()
    t_param = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(p).max()
    best_local = p.copy()
    it = 0
    current_power = base_power

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 40)

        b = fft_convolve_self(y)
        cur = b.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Top-k subgradient
        top_indices = np.argsort(b)[::-1][:top_k]
        top_vals = b[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        grad = 2.0 * compute_cross_corr(w_full, y)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            # Try power smoothing gradient instead
            if use_power:
                current_power = min(current_power * power_growth, 256.0)
                _, grad = power_obj_grad_on_simplex(y, current_power)
                gnorm = np.linalg.norm(grad)
                if gnorm < 1e-15:
                    break

        step = 2.0 / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad, 1.0)

        if it % 5 == 0:
            b_x = fft_convolve_self(x_new)
            if b_x.max() < best_score:
                best_score = b_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        y = x_new + ((t_param - 1.0) / t_new) * (x_new - y)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = np.ones(n) / n
        else:
            y = y / sy
        p = x_new
        t_param = t_new

        # Occasional restart with perturbation
        if it % 5000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.001 / np.sqrt(n), n), 1.0)
            t_param = 1.0

    return best_local, best_score


def run_frank_wolve(p_init, n, max_time, rng, use_power=True, base_power=4):
    """
    Frank-Wolfe (conditional gradient) for peak reduction on the simplex.
    Alternates between FW steps and Nesterov-like accelerated steps.
    """
    p = np.clip(p_init, 0.0, 1.0).copy().astype(np.float64)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    y = p.copy()
    t_param = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(p).max()
    best_local = p.copy()
    it = 0
    current_power = base_power
    mw = 0.0  # memorized weight for line search

    while time.monotonic() - start < max_time:
        it += 1
        current_power = min(base_power * (1.2 ** (it // 100)), 512.0)

        b = fft_convolve_self(y)
        cur = b.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Compute gradient using power smoothing
        b_clipped = np.maximum(b, 0.0)
        sp = np.sum(b_clipped ** current_power)
        if sp < 1e-300:
            break
        f_p = sp ** (1.0 / current_power)
        coeff = 2.0 * sp ** (1.0 / current_power - 1.0)
        w = b_clipped ** (current_power - 1.0)
        grad = coeff * compute_cross_corr(w, y)

        # Frank-Wolfe: find min and max gradient positions
        g_min = grad.min()
        g_max = grad.max()
        j_min = int(np.argmin(grad))
        j_max = int(np.argmax(grad))

        # Move toward the best vertex (min gradient)
        e_min = np.zeros(n)
        e_min[j_min] = 1.0

        # Line search or fixed step
        gamma = 2.0 / (it + 2.0)  # standard FW step
        x_new = (1.0 - gamma) * y + gamma * e_min
        x_new = project_simplex(x_new, 1.0)

        # Also try moving mass from j_max to j_min (direct swap)
        gamma2 = min(1.0, 0.1 / (1.0 + 0.01 * it))
        x_swap = y.copy()
        move_amount = min(y[j_max], gamma2)
        x_swap[j_max] -= move_amount
        x_swap[j_min] += move_amount
        x_swap = project_simplex(x_swap, 1.0)

        # Choose the better of the two
        b_fw = fft_convolve_self(x_new)
        b_sw = fft_convolve_self(x_swap)
        if b_sw.max() < b_fw.max():
            x_new = x_swap

        if x_new.max() < best_score:
            pass  # checked below
        b_new = fft_convolve_self(x_new)
        if b_new.max() < best_score:
            best_score = b_new.max()
            best_local = x_new.copy()

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        y_prev = y.copy()
        y = x_new + ((t_param - 1.0) / t_new) * (x_new - y_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = np.ones(n) / n
        else:
            y = y / sy
        p = x_new
        t_param = t_new

    return best_local, best_score


def safe_normalize_to_prob(a):

    return a / s

                        0.2 * np.cos(2*np.pi*5*tt + 1.0) +
                        0.15 * np.sin(2*np.pi*9*tt + 2.0) +
                        0.1 * np.cos(2*np.pi*17*tt + 3.0) +
                        0.05 * np.sin(2*np.pi*31*tt)))
    other_inits.append(("bump", np.exp(-0.5*((tt-0.25)*8.0)**2) +
                        0.6*np.exp(-0.5*((tt-0.5)*6.0)**2) +
                        0.4*np.exp(-0.5*((tt-0.75)*8.0)**2) + 0.02))
    other_inits.append(("triangle_mod", np.abs(1.0 - 4.0*np.abs(tt - 0.25)) +
                        0.5*np.abs(1.0 - 4.0*np.abs(tt - 0.75)) + 0.01))

    # --- Phase 1: L-BFGS with power continuation (primary) ---
    phase1_time = min(1200.0, (deadline - time.monotonic()) * 0.55)
    print("\n=== Phase 1: L-BFGS Power Continuation (%.0f sec) ===" % phase1_time)

    all_inits = [("given", p_given)] + [(name, safe_normalize_to_prob(arr)) for name, arr in other_inits]

    per_init = phase1_time / len(all_inits)
    per_init = max(per_init, 15.0)

    phase1_results = []

    for name, p_init in all_inits:
        if time.monotonic() > deadline - 300:
            break

        t0 = time.monotonic()
        S = np.sqrt(2.0 * float(n))
        a_init = p_init * S

        power_schedule = [2, 3, 4, 6, 8, 12, 16, 24, 32, 48, 64, 96, 128, 192, 256, 384, 512, 768]

        a_new, score = run_lbfgs_power(
            a_init, n, S, per_init, rng,
            power_schedule=power_schedule,
            maxiter_per_power=250
        )
        p_new = safe_normalize_to_prob(a_new)
        obj_new = fft_convolve_self(p_new).max() * 2.0 * n

        phase1_results.append((name, p_new, obj_new))

        if obj_new < global_best:
            global_best = obj_new
            global_p = p_new.copy()
            print("  [NEW BEST] %s: %.8f (%.1fs)" % (name, global_best, time.monotonic() - t0))
        else:
            print("  %-12s: %.8f (best=%.8f, %.1fs)" % (name, obj_new, global_best, time.monotonic() - t0))

    print("\nAfter Phase 1: global_best=%.8f" % global_best)

    # --- Phase 2: Nesterov top-k on simplex (different local minima) ---
    remaining = deadline - time.monotonic()
    if remaining > 100:
        phase2_time = min(400.0, remaining * 0.3)
        print("\n=== Phase 2: Nesterov Top-K on Simplex (%.0f sec) ===" % phase2_time)

        # Run from the best of phase 1
        p_start = global_p.copy()
        p_new, score = run_nesterov_toppk(
            p_start, n, phase2_time * 0.6, rng,
            top_k_init=8, top_k_max=512,
            use_power=True, base_power=4, power_growth=1.4
        )
        obj_new = score * 2.0 * n
        if obj_new < global_best:
            global_best = obj_new
            global_p = p_new.copy()
            print("  Nesterov: %.8f [NEW BEST]" % global_best)
        else:
            print("  Nesterov: %.8f" % obj_new)

        # Also run from a perturbed version
        remaining2 = deadline - time.monotonic()
        if remaining2 > 60:
            p_pert = project_simplex(global_p + rng.normal(0, 0.01/np.sqrt(n), n), 1.0)
            p_new2, score2 = run_nesterov_toppk(
                p_pert, n, min(200.0, remaining2 * 0.5), rng,
                top_k_init=16, top_k_max=1024,
                use_power=True, base_power=8, power_growth=1.5
            )
            obj_new2 = score2 * 2.0 * n
            if obj_new2 < global_best:
                global_best = obj_new2
                global_p = p_new2.copy()
                print("  Nesterov perturbed: %.8f [NEW BEST]" % global_best)
            else:
                print("  Nesterov perturbed: %.8f" % obj_new2)

    print("\nAfter Phase 2: global_best=%.8f" % global_best)

    # --- Phase 3: Frank-Wolve hybrid ---
    remaining = deadline - time.monotonic()
    if remaining > 80:
        phase3_time = min(300.0, remaining * 0.25)
        print("\n=== Phase 3: Frank-Wolfe Hybrid (%.0f sec) ===" % phase3_time)

        p_new, score = run_frank_wolve(
            global_p, n, phase3_time, rng,
            use_power=True, base_power=4
        )
        obj_new = score * 2.0 * n
        if obj_new < global_best:
            global_best = obj_new
            global_p = p_new.copy()
            print("  FW: %.8f [NEW BEST]" % global_best)
        else:
            print("  FW: %.8f" % obj_new)

    print("\nAfter Phase 3: global_best=%.8f" % global_best)

    # --- Phase 4: Iterated perturbation + L-BFGS restart ---
    remaining = deadline - time.monotonic()
    if remaining > 60:
        phase4_time = min(400.0, remaining * 0.4)
        print("\n=== Phase 4: Perturbation + L-BFGS Restart (%.0f sec) ===" % phase4_time)

        n_perturb = max(4, int(phase4_time / 80))
        perturb_budget = phase4_time / n_perturb

        S = np.sqrt(2.0 * float(n))

        for p_idx in range(n_perturb):
            if time.monotonic() > deadline - 60:
                break

            # Varying noise scales
            noise_scales = [0.02, 0.05, 0.1, 0.03]
            noise_scale = noise_scales[p_idx % len(noise_scales)]

            a_pert = global_p * S + rng.normal(0, noise_scale * S / np.sqrt(n), n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert * S / a_pert.sum()

            # Also try structured perturbations
            if p_idx % 3 == 1:
                # Flip a random subset
                n_flip = n // 4
                flip_idx = rng.choice(n, n_flip, replace=False)
                a_struct = global_p * S
                a_struct[flip_idx] *= 0.3
                a_struct = a_struct * S / a_struct.sum()
                a_pert = a_struct
            elif p_idx % 3 == 2:
                # Frequency-domain perturbation
                fsize = 1 << (2 * n - 1).bit_length()
                A = fft(a_pert, fsize)
                # Add random phase to non-DC components
                phase_pert = rng.uniform(0, 2*np.pi, fsize)
                A[1:] *= np.exp(1j * 0.1 * phase_pert[1:])
                a_struct = np.real(ifft(A)[:n])
                a_struct = np.maximum(a_struct, 0.0)
                if a_struct.sum() > 1e-10:
                    a_pert = a_struct * S / a_struct.sum()

            a_new, score = run_lbfgs_power(
                a_pert, n, S, perturb_budget, rng,
                power_schedule=[8, 16, 32, 48, 64, 96, 128, 192, 256],
                maxiter_per_power=200
            )
            p_new = safe_normalize_to_prob(a_new)
            obj_new = fft_convolve_self(p_new).max() * 2.0 * n

            if obj_new < global_best:
                global_best = obj_new
                global_p = p_new.copy()
                print("  Pert %d (scale=%.3f): %.8f [NEW BEST]" % (p_idx, noise_scale, global_best))
            else:
                print("  Pert %d: %.8f" % (p_idx, obj_new))

    print("\nAfter Phase 4: global_best=%.8f" % global_best)

    # --- Phase 5: Final intensive polishing ---
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\n=== Phase 5: Final Polishing (%.0f sec) ===" % remaining)

        S = np.sqrt(2.0 * float(n))
        a_final = global_p * S

        # High-power L-BFGS
        seg1 = min(remaining * 0.5, remaining - 10)
        a_new, score = run_lbfgs_power(
            a_final, n, S, seg1, rng,
            power_schedule=[32, 48, 64, 96, 128, 192, 256, 384, 512, 768, 1024],
            maxiter_per_power=300
        )
        p_new = safe_normalize_to_prob(a_new)
        obj_new = fft_convolve_self(p_new).max() * 2.0 * n
        if obj_new < global_best:
            global_best = obj_new
            global_p = p_new.copy()
            print("  High-power L-BFGS: %.8f [NEW BEST]" % global_best)
        else:
            print("  High-power L-BFGS: %.8f" % obj_new)

        # Nesterov polish
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            p_new2, score2 = run_nesterov_toppk(
                global_p, n, remaining2, rng,
                top_k_init=32, top_k_max=2048,
                use_power=True, base_power=16, power_growth=1.5
            )
            obj_new2 = score2 * 2.0 * n
            if obj_new2 < global_best:
                global_best = obj_new2
                global_p = p_new2.copy()
                print("  Final Nesterov: %.8f [NEW BEST]" % global_best)
            else:
                print("  Final Nesterov: %.8f" % obj_new2)

    # --- Phase 6: Very final micro-optimization with coordinate descent ---
    remaining = deadline - time.monotonic()
    if remaining > 15 and global_p is not None:
        print("\n=== Phase 6: Coordinate Descent Polish (%.0f sec) ===" % remaining)
        p = global_p.copy()
        obj_cur = fft_convolve_self(p).max() * 2.0 * n
        cd_start = time.monotonic()

        while time.monotonic() - cd_start < remaining - 5:
            b = fft_convolve_self(p)
            peak_pos = int(np.argmax(b))
            # Find the element that most reduces the peak if decreased
            # and the element that is "safest" to increase
            # Contribution of p_j to b[peak_pos]: 2 * p[peak_pos - j] (if valid)
            contrib = np.zeros(n)
            valid_j = np.arange(max(0, peak_pos - n + 1), min(n, peak_pos + 1))
            for j in valid_j:
                if 0 <= peak_pos - j < n:
                    contrib[j] = 2.0 * p[peak_pos - j]
                else:
                    contrib[j] = 0.0

            # Find best j to decrease (high contribution) and j to increase (low contribution)
            j_dec = int(np.argmax(contrib * p))  # reduce where p is high AND contributes to peak
            j_inc = int(np.argmin(contrib))  # increase where contribution is minimal

            if j_dec == j_inc:
                break

            # Try several step sizes
            best_step = 0
            best_obj = obj_cur
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                move = p[j_dec] * frac
                if move < 1e-15:
                    break
                p_trial = p.copy()
                p_trial[j_dec] -= move
                p_trial[j_inc] += move
                b_trial = fft_convolve_self(p_trial)
                obj_trial = b_trial.max() * 2.0 * n
                if obj_trial < best_obj:
                    best_obj = obj_trial
                    best_step = frac

            if best_step > 0:
                move = p[j_dec] * best_step
                p[j_dec] -= move
                p[j_inc] += move
                obj_cur = best_obj
                if obj_cur < global_best:
                    global_best = obj_cur
                    global_p = p.copy()
            else:
                break

        print("  CD final: %.8f" % global_best)

    print("\n=== FINAL: global_best=%.8f ===" % global_best)

    # --- Convert to final answer ---
    if global_p is not None:
        n_final = len(global_p)
        # Scale to get a good absolute sequence (the objective is scale-invariant)
        # Use sum = sqrt(2n) for the standard normalization
        S = np.sqrt(2.0 * float(n_final))
        final_a = global_p * S
        final_a = np.clip(final_a, 0.0, 1000.0)
    else:
        final_a = np.ones(256, dtype=np.float64)

    # Verify
    final_list = [float(x) for x in final_a]
    ver = evaluate_sequence(final_list)
    print("\nVerification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        final_a = np.ones(256, dtype=np.float64)
        final_list = [1.0] * 256

    # Ensure valid
    if not final_a.any():
        final_a = np.ones(256, dtype=np.float64)
        final_list = [1.0] * 256

    print("Returning sequence of length %d, score=%.8f" % (len(final_list), ver if np.isfinite(ver) else float('inf')))
    elapsed = time.monotonic() - (deadline - total_budget)
    print("Total time: %.1fs" % elapsed)
    return final_list
