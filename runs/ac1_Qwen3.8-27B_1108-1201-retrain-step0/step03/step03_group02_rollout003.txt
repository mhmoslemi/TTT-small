"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 using adaptive projected subgradient

Key improvements over previous approach:

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_toppk(a, c, k):
    """
    Subgradient of max(c) using top-k active peaks.
    g[m] = (2/k) * sum_{j in top_k} a[k_j - m] for valid indices.
    """
    n = len(a)
    if k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / k
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return g


def subgradient_soft(a, c, beta):
    """Smoothed subgradient using log-sum-exp. grad[m] = (2/beta)*sum_k softmax_k * a[k-m]."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def normalize_seq(a, n):
    """Clip to [0, 1000] and normalize to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
