"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.
Strategy: Annealed smooth max (log-sum-exp) with FISTA acceleration,
followed by top-k focused refinement and coordinate-wise local search.
Multi-start with different initializations for robustness.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad(a, weights):
    """
    Compute gradient of sum_k weights[k] * (convolve(a,a))[k] w.r.t. a.
    Returns array of length n.
    """
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def soft_max_weights(c, T):
    """Compute softmax weights with temperature T."""
    c_max = c.max()
    exp_val = np.exp((c - c_max) / T)
    total = exp_val.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_val / total


def topk_weights(c, k):
    """Normalized top-k weights."""
    k = min(k, len(c))
    idx = np.argpartition(c, -k)[-k:]
    vals = c[idx]
    total = vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    w = np.zeros(len(c))
    w[idx] = vals / total
    return w


def annealed_fista(a_init, n, S, max_time, rng, T_init=None, T_min=0.01):
    """
    FISTA with annealed smooth max.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    
    if T_init is None:
        c = fft_convolve_self(a)
        T_init = max(c.max() - c.min(), 0.1)
    
    it = 0
    stall = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Anneal temperature
        progress = min(1.0, it / max(1, len(a)))
        T = T_init * (T_min / T_init) ** progress
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score - 1e-10:
            best_score = cur_max
            best_local = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute weights and gradient
        w = soft_max_weights(c, T)
        grad = compute_grad(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: use Lipschitz-like bound
        # The Lipschitz constant of the gradient is bounded by 2 * max(w) * 2n (roughly)
        # Use a conservative step
        step = S / (n * 4.0 * max(w.max() * 2 * n, 1.0))
        
        x_new = project_simplex(y - step * grad, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Reset momentum if stalled too long
        if stall > 2000:
            y = a.copy()
            t = 1.0
            stall = 0
            # Small perturbation
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
    
    return best_local, best_score


def topk_fista(a_init, n, S, max_time, rng, k_init=1, k_max=64, k_rate=100):

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_local = a.copy()

def coordinate_local_search(a_init, n, S, max_time, rng, n_pairs_per_iter=100):
    """
    Coordinate-wise local search: adjust pairs of coordinates to reduce peak.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    
    # Precompute the "influence" of each coordinate on the peak
    # For pair (i, j), moving delta from i to j changes the convolution by:
    # delta * (a_shifted_by_i - a_shifted_by_j) at each lag
    
    while time.monotonic() - start < max_time:
        # Get current peak location
        peak_idx = c.argmax()
        
        # Try random pairs
        for _ in range(n_pairs_per_iter):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Try moving a small amount from the larger to the smaller
            if a[i] < a[j]:
                i, j = j, i
            
            # Try different fractions
            for frac in [0.01, 0.05, 0.1]:
                delta = frac * a[i]
                if delta > a[i]:
                    continue
                
                a[i] -= delta
                a[j] += delta
                
                # Quick check: compute new peak (approximate)
                # Full recomputation is expensive, so we do it periodically
                c_new = fft_convolve_self(a)
                new_max = c_new.max()
                
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_local = a.copy()
                    c = c_new
                else:
                    # Revert
                    a[i] += delta
                    a[j] -= delta
                    break
        
        # Full recomputation every few iterations
        c = fft_convolve_self(a)
        if c.max() < best_score:
            best_score = c.max()
            best_local = a.copy()
    
    return best_local, best_score

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad(a, weights):
    """
    Gradient of sum_k weights[k] * (convolve(a,a))[k] w.r.t. a.
    Uses the identity: grad_j = 2 * sum_k w_k * a[k-j] (valid k).
    Computed via FFT cross-correlation.
    """
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def soft_max_weights(c, T):
    """Softmax weights with temperature T. As T->0, concentrates on max."""
    c_max = c.max()
    exp_val = np.exp((c - c_max) / T)
    total = exp_val.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_val / total


def topk_weights(c, k):
    """Normalized top-k weights."""
    k = min(k, len(c))
    idx = np.argpartition(c, -k)[-k:]
    vals = c[idx]
    total = vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    w = np.zeros(len(c))
    w[idx] = vals / total
    return w


def smooth_max_value(c, T):
    """Compute the smooth max value: T * log(sum(exp(c/T)))."""
    c_max = c.max()
    return c_max + T * np.log(np.sum(np.exp((c - c_max) / T)))


def annealed_fista_bt(a_init, n, S, max_time, rng, T_init=None, T_min=0.005):
    """
    FISTA with annealed smooth max and Armijo backtracking line search.
    More robust than fixed-step approaches.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()

    if T_init is None:
        spread = c.max() - np.percentile(c, 90)
        T_init = max(spread * 0.5, c.max() * 0.05, 0.01)

    it = 0
    last_best_it = 0
    step = S / (4.0 * n)  # Conservative initial step
    armijo_c = 0.01  # Armijo parameter
    decay_rate = 0.9997  # Temperature decay per iteration

    while time.monotonic() - start < max_time:
        it += 1
        T = max(T_min, T_init * (decay_rate ** it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score - 1e-10:
            best_score = cur_max
            best_local = y.copy()
            last_best_it = it

        w = soft_max_weights(c, T)
        grad = compute_grad(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Armijo backtracking
        alpha = step
        f_cur = smooth_max_value(c, T)
        accepted = False
        for _ in range(8):
            x_trial = project_simplex(y - alpha * grad, S)
            c_trial = fft_convolve_self(x_trial)
            f_trial = smooth_max_value(c_trial, T)
            if f_trial < f_cur - armijo_c * alpha * gnorm * np.linalg.norm(grad):
                accepted = True
                break
            alpha *= 0.5

        if not accepted:
            # Just take a small step
            x_trial = project_simplex(y - alpha * grad, S)
            accepted = True

        x_new = x_trial
        step = alpha  # Update step size from successful backtracking

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Stall detection with perturbed restart
        if it - last_best_it > 5000:
            y = best_local.copy()
            a = best_local.copy()
            t = 1.0
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            step = S / (4.0 * n)
            last_best_it = it

    return best_local, best_score


def topk_fista_adaptive(a_init, n, S, max_time, rng, k_init=1, k_max=32, k_rate=150):
    """
    Top-k FISTA with adaptive step size based on improvement rate.
    Gradually increases k to shift from peak-focused to global optimization.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    it = 0
    step = S / (2.0 * n)
    best_step = step
    plateaus = 0
    last_best_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        k = min(k_max, k_init + it // k_rate)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score - 1e-11:
            best_score = cur_max
            best_local = y.copy()
            last_best_it = it
            best_step = step
            step = min(step * 1.01, S / n)
            plateaus = 0
        else:
            plateaus += 1
            if plateaus > 200:
                step *= 0.95
                plateaus = 0

        w = topk_weights(c, k)
        grad = compute_grad(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Normalized gradient step
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        # Check if actually improved (periodic full check)
        if it % 10 == 0:
            c_new = fft_convolve_self(x_new)
            if c_new.max() < best_score:
                best_score = c_new.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Restart if stuck
        if it - last_best_it > 8000:
            y = best_local.copy()
            a = best_local.copy()
            t = 1.0
            step = best_step * 0.5
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            last_best_it = it

    return best_local, best_score


def focused_peak_descent(a_init, n, S, max_time, rng):
    """
    Very aggressive peak-focused descent: only optimizes the single worst lag.
    Uses small steps and frequent convergence checks.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    it = 0
    step = S / (n * 4.0)

    while time.monotonic() - start < max_time:
        it += 1

        # Focus only on the top 1-3 lags
        k = min(3, 1 + it // 2000)
        w = topk_weights(c, k)
        grad = compute_grad(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Try the step
        x_new = project_simplex(a - step * grad / max(gnorm, 1e-15), S)
        c_new = fft_convolve_self(x_new)

        if c_new.max() < best_score - 1e-12:
            best_score = c_new.max()
            best_local = x_new.copy()
            a = x_new
            c = c_new
            step = min(step * 1.002, S / (2.0 * n))
        else:
            # Try half step
            x_half = project_simplex(a - 0.5 * step * grad / max(gnorm, 1e-15), S)
            c_half = fft_convolve_self(x_half)
            if c_half.max() < best_score - 1e-12:
                best_score = c_half.max()
                best_local = x_half.copy()
                a = x_half
                c = c_half
                step *= 0.9
            else:
                step *= 0.85
                # Occasional perturbation to escape
                if it % 1000 == 0:
                    noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                    a = project_simplex(a + noise, S)
                    c = fft_convolve_self(a)

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def try_binary_structure(n, S, rng):
    """Generate a structured binary-like sequence as alternative initialization."""
    # Try a sequence with values in {0, S/n * ratio} for various ratios
    # Using a "thickened" random pattern
    base = (rng.random(n) > 0.5).astype(np.float64)
    # Smooth it slightly
    kernel = np.ones(7) / 7.0
    base = np.convolve(base, kernel, mode='same')
    base = safe_normalize(base, n)
    return base
