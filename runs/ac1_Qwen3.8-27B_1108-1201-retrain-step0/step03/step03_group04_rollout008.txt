def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_autocorr(a):

    n = len(a)
    fsize = 1 << (2 * n - 1 - 1).bit_length()

    c = np.real(ifft(A * np.conj(A)))
    # Linear autocorrelation
    ac = np.zeros(2 * n - 1)
    ac[:n] = c[-(n-1):]  # negative lags (reversed)
    # Wait, let me be more careful
    # np.convolve(a, a, 'full') gives the linear convolution
    # c[k] = sum_i a[i] * a[i+k-(n-1)] for k=0..2n-2
    # Actually, fft_convolve_self already does this correctly
    return c[:2*n-1]

def compute_gradient_fft(a, weights):
    """Compute gradient given weights on convolution lags using FFT."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total

def toppk_weights(c, top_k):
    """Compute normalized top-k weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights

def spectral_flatten(a, n_iter=100, alpha=0.5, floor=0.01):
    """
    Gerchberg-Saxton style spectral flattening.
    Iteratively: FFT -> flatten power spectrum -> IFFT -> project to non-negative -> rescale
    """
    a = a.copy()
    S = a.sum()
    n = len(a)
    fsize = 1 << (2 * n - 1 - 1).bit_length()
    
    for _ in range(n_iter):
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        # Flatten: target uniform power spectrum
        # Apply inverse of power spectrum to the signal
        # A_new = A * (1/sqrt(power))^alpha = A * |A|^(-alpha)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        new_mag = np.maximum(mag, floor) ** (-alpha)
        A_new = phase * new_mag
        a_new = np.real(ifft(A_new, n))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new * (S / s)
    
    return a

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Spectral flattening (GS-style) initialization, FISTA with backtracking
line search and very high beta, exact top-1 peak gradient polishing,
coordinate balancing at peak lags, multi-n exploration, and diverse restarts.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient given weights on convolution lags using FFT."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Compute normalized top-k weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights


def spectral_flatten(a, n_iter=80, alpha=0.5, floor=0.01):
    """
    Gerchberg-Saxton style spectral flattening.
    Iteratively: FFT -> compress power spectrum -> IFFT -> project non-negative -> rescale.
    """
    a = a.copy()
    S = a.sum()
    n = len(a)
    fsize = 1 << (2 * n).bit_length()

    for it in range(n_iter):
        A = fft(a, fsize)
        power = np.abs(A) ** 2
        # Target: uniform power spectrum. Apply |A|^(-alpha) to compress dynamic range.
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-30)
        new_mag = np.maximum(mag, floor) ** (-alpha)
        A_new = phase * new_mag
        a_new = np.real(ifft(A_new, n))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new * (S / s)

    return a

                    transfer = 0.05 * (a[j] - a[i])
                    a[j] -= transfer
                    a[i] += transfer

            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a * (S / s)

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def evaluate_fast(a):
    """Quick evaluation: returns 2*n*max_conv/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def propose_candidate(seed=42, budget_s=2390, **kwargs):
    """
    Minimize 2*n*max_conv/sum^2 over non-negative sequences.
    Multi-phase: spectral flatten init, FISTA backtrack, top-1 exact,
    peak balancing, multi-n, diverse restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s - 25.0)

    global_best_obj = np.inf
    global_a = None
    global_n = 0

    # Get parent sequence
    parent_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            parent_a = np.clip(given, 0, 1000)
    except Exception:
        pass

    def try_update(a, label):
        nonlocal global_best_obj, global_a, global_n
        a = safe_normalize(a, len(a))
        obj = evaluate_fast(a)
        if obj < global_best_obj:
            global_best_obj = obj
            global_a = a.copy()
            global_n = len(a)
            print("  [%.6f] %s (n=%d)" % (obj, label, len(a)))
            return True
        return False

    print("=== Starting optimization, budget=%ds ===" % budget_s)
    t_global_start = time.monotonic()

    # Determine n values to try
    n_candidates = []
    if parent_a is not None:
        n_candidates.append(len(parent_a))
    for n_try in [4096, 8192, 16384]:
        if n_try not in n_candidates:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n for n in n_candidates if 1024 <= n <= 32768))
    print("n candidates: %s" % n_candidates)

    # === Phase 1: Initialization (10% of time) ===
    init_budget = 0.10 * (deadline - time.monotonic())
    print("Phase 1: Initialization (%.0f sec)" % init_budget)

    best_init_a = None
    best_init_obj = np.inf

    for n in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        S = np.sqrt(2.0 * float(n))
        inits = []

        # From parent
        if parent_a is not None:
            if len(parent_a) == n:
                inits.append(('parent', safe_normalize(parent_a, n)))
            else:
                parent_resized = np.interp(
                    np.linspace(0, 1, n, endpoint=False),
                    np.linspace(0, 1, len(parent_a), endpoint=False),
                    parent_a)
                inits.append(('parent_rs', safe_normalize(parent_resized, n)))

        # Spectral flattening from random
        a_rand = rng.random(n) * 0.5 + 0.1
        a_sf = spectral_flatten(safe_normalize(a_rand, n), n_iter=60, alpha=0.5)
        inits.append(('spec_rand', safe_normalize(a_sf, n)))

        # Spectral flattening from chirp-like
        tt = np.linspace(0, 1, n, endpoint=False)
        a_chirp = 0.5 + 0.5 * np.cos(2 * np.pi * (2.0 * tt + 1.5 * tt * tt))
        a_sf2 = spectral_flatten(safe_normalize(a_chirp, n), n_iter=60, alpha=0.5)
        inits.append(('spec_chirp', safe_normalize(a_sf2, n)))

        # Spectral flattening from windowed noise
        a_noise = rng.normal(0, 1, n)
        window = 0.5 * (1 + np.cos(2 * np.pi * tt))
        a_win_noise = np.abs(a_noise) * window + 0.05
        a_sf3 = spectral_flatten(safe_normalize(a_win_noise, n), n_iter=60, alpha=0.4)
        inits.append(('spec_win', safe_normalize(a_sf3, n)))

        for name, a_init in inits:
            obj = evaluate_fast(a_init)
            if obj < best_init_obj:
                best_init_obj = obj
                best_init_a = a_init.copy()
                best_init_n = n
            print("  init n=%d %s: obj=%.6f" % (n, name, obj))

    if best_init_a is not None:
        try_update(best_init_a, "best_init")
    else:
        # Fallback
        n = 8192
        global_a = np.ones(n) * 0.5
        global_a = safe_normalize(global_a, n)
        global_n = n
        global_best_obj = evaluate_fast(global_a)
        try_update(global_a, "fallback_uniform")

    # === Phase 2: FISTA with backtracking (30% of time) ===
    remaining = deadline - time.monotonic()
    phase2_time = 0.30 * remaining
    print("Phase 2: FISTA backtrack (%.0f sec)" % phase2_time)
    n = global_n
    S = np.sqrt(2.0 * float(n))

    a2, _ = run_fista_backtrack(global_a, n, S, phase2_time, rng,
                                 beta_init=20.0, beta_max=20000.0, use_topk=False)
    try_update(a2, "fista_softmax")

    # === Phase 3: Top-k FISTA (15% of time) ===
    remaining = deadline - time.monotonic()
    phase3_time = 0.15 * remaining
    print("Phase 3: Top-k FISTA (%.0f sec)" % phase3_time)

    a3, _ = run_fista_backtrack(global_a, n, S, phase3_time, rng,
                                 beta_init=10.0, beta_max=10000.0,
                                 use_topk=True, top_k_init=1, top_k_max=100)
    try_update(a3, "fista_topk")

    # === Phase 4: Top-1 exact peak (20% of time) ===
    remaining = deadline - time.monotonic()
    phase4_time = 0.20 * remaining
    print("Phase 4: Top-1 exact (%.0f sec)" % phase4_time)

    a4, _ = run_top1_exact(global_a, n, S, phase4_time, rng)
    try_update(a4, "top1_exact")

    # === Phase 5: Peak balancing (10% of time) ===
    remaining = deadline - time.monotonic()
    phase5_time = 0.10 * remaining
    if phase5_time > 5:
        print("Phase 5: Peak balancing (%.0f sec)" % phase5_time)
        a5, _ = run_balance_peaks(global_a, n, S, phase5_time, rng)
        try_update(a5, "balance")
        # Follow with quick top-1
        if time.monotonic() < deadline - 10:
            t_bal = deadline - time.monotonic() - 10
            a5b, _ = run_top1_exact(global_a, n, S, min(t_bal, 60), rng)
            try_update(a5b, "balance+top1")

    # === Phase 6: Diverse restarts (15% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("Phase 6: Restart (%.0f sec)" % remaining)
        n_restarts = max(3, int(remaining / 80))
        per_restart = (remaining - 20) / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break

            if r % 5 == 0:
                perturbed = global_a + rng.normal(0, 0.004 * S / np.sqrt(n), n)
            elif r % 5 == 1:
                smooth = np.convolve(rng.normal(0, 1, n), np.ones(64) / 64, mode='same')
                perturbed = global_a + 0.003 * S / np.sqrt(n) * smooth
            elif r % 5 == 2:
                tt = np.linspace(0, 1, n, endpoint=False)
                freq = 2.0 + 1.3 * r
                perturbed = global_a + 0.006 * S / np.sqrt(n) * np.sin(2 * np.pi * freq * tt + rng.random() * 6.28)
            elif r % 5 == 3:
                perturbed = global_a * (1.0 + rng.normal(0, 0.03, n))
            else:
                # Spectral perturbation
                A = fft(global_a, 1 << (2 * n).bit_length())
                A_flat = A * (1.0 + rng.normal(0, 0.1, len(A)))
                perturbed = np.real(ifft(A_flat, n))
                perturbed = np.maximum(perturbed, 0)

            perturbed = safe_normalize(perturbed, n)

            # Quick optimization: FISTA then top-1
            t_f = min(per_restart * 0.6, 100)
            a_r, _ = run_fista_backtrack(perturbed, n, S, t_f, rng,
                                          beta_init=50.0, beta_max=15000.0)
            try_update(a_r, "restart_%d_fista" % r)

            if time.monotonic() < deadline - 15:
                t_t = min(per_restart * 0.4, 60)
                a_r2, _ = run_top1_exact(global_a, n, S, t_t, rng)
                try_update(a_r2, "restart_%d_top1" % r)

    # === Phase 7: Final ultra-fine polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("Phase 7: Final polish (%.0f sec)" % remaining)
        # Sweep: try different step multipliers
        for mult in [0.5, 0.7, 1.0, 1.5, 2.0]:
            if time.monotonic() > deadline - 5:
                break
            t_sub = (deadline - time.monotonic() - 5) / (5 - (int(mult * 10) % 5))
            a_sub, _ = run_top1_exact(global_a, n, S, min(t_sub, 30), rng)
            try_update(a_sub, "polish_mult=%.1f" % mult)

    # === Final ===
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    final_obj = evaluate_fast(global_a)
    print("=== FINAL: n=%d, objective=%.6f, time=%.0fs ===" % (
        global_n, final_obj, time.monotonic() - t_global_start))

    # Verify with the actual evaluator
    try:
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified: %.6f" % ver)
        else:
            print("Verification failed, using fallback")
            final_list = [0.5] * 256
            print("Fallback: %.6f" % evaluate_sequence(final_list))
    except Exception as e:
        print("Verification error: %s" % e)

    return final_list
