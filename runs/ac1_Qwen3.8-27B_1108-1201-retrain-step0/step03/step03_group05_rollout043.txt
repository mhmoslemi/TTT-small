"""
Minimize 2*n*max_conv / sum^2 for non-negative sequences.

Approach:
- p-norm smoothed objective with L-BFGS-B (continuation on p, adaptive schedule)
- Multiple diverse initializations (frequency-based, structured, given sequence)
- Nesterov-accelerated top-k subgradient polishing
- Peak-shaving via targeted coordinate updates
- FFT-accelerated gradient computation
- Innovative "peak equalization" initialization strategy

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_cross_corr(w, a):
    """Compute cross-correlation (w * a)[j] = sum_k w[k] * a[k-j] via FFT."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return np.real(ifft(np.conj(W) * A))[:n]


def pnorm_obj_grad_z(z, S, p):
    """
    Compute p-norm of self-convolution and gradient w.r.t. softmax params z.
    a = S * softmax(z), obj = (sum c^p)^(1/p)
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c = np.maximum(c, 0.0)

    if p == np.inf:
        max_idx = np.argmax(c)
        w = np.zeros(2 * n - 1)
        w[max_idx] = 1.0
        obj = c[max_idx]
    else:
        c_clip = np.maximum(c, 1e-30)
        c_pow = c_clip ** (p - 1)
        sum_cpow = c_pow.sum()
        if sum_cpow < 1e-30:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w = c_pow / sum_cpow
        obj = c_clip ** p
        obj = obj.sum() ** (1.0 / p)

    # Gradient via cross-correlation
    cross = compute_cross_corr(w, a)
    # For p != inf: obj = (sum c^p)^(1/p), d obj/d a = (sum c^p)^(1/p-1) * sum_k c[k]^(p-1) * dc[k]/da
    # dc[k]/da vector = 2 * cross (as derived)
    # So grad_a = (sum c^p)^(1/p - 1) * 2 * (sum_k w[k]*c[k]^(p-1) ... )
    # Actually: grad_a = (sum c^p)^(1/p-1) * sum_k c[k]^(p-1) * 2*cross[k]... no
    # Let me re-derive: obj = (sum_k c_k^p)^(1/p)
    # d obj/d a[j] = (1/p)(sum c^p)^(1/p-1) * sum_k p*c_k^(p-1)*dc_k/da[j]
    # = (sum c^p)^(1/p-1) * sum_k c_k^(p-1) * dc_k/da[j]
    # = (sum c^p)^(1/p-1) * 2 * sum_k c_k^(p-1) * cross_k[j]  ... no
    # dc_k/da[j] gives a vector, and sum_k c_k^(p-1) * (dc_k/da) is a vector:
    # [sum_k c_k^(p-1) * dc_k/da[j] for each j]
    # = 2 * sum_k c_k^(p-1) * a[k-j] (by our derivation)
    # = 2 * (w_unnorm * a) cross-correlation where w_unnorm[k] = c[k]^(p-1)
    # = 2 * sum_cpow * (w * a)[j] where w[k] = c[k]^(p-1)/sum_cpow
    # = 2 * sum_cpow * cross[j]
    # So grad_a = (sum c^p)^(1/p-1) * 2 * sum_cpow * cross
    # And sum_cpow = sum c[k]^(p-1) = c_pow.sum()

    if p != np.inf:
        sum_cp = (c_clip ** p).sum()
        # (sum c^p)^(1/p-1) = sum_cp^(1/p-1)
        prefactor = sum_cp ** (1.0 / p - 1.0)
        grad_a = 2.0 * prefactor * sum_cpow * cross
    else:
        grad_a = 2.0 * cross

    # Project to z-space: grad_z = a * (grad_a - dot(grad_a, a)/S)
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs_pnorm(a_init, n, S, max_time, rng, p_schedule=None, maxiter_per_p=200):
    """
    Run L-BFGS-B with p-norm continuation.
    Returns best (a, max_conv) found.
    """
    if p_schedule is None:
        p_schedule = [4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for pi, p in enumerate(p_schedule):
        if time.monotonic() - start > max_time - 3:
            break

        # At high p, try multiple attempts with perturbation
        n_attempts = 1
        if p >= 512 and max_time > 120:
            n_attempts = 2
        elif p >= 128 and max_time > 60:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 3:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.01, n)
                z = z - z.mean()

            def fun(z_val):
                return pnorm_obj_grad_z(z_val, S, p)[0]

            def jac(z_val):
                return pnorm_obj_grad_z(z_val, S, p)[1]

            t0 = time.monotonic()
            try:
                result = scipy_minimize(
                    fun, z,
                    jac=jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_p,
                        'maxfun': maxiter_per_p * 3,
                        'ftol': 1e-14,
                        'gtol': 1e-8,
                        'maxls': 50,
                        'iprint': -1
                    }
                )
            except Exception:
                break

            z = result.x
            z = z - z.mean()

            # Convert back to a
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

        if pi % 2 == 0:
            elapsed = time.monotonic() - start
            print("    p=%4d: max_conv=%.8f (total %.1fs)" % (p, best_score, elapsed))

    return best_a, best_score


def run_topk_nesterov(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=256):
    """Nesterov-accelerated top-k subgradient with adaptive step."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    prev_score = np.inf
    stall_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Stall detection
        if cur > prev_score * 0.999:
            stall_count += 1
        else:
            stall_count = 0
        prev_score = cur

        if stall_count > 200:
            # Perturb and reset momentum
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t_mom = 1.0
            stall_count = 0
            continue

        # Top-k weights
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        cross = compute_cross_corr(w_full, a)
        grad = 2.0 * cross
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size with backtracking
        step_base = 2.0 * S / (n * (1.0 + 0.001 * it))
        step = step_base

        # Backtracking line search (simple: try a few step sizes)
        best_x = None
        best_val = np.inf
        for mult in [1.0, 0.5, 0.25, 0.1]:
            x_try = project_simplex(y - step * mult * grad / max(gnorm, 1e-15), S)
            c_try = fft_convolve_self(x_try)
            val_try = c_try.max()
            if val_try < best_val:
                best_val = val_try
                best_x = x_try
            if val_try < cur * 0.999:
                break

        if best_x is None:
            best_x = a
        x_new = best_x

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = np.ones(n) * S / n
        else:
            y = y * S / s_y
        a = x_new
        t_mom = t_new

        # Occasional random restart
        if it % 3000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t_mom = 1.0

    return best_local, best_score


def run_peak_shaving(a_init, n, S, max_time, rng):
    """
    Peak shaving: iteratively identify top peaks and take steps to reduce them.
    Uses exact subgradient of the max function.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()

        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # Get top peaks
        n_peaks = min(8, 2 * n - 1)
        top_peaks = np.argsort(c)[::-1][:n_peaks]
        top_vals = c[top_peaks]

        # Weight by peak value
        w = top_vals / top_vals.sum()

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_peaks):
            w_full[int(k)] = w[i]

        cross = compute_cross_corr(w_full, a)
        grad = 2.0 * cross
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step
        step = S / (np.sqrt(n) * (1.0 + 0.01 * it))
        a_new = project_simplex(a - step * grad / max(gnorm, 1e-15), S)

        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur:
            a = a_new
        else:
            # Try smaller step
            a_new2 = project_simplex(a - 0.1 * step * grad / max(gnorm, 1e-15), S)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < cur:
                a = a_new2
            elif it % 100 == 0:
                # Add small perturbation to escape
                a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def generate_freq_init(n, n_freq, amplitudes, freqs, phases, rng):
    """Generate initialization from frequency components."""
    j = np.arange(n)
    a = np.ones(n)
    for i in range(n_freq):
        a += amplitudes[i] * np.cos(2 * np.pi * freqs[i] * j / n + phases[i])
    return np.maximum(a, 0.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    # Determine sequence length
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            n = len(given)
        else:
            n = 8192
    except Exception:
        n = 8192

    # Try different lengths: use the given length primarily
    S = np.sqrt(2.0 * float(n))
    print("=== n=%d, S=%.4f, budget=%.0fs ===" % (n, S, budget_s))
    print("Target: 1.5027")

    global_best = np.inf
    global_a = None

    def update_global(a, score):
        nonlocal global_best, global_a
        if score < global_best:
            global_best = score
            global_a = a.copy()
            return True
        return False

    # --- Generate initializations ---
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    j_idx = np.arange(n)

    # Given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0.0, 1000.0)))
    except Exception:
        pass

    # Frequency-based initializations (multiple diverse sets)
    # Strategy: use frequencies that are "incommensurate" (ratio of small primes)
    prime_freqs = [3, 7, 11, 13, 17, 19, 23, 29, 31, 37]

    for trial in range(4):
        n_freq = 5 + trial
        selected = prime_freqs[:min(n_freq, len(prime_freqs))]
        if trial >= 2:
            # Shuffle for diversity
            selected = list(prime_freqs)
            rng.shuffle(selected)
            selected = selected[:n_freq]
        amps = 0.3 / np.sqrt(n_freq) * rng.uniform(0.5, 1.0, n_freq)
        freqs = np.array(selected, dtype=np.float64)
        phas = rng.uniform(0, 2 * np.pi, n_freq)
        a_init = generate_freq_init(n, n_freq, amps, freqs, phas, rng)
        inits.append(("freq%d" % trial, a_init))

    # Sparse pattern initializations
    for sparsity in [0.3, 0.5, 0.7]:
        mask = rng.random(n) < (1 - sparsity)
        a_sparse = mask.astype(np.float64) * rng.uniform(0.5, 1.5, n)
        inits.append(("sparse%.1f" % sparsity, a_sparse))

    # Window function initializations
    inits.append(("hann", 0.5 + 0.5 * np.cos(2 * np.pi * tt)))
    inits.append(("tri", 1.0 - 2.0 * np.abs(tt - 0.5)))
    inits.append(("bump", np.exp(-2.0 * (tt - 0.5)**2) + 0.1))

    # Random
    inits.append(("rand", rng.random(n) + 0.1))

    # "Anti-peak" initialization: try to equalize convolution peaks
    # Start uniform, then push down the elements most responsible for the peak
    a_uniform = np.ones(n)
    c_uniform = fft_convolve_self(a_uniform)
    peak_idx = c_uniform.argmax()
    # Elements most contributing to peak are a[peak_idx-j] for j
    contrib = np.zeros(n)
    lo = max(0, peak_idx - n + 1)
    hi = min(n - 1, peak_idx)
    for jj in range(lo, hi + 1):
        contrib[jj] += 1.0
    a_antipeak = 1.0 - 0.3 * contrib / contrib.max()
    inits.append(("antipeak", np.maximum(a_antipeak, 0.01)))

    n_inits = len(inits)
    print("Generated %d initializations" % n_inits)

    # --- Phase 1: L-BFGS with p-norm on diverse inits ---
    remaining = deadline - time.monotonic()
    # Allocate time: 60% to phase 1, 20% to phase 2, 20% to phase 3
    phase1_time = remaining * 0.55
    per_init = phase1_time / n_inits
    per_init = max(per_init, 15.0)

    print("\n=== Phase 1: L-BFGS p-norm (%.0f sec each, %d inits) ===" % (per_init, n_inits))
    phase1_results = {}

    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break

        t0 = time.monotonic()
        a_norm = safe_normalize(arr, n)
        c = fft_convolve_self(a_norm)
        init_score = c.max()

        if init_score < global_best:
            global_best = init_score
            global_a = a_norm.copy()

        # L-BFGS continuation
        p_sched = [4, 8, 16, 32, 64, 128, 256, 512]
        if per_init > 60:
            p_sched = [4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048]

        a_new, score = run_lbfgs_pnorm(
            a_norm, n, S, per_init, rng,
            p_schedule=p_sched, maxiter_per_p=150
        )
        a_new = safe_normalize(a_new, n)
        c_new = fft_convolve_self(a_new)
        final_score = c_new.max()

        phase1_results[name] = (a_new, final_score)

        improved = update_global(a_new, final_score)
        tag = "OK" if improved else "  "
        print("  [%s] %s: %.8f (init=%.4f, %.1fs)" % (tag, name, final_score, init_score, time.monotonic() - t0))

    print("\n=== After Phase 1: best=%.8f ===" % global_best)

    # --- Phase 2: Cross-pollination + targeted L-BFGS on best candidates ---
    if global_a is not None:
        remaining2 = deadline - time.monotonic()
        if remaining2 > 60:
            print("\n=== Phase 2: Cross-pollination + deep optimization (%.0f sec) ===" % remaining2)

            sorted_results = sorted(phase1_results.items(), key=lambda x: x[1][1])
            n_deep = max(2, int(remaining2 / 80))
            deep_budget = remaining2 * 0.5 / n_deep

            for di in range(n_deep):
                if time.monotonic() > deadline - 120:
                    break

                idx = di % len(sorted_results)
                a_base = sorted_results[idx][1][0]

                # Try blending with the global best
                if di > 0 and global_a is not None:
                    for blend in [0.3, 0.5, 0.7]:
                        if time.monotonic() > deadline - 60:
                            break
                        a_blend = blend * global_a + (1 - blend) * a_base
                        a_blend = safe_normalize(a_blend, n)
                        c_blend = fft_convolve_self(a_blend)
                        if c_blend.max() < global_best:
                            global_best = c_blend.max()
                            global_a = a_blend.copy()
                            print("  Blend %.1f: %.8f" % (blend, global_best))

                # Deep L-BFGS with high p
                a_new, score = run_lbfgs_pnorm(
                    safe_normalize(a_base, n), n, S, deep_budget, rng,
                    p_schedule=[32, 64, 128, 256, 512, 1024, 2048, 4096],
                    maxiter_per_p=200
                )
                a_new = safe_normalize(a_new, n)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < global_best:
                    global_best = c_new.max()
                    global_a = a_new.copy()
                    print("  Deep %d: %.8f" % (di, global_best))

    print("\n=== After Phase 2: best=%.8f ===" % global_best)

    # --- Phase 3: Top-k Nesterov polishing + Peak shaving ---
    if global_a is not None:
        remaining3 = deadline - time.monotonic()
        if remaining3 > 30:
            print("\n=== Phase 3: Polishing (%.0f sec) ===" % remaining3)

            # Top-k Nesterov
            seg1 = min(remaining3 * 0.4, 180)
            if seg1 > 10:
                a_pol, score_pol = run_topk_nesterov(
                    global_a, n, S, seg1, rng,
                    top_k_init=8, top_k_max=128
                )
                a_pol = safe_normalize(a_pol, n)
                c_pol = fft_convolve_self(a_pol)
                if c_pol.max() < global_best:
                    global_best = c_pol.max()
                    global_a = a_pol.copy()
                    print("  Top-k Nesterov: %.8f" % global_best)

            # Peak shaving
            remaining3b = deadline - time.monotonic()
            if remaining3b > 15:
                a_ps, score_ps = run_peak_shaving(
                    global_a, n, S, remaining3b * 0.5, rng
                )
                a_ps = safe_normalize(a_ps, n)
                c_ps = fft_convolve_self(a_ps)
                if c_ps.max() < global_best:
                    global_best = c_ps.max()
                    global_a = a_ps.copy()
                    print("  Peak shaving: %.8f" % global_best)

    # --- Phase 4: Final high-p L-BFGS polish ---
    if global_a is not None:
        remaining4 = deadline - time.monotonic()
        if remaining4 > 20:
            print("\n=== Phase 4: Final L-BFGS polish (%.0f sec) ===" % remaining4)
            a_final, score_final = run_lbfgs_pnorm(
                global_a, n, S, remaining4, rng,
                p_schedule=[128, 256, 512, 1024, 2048, 4096, 8192],
                maxiter_per_p=300
            )
            a_final = safe_normalize(a_final, n)
            c_final = fft_convolve_self(a_final)
            if c_final.max() < global_best:
                global_best = c_final.max()
                global_a = a_final.copy()
                print("  Final L-BFGS: %.8f" % global_best)

    # --- Verification ---
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\n=== Verification: %.6f ===" % (ver if np.isfinite(ver) else float('inf')))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        global_a = np.ones(512, dtype=np.float64)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, score=%.8f" % (len(final_list), global_best))
    return final_list
