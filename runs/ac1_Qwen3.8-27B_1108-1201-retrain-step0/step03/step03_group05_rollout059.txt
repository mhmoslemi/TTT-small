Algorithm:

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_cross_corr(w, a):
    """Compute sum_k w[k] * a[k-j] for j=0..n-1 via FFT (valid index range)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return np.real(ifft(W * np.conj(A)))[:n]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(w_sorted)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w_sorted
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z (unconstrained softmax params)."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta
    w = exp_bc / sum_exp_bc
    grad_a = 2.0 * compute_cross_corr(w, a)
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)
    return obj, grad_z


def run_lbfgs_continuation(a_init, n, S, max_time, rng, beta_schedule=None, maxiter_per_beta=250):
    """Run L-BFGS-B with continuation on beta. Returns best (a, score)."""
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break

        n_attempts = 1
        if beta >= 500 and max_time > 60:
            n_attempts = 2

            if time.monotonic() - start > max_time - 3:
                break
            if attempt > 0:
                z_pert = z + rng.normal(0, 0.015, n)
                z_pert = z_pert - z_pert.mean()
            else:
                z_pert = z.copy()

            def _fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def _jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            t0 = time.monotonic()
            try:

                    _fun, z_pert, jac=_jac, method='L-BFGS-B',

                )
            except Exception:
                break

            z = result.x
            z = z - z.mean()
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)

            a_new = np.maximum(a_new, 0.0)

            score = c.max()
            if score < best_score:

                best_a = a_new.copy()

def active_set_subgradient(a_init, n, S, max_time, rng, init_step=None):
    """
    Nesterov-accelerated subgradient with adaptive active set and backtracking.
    Focuses on the top-k lags where c_k is largest.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    top_k = 20

    if init_step is None:
        step = 1.0 / (S * np.sqrt(n))
    else:
        step = init_step

    while time.monotonic() - start < max_time:
        it += 1
        # Grow top_k slowly
        top_k = min(256, 10 + it // 50)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = y.copy()

        # Find active set: lags where c_k is close to max
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[min(top_k - 1, len(c_sorted) - 1)] * 0.95
        active_mask = c >= threshold
        n_active = active_mask.sum()
        if n_active == 0:
            active_mask = np.zeros(2 * n - 1, dtype=bool)
            active_mask[np.argmax(c)] = 1
            n_active = 1

        # Use sharpened weights on active set
        c_active = c[active_mask]
        # Exponential weighting (sharper than uniform)
        w_alpha = 30.0  # sharpening parameter
        w_vals = np.exp(w_alpha * (c_active - c_active.max()))
        w_vals = w_vals / w_vals.sum()

        w_full = np.zeros(2 * n - 1)
        active_indices = np.where(active_mask)[0]
        for i, k in enumerate(active_indices):
            w_full[int(k)] = w_vals[i]

        # Compute gradient
        grad = 2.0 * compute_cross_corr(w_full, a)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step_size = step / np.sqrt(1 + 0.5 * it)
        alpha = step_size
        a_curr = a.copy()
        c_curr = fft_convolve_self(a_curr)
        obj_curr = c_curr.max()

        for _ in range(30):
            x_trial = project_simplex(a_curr - alpha * grad, S)
            c_trial = fft_convolve_self(x_trial)
            obj_trial = c_trial.max()
            if obj_trial < obj_curr - 1e-12:
                break
            alpha *= 0.5

        if alpha < 1e-10:
            alpha = step_size / (1 + 0.1 * it)
            x_trial = project_simplex(a_curr - alpha * grad, S)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t - 1.0) / t_new
        y = np.maximum(x_trial + momentum * (x_trial - a), 0.0)
        a = x_trial
        t = t_new

        # Periodic check
        if it % 200 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() < best_score:
                best_score = c_check.max()
                best_a = a.copy()

        # Random perturbation for escaping
        if it % 2000 == 0 and rng.random() < 0.15:
            y = project_simplex(y + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            a = y.copy()
            t = 1.0

    return best_a, best_score


def coordinate_descent(a_init, n, S, max_time, rng, window=5, n_trials=8):
    """
    Coordinate descent: try changing each coordinate to nearby values.
    Uses a random subset of coordinates per pass for efficiency.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    # Pre-compute current convolution
    c_current = fft_convolve_self(a)
    current_max = c_current.max()
    pass_count = 0

    while time.monotonic() - start < max_time:
        pass_count += 1
        # Number of coordinates to try per pass (increasing over time)
        n_coords = min(n, 200 + pass_count * 50)
        if n_coords >= n:
            indices = np.arange(n)
        else:
            indices = rng.choice(n, size=n_coords, replace=False)
            np.sort(indices)

        improved = False
        for j in indices:
            if time.monotonic() - start > max_time - 0.5:
                break

            orig_val = a[j]
            if orig_val < 1e-10:
                continue

            # Try several candidate values
            candidates = []
            delta = orig_val * 0.1
            for frac in [0.7, 0.85, 1.15, 1.3]:
                candidates.append(orig_val * frac)
            for d in [-delta, delta, -2*delta, 2*delta]:
                candidates.append(max(0.0, orig_val + d))
            # Also try small absolute changes
            for d in [-0.01 * S / n, 0.01 * S / n]:
                candidates.append(max(0.0, orig_val + d))

            best_trial = orig_val
            best_trial_score = current_max

            for cv in candidates:
                if cv <= 0 and orig_val <= 0:
                    continue
                a[j] = cv
                # Quick evaluation: only need to check affected lags
                # For efficiency, compute full convolution only for promising candidates
                # Here we do full convolution (expensive but safe)
                c_new = fft_convolve_self(a)
                score = c_new.max()
                if score < best_trial_score:
                    best_trial_score = score
                    best_trial = cv

            a[j] = best_trial
            if best_trial != orig_val:
                improved = True
                c_current = fft_convolve_self(a)
                current_max = c_current.max()
                if current_max < best_score:
                    best_score = current_max
                    best_a = a.copy()

            n_perturb = min(20, n // n)
            for _ in range(n_perturb):

            current_max = c_current.max()
            if current_max < best_score:
                best_score = current_max
                best_a = a.copy()

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def block_mutation(a, n, rng, block_size=64):
    """Replace a random block with new random values."""
    a_new = a.copy()
    start = rng.integers(0, max(1, n - block_size))
    end = start + block_size
    S = np.sqrt(2.0 * float(n))
    # Generate new block values (exponential-like distribution)
    new_block = rng.exponential(S / n, size=block_size)
    a_new[start:end] = new_block
    return safe_normalize(a_new, n)

    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(min(budget_s, 2380))
    deadline = time.monotonic() + budget_s

    global_best = np.inf
    global_a = None

    # --- Get initial sequence ---
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0.0, 1000.0)))
    except Exception:
        pass

    if inits:
        main_len = len(inits[0][1])
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * float(n))
    tt = np.linspace(0, 1, n, endpoint=False)

    # Additional initializations
    inits.append(("rand", rng.random(n) + 0.1))
    inits.append(("structured", 1.0 + 0.3 * np.sin(2 * np.pi * 5 * tt) +
                  0.2 * np.cos(2 * np.pi * 11 * tt) + 0.1 * np.sin(2 * np.pi * 17 * tt)))
    inits.append(("bump", np.exp(-0.5 * ((tt - 0.3) * 8.0) ** 2) +
                   0.6 * np.exp(-0.5 * ((tt - 0.7) * 8.0) ** 2) + 0.05))

    n_inits = len(inits)
    print("=== Optimization: n=%d, %d inits, budget=%.0fs ===" % (n, n_inits, budget_s))
    print("Target: 1.5027, Previous: 1.508805")

    # --- Phase 1: L-BFGS continuation (55% of budget split among inits) ---
    phase1_budget = (deadline - time.monotonic()) * 0.40
    per_init = max(15.0, phase1_budget / n_inits)

    print("\n=== Phase 1: L-BFGS Continuation ===")
    phase1_results = []

    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break
        t0 = time.monotonic()
        a_norm = safe_normalize(arr, n)
        c = fft_convolve_self(a_norm)
        init_score = c.max()

        if init_score < global_best:
            global_best = init_score
            global_a = a_norm.copy()

        # Use adaptive continuation
        if init_score < 1.6:
            # Already good, use finer schedule
            beta_sched = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
            maxit = 300
        else:
            beta_sched = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
            maxit = 250

        a_new, score = run_lbfgs_continuation(
            a_norm, n, S, per_init, rng,
            beta_schedule=beta_sched, maxiter_per_beta=maxit
        )
        a_new = safe_normalize(a_new, n)
        c_new = fft_convolve_self(a_new)
        final_score = c_new.max()
        phase1_results.append((name, a_new, final_score))

        if final_score < global_best:
            global_best = final_score
            global_a = a_new.copy()
            print("  [IMPROVE] %s: %.8f (was %.4f, %.1fs)" %
                  (name, global_best, init_score, time.monotonic() - t0))
        else:
            print("  %s: %.8f (not better than %.8f, %.1fs)" %
                  (name, final_score, global_best, time.monotonic() - t0))

    print("\n--- After Phase 1: best=%.8f ---" % global_best)

    # --- Phase 2: Active-set subgradient refinement ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        phase2_budget = min(remaining * 0.30, 600)
        if phase2_budget > 30:
            print("\n=== Phase 2: Active-Set Subgradient (%.0fs) ===" % phase2_budget)

            # Run on best from phase 1
            a_sub, score_sub = active_set_subgradient(
                global_a, n, S, phase2_budget, rng
            )
            a_sub = safe_normalize(a_sub, n)
            c_sub = fft_convolve_self(a_sub)
            if c_sub.max() < global_best:
                global_best = c_sub.max()
                global_a = a_sub.copy()
                print("  Active-set: %.8f" % global_best)
            else:
                print("  Active-set: %.8f (no improvement, best=%.8f)" % (c_sub.max(), global_best))

    print("\n--- After Phase 2: best=%.8f ---" % global_best)

    # --- Phase 3: Coordinate descent + perturbations ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        phase3_budget = min(remaining * 0.25, 400)
        if phase3_budget > 20:
            print("\n=== Phase 3: Coordinate Descent + Perturbations (%.0fs) ===" % phase3_budget)

            # First: coordinate descent on current best
            cd_budget = phase3_budget * 0.5
            a_cd, score_cd = coordinate_descent(
                global_a, n, S, cd_budget, rng
            )
            a_cd = safe_normalize(a_cd, n)
            c_cd = fft_convolve_self(a_cd)
            if c_cd.max() < global_best:
                global_best = c_cd.max()
                global_a = a_cd.copy()
                print("  CoordDescent: %.8f" % global_best)
            else:
                print("  CoordDescent: %.8f (no improvement)" % c_cd.max())

            # Then: perturbation + mini L-BFGS
            remaining3 = deadline - time.monotonic()
            if remaining3 > 30:
                n_perturb = max(2, int(remaining3 / 60))
                perturb_budget = remaining3 * 0.7 / n_perturb

                for p in range(n_perturb):
                    if time.monotonic() > deadline - 60:
                        break

                    # Different perturbation strategies
                    strategy = p % 3
                    if strategy == 0:
                        # Gaussian perturbation
                        noise_scale = 0.03 / (1.0 + 0.2 * p)
                        a_pert = global_a + rng.normal(0, noise_scale * S / np.sqrt(n), n)
                        a_pert = np.maximum(a_pert, 0.0)
                    elif strategy == 1:
                        # Block mutation
                        a_pert = block_mutation(global_a, n, rng, block_size=max(32, n // 32))
                    else:
                        # Sparse perturbation (change a few coordinates)
                        a_pert = global_a.copy()
                        n_sparse = max(10, n // 100)
                        sparse_idx = rng.choice(n, size=n_sparse, replace=False)
                        for j in sparse_idx:
                            a_pert[j] = max(0.0, a_pert[j] * (1 + rng.normal(0, 0.3)))

                    a_pert = safe_normalize(a_pert, n)

                    # Quick L-BFGS on perturbed
                    a_new, score_new = run_lbfgs_continuation(
                        a_pert, n, S, perturb_budget, rng,
                        beta_schedule=[50, 100, 300, 800, 2000, 5000],
                        maxiter_per_beta=150
                    )
                    a_new = safe_normalize(a_new, n)
                    c_new = fft_convolve_self(a_new)
                    if c_new.max() < global_best:
                        global_best = c_new.max()
                        global_a = a_new.copy()
                        print("  Perturb %d (strat=%d): %.8f" % (p, strategy, global_best))

    print("\n--- After Phase 3: best=%.8f ---" % global_best)

    # --- Phase 4: Cross-pollination between best results ---
    if global_a is not None and len(phase1_results) >= 2:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print("\n=== Phase 4: Cross-Pollination (%.0fs) ===" % remaining)
            sorted_results = sorted(phase1_results, key=lambda x: x[2])
            best_a1 = sorted_results[0][1]
            best_a2 = sorted_results[1][1] if len(sorted_results) > 1 else global_a

            n_xpoll = max(2, int(remaining / 50))
            xpoll_budget = remaining * 0.6 / n_xpoll

            for xp in range(n_xpoll):
                if time.monotonic() > deadline - 30:
                    break
                for blend in [0.3, 0.5, 0.7]:
                    if time.monotonic() > deadline - 15:
                        break
                    a_blend = blend * best_a1 + (1 - blend) * best_a2
                    a_blend = safe_normalize(a_blend, n)
                    a_new, score = run_lbfgs_continuation(
                        a_blend, n, S, min(xpoll_budget / 3, 20), rng,
                        beta_schedule=[20, 50, 200, 500, 2000],
                        maxiter_per_beta=80
                    )
                    a_new = safe_normalize(a_new, n)
                    c_new = fft_convolve_self(a_new)
                    if c_new.max() < global_best:
                        global_best = c_new.max()
                        global_a = a_new.copy()
                        print("  XPoll %d (blend=%.1f): %.8f" % (xp, blend, global_best))

    print("\n--- After Phase 4: best=%.8f ---" % global_best)

    # --- Phase 5: Final intensive polishing ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print("\n=== Phase 5: Final Polishing (%.0fs) ===" % remaining)

            # High-beta L-BFGS
            seg1 = min(remaining * 0.5, 120)
            a_polish, score_polish = run_lbfgs_continuation(
                global_a, n, S, seg1, rng,
                beta_schedule=[500, 1000, 2000, 5000, 10000, 20000],
                maxiter_per_beta=200
            )
            a_polish = safe_normalize(a_polish, n)
            c_polish = fft_convolve_self(a_polish)
            if c_polish.max() < global_best:
                global_best = c_polish.max()
                global_a = a_polish.copy()
                print("  Final LBFGS: %.8f" % global_best)

            # One more round of active-set subgradient on the polished result
            remaining2 = deadline - time.monotonic()
            if remaining2 > 15:
                a_fin, score_fin = active_set_subgradient(
                    global_a, n, S, min(remaining2 * 0.7, 60), rng
                )
                a_fin = safe_normalize(a_fin, n)
                c_fin = fft_convolve_self(a_fin)
                if c_fin.max() < global_best:
                    global_best = c_fin.max()
                    global_a = a_fin.copy()
                    print("  Final ActiveSet: %.8f" % global_best)

            # Final coordinate descent
            remaining3 = deadline - time.monotonic()
            if remaining3 > 10:
                a_cd2, score_cd2 = coordinate_descent(
                    global_a, n, S, min(remaining3 * 0.8, 30), rng
                )
                a_cd2 = safe_normalize(a_cd2, n)
                c_cd2 = fft_convolve_self(a_cd2)
                if c_cd2.max() < global_best:
                    global_best = c_cd2.max()
                    global_a = a_cd2.copy()
                    print("  Final CoordDescent: %.8f" % global_best)

    # --- Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: score = %s" % ("%.8f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("\nReturning sequence of length %d, score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
