Population-based optimization with adaptive softmax, simulated annealing,
multi-scale refinement, and peak-aware coordinate descent to minimize
2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_from_conv(a, c):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def softmax_subgrad(c, a, temperature):
    """Compute smoothed subgradient using softmax weights."""
    n = len(a)
    c_max = c.max()
    w = np.exp(temperature * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        # All weights are ~0, use uniform over top few
        top_k = 10
        top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / top_k
        w_sum = 1.0
    else:
        w /= w_sum
    
    # grad[m] = 2 * sum_k w[k] * a[k-m]
    # This is 2 * (w * a_reversed) convolved, taking the middle part
    a_rev = a[::-1]
    out_len = len(c) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad

        mom = mom0 + (mom1 - mom0) * p

        if grad_mode == 'softmax':
            grad = smooth_subgrad(c, a, T)
        elif grad_mode == 'pair':
            grad = peak_pair_grad(c, a, n_top=max(1, int(5 / (0.1 + p))))
        else:
            grad = smooth_subgrad(c, a, T)
            if p > 0.5:
                grad2 = peak_pair_grad(c, a, n_top=3)
                grad = 0.5 * (grad / (np.linalg.norm(grad) + 1e-30) +
                              grad2 / (np.linalg.norm(grad2) + 1e-30))

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn

        v = mom * v + (1.0 - mom) * grad
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        v /= vn

        taken = False
        for bt in range(6):
            e = eta * (0.65 ** bt)
            a_new = a * np.exp(-e * v)
            a_new = np.maximum(a_new, 1e-300)
            a_new *= S / a_new.sum()
            c_new = fft_conv_self(a_new)
            o_new = obj_val(a_new, c_new)

            if o_new < best_obj * (1.0 - 1e-11):
                a, c = a_new, c_new
                best_obj = o_new
                best_a = a.copy()
                stall = 0
                taken = True
                break
            elif p < 0.55 and T > 2.0:
                accept_prob = np.exp(-(o_new - best_obj) / max(T * 0.003, 0.001))
                if np.random.random() < accept_prob:
                    a, c = a_new, c_new
                    taken = True
                    stall += 1
                    break

        if not taken:
            a = a * np.exp(-eta * 0.04 * v)
            a = np.maximum(a, 1e-300)
            a *= S / a.sum()
            c = fft_conv_self(a)
            stall += 1

        if stall >= perturb_every:
            scale = a.max()
            if scale < 1e-10:
                scale = S / n
            noise = np.random.randn(n) * perturb_scale * scale
            a = a + noise
            a = np.maximum(a, 0.0)
            a *= S / a.sum()
            c = fft_conv_self(a)
            v *= 0.05
            stall = 0
            if verbose:
                print(f"        [perturb] it={it} best={best_obj:.6f} t={elapsed:.0f}s")

        it += 1
        if verbose and it % 5000 == 0:
            print(f"        it={it} cur={obj_val(a, c):.6f} best={best_obj:.6f} "
                  f"T={T:.3f} eta={eta:.4f} t={elapsed:.0f}s")

    return best_a, best_obj


def coord_descent_peaks(a0, time_budget, max_coords=60):
    """Targeted coordinate descent on peak-contributing coordinates."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_conv_self(a)
    best_obj = obj_val(a, c)
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        c_sorted = np.sort(c)[::-1]
        thresh = c_sorted[min(40, len(c_sorted) - 1)]
        peak_idx = np.where(c >= thresh)[0]

        contrib = np.zeros(n)
        for k in peak_idx[:25]:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1):
                contrib[m] += a[m] * a[k - m]

        order = np.argsort(contrib)[::-1]
        improved = False

        for i in order[:max_coords]:
            if time.monotonic() - t0 > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            for f in [0.78, 0.88, 0.94, 1.06, 1.15]:
                a[i] = orig * f
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                an = a * (S / s)
                cn = fft_conv_self(an)
                on = obj_val(an, cn)
                if on < best_obj:
                    best_obj = on
                    best_a = an.copy()
                    a, c = an, cn
                    improved = True
                    break
                a[i] = orig

        if not improved:
            for _ in range(40):
                if time.monotonic() - t0 > time_budget:
                    break
                i = np.random.randint(n)
                orig = a[i]
                if orig < 1e-12:
                    continue
                f = np.random.uniform(0.7, 1.3)
                a[i] = orig * f
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                an = a * (S / s)
                cn = fft_conv_self(an)
                on = obj_val(an, cn)
                if on < best_obj:
                    best_obj = on
                    best_a = an.copy()
                    a, c = an, cn
                    improved = True
                a[i] = orig

        if not improved:
            break

    return best_a, best_obj


def generate_diverse_cands(n, rng):
    """Generate diverse initial candidates."""
    S = np.sqrt(2.0 * n)
    i_arr = np.arange(n)
    cands = []

    a = np.ones(n) * S / n
    cands.append(('uniform', a))

    for freq in [3, 7, 11, 17]:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * i_arr / n)
        a = np.maximum(a, 0.005)
        a *= S / a.sum()
        cands.append((f'cos{freq}', a))

    a = 1.0 + 0.3 * np.sin(np.pi * i_arr / n)
    a = np.maximum(a, 0.005)
    a *= S / a.sum()
    cands.append(('sin_half', a))

    a = rng.random(n)
    a = np.maximum(a, 0.005)
    a *= S / a.sum()
    cands.append(('rand', a))

    centers = rng.uniform(n * 0.1, n * 0.9, size=3)
    widths = rng.uniform(n * 0.04, n * 0.12, size=3)
    a = np.zeros(n)
    for ct, wd in zip(centers, widths):
        a += np.exp(-0.5 * ((i_arr - ct) / wd) ** 2)
    a = np.maximum(a, 0.005)
    a *= S / a.sum()
    cands.append(('gauss3', a))

    a = 1.0 + 0.25 * np.cos(2 * np.pi * 7 * i_arr / n) + 0.15 * np.cos(2 * np.pi * 19 * i_arr / n)
    a = np.maximum(a, 0.005)
    a *= S / a.sum()
    cands.append(('cos7+19', a))

    return cands


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search: diverse initialization, adaptive-temperature softmax
    subgradient with SA, peak-pair coordinate descent, and structured restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 12.0)
    global_best = np.inf
    global_a = None
    n = 4096
    S = np.sqrt(2.0 * n)

    # Load given sequence
    given_a = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        raw = np.clip(raw, 0.0, 1000.0)
        if raw.size >= 16 and raw.sum() > 0.01:
            if raw.size != n:
                idx = np.linspace(0, raw.size - 1, n).astype(int)
                raw = raw[idx]
            given_a = raw * (S / raw.sum())
            ver = evaluate_sequence(given_a.tolist())
            print(f"Given: obj={ver:.6f}")
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print(f"Load error: {e}")

    # Generate candidates
    cands = generate_diverse_cands(n, rng)
    if given_a is not None:
        cands.insert(0, ('given', given_a.copy()))

    # Evaluate all candidates
    print(f"\nInitial candidates ({len(cands)}):")
    for name, a_c in cands:
        c_t = fft_conv_self(a_c)
        o_t = obj_val(a_c, c_t)
        print(f"  {name}: {o_t:.6f}")
        if o_t < global_best:
            global_best = o_t
            global_a = a_c.copy()

    # Phase 1: Quick optimization on each candidate (12%)
    rem = deadline - time.monotonic()
    p1_total = 0.12 * rem
    per_cand = max(2.0, p1_total / len(cands))
    print(f"\nPhase 1: Quick opt ({per_cand:.1f}s/cand, {p1_total:.0f}s total)")

    for name, a_c in cands:
        if time.monotonic() > deadline - 80:
            break
        ab, ob = optimize_adaptive(a_c, per_cand, 60, 5, 0.35, 0.12, 0.7, 0.88,
                                    perturb_every=300, perturb_scale=0.05, verbose=False)
        tag = " *" if ob < global_best else ""
        print(f"  {name}: {ob:.6f}{tag}")
        if ob < global_best:
            global_best = ob
            global_a = ab.copy()

    # Phase 2: Main optimization (48% of remaining)
    rem = deadline - time.monotonic() - 30
    p2 = 0.48 * rem
    print(f"\nPhase 2: Main ({p2:.0f}s)")

    if global_a is not None and p2 > 10:
        # 2a: High-T exploration with softmax
        t2a = p2 * 0.33
        ab, ob = optimize_adaptive(global_a, t2a, 150, 8, 0.4, 0.18, 0.7, 0.85,
                                    perturb_every=400, perturb_scale=0.06,
                                    grad_mode='softmax', verbose=True)
        tag = " *" if ob < global_best else ""
        print(f"  2a (high-T softmax): {ob:.6f}{tag}")
        if ob < global_best:
            global_best, global_a = ob, ab.copy()

        # Short coord descent
        t_cd = min(5.0, rem * 0.02)
        if t_cd > 2:
            ab, ob = coord_descent_peaks(global_a, t_cd, max_coords=50)
            if ob < global_best:
                global_best, global_a = ob, ab.copy()
                print(f"  2a-cd: {ob:.6f} *")

        # 2b: Medium-T with mixed gradient
        t2b = p2 * 0.34
        ab, ob = optimize_adaptive(global_a, t2b, 15, 1.0, 0.22, 0.08, 0.85, 0.93,
                                    perturb_every=350, perturb_scale=0.04,
                                    grad_mode='mixed', verbose=True)
        tag = " *" if ob < global_best else ""
        print(f"  2b (med-T mixed): {ob:.6f}{tag}")
        if ob < global_best:
            global_best, global_a = ob, ab.copy()

        # 2c: Low-T exploitation with pair gradient
        t2c = p2 * 0.33
        ab, ob = optimize_adaptive(global_a, t2c, 3, 0.1, 0.14, 0.03, 0.92, 0.97,
                                    perturb_every=250, perturb_scale=0.02,
                                    grad_mode='pair', verbose=True)
        tag = " *" if ob < global_best else ""
        print(f"  2c (low-T pair): {ob:.6f}{tag}")
        if ob < global_best:
            global_best, global_a = ob, ab.copy()

    # Phase 3: Coordinate descent (28% of remaining)
    rem = deadline - time.monotonic() - 20
    p3 = 0.28 * rem
    print(f"\nPhase 3: Coord descent ({p3:.0f}s)")
    if global_a is not None and p3 > 5:
        ab, ob = coord_descent_peaks(global_a, p3, max_coords=80)
        tag = " *" if ob < global_best else ""
        print(f"  coord: {ob:.6f}{tag}")
        if ob < global_best:
            global_best, global_a = ob, ab.copy()

    # Phase 4: Final fine-tuning (30% of remaining)
    rem = deadline - time.monotonic() - 10
    p4 = 0.35 * rem
    print(f"\nPhase 4: Final polish ({p4:.0f}s)")
    if global_a is not None and p4 > 3:
        ab, ob = optimize_adaptive(global_a, p4, 1.0, 0.05, 0.08, 0.015,
                                    0.95, 0.98, perturb_every=200,
                                    perturb_scale=0.01, grad_mode='mixed', verbose=True)
        tag = " *" if ob < global_best else ""
        print(f"  final: {ob:.6f}{tag}")
        if ob < global_best:
            global_best, global_a = ob, ab.copy()

    # Phase 5: Final coord descent if time remains
    rem = deadline - time.monotonic() - 5
    if rem > 3 and global_a is not None:
        ab, ob = coord_descent_peaks(global_a, rem, max_coords=100)
        if ob < global_best:
            global_best, global_a = ob, ab.copy()
            print(f"  final-cd: {ob:.6f} *")

    # Final verification
    if global_a is None:
        global_a = np.ones(n) * S / n
    global_a = np.clip(global_a.astype(np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(n) * S / n

    final = global_a.tolist()
    ver = evaluate_sequence(final)
    print(f"\nRESULT: local={global_best:.6f} verified={ver:.6f}")

    if not np.isfinite(ver):
        final = [1.0] * n
        ver = evaluate_sequence(final)
        print(f"fallback: {ver:.6f}")

    return final
