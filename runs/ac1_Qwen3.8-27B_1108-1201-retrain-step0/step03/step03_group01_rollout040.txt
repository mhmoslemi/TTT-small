```python
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def gerchberg_saxe(a, N, n_iter, S):
    """
    Alternating projection: flat spectrum <-> non-negativity + sum constraint.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = np.inf
    
    for it in range(n_iter):
        # Step 1: Go to frequency domain
        A = fft(a, N)
        
        # Step 2: Flatten the spectrum (set magnitude to constant)
        # Use the current mean magnitude as the target
        mag_mean = np.mean(np.abs(A))
        A_new = mag_mean * np.exp(1j * np.angle(A))
        
        # Step 3: Back to time domain
        a_new = np.real(ifft(A_new))[:n]
        
        # Step 4: Project to non-negative simplex
        a_new = project_simplex(a_new, S)
        
        # Check improvement
        c = fft_convolve_self(a_new)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score
```

```python
"""
Minimize 2*n*max(convolve(a,a)) / sum(a)^2 using:
1. Multi-n search (try different sequence lengths)
2. Modified Gerchberg-Saxe spectral shaping with adaptive correction
3. Log-domain Nesterov LSE optimization
4. Peak-chasing descent that directly targets the maximum convolution value
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a, N=None):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    if N is None:
        N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]

def spectral_shape_step(a, N, c, top_k, alpha, S):
    """
    One step of spectral shaping: modify |FFT(a)|^2 to reduce peak convolution.
    Returns new a projected to non-negative simplex.
    """
    n = len(a)
    A = fft(a, N)
    P = np.abs(A) ** 2
    
    # Find the top peaks in the convolution
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    
    weights = top_vals / total
    
    # Construct the effective cosine
    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
    
    # Apply exponential correction (softer than linear)
    # Normalize cos_eff to have zero mean
    cos_eff = cos_eff - cos_eff.mean()
    max_ce = np.abs(cos_eff).max()
    if max_ce < 1e-15:
        return a
    cos_eff = cos_eff / max_ce
    
    # Modify the power spectrum
    P_new = P * np.exp(-alpha * cos_eff)
    P_new = np.maximum(P_new, 1e-30)
    
    # IFFT with modified magnitude, keeping phase
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    
    # Project to non-negative simplex
    a_new = project_simplex(a_new, S)
    return a_new

def spectral_shaping_loop(a, N, S, n_iter, rng, alpha_init=0.5, top_k_init=5):
    """
    Run spectral shaping for n_iter iterations with adaptive parameters.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = np.inf
    c = fft_convolve_self(a, N)
    
    for it in range(n_iter):
        c = fft_convolve_self(a, N)
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        
        # Adaptive parameters
        top_k = min(50, top_k_init + it // 20)
        # Reduce alpha over time for convergence
        alpha = alpha_init * 0.999 ** it
        
        a = spectral_shape_step(a, N, c, top_k, alpha, S)
        
        # Periodic perturbation
        if it % 200 == 100 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
    
    return best_a, best_score

def log_nesterov_lse(a_init, n, S, N, max_time, rng, beta_init=5.0, beta_max=500.0):
    """
    Nesterov-accelerated gradient descent in log-domain with LSE smoothing.
    """
    # Initialize in log-domain
    a = np.clip(a_init, 1e-15, 1000.0).copy()
    s = a.sum()
    a = a * (S / s)
    b = np.log(a / S + 1e-30)  # log of normalized values
    
    y = b.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.002 * it))
        lr = 1.0 / (beta * (1.0 + 0.001 * it))
        
        # Compute a from log-domain
        b_curr = np.clip(y, -50, 50)
        a_curr = np.exp(b_curr - b_curr.max())
        a_curr = a_curr / a_curr.sum() * S
        
        # Compute convolution
        c = fft_convolve_self(a_curr, N)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a_curr.copy()
        
        # LSE gradient
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total
        
        # Gradient of LSE objective w.r.t. a
        # d(c_k)/da_j = a_{k-j} + a_{j-k} (for valid indices)
        # In FFT: grad_a = 2 * IFFT(FFT(w) * conj(FFT(a)))
        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a_curr, fsize)
        grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Convert to log-domain gradient
        # db_j = d(da_j / a_j) approximately
        # da_j / db_j = a_j * (1 - a_j/S) (for softmax parameterization)
        # But we're using a simpler parameterization: a_j = S * exp(b_j) / sum(exp(b))
        # da_j/db_j = a_j * (1 - a_j/S)
        da_db = a_curr * (1.0 - a_curr / S)
        da_db = np.maximum(da_db, 1e-15)
        grad_b = grad_a / da_db
        
        # Nesterov step
        y_new = y - lr * grad_b
        y_new = np.clip(y_new, -50, 50)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = y_new + ((t - 1.0) / t_new) * (y_new - b)
        y = np.clip(y, -50, 50)
        b = b.copy()
        
        # Periodic perturbation
        if it % 500 == 250 and rng.random() < 0.15:
            y = y + rng.normal(0, 0.01, n)
            t = 1.0
    
    return best_a, best_score

        c_max = c.max()

            best_score = c_max
            best_a = a.copy()
        
        # Find the peak position
        k_peak = int(np.argmax(c))
        
        # Construct a targeted correction: reduce power at frequencies

        A = fft(a, N)
        P = np.abs(A) ** 2
        m = np.arange(N)
        cos_k = np.cos(2.0 * np.pi * m * k_peak / N)

        alpha = 0.1 * (1.0 + 0.1 * min(it, 10))
        P_new = P * np.exp(-alpha * cos_k)
        P_new = np.maximum(P_new, 1e-30)
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new, S)
        
        # Also try a small time-domain adjustment: reduce values at
        # positions that contribute most to the peak

            contrib = np.zeros(n)

                j = k_peak - i

                    contrib[i] = a[i] * a[j]
            # Reduce the top contributors slightly
            top_contrib_idx = np.argsort(contrib)[::-1][:max(1, n//10)]
            a[top_contrib_idx] *= 0.99
            a = project_simplex(a, S)
    
    return best_a, best_score

def make_smooth_init(n, rng):
    """Create a smooth initialization that's close to optimal."""
    t = np.arange(n) / n
    # Multi-frequency smooth function
    a = 1.0
    for freq in [2, 3, 5, 7, 11]:
        a += 0.3 / freq * np.sin(2 * np.pi * freq * t + rng.random() * 2 * np.pi)
    a = np.clip(a, 0.01, None)
    return a

def make_gaussian_init(n, rng, n_bumps=5):
    """Create a multi-bump Gaussian initialization."""
    a = np.zeros(n)
    for _ in range(n_bumps):
        mu = rng.random()
        sigma = 0.05 + rng.random() * 0.15
        h = 0.5 + rng.random()
        t = np.arange(n) / n
        a += h * np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    a += 0.05  # small floor to avoid zeros
    return a

def safe_normalize(a, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)
```

```python
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using aggressive spectral shaping,
Nesterov subgradient, and adaptive peak-chasing with multi-start strategy.
Key improvements: adaptive alpha based on non-negativity margin, direct peak targeting,

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a, N):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_gradient(a, c, N, w):
    """Compute gradient of weighted sum of convolution values w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
    else:
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            return a
        weights = top_vals / total
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    # Normalize
    cos_eff = cos_eff - cos_eff.mean()
    max_ce = np.abs(cos_eff).max()
    if max_ce < 1e-15:
        return a
    cos_eff = cos_eff / max_ce

    # Apply exponential correction
    P_new = P * np.exp(-alpha * cos_eff)
    P_new = np.maximum(P_new, 1e-30)

    # IFFT
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]

    # Project to non-negative simplex
    a_new = project_simplex(a_new_full, S)
    return a_new


def spectral_shaping_v2(a_init, N, S, max_time, rng, alpha_init=0.6, top_k_init=3):
    """
    Aggressive spectral shaping with adaptive alpha and periodic restarts.
    """
    n = len(a_init)
    a = np.clip(a_init, 0.0, None).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_score = np.inf
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a, N)
        c_max = c.max()

        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        if best_score < 1.01:
            break

        # Adaptive alpha: larger when further from optimal
        mean_c = c[:2*n-1].mean()
        excess_ratio = max(0.0, (c_max - mean_c) / max(mean_c, 1e-10))
        alpha = alpha_init * (0.5 + 0.5 * min(1.0, excess_ratio))
        # Decay over time
        alpha *= max(0.3, 1.0 - 0.0001 * it)

        top_k = min(30, top_k_init + it // 30)

        a = spectral_step_aggressive(a, N, c, alpha, top_k, S, direct_peak=(it % 3 == 0))

        # Periodic large perturbation to escape local minima
        if it % 500 == 250:
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
        elif it % 500 == 400:
            # More aggressive: random reweighting
            a = a * rng.uniform(0.5, 1.5, n)
            a = project_simplex(a, S)

    return best_a, best_score


def nesterov_subgrad(a_init, N, S, max_time, rng, top_k_init=5, top_k_max=200):
    """
    Nesterov-accelerated projected subgradient method.
    Uses top-k active lags for the gradient.
    """
    n = len(a_init)
    a = np.clip(a_init, 0.0, None).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    momentum = 0.0
    mu = 0.85
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 20)

        c = fft_convolve_self(y, N)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        if best_score < 1.01:
            break

        # Top-k weights
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        w = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w[int(k)] = top_vals[i] / total

        # Gradient
        grad = fft_gradient(y, c, N, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = S / (n * (1.0 + 0.002 * it))

        # Nesterov-style momentum
        x_new = y - step * grad / max(gnorm, 1e-15)
        x_new = project_simplex(x_new, S)

        # Momentum update
        y = x_new + momentum * (x_new - y)
        y = np.maximum(y, 0.0)
        y = y / max(y.sum(), 1e-15) * S
        a = x_new

        # Periodic perturbation
        if it % 1000 == 500 and rng.random() < 0.2:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            a = y.copy()

    return best_local, best_score


def lse_refine(a_init, N, S, max_time, rng, beta_init=10.0, beta_max=1000.0):
    """
    FISTA with log-sum-exp smoothed objective for fine-grained optimization.
    """
    n = len(a_init)
    a = np.clip(a_init, 0.0, None).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))
        lr = 1.0 / (beta * 2.0)

        c = fft_convolve_self(y, N)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        if best_score < 1.01:
            break

        # LSE weights
        c_max = c.max()
        # Shift for numerical stability
        shifted = beta * (c - c_max)
        shifted = np.clip(shifted, -500, 500)
        exp_bc = np.exp(shifted)
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        # Gradient
        grad = fft_gradient(y, c, N, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = y - lr * grad
        x_new = project_simplex(x_new, S)

        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 2000 == 1000 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0
            a = y.copy()

    return best_local, best_score


def coordinate_refine(a_init, N, S, max_time, rng):
    """
    Coordinate descent: try random blocks of coordinates and optimize them.
    """
    n = len(a_init)
    a = np.clip(a_init, 0.0, None).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_score = np.inf
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a, N)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        if best_score < 1.01:
            break

        # Pick a random block to "shake up"
        block_size = max(10, n // 50)
        start_idx = rng.integers(0, n - block_size)
        end_idx = start_idx + block_size

        # Try several random perturbations of the block
        best_block = a[start_idx:end_idx].copy()
        best_block_score = cur

        for trial in range(20):
            # Random perturbation
            perturbation = rng.normal(0, 0.01 * S / np.sqrt(n), block_size)
            candidate = a.copy()
            candidate[start_idx:end_idx] = np.maximum(0, a[start_idx:end_idx] + perturbation)
            candidate = project_simplex(candidate, S)

            c_cand = fft_convolve_self(candidate, N)
            if c_cand.max() < best_block_score:
                best_block_score = c_cand.max()
                best_block = candidate[start_idx:end_idx].copy()

        a[start_idx:end_idx] = best_block
        a = project_simplex(a, S)

    return best_a, best_score


def make_init_smooth(n, rng):
    """Multi-frequency smooth initialization."""
    t = np.arange(n) / n
    a = np.ones(n)
    freqs = [2, 3, 5, 7, 11, 13, 17]
    for f in freqs:
        a += 0.2 / f * np.sin(2 * np.pi * f * t + rng.random() * 2 * np.pi)
    a = np.clip(a, 0.01, None)
    return a


def make_init_multibump(n, rng, n_bumps=7):
    """Multi-bump Gaussian initialization."""
    a = np.zeros(n)
    t = np.arange(n) / n
    for _ in range(n_bumps):
        mu = rng.random()
        sigma = 0.03 + rng.random() * 0.1
        h = 0.3 + rng.random() * 0.7
        a += h * np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    a += 0.02
    return a


def make_init_structured(n, rng):
    """
    Structured init: use a pattern that avoids periodicity.
    Based on the idea that irrational frequency ratios give good spreading.
    """
    t = np.arange(n) / n
    # Use golden ratio related frequencies
    phi = (1 + np.sqrt(5)) / 2
    a = 1.0
    a += 0.3 * np.sin(2 * np.pi * phi * t)
    a += 0.2 * np.sin(2 * np.pi * phi * phi * t + 0.5)
    a += 0.15 * np.sin(2 * np.pi * 1 / phi * t + 1.0)
    a += 0.1 * np.cos(2 * np.pi * (1 + 1/phi) * t)
    a += rng.normal(0, 0.05, n)
    a = np.clip(a, 0.01, None)
    return a


def safe_normalize(a, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def optimize_single_n(n, S, N, time_budget, rng):
    """
    Full optimization pipeline for a single n.
    Returns (best_a, best_score).
    """
    global_best = np.inf
    global_a = np.ones(n) * S / n
    t0 = time.monotonic()

    # --- Phase 1: Spectral shaping from multiple initializations ---
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            if len(given) == n:
                inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
            else:
                # Resample
                x_old = np.linspace(0, 1, len(given))
                x_new = np.linspace(0, 1, n)
                resampled = np.interp(x_new, x_old, np.clip(given, 0, 1000))
                inits.append(("given_rs", resampled))
    except Exception:
        pass

    inits.append(("smooth", make_init_smooth(n, rng)))
    inits.append(("bump", make_init_multibump(n, rng, 7)))
    inits.append(("struct", make_init_structured(n, rng)))

    n_inits = len(inits)
    phase1_budget = time_budget * 0.35
    per_init = phase1_budget / n_inits

    print("  Phase 1: %d inits, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() - t0 > t0 + phase1_budget:
            break

        a_norm = safe_normalize(arr, S)
        t_init_start = time.monotonic()

        # Spectral shaping
        ss_time = min(per_init * 0.6, 60.0)
        a_ss, score_ss = spectral_shaping_v2(a_norm, N, S, ss_time, rng,
                                              alpha_init=0.5, top_k_init=3)
        a_ss = safe_normalize(a_ss, S)
        c_ss = fft_convolve_self(a_ss, N)
        if c_ss.max() < global_best:
            global_best = c_ss.max()
            global_a = a_ss.copy()
            print("    [%s] SS: %.6f (t=%.1f)" % (name, global_best, time.monotonic() - t_init_start))

        # Quick Nesterov polish
        rem = per_init - (time.monotonic() - t_init_start)
        if rem > 3:
            a_ng, _ = nesterov_subgrad(a_ss, N, S, min(rem, 15.0), rng,
                                        top_k_init=5, top_k_max=50)
            a_ng = safe_normalize(a_ng, S)
            c_ng = fft_convolve_self(a_ng, N)
            if c_ng.max() < global_best:
                global_best = c_ng.max()
                global_a = a_ng.copy()
                print("    [%s] NG: %.6f" % (name, global_best))

    # --- Phase 2: Deep optimization on best candidate ---
    rem_total = (t0 + time_budget) - time.monotonic()
    if rem_total > 10:
        print("  Phase 2: Deep optimization (budget=%.0f sec)" % rem_total)

        a_curr = global_a.copy()
        stall = 0
        last_best = global_best
        it = 0

        while time.monotonic() < t0 + time_budget - 5:
            it += 1
            # Alternate between methods
            method = it % 4

            if method == 0 or method == 1:
                # Spectral shaping
                seg_time = min(20.0, (t0 + time_budget - 5) - time.monotonic())
                if seg_time < 2:
                    break
                a_curr, _ = spectral_shaping_v2(a_curr, N, S, seg_time, rng,
                                                 alpha_init=0.3 / (1 + 0.05 * it),
                                                 top_k_init=5)
            elif method == 2:
                # Nesterov
                seg_time = min(30.0, (t0 + time_budget - 5) - time.monotonic())
                if seg_time < 3:
                    break
                a_curr, _ = nesterov_subgrad(a_curr, N, S, seg_time, rng,
                                              top_k_init=10 + 5 * it, top_k_max=100)
            else:
                # LSE refine
                seg_time = min(20.0, (t0 + time_budget - 5) - time.monotonic())
                if seg_time < 3:
                    break
                a_curr, _ = lse_refine(a_curr, N, S, seg_time, rng,
                                        beta_init=20.0 + 10 * it, beta_max=500.0)

            a_curr = safe_normalize(a_curr, S)
            c_curr = fft_convolve_self(a_curr, N)

            if c_curr.max() < global_best:
                global_best = c_curr.max()
                global_a = a_curr.copy()
                last_best = global_best
                stall = 0
                if it % 3 == 0:
                    print("    [iter %d] %.6f" % (it, global_best))
            else:
                stall += 1

            if stall >= 3:
                # Restart with perturbation
                a_curr = project_simplex(a_curr + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
                stall = 0

    # --- Phase 3: Final polish ---
    rem = (t0 + time_budget) - time.monotonic()
    if rem > 5:
        print("  Phase 3: Final polish (budget=%.0f sec)" % rem)

        # High-beta LSE for very fine optimization
        seg = min(rem * 0.4, 60.0)
        a_p, _ = lse_refine(global_a, N, S, seg, rng, beta_init=100.0, beta_max=2000.0)
        a_p = safe_normalize(a_p, S)
        c_p = fft_convolve_self(a_p, N)
        if c_p.max() < global_best:
            global_best = c_p.max()
            global_a = a_p.copy()
            print("    LSE polish: %.6f" % global_best)

        # Final spectral micro-adjustments
        rem2 = (t0 + time_budget) - time.monotonic()
        if rem2 > 3:
            a_f, _ = spectral_shaping_v2(global_a, N, S, min(rem2, 30.0), rng,
                                          alpha_init=0.1, top_k_init=10)
            a_f = safe_normalize(a_f, S)
            c_f = fft_convolve_self(a_f, N)
            if c_f.max() < global_best:
                global_best = c_f.max()
                global_a = a_f.copy()
                print("    Final SS: %.6f" % global_best)

    return global_a, global_best


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Multi-phase: spectral shaping + Nesterov + LSE with adaptive parameters.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    print("=== Starting: budget=%.0f sec, seed=%d ===" % (budget_s, seed))

    global_best = np.inf
    global_a = None
    global_n = 8192

    # Determine n
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            global_n = len(given)
    except Exception:
        global_n = 8192

    n = global_n
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d" % (n, S, N))

    # Main optimization
    a_best, score_best = optimize_single_n(n, S, N, deadline - time.monotonic(), rng)

    if a_best is not None and score_best < global_best:
        global_best = score_best
        global_a = a_best.copy()

    # Verify and return
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            global_best = ver
        else:
            print("  WARNING: evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
            global_best = evaluate_sequence([float(x) for x in global_a])
    else:
        global_a = np.ones(256, dtype=np.float64)
        global_best = evaluate_sequence([float(x) for x in global_a])

    print("=== Done: length=%d, score=%.6f ===" % (len(global_a), global_best))
    return [float(x) for x in global_a]
```
