def solve_lp(a_t, S, n):
    """
    Solve the linearized LP.
    Variables: a_0, ..., a_{n-1}, M (index n)
    Constraints:
      2 * sum_j a_{k-j}^{(t)} * a_j - M <= 0  for k in S
      sum_j a_j = 1
      a_j >= 0
    Objective: minimize M
    """
    # c_matrix[i, j] = 2 * a_{S[i] - j}^{(t)} (0 if out of bounds)
    c_matrix = np.zeros((len(S), n))
    for i, k in enumerate(S):
        k = int(k)
        for j in range(max(0, k-n+1), min(n, k+1)):
            c_matrix[i, j] = 2.0 * a_t[k-j]
    
    # LP: min M = [0, ..., 0, 1] @ [a, M]
    # s.t. c_matrix @ a - M <= 0  => [c_matrix, -1] @ [a, M] <= 0
    #      sum a = 1  => [1, ..., 1, 0] @ [a, M] = 1
    #      a >= 0, M free (but M >= 0 implicitly)
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M
    
    A_ub = np.hstack([c_matrix, -np.ones((len(S), 1))])
    b_ub = np.zeros(len(S))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        a_new = result.x[:n]
        M_new = result.x[n]
    else:
        a_new = a_t
        M_new = np.inf
    
    return a_new, M_new

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using linearized active-set LP (outer approximation).

Key idea: At each iteration, linearize the self-convolution c[k] = sum_i a_i * a_{k-i}
around the current point and solve the resulting LP to find a new point. The active set
of "tight" lags grows over time, leading to convergence to a KKT point.

Combined with:
- Structured initializations (anti-palindromic, periodic, two-level)
- Damped LP steps with adaptive damping
- Periodic FISTA polish with LSE smoothing
- Multi-length exploration
- Spectral-flatness motivated initializations

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) * S / n
    if abs(s - S) < 1e-12:
        return v
    # Water-filling
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def build_and_solve_lp(a_t, active_lags, n):
    """
    Solve the linearized LP:
      min M
      s.t. 2 * sum_j a_t[k-j] * a_j <= M   for k in active_lags
           sum a_j = 1
           a_j >= 0
    Returns (a_new, M_lp, success).
    """
    K = len(active_lags)
    # Build constraint matrix G: G[i, j] = 2 * a_t[active_lags[i] - j]
    # Only non-zero when 0 <= active_lags[i] - j < n
    G = np.zeros((K, n))
    for i, k in enumerate(active_lags):
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n, k + 1)
        if j_min < j_max:
            G[i, j_min:j_max] = 2.0 * a_t[k - (np.arange(j_min, j_max))]

    # LP in variables x = [a_0, ..., a_{n-1}, M]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_ub = np.hstack([G, -np.ones((K, 1))])
    b_ub = np.zeros(K)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 30})

    if result.success:
        a_new = np.maximum(result.x[:n], 0.0)
        M_lp = result.x[n]
        return a_new, M_lp, True
    else:
        return a_t, np.inf, False

    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s  # Normalize to sum = 1

    best_score = np.inf

    stall = 0

        obj = 2.0 * n * cur_max  # Since sum(a) = 1

            stall = 0
        elif it % 10 == 0:
            stall += 1

        top_lags = np.argsort(c)[::-1][:K]

            alpha *= 0.6
            if obj_new > obj * 1.01:
                # Revert
                a = (1.0 - 0.1 * alpha) * best_a + 0.1 * alpha * a_lp
                a = np.maximum(a, 0.0)
                a = a / a.sum()

        # If stalled, try a perturbation
        if stall > 30 and rng.random() < 0.3:
            noise_scale = 0.005 / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), 1.0)
            alpha = 0.5
            stall = 0

        # Check for convergence
        if it > 100 and abs(obj - best_score) < 1e-10 * best_score:
            break

    # Final normalize to sum = sqrt(2n) for the evaluator
    best_a_final = best_a * S_target
    return best_a_final, best_score


def run_fista_polish(a_init, n, max_time, rng, beta_init=20.0, beta_max=500.0):
    """FISTA with LSE smoothing for polishing."""
    S_ = np.sqrt(2.0 * float(n))
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S_ / n
    else:
        a = a * (S_ / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = beta_init * 2.0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S_)

        c_x = fft_convolve_self(x_new)
        if c_x.max() > cur * 1.01:
            L *= 1.5
        elif it % 10 == 0:
            L = max(L * 0.95, beta * 2.0)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    return best_local, best_score


def run_topk_subgradient(a_init, n, max_time, rng, top_k=20, step_init=0.1):
    """Top-k subgradient with momentum and adaptive step."""
    S_ = np.sqrt(2.0 * float(n))
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S_ / n
    else:
        a = a * (S_ / s)

    v = np.zeros(n)  # momentum
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = step_init * S_ / np.sqrt(n)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Top-k subgradient
        top_lags = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_lags:
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(n, k + 1)
            grad[j_min:j_max] += 2.0 * a[k - np.arange(j_min, j_max)]
        grad /= len(top_lags)

        # Momentum update
        v = 0.9 * v + 0.1 * grad
        a_new = a - step * v
        a_new = project_simplex(a_new, S_)

        c_new = fft_convolve_self(a_new)
        if c_new.max() < best_score:
            best_score = c_new.max()
            best_local = a_new.copy()
        a = a_new

        # Decay step
        step *= 0.9995

    return best_local, best_score


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())
    return a


def make_periodic_init(n, rng, S, num_components=3):
    """Periodic initialization with multiple sinusoidal components."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_components):
        freq = rng.integers(2, 30)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.4
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_two_level_init(n, rng, S):
    """Two-level sequence: a_i in {low, high} with random pattern."""
    low = rng.random() * 0.3 + 0.05
    high = rng.random() * 0.5 + 0.8
    pattern = (rng.random(n) < 0.5).astype(np.float64)
    a = low + (high - low) * pattern
    # Add some smoothness
    kernel = np.ones(5) / 5
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_spectral_flat_init(n, rng, S):
    """Try to make the power spectrum flat (minimizes peak autocorrelation)."""
    # Start with a random phase spectrum
    fsize = 1 << (2 * n - 1).bit_length()
    # |A(w)|^2 should be flat => |A(w)| = const for all w
    # So A(w) = const * exp(i*phi(w)) for random phases
    # But we need a >= 0, so this is just an initialization heuristic
    phases = rng.uniform(0, 2 * np.pi, fsize)
    # Make spectrum conjugate-symmetric for real signal
    A = np.exp(1j * phases)
    A[0] = 1.0
    for w in range(1, fsize // 2 + 1):
        A[fsize - w] = np.conj(A[w])
    a = np.real(ifft(A))[:n]
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_w_shape_init(n, rng, S):
    """W-shaped initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    k = rng.integers(2, 6)
    a = np.abs(np.sin(2 * np.pi * k * tt)) + 0.05
    a = a * (S / a.sum())
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    S = np.sqrt(2.0 * float(n))
    inits = []

    if parent is not None and len(parent) >= 64:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if p.sum() > 0:
            inits.append(("parent", p * (S / p.sum())))

    for i in range(3):
        inits.append(("anti_pal_%d" % i, make_anti_palindromic_init(n, rng, S)))
    for i in range(2):
        inits.append(("periodic_%d" % i, make_periodic_init(n, rng, S)))
    for i in range(2):
        inits.append(("two_level_%d" % i, make_two_level_init(n, rng, S)))
    for i in range(2):
        inits.append(("wshape_%d" % i, make_w_shape_init(n, rng, S)))
    inits.append(("spectral", make_spectral_flat_init(n, rng, S)))

    # Random
    rand_a = rng.random(n) + 0.05
    inits.append(("random", rand_a * (S / rand_a.sum())))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses linearized active-set LP as the primary optimizer, complemented by
    FISTA polishing and top-k subgradient methods.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    primary_n = 8192
    if parent is not None and len(parent) >= 64:
        primary_n = len(parent)

    print("Primary length: %d, parent available: %s" % (primary_n, parent is not None))

    # === Phase 1: LP-based optimization from multiple initializations ===
    inits = generate_initializations(primary_n, rng, parent)
    n_inits = len(inits)

    # Allocate time: 60% for LP optimization, 20% for FISTA polish, 20% reserve
    total_time = max(60.0, deadline - time.monotonic())
    lp_time_each = total_time * 0.55 / max(1, n_inits)
    print("=== Phase 1: %d inits, %.1f sec each for LP ===" % (n_inits, lp_time_each))

    lp_results = []

    for name, arr in inits:
        if time.monotonic() > deadline - 120:
            break
        n = len(arr)
        if n < 64:
            continue
        t0 = time.monotonic()
        seg_time = min(lp_time_each, deadline - time.monotonic() - 100)
        if seg_time < 5:
            break

        a_lp, score_lp = run_lp_optimization(arr, n, seg_time, rng, verbose=(name == "parent"))
        obj_lp = 2.0 * n * (score_lp)  # score_lp is max_b when sum=1, obj = 2n*max_b
        # Actually score_lp is already 2*n*max_b (since we normalize sum=1)
        # Wait no: in run_lp_optimization, obj = 2.0 * n * cur_max where cur_max = c.max()
        # and sum(a) = 1. So obj = 2*n*max_b / sum^2 = 2*n*max_b. Correct.
        if obj_lp < global_best:
            global_best = obj_lp
            global_a = a_lp.copy()
            print("  LP [%s]: obj=%.8f (elapsed %.1fs)" % (name, obj_lp, time.monotonic() - t0))
        lp_results.append((obj_lp, a_lp, name))

    # Sort by score
    lp_results.sort(key=lambda x: x[0])
    print("=== After Phase 1: best=%.8f ===" % global_best)

    # === Phase 2: FISTA polish on the best LP result ===
    if global_a is not None and time.monotonic() < deadline - 60:
        n = len(global_a)
        remaining = deadline - time.monotonic()
        print("=== Phase 2: FISTA polish (remaining %.0f sec) ===" % remaining)

        # Run FISTA with increasing beta
        for beta_seq in [10.0, 50.0, 200.0]:
            if time.monotonic() > deadline - 30:
                break
            seg = min(60.0, max(10.0, (deadline - time.monotonic()) * 0.25))
            a_f, s_f = run_fista_polish(global_a, n, seg, rng,
                                         beta_init=beta_seq, beta_max=beta_seq * 10)
            # Evaluate
            S_ = np.sqrt(2.0 * n)
            s = a_f.sum()
            if s > 0:
                c_f = fft_convolve_self(a_f)
                obj_f = 2.0 * n * c_f.max() / (s * s)
            else:
                obj_f = np.inf
            if obj_f < global_best:
                global_best = obj_f
                global_a = a_f.copy()
                print("  FISTA beta=%.0f: obj=%.8f" % (beta_seq, obj_f))

    # === Phase 3: Top-k subgradient polish ===
    if global_a is not None and time.monotonic() < deadline - 40:
        n = len(global_a)
        S_ = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        print("=== Phase 3: Top-k subgradient (remaining %.0f sec) ===" % remaining)
        seg3 = min(80.0, max(10.0, remaining * 0.4))
        a_sg, s_sg = run_topk_subgradient(global_a, n, seg3, rng, top_k=30)
        s = a_sg.sum()
        if s > 0:
            c_sg = fft_convolve_self(a_sg)
            obj_sg = 2.0 * n * c_sg.max() / (s * s)
        else:
            obj_sg = np.inf
        if obj_sg < global_best:
            global_best = obj_sg
            global_a = a_sg.copy()
            print("  TopK subgrad: obj=%.8f" % obj_sg)

    # === Phase 4: Another round of LP on the best result ===
    if global_a is not None and time.monotonic() < deadline - 50:
        n = len(global_a)
        # Normalize for LP (needs sum=1)
        a_norm = global_a / global_a.sum()
        remaining = deadline - time.monotonic()
        print("=== Phase 4: LP refinement (remaining %.0f sec) ===" % remaining)
        seg4 = min(120.0, max(20.0, remaining * 0.6))
        a_lp2, score_lp2 = run_lp_optimization(a_norm, n, seg4, rng, verbose=False)
        obj_lp2 = 2.0 * n * score_lp2
        if obj_lp2 < global_best:
            global_best = obj_lp2
            global_a = a_lp2.copy()
            print("  LP refinement: obj=%.8f" % obj_lp2)

    # === Phase 5: Final FISTA polish ===
    if global_a is not None and time.monotonic() < deadline - 20:
        n = len(global_a)
        remaining = deadline - time.monotonic()
        seg5 = max(5.0, remaining * 0.7)
        a_f2, s_f2 = run_fista_polish(global_a, n, seg5, rng,
                                       beta_init=200.0, beta_max=2000.0)
        s = a_f2.sum()
        if s > 0:
            c_f2 = fft_convolve_self(a_f2)
            obj_f2 = 2.0 * n * c_f2.max() / (s * s)
        else:
            obj_f2 = np.inf
        if obj_f2 < global_best:
            global_best = obj_f2
            global_a = a_f2.copy()
            print("  Final FISTA: obj=%.8f" % obj_f2)

    # === Phase 6: Try longer sequence ===
    if global_a is not None and time.monotonic() < deadline - 100:
        n_long = primary_n * 2
        if n_long <= 16384:
            print("=== Phase 6: Try n=%d ===" % n_long)
            S_long = np.sqrt(2.0 * n_long)
            # Interpolate
            tt_old = np.linspace(0, 1, len(global_a))
            tt_new = np.linspace(0, 1, n_long)
            a_long_init = np.interp(tt_new, tt_old, global_a)
            a_long_init = np.maximum(a_long_init, 0.01)
            a_long_init = a_long_init * (S_long / a_long_init.sum())

            seg6 = min(80.0, max(20.0, (deadline - time.monotonic()) * 0.5))
            a_l, s_l = run_lp_optimization(a_long_init, n_long, seg6, rng, verbose=False)
            obj_l = 2.0 * n_long * s_l
            if obj_l < global_best:
                global_best = obj_l
                global_a = a_l.copy()
                print("  Long n=%d: obj=%.8f" % (n_long, obj_l))

    # === Final verification ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64) * 1.0

    # Ensure validity
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Verification: %.8f (internal: %.8f), length=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: evaluate_sequence rejected, using fallback")
        global_a = np.ones(1024, dtype=np.float64) * 1.0
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)

    print("=== Final score: %.8f ===" % (ver if np.isfinite(ver) else float('inf')))
    return final_list
