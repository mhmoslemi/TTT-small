Actually, I realize I should be more careful about the Gerchberg-Saxton approach. The standard GS algorithm alternates between:
- Enforcing the magnitude constraint in the frequency domain
- Enforcing the support constraint in the time domain

In our case:
- Frequency domain: we want |A(ω)| to be as flat as possible (not a hard constraint, but a soft one)
- Time domain: we want a ≥ 0 and sum(a) = S (hard constraints)

So the algorithm would be:
1. Compute A = FFT(a)
2. Compute |A| and its mean μ
3. Set |A_new| = |A| * (μ / |A|)^α for some α ∈ (0, 1) (partial flattening)
   Or: |A_new| = (1-α)*|A| + α*μ (interpolation toward flat)
4. A_new = |A_new| * exp(i*arg(A))
5. a_new = IFFT(A_new).real
6. Project a_new onto {a ≥ 0, sum = S} (simplex projection)
7. Set a = a_new, go to 1

The parameter α controls how aggressively we flatten. Small α = gentle, large α = aggressive.

This should work well because:
- It directly targets the spectral flatness (equivalent to autocorrelation flatness)
- The simplex projection is O(n)
- Each iteration is O(n log n) due to FFTs
- It can escape local minima that trap gradient methods (because it makes large changes in the frequency domain)

```python
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_objective(a):
    """Compute 2*n*max_c / sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def gs_step(a, alpha, S):
    """One Gerchberg-Saxton step for spectral flattening."""
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    # Only consider the first n frequencies (since a is real, the spectrum is symmetric)
    # Actually, for a real sequence, |A[k]| = |A[n-k]| (for size n FFT)
    # But we're using a larger FFT (fsize), so the symmetry is different
    # Let's just flatten the full spectrum
    mu = mag.mean()
    # Partial flattening: interpolate between current magnitude and mean
    mag_new = (1 - alpha) * mag + alpha * mu
    # Ensure non-negative
    mag_new = np.maximum(mag_new, 1e-10)
    phase = np.angle(A)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    # Project onto simplex
    a_new = project_simplex(a_new, S)
    return a_new

def nesterov_subgradient(a_init, n, S, max_time, rng, top_k=100, alpha_init=0.1):
    """Nesterov-accelerated subgradient method."""
    a = a_init.copy()
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = compute_objective(a)
    best_a = a.copy()
    it = 0
    
    # Adaptive step size
    A_est = 2.0 * S  # Rough Lipschitz constant estimate
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_score = 2.0 * n * cur_max / S**2
        if cur_score < best_score:
            best_score = cur_score
            best_a = y.copy()
        
        # Top-k weights
        k = min(top_k, 2 * n - 1)
        top_indices = np.argsort(c)[::-1][:k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Gradient: grad_a = 2 * (w * a) (cross-correlation)
        # Using the weights concentrated on top-k
        w_full = np.zeros(2 * n - 1)
        for i, idx in enumerate(top_indices):
            w_full[int(idx)] = weights[i]
        
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Adaptive step size
        step = 2.0 * S / (A_est * (1.0 + it * 0.01))
        # Or use the theoretical step: step = 2 / (L * t) where L is Lipschitz constant
        # L for max_conv on the simplex is roughly 2*max(a)*n
        
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic restart
        if it % 100 == 0 and rng.random() < 0.1:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            y = project_simplex(best_a + noise, S)
            a = y.copy()
            t = 1.0
    
    return best_a, best_score

def lbfgs_smoothing(a_init, n, S, max_time, rng, beta_start=100, beta_end=5000):
    """L-BFGS with log-sum-exp smoothing continuation."""
    a = a_init.copy()
    a = np.maximum(a, 1e-10)
    a = a * S / a.sum()
    
    # Convert to log parameters
    z = np.log(a / S)
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    
    # Beta schedule
    n_betas = 8
    betas = np.geomspace(beta_start, beta_end, n_betas)
    time_per_beta = max_time / n_betas
    
    for beta in betas:
        if time.monotonic() - start > max_time - 2:
            break
        
        def fun(z_val):
            n_ = len(z_val)
            z_s = z_val - z_val.max()
            exp_z = np.exp(z_s)
            a_ = S * exp_z / exp_z.sum()
            a_ = np.maximum(a_, 1e-15)
            a_ = a_ * S / a_.sum()
            c = fft_convolve_self(a_)
            c_max = c.max()
            # Log-sum-exp
            exp_bc = np.exp(beta * (c - c_max))
            sum_exp = exp_bc.sum()
            return (c_max + np.log(sum_exp)) / beta
        
        def jac(z_val):
            n_ = len(z_val)
            z_s = z_val - z_val.max()
            exp_z = np.exp(z_s)
            a_ = S * exp_z / exp_z.sum()
            a_ = np.maximum(a_, 1e-15)
            a_ = a_ * S / a_.sum()
            c = fft_convolve_self(a_)
            c_max = c.max()
            exp_bc = np.exp(beta * (c - c_max))
            sum_exp = exp_bc.sum()
            w = exp_bc / sum_exp
            # grad_a = 2 * convolve(w, a) (cross-correlation)
            fsize = 1 << (len(w) + n_ - 2).bit_length()
            W = fft(w, fsize)
            A = fft(a_, fsize)
            grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n_]
            dot = np.dot(grad_a, a_)
            grad_z = a_ * (grad_a - dot / S)
            return grad_z
        
        t0 = time.monotonic()
        maxiter = max(50, int(time_per_beta * 20))
        try:
            result = scipy_minimize(
                fun, z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'iprint': -1}
            )
            z = result.x
        except Exception:
            pass
        
        # Convert back
        z_s = z - z.max()
        exp_z = np.exp(z_s)
        a = S * exp_z / exp_z.sum()
        a = np.maximum(a, 0.0)
        a = a * S / a.sum()
        
        score = compute_objective(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        z = z - z.mean()
    
    return best_a, best_score
```

```python
"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm combines:
1. Gerchberg-Saxton spectral flattening (new: directly targets flat autocorrelation via FFT)
2. Nesterov-accelerated subgradient with adaptive step size and large top-k
3. L-BFGS with log-sum-exp continuation for high-precision polishing
4. Random restarts and cross-pollination to escape local minima
5. Multi-length exploration (n=8192 primary, n=16384 if time permits)

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_objective(a):
    """Compute 2*n*max_c / sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (S * S))


def gs_spectral_flatten(a, alpha_schedule, n_iters, S, rng, deadline):
    """
    Gerchberg-Saxton spectral flattening.
    Interpolates |FFT(a)| toward its mean to flatten the autocorrelation.
    """
    n = len(a)
    a_curr = a.copy()
    best_a = a.copy()

    fsize = 1 << (2 * n - 1).bit_length()

        else:

            alpha = alpha_schedule[idx]
        
        # FFT
        A = fft(a_curr, fsize)
        mag = np.abs(A)
        
        # Flatten: interpolate magnitude toward mean
        # Use a stronger flattening for high-magnitude bins
        mu = mag.mean()
        # Soft flattening: reduce peaks more than troughs
        ratio = mag / (mu + 1e-30)

def nesterov_accelerated(a_init, n, S, max_time, rng, top_k=200, step_scale=1.0):

    """

        A = fft(a, fsize_cross)

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break
        
        t0 = time.monotonic()
        remaining_for_beta = max(5, max_time - (time.monotonic() - start) - 3)

            z_s = z_val - z_val.max()
            exp_z = np.exp(z_s)
            a_ = S * exp_z / exp_z.sum()
            a_ = np.maximum(a_, 1e-15)
            a_ = a_ * S / a_.sum()
            c = fft_convolve_self(a_)
            c_max = c.max()
            exp_bc = np.exp(beta * (c - c_max))
            sum_exp_bc = exp_bc.sum()
            return float((c_max + np.log(sum_exp_bc)) / beta)
        
        def jac(z_val):
            z_s = z_val - z_val.max()
            exp_z = np.exp(z_s)
            a_ = S * exp_z / exp_z.sum()
            a_ = np.maximum(a_, 1e-15)
            a_ = a_ * S / a_.sum()
            c = fft_convolve_self(a_)
            c_max = c.max()
            exp_bc = np.exp(beta * (c - c_max))
            sum_exp_bc = exp_bc.sum()
            w = exp_bc / sum_exp_bc
            fsize = 1 << (len(w) + len(z_val) - 2).bit_length()
            W = fft(w, fsize)
            A = fft(a_, fsize)
            grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:len(z_val)]
            dot = np.dot(grad_a, a_)
            grad_z = a_ * (grad_a - dot / S)
            return grad_z

        try:

                    'gtol': 1e-12,
                    'maxls': 50,
                    'iprint': -1
                }
            )
            z = result.x
        except Exception as e:
            print("      L-BFGS error: %s" % str(e))
            break
        
        z = z - z.mean()

        exp_z = np.exp(z_s)
        a = S * exp_z / exp_z.sum()
        a = np.maximum(a, 0.0)
        a = a * S / a.sum()
        
        score = compute_objective(a)
        elapsed = time.monotonic() - t0
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        print("      beta=%5d: score=%.8f (%.1fs)" % (beta, best_score, elapsed))

        for p in range(n_perturb):
            if time.monotonic() > deadline - 20:
                break
            
            noise_scale = 0.01 / (1.0 + 0.5 * p)
            a_pert = global_a + rng.normal(0, noise_scale * S / np.sqrt(n), n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert * S / a_pert.sum()
            
            # Quick L-BFGS
            a_new, score = lbfgs_polish(
                a_pert, n, S, per_perturb, rng,
                beta_schedule=[100, 500, 2000, 10000],
                maxiter_per_beta=200
            )
            update_best(a_new, score)
            if p % 2 == 0:
                print("  Perturb %d (scale=%.4f): %.8f" % (p, noise_scale, score))
    
    # === Verification ===
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\n=== Verification ===")
        print("evaluate_sequence: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  Rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64)
            global_best = np.inf
    else:
        global_a = np.ones(256, dtype=np.float64)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
    
    print("\n=== Final Result ===")
    print("Length: %d" % len(global_a))
    print("Computed score: %.8f" % global_best)
    print("Time used: %.1fs" % (time.monotonic() - (deadline - budget_s + 25)))
    
    return [float(x) for x in global_a]
```
