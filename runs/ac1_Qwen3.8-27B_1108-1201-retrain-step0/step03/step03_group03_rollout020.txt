import numpy as np

from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, beta):

def make_anti_palindromic_init(n, rng, S):
    """Create an anti-palindromic initialization."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = max(0.0, mu - h[i])
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        return np.ones(n) * S / n
    return a * (S / a.sum())

def upsample(a, n_new):
    """Upsample a sequence to new length using linear interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    tt_old = np.linspace(0, 1, n_old)
    tt_new = np.linspace(0, 1, n_new)
    a_new = np.interp(tt_new, tt_old, a)
    return np.maximum(a_new, 0.0)

            x_new = project_simplex(y - step * grad, S)

        # Adaptive L
        c_x = fft_convolve_self(x_new)
        if c_x.max() > f_y * 1.01:
            L *= 1.3
        elif it % 100 == 0:
            L = max(L * 0.97, beta * 1.5)

        # Momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

    return best_local, best_score


def run_cutting_plane(a_init, n, S, max_time, rng, max_cuts=50):
    """Cutting plane method: LP to minimize max of linearized convolution values."""
    from scipy.optimize import linprog

    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    cut_lags = []
    cut_grads = []
    cut_offsets = []

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Find top lags
        top_k = min(max_cuts, 20 + it // 10)
        top_indices = np.argsort(c)[::-1][:top_k]

        # Build LP: minimize t
        # Variables: [a_0, ..., a_{n-1}, t]
        # Constraints:
        #   sum(a) = S  (equality)
        #   a >= 0      (bounds)
        #   for each cut: sum_i g_i * a[i] <= t + offset_i
        #     i.e., sum_i g_i * a[i] - t <= offset_i

        # For each top lag, compute subgradient at current a
        A_ub_rows = []
        b_ub_vals = []

        for lag in top_indices:
            lag = int(lag)
            g = subgradient_at_lag_fast(a, lag)
            # Linearization: (a'*a')[lag] ≈ c[lag] + sum_i g[i]*(a'[i] - a[i])
            # = c[lag] + sum_i g[i]*a'[i] - sum_i g[i]*a[i]
            # We want: c[lag] + sum_i g[i]*a'[i] - sum_i g[i]*a[i] <= t
            # => sum_i g[i]*a'[i] - t <= sum_i g[i]*a[i] - c[lag]
            offset = g.dot(a) - c[lag]
            # Row: [g[0], ..., g[n-1], -1]  <=  offset
            row = np.zeros(n + 1)
            row[:n] = g
            row[n] = -1.0
            A_ub_rows.append(row)
            b_ub_vals.append(offset)

        if not A_ub_rows:
            break

        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)

        # Objective: minimize t => c_objective = [0, 0, ..., 0, 1]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        # Equality: sum(a) = S
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])

        # Bounds: a[i] >= 0, t >= 0
        bounds = [(0, None)] * n + [(0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                a_new = res.x[:n]
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a_new = a_new * (S / s_new)
                    c_new = fft_convolve_self(a_new)
                    if c_new.max() < best_score:
                        best_score = c_new.max()
                        best_local = a_new.copy()
                    a = a_new
            else:
                # LP failed, try a smaller step
                break
        except Exception:
            break

        # Check convergence
        if it > 50 and abs(cur - best_score) < 1e-8:
            break

    return best_local, best_score


def run_coordinate_descent(a_init, n, S, max_time, rng, batch_size=50):
    """Coordinate descent: transfer mass from high-responsibility to low-responsibility elements."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    c = fft_convolve_self(a)

    while time.monotonic() - start < max_time:
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Find top lags
        top_lags = np.argsort(c)[::-1][:5]

        # For each top lag, find elements with high responsibility
        # and transfer mass to elements with low responsibility
        for _ in range(batch_size):
            lag = int(top_lags[np.random.randint(len(top_lags))])
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min >= i_max:
                continue

            indices = np.arange(i_min, i_max + 1)
            partner = lag - indices
            responsibilities = a[partner].copy()

            # Find the element with highest responsibility to reduce
            # and one with low responsibility to increase
            idx_order = np.argsort(responsibilities)[::-1]
            if len(idx_order) < 2:
                continue

            src_local = idx_order[0]
            dst_candidates = idx_order[len(idx_order)//2:]
            if len(dst_candidates) == 0:
                dst_candidates = [idx_order[-1]]
            dst_local = dst_candidates[np.random.randint(len(dst_candidates))]

            src = indices[src_local]
            dst = indices[dst_local]

            # Transfer mass from src to dst
            transfer = min(a[src], 0.01 * S / n) * rng.uniform(0.5, 1.5)
            if transfer < 1e-15:
                continue

            a_old_src = a[src]
            a_old_dst = a[dst]
            a[src] -= transfer
            a[dst] += transfer

            if a[src] < 0:
                a[dst] += a[src]
                a[src] = 0.0

            # Quick check: compute new max at the target lag
            # (a*a)[lag] = sum_i a[i]*a[lag-i]
            new_val = 0.0
            for i in range(max(0, lag - n + 1), min(lag, n - 1) + 1):
                p = lag - i
                if 0 <= p < n:
                    new_val += a[i] * a[p]

            # Also check neighbor lags
            max_neighbor = 0.0
            for dlag in [-2, -1, 0, 1, 2]:
                lk = lag + dlag
                if 0 <= lk < 2 * n - 1:
                    val = 0.0
                    for i in range(max(0, lk - n + 1), min(lk, n - 1) + 1):
                        p = lk - i
                        if 0 <= p < n:
                            val += a[i] * a[p]
                    max_neighbor = max(max_neighbor, val)

            if max_neighbor > cur * 1.005:
                # Revert
                a[src] = a_old_src
                a[dst] = a_old_dst
            else:
                # Accept, update top_lags occasionally
                top_lags = np.argsort(c)[::-1][:5]

        # Update full convolution periodically
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

    return best_local, best_score


def run_simulated_annealing(a_init, n, S, max_time, rng, T_init=1.0, T_min=0.01, cool_rate=0.999):
    """Simulated annealing with structured moves."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    c = fft_convolve_self(a)
    cur = c.max()
    start = time.monotonic()
    best_score = cur
    best_local = a.copy()
    T = T_init

    while time.monotonic() - start < max_time:
        # Choose a random structured move
        move_type = rng.random()
        a_trial = a.copy()

        if move_type < 0.4:
            # Transfer mass between two random elements
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            delta = rng.normal(0, 0.01 * S / n)
            a_trial[i] -= delta
            a_trial[j] += delta
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial * (S / a_trial.sum())
        elif move_type < 0.7:
            # Add Gaussian noise to a random block
            block_size = rng.integers(16, 128)
            start_idx = rng.integers(0, max(1, n - block_size))
            noise = rng.normal(0, 0.005 * S / n, block_size)
            a_trial[start_idx:start_idx + block_size] += noise
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial * (S / a_trial.sum())
        elif move_type < 0.9:
            # Swap two elements
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            a_trial[i], a_trial[j] = a_trial[j], a_trial[i]
        else:
            # Anti-palindromic perturbation
            i = rng.internels(n // 2)
            j = n - 1 - i
            delta = rng.normal(0, 0.01 * S / n)
            a_trial[i] += delta
            a_trial[j] -= delta
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial * (S / a_trial.sum())

        c_trial = fft_convolve_self(a_trial)
        new_val = c_trial.max()

        # Accept/reject
        delta_f = new_val - cur
        if delta_f < 0 or rng.random() < np.exp(-delta_f / max(T, 1e-10)):
            a = a_trial
            c = c_trial
            cur = new_val
            if cur < best_score:
                best_score = cur
                best_local = a.copy()

        T = max(T_min, T * cool_rate)

    return best_local, best_score


def run_anti_palindromic_opt(a_init, n, S, max_time, rng):
    """Optimize under the anti-palindromic constraint a[i] + a[n-1-i] = mu."""
    half = n // 2
    mu = S / half  # Since sum = mu * half (for even n)

    # Initialize first half
    h = np.clip(a_init[:half], 0, mu).copy()
    h = np.maximum(h, 0.0)
    h = np.minimum(h, mu)
    # Adjust to make sum(h) = S/2 (since total sum = 2*sum(h) for even n)
    target_half_sum = S / 2.0
    h_sum = h.sum()
    if h_sum > 1e-15:
        # Scale and clip
        h = h * (target_half_sum / h_sum)
        h = np.clip(h, 0, mu)
        # Iteratively adjust
        for _ in range(10):
            h = np.minimum(h, mu)
            diff = target_half_sum - h.sum()
            if abs(diff) < 1e-10:
                break
            # Distribute difference to elements that aren't at bounds
            free_mask = (h > 1e-10) & (h < mu - 1e-10)
            if np.any(free_mask):
                h[free_mask] += diff / free_mask.sum()
            else:
                break

    start = time.monotonic()
    best_score = np.inf
    best_h = h.copy()
    best_full = None
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Build full sequence
        a = np.zeros(n)
        a[:half] = h
        for i in range(half):
            a[n - 1 - i] = mu - h[i]
        if n % 2 == 1:
            a[half] = mu / 2

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_h = h.copy()
            best_full = a.copy()

        # Compute subgradient for the first half
        # The gradient of max_k (a*a)[k] w.r.t. h[i]
        # a[i] = h[i] for i < half
        # a[n-1-i] = mu - h[i] for i < half
        # So d(a*a)[k]/d(h[i]) = d(a*a)[k]/d(a[i]) * 1 + d(a*a)[k]/d(a[n-1-i]) * (-1)
        # = gradient[k][i] - gradient[k][n-1-i]

        top_lags = np.argsort(c)[::-1][:10]
        grad_h = np.zeros(half)
        for lag in top_lags:
            lag = int(lag)
            # Subgradient w.r.t. full a
            sg_full = subgradient_at_lag_fast(a, lag)
            # Project to h
            gh = sg_full[:half] - sg_full[n-1-half:]
            # Weight by softmax of convolution values
            w = np.exp(50 * (c[lag] - c[top_lags[0]]))
            grad_h += w * gh

        gnorm = np.linalg.norm(grad_h)
        if gnorm < 1e-14:
            break

        # Projected gradient step on the polytope {0 <= h <= mu, sum(h) = S/2}
        lr = 0.01 * S / (n * max(gnorm / (S/n), 0.01))
        h_new = h - lr * grad_h
        # Project to box [0, mu] and simplex
        h_new = np.clip(h_new, 0, mu)
        # Adjust sum
        for _ in range(5):
            h_new = np.clip(h_new, 0, mu)
            diff = target_half_sum - h_new.sum()
            if abs(diff) < 1e-12:
                break
            free = (h_new > 1e-10) & (h_new < mu - 1e-10)
            if np.any(free):
                h_new[free] += diff / free.sum()
            else:
                break

        h = h_new

        # Periodic perturbation
        if it % 500 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, 0.005 * S / n, half)
            h = np.clip(h + noise, 0, mu)
            for _ in range(5):
                h = np.clip(h, 0, mu)
                diff = target_half_sum - h.sum()
                if abs(diff) < 1e-12:
                    break
                free = (h > 1e-10) & (h < mu - 1e-10)
                if np.any(free):
                    h[free] += diff / free.sum()

    if best_full is None:
        best_full = np.zeros(n)
        best_full[:half] = best_h
        for i in range(half):
            best_full[n - 1 - i] = mu - best_h[i]

    return best_full, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (a.sum() ** 2)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best_obj = np.inf
    global_a = None

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    target_n = 8192
    print("Target n=%d, parent available: %s (len=%s)" % (target_n, parent is not None, len(parent) if parent is not None else 0))

    # === Phase 1: Multi-scale initialization ===
    print("=== Phase 1: Multi-scale initialization ===")
    scales = [64, 256, 1024, 4096]
    current_a = None
    scale_times = [10, 30, 60, 120]  # Time budget for each scale

    for idx, n_scale in enumerate(scales):
        if time.monotonic() > deadline - 1500:
            break
        S_scale = np.sqrt(2.0 * n_scale)

        if current_a is None:
            # Initial: mix of anti-palindromic and periodic
            a1 = make_anti_palindromic_init(n_scale, rng, S_scale)
            a2 = make_periodic_init(n_scale, rng, S_scale)
            current_a = (a1 + a2) / 2.0
            current_a = safe_normalize(current_a, n_scale)
        else:
            # Upsample
            current_a = upsample(current_a, n_scale)
            current_a = safe_normalize(current_a, n_scale)

        t0 = time.monotonic()
        max_t = min(scale_times[idx], max(5.0, (deadline - 1500 - time.monotonic())))
        a_opt, score = run_fista_v2(current_a, n_scale, S_scale, max_t, rng,
                                     beta_init=1.0, beta_max=200.0, beta_rate=0.002)
        current_a = safe_normalize(a_opt, n_scale)

        obj = compute_objective(current_a)
        elapsed = time.monotonic() - t0
        print("  Scale n=%d: obj=%.6f, max_conv=%.4f (%.1fs)" % (n_scale, obj, score, elapsed))

    # === Phase 2: Main optimization at target size ===
    print("=== Phase 2: Main optimization at n=%d ===" % target_n)
    if current_a is not None:
        a_main = upsample(current_a, target_n)
    else:
        a_main = make_anti_palindromic_init(target_n, rng, np.sqrt(2.0 * target_n))

    # Also try from parent if available
    if parent is not None and len(parent) >= 64:
        a_parent = upsample(parent, target_n)
        a_parent = safe_normalize(a_parent, target_n)
    else:
        a_parent = None

    S_main = np.sqrt(2.0 * target_n)

    # Run FISTA on main initialization
    remaining_for_main = max(100.0, (deadline - time.monotonic()) - 600.0)
    seg1 = remaining_for_main * 0.40
    a_f1, s_f1 = run_fista_v2(a_main, target_n, S_main, seg1, rng,
                                beta_init=0.5, beta_max=400.0, beta_rate=0.0008)
    a_f1 = safe_normalize(a_f1, target_n)
    obj_f1 = compute_objective(a_f1)
    if obj_f1 < global_best_obj:
        global_best_obj = obj_f1
        global_a = a_f1.copy()
    print("  FISTA main: obj=%.6f" % obj_f1)

    # Run FISTA on parent (if different)
    if a_parent is not None and remaining_for_main > 100:
        seg2 = remaining_for_main * 0.30
        a_f2, s_f2 = run_fista_v2(a_parent, target_n, S_main, seg2, rng,
                                    beta_init=0.5, beta_max=400.0, beta_rate=0.0008)
        a_f2 = safe_normalize(a_f2, target_n)
        obj_f2 = compute_objective(a_f2)
        if obj_f2 < global_best_obj:
            global_best_obj = obj_f2
            global_a = a_f2.copy()
        print("  FISTA parent: obj=%.6f" % obj_f2)

    # Anti-palindromic optimization
    remaining_anti = max(30.0, (deadline - time.monotonic()) - 400.0)
    if remaining_anti > 30:
        a_anti_init = make_anti_palindromic_init(target_n, rng, S_main)
        # Blend with best found so far
        if global_a is not None:
            a_anti_init = 0.5 * a_anti_init + 0.5 * global_a
            # Enforce anti-palindromic
            half = target_n // 2
            mu = S_main / half
            h = (a_anti_init[:half] + (mu - a_anti_init[n-1-half:])) / 2.0
            h = np.clip(h, 0, mu)
            a_anti_init = np.zeros(target_n)
            a_anti_init[:half] = h
            for i in range(half):
                a_anti_init[n - 1 - i] = mu - h[i]
            a_anti_init = safe_normalize(a_anti_init, target_n)

        seg_anti = min(remaining_anti, 200.0)
        a_ant, s_ant = run_anti_palindromic_opt(a_anti_init, target_n, S_main, seg_anti, rng)
        a_ant = safe_normalize(a_ant, target_n)
        obj_ant = compute_objective(a_ant)
        if obj_ant < global_best_obj:
            global_best_obj = obj_ant
            global_a = a_ant.copy()
        print("  Anti-palindromic: obj=%.6f" % obj_ant)

    # === Phase 3: Polishing ===
    print("=== Phase 3: Polishing ===")
    if global_a is None:
        global_a = a_f1.copy()
        global_best_obj = obj_f1

    n = len(global_a)
    S = np.sqrt(2.0 * n)
    remaining = deadline - time.monotonic()
    print("  Remaining time: %.0fs" % remaining)

    # 3a: Continued FISTA with high beta (exploitation)
    seg3a = max(10.0, remaining * 0.25)
    a_p1, s_p1 = run_fista_v2(global_a, n, S, seg3a, rng,
                                beta_init=50.0, beta_max=800.0, beta_rate=0.001)
    a_p1 = safe_normalize(a_p1, n)
    obj_p1 = compute_objective(a_p1)
    if obj_p1 < global_best_obj:
        global_best_obj = obj_p1
        global_a = a_p1.copy()
    print("  Polish FISTA: obj=%.6f" % obj_p1)

    # 3b: Coordinate descent
    remaining2 = deadline - time.monotonic()
    if remaining2 > 30:
        seg3b = min(remaining2 * 0.25, 200.0)
        a_p2, s_p2 = run_coordinate_descent(global_a, n, S, seg3b, rng, batch_size=30)
        a_p2 = safe_normalize(a_p2, n)
        obj_p2 = compute_objective(a_p2)
        if obj_p2 < global_best_obj:
            global_best_obj = obj_p2
            global_a = a_p2.copy()
        print("  Coord descent: obj=%.6f" % obj_p2)

    # 3c: Simulated annealing
    remaining3 = deadline - time.monotonic()
    if remaining3 > 30:
        seg3c = min(remaining3 * 0.20, 150.0)
        a_p3, s_p3 = run_simulated_annealing(global_a, n, S, seg3c, rng,
                                              T_init=0.01, T_min=0.001, cool_rate=0.9995)
        a_p3 = safe_normalize(a_p3, n)
        obj_p3 = compute_objective(a_p3)
        if obj_p3 < global_best_obj:
            global_best_obj = obj_p3
            global_a = a_p3.copy()
        print("  Sim anneal: obj=%.6f" % obj_p3)

    # 3d: Cutting plane
    remaining4 = deadline - time.monotonic()
    if remaining4 > 60:
        seg3d = min(remaining4 * 0.30, 200.0)
        a_p4, s_p4 = run_cutting_plane(global_a, n, S, seg3d, rng, max_cuts=30)
        a_p4 = safe_normalize(a_p4, n)
        obj_p4 = compute_objective(a_p4)
        if obj_p4 < global_best_obj:
            global_best_obj = obj_p4
            global_a = a_p4.copy()
        print("  Cutting plane: obj=%.6f" % obj_p4)

    # 3e: Final FISTA polish
    remaining5 = deadline - time.monotonic()
    if remaining5 > 20:
        a_p5, s_p5 = run_fista_v2(global_a, n, S, min(remaining5 * 0.9, 100.0), rng,
                                    beta_init=200.0, beta_max=1000.0, beta_rate=0.002)
        a_p5 = safe_normalize(a_p5, n)
        obj_p5 = compute_objective(a_p5)
        if obj_p5 < global_best_obj:
            global_best_obj = obj_p5
            global_a = a_p5.copy()
        print("  Final FISTA: obj=%.6f" % obj_p5)

    # === Phase 4: Try a slightly different length ===
    remaining6 = deadline - time.monotonic()
    if remaining6 > 100:
        n_alt = 16384
        print("=== Trying n=%d ===" % n_alt)
        if global_a is not None:
            a_alt = upsample(global_a, n_alt)
        else:
            a_alt = make_anti_palindromic_init(n_alt, rng, np.sqrt(2.0 * n_alt))
        a_alt = safe_normalize(a_alt, n_alt)
        seg_alt = min(remaining6 * 0.5, 200.0)
        a_alt_opt, _ = run_fista_v2(a_alt, n_alt, np.sqrt(2.0 * n_alt), seg_alt, rng,
                                     beta_init=10.0, beta_max=300.0, beta_rate=0.001)
        a_alt_opt = safe_normalize(a_alt_opt, n_alt)
        obj_alt = compute_objective(a_alt_opt)
        if obj_alt < global_best_obj:
            global_best_obj = obj_alt
            global_a = a_alt_opt.copy()
            print("  Alt length n=%d: obj=%.6f (NEW BEST)" % (n_alt, obj_alt))
        else:
            print("  Alt length n=%d: obj=%.6f (no improvement)" % (n_alt, obj_alt))

    # === Final verification ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Verification: eval=%.6f, internal_obj=%.6f, len=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best_obj, len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: evaluate_sequence rejected, using fallback")
        global_a = np.ones(2048, dtype=np.float64) * 0.1
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback eval: %.6f" % (ver if np.isfinite(ver) else float('inf')))

    print("=== FINAL: %.6f (length=%d) ===" % (ver if np.isfinite(ver) else float('inf'), len(final_list)))
    return final_list
