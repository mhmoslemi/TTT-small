def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    # Water-filling / bisection approach
    lo = v.min() if n > 0 else 0.0
    hi = v.max() if n > 0 else 0.0
    for _ in range(60):
        mid = (lo + hi) / 2
        clipped = np.maximum(v - mid, 0.0)
        s = clipped.sum()
        if abs(s - S) < 1e-15:
            break
        if s > S:
            lo = mid
        else:
            hi = mid
    result = np.maximum(v - mid, 0.0)
    # Fix rounding
    diff = S - result.sum()
    if abs(diff) > 1e-15:
        # Distribute the diff among non-zero elements
        mask = result > 0
        result[mask] += diff / mask.sum()
    return np.maximum(result, 0.0)
```

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} in O(n) using the standard algorithm."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cumsum = np.cumsum(u)
    rho_candidates = np.arange(1, n+1, dtype=np.float64)
    theta_candidates = (cumsum - S) / rho_candidates
    valid = theta_candidates < u  # Note: < not <=
    if not np.any(valid):
        # All in the support
        theta = (v.sum() - S) / n
        return np.maximum(v - theta, 0.0)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)
```
