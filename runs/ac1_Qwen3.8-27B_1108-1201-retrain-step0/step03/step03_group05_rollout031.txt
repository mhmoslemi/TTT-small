def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT. Returns array of length n+m-1."""
    n, m = len(a), len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (S * S)

    if s < 1e-15:

        res = linprog(
            c=-obj,
            A_ub=A_mat,
            b_ub=np.full(len(active), M),
            bounds=[(0, None)] * n,
            method='highs'
        )
        
        if not res.success:
            break
        
        g_solution = res.x
        
        # Check if all constraints are satisfied
        b_test = fft_convolve(a, g_solution)
        violations = b_test > M * (1 + 1e-8)
        
        if not violations.any():
            break
        
        # Add violated constraints (up to 100 at a time)
        violated_idx = np.where(violations)[0]
        # Add the worst violations
        violation_amounts = b_test[violated_idx] - M
        worst = violated_idx[np.argsort(violation_amounts)[::-1][:100]]
        for k in worst:
            if k not in active:
                active.append(int(k))
        
        active = list(set(active))
    
    return g_solution, len(active)


def lp_local_search(a, n_directions=200, t_init=0.1, t_min=0.01, timeout=600):
    """
    LP-based local search.
    Iteratively finds the best direction g and takes a step.
    """
    n = len(a)
    S = a.sum()
    a = normalize(a, S).copy()
    
    start_time = time.monotonic()
    best_a = a.copy()
    best_obj = compute_objective(a)
    t = t_init
    
    for it in range(n_directions):
        if time.monotonic() - start_time > timeout:
            break
        
        b = fft_convolve_self(a)
        M = b.max()
        
        # Solve LP for direction
        g, n_active = solve_lp_direction(a, M, n_constraints=300, max_iters=10, timeout=15)
        
        if g is None or g.sum() < 1e-10:
            t = max(t * 0.5, t_min)
            continue
        
        # Scale g to have same sum as a
        g = g * (S / g.sum())
        
        # Try the step
        a_new = (1 - t) * a + t * g
        a_new = normalize(a_new, S)
        
        obj_new = compute_objective(a_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            # Increase t slightly (aggressive)
            t = min(t * 1.2, 0.5)
        else:
            # Decrease t
            t = max(t * 0.7, t_min)
            
            # If t is too small, try a different approach
            if t <= t_min * 1.1:
                # Try a smaller step in a random direction
                a_perturb = a + np.random.normal(0, 0.01 * S / np.sqrt(n), n)
                a_perturb = normalize(a_perturb, S)
                obj_perturb = compute_objective(a_perturb)
                if obj_perturb < best_obj:
                    best_obj = obj_perturb
                    best_a = a_perturb.copy()
                    a = a_perturb.copy()
                    t = t_init
        
        if it % 20 == 0:
            print("    LP iter %d: obj=%.8f, t=%.3f, M=%.4f" % (it, best_obj, t, M))
    
    return best_a, best_obj


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z (softmax params)."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp
    
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta
    
    w = exp_bc / sum_exp_bc
    # grad_a = 2 * conv(w, a) (cross-correlation)
    grad_a = 2.0 * fft_convolve(w, a)[:n]
    
    # grad_z[k] = a[k] * (grad_a[k] - dot(grad_a, a)/S)
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)
    
    return obj, grad_z


def lbfgs_polish(a_init, S, max_time, beta_start=100, beta_end=2000, n_betas=5):
    """Polish using L-BFGS-B on softmax parameterization with continuation."""
    n = len(a_init)
    a = normalize(a_init, S)
    
    if a.max() < 1e-15:
        z = np.zeros(n)
    else:
        z = np.log(np.maximum(a, 1e-30) / S)
        z = z - z.mean()
    
    betas = np.geomspace(beta_start, beta_end, n_betas)
    start_time = time.monotonic()
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    
    for beta in betas:
        if time.monotonic() - start_time > max_time - 5:
            break
        
        def fun(z_val):
            return softmax_obj_grad(z_val, beta, S)[0]
        
        def jac(z_val):
            return softmax_obj_grad(z_val, beta, S)[1]
        
        maxiter = max(20, int(30 * max_time / (n_betas * n / 8192)))
        
        try:
            result = scipy_minimize(
                fun, z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'maxfun': maxiter * 2,
                         'ftol': 1e-14, 'gtol': 1e-9, 'maxls': 30, 'iprint': -1}
            )
            z = result.x
            z = z - z.mean()
        except Exception:
            pass
        
        # Convert back
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = S * exp_z / exp_z.sum()
        a = np.maximum(a, 0.0)
        a = normalize(a, S)
        
        c = fft_convolve_self(a)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def nesterov_subgrad(a_init, S, max_time, top_k=64, restart_every=2000):
    """Nesterov-accelerated top-k subgradient method."""
    n = len(a_init)
    a = normalize(a_init, S).copy()
    y = a.copy()
    t = 1.0
    start_time = time.monotonic()
    best_a = a.copy()
    best_score = compute_objective(a)
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        c = fft_convolve_self(y)
        cur = c.max()
        S_cur = y.sum()
        obj = 2.0 * n * cur / (S_cur * S_cur) if S_cur > 0.01 else np.inf
        if obj < best_score:
            best_score = obj
            best_a = y.copy()
        
        # Top-k subgradient
        k_eff = min(top_k + it // 500, n)
        top_indices = np.argsort(c)[::-1][:k_eff]
        top_vals = np.maximum(c[top_indices], 1e-15)
        weights = top_vals / top_vals.sum()
        
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]
        
        grad = 2.0 * fft_convolve(w_full, a)[:n]
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-15:
            break
        
        step = 2.0 * S / (np.sqrt(n) * (1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % restart_every == 0:
            # Restart with perturbation
            y = project_simplex(y + np.random.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_a, best_score
