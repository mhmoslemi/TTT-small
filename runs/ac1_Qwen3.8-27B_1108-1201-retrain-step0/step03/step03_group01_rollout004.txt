"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Improved spectral shaping with flat-spectrum target
2. FISTA with adaptive log-sum-exp smoothing
3. L-BFGS-B with softmax parametrization for final polish
4. Coordinate descent for ultra-fine tuning
5. Multi-length search (n=4096, 8192, 16384)
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:size]


def conv_gradient(a, w):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    grad[m] = 2 * sum_k w[k] * a[k-m] (for valid k-m)
    This is 2 * cross_correlation(w, a)[:n]
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(np.conj(W) * A))[:n]
    return 2.0 * corr


def lse_obj_grad(a, beta, S):
    """
    Log-sum-exp smoothed objective and gradient.
    f(a) = (1/beta) * log(sum_k exp(beta * C[k]))  where C = conv(a,a)
    grad = (2/beta) * sum_k softmax(beta*C)[k] * (correlation of a with e_k)
         = 2 * conv_gradient(a, softmax(beta*C))
    (The 1/beta factor is absorbed since we only need gradient direction)
    Actually: f = (1/beta)*log(sum exp(beta*C))
    df/da = (1/beta) * sum_k (beta*exp(beta*C[k])/sum(exp(beta*C))) * dC[k]/da
          = sum_k softmax_k * dC[k]/da
    So grad = conv_gradient(a, softmax(beta*C))
    """
    n = len(a)
    c = compute_conv(a)
    c_max = c.max()
    shifted = np.clip(beta * (c - c_max), -500, 500)
    exp_c = np.exp(shifted)
    total = exp_c.sum()
    if total < 1e-300:
        return c_max, np.zeros(n)
    w = exp_c / total
    obj = c_max + np.log(total) / beta
    grad = conv_gradient(a, w)
    return obj, grad


def fista_lse(a_init, n, S, max_time, rng, beta_start=10.0, beta_end=2000.0,
              n_beta_steps=15, ftol=1e-14):
    """
    FISTA with log-sum-exp smoothing, increasing beta.
    """
    a = np.clip(a_init.copy(), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_cmax = compute_conv(a).max()
    start_time = time.monotonic()

    for bi in range(n_beta_steps):
        if time.monotonic() - start_time > max_time:
            break
        # Log-spaced beta
        frac = bi / max(1, n_beta_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, 5000.0)

        # FISTA inner loop
        x = a.copy()
        y = a.copy()
        t = 1.0
        L = beta  # Lipschitz constant (rough)
        it = 0
        beta_start_time = time.monotonic()
        max_inner_time = max_time * max(0.1, 0.6 / (bi + 2))

        while time.monotonic() - beta_start_time < max_inner_time:
            it += 1
            # Gradient step
            _, grad = lse_obj_grad(y, beta, S)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            step_size = 1.0 / (L * 2.0)
            x_new = project_simplex(y - step_size * grad, S)

            # Check improvement
            if it % 10 == 0:
                c_new = compute_conv(x_new)
                if c_new.max() < best_cmax:
                    best_cmax = c_new.max()
                    best_a = x_new.copy()
                    print(f"    FISTA beta={beta:.1f} it={it}: max_conv={best_cmax:.8f}")

            # Nesterov update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = x_new + ((t - 1.0) / t_new) * (x_new - x)
            y = np.maximum(y, 0.0)
            # Renormalize
            ys = y.sum()
            if ys > 1e-15:
                y = y * (S / ys)
            x = x_new
            t = t_new

            # Adapt L
            L = min(L * 1.02, beta * 10)

        # Update a to best found so far
        a = best_a.copy()
        a = a * (S / a.sum())

        # Noise to escape local minima
        if bi > 3 and rng.random() < 0.4:
            noise_scale = 0.005 * S / np.sqrt(n) * (1.0 - frac)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best_cmax


def spectral_shaping_improved(a_init, n, S, N_fft, max_time, rng,
                               n_iter_max=5000, target_score=1.45):
    """
    Improved spectral shaping: drive the spectrum toward flat while
    maintaining non-negativity in time domain.
    """
    a = np.clip(a_init.copy(), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_cmax = compute_conv(a).max()
    start_time = time.monotonic()

    m = np.arange(N_fft)

    for it in range(n_iter_max):
        if time.monotonic() - start_time > max_time:
            break

        a_pad = np.zeros(N_fft)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = c.max()

        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
            if it % 500 == 0:
                print(f"    SS it={it}: max_conv={best_cmax:.8f}")

        score = 2.0 * n * best_cmax / S ** 2
        if score <= target_score:
            break

        # Compute target: flat spectrum
        # P_target = S^2 / N_fft (uniform)
        # But we want to reduce peaks in autocorrelation
        # Strategy: reduce P at frequencies that contribute to autocorrelation peaks

        # Find top autocorrelation lags
        top_k = min(20, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # Effective cosine
        cos_eff = np.zeros(N_fft)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N_fft)

        # Adaptive alpha
        excess = max(0.0, c_max - S * S / (2.0 * n)) / max(c_max, 1e-10)
        alpha = 0.15 * (0.3 + 0.7 * excess)

        # Modify spectrum
        P_new = np.maximum(P * (1.0 - alpha * cos_eff), 1e-20)

        # Also pull toward flat spectrum (spectral regularization)
        P_flat = S * S / N_fft
        blend = 0.02
        P_new = (1.0 - blend) * P_new + blend * P_flat

        # Preserve phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]

        # Project to simplex
        a = project_simplex(a_new_full, S)

        # Occasional noise
        if it % 500 == 250 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

    return best_a, best_cmax


def lbfgs_softmax(a_init, n, S, max_time, beta):
    """
    L-BFGS-B optimization using softmax parametrization.
    a = S * softmax(x), optimize over x (unconstrained).
    """
    a0 = np.clip(a_init.copy(), 1e-10, 1000)
    x0 = np.log(a0 / a0.sum())  # Initial log-softmax parameters

    best_a = a_init.copy()
    best_cmax = compute_conv(a_init).max()
    start_time = time.monotonic()

    def obj_grad(x):
        # Compute a from softmax
        x_shifted = x - x.max()
        ex = np.exp(x_shifted)
        a = S * ex / ex.sum()

        # Compute gradient of smoothed objective w.r.t. a
        c = compute_conv(a)
        c_max = c.max()
        shifted = np.clip(beta * (c - c_max), -500, 500)
        exp_c = np.exp(shifted)
        total = exp_c.sum()
        if total < 1e-300:
            return c_max, np.zeros(n)
        w = exp_c / total
        obj = c_max + np.log(total) / beta

        # Gradient w.r.t. a
        grad_a = conv_gradient(a, w)

        # Chain rule: da[i]/dx[j] = a[i]*(delta_ij - a[j]/S)
        # df/dx[j] = grad_a[j]*a[j] - a[j]/S * sum_i grad_a[i]*a[i]
        dot = np.dot(grad_a, a) / S
        grad_x = grad_a * a - a * dot

        return obj, grad_x

    # Run L-BFGS-B
    n_iter_budget = int(max_time / 0.05)  # ~50ms per iteration estimate
    n_iter = min(n_iter_budget, 1000)

    try:
        res = minimize(obj_grad, x0, method='L-BFGS-B',
                       options={'maxiter': n_iter, 'ftol': 1e-15,
                                'gtol': 1e-10, 'maxfun': 5000})
        x_opt = res.x
        x_shifted = x_opt - x_opt.max()
        ex = np.exp(x_shifted)
        a_opt = S * ex / ex.sum()
        a_opt = np.clip(a_opt, 0, 1000)
        c_opt = compute_conv(a_opt)
        if c_opt.max() < best_cmax:
            best_cmax = c_opt.max()
            best_a = a_opt.copy()
            print(f"    L-BFGS-B: max_conv={best_cmax:.8f}")
    except Exception as e:
        print(f"    L-BFGS-B error: {e}")

    # Check time
    elapsed = time.monotonic() - start_time
    if elapsed < max_time * 0.7:
        # Try again with higher beta
        try:
            res2 = minimize(obj_grad, x_opt, method='L-BFGS-B',
                           options={'maxiter': 500, 'ftol': 1e-15,
                                    'gtol': 1e-10, 'maxfun': 3000})
            # Monkey-patch beta for second run - need to use closure-free approach
        except Exception:
            pass

    x_shifted = x0 - x0.max()
    # Use the best x found
    a_final = np.clip(a_init.copy(), 1e-10, 1000)
    return best_a, best_cmax


def coordinate_descent_polish(a_init, n, S, max_time, rng):
    """
    Coordinate descent: optimize small groups of coordinates at a time.
    For each coordinate (or small group), find the value that minimizes the peak.
    """
    a = np.clip(a_init.copy(), 0, 1000)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_a = a.copy()
    best_cmax = compute_conv(a).max()
    start_time = time.monotonic()
    rng_local = np.random.default_rng(42)
    order = rng_local.permutation(n)
    pos = 0

    group_size = max(1, n // 2000)  # Process groups of ~group_size elements

    while time.monotonic() - start_time < max_time:
        i = order[pos % n]
        pos += 1

        if pos % (n * 5) == 0:
            # Re-randomize order
            order = rng_local.permutation(n)

        # Try perturbing a[i]
        a_old = a[i]
        # Try a few candidate values
        candidates = [a_old * 0.5, a_old * 0.8, a_old * 1.2, a_old * 1.5,
                      a_old + S / n * 0.1, a_old - S / n * 0.1, S / n]
        candidates = [max(0, c) for c in candidates]

        best_val = a_old
        best_local = best_cmax

        for cv in candidates:
            a[i] = cv
            # Renormalize
            a = a * (S / a.sum())
            c = compute_conv(a)
            if c.max() < best_local:
                best_local = c.max()
                best_val = cv
            a[i] = a_old  # Restore

        if best_val != a_old:
            a[i] = best_val
            a = a * (S / a.sum())
            c = compute_conv(a)
            if c.max() < best_cmax:
                best_cmax = c.max()
                best_a = a.copy()

        # Every 5000 steps, check global improvement
        if pos % 5000 == 0 and pos % n == 0:
            print(f"    CD it={pos//n}: max_conv={best_cmax:.8f}")

    return best_a, best_cmax


def diverse_init(n, rng, S):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Random
    inits.append(("rand", (rng.random(n) + 0.05) * S / (rng.random(n) + 0.05).sum() * np.abs(rng.random(n) + 0.05).sum() / n * n / np.abs(rng.random(n) + 0.05).sum()))
    # Simpler: just normalize
    a = rng.random(n) + 0.05
    inits[0] = ("rand", a * S / a.sum())

    # Raised cosine
    a = np.sin(np.pi * tt) ** 2 + 0.01
    inits.append(("cosine", a * S / a.sum()))

    # Triangle
    a = np.sin(np.pi * tt) + 0.01
    inits.append(("triangle", a * S / a.sum()))

    # Multi-sine
    a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt) + 0.1 * np.sin(2 * np.pi * 13 * tt)
    a = np.maximum(a, 0.01)
    inits.append(("multi_sin", a * S / a.sum()))

    # Chirp
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (3 * tt + 0.3 * tt ** 2)) + 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    inits.append(("chirp", a * S / a.sum()))

    # Bell
    a = np.exp(-2.0 * (tt - 0.5) ** 2) + 0.05
    inits.append(("bell", a * S / a.sum()))

    # Two-bump
    a = np.exp(-5.0 * (tt - 0.3) ** 2) + np.exp(-5.0 * (tt - 0.7) ** 2) + 0.05
    inits.append(("two_bump", a * S / a.sum()))

    # Flat with noise
    a = 1.0 + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.01)
    inits.append(("flat_noise", a * S / a.sum()))

    return inits


def safe_score(a, n):
    """Compute the actual objective score."""
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = compute_conv(a)
    return 2.0 * n * c.max() / (s ** 2)


def optimize_for_length(n, S, N_fft, budget, rng, start_a=None):
    """Full optimization pipeline for a given length."""
    print(f"  Optimizing n={n}, S={S:.4f}, budget={budget:.0f}s")
    t_start = time.monotonic()

    best_a = None
    best_score = np.inf

    # Gather initializations
    inits = []
    if start_a is not None and len(start_a) == n:
        a0 = np.clip(start_a, 0, 1000).astype(np.float64)
        s0 = a0.sum()
        if s0 > 0.01:
            a0 = a0 * S / s0
            inits.append(("start", a0))

    for name, arr in diverse_init(n, rng, S):
        inits.append((name, arr))

    n_inits = len(inits)
    per_init = budget / (n_inits + 2)  # +2 for polish phases
    print(f"  {n_inits} initializations, ~{per_init:.1f}s each")

    # Phase 1: Spectral shaping + FISTA on each init
    for name, arr in inits:
        elapsed = time.monotonic() - t_start
        remaining = budget - elapsed - 60  # Reserve 60s for polish
        if remaining <= 10:
            break

        time_budget = min(per_init, remaining)

        # Spectral shaping
        ss_time = time_budget * 0.4
        a_ss, ss_score = spectral_shaping_improved(arr, n, S, N_fft, ss_time, rng)
        sc = safe_score(a_ss, n)
        if sc < best_score:
            best_score = sc
            best_a = a_ss.copy()
            print(f"    [{name}] SS: score={sc:.6f}")

        # FISTA on best so far
        fista_time = time_budget * 0.6
        if fista_time > 5:
            a_f, f_score = fista_lse(a_ss, n, S, fista_time, rng,
                                     beta_start=5.0, beta_end=1000.0, n_beta_steps=10)
            sc = safe_score(a_f, n)
            if sc < best_score:
                best_score = sc
                best_a = a_f.copy()
                print(f"    [{name}] FISTA: score={sc:.6f}")

    # Phase 2: Focused optimization
    if best_a is None:
        best_a = np.ones(n) * S / n
        best_score = 2.0

    print(f"  After Phase 1: score={best_score:.6f}")
    t_phase2 = time.monotonic()
    phase2_budget = (t_start + budget) - t_phase2 - 30  # Reserve 30s for final polish

    # Long FISTA run
    if phase2_budget > 20:
        a_f2, _ = fista_lse(best_a, n, S, phase2_budget * 0.4, rng,
                            beta_start=50.0, beta_end=3000.0, n_beta_steps=10)
        sc = safe_score(a_f2, n)
        if sc < best_score:
            best_score = sc
            best_a = a_f2.copy()
            print(f"  Phase 2 FISTA: score={sc:.6f}")

    # Spectral shaping with fine params
    rem = (t_start + budget) - time.monotonic() - 20
    if rem > 10:
        a_s2, _ = spectral_shaping_improved(best_a, n, S, N_fft, min(rem, 120), rng,
                                            target_score=best_score * 0.999)
        sc = safe_score(a_s2, n)
        if sc < best_score:
            best_score = sc
            best_a = a_s2.copy()
            print(f"  Phase 2 SS fine: score={sc:.6f}")

    # Phase 3: L-BFGS-B polish
    rem = (t_start + budget) - time.monotonic() - 10
    if rem > 10:
        a_lb, _ = lbfgs_softmax(best_a, n, S, min(rem, 120), beta=500.0)
        sc = safe_score(a_lb, n)
        if sc < best_score:
            best_score = sc
            best_a = a_lb.copy()
            print(f"  L-BFGS-B: score={sc:.6f}")

    # Phase 4: Final FISTA with very high beta
    rem = (t_start + budget) - time.monotonic()
    if rem > 10:
        a_f3, _ = fista_lse(best_a, n, S, rem * 0.6, rng,
                            beta_start=500.0, beta_end=5000.0, n_beta_steps=5)
        sc = safe_score(a_f3, n)
        if sc < best_score:
            best_score = sc
            best_a = a_f3.copy()
            print(f"  Final FISTA: score={sc:.6f}")

    # Phase 5: Coordinate descent
    rem = (t_start + budget) - time.monotonic()
    if rem > 15:
        a_cd, _ = coordinate_descent_polish(best_a, n, S, rem, rng)
        sc = safe_score(a_cd, n)
        if sc < best_score:
            best_score = sc
            best_a = a_cd.copy()
            print(f"  CD polish: score={sc:.6f}")

    print(f"  Done n={n}: best score={best_score:.6f}")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for non-negative sequence minimizing evaluate_sequence.
    Multi-length search with spectral shaping + FISTA + L-BFGS-B + CD.
    """
    global height_sequence_1
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_global_start = time.monotonic()
    deadline = t_global_start + max(0.0, budget_s - 30.0)

    overall_best_a = None
    overall_best_score = np.inf

    # Get initial sequence if available
    given = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if h.size >= 64 and np.sum(np.clip(h, 0, 1000)) > 0.01:
            given = np.clip(h, 0, 1000)
    except Exception:
        pass

    # Determine lengths to try
    if given is not None:
        n_given = len(given)
        lengths = [n_given]
        # Add nearby lengths
        for delta in [-2048, 2048, -4096, 4096]:
            nl = n_given + delta
            if nl >= 1024 and nl <= 32768:
                lengths.append(nl)
        # Also try powers of 2
        for n_try in [4096, 8192, 16384]:
            if n_try not in lengths and 1024 <= n_try <= 32768:
                lengths.append(n_try)
    else:
        lengths = [4096, 8192, 16384]

    # Deduplicate and sort
    lengths = sorted(set(lengths))

    # Allocate time: give more time to the given length if available
    n_lengths = len(lengths)
    if given is not None:
        # Give 50% to given length, rest split
        time_alloc = {}
        for i, nl in enumerate(lengths):
            if nl == n_given:
                time_alloc[nl] = (deadline - time.monotonic()) * 0.5
            else:
                time_alloc[nl] = (deadline - time.monotonic()) * 0.5 / (n_lengths - 1)
    else:
        # Equal split with more time for middle length
        weights = [1.0] * n_lengths
        # Give more to 8192 if present
        for i, nl in enumerate(lengths):
            if nl == 8192:
                weights[i] = 1.5
        total_w = sum(weights)
        total_time = deadline - time.monotonic()
        time_alloc = {nl: total_time * w / total_w for nl, w in zip(lengths, weights)}

    # Run optimization for each length
    for nl in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        alloc = min(time_alloc[nl], remaining - 30)
        if alloc < 30:
            continue

        S = np.sqrt(2.0 * float(nl))
        N_fft = 1 << (2 * nl - 1).bit_length()

        # Pass the given sequence if it matches this length
        start_a = None
        if given is not None and len(given) == nl:
            start_a = given

        try:
            a_best, score_best = optimize_for_length(nl, S, N_fft, alloc, rng, start_a)
            if score_best < overall_best_score:
                overall_best_score = score_best
                overall_best_a = a_best.copy()
                print(f"  *** NEW GLOBAL BEST: n={nl}, score={score_best:.6f} ***")
        except Exception as e:
            print(f"  Error for n={nl}: {e}")

    # Final verification
    if overall_best_a is None:
        # Fallback
        overall_best_a = np.ones(256, dtype=np.float64)
        overall_best_score = 2.0
    else:
        overall_best_a = np.clip(overall_best_a, 0.0, 1000.0)
        if overall_best_a.sum() < 0.01:
            overall_best_a = np.ones(256, dtype=np.float64)

    # Verify with evaluate_sequence
    final_list = [float(x) for x in overall_best_a]
    try:
        ver = evaluate_sequence(final_list)
        print(f"Verification: {ver:.6f}" if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            overall_best_a = np.ones(256, dtype=np.float64)
            final_list = [float(x) for x in overall_best_a]
    except Exception as e:
        print(f"  Verification error: {e}")

    print(f"Returning sequence of length {len(final_list)}, score={overall_best_score:.6f}")
    return final_list
