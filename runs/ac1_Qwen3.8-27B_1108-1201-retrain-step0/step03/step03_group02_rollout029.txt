import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_gradient(a, c, method='soft', beta=100, top_k=64):
    """
    Compute the subgradient of max(conv) w.r.t. a.
    Uses soft-max (log-sum-exp) or top-k approximation.
    """
    n = len(a)
    if method == 'soft':
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            return np.zeros(n)
        w = exp_bc / total
    else:  # top-k
        k = min(top_k, len(c))
        top_indices = np.argsort(c)[::-1][:k]
        w = np.zeros(len(c))
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            return np.zeros(n)
        for i, idx in enumerate(top_indices):
            w[idx] = top_vals[i] / total
    
    # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m] (for valid k, m)
    # This is 2 * cross-correlation of w and a

    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

Algorithm:

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_soft(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad


def compute_grad_toppk(a, c, k):
    """Gradient using top-k active lags (hard subgradient)."""
    n = len(a)
    k = min(k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(len(c))
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w_full, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad


def spectral_smooth(a, n_smooth):
    """Apply spectral smoothing: low-pass filter in frequency domain, then re-project."""
    n = len(a)
    fsize = 1 << (n - 1).bit_length()
    A = fft(a, fsize)
    # Smooth the spectrum (reduce high-frequency content)
    freqs = np.fft.fftfreq(fsize)
    cutoff = n_smooth / float(fsize)
    mask = np.ones(fsize)
    mask[np.abs(freqs) > cutoff] = 0.3
    A_smooth = A * mask
    a_smooth = np.real(ifft(A_smooth))[:n]
    return a_smooth


def objective_value(a):
    """Compute 2n*max(conv)/sum^2 for the given sequence."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def run_fista_smooth(a_init, n, S, max_time, rng, beta_start=5.0, beta_end=100.0,
                     noise_scale=0.003, use_spectral=True):
    """
    FISTA with log-sum-exp smoothing and adaptive beta.
    Minimizes the smoothed max-convolution objective.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = objective_value(a)
    best_local = a.copy()
    it = 0
    step = 1.0
    stall_count = 0
    best_at_stall = best_score

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: linearly interpolate from beta_start to beta_end
        frac = min(1.0, 3.0 * (time.monotonic() - start) / max_time)
        beta = beta_start + (beta_end - beta_start) * (frac ** 0.7)

        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_score:
            best_score = cur_obj
            best_local = y.copy()
            stall_count = 0
            best_at_stall = best_score
        else:
            stall_count += 1

        # Compute gradient
        grad = compute_grad_soft(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = S / (beta * gnorm * (1.0 + 0.001 * it))
        step = min(step, S / (2.0 * np.sqrt(n)))
        step = max(step, 1e-10 * S)

        x_new = project_simplex(y - step * grad, S)

        # Spectral smoothing every 500 iterations
        if use_spectral and it % 500 == 0:
            a_smoothed = spectral_smooth(x_new, max(4, n // 64))
            a_smoothed = project_simplex(a_smoothed, S)
            obj_smooth = objective_value(a_smoothed)
            if obj_smooth < best_score:
                best_score = obj_smooth
                best_local = a_smoothed.copy()
                x_new = a_smoothed.copy()

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise injection to escape local minima
        if stall_count > 1000 and rng.random() < 0.15:
            noise_std = noise_scale * S / np.sqrt(n)
            noise = rng.normal(0, noise_std, n)
            y = project_simplex(y + noise, S)
            a = y.copy()
            t_mom = 1.0
            stall_count = 0

    return best_local, best_score


def run_toppk_refine(a_init, n, S, max_time, rng, k_start=8, k_end=256, noise_scale=0.001):

    Top-k subgradient refinement for precision finishing.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = objective_value(a)
    best_local = a.copy()
    it = 0
    stall_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, (time.monotonic() - start) / max_time)
        k = min(k_end, max(k_start, int(k_start + (k_end - k_start) * frac ** 0.5)))

        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_score:
            best_score = cur_obj
            best_local = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        grad = compute_grad_toppk(y, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: decrease over time
        step = 2.0 * S / (n * (1.0 + 0.005 * it))
        step = min(step, 0.5 * S / max(gnorm, 1e-15))
        step = max(step, 1e-12 * S)

        x_new = project_simplex(y - step * grad, S)

        # Backtracking: if objective worsens, halve the step

        if c_new.max() > c.max() * 1.005:
            x_new = 0.5 * y + 0.5 * x_new
            x_new = project_simplex(x_new, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)

            t_mom = 1.0

        if s_norm > 1e-15:
            a_norm = a_norm * (S / s_norm)

        c = fft_convolve_self(a_norm)
        obj = 2.0 * n_i * c.max() / (S * S)

        if obj < global_best:
            global_best = obj
            global_a = a_norm.copy()
            print("  [%s] n=%d: obj=%.6f" % (name, n_i, obj))
        else:
            print("  [%s] n=%d: obj=%.6f (best: %.6f)" % (name, n_i, obj, global_best))

    print("=== After exploration: global_best=%.6f ===" % global_best)

    # Phase 2: Deep polish on the best candidate
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = deadline - time.monotonic()

        if remaining > 15:
            print("=== Phase 2: Deep Polish (remaining %.1f sec) ===" % remaining)

            # Polish 1: High-beta FISTA
            seg1 = remaining * 0.20
            a_p1, s1 = run_fista_smooth(global_a, n_best, S_best, seg1, rng,
                                         beta_start=30.0, beta_end=200.0,
                                         noise_scale=0.001, use_spectral=True)
            a_p1 = np.clip(a_p1, 0, 1000)
            sp = a_p1.sum()
            if sp > 1e-15:
                a_p1 = a_p1 * (S_best / sp)
            c1 = fft_convolve_self(a_p1)
            obj1 = 2.0 * n_best * c1.max() / (S_best * S_best)
            if obj1 < global_best:
                global_best = obj1
                global_a = a_p1.copy()
                print("  Polish FISTA: %.6f" % global_best)

            # Polish 2: Top-k with moderate k
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                seg2 = remaining2 * 0.30
                a_p2, s2 = run_toppk_refine(global_a, n_best, S_best, seg2, rng,
                                             k_start=16, k_end=64, noise_scale=0.0005)
                a_p2 = np.clip(a_p2, 0, 1000)
                sp = a_p2.sum()
                if sp > 1e-15:
                    a_p2 = a_p2 * (S_best / sp)
                c2 = fft_convolve_self(a_p2)
                obj2 = 2.0 * n_best * c2.max() / (S_best * S_best)
                if obj2 < global_best:
                    global_best = obj2
                    global_a = a_p2.copy()
                    print("  Polish top-k mod: %.6f" % global_best)

            # Polish 3: Top-k with large k (very precise)
            remaining3 = deadline - time.monotonic()
            if remaining3 > 10:
                seg3 = remaining3 * 0.40
                a_p3, s3 = run_toppk_refine(global_a, n_best, S_best, seg3, rng,
                                             k_start=64, k_end=512, noise_scale=0.0002)
                a_p3 = np.clip(a_p3, 0, 1000)
                sp = a_p3.sum()
                if sp > 1e-15:
                    a_p3 = a_p3 * (S_best / sp)
                c3 = fft_convolve_self(a_p3)
                obj3 = 2.0 * n_best * c3.max() / (S_best * S_best)
                if obj3 < global_best:
                    global_best = obj3
                    global_a = a_p3.copy()
                    print("  Polish top-k large: %.6f" % global_best)

            # Polish 4: Final micro-refinement
            remaining4 = deadline - time.monotonic()
            if remaining4 > 5:
                a_p4, s4 = run_toppk_refine(global_a, n_best, S_best, remaining4, rng,
                                             k_start=128, k_end=1024, noise_scale=0.0001)
                a_p4 = np.clip(a_p4, 0, 1000)
                sp = a_p4.sum()
                if sp > 1e-15:
                    a_p4 = a_p4 * (S_best / sp)
                c4 = fft_convolve_self(a_p4)
                obj4 = 2.0 * n_best * c4.max() / (S_best * S_best)
                if obj4 < global_best:
                    global_best = obj4
                    global_a = a_p4.copy()
                    print("  Final micro: %.6f" % global_best)

    # Fallback
    if global_a is None:
        global_a = np.ones(8192, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification: obj=%.6f" % ver if np.isfinite(ver) else "Verification: inf")

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, verified obj=%.6f" % (len(final_list), ver))
    return final_list
