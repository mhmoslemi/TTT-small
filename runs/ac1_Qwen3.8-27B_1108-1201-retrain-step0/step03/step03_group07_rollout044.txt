"""
Multi-strategy peak-flattening optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_ab(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def softmax_gradient(c, a, temperature):
    """Compute smooth gradient using softmax weighting over convolution peaks."""
    n = len(a)
    # Shift c to avoid overflow in exp
    c_shifted = c - c.max()
    weights = np.exp(temperature * c_shifted)
    weights /= weights.sum()

    # grad[m] = 2 * sum_k weights[k] * a[k-m]
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1 - 1).bit_length()
    W = fft(weights, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad, weights


def topk_gradient(c, a, top_k, soft_temp=0.0):
    """Compute subgradient using top-k peaks with optional soft weighting."""
    n = len(a)
    top_k = max(1, min(top_k, len(c)))
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    c_max = c[top_idx[0]]

    if soft_temp > 0:
        w = np.exp(soft_temp * (c[top_idx] - c_max))
        w /= w.sum()
    else:
        w = np.full(top_k, 1.0 / top_k)

    w_full = np.zeros(len(c))
    w_full[top_idx] = w

    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1 - 1).bit_length()
    W = fft(w_full, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def peak_equalization_gradient(c, a, threshold_frac=0.95):
    """Gradient that targets all peaks above a threshold fraction of max."""
    n = len(a)
    c_max = c.max()
    threshold = threshold_frac * c_max
    mask = c >= threshold
    if mask.sum() == 0:
        mask[np.argmax(c)] = True

    # Weight proportional to how far above threshold
    excess = np.where(mask, c - threshold + c_max * 0.01, 0.0)
    total_excess = excess.sum()
    if total_excess < 1e-15:
        return topk_gradient(c, a, 1)
    weights = excess / total_excess

    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1 - 1).bit_length()
    W = fft(weights, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad

            c_mean = c.mean()
            spread = (c_max - c_mean) / (c_max + 1e-15)
            temp_cur = softmax_temp * (1.0 + spread)
            grad, weights = softmax_gradient(c, a, temp_cur)
        else:
            grad = topk_gradient(c, a, top_k_cur)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
            eta = min(eta * 1.01, eta0 * 3.0)

                    a = a_pert
                    c = c_pert

                c = c_pert

def rl_peak_chase(a0, time_budget, lr=0.003):
    """Focus exclusively on the single highest peak."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    v = np.zeros(n)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        k_max = np.argmax(c)
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for m in range(lo, hi + 1):
            grad[m] += 2.0 * a[k_max - m]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        v = 0.92 * v + 0.08 * grad

        a = a * np.exp(-lr * v)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())
        c = fft_convolve_self(a)

        cur_obj = obj_value(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

    return best_a, best_obj


def coordinate_polish(a0, time_budget):
    """Fine-grained coordinate descent polishing."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()

    # Process indices in order of gradient contribution
    k_max = np.argmax(c)
    idx_order = np.argsort(-a, kind='stable')

    for _ in range(50):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        for i in idx_order[:min(n, 2048)]:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            for factor in [0.8, 0.9, 1.1, 1.2]:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_tmp = a * (S / s)
                c_new = fft_convolve_self(a_tmp)
                obj_new = obj_value(a_tmp, c_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_tmp.copy()
                    a = a_tmp
                    c = c_new
                    improved = True
                    break
            # Renormalize
            s = a.sum()
            if s > 0:
                a = a * (S / s)
        if not improved:
            break

    return best_a, best_obj

        # Sub-phase 2a: Softmax with broad peaks

        # Sub-phase 2c: Focused single peak chasing

        # Sub-phase 2d: Very fine mirror descent

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(256) * 0.5

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)

    if not np.isfinite(ver):
