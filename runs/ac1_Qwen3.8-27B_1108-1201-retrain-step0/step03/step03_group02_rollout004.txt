import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size-1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A*A))
    return c[:size]

def objective_from_conv(a, c):
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (S*S)

    Solve LP to find direction g that is 'compatible' with a.
    Maximize sum(g) subject to:
      g >= 0
      (a*g)_k <= M for k in active set
    Returns g (the LP solution).
    """
    n = len(a)
    c = fft_convolve_self(a)
    M = c.max()
    
    # Find active constraints: top peaks and near-peaks
    threshold = M * (1.0 - mpsilon)
    active = np.where(c >= threshold)[0]
    
    if len(active) > top_k:
        # Keep only the top peaks
        active = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix: A_eq @ g <= b

    # This is a row with nonzeros at positions where a is nonzero
    
    # Actually, let me think about this more carefully.
    # (a*g)_k = sum_{i} a_i * g_{k-i} where 0 <= i < n and 0 <= k-i < n
    # So for a given k, the coefficients on g_j are a_{k-j} for j in [max(0,k-n+1), min(n-1,k)]

    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in active:

        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):

            cols.append(j)

    A_data = []
    A_rows = []
    A_cols = []

        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A_data.append(a[k - j])
            A_rows.append(ci)
            A_cols.append(j)
    
    A_mat = csr_matrix((A_data, (A_rows, A_cols)), shape=(n_constraints, n))

Algorithm:
1. Start from parent sequence (height_sequence_1) and diverse asymmetric initializations
2. Phase 1: Robust subgradient with exact backtracking line search (fast, reliable)

4. Phase 3: LP-based direction finding for coarse jumps (solves reduced LP on active peaks)

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_from_conv(n, max_c, S):
    """Compute objective from known max convolution and sum."""
    if S < 1e-15:
        return np.inf
    return 2.0 * n * max_c / (S * S)

def soft_max_grad(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_subgradient_bt(a_init, n, S, max_time, rng, top_k=8, beta_init=5.0, beta_max=200.0):
    """
    Hybrid subgradient with backtracking line search.
    Alternates between exact subgradient (top-k peaks) and soft-max approximation.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    no_improve = 0
    patience = 500

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()
        obj = obj_from_conv(n, M, S)

        if obj < best_score - 1e-10:
            best_score = obj
            best_local = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            # Perturb and reset
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            no_improve = 0
            continue

        # Choose between subgradient and soft-max
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))
        if it % 3 == 0:
            grad = soft_max_grad(a, c, beta)
        else:
            grad = subgradient_toppk(a, c, top_k)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        grad_normed = grad / gnorm

        # Backtracking line search
        step = S / (2.0 * np.sqrt(n) * (1.0 + 0.005 * it))
        improved = False
        for _ in range(8):
            a_try = project_simplex(a - step * grad, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < M * (1.0 - 1e-8):
                a = a_try
                improved = True
                break
            step *= 0.5

        if not improved:
            # Try a smaller fixed step
            step = S / (10.0 * n)
            a_try = project_simplex(a - step * grad, S)
            a_try = np.clip(a_try, 0, None)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a = a_try * (S / s_try)

        if it % 200 == 0:
            elapsed = time.monotonic() - start
            print("      SG(iter %d, %.0fs): obj=%.6f" % (it, elapsed, best_score))

    return best_local, best_score


def run_fista_adaptive(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=500.0):
    """
    Adaptive FISTA with periodic restarts and noise injection.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    # FISTA state
    y = a.copy()
    t = 1.0
    it = 0
    no_improve = 0
    patience = 3000

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))

        c = fft_convolve_self(y)
        M = c.max()
        obj = obj_from_conv(n, M, S)

        if obj < best_score - 1e-10:
            best_score = obj
            best_local = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            # Restart with perturbation
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            y = project_simplex(best_local + noise, S)
            t = 1.0
            no_improve = 0
            continue

        grad = soft_max_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta * 2.5)
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 5000 == 0:
            elapsed = time.monotonic() - start
            print("      FISTA(iter %d, %.0fs): obj=%.6f" % (it, elapsed, best_score))

    return best_local, best_score


def solve_lp_direction(a, S, top_k=20, m_relax=0.02, time_limit=10):
    """
    Solve reduced LP: maximize sum(g) s.t. g>=0, (a*g)_k <= M*(1+m_relax) for top peaks.
    Returns g or None.
    """
    n = len(a)
    c = fft_convolve_self(a)
    M = c.max()
    
    # Active constraints: top peaks
    active_idx = np.argsort(c)[::-1][:top_k]
    M_relaxed = M * (1.0 + m_relax)
    
    # Build constraint matrix
    # Row for constraint k: sum_{j: 0<=j<n, 0<=k-j<n} a_{k-j} * g_j <= M_relaxed
    n_con = len(active_idx)
    A_data = []
    A_rows = []
    A_cols = []
    
    for ci, k in enumerate(active_idx):
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max < j_min:
            continue
        # a[k-j] for j in [j_min, j_max]
        # = a[k-j_min], a[k-j_min-1], ..., a[k-j_max]
        # = a[k-j_max : k-j_min+1][::-1]
        idx = np.arange(j_min, j_max + 1)
        a_vals = a[k - idx][::-1]  # a[k-j_max], ..., a[k-j_min]
        A_data.extend(a_vals.tolist())
        A_rows.extend([ci] * len(idx))
        A_cols.extend(idx.tolist())
    
    if not A_data:
        return None
    
    A_mat = csr_matrix((A_data, (A_rows, A_cols)), shape=(n_con, n))
    
    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)
    bounds = [(0.0, S * 5.0)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_mat, b_ub=np.full(n_con, M_relaxed),
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_limit,
                                  'mip_rel_gap': 0.0})
        if result.success and result.x is not None:
            g = np.clip(result.x, 0, None)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    
    return None


def run_lp_optimize(a_init, n, S, max_time, rng, top_k=15, m_relax=0.03):
    """Iterative LP-based optimization with line search."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()
        obj = obj_from_conv(n, M, S)

        if obj < best_score - 1e-10:
            best_score = obj
            best_local = a.copy()

        g = solve_lp_direction(a, S, top_k=top_k, m_relax=m_relax, time_limit=15)
        if g is None:
            break

        g = g * (S / g.sum())

        # Line search: find best t in a_try = (1-t)*a + t*g
        best_obj = obj
        best_a = a.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t) * a + t * g
            a_try = np.clip(a_try, 0, None)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try * (S / s_try)
            c_try = fft_convolve_self(a_try)
            obj_try = obj_from_conv(n, c_try.max(), S)
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

        if best_obj >= obj * (1 - 1e-10):
            # No improvement, increase relaxation
            m_relax = min(0.2, m_relax * 1.5)
            if m_relax > 0.15:
                break
            continue

        a = best_a

        if it % 5 == 0:
            elapsed = time.monotonic() - start
            print("      LP(iter %d, %.0fs): obj=%.6f" % (it, elapsed, best_score))

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    starts.append(("alt_exp4", np.exp(4.0 * tt_alt)))
    starts.append(("alt_exp4b", np.exp(4.0 * (1.0 - tt_alt))))
    starts.append(("alt_ramp", tt_alt + 0.1))
    starts.append(("alt_ramp_b", 1.0 - tt_alt + 0.1))

    # --- Phase 1: Quick screening of all starts ---
    print("=== Phase 1: Quick Screening (%d starts) ===" % len(starts))
    t0 = time.monotonic()

    screened = []
    for name, arr in starts:
        if time.monotonic() > deadline - 300:
            break
        n_i = len(arr)
        if n_i < 32:
            continue
        S_i = np.sqrt(2.0 * float(n_i))
        a_norm = safe_normalize(arr, n_i)
        obj = quick_evaluate(a_norm)
        screened.append((obj, name, a_norm, n_i, S_i))
        if obj < global_best:
            global_best = obj
            global_a = a_norm.copy()

    screened.sort(key=lambda x: x[0])
    elapsed_screen = time.monotonic() - t0
    print("  Screening done in %.1fs. Best: %.6f (%s)" %
          (elapsed_screen, screened[0][0], screened[0][1]))

    # --- Phase 2: Optimize top candidates with subgradient+backtracking ---
    print("=== Phase 2: Subgradient Optimization ===")
    remaining = deadline - time.monotonic()
    n_optimize = min(6, len(screened))
    per_cand = remaining * 0.45 / max(1, n_optimize)

    for i in range(n_optimize):
        if time.monotonic() > deadline - 120:
            break
        obj_init, name, a_init, n_i, S_i = screened[i]
        alloc = min(per_cand, deadline - time.monotonic() - 60)
        if alloc < 10:
            break

        print("  [%s] n=%d, init_obj=%.6f, alloc=%.0fs" % (name, n_i, obj_init, alloc))

        # Run subgradient with backtracking
        a_opt, score = run_subgradient_bt(a_init, n_i, S_i, alloc * 0.6, rng,
                                           top_k=8, beta_init=10.0, beta_max=300.0)
        a_opt = safe_normalize(a_opt, n_i)
        obj = quick_evaluate(a_opt)
        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
            print("    SG improved: %.6f" % global_best)

        # Continue with FISTA on the result
        remaining_cand = min(alloc * 0.4, deadline - time.monotonic() - 60)
        if remaining_cand > 5:
            a_f, score_f = run_fista_adaptive(a_opt, n_i, S_i, remaining_cand, rng,
                                               beta_init=20.0, beta_max=400.0)
            a_f = safe_normalize(a_f, n_i)
            obj_f = quick_evaluate(a_f)
            if obj_f < global_best:
                global_best = obj_f
                global_a = a_f.copy()
                print("    FISTA improved: %.6f" % global_best)

    # --- Phase 3: LP-based optimization on the best ---
    print("=== Phase 3: LP Optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        alloc_lp = min(remaining * 0.35, 600.0)
        print("  LP on n=%d, alloc=%.0fs, current best=%.6f" % (n_best, alloc_lp, global_best))

        a_lp, score_lp = run_lp_optimize(global_a, n_best, S_best, alloc_lp, rng,
                                          top_k=12, m_relax=0.05)
        a_lp = safe_normalize(a_lp, n_best)
        obj_lp = quick_evaluate(a_lp)
        if obj_lp < global_best:
            global_best = obj_lp
            global_a = a_lp.copy()
            print("  LP improved: %.6f" % global_best)

    # --- Phase 4: Final polish ---
    print("=== Phase 4: Final Polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 10 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))

        # Subgradient with very small top_k for precision
        alloc_final = remaining * 0.5
        a_fin, score_fin = run_subgradient_bt(global_a, n_best, S_best, alloc_final, rng,
                                               top_k=3, beta_init=50.0, beta_max=800.0)
        a_fin = safe_normalize(a_fin, n_best)
        obj_fin = quick_evaluate(a_fin)
        if obj_fin < global_best:
            global_best = obj_fin
            global_a = a_fin.copy()
            print("  Final SG: %.6f" % global_best)

        # One more LP round
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            a_lp2, _ = run_lp_optimize(global_a, n_best, S_best, min(remaining2 * 0.4, 120.0), rng,
                                        top_k=8, m_relax=0.02)
            a_lp2 = safe_normalize(a_lp2, n_best)
            obj_lp2 = quick_evaluate(a_lp2)
            if obj_lp2 < global_best:
                global_best = obj_lp2
                global_a = a_lp2.copy()
                print("  Final LP: %.6f" % global_best)

    # --- Phase 5: Random restarts from best (if time remains) ---
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        print("=== Phase 5: Random Restarts (%.0fs) ===" % remaining)
        n_restarts = min(int(remaining / 30), 5)

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            noise = rng.normal(0, 0.02 * S_best / np.sqrt(n_best), n_best)
            a_restart = project_simplex(global_a + noise, S_best)
            alloc_r = min(25.0, deadline - time.monotonic() - 10)
            a_r, score_r = run_subgradient_bt(a_restart, n_best, S_best, alloc_r, rng,
                                               top_k=5, beta_init=20.0, beta_max=200.0)
            a_r = safe_normalize(a_r, n_best)
            obj_r = quick_evaluate(a_r)
            if obj_r < global_best:
                global_best = obj_r
                global_a = a_r.copy()
                print("  Restart %d: %.6f" % (ri, global_best))

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
    print("Internal: %.6f, len=%d" % (global_best, len(final_list)))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        global_a = np.ones(4096, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback verification: %.6f" % ver)

    return final_list
