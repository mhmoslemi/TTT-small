Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses basin hopping + adaptive FISTA + crossover for global search,
exploiting the non-convexity of the self-convolution max function.

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    # Standard simplex projection
    if v.sum() <= S:
        # Already under, just add uniformly? No, we need sum exactly S.
        # Actually for projection onto {x>=0, sum(x)=S}:
        pass
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    rho = np.arange(1, n + 1)
    cond = w - (cs - S) / rho > 0
    if not np.any(cond):
        theta = cs[-1] / n
    else:
        rho_val = np.max(rho[cond])
        theta = (cs[rho_val - 1] - S) / rho_val
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_value_and_grad(a, c, beta):
    """Compute LSE value and gradient."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    lse = cm + np.log(tot) / beta
    if tot < 1e-300:
        return lse, np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse, grad


def local_optimize(a_init, n, S, max_time, rng, 
                   beta_start=2.0, beta_end=500.0,
                   use_backtrack=True, jump_prob=0.02,
                   jump_scale=0.1):
    """
    Local optimization using adaptive LSE-smoothed proximal gradient
    with occasional jumps (basin hopping).
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: increase gradually
        progress = min(1.0, it / 5000.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = y.copy()
            last_improve = it
        
        # Compute gradient
        lse_val, grad = lse_value_and_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size
        if use_backtrack:
            # Approximate Lipschitz constant
            L = 2.0 * beta * np.max(c) / max(gnorm, 1e-10)
            L = max(L, 1.0)
            step = 1.0 / (L * 1.1)
            step = min(step, S / (gnorm + 1e-10))
        else:
            step = 1.0 / (2.0 * beta)
            step = min(step, S / (gnorm + 1e-10) * 0.5)
        
        # FISTA step
        x_new = project_simplex(y - step * grad, S)
        
        # Momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Basin hopping: occasional large jump
        if it - last_improve > 2000 and rng.random() < jump_prob:
            # Large perturbation
            jump = rng.normal(0, jump_scale * S / np.sqrt(n), n)
            y = project_simplex(y + jump, S)
            a = y.copy()
            t_mom = 1.0
            last_improve = it
        
        # Periodic small perturbation to avoid stalling
        if it % 1000 == 0:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = max(1.0, t_mom * 0.9)
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def exact_subgradient_step(a, c, S, step):
    """Take a step using exact subgradient at argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-14:
        return a
    return project_simplex(a - step * grad / gnorm, S)


def polish_exact(a_init, n, S, max_time, rng):
    """Polish using exact subgradients with decreasing step size."""
    a = normalize_sum(a_init, n)

    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Step size decreases over time
        step = S / (n * (1.0 + 0.01 * it))
        
        # Use multiple argmax lags
        k = min(64, 8 + it // 100)
        top_idx = np.argsort(c)[::-1][:k]
        
        # Average gradient over top-k lags
        grad = np.zeros(n)
        for ki in top_idx:
            ki = int(ki)
            lo = max(0, ki - n + 1)
            hi = min(ki, n - 1)
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * a[ki - idx]
        grad /= k
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        a = project_simplex(a - step * grad / gnorm, S)
        
        # Occasional small random perturbation
        if it % 500 == 0 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
    
    return best_a, best

**Approach: Direct max-convolution minimization via iterative redistribution**

The key idea: if we identify the lag k where the convolution is maximal, we can reduce it by redistributing mass from the "contributing" indices to "non-contributing" indices.

Specifically, if c_k is the max, then c_k = sum_i a_i * a_{k-i}. To reduce c_k, we want to reduce the products a_i * a_{k-i}. This can be done by:

"""

Uses basin-hopping proximal gradient with adaptive LSE smoothing,

Exploits non-convexity via aggressive exploration and annealing.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    rho = np.arange(1, n + 1)
    cond = w - (cs - S) / rho > 0
    if not np.any(cond):
        theta = cs[-1] / n
    else:
        rho_val = int(np.max(rho[cond]))
        theta = (cs[rho_val - 1] - S) / rho_val
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_grad(a, c, beta):
    """Gradient of LSE(beta * c)."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgrad(a, c, k_max):
    """Average subgradient over top-k lags."""
    n = len(a)
    k = min(k_max, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for ki in top_idx:
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(ki, n - 1)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[ki - idx]
    grad /= k
    return grad


def pair_redistribute(a, c, n, S, k_top=32):
    """
    For the top lags, try to reduce convolution by redistributing mass
    within pairs (i, k-i). Returns modified a.
    """
    a = a.copy()
    top_lags = np.argsort(c)[::-1][:k_top]
    
    for ki in top_lags:
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(ki, n - 1)
        
        # For each pair (i, ki-i) with i < ki-i, try to reduce the product
        for i in range(lo, hi + 1):
            j = ki - i
            if i >= j:
                continue
            ai = a[i]
            aj = a[j]
            if ai < 1e-15 or aj < 1e-15:
                continue
            # Product is ai*aj (counted twice in c_ki since i≠j)
            # To reduce: make them more unequal while preserving ai+aj
            # Move mass from larger to smaller (or vice versa to create more disparity)
            # Actually to REDUCE product with fixed sum, push toward extremes
            # But we can't go below 0. So: move a fraction of the smaller to the larger
            frac = 0.3
            move = frac * min(ai, aj)
            if ai > aj:
                a[i] += move
                a[j] -= move
            else:
                a[j] += move
                a[i] -= move
    
    # Re-project to maintain constraints
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a *= S / s
    return a

            best = cur
            best_a = a.copy()
        
        # Very fine subgradient with many top lags
        k_max = min(256, 32 + it // 20)
        grad = topk_subgrad(a, c, k_max)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Decreasing step size
        step = S / (gnorm * (1.0 + 0.03 * it))
        step = min(step, 0.5)
        
        a = project_simplex(a - step * grad, S)
        
        # Tiny perturbations to escape degenerate points
        if it % 300 == 0:
            a = project_simplex(a + rng.normal(0, 1e-4 * S / np.sqrt(n), n), S)
    
    return best_a, best


def gen_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Structured
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * tt))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1 + 0.4 * np.sin(2 * np.pi * 2 * tt))
    inits.append(1 + 0.3 * np.sin(2 * np.pi * 3 * tt + 0.7))
    inits.append(1 + 0.5 * np.cos(4 * np.pi * tt))
    
    # Gaussian bump
    inits.append(np.exp(-8 * (tt - 0.5) ** 2))
    
    # Random
    inits.append(rng.random(n) + 0.05)
    inits.append(np.abs(rng.normal(0, 1, n)) + 0.05)
    inits.append(np.exp(rng.normal(0, 0.5, n)) + 0.01)
    
    return [np.maximum(x, 0.01) for x in inits]
