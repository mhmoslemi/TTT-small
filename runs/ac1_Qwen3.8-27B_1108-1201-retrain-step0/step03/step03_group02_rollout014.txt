Minimize 2*n*max(conv(a,a)) / sum(a)^2 over non-negative sequences.

Approach:
- Multiple asymmetric initializations (Beta-distribution shapes, ramps, parent)
- Projected subgradient descent with adaptive peak weighting
- Backtracking line search for guaranteed progress
- Periodic perturbation to escape local minima
- Final coordinate-descent polish

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    if v.sum() <= 0:
        return np.full(n, S / n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    # Fix numerical errors
    c[np.abs(c) < 1e-12] = 0.0
    return c[:size]


def compute_subgradient_fft(a, w_full):
    """
    Compute subgradient of max_k c[k] given peak weights w_full.
    w_full has length 2n-1, with w_full[k] = weight for peak k.
    Returns gradient vector of length n.
    grad[m] = 2 * sum_k w_full[k] * a[k-m] (for valid k-m)
    """
    n = len(a)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_from_maxc(max_c, n):
    """Compute objective from max convolution value (assuming sum = sqrt(2n))."""
    S = np.sqrt(2.0 * float(n))
    return 2.0 * n * max_c / (S * S)

    best_score = np.inf

    top_k_max = 512

    p_decay = 0.999

        it += 1
        
        # Compute convolution
        c = fft_convolve_self(a)
        cur_max = c.max()

            obj = objective_from_maxc(cur_max, n)
            print("    it=%d: max_c=%.6f, obj=%.6f" % (it, cur_max, obj))
        
        # Adaptive parameters

        p = max(p_min, p_init * (p_decay ** it))
        
        # Select top peaks
        if top_k >= 2 * n - 1:
            peak_indices = np.arange(2 * n - 1)
            peak_vals = c
        else:
            peak_indices = np.argpartition(c, -top_k)[-top_k:]
            peak_vals = c[peak_indices]
        
        # Weight peaks by value^p
        peak_vals_pos = np.maximum(peak_vals, 1e-30)
        w = peak_vals_pos ** p
        w = w / w.sum()

        w_full = np.zeros(2 * n - 1)
        w_full[peak_indices] = w
        
        # Compute subgradient
        grad = compute_subgradient_fft(a, w_full)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-20:

        # Step size (diminishing)

            candidate = project_simplex(a - factor * step * grad, S)
            c_cand = fft_convolve_self(candidate)

                a_new = candidate

        a = a_new

            lo = max(0, worst_k - n + 1)

Algorithm:
1. Diverse asymmetric initializations (Beta shapes, ramps, parent)

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    c[np.abs(c) < 1e-12] = 0.0
    return c[:size]


def subgradient_fft(a, w_full):
    """grad[m] = 2 * sum_k w_full[k] * a[k-m]"""
    n = len(a)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def optimize_sequence(a_init, n, S, max_time, rng):
    """Main subgradient optimizer."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    stagnation = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Adaptive parameters
        top_k = min(512, 4 + it // 10)
        p = max(0.5, 4.0 * (0.9995 ** it))
        
        # Peak selection and weighting
        if top_k >= c.size:
            peak_idx = np.arange(c.size)
            peak_vals = c.copy()
        else:
            peak_idx = np.argpartition(c, -top_k)[-top_k:]
            peak_vals = c[peak_idx]
        
        pv = np.maximum(peak_vals, 1e-30)
        w = pv ** p
        ws = w.sum()
        if ws < 1e-30:
            break
        w = w / ws
        
        w_full = np.zeros(c.size)
        w_full[peak_idx] = w
        
        grad = subgradient_fft(a, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-20:
            break
        
        # Step size
        step = S / (np.sqrt(n) * np.sqrt(1.0 + 0.05 * it))
        
        # Backtracking
        a_new = None
        for factor in [1.0, 0.5, 0.25, 0.1, 0.01]:
            cand = project_simplex(a - factor * step * grad, S)
            c_cand = fft_convolve_self(cand)
            if c_cand.max() < cur_max * (1.0 - 1e-10):
                a_new = cand
                break
        
        if a_new is not None:
            a = a_new
        else:
            # Stuck: perturb
            if stagnation > 100:
                noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
                stagnation = 0
            elif it % 100 == 0:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
    
    return best_a, best_score
