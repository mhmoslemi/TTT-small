    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    # g[j] = 2 * (w * a_rev)[n-1+j] where a_rev = a[::-1]
    a_rev = a[::-1]
    fsize = 1 << (3 * n - 3).bit_length()
    W = fft(w, fsize)
    Ar = fft(a_rev, fsize)
    conv = np.real(ifft(W * Ar))
    grad = 2.0 * conv[n - 1: 2 * n - 1]
    return grad


def fista_lse(a_init, n, S, max_time, rng, beta_start=5.0, beta_end=5000.0, L_init=None):
    """FISTA with LSE smoothing, adaptive beta and Lipschitz constant."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    L = L_init if L_init is not None else beta_start * 4.0

    start = time.monotonic()
    it = 0
    best_score = np.inf
    best_a = a.copy()
    stall_count = 0
    last_best_val = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max_time)
        beta = beta_start * (beta_end / beta_start) ** (frac ** 0.7)

        c = fft_convolve_self(y)
        cur_max = c.max()
        true_score = 2.0 * n * cur_max / (S * S)

        if true_score < best_score - 1e-10:
            best_score = true_score
            best_a = y.copy()
            stall_count = 0
            last_best_val = cur_max
        elif it % 200 == 0:
            if cur_max > last_best_val * 1.0005:
                stall_count += 1
            else:
                stall_count = max(0, stall_count - 1)

        grad = compute_lse_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / L
        x = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x + ((t - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        a = x
        t = t_new

        if it % 200 == 0:
            L = max(L * 0.92, beta * 2.0 * S * 0.5)

        if stall_count > 80 and rng.random() < 0.15:
            noise_scale = 0.015 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t = 1.0
            stall_count = 0

    return best_a, best_score


def subgradient_topk(a, n, S, max_time, rng, k_init=3, k_max=64):
    """Subgradient method targeting top-k lags with adaptive step."""
    a = np.clip(a, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    it = 0
    best_score = np.inf
    best_a = a.copy()
    avg_step = 2.0 * S / (n * 10.0)

    while time.monotonic() - start < max_time:
        it += 1
        k = min(k_max, k_init + it // 500)

        c = fft_convolve_self(a)
        cur_max = c.max()
        true_score = 2.0 * n * cur_max / (S * S)
        if true_score < best_score:
            best_score = true_score
            best_a = a.copy()

        # Softmax-weighted top-k gradient
        top_idx = np.argsort(c)[::-1][:k]
        tv = c[top_idx]
        w_k = np.exp(50.0 * (tv - tv.max()))
        w_k = w_k / w_k.sum()
        # Build full weight vector
        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = w_k
        # Gradient = 2 * cross-correlation
        a_rev = a[::-1]
        fsize = 1 << (3 * n - 3).bit_length()
        W = fft(w_full, fsize)
        Ar = fft(a_rev, fsize)
        conv = np.real(ifft(W * Ar))
        grad = 2.0 * conv[n - 1: 2 * n - 1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = avg_step / max(gnorm / (S / n), 0.1) * 0.5
        a_new = project_simplex(a - step * grad, S)

        if it % 5 == 0:
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max:
                a = a_new
                avg_step = min(avg_step * 1.05, 2.0 * S / n)
            else:
                avg_step *= 0.85

    return best_a, best_score


def peak_shave(a, n, S, max_time, rng):
    """Greedy peak reduction: transfer mass from expensive to cheap coordinates."""
    a = np.clip(a, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        true_score = 2.0 * n * cur_max / (S * S)
        if true_score < best_score:
            best_score = true_score
            best_a = a.copy()

        top_lags = np.argsort(c)[::-1][:3]
        improved = False

        for lag in top_lags:
            lag = int(lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue
            indices = np.arange(i_min, i_max + 1)
            partner = lag - indices
            products = a[indices] * a[partner]
            if products.max() <= 0:
                continue
            # Find coordinate with largest contribution
            j_star = indices[np.argmax(products)]
            p_star = partner[np.argmax(products)]
            # Find coordinate with smallest contribution (and positive value)
            min_prod = np.min(products)
            if min_prod >= products.max() * 0.9:
                continue
            j_min = indices[np.argmin(products)]
            p_min = partner[np.argmin(products)]

            # Transfer from j_star to j_min
            delta = 0.005 * max(a[j_star], a[p_star])
            a_new = a.copy()
            if a[j_star] >= a[p_star]:
                a_new[j_star] = max(0, a[j_star] - delta)
            else:
                a_new[p_star] = max(0, a[p_star] - delta)
            target = j_min if a[j_min] >= a[p_min] else p_min
            a_new[target] += delta
            a_new = a_new * (S / max(a_new.sum(), 1e-15))

            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max * 0.9999:
                a = a_new
                improved = True

        if not improved and it % 10 == 0:
            # Perturb to escape
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_a, best_score


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.05
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_periodic_init(n, rng, S, num_freqs=4):
    """Periodic with multiple sinusoidal components."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 25)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.5
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_w_shape_init(n, rng, S):
    """W/M shaped initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    k = rng.integers(2, 6)
    a = np.abs(np.sin(np.pi * k * tt)) + 0.05
    a = a * (S / a.sum())
    return a


def make_bimodal_init(n, rng, S):
    """Two Gaussian bumps."""
    tt = np.arange(n, dtype=np.float64) / n
    c1 = rng.uniform(0.2, 0.4)
    c2 = 1.0 - c1
    w1 = rng.uniform(0.02, 0.08)
    w2 = rng.uniform(0.02, 0.08)
    a = np.exp(-((tt - c1) / w1) ** 2) + np.exp(-((tt - c2) / w2) ** 2) + 0.01
    a = a * (S / a.sum())
    return a


def objective_value(a):
    """Compute the true objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    # Get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0.0, 1000.0)
    except Exception:
        pass

    print("Parent available: %s, length: %s" % (parent is not None,
          len(parent) if parent is not None else 0))

    global_best = np.inf
    global_a = None

    def update_best(a_cand, label=""):
        nonlocal global_best, global_a
        a_cand = np.clip(a_cand, 0.0, 1000.0)
        if a_cand.sum() < 0.01:
            return
        obj = objective_value(a_cand)
        if obj < global_best:
            global_best = obj
            global_a = a_cand.copy()
            if label:
                print("  New best: %.6f (%s)" % (obj, label))
        return obj

    # === Phase 1: Main FISTA optimization from parent ===
    if parent is not None and len(parent) >= 64:
        n = len(parent)
    else:
        n = 8192
    S = np.sqrt(2.0 * n)
    a_start = parent.copy() if parent is not None and len(parent) >= 64 else np.ones(n) * S / n
    a_start = a_start * (S / max(a_start.sum(), 1e-15))

    print("=== Phase 1: FISTA from parent, n=%d ===" % n)
    t0 = time.monotonic()
    remaining = deadline - t0
    seg1 = min(remaining * 0.40, 800.0)

    a1, s1 = fista_lse(a_start, n, S, seg1, rng,
                        beta_start=5.0, beta_end=3000.0)
    update_best(a1, "FISTA parent")
    print("  Phase 1 done in %.0fs, best=%.6f" % (time.monotonic() - t0, global_best))

    # === Phase 2: Subgradient polish ===
    t0 = time.monotonic()
    remaining = deadline - t0
    seg2 = min(remaining * 0.15, 300.0)
    a2, s2 = subgradient_topk(global_a, n, S, seg2, rng, k_init=3, k_max=80)
    update_best(a2, "Subgradient")
    print("  Phase 2 done in %.0fs, best=%.6f" % (time.monotonic() - t0, global_best))

    # === Phase 3: Peak shaving ===
    t0 = time.monotonic()
    remaining = deadline - t0
    seg3 = min(remaining * 0.10, 200.0)
    a3, s3 = peak_shave(global_a, n, S, seg3, rng)
    update_best(a3, "PeakShave")
    print("  Phase 3 done in %.0fs, best=%.6f" % (time.monotonic() - t0, global_best))

    # === Phase 4: High-beta FISTA polish ===
    t0 = time.monotonic()
    remaining = deadline - t0
    seg4 = min(remaining * 0.15, 400.0)
    a4, s4 = fista_lse(global_a, n, S, seg4, rng,
                        beta_start=50.0, beta_end=8000.0,
                        L_init=50.0 * 2.0 * S * 0.5)
    update_best(a4, "FISTA hi-beta")
    print("  Phase 4 done in %.0fs, best=%.6f" % (time.monotonic() - t0, global_best))

    # === Phase 5: Alternative starts ===
    t0 = time.monotonic()
    remaining = deadline - t0
    if remaining > 100:
        print("=== Phase 5: Alternative starts ===")
        inits = []
        inits.append(("anti_pal", make_anti_palindromic_init(n, rng, S)))
        inits.append(("periodic", make_periodic_init(n, rng, S)))
        inits.append(("wshape", make_w_shape_init(n, rng, S)))
        inits.append(("bimodal", make_bimodal_init(n, rng, S)))

        per_start = remaining * 0.6 / len(inits)
        for name, arr in inits:
            if time.monotonic() > deadline - 50:
                break
            seg = min(per_start, 200.0)
            a_alt, s_alt = fista_lse(arr, n, S, seg, rng,
                                      beta_start=5.0, beta_end=2000.0)
            update_best(a_alt, "alt:%s" % name)

    # === Phase 6: Try longer sequence ===
    t0 = time.monotonic()
    remaining = deadline - t0
    if remaining > 120:
        n_long = n * 2 if n <= 16384 else n
        if n_long != n:
            print("=== Phase 6: Trying n=%d ===" % n_long)
            S_long = np.sqrt(2.0 * n_long)
            if global_a is not None:
                from numpy import interp
                tt_old = np.linspace(0, 1, len(global_a))
                tt_new = np.linspace(0, 1, n_long)
                a_long = interp(tt_new, tt_old, global_a)
                a_long = np.maximum(a_long, 0.0)
            else:
                a_long = make_anti_palindromic_init(n_long, rng, S_long)
            a_long = a_long * (S_long / max(a_long.sum(), 1e-15))

            seg_long = min(remaining * 0.35, 300.0)
            a_l, s_l = fista_lse(a_long, n_long, S_long, seg_long, rng,
                                  beta_start=5.0, beta_end=3000.0)
            obj_l = objective_value(a_l)
            if obj_l < global_best:
                global_best = obj_l
                global_a = a_l.copy()
                print("  Long seq n=%d: %.6f" % (n_long, global_best))

    # === Phase 7: Final polish ===
    t0 = time.monotonic()
    remaining = deadline - t0
    if remaining > 20 and global_a is not None:
        n_f = len(global_a)
        S_f = np.sqrt(2.0 * n_f)
        seg_f = min(remaining * 0.7, 300.0)
        print("=== Phase 7: Final polish, n=%d, %.0fs ===" % (n_f, seg_f))
        a_f, s_f = fista_lse(global_a, n_f, S_f, seg_f, rng,
                              beta_start=100.0, beta_end=10000.0,
                              L_init=100.0 * 2.0 * S_f * 0.5)
        update_best(a_f, "Final polish")

        # One more peak shave
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a_ps, s_ps = peak_shave(global_a, n_f, S_f, min(remaining2 * 0.5, 60.0), rng)
            update_best(a_ps, "Final peak shave")

    # === Final verification ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Final result: %.6f (len=%d) ===" % (ver if np.isfinite(ver) else float('inf'), len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: evaluator rejected, using fallback")
        final_list = [1.0] * 1024

    return final_list
