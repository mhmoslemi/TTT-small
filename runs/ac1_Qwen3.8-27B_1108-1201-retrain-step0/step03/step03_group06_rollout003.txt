"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multiplicative Equalization (ME) - equalizes subgradient across coordinates
2. Approximate LP best-response (top-k constraints only)
3. L-BFGS-B with smooth p-norm objective
4. Coordinate descent polish
5. Multi-start with diverse initializations
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def compute_subgrad_smooth(c, a, T):
    """Compute smoothed subgradient: 2 * sum_k softmax(T*c)[k] * a[k-m]"""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    scaled = T * c / c_max - T
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    fsize = 1 << (2 * len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]

def multiplicative_equalization(a_init, n, S, time_budget, 
                                 T_init=20.0, T_final=500.0,
                                 eta_init=0.3, eta_final=0.05,
                                 n_restarts=5, verbose=False):
    """
    Multiplicative Equalization algorithm.
    Key idea: at optimum, all positive coordinates should have equal subgradient.
    Update: a_i <- a_i * (lambda / s_i)^eta where s_i is subgradient at i.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.5)
        eta = eta_init * (eta_final / eta_init) ** progress
        
        # Compute autocorrelation
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > max_no_improve:
            scale = 0.02 * max(0.1, 1.0 - progress)
            rng = np.random.default_rng(step * 7919 + 13)
            a = best_a + scale * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            no_improve = 0
            step += 1
            continue
        
        # Compute smoothed subgradient
        g = compute_subgrad_smooth(c, a, T)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Multiplicative equalization update
        # Target: all positive a_i should have g_i = lambda (the average)
        positive_mask = a > 1e-12
        if positive_mask.sum() > 0:
            lambda_val = g[positive_mask].mean()
            if lambda_val > 1e-15:
                ratio = lambda_val / np.maximum(g, 1e-15)
                # Clip ratio to avoid extreme updates
                ratio = np.clip(ratio, 0.1, 10.0)
                a = a * ratio ** eta
            else:
                # If all subgradients are ~0, add small noise
                a = a * (1.0 + 1e-4 * np.random.default_rng(step).standard_normal(n))
        
        a = normalize_nonneg(a, S)
        step += 1
        
        if verbose and step % 200 == 0:
            print("    ME step %d: obj=%.7f, T=%.1f, eta=%.3f" % (step, best_obj, T, eta))
    
    return best_a, best_obj


def lp_best_response(a, M, top_k=100, max_iter=50):
    """
    Solve approximate LP: maximize sum(g) subject to top-k convolution constraints.
    Returns the best response g (not normalized).
    """
    n = len(a)
    c = fft_convolve_self(a)
    
    # Get top-k peak positions
    peak_indices = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix A_eq: A_eq[l, j] = a[j+l] if 0 <= j+l < n else 0
    # We want: A_eq @ g <= M for each peak lag
    # For lag l (0-indexed in c, which is l - (n-1)):
    #   c_l = sum_{j=max(0,l')}^{min(n-1, n-1+l')} a[j] * g[j-l']
    #   where l' = l - (n-1)
    
    # Actually, c[k] for k=0,...,2n-2 corresponds to lag (k - (n-1))
    # c[k] = sum_j a[j] * a[j - (k-(n-1))]
    #       = sum_j a[j] * a[j + (n-1-k)]
    
    # For the LP, we want constraints: for each peak index k:
    #   sum_j a[j + (n-1-k)] * g[j] <= M  (where a[j+(n-1-k)] is defined)
    #   i.e., sum_j a[j + offset] * g[j] <= M
    #   where offset = n-1-k
    
    A_rows = []
    for k in peak_indices:
        offset = (n - 1) - k  # This is the lag
        # c[k] = sum_j a[j] * a[j - offset] = sum_j a[j + offset] * g[j]
        # For each j, the coefficient is a[j + offset] if 0 <= j+offset < n
        row = np.zeros(n)
        if offset >= 0:
            # j + offset >= 0 always, and j + offset < n means j < n - offset
            valid_j = np.arange(0, n - offset)
            row[valid_j] = a[valid_j + offset]
        else:
            # j + offset >= 0 means j >= -offset = |offset|
            valid_j = np.arange(-offset, n)
            row[valid_j] = a[valid_j + offset]
        A_rows.append(row)
    
    A_constraint = np.array(A_rows)
    b_constraint = np.full(top_k, M)
    
    # Maximize sum(g) = -minimize -sum(g)
    c_obj = -np.ones(n)
    
    result = linprog(c_obj, A_ub=A_constraint, b_ub=b_constraint, 
                     bounds=[(0, None)] * n, method='highs')
    
    if result.success:
        return result.x
    else:
        # Fallback: return uniform
        return np.full(n, 1.0)


def lp_iteration(a_init, n, S, time_budget, top_k=50, mixing=0.3, 
                 verbose=False):
    """
    LP-based best response iteration.
    a_{t+1} = (1-mixing)*a_t + mixing*g_t where g_t is LP best response.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    iteration = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget:
            break
            
        c = fft_convolve_self(a)
        M = c.max()
        
        # Solve LP
        t0 = time.monotonic()
        g = lp_best_response(a, M, top_k=top_k)
        t_lp = time.monotonic() - t0
        
        if g is None or not np.all(np.isfinite(g)):
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g * (S / g_sum)
        
        # Mix
        a_new = (1.0 - mixing) * a + mixing * g
        a_new = normalize_nonneg(a_new, S)
        
        c_new = fft_convolve_self(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller mixing
            if no_improve > 10:
                mixing *= 0.5
                if mixing < 0.01:
                    break
            continue
        
        iteration += 1
        if verbose and iteration % 5 == 0:
            print("    LP iter %d: obj=%.7f, lp_time=%.2fs, mixing=%.3f" % (iteration, best_obj, t_lp, mixing))
    
    return best_a, best_obj


def smooth_pnorm_objective(a, n, S, p=100.0):
    """Compute smooth p-norm approximation of max(conv(a,a))."""
    c = fft_convolve_self(a)
    # p-norm approximation of max
    c_pos = np.maximum(c, 1e-30)
    val = (np.sum(c_pos ** p)) ** (1.0 / p)
    return val

def smooth_pnorm_gradient(a, n, S, p=50.0):
    """Compute gradient of p-norm approximation."""
    c = fft_convolve_self(a)
    c_pos = np.maximum(c, 1e-30)
    # d/d(a) of (sum c^p)^(1/p) = (1/p) * (sum c^p)^(1/p - 1) * p * c^(p-1) * dc/da
    # = (sum c^p)^(1/p - 1) * c^(p-1) * dc/da
    norm_val = (np.sum(c_pos ** p)) ** (1.0 / p)
    weights = c_pos ** (p - 1) * (1.0 / (norm_val ** (p - 1)))
    # weights[l] / sum(weights) * p ... actually let me be more careful
    # f = (sum_l c_l^p)^(1/p)
    # df/dc_l = c_l^(p-1) / (sum c^p)^(1 - 1/p)
    # df/da_i = sum_l df/dc_l * dc_l/da_i
    # dc_l/da_i = 2 * sum over valid j: a[j] where l corresponds to lag involving i
    
    # dc_l/da_i = 2 * (a[i-l'] + a[i+l'])/2 ... no
    # c_l = sum_j a_j * a_{j-l}
    # dc_l/da_i: i appears if i = j (then contrib is a_{i-l}) or i = j-l (then contrib is a_i)
    # So dc_l/da_i = a_{i-l} + a_{i+l} (where defined)
    # Sum over l: sum_l w_l * (a_{i-l} + a_{i+l})
    
    # This is the same as the subgradient computation!
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def lbfgs_smooth(a_init, n, S, time_budget, p=50.0, max_iter=200):
    """Use L-BFGS-B to minimize smooth p-norm objective in log-space."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())
    
    def obj_and_grad(theta):
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            return 1e10, np.zeros(n)
        a_local = S * exp_theta / total
        c = fft_convolve_self(a_local)
        c_pos = np.maximum(c, 1e-30)
        val = (np.sum(c_pos ** p)) ** (1.0 / p)

        norm_p1 = (np.sum(c_pos ** p)) ** (1.0 - 1.0/p)
        w = c_pos ** (p - 1) / norm_p1
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a_local, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g_a = 2.0 * corr[:n]
        
        # Chain rule: dtheta
        dot = np.dot(g_a, a_local)
        g_theta = g_a * a_local - (a_local / S) * dot
        
        return val, g_theta
    
    start = time.monotonic()
    result = minimize(obj_and_grad, theta, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'maxfun': 500,
                               'ftol': 1e-10, 'gtol': 1e-8})
    
    elapsed = time.monotonic() - start
    theta_final = result.x
    theta_shifted = theta_final - theta_final.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    a_final = S * exp_theta / total
    a_final = normalize_nonneg(a_final, S)
    
    c_final = fft_convolve_self(a_final)
    return a_final, c_final.max()


def coordinate_descent_polish(a, n, S, time_budget, max_sweeps=20):
    """Fast coordinate descent polish."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(999)
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-10:
                continue
            best_val = orig
            best_obj_local = best_obj
            for factor in [0.5, 0.7, 1.3, 1.5, 0.3, 2.0]:
                a[i] = orig * factor
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_obj_local - 1e-12:
                    best_obj_local = obj_check
                    best_val = orig * factor
            a[i] = best_val
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            if c.max() < best_obj - 1e-12:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        if not improved and sweep >= 2:
            break
    
    return best_a, best_obj


def get_initialization(name, n, S, rng):
    """Generate various initializations."""
    i = np.arange(n)
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "cosine_mix":
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * i / n + rng.random() * 2) + \
            0.2 * np.cos(2 * np.pi * 7 * i / n + rng.random() * 2) + \
            0.1 * np.cos(2 * np.pi * 13 * i / n + rng.random() * 2) + \
            0.1 * np.cos(2 * np.pi * 17 * i / n + rng.random() * 2)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "chirp":
        # Linear chirp - has good autocorrelation properties
        a = 1.0 + 0.2 * np.sin(2 * np.pi * (i ** 2) / (2 * n))
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "m_seq":
        # Pseudo-random (m-sequence-like)
        a = 0.5 + 0.5 * rng.random(n)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)

Now, can we beat 2? We need max_b < sum_a^2/(2n)... wait no, we need 2n*max_b/sum_a^2 < 2, i.e., max_b < sum_a^2/n.

But b[n-1] = sum a[j]*a[n-1-j]. By Cauchy-Schwarz:
(sum a[j]*a[n-1-j])^2 <= (sum a[j]^2) * (sum a[n-1-j]^2) = (sum a[j]^2)^2

So b[n-1] <= sum a[j]^2. And by Cauchy-Sch, sum a[j]^2 >= sum_a^2/n.

So b[n-1] can be anywhere from 0 to sum a[j]^2, and sum a[j]^2 >= sum_a^2/n.

The question is: can max_b < sum_a^2/n?

We need ALL b[k] < sum_a^2/n.

Note that sum_{k=0}^{2n-2} b[k] = (sum a)^2 = sum_a^2.
There are 2n-1 terms, so the average is sum_a^2/(2n-1).

If all b[k] < sum_a^2/n, then sum_a^2/(2n-1) < sum_a^2/n, which is true for n > 1. So it's not immediately impossible.

But we also have b[n-1] = sum a[j]*a[n-1-j]. If a is symmetric (a[j] = a[n-1-j]), then b[n-1] = sum a[j]^2 >= sum_a^2/n.

So for symmetric sequences, we can't beat 2. But for asymmetric sequences, b[n-1] could be less than sum_a^2/n!
