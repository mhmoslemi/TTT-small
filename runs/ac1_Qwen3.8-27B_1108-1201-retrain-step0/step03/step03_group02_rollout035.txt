def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)

def multi_peak_gradient(p, c, weights):
    """
    Compute gradient of sum_k weights[k]*c[k] w.r.t. p.
    weights is a vector of length 2n-1 (same as c).

    Computed via FFT cross-correlation.

    fsize = 1 << (len(weights) + n - 1).bit_length()
    W = fft(weights, fsize)
    P = fft(p, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(P)))[:n]

def fienup_step(p, target_max, n):
    """
    One Fienup error reduction step.
    1. Compute conv = p*p.
    2. Peak-shave: conv_target = min(conv, target_max).
    3. Invert: find new_p such that new_p*new_p ≈ conv_target.
    4. Clip and normalize.
    Returns (new_p, conv).
    """
    conv = fft_convolve_self(p)
    # Peak shave with soft transition
    conv_target = np.minimum(conv, target_max)
    # Ensure non-negative (conv can have tiny negative values due to FFT)
    conv_target = np.maximum(conv_target, 0)
    
    # Invert via FFT: FFT(new_p)^2 = FFT(conv_target)
    fsize = 1 << (2 * n - 1).bit_length()
    C = fft(conv_target, fsize)
    P_new = np.sqrt(C)  # complex square root
    new_p = np.real(ifft(P_new))[:n]
    new_p = np.maximum(new_p, 0)
    s = new_p.sum()
    if s < 1e-15:
        return p, conv
    new_p = new_p / s
    return new_p, conv

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Hybrid approach combining:
1. Fienup-style error reduction (global convolution-domain steps via FFT inversion)
2. Projected gradient with adaptive softmax-weighted multi-peak subgradient

4. Final high-precision polish with top-peak focus

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_gradient(p, weights):
    """
    Compute gradient of sum_k weights[k]*c[k] w.r.t. p, where c = p*p (full conv).
    d(c[k])/dp[m] = 2*p[k-m] (if 0<=k-m<n), so:
    grad[m] = 2 * sum_k weights[k] * p[k-m]
    Computed via FFT cross-correlation.
    """
    n = len(p)
    fsize = 1 << (len(weights) + n - 1).bit_length()
    W = fft(weights, fsize)
    P = fft(p, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(P)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights over the convolution values."""
    c_safe = c - c.max()
    exp_bc = np.exp(beta * c_safe)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.ones(len(c)) / len(c)
    return exp_bc / total


def fienup_invert(c_target, n):
    """
    Invert a target convolution to find a sequence p such that p*p ≈ c_target.
    Uses FFT: FFT(p)^2 = FFT(c_target), so p = IFFT(sqrt(FFT(c_target))).
    """
    fsize = 1 << (2 * n - 1).bit_length()
    C = fft(c_target, fsize)
    # Complex square root
    P_new = np.sqrt(C)
    new_p = np.real(ifft(P_new))[:n]
    return new_p


def objective_from_conv(c_max, n, S):
    """Compute objective from max convolution value, n, and sum S."""
    if S < 1e-15:
        return np.inf
    return 2.0 * n * c_max / (S * S)

                top_k_start=1, top_k_end=64)
            a_fin3 = scale_to_target(p_fin3b, len(p_fin3b))
            c_fin3 = fft_convolve_self(a_fin3)
            obj_fin3 = objective_from_conv(c_fin3.max(), len(a_fin3), a_fin3.sum())
            if obj_fin3 < global_best:
                global_best = obj_fin3
                global_a = a_fin3.copy()
                print("  Fine gradient: obj=%.6f" % obj_fin3)

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Final: length=%d, internal=%.6f, verified=%s" %
          (len(final_list), global_best,
           "%.6f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        fallback = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in fallback]

    return final_list
