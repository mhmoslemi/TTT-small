Minimize 2*n*max_conv/sum^2 over non-negative sequences using
adaptive FISTA with LSE smoothing, multi-start exploration,
and targeted polishing.

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def exact_subgradient(a, c):
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad

def topk_lse_gradient(a, c, beta, topk=64):
    """LSE gradient using only top-k lags for efficiency."""
    n = len(a)
    k = min(topk, len(c))
    idx = np.argsort(c)[::-1][:k]
    cm = c[idx].max()
    e = np.exp(beta * (c[idx] - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w_full = np.zeros(len(c))
    w_full[idx] = e / tot
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

                   beta_iters=5000, momentum='nesterov', topk=None):

    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    mu = 0.9  # fixed momentum
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta: exponential schedule
        progress = min(1.0, it / beta_iters)
        beta = beta_init * (beta_final / beta_init) ** progress
        beta = min(beta, 10000.0)

        else:

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:

        # Step size

        x_new = project_simplex(y - step * grad, S)
        
        # Check improvement

        cur = c_new.max()
        if cur < best:

            best_a = x_new.copy()
            last_improve = it

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))

            a = x_new
            t = t_new
        else:  # fixed momentum

            a = x_new

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using
adaptive FISTA with LSE smoothing, top-k gradient approximation
for numerical stability, multi-start exploration, and targeted
polishing with exact subgradient refinement.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Full LSE gradient (use when beta is moderate)."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500.0, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_lse_gradient(a, c, beta, topk=256):
    """LSE gradient using only top-k lags (stable for large beta)."""
    n = len(a)
    k = min(topk, len(c))
    idx = np.argsort(c)[::-1][:k]
    cm = c[idx].max()
    scaled = beta * (c[idx] - cm)
    scaled = np.clip(scaled, -500.0, 500.0)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w_full = np.zeros(len(c))
    w_full[idx] = e / tot
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def topk_subgradient(a, c, k=8):
    """Subgradient averaged over top-k lags."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.exp(50.0 * (vals - vals.max()))
    w = w / w.sum()
    grad = np.zeros(n)
    for ii in range(k):
        lag = int(idx[ii])
        lo = max(0, lag - n + 1)
        hi = min(lag, n - 1)
        Iidx = np.arange(lo, hi + 1)
        grad[Iidx] += w[ii] * 2.0 * a[lag - Iidx]
    return grad

    inits.append(rng.random(n) ** 2 + 0.05)
    inits.append(np.exp(rng.normal(0, 0.5, n)))
    inits.append(np.abs(rng.normal(0, 1, n)) + 0.1)

    # Block structures
    for nblock in [2, 4, 6, 8]:
        block_len = n // nblock
        heights = rng.uniform(0.3, 1.5, nblock)
        block_seq = np.repeat(heights, block_len)
        if len(block_seq) < n:
            block_seq = np.concatenate([block_seq, np.full(n - len(block_seq), heights[-1])])
        inits.append(block_seq[:n])

    # Triangle
    inits.append(1.0 - 2.0 * np.abs(tt - 0.5))

    # V-shaped (minimum in center - helps reduce center conv peak)
    inits.append(0.3 + 1.4 * np.abs(tt - 0.5) * 2.0)

    # W-shaped
    inits.append(0.3 + 0.8 * np.abs(np.sin(4 * np.pi * tt)) ** 0.5)

    return [np.maximum(x, 0.001) for x in inits]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    global_best = np.inf
    global_a = None

    # Try to get the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose main length
    if given is not None and 2048 <= len(given) <= 16384:
        main_len = len(given)
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * n)

    print("Starting optimization with n=%d, budget=%.0fs" % (n, budget_s))

    # Collect initializations
    inits = []
    if given is not None:
        inits.append(("given", np.clip(given, 0, 1000).copy()))

    gen_inits = generate_initializations(n, rng)
    for i, g in enumerate(gen_inits):
        inits.append(("gen_%d" % i, g))

    n_inits = len(inits)
    usable_time = max(90.0, deadline - time.monotonic())

    # Phase 1: Diverse exploration (55% of time)
    exploration_time = usable_time * 0.55
    time_weights = np.ones(n_inits)
    if given is not None:
        time_weights[0] = 3.0
    # Give more time to structured inits (indices 1-10)
    for i in range(1, min(11, n_inits)):
        time_weights[i] = 1.5
    time_weights /= time_weights.sum()

    print("=== Phase 1: Exploration (%d inits, %.0fs) ===" % (n_inits, exploration_time))

    p1_start = time.monotonic()
    for i, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - usable_time * 0.40 - 20:
            break
        arr_n = len(arr)
        if arr_n < 128:
            continue
        S_i = np.sqrt(2.0 * arr_n)
        t_alloc = exploration_time * time_weights[i]
        if t_alloc < 5:
            continue

        # Vary parameters for diversity
        seed_offset = i * 7919
        sub_rng = np.random.default_rng(seed + seed_offset)

        beta_init = 5.0 + 3.0 * (i % 5)
        beta_fin = 2000.0 + 2000.0 * (i % 3)
        momentum = 'nesterov' if i % 2 == 0 else 'fixed'
        step_scale = 0.8 + 0.1 * (i % 4)

        a_opt, score = fista_optimize(arr, arr_n, S_i, t_alloc, sub_rng,
                                       beta_init=beta_init,
                                       beta_final=beta_fin,
                                       beta_iters=max(1000, int(t_alloc * 1.5)),
                                       momentum=momentum,
                                       step_scale=step_scale)
        a_norm = normalize_sum(a_opt, arr_n)
        c_chk = fft_convolve_self(a_norm)
        real_score = c_chk.max()

        if real_score < global_best:
            global_best = real_score
            global_a = a_norm.copy()
            elapsed = time.monotonic() - p1_start
            print("  [%s] n=%d: %.6f (elapsed %.0fs)" % (name, arr_n, global_best, elapsed))

    print("=== After exploration: best=%.6f, elapsed=%.0fs ===" % (global_best, time.monotonic() - p1_start))

    # Phase 2: Intensive polishing (35% of time)
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        remaining = deadline - time.monotonic()
        polish_time = remaining * 0.65

        print("=== Phase 2: Polishing (remaining %.0fs) ===" % remaining)

        # Round 1: High-beta FISTA from best
        r1_time = polish_time * 0.35
        if r1_time > 3:
            a1, s1 = fista_optimize(global_a, n_best, S_best, r1_time, rng,
                                     beta_init=50.0, beta_final=10000.0,
                                     beta_iters=3000, momentum='nesterov',
                                     step_scale=1.2)
            a1 = normalize_sum(a1, n_best)
            c1 = fft_convolve_self(a1)
            if c1.max() < global_best:
                global_best = c1.max()
                global_a = a1.copy()
                print("  After polish round 1: %.6f" % global_best)

        # Round 2: Subgradient polishing
        remaining2 = deadline - time.monotonic()
        r2_time = remaining2 * 0.35
        if r2_time > 3:
            a2, s2 = subgradient_polish(global_a, n_best, S_best, r2_time, rng,
                                         topk_lags=24, step_decay=0.9997, mu=0.9)
            a2 = normalize_sum(a2, n_best)
            c2 = fft_convolve_self(a2)
            if c2.max() < global_best:
                global_best = c2.max()
                global_a = a2.copy()
                print("  After subgradient polish: %.6f" % global_best)

        # Round 3: Very high beta FISTA
        remaining3 = deadline - time.monotonic()
        r3_time = remaining3 * 0.50
        if r3_time > 3:
            a3, s3 = fista_optimize(global_a, n_best, S_best, r3_time, rng,
                                     beta_init=200.0, beta_final=30000.0,
                                     beta_iters=2000, momentum='nesterov',
                                     step_scale=1.5, topk_threshold=150)
            a3 = normalize_sum(a3, n_best)
            c3 = fft_convolve_self(a3)
            if c3.max() < global_best:
                global_best = c3.max()
                global_a = a3.copy()
                print("  After high-beta polish: %.6f" % global_best)

    # Phase 3: Try alternative perturbations of best (if time remains)
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("=== Phase 3: Perturbation search (%.0fs) ===" % remaining)

        # Try several random perturbations and short FISTA runs
        for trial in range(5):
            if time.monotonic() > deadline - 15:
                break
            pert = global_a + rng.normal(0, 0.02 * S_best / np.sqrt(n_best), n_best)
            pert = project_simplex(pert, S_best)
            t_trial = min(10.0, (deadline - time.monotonic() - 10) / (5 - trial))
            if t_trial < 2:
                break
            a_p, s_p = fista_optimize(pert, n_best, S_best, t_trial, rng,
                                       beta_init=100.0, beta_final=8000.0,
                                       beta_iters=1500, momentum='nesterov',
                                       step_scale=1.0)
            a_p = normalize_sum(a_p, n_best)
            c_p = fft_convolve_self(a_p)
            if c_p.max() < global_best:
                global_best = c_p.max()
                global_a = a_p.copy()
                print("  Perturb trial %d: %.6f" % (trial, global_best))

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("Verification failed, using fallback")
            global_a = np.ones(2048, dtype=np.float64) * (np.sqrt(2 * 2048) / 2048)

    if global_a is None:
        global_a = np.ones(2048, dtype=np.float64) * (np.sqrt(2 * 2048) / 2048)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(2048, dtype=np.float64) * (np.sqrt(2 * 2048) / 2048)

    print("Returning sequence of length %d, best internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
