"""

Uses anti-palindromic structure, adaptive FISTA with progressive beta,

- Coordinate descent polish on contributing pairs

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lag_gradient(a, c, k):
    """Exact gradient of conv(a,a)[k] w.r.t. a."""
    n = len(a)
    k = int(k)
    g = np.zeros(n)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    if i_min > i_max:
        return g
    indices = np.arange(i_min, i_max + 1)
    partners = k - indices
    g[indices] += a[partners]
    valid = (partners >= 0) & (partners < n) & (partners != indices)
    g[partners[valid]] += a[indices[valid]]
    return g


def smooth_max_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_smooth_gradient(a, c, top_k, temperature=1.0):
    """Gradient using top-k lags with softmax weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices].copy()
    tv_max = top_vals.max()
    inv_temp = 1.0 / max(temperature, 1e-10)
    weights = np.exp((top_vals - tv_max) * inv_temp)
    weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista_progressive(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=2000.0):
    """FISTA with progressive beta and adaptive step size."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = 10.0
    stall = 0
    last_best_time = start

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            stall = 0
            last_best_time = time.monotonic()
        elif time.monotonic() - last_best_time > 10.0:
            stall += 1
            last_best_time = time.monotonic()

        grad = smooth_max_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / max(L, gnorm / S * 10)
        x_new = project_simplex(y - step * grad, S)

        # Backtracking
        c_y = c
        f_y = c_y.max()
        for _bt in range(3):
            c_x = fft_convolve_self(x_new)
            if c_x.max() <= f_y + 1e-6:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if c_x.max() > f_y * 1.001:
            L *= 1.3
        elif it % 50 == 0:
            L = max(L * 0.92, 1.0)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Warm restart on stall
        if stall > 5 and rng.random() < 0.4:
            y = a.copy()
            t_mom = 1.0
            stall = 0
            noise = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)

    return best_local, best_score


def run_waterfill(a_init, n, S, max_time, rng, K_start=1, K_end=100):
    """Water-filling: progressively balance top K lags."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (n * 5.0)
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / max(1, int(max_time * 10)))
        K = int(K_start + (K_end - K_start) * progress)

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()
            stall = 0
        else:
            stall += 1

        # Compute gradient as weighted sum of top K lag gradients
        top_lags = np.argsort(c)[::-1][:K]
        top_vals = c[top_lags]
        # Weight by how much each exceeds the median of top K
        med = np.median(top_vals)
        excess = np.maximum(top_vals - med, 0) + top_vals.max() * 0.01
        weights = excess / excess.sum()

        grad = np.zeros(n)
        for j, lag in enumerate(top_lags):
            grad += weights[j] * lag_gradient(a, c, int(lag))

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step
        step = 2.0 * S / (n * max(gnorm / S, 0.1) * (1.0 + 0.001 * it))
        x_new = project_simplex(a - step * grad, S)

        # Check if we improved
        if it % 5 == 0:
            c_new = fft_convolve_self(x_new)
            if c_new.max() > cur * 1.005:
                step *= 0.6
                x_new = project_simplex(a - step * grad, S)

        a = x_new

        # Periodic perturbation
        if stall > 20 and rng.random() < 0.3:
            noise = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)
            stall = 0
            step = S / (n * 5.0)

    c_final = fft_convolve_self(a)
    if c_final.max() < best_score:
        best_score = c_final.max()
        best_local = a.copy()
    return best_local, best_score


def run_peak_shave(a_init, n, S, max_time, rng):
    """Directly shave the peak by targeting contributing pairs."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    scale = 0.02 * S / n

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()

        # Get top 5 lags
        top_lags = np.argsort(c)[::-1][:5]

        for lag in top_lags:
            lag = int(lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue
            indices = np.arange(i_min, i_max + 1)
            partners = lag - indices
            products = a[indices] * a[partners]
            # Find the pair with maximum product
            max_idx = np.argmax(products)
            idx_hi = indices[max_idx]
            pidx_hi = partners[max_idx]

            # Try to reduce the larger element of the pair
            if a[idx_hi] >= a[pidx_hi]:
                reduce_idx = idx_hi
            else:
                reduce_idx = pidx_hi

            # Find a low element to add to (not contributing to top lags)
            a_median = np.median(a)
            low_samples = np.where(a < a_median * 0.7)[0]
            if len(low_samples) == 0:
                low_samples = np.where(a < a_median)[0]
            if len(low_samples) == 0:
                continue

            add_idx = low_samples[rng.integers(len(low_samples))]

            # Try the swap
            delta = scale * a[reduce_idx]
            a_trial = a.copy()
            a_trial[reduce_idx] -= delta
            a_trial[add_idx] += delta
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial * (S / a_trial.sum()) if a_trial.sum() > 0 else a

            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur - 1e-10:
                a = a_trial

        # Occasionally try the opposite: increase a contributing element
        if it % 10 == 0 and rng.random() < 0.2:
            lag = int(top_lags[0])
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            indices = np.arange(i_min, i_max + 1)
            partners = lag - indices
            # Try reducing a non-peak contributor elsewhere
            all_lags_contrib = np.zeros(n)
            for lag2 in top_lags:
                lag2 = int(lag2)
                i2_min = max(0, lag2 - (n - 1))
                i2_max = min(lag2, n - 1)
                if i2_min <= i2_max:
                    idx2 = np.arange(i2_min, i2_max + 1)
                    all_lags_contrib[idx2] += a[lag2 - idx2] * 0.5
                    p2 = lag2 - idx2
                    valid2 = (p2 >= 0) & (p2 < n)
                    all_lags_contrib[p2[valid2]] += a[idx2[valid2]] * 0.5

            safe_indices = np.where(all_lags_contrib < np.median(all_lags_contrib) * 0.5)[0]
            if len(safe_indices) > 0:
                src = safe_indices[rng.integers(len(safe_indices))]
                dst = np.argmax(a)
                delta = 0.01 * S / n
                a_trial = a.copy()
                a_trial[src] += delta
                a_trial[dst] -= delta
                a_trial = np.maximum(a_trial, 0.0)
                if a_trial.sum() > 0:
                    a_trial = a_trial * (S / a_trial.sum())
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < best_score:
                    a = a_trial

    return best_local, best_score


def run_coord_descent(a_init, n, S, max_time, rng):
    """Coordinate descent: optimize pairs of coordinates."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    c = fft_convolve_self(a)
    best_score = c.max()

    # Precompute which lags each coordinate affects (top lags only)
    top_K = 10
    top_lags = np.argsort(c)[::-1)[:top_K] if False else np.argsort(c)[::-1][:top_K]

    while time.monotonic() - start < max_time:
        # Randomly pick a coordinate
        i = rng.integers(n)
        # Try modifying a[i]
        a_i = a[i]
        # The change in conv[k] when a[i] changes by delta is:
        # For each lag k, the change is delta * (a[k-i] + a[i-k]) (if valid)
        # We want to find delta that reduces the max
        # Binary search on delta in [-a_i, a_i] (keeping a[i] >= 0)
        # Try a few candidate deltas
        candidates = [0.5 * a_i, -0.5 * a_i, 0.1 * S / n, -0.1 * S / n,
                      0.01 * S / n, -0.01 * S / n]
        best_delta = 0.0
        best_trial_score = best_score + 1e-10  # Start with no improvement

        for delta in candidates:
            new_val = a_i + delta
            if new_val < 0:
                delta = -a_i + 0.01 * 0.01 * S / n
                new_val = max(0, delta)
            if new_val > 1000:
                continue
            trial = a.copy()
            trial[i] = new_val
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 0.01:
                continue
            trial = trial * (S / ts)
            c_trial = fft_convolve_self(trial)
            if c_trial.max() < best_trial_score:
                best_trial_score = c_trial.max()
                best_delta = delta

        if best_delta != 0:
            a[i] = max(0, a[i] + best_delta)
            a = np.maximum(a, 0.0)
            if a.sum() > 0:
                a = a * (S / a.sum())
            c = fft_convolve_self(a)
            if c.max() < best_score - 1e-12:
                best_score = c.max()
                best_local = a.copy()

    return best_local, best_score


def make_anti_palinit(n, rng, S, mode=0):
    """Anti-palindromic initialization with various internal structures."""
    half = n // 2
    tt = np.arange(half, dtype=np.float64) / half

    if mode == 0:
        # Smooth cosine
        h = 0.5 + 0.4 * np.cos(2 * np.pi * rng.random() * 3 * tt + rng.random() * np.pi)
    elif mode == 1:
        # Sawtooth
        h = 0.3 + 0.5 * (tt + rng.random() * 0.3) % 1.0
    elif mode == 2:
        # Gaussian bump
        center = rng.random()
        width = 0.1 + 0.2 * rng.random()
        h = 0.2 + 0.8 * np.exp(-0.5 * ((tt - center) / width) ** 2)
    elif mode == 3:
        # Multi-frequency
        h = 0.5
        for _ in range(3):
            f = rng.integers(1, 8)
            p = rng.random() * 2 * np.pi
            h += 0.15 * np.cos(2 * np.pi * f * tt + p)
    elif mode == 4:
        # Step-like
        h = np.where(tt < rng.random(), 0.8, 0.2)
        h = h + 0.1 * rng.random(half)
    else:
        h = 0.3 + 0.6 * rng.random(half)

    mu = S / half
    a = np.zeros(n)
    a[:half] = h * (mu / h.mean())
    for i in range(half):
        a[n - 1 - i] = mu - a[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_periodic_anti(n, rng, S, period=8):
    """Periodic anti-palindromic: repeat a small anti-palindromic block."""
    # Create a small anti-palindromic block of length `period`

    w = np.maximum(w, 0.01)

    # Tile to length n
    reps = n // period
    a = np.tile(w, reps)
    if n % period != 0:
        a = np.concatenate([a, w[:n % period]])
    a = a[:n]
    # Make it anti-palindromic
    c = a[0] + a[n - 1]  # approx
    for i in range(n // 2):
        pair_sum = a[i] + a[n - 1 - i]
        target = a.mean() * 2 / max(1, 1)  # just normalize
    # Simpler: just normalize
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum()) if a.sum() > 0 else np.ones(n) * S / n
    return a


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def objective_value(a):
    """Compute the objective 2*n*max_b/(sum_a)^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses multi-phase approach: exploration with diverse inits, refinement with
    water-filling, and final polish with peak shaving and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)

    global_best_obj = np.inf
    global_a = None
    global_n = 0

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 32 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            parent = np.clip(given, 0, 1000)
            print("Loaded parent of length %d" % len(parent))
    except Exception:
        pass

    # Primary length
    if parent is not None:
        primary_n = len(parent)
    else:
        primary_n = 8192

    print("Primary length: %d, total budget: %.0fs" % (primary_n, budget_s))
    t_start = time.monotonic()

    S = np.sqrt(2.0 * primary_n)

    # === Generate diverse initializations ===
    inits = []

    # Parent (if available) - most important
    if parent is not None and len(parent) == primary_n:
        inits.append(("parent", safe_normalize(parent, primary_n)))
    elif parent is not None:
        # Resize parent
        from numpy import interp
        tt_old = np.linspace(0, 1, len(parent))
        tt_new = np.linspace(0, 1, primary_n)
        p_resized = interp(tt_new, tt_old, parent)
        inits.append(("parent_resized", safe_normalize(p_resized, primary_n)))

    # Anti-palindromic with different modes
    for mode in range(5):
        inits.append(("anti_%d" % mode, make_anti_palinit(primary_n, rng, S, mode)))

    # Periodic anti
    inits.append(("per_anti", make_periodic_anti(primary_n, rng, S, period=8)))
    inits.append(("per_anti16", make_periodic_anti(primary_n, rng, S, period=16)))

    # Random
    rand_a = 0.2 + 0.8 * rng.random(primary_n)
    inits.append(("random", rand_a * (S / rand_a.sum())))

    print("Generated %d initializations" % len(inits))

    # === Phase 1: FISTA from each init (short time each) ===
    n_inits = len(inits)
    phase1_total = (deadline - time.monotonic()) * 0.45
    per_init = max(5.0, phase1_total / n_inits)

    print("=== Phase 1: FISTA exploration (%.0f sec each) ===" % per_init)

    phase1_best = None
    phase1_best_obj = np.inf

    for name, arr in inits:
        if time.monotonic() > deadline - 60:
            break
        n = len(arr)
        S_local = np.sqrt(2.0 * n)
        t0 = time.monotonic()

        a_f, s_f = run_fista_progressive(arr, n, S_local, per_init, rng,
                                           beta_init=5.0, beta_max=500.0)
        a_f = safe_normalize(a_f, n)
        obj_f = objective_value(a_f)
        elapsed = time.monotonic() - t0

        if obj_f < phase1_best_obj:
            phase1_best_obj = obj_f
            phase1_best = a_f.copy()
            print("  Phase1 [%s] n=%d: obj=%.6f (%.1fs)" % (name, n, obj_f, elapsed))

        if obj_f < global_best_obj:
            global_best_obj = obj_f
            global_a = a_f.copy()
            global_n = n

    print("=== After Phase 1: best_obj=%.6f ===" % global_best_obj)

    # === Phase 2: Water-filling refinement on best ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 30:
        n = len(global_a)
        S_local = np.sqrt(2.0 * n)
        seg2 = remaining * 0.30
        print("=== Phase 2: Water-filling (%.0f sec, n=%d) ===" % (seg2, n))

        a_wf, s_wf = run_waterfill(global_a, n, S_local, seg2, rng,
                                    K_start=1, K_end=80)
        a_wf = safe_normalize(a_wf, n)
        obj_wf = objective_value(a_wf)
        if obj_wf < global_best_obj:
            global_best_obj = obj_wf
            global_a = a_wf.copy()
            global_n = n
            print("  WaterFill: obj=%.6f" % obj_wf)
        else:
            print("  WaterFill: obj=%.6f (no improvement)" % obj_wf)

    # === Phase 3: Peak shaving ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 20:
        n = len(global_a)
        S_local = np.sqrt(2.0 * n)
        seg3 = remaining * 0.25
        print("=== Phase 3: Peak shaving (%.0f sec) ===" % seg3)

        a_ps, s_ps = run_peak_shave(global_a, n, S_local, seg3, rng)
        a_ps = safe_normalize(a_ps, n)
        obj_ps = objective_value(a_ps)
        if obj_ps < global_best_obj:
            global_best_obj = obj_ps
            global_a = a_ps.copy()
            global_n = n
            print("  PeakShave: obj=%.6f" % obj_ps)
        else:
            print("  PeakShave: obj=%.6f (no improvement)" % obj_ps)

    # === Phase 4: Coordinate descent polish ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 15:
        n = len(global_a)
        S_local = np.sqrt(2.0 * n)
        seg4 = remaining * 0.20
        print("=== Phase 4: Coord descent (%.0f sec) ===" % seg4)

        a_cd, s_cd = run_coord_descent(global_a, n, S_local, seg4, rng)
        a_cd = safe_normalize(a_cd, n)
        obj_cd = objective_value(a_cd)
        if obj_cd < global_best_obj:
            global_best_obj = obj_cd
            global_a = a_cd.copy()
            global_n = n
            print("  CoordDesc: obj=%.6f" % obj_cd)
        else:
            print("  CoordDesc: obj=%.6f (no improvement)" % obj_cd)

    # === Phase 5: Final FISTA polish with high beta ===
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 10:
        n = len(global_a)
        S_local = np.sqrt(2.0 * n)
        seg5 = remaining * 0.40
        print("=== Phase 5: Final FISTA polish (%.0f sec) ===" % seg5)

        a_pf, s_pf = run_fista_progressive(global_a, n, S_local, seg5, rng,
                                            beta_init=50.0, beta_max=3000.0)
        a_pf = safe_normalize(a_pf, n)
        obj_pf = objective_value(a_pf)
        if obj_pf < global_best_obj:
            global_best_obj = obj_pf
            global_a = a_pf.copy()
            global_n = n
            print("  FinalFISTA: obj=%.6f" % obj_pf)
        else:
            print("  FinalFISTA: obj=%.6f (no improvement)" % obj_pf)

        # One more water-fill
        remaining2 = deadline - time.monotonic()
        if remaining2 > 5:
            a_wf2, _ = run_waterfill(global_a, n, S_local, remaining2 * 0.6, rng,
                                      K_start=5, K_end=200)
            a_wf2 = safe_normalize(a_wf2, n)
            obj_wf2 = objective_value(a_wf2)
            if obj_wf2 < global_best_obj:
                global_best_obj = obj_wf2
                global_a = a_wf2.copy()
                global_n = n
                print("  FinalWaterFill: obj=%.6f" % obj_wf2)

    # === Fallback ===
    if global_a is None:
        global_a = np.ones(4096, dtype=np.float64)
        global_n = 4096

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64) * 1.0
        global_n = 1024

    # === Verification ===
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Verification: evaluate=%.6f, internal=%.6f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best_obj, global_n))

    if not np.isfinite(ver):
        print("  WARNING: evaluate rejected, using safe fallback")
        global_a = np.ones(2048, dtype=np.float64) * 1.0
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback evaluate: %.6f" % (ver if np.isfinite(ver) else float('inf')))

    print("=== FINAL: %.6f, length=%d ===" % (ver if np.isfinite(ver) else float('inf'), len(global_a)))
    return final_list
