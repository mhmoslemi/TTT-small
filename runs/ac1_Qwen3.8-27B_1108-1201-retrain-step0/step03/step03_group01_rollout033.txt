"""
Minimize 2*n*max_conv/sum^2 using FISTA with adaptive softmax smoothing + spectral kicks + coordinate polish.

Key differences from previous approach:
1. Softmax-smoothed max with adaptive beta (vs fixed top-k)
2. Backtracking line search for step size
3. Spectral "kicks" using a novel phase-randomization technique
4. Final coordinate descent polish
5. Multi-length exploration (try n=8192 and n=16384)

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def full_convolve(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_w(c, beta):
    """Compute softmax weights over the convolution values."""
    c_shifted = c - c.max()
    exp_c = np.exp(beta * c_shifted)
    total = exp_c.sum()
    if total < 1e-300:
        return np.ones(len(c)) / len(c)
    return exp_c / total


def compute_grad(a, p, fsize=None):
    """
    Compute gradient of sum_k p[k]*c[k] with respect to a.
    c[k] = sum_j a[j]*a[k-j]
    grad[i] = sum_k p[k]*(a[k-i] + a[i-k]) (approx, ignoring diagonal)
    = 2 * cross_correlation(p, a)[i]
    """
    n = len(a)
    if fsize is None:
        fsize = 1 << (2 * n - 1).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    # cross-correlation (p ⋆ a)[i] = IFFT(conj(P) * A)[:n]
    # For real signals: real part of this
    grad = 2.0 * np.real(ifft(np.conj(P) * A))[:n]
    return grad


def fista_softmax(a_init, n, S, max_time, rng, beta_start=20.0, beta_end=500.0,
                   beta_growth=1.05, verbose=True):
    """
    FISTA with softmax-smoothed max objective and increasing beta.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    beta = beta_start
    fsize = 1 << (2 * n - 1).bit_length()
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 1.0 / (2.0 * beta)
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = full_convolve(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            if verbose and it % 5000 == 0:
                print("    FISTA it=%d, beta=%.1f, score=%.6f" % (it, beta, best_score))
        
        # Compute softmax-smoothed gradient
        p = softmax_w(c, beta)
        grad = compute_grad(y, p, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search (simplified: use fixed step with decay)
        # L is approximately 2*beta, but effective L on simplex is smaller
        step = 1.0 / (2.0 * beta) * min(1.0, 50.0 / it)
        
        x_new = project_simplex(y - step * grad, S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Renormalize y to maintain sum = S (for stability)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
        else:
            y = y * (S / y_sum)
        a = x_new
        t = t_new
        
        # Increase beta
        beta *= beta_growth
        if beta > beta_end:
            beta = beta_end
        
        # Occasional perturbation
        if it % 5000 == 0 and rng.random() < 0.2:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            beta = max(beta_start, beta * 0.5)  # Reset beta on perturbation
    
    return best_local, best_score


def spectral_kick(a, N, rng, n_kicks=10, alpha=0.3):
    """
    Apply spectral randomization kicks to escape local minima.
    Randomize the phase of the FFT at select frequencies, then project back.
    """
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score = full_convolve(a).max()
    
    for _ in range(n_kicks):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        phase = np.angle(A)
        mag = np.abs(A)
        
        # Randomize a subset of phases
        n_randomize = n // 4
        indices = rng.choice(N, size=n_randomize, replace=False)
        phase[indices] = rng.uniform(0, 2 * np.pi, n_randomize)
        
        # Also slightly randomize magnitudes
        mag[indices] *= (1 + rng.normal(0, 0.1, n_randomize))
        
        A_new = mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new + 0.01 * S / n, S)
        
        c = full_convolve(a_new)
        if c.max() < best_score:
            best_score = c.max()
            best_a = a_new.copy()
    
    return best_a, best_score


def coordinate_polish(a_init, n, S, max_time, rng, n_rounds=5):
    """
    Coordinate descent polish: iteratively update each element to reduce the peak.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = full_convolve(a).max()
    best_a = a.copy()
    
    for rnd in range(n_rounds):
        if time.monotonic() - start > max_time:
            break
        
        # Random order for coordinate updates
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > max_time:
                break
            
            # Try a few candidate values for a[i]
            old_val = a[i]
            # The optimal a[i] for minimizing max(c) subject to sum = S
            # is roughly: we want to reduce the contribution to the peak lags
            # Simple heuristic: try a few values around old_val
            candidates = [old_val * 0.8, old_val * 0.9, old_val, old_val * 1.1, old_val * 1.2, 
                         S / n, 0.0, old_val * 0.5]
            
            best_val = old_val
            best_local_score = np.inf
            
            for cv in candidates:
                if cv < 0 or cv > 1000:
                    continue
                a[i] = cv
                # Renormalize
                a_sum = a.sum()
                if a_sum < 1e-15:
                    a[i] = old_val
                    continue
                # Quick score check (expensive but we're at the end)
                # Use a mini-convolution: only check the top few lags
                # For speed, just check the current peak
                c = full_convolve(a)
                sc = c.max()
                if sc < best_local_score:
                    best_local_score = sc
                    best_val = cv
            
            a[i] = best_val
            a_sum = a.sum()
            if a_sum < 1e-15:
                a[i] = old_val
            # Don't renormalize at each step (would be too slow)
        
        # Renormalize after each round
        a_sum = a.sum()
        if a_sum > 1e-15:
            a = a * (S / a_sum)
        
        c = full_convolve(a)
        if c.max() < best_score:
            best_score = c.max()
            best_a = a.copy()
            print("    Coord polish round %d: %.6f" % (rnd, best_score))
    
    return best_a, best_score


def spectral_flatten(a, N, n_iter, alpha, rng):
    """
    Spectral flattening: iteratively reduce the dynamic range of |FFT(a)|^2
    to flatten the autocorrelation.
    """
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score = full_convolve(a).max()
    
    for it in range(n_iter):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = c.max()
        
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        
        # Compute the "excess" at each frequency
        P_mean = P.mean()
        # Reduce peaks in the power spectrum
        P_new = P.copy()
        threshold = P_mean * 1.5
        P_new[P_new > threshold] = threshold + alpha * (P_new[P_new > threshold] - threshold)
        # Or: iterative Bussgang-like approach
        # P_new = P * (1 - alpha * (P - P_mean) / P_mean) for P > P_mean
        
        # Actually, let's use a more principled approach:
        # We want to minimize max(c) = max(IFFT(|A|^2))
        # In the frequency domain, this is equivalent to making |A|^2 as flat as possible
        # subject to the constraint that ifft(A) is non-negative
        
        # Simple approach: reduce the variance of P
        P_std = P.std()
        P_mean_val = P.mean()
        P_target = P_mean_val + 0.5 * P_std * np.sign(P - P_mean_val) * np.minimum(1.0, alpha)
        P_new = np.maximum(P * (1 - alpha) + P_target * alpha, 0)
        
        # Preserve phase, update magnitude
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)
    
    return best_a, best_score
