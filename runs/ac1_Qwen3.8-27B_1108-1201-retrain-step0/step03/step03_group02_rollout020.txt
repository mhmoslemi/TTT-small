import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.zeros(n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Strategy:

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.zeros(n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_weighted(a, c, weights):
    """
    Compute subgradient using given weights w[k] over lags.
    grad[m] = 2 * sum_k w[k] * a[k-m] (for valid indices).
    weights should sum to 1 and have length 2n-1.
    """
    n = len(a)
    fsize = 1 << (len(weights) + n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def grad_softmax(a, c, beta):
    """Subgradient using softmax weighting over lags."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    return fft_grad_weighted(a, c, w)


def grad_toppk(a, c, top_k):
    """Subgradient using top-k active lags (uniform weight among them)."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    w_full = np.zeros(2 * n - 1)
    w_full[top_indices] = 1.0 / top_k
    return fft_grad_weighted(a, c, w_full)


def grad_slack_weighted(a, c, power=2.0):
    """
    Subgradient weighted by slack (mimics LP dual).
    Lags with small slack (close to max) get more weight.
    weight[k] = (c[k] / c_max)^power, normalized.
    """
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    w = (c / c_max) ** power
    w_total = w.sum()
    if w_total < 1e-300:
        return np.zeros(n)
    w = w / w_total
    return fft_grad_weighted(a, c, w)


def objective(a):
    """Compute 2n * max(conv) / sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def fista_optimize(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=300.0,
                   top_k_init=4, top_k_max=256, restart_interval=400,
                   noise_scale=0.003, use_backtrack=True):

    FISTA with adaptive soft-max, backtracking line search, and periodic restarts.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stagnation_count = 0
    last_improvement = best_score
    noise_level = noise_scale

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))
        k = min(top_k_max, top_k_init + it // 100)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stagnation_count = 0
            if cur < last_improvement * 0.999:
                last_improvement = cur
                noise_level = max(0.0001, noise_level * 0.9)
        else:
            stagnation_count += 1

        # Adaptive: if stagnating, increase noise and reset
        if stagnation_count > restart_interval:
            noise = rng.normal(0, noise_level * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
            stagnation_count = 0
            noise_level = min(0.02, noise_level * 1.5)

        # Gradient
        if it < 200:
            grad = grad_toppk(y, c, k)
        else:
            grad = grad_softmax(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size with backtracking
        if use_backtrack:
            step = S / (4.0 * np.sqrt(n))
            # Backtrack up to 30 times
            accepted = False
            for bt in range(30):
                x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
                c_new = fft_convolve_self(x_new)
                if c_new.max() <= cur * 1.001 + 1e-15:
                    accepted = True
                    break
                step *= 0.5
            if not accepted:

                x_new = project_simplex(y - 1e-10 * grad / max(gnorm, 1e-15), S)
        else:

        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Check if y has valid sum
        if y.sum() < 0.001:
            y = a.copy()

    return best_local, best_score

        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Slack-weighted gradient (higher power = more focused on tightest constraints)
        p = power + 0.5 * min(it / 500.0, 3.0)

    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        k = min(512, 64 + it // 50)
        grad = grad_toppk(y, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Very small step
        step = S / (16.0 * np.sqrt(n) * (1.0 + it / 5000.0))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        c_new = fft_convolve_self(x_new)
        if c_new.max() > cur:
            # Backtrack
            step *= 0.25
            x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        y = x_new

    return best_local, best_score


def generate_parametric(n, alpha, beta, gamma, kind="gauss"):
    """Generate a parametric asymmetric sequence."""
    t = np.linspace(0, 1, n, endpoint=False)
    if kind == "gauss":
        # Skewed Gaussian: a(t) = exp(-alpha*(t-beta)^2 + gamma*t)
        a = np.exp(-alpha * (t - beta) ** 2 + gamma * t)
    elif kind == "exp_ramp":
        # Exponential ramp: a(t) = exp(alpha * (t - beta) + gamma * t^2)
        a = np.exp(alpha * (t - beta) + gamma * t * t)
    elif kind == "power":
        # Power law with offset: a(t) = (t + 0.01)^alpha * exp(gamma * (1-t)*beta)
        a = (t + 0.01) ** alpha * np.exp(gamma * (1.0 - t) * beta)
    elif kind == "bimodal":
        # Two peaks at different heights: a(t) = exp(-alpha*(t-beta1)^2) + gamma*exp(-alpha*(t-beta2)^2)
        b1 = min(beta, 0.4)
        b2 = max(beta + 0.3, 0.7)
        a = np.exp(-alpha * (t - b1) ** 2) + gamma * np.exp(-alpha * (t - b2) ** 2)
    else:
        a = np.ones(n)
    return np.maximum(a, 0.01 * a.max() if a.max() > 0 else 0.1)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses multi-phase optimization: parametric sweep, FISTA, LP-inspired refinement, micro-polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # --- Load parent sequence if available ---
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
            parent = parent / parent.sum()
    except Exception:
        pass

    # Determine sequence length
    if parent is not None and len(parent) >= 1024:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * float(n))  # Normalization (objective is scale-invariant)

    # --- Phase 1: Parametric sweep (10% time) ---
    phase1_time = (deadline - time.monotonic()) * 0.10
    print("=== Phase 1: Parametric sweep (n=%d, %.1f sec) ===" % (n, phase1_time))

    candidates = []

    if parent is not None:
        p = parent * S
        p = np.clip(p, 0, 1000)
        obj_p = 2.0 * n * fft_convolve_self(p).max() / (p.sum() ** 2)
        candidates.append(("parent", p, obj_p))
        print("  Parent: obj=%.6f" % obj_p)

    # Parametric families
    t0 = time.monotonic()
    alphas = [2.0, 4.0, 6.0, 8.0]
    betas = [0.1, 0.2, 0.3, 0.4, 0.6, 0.7, 0.8]
    gammas = [-3.0, -2.0, -1.0, 0.0, 1.0, 2.0, 3.0]

    for alpha in alphas:
        for beta in betas:
            for gamma in gammas:
                if time.monotonic() - t0 > phase1_time * 0.6:
                    break
                a = generate_parametric(n, alpha, beta, gamma, "gauss")
                a = a * (S / a.sum())
                a = np.clip(a, 0, 1000)
                c = fft_convolve_self(a)
                obj = 2.0 * n * c.max() / (a.sum() ** 2)
                if obj < 2.0:  # Only keep interesting candidates (better than uniform)
                    candidates.append(("gauss_a%.0f_b%.1f_g%.0f" % (alpha, beta, gamma), a, obj))

            if time.monotonic() - t0 > phase1_time * 0.6:
                break
        if time.monotonic() - t0 > phase1_time * 0.6:
            break

    # Also try bimodal
    t0 = time.monotonic()
    for alpha in [4.0, 8.0, 12.0]:
        for beta in [0.3, 0.4, 0.5]:
            for gamma in [0.3, 0.5, 0.7, 1.0]:
                if time.monotonic() - t0 > phase1_time * 0.3:
                    break
                a = generate_parametric(n, alpha, beta, gamma, "bimodal")
                a = a * (S / a.sum())
                a = np.clip(a, 0, 1000)
                c = fft_convolve_self(a)
                obj = 2.0 * n * c.max() / (a.sum() ** 2)
                if obj < 2.0:
                    candidates.append(("bimodal_a%.0f_b%.1f_g%.1f" % (alpha, beta, gamma), a, obj))
            if time.monotonic() - t0 > phase1_time * 0.3:
                break
        if time.monotonic() - t0 > phase1_time * 0.3:
            break

    print("  Found %d candidates with obj < 2.0" % len([c for c in candidates if c[2] < 2.0]))

    # Sort by objective and pick top few
    candidates.sort(key=lambda x: x[2])
    for name, a, obj in candidates[:5]:
        print("  %s: obj=%.6f" % (name, obj))

    # Start from the best candidate
    if candidates:
        best_start_name, best_start_a, best_start_obj = candidates[0]
    elif parent is not None:
        best_start_name = "parent"
        best_start_a = parent * S
        best_start_obj = 2.0
    else:
        best_start_name = "uniform"
        best_start_a = np.ones(n) * S / n
        best_start_obj = 2.0

    global_best = best_start_obj
    global_a = best_start_a.copy()

    # --- Phase 2: Main FISTA optimization (55% time) ---
    remaining = deadline - time.monotonic()
    phase2_time = remaining * 0.55
    print("=== Phase 2: FISTA optimization (%.1f sec) ===" % phase2_time)

    # Run FISTA from the best starting point
    a2, score2 = fista_optimize(
        best_start_a, n, S, phase2_time * 0.7, rng,
        beta_init=5.0, beta_max=200.0,
        top_k_init=4, top_k_max=128,
        restart_interval=300, noise_scale=0.003,
        use_backtrack=True
    )
    a2 = np.clip(a2, 0, 1000)
    s2 = a2.sum()
    if s2 < 0.01:
        a2 = global_a
    obj2 = 2.0 * n * fft_convolve_self(a2).max() / (s2 * s2)
    print("  FISTA result: obj=%.6f" % obj2)
    if obj2 < global_best:
        global_best = obj2
        global_a = a2.copy()

    # Run FISTA from second-best starting point (if different)
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(candidates) > 1 and candidates[1][1] is not best_start_a:
        phase2b_time = remaining * 0.3
        a2b, score2b = fista_optimize(
            candidates[1][1], n, S, phase2b_time, rng,
            beta_init=10.0, beta_max=300.0,
            top_k_init=8, top_k_max=256,
            restart_interval=200, noise_scale=0.002,
            use_backtrack=True
        )
        a2b = np.clip(a2b, 0, 1000)
        s2b = a2b.sum()
        if s2b >= 0.01:
            obj2b = 2.0 * n * fft_convolve_self(a2b).max() / (s2b * s2b)
            print("  FISTA (2nd start): obj=%.6f" % obj2b)
            if obj2b < global_best:
                global_best = obj2b
                global_a = a2b.copy()

    # --- Phase 3: Slack-weighted refinement (25% time) ---
    remaining = deadline - time.monotonic()
    phase3_time = remaining * 0.45
    if phase3_time > 5:
        print("=== Phase 3: Slack refinement (%.1f sec) ===" % phase3_time)
        a3, score3 = slack_refinement(
            global_a, n, S, phase3_time, rng, power=2.0
        )
        a3 = np.clip(a3, 0, 1000)
        s3 = a3.sum()
        if s3 >= 0.01:
            obj3 = 2.0 * n * fft_convolve_self(a3).max() / (s3 * s3)
            print("  Slack refinement: obj=%.6f" % obj3)
            if obj3 < global_best:
                global_best = obj3
                global_a = a3.copy()

        # Second round of slack refinement with higher power
        remaining = deadline - time.monotonic()
        if remaining > 10:
            phase3b_time = remaining * 0.6
            a3b, score3b = slack_refinement(
                global_a, n, S, phase3b_time, rng, power=4.0
            )
            a3b = np.clip(a3b, 0, 1000)
            s3b = a3b.sum()
            if s3b >= 0.01:
                obj3b = 2.0 * n * fft_convolve_self(a3b).max() / (s3b * s3b)
                print("  Slack refinement (high power): obj=%.6f" % obj3b)
                if obj3b < global_best:
                    global_best = obj3b
                    global_a = a3b.copy()

    # --- Phase 4: Micro-polish (remaining time) ---
    remaining = deadline - time.monotonic()
    if remaining > 5:
        print("=== Phase 4: Micro-polish (%.1f sec) ===" % remaining)
        a4, score4 = micro_polish(global_a, n, S, remaining, rng)
        a4 = np.clip(a4, 0, 1000)
        s4 = a4.sum()
        if s4 >= 0.01:
            obj4 = 2.0 * n * fft_convolve_self(a4).max() / (s4 * s4)
            print("  Micro-polish: obj=%.6f" % obj4)
            if obj4 < global_best:
                global_best = obj4
                global_a = a4.copy()

    # --- Final verification ---
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        global_a = np.ones(n, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)

    print("Returning sequence of length %d, internal score=%.6f, verified=%.6f" %
          (len(final_list), global_best, ver if np.isfinite(ver) else -1))
    return final_list
