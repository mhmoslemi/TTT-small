"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using active-set QP with CG iteration.

Key innovations over previous approach:
- Proper QP formulation with adaptive active set
- CG solver for the QP (exploiting Hankel structure)
- Adaptive regularization parameter
- Anti-palindromic structural initialization
- Multi-length exploration

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        return np.zeros(n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta_val, 0.0)


def apply_hessian(x, active_lags, mu, n):
    """Compute H @ x where H = (1/mu)*I + (2/m)*sum_{k in AS} K_k.
    K_k has (K_k)_{i,j} = 1 if i+j = k (and indices in [0,n-1])."""
    m = len(active_lags)
    result = x / mu
    for k in active_lags:
        k = int(k)
        i_low = max(0, k - n + 1)
        i_high = min(k, n - 1)
        if i_low > i_high:
            continue
        # (Hx)[i_low:i_high+1] += (2/m) * x[k-i_low : k-i_high-1 : -1]
        result[i_low:i_high+1] += (2.0 / m) * x[k - i_low : k - i_high - 1 : -1]
    return result


def cg_solve(H_func, b, n, max_iter=50, tol=1e-10):
    """Conjugate gradient for solving H x = b, where H is symmetric positive definite.
    H_func(x) returns H @ x."""
    x = np.zeros(n, dtype=np.float64)
    r = b.copy()
    p = r.copy()
    rs_old = np.dot(r, r)
    
    for i in range(max_iter):
        if np.sqrt(rs_old) < tol * (np.dot(b, b) + 1e-30):
            break
        Hp = H_func(p)
        alpha = rs_old / (np.dot(p, Hp) + 1e-30)
        x += alpha * p
        r -= alpha * Hp
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * (np.dot(b, b) + 1e-30):
            break
        p = r + (rs_new / (rs_old + 1e-30)) * p
        rs_old = rs_new
    
    return x


def solve_qp(a_old, active_lags, S, mu, n):
    """Solve the active-set QP:
    minimize (1/m)*sum_{k in AS} a^T K_k a + (1/(2*mu))*||a - a_old||^2
    subject to sum(a) = S (non-negativity handled by post-projection)
    
    Returns a_new (before non-negativity projection)."""
    m = len(active_lags)
    
    # H = (2/m)*sum_{k in AS} K_k + (1/mu)*I
    # b = (1/mu) * a_old
    b = a_old / mu
    
    def H_func(x):
        return apply_hessian(x, active_lags, mu, n)
    
    # Solve H x = b
    x1 = cg_solve(H_func, b, n, max_iter=100, tol=1e-12)
    
    # Solve H y = 1
    ones = np.ones(n, dtype=np.float64)
    x2 = cg_solve(H_func, ones, n, max_iter=100, tol=1e-12)
    
    # lambda = (S - 1' x1) / (1' x2)
    sum_x1 = np.sum(x1)
    sum_x2 = np.sum(x2)
    if abs(sum_x2) < 1e-15:
        return a_old.copy()
    lam = (S - sum_x1) / sum_x2
    
    a_new = x1 + lam * x2
    return a_new


def make_anti_palindromic_init(n, rng, S):
    """Create anti-palindromic initialization."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    return a


def make_periodic_init(n, rng, S, num_freqs=4):
    """Periodic initialization with multiple sinusoidal components."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 30)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.5
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a * (S / a.sum())


def make_edge_concentrated_init(n, rng, S):
    """Concentrate mass near edges (exploits anti-palindromic structure)."""
    tt = np.arange(n, dtype=np.float64) / n
    sigma = rng.random() * 0.1 + 0.02
    a = np.exp(-((tt - 0.0) / sigma) ** 2) + np.exp(-((tt - 1.0) / sigma) ** 2)
    a = a + 0.01 * rng.random(n)
    a = np.maximum(a, 0.0)
    return a * (S / a.sum())


def zc_sequence_real(n):
    """Generate real part of a Zadoff-Chu-like sequence as initialization."""
    # Zadoff-Chu: x[k] = exp(-j*pi*m*k*(k+1)/N) for k=0..N-1
    # For real non-negative, take abs or use a modified version
    m = 1
    kk = np.arange(n, dtype=np.float64)
    phase = -np.pi * m * kk * (kk + 1) / n
    # Use |cos| for a non-negative version with good autocorrelation properties
    a = np.abs(np.cos(phase)) + 0.1
    return a


def run_active_set_qp(a_init, n, S, max_time, rng, mu_init=0.1, max_active=30):
    """Main optimization loop using active-set QP with CG."""
    a = np.clip(a_init, 0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    mu = mu_init
    stall_count = 0
    last_improve = np.inf
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        max_c = c.max()
        
        if max_c < best_score:
            best_score = max_c
            best_local = a.copy()
            stall_count = 0
            # Slowly decrease mu for more aggressive steps
            mu = max(mu * 0.995, 0.001)
        else:
            if it % 10 == 0:
                if max_c < last_improve - 1e-12:
                    last_improve = max_c
                    stall_count = 0
                else:
                    stall_count += 1
            # If stalled, increase mu (smaller steps)
            if stall_count > 5:
                mu = min(mu * 1.2, 10.0)
                stall_count = 0
        
        # Determine active set
        # Adaptive threshold
        eps = max(0.001, 0.02 * (1.0 + 0.1 * (stall_count / 10.0)))
        threshold = max_c - eps * max_c
        active_mask = c >= threshold
        active_lags = np.where(active_mask)[0]
        
        # Cap active set size
        if len(active_lags) > max_active:
            active_lags = np.argsort(c)[::-1][:max_active]
        # Ensure at least 2 lags
        if len(active_lags) < 2:
            active_lags = np.argsort(c)[::-1][:2]
        
        # Solve QP
        a_qp = solve_qp(a, active_lags, S, mu, n)
        
        # Project onto non-negativity and sum constraint
        a_new = project_simplex(a_qp, S)
        
        # Evaluate and accept/reject
        c_new = fft_convolve_self(a_new)
        max_new = c_new.max()
        
        if max_new < max_c:
            a = a_new
        else:
            # Try a fraction of the step
            frac = 0.5
            for _ in range(3):
                a_try = project_simplex(a + frac * (a_qp - a), S)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < max_c:
                    a = a_try
                    break
                frac *= 0.5
            else:
                # No improvement, increase mu
                mu = min(mu * 1.5, 10.0)
        
        # Periodic perturbation
        if stall_count > 20 and rng.random() < 0.2:
            noise_scale = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            mu = mu_init
            stall_count = 0
    
    return best_local, best_score


def run_lse_fista(a_init, n, S, max_time, rng, beta_start=5.0, beta_end=200.0):
    """FISTA with LSE smoothing as a backup/complement to the QP method."""
    a = np.clip(a_init, 0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (n * 10.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_time * 100))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        # LSE gradient
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            w = np.zeros(len(c))
            w[c.argmax()] = 1.0
        else:
            w = exp_bc / total
        
        # Gradient via FFT
        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        Y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]
        
        # Adaptive step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 2.0 * S / (n * (1.0 + 0.005 * it) * max(gnorm / (S/n), 0.01))
        
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    
    return best_local, best_score

        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        max_c = c.max()
        if max_c < best_score:
            best_score = max_c
            best_local = a.copy()
        
        # Find the lag with maximum convolution
        top_lag = int(c.argmax())
        
        # Find pairs contributing to this lag
        i_min = max(0, top_lag - n + 1)
        i_max = min(top_lag, n - 1)
        if i_min > i_max:
            break
        
        indices = np.arange(i_min, i_max + 1)
        partners = top_lag - indices
        products = a[indices] * a[partners]
        
        # Try reducing the element with largest contribution
        top_idx = indices[np.argmax(products)]
        partner_idx = top_lag - top_idx
        
        # Try reducing a[top_idx] by transferring to a small element
        if a[top_idx] < 1e-10:
            continue
        
        delta = a[top_idx] * 0.1  # Try reducing by 10%
        
        # Find where to add (element with smallest value)
        small_idx = int(np.argmin(a))
        
        a_try = a.copy()
        a_try[top_idx] -= delta
        a_try[small_idx] += delta
        a_try = np.maximum(a_try, 0.0)
        a_try = a_try * (S / a_try.sum())
        
        c_try = fft_convolve_self(a_try)
        if c_try.max() < max_c:
            a = a_try
        else:
            # Try the partner
            a_try2 = a.copy()
            a_try2[partner_idx] -= a_try2[partner_idx] * 0.1
            if a_try2[partner_idx] < 0:
                a_try2[partner_idx] = 0
            a_try2[small_idx] += delta
            a_try2 = np.maximum(a_try2, 0.0)
            a_try2 = a_try2 * (S / a_try2.sum())
            c_try2 = fft_convolve_self(a_try2)
            if c_try2.max() < max_c:
                a = a_try2
    
    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
