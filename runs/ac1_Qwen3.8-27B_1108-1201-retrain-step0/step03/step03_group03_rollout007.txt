def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.full(n, S / n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def project_anti_palindromic(v, S, n):
    """Project onto {a >= 0, sum(a) = S, a[i] + a[n-1-i] = c}."""
    half = n // 2
    a = np.asarray(v, dtype=np.float64).copy()

def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_exact(a, c, top_k=5, temperature=0.0):
    """
    Exact subgradient of max(conv(a,a)) using top-k lags.
    grad[i] = sum_j w_j * a[k_j - i] for valid indices.
    """
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    
    if temperature > 0:
        scale = temperature
        vals = c[top_indices]
        w = np.exp((vals - vals.max()) / scale)
        w = w / w.sum()
    else:
        w = np.ones(k) / k
    
    grad = np.zeros(n)
    for idx, wi in zip(top_indices, w):
        lag = int(idx)
        # conv[lag][i] contribution: a[i] * a[lag - i]
        # d(conv[lag])/d(a[i]) = a[lag-i] for valid i
        i_min = max(0, lag - (n - 1))
        i_max = min(lag, n - 1)
        if i_min <= i_max:
            indices = np.arange(i_min, i_max + 1)
            partners = lag - indices
            grad[i_min:i_max + 1] += wi * a[partners]
    
    return grad


def compute_gradient_lse(a, c, beta):
    """LSE-based gradient: (1/beta)*log(sum exp(beta*c)) gradient."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

            grad_lse = compute_gradient_lse(y, c, beta)
            k = min(top_k + it // 500, 32)
            grad_exact = compute_gradient_exact(y, c, top_k=k)
            # Normalize both to same scale
            gnorm_lse = np.linalg.norm(grad_lse)
            gnorm_exact = np.linalg.norm(grad_exact)
            if gnorm_lse > 1e-14 and gnorm_exact > 1e-14:
                blend = 0.5 + 0.5 * min(1.0, it / 2000)  # start LSE, move to exact
                grad = blend * (grad_exact / gnorm_exact) + (1 - blend) * (grad_lse / gnorm_lse)
                grad = grad * max(gnorm_lse, gnorm_exact)
            else:
                grad = grad_lse

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        f_y = cur
        x_new = None
        for bt in range(8):
            if use_anti_pal:
                x_candidate = project_anti_palindromic(y - step * grad, S, n)
            else:
                x_candidate = project_simplex(y - step * grad, S)
            if bt == 0:
                x_new = x_candidate
                break
            c_cand = fft_convolve_self(x_candidate)
            f_cand = c_cand.max()
            if f_cand <= f_y + 1e-4:
                x_new = x_candidate
                break
            step *= 0.5

        if x_new is None:
            if use_anti_pal:
                x_new = project_anti_palindromic(y - step * grad, S, n)
            else:
                x_new = project_simplex(y - step * grad, S)

        # Update step for next iteration
        c_x = fft_convolve_self(x_new)
        if c_x.max() > f_y * 1.001:
            step *= 0.8
        elif it % 50 == 0:
            step *= 1.02

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Restart momentum periodically
        if stall_count > 20:
            t_momentum = 1.0
            y = a.copy()
            stall_count = 0
            # Structured perturbation
            if rng.random() < 0.5:
                # Fourier mode perturbation
                freq = rng.integers(2, max(3, n // 100))
                perturbation = np.cos(2 * np.pi * freq * np.arange(n) / n + rng.random() * 2 * np.pi)
                scale = 0.02 * S / n
                if use_anti_pal:
                    y = project_anti_palindromic(y + scale * perturbation, S, n)
                else:
                    y = project_simplex(y + scale * perturbation, S)
            else:
                scale = 0.01 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, scale, n), S)

    return best_local, best_score


def run_pairwise_cd(a_init, n, S, max_time, rng):
    """
    Coordinate descent: for the peak lag, find the pair (i, j) with largest
    contribution a[i]*a[j] and try to reduce it by redistributing mass.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()

        # Find top lags
        top_lags = np.argsort(c)[::-1][:min(5, len(c))]
        
        improved = False
        for lag in top_lags:
            lag = int(lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue
            indices = np.arange(i_min, i_max + 1)
            partners = lag - indices
            products = a[indices] * a[partners]
            
            # Find the top contributing pairs
            top_products = np.argsort(products)[::-1]
            for tp in top_products[:10]:
                i = indices[tp]
                j = partners[tp]
                if i == j:
                    continue
                # Try to reduce a[i]*a[j] by transferring mass
                # Reduce the larger one, add to a random small element
                ai = int(i)
                aj = int(j)
                
                # Find a "donor" element (large) and "recipient" (small)
                # Strategy: reduce max(a[i], a[j]) and add to a small element not at this lag
                if a[ai] >= a[aj]:
                    donor = ai
                    other = aj
                else:
                    donor = aj
                    other = ai
                
                # Find recipient: small element that doesn't contribute much to top lags
                candidate_mask = np.ones(n, dtype=bool)
                candidate_mask[donor] = False
                candidate_mask[other] = False
                # Also exclude elements that pair with large elements at top lags
                for tl in top_lags[:3]:
                    tl = int(tl)
                    # Elements that would create new peaks
                    pm = max(0, tl - donor)
                    pl = min(tl, n - 1 - (n - 1 - tl))  # not quite right, just skip for simplicity
                
                small_elements = np.where(candidate_mask & (a < np.median(a)))[0]
                if len(small_elements) == 0:
                    small_elements = np.where(candidate_mask)[0]
                
                if len(small_elements) == 0:
                    continue
                
                recipient = small_elements[rng.integers(len(small_elements))]
                
                # Transfer: reduce donor, increase recipient
                for frac in [0.01, 0.05, 0.1, 0.2]:
                    transfer = frac * a[donor]
                    if a[donor] - transfer < 0:
                        continue
                    a_new = a.copy()
                    a_new[donor] -= transfer
                    a_new[recipient] += transfer
                    c_new = fft_convolve_self(a_new)
                    if c_new.max() < cur * (1 - 1e-6):
                        a = a_new
                        cur = c_new.max()
                        improved = True
                        break
        
        if not improved and it % 100 == 0:
            # Add small random perturbation
            scale = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, scale, n), S)

    return best_local, best_score


def run_fista_restart(a_init, n, S, max_time, rng):
    """
    FISTA with periodic full restarts and adaptive beta.
    Different from previous: uses exact gradient + periodic anti-palindromic projection.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    # Run multiple FISTA segments with restarts
    n_segments = 5
    seg_time = max_time / n_segments
    
    for seg in range(n_segments):
        seg_start = time.monotonic()
        seg_deadline = min(start + max_time, seg_start + seg_time)
        
        # Alternate between free and anti-palindromic constraint
        use_anti = (seg % 3 == 2)
        beta_start = 5.0 * (seg + 1)
        
        a_seg, s_seg = run_proximal_gradient(
            a, n, S, seg_deadline - time.monotonic(), rng,
            use_anti_pal=use_anti, approach="mixed",
            top_k=8, beta_init=beta_start, beta_max=300.0 + 100 * seg
        )
        
        a_seg = np.clip(a_seg, 0, 1000)
        s_seg_sum = a_seg.sum()
        if s_seg_sum > 0:
            a_seg = a_seg * (S / s_seg_sum)
        
        c_seg = fft_convolve_self(a_seg)
        if c_seg.max() < best_score:
            best_score = c_seg.max()
            best_local = a_seg.copy()
            a = a_seg.copy()
        else:
            # Keep better of current and segment result
            c_cur = fft_convolve_self(a)
            if c_cur.max() < c_seg.max():
                # Keep a
                pass
            else:
                a = a_seg.copy()
    
    return best_local, best_score


def make_structured_init(n, rng, S, method="anti_pal"):
    """Generate diverse structured initializations."""
    if method == "anti_pal":
        half = n // 2
        # Use power-law distribution for the first half
        alpha = rng.random() * 2.0 + 0.5
        tt = np.arange(half, dtype=np.float64) / half
        h = tt ** alpha + 0.1 * (1 - tt) ** alpha + 0.05
        c = S / half if n % 2 == 0 else S / (half + 0.5)
        h = h * (c / 2 / h.mean())  # normalize so mean is c/2
        h = np.clip(h, 0.0, c)
        a = np.zeros(n)
        a[:half] = h
        a[half:] = c - h[::-1]
        if n % 2 == 1:
            a[half] = c / 2
        a = np.maximum(a, 0)
        
    elif method == "bimodal":
        # Two peaks
        center1 = n // 3
        center2 = 2 * n // 3
        sigma = n * rng.random() * 0.1 + n * 0.01
        a = np.exp(-0.5 * ((np.arange(n) - center1) / sigma) ** 2)
        a += np.exp(-0.5 * ((np.arange(n) - center2) / sigma) ** 2) + 0.05
        a = np.maximum(a, 0)
        
    elif method == "fractal":
        # Multiscale structure
        a = np.ones(n)
        for scale in [n, n//4, n//16, n//64]:
            freq = rng.integers(2, max(3, n // scale))
            amp = 0.3 / np.log2(scale) if scale > 1 else 0.1
            a += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.random() * 2 * np.pi)
        a = np.maximum(a, 0.01)
        
    elif method == "sparse_peak":
        # Mostly flat with a few peaks
        a = np.ones(n) * 0.5
        n_peaks = rng.integers(3, 10)
        for _ in range(n_peaks):
            pos = rng.integers(0, n)
            width = rng.integers(1, n // 50 + 1)
            height = rng.random() * 3 + 1
            a[pos - width:pos + width + 1] += height * np.exp(-0.5 * (np.arange(-width, width+1)/width)**2)
        a = np.maximum(a, 0)
        
    else:
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a * (S / s)


def objective_value(a):
    """Compute the actual objective (scale-invariant)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    
    Strategy: Multi-scale exploration + intensive proximal gradient with
    exact subgradients + pairwise coordinate descent + anti-palindromic constraints.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(1.0, budget_s - 15.0)
    
    global_best = np.inf
    global_a = None
    
    # Try to get parent
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 1.0:
            parent = np.clip(given, 0, 1000)
            print("Parent available: length=%d, sum=%.2f" % (len(parent), parent.sum()))
    except Exception:
        pass

    # === Phase 1: Multi-scale exploration (30% of time) ===
    phase1_time = (deadline - time.monotonic()) * 0.30
    print("=== Phase 1: Multi-scale exploration (%.0f sec) ===" % phase1_time)
    
    n_candidates = [4096, 8192, 16384]
    if parent is not None and len(parent) in [4096, 8192, 16384]:
        # Put parent length first
        n_candidates = [len(parent)] + [x for x in n_candidates if x != len(parent)]
    
    per_n_time = phase1_time / len(n_candidates)
    
    for n in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        S = np.sqrt(2.0 * n)
        remaining = min(per_n_time, deadline - time.monotonic() - 30)
        if remaining < 5:
            continue
        
        print("  Trying n=%d (%.0f sec)" % (n, remaining))
        
        # Try different initializations
        init_methods = ["anti_pal", "bimodal", "fractal"]
        if parent is not None and n == len(parent):
            init_methods = ["parent"] + init_methods
        elif parent is not None:
            init_methods = ["interp"] + init_methods
        
        per_init = remaining / len(init_methods)
        
        for method in init_methods:
            if time.monotonic() > deadline - 60:
                break
            if method == "parent" and parent is not None:
                a_init = parent.copy()
            elif method == "interp" and parent is not None:
                from numpy import interp
                tt_old = np.linspace(0, 1, len(parent))
                tt_new = np.linspace(0, 1, n)
                a_init = interp(tt_new, tt_old, parent)
                a_init = np.maximum(a_init, 0)
                s = a_init.sum()
                if s > 0:
                    a_init = a_init * (S / s)
            else:
                a_init = make_structured_init(n, rng, S, method)
            
            # Quick FISTA
            a_res, s_res = run_proximal_gradient(
                a_init, n, S, per_init * 0.6, rng,
                use_anti_pal=(method == "anti_pal"),
                approach="mixed", top_k=5, beta_init=3.0, beta_max=200.0
            )
            a_res = np.clip(a_res, 0, 1000)
            s_sum = a_res.sum()
            if s_sum > 0:
                a_res = a_res * (S / s_sum)
            
            obj = objective_value(a_res)
            c = fft_convolve_self(a_res)
            max_b = c.max()
            
            if max_b < global_best:
                global_best = max_b
                global_a = a_res.copy()
                print("    n=%d [%s]: max_b=%.6f, obj=%.6f" % (n, method, max_b, obj))
            else:
                print("    n=%d [%s]: max_b=%.6f" % (n, method, max_b))
    
    print("=== After Phase 1: best max_b=%.6f (n=%d) ===" % (global_best, len(global_a) if global_a is not None else 0))
    
    # === Phase 2: Focused optimization on best (50% of time) ===
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        phase2_time = (deadline - time.monotonic()) * 0.50
        print("\n=== Phase 2: Focused optimization (n=%d, %.0f sec) ===" % (n, phase2_time))
        
        # 2a: Long FISTA with mixed gradient
        seg2a = phase2_time * 0.40
        a_f, s_f = run_fista_restart(global_a, n, S, seg2a, rng)
        a_f = np.clip(a_f, 0, 1000)
        s_f = a_f.sum()
        if s_f > 0:
            a_f = a_f * (S / s_f)
        c_f = fft_convolve_self(a_f)
        if c_f.max() < global_best:
            global_best = c_f.max()
            global_a = a_f.copy()
            print("  2a FISTA restart: max_b=%.6f" % global_best)
        
        # 2b: Exact subgradient with high top_k
        remaining2 = min(deadline - time.monotonic() - 60, phase2_time * 0.30)
        if remaining2 > 5:
            a_e, s_e = run_proximal_gradient(
                global_a, n, S, remaining2, rng,
                use_anti_pal=False, approach="exact", top_k=10,
                beta_init=1.0, beta_max=100.0
            )
            a_e = np.clip(a_e, 0, 1000)
            s_e = a_e.sum()
            if s_e > 0:
                a_e = a_e * (S / s_e)
            c_e = fft_convolve_self(a_e)
            if c_e.max() < global_best:
                global_best = c_e.max()
                global_a = a_e.copy()
                print("  2b Exact subgrad: max_b=%.6f" % global_best)
        
        # 2c: Pairwise coordinate descent
        remaining2c = min(deadline - time.monotonic() - 60, phase2_time * 0.30)
        if remaining2c > 5:
            a_p, s_p = run_pairwise_cd(global_a, n, S, remaining2c, rng)
            a_p = np.clip(a_p, 0, 1000)
            s_p = a_p.sum()
            if s_p > 0:
                a_p = a_p * (S / s_p)
            c_p = fft_convolve_self(a_p)
            if c_p.max() < global_best:
                global_best = c_p.max()
                global_a = a_p.copy()
                print("  2c Pairwise CD: max_b=%.6f" % global_best)
    
    # === Phase 3: Try larger n for more degrees of freedom (15% of time) ===
    if time.monotonic() < deadline - 120 and global_a is not None:
        n_current = len(global_a)
        n_long = n_current * 2
        if n_long > 65536:
            n_long = max(n_current, 32768)
        if n_long != n_current and n_long <= 65536:
            S_long = np.sqrt(2.0 * n_long)
            phase3_time = min(120.0, (deadline - time.monotonic()) * 0.15)
            if phase3_time > 20:
                print("\n=== Phase 3: Larger n=%d (%.0f sec) ===" % (n_long, phase3_time))
                from numpy import interp
                tt_old = np.linspace(0, 1, n_current)
                tt_new = np.linspace(0, 1, n_long)
                a_long_init = interp(tt_new, tt_old, global_a)
                a_long_init = np.maximum(a_long_init, 0)
                s_l = a_long_init.sum()
                if s_l > 0:
                    a_long_init = a_long_init * (S_long / s_l)
                
                a_l, s_l = run_proximal_gradient(
                    a_long_init, n_long, S_long, phase3_time * 0.7, rng,
                    use_anti_pal=False, approach="mixed", top_k=8,
                    beta_init=3.0, beta_max=200.0
                )
                a_l = np.clip(a_l, 0, 1000)
                s_l = a_l.sum()
                if s_l > 0:
                    a_l = a_l * (S_long / s_l)
                c_l = fft_convolve_self(a_l)
                max_b_l = c_l.max()
                # Compare objectives (scale-invariant)
                obj_new = 2.0 * n_long * max_b_l / (s_l ** 2)
                obj_cur = 2.0 * n_current * global_best / (S_current ** 2) if 'S_current' in dir() else 2.0 * n_current * global_best / (np.sqrt(2.0 * n_current) ** 2)
                # Actually just compare max_b since they're normalized to same S
                if max_b_l < global_best:
                    global_best = max_b_l
                    global_a = a_l.copy()
                    print("  Longer n=%d: max_b=%.6f" % (n_long, max_b_l))
                else:
                    print("  Longer n=%d: max_b=%.6f (no improvement)" % (n_long, max_b_l))
    
    # === Phase 4: Final intensive polishing (remaining time) ===
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            print("\n=== Phase 4: Final polishing (%.0f sec) ===" % remaining)
            
            # 4a: Very high beta LSE (nearly exact max)
            seg4a = remaining * 0.30
            a_hb, _ = run_proximal_gradient(
                global_a, n, S, seg4a, rng,
                use_anti_pal=False, approach="lse",
                top_k=5, beta_init=100.0, beta_max=2000.0
            )
            a_hb = np.clip(a_hb, 0, 1000)
            s_hb = a_hb.sum()
            if s_hb > 0:
                a_hb = a_hb * (S / s_hb)
            c_hb = fft_convolve_self(a_hb)
            if c_hb.max() < global_best:
                global_best = c_hb.max()
                global_a = a_hb.copy()
                print("  4a High-beta LSE: max_b=%.6f" % global_best)
            
            # 4b: Anti-palindromic constrained optimization
            remaining4b = max(5, deadline - time.monotonic() - 10)
            seg4b = remaining4b * 0.30
            a_ap, _ = run_proximal_gradient(
                global_a, n, S, seg4b, rng,
                use_anti_pal=True, approach="exact", top_k=5,
                beta_init=1.0, beta_max=100.0
            )
            a_ap = np.clip(a_ap, 0, 1000)
            s_ap = a_ap.sum()
            if s_ap > 0:
                a_ap = a_ap * (S / s_ap)
            c_ap = fft_convolve_self(a_ap)
            if c_ap.max() < global_best:
                global_best = c_ap.max()
                global_a = a_ap.copy()
                print("  4b Anti-pal: max_b=%.6f" % global_best)
            
            # 4c: More pairwise CD
            remaining4c = max(5, deadline - time.monotonic() - 10)
            seg4c = remaining4c * 0.40
            a_pc, _ = run_pairwise_cd(global_a, n, S, seg4c, rng)
            a_pc = np.clip(a_pc, 0, 1000)
            s_pc = a_pc.sum()
            if s_pc > 0:
                a_pc = a_pc * (S / s_pc)
            c_pc = fft_convolve_self(a_pc)
            if c_pc.max() < global_best:
                global_best = c_pc.max()
                global_a = a_pc.copy()
                print("  4c Pairwise CD: max_b=%.6f" % global_best)
    
    # === Final verification ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\n=== Final: evaluate=%.6f, internal max_b=%.6f, n=%d ===" % (
        ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))
    
    if not np.isfinite(ver):
        print("  WARNING: evaluate rejected, using fallback")
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback evaluate=%.6f" % ver)
    
    return final_list
