"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using an adaptive log-sum-exp smoothed
objective with top-K peak ensemble, multi-scale initialization, and
coordinate descent polish.

Key innovations over previous approach:
1. Adaptive temperature based on convolution flatness
2. Top-K peak ensemble for more aggressive peak reduction
3. Multi-scale initialization (optimize at smaller n first, then upsample)
4. Two-stage optimization (shape phase + peak phase)
5. Nesterov momentum with adaptive restart
6. Peak-aware coordinate descent
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_c(c, a, w):
    """Compute 2 * cross_corr(w, a) using FFT.
    w is the weight vector (same length as c).
    Returns gradient vector of length n.
    """
    n = len(a)
    # Need linear cross-correlation: sum_k w[k] * a[k-j]
    # = (w * reverse(a))[n-1-j] ... let me just use the standard approach
    # cross_corr(w, a)[j] = sum_k w[k] * a[k-j]
    # This is the same as the convolution w * a_rev evaluated at appropriate index
    # Simpler: use ifft(W * conj(A)) with zero-padding
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def get_top_k_peaks(c, a, k=5):
    """Get the top k peak lags and their weights (softmax within top-k)."""
    # Get indices of top k peaks
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    # Use softmax weighting
    max_val = top_vals.max()
    weights = np.exp(top_vals - max_val)
    weights /= weights.sum()
    return top_idx, top_vals, weights


def run_adaptive_adam(a_init, n, S, time_budget, 
                      T_init=10.0, T_max=3000.0,
                      lr_init=0.05, beta1=0.9, beta2=0.999,
                      top_k=5, verbose=False, seed=42):
    """
    Adaptive log-Adam optimizer with:
    - Adaptive temperature based on convolution flatness
    - Top-K peak ensemble for gradient computation
    - Nesterov-like momentum (via Adam)
    - Periodic "hard" steps
    """
    rng = np.random.default_rng(seed)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())  # log of proportions
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Convert theta to a
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
        else:
            a = S * exp_theta / total
        
        # Compute autocorrelation
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > max_no_improve:
            noise = 0.01 * max(0.1, 1.0 - progress)
            a = best_a + noise * rng.standard_normal(n)
            a = np.maximum(a, 1e-15)
            a = a * (S / a.sum())
            theta = np.log(a / a.sum())
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            continue
        
        # Adaptive temperature
        c_mean = c.mean()
        c_max = c.max()
        flatness = c_mean / max(c_max, 1e-30)  # closer to 1 = flatter
        
        # T increases as we want to focus more on the peaks
        # But also adapt: if already flat, we can use higher T
        base_T = T_init * (T_max / T_init) ** (progress ** 0.5)
        # If flatness is high, we can use a lower T (more uniform weighting)
        # If flatness is low, use higher T (focus on peaks)
        T = base_T * (1.0 / max(flatness, 0.5))
        T = np.clip(T, T_init, T_max)
        
        # Compute weights: combination of log-sum-exp and top-K
        # Log-sum-exp weights
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w_lse = np.exp(scaled)
        w_lse /= w_lse.sum()
        
        # Top-K weights (more concentrated)
        top_idx, top_vals, top_weights = get_top_k_peaks(c, a, k=top_k)
        w_topk = np.zeros(len(c))
        for i, idx in enumerate(top_idx):
            w_topk[idx] = top_weights[i]
        
        # Blend: more top-K as progress increases
        alpha = min(1.0, progress * 1.5)  # 0 to 1
        w = (1 - alpha) * w_lse + alpha * w_topk
        
        # Compute gradient in a-space
        g_a = compute_grad_c(c, a, w)
        
        # Chain rule for log-parameters
        # a_j = S * exp(theta_j) / sum(exp(theta))
        # da_j/dtheta_l = a_j * (delta_jl - a_l / S)
        # dL/dtheta_l = sum_j (dL/da_j) * (da_j/dtheta_l)
        # = sum_j g_a[j] * a_j * (delta_jl - a_l/S)
        # = g_a[l]*a[l] - (a[l]/S) * sum_j g_a[j]*a[j]
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot
        
        # Adam update
        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break
        
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr = lr_init * (0.5 + 0.5 * (1.0 - progress))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if verbose and t % 2000 == 0:
            print("    Adam t=%d: obj=%.7f, T=%.1f, flat=%.4f, no_improve=%d" % (t, best_obj, T, flatness, no_improve))
    
    return best_a, best_obj


def run_hard_subgradient(a_init, n, S, time_budget,
                         lr_init=0.05, momentum=0.9,
                         top_k=3, noise_scale=0.005,
                         verbose=False, seed=42):
    """
    Hard subgradient method: use exact subgradient at the top-K peaks.
    More aggressive than the soft (log-sum-exp) approach.
    """
    rng = np.random.default_rng(seed)
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    a_prev = a.copy()
    beta = momentum
    start = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 100
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        
        # Nesterov lookahead
        y = a + (beta - 1.0) / (1.0 + beta) * (a - a_prev)
        y = normalize_nonneg(y, S)
        
        c_y = fft_convolve_self(y)
        
        # Track best
        c_cur = fft_convolve_self(a)
        cur_obj = c_cur.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart
        if no_improve > max_no_improve:
            scale = noise_scale * max(0.1, 1.0 - progress)
            a = best_a + scale * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            a_prev = a.copy()
            no_improve = 0
            step += 1
            continue
        
        # Compute subgradient at top-K peaks
        top_idx, top_vals, top_weights = get_top_k_peaks(c_y, a, k=top_k)
        w = np.zeros(len(c_y))
        for i, idx in enumerate(top_idx):
            w[idx] = top_weights[i]
        
        g = compute_grad_c(c_y, a, w)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g_dir = g / gnorm
        
        # Backtracking line search
        lr = lr_init * (1.0 - 0.5 * progress)
        best_step_a = a.copy()
        best_step_obj = cur_obj
        
        for i in range(10):
            a_trial = a - lr * g_dir
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            
            if obj_trial < cur_obj - 1e-8 * cur_obj:
                best_step_a = a_trial
                best_step_obj = obj_trial
                break
            lr *= 0.5
        
        a_prev = a.copy()
        a = best_step_a
        step += 1
        
        if verbose and step % 500 == 0:
            print("    HardSG step=%d: obj=%.7f, no_improve=%d" % (step, best_obj, no_improve))
    
    return best_a, best_obj


def coordinate_descent_peak_aware(a, n, S, time_budget, verbose=False, seed=123):
    """
    Coordinate descent that prioritizes coordinates contributing to top peaks.
    """
    rng = np.random.default_rng(seed)
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    start = time.monotonic()
    
    sweep = 0
    max_sweeps = 100
    
    while time.monotonic() - start < time_budget and sweep < max_sweeps:
        # Identify top peaks and "responsible" coordinates
        top_idx, top_vals, _ = get_top_k_peaks(c, a, k=5)
        
        # A coordinate j is "responsible" for peak at lag k if a[j] and a[k-j] are both significant
        # For simplicity, just use all coordinates but in an order biased toward edges
        # (edges contribute more to edge peaks)
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            original = a[i]
            if original < 1e-12:
                continue
            
            best_local = original
            best_local_obj = best_obj
            
            # Try multiplicative changes
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]:
                a[i] = original * factor
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-12:
                    best_local_obj = obj_check
                    best_local = original * factor
            
            a[i] = best_local
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            
            if c.max() < best_obj - 1e-12:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        sweep += 1
        if not improved and sweep >= 5:
            break
    
    return best_a, best_obj


def get_initialization(name, n, S, rng):
    """Generate various initializations."""
    i = np.arange(n)
    t = i / n
    
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "raised_cosine":
        # Raised cosine (Hann-like) window
        a = 0.5 * (1 + np.cos(2 * np.pi * t - np.pi))  # = sin^2(pi*t)
        a = np.maximum(a, 1e-6)
        return a * (S / a.sum())
    elif name == "hann":
        a = 0.5 * (1 - np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-6)
        return a * (S / a.sum())
    elif name == "gaussian":
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-6)
        return a * (S / a.sum())
    elif name == "cosine_mix":
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * t + rng.random() * 2) + \
            0.2 * np.cos(2 * np.pi * 7 * t + rng.random() * 2) + \
            0.1 * np.cos(2 * np.pi * 13 * t + rng.random() * 2)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "flat_spectrum":
        # Try to make |A(w)|^2 flat
        a = 1.0 + 0.15 * np.sin(2 * np.pi * 5 * t + rng.random()) + \
            0.1 * np.cos(2 * np.pi * 11 * t + rng.random())
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "inverse_triangular":
        # Counter the triangular convolution: make edges larger
        a = 1.0 + 0.5 * np.abs(2*t - 1)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def upsample(a_small, n_large):
    """Upsample a small sequence to a larger one using linear interpolation."""
    n_small = len(a_small)
    if n_small == n_large:
        return a_small.copy()
    # Linear interpolation
    old_indices = np.arange(n_small) / (n_small - 1)
    new_indices = np.arange(n_large) / (n_large - 1)
    a_large = np.interp(new_indices, old_indices, a_small)
    return a_large


def try_zero_masking(a, n, S, n_zero=30, verbose=False):
    """Try setting the smallest coordinates to zero."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    current_obj = c.max()
    
    sorted_idx = np.argsort(a)
    candidates = sorted_idx[:n_zero * 5]
    
    best_obj = current_obj
    best_a = a.copy()
    
    for idx in candidates:
        if a[idx] < 1e-10:
            continue
        saved = a[idx]
        a[idx] = 0.0
        a = normalize_nonneg(a, S)
        c_check = fft_convolve_self(a)
        obj_check = c_check.max()
        if obj_check < best_obj - 1e-12:
            best_obj = obj_check
            best_a = a.copy()
        else:
            a[idx] = saved
            a = normalize_nonneg(a, S)
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive log-sum-exp smoothed
objective with top-K peak ensemble blending, multi-scale initialization,
hard/soft hybrid subgradient methods, and peak-aware coordinate descent.

Key improvements over previous approach:
1. Top-K peak ensemble blended with log-sum-exp for more aggressive peak targeting
2. Adaptive temperature based on convolution flatness ratio
3. Multi-scale initialization (optimize at n/4 first, then upsample)
4. Hard subgradient method alternating with soft (log-sum-exp) method
5. Raised cosine and inverse-triangular initializations
6. Extended coordinate descent with more factors and sweeps
7. Strategic n=8192 exploration alongside n=4096
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_c(w, a):
    """Compute 2 * cross_corr(w, a) where cross_corr(j) = sum_k w[k]*a[k-j].
    w has length 2n-1, a has length n. Returns length n.
    """
    n = len(a)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def get_top_k_weights(c, k=5):
    """Get weights concentrated on top-k peaks using softmax within top-k."""
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    max_val = top_vals.max()
    weights = np.exp(top_vals - max_val)
    weights /= weights.sum()
    w = np.zeros(len(c))
    for i, idx in enumerate(top_idx):
        w[idx] = weights[i]
    return w


def run_adaptive_adam(a_init, n, S, time_budget,
                      T_init=10.0, T_max=3000.0,
                      lr_init=0.05, beta1=0.9, beta2=0.999,
                      top_k=5, blend_rate=1.0,
                      verbose=False, seed=42):
    """
    Adaptive log-Adam optimizer with top-K peak ensemble blending.
    """
    rng = np.random.default_rng(seed)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    no_improve = 0
    max_no_improve = 250

    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Convert theta to a
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
        else:
            a = S * exp_theta / total

        # Compute convolution
        c = fft_convolve_self(a)
        cur_obj = c.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve > max_no_improve:
            noise = 0.008 * max(0.1, 1.0 - progress)
            a = best_a + noise * rng.standard_normal(n)
            a = np.maximum(a, 1e-15)
            a = a * (S / a.sum())
            theta = np.log(a / a.sum())
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            continue

        # Adaptive temperature
        c_mean = c.mean()
        c_max = c.max()
        flatness = c_mean / max(c_max, 1e-30)

        base_T = T_init * (T_max / T_init) ** (progress ** 0.6)
        T = base_T * (1.0 / max(flatness, 0.3))
        T = np.clip(T, T_init * 0.5, T_max * 1.5)

        # Log-sum-exp weights
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w_lse = np.exp(scaled)
        w_lse_sum = w_lse.sum()
        if w_lse_sum < 1e-300:
            w_lse = np.zeros(len(c))
            w_lse[np.argmax(c)] = 1.0
        else:
            w_lse /= w_lse_sum

        # Top-K weights
        w_topk = get_top_k_weights(c, k=top_k)

        # Blend: more top-K as progress increases
        alpha = min(1.0, progress * blend_rate)
        w = (1.0 - alpha) * w_lse + alpha * w_topk

        # Compute gradient in a-space
        g_a = compute_grad_c(w, a)

        # Chain rule for log-parameters
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot

        # Adam update
        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break

        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_init * (0.4 + 0.6 * (1.0 - progress))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if verbose and t % 3000 == 0:
            print("    AdamL t=%d: obj=%.7f, T=%.1f, flat=%.4f" % (t, best_obj, T, flatness))

def run_hard_sg(a_init, n, S, time_budget,
                lr_init=0.05, momentum=0.85,
                top_k=3, noise_scale=0.005,
                verbose=False, seed=42):
    """
    Hard subgradient with Nesterov momentum at top-K peaks.
    """
    rng = np.random.default_rng(seed)
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    a_prev = a.copy()
    beta = momentum
    start = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 120

    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)

        # Nesterov lookahead
        y = a + (beta - 1.0) / (1.0 + beta) * (a - a_prev)
        y = normalize_nonneg(y, S)
        c_y = fft_convolve_self(y)

        c_cur = fft_convolve_self(a)
        cur_obj = c_cur.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            scale = noise_scale * max(0.1, 1.0 - progress)
            a = best_a + scale * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            a_prev = a.copy()
            no_improve = 0
            step += 1
            continue

        # Subgradient at top-K peaks
        w = get_top_k_weights(c_y, k=top_k)
        g = compute_grad_c(w, a)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g_dir = g / gnorm

        # Backtracking line search
        lr = lr_init * (1.0 - 0.4 * progress)
        best_step_a = a.copy()

        for i in range(12):
            a_trial = a - lr * g_dir
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()

            if obj_trial < cur_obj - 1e-8 * cur_obj:
                best_step_a = a_trial
                break
            lr *= 0.4

        a_prev = a.copy()
        a = best_step_a
        step += 1

        if verbose and step % 500 == 0:
            print("    HardSG step=%d: obj=%.7f" % (step, best_obj))

    return best_a, best_obj


def coordinate_descent_extensive(a, n, S, time_budget, verbose=False, seed=123):
    """Extensive coordinate descent with many factors."""
    rng = np.random.default_rng(seed)
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    start = time.monotonic()

    sweep = 0
    max_sweeps = 200
    factors = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0, 0.4, 0.6, 1.2]

    while time.monotonic() - start < time_budget and sweep < max_sweeps:
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            original = a[i]
            if original < 1e-14:
                continue

            best_local = original
            best_local_obj = best_obj

            for factor in factors:
                a[i] = original * factor
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-13:
                    best_local_obj = obj_check
                    best_local = original * factor

            a[i] = best_local
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)

            if c.max() < best_obj - 1e-13:
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        sweep += 1
        if verbose and sweep % 20 == 0:
            print("    CD sweep %d: obj=%.7f" % (sweep, best_obj))
        if not improved and sweep >= 8:
            break

    return best_a, best_obj


def get_initialization(name, n, S, rng):
    """Generate various initializations."""
    i = np.arange(n, dtype=np.float64)
    t = i / n

    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "raised_cosine":
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-6)
        return a * (S / a.sum())
    elif name == "hann":
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-6)
        return a * (S / a.sum())
    elif name == "cosine_mix":
        a = (1.0 + 0.3 * np.cos(2 * np.pi * 3 * t + rng.random() * 2) +
             0.2 * np.cos(2 * np.pi * 7 * t + rng.random() * 2) +
             0.1 * np.cos(2 * np.pi * 17 * t + rng.random() * 2))
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "flat_spectrum":
        a = (1.0 + 0.15 * np.sin(2 * np.pi * 5 * t + rng.random()) +
             0.1 * np.cos(2 * np.pi * 13 * t + rng.random()))
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "inverse_triangular":
        a = 1.0 + 0.6 * np.abs(2.0 * t - 1.0)
        return a * (S / a.sum())
    elif name == "gaussian_wide":
        a = np.exp(-0.5 * ((t - 0.5) / 0.4) ** 2)
        a = np.maximum(a, 1e-6)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def upsample_linear(a_small, n_large):
    """Upsample using linear interpolation."""
    n_small = len(a_small)
    if n_small == n_large:
        return a_small.copy()
    old_t = np.linspace(0, 1, n_small)
    new_t = np.linspace(0, 1, n_large)
    a_large = np.interp(new_t, old_t, a_small)
    return a_large


def try_zero_masking(a, n, S, n_zero=30):
    """Try zeroing smallest coordinates if it helps."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    sorted_idx = np.argsort(a)
    candidates = sorted_idx[:n_zero * 5]

    for idx in candidates:
        if a[idx] < 1e-12:
            continue
        saved = a[idx]
        a[idx] = 0.0
        a = normalize_nonneg(a, S)
        c_check = fft_convolve_self(a)
        obj_check = c_check.max()
        if obj_check < best_obj - 1e-12:
            best_obj = obj_check
            best_a = a.copy()
        else:
            a[idx] = saved
            a = normalize_nonneg(a, S)

    return best_a, best_obj
