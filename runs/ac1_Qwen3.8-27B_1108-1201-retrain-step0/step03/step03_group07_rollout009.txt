    a_pad = np.zeros(L)
    a_pad[:n] = a
    grad_a = 2.0 * np.real(ifft(fft(w_pad) * conj(fft(a_pad))))[:n]

    # Chain rule through softmax parameterization:
    # a_i = S * softmax(x)_i, so da_i/dx_j = a_i*(delta_ij - a_j/S)
    # dlse/dx_j = sum_i grad_a_i * a_i * (delta_ij - a_j/S)
    #            = grad_a_j * a_j - (a_j/S) * sum_i grad_a_i * a_i
    dot_ga_a = np.dot(grad_a, a)
    grad_x = grad_a * a - (a / S) * dot_ga_a

    return lse, grad_x


def lbe_anneal(a_init, S, T_start_frac, T_end_frac, n_temps,
               max_iter_per_temp, time_limit, verbose=True):
    """
    Run L-BFGS with exponential temperature annealing.
    T_start = T_start_frac * c_max, T_end = T_end_frac * c_max.
    """
    n = len(a_init)
    a = a_init.copy()
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a * (S / a.sum())

    # Convert to x parameter: p = a/S, x = log(p)
    p = a / S
    p = np.maximum(p, 1e-300)
    x = np.log(p)

    start_time = time.monotonic()
    best_obj = obj_exact(a)
    best_x = x.copy()

    for t_idx in range(n_temps):
        if time.monotonic() - start_time > time_limit:
            break

        # Exponential annealing
        if n_temps == 1:
            frac = 0.0
        else:
            frac = t_idx / (n_temps - 1)
        T_frac = T_start_frac * (T_end_frac / T_start_frac) ** frac

        a_cur = softmax_parameterize(x, S)
        c = fft_convolve_self(a_cur)
        c_max = c.max()
        T_actual = T_frac * c_max

        if T_actual < 1e-10:
            T_actual = 1e-10

        # L-BFGS at this temperature
        state = {'T': T_actual, 'S': S}

        def fun_local(x_local):
            val, _ = lse_obj_and_grad_x(x_local, state['S'], state['T'])
            return val

        def grad_local(x_local):
            _, g = lse_obj_and_grad_x(x_local, state['S'], state['T'])
            return g

        try:
            result = minimize(
                fun_local, x, jac=grad_local, method='L-BFGS-B',
                options={'maxiter': max_iter_per_temp, 'gtol': 1e-7,
                         'maxls': 30, 'iprint': -1}
            )
            x = result.x
        except Exception:
            pass

        a_new = softmax_parameterize(x, S)
        obj_new = obj_exact(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_x = x.copy()

        if verbose:
            elapsed = time.monotonic() - start_time
            print(f"        T/T0={T_frac:.4f}, obj={obj_new:.6f}, "
                  f"best={best_obj:.6f}, t={elapsed:.1f}s")

    a_best = softmax_parameterize(best_x, S)
    return a_best, obj_exact(a_best)


def multiplicative_polish(a_init, S, time_budget, lr_init=0.03, verbose=True):
    """
    Multiplicative (mirror descent) polish chasing the single worst peak.
    Uses momentum and adaptive learning rate.
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = obj_exact(a)
    best_a = a.copy()
    start = time.monotonic()
    v = np.zeros(n)
    lr = lr_init
    stalled = 0

    while time.monotonic() - start < time_budget:
        k_max = int(np.argmax(c))

        # Subgradient of c[k_max] w.r.t. a
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for i in range(lo, hi + 1):
            grad[i] = 2.0 * a[k_max - i]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Momentum
        v = 0.9 * v + 0.1 * grad
        vnorm = np.linalg.norm(v)
        if vnorm > 0:
            v = v / vnorm

        # Multiplicative update
        a = a * np.exp(-lr * v)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())

        c = fft_convolve_self(a)
        cur_obj = obj_exact(a)

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            stalled = 0
            lr = min(lr * 1.05, lr_init * 3.0)
        else:
            stalled += 1
            if stalled % 50 == 0:
                lr *= 0.7
            if stalled > 200:
                # Perturb to escape
                noise = rng_global.randn(n) * 0.03 * S / np.sqrt(n)
                a = a + noise
                a = np.maximum(a, 1e-300)
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                v *= 0.3
                lr = lr_init * 0.5
                stalled = 0

        if verbose and int(time.monotonic() - start) % 10 == 0:
            elapsed = time.monotonic() - start
            if int(elapsed) != int(elapsed - 1):
                print(f"          obj={cur_obj:.6f}, best={best_obj:.6f}, "
                      f"lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a, best_obj


def smooth_peaks_polish(a_init, S, time_budget, top_k=5, lr_init=0.02, verbose=True):
    """
    Multiplicative polish using top-k peaks (softer than single peak).
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = obj_exact(a)
    best_a = a.copy()
    start = time.monotonic()
    v = np.zeros(n)
    lr = lr_init
    stalled = 0

    while time.monotonic() - start < time_budget:
        # Top-k peaks with exponential weighting
        sorted_idx = np.argsort(c)[::-1]
        top_idx = sorted_idx[:top_k]
        c_max = c[top_idx[0]]
        weights = np.exp(0.5 * (c[top_idx] - c_max))
        weights /= weights.sum()

        # Weighted subgradient
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            w_i = 2.0 * weights[i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += w_i * a[k - j]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        v = 0.85 * v + 0.15 * grad
        vnorm = np.linalg.norm(v)
        if vnorm > 0:
            v = v / vnorm

        a = a * np.exp(-lr * v)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())

        c = fft_convolve_self(a)
        cur_obj = obj_exact(a)

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            stalled = 0
            lr = min(lr * 1.03, lr_init * 3.0)
        else:
            stalled += 1
            if stalled % 100 == 0:
                lr *= 0.7
            if stalled > 500:
                noise = rng_global.randn(n) * 0.02 * S / np.sqrt(n)
                a = a + noise
                a = np.maximum(a, 1e-300)
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                v *= 0.3
                lr = lr_init * 0.5
                stalled = 0

        if verbose and int(time.monotonic() - start) % 10 == 0:
            elapsed = time.monotonic() - start
            if int(elapsed) != int(elapsed - 1):
                print(f"          smooth: obj={cur_obj:.6f}, best={best_obj:.6f}, "
                      f"lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a, best_obj


def make_initializations(n, S, rng):
    """Generate diverse initializations for length n with sum S."""
    cands = []
    i = np.arange(n)

    # Cosine mixture
    a1 = (1.0 + 0.3 * np.cos(2 * np.pi * 3 * i / n) +
          0.2 * np.cos(2 * np.pi * 7 * i / n) +
          0.1 * np.cos(2 * np.pi * 13 * i / n))
    a1 = np.maximum(a1, 0.01)
    a1 *= S / a1.sum()
    cands.append(a1)

    # Random
    a2 = rng.random(n) + 0.1
    a2 *= S / a2.sum()
    cands.append(a2)

    # Two Gaussian bumps
    a3 = (np.exp(-0.5 * ((i - n * 0.3) / (n * 0.12)) ** 2) +
          np.exp(-0.5 * ((i - n * 0.7) / (n * 0.12)) ** 2))
    a3 = np.maximum(a3, 0.01)
    a3 *= S / a3.sum()
    cands.append(a3)

    # Periodic modulation
    a4 = 1.0 + 0.4 * np.abs(np.sin(2 * np.pi * 5 * i / n))
    a4 = np.maximum(a4, 0.01)
    a4 *= S / a4.sum()
    cands.append(a4)

    return cands


# Global RNG for perturbations
rng_global = np.random.default_rng(12345)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase: explore -> L-BFGS anneal -> multiplicative polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    global_best = np.inf
    global_a = None

    def try_update(a, obj):
        nonlocal global_best, global_a
        if obj < global_best and np.isfinite(obj):
            global_best = obj
            global_a = a.copy()

    # Load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw.astype(np.float64)
            n_g = len(given_a)
            S_g = np.sqrt(2.0 * n_g)
            given_a = given_a * (S_g / given_a.sum())
            obj_g = obj_exact(given_a)
            print(f"Given: n={n_g}, obj={obj_g:.6f}")
            try_update(given_a, obj_g)
    except Exception as e:
        print(f"No given sequence: {e}")

    # === Phase 1: Quick exploration (12% of time) ===
    remaining = deadline - time.monotonic()
    phase1_time = 0.12 * remaining
    print(f"\n=== Phase 1: Exploration ({phase1_time:.0f}s) ===")

    # Try different lengths
    lengths_to_try = [4096, 8192]
    per_len_time = phase1_time / len(lengths_to_try)

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        S_try = np.sqrt(2.0 * n_try)
        inits = make_initializations(n_try, S_try, rng)

        if given_a is not None and len(given_a) == n_try:
            inits.insert(0, given_a.copy())

        per_init_time = per_len_time / len(inits)
        for idx, a_init in enumerate(inits):
            if time.monotonic() > deadline - 60:
                break
            obj_init = obj_exact(a_init)
            print(f"  n={n_try}, init{idx}: obj={obj_init:.6f}")

            # Quick L-BFGS
            c = fft_convolve_self(a_init)
            T_start = 0.15
            T_end = 0.03
            t0 = time.monotonic()
            a_best, obj_best = lbe_anneal(
                a_init, S_try, T_start, T_end,
                n_temps=3, max_iter_per_temp=100,
                time_limit=per_init_time, verbose=False
            )
            elapsed = time.monotonic() - t0
            print(f"    -> obj={obj_best:.6f} ({elapsed:.1f}s)")
            try_update(a_best, obj_best)

    # === Phase 2: Main L-BFGS optimization (55% of time) ===
    remaining = deadline - time.monotonic()
    phase2_time = 0.55 * remaining
    print(f"\n=== Phase 2: Main L-BFGS ({phase2_time:.0f}s) ===")

    if global_a is not None:
        n = len(global_a)
        S = global_a.sum()
        c = fft_convolve_self(global_a)
        c_max = c.max()
        print(f"  Starting: n={n}, obj={obj_exact(global_a):.6f}, c_max={c_max:.4f}")

        # Temperature annealing: broad to sharp
        T_start = 0.10
        T_end = 0.003
        n_temps = 20
        max_iter = 400

        # Multiple restarts
        n_restarts = 3
        per_restart = phase2_time / n_restarts

        for restart in range(n_restarts):
            if time.monotonic() > deadline - 120:
                break

            if restart == 0:
                a_start = global_a.copy()
            elif restart == 1:
                a_start = global_a + rng_global.randn(n) * 0.03 * S / np.sqrt(n)
                a_start = np.maximum(a_start, 1e-10)
                a_start = a_start * (S / a_start.sum())
            else:
                # More aggressive perturbation
                a_start = global_a.copy()
                mask = rng_global.random(n) < 0.1
                a_start[mask] *= rng_global.uniform(0.3, 3.0, mask.sum())
                a_start = np.maximum(a_start, 1e-10)
                a_start = a_start * (S / a_start.sum())

            print(f"  Restart {restart+1}/{n_restarts}:")
            a_best, obj_best = lbe_anneal(
                a_start, S, T_start, T_end,
                n_temps=n_temps, max_iter_per_temp=max_iter,
                time_limit=per_restart, verbose=True
            )
            print(f"    -> obj={obj_best:.6f}")
            try_update(a_best, obj_best)

    # === Phase 3: Multiplicative polish (25% of time) ===
    remaining = deadline - time.monotonic()
    phase3_time = min(0.25 * (budget_s - 25), remaining * 0.7)
    phase3_time = max(5.0, phase3_time)
    print(f"\n=== Phase 3: Polish ({phase3_time:.0f}s) ===")

    if global_a is not None:
        n = len(global_a)
        S = global_a.sum()

        # 3a: Single-peak multiplicative (50%)
        t3a = phase3_time * 0.50
        print(f"  3a: Single-peak polish ({t3a:.0f}s)")
        a_best, obj_best = multiplicative_polish(global_a, S, t3a, lr_init=0.03)
        print(f"    -> obj={obj_best:.6f}")
        try_update(a_best, obj_best)

        # 3b: Top-k smooth peaks (30%)
        t3b = phase3_time * 0.30
        if t3b > 3:
            print(f"  3b: Top-k polish ({t3b:.0f}s)")
            a_best, obj_best = smooth_peaks_polish(global_a, S, t3b, top_k=5)
            print(f"    -> obj={obj_best:.6f}")
            try_update(a_best, obj_best)

        # 3c: Final fine L-BFGS (20%)
        t3c = phase3_time * 0.20
        if t3c > 3:
            c = fft_convolve_self(global_a)
            print(f"  3c: Fine L-BFGS ({t3c:.0f}s)")
            a_best, obj_best = lbe_anneal(
                global_a, S, 0.02, 0.002,
                n_temps=5, max_iter_per_temp=300,
                time_limit=t3c, verbose=True
            )
            print(f"    -> obj={obj_best:.6f}")
            try_update(a_best, obj_best)

    # === Final short multiplicative sweep ===
    if time.monotonic() < deadline - 15 and global_a is not None:
        n = len(global_a)
        S = global_a.sum()
        t_final = deadline - time.monotonic() - 10
        if t_final > 5:
            print(f"\n=== Final sweep ({t_final:.0f}s) ===")
            a_best, obj_best = multiplicative_polish(global_a, S, t_final, lr_init=0.01)
            print(f"  -> obj={obj_best:.6f}")
            try_update(a_best, obj_best)

    # === Final verification ===
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: local_obj={global_best:.6f}, verified={ver:.6f}")

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.6f}")

    return final_a
