"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using continuation L-BFGS on the
log-sum-exp smooth approximation in softmax-parameterized space,
combined with coordinate descent polishing and random restarts.

Key innovations over previous approach:
1. L-BFGS (quasi-Newton) instead of subgradient - uses curvature information
2. Continuation method: gradually increase T from smooth to sharp
3. Softmax parameterization ensures constraints are automatically satisfied
4. Efficient FFT-based gradient computation
5. Adaptive stopping and multiple restarts for robustness
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = rfft(a, fsize)
    c = irfft(A * A, fsize)
    return c[:size]


def softmax_theta_to_a(theta, S):
    """Convert theta to a = S * softmax(theta)."""
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    if total < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * exp_theta / total


def a_to_theta(a):
    """Convert a to theta (log of proportions)."""
    a = np.maximum(a, 1e-30)
    return np.log(a / a.sum())


def smooth_obj_and_grad(theta, T, S):
    """Compute log-sum-exp smooth objective and gradient in theta-space.
    
    Objective: F = (1/T) * log(sum_m exp(T * c[m])) where c = conv(a,a)
    Gradient w.r.t. theta: a_j * (g_j - (a·g)/S) where g = gradient w.r.t. a
    """
    n = len(theta)
    a = softmax_theta_to_a(theta, S)
    
    # Compute convolution
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = rfft(a, fsize)
    c = irfft(A * A, fsize)[:size]
    
    # Compute objective (log-sum-exp)
    c_scaled = T * c
    c_max = c_scaled.max()
    log_sum_exp = c_max + np.log(np.exp(c_scaled - c_max).sum())
    F = log_sum_exp / T
    
    # Compute softmax weights
    p = np.exp(c_scaled - c_max)
    p /= p.sum()
    
    # Compute gradient w.r.t. a:
    # g[m] = sum_k p[k] * (a[k-m] + a[m-k])
    # = h1[m] + h2[m] where:
    #   h1[m] = sum_j a[j] * p[j+m]  (cross-correlation of a and p)
    #   h2[m] = sum_j a[j] * p[m-j]  (convolution of a and p)
    
    N = 1 << (3 * n - 2).bit_length()
    A_full = rfft(a, N)
    P = rfft(p, N)
    
    # Cross-correlation: ifft(conj(fft(a)) * fft(p))
    h1_full = irfft(np.conj(A_full) * P, N)
    h1 = h1_full[:n]
    
    # Convolution: ifft(fft(a) * fft(p))
    h2_full = irfft(A_full * P, N)
    h2 = h2_full[:n]
    
    g_a = h1 + h2
    
    # Gradient w.r.t. theta: dF/dtheta_j = a_j * (g_a_j - (a·g_a)/S)
    dot = np.dot(a, g_a)
    g_theta = a * (g_a - dot / S)
    
    return F, g_theta


def run_lbfgs(a_init, n, S, T, time_budget, verbose=False):
    """Run L-BFGS on the smooth objective for a given temperature T."""
    theta_init = a_to_theta(a_init)
    start = time.monotonic()
    
    # Callback to check time
    state = {'last_check': time.monotonic(), 'timeout': False}
    
    def callback(xk):
        if time.monotonic() - state['last_check'] > 2.0:
            state['last_check'] = time.monotonic()
            if time.monotonic() - start > time_budget:
                state['timeout'] = True
                return True  # Stop optimization
        return False
    
    result = minimize(
        smooth_obj_and_grad,
        theta_init,
        jac=True,  # smooth_obj_and_grad returns (f, g)
        method='L-BFGS-B',
        options={
            'maxiter': 10000,
            'ftol': 1e-14,
            'gtol': 1e-10,
            'maxfun': 1000000,
            'iprint': 0,
        },
        callback=callback
    )
    
    elapsed = time.monotonic() - start
    a_result = softmax_theta_to_a(result.x, S)
    
    # Evaluate true objective
    c = fft_convolve_self(a_result)
    true_obj = c.max()
    
    if verbose:
        print("    L-BFGS T=%d: F=%.7f, true_max=%.7f, time=%.1fs, status=%s" % (
            T, result.fun, true_obj, elapsed, result.message))
    
    return a_result, true_obj, result.fun


def coordinate_descent(a, n, S, time_budget, verbose=False):
    """Coordinate descent polish."""
    a = a.copy().astype(np.float64)
    S_target = S
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    sweep = 0
    max_sweeps = 100
    
    while sweep < max_sweeps and time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            original = a[i]
            if original < 1e-15:
                continue
            
            best_factor = 1.0
            best_local_obj = best_obj
            
            # Try multiplicative changes
            for factor in [0.7, 0.85, 1.0, 1.18, 1.4, 0.5, 1.67]:
                a[i] = original * factor
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    break
                a = a * (S_target / s)
                
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                
                if obj_check < best_local_obj:
                    best_local_obj = obj_check
                    best_factor = factor
            
            a[i] = original * best_factor
            a = np.maximum(a, 0.0)
            a = a * (S_target / a.sum())
            c = fft_convolve_self(a)
            
            if c.max() < best_obj:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        if not improved and sweep >= 5:
            break
    
    return best_a, best_obj
