def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_subgradient(a, c, top_K):
    """Compute subgradient of max(c) using top-K peaks."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_K]

    # This is 2 * cross_correlation(1_top_idx, a)
    # cross_correlation(p, a)[m] = sum_k p[k] * a[k-m]
    # = sum_k p[k] * a[(k-m) mod ...] -- careful with boundaries
    
    # Use FFT: cross_correlation(p, a)[m] = ifft(fft(p) * conj(fft(a)))
    # Wait, let me verify the convention.
    # (p * a)[k] = sum_i p[i] * a[k-i]  (full convolution)
    # cross_correlation(p, a)[m] = sum_k p[k] * a[k-m] = sum_k p[k] * a[(k-m)]
    # Let j = k-m: = sum_j p[j+m] * a[j]
    # This is the cross-correlation, which equals the convolution of p_reversed with a.
    # p_reversed[i] = p[n_p-1-i] (for circular), or we can use:
    # cross_correlation(p, a)[m] = (ifft(fft(p) * conj(fft(a))))[m]
    
    # Let me verify: fft(p * a) = fft(p) * fft(a) (circular convolution)
    # For linear cross-correlation:
    # xc[m] = sum_k p[k] * a[k-m]
    # Let's compute via FFT: ifft(P * conj(A))[m] where P = fft(p), A = fft(a)
    # ifft(P * conj(A))[m] = sum_k P[k] * conj(A[k]) * exp(2*pi*i*m*k/N) / N
    # This equals the circular cross-correlation of p and a at lag m.
    # For linear cross-correlation (no wrap-around), we need to zero-pad.
    
    p = np.zeros(2 * n - 1)
    for k in top_idx:
        p[int(k)] = 1.0
    
    fsize = 1 << (len(p) + n - 1).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    xc = np.real(ifft(P * np.conj(A)))[:n]
    return 2.0 * xc

def compute_subgradient_toppk(a, c, top_K):

def armijo_step(a, sg, c_current, S, step_max=1.0, c_armijo=1e-4, max_backtracks=20):
    """Armijo backtracking line search on the simplex."""
    n = len(a)
    # We want to minimize max(conv(a,a)) subject to a >= 0, sum(a) = S
    # The descent direction is -sg (negate the subgradient)
    # We check: f(a_new) <= f(a_old) - c * step * ||sg||^2 (for the non-smooth case, we use a simpler criterion)
    # For non-smooth functions, we use: f(a_new) <= f(a_old) - c * step * <sg, a_old - a_new>
    # where <sg, a_old - a_new> >= 0 (since we're moving in the descent direction)
    
    # Simpler: just check f(a_new) < f(a_old) with some margin
    step = step_max
    a_new = project_simplex(a - step * sg, S)
    c_new = fft_convolve_self(a_new)
    max_new = c_new.max()
    
    backtrack = 0
    while max_new > c_current * (1.0 - 1e-8 * step) and backtrack < max_backtracks:
        step *= 0.5
        a_new = project_simplex(a - step * sg, S)
        c_new = fft_convolve_self(a_new)

        backtrack += 1

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient_toppk(a, c, top_K):
    """
    Compute subgradient of max(c) at a using top-K peak positions.
    sg[m] = 2 * sum_{k in topK} a[k-m] * 1(0 <= k-m < n)
    Computed via FFT cross-correlation.
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_K]
    p = np.zeros(2 * n - 1)
    p[top_idx] = 1.0
    fsize = 1 << (len(p) + n - 1).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    xc = np.real(ifft(P * np.conj(A)))[:n]
    return 2.0 * xc

    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for top-K peaks, b >= 0.
    Returns normalized direction b or None.
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix A: K x n
    # A[i, j] = a[k_i - j] if 0 <= k_i - j < n, else 0
    A_mat = np.zeros((len(top_idx), n))
    for i, k in enumerate(top_idx):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A_mat[i, j_start:j_end + 1] = a[j_start:j_end + 1]
    
    # LP: max sum(b) s.t. A_mat @ b <= M, b >= 0
    # = min -sum(b) s.t. A_mat @ b <= M, b >= 0
    from scipy.optimize import linprog
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A_mat, b_ub=np.full(len(top_idx), M),
                     bounds=bounds, method='highs')
    
    if result.success and result.x is not None:
        b = result.x
        s = b.sum()
        if s > 1e-10:
            return b / s
    return None
