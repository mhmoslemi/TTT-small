Minimize 2*n*max(conv(a,a))/sum(a)^2 using Adam optimizer with softmax-smoothed

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_objective_and_grad(a, c, tau):
    """
    Compute softmax-smoothed objective and its gradient w.r.t. a.
    Returns (obj, grad_a) where grad_a is the gradient of the smoothed objective.
    """
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf, np.zeros(n)
    
    # Softmax weights over c
    c_max = c.max()
    w = np.exp((c - c_max) / tau)
    w_sum = w.sum()
    w = w / w_sum
    
    # Smoothed max
    c_smooth = w @ c
    
    # Objective value (use true max for tracking, smoothed for gradient)
    obj_true = 2.0 * n * c_max / (S * S)
    
    # Gradient of smoothed objective:
    # f = 2n * c_smooth / S^2
    # df/da_m = 2n * (dc_smooth/da_m * S^2 - c_smooth * 2S) / S^4
    #          = 2n * (dc_smooth/da_m * S - 2*c_smooth) / S^3
    
    # dc_smooth/da_m = sum_k w_k * dc_k/da_m
    # dc_k/da_m = 2 * a[k-m] (for valid m, k-m)
    #            = 2 * conv(w, a_rev)[n-1+m]
    
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 2).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    
    dc_da = 2.0 * full_conv[n - 1:2 * n - 1]
    
    grad_a = 2.0 * n * (dc_da * S - 2.0 * c_smooth) / (S * S * S)
    
    return obj_true, grad_a

    Adam optimization in log-parameter space.

    n = len(a)

    obj = 2.0 * n * c.max() / (a.sum() ** 2)

    best_theta = theta.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau = tau0
    
    start = time.monotonic()
    no_improve = 0
    patience = 200
    
    while time.monotonic() - start < time_budget:
        # Compute gradient
        obj_curr, grad_a = compute_objective_and_grad(a, c, tau)
        
        if not np.isfinite(obj_curr):
            # Reset
            theta = theta0.copy() + 0.1 * np.random.randn(n)
            a = np.exp(theta)
            a = a * S_target / a.sum()
            theta = np.log(np.maximum(a, 1e-300))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            c = fft_convolve_self(a)
            continue
        
        # Chain rule: grad_theta = grad_a * a
        grad_theta = grad_a * a
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

    else:

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using Adam optimizer with softmax-smoothed
max, log-parameterization for positivity, adaptive temperature annealing,
multi-start with tournament selection, and L-BFGS polish.
Key differences from prior approach: exact gradient (not top-k subgradient),
Adam with per-parameter adaptive LR, log-space optimization, and L-BFGS refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_max_and_grad(a, c, tau):
    """
    Compute softmax-smoothed max of c and its gradient w.r.t. a.
    Returns (c_smooth, dc_smooth_da).
    """
    n = len(a)
    c_max = c.max()
    w = np.exp((c - c_max) / tau)
    w_sum = w.sum()
    w = w / w_sum
    c_smooth = w @ c

    # dc_smooth/da_m = 2 * sum_k w_k * a[k-m] (for valid indices)
    # = 2 * conv(w, a_rev)[n-1+m]
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 2).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    dc_da = 2.0 * full_conv[n - 1:2 * n - 1]

    return c_smooth, dc_da, w


def compute_grad_theta(theta, tau):
    """
    Compute gradient of the smoothed objective w.r.t. theta (log-space).
    f(a) = 2n * c_smooth / S^2, a = exp(theta)
    """
    a = np.exp(theta)
    a = np.maximum(a, 1e-300)
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.zeros(n)

    c = fft_convolve_self(a)
    c_smooth, dc_da, w = softmax_max_and_grad(a, c, tau)

    # df/da_m = 2n * (dc_da_m * S - 2*c_smooth) / S^3
    grad_a = 2.0 * n * (dc_da * S - 2.0 * c_smooth) / (S * S * S)

    # Chain rule for log-space
    grad_theta = grad_a * a
    return grad_theta


def obj_value(a):
    """Compute the true objective."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def adam_optimize(theta0, time_budget, tau0=5.0, tau_end=0.01, tau_decay=0.999,
                  lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8,
                  warmup=50, patience=150, verbose=False, tag=""):
    """
    Adam optimization in log-parameter space with softmax-smoothed max.
    """
    a = np.exp(np.clip(theta0, -30, 15))
    n = len(a)
    if a.sum() < 1e-10:
        a = np.exp(np.zeros(n))
    a = a * np.sqrt(2.0 * n) / a.sum()
    theta = np.log(a)

    c = fft_convolve_self(a)
    obj = 2.0 * n * c.max() / (a.sum() ** 2)

    best_obj = obj
    best_theta = theta.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau = tau0
    no_improve = 0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 0.5:
            break

        # Compute gradient in theta space
        grad_theta = compute_grad_theta(theta, tau)

        if not np.all(np.isfinite(grad_theta)):
            theta = best_theta + 0.02 * np.random.randn(n)
            a = np.exp(np.clip(theta, -30, 15))
            a = a * np.sqrt(2.0 * n) / a.sum()
            theta = np.log(a)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            c = fft_convolve_self(a)
            obj = 2.0 * n * c.max() / (a.sum() ** 2)
            no_improve = 0
            continue

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr_t = lr * min(1.0, t / warmup)
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        theta = np.clip(theta, -30, 15)

        # Convert back
        a = np.exp(theta)
        S = a.sum()
        if S < 1e-10 or not np.isfinite(S):
            theta = best_theta + 0.01 * np.random.randn(n)
            a = np.exp(np.clip(theta, -30, 15))
            a = a * np.sqrt(2.0 * n) / a.sum()
            theta = np.log(np.maximum(a, 1e-300))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            continue

        c = fft_convolve_self(a)
        obj = 2.0 * n * c.max() / (S * S)

        if obj < best_obj:
            best_obj = obj
            best_theta = theta.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Decay temperature
        tau = max(tau_end, tau * tau_decay)

        # Restart if stuck
        if no_improve > patience:
            scale = 0.03 + 0.02 * (no_improve / patience)
            theta = best_theta + scale * np.random.randn(n)
            a = np.exp(np.clip(theta, -30, 15))
            a = a * np.sqrt(2.0 * n) / a.sum()
            theta = np.log(np.maximum(a, 1e-300))
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            tau = max(tau * 2, tau0 * 0.3)
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    {tag}[restart] t={elapsed:.1f}s, best={best_obj:.6f}")

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"    {tag}t={elapsed:.1f}s, obj={obj:.6f}, best={best_obj:.6f}, tau={tau:.4f}")

    return best_theta, best_obj


def lbfgs_polish(theta0, time_budget, maxiter=500, verbose=False, tag=""):
    """
    L-BFGS polish in log-space. Uses softmax-smoothed objective with low tau.
    """
    from scipy.optimize import minimize

    a = np.exp(np.clip(theta0, -30, 15))
    a = a * np.sqrt(2.0 * a.size) / a.sum()
    theta = np.log(a).copy()
    n = len(a)

    best_obj = 2.0 * n * fft_convolve_self(a).max() / (a.sum() ** 2)
    best_theta = theta.copy()

    start = time.monotonic()
    tau_current = 0.1

    for round_idx in range(5):
        if time.monotonic() - start > time_budget - 1:
            break

        tau_current = max(0.01, tau_current * 0.5)

        def fun_and_grad(th):
            th = np.clip(th, -30, 15)
            a_local = np.exp(th)
            S_local = a_local.sum()
            if S_local < 1e-10:
                return 1e10, np.zeros_like(th)
            c_local = fft_convolve_self(a_local)
            cs, dc_da, _ = softmax_max_and_grad(a_local, c_local, tau_current)
            f_val = 2.0 * n * cs / (S_local * S_local)
            grad_a = 2.0 * n * (dc_da * S_local - 2.0 * cs) / (S_local * S_local * S_local)
            grad_th = grad_a * a_local
            return f_val, grad_th

        remaining = time_budget - (time.monotonic() - start) - 1
        iters_this_round = min(maxiter, max(10, int(remaining * 100)))

        try:
            res = minimize(fun_and_grad, theta, method='L-BFGS-B',
                           jac=True, options={'maxiter': iters_this_round,
                                              'gtol': 1e-8, 'maxfun': iters_this_round})
            theta_new = np.clip(res.x, -30, 15)
            a_new = np.exp(theta_new)
            obj_new = 2.0 * n * fft_convolve_self(a_new).max() / (a_new.sum() ** 2)

            if obj_new < best_obj:
                best_obj = obj_new
                best_theta = theta_new.copy()
                theta = theta_new.copy()
                if verbose:
                    elapsed = time.monotonic() - start
                    print(f"    {tag}[lbfgs r{round_idx}] t={elapsed:.1f}s, obj={best_obj:.6f}")
        except Exception as e:
            if verbose:
                print(f"    {tag}[lbfgs error] {e}")
            break

    return best_theta, best_obj


def generate_initialization(n, strategy, rng):
    """Generate initial theta vector based on strategy."""
    i = np.arange(n)

    if strategy == 'cosine1':
        a = 1.0 + 0.5 * np.cos(2 * np.pi * 7.0 * i / n)
    elif strategy == 'cosine2':
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 13.0 * i / n + 0.7)
    elif strategy == 'cosine3':
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5.0 * i / n) + 0.2 * np.cos(2 * np.pi * 17.0 * i / n)
    elif strategy == 'gaussian_humps':
        a = np.ones(n) * 0.1
        for _ in range(rng.integers(2, 5)):
            center = rng.random() * n
            width = n * rng.uniform(0.05, 0.15)
            a += np.exp(-0.5 * ((i - center) / width) ** 2)
    elif strategy == 'random_smooth':
        a = np.abs(rng.standard_normal(n)) + 0.05
        # Smooth slightly
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif strategy == 'lowpass':
        a = 1.0 + 0.3 * np.sin(2 * np.pi * i / n) + 0.2 * np.cos(4 * np.pi * i / n) + 0.1 * np.sin(6 * np.pi * i / n)
    elif strategy == 'comb':
        period = int(rng.integers(16, 64))
        a = np.where((i % period) < period // 2, 1.0, 0.3)
    elif strategy == 'ramp_mod':
        a = 1.0 + 0.4 * np.sin(np.pi * i / n)
    elif strategy == 'barker_like':
        # Alternating with varying amplitude
        a = np.where(i % 2 == 0, 1.0, 0.3)
        a = a * (1.0 + 0.2 * np.cos(2 * np.pi * 3.0 * i / n))
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.01)
    a = a * np.sqrt(2.0 * n) / a.sum()
    theta = np.log(a)
    return theta

    remaining = deadline - time.monotonic()
    main_time = 0.55 * remaining
    print(f"\nPhase 2: Deep optimization ({main_time:.0f}s)")

    candidates.sort(key=lambda x: x[0])
    top_k = min(3, len(candidates))
    top_candidates = candidates[:top_k]
    per_candidate_time = main_time / top_k

    for i, (obj_c, theta_c, name) in enumerate(top_candidates):
        if time.monotonic() > deadline - 30:
            break
        remaining_here = min(per_candidate_time, deadline - time.monotonic() - 20)
        if remaining_here < 5:
            break

        print(f"  Deep opt candidate {i} ({name}): obj={obj_c:.6f}, time={remaining_here:.0f}s")

        # Two-stage: broad then narrow
        half_time = remaining_here / 2

        # Stage A: moderate temperature
        theta_best, obj_best = adam_optimize(
            theta_c, half_time,
            tau0=3.0, tau_end=0.05, tau_decay=0.9997,
            lr=0.004, beta1=0.9, beta2=0.999,
            patience=200, tag=f"d{i}a"
        )
        if obj_best < global_best:
            global_best = obj_best
            global_theta = theta_best.copy()
            print(f"    stage A -> {obj_best:.6f}")

        # Stage B: low temperature
        remaining_b = deadline - time.monotonic() - 20
        if remaining_b > 3:
            theta_best, obj_best = adam_optimize(
                theta_best, remaining_b,
                tau0=0.5, tau_end=0.005, tau_decay=0.9999,
                lr=0.002, beta1=0.92, beta2=0.999,
                patience=150, tag=f"d{i}b"
            )
            if obj_best < global_best:
                global_best = obj_best
                global_theta = theta_best.copy()
                print(f"    stage B -> {obj_best:.6f}")

    # --- Phase 3: L-BFGS polish (17% of time) ---
    remaining = deadline - time.monotonic()
    polish_time = 0.17 * remaining
    print(f"\nPhase 3: L-BFGS polish ({polish_time:.0f}s)")

    if global_theta is not None and polish_time > 5:
        theta_best, obj_best = lbfgs_polish(
            global_theta, polish_time,
            maxiter=200, verbose=True, tag="L"
        )
        if obj_best < global_best:
            global_best = obj_best
            global_theta = theta_best.copy()
            print(f"  L-BFGS -> {obj_best:.6f}")

    # --- Phase 4: Final fine Adam (10% of time) ---
    remaining = deadline - time.monotonic() - 20.0
    print(f"\nPhase 4: Final polish ({remaining:.0f}s)")

    if global_theta is not None and remaining > 5:
        theta_best, obj_best = adam_optimize(
            global_theta, remaining,
            tau0=0.1, tau_end=0.001, tau_decay=0.99995,
            lr=0.001, beta1=0.95, beta2=0.999,
            patience=100, tag="fin"
        )
        if obj_best < global_best:
            global_best = obj_best
            global_theta = theta_best.copy()
            print(f"  Final -> {obj_best:.6f}")

    # Try again with L-BFGS if time
    remaining = deadline - time.monotonic() - 20.0
    if global_theta is not None and remaining > 5:
        theta_best, obj_best = lbfgs_polish(
            global_theta, remaining,
            maxiter=100, verbose=True, tag="L2"
        )
        if obj_best < global_best:
            global_best = obj_best
            global_theta = theta_best.copy()
            print(f"  L2 -> {obj_best:.6f}")

    # --- Final verification ---
    if global_theta is None:
        global_theta = np.zeros(n)

    final_a = np.exp(np.clip(global_theta, -30, 15))
    final_a = np.clip(final_a, 0.0, 1000.0)
    if final_a.sum() < 0.01:
        final_a = np.ones(256)

    final_list = final_a.tolist()
    ver = evaluate_sequence(final_list)
    print(f"\nFINAL: local_obj={global_best:.6f}, verified={ver:.6f}")

    if not np.isfinite(ver):
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
        print(f"Fallback: obj={ver:.6f}")

    return final_list
