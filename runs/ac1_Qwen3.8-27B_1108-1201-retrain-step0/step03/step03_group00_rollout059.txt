def line_search(y, d, S, M_current, pre, prev_alpha, trial_limit=6, cap_base=0.5):

        return None, M_current, 0.0
    u = d / nrm
    neg = u < 0.0
    cap = cap_base
    if np.any(neg):
        denom = -u[neg]
        denom = np.maximum(denom, 1e-300)
        max_alpha = 0.9 * float(np.min(y[neg] / denom))
        if max_alpha < cap:
            cap = max_alpha
    if cap <= 1e-10:
        # Try unblocking
        u2 = u.copy()
        blocked = (y < 1e-9) & (u < 0.0)
        if np.any(blocked):
            u2[blocked] = 0.0
            u2 = u2 - u2.mean()
            nrm2 = float(np.linalg.norm(u2))
            if nrm2 > 1e-12:
                u = u2 / nrm2
                neg = u < 0.0
                if np.any(neg):
                    denom = np.maximum(-u[neg], 1e-300)
                    cap = min(cap_base, 0.9 * float(np.min(y[neg] / denom)))

        raw.append(min(0.5*prev_alpha, cap))

            opts.append(float(val))

def perturb(a, sigma, rng, S):
    n = len(a)
    if sigma <= 0:
       return a.copy()
    # log-normal multiplicative
    v = a * np.exp(rng.normal(0.0, sigma, n))
    # additive for zeros with small scale
    base = max(1e-8, float(np.mean(a)))
    v = v + rng.normal(0.0, max(0.0, sigma) * base * 0.05, n)

def resample_sequence(a, n):
    a=np.asarray(a,float)
    m=len(a)
    if m==n: return a.copy()
    if m==0: return np.ones(n)
    x_old=np.linspace(0.0,1.0,m,endpoint=False)
    x_new=np.linspace(0.0,1.0,n,endpoint=False)
    return np.interp(x_new, x_old, a)

```python
"""
Search non-negative sequences minimizing
    2*n*max(conv(a,a)) / sum(a)^2

Method:
- Fix scale to sum(a) = sqrt(2n), so objective = max coefficient.
- Smooth max by softmax / plateau / top-k weights over convolution lags.
- Use corrected FFT convolution and gradient of weighted self-convolution.
- Alternating projected gradient line search on the exact max objective.
- Lightweight dual weight refinement to approximate KKT active-set weights.
- Multi-start, noise/mixing restarts, polishing, and optional length expansion.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def next_power_two(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def make_pre(n):
    size = 2 * n - 1
    L = next_power_two(size)
    return (L, size)


def project_sum(v, S):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s <= 1e-15:
        return np.full(len(v), S / max(1, len(v)), dtype=np.float64)
    return v * (S / s)


def convolve_self_fast(a, pre):
    L, size = pre
    A = fft(a, L)
    c = ifft(A * A).real
    return c[:size]


def convolve_self_and_a(a, pre):
    L, size = pre
    A = fft(a, L)
    c = ifft(A * A).real
    return c[:size], A


def objective_score(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = len(a)
    if n == 0:
        return float("inf")
    if not np.all(np.isfinite(a)):
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    pre = make_pre(n)
    c = convolve_self_fast(a, pre)
    if not np.all(np.isfinite(c)):
        return float("inf")
    m = float(c.max())
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))


def softmax_weights(c, beta):
    n = len(c)
    if n == 0:
        return np.array([], dtype=np.float64)
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        w = np.zeros(n, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    arg = beta * (c - cmax)
    arg = np.clip(arg, -745.0, 0.0)
    e = np.exp(arg)
    total = float(e.sum())
    if total <= 0.0 or not np.isfinite(total):
        w = np.zeros(n, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / total


def topk_weights(c, k, power=1.0):
    n = len(c)
    k = int(k)
    if k <= 0:
        k = 1
    if k >= n:
        w = np.maximum(c, 0.0)
    else:
        idx = np.argpartition(c, -k)[-k:]
        w = np.zeros(n, dtype=np.float64)
        w[idx] = np.maximum(c[idx], 0.0)
    if power != 1.0:
        w = np.power(np.maximum(w, 0.0), power)
    total = float(w.sum())
    if total <= 0.0 or not np.isfinite(total):
        w = np.zeros(n, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / total
    return w


def plateau_weights(c, tol, power=1.0):
    n = len(c)
    cmax = float(np.max(c))
    if cmax <= 0.0 or not np.isfinite(cmax):
        w = np.zeros(n, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    tol = max(float(tol), 1e-12)
    thr = cmax - tol
    mask = c >= thr
    mcnt = int(np.count_nonzero(mask))
    if mcnt < 8:
        k = max(8, min(n, n // 200))
        return topk_weights(c, k, power)
    w = np.zeros(n, dtype=np.float64)
    vals = np.maximum(c[mask] - thr, 0.0)
    if power != 1.0:
        vals = np.power(vals, power)
    w[mask] = vals
    total = float(w.sum())
    if total <= 0.0 or not np.isfinite(total):
        return topk_weights(c, max(8, mcnt), power)
    return w / total


def weighted_gradient(a, w, A, pre):
    L, _ = pre
    n = len(a)
    W = fft(w, L)
    corr = ifft(W * np.conj(A)).real
    return 2.0 * corr[:n]


def lag_gradient(a, k):
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k - idx]
    return g


def refine_weights(a, w, A, pre, iters=2, eta=1.0):
    n = len(a)
    L, size = pre
    w = np.asarray(w, dtype=np.float64)
    active = w > 0.0
    if np.count_nonzero(active) == 0:
        return w

    a_scale = max(1e-12, float(np.max(a)))
    for _ in range(int(iters)):
        g = weighted_gradient(a, w, A, pre)
        if not np.all(np.isfinite(g)):
            break
        h = 0.5 * g
        support = a > (1e-7 * a_scale + 1e-12)
        nsup = int(np.count_nonzero(support))
        if nsup == 0:
            break
        lam = float(h[support].mean())
        r = np.zeros(n, dtype=np.float64)
        r[support] = lam - h[support]
        scale_r = float(np.max(np.abs(r)))
        if scale_r <= 1e-14:
            break

        R = fft(r, L)
        p = ifft(R * A).real[:size]
        p_act = p[active]
        if p_act.size == 0:
            break
        pmax = float(np.max(np.abs(p_act)))
        if pmax <= 1e-14:
            break
        inc = eta * (p_act / pmax)
        inc = np.clip(inc, -50.0, 50.0)

        w_new = w.copy()
        w_new[active] = w_new[active] * np.exp(inc)
        total = float(w_new.sum())
        if total <= 0.0 or not np.isfinite(total):
            break
        w = w_new / total
    return w


def line_search(y, d, S, M_current, pre, prev_alpha, trial_limit=6, cap_base=0.6):
    d = np.asarray(d, dtype=np.float64)
    nrm = float(np.linalg.norm(d))
    if nrm < 1e-15 or not np.isfinite(nrm):
        return None, float(M_current), 0.0

    u = d / nrm
    cap = float(cap_base)

    neg = u < 0.0
    if np.any(neg):
        denom = np.maximum(-u[neg], 1e-300)
        max_alpha = 0.9 * float(np.min(y[neg] / denom))
        if max_alpha < cap:
            cap = max_alpha

    if cap <= 1e-10:
        u2 = u.copy()
        blocked = (y < 1e-9) & (u < 0.0)
        if np.any(blocked):
            u2[blocked] = 0.0
            u2 = u2 - u2.mean()
            nrm2 = float(np.linalg.norm(u2))
            if nrm2 > 1e-12:
                u = u2 / nrm2
                neg = u < 0.0
                if np.any(neg):
                    denom = np.maximum(-u[neg], 1e-300)
                    cap = min(float(cap_base), 0.9 * float(np.min(y[neg] / denom)))
                else:
                    cap = float(cap_base)

    if cap <= 1e-12:
        return None, float(M_current), 0.0

    raw = [cap, 0.5 * cap, 0.25 * cap, 0.125 * cap, 0.0625 * cap,
           0.03125 * cap, 0.015625 * cap, 0.0078125 * cap]
    if prev_alpha > 0.0 and np.isfinite(prev_alpha):
        raw.append(min(float(prev_alpha), cap))
        raw.append(min(0.5 * float(prev_alpha), cap))
        raw.append(min(2.0 * float(prev_alpha), cap))

    opts = []
    for val in raw:
        if val > 1e-12 and val <= cap + 1e-12:
            opts.append(float(val))
    opts.sort(reverse=True)

    unique = []
    for val in opts:
        if not unique or abs(val - unique[-1]) > 1e-12:
            unique.append(val)
    opts = unique[:int(trial_limit)]

    best_val = float(M_current)
    best_cand = None
    best_alpha = 0.0

    for alpha in opts:
        cand = y + alpha * u
        cand = np.maximum(cand, 0.0)
        s = float(cand.sum())
        if s <= 1e-15:
            continue
        cand = cand * (S / s)
        c = convolve_self_fast(cand, pre)
        M2 = float(c.max())
        if np.isfinite(M2) and M2 < best_val:
            best_val = M2
            best_cand = cand.copy()
            best_alpha = alpha

    return best_cand, best_val, best_alpha


def perturb(a, sigma, rng, S):
    n = len(a)
    a = np.asarray(a, dtype=np.float64)
    if sigma <= 0.0:
        v = a.copy()
    else:
        v = a * np.exp(rng.normal(0.0, sigma, n))
        base = max(1e-8, float(np.mean(a)))
        v = v + rng.normal(0.0, sigma * base * 0.05, n)
    return project_sum(v, S)


def resample_sequence(a, n):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if len(a) == 0:
        return np.ones(n, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    m = len(a)
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(x_new, x_old, a)


def generate_initials(n, rng):
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    inits = []
    inits.append(np.ones(n, dtype=np.float64))
    inits.append(0.5 + 0.5 * np.cos(2.0 * np.pi * t))
    inits.append(0.54 + 0.46 * np.cos(2.0 * np.pi * t))
    inits.append(0.42 - 0.5 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t))
    inits.append(np.sin(np.pi * t) ** 2)
    inits.append(np.sin(np.pi * t) ** 4)
    inits.append((t ** 2) * ((1.0 - t) ** 2) + 1e-6)
    inits.append((t ** 3) * ((1.0 - t) ** 1) + 1e-6)
    inits.append(1.0 + 0.3 * np.sin(4.0 * np.pi * t) + 0.2 * np.cos(10.0 * np.pi * t))
    inits.append(rng.random(n) + 0.05)
    inits.append(np.exp(rng.normal(0.0, 0.4, n)))

    for nb in (3, 5, 7, 9):
        heights = rng.uniform(0.2, 1.5, nb)
        old_t = np.linspace(0.0, 1.0, nb, endpoint=False)
        inits.append(np.interp(t, old_t, heights))
    return inits


def run_optimizer(a_init, n, S, max_time, rng, name="", beta_init=20.0,
                  beta_max=3000.0, beta_rate=0.0012, trial_limit=6,
                  cap_base=0.6, use_momentum=True, refine_every=0,
                  verbose=True):
    a_init = np.asarray(a_init, dtype=np.float64).reshape(-1)
    if len(a_init) != n:
        a_init = resample_sequence(a_init, n)

    x = project_sum(a_init, S)
    y = x.copy()
    x_prev = x.copy()
    pre = make_pre(n)
    start = time.monotonic()
    end = start + max(float(max_time), 1.0)

    c0 = convolve_self_fast(x, pre)
    best_M = float(c0.max())
    best_a = x.copy()

    t = 1.0
    prev_alpha = 0.05
    it = 0
    last_best_it = 0
    restarts = 0
    last_print_it = 0

    k_list = []
    for kv in (128, 256, 512, 1024, 2048, 4096, 8192, 16384):
        if kv < 2 * n:
            k_list.append(kv)
    if not k_list:
        k_list.append(max(1, n))

    delta_fracs = (0.001, 0.003, 0.008, 0.02, 0.05)

    if verbose:
        print("OPT start %s n=%d init_score=%.10f" % (name, n, best_M), flush=True)

    while time.monotonic() < end - 2.0:
        it += 1

        if it - last_best_it > 4000:
            if restarts % 2 == 0:
                noise = 0.25 / (1.0 + 0.6 * restarts)
                x = perturb(best_a, noise, rng, S)
            else:
                mix = 0.2 + 0.4 * float(rng.random())
                v = best_a + mix * (x - best_a)
                x = project_sum(v, S)
            y = x.copy()
            x_prev = x.copy()
            t = 1.0
            prev_alpha = 0.05
            restarts += 1
            last_best_it = it
            if verbose and restarts < 30:
                print("OPT %s restart %d at iter %d" % (name, restarts, it), flush=True)

        c, A = convolve_self_and_a(y, pre)
        if not (np.all(np.isfinite(c)) and np.all(np.isfinite(A))):
            x = best_a.copy()
            y = x.copy()
            x_prev = x.copy()
            t = 1.0
            continue

        if c.size and float(c.min()) < 0.0:
            c_w = c.copy()
            c_w[c_w < 0.0] = 0.0
        else:
            c_w = c

        M = float(c.max())
        if not np.isfinite(M) or M <= 0.0:
            break

        if M < best_M - 1e-12:
            best_M = M
            best_a = y.copy()
            last_best_it = it

        exp_arg = beta_rate * float(it)
        if exp_arg > 50.0:
            beta = float(beta_max)
        else:
            beta = float(beta_init) * float(np.exp(exp_arg))
            if not np.isfinite(beta) or beta > beta_max:
                beta = float(beta_max)
            if beta < 1.0:
                beta = 1.0

        r = float(rng.random())
        if M <= best_M * (1.0 + 1e-6):
            if r < 0.22:
                idx_choice = (it // 900) % len(k_list)
                k = k_list[idx_choice]
                power = 1.0 if (it // 900) % 2 == 0 else 2.0
                w = topk_weights(c_w, k, power)
            elif r < 0.47:
                df = delta_fracs[(it // 500) % len(delta_fracs)]
                w = plateau_weights(c_w, max(1e-10, M * df), 1.0)
            else:
                w = softmax_weights(c_w, beta)
        else:
            w = softmax_weights(c_w, max(1.0, 0.2 * beta))

        if refine_every > 0 and it % refine_every == 0:
            if np.count_nonzero(w > 0.0) >= 32:
                w = refine_weights(y, w, A, pre, iters=2, eta=1.0)

        g = weighted_gradient(y, w, A, pre)
        if not np.all(np.isfinite(g)):
            g = lag_gradient(y, int(np.argmax(c_w)))

        d = -g
        d = d - d.sum() / d.size
        if not np.all(np.isfinite(d)):
            x = perturb(best_a, 0.1, rng, S)
            y = x.copy()
            x_prev = x.copy()
            t = 1.0
            prev_alpha = 0.05
            continue

        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            x = perturb(best_a, 0.1, rng, S)
            y = x.copy()
            x_prev = x.copy()
            t = 1.0
            prev_alpha = 0.05
            continue

        cand, Mnew, alpha = line_search(y, d, S, M, pre, prev_alpha, trial_limit, cap_base)
        if cand is None:
            kk = int(np.argmax(c_w))
            g2 = lag_gradient(y, kk)
            d2 = -g2
            d2 = d2 - d2.sum() / d2.size
            cand2, Mnew2, alpha2 = line_search(y, d2, S, M, pre, prev_alpha, trial_limit, cap_base)
            if cand2 is not None:
                cand, Mnew, alpha = cand2, Mnew2, alpha2
            else:
                noise = 0.03 / (1.0 + 0.5 * restarts)
                y = perturb(y, noise, rng, S)
                x = y.copy()
                x_prev = y.copy()
                t = 1.0
                prev_alpha = 0.03
                continue

        if Mnew < best_M - 1e-12:
            best_M = Mnew
            best_a = cand.copy()
            last_best_it = it

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y_new = cand + ((t - 1.0) / t_new) * (cand - x_prev)
            y_new = project_sum(y_new, S)
            if not np.all(np.isfinite(y_new)):
                y_new = cand.copy()
            x_prev = cand
            x = cand
            t = t_new
            y = y_new
        else:
            x = cand
            x_prev = cand.copy()
            y = cand.copy()
            t = 1.0

        prev_alpha = alpha

        if it - last_best_it > 1500:
            t = 1.0
            y = x.copy()

        if verbose and (it == 1 or it - last_print_it >= 2000):
            print("OPT %s iter=%d best=%.10f current=%.10f alpha=%.5g" %
                  (name, it, best_M, M, alpha), flush=True)
            last_print_it = it

    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = project_sum(best_a, S)
    cbest = convolve_self_fast(best_a, pre)
    if np.all(np.isfinite(cbest)):
        bm = float(cbest.max())
        if np.isfinite(bm) and bm < best_M:
            best_M = bm

    if verbose:
        print("OPT end %s best=%.10f" % (name, best_M), flush=True)
    return best_a, best_M


def propose_candidate():

        parent = None
```
