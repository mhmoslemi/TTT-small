    a = np.maximum(a, 0.0)
    a *= S / a.sum()

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based active peak elimination
combined with diverse initialization and multi-phase optimization.

Key innovation over subgradient methods: Instead of computing a single
subgradient direction targeting the worst peak, we solve a Linear Program
that finds the optimal trade-off direction across all simultaneously active
peaks. This leverages the linear structure of the first-order model of
the quadratic autocorrelation, enabling more effective coordinated reduction
of multiple peaks per iteration.

Phases:
  1. Diverse initialization (spectral, structured, given)
  2. LP-based active peak elimination (main optimization)
  3. Greedy peak shaving polish
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def normalized_max_b(a, c=None):
    """Return max_b assuming a is normalized to sum=sqrt(2n)."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    S_target = np.sqrt(2.0 * n)
    scale = S_target / S
    return (c * scale * scale).max()


def build_grad_matrix(a, active_indices):
    """
    Build gradient matrix G where G[j, m] = dc[k_j]/da[m].
    dc[k]/da[m] = 2*a[k-m] if 0 <= k-m < n, else 0.
    """
    n = len(a)
    rows = []
    cols = []
    data = []
    for j, k in enumerate(active_indices):
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        for m in range(m_lo, m_hi + 1):
            rows.append(j)
            cols.append(m)
            data.append(2.0 * a[k - m])
    if not data:
        return csr_matrix((len(active_indices), n))
    return csr_matrix((data, (rows, cols)), shape=(len(active_indices), n))


def solve_lp_direction(a, c, n_active=80, trust_radius=0.1):
    """
    Solve LP to find direction d minimizing max of active peaks (linearized).
    min t s.t. c[k] + G[k,:].d <= t for active k, a+d>=0, |d|<=r, sum(d)=0
    """
    n = len(a)
    max_b = c.max()
    threshold = max_b - 0.08 * max_b
    active_mask = c >= threshold
    active_indices = np.where(active_mask)[0]

    if len(active_indices) > n_active:
        order = np.argsort(c[active_indices])[::-1]
        worst = active_indices[order[:n_active]]
    else:
        worst = active_indices

    if len(worst) == 0:
        return None

    G = build_grad_matrix(a, worst)
    c_active = c[worst]

    d = cp.Variable(n)
    t = cp.Variable()

    constraints = [
        cp.sum(d) == 0,
        d <= trust_radius,
        -d <= np.maximum(a, 1e-12),
    ]

    # c[k] + G[k,:].d <= t  =>  G[k,:].d - t <= -c[k]
    constraints.append(G @ d - t <= -c_active)

    prob = cp.Problem(cp.Minimize(t), constraints)

    try:
        prob.solve(solver=cp.CLARABEL, max_iter=500)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=5000)
        except Exception:
            return None

    if d.value is None:
        return None

    d_val = d.value.copy()
    d_val = np.clip(d_val, -trust_radius, trust_radius)
    d_val -= d_val.sum() / n
    return d_val


def greedy_peak_shave(a, S, max_iters=2000, verbose=False):
    """
    Greedy: find worst lag, reduce the largest contributing pair.
    """
    n = len(a)
    a = a.copy()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > max_iters * 0.05:
            break
        k_max = np.argmax(c)
        lag = k_max - (n - 1)
        abs_lag = abs(lag)

        if abs_lag == 0:
            # c[n-1] = sum a[i]^2; reduce largest a[i]
            idx = np.argmax(a)
            a[idx] *= 0.92
        else:
            if lag > 0:
                prods = a[:n - abs_lag] * a[abs_lag:]
            else:
                prods = a[abs_lag:] * a[:n - abs_lag]
            # Reduce top contributors
            top = np.argsort(prods)[::-1][:5]
            for idx in top:
                if lag > 0:
                    if a[idx] >= a[idx + abs_lag]:
                        a[idx] *= 0.92
                    else:
                        a[idx + abs_lag] *= 0.92
                else:
                    if a[idx + abs_lag] >= a[idx]:
                        a[idx + abs_lag] *= 0.92
                    else:
                        a[idx] *= 0.92

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a *= S / s
        c = fft_convolve_self(a)
        if c.max() < best_obj:
            best_obj = c.max()
            best_a = a.copy()

        if verbose and it % 200 == 0:
            print(f"      greedy iter={it}, obj={c.max():.6f}, best={best_obj:.6f}")

    return best_a, best_obj


def lp_optimize(a0, time_budget, n_active=80, trust_radius=0.1, nperturb=0.02, verbose=True):
    """Main LP-based optimization loop."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.maximum(a, 0.0)
    a *= S / a.sum()

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    best_iter = 0

    start = time.monotonic()
    iter_no = 0
    current_trust = trust_radius

    while time.monotonic() - start < time_budget:
        d_val = solve_lp_direction(a, c, n_active=n_active, trust_radius=current_trust)
        if d_val is None:
            d_val = -np.ones(n) * 0.001
        d_val -= d_val.sum() / n

        # Line search
        best_step = 0.0
        best_trial_obj = c.max()
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_trial = a + alpha * d_val
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial *= S / s
            c_trial = fft_convolve_self(a_trial)
            trial_obj = c_trial.max()
            if trial_obj < best_trial_obj:
                best_trial_obj = trial_obj
                best_step = alpha

        if best_step > 0:
            a = a + best_step * d_val
            a = np.maximum(a, 0.0)
            a *= S / a.sum()
            c = fft_convolve_self(a)
            cur_obj = c.max()

            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                best_iter = iter_no
                current_trust = max(0.01, current_trust * 0.97)
            else:
                current_trust = max(0.01, current_trust * 0.9)
        else:
            current_trust = max(0.01, current_trust * 0.8)

        # Stagnation: perturb
        if iter_no - best_iter > 30:
            noise = np.random.randn(n) * nperturb * S / np.sqrt(n)
            a += noise
            a = np.maximum(a, 0.0)
            a *= S / a.sum()
            c = fft_convolve_self(a)
            current_trust = trust_radius
            best_iter = iter_no
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    [perturb] iter={iter_no}, obj={c.max():.6f}, best={best_obj:.6f}, t={elapsed:.1f}s")

        if verbose and iter_no % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    LP iter={iter_no}, obj={c.max():.6f}, best={best_obj:.6f}, "
                  f"trust={current_trust:.4f}, t={elapsed:.1f}s")

        iter_no += 1

    return best_a, best_obj


def subgradient_polish(a0, time_budget, top_k=3, eta0=0.05, momentum=0.9, verbose=False):
    """Fine-tuning subgradient method for the final phase."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.maximum(a, 0.0)
    a *= S / a.sum()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    v = np.zeros(n)
    eta = eta0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        # Subgradient of max: average over top_k peaks
        top_idx = np.argsort(c)[::-1][:top_k]
        c_max = c[top_idx[0]]
        w = np.exp(2.0 * (c[top_idx] - c_max) / max(c_max, 1e-10))
        w /= w.sum()

        # Gradient
        grad = np.zeros(n)
        for j, k in enumerate(top_idx):
            m_lo = max(0, k - (n - 1))
            m_hi = min(n - 1, k)
            for m in range(m_lo, m_hi + 1):
                grad[m] += 2.0 * w[j] * a[k - m]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v /= vnorm

        a = a * np.exp(-eta * v)
        a = np.maximum(a, 1e-30)
        a *= S / a.sum()
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        else:
            eta *= 0.95

        eta *= 0.9999
        if verbose:
            elapsed = time.monotonic() - start
            if int((iter_no:=0)) % 100 == 0:
                pass

    return best_a, best_obj


def generate_initializations(n, rng):
    """Generate diverse initializations for a given length n."""
    S = np.sqrt(2.0 * n)
    i = np.arange(n)
    candidates = []

    # Uniform
    a = np.ones(n) * S / n
    candidates.append(('uniform', a.copy()))

    # Cosine-like
    for freq in [3, 7, 13]:
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * i / n)
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
        candidates.append((f'cos{freq}', a.copy()))

    # Two humps
    a = (np.exp(-0.5 * ((i - 0.25 * n) / (n * 0.1)) ** 2) +
         np.exp(-0.5 * ((i - 0.75 * n) / (n * 0.1)) ** 2))
    a = np.maximum(a, 0.01)
    a *= S / a.sum()
    candidates.append(('2hump', a.copy()))

    # Random smooth
    a = np.abs(fft(rng.standard_normal(4 * n)).real[:n])
    a = np.maximum(a, 0.01)
    a *= S / a.sum()
    candidates.append(('rand_smooth', a.copy()))

    # Power-law modulated
    a = (1.0 + 0.3 * np.sin(2.0 * np.pi * 5 * i / n) * np.sin(2.0 * np.pi * 11 * i / n))
    a = np.maximum(a, 0.01)
    a *= S / a.sum()
    candidates.append(('modulated', a.copy()))

    return candidates
