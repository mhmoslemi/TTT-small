"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FISTA with adaptive log-sum-exp smoothing, structured perturbations,
field-equalization steps, and coordinate descent polish.
Key innovations: damped momentum, restoration-based FISTA, spectral
field equalization for the active lags, and aggressive high-beta polishing.
"""
import numpy as np
import time
import math
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = w[-1] - S / n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def hard_topk_gradient(a, c, top_k):
    """Gradient using exactly top-k active lags (uniform weighting)."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    w_full = np.zeros(2 * n - 1)
    w_full[top_indices] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def smooth_topk_gradient(a, c, top_k, temperature):
    """Gradient using top-k lags with softmax weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices].copy()
    tv_max = top_vals.max()
    inv_temp = 1.0 / max(temperature, 1e-10)
    weights = np.exp(inv_temp * top_vals - inv_temp * tv_max)
    weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def field_equalization_step(a, c, n, S, top_k=10, relaxation=0.5):
    """
    Field equalization: find direction that equalizes the 'field' from top lags.
    Uses FFT trick: in Fourier domain, a_new ~ 1 / FFT(v) where v is the top-lag indicator.
    """
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices]
    # Softmax weights on the top lags
    tv_max = top_vals.max()
    weights = np.exp(top_vals - tv_max)
    weights = weights / weights.sum()
    
    v = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        v[int(idx)] = weights[i]
    
    # Make v symmetric (convolution weight should be symmetric)
    v_sym = 0.5 * (v + v[::-1])
    
    N = 1 << (3 * n - 2).bit_length()
    V = fft(v_sym, N)
    ones = np.zeros(N)
    ones[:n] = 1.0
    O = fft(ones, N)
    
    # Avoid division by zero
    V_safe = np.where(np.abs(V) > 1e-15, V, 1e-15 + 0j)
    A_new = O / V_safe
    a_new = np.real(ifft(A_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    
    if a_new.sum() < 1e-15:
        return a.copy()
    a_new = a_new * (S / a_new.sum())
    
    # Relaxation step
    a_relaxed = (1 - relaxation) * a + relaxation * a_new
    a_relaxed = np.maximum(a_relaxed, 0.0)
    if a_relaxed.sum() < 1e-15:
        return a.copy()
    a_relaxed = a_relaxed * (S / a_relaxed.sum())
    
    return a_relaxed


def run_fista_improved(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=2000.0,
                       momentum_damp=0.95, restore_freq=200, perturb_every=150):
    """
    Improved FISTA with damped momentum, periodic restoration, and structured perturbations.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    L = beta_init * 4.0  # Initial Lipschitz estimate
    stall_count = 0
    last_best = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        # Exponential beta schedule
        beta = min(beta_max, beta_init * math.exp(0.005 * it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = y.copy()
            stall_count = 0
            last_best = cur_max
        elif it % 20 == 0:
            stall_count += 1

        # Gradient computation
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / L
        f_y = cur_max
        x_new = None
        for bt in range(6):
            x_candidate = project_simplex(y - step * grad, S)
            if bt > 0:
                c_cand = fft_convolve_self(x_candidate)
                # Use smoothed objective for backtracking
                c_cand_smooth = (1.0 / beta) * np.log(np.sum(np.exp(beta * (c_cand - c_cand.max()))) + 1e-300) + c_cand.max()
                c_y_smooth = (1.0 / beta) * np.log(np.sum(np.exp(beta * (c - c.max()))) + 1e-300) + c.max()
                if c_cand_smooth <= c_y_smooth - 1e-4 * step * gnorm * gnorm:
                    x_new = x_candidate
                    break
                step *= 0.5
            else:
                x_new = x_candidate

        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        # Adaptive L update
        c_x = fft_convolve_self(x_new)
        if c_x.max() > f_y * 1.1:
            L *= 2.0
        elif it % 50 == 0:
            L = max(L * 0.98, beta * 2.0)

        # Damped Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        alpha = (t_momentum - 1.0) / t_new * momentum_damp
        y = x_new + alpha * (x_new - a)
        y = np.maximum(y, 0.0)
        if y.sum() < 1e-15:
            y = a.copy()
        else:
            y = y * (S / y.sum())
        a = x_new
        t_momentum = t_new

        # Periodic restoration to best
        if it % restore_freq == 0:
            y = best_a.copy()
            a = best_a.copy()
            t_momentum = 1.0

        # Structured perturbation to escape local minima
        if stall_count > 40 and it % perturb_every == 0:
            # Add a random sinusoidal component
            freq = rng.integers(2, max(3, n // 100))
            phase = rng.random() * 2 * np.pi
            amp = 0.05 * S / n
            tt = np.arange(n, dtype=np.float64)
            perturbation = amp * np.sin(2 * np.pi * freq * tt / n + phase)
            y = project_simplex(y + perturbation, S)
            t_momentum = 1.0
            stall_count = 0
        elif stall_count > 80 and rng.random() < 0.2:
            # Larger random perturbation
            noise_scale = 0.02 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_momentum = 1.0
            stall_count = 0

    # Final: compare with best_a
    c_best = fft_convolve_self(best_a)
    return best_a, c_best.max()

            best_score = cur_max
            best_a = a.copy()

        # Field equalization step with adaptive relaxation
        relaxation = max(0.1, 0.8 * (1.0 - it / max_time))
        a_new = field_equalization_step(a, c, n, S, top_k=top_k, relaxation=relaxation)
        
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller relaxation
            a_new2 = field_equalization_step(a, c, n, S, top_k=top_k, relaxation=0.1)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < cur_max:
                a = a_new2
            else:
                break  # No improvement possible with this method

    return best_a, best_score


def run_coordinate_descent(a_init, n, S, max_time, rng):
    """Coordinate descent targeting top lags by moving mass between pairs."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()

        # Find top lag
        k_star = int(np.argmax(c))
        
        # Find the contributing pairs for this lag
        i_min = max(0, k_star - (n - 1))
        i_max = min(k_star, n - 1)
        if i_min > i_max:
            break
        indices = np.arange(i_min, i_max + 1)
        partner = k_star - indices
        products = a[indices] * a[partner]
        
        # Find the pair with the largest product
        max_prod_idx = int(np.argmax(products))
        p = int(indices[max_prod_idx])
        q = int(partner[max_prod_idx])
        
        # Try to move mass: decrease the larger, increase a random small element
        if p == q:
            continue
        
        # Find a good target: element with small contribution to the peak
        target_scores = np.zeros(n)
        target_scores[indices] = products
        # We want to move mass TO an element that doesn't contribute much to ANY top lag
        top_lags = np.argsort(c)[::-1][:5]
        for lag in top_lags:
            lag = int(lag)
            li_min = max(0, lag - (n - 1))
            li_max = min(lag, n - 1)
            if li_min <= li_max:
                li_indices = np.arange(li_min, li_max + 1)
                li_partner = lag - li_indices
                li_products = a[li_indices] * a[li_partner]
                target_scores[li_indices] += li_products
                target_scores[li_partner] += li_products
        
        # Choose target from elements with lowest target score
        sorted_targets = np.argsort(target_scores)
        # Try a few targets
        improved = False
        for trial in range(5):
            target_idx = int(sorted_targets[trial])
            if target_idx == p or target_idx == q:
                continue
            
            # Move delta from the larger of (p, q) to target_idx
            donor = p if a[p] >= a[q] else q
            delta = 0.1 * a[donor]
            if delta < 1e-15:
                continue
            
            a_trial = a.copy()
            a_trial[donor] -= delta
            a_trial[target_idx] += delta
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial * (S / a_trial.sum())
            
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_max:
                a = a_trial
                cur_max = c_trial.max()
                improved = True
                break
        
        if not improved and it % 100 == 0:
            # Try random perturbation
            noise = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)

    return best_a, best_score


def make_anti_palindromic_init(n, rng, S):
    """Create anti-palindromic initialization."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n) * S / n
    a = a * (S / a.sum())
    return a


def make_multi_sinusoidal_init(n, rng, S, num_freqs=5):
    """Create initialization with multiple sinusoidal components."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, max(3, n // 32))
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.3
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
    1. Start from parent or structured initialization.
    2. Run improved FISTA (damped momentum, adaptive beta, structured perturbations).
    3. Apply field equalization for structural improvement.
    4. Polish with coordinate descent on top lags.
    5. Final high-beta FISTA with very small steps.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    n = 8192
    if parent is not None and len(parent) >= 64:
        n = len(parent)

    S = np.sqrt(2.0 * n)
    print("Starting: n=%d, parent=%s, budget=%.0fs" % (n, "yes" if parent is not None else "no", budget_s))

    # Initial sequence
    if parent is not None:
        a_init = safe_normalize(parent, n)
    else:
        a_init = make_anti_palindromic_init(n, rng, S)

    c_init = fft_convolve_self(a_init)
    init_score = c_init.max()
    global_best = init_score
    global_a = a_init.copy()
    print("Initial score: %.6f" % init_score)

    # === Phase 1: Main FISTA optimization (55% of time) ===
    phase1_time = max(30.0, (deadline - time.monotonic()) * 0.55)
    print("\n=== Phase 1: FISTA improved (%.0f sec) ===" % phase1_time)
    
    a1, s1 = run_fista_improved(a_init, n, S, phase1_time, rng,
                                  beta_init=5.0, beta_max=1500.0,
                                  momentum_damp=0.95, restore_freq=150, perturb_every=100)
    a1 = safe_normalize(a1, n)
    c1 = fft_convolve_self(a1)
    score1 = c1.max()
    if score1 < global_best:
        global_best = score1
        global_a = a1.copy()
        print("Phase 1 FISTA: %.6f (improvement: %.4f%%)" % (score1, 100*(init_score-score1)/init_score))

    # === Phase 2: Hard top-k FISTA (15% of time) ===
    remaining = deadline - time.monotonic()
    phase2_time = max(10.0, remaining * 0.15)
    print("\n=== Phase 2: Hard top-k FISTA (%.0f sec) ===" % phase2_time)
    
    a2, s2 = run_fista_hard(global_a, n, S, phase2_time, rng,
                              top_k_init=3, top_k_max=50)
    a2 = safe_normalize(a2, n)
    c2 = fft_convolve_self(a2)
    score2 = c2.max()
    if score2 < global_best:
        global_best = score2
        global_a = a2.copy()
        print("Phase 2: %.6f" % score2)

    # === Phase 3: Field equalization (10% of time) ===
    remaining = deadline - time.monotonic()
    phase3_time = max(5.0, remaining * 0.10)
    print("\n=== Phase 3: Field equalization (%.0f sec) ===" % phase3_time)
    
    a3, s3 = run_field_equalization(global_a, n, S, phase3_time, rng, top_k=8)
    a3 = safe_normalize(a3, n)
    c3 = fft_convolve_self(a3)
    score3 = c3.max()
    if score3 < global_best:
        global_best = score3
        global_a = a3.copy()
        print("Phase 3: %.6f" % score3)

    # === Phase 4: Coordinate descent (10% of time) ===
    remaining = deadline - time.monotonic()
    phase4_time = max(5.0, remaining * 0.10)
    print("\n=== Phase 4: Coordinate descent (%.0f sec) ===" % phase4_time)
    
    a4, s4 = run_coordinate_descent(global_a, n, S, phase4_time, rng)
    a4 = safe_normalize(a4, n)
    c4 = fft_convolve_self(a4)
    score4 = c4.max()
    if score4 < global_best:
        global_best = score4
        global_a = a4.copy()
        print("Phase 4: %.6f" % score4)

    # === Phase 5: Final high-beta FISTA polish (remaining time) ===
    remaining = deadline - time.monotonic()
    phase5_time = max(5.0, remaining * 0.60)
    print("\n=== Phase 5: Final high-beta polish (%.0f sec) ===" % phase5_time)
    
    a5, s5 = run_fista_improved(global_a, n, S, phase5_time, rng,
                                  beta_init=50.0, beta_max=3000.0,
                                  momentum_damp=0.9, restore_freq=100, perturb_every=200)
    a5 = safe_normalize(a5, n)
    c5 = fft_convolve_self(a5)
    score5 = c5.max()
    if score5 < global_best:
        global_best = score5
        global_a = a5.copy()
        print("Phase 5: %.6f" % score5)

    # === Phase 6: Ultra-fine final polish ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        phase6_time = max(5.0, remaining * 0.70)
        print("\n=== Phase 6: Ultra-fine polish (%.0f sec) ===" % phase6_time)
        
        a6, s6 = run_fista_improved(global_a, n, S, phase6_time, rng,
                                      beta_init=200.0, beta_max=5000.0,
                                      momentum_damp=0.85, restore_freq=50, perturb_every=500)
        a6 = safe_normalize(a6, n)
        c6 = fft_convolve_self(a6)
        score6 = c6.max()
        if score6 < global_best:
            global_best = score6
            global_a = a6.copy()
            print("Phase 6: %.6f" % score6)

    # === Attempt with different initialization if time allows ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("\n=== Extra: Try different init (%.0f sec) ===" % min(30.0, remaining))
        a_extra = make_multi_sinusoidal_init(n, rng, S, num_freqs=4)
        a_extra, _ = run_fista_improved(a_extra, n, S, min(25.0, remaining * 0.5), rng,
                                         beta_init=10.0, beta_max=500.0,
                                         momentum_damp=0.95, restore_freq=100, perturb_every=80)
        a_extra = safe_normalize(a_extra, n)
        c_extra = fft_convolve_self(a_extra)
        if c_extra.max() < global_best:
            global_best = c_extra.max()
            global_a = a_extra.copy()
            print("Extra init: %.6f" % global_best)

    # === Final verification ===
    print("\n=== Final verification ===")
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    internal_score = 2.0 * len(global_a) * fft_convolve_self(global_a).max() / (global_a.sum()**2)
    print("Internal score: %.8f, length=%d" % (internal_score, len(global_a)))
    
    ver = evaluate_sequence(final_list)
    print("evaluate_sequence: %.8f" % (ver if np.isfinite(ver) else float('inf')))

    if not np.isfinite(ver):
        print("WARNING: sequence rejected, using fallback")
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback evaluate_sequence: %.8f" % ver)

    print("DONE. Final score: %.8f" % (ver if np.isfinite(ver) else float('inf')))
    return final_list
