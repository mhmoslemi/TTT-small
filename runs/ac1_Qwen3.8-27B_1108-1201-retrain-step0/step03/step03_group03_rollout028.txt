For the KKT refinement, here's the detailed algorithm:
1. Compute c = conv(a, a).
2. Find the threshold: t = c.max() - margin (where margin is small, like 0.01 * c.max()).
3. Active lags: K = {k : c[k] >= t}.

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key innovations over previous approach:
1. KKT-condition-based refinement: identifies active lags and enforces
   first-order optimality via partner-averaging fixed-point iterations
   computed efficiently via FFT.
2. Weighted multi-lag balancing: uses convex combinations of top-lag
   gradients to balance multiple near-maximal lags simultaneously.
3. Adaptive FISTA with faster beta scheduling and dual tracking.
4. Targeted coordinate descent on partner pairs.
5. Multi-length exploration with interpolation-based parent transfer.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_lse_grad(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_weighted_topk_grad(a, c, top_k, power=2.0):
    """Gradient using top-k lags with power-weighted softmax."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices]
    # Power-weighted: emphasize lags closer to max
    tv_min = top_vals.min()
    tv_range = top_vals.max() - tv_min
    if tv_range < 1e-15:
        weights = np.ones(k) / k
    else:
        normalized = (top_vals - tv_min) / tv_range
        weights = normalized ** power
        weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_kkt_target(a, c, n, margin_frac=0.03, power=2.0):
    """
    Compute the KKT target vector via FFT.
    target[i] = weighted average of a[k-i] over active lags k.
    Active lags are those with c[k] >= c.max() * (1 - margin_frac).
    Weights are c[k]^power for active k.
    """
    c_max = c.max()
    threshold = c_max * (1.0 - margin_frac)
    active_mask = c >= threshold
    K = np.where(active_mask)[0]
    if len(K) == 0:
        return a.copy()
    
    # Compute weights
    w_vals = c[K] ** power
    w_sum = w_vals.sum()
    if w_sum < 1e-300:
        w_vals = np.ones(len(K))
        w_sum = len(K)
    else:
        w_vals = w_vals / w_sum
    
    # Build weighted indicator: p[k] = w_vals[i] if k == K[i] else 0
    p = np.zeros(2 * n - 1)
    p[K] = w_vals
    
    # Compute target[i] = sum_j a[j] * p[j+i] via FFT
    # As derived: target[i] = conv(a, reversed(p))[2n-2-i]
    r = p[::-1]
    N = 1 << (3 * n - 2).bit_length()
    A_ft = fft(a, N)
    R_ft = fft(r, N)
    conv_result = np.real(ifft(A_ft * R_ft))[:3 * n - 2]
    
    # target[i] = conv_result[2n-2-i] for i = 0..n-1
    target = conv_result[2 * n - 2 : n - 2 : -1]
    # Ensure correct length
    if len(target) < n:
        target = np.concatenate([target, np.zeros(n - len(target))])
    else:
        target = target[:n]
    
    return target

def run_kkt_refinement(a_init, n, S, max_time, rng, initial_alpha=0.5):
    """
    KKT-condition-based refinement.
    At each iteration:
    1. Compute convolution and identify active lags (near-max).
    2. Compute target vector via FFT (weighted partner averaging).
    3. Mix current with target and project onto simplex.
    Uses adaptive step size and momentum.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n * S
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    alpha = initial_alpha
    momentum = 0.3
    v = np.zeros(n)
    it = 0
    margin_frac = 0.02
    power = 2.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()

        # Progressively tighten the active set
        progress = min(1.0, it / max(1, int(max_time * 20)))
        margin_frac = 0.05 * (1 - progress) + 0.005 * progress
        power = 2.0 + 3.0 * progress

        # Compute target
        target = compute_kkt_target(a, c, n, margin_frac=margin_frac, power=power)

        # Momentum update
        v = momentum * v + (1.0 - momentum) * (target - a)
        a_new = a + alpha * v
        a_new = project_simplex(a_new, S)

        # Check improvement
        c_new = fft_convolve_self(a_new)
        new_max = c_new.max()

        if new_max < cur * (1 + 1e-10):
            a = a_new
            # Slightly increase alpha if improving
            alpha = min(0.9, alpha * 1.01)
        else:
            # Decrease alpha on failure
            alpha *= 0.8
            # If alpha gets too small, try a restart with noise
            if alpha < 0.05:
                noise = 0.005 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)
                alpha = 0.3
                v = np.zeros(n)

        # Adaptive margin: widen if many lags are close
        if it % 50 == 0:
            top5 = np.sort(c)[::-1][:5]
            if len(top5) >= 2 and top5[1] > top5[0] * 0.99:
                margin_frac = min(0.1, margin_frac * 1.5)

    return best_local, best_score

    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()

        eta = eta_init / (1.0 + 0.005 * it ** 0.5)
        grad = compute_weighted_topk_grad(a, c, top_k, power=2.0)
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break

        log_update = -eta * grad / gmax
        log_update = np.clip(log_update, -3.0, 3.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) / n * S
        else:
            a = a_new * (S / s_new)

    return best_local, best_score


def make_anti_palindromic_init(n, rng, S, spread=1.0):
    """Create anti-palindromic initialization with controlled spread."""
    half = n // 2
    # Use a smooth profile for the first half
    tt = np.linspace(0, 1, half)
    # Mix of sinusoidal and random
    freq = rng.integers(1, 8)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
    h = h + spread * rng.random(half) * 0.3
    h = np.maximum(h, 0.01)

    mu = S / half
    a = np.zeros(n)
    a[:half] = h * (mu / h.sum()) * h.sum()  # placeholder
    # Proper scaling: a[:half] = h * scale, a[n-1-i] = mu - h * scale
    # sum = half * mu regardless of h values (as long as non-negative)
    scale = mu / h.max() if h.max() > 0 else 1.0
    a[:half] = h * scale
    for i in range(half):
        a[n - 1 - i] = max(0.0, mu - a[i])
    if n % 2 == 1:
        a[half] = mu / 2

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total > 1e-15:
        a = a * (S / total)
    else:
        a = np.ones(n) / n * S
    return a


def make_structured_init(n, rng, S):
    """Create initialization based on known good sequence properties."""
    tt = np.arange(n, dtype=np.float64) / n
    # Multi-frequency anti-symmetric blend
    num_freqs = rng.integers(2, 6)
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, max(3, n // 20))
        phase = rng.random() * np.pi  # Phase in [0, pi] for anti-palindromic
        amp = rng.random() * 0.4 + 0.1
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_balanced_binary_init(n, rng, S):
    """Create an initialization close to balanced binary (0 or c pattern)."""
    # Start with random binary, then smooth
    binary = (rng.random(n) > 0.5).astype(np.float64)
    # Smooth with a small window
    kernel_size = max(3, n // 512)
    kernel = np.ones(kernel_size) / kernel_size
    smoothed = np.convolve(binary, kernel, mode='same')
    a = np.maximum(smoothed, 0.01)
    a = a * (S / a.sum())
    return a


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) / n * S
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(10.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    primary_n = 8192
    if parent is not None and len(parent) >= 64:
        primary_n = len(parent)

    print("Primary length: %d, parent available: %s" % (primary_n, parent is not None))

    S = np.sqrt(2.0 * primary_n)
    n = primary_n

    # === Phase 1: Initialize and run FISTA ===
    # Use parent if available, otherwise anti-palindromic
    if parent is not None:
        a_start = safe_normalize(parent, n)
    else:
        a_start = make_anti_palindromic_init(n, rng, S)

    t0 = time.monotonic()
    phase1_budget = max(30.0, (deadline - t0) * 0.35)
    print("=== Phase 1: FISTA (budget %.0fs) ===" % phase1_budget)

    a_f, s_f = run_fista_improved(a_start, n, S, phase1_budget, rng,
                                   beta_init=10.0, beta_max=400.0,
                                   use_weighted=True, top_k_init=5, top_k_max=48)
    a_f = safe_normalize(a_f, n)
    c_f = fft_convolve_self(a_f)
    score_f = c_f.max()
    obj_f = 2.0 * n * score_f / S ** 2
    if score_f < global_best:
        global_best = score_f
        global_a = a_f.copy()
    print("  FISTA done: max_b=%.6f, obj=%.6f" % (score_f, obj_f))

    # === Phase 2: KKT Refinement ===
    remaining = deadline - time.monotonic()
    phase2_budget = max(20.0, remaining * 0.30)
    print("=== Phase 2: KKT Refinement (budget %.0fs) ===" % phase2_budget)

    a_k, s_k = run_kkt_refinement(a_f, n, S, phase2_budget, rng, initial_alpha=0.4)
    a_k = safe_normalize(a_k, n)
    c_k = fft_convolve_self(a_k)
    score_k = c_k.max()
    if score_k < global_best:
        global_best = score_k
        global_a = a_k.copy()
    print("  KKT done: max_b=%.6f, obj=%.6f" % (score_k, 2.0 * n * score_k / S ** 2))

    # === Phase 3: Mirror Descent ===
    remaining = deadline - time.monotonic()
    phase3_budget = max(10.0, remaining * 0.15)
    print("=== Phase 3: Mirror Descent (budget %.0fs) ===" % phase3_budget)

    a_m, s_m = run_mirror_descent(global_a, n, S, phase3_budget, rng,
                                   eta_init=0.05, top_k=15)
    a_m = safe_normalize(a_m, n)
    c_m = fft_convolve_self(a_m)
    score_m = c_m.max()
    if score_m < global_best:
        global_best = score_m
        global_a = a_m.copy()
    print("  Mirror done: max_b=%.6f, obj=%.6f" % (score_m, 2.0 * n * score_m / S ** 2))

    # === Phase 4: Local Search ===
    remaining = deadline - time.monotonic()
    phase4_budget = max(10.0, remaining * 0.20)
    print("=== Phase 4: Local Search (budget %.0fs) ===" % phase4_budget)

    a_l, s_l = run_local_search(global_a, n, S, phase4_budget, rng)
    a_l = safe_normalize(a_l, n)
    c_l = fft_convolve_self(a_l)
    score_l = c_l.max()
    if score_l < global_best:
        global_best = score_l
        global_a = a_l.copy()
    print("  Local search done: max_b=%.6f, obj=%.6f" % (score_l, 2.0 * n * score_l / S ** 2))

    # === Phase 5: Final FISTA polish with high beta ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        phase5_budget = remaining * 0.5
        print("=== Phase 5: Final FISTA polish (budget %.0fs) ===" % phase5_budget)
        a_p, s_p = run_fista_improved(global_a, n, S, phase5_budget, rng,
                                       beta_init=100.0, beta_max=800.0,
                                       use_weighted=True, top_k_init=10, top_k_max=80)
        a_p = safe_normalize(a_p, n)
        c_p = fft_convolve_self(a_p)
        score_p = c_p.max()
        if score_p < global_best:
            global_best = score_p
            global_a = a_p.copy()
        print("  Final FISTA: max_b=%.6f, obj=%.6f" % (score_p, 2.0 * n * score_p / S ** 2))

    # === Phase 6: One more KKT pass on best ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        phase6_budget = remaining * 0.6
        print("=== Phase 6: Final KKT pass (budget %.0fs) ===" % phase6_budget)
        a_f2, s_f2 = run_kkt_refinement(global_a, n, S, phase6_budget, rng,
                                         initial_alpha=0.3)
        a_f2 = safe_normalize(a_f2, n)
        c_f2 = fft_convolve_self(a_f2)
        score_f2 = c_f2.max()
        if score_f2 < global_best:
            global_best = score_f2
            global_a = a_f2.copy()
        print("  Final KKT: max_b=%.6f, obj=%.6f" % (score_f2, 2.0 * n * score_f2 / S ** 2))

    # === Phase 7: Try different length if time permits ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and n <= 16384:
        n_alt = n * 2 if n * 2 <= 16384 else max(4096, n // 2)
        if n_alt != n:
            S_alt = np.sqrt(2.0 * n_alt)
            print("=== Phase 7: Trying n=%d (budget %.0fs) ===" % (n_alt, remaining * 0.4))
            if parent is not None:
                from numpy import interp
                tt_old = np.linspace(0, 1, len(parent))
                tt_new = np.linspace(0, 1, n_alt)
                a_alt_init = np.maximum(interp(tt_new, tt_old, parent), 0.01)
            else:
                a_alt_init = make_anti_palindromic_init(n_alt, rng, S_alt)

            a_alt, s_alt = run_fista_improved(a_alt_init, n_alt, S_alt,
                                              remaining * 0.3, rng,
                                              beta_init=10.0, beta_max=300.0)
            a_alt = safe_normalize(a_alt, n_alt)
            c_alt = fft_convolve_self(a_alt)
            obj_alt = 2.0 * n_alt * c_alt.max() / S_alt ** 2
            obj_cur = 2.0 * n * global_best / S ** 2
            if obj_alt < obj_cur:
                global_best = c_alt.max()
                global_a = a_alt.copy()
                n = n_alt
                S = S_alt
                print("  Alt length improved: obj=%.6f" % obj_alt)

            # KKT on alt
            if remaining * 0.3 > 10:
                a_alt2, _ = run_kkt_refinement(a_alt, n_alt, S_alt,
                                               remaining * 0.15, rng)
                a_alt2 = safe_normalize(a_alt2, n_alt)
                c_alt2 = fft_convolve_self(a_alt2)
                obj_alt2 = 2.0 * n_alt * c_alt2.max() / S_alt ** 2
                if obj_alt2 < obj_cur:
                    global_best = c_alt2.max()
                    global_a = a_alt2.copy()
                    n = n_alt
                    S = S_alt
                    print("  Alt KKT improved: obj=%.6f" % obj_alt2)

    # === Verification ===
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Verification: evaluate=%.6f, internal_max_b=%.6f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: evaluate_sequence rejected, using fallback")
        global_a = np.ones(2048, dtype=np.float64) * 1.0
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback score: %.6f" % (ver if np.isfinite(ver) else float('inf')))

    return final_list
