```python
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_convolve_ab(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]

def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def solve_direction_lp(f, M, n):
    """
    Solve: maximize sum(b) subject to b >= 0, conv(f, b)[k] <= M for all k.
    Returns the optimal b and its value.
    """
    # conv(f, b)[k] = sum_{i: valid} f[i] * b[k-i]
    # For k = 0..2n-2:
    #   k=0: f[0]*b[0]
    #   k=1: f[0]*b[1] + f[1]*b[0]
    #   ...
    #   k=n-1: f[0]*b[n-1] + f[1]*b[n-2] + ... + f[n-1]*b[0]
    #   k=n: f[1]*b[n-1] + ... + f[n-1]*b[1]
    #   ...
    #   k=2n-2: f[n-1]*b[n-1]
    
    # Build constraint matrix A (shape: (2n-1, n))
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # i.e., A[k, j] = f[j] if max(0, k-n+1) <= j <= min(n-1, k)
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # A[k, j] = f[j] for k = j, j+1, ..., j+n-1
        # i.e., from row j to row j+n-1
        A[j:j+n, j] = f[j]
    
    # Maximize sum(b) => minimize -sum(b)
    c_lin = -np.ones(n)
    
    res = linprog(c_lin, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        b = res.x
    else:
        b = np.zeros(n)
    
    return b
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding
with the SOTA linear programming approach, combined with multiplicative
polishing and multi-phase optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_ab(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(f, n):
    """Build the convolution constraint matrix A of shape (2n-1, n).
    A[k, j] = f[j] if 0 <= k-j < n, else 0.
    """
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f[j]
    return A


def solve_direction_lp(f, M, n):
    """
    Solve: maximize sum(b) subject to b >= 0, A @ b <= M.
    Returns optimal b.
    """
    A = build_conv_matrix(f, n)
    c_lin = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_lin, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return np.zeros(n)

        t0 = time.monotonic()

        b_dir = solve_direction_lp(a, M, n)

            # LP gave zero direction, add noise
            break

        b_norm = b_dir * (S / b_dir.sum())
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding (SOTA LP approach),
combined with fast multiplicative polishing, heuristic directions, and multi-phase
optimization with adaptive restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_ab(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(f, n):
    """Build convolution constraint matrix A of shape (2n-1, n). A[k,j] = f[j] if 0<=k-j<n."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f[j]
    return A


def solve_lp_direction(f, M, n, max_time=5.0):
    """
    Solve LP: maximize sum(b) s.t. A@b <= M, b >= 0.
    Returns direction b, or None if LP is too slow.
    """
    t0 = time.monotonic()
    # Check time before
    if time.monotonic() - t0 > max_time:
        return None
    
    A = build_conv_matrix(f, n)
    c_lin = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_lin, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': max_time})
    
    if time.monotonic() - t0 > max_time * 1.5:
        return None
    
    if res.success:
        b = res.x
        if b.sum() > 1e-10:
            return b
    return None


def heuristic_direction(f, M, n):
    """
    Fast heuristic direction: b[j] = M/(n*f[j]) which is always feasible.
    Then adjust to push more mass toward edges where constraints are slacker.
    """
    f_safe = np.maximum(f, 1e-10 * f.max())
    # Base direction: uniform filling
    b = M / (n * f_safe)
    
    # Edge boost: constraints near edges have fewer terms, so we can push more there
    # For position j, the "bottleneck" constraint is the one with the most terms
    # that still has the least slack. Approximate: positions near center are more constrained.
    # Boost edges.
    i = np.arange(n)
    edge_factor = 1.0 + 0.5 * (1.0 - np.sin(np.pi * i / (n - 1)))
    b = b * edge_factor
    
    return b


def compute_subgrad_fft(c, a, top_k, soft_T=0.0):
    """Compute subgradient of max c (top-k weighted) via FFT."""
    n = len(a)
    top_k = max(1, min(top_k, len(c)))
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    c_max = c[top_idx[0]]
    
    if soft_T > 0:
        w = np.exp(soft_T * (c[top_idx] - c_max))
        w /= w.sum()
    else:
        w = np.full(top_k, 1.0 / top_k)
    
    w_full = np.zeros(len(c))
    w_full[top_idx] = w
    
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1 - 1).bit_length()
    W = fft(w_full, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad

    a_rand = np.maximum(a_rand, 0.01)
    a_rand *= S_init / a_rand.sum()
    candidates.append(a_rand)
    
    # Two-hump Gaussian
    a_2h = (np.exp(-0.5 * ((i - n * 0.2) / (n * 0.08)) ** 2) +
            np.exp(-0.5 * ((i - n * 0.8) / (n * 0.08)) ** 2) + 0.05)
    a_2h = np.maximum(a_2h, 0.01)
    a_2h *= S_init / a_2h.sum()
    candidates.append(a_2h)
    
    # "Flat target" initialization: try to make convolution flat
    # Start with alternating high/low
    a_alt = np.ones(n)
    a_alt[0::2] = 1.5
    a_alt[1::2] = 0.5
    a_alt = np.maximum(a_alt, 0.01)
    a_alt *= S_init / a_alt.sum()
    candidates.append(a_alt)
    
    per_candidate_time = max(3.0, explore_time / len(candidates))
    for idx, a_cand in enumerate(candidates):
        if time.monotonic() > deadline - 60:
            break
        print(f"  Candidate {idx}: quick optimization ({per_candidate_time:.0f}s)...")
        a_best, obj_best = multiplicative_search(
            a_cand, per_candidate_time,
            eta0=0.5, momentum=0.8,
            top_k_start=500, top_k_end=50,
            soft_T=3.0, stagnation_limit=300,
            verbose=False, name=f"exp{idx}"
        )
        print(f"  Candidate {idx}: obj={obj_best:.6f}")
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> New best: {global_best:.6f}")
    
    # === Phase 2: LP-based optimization (45% of time) ===
    lp_time = 0.45 * (deadline - time.monotonic())
    print(f"\n{'='*60}")
    print(f"Phase 2: LP-based optimization ({lp_time:.0f}s)")
    print(f"{'='*60}")
    
    if global_a is not None and lp_time > 20:
        # Split into two sub-phases with different LP settings
        lp_phase1_time = lp_time * 0.5
        lp_phase2_time = lp_time * 0.5
        
        print("  2a: LP optimization (broad)")
        a_best, obj_best = lp_optimization(
            global_a, lp_phase1_time, verbose=True, name="LP-broad"
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")
        
        print("  2b: LP optimization (continued)")
        a_best, obj_best = lp_optimization(
            global_a, lp_phase2_time, verbose=True, name="LP-tight"
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")
    
    # === Phase 3: Intensive polishing (35% of time) ===
    polish_time = deadline - time.monotonic() - 15.0
    print(f"\n{'='*60}")
    print(f"Phase 3: Polishing ({polish_time:.0f}s)")
    print(f"{'='*60}")
    
    if global_a is not None and polish_time > 10:
        # 3a: Broad flattening (many peaks, soft)
        t_3a = polish_time * 0.3
        print(f"  3a: Broad multiplicative ({t_3a:.0f}s)")
        a_best, obj_best = multiplicative_search(
            global_a, t_3a,
            eta0=0.3, momentum=0.85,
            top_k_start=1000, top_k_end=200,
            soft_T=10.0, stagnation_limit=1000,
            verbose=True, name="broad"
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")
        
        # 3b: Medium focus
        t_3b = polish_time * 0.3
        print(f"  3b: Medium multiplicative ({t_3b:.0f}s)")
        a_best, obj_best = multiplicative_search(
            global_a, t_3b,
            eta0=0.2, momentum=0.9,
            top_k_start=50, top_k_end=5,
            soft_T=0.0, stagnation_limit=500,
            verbose=True, name="medium"
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")
        
        # 3c: Single peak chase
        t_3c = polish_time * 0.2
        print(f"  3c: Single peak chase ({t_3c:.0f}s)")
        a_best, obj_best = single_peak_chase(
            global_a, t_3c, lr=0.005, verbose=True, name="peak"
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")
        
        # 3d: Fine multiplicative (very small steps)
        t_3d = polish_time * 0.2
        print(f"  3d: Fine multiplicative ({t_3d:.0f}s)")
        a_best, obj_best = multiplicative_search(
            global_a, t_3d,
            eta0=0.05, momentum=0.95,
            top_k_start=3, top_k_end=1,
            soft_T=0.0, stagnation_limit=300,
            verbose=False, name="fine"
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")
    
    # === Final verification ===
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\n{'='*60}")
    print(f"FINAL: local_obj={global_best:.6f}, verified={ver:.6f}")
    print(f"{'='*60}")
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.6f}")
    
    return final_a
```
