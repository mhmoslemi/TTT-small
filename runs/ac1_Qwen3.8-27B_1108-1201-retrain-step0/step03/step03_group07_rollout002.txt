from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time

def fft_convolve_self(a):

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)

def lp_direction(a, M):
    """
    Find g that maximizes sum(g) subject to g >= 0 and (a * g)[k] <= M for all k.
    Uses scipy linprog.
    """
    n = len(a)
    # Constraint matrix: (2n-1) x n
    # A_eq[k, j] = a[j] if 0 <= k-j < n, else 0
    # We want A_eq @ g <= M (scalar, applied to all rows)
    
    # Build the constraint matrix in a memory-efficient way
    # Row k has non-zeros at j = max(0, k-n+1) to min(n-1, k)
    
    # Use sparse matrix
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: A @ g <= M (scalar)
    # In linprog: A_ub @ x <= b_ub
    b_ub = np.full(2 * n - 1, M)
    
    # Bounds: 0 <= g[j] <= inf
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return None

Uses a linear program at each step to find a descent direction, combined with

import numpy as np
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def objective(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def build_constraint_matrix(a):
    """Build sparse constraint matrix for the LP.
    Row k: A[k, j] = a[j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
