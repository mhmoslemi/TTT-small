"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based alternating optimization
with fast local search polishing and multi-start strategy.

Key innovations:
1. LP-based alternating method (exact best response via linear programming)
2. O(n) convolution update for fast single-coordinate local search
3. Multi-scale approach: global LP optimization + local coordinate descent
4. Multiple restarts with diverse initializations
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy import sparse
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def update_conv_single_change(c, a, i, delta, n):
    """Update self-convolution after changing a[i] by delta.
    Returns new convolution array.
    conv_new[k] = conv_old[k] + 2*delta*a[k-i] + delta^2*(1 if k==2i else 0)
    Valid when 0 <= k-i < n.
    """
    c_new = c.copy()
    k_min = max(0, i)
    k_max = min(2*n-2, i+n-1)
    for k in range(k_min, k_max+1):
        c_new[k] += 2*delta*a[k-i]
    if 0 <= 2*i <= 2*n-2:
        c_new[2*i] += delta*delta
    return c_new


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def build_conv_matrix(n, f):
    """Build the (2n-1) x n matrix F where (F@g)[k] = (f*g)[k]."""
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    return sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))


def solve_lp_best_response(f, M, n, solver=cp.GLPK):
    """Solve LP: maximize sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    F = build_conv_matrix(n, f)
    g = cp.Variable(n)
    constraints = [F @ g <= M, g >= 0]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return None
    if g.value is None:
        return None
    return np.maximum(g.value, 0)

def lp_alternating(a_init, n, S, time_budget, t_init=0.1, t_min=0.01,
                   max_iters=50, verbose=True):
    """LP-based alternating optimization."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    F = None  # Cache the matrix
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 5:
            break
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        M = c.max()
        
        # Solve LP
        t0 = time.monotonic()
        g = solve_lp_best_response(a, M, n)
        lp_time = time.monotonic() - t0
        
        if g is None or g.sum() < 1e-10:
            break
        
        # Rescale g
        g_scaled = g * (S / g.sum())
        
        # Update
        progress = (time.monotonic() - start) / time_budget
        t = t_init * (t_min / t_init) ** progress
        a_new = (1 - t) * a + t * g_scaled
        a_new = normalize_nonneg(a_new, S)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_obj:
            a = a_new
        else:
            # Try smaller step
            t = t * 0.5
            a_new = (1 - t) * a + t * g_scaled
            a_new = normalize_nonneg(a_new, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_obj:
                a = a_new
            else:
                # No improvement, break
                if verbose:
                    print("    LP alt it %d: no improvement, breaking" % it)
                break
        
        if verbose:
            print("    LP alt it %d: obj=%.7f (M=%.7f), t=%.4f, lp_time=%.1f" % (
                it, best_obj, M, t, lp_time))
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based alternating optimization
(Exact best response via HiGHS LP solver) combined with fast vectorized local search.

Key innovations over previous approach:
1. LP-based alternating: solve exact LP for best response at each step (HiGHS solver)
2. Vectorized O(n) convolution update for ultra-fast local search
3. Pre-computed sparse matrix structure for repeated LP solves
4. Adaptive phase allocation and multi-restart strategy

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy import sparse
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def build_conv_sparse_structure(n):
    """Pre-compute sparse structure for convolution matrix F where (F@g)[k] = (f*g)[k]."""
    total_nz = n * n
    rows = np.zeros(total_nz, dtype=np.int32)
    cols = np.zeros(total_nz, dtype=np.int32)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        count = j_max - j_min + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = np.arange(j_min, j_max + 1)
        idx += count
    return rows, cols


def make_conv_matrix(f, n, rows, cols):
    """Create CSR convolution matrix using pre-computed structure."""
    data = f[rows - cols]
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_best_response(f, M, n, rows, cols):
    """Solve LP: maximize sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    F = make_conv_matrix(f, n, rows, cols)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=F, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def update_conv_single_change(c, a, i, delta, n):
    """Vectorized update of self-convolution after changing a[i] by delta."""
    c_new = c.copy()
    k_min = i
    k_max = min(2 * n - 1, i + n)
    if k_min < 2 * n - 1 and k_max > 0:
        k_min = max(0, k_min)
        k_max = min(2 * n - 1, k_max)
        idx = np.arange(k_min, k_max)
        c_new[k_min:k_max] += 2.0 * delta * a[idx - i]
    if 0 <= 2 * i < 2 * n - 1:
        c_new[2 * i] += delta * delta
    return c_new


def fast_local_search(a, n, S, time_budget, verbose=True):
    """Fast local search with O(n) vectorized convolution updates."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 8000
    rng = np.random.default_rng(777)

    factors_high = np.array([0.7, 0.85, 0.92, 1.08, 1.15, 1.3, 1.5])
    factors_low = np.array([0.1, 0.3, 0.5])

    while time.monotonic() - start < time_budget and no_improve < max_no_improve:
        step += 1
        i = rng.integers(0, n)

        if a[i] < 1e-10 * S:
            # Try to increase a small coordinate
            new_val = rng.random() * 0.1 * S / n
            delta = new_val - a[i]
            c_new = update_conv_single_change(c, a, i, delta, n)
            if c_new.max() < best_obj - 1e-12:
                a[i] = new_val
                c = c_new
                best_obj = c.max()
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            continue

        # Try multiplicative changes
        if rng.random() < 0.7:
            factor = rng.choice(factors_high)
        else:
            factor = rng.choice(factors_low)

        new_val = a[i] * factor
        delta = new_val - a[i]
        c_new = update_conv_single_change(c, a, i, delta, n)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-12:
            a[i] = new_val
            c = c_new
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and step % 50000 == 0:
            elapsed = time.monotonic() - start
            print("    FLA step %d: obj=%.7f, rate=%d/s, no_improve=%d" % (

    return best_a, best_obj

                   max_iters=100, verbose=True):

    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    # Pre-compute sparse structure
    rows, cols = build_conv_sparse_structure(n)

    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget - 10:
            break

        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()

        M = c.max()

        # Solve LP for best response
        t0 = time.monotonic()
        g = solve_lp_best_response(a, M, n, rows, cols)
        lp_time = time.monotonic() - t0

        if g is None or g.sum() < 1e-10:
            if verbose:
                print("    LP it %d: LP failed, stopping" % it)
            break

        # Rescale g to match normalization
        g_scaled = g * (S / g.sum())

        # Compute step size (diminishing)
        progress = (time.monotonic() - start) / max(time_budget, 1.0)
        t_step = t_init * (t_min / t_init) ** min(progress, 1.0)

        # Update with backtracking
        a_new = (1.0 - t_step) * a + t_step * g_scaled
        a_new = normalize_nonneg(a_new, S)
        c_new = fft_convolve_self(a_new)
        obj_new = c_new.max()

        accepted = False
        while t_step > t_min * 0.1:
            if obj_new < cur_obj - 1e-14:
                accepted = True
                break
            t_step *= 0.5
            a_new = (1.0 - t_step) * a + t_step * g_scaled
            a_new = normalize_nonneg(a_new, S)
            c_new = fft_convolve_self(a_new)
            obj_new = c_new.max()

        if accepted:
            a = a_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a.copy()
        else:
            if verbose:
                print("    LP it %d: no improvement (M=%.7f), breaking" % (it, M))
            break

        if verbose:
            print("    LP it %d: obj=%.7f, M=%.7f, t=%.4f, lp_time=%.1fs" % (
                it, best_obj, M, t_step, lp_time))

    return best_a, best_obj


def get_initialization(name, n, S, rng):
    """Generate various initializations."""
    i = np.arange(n)
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "gaussian":
        a = np.exp(-0.5 * ((i - n / 2) / (n / 5)) ** 2)
        return a * (S / a.sum())
    elif name == "cosine_mix":
        a = (1.0 + 0.3 * np.cos(2 * np.pi * 3 * i / n + rng.random() * 2) +
             0.15 * np.cos(2 * np.pi * 7 * i / n + rng.random() * 2) +
             0.08 * np.cos(2 * np.pi * 13 * i / n + rng.random() * 2))
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "broad_gaussian":
        a = np.exp(-0.5 * ((i - n / 2) / (n / 3)) ** 2)
        return a * (S / a.sum())
    elif name == "two_hump":
        a = (np.exp(-0.5 * ((i - n * 0.3) / (n / 8)) ** 2) +
             np.exp(-0.5 * ((i - n * 0.7) / (n / 8)) ** 2))
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
    1. LP-based alternating optimization (exact best response via HiGHS)
    2. Fast vectorized local search with O(n) convolution updates
    3. Adaptive phasing with multiple restarts
    """
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 20.0)
    deadline = time.monotonic() + total_budget

    global_best = np.inf
    global_a = None

    # Load given sequence as starting point
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: " + str(e))

    # Determine working n
    if given_a is not None:
        n = len(given_a)
    else:
        n = 4096
    S = np.sqrt(2.0 * float(n))

    # ============ PHASE 1: LP-based alternating (50% time) ============
    p1_budget = 0.50 * total_budget
    print("\n=== Phase 1: LP Alternating (budget=%.0fs, n=%d) ===" % (p1_budget, n))

    a = global_a.copy() if global_a is not None else get_initialization("uniform", n, S, rng)
    a = normalize_nonneg(a, S)

    if p1_budget > 30:
        a_best, obj_best = lp_alternating(a, n, S, p1_budget, t_init=0.25, t_min=0.01, max_iters=80)
        a_best = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_best.copy()
            print("  Phase 1 result: obj=%.7f" % obj)

    # ============ PHASE 2: Fast local search (25% time) ============
    p2_budget = 0.25 * total_budget
    remaining = deadline - time.monotonic()
    p2_budget = min(p2_budget, max(0, remaining - 10))
    print("\n=== Phase 2: Fast Local Search (budget=%.0fs) ===" % p2_budget)

    if global_a is not None and p2_budget > 10:
        a = global_a.copy()
        a = normalize_nonneg(a, S)
        a_best, obj_best = fast_local_search(a, n, S, p2_budget)
        a_best = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_best.copy()
            print("  Phase 2 result: obj=%.7f" % obj)

    # ============ PHASE 3: LP polish with small steps (15% time) ============
    p3_budget = 0.15 * total_budget
    remaining = deadline - time.monotonic()
    p3_budget = min(p3_budget, max(0, remaining - 10))
    print("\n=== Phase 3: LP Polish (budget=%.0fs) ===" % p3_budget)

    if global_a is not None and p3_budget > 15:
        a = global_a.copy()
        a = normalize_nonneg(a, S)
        a_best, obj_best = lp_alternating(a, n, S, p3_budget, t_init=0.08, t_min=0.003, max_iters=60)
        a_best = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_best.copy()
            print("  Phase 3 result: obj=%.7f" % obj)

    # ============ PHASE 4: Final local search (10% time) ============
    p4_budget = max(0, deadline - time.monotonic() - 5)
    print("\n=== Phase 4: Final Local Search (budget=%.0fs) ===" % p4_budget)

    if global_a is not None and p4_budget > 5:
        a = global_a.copy()
        a = normalize_nonneg(a, S)
        a_best, obj_best = fast_local_search(a, n, S, p4_budget)
        a_best = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_best.copy()
            print("  Phase 4 result: obj=%.7f" % obj)

    # ============ PHASE 5: Try alternative starting points if time remains ============
    remaining = deadline - time.monotonic() - 5
    if remaining > 60:
        print("\n=== Phase 5: Alternative starts (budget=%.0fs) ===" % remaining)
        init_names = ["gaussian", "cosine_mix", "two_hump", "random", "broad_gaussian"]
        per_attempt = max(15.0, remaining / (len(init_names) + 1))

        for init_name in init_names:
            if time.monotonic() > deadline - 10:
                break
            if per_attempt < 10:
                break

            a_init = get_initialization(init_name, n, S, rng)
            # Quick LP optimization
            a_best, obj_best = lp_alternating(a_init, n, S, per_attempt,
                                              t_init=0.3, t_min=0.02, max_iters=30, verbose=False)
            a_best = normalize_nonneg(a_best, S)
            obj = evaluate_sequence(a_best.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_best.copy()
                print("  Alt start %s: obj=%.7f" % (init_name, obj))
            else:
                print("  Alt start %s: obj=%.7f (no improvement)" % (init_name, obj))

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f (verified=%.7f, n=%d)" % (global_best, ver, len(final_a)))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)

    return final_a
