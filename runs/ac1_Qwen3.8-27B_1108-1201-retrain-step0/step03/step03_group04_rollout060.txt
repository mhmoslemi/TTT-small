"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization combining spectral initialization,
Adam in softmax parameterization, subgradient polishing with peak
equalization, and diversified restarts across multiple sequence lengths.

import numpy as np
import time
from scipy.fft import fft, ifft
import math


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient of weighted sum of convolution values w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Compute normalized top-k weights (peak values as weights)."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights


def indicator_weights(c, top_k, threshold=None):
    """Uniform weights on top-k peaks."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    weights = np.zeros(len(c))
    weights[top_indices] = 1.0 / top_k
    return weights


def spectral_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    # Generate random phases with flat magnitude
    fsize = n
    A = np.ones(fsize, dtype=np.complex128)
    # Random phases
    phases = rng.uniform(0, 2*np.pi, fsize)
    A *= np.exp(1j * phases)
    # Ensure Hermitian symmetry for real output
    if fsize % 2 == 0:
        A[0] = 1.0 + 0j
        A[fsize//2] = 1.0 + 0j
    a_complex = np.real(ifft(A))
    # Take absolute value to ensure non-negativity
    a = np.abs(a_complex)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def random_init(n, rng):
    """Random non-negative initialization."""
    a = rng.exponential(1.0, n)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    else:
        a = a / s
    return a


def run_adam_softmax(a_init, n, S, max_time, rng, lr=0.01, top_k=5, smooth_beta=10.0):
    """Adam optimizer in softmax parameterization."""
    # Initialize u from a_init: a = S * softmax(u), so u = log(a/S) + const
    a = np.clip(a_init, 1e-10, None).astype(np.float64)
    s = a.sum()
    a = a * (S / s)
    u = np.log(a / S + 1e-30)
    
    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    
    # Precompute for FFT
    fsize = 1 << (2*n - 1 - 1).bit_length()
    conv_len = 2*n - 1
    
    while time.monotonic() - start < max_time:
        t += 1
        
        # Compute current a from u
        u_max = u.max()
        exp_u = np.exp(u - u_max)
        Z = exp_u.sum()
        a = S * exp_u / Z
        
        # Compute convolution
        A = fft(a, fsize)
        B = np.real(ifft(A * A))[:conv_len]
        
        # Compute smoothed max
        b_max = B.max()
        if b_max < best_score:
            best_score = b_max
            best_a = a.copy()
        
        # Gradient of smoothed max w.r.t. B
        # Use top-k or softmax
        w_B = softmax_weights(B, smooth_beta)
        
        # d(B[k])/d(a[j]) = a[k-j] + a[j-k] (for valid indices) - this is the gradient
        # Actually, d(a*a)[k]/d(a[j]) = a[(k-j) mod something]... 
        # For linear convolution: d(a*a)[k]/d(a[j]) = sum_delta a[k-j+delta] ... 
        # Let me use the FFT approach
        grad_a = compute_gradient_fft(a, w_B)
        
        # d(a)/d(u): a[j] = S * exp(u[j]) / Z
        # da[j]/du[i] = a[j] * (delta[ji] - a[i]/sum(a)) = a[j] * (delta[ji] - 1/n)
        # (since sum(a) = S, a[i]/sum(a) = a[i]/S... wait)
        # Actually: a[j] = S * exp(u[j]) / Z where Z = sum(exp(u))
        # da[j]/du[i] = S * exp(u[j]) / Z^2 * (Z*delta[ji] - exp(u[i]))
        # = a[j] * (delta[ji] - exp(u[i])/Z)
        # = a[j] * (delta[ji] - a[i]/S)
        
        # So grad_u[i] = sum_j grad_a[j] * da[j]/du[i]
        # = grad_a[i] * a[i] * (1 - a[i]/S) + sum_{j!=i} grad_a[j] * a[j] * (-a[i]/S)
        # = grad_a[i] * a[i] * (1 - a[i]/S) - a[i]/S * (sum_j grad_a[j]*a[j] - grad_a[i]*a[i])
        # = grad_a[i] * a[i] * (1 - a[i]/S) - a[i]/S * (sum_j grad_a[j]*a[j]) + a[i]/S * grad_a[i] * a[i]
        # = grad_a[i] * a[i] - grad_a[i] * a[i]^2/S - a[i]/S * sum_j grad_a[j]*a[j] + a[i]*grad_a[i]*a[i]/S
        # = grad_a[i] * a[i] - a[i]/S * sum_j grad_a[j]*a[j]
        # = a[i] * (grad_a[i] - mean_like(grad_a * a))
        # where mean_like = sum_j grad_a[j]*a[j] / S
        
        dot = np.dot(grad_a, a) / S
        grad_u = a * (grad_a - dot)
        
        # Also need to account for the 1/Z^2 factor in the objective
        # objective = 2n * max_b / S^2 where max_b = max B
        # d(objective)/d(a) = 2n/S^2 * grad_a (for the active peak)
        # But we're minimizing max_b directly (since S is fixed), so grad_a is sufficient.
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * grad_u**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Re-center u (for numerical stability)
        if t % 100 == 0:
            u = u - u.mean()
        
        # Check for NaN
        if np.any(~np.isfinite(u)):
            u = np.zeros(n)
            m = np.zeros(n)
            v = np.zeros(n)
    
    return best_a, best_score


def run_subgradient_polish(a_init, n, S, max_time, rng, mode='topk', top_k_init=1, top_k_max=50):
    """Subgradient descent with adaptive step size and peak equalization."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (n * 5.0)
    stall = 0
    
    fsize = 1 << (2*n - 1 - 1).bit_length()
    conv_len = 2*n - 1
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)
        
        # Compute convolution
        A = fft(y, fsize)
        B = np.real(ifft(A * A))[:conv_len]
        cur = B.max()
        
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            stall = 0
        else:
            stall += 1
        
        # Choose weights
        if mode == 'topk':
            weights = toppk_weights(B, top_k)
        elif mode == 'indicator':
            weights = indicator_weights(B, top_k)
        else:
            weights = softmax_weights(B, min(500, 20 * it + 10))
        
        grad = compute_gradient_fft(y, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        if stall > 200:
            step *= 0.7
            stall = 0
        elif it % 50 == 0:
            # Try to increase step if we've been improving
            step = min(step * 1.02, S / n)
        
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        # Check improvement
        if it % 3 == 0:
            A_new = fft(x_new, fsize)
            B_new = np.real(ifft(A_new * A_new))[:conv_len]
            if B_new.max() < cur:
                pass  # Good, accept
            else:
                # Try half step
                x_half = project_simplex(y - 0.5 * step * grad / max(gnorm, 1e-15), S)
                A_half = fft(x_half, fsize)
                B_half = np.real(ifft(A_half * A_half))[:conv_len]
                if B_half.max() < cur:
                    x_new = x_half
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum**2))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new
        
        # Perturbation when stalled
        if stall > 1000 and it % 500 == 0:
            noise_scale = 0.005 * S / np.sqrt(n)
            if rng.random() < 0.5:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            else:
                # Fourier perturbation
                freq = rng.integers(1, n//2)
                amp = noise_scale * np.sqrt(n)
                tt = np.arange(n)
                pert = amp * np.cos(2*np.pi*freq*tt/n + rng.uniform(0, 2*np.pi))
                y = project_simplex(y + pert, S)
            t_momentum = 1.0
    
    return best_local, best_score


def run_peak_equalization(a_init, n, S, max_time, rng):
    """Targeted peak equalization: make top peaks equal."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    fsize = 1 << (2*n - 1 - 1).bit_length()
    conv_len = 2*n - 1
    
    while time.monotonic() - start < max_time:
        it += 1
        
        A = fft(a, fsize)
        B = np.real(ifft(A * A))[:conv_len]
        cur = B.max()
        
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()
        
        # Find top peaks and their "excess"
        top_k = min(10, 3 + it // 200)
        top_idx = np.argpartition(B, -top_k)[-top_k:]
        top_vals = B[top_idx]
        mean_top = top_vals.mean()
        
        # Weight peaks by their excess over mean
        excess = np.maximum(top_vals - mean_top, 0.0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            # All top peaks are approximately equal, use uniform weights
            weights = np.zeros(conv_len)
            weights[top_idx] = 1.0 / top_k
        else:
            weights = np.zeros(conv_len)
            weights[top_idx] = excess / total_excess
        
        grad = compute_gradient_fft(a, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Small step
        step = S / (n * 10.0) / (1.0 + 0.001 * it)
        x_new = project_simplex(a - step * grad / max(gnorm, 1e-15), S)
        
        # Verify improvement
        A_new = fft(x_new, fsize)
        B_new = np.real(ifft(A_new * A_new))[:conv_len]
        if B_new.max() < cur:
            a = x_new
        elif it % 100 == 0:
            # Accept anyway to avoid getting stuck
            a = project_simplex(a + rng.normal(0, 0.001*S/np.sqrt(n), n), S)
    
    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def evaluate_bound(a, n):
    """Compute the objective value for a sequence."""
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:

        grad_a = compute_gradient_fft(a, w_B)

        dot = np.dot(grad_a, a) / S
        grad_u = a * (grad_a - dot)

        gnorm = np.linalg.norm(grad_u)
        if gnorm < 1e-15:
            break

        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 200 == 0:
            u = u - u.mean()

        if np.any(~np.isfinite(u)):
            u = np.zeros(n)
            m = np.zeros(n)
            v = np.zeros(n)
            break

    return best_a, best_score


def run_subgradient(a_init, n, S, max_time, rng, top_k_init=1, top_k_max=20,
                    use_momentum=True):
    """Adaptive subgradient with backtracking and peak equalization."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (n * 8.0)
    stall = 0
    best_stall_score = np.inf

    fsize = 1 << (2 * n - 1 - 1).bit_length()
    conv_len = 2 * n - 1

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 150)

        A = fft(y, fsize)
        B = np.real(ifft(A * A))[:conv_len]
        cur = B.max()

        if cur < best_score - 1e-13:
            best_score = cur
            best_local = y.copy()
            stall = 0
            best_stall_score = cur
        else:
            stall += 1

        # Peak equalization weighting: emphasize peaks that are above the median of top peaks
        if top_k <= 5:
            weights = indicator_weights(B, top_k)
        else:
            weights = toppk_weights(B, top_k)

        grad = compute_gradient_fft(y, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step: backtrack if no improvement
        if stall > 300:
            step *= 0.5
            stall = 0
        elif it % 100 == 0 and stall < 50:
            step = min(step * 1.05, S / (n * 2.0))

        dir_grad = grad / max(gnorm, 1e-15)
        x_new = project_simplex(y - step * dir_grad, S)

        # Verify with occasional check
        if it % 5 == 0:
            A_x = fft(x_new, fsize)
            B_x = np.real(ifft(A_x * A_x))[:conv_len]
            if B_x.max() > cur * 1.01:
                # Reject, try smaller step
                x_new = project_simplex(y - 0.3 * step * dir_grad, S)

        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new
        else:
            a = x_new
            y = x_new.copy()

        # Perturbation when significantly stalled
        if stall > 2000 and it % 1000 == 0:
            noise_scale = 0.003 * S / np.sqrt(n)
            r = rng.random()
            if r < 0.3:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            elif r < 0.6:
                freq = rng.integers(1, n // 4)
                amp = noise_scale * np.sqrt(n) * 0.5
                tt = np.arange(n)
                pert = amp * np.cos(2 * np.pi * freq * tt / n + rng.uniform(0, 2 * np.pi))
                y = project_simplex(y + pert, S)
            else:
                # Block shuffle
                block = max(16, n // 128)
                i = rng.integers(0, n - 2 * block)
                j = rng.integers(0, n - 2 * block)
                if abs(i - j) > block:
                    y[i:i+block], y[j:j+block] = y[j:j+block].copy(), y[i:i+block].copy()
                    y = project_simplex(y, S)
            t_mom = 1.0

    return best_local, best_score


def run_fine_polish(a_init, n, S, max_time, rng):
    """Very fine-grained top-1 polishing with tiny steps."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (n * 20.0)

    fsize = 1 << (2 * n - 1 - 1).bit_length()
    conv_len = 2 * n - 1

    while time.monotonic() - start < max_time:
        it += 1

        A = fft(a, fsize)
        B = np.real(ifft(A * A))[:conv_len]
        cur = B.max()

        if cur < best_score - 1e-14:
            best_score = cur
            best_local = a.copy()

        # Only the single peak
        peak_idx = np.argmax(B)
        weights = np.zeros(conv_len)
        weights[peak_idx] = 1.0

        grad = compute_gradient_fft(a, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        x_new = project_simplex(a - step * grad / gnorm, S)

        # Check if it actually helps
        A_x = fft(x_new, fsize)
        B_x = np.real(ifft(A_x * A_x))[:conv_len]
        if B_x.max() < cur:
            a = x_new
            step = min(step * 1.01, S / (n * 5.0))
        else:
            step *= 0.8
            # Micro perturbation to escape
            if it % 200 == 0:
                a = project_simplex(a + rng.normal(0, 0.0005 * S / np.sqrt(n), n), S)

    return best_local, best_score


def run_multiscale(a_init, n, S, max_time, rng):
    """Multi-timescale: alternate between coarse (large top-k) and fine (top-1) optimization."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    fsize = 1 << (2 * n - 1 - 1).bit_length()
    conv_len = 2 * n - 1

    coarse_time = 0.005  # seconds per coarse step
    fine_time = 0.001     # seconds per fine step
    cycle = 0

    while time.monotonic() - start < max_time:
        cycle += 1
        # Coarse phase: large top-k, moderate steps
        top_k_coarse = min(50, 5 + cycle * 2)
        step_coarse = S / (n * 5.0) / (1.0 + 0.01 * cycle)

        A = fft(a, fsize)
        B = np.real(ifft(A * A))[:conv_len]
        cur = B.max()
        if cur < best_score - 1e-13:
            best_score = cur
            best_local = a.copy()

        w = softmax_weights(B, min(100, 5 + cycle))
        grad = compute_gradient_fft(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            a = project_simplex(a - step_coarse * grad / gnorm, S)

        # Fine phase: top-1, small steps
        step_fine = S / (n * 30.0)
        for _ in range(3):
            A = fft(a, fsize)
            B = np.real(ifft(A * A))[:conv_len]
            cur = B.max()
            if cur < best_score - 1e-14:
                best_score = cur
                best_local = a.copy()

            peak_idx = np.argmax(B)
            w1 = np.zeros(conv_len)
            w1[peak_idx] = 1.0
            g1 = compute_gradient_fft(a, w1)
            g1_norm = np.linalg.norm(g1)
            if g1_norm > 1e-15:
                x_new = project_simplex(a - step_fine * g1 / g1_norm, S)
                A_x = fft(x_new, fsize)
                B_x = np.real(ifft(A_x * A_x))[:conv_len]
                if B_x.max() <= cur * 1.001:
                    a = x_new

    return best_local, best_score
