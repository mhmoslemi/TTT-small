from scipy.sparse import coo_matrix
import cvxpy as cp

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:

    return 2.0 * n * c.max() / (S * S)

def solve_lp_direction(f, M):
    """Solve LP to find direction g that maximizes sum(g) subject to f*g peak <= M."""
    n = len(f)
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            j = k - i  # g[j] coefficient
            if fi > 0:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    A = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: maximize sum(g) subject to A @ g <= M, g >= 0

    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= M, g >= 0]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CBC, threads=1)
    if g.value is None:

    return np.maximum(g.value, 0)

def softmax_grad(c, a, temperature):
    """Gradient of log-sum-exp approximation of max(c)."""
    n = len(a)

    w = np.exp(c_shifted / temperature)

    W = fft(w, fsize)

def topk_grad(c, a, top_k):

def solve_relaxed_lp(f, M, n_top=150):

                cols_f.append(j)

            grad = grad_sm
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # LP direction blend
        if lp_g is not None and lp_g.sum() > 1e-10:
            g_target = normalize(lp_g, S)
            lp_dir = g_target - a
            lp_norm = np.linalg.norm(lp_dir)
            if lp_norm > 1e-10:
                lp_dir = lp_dir / lp_norm
                grad = 0.55 * grad + 0.45 * lp_dir
                gn2 = np.linalg.norm(grad)
                if gn2 > 1e-15:
                    grad = grad / gn2
            lp_g = None
        
        # Momentum
        v = mom_cur * v + (1.0 - mom_cur) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm
        
        # Line search
        cur_obj = obj_value(a, c)
        accepted = False
        for bt in range(10):
            eta_try = eta * (0.75 ** bt)
            a_trial = a * np.exp(-eta_try * v)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = normalize(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)
            if obj_trial <= cur_obj * (1.0 + 1e-10):
                a = a_trial
                c = c_trial
                accepted = True
                break
        
        if not accepted:
            eta = max(eta * 0.8, 0.005)
            a = a * np.exp(-eta * v)
            a = np.maximum(a, 1e-300)
            a = normalize(a, S)
            c = fft_convolve_self(a)
        
        cur_obj = obj_value(a, c)
        improved = cur_obj < best_obj - 1e-10
        
        if improved:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
            recent_improvements.append(1)
            eta = min(eta * 1.012, 0.5)
        else:
            recent_improvements.append(0)
            eta = max(eta * 0.985, 0.005)
        
        if len(recent_improvements) > 80:
            recent_improvements = recent_improvements[-80:]
        
        # Adaptive temperature
        if len(recent_improvements) >= 40:
            rate = np.mean(recent_improvements[-40:])
            if rate < 0.03:
                temp_cur *= 1.5
            elif rate > 0.6:
                temp_cur = max(0.1, temp_cur * 0.85)
        
        # Stagnation handling
        if iter_no - best_iter > 700:
            M = c.max()
            lp_g = solve_relaxed_lp(a, M, n_top=120)
            
            if lp_g is not None and lp_g.sum() > 1e-10:
                g_target = normalize(lp_g, S)
                best_blend_obj = cur_obj
                best_blend_a = a.copy()
                for t in [0.15, 0.25, 0.35, 0.5, 0.7]:
                    a_new = (1.0 - t) * a + t * g_target
                    a_new = np.maximum(a_new, 0.0)
                    a_new = normalize(a_new, S)
                    c_new = fft_convolve_self(a_new)
                    obj_new = obj_value(a_new, c_new)
                    if obj_new < best_blend_obj:
                        best_blend_obj = obj_new
                        best_blend_a = a_new.copy()
                
                if best_blend_obj < cur_obj * 0.999:
                    a = best_blend_a
                    c = fft_convolve_self(a)
                    cur_obj = obj_value(a, c)
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                        best_iter = iter_no
                else:
                    # Structured perturbation
                    k_peak = np.argmax(c)
                    contribs = np.zeros(n)
                    lo = max(0, k_peak - n + 1)
                    hi = min(n - 1, k_peak)
                    for i in range(lo, hi + 1):
                        contribs[i] = a[i] * a[k_peak - i]
                    top_idx = np.argsort(contribs)[::-1][:max(10, n // 8)]
                    a[top_idx] *= 0.8
                    a = np.maximum(a, 0.0)
                    a = normalize(a, S)
                    c = fft_convolve_self(a)
                    cur_obj = obj_value(a, c)
            else:
                noise_scale = 0.05 * S / np.sqrt(n)
                noise = np.random.randn(n) * noise_scale
                a = a + noise
                a = np.maximum(a, 0.0)
                a = normalize(a, S)
                c = fft_convolve_self(a)
                cur_obj = obj_value(a, c)
            
            v *= 0.2
            eta = 0.2
            best_iter = iter_no
            recent_improvements = []
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    [stagnation] iter={iter_no}, obj={cur_obj:.6f}, "
                      f"best={best_obj:.6f}, t={elapsed:.1f}s")
        
        # Periodic LP
        if iter_no % lp_every == 0 and iter_no > 100 and lp_g is None:
            M = c.max()
            lp_g = solve_relaxed_lp(a, M, n_top=120)
        
        eta *= 0.99998
        iter_no += 1
        
        if verbose and iter_no % 1500 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"eta={eta:.4f}, temp={temp_cur:.2f}, t={elapsed:.1f}s")
    
    return best_a, best
