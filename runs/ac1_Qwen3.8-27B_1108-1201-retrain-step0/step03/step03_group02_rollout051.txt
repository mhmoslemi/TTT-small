import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_objective(a):
    """Compute 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def soft_max_grad(a, c, beta):
    """Gradient of soft-max approximation to max(c) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(2 * n - 1)
    w = exp_bc / total
    return w


def grad_from_weights(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def top_k_weights(c, k):
    """Get weights for top-k peaks (normalized to sum to 1)."""
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(2 * len(c) - 1)
    w = np.zeros(2 * len(c) - 1)
    for i, idx in enumerate(top_idx):
        w[int(idx)] = top_vals[i] / total
    return w


def run_hybrid_fista(a_init, n, S, max_time, rng, beta_start=5.0, beta_end=40.0):
    """
    FISTA with adaptive soft-max and periodic stochastic restarts.
    Uses moderate beta to balance peaks.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stagnation = 0
    last_improve = start

    while time.monotonic() - start < max_time:
        it += 1
        frac = (time.monotonic() - start) / max_time
        beta = beta_start + (beta_end - beta_start) * min(1.0, frac * 1.5)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_local = y.copy()
            last_improve = time.monotonic()
            stagnation = 0
        else:
            stagnation += 1

        # Stochastic restart if stuck
        if stagnation > 3000 and rng.random() < 0.1:
            noise_scale = 0.01 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            stagnation = 0
            continue

        w = soft_max_grad(y, c, beta)
        grad = grad_from_weights(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size with backtracking
        step = S / (beta * gnorm)
        step = min(step, 2.0 * S / (beta * np.sqrt(n)))
        step = max(step, 1e-8)

        x_new = project_simplex(y - step * grad, S)

        # Backtracking: verify we didn't make it much worse
        if it % 5 == 0:
            c_new = fft_convolve_self(x_new)
            if c_new.max() > cur_max * 1.05:
                x_new = project_simplex(0.5 * y + 0.5 * x_new, S)

        if it % 20 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    return best_local, best_score


def run_peak_pincer(a_init, n, S, max_time, rng, k_start=3, k_end=32):
    """
    Aggressive peak-pincer: focus on reducing the top-k peaks.
    k increases over time for broader coverage.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stagnation = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = (time.monotonic() - start) / max_time
        k = min(k_end, k_start + int(frac * (k_end - k_start)))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_local = y.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 2000 and rng.random() < 0.05:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            stagnation = 0
            continue

        w = top_k_weights(c, k)
        grad = grad_from_weights(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.5 * S / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        # Backtrack if worse
        if it % 3 == 0:
            c_new = fft_convolve_self(x_new)
            if c_new.max() > cur_max * 1.02:
                x_new = project_simplex(0.7 * y + 0.3 * x_new, S)

        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    return best_local, best_score


def run_coord_descent(a_init, n, S, max_time, rng):
    """
    Coordinate descent: optimize subsets of coordinates at a time.
    For each coordinate j, find the delta that minimizes the max convolution.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = a.max()  # placeholder, will recompute
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        # Find top peaks
        top_peaks = np.argsort(c)[::-1][:5]

        # For each top peak, try to reduce it by adjusting a few coordinates
        for pk in top_peaks:
            if time.monotonic() - start > max_time - 5:
                break

            # The gradient of c[pk] w.r.t. a[j] is a[pk-j] + a[j-pk] (valid indices)
            # We want to reduce a[j] where a[pk-j] > 0 (and increase elsewhere to maintain sum)
            # Simpler: find the j that most reduces c[pk] and adjust it

            # Compute gradient for this peak
            g = np.zeros(n)
            for j in range(n):
                val = 0.0
                if 0 <= pk - j < 2 * n - 1:
                    val += a[pk - j] if (pk - j) < n else 0
                if 0 <= j - pk < 2 * n - 1:
                    val += a[j - pk] if (j - pk) < n else 0
                g[j] = val

            # Find the coordinate with largest positive contribution to c[pk]
            # Decreasing it will most reduce c[pk]
            pos = np.argmax(g)
            decrease_amt = 0.01 * S / n

            # Try decreasing a[pos] and redistributing
            if a[pos] > decrease_amt:
                a_trial = a.copy()
                a_trial[pos] -= decrease_amt
                # Redistribute to random coordinates
                targets = rng.choice(n, size=min(10, n), replace=False)
                a_trial[targets] += decrease_amt / len(targets)
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = project_simplex(a_trial, S)

                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur_max:
                    a = a_trial

    return best_local, best_score


def make_initializations(n, rng, parent=None):
    """Generate diverse asymmetric initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Parent sequence
    if parent is not None:
        p = np.clip(parent, 0, 1000)
        if len(p) == n and p.sum() > 0:
            inits.append(("parent", p.copy()))

    # Exponential (front and back loaded)
    for alpha in [1.5, 3.0, 5.0, 8.0]:
        inits.append(("exp_%s" % alpha, np.exp(-alpha * tt) + 0.01))
        inits.append(("expb_%s" % alpha, np.exp(-alpha * (1 - tt)) + 0.01))

    # Power law
    for p in [0.5, 1.5, 3.0]:
        inits.append(("pow_%s" % p, (tt + 0.01) ** p))
        inits.append(("powb_%s" % p, (1 - tt + 0.01) ** p))

    # Step functions
    for frac in [0.25, 0.4, 0.6]:
        step_arr = np.ones(n)
        step_arr[:int(n * frac)] = 10.0
        inits.append(("step_%s" % frac, step_arr))
        step_arr2 = np.ones(n)
        step_arr2[int(n * (1 - frac)):] = 10.0
        inits.append(("stepb_%s" % frac, step_arr2))

    # Double bump (mass at 1/4 and 3/4)
    center1 = n / 4
    center2 = 3 * n / 4
    width = n / 16
    bump1 = np.exp(-((np.arange(n) - center1) ** 2) / (2 * width ** 2))
    bump2 = np.exp(-((np.arange(n) - center2) ** 2) / (2 * width ** 2))
    inits.append(("dbl_bump", bump1 + 1.5 * bump2))
    inits.append(("dbl_bump2", 1.5 * bump1 + bump2))

    # Triple bump
    bump3 = np.exp(-((np.arange(n) - n / 2) ** 2) / (2 * (n / 20) ** 2))
    inits.append(("trp_bump", bump1 + bump2 + 0.5 * bump3))

    # Oscillatory + trend
    inits.append(("osc_trend", 1 + 0.3 * np.sin(2 * np.pi * 7 * tt) + 0.5 * tt))
    inits.append(("osc2", 1 + 0.4 * np.cos(2 * np.pi * 13 * tt + 1) + 0.3 * (1 - tt)))

    # Random with structure
    for seed_off in [0, 1, 2]:
        r = np.random.default_rng(42 + seed_off * 100)
        coarse = r.random(n // 64 + 1)
        # Interpolate to full length
        fine = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(coarse)), coarse)
        fine = fine + 0.1 * r.random(n)
        inits.append(("rand_struct_%d" % seed_off, fine + 0.1))

    # Sharp front edge (very asymmetric)
    inits.append(("sharp_front", np.exp(-10 * tt) * 5 + 0.1))
    inits.append(("sharp_back", np.exp(-10 * (1 - tt)) * 5 + 0.1))

    # Logarithmic ramp
    inits.append(("log_ramp", np.log(1 + 50 * tt)))
    inits.append(("log_ramp_b", np.log(1 + 50 * (1 - tt))))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses hybrid FISTA + peak-pincer + coordinate descent with diverse starts.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * float(n))

    # Generate initializations
    inits = make_initializations(n, rng, parent)

    # Evaluate initializations quickly
    print("=== Evaluating %d initializations (n=%d) ===" % (len(inits), n))
    init_scores = []
    for name, arr in inits:
        arr = np.clip(arr, 0, 1000)
        s = arr.sum()
        if s < 0.01:
            continue
        arr_norm = arr * (S / s)
        c = fft_convolve_self(arr_norm)
        score = c.max()
        init_scores.append((score, name, arr_norm.copy()))
        if score < global_best:
            global_best = score
            global_a = arr_norm.copy()

    init_scores.sort(key=lambda x: x[0])
    print("Top initializations:")
    for score, name, _ in init_scores[:5]:
        print("  %s: %.6f" % (name, score))
    print("Best init score: %.6f" % global_best)

    usable_time = max(10.0, deadline - time.monotonic())

    # Phase 1: Explore top initializations (40% of time)
    explore_time = usable_time * 0.35
    top_inits = init_scores[:8]
    per_init_time = explore_time / len(top_inits)

    print("\n=== Phase 1: Exploration (%d inits, ~%.1f sec each) ===" % (len(top_inits), per_init_time))

    for score, name, arr in top_inits:
        if time.monotonic() > deadline - 120:
            break
        t0 = time.monotonic()
        t_seg = min(per_init_time, deadline - time.monotonic() - 120)
        if t_seg < 5:
            break

        # Run hybrid FISTA
        a_opt, s_opt = run_hybrid_fista(arr, n, S, t_seg * 0.7, rng,
                                         beta_start=5.0, beta_end=35.0)
        a_opt = project_simplex(a_opt, S)
        c_opt = fft_convolve_self(a_opt)
        if c_opt.max() < global_best:
            global_best = c_opt.max()
            global_a = a_opt.copy()
            print("  [%s] FISTA: %.6f" % (name, global_best))

        remaining = deadline - time.monotonic() - 120
        if remaining > 5:
            # Quick peak pincer follow-up
            a_opt2, s_opt2 = run_peak_pincer(a_opt, n, S, remaining * 0.3, rng,
                                              k_start=2, k_end=16)
            a_opt2 = project_simplex(a_opt2, S)
            c_opt2 = fft_convolve_self(a_opt2)
            if c_opt2.max() < global_best:
                global_best = c_opt2.max()
                global_a = a_opt2.copy()
                print("  [%s] +PeakPincer: %.6f" % (name, global_best))

    print("\nAfter Phase 1: best = %.6f" % global_best)

    # Phase 2: Deep optimization on best candidate (45% of time)
    remaining = deadline - time.monotonic() - 60
    if remaining > 10 and global_a is not None:
        print("\n=== Phase 2: Deep optimization (%.1f sec) ===" % remaining)

        # 2a: Higher beta FISTA
        seg = remaining * 0.25
        a_p, s_p = run_hybrid_fista(global_a, n, S, seg, rng,

            global_a = a_p.copy()
            print("  High-beta FISTA: %.6f" % global_best)

        # 2b: Peak pincer with increasing k

            a_p2 = project_simplex(a_p2, S)

    remaining = deadline - time.monotonic() - 10
    if remaining > 3 and global_a is not None:

        # Very focused peak pincer (top-1 to top-4)
        a_fp, s_fp = run_peak_pincer(global_a, n, S, remaining * 0.5, rng,
                                      k_start=1, k_end=4)
        a_fp = project_simplex(a_fp, S)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:

    final_list = [float(x) for x in global_a]

        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback verified: %.6f" % ver)

    return final_list
