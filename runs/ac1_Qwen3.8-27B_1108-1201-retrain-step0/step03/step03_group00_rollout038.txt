"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using truncated central binomial initialization
and adaptive FISTA with hybrid gradients.

Key insight: The sequence a[k] = binom(2k,k)/4^k has a nearly-constant linear
convolution in its "rise" portion (approaching 1 for all lags), giving score ~pi/2.
Starting FISTA from this point (with optional tapering) allows rapid convergence
to optima well below 1.51. Multi-scale coarse-to-fine optimization and
diverse gradient modes (LSE, top-k, exact subgradient) are combined.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(a, c, k, power=1.0):
    """Top-k weighted gradient."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def hybrid_gradient(a, c, beta, it, rng, topk):
    """Adaptive gradient: blend LSE with top-k and exact subgradient."""
    n = len(a)
    g_lse = lse_gradient(a, c, beta)
    
    # Increasing emphasis on exact subgradient as we converge
    alpha_exact = min(0.5, 0.01 * it / 1000.0)
    alpha_topk = min(0.3, 0.005 * it / 1000.0)
    alpha_lse = 1.0 - alpha_exact - alpha_topk
    
    g = alpha_lse * g_lse
    
    if alpha_topk > 0.05:
        g_topk = topk_gradient(a, c, topk, power=2.0)
        g += alpha_topk * g_topk
    
    if alpha_exact > 0.05:
        g_exact = exact_subgradient(a, c)
        g += alpha_exact * g_exact
    
    return g


def run_fista(a_init, n, S, max_time, rng, mode="hybrid",
              beta_init=10.0, beta_rate=0.003, beta_max=2000.0,
              topk_init=64, topk_growth=0.5):
    """FISTA with adaptive smoothing and hybrid gradients."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    topk = topk_init

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        topk = min(2 * n - 1, int(topk_init + topk_growth * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        if mode == "hybrid":
            grad = hybrid_gradient(y, c, beta, it, rng, topk)
        elif mode == "topk":
            grad = topk_gradient(y, c, topk, power=1.5)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 1.0 / (2.0 * beta * 0.5 + gnorm / S + 1e-10)
        step = min(step, 2.0 * S / max(gnorm, 1e-10))

        best_a = a.copy()

    return best_a, best


def binomial_sequence(n, taper="none", alpha=0.0):
    """Generate truncated central binomial sequence with optional taper.
    
    a[k] = binom(2k,k)/4^k * taper(k)
    
    This sequence has nearly-constant linear convolution in its rise portion.
    """
    k = np.arange(n, dtype=np.float64)
    
    # Compute binom(2k,k)/4^k using log for numerical stability
    # log(binom(2k,k)) = log(2k!) - 2*log(k!)
    # Use scipy.special or manual
    from scipy.special import gammaln
    log_c = gammaln(2 * k + 1) - 2 * gammaln(k + 1) - k * np.log(4.0)
    c = np.exp(log_c)
    
    # Handle k=0
    c[0] = 1.0
    
    # Apply taper
    if taper == "linear":
        t = np.maximum(1.0 - k / (n - 1), 0.0)
        c = c * t
    elif taper == "hann":
        t = 0.5 * (1 + np.cos(2 * np.pi * k / (n - 1)))
        c = c * t
    elif taper == "exp":
        t = np.exp(-alpha * k / n)
        c = c * t
    elif taper == "sqrt":
        t = np.sqrt(np.maximum(1.0 - k / (n - 1), 0.01))
        c = c * t
    elif taper == "power":
        t = np.power(np.maximum(1.0 - k / (n - 1), 0.0), alpha)
        c = c * t
    
    return np.maximum(c, 0.0)


def generate_initializations(n, rng):
    """Generate diverse initializations including binomial-based ones."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Binomial sequences (primary)
    inits.append(("binom_plain", binomial_sequence(n, taper="none")))
    inits.append(("binom_linear", binomial_sequence(n, taper="linear")))
    inits.append(("binom_hann", binomial_sequence(n, taper="hann")))
    inits.append(("binom_exp", binomial_sequence(n, taper="exp", alpha=2.0)))
    inits.append(("binom_sqrt", binomial_sequence(n, taper="sqrt")))
    inits.append(("binom_power", binomial_sequence(n, taper="power", alpha=0.5)))
    
    # Symmetric binomial (forward + reverse)
    b = binomial_sequence(n, taper="none")
    inits.append(("binom_sym", b + b[::-1]))
    
    # Tapered symmetric

    inits.append(("binom_sym_lin", b2 + b2[::-1]))
    
    # Window functions (from parent, for diversity)

    inits.append(("sin3", 1.0 + 0.4 * np.sin(2 * np.pi * 3 * tt + 0.5)))
    
    # Random
    inits.append(("rand", rng.random(n) + 0.1))
    inits.append(("rand2", np.exp(rng.normal(0, 0.3, n))))
    
    return [np.maximum(x, 1e-6) for x in inits]


def downsample_upsample(a, n_small):
    """Downsample by averaging, then upsample by interpolation."""
    n = len(a)
    if n_small >= n:
        return a.copy()
    nblocks = n // n_small
    a_small = a[:nblocks * n_small].reshape(n_small, nblocks).mean(axis=1)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n)
    return np.interp(t_new, t_old, a_small)

                                        beta_max=500.0, topk_init=32, topk_growth=0.8)
            a_alt2 = normalize_sum(a_alt2, alt_n2)
            c_alt2 = fft_convolve_self(a_alt2)
            if c_alt2.max() < global_best:
                global_best = c_alt2.max()
                global_a = a_alt2.copy()
                print("  Alt n=%d: %.6f" % (alt_n2, global_best))
    
    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("Verification failed, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    if global_a is None:
        # Fallback: binomial sequence at n=4096
        global_a = binomial_sequence(4096, taper="none")
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    
    print("Returning sequence of length %d, score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
