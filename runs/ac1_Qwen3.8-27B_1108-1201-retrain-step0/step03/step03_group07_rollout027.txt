Let me now think about the Nesterov mirror descent more carefully.

**Nesterov Mirror Descent for `min f(x) s.t. x >= 0, sum(x) = S`:**

The mirror map is `phi(x) = sum x_i log(x_i)` (negative entropy).
The mirror descent update is: `x_{k+1} = argmin_{x >= 0, sum(x)=S} { <g_k, x> + (1/eta) * KL(x || x_k) }`

This has the closed form: `x_{k+1}[i] = S * x_k[i] * exp(-eta * g_k[i]) / sum_j x_k[j] * exp(-eta * g_k[j])`

Or equivalently: `x_{k+1} = x_k * exp(-eta * g_k)`, then renormalize to sum S.

With Nesterov momentum:

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive subgradient mirror descent
with log-sum-exp smoothing, momentum, structured peak-based perturbations,
and multi-phase optimization (explore -> exploit -> polish).
Uses FFT for fast convolution and gradient computation.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_conv_ab(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def compute_obj(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_conv_self(a)
    return 2.0 * n * c.max() / (S * S)


def smooth_max_weights(c, T):
    """Get softmax weights for smooth max approximation."""
    c_max = c.max()
    if T < 1e-8:
        w = np.zeros(len(c))
        idx = np.where(np.abs(c - c_max) < 1e-10)[0]
        w[idx] = 1.0 / len(idx)
        return w
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_s = np.exp(shifted)
    s = exp_s.sum()
    if s < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return w
    return exp_s / s


def compute_smooth_grad(c, a, T):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    w = smooth_max_weights(c, T)
    a_rev = a[::-1]
    fsize = 1 << (2 * n - 1 + n - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    conv_result = np.real(ifft(W * R))
    grad = 2.0 * conv_result[n - 1:2 * n - 1]
    return grad


def mirror_step(a, grad, eta, S):
    """Multiplicative (mirror) descent step with renormalization."""
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-300)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new * (S / s)


def structured_perturb(a, S, c, rng, strength=0.05):
    """
    Peak-based structured perturbation: move mass from coordinates
    that contribute most to peaks to those that contribute least.
    """
    n = len(a)
    c_max = c.max()
    # Find peaks within 5% of max
    threshold = c_max * 0.95
    peak_mask = c >= threshold
    num_peaks = peak_mask.sum()
    if num_peaks < 1:
        return a.copy()

    # Compute effect of each coordinate on the peaks
    # effect[m] = sum_{k: peak} a[k-m] (how much a[m] contributes to peaks)
    w = peak_mask.astype(np.float64)
    w /= num_peaks
    a_rev = a[::-1]
    fsize = 1 << (2 * n - 1 + n - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    conv_result = np.real(ifft(W * R))
    effect = conv_result[n - 1:2 * n - 1]

    # Normalize effect
    e_max = effect.max()
    if e_max < 1e-15:
        return a.copy()
    effect_norm = effect / e_max

    # Randomly pick some coordinates to reduce (high effect) and add (low effect)
    num_transfer = max(1, n // 50)
    # Sort by effect
    order = np.argsort(effect_norm)
    low_idx = order[:num_transfer]
    high_idx = order[-num_transfer:]

    # Transfer mass
    transfer_total = 0.0
    for m in high_idx:
        if a[m] > 1e-10:
            transfer = a[m] * strength
            a[m] -= transfer
            transfer_total += transfer

    if transfer_total < 1e-15:
        return a

    for m in low_idx:
        a[m] += transfer_total / num_transfer

    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())
    return a


def random_perturb(a, S, rng, scale=0.03):
    """Gaussian perturbation preserving non-negativity and sum."""
    n = len(a)
    noise = rng.standard_normal(n) * scale * S / np.sqrt(n)
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new * (S / s)

def coordinate_reduce_peak(a, S, c, k_peak, num_reduce=50):
    """
    Try reducing specific coordinates that contribute most to the peak at k_peak.
    Returns improved sequence or original.
    """
    n = len(a)
    a_new = a.copy()
    best_obj = compute_obj(a, c)

    # Find coordinates that contribute most to peak k_peak
    # c[k_peak] = sum_i a[i] * a[k_peak - i]
    contributions = np.zeros(n)
    for i in range(n):
        j = k_peak - i
        if 0 <= j < n:
            contributions[i] = a[i] * a[j]

    # Sort by contribution (descending)
    top_idx = np.argsort(contributions)[::-1][:num_reduce]

    # Try reducing each top contributor
    for idx in top_idx:
        orig = a_new[idx]
        if orig < 1e-15:
            continue
        # Try a few reduction factors
        for factor in [0.8, 0.9, 0.95]:
            a_new[idx] = orig * factor
            # Distribute the removed mass to random coordinates
            removed = orig - a_new[idx]
            if removed > 0:
                # Add to coordinates with low contribution
                low_idx = np.argsort(contributions)[:max(1, num_reduce // 2)]
                a_new[low_idx] += removed / len(low_idx)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new * (S / s)
            c_new = fft_conv_self(a_new)
            obj_new = compute_obj(a_new, c_new)
            if obj_new < best_obj:
                best_obj = obj_new
            else:
                a_new[idx] = orig
                break

    return a_new, best_obj

    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    global_best = np.inf
    global_a = None

    # --- Load given sequence ---
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw.copy()
            n_given = len(given_a)
            S_target = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_target / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print(f"Given sequence: n={n_given}, obj={ver:.6f}")
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print(f"Could not load given sequence: {e}")
        # Fallback: create a reasonable starting point
        n_fb = 4096
        given_a = np.ones(n_fb) + 0.3 * np.cos(2 * np.pi * 17 * np.arange(n_fb) / n_fb)
        given_a = np.maximum(given_a, 0.01)
        S_fb = np.sqrt(2.0 * n_fb)
        given_a = given_a * (S_fb / given_a.sum())
        ver = evaluate_sequence(given_a.tolist())
        print(f"Fallback sequence: n={n_fb}, obj={ver:.6f}")
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = given_a.copy()

    # --- Phase 1: Quick exploration with different initializations (15% time) ---
    remaining = deadline - time.monotonic()
    explore_time = 0.15 * remaining
    print(f"\nPhase 1: Exploration ({explore_time:.0f}s)")

    # Try a few alternative initializations
    if given_a is not None:
        n_base = len(given_a)
        S_base = np.sqrt(2.0 * n_base)
        candidates = [('given', given_a.copy())]

        # Cosine with different frequencies
        i_arr = np.arange(n_base)
        for freq in [11, 23, 37]:
            a_cos = 1.0 + 0.4 * np.cos(2 * np.pi * freq * i_arr / n_base)
            a_cos = np.maximum(a_cos, 0.01)
            a_cos *= S_base / a_cos.sum()
            candidates.append((f'cos{freq}', a_cos))

        # Random
        a_rand = rng.random(n_base) + 0.01
        a_rand *= S_base / a_rand.sum()
        candidates.append(('rand', a_rand))

        # Two Gaussian bumps
        a_2g = (np.exp(-0.5 * ((i_arr - 0.3 * n_base) / (0.1 * n_base)) ** 2) +
                np.exp(-0.5 * ((i_arr - 0.7 * n_base) / (0.1 * n_base)) ** 2) + 0.01)
        a_2g *= S_base / a_2g.sum()
        candidates.append(('2gauss', a_2g))

        # Alternate high/low pattern
        a_alt = np.ones(n_base)
        a_alt[::2] *= 1.3
        a_alt[1::2] *= 0.7
        a_alt = np.maximum(a_alt, 0.01)
        a_alt *= S_base / a_alt.sum()
        candidates.append(('alt', a_alt))
    else:
        candidates = []

    per_cand_time = max(5.0, explore_time / len(candidates))
    for name, a_cand in candidates:
        if time.monotonic() > deadline - 120:
            break
        t0 = time.monotonic()
        a_best, obj_best = main_optimize(
            a_cand, min(per_cand_time, deadline - time.monotonic() - 120),
            T_start=30.0, T_end=5.0,
            eta_start=0.25, eta_end=0.05,
            momentum=0.8, stagnation_limit=800,
            verbose=False
        )
        elapsed = time.monotonic() - t0
        print(f"  {name} (n={len(a_best)}): obj={obj_best:.6f} ({elapsed:.1f}s)")
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

    # --- Phase 2: Main optimization (60% time) ---
    remaining = deadline - time.monotonic()
    main_time = 0.60 * remaining
    print(f"\nPhase 2: Main optimization ({main_time:.0f}s)")

    if global_a is not None and main_time > 30:
        # Split into sub-phases with different parameters
        n_sub = 4
        sub_time = main_time / n_sub

        params_list = [
            # (T_start, T_end, eta_start, eta_end, momentum, stagnation)
            (80.0, 10.0, 0.3, 0.05, 0.82, 1000),   # Broad
            (40.0, 5.0, 0.25, 0.03, 0.85, 800),    # Medium
            (20.0, 2.0, 0.2, 0.02, 0.88, 600),     # Narrow
            (10.0, 0.5, 0.15, 0.01, 0.92, 500),    # Very narrow
        ]

        for phase_i, (T_s, T_e, e_s, e_e, mom, stall) in enumerate(params_list):
            if time.monotonic() > deadline - 60:
                break
            t_avail = min(sub_time, deadline - time.monotonic() - 60)
            if t_avail < 10:
                break
            print(f"  Phase 2.{phase_i}: T=[{T_s},{T_e}], eta=[{e_s},{e_e}], "
                  f"mom={mom}, time={t_avail:.0f}s")
            a_best, obj_best = main_optimize(
                global_a, t_avail,
                T_start=T_s, T_end=T_e,
                eta_start=e_s, eta_end=e_e,
                momentum=mom, stagnation_limit=stall,
                verbose=True
            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print(f"    -> improved: {obj_best:.6f}")
            else:
                print(f"    -> no improvement (best remains {global_best:.6f})")

    # --- Phase 3: Polish (25% time) ---
    remaining = deadline - time.monotonic()
    polish_time = 0.25 * remaining
    print(f"\nPhase 3: Polish ({polish_time:.0f}s)")

    if global_a is not None and polish_time > 10:
        # 3a: Fast peak-focused polish
        p_time = polish_time * 0.5
        print(f"  3a: Fast polish ({p_time:.0f}s)")
        a_best, obj_best = fast_polkish(
            global_a, min(p_time, deadline - time.monotonic() - 30),
            verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

        # 3b: Another round of main optimize with very narrow focus
        p_time = polish_time * 0.3
        if time.monotonic() < deadline - 30:
            print(f"  3b: Fine mirror descent ({p_time:.0f}s)")
            a_best, obj_best = main_optimize(
                global_a, min(p_time, deadline - time.monotonic() - 30),
                T_start=5.0, T_end=0.3,
                eta_start=0.08, eta_end=0.005,
                momentum=0.93, stagnation_limit=400,
                verbose=True
            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print(f"    -> {obj_best:.6f}")

        # 3c: Peak reduction via coordinate adjustment
        p_time = polish_time * 0.2
        if time.monotonic() < deadline - 15:
            print(f"  3c: Coordinate peak reduction ({p_time:.0f}s)")
            a_cur = global_a.copy()
            S_cur = a_cur.sum()
            c_cur = fft_conv_self(a_cur)
            obj_cur = compute_obj(a_cur, c_cur)
            k_peak = np.argmax(c_cur)
            a_new, obj_new = coordinate_reduce_peak(
                a_cur, S_cur, c_cur, k_peak, num_reduce=100
            )
            if obj_new < global_best:
                global_best = obj_new
                global_a = a_new.copy()
                print(f"    -> {obj_new:.6f}")

    # --- Phase 4: Final attempt with different n (if time remains) ---
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        print(f"\nPhase 4: Try different length ({remaining:.0f}s)")
        # Try doubling the length by upsampling
        n_cur = len(global_a)
        n_new = n_cur * 2
        S_new = np.sqrt(2.0 * n_new)
        # Upsample: repeat each element
        a_up = np.repeat(global_a, 2)
        a_up *= S_new / a_up.sum()
        # Add small random perturbation
        a_up += rng.standard_normal(n_new) * 0.01 * S_new / np.sqrt(n_new)
        a_up = np.maximum(a_up, 0.0)
        a_up *= S_new / a_up.sum()

        obj_up = compute_obj(a_up)
        print(f"  Upsampled n={n_new}: initial obj={obj_up:.6f}")

        if obj_up < global_best + 0.05:  # Only if promising
            t_avail = min(remaining - 20, 300)
            a_best, obj_best = main_optimize(
                a_up, t_avail,
                T_start=50.0, T_end=2.0,
                eta_start=0.25, eta_end=0.02,
                momentum=0.85, stagnation_limit=800,
                verbose=True
            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print(f"  -> improved with n={n_new}: {obj_best:.6f}")

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    # Trim trailing zeros for compactness (optional)
    # Actually keep all elements as they are

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: local_obj={global_best:.6f}, verified={ver:.6f}, n={len(final_a)}")

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.6f}")

    return final_a
