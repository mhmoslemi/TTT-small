"""
Minimize 2*n*max_conv/sum^2 using FISTA with LSE smoothing and LP-based minimax polishing.
Key insight: For non-negative sequences, we can use linearized convex constraints in an LP
to equalize the top convolution lags, achieving faster convergence than pure gradient methods.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def run_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_rate=0.005, beta_max=500.0):
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 5000 == 0 and rng.random() < 0.1:
            noise = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t = 1.0
    
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()

def lp_polish(a_init, n, S, max_time, rng, n_active=30, n_iters=200):
    """LP-based minimax polishing."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    for iteration in range(n_iters):

"""
Minimize 2*n*max(self_conv)/sum^2 for non-negative sequences.
Strategy: FISTA with LSE smoothing for coarse optimization, followed by
LP-based minimax polishing to equalize active convolution lags.
Key innovation: Linearized convex constraints in an LP directly target
the top lags, achieving faster convergence than pure gradient methods
in the "last mile" of optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix numerical drift
    diff = S - result.sum()
    if abs(diff) > 1e-12:
        result += diff / n
        result = np.maximum(result, 0.0)
    return result


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    grad[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_rate=0.004, beta_max=500.0):
    """FISTA with LSE smoothing and periodic exact subgradient kicks."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Hybrid: use exact subgradient periodically
        if it > 200 and (it % 500 == 0) and rng.random() < 0.3:
            grad = exact_subgradient(y, c)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = S / (gnorm * (1.0 + 0.01 * np.log1p(it / 100.0)))
        else:
            grad = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic noise injection to escape local minima
        if it % 8000 == 0 and (it - last_improve > 3000) and rng.random() < 0.3:
            decay = np.exp(-it / 30000.0)
            noise = 0.003 * S / np.sqrt(n) * max(decay, 0.1)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t = 1.0

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def lp_polish(a_init, n, S, max_time, rng, n_active=25, max_iters=5000):
    """LP-based minimax polishing with damped steps."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    no_improve_count = 0

    for iteration in range(max_iters):
        if time.monotonic() - start > max_time - 1.0:
            break

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best:
            best = cur_max
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > 200:
            break

        # Find active set
        idx = np.argsort(c)[::-1][:n_active]

        # Build LP constraint matrix
        A_ub = np.zeros((n_active, n + 1))
        b_ub = np.zeros(n_active)
        for ii, ki in enumerate(idx):
            lo = max(0, ki - n + 1)
            hi = min(ki, n - 1)
            seg = a[ki - hi:ki - lo + 1][::-1]
            A_ub[ii, lo:hi + 1] = 2.0 * seg
            A_ub[ii, n] = -1.0
            b_ub[ii] = c[ki]

        # Equality: sum(a) = S
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])

        # Objective: minimize M
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0.0, None)] * (n + 1)

        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
        except Exception:
            break

        if not result.success:
            # Try with fewer active constraints
            if n_active > 5:
                n_active = max(5, n_active // 2)
                continue
            break

        a_lp = result.x[:n]
        a_lp = np.maximum(a_lp, 0.0)

        # Damped step with line search
        improved = False
        for t_frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_try = (1.0 - t_frac) * a + t_frac * a_lp
            a_try = project_simplex(a_try, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < cur_max - 1e-12:
                a = a_try
                improved = True
                break

        if not improved:
            # Try a small random perturbation
            if iteration % 100 == 99 and rng.random() < 0.5:
                noise = 0.001 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)
            elif no_improve_count > 100:
                break

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()
