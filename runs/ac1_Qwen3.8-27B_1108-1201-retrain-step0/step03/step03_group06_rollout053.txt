"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based peak reduction.

Key innovation: Instead of gradient-based methods, use the LP formulation from
the SOTU to find the optimal direction at each step. The LP maximizes the L1 norm
of a non-negative sequence b subject to (a*b)_∞ ≤ M, where M is the current peak.
Moving towards the LP solution reduces the peak.

Combined with:
- Structured initializations (Gaussian, cosine, flat spectrum)
- Line search for step size
- Coordinate descent polishing
- Multi-start exploration

import numpy as np
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def build_conv_matrix(n, a):
    """Build sparse matrix A where (A @ b)_k = sum_j a_j * b_{k-j} (linear convolution).
    Shape: (2n-1, n)
    """
    # A[k, j] = a[j] if max(0, k-n+1) <= j <= min(n-1, k), else 0
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_peak_reduction(a, M):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0.
    Returns optimal b and the optimal value.
    """
    n = len(a)
    A = build_conv_matrix(n, a)
    
    # Objective: maximize sum(b) -> minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Inequality: A @ b <= M (scalar M broadcast)
    b_ub = np.full(2 * n - 1, M)
    
    # Bounds: 0 <= b_j (no upper bound)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x, -res.fun
    else:
        return a * (np.sqrt(2 * n) / a.sum()), np.sqrt(2 * n)  # fallback: return a itself

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based peak reduction with
line search, combined with structured initialization and coordinate descent polishing.

The key innovation is using a linear programming subproblem at each iteration:
  max sum(b) s.t. (a*b)_k <= M for all k, b >= 0
where M is the current peak autocorrelation. Moving towards the LP solution
reduces the peak, providing well-directed steps that avoid the local minima
trapping gradient methods.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def build_conv_structure(n):
    """Precompute row/col indices for the convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        rows_list.extend([k] * (j_end - j_start + 1))
        cols_list.extend(range(j_start, j_end + 1))
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def build_conv_matrix(n, a, row_struct=None, col_struct=None):
    """Build sparse convolution constraint matrix."""
    if row_struct is None:
        row_struct, col_struct = build_conv_structure(n)
    data = a[col_struct].copy()
    return csr_matrix((data, (row_struct, col_struct)), shape=(2*n-1, n))


def solve_lp(a, M, row_struct, col_struct, upper_bounds=None):
    """Solve LP: max sum(b) s.t. A@b <= M, 0 <= b <= ub."""
    n = len(a)
    A = build_conv_matrix(n, a, row_struct, col_struct)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    if upper_bounds is not None:
        bounds = list(zip(np.zeros(n), upper_bounds))
    else:
        bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x, -res.fun
    return None, None


def run_lp_iterations(a_init, n, S, time_budget, max_iters=2000, verbose=True):
    """LP-based peak reduction with line search."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    row_struct, col_struct = build_conv_structure(n)
    
    start = time.monotonic()
    it = 0
    stagnant = 0
    
    while time.monotonic() - start < time_budget and it < max_iters:
        M = c.max()
        if M < 1e-15:
            break
        
        # Compute upper bounds for LP variables
        upper_bounds = np.where(a > 1e-15, M / np.maximum(a, 1e-15), 10.0 * S / n)
        
        g0, lp_val = solve_lp(a, M, row_struct, col_struct, upper_bounds)
        
        if g0 is None:
            # LP failed, break
            break
        
        g0 = normalize_nonneg(g0, S)
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - start
            print("    LP iter %d: M=%.8f, elapsed=%.1fs" % (it, M, elapsed))
        
        # Line search
        best_t = 0.0
        best_trial_obj = best_obj
        for t in [0.3, 0.5, 0.7, 1.0, 0.2, 0.1, 0.4]:
            a_trial = (1 - t) * a + t * g0
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial * (S / s)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial < best_trial_obj - 1e-12:
                best_trial_obj = obj_trial
                best_t = t
        
        if best_t == 0.0:
            stagnant += 1
            if stagnant > 5:
                # Try perturbation
                rng = np.random.default_rng(it * 7919)
                a = a + 0.005 * rng.standard_normal(n)
                a = normalize_nonneg(a, S)
                c = fft_convolve_self(a)
                stagnant = 0
            it += 1
            continue
        
        a = (1 - best_t) * a + best_t * g0
        a = normalize_nonneg(a, S)
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        it += 1
    
    if verbose:
        print("    LP done: iters=%d, best_obj=%.8f" % (it, best_obj))
    
    return best_a, best_obj


def coordinate_descent_peaks(a, n, S, time_budget, n_peaks=5, verbose=False):
    """Coordinate descent targeting the worst peaks."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(42)
    sweep = 0
    
    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        # Find top peak positions
        peak_indices = np.argsort(c)[-n_peaks:][::-1]
        
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            orig = a[i]
            if orig < 1e-12:
                continue
            
            # Try different scales
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-15:
                    a[i] = orig
                    break
                a_scaled = a * (S / s)
                c_check = fft_convolve_self(a_scaled)
                if c_check.max() < cur_obj - 1e-12:
                    cur_obj = c_check.max()
                    a = a_scaled
                    improved = True
                    break
            a[i] = orig
            s = a.sum()
            if s > 0:
                a = a * (S / s)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        if not improved and sweep > 2:
            break
        sweep += 1
    
    return best_a, best_obj


def run_adam_smooth(a_init, n, S, time_budget, T_init=10.0, T_final=500.0, lr=0.05):
    """Adam on log-parameters with smooth (softmax) objective."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    theta = np.log(a) - np.log(S)  # log proportions
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** progress
        
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        a = S * exp_theta / total
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Smooth objective gradient
        c_max = c.max()
        scaled = T * c / c_max
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # g_a[m] = 2 * sum_k w[k] * a[k-m]
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g_a = 2.0 * corr[:n]
        
        # Chain rule for log params
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot
        
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * (g_theta ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr_t = lr * (1.0 + 0.5 * (1.0 - progress))
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
    
    return best_a, best_obj


def get_initialization(name, n, S, rng):
    i = np.arange(n)
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "gaussian":
        a = np.exp(-0.5 * ((i - n/2) / (n/8)) ** 2)
        return a * (S / a.sum())
    elif name == "cosine":
        phases = rng.random(5) * 2 * np.pi
        a = np.ones(n)
        for k, p in enumerate(phases):
            a += 0.1 / (k+1) * np.cos(2*np.pi*(k+3)*i/n + p)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based peak reduction.

Key innovation: At each iteration, solve a linear program
  max sum(b) s.t. (a*b)_k <= M for all k, b >= 0
where M is the current peak autocorrelation. Moving towards the LP solution
gives well-directed steps that reduce the peak, avoiding local minima that
trap gradient methods. Combined with structured initialization, line search,
and coordinate descent polishing.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def build_conv_structure(n):
    """Precompute row/col indices for the convolution constraint matrix (2n-1 x n)."""
    rows_parts = []
    cols_parts = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        length = j_end - j_start + 1
        rows_parts.append(np.full(length, k, dtype=np.int32))
        cols_parts.append(np.arange(j_start, j_end + 1, dtype=np.int32))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    return rows, cols


def build_conv_matrix(n, a, row_struct, col_struct):
    """Build sparse convolution constraint matrix with precomputed structure."""
    data = a[col_struct].copy()
    return csr_matrix((data, (row_struct, col_struct)), shape=(2 * n - 1, n))


def solve_lp_peak(a, M, row_struct, col_struct):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0."""
    n = len(a)
    A = build_conv_matrix(n, a, row_struct, col_struct)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    # Add upper bounds to help the solver
    ub = np.where(a > 1e-12, M / np.maximum(a, 1e-12), 100.0 * M / max(a.sum(), 1e-12))
    bounds = list(zip(np.zeros(n), ub))
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 8})
    if res.status == 0:
        return res.x, -res.fun
    elif res.status == 9:
        # Time limit hit, return current iterate if available
        if res.x is not None:
            return res.x, None
    return None, None


def run_lp_iterations(a_init, n, S, time_budget, verbose=True):
    """LP-based peak reduction iterations with line search."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    row_struct, col_struct = build_conv_structure(n)
    
    start = time.monotonic()
    it = 0
    stagnant = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        M = c.max()
        if M < 1e-15:
            break
        
        t0 = time.monotonic()
        g0, lp_val = solve_lp_peak(a, M, row_struct, col_struct)
        lp_time = time.monotonic() - t0
        
        if g0 is None:
            if verbose:
                print("    LP iter %d: solver failed, breaking" % it)
            break
        
        g0 = normalize_nonneg(g0, S)
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - start
            print("    LP iter %d: M=%.8f, lp_sum=%.4f, t=%.2fs, elapsed=%.0fs" % (it, M, g0.sum(), lp_time, elapsed))
        
        # Line search for best step size
        best_t = 0.0
        best_trial_obj = best_obj
        step_candidates = [0.3, 0.5, 0.7, 1.0, 0.2, 0.4, 0.1, 0.6, 0.8]
        
        for t in step_candidates:
            a_trial = (1.0 - t) * a + t * g0
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial * (S / s)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial < best_trial_obj - 1e-11:
                best_trial_obj = obj_trial
                best_t = t
        
        if best_t == 0.0:
            stagnant += 1
            if stagnant > 8:
                # Perturb to escape
                noise_scale = 0.003 * (1 + stagnant * 0.5)
                a = a + noise_scale * rng.standard_normal(n)
                a = normalize_nonneg(a, S)
                c = fft_convolve_self(a)
                stagnant = 0
            it += 1
            continue
        
        # Apply the step
        a = (1.0 - best_t) * a + best_t * g0
        a = normalize_nonneg(a, S)
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        it += 1
    
    if verbose:
        print("    LP done: iters=%d, best_obj=%.8f, time=%.0fs" % (it, best_obj, time.monotonic() - start))
    
    return best_a, best_obj


def run_adam_smooth(a_init, n, S, time_budget, T_init=10.0, T_final=500.0, lr_init=0.05):
    """Adam optimizer on log-parameters with smooth peak objective."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    theta = np.log(a / S)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** progress
        
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
        else:
            a = S * exp_theta / total
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Compute smooth gradient
        c_max = max(c.max(), 1e-15)
        scaled = T * c / c_max
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        
        # g_a[m] = 2 * sum_k w[k] * a[k-m]
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g_a = 2.0 * corr[:n]
        
        # Chain rule for log-parameters
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot
        
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr_t = lr_init * (0.5 + 0.5 * (1.0 - progress))
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
    
    return best_a, best_obj


def coordinate_descent_peaks(a, n, S, time_budget, n_peaks=3, verbose=False):
    """Coordinate descent targeting worst autocorrelation peaks."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    rng = np.random.default_rng(777)
    sweep = 0

    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            orig = a[i]
            if orig < 1e-12:
                # Try small positive values

                    a[i] = val
                    s = a.sum()
                    a_s = a * (S / s)
                    c_ch = fft_convolve_self(a_s)
                    if c_ch.max() < cur_obj - 1e-12:
                        cur_obj = c_ch.max()
                        a = a_s
                        improved = True
                        orig = val
                        break
                a[i] = orig
                continue
            
            best_local = orig
            best_local_obj = cur_obj
            
            for factor in factors:
                a[i] = orig * factor

                if s < 1e-15:

                a_s = a * (S / s)

                    best_local = orig * factor

def get_initialization(name, n, S, rng):

        return a * (S / a.sum())
    elif name == "gaussian":

    elif name == "cosine_mix":

def run_subgradient_direct(a_init, n, S, time_budget, T_init=50.0, T_final=1000.0, lr_init=0.08, momentum=0.85):
    """Direct subgradient descent with peak weighting and momentum."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    a_prev = a.copy()
    beta = momentum
    start = time.monotonic()
    step = 0
    no_improve = 0
    rng = np.random.default_rng(999)
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.8)
        
        # Nesterov lookahead
        y = a + (beta - 1.0) / (1.0 + beta) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        else:
            y = a.copy()
        
        c_y = fft_convolve_self(y)
        c_y_max = c_y.max()
        
        # Weighted subgradient
        scaled = T * c_y / max(c_y_max, 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c_y))
            w[np.argmax(c_y)] = 1.0
        else:
            w /= w_sum
        
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g = 2.0 * corr[:n]
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g_dir = g / gnorm
        
        # Backtracking line search
        lr = lr_init * (1.0 - 0.5 * progress)
        c_cur = fft_convolve_self(a)
        cur_obj = c_cur.max()
        
        best_step_a = a.copy()
        best_step_obj = cur_obj
        
        for i in range(10):

            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial < cur_obj - 1e-10 * cur_obj:
                best_step_a = a_trial
                best_step_obj = obj_trial
                break
            lr *= 0.5
        
        a_prev = a.copy()
        a = best_step_a
        step += 1
        no_improve += 1
        
        if best_step_obj < best_obj - 1e-12:
            best_obj = best_step_obj
            best_a = a.copy()
            no_improve = 0
        
        if no_improve > 200:
            scale = 0.005 * max(0.1, 1.0 - progress)
            a = best_a + scale * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            a_prev = a.copy()
            no_improve = 0
    
    return best_a, best_obj

    print("\n=== Phase 1: Exploration (budget=%.0fs, n=%d) ===" % (p1_budget, n_main))
    
    init_names = ["gaussian", "cosine_mix", "random", "two_hump", "uniform"]
    per_attempt = max(5.0, p1_budget / (len(init_names) + 1))
    
    for init_name in init_names:
        if time.monotonic() > p1_deadline - 5:
            break
        a_init = get_initialization(init_name, n_main, S_main, rng)
        t0 = time.monotonic()
        a_best, _ = run_adam_smooth(a_init, n_main, S_main, min(per_attempt, 35), T_init=10, T_final=300, lr_init=0.04)
        a_check = normalize_nonneg(a_best, S_main)
        ver = evaluate_sequence(a_check.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_check.copy()
            print("  %s: obj=%.8f (%.1fs)" % (init_name, ver, time.monotonic() - t0))
    
    # Also try the given sequence with quick optimization
    if given_a is not None and time.monotonic() < p1_deadline - 5:
        a_best, _ = run_adam_smooth(given_a, n_main, S_main, min(20, p1_deadline - time.monotonic() - 3), T_init=30, T_final=300, lr_init=0.03)
        a_check = normalize_nonneg(a_best, S_main)
        ver = evaluate_sequence(a_check.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_check.copy()
            print("  Given+Adam: obj=%.8f" % ver)
    
    # ============ PHASE 2: LP Optimization (50%) ============
    p2_budget = 0.50 * total_budget
    p2_deadline = time.monotonic() + p2_budget
    remaining_after_p1 = p2_deadline - time.monotonic()
    if remaining_after_p1 < p2_budget:
        p2_budget = max(30, remaining_after_p1)
        p2_deadline = time.monotonic() + p2_budget
    
    print("\n=== Phase 2: LP Optimization (budget=%.0fs) ===" % p2_budget)
    print("  Current best: %.8f" % global_best)
    
    if global_a is not None and p2_budget > 30:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a_current = normalize_nonneg(global_a, S)
        
        # Build structure once
        row_struct, col_struct = build_conv_structure(n)
        print("  Conv matrix: %d vars, %d constraints, %d nonzeros" % (n, 2*n-1, len(row_struct)))
        
        # First LP round
        lp_budget_1 = 0.6 * p2_budget
        a_current, obj_lp1 = run_lp_iterations(a_current, n, S, lp_budget_1, verbose=True)
        a_current = normalize_nonneg(a_current, S)
        ver = evaluate_sequence(a_current.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_current.copy()
            print("  After LP round 1: obj=%.8f" % ver)
        
        # Second LP round with different perturbation
        remaining = p2_deadline - time.monotonic()
        if remaining > 20:
            a_perturbed = global_a + 0.002 * rng.standard_normal(n)
            a_perturbed = normalize_nonneg(a_perturbed, S)
            a_current, obj_lp2 = run_lp_iterations(a_perturbed, n, S, remaining, verbose=True)
            a_current = normalize_nonneg(a_current, S)
            ver = evaluate_sequence(a_current.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_current.copy()
                print("  After LP round 2: obj=%.8f" % ver)
    
    # ============ PHASE 3: Polish (20%) ============
    p3_budget = deadline - time.monotonic()
    if p3_budget < 10:
        p3_budget = max(5, 0.1 * total_budget)
    
    print("\n=== Phase 3: Polish (budget=%.0fs) ===" % p3_budget)
    
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        
        # Coordinate descent
        cd_budget = 0.4 * p3_budget
        a_cd, _ = coordinate_descent_peaks(global_a, n, S, cd_budget, n_peaks=3, verbose=True)
        a_cd = normalize_nonneg(a_cd, S)
        ver = evaluate_sequence(a_cd.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_cd.copy()
            print("  After CD: obj=%.8f" % ver)
        
        # Subgradient polish
        remaining = p3_budget - (time.monotonic() - (time.monotonic() - p3_budget))
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            a_sg, _ = run_subgradient_direct(global_a, n, S, min(remaining * 0.5, 60), T_init=100, T_final=2000, lr_init=0.04, momentum=0.85)
            a_sg = normalize_nonneg(a_sg, S)
            ver = evaluate_sequence(a_sg.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_sg.copy()
                print("  After SG: obj=%.8f" % ver)
        
        # Final short LP if time remains
        remaining = deadline - time.monotonic() - 5
        if remaining > 30:
            n = len(global_a)
            S = np.sqrt(2.0 * n)
            row_struct, col_struct = build_conv_structure(n)
            a_lp3, _ = run_lp_iterations(global_a, n, S, min(remaining * 0.6, 60), verbose=False)
            a_lp3 = normalize_nonneg(a_lp3, S)
            ver = evaluate_sequence(a_lp3.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_lp3.copy()
                print("  After LP3: obj=%.8f" % ver)
            
            # Final tiny CD
            remaining2 = deadline - time.monotonic() - 2
            if remaining2 > 5:
                a_final, _ = coordinate_descent_peaks(global_a, n, S, remaining2, n_peaks=2)
                a_final = normalize_nonneg(a_final, S)
                ver = evaluate_sequence(a_final.tolist())
                if np.isfinite(ver) and ver < global_best:
                    global_best = ver
                    global_a = a_final.copy()
                    print("  Final CD: obj=%.8f" % ver)
    
    # ============ FINAL ============
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\n=== FINAL: obj=%.8f (n=%d) ===" % (ver if np.isfinite(ver) else np.inf, len(final_a)))
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        print("Fallback used")
    
    return final_a
