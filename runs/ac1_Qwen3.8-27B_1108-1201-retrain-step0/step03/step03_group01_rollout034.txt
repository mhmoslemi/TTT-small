def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)

    return np.maximum(v - theta, 0.0)

"""
Minimize 2*n*max_conv/sum^2 using multi-scale FISTA-LSE with annealing schedule,
spectral equalization kicks, and cyclic autocorrelation warm-start.

Key improvements over previous approach:
1. Multi-scale optimization (64->512->2048->8192) for better initialization
2. Simulated annealing on LSE temperature for escaping local minima
3. Cyclic autocorrelation (periodogram flattening) as warm-start objective
4. Adaptive step size with backtracking
5. Try both n=8192 and n=16384, return best
6. Structured spectral kicks when stalled
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def cyclic_autocorr(a, fsize):
    """Cyclic autocorrelation via periodogram."""
    n = len(a)
    A = fft(a, fsize)
    P = np.abs(A) ** 2
    r = np.real(ifft(P, fsize))[:n]
    return r


def linear_score(a):
    """Compute the actual evaluation score for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (s ** 2))


def fista_lse_anneal(a_init, n, S, max_time, rng, fsize, beta_min=3.0, beta_max=500.0,
                     cycle_period=30.0, anneal_depth=0.5):
    """
    FISTA with LSE and simulated annealing on temperature.
    Beta oscillates: slowly increases, but periodically drops (annealing cycles).
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    step = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)

        # Annealing schedule: base beta increases, but periodically drops
        base_beta = beta_min + (beta_max - beta_min) * progress
        # Annealing cycle: every cycle_period seconds, beta drops
        cycle_phase = (elapsed % cycle_period) / cycle_period
        # Smooth annealing: beta = base * (1 - depth * exp(-5*cycle_phase))
        anneal_factor = 1.0 - anneal_depth * np.exp(-3.0 * cycle_phase)
        beta = max(2.0, base_beta * anneal_factor)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            last_improve = it

        c_max = c.max()
        # LSE computation
        shifted = beta * (c - c_max)
        shifted = np.clip(shifted, -500, 0)  # Prevent overflow
        exp_bc = np.exp(shifted)
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        # Gradient: 2 * cross-correlation of w with y
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 1.0 / (beta * 0.5 + 1.0)
        step = min(step, 2.0 / (beta + 1.0))

        x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t - 1.0) / t_new
        y = x_new + momentum * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Check progress and perturb if stalled
        if (it - last_improve) > 20000 and rng.random() < 0.1:
            noise_scale = 0.005 + 0.01 * progress
            y = project_simplex(y + rng.normal(0, noise_scale * S / np.sqrt(n), n), S)
            t = 1.0

        # Log progress
        if it % 500000 == 0:
            print("    FISTA it=%d, beta=%.1f, score=%.6f, elapsed=%.1fs" % (it, beta, best_score, elapsed))

    return best_local, best_score


def spectral_kick(a, n, S, fsize, n_steps, rng):
    """
    Spectral equalization: directly flatten the periodogram to reduce
    cyclic autocorrelation peaks. Projects back to non-negative simplex.
    """
    a_curr = a.copy()
    best_a = a.copy()
    best_score = np.inf

    for i in range(n_steps):
        A = fft(a_curr, fsize)
        P = np.abs(A) ** 2
        r = np.real(ifft(P, fsize))[:n]

        # Remove DC component (mean)
        r_dc = r.copy()
        r_dc[0] = 0.0
        r_max = r_dc.max()

        if r_max < 1e-10:
            break

        # Compute excess autocorrelation
        r_mean = r_dc.mean()
        r_excess = np.maximum(r_dc - r_mean, 0.0)

        # Modify spectrum to reduce excess
        alpha = 0.15 * (1.0 + 0.5 * np.sin(0.1 * i))
        R_excess = fft(r_excess, fsize)
        P_new = np.maximum(P - alpha * R_excess.real, 0.0)

        # Preserve DC
        P_new[0] = P[0]

        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_curr = project_simplex(a_new, S)

        # Check linear score every 10 steps
        if i % 10 == 0:
            c = fft_convolve_self(a_curr)
            if c.max() < best_score:
                best_score = c.max()
                best_a = a_curr.copy()

    c_final = fft_convolve_self(a_curr)
    if c_final.max() < best_score:
        best_score = c_final.max()
        best_a = a_curr.copy()

    return best_a, best_score


def cyclic_fista(a_init, n, S, max_time, rng, fsize):
    """
    FISTA optimizing the cyclic autocorrelation peak (easier problem).
    Used as warm-start before linear optimization.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = 5.0 + 95.0 * progress

        A = fft(y, fsize)
        P = np.abs(A) ** 2
        r = np.real(ifft(P, fsize))[:n]
        # Zero out DC for the objective
        r_obj = r.copy()
        r_obj[0] = 0.0
        r_max = r_obj.max()

        if r_max < best_score:
            best_score = r_max
            best_local = y.copy()

        # LSE gradient for cyclic autocorrelation
        shifted = beta * (r_obj - r_max)
        shifted = np.clip(shifted, -500, 0)
        exp_r = np.exp(shifted)
        total = exp_r.sum()
        if total < 1e-300:
            break
        w = exp_r / total
        w[0] = 0.0  # Don't optimize DC

        W = fft(w, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta * 0.5 + 1.0)
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t - 1.0) / t_new
        y = x_new + momentum * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    return best_local, best_score


def upscale_sequence(a_small, n_large, rng):
    """Upscale a sequence by interpolation + projection."""
    n_small = len(a_small)
    # Linear interpolation
    x_small = np.linspace(0, 1, n_small, endpoint=False)
    x_large = np.linspace(0, 1, n_large, endpoint=False)
    a_large = np.interp(x_large, x_small, a_small)
    # Add small noise to break symmetry
    a_large += rng.normal(0, 0.01 * a_large.mean(), n_large)
    return project_simplex(a_large, a_small.sum() * (n_large / n_small))


def multi_segment_init(n, k, rng):
    """Create a multi-segment initialization."""
    seg_len = n // k
    remainder = n % k
    best_score = np.inf
    best_heights = None
    for trial in range(100):
        heights = rng.random(k) + 0.1
        a = np.zeros(n)
        for i in range(k):
            start = i * seg_len + min(i, remainder)
            end = start + seg_len + (1 if i < remainder else 0)
            a[start:min(end, n)] = heights[i]
        c = fft_convolve_self(a)
        score = c.max() / (a.sum() ** 2)
        if score < best_score:
            best_score = score
            best_heights = heights.copy()

    a = np.zeros(n)
    for i in range(k):
        start = i * seg_len + min(i, remainder)
        end = start + seg_len + (1 if i < remainder else 0)
        a[start:min(end, n)] = best_heights[i]
    return a


def safe_normalize(a, n, target_sum=None):
    """Clamp to [0,1000] and rescale."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if target_sum is None:
        target_sum = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * target_sum / n
    return a * (target_sum / s)

            sc2 = 2.0 * n2 * c2.max() / S2 ** 2
            print("  n=16384 after opt: %.6f" % sc2)
            if sc2 < global_best:
                global_best = sc2
                global_a = a2.copy()

    # --- Final verification ---
    print("=== Final ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification score: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
            final_list = [float(x) for x in global_a]
    else:
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, best internal score=%.6f" % (len(global_a), global_best))
    return final_list
