    for x in sequence:

        if np.isnan(x) or np.isinf(x):
            return np.inf

    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def topk_gradient(a, c, k, power=1.0):
    """Top-k gradient with power-law weighting on values."""
    n = len(a)

    idx = np.argsort(c)[::-1][:k]

def generate_initializations(n, rng):

    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 1 * tt) + 0.2 * np.sin(2 * np.pi * 4 * tt + 1.0))

    inits.append(rng.random(n) + 0.1)
    inits.append(rng.random(n) ** 2 + 0.05)
    inits.append(np.exp(rng.normal(0, 0.5, n)))

    # Block structures (3, 5, 7 blocks)
    for nblock in [3, 5, 7, 9]:
        block_len = n // nblock

        block_seq = np.repeat(heights, block_len)

    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits.append(alt)

    # Triangle
    inits.append(1.0 - 2.0 * np.abs(tt - 0.5))

    return [np.maximum(x, 0.01) for x in inits]


def downsample_and_upsample(a, n_small):
    """Downsample a to n_small by averaging, then back to n by interpolation."""
    n = len(a)
    if n_small >= n:
        return a.copy()
    nblocks = n // n_small
    a_small = a[:nblocks * n_small].reshape(n_small, nblocks).mean(axis=1)
    # Upsample by linear interpolation
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n)
    a_up = np.interp(t_new, t_old, a_small)
    return a_up
