    n = len(v)

    c = np.real(ifft(A * A))
    return c[:size]


def softmax_weights(c, beta):
    """Compute softmax(beta * c) weights."""
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def compute_grad_weights(a, weights, beta):
    """Compute gradient: 2*beta * cross_correlation(a, weights)[:n]."""
    n = len(a)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * beta * corr[:n]


def compute_toppk_grad(a, c, top_k, scale=1.0):
    """Compute subgradient from top-k lags. Returns 2 * cross_correlation(a, w)."""
    n = len(a)
    top_k = min(top_k, len(c))
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    w = np.zeros(len(c))
    vals = c[top_idx]
    # Weight by value (or uniformly)
    total = vals.sum()
    if total < 1e-30:
        w[top_idx] = 1.0 / top_k
    else:
        w[top_idx] = vals / total
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * scale * corr[:n]


def run_lse_optimization(a_init, n, S, max_time, rng, beta_start=5.0, beta_end=500.0, 
                          step_init=None, use_nesterov=True):
    """
    Progressive-beta log-sum-exp minimization with adaptive step size.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    if step_init is None:
        step_init = S / (n * 4.0)

    # BB step parameters
    a_prev = a.copy()
    step = step_init

    while time.monotonic() - start < max_time:
        it += 1
        
        # Progressive beta
        frac = min(1.0, it / (max_time * 20))  # increase over ~20s worth of iterations
        beta = beta_start * (beta_end / beta_start) ** frac
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = y.copy()

        w = softmax_weights(c, beta)
        grad = compute_grad_weights(y, w, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size (BB-like)
        # Try step, if worsens, halve
        x_new = project_simplex(y - step * grad / gnorm, S)
        
        # Nesterov momentum
        if use_nesterov:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y_new = x_new + ((t - 1.0) / t_new) * (x_new - a)
            y_new = np.maximum(y_new, 0.0)
            # Check if momentum helps
            c_check = fft_convolve_self(y_new)
            if c_check.max() < c.max():
                y = y_new
            else:
                y = x_new
                t = 1.0
            a = x_new
            t = t_new if c_check.max() < c.max() else 1.0
        else:
            y = x_new
            a = x_new
            t = 1.0

        # BB step size update
        if it > 1 and np.linalg.norm(a - a_prev) > 1e-15:
            gamma = np.linalg.norm(a - a_prev) ** 2 / max(np.dot(a - a_prev, grad), 1e-15)
            step = 0.5 * (step + gamma / max(gnorm, 1e-15))
            step = np.clip(step, 1e-6, S)
        a_prev = a.copy()

        # Periodic perturbation
        if it % 2000 == 0 and rng.random() < 0.2:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0

    return best_local, best_score


def run_toppk_nesterov(a_init, n, S, max_time, rng, top_k_init=1, top_k_max=50,
                        top_k_rate=100, step_scale=1.0):
    """Nesterov top-k subgradient with decay."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    base_step = step_scale * S / n

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // top_k_rate)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = y.copy()

        grad = compute_toppk_grad(y, c, top_k, scale=1.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = base_step / (1.0 + 0.002 * it)
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 3000 == 0 and rng.random() < 0.15:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0

    return best_local, best_score


def run_coordinate_descent(a_init, n, S, max_time, rng, n_coords_per_iter=64,
                            n_iterations_budget=5000):
    """
    Coordinate descent: for each selected coordinate i, optimize a[i]
    to minimize the peak convolution value.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    # Precompute the "base" convolution (without the variable coordinate)
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = a.copy()

        # Select coordinates to update (those near the peak lag)
        peak_lag = np.argmax(c)
        # Update coordinates i where i or peak_lag-i is "active"
        # For lag=peak_lag: conv[peak_lag] = sum_j a[j]*a[peak_lag-j]
        # Coordinates involved: j and peak_lag-j for j=0..n-1 (valid)
        
        # Pick a subset of coordinates to update
        if rng.random() < 0.5:
            # Focus on coordinates involved in the peak lag
            j_vals = np.arange(n)
            valid_j = (j_vals <= peak_lag) & (peak_lag - j_vals < n) & (peak_lag - j_vals >= 0)
            candidates = j_vals[valid_j]
            if len(candidates) > 0:
                # Pick the ones with largest contribution
                contributions = a[candidates] * a[peak_lag - candidates]
                top_idx = np.argsort(contributions)[-n_coords_per_iter:]
                coords = candidates[top_idx]
            else:
                coords = rng.choice(n, n_coords_per_iter, replace=False)
        else:
            coords = rng.choice(n, n_coords_per_iter, replace=False)

        for i in coords:
            # Optimize a[i] to minimize max(conv)
            # Save current value
            ai = a[i]
            
            # The convolution as a function of a[i]:
            # conv[k] = base_k + a[i] * a[k-i] (for k != 2i)
            # conv[2i] = base_2i + a[i]^2 (where base_2i excludes the a[i]^2 term)
            
            # Compute base convolution (with a[i] = 0)
            a[i] = 0.0
            c_base = fft_convolve_self(a)
            
            # The "cross" term: for each k, the contribution from a[i] is:
            # If k-i is valid (0 <= k-i < n): a[k-i]
            # Plus if k != i (i.e., there's a separate term): ... 
            # Actually: conv[k] with a[i] restored = c_base
