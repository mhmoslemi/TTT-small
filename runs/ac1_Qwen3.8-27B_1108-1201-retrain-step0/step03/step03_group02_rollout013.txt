import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def grad_logsumexp(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def compute_objective(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

- Progressive refinement: increase K, decrease step size

import numpy as np

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def cross_corr_w_a(w, a):
    """
    Compute 2 * cross_correlation(w, a) where w has length 2n-1.
    Result[m] = 2 * sum_k w[k] * a[k-m] for valid k, m.
    This equals the gradient of sum_k w[k]*c[k] w.r.t. a, where c = a*a.
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    result = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return result


def grad_single_peak(a, c, k):
    """Gradient of c[k] = sum_i a[i]*a[k-i] w.r.t. a."""
    n = len(a)
    w = np.zeros(2 * n - 1)
    w[int(k)] = 1.0
    return cross_corr_w_a(w, a)


def grad_toppk_weighted(a, c, K, weights=None):
    """
    Weighted sum of gradients of top-K peaks.
    If weights is None, use peak values as weights (normalized).
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:K]
    if weights is None:
        wvals = c[top_idx]
        wsum = wvals.sum()
        if wsum < 1e-30:
            return np.zeros(n)
        weights = wvals / wsum
    
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]
    
    return cross_corr_w_a(w_full, a)


def grad_logsumexp(a, c, beta):
    """Gradient of (1/beta)*log(sum_j exp(beta*c[j])) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:

    w = exp_bc / total
    return cross_corr_w_a(w, a)


def min_norm_direction(g_list, max_iter=200):
    """
    Given a list of gradient vectors g_1, ..., g_m (each of length n),
    find lambda >= 0, sum(lambda)=1 minimizing ||sum lambda_j g_j||^2.
    Returns the optimal combination direction (as a vector of length n).
    Uses projected gradient on the simplex.
    """
    m = len(g_list)
    if m == 1:
        return g_list[0]
    
    # G is (n, m)
    G = np.stack(g_list, axis=-1)  # shape (n, m)
    GtG = G.T @ G  # (m, m)
    
    lam = np.ones(m) / m
    step = 1.0 / (np.trace(GtG) / m + 1e-15)
    
    for _ in range(max_iter):
        grad_lam = GtG @ lam  # gradient of (1/2) lambda^T GtG lambda
        lam_new = project_simplex(lam - step * grad_lam, 1.0)
        if np.linalg.norm(lam_new - lam) < 1e-12:
            break
        lam = lam_new
    
    return G @ lam

    return 2.0 * n * c.max() / (S * S)
