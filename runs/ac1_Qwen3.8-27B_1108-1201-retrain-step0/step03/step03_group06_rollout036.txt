import numpy as np

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)

    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)

def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def get_subgrad_weights(c, T):
    """Get exponential weights for subgradient computation."""
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return w
    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    return w


def compute_grad(a, w, n):
    """Compute gradient of weighted autocorrelation."""
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]

def spectral_shaping(a, n, S, time_budget, niter=300):
    """Flatten convolution by iteratively modifying the power spectrum."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()
    best_a = a.copy()
    c = fft_convolve_self(a)
    best_obj = c.max()

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

    for it in range(niter):
        if time.monotonic() - start > time_budget:
            break

        A = fft(a, fsize)
        r = np.real(ifft(A * A, fsize))[:size]
        r_max = r.max()

        # Flatten: cap peaks at 95% of current max
        r_mod = np.minimum(r, 0.95 * r_max)
        # Ensure the sum is preserved (approximately)
        r_scale = r.sum() / max(r_mod.sum(), 1e-30)
        r_mod = r_mod * r_scale

        # Recompute spectrum from modified r
        spec = np.real(fft(r_mod, fsize))
        spec = np.maximum(spec, 0.0)
        new_abs_A = np.sqrt(spec / fsize)

        # Keep original phase
        phase_A = np.angle(A)
        A_new = new_abs_A * np.exp(1j * phase_A)
        a_new = np.real(ifft(A_new, fsize))[:n]

        a_new = normalize_nonneg(a_new, S)
        c_new = fft_convolve_self(a_new)

            best_obj = c_new.max()
            best_a = a_new.copy()

                a[m] = saved * frac
                a = normalize_nonneg(a, S)
                c_t = fft_convolve_self(a)
                if c_t.max() < best_obj - 1e-14:
                    best_obj = c_t.max()
                    best_a = a.copy()
                    improved = True
                    break
            if not improved:
                a[m] = best_a[m]
                a = normalize_nonneg(a, S)
            else:
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                rng_local = np.random.default_rng(rnd * 7 + 3)
                a = best_a + 0.002 * rng_local.standard_normal(n)
                a = normalize_nonneg(a, S)
                stall = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 25.0)
    deadline = time.monotonic() + total_budget

    global_best = np.inf
    global_a = None

    # Load given sequence
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: " + str(e))

    if global_a is None:
        n = 4096
        S = np.sqrt(2.0 * n)
        global_a = rng.random(n)
        global_a = normalize_nonneg(global_a, S)
        ver = evaluate_sequence(global_a.tolist())
        global_best = ver
        print("Random init: obj=%.7f" % ver)

    n = len(global_a)
    S = np.sqrt(2.0 * float(n))

    # Phase 1: FISTA moderate T - warm up and get overall shape (20%)
    p1_budget = 0.20 * total_budget
    print("Phase 1: FISTA moderate T (%.0fs, n=%d)" % (p1_budget, n))
    a_best, _ = fista_optimize(global_a, n, S, p1_budget,
                                T_init=5.0, T_end=150.0, lr_init=0.06)
    a_best = normalize_nonneg(a_best, S)
    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  Phase 1: obj=%.7f" % ver)

    # Phase 2: FISTA high T - focus on peak reduction (30%)
    p2_budget = 0.30 * total_budget
    print("Phase 2: FISTA high T (%.0fs)" % p2_budget)
    a_best, _ = fista_optimize(global_a, n, S, p2_budget,
                                T_init=80.0, T_end=1500.0, lr_init=0.03)
    a_best = normalize_nonneg(a_best, S)
    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  Phase 2: obj=%.7f" % ver)

    # Phase 3: Peak shaving - direct attack on the peak (20%)
    p3_budget = 0.20 * total_budget
    print("Phase 3: Peak shaving (%.0fs)" % p3_budget)
    a_best, _ = peak_shave(global_a, n, S, p3_budget)
    a_best = normalize_nonneg(a_best, S)
    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  Phase 3: obj=%.7f" % ver)

    # Phase 4: Spectral shaping (10%)
    p4_budget = 0.10 * total_budget
    print("Phase 4: Spectral shaping (%.0fs)" % p4_budget)
    a_best, _ = spectral_shaping(global_a, n, S, p4_budget)
    a_best = normalize_nonneg(a_best, S)
    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  Phase 4: obj=%.7f" % ver)

    # Phase 5: Coordinate polish (10%)
    p5_budget = 0.10 * total_budget
    print("Phase 5: Coordinate polish (%.0fs)" % p5_budget)
    a_best, _ = coordinate_polish(global_a, n, S, p5_budget)
    a_best = normalize_nonneg(a_best, S)
    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  Phase 5: obj=%.7f" % ver)

    # Phase 6: Remaining time - alternating FISTA + peak shave
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("Phase 6: Final refinement (%.0fs)" % remaining)
        # Alternate between FISTA (very high T) and peak shaving
        chunk = max(5.0, remaining / 6.0)
        for i in range(int(remaining / chunk) + 1):
            if time.monotonic() > deadline - 5:
                break
            # FISTA with very high T
            t_left = deadline - time.monotonic()
            if t_left < 3:
                break
            a_best, _ = fista_optimize(global_a, n, S, min(chunk * 0.6, t_left),
                                        T_init=300.0, T_end=5000.0, lr_init=0.015,
                                        max_no_improve=100)
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  P6-FISTA %d: obj=%.7f" % (i, ver))

            # Peak shave
            t_left = deadline - time.monotonic()
            if t_left < 3:
                break
            a_best, _ = peak_shave(global_a, n, S, min(chunk * 0.4, t_left))
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  P6-Shave %d: obj=%.7f" % (i, ver))

    # Final verification
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f" % ver)

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)

    return final_a
