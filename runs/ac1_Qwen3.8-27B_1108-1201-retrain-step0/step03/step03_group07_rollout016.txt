import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_cross_correlate_h(a, c):
    """
    Compute h[i] = sum_{j=0}^{n-1} a[j] * c[i+j] for i=0,...,n-1
    Using FFT. c has length 2n-1.
    """
    n = len(a)
    a_rev = a[::-1]
    # h[i] = (a_rev * c)[n-1+i]
    out_len = n + len(c) - 1  # n + (2n-1) - 1 = 3n-2
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a_rev, fsize)
    C = fft(c, fsize)
    conv = np.real(ifft(A * C))
    return conv[n-1:2*n-1]  # length n

def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def boltzmann_optimize(a0, time_budget, eta0=1.0, eta_decay=0.9999,
                       target_mode='mean', quantile=0.9, 
                       verbose=True, stagnation_limit=2000,
                       perturb_every=1000, noise_scale=0.05):

    Boltzmann-style optimization with full excess gradient.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    best_iter = 0
    
    eta = eta0
    iter_no = 0
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        # Choose target level
        if target_mode == 'mean':
            T = c.mean()
        elif target_mode == 'quantile':
            T = np.quantile(c, quantile)
        elif target_mode == 'max':
            T = c.max()
        else:  # adaptive: start with mean, transition to quantile
            progress = min(1.0, iter_no / 5000)
            T = c.mean() * (1 - progress) + np.quantile(c, quantile) * progress
        
        c_excess = c - T
        # Compute h[i] = sum_j a[j] * c_excess[i+j]
        h = fft_cross_correlate_h(a, c_excess)

        # = (raw correlation) - T * S
        # Actually, let me compute it directly as the excess:
        # g[i] = sum_j a[j] * (c[i+j] - T) = h - T*S
        # But h already has the -T included since we passed c_excess.
        # So h IS the excess gradient.
        
        g = h  # This is the excess gradient
        
        # Normalize gradient for stability
        g_max = np.abs(g).max()
        if g_max < 1e-15:
            break
        g_norm = g / g_max
        
        # Multiplicative (mirror) update
        a_new = a * np.exp(-eta * g_norm)
        a_new = np.maximum(a_new, 1e-10)

        a_new = a_new * (S / S_new)

        obj_new = obj_value(a_new, c_new)

            c = c_new
            best_obj = obj_new

            c_half = fft_convolve_self(a_half)
            obj_half = obj_value(a_half, c_half)

                c = c_half
                if obj_half < best_obj:
                    best_obj = obj_half
                    best_a = a.copy()
                    best_iter = iter_no
            # Decrease eta on failure
            eta = max(eta * 0.7, eta0 * 0.01)

            a = np.maximum(a, 1e-10)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            eta = eta0
            best_iter = iter_no

import time
from scipy.fft import fft, ifft

def fft_conv_self(a):
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size-1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A*A))
    return c[:size]

def compute_excess_grad(a, c, T):
    """Compute grad[m] = 2*sum_j a[j]*(c[m+j]-T) for m=0..n-1"""
    n = len(a)
    # c_excess = c - T
    c_excess = c - T
    # We need h[m] = sum_{j=0}^{n-1} a[j] * c_excess[m+j]
    # This is a cross-correlation: h = (a reversed) convolved with c_excess, 
    # extracted at the right positions.
    a_rev = a[::-1]
    out_len = n + len(c_excess) - 1
    fsize = 1 << (out_len-1).bit_length()
    A = fft(a_rev, fsize)
    C = fft(c_excess, fsize)
    conv = np.real(ifft(A*C))
    h = conv[n-1:2*n-1]  # length n
    return 2.0 * h

            a_new *= S / a_new.sum()

            obj_new = obj_value(a_new, c_new)

                c = c_new

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using full-excess-gradient mirror descent
with progressive target levels, multi-start exploration, and coordinate polishing.
Key innovation: uses the FULL autocorrelation excess gradient (not just top-k peaks)
to compute a rich update direction in O(n log n) per step, enabling faster convergence
to flatter autocorrelation profiles.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_full_grad(a, c_excess):
    """
    Compute grad[m] = 2 * sum_{j=0}^{n-1} a[j] * c_excess[m+j] for m=0..n-1.
    This is the gradient of sum_k c[k] * (c[k] - T) w.r.t. a, evaluated via FFT.
    c_excess = c - T (precomputed).
    """
    n = len(a)
    a_rev = a[::-1]
    c_len = len(c_excess)
    out_len = n + c_len - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a_rev, fsize)
    C = fft(c_excess, fsize)
    conv = np.real(ifft(A * C))
    return 2.0 * conv[n - 1:2 * n - 1]


def obj_from_conv(c, S, n):
    """Compute objective from convolution and known sum."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def boltzmann_optimize(a0, time_budget, eta0=2.0, verbose=True,
                       stagnation_limit=3000, progress_ramp=15000):
    """
    Mirror descent with full excess gradient and progressive target.
    Phases: flatten (T=mean) -> reduce tails (T=quantile) -> attack peak (T=high quantile).
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 1e-10)
    a *= S / a.sum()

    c = fft_conv_self(a)
    cur_obj = obj_from_conv(c, S, n)
    best_obj = cur_obj
    best_a = a.copy()
    best_iter = 0
    iter_no = 0
    start = time.monotonic()
    eta = eta0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        # Progressive target selection
        progress = min(1.0, iter_no / progress_ramp)
        if progress < 0.25:
            T = c.mean()
        elif progress < 0.5:
            T = c.mean() * (1 - 2 * (progress - 0.25)) + np.quantile(c, 0.85) * (2 * (progress - 0.25))
        elif progress < 0.75:
            T = np.quantile(c, 0.85) * (1 - 4 * (progress - 0.5)) + np.quantile(c, 0.95) * (4 * (progress - 0.5))
        else:
            T = np.quantile(c, 0.95) * (1 - (progress - 0.75) / 0.25) + c.max() * (progress - 0.75) / 0.25

        c_excess = c - T
        grad = compute_full_grad(a, c_excess)

        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break
        grad = grad / gmax

        # Line search with backtracking
        accepted = False
        for bt in range(6):
            eta_try = eta * (0.5 ** bt)
            a_new = a * np.exp(-eta_try * grad)
            a_new = np.maximum(a_new, 1e-10)
            S_new = a_new.sum()
            if S_new < 1e-10:
                continue
            a_new *= S / S_new
            c_new = fft_conv_self(a_new)
            obj_new = obj_from_conv(c_new, S, n)

            if obj_new < cur_obj * (1.0 + 1e-9):
                a = a_new
                c = c_new
                cur_obj = obj_new
                accepted = True
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
                    best_iter = iter_no
                    no_improve = 0
                else:
                    no_improve += 1
                eta = min(eta * 1.03, eta0 * 20.0)
                break

        if not accepted:
            no_improve += 1
            eta = max(eta * 0.4, eta0 * 0.01)

        # Stagnation: perturb to escape local minimum
        if no_improve > stagnation_limit:
            # Structured perturbation: focus noise on high-value coordinates
            weight = a + 0.1 * a.mean()
            weight /= weight.sum()
            noise = np.random.randn(n) * np.sqrt(0.02 * S / n)
            noise *= (a / a.mean() + 0.5)
            a = a + noise
            a = np.maximum(a, 1e-10)
            a *= S / a.sum()
            c = fft_conv_self(a)
            cur_obj = obj_from_conv(c, S, n)
            eta = eta0
            no_improve = 0
            best_iter = iter_no

        eta *= 0.99997
        iter_no += 1

        if verbose and iter_no % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"eta={eta:.3f}, T={T:.4f}, t={elapsed:.0f}s")

    return best_a, best_obj


def coordinate_peel(a0, time_budget, verbose=True):
    """
    Coordinate descent: iteratively reduce coordinates that contribute most to peaks.
    For each peak location k*, the gradient w.r.t. a[m] is 2*a[k*-mod(n)] (circular approx).
    We reduce a[m] where the "damage" to the peak is highest, and compensate elsewhere.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_conv_self(a)
    best_obj = obj_from_conv(c, S, n)
    best_a = a.copy()
    start = time.monotonic()
    sweep = 0

    while time.monotonic() - start < time_budget:
        improved_this_sweep = False

        # Process top 20 peaks
        peak_indices = np.argsort(c)[::-1][:20]

        for k in peak_indices:
            if time.monotonic() - start > time_budget:
                break

            # Gradient of c[k] w.r.t. a[m]: 2*a[k-m] for valid m
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi < lo:
                continue

            # Try reducing the coordinate with highest contribution
            # Contribution of a[m] to c[k] is a[m]*a[k-m]
            m_vals = np.arange(lo, hi + 1)
            contributions = a[m_vals] * a[k - m_vals]
            m_best = m_vals[np.argmax(contributions)]

            # Try reducing a[m_best] by various amounts
            orig = a[m_best]
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
                a[m_best] = orig * factor
                a = np.maximum(a, 0.0)
                S_curr = a.sum()
                if S_curr < 1e-10:
                    a[m_best] = orig
                    break
                a_norm = a * (S / S_curr)
                c_new = fft_conv_self(a_norm)
                obj_new = obj_from_conv(c_new, S, n)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_norm.copy()
                    a = a_norm
                    c = c_new
                    improved_this_sweep = True
                    break
            a[m_best] = a[m_best]  # restore if not accepted
            # Renormalize
            a = np.maximum(a, 0.0)
            S_curr = a.sum()
            if S_curr > 0:
                a *= S / S_curr

        sweep += 1
        if not improved_this_sweep and sweep > 5:
            break

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    coord sweep={sweep}, best_obj={best_obj:.6f}, t={elapsed:.0f}s")

    return best_a, best_obj


def two_point_swap(a0, time_budget, verbose=True):
    """
    Try 2-point modifications: transfer mass from a[i] to a[j].
    This preserves the sum constraint exactly and maintains non-negativity.
    Focus on pairs that reduce the peak most effectively.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_conv_self(a)
    best_obj = obj_from_conv(c, S, n)
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    iter_no = 0
    while time.monotonic() - start < time_budget:
        # Find current peak
        k_max = np.argmax(c)

        # The peak c[k_max] = sum_i a[i]*a[k_max-i]
        # We want to reduce it. The most effective single change is to
        # reduce a[m] where a[m]*a[k_max-m] is largest.
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)

        if hi >= lo:
            m_vals = np.arange(lo, hi + 1)
            contribs = a[m_vals] * a[k_max - m_vals]
            # Pick top few candidates for reduction
            top_m = m_vals[np.argsort(contribs)[::-1][:5]]

            for m in top_m:
                # Transfer mass from m to a random other position
                # or to a position that reduces other peaks
                for _ in range(3):
                    j = rng.integers(0, n)
                    if j == m or a[m] < 1e-10:
                        continue
                    # Try transferring a fraction of a[m] to a[j]
                    for frac in [0.1, 0.05, 0.02]:
                        delta = a[m] * frac
                        if delta < 1e-10:
                            continue
                        a[m] -= delta
                        a[j] += delta
                        if a[m] < 0:
                            a[m] += delta
                            a[j] -= delta
                            continue
                        c_new = fft_conv_self(a)
                        obj_new = obj_from_conv(c_new, S, n)
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_a = a.copy()
                            c = c_new
                            c_max = c.max()
                            iter_no += 1
                            break
                        else:
                            a[m] += delta
                            a[j] -= delta

        iter_no += 1
        if verbose and iter_no % 500 == 0:
            print(f"    swap iter={iter_no}, best={best_obj:.6f}, t={time.monotonic()-start:.0f}s")

    return best_a, best_obj


def make_initialization(n, kind='mixed', seed=42):

    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * n)
    i = np.arange(n)

    if kind == 'uniform':
        a = np.ones(n) * S / n
    elif kind == 'cos7':
        a = 1.0 + 0.4 * np.cos(2 * np.pi * 7 * i / n)
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
    elif kind == 'cos3':
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * i / n)
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
    elif kind == 'cos13':
        a = 1.0 + 0.5 * np.cos(2 * np.pi * 13 * i / n)
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
    elif kind == 'sin2':
        a = 1.0 + 0.4 * np.sin(2 * np.pi * 2 * i / n)
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
    elif kind == 'chirp':
        # Linear chirp: frequency increases linearly
        freq_start = 3.0
        freq_end = 11.0
        phase = 2 * np.pi * (freq_start * i / n + (freq_end - freq_start) * i**2 / (2 * n**2))
        a = 1.0 + 0.3 * np.cos(phase)
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
    elif kind == 'inverse_gauss':
        center = n / 2.0
        sigma = n * 0.3
        a = 1.0 + 0.5 * np.exp(-0.5 * ((i - center) / sigma) ** 2)
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
    elif kind == 'noise':
        a = rng.random(n)
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
    elif kind == 'two_hump':
        a = (np.exp(-0.5 * ((i - n * 0.3) / (n * 0.1)) ** 2) +
             np.exp(-0.5 * ((i - n * 0.7) / (n * 0.1)) ** 2))
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
    elif kind == 'ramp':
        a = (i + 1) / n
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
    elif kind == 'mixed':
        # Mix of cosine and noise
        a = (1.0 + 0.3 * np.cos(2 * np.pi * 5 * i / n) +
             0.1 * rng.random(n))
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
    elif kind == 'flat_top':
        # Flat in middle, smooth edges (Hamming-like window inverted)
        window = 0.54 + 0.46 * np.cos(2 * np.pi * i / n)
        a = 1.0 / np.maximum(window, 0.1)  # Inverse to create flat top
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
    else:
        a = np.ones(n) * S / n

    return a

    # --- Load given sequence if available ---
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw.copy()
            n_given = len(given_a)
            ver = evaluate_sequence(given_a.tolist())
            print(f"Given sequence: n={n_given}, obj={ver:.6f}")
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print(f"Could not load given sequence: {e}")

    # --- Phase 1: Exploration with diverse initializations (15% of time) ---
    explore_time = 0.15 * (deadline - time.monotonic())
    print(f"\n=== Phase 1: Exploration ({explore_time:.0f}s) ===")

    n_main = 4096
    if given_a is not None:
        n_main = len(given_a)

    initializations = [
        ('given', given_a),
        ('cos7', make_initialization(n_main, 'cos7', 1)),
        ('cos3', make_initialization(n_main, 'cos3', 2)),
        ('cos13', make_initialization(n_main, 'cos13', 3)),
        ('chirp', make_initialization(n_main, 'chirp', 4)),
        ('mixed', make_initialization(n_main, 'mixed', 5)),
        ('flat_top', make_initialization(n_main, 'flat_top', 6)),
        ('noise', make_initialization(n_main, 'noise', 7)),
        ('two_hump', make_initialization(n_main, 'two_hump', 8)),
    ]

    # Also try different lengths
    for n_alt in [2048, 8192]:
        if n_alt != n_main:
            initializations.append((f'cos7_n{n_alt}', make_initialization(n_alt, 'cos7', 9)))
            initializations.append((f'mixed_n{n_alt}', make_initialization(n_alt, 'mixed', 10)))

    # Remove None entries
    initializations = [(name, a) for name, a in initializations if a is not None]

    per_init_time = explore_time / len(initializations)
    for name, a_init in initializations:
        if time.monotonic() > deadline - 120:
            break
        t0 = time.monotonic()
        a_best, obj_best = boltzmann_optimize(
            a_init, per_init_time,
            eta0=2.0, verbose=False,
            stagnation_limit=1000, progress_ramp=5000
        )
        elapsed = time.monotonic() - t0
        print(f"  {name}: obj={obj_best:.6f} ({elapsed:.1f}s)")
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

    # --- Phase 2: Main optimization on best candidate (55% of time) ---
    main_time = 0.55 * (deadline - time.monotonic())
    print(f"\n=== Phase 2: Main optimization ({main_time:.0f}s) ===")

    if global_a is not None and main_time > 20:
        n = len(global_a)
        S = global_a.sum()

        # Sub-phase 2a: Full flattening (broad target)
        t_2a = main_time * 0.35
        print(f"  2a: Flattening phase ({t_2a:.0f}s)")
        a_best, obj_best = boltzmann_optimize(
            global_a, t_2a,
            eta0=2.0, verbose=True,
            stagnation_limit=2000, progress_ramp=50000
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {global_best:.6f}")

        # Sub-phase 2b: Peak reduction (narrow target)
        t_2b = main_time * 0.35
        print(f"  2b: Peak reduction ({t_2b:.0f}s)")
        a_best, obj_best = boltzmann_optimize(
            global_a, t_2b,
            eta0=1.0, verbose=True,
            stagnation_limit=1500, progress_ramp=3000
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {global_best:.6f}")

        # Sub-phase 2c: Fine tuning
        t_2c = main_time * 0.30
        print(f"  2c: Fine tuning ({t_2c:.0f}s)")
        a_best, obj_best = boltzmann_optimize(
            global_a, t_2c,
            eta0=0.5, verbose=True,
            stagnation_limit=1000, progress_ramp=1000
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {global_best:.6f}")

    # --- Phase 3: Polish (25% of time) ---
    polish_time = deadline - time.monotonic() - 15.0
    print(f"\n=== Phase 3: Polish ({polish_time:.0f}s) ===")

    if global_a is not None and polish_time > 5:
        # 3a: Coordinate peeling
        t_3a = polish_time * 0.35
        print(f"  3a: Coordinate peeling ({t_3a:.0f}s)")
        a_best, obj_best = coordinate_peel(global_a, t_3a, verbose=True)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {global_best:.6f}")

        # 3b: Two-point swaps
        t_3b = polish_time * 0.30
        print(f"  3b: Two-point swaps ({t_3b:.0f}s)")
        a_best, obj_best = two_point_swap(global_a, t_3b, verbose=True)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {global_best:.6f}")

        # 3c: Final boltzmann pass with very small steps
        t_3c = polish_time * 0.35
        print(f"  3c: Final fine pass ({t_3c:.0f}s)")
        a_best, obj_best = boltzmann_optimize(
            global_a, t_3c,
            eta0=0.3, verbose=True,
            stagnation_limit=800, progress_ramp=500
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {global_best:.6f}")

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    # Verify with the evaluator
    final_list = global_a.tolist()
    ver = evaluate_sequence(final_list)
    print(f"\nFINAL: local_best={global_best:.6f}, verified={ver:.6f}, n={len(global_a)}")

    if not np.isfinite(ver):
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
        print(f"Fallback: obj={ver:.6f}")

    return final_list
