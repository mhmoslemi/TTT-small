"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Strategy: Adaptive active-set projected gradient with:
- Log-sum-exp smoothing with adaptive beta
- Barzilai-Borwein step sizes with backtracking line search
- KKT-inspired peak equalization via dynamic active-set weighting

- Coordinate descent final polish

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.asarray(v, dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_active_gradient(a, c, active_lags, weights=None):
    """
    Compute gradient of weighted sum of active convolution values.
    grad[j] = 2 * sum_k w_k * a[k-j] (where 0 <= k-j < n)
    Use FFT for efficiency.
    """
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    w_full = np.zeros(size)
    if weights is None:
        w_full[active_lags] = 1.0
    else:
        w_full[active_lags] = weights
    
    # grad[j] = 2 * sum_k w_full[k] * a[k-j] = 2 * cross_corr(w_full, a)[j]
    # cross_corr(w, a)[j] = sum_k w[k] * a[k-j]
    # = ifft(conj(fft(w)) * fft(a))
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return grad


def lse_subgradient(a, c, beta):
    """Compute gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return grad


def make_anti_palindromic_init(n, rng, S):
    """Create anti-palindromic init: a[i] + a[n-1-i] = C."""
    half = n // 2
    h = (rng.random(half) + 0.1) 
    mu = S / half
    a = np.zeros(n)
    a[:half] = h * (mu * half / h.sum())
    for i in range(half):
        a[n - 1 - i] = mu * half / h.sum() - a[i]
    if n % 2 == 1:
        a[half] = mu * half / h.sum() * 0.5
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-15:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_smooth_anti_pal(n, rng, S):
    """Smooth anti-palindromic: a[i] = C/2 + A*sin(pi*i/(n-1) + phase)."""
    half = n // 2
    mu = 2.0 * S / n
    phase = rng.random() * np.pi
    amp = rng.random() * 0.3 * mu
    tt = np.arange(n, dtype=np.float64)
    a = mu / 2.0 + amp * np.sin(np.pi * tt / max(n - 1, 1) + phase)
    # Make it anti-palindromic: enforce a[i] + a[n-1-i] = mu

    a_anti = np.maximum(a_anti, 0.0)
    if a_anti.sum() > 1e-15:
        a_anti = a_anti * (S / a_anti.sum())
    else:

    return a_anti

def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:

    return a * (S / s)


def run_projected_gradient(a_init, n, S, max_time, rng,

    if s < 1e-15:

    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    # BB state
    a_prev = a.copy()
    g_prev = None
    step = S / (n * 5.0)
    
    # Adaptive parameters
    beta = beta_start
    stall_count = 0
    last_improve = np.inf
    
    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = beta_start + (beta_end - beta_start) * progress * progress
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_local = a.copy()
            stall_count = 0
            last_improve = cur_max
        else:
            stall_count += 1
        
        # Compute gradient

            grad = lse_subgradient(a, c, beta)
        else:

                step_bb = np.clip(step_bb, 0.1 * step, 5.0 * step)

        a_prev = a.copy()
        g_prev = grad.copy()

        accepted = False

            step_try = step * (0.5 ** bt)

                a = a_trial
                accepted = True
                break

            stall_count = 0
            g_prev = None
            a_prev = a.copy()
    
    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, eta=0.1, top_k=20):
    """Mirror descent (multiplicative weights) with top-k subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    patience = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_local = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 5000:
            break
        
        eta_cur = eta / (1.0 + 0.0005 * it)
        
        # Top-k gradient with softmax weighting
        k = min(top_k, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:k]
        top_vals = c[top_idx]
        tv_max = top_vals.max()
        w = np.exp((top_vals - tv_max) / max(1.0, top_vals.max() * 0.1))
        w = w / w.sum()
        
        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = w
        fsize = 1 << ((2 * n - 1) - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
        
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break
        
        log_update = -eta_cur * grad / gmax
        log_update = np.clip(log_update, -3.0, 3.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_new * (S / s_new)
    
    return best_local, best_score


def run_coordinate_polish(a_init, n, S, max_time, rng, passes=3):
    """
    Coordinate descent: for each coordinate, find optimal value
    that minimizes the max convolution while maintaining sum constraint.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    for p in range(passes):
        if time.monotonic() - start > max_time:
            break
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        
        # Process coordinates in random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > max_time:
                break
            
            old_val = a[i]
            if old_val < 1e-15:
                # Try making it positive
                a[i] = S / n * 0.01
                s = a.sum()
                a = a * (S / s)
                c_new = fft_convolve_self(a)
                if c_new.max() < cur_max:
                    cur_max = c_new.max()
                else:
                    a[i] = 0.0
                continue
            
            # Binary search for optimal a[i]
            # b[k] is linear in a[i] for k where a[i] participates
            # Maximize the range where max_b doesn't increase
            lo = 0.0
            hi = old_val * 3.0
            
            # Find the best value
            best_v = old_val
            best_b = cur_max
            
            # Quick check: try reducing
            for frac in [0.5, 0.7, 0.3, 1.2, 0.8]:
                trial = old_val * frac
                a[i] = trial
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a_scaled = a * (S / s)
                c_trial = fft_convolve_self(a_scaled)
                b_trial = c_trial.max()
                if b_trial < best_b:
                    best_b = b_trial
                    best_v = trial
            
            a[i] = best_v
            s = a.sum()
            a = a * (S / s)
    
    c_final = fft_convolve_self(a)
    final_score = c_final.max()
    if final_score < best_score:
        best_score = final_score
        best_local = a.copy()
    
    return best_local, best_score


def run_peak_equalization(a_init, n, S, max_time, rng):
    """
    Target peak equalization: try to make all top peaks equal.
    Use Newton-like updates on the active set.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (n * 10.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_local = a.copy()
        
        # Find all peaks within 2% of max
        threshold = cur_max * 0.98
        active_lags = np.where(c >= threshold)[0]
        if len(active_lags) == 0:
            active_lags = np.argsort(c)[::-1][:5]
        elif len(active_lags) > 200:
            active_lags = np.argsort(c)[::-1][:200]
        
        # Compute gradient pointing "equally" against all active peaks
        # Weight by inverse of current value (peaks that are already smaller get less weight)
        active_vals = c[active_lags]
        # Normalize so that the "weakest" active peak has weight 1
        w = active_vals / active_vals.min()
        w = 1.0 / (w + 1.0)  # Inverse weighting: smaller peaks get more push
        w = w / w.sum()
        
        grad = compute_active_gradient(a, c, active_lags, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step with occasional line search
        alpha = step / gnorm if gnorm > 0 else step
        a_new = project_simplex(a - alpha * grad, S)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() < cur_max:
            a = a_new
            step = np.min(step * 1.02, S / (n * 0.05))
        else:
            # Try smaller step
            a_new2 = project_simplex(a - alpha * 0.5 * grad, S)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < cur_max:
                a = a_new2
            step *= 0.7
        
        # Add diversity every 2000 iterations
        if it % 2000 == 0 and rng.random() < 0.3:
            noise = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)
    
    return best_local, best_score


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    S = np.sqrt(2.0 * n)
    inits = []
    
    if parent is not None and len(parent) >= 64:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if p.sum() > 0:
            inits.append(("parent", p * (S / p.sum())))
    
    # Anti-palindromic variants
    for i in range(3):
        inits.append(("anti_pal_%d" % i, make_anti_palindromic_init(n, rng, S)))
    
    # Smooth anti-palindromic
    for i in range(3):
        inits.append(("smooth_anti_%d" % i, make_smooth_anti_pal(n, rng, S)))
    
    # Periodic
    for i in range(2):
        inits.append(("periodic_%d" % i, make_periodic_init(n, rng, S)))
    
    # Random
    inits.append(("random", make_random_init(n, rng, S)))
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Multi-phase optimization with diverse methods and initializations.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 10.0)
    
    global_best = np.inf
    global_a = None
    
    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    # Primary length
    primary_n = 8192
    if parent is not None and len(parent) >= 64:
        primary_n = len(parent)
    
    print("Primary length: %d, parent available: %s" % (primary_n, parent is not None))
    
    S = np.sqrt(2.0 * primary_n)
    
    # === PHASE 1: Initial optimization from best starting points ===
    # Start with parent first (most likely to be good)
    type_plan_1 = []
    if parent is not None:
        type_plan_1.append(("parent_direct", parent * (S / parent.sum())))
    
    inits = generate_initializations(primary_n, rng, parent)
    
    # Allocate time: 60% to phase 1
    usable_time = max(30.0, deadline - time.monotonic())
    phase1_time = usable_time * 0.55
    
    # First: optimize the parent (or best init) with the main method
    print("=== Phase 1: Main optimization ===")
    t0 = time.monotonic()
    
    # Get the best starting point
    start_point = None
    if parent is not None:
        start_point = parent * (S / parent.sum())
    else:
        start_point = inits[0][1]
    
    # Run main projected gradient for a good chunk of time
    seg1 = phase1_time * 0.5
    a1, s1 = run_projected_gradient(start_point, primary_n, S, seg1, rng,
                                     beta_start=10.0, beta_end=300.0,
                                     use_lse=True, active_threshold=0.98)
    a1 = safe_normalize(a1, primary_n)
    c1 = fft_convolve_self(a1)
    if c1.max() < global_best:
        global_best = c1.max()
        global_a = a1.copy()
        print("  PG-LSE: %.6f (t=%.0fs)" % (global_best, time.monotonic() - t0))
    
    # Phase 1b: Active set refinement
    if global_a is not None and time.monotonic() < deadline - 300:
        seg1b = phase1_time * 0.25
        t1 = time.monotonic()
        a2, s2 = run_peak_equalization(global_a, primary_n, S, seg1b, rng)
        a2 = safe_normalize(a2, primary_n)
        c2 = fft_convolve_self(a2)
        if c2.max() < global_best:
            global_best = c2.max()
            global_a = a2.copy()
            print("  PeakEq: %.6f (t=%.0fs)" % (global_best, time.monotonic() - t1))
    
    # Phase 1c: Mirror descent
    if global_a is not None and time.monotonic() < deadline - 200:
        seg1c = phase1_time * 0.15
        t2 = time.monotonic()
        a3, s3 = run_mirror_descent(global_a, primary_n, S, seg1c, rng,
                                     eta=0.05, top_k=50)
        a3 = safe_normalize(a3, primary_n)
        c3 = fft_convolve_self(a3)
        if c3.max() < global_best:
            global_best = c3.max()
            global_a = a3.copy()
            print("  Mirror: %.6f (t=%.0fs)" % (global_best, time.monotonic() - t2))
    
    # Try some other initializations if time permits
    if time.monotonic() < deadline - 400:
        for name, arr in inits:
            if time.monotonic() > deadline - 350:
                break
            if "parent" in name:
                continue  # Already tried
            rem = min(30.0, (deadline - time.monotonic()) * 0.1)
            if rem < 5:
                break
            t3 = time.monotonic()
            a_t, s_t = run_projected_gradient(arr, primary_n, S, rem, rng,
                                               beta_start=20.0, beta_end=200.0,
                                               use_lse=False, active_threshold=0.95)
            a_t = safe_normalize(a_t, primary_n)
            c_t = fft_convolve_self(a_t)
            if c_t.max() < global_best:
                global_best = c_t.max()
                global_a = a_t.copy()
                print("  Alt [%s]: %.6f (t=%.0fs)" % (name, global_best, time.monotonic() - t3))
    
    print("=== After Phase 1: best=%.6f ===" % global_best)
    
    # === PHASE 2: Intensive polishing of the best solution ===
    if global_a is not None:

        seg2a = max(10.0, remaining * 0.20)

        c_p1 = fft_convolve_self(a_p1)

        seg2b = max(10.0, remaining2 * 0.15)

            print("  Polish PeakEq: %.6f (t=%.0fs)" % (global_best, time.monotonic() - t5))
        
        # 2c: Mirror descent with larger top_k
        remaining3 = deadline - time.monotonic()
        seg2c = max(5.0, remaining3 * 0.10)
        t6 = time.monotonic()
        a_p3, _ = run_mirror_descent(global_a, n, S_cur, seg2c, rng,
                                      eta=0.03, top_k=100)
        a_p3 = safe_normalize(a_p3, n)
        c_p3 = fft_convolve_self(a_p3)
        if c_p3.max() < global_best:
            global_best = c_p3.max()
            global_a = a_p3.copy()
            print("  Polish Mirror: %.6f (t=%.0fs)" % (global_best, time.monotonic() - t6))

    # === PHASE 3: Multi-restart exploration ===
    if global_a is not None and time.monotonic() < deadline - 100:
        remaining = deadline - time.monotonic()
        print("=== Phase 3: Multi-restart (remaining %.0f sec) ===" % remaining)
        
        n = len(global_a)
        S_cur = np.sqrt(2.0 * n)
        num_restarts = min(5, int(remaining / 40))
        
        for r in range(num_restarts):
            if time.monotonic() > deadline - 80:
                break
            # Different perturbation strategies
            if r == 0:
                # Small Gaussian noise
                perturbed = global_a + rng.normal(0, 0.01 * S_cur / np.sqrt(n), n)
            elif r == 1:
                # Anti-palindromic perturbation
                perturbed = global_a.copy()
                half = n // 2
                delta = rng.random(half) * 0.02 * S_cur / n
                perturbed[:half] += delta
                perturbed[n-1-half:] -= delta
            elif r == 2:
                # Swap-based perturbation
                perturbed = global_a.copy()
                idx1 = rng.integers(0, n)
                idx2 = rng.integers(0, n)
                swap_amount = 0.01 * S_cur
                perturbed[idx1] += swap_amount
                perturbed[idx2] -= swap_amount
            elif r == 3:
                # Frequency domain perturbation
                A = fft(global_a, n)
                mask = (np.abs(np.arange(n) - n//2) > n//10)
                A[mask] *= (1 + 0.05 * rng.standard_normal(np.sum(mask)))
                perturbed = np.real(ifft(A)[:n])
            else:
                # Sparse perturbation
                perturbed = global_a.copy()
                sparse_idx = rng.choice(n, size=n//10, replace=False)
                perturbed[sparse_idx] += rng.normal(0, 0.02 * S_cur / np.sqrt(n), len(sparse_idx))
            
            perturbed = np.maximum(perturbed, 0.0)
            perturbed = perturbed * (S_cur / perturbed.sum())
            
            seg_r = min(30.0, (deadline - time.monotonic()) * 0.25)
            t8 = time.monotonic()
            a_r, _ = run_projected_gradient(perturbed, n, S_cur, seg_r, rng,

            if c_r.max() < global_best:
                global_best = c_r.max()
                global_a = a_r.copy()
                print("  Restart %d: %.6f" % (r, global_best))
    
    # === PHASE 4: Final fine-tuning ===
    if global_a is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            n = len(global_a)
            S_cur = np.sqrt(2.0 * n)
            print("=== Phase 4: Final fine-tuning (%.0f sec) ===" % remaining)
            
            # Very careful gradient with tiny steps
            a = global_a.copy()
            step = S_cur / (n * 50.0)

                if c_new.max() < cur:

                    step *= 0.7

            if c_final.max() < global_best:

                global_a = a.copy()
                print("  Final tune: %.6f" % global_best)

    final_list = [float(x) for x in global_a]

    if not np.isfinite(ver):
        print("  WARNING: evaluate_sequence rejected, using fallback")
        global_a = np.ones(1024, dtype=np.float64) * 1.0
        final_list = [float(x) for x in global_a]
    
    ver2 = evaluate_sequence(final_list)
    print("=== Final: %.6f ===" % (ver2 if np.isfinite(ver2) else float('inf')))
    
    return final_list
