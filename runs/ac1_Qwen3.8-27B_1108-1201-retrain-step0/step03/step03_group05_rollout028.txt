    """
    Solve: max sum(b) s.t. b >= 0, (a*b)[k] <= M for k in active_set
    Using cutting plane: start with top constraints, add more as needed.
    """
    import cvxpy as cp
    
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    
    # Start with top-k constraints
    c = np.convolve(a, a)
    top_k = np.argsort(c)[::-1][:max_constr]
    
    constraints = []
    for k in top_k:
        # (a*b)[k] = sum_j a[j]*b[k-j]
        # For k-j in [0, n-1], i.e., j in [k-n+1, k]
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        coeffs = a[j_min:j_max+1]
        vars_slice = b[j_min:j_max+1]
        constraints.append(cp.sum(cp.multiply(coeffs, vars_slice)) <= M)
    
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    # Solve
    prob.solve(solver=solver)
    
    # Check for violated constraints and add them (cutting plane)
    for _ in range(20):
        b_val = b.value
        c_new = np.convolve(a, b_val)
        violated = np.where(c_new > M + 1e-6)[0]
        if len(violated) == 0:
            break
        # Add most violated constraints
        add_k = violated[np.argsort(c_new[violated])[::-1]][:10]
        for k in add_k:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)

        prob = cp.Problem(objective, constraints)
        prob.solve(solver=solver)
    
    return b.value

Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Algorithm:
- Nesterov-accelerated proximal gradient on log-sum-exp smoothed objective
- Adaptive temperature (beta) continuation
- FFT-based O(n log n) gradient computation
- LP-based cutting plane for escaping local minima
- Multiple initializations and restarts

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_bg_grad(a, w, n):
    """
    Compute grad of smoothed max w.r.t. a.
    w: probability distribution over convolution indices (length 2n-1)
    Returns gradient of length n.
    grad[j] = sum_k w[k] * a[k-j] (cross-correlation)
    """
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = np.real(ifft(W * A))[:n]
    return grad


def smooth_obj_grad(a, beta, S):
    """
    Compute log-sum-exp smoothed objective and gradient.
    Returns (obj, grad_a).
    """
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp = exp_bc.sum()
    obj = (c_max + np.log(sum_exp)) / beta
    
    w = exp_bc / sum_exp
    grad = compute_bg_grad(a, w, n)
    
    return obj, grad

    x_prev = a.copy()

    best_a = a.copy()
    t = 1.0

    c0 = fft_convolve_self(a)
    L = np.linalg.norm(a) * np.sqrt(len(a)) * 2  # rough upper bound

            break

        y = best_a.copy()
        x_prev = best_a.copy()
        t = 1.0
        best_a_local = best_a.copy()
        best_score_local = best_score

        step = 1.0 / L
        
        while time.monotonic() - start < max_time:

            obj_y, grad_y = smooth_obj_grad(y, beta, S)

            x_new = y - step * grad_y
            x_new = project_simplex(x_new, S)
            
            # Check at x_new
            c_x = fft_convolve_self(x_new)
            score_x = c_x.max()
            if score_x < best_score_local:

        print("  beta=%d: score=%.8f" % (beta, best_score))

"""

Algorithm:
- Nesterov-accelerated proximal gradient on log-sum-exp smoothed objective
- Adaptive temperature continuation (beta increases over time)
- FFT-based O(n log n) gradient computation
- Multiple initializations with diverse structures
- Periodic restarts with Gaussian perturbation
- Final polishing with exact subgradient (top-1 active set)

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_grad_lse(a, w):
    """
    Compute grad of log-sum-exp smoothed max w.r.t. a.
    w: probability distribution over convolution indices (length 2n-1)
    grad[j] = sum_m a[m] * w[m+j]  (cross-correlation)
    """
    n = len(a)
    fsize = 1 << (3 * n - 3).bit_length()
    a_flip = a[::-1]
    AF = fft(a_flip, fsize)
    WF = fft(w, fsize)
    conv = np.real(ifft(AF * WF))
    return conv[n-1 : 2*n-1]


def lse_obj_and_grad(a, beta):
    """Compute log-sum-exp objective and gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp = exp_bc.sum()
    obj = (c_max + np.log(sum_exp)) / beta
    w = exp_bc / sum_exp
    grad = compute_grad_lse(a, w)
    return obj, grad


def nesterov_lse(a_init, n, S, beta, max_time, start_time, 
                  n_iters=None, step_scale=1.0, label=""):
    """
    Nesterov-accelerated gradient descent on LSE smoothed objective.
    Returns (best_a, best_score).
    """
    a = a_init.copy().astype(np.float64)
    y = a.copy()
    x_prev = a.copy()
    t = 1.0
    
    # Estimate Lipschitz constant
    c0 = fft_convolve_self(y)
    max_a = np.max(np.abs(y))
    L = beta * 4.0 * max_a + 0.1  # conservative bound
    step = step_scale / L
    
    best_a = a.copy()
    best_score = c0.max()
    it = 0
    
    while (n_iters is None or it < n_iters) and time.monotonic() - start_time < max_time:
        it += 1
        
        # Compute gradient at y
        obj_y, grad_y = lse_obj_and_grad(y, beta)
        
        # Gradient step
        x_new = y - step * grad_y
        x_new = project_simplex(x_new, S)
        
        # Evaluate at x_new

            c_new = fft_convolve_self(x_new)
            score_new = c_new.max()
            if score_new < best_score:
                best_score = score_new
                best_a = x_new.copy()
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        moment = (t - 1.0) / t_new
        y = x_new + moment * (x_new - x_prev)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        
        x_prev = x_new
        t = t_new
