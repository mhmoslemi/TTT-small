def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)

        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    """

    This is the cross-correlation needed for gradient computation.
    """
    n = len(a)
    fsize = 1 << (n + len(w) - 1).bit_length()
    A = fft(a, fsize)
    W = fft(w, fsize)
    result = np.real(ifft(np.conj(A) * W))[:n]
    return result


def softmax_obj_grad(z, beta, S):
    """
    Compute smoothed objective and gradient w.r.t. z (softmax params).
    a = S * softmax(z), objective = 2n * (1/beta)*log(sum exp(beta*c_k)) / S^2
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    bc = beta * (c - c_max)
    exp_bc = np.exp(bc)
    sum_exp_bc = exp_bc.sum()
    L = (c_max + np.log(sum_exp_bc)) / beta

    # Objective: 2n * L / S^2
    obj = 2.0 * n * L / (S * S)

    # w_k = exp(beta*c_k) / sum(exp(beta*c_k))
    w = exp_bc / sum_exp_bc

    # dL/da_j = 2 * sum_k w[k] * a[k-j] = 2 * cross_corr
    cross = compute_cross_corr(a, w)
    dL_da = 2.0 * cross

    # dF/da_j = 2n * (S * dL_da_j - 2L) / S^3
    dF_da = 2.0 * n * (S * dL_da - 2.0 * L) / (S ** 3)

    # grad_z[k] = a[k] * (dF_da[k] - dot(dF_da, a)/S)
    dot = np.dot(dF_da, a)
    grad_z = a * (dF_da - dot / S)

    return obj, grad_z

                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, lr_scale=1.0, top_k=64):
    """
    Mirror descent (multiplicative weights) with adaptive step size.
    a_new[j] = a[j] * exp(-eta * g[j]), then normalize.
    """
    a = np.clip(a_init, 1e-15, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    eta = lr_scale / (n ** 0.5)
    it = 0
    patience = 0
    prev_score = best_score

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()

        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience > 200:
            # Try a random restart
            a = best_a + rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = np.maximum(a, 1e-10)
            a = a * S / a.sum()
            eta *= 0.5
            patience = 0
            continue

        # Compute top-k weights
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        cross = compute_cross_corr(a, w_full)
        dL_da = 2.0 * cross
        # L approximation: weighted mean of top-k
        L_approx = np.dot(weights, top_vals)

        dF_da = 2.0 * n * (S * dL_da - 2.0 * L_approx) / (S ** 3)

        # Adaptive step size
        gmax = np.abs(dF_da).max()
        if gmax < 1e-15:
            break
        step = eta / gmax

        # Multiplicative update
        a_new = a * np.exp(-step * dF_da)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new * S / a_new.sum()

        # Check improvement
        c_new = fft_convolve_self(a_new)
        score_new = c_new.max()

        if score_new < cur:
            a = a_new
            # Slightly increase eta if improving fast
            if rng.random() < 0.1:
                eta *= 1.05
        else:
            # Take a smaller step
            a_half = a * np.exp(-0.25 * step * dF_da)
            a_half = np.maximum(a_half, 1e-30)
            a_half = a_half * S / a_half.sum()
            c_half = fft_convolve_self(a_half)
            if c_half.max() < cur:
                a = a_half
            else:
                eta *= 0.5
                if eta < 1e-10:
                    break
            a = a  # keep old if no improvement

        # Periodic small random perturbation
        if it % 500 == 0 and rng.random() < 0.15:
            a = a + rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = np.maximum(a, 1e-10)
            a = a * S / a.sum()

    return best_a, best_score


def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=8, top_k_max=256):
    """Nesterov-accelerated top-k subgradient with FFT gradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 60)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        cross = compute_cross_corr(a, w_full)
        grad = 2.0 * cross * S - 2.0 * np.dot(weights, top_vals)
        # Actually let me use the proper gradient
        L_approx = np.dot(weights, top_vals)
        grad = 2.0 * n * (S * 2.0 * cross - 2.0 * L_approx) / (S ** 3)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 3000 == 0 and rng.random() < 0.05:
            y = project_simplex(y + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def try_symmetric_optimization(a_init, n, S, max_time, rng):
    """
    Optimize with symmetry constraint a[j] = a[n-1-j].
    Halves the variable space.
    """
    if n < 4:
        return a_init, np.inf
    
    half = (n + 1) // 2
    # Extract half-sequence
    a = np.clip(a_init, 1e-15, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    
    # Force symmetry
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 1e-15)
    a_sym = a_sym * S / a_sym.sum()
    
    # Work on the half-sequence
    if n % 2 == 0:
        half_n = n // 2
        h = a_sym[:half_n].copy()
    else:
        half_n = (n + 1) // 2
        h = a_sym[:half_n].copy()
    
    h = h * S / (2.0 * h.sum() - (h[half_n - 1] if n % 2 == 1 else 0))
    
    # Define objective in terms of h
    def build_full(h_val):
        if n % 2 == 0:
            full = np.concatenate([h_val, h_val[::-1]])
        else:
            full = np.concatenate([h_val, h_val[-2::-1]])
        return full
    
    def sym_obj_grad(z):
        h_val = np.exp(z - z.max())
        h_val = h_val / h_val.sum()
        h_val = h_val * S / 2.0  # rough normalization
        full = build_full(h_val)
        full = full * S / full.sum()
        c = fft_convolve_self(full)
        c_max = c.max()
        beta = 100
        bc = beta * (c - c_max)
        exp_bc = np.exp(bc)
        sum_ebc = exp_bc.sum()
        L = (c_max + np.log(sum_ebc)) / beta
        obj = 2.0 * len(full) * L / (S * S)
        # For simplicity, use finite differences for gradient in this helper
        return obj
    
    # Just run L-BFGS on the half-sequence
    h_start = np.log(h / (S/2) + 1e-30)
    h_start = h_start - h_start.mean()
    
    best_score = np.inf
    best_h = h.copy()
    
    for beta in [50, 200, 500, 2000]:
        if time.monotonic() + 30 > start_mirror:
            break
        
        def fun_sym(z_val):
            h_val = np.exp(z_val - z_val.max())
            h_val = h_val / h_val.sum()
            full = build_full(h_val)
            full = full * S / full.sum()
            c = fft_convolve_self(full)
            c_max = c.max()
            bc = beta * (c - c_max)
            exp_bc = np.exp(bc)
            L = (c_max + np.log(exp_bc.sum())) / beta
            return 2.0 * n * L / (S * S)
        
        def jac_sym(z_val):
            h_val = np.exp(z_val - z_val.max())
            h_val = h_val / h_val.sum()
            full = build_full(h_val)
            full = full * S / full.sum()
            c = fft_convolve_self(full)
            c_max = c.max()
            bc = beta * (c - c_max)
            exp_bc = np.exp(bc)
            sum_ebc = exp_bc.sum()
            w = exp_bc / sum_ebc
            L = (c_max + np.log(sum_ebc)) / beta
            
            cross = compute_cross_corr(full, w)
            dL_da = 2.0 * cross
            dF_da = 2.0 * n * (S * dL_da - 2.0 * L) / (S ** 3)
            
            # Map gradient back to h (first half)
            if n % 2 == 0:
                grad_h = dF_da[:half_n].copy()
                grad_h += dF_da[half_n:][::-1]
            else:
                grad_h = dF_da[:half_n].copy()
                grad_h += dF_da[half_n-1:0:-1]
            
            # Softmap Jacobian
            dot = np.dot(dF_da, full)
            dF_dh = grad_h
            grad_z = h_val * (dF_dh - np.dot(dF_dh, h_val) / h_val.sum())
            return grad_z
        
        try:
            result = scipy_minimize(
                fun_sym, h_start,
                jac=jac_sym,
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10, 'iprint': -1}
            )
            h_start = result.x
            h_new = np.exp(h_start - h_start.max())
            h_new = h_new / h_new.sum()
            full_new = build_full(h_new)
            full_new = full_new * S / full_new.sum()
            score = fft_convolve_self(full_new).max()
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
        except Exception:
            pass
    
    if best_score < np.inf:
        if n % 2 == 0:
            full = np.concatenate([best_h, best_h[::-1]])
        else:
            full = np.concatenate([best_h, best_h[-2::-1]])
        full = full * S / full.sum()
        return full, best_score
    return a_init, np.inf


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # --- Get initial sequence ---
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0.0, 1000.0)))
    except Exception:
        pass

    if inits:
        main_len = len(inits[0][1])
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * float(n))
    tt = np.linspace(0, 1, n, endpoint=False)

    # Add diverse initializations
    inits.append(("rand", rng.random(n) + 0.1))
    inits.append(("raised_cos", 0.5 + 0.5 * np.cos(2 * np.pi * tt)))
    inits.append(("gauss", np.exp(-2.0 * (tt - 0.5) ** 2) + 0.02))
    inits.append(("sin_multi", 1.0 + 0.2 * np.sin(2*np.pi*3*tt) +
                  0.15 * np.cos(2*np.pi*7*tt) + 0.1 * np.sin(2*np.pi*11*tt) +
                  0.08 * np.cos(2*np.pi*17*tt)))
    inits.append(("triangle", np.maximum(1.0 - 2.0 * np.abs(tt - 0.5), 0.0) + 0.05))

    n_inits = len(inits)
    print("=== Optimization: n=%d, %d inits, budget=%.0fs ===" % (n, n_inits, budget_s))
    print("Target: 1.5027, Previous best: 1.508805")

    # --- Phase 1: L-BFGS continuation on each init ---
    usable = deadline - time.monotonic()
    per_init = usable * 0.40 / n_inits
    per_init = max(per_init, 8.0)

    print("\n=== Phase 1: L-BFGS Continuation (%.1f sec each) ===" % per_init)

    phase1_best = {}

    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break

        t0 = time.monotonic()
        a_norm = safe_normalize(arr, n)
        c = fft_convolve_self(a_norm)
        init_score = c.max()
        init_obj = 2.0 * n * init_score / (a_norm.sum() ** 2)

        if init_obj < global_best:
            global_best = init_obj
            global_a = a_norm.copy()

        # L-BFGS continuation
        a_new, score = run_lbfgs_continuation(
            a_norm, n, S, per_init, rng,
            beta_schedule=[5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000],
            maxiter_per_beta=150
        )
        a_new = safe_normalize(a_new, n)
        c_new = fft_convolve_self(a_new)
        final_obj = 2.0 * n * c_new.max() / (a_new.sum() ** 2)

        phase1_best[name] = (a_new, final_obj)

        if final_obj < global_best:
            global_best = final_obj
            global_a = a_new.copy()
            print("  [OK] %s: %.8f (init=%.6f, %.1fs)" %
                  (name, global_best, init_obj, time.monotonic() - t0))
        else:
            print("  [%s: %.8f (init=%.6f, best=%.6f, %.1fs)]" %
                  (name, final_obj, init_obj, global_best, time.monotonic() - t0))

    print("\n=== After Phase 1: global_best=%.8f ===" % global_best)

    # --- Phase 2: Mirror descent on best solution ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            md_time = min(remaining * 0.25, 300)
            print("\n=== Phase 2: Mirror Descent (%.1f sec) ===" % md_time)
            a_md, score_md = run_mirror_descent(
                global_a, n, S, md_time, rng,
                lr_scale=1.0, top_k=128
            )
            a_md = safe_normalize(a_md, n)
            obj_md = 2.0 * n * score_md / (a_md.sum() ** 2)
            if obj_md < global_best:
                global_best = obj_md
                global_a = a_md.copy()
                print("  Mirror descent: %.8f" % global_best)
            else:
                print("  Mirror descent: %.8f (no improvement)" % obj_md)

    print("\n=== After Phase 2: global_best=%.8f ===" % global_best)

    # --- Phase 3: Nesterov top-k ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            nesterov_time = min(remaining * 0.15, 120)
            print("\n=== Phase 3: Nesterov Top-K (%.1f sec) ===" % nesterov_time)
            a_nest, score_nest = run_nesterov_toppk(
                global_a, n, S, nesterov_time, rng,
                top_k_init=8, top_k_max=256
            )
            a_nest = safe_normalize(a_nest, n)
            obj_nest = 2.0 * n * score_nest / (a_nest.sum() ** 2)
            if obj_nest < global_best:
                global_best = obj_nest
                global_a = a_nest.copy()
                print("  Nesterov: %.8f" % global_best)
            else:
                print("  Nesterov: %.8f (no improvement)" % obj_nest)

    print("\n=== After Phase 3: global_best=%.8f ===" % global_best)

    # --- Phase 4: Perturbation + L-BFGS restart ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print("\n=== Phase 4: Perturbation + L-BFGS (%.1f sec) ===" % remaining)

            n_perturb = max(4, int(remaining / 50))
            perturb_budget = remaining * 0.6 / n_perturb

            for p in range(n_perturb):
                if time.monotonic() > deadline - 60:
                    break

                # Multiple noise scales, decreasing
                noise_scales = [0.02, 0.01, 0.005, 0.002]
                noise_scale = noise_scales[p % len(noise_scales)]
                a_pert = global_a + rng.normal(0, noise_scale * S / np.sqrt(n), n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = safe_normalize(a_pert, n)

                a_new, score = run_lbfgs_continuation(
                    a_pert, n, S, perturb_budget, rng,
                    beta_schedule=[50, 200, 500, 1000, 2000, 5000],
                    maxiter_per_beta=150
                )
                a_new = safe_normalize(a_new, n)
                c_new = fft_convolve_self(a_new)
                obj_new = 2.0 * n * c_new.max() / (a_new.sum() ** 2)
                if obj_new < global_best:
                    global_best = obj_new
                    global_a = a_new.copy()
                    print("  Perturb %d (scale=%.4f): %.8f" % (p, noise_scale, global_best))
                else:
                    print("  Perturb %d (scale=%.4f): %.8f (no improve)" % (p, noise_scale, obj_new))

    print("\n=== After Phase 4: global_best=%.8f ===" % global_best)

    # --- Phase 5: Cross-pollination ---
    if global_a is not None and len(phase1_best) >= 2:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print("\n=== Phase 5: Cross-pollination (%.1f sec) ===" % remaining)

            sorted_results = sorted(phase1_best.items(), key=lambda x: x[1][1])
            n_xpoll = max(2, int(remaining / 40))
            xp_budget = remaining * 0.6 / n_xpoll

            for xp in range(n_xpoll):
                if time.monotonic() > deadline - 30:
                    break

                idx1 = xp % len(sorted_results)
                idx2 = (xp + 1) % len(sorted_results)
                if idx2 == idx1:
                    idx2 = (idx2 + 1) % len(sorted_results)

                a1 = sorted_results[idx1][1][0]
                a2 = sorted_results[idx2][1][0]

                for blend in [0.3, 0.5, 0.7]:
                    if time.monotonic() > deadline - 20:
                        break
                    a_blend = blend * a1 + (1 - blend) * a2
                    a_blend = safe_normalize(a_blend, n)

                    a_new, score = run_lbfgs_continuation(
                        a_blend, n, S, xp_budget / 3, rng,
                        beta_schedule=[20, 100, 500, 2000],
                        maxiter_per_beta=100
                    )
                    a_new = safe_normalize(a_new, n)
                    c_new = fft_convolve_self(a_new)
                    obj_new = 2.0 * n * c_new.max() / (a_new.sum() ** 2)
                    if obj_new < global_best:
                        global_best = obj_new
                        global_a = a_new.copy()
                        print("  XPoll %d (blend=%.1f): %.8f" % (xp, blend, global_best))

    print("\n=== After Phase 5: global_best=%.8f ===" % global_best)

    # --- Phase 6: Final high-beta L-BFGS polishing ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print("\n=== Phase 6: Final L-BFGS Polish (%.1f sec) ===" % remaining)
            a_final, score = run_lbfgs_continuation(
                global_a, n, S, remaining, rng,
                beta_schedule=[200, 500, 1000, 2000, 5000, 10000, 20000],
                maxiter_per_beta=200
            )
            a_final = safe_normalize(a_final, n)
            c_final = fft_convolve_self(a_final)
            obj_final = 2.0 * n * c_final.max() / (a_final.sum() ** 2)
            if obj_final < global_best:
                global_best = obj_final
                global_a = a_final.copy()
                print("  Final L-BFGS: %.8f" % global_best)
            else:
                print("  Final L-BFGS: %.8f (no improvement)" % obj_final)

    # --- Phase 7: Try symmetric optimization ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("\n=== Phase 7: Symmetric Optimization (%.1f sec) ===" % remaining)
            a_sym, score_sym = try_symmetric_optimization(
                global_a, n, S, remaining, rng
            )
            a_sym = safe_normalize(a_sym, n)
            obj_sym = 2.0 * n * score_sym / (a_sym.sum() ** 2)
            if obj_sym < global_best:
                global_best = obj_sym
                global_a = a_sym.copy()
                print("  Symmetric: %.8f" % global_best)
            else:
                print("  Symmetric: %.8f (no improvement)" % obj_sym)

    # --- Phase 8: Final mirror descent + L-BFGS ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print("\n=== Phase 8: Final Combined Polish (%.1f sec) ===" % remaining)
            half_time = remaining / 2.0

            a_md2, score_md2 = run_mirror_descent(
                global_a, n, S, half_time, rng,
                lr_scale=0.5, top_k=64
            )
            a_md2 = safe_normalize(a_md2, n)
            obj_md2 = 2.0 * n * score_md2 / (a_md2.sum() ** 2)
            if obj_md2 < global_best:
                global_best = obj_md2
                global_a = a_md2.copy()
                print("  MD2: %.8f" % global_best)

            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_pol, score_pol = run_lbfgs_continuation(
                    global_a, n, S, remaining2, rng,
                    beta_schedule=[500, 2000, 5000, 20000],
                    maxiter_per_beta=300
                )
                a_pol = safe_normalize(a_pol, n)
                c_pol = fft_convolve_self(a_pol)
                obj_pol = 2.0 * n * c_pol.max() / (a_pol.sum() ** 2)
                if obj_pol < global_best:
                    global_best = obj_pol
                    global_a = a_pol.copy()
                    print("  Polish L-BFGS: %.8f" % global_best)

    # --- Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_score = evaluate_sequence([float(x) for x in global_a])
    print("\n=== FINAL: length=%d, score=%.8f ===" % (len(global_a), final_score))
    return [float(x) for x in global_a]
