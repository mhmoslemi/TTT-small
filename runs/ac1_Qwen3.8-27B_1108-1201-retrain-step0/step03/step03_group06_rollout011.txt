```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative refinement.

At each iteration, solve an LP to find the best direction g that maximizes
sum(g) subject to bounded cross-correlation with the current a. Then update
a by blending toward g. This is based on the state-of-the-art approach.

Key innovations over previous program:
1. LP-based direction finding (much more principled than subgradient)
2. Adaptive step size via bisection (ensures monotone decrease)
3. Sparse constraint matrix for efficient LP solves
4. Multiple restarts and polish phases
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def build_constraint_matrix(a, n):
    """Build sparse constraint matrix for (a*g)_k <= M.
    A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    Shape: (2n-1, n)
    """
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_direction(a, M, n):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns optimal g (unnormalized).
    """
    A = build_constraint_matrix(a, n)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def normalize_nonneg(a, S):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def evaluate_obj(a):
    n = len(a)
    S = np.sqrt(2.0 * n)
    a_norm = normalize_nonneg(a, S)
    c = fft_convolve_self(a_norm)
    return c.max()
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative refinement.

Algorithm: At each iteration, solve the LP
    max sum(g) s.t. (a*g)_k <= M for all k, g >= 0
where M = max((a*a)_k). The LP solution g* gives the best direction
to move. We then blend a toward g* with an adaptive step size.

This is based on the state-of-the-art convex optimization approach for
flat autocorrelation sequences.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def precompute_lp_structure(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows_list.append(k)
            cols_list.append(j)
    return (np.array(rows_list, dtype=np.int64),
            np.array(cols_list, dtype=np.int64))


def solve_lp_direction(a, M, rows, cols, n):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0.
    A[k,j] = a[k-j] if valid.
    """
    vals = a[rows - cols]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_refine(a, n, S, time_budget, rows, cols,
              step_init=0.5, verbose=True):
    """LP-based iterative refinement."""
    a = normalize_nonneg(a.copy().astype(np.float64), S)
    c = fft_convolve_self(a)
    M = c.max()
    best_a = a.copy()
    best_M = M
    start = time.monotonic()
    step = step_init
    stall = 0
    max_stall = 20

    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        t0 = time.monotonic()

        g = solve_lp_direction(a, M, rows, cols, n)
        if g is None or g.sum() < 1e-10:
            if verbose:
                print("  LP failed at iter %d" % it)
            break

        g_norm = g * (S / g.sum())

        # Try step size t
        a_trial = (1.0 - step) * a + step * g_norm
        a_trial = normalize_nonneg(a_trial, S)
        c_trial = fft_convolve_self(a_trial)
        M_trial = c_trial.max()

        if M_trial < M - 1e-14:
            # Accept
            a = a_trial
            M = M_trial
            stall = 0
            if M < best_M:
                best_M = M
                best_a = a.copy()
            # Try increasing step
            if step < 0.9:
                step = min(1.0, step * 1.2)
        else:
            # Reject, reduce step
            step *= 0.5
            stall += 1
            if stall > max_stall or step < 1e-4:
                break

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            lp_time = time.monotonic() - t0
            print("  LP iter %d: M=%.8f, step=%.3f, lp_time=%.1fs, elapsed=%.0fs"
                  % (it, best_M, step, lp_time, elapsed))

    return best_a, best_M


def subgradient_refine(a, n, S, time_budget, T_init=100, T_final=2000,
                       lr_init=0.05, verbose=True):
    """Subgradient descent as a polish method."""
    a = normalize_nonneg(a.copy().astype(np.float64), S)
    c = fft_convolve_self(a)
    best_M = c.max()
    best_a = a.copy()
    start = time.monotonic()
    momentum = 0.9
    a_prev = a.copy()
    beta = momentum

    it = 0
    while time.monotonic() - start < time_budget:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.6)

        c = fft_convolve_self(a)
        c_max = c.max()
        if c_max < 1e-30:
            break

        if c_max < best_M - 1e-12:
            best_M = c_max
            best_a = a.copy()

        # Weighted subgradient
        scaled = T * c / c_max
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

        # Gradient: g[m] = 2 * sum_k w[k] * a[k-m]
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g = 2.0 * corr[:n]

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Nesterov
        a_prev_old = a_prev.copy()
        a_prev = a.copy()
        y = a + (beta - 1.0) / (1.0 + beta) * (a - a_prev_old)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)

        # Line search
        lr = lr_init * (0.5 + 0.5 * (1.0 - progress))
        best_a_trial = a.copy()
        best_obj = c_max
        for _ in range(8):
            a_trial = y - lr * g / gnorm
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a_trial = a_trial
            lr *= 0.5

        a = best_a_trial
        beta = 0.5 * (1 + np.sqrt(1 + 4 * beta * beta))

        if verbose and it % 20 == 0:
            print("  SG iter %d: best_M=%.8f, T=%.0f, elapsed=%.0fs"
                  % (it, best_M, T, time.monotonic() - start))

    return best_a, best_M
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative refinement.

At each iteration, solve the LP:
    max sum(g)  s.t.  (a*g)_k <= M for all k,  g >= 0
where M = max_k (a*a)_k is the current autocorrelation peak.
The LP solution gives the optimal descent direction. We blend toward it
with adaptive step size, ensuring monotone decrease in the objective.

Key innovations vs. previous program:
1. LP-based direction finding (principled convex subproblem, not subgradient)
2. Precomputed sparse matrix structure for efficient per-iteration updates
3. Adaptive step-size with stall detection
4. Multi-phase strategy: LP refine → LP polish → subgradient fine-tune
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def precompute_csr(n):
    """Precompute CSR structure (indptr, indices) for (a*g)_k constraint matrix."""
    n_rows = 2 * n - 1
    indptr = np.zeros(n_rows + 1, dtype=np.int64)
    for k in range(n_rows):
        if k < n:
            count = k + 1
        else:
            count = 2 * n - 1 - k
        indptr[k + 1] = indptr[k] + count

    nnz = indptr[-1]
    indices = np.zeros(nnz, dtype=np.int64)
    for k in range(n_rows):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        start = indptr[k]
        cnt = j_end - j_start + 1
        indices[start:start + cnt] = np.arange(j_start, j_end + 1, dtype=np.int64)
    return indptr, indices


def compute_data(a, indptr, indices, n):
    """Compute data array: data[i] = a[row_i - col_i]."""
    nnz = len(indices)
    pos = np.arange(nnz)
    k = np.repeat(np.arange(2 * n - 1, dtype=np.int64), np.diff(indptr))
    j = indices
    return a[k - j]


def lp_refine(a, n, S, time_budget, indptr, indices,
              step_init=0.4, verbose=True):
    """LP-based iterative refinement with adaptive step size."""
    a = normalize_nonneg(a.copy().astype(np.float64), S)
    c = fft_convolve_self(a)
    M = c.max()
    best_a = a.copy()
    best_M = M
    start = time.monotonic()
    step = step_init
    stall = 0
    max_stall = 30
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        t0 = time.monotonic()

        # Build constraint matrix
        data = compute_data(a, indptr, indices, n)
        A = csr_matrix((data, indices, indptr), shape=(2 * n - 1, n))

        # Solve LP
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max(5, time_budget / 100)})
        lp_time = time.monotonic() - t0

        if not res.success or res.x is None:
            if verbose:
                print("  LP failed at iter %d (%s)" % (it, res.message))
            break

        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break

        g_norm = g * (S / g_sum)

        # Try the step
        a_trial = (1.0 - step) * a + step * g_norm
        a_trial = normalize_nonneg(a_trial, S)
        c_trial = fft_convolve_self(a_trial)
        M_trial = c_trial.max()

        if M_trial < M - 1e-13:
            a = a_trial
            M = M_trial
            stall = 0
            if M < best_M - 1e-14:
                best_M = M
                best_a = a.copy()
            if step < 0.95:
                step = min(1.0, step * 1.15)
        else:
            step *= 0.5
            stall += 1
            if stall > max_stall or step < 1e-5:
                break

        if verbose and (it % 5 == 0 or it <= 3):
            elapsed = time.monotonic() - start
            obj_val = 2.0 * n * best_M / (S * S)
            print("  LP iter %d: M=%.8f obj=%.6f step=%.3f lp=%.1fs elapsed=%.0fs"
                  % (it, best_M, obj_val, step, lp_time, elapsed))

    return best_a, best_M


def subgradient_refine(a, n, S, time_budget, T_init=200, T_final=5000,
                       lr_init=0.04, verbose=True):
    """Nesterov accelerated weighted subgradient descent for polish."""
    a = normalize_nonneg(a.copy().astype(np.float64), S)
    c = fft_convolve_self(a)
    best_M = c.max()
    best_a = a.copy()
    start = time.monotonic()
    beta = 0.9
    a_prev = a.copy()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.6)

        c = fft_convolve_self(a)
        c_max = c.max()
        if c_max < best_M - 1e-12:
            best_M = c_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break

        scaled = T * c / c_max
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g = 2.0 * corr[:n]

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        a_old = a.copy()
        y = a + (beta - 1.0) / (1.0 + beta) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        a_prev = a_old

        lr = lr_init * (0.4 + 0.6 * (1.0 - progress))
        best_a_trial = a.copy()
        best_obj = c_max
        for _ in range(10):
            a_trial = y - lr * g / gnorm
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a_trial = a_trial
                break
            lr *= 0.5

        a = best_a_trial
        beta = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * beta * beta))

        if verbose and it % 30 == 0:
            obj_val = 2.0 * n * best_M / (S * S)
            print("  SG iter %d: obj=%.6f T=%.0f elapsed=%.0fs"
                  % (it, obj_val, T, time.monotonic() - start))

    return best_a, best_M


def coordinate_polish(a, n, S, time_budget, verbose=True):
    """Simple coordinate descent for final polish."""
    a = normalize_nonneg(a.copy().astype(np.float64), S)
    c = fft_convolve_self(a)
    best_M = c.max()
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_obj = best_M
            for factor in [0.7, 1.4, 0.5, 1.6, 0.85, 1.15]:
                a[i] = orig * factor
                a = normalize_nonneg(a, S)
                c_ch = fft_convolve_self(a)
                if c_ch.max() < best_obj - 1e-13:
                    best_obj = c_ch.max()
                    best_val = orig * factor
            a[i] = best_val
            if best_val != orig:
                a = normalize_nonneg(a, S)
                c = fft_convolve_self(a)
                if c.max() < best_M - 1e-13:
                    best_M = c.max()
                    best_a = a.copy()
                    improved = True
        if not improved:
            break

    return best_a, best_M
```
