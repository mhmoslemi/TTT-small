from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_a(a, w):
    """Compute grad_a[j] = 2 * sum_k w[k] * a[j-k] (cross-correlation via FFT)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

        # Preserve phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        
        # Ensure conjugate symmetry (for real output)
        A_new[0] = np.abs(A_new[0]) + 0j
        for k in range(1, fsize // 2 + 1):
            A_new[fsize - k] = np.conj(A_new[k])
        
        # Time domain
        a_new = np.real(ifft(A_new))[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * (S / s_new)
        
        # Check if it actually improved
        c_new = fft_convolve_self(a_new)
        if c_new.max() < best_score:
            best_score = c_new.max()
            best_a = a_new.copy()
        
        a_work = a_new
    
    return best_a


def subgradient_step(a, S, n_steps=100, lr_decay=0.999):
    """Direct subgradient steps targeting the maximizing lag(s)."""
    n = len(a)
    a_work = a.copy().astype(np.float64)
    s = a_work.sum()
    if s < 1e-15:
        return a
    a_work = a_work * (S / s)
    
    best_score = fft_convolve_self(a_work).max()
    best_a = a_work.copy()
    lr = 1.0 / (np.sqrt(n) * 10)
    
    for it in range(n_steps):
        c = fft_convolve_self(a_work)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a_work.copy()
        
        # Top-k lags
        top_k = min(16, max(2, int(np.sqrt(n) / 4)))
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Compute gradient using weighted combination
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]
        
        grad = compute_grad_a(a_work, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step
        a_new = a_work - lr * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * (S / s_new)
        
        a_work = a_new
        lr *= lr_decay
    
    return best_a


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp
    
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta
    
    w = exp_bc / sum_exp_bc
    grad_a = compute_grad_a(a, w)
    
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)
    
    return obj, grad_z


def run_lbfgs(a_init, n, S, max_time, rng, beta_schedule=None, maxiter_per_beta=200):
    """L-BFGS-B with continuation on beta."""
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
    
    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()
    
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    
    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break
        
        n_attempts = 1
        if beta >= 500 and max_time > 30:
            n_attempts = 2
        
        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 3:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.01, n)
                z = z - z.mean()
            
            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]
            
            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]
            
            try:
                result = scipy_minimize(
                    fun, z, jac=jac, method='L-BFGS-B',
                    options={'maxiter': maxiter_per_beta, 'maxfun': maxiter_per_beta * 3,
                             'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50, 'iprint': -1}
                )
                z = result.x
                z = z - z.mean()
            except Exception:
                break
            
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s
            
            c = fft_convolve_self(a)
            score = c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def spectral_refine_interleaved(a_init, n, S, max_time, rng):
    """Interleave spectral equalization with L-BFGS for better convergence."""
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    
    # Phase 1: Aggressive spectral equalization
    gamma_schedule = [0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.45, 0.5]
    
    for gamma in gamma_schedule:
        if time.monotonic() - start > max_time * 0.5:
            break
        a = spectral_equalize(a, gamma=gamma, n_iter=30, S=S)
        a = safe_normalize(a, n)
        c = fft_convolve_self(a)
        if c.max() < best_score:
            best_score = c.max()
            best_a = a.copy()
        
        # Short L-BFGS polish after each spectral step
        remaining = max_time * 0.5 - (time.monotonic() - start)
        if remaining > 5:
            a, score = run_lbfgs(a, n, S, min(remaining * 0.4, 20), rng,
                                 beta_schedule=[50, 200, 500], maxiter_per_beta=80)
            a = safe_normalize(a, n)
            c = fft_convolve_self(a)
            if c.max() < best_score:
                best_score = c.max()
                best_a = a.copy()
    
    print("    Spectral refine: %.8f" % best_score)
    
    # Phase 2: Fine-tune with subgradient + L-BFGS
    remaining = max_time - (time.monotonic() - start)
    if remaining > 10 and best_a is not None:
        a = subgradient_step(best_a, S, n_steps=200)
        a = safe_normalize(a, n)
        c = fft_convolve_self(a)
        if c.max() < best_score:
            best_score = c.max()
            best_a = a.copy()
        
        remaining2 = max_time - (time.monotonic() - start)
        if remaining2 > 10:
            a, score = run_lbfgs(best_a, n, S, remaining2, rng,
                                 beta_schedule=[200, 500, 1000, 2000, 5000, 8000],
                                 maxiter_per_beta=200)
            a = safe_normalize(a, n)
            c = fft_convolve_self(a)
            if c.max() < best_score:
                best_score = c.max()
                best_a = a.copy()
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)
    
    global_best = np.inf
    global_a = None
    
    # Get initial sequence
    n = 8192
    S = np.sqrt(2.0 * float(n))
    tt = np.linspace(0, 1, n, endpoint=False)
    
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", safe_normalize(np.clip(given, 0.0, 1000.0), n)))
    except Exception:
        pass
    
    if not inits:
        n = 8192
        S = np.sqrt(2.0 * float(n))
        tt = np.linspace(0, 1, n, endpoint=False)
        inits.append(("uniform", np.ones(n)))
    
    # Add structured initializations based on frequency analysis
    # Multi-frequency with irrational frequency ratios for spread spectrum
    inits.append(("multi_freq", 
                  1.0 + 0.25 * np.cos(2 * np.pi * 3 * tt) +
                  0.20 * np.cos(2 * np.pi * 7 * tt + 0.5) +
                  0.15 * np.cos(2 * np.pi * 13 * tt + 1.0) +
                  0.10 * np.cos(2 * np.pi * 21 * tt + 1.5) +
                  0.08 * np.cos(2 * np.pi * 33 * tt + 2.0)))
    
    # Quadratic (parabolic) shape - minimizes center lag by AM-GM
    inits.append(("parabola", np.maximum(1.0 - 4.0 * (tt - 0.5) ** 2, 0.01)))
    
    # Sine-squared (always non-negative, good spectral properties)
    inits.append(("sin2", 0.5 + 0.4 * np.sin(np.pi * tt) ** 2 + 0.1 * np.cos(2 * np.pi * 5 * tt)))
    
    # Gaussian bump (concentrated in middle)
    inits.append(("gauss", np.exp(-8.0 * (tt - 0.5) ** 2) + 0.05))
    
    # Random smooth (low-pass filtered noise)
    noise = rng.standard_normal(n)
    from scipy.ndimage import gaussian_filter1d
    smooth_noise = gaussian_filter1d(noise, sigma=10)
    inits.append(("smooth_rand", 1.0 + 0.3 * smooth_noise / (np.std(smooth_noise) + 1e-10)))
    
    n_inits = len(inits)
    print("=== Optimization: n=%d, %d inits, budget=%.0fs ===" % (n, n_inits, budget_s))
    print("Target: 1.5027, Previous best: 1.508805")
    
    # Initialize from given sequence first if available
    if inits and inits[0][0] == "given":
        a0 = inits[0][1]
        c0 = fft_convolve_self(a0)
        global_best = c0.max()
        global_a = a0.copy()
        print("Initial (given): max_conv=%.6f, score=%.6f" % (global_best, 2 * n * global_best / S ** 2))
    
    # === Phase 1: Spectral equalization on best init ===
    usable = deadline - time.monotonic()
    
    # First, try spectral equalization on the best initial
    if global_a is not None:
        t0 = time.monotonic()
        a_spec, score_spec = spectral_refine_interleaved(
            global_a, n, S, min(usable * 0.3, 300), rng)
        a_spec = safe_normalize(a_spec, n)
        c_spec = fft_convolve_self(a_spec)
        if c_spec.max() < global_best:
            global_best = c_spec.max()
            global_a = a_spec.copy()
            print("  Spectral on given: %.8f (score=%.6f, %.1fs)" %
                  (global_best, 2 * n * global_best / S ** 2, time.monotonic() - t0))
        else:
            print("  Spectral on given: %.8f (no improvement, %.1fs)" %
                  (c_spec.max(), time.monotonic() - t0))
    
    # === Phase 2: Try other initializations with spectral + L-BFGS ===
    remaining = deadline - time.monotonic()
    other_inits = inits[1:] if (inits and inits[0][0] == "given") else inits
    
    per_init = max(15.0, remaining * 0.4 / len(other_inits))
    print("\n=== Phase 2: Other initializations (%.1fs each) ===" % per_init)
    
    for name, arr in other_inits:
        if time.monotonic() > deadline - 120:
            break
        t0 = time.monotonic()
        a_norm = safe_normalize(arr, n)
        c_init = fft_convolve_self(a_norm)
        init_score = c_init.max()
        
        if init_score < global_best:
            global_best = init_score
            global_a = a_norm.copy()
        
        # Spectral equalization
        a_spec = spectral_equalize(a_norm, gamma=0.3, n_iter=40, S=S)
        a_spec = safe_normalize(a_spec, n)
        c_spec = fft_convolve_self(a_spec)
        spec_score = c_spec.max()
        
        # L-BFGS polish
        a_lb, score_lb = run_lbfgs(a_spec, n, S, per_init * 0.5, rng,
                                   beta_schedule=[20, 50, 100, 300, 800],
                                   maxiter_per_beta=100)
        a_lb = safe_normalize(a_lb, n)
        c_lb = fft_convolve_self(a_lb)
        lb_score = c_lb.max()
        
        best_local = min(spec_score, lb_score)
        best_local_a = a_spec if spec_score < lb_score else a_lb
        
        if best_local < global_best:
            global_best = best_local
            global_a = best_local_a.copy()
            print("  [OK] %s: %.8f (init=%.4f, spec=%.4f, lbfgs=%.4f, %.1fs)" %
                  (name, global_best, init_score, spec_score, lb_score, time.monotonic() - t0))
        else:
            print("  %s: best=%.6f (no improvement, %.1fs)" %
                  (name, best_local, time.monotonic() - t0))
    
    print("\n=== After Phase 2: global_best=%.8f (score=%.6f) ===" %
          (global_best, 2 * n * global_best / S ** 2))
    
    # === Phase 3: Aggressive spectral equalization on global best ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("\n=== Phase 3: Deep spectral refinement (%.1fs) ===" % remaining)
            
            # Multiple rounds of spectral equalization with increasing gamma
            a_work = global_a.copy()
            for round_idx in range(3):
                if time.monotonic() > deadline - 60:
                    break
                gamma = 0.3 + 0.1 * round_idx
                a_work = spectral_equalize(a_work, gamma=gamma, n_iter=50, S=S)
                a_work = safe_normalize(a_work, n)
                c = fft_convolve_self(a_work)
                if c.max() < global_best:
                    global_best = c.max()
                    global_a = a_work.copy()
                    print("  Round %d (gamma=%.2f): %.8f" % (round_idx, gamma, global_best))
            
            # Subgradient attack
            if time.monotonic() < deadline - 30:
                a_sg = subgradient_step(global_a, S, n_steps=500)
                a_sg = safe_normalize(a_sg, n)
                c = fft_convolve_self(a_sg)
                if c.max() < global_best:
                    global_best = c.max()
                    global_a = a_sg.copy()
                    print("  Subgradient: %.8f" % global_best)
    
    # === Phase 4: Final L-BFGS polish ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print("\n=== Phase 4: Final L-BFGS (%.1fs) ===" % remaining)
            a_final, score_final = run_lbfgs(
                global_a, n, S, remaining, rng,
                beta_schedule=[100, 300, 500, 1000, 2000, 5000, 10000],
                maxiter_per_beta=200)
            a_final = safe_normalize(a_final, n)
            c_final = fft_convolve_self(a_final)
            if c_final.max() < global_best:
                global_best = c_final.max()
                global_a = a_final.copy()
                print("  Final L-BFGS: %.8f" % global_best)
    
    # === Phase 5: Fine subgradient + spectral micro-adjustments ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print("\n=== Phase 5: Micro-refinement (%.1fs) ===" % remaining)
            
            # Very gentle spectral equalization
            a_micro = spectral_equalize(global_a, gamma=0.1, n_iter=30, S=S)
            a_micro = safe_normalize(a_micro, n)
            c = fft_convolve_self(a_micro)
            if c.max() < global_best:
                global_best = c.max()
                global_a = a_micro.copy()
                print("  Micro spectral: %.8f" % global_best)
            
            # Short L-BFGS at very high beta
            remaining2 = deadline - time.monotonic()
            if remaining2 > 5:
                a_fin, _ = run_lbfgs(global_a, n, S, remaining2, rng,
                                     beta_schedule=[2000, 5000, 10000],
                                     maxiter_per_beta=150)
                a_fin = safe_normalize(a_fin, n)
                c = fft_convolve_self(a_fin)
                if c.max() < global_best:
                    global_best = c.max()
                    global_a = a_fin.copy()
                    print("  Final micro L-BFGS: %.8f" % global_best)
    
    # === Verification ===
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: score = %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  Fallback to uniform")
            global_a = np.ones(256, dtype=np.float64)
        
        final_score = 2 * len(global_a) * fft_convolve_self(global_a).max() / (global_a.sum() ** 2)
        print("Final: n=%d, max_conv=%.6f, score=%.6f" %
              (len(global_a), fft_convolve_self(global_a).max(), final_score))
    else:
        global_a = np.ones(256, dtype=np.float64)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
    
    print("Time used: %.1fs" % (time.monotonic() - (deadline - budget_s + 15)))
    return [float(x) for x in global_a]
