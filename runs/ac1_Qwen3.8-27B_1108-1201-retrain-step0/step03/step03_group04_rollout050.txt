            w = softmax_w(c, beta)
        else:
            top_k = min(8, 1 + int(beta / 100))
            w = toppk_w(c, top_k)

        grad = grad_fft(y, w)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Backtracking line search
        alpha = step
        accepted = False
        for _ in range(5):
            x_try = simplex_proj(y - alpha * grad / gn, S)
            c_try = ff_conv(x_try)
            if c_try.max() < cur_max * 1.001:  # allow tiny increase
                y_new = x_try
                accepted = True
                break
            alpha *= 0.5

        if not accepted:
            # Small random perturbation to escape
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y_new = simplex_proj(y + noise, S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = y_new + ((t_mom - 1.0) / t_new) * (y_new - y)
        y = np.maximum(y, 0.0)
        # Re-project to maintain sum
        y = simplex_proj(y, S)
        t_mom = t_new

        # Update step size
        if accepted and alpha > step:
            step = min(step * 1.05, S / n)
        elif stall > 500:
            step *= 0.7

        # Periodic jump
        if stall > 8000 and rng.random() < 0.1:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            y = simplex_proj(best_a + noise, S)
            t_mom = 1.0
            stall = 0

    return best_a, best_c


def run_topk_refine(a0, n, S, max_time, rng, k_start=1, k_end=6):
    """Focused top-k refinement with decreasing step size."""
    a = normalize_seq(a0, n).copy()
    start = time.monotonic()
    best_c = ff_conv(a).max()
    best_a = a.copy()
    it = 0
    step = S / (n * 3.0)

    while time.monotonic() - start < max_time:
        it += 1
        k = min(k_end, k_start + it // 3000)

        c = ff_conv(a)
        cur = c.max()
        if cur < best_c - 1e-13:
            best_c = cur
            best_a = a.copy()

        w = toppk_w(c, k)
        grad = grad_fft(a, w)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        # Decreasing step
        alpha = step / (1.0 + 0.001 * it)
        x_new = simplex_proj(a - alpha * grad / gn, S)
        c_new = ff_conv(x_new)

        if c_new.max() < cur:
            a = x_new
        else:
            # Try half step
            x_half = simplex_proj(a - 0.5 * alpha * grad / gn, S)
            c_half = ff_conv(x_half)
            if c_half.max() < cur:
                a = x_half
            elif it % 100 == 0 and rng.random() < 0.3:
                a = simplex_proj(a + rng.normal(0, 0.0005 * S / np.sqrt(n), n), S)

    return best_a, best_c


def spectral_perturb(a, rng, strength=0.02):
    """Generate a structured perturbation in the frequency domain."""
    n = len(a)
    fsize = 1 << (n - 1).bit_length()
    A = fft(a, fsize)
    # Perturb the magnitude in a structured way
    phase_noise = rng.normal(0, strength, fsize)
    mag_noise = 1.0 + rng.normal(0, strength * 0.5, fsize)
    A_pert = A * mag_noise * np.exp(1j * phase_noise)
    a_new = np.real(ifft(A_pert, n))
    return np.maximum(a_new, 0.0)


def run_multi_restart(a_best, n, S, max_time, rng, n_restarts=4):
    """Try multiple diverse restarts from the best solution."""
    best_a = a_best.copy()
    best_c = ff_conv(a_best).max()
    per_restart = max_time / n_restarts

    for r in range(n_restarts):
        if time.monotonic() > time.monotonic() + 0:
            pass
        remaining = max_time - (time.monotonic() - (time.monotonic() - max_time + per_restart * r))
        # Choose perturbation strategy
        if r == 0:
            perturbed = simplex_proj(a_best + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
        elif r == 1:
            perturbed = spectral_perturb(a_best, rng, 0.03)
            perturbed = simplex_proj(perturbed, S)
        elif r == 2:
            # Smooth perturbation
            kernel = np.exp(-np.arange(-32, 33) ** 2 / 64.0)
            kernel /= kernel.sum()
            noise = np.convolve(rng.normal(0, 1, n), kernel, mode='same')
            perturbed = simplex_proj(a_best + 0.004 * S / np.sqrt(n) * noise, S)
        else:
            # Mirror + perturb
            mirrored = a_best[::-1]
            mix = 0.7 * a_best + 0.3 * mirrored
            perturbed = simplex_proj(mix + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)

        # Short optimization from perturbed start
        a_r, c_r = run_adaptive_subgrad(perturbed, n, S, per_restart * 0.7, rng,
                                         beta_start=50.0, beta_end=200.0)
        if c_r < best_c:
            best_c = c_r
            best_a = a_r.copy()
        # Quick polish
        a_r2, c_r2 = run_topk_refine(a_r, n, S, per_restart * 0.3, rng, k_start=1, k_end=4)
        if c_r2 < best_c:
            best_c = c_r2
            best_a = a_r2.copy()

    return best_a, best_c


def propose_candidate(seed=42, budget_s=2385, **kwargs):
    """
    Multi-phase optimization: adaptive subgradient → top-k refine →
    multi-restart → ultra-fine polish. Uses backtracking, adaptive
    temperature, and spectral perturbation for diversity.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s - 25.0)

    # Get starting sequence
    given_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given_a = np.clip(given, 0, 1000)
    except Exception:
        pass

    if given_a is not None:
        n = len(given_a)
    else:
        n = 8192
        given_a = np.ones(n) * 0.5

    S = np.sqrt(2.0 * float(n))
    a_start = normalize_seq(given_a, n)
    c_start = ff_conv(a_start)
    initial_score = c_start.max()

    global_best_a = a_start.copy()
    global_best_c = initial_score
    obj = 2.0 * n * global_best_c / (S * S)
    print("n=%d, initial max_conv=%.6f, objective=%.6f" % (n, initial_score, obj))

    # === Phase 1: Adaptive subgradient (40% of budget) ===
    p1_time = 0.40 * (deadline - time.monotonic())
    if p1_time > 10:
        print("Phase 1: Adaptive subgradient (%.0fs)" % p1_time)
        t0 = time.monotonic()
        a1, c1 = run_adaptive_subgrad(a_start, n, S, p1_time, rng,
                                       beta_start=25.0, beta_end=500.0)
        if c1 < global_best_c:
            global_best_c = c1
            global_best_a = a1.copy()
            obj = 2.0 * n * c1 / (S * S)
            print("  P1: max_conv=%.6f, obj=%.6f [%.0fs]" % (c1, obj, time.monotonic() - t0))

    # === Phase 2: Continue from best with different params (20%) ===
    remaining = deadline - time.monotonic()
    p2_time = 0.20 * remaining
    if p2_time > 10:
        print("Phase 2: Refined subgradient (%.0fs)" % p2_time)
        t0 = time.monotonic()
        # Restart from best with lower beta range for broader exploration
        a2, c2 = run_adaptive_subgrad(global_best_a, n, S, p2_time, rng,
                                       beta_start=15.0, beta_end=200.0)
        if c2 < global_best_c:
            global_best_c = c2
            global_best_a = a2.copy()
            obj = 2.0 * n * c2 / (S * S)
            print("  P2: max_conv=%.6f, obj=%.6f [%.0fs]" % (c2, obj, time.monotonic() - t0))

    # === Phase 3: Top-k focused refinement (15%) ===
    remaining = deadline - time.monotonic()
    p3_time = 0.15 * remaining
    if p3_time > 5:
        print("Phase 3: Top-k refinement (%.0fs)" % p3_time)
        t0 = time.monotonic()
        a3, c3 = run_topk_refine(global_best_a, n, S, p3_time, rng, k_start=1, k_end=6)
        if c3 < global_best_c:
            global_best_c = c3
            global_best_a = a3.copy()
            obj = 2.0 * n * c3 / (S * S)
            print("  P3: max_conv=%.6f, obj=%.6f [%.0fs]" % (c3, obj, time.monotonic() - t0))

    # === Phase 4: Multi-restart exploration (15%) ===
    remaining = deadline - time.monotonic()
    p4_time = 0.15 * remaining
    if p4_time > 10:
        print("Phase 4: Multi-restart (%.0fs)" % p4_time)
        t0 = time.monotonic()
        a4, c4 = run_multi_restart(global_best_a, n, S, p4_time, rng, n_restarts=4)
        if c4 < global_best_c:
            global_best_c = c4
            global_best_a = a4.copy()
            obj = 2.0 * n * c4 / (S * S)
            print("  P4: max_conv=%.6f, obj=%.6f [%.0fs]" % (c4, obj, time.monotonic() - t0))

    # === Phase 5: Final ultra-fine polish (10%) ===
    remaining = deadline - time.monotonic()
    if remaining > 3:
        print("Phase 5: Final polish (%.0fs)" % remaining)
        t0 = time.monotonic()
        a5, c5 = run_topk_refine(global_best_a, n, S, remaining, rng, k_start=1, k_end=3)
        if c5 < global_best_c:
            global_best_c = c5
            global_best_a = a5.copy()
            obj = 2.0 * n * c5 / (S * S)
            print("  P5: max_conv=%.6f, obj=%.6f [%.0fs]" % (c5, obj, time.monotonic() - t0))

    # Final output
    final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
    final_obj = 2.0 * n * global_best_c / (S * S)
    print("FINAL: n=%d, max_conv=%.6f, objective=%.6f" % (len(final_list), global_best_c, final_obj))

    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: %.6f" % ver)
    else:
        print("Verification failed! Using fallback.")
        final_list = [0.5] * 256

    return final_list
