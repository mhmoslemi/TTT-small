"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses multi-phase FISTA with adaptive LSE smoothing, asymmetric initializations,
top-k peak targeting, and backtracking line search. Key insight: the convolution
center c[n-1] = sum a[i]*a[n-1-i] can be made small via asymmetry, allowing
the max to approach the theoretical lower bound of 2n/(2n-1) ~ 1.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def lse_grad(a, c, beta):
    """Gradient of LSE approximation of max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted -= shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return np.real(ifft(W * np.conj(A)))[:n]


def topk_grad(a, c, K, power=1.0):
    """Gradient targeting top-K peaks with power weighting."""
    n = len(a)
    K = min(K, len(c))
    idx = np.argpartition(c, -K)[-K:]
    vals = c[idx]
    w = np.power(np.maximum(vals, 1e-30), power)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return np.real(ifft(W * np.conj(A)))[:n]


def exact_subgrad(a, c):
    """Exact subgradient at argmax lag (one-hot)."""
    n = len(a)
    k = int(np.argmax(c))
    # c[k] = sum_i a[i]*a[k-i], grad[j] = a[k-j] + a[j-k] (with zero-padding)
    grad = np.zeros(n)
    # grad[j] = a[k-j] if 0<=k-j<n, plus a[j-k] if 0<=j-k<n
    for j in range(n):
        g = 0.0
        if 0 <= k - j < n:
            g += a[k - j]
        if 0 <= j - k < n:
            g += a[j - k]
        grad[j] = g
    return grad


def fista_optimize(a_init, n, S, max_time, rng, mode="adaptive",
                   beta_init=10.0, beta_rate=0.003, beta_max=2000.0,
                   k_init=2, k_max=128, backtracking=True):
    """
    FISTA with adaptive gradient selection.
    mode: 'lse', 'topk', 'adaptive' (starts topk, transitions to lse)
    """
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = compute_obj(a)
    best_a = a.copy()
    it = 0
    last_improve = 0
    c = fft_convolve_self(y)

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive K for topk
        K = min(k_max, k_init + it // 200)

        # Choose gradient
        if mode == "adaptive":
            if it < 200:
                grad = topk_grad(y, c, min(K, 4), power=1.0)
            elif it < 1000:
                grad = topk_grad(y, c, K, power=1.5)
            else:
                beta = min(beta_max, beta_init * np.exp(beta_rate * it))
                grad = lse_grad(y, c, beta)
        elif mode == "topk":
            grad = topk_grad(y, c, K, power=1.5)
        else:
            beta = min(beta_max, beta_init * np.exp(beta_rate * it))
            grad = lse_grad(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        grad = grad / gnorm

        # Step size with backtracking
        if backtracking:
            step = S / (n * 0.5)
            # Try to take a step, check if objective improved
            for _ in range(10):
                x_new = project_simplex(y - step * grad, S)
                c_new = fft_convolve_self(x_new)
                obj_new = 2.0 * n * c_new.max() / S**2
                obj_cur = 2.0 * n * c.max() / S**2
                if obj_new <= obj_cur + 1e-10:
                    c = c_new
                    break
                step *= 0.5
                x_new = project_simplex(y - step * grad, S)
                c_new = fft_convolve_self(x_new)
        else:
            step = S / (n * (1.0 + 0.01 * np.log1p(it)))
            x_new = project_simplex(y - step * grad, S)
            c = fft_convolve_self(x_new)

        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-project to maintain feasibility
        y = project_simplex(y, S)
        a = x_new
        t = t_new

        # Track best
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / S**2
        if cur_obj < best - 1e-10:
            best = cur_obj
            best_a = a.copy()
            last_improve = it

        # Stuck detection: inject noise
        if it - last_improve > 500:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it

        # Periodic restart with smaller noise
        if it % 2000 == 0:
            decay = np.exp(-it / 15000.0)
            noise_scale = 0.003 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    return best_a, best


def generate_inits(n, rng, given=None):
    """Generate diverse initializations emphasizing asymmetry."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Asymmetric ramps (key for reducing center convolution)
    inits.append(tt + 0.1)
    inits.append(tt**2 + 0.1)
    inits.append(np.exp(3.0 * tt) * 0.1 + 0.1)
    inits.append(1.0 - tt + 0.1)
    inits.append(1.0 / (tt + 0.5))

    # Window functions (symmetric, for comparison)
    inits.append(0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.05)
    inits.append(np.sin(np.pi * tt)**2 + 0.05)
    inits.append(np.sin(np.pi * tt)**4 + 0.05)
    inits.append(np.exp(-4.0 * (tt - 0.5)**2) + 0.05)

    # Multi-sinusoidal with phase
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt + 0.7) + 0.15 * np.cos(2 * np.pi * 7 * tt))
    inits.append(1.0 + 0.25 * np.sin(2 * np.pi * 2 * tt) * np.cos(2 * np.pi * 5 * tt + 1.0))

    # Random structures
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.4, n)) * 0.5 + 0.1)

    # Block structures (asymmetric)
    for nblock in [3, 5]:
        block_len = n // nblock
        heights = np.linspace(0.3, 1.5, nblock)
        if rng.random() < 0.5:
            heights = heights[::-1]
        block_seq = np.repeat(heights, block_len)
        if len(block_seq) < n:
            block_seq = np.concatenate([block_seq, np.full(n - len(block_seq), heights[-1])])
        inits.append(block_seq[:n])

    # Triangle (pointing right)
    tri = np.minimum(tt, 1.0 - tt) * 2.0
    inits.append(tri + 0.1)

    # W-shape
    w_shape = 1.0 - 0.4 * np.cos(4 * np.pi * tt) + 0.2 * tt
    inits.append(w_shape + 0.1)

    # Given sequence if available
    if given is not None:
        g = np.clip(np.asarray(given, dtype=np.float64).reshape(-1), 0.0, 1000.0)
        if len(g) == n and g.sum() > 0:
            inits.insert(0, g)

    return [np.maximum(x, 0.01) for x in inits]


def polish(a_init, n, S, max_time, rng):
    """Multi-strategy polishing."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    start = time.monotonic()
    best = compute_obj(a)
    best_a = a.copy()

    # Strategy 1: High-beta LSE FISTA
    seg1 = max(2.0, (max_time - (time.monotonic() - start)) * 0.4)
    a1, s1 = fista_optimize(a, n, S, seg1, rng, mode="lse",
                            beta_init=50.0, beta_rate=0.002, beta_max=3000.0,
                            backtracking=False)
    c1 = fft_convolve_self(a1)
    obj1 = 2.0 * n * c1.max() / S**2
    if obj1 < best:
        best = obj1
        best_a = a1.copy()

    # Strategy 2: Top-k with growing K
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        seg2 = max(2.0, remaining * 0.35)
        a2, s2 = fista_optimize(best_a, n, S, seg2, rng, mode="topk",
                                k_init=8, k_max=256, backtracking=False)
        c2 = fft_convolve_self(a2)
        obj2 = 2.0 * n * c2.max() / S**2
        if obj2 < best:
            best = obj2
            best_a = a2.copy()

    # Strategy 3: Very fine adaptive with backtracking
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        seg3 = max(2.0, remaining * 0.3)
        a3, s3 = fista_optimize(best_a, n, S, seg3, rng, mode="adaptive",
                                beta_init=100.0, beta_rate=0.001, beta_max=5000.0,
                                k_init=4, k_max=64, backtracking=True)
        c3 = fft_convolve_self(a3)
        obj3 = 2.0 * n * c3.max() / S**2
        if obj3 < best:
            best = obj3
            best_a = a3.copy()

    return best_a, best
