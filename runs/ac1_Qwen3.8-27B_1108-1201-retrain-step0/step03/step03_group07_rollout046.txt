"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive temperature soft-max gradient
descent with diverse restarts and structural perturbations.
Key innovations over previous approach:
- Principled log-sum-exp temperature annealing (broad -> narrow peak focus)
- Exact gradient via FFT convolution of weight vector
- Targeted structural perturbations (sinusoidal, spectral) for restarts
- Multi-phase: explore diverse basins, then focused exploitation, then fine polish
- Adaptive step size with aggressive backtracking
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def compute_gradient(c, a, temperature):
    """
    Compute gradient of soft_max(c, T) w.r.t. a.
    soft_max = (1/T) * log(sum exp(T*c))
    d(soft_max)/d(a[m]) = sum_k p_k * 2*a[k-m]
    where p_k = exp(T*(c_k - c_max)) / sum_j exp(T*(c_j - c_max))
    """
    n = len(a)
    c_max = c.max()

    if temperature < 1e-10:
        # Near-hard max: find all peaks within epsilon of max
        threshold = c_max - max(c_max * 1e-6, c_max * 0.001)
        mask = c >= threshold
        if mask.sum() == 0:
            mask[np.argmax(c)] = True
        w = np.zeros(len(c))
        w[mask] = 1.0 / mask.sum()
    else:
        exp_c = np.exp(temperature * (c - c_max))
        s = exp_c.sum()
        if s < 1e-300:
            idx = np.argmax(c)
            w = np.zeros(len(c))
            w[idx] = 1.0
        else:
            w = exp_c / s

    # grad[m] = 2 * sum_k w[k] * a[k-m] (valid indices only)
    # = 2 * conv(w, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    return 2.0 * full_conv[n - 1:2 * n - 1]


def optimize_sequence(a0, time_budget, T_start=50.0, T_end=500.0,
                      eta0=0.3, momentum_start=0.8, momentum_end=0.95,
                      perturb_interval=1500, verbose=True, name=""):
    """
    Adaptive temperature mirror descent optimization.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)
    eta = eta0
    iter_no = 0
    start = time.monotonic()
    momentum = momentum_start

    while time.monotonic() - start < time_budget:
        elapsed_frac = min(1.0, (time.monotonic() - start) / time_budget)

        # Adaptive temperature: start broad, end narrow
        T_cur = T_start + (T_end - T_start) * (elapsed_frac ** 0.5)

        # Ramp momentum
        mom_cur = momentum_start + (momentum_end - momentum_start) * elapsed_frac

        # Compute gradient
        grad = compute_gradient(c, a, T_cur)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Momentum update
        v = mom_cur * v + (1.0 - mom_cur) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        # Backtracking line search (multiplicative updates)
        current_obj = obj_value(a, c)
        accepted = False
        for bt in range(10):
            eta_try = eta * (0.7 ** bt)
            log_update = -eta_try * v
            # Clamp to avoid underflow
            log_update = np.clip(log_update, -50, 50)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial * (S / a_trial.sum())

            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)

            if obj_trial <= current_obj * (1.0 + 1e-12):
                a = a_trial
                c = c_trial
                accepted = True
                break

        if not accepted:
            # Very small step
            log_update = -eta * 0.05 * v
            log_update = np.clip(log_update, -50, 50)
            a = a * np.exp(log_update)
            a = np.maximum(a, 1e-300)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)

        cur_obj = obj_value(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
            eta = min(eta * 1.03, eta0 * 3.0)
        else:
            eta = max(eta * 0.92, eta0 * 0.05)

        # Stagnation: structural perturbation
        if iter_no - best_iter > perturb_interval:
            pert_type = iter_no % 4
            if pert_type == 0:
                # Sinusoidal perturbation
                freq = np.random.randint(2, max(3, n // 100))
                phase = np.random.uniform(0, 2 * np.pi)
                i = np.arange(n)
                pert = 0.1 * np.cos(2 * np.pi * freq * i / n + phase)
                a = a + pert * (S / n)
            elif pert_type == 1:
                # Gaussian noise
                noise = np.random.randn(n) * (S / np.sqrt(n)) * 0.05
                a = a + noise
            elif pert_type == 2:
                # Spectral perturbation: modify random frequency bins
                A = fft(a)
                idx = np.random.choice(n, size=max(1, n // 50), replace=False)
                A[idx] *= (1 + 0.1 * np.random.randn(len(idx)))
                a = np.real(ifft(A))
            else:
                # Local bump
                center = np.random.randint(0, n)
                width = np.random.randint(n // 20, n // 5)
                i = np.arange(n)
                bump = 0.05 * (S / n) * np.exp(-0.5 * ((i - center) / width) ** 2)
                a = a + bump

            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.2
            eta = eta0
            best_iter = iter_no

            if verbose:
                elapsed = time.monotonic() - start
                print(f"    [{name}] perturb iter={iter_no}, obj={cur_obj:.6f}, "
                      f"best={best_obj:.6f}, t={elapsed:.1f}s")

        # Slow eta decay
        eta *= 0.99998

        iter_no += 1
        if verbose and iter_no % 3000 == 0:
            elapsed = time.monotonic() - start
            print(f"    [{name}] iter={iter_no}, obj={cur_obj:.6f}, "
                  f"best={best_obj:.6f}, T={T_cur:.0f}, eta={eta:.4f}, t={elapsed:.1f}s")

    return best_a, best_obj


def fine_polish(a0, time_budget, verbose=True):
    """
    Fine polishing: focus on the single worst peak with very small steps.
    Alternates between reducing the worst peak and checking for new worst peaks.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    v = np.zeros(n)
    eta = 0.05

    while time.monotonic() - start < time_budget:
        # Find the worst peak
        k_max = int(np.argmax(c))
        c_max_val = c[k_max]

        # Compute gradient for just this peak
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo <= hi:
            m_vals = np.arange(lo, hi + 1)
            grad[m_vals] = 2.0 * a[k_max - m_vals]

        # Also check nearby peaks (within 10% of max)
        threshold = c_max_val * 0.9
        nearby = np.where(c >= threshold)[0]
        for k in nearby[:10]:  # Limit to top 10 nearby peaks
            if k == k_max:
                continue
            lo_k = max(0, k - n + 1)
            hi_k = min(n - 1, k)
            if lo_k <= hi_k:
                m_vals = np.arange(lo_k, hi_k + 1)
                grad[m_vals] += 2.0 * a[k - m_vals] * 0.5

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Momentum
        v = 0.9 * v + 0.1 * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        # Small multiplicative step with backtracking
        current_obj = obj_value(a, c)
        for bt in range(8):
            eta_try = eta * (0.6 ** bt)
            log_update = np.clip(-eta_try * v, -20, 20)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)
            if obj_trial <= current_obj:
                a = a_trial
                c = c_trial
                break
        else:
            break  # No improvement even with tiny step

        cur_obj = obj_value(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            eta = min(eta * 1.05, 0.3)
        else:
            eta = max(eta * 0.8, 0.005)

    return best_a, best_obj


def spectral_initialization(n, rng, num_harmonics=None):
    """
    Create a non-negative sequence with approximately flat power spectrum.
    Uses random phases in frequency domain.
    """
    if num_harmonics is None:
        num_harmonics = n
    A = np.zeros(n, dtype=np.complex128)
    # DC component
    A[0] = rng.uniform(1, 3)
    # Random phases for other bins
    for k in range(1, num_harmonics):
        mag = rng.uniform(0.5, 1.5)
        phase = rng.uniform(0, 2 * np.pi)
        A[k] = mag * np.exp(1j * phase)
        A[n - k] = np.conj(A[k])
    a = np.real(ifft(A))
    a = np.abs(a)  # Make non-negative
    a = np.maximum(a, 1e-6)
    S = np.sqrt(2.0 * n)
    a = a * S / a.sum()
    return a


def cosine_initialization(n, freq, amplitude, rng):
    """Create a cosine-modulated initialization."""
    i = np.arange(n)
    a = 1.0 + amplitude * np.cos(2 * np.pi * freq * i / n + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.01)
    S = np.sqrt(2.0 * n)
    a = a * S / a.sum()
    return a
