import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_topk_gradient(c, a, K=1, p=2.0):
    """Compute weighted gradient targeting top-K peaks.
    
    Returns gradient vector g where g[j] = 2 * sum_i w_i * a[m_i - j]
    (with a extended as 0 outside [0, n-1]).
    """
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    
    # Find top-K indices
    topk_idx = np.argpartition(c, -K)[-K:]
    topk_idx = topk_idx[np.argsort(c[topk_idx])[::-1]]  # sort descending
    
    # Compute weights
    topk_vals = c[topk_idx]
    weights = (topk_vals / topk_vals[0]) ** p
    
    # Compute gradient: g[j] = 2 * sum_i w_i * a[m_i - j]
    # where a[m_i - j] is 0 if m_i - j is outside [0, n-1]
    g = np.zeros(n)
    for i, m in enumerate(topk_idx):
        if i == 0:
            w = weights[i]
        else:
            w = weights[i]
        # g[j] += 2 * w * a[m - j] for j in [max(0, m-n+1), min(n-1, m)]
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            # a[m-j] for j in [lo, hi] → indices in a: [m-hi, m-lo]
            idx = m - hi + (np.arange(hi - lo + 1))
            g[lo:hi+1] += 2.0 * w * a[np.clip(idx, 0, n-1)]
    
    return g


def compute_topk_gradient_fft(c, a, K=1, p=2.0):
    """Compute weighted gradient using FFT for efficiency."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    
    # Find top-K indices
    topk_idx = np.argpartition(c, -K)[-K:]
    topk_idx = topk_idx[np.argsort(c[topk_idx])[::-1]]
    
    # Compute weights
    topk_vals = c[topk_idx]
    weights = (topk_vals / topk_vals[0]) ** p
    
    # Build the weight vector
    w = np.zeros(2 * n - 1)
    for i, m in enumerate(topk_idx):
        w[m] = weights[i]
    
    # g[j] = 2 * sum_m w[m] * a[m-j] (with a=0 outside [0,n-1])
    # This is 2 * (w * a)_j where * denotes cross-correlation
    # Using FFT: cross_corr(w, a)[j] = sum_m w[m] * a[m-j]
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def run_adam_topk(a_init, n, S, time_budget, K_start=1, K_end=50,
                  p=2.0, lr=0.05, beta1=0.9, beta2=0.999,
                  warmup_steps=100, verbose=False):
    """Adam optimization with top-K peak targeting."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Adaptive K
        K = int(K_start + (K_end - K_start) * progress ** 0.5)
        K = max(1, K)
        
        # Compute autocorrelation
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Compute gradient
        g = compute_topk_gradient_fft(c, a, K=K, p=p)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        # Learning rate with warmup
        if t < warmup_steps:
            lr_t = lr * t / warmup_steps
        else:
            lr_t = lr * (1.0 - 0.5 * progress)
        
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = normalize_nonneg(a, S)
        
        if verbose and t % 10000 == 0:
            print("    Adam-TK step %d: obj=%.7f, K=%d" % (t, best_obj, K))
    
    return best_a, best_obj


def run_softmax_adam(a_init, n, S, time_budget, T_start=10, T_end=2000,
                     lr=0.03, beta1=0.9, beta2=0.999, verbose=False):
    """Adam with softmax smoothing (similar to previous but improved)."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())
    
    n_par = n
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    eps = 1e-8
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** (progress ** 0.6)
        
        # Softmax weights
        c_max = c.max()
        if c_max < 1e-30:
            break
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        
        # Gradient
        fsize = 1 << (3 * n_par - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g_a = 2.0 * corr[:n_par]
        
        # Chain rule for log-params
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot
        
        # Adam
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr_t = lr * (1.0 - 0.5 * progress)
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n_par, S / n_par)
        else:
            a = S * exp_theta / total
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        if verbose and t % 10000 == 0:
            print("    Softmax step %d: obj=%.7f, T=%.1f" % (t, best_obj, T))
    
    return best_a, best_obj


def coordinate_descent(a, n, S, time_budget, max_sweeps=100, verbose=False):
    """Coordinate descent with adaptive factor selection."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(42)
    factors = [0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5, 0.3, 1.0/0.3]
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                    break
            
            original = a[i]
            if original < 1e-10:
                continue
            
            best_local = original
            best_local_obj = best_obj
            
            for factor in factors:
                a[i] = original * factor
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-13:
                    best_local_obj = obj_check
                    best_local = original * factor
            
            a[i] = best_local
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            
            if c.max() < best_obj - 1e-13:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        if not improved and sweep >= 5:
            break
    
    return best_a, best_obj


def spectral_shaping(a, n, S, time_budget, verbose=False):
    """Try to flatten the power spectrum to reduce peaks."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()
    
    # Work in log-frequency domain
    A = fft(a, 2 * n)
    # Target: make |A[k]| more uniform for k != 0
    # |A[0]| = S (fixed)
    
    # Use iterative approach: adjust a to make PSD flatter
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.02
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        
        A = fft(a, 2 * n)
        P = np.abs(A) ** 2
        
        # Target PSD: ideal would be flat, but DC is fixed
        # Use log-target: minimize variance of log(P[1:])
        # But that might be too aggressive. Instead, penalize peaks in time domain
        # using the actual objective
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Gradient of time-domain peak
        m_star = np.argmax(c)
        # g[j] = 2 * a[m_star - j] (with a=0 outside)
        g = np.zeros(n)
        lo = max(0, m_star - n + 1)
        hi = min(n - 1, m_star)
        if lo <= hi:
            idx = m_star - hi + np.arange(hi - lo + 1)
            g[lo:hi+1] = 2.0 * a[np.clip(idx, 0, n-1)]
        
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr_t = lr * (1.0 - 0.5 * progress)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = normalize_nonneg(a, S)
        
        if t % 50000 == 0 and verbose:
            print("    Spectral step %d: obj=%.7f" % (t, best_obj))
    
    return best_a, best_obj
