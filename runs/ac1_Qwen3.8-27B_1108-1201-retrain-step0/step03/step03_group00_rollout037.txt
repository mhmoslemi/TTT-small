import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    sum_a = a.sum()
    if sum_a < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (sum_a ** 2)


def lse_obj_and_grad(a, c, beta):
    """Compute LSE approximation of max(c) and its gradient w.r.t. a."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    # Guard against overflow
    e = np.exp(shifted - shifted.max())
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(n)
    lse = cm + np.log(tot) / beta
    w = e / tot
    # Gradient: d/d a_j [lse] = sum_k w_k * 2 * a_{k-j} (for valid k-j)
    # This is 2 * convolve(w, a)[:n]
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse, grad


def topk_obj_and_grad(a, c, k, power=1.0):
    """Top-k approximation: sum of top k values raised to power, normalized."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    
    if power == 1.0:
        obj = vals.sum() / k
    else:
        obj = np.mean(np.power(np.maximum(vals, 1e-30), power)) ** (1.0 / power)
    
    # Gradient weights
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        w = w / w.sum()
    
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return obj, grad


def exact_subgradient(a, c, top_frac=0.0):
    """Exact subgradient: average gradient of top (active) lags."""
    n = len(a)
    max_val = c.max()
    # Find all lags within a small tolerance of the max
    tol = max_val * top_frac if top_frac > 0 else 0.0
    active = np.where(c >= max_val - tol)[0]
    
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return c[active].mean(), grad

Wait, I need to reconsider the step size in FISTA. The issue is that the gradient of the LSE approximation has a specific scale, and I need to choose the step size correctly.

The LSE approximation: `f(a) = (1/beta) * log(sum_k exp(beta * (a*a)[k]))`

The gradient: `grad_i = sum_k w_k * 2 * (a*a)_k' ` where `(a*a)_k' = d(a*a)[k]/da_i = a[k-i] + a[i-k]` (for valid indices)

So `grad = 2 * convolve(w, a)[:n]` (approximately).

The Lipschitz constant of this gradient: `L = beta * max_k (Hessian of (a*a)_k) `... this is complex. Let me just use a practical approach: normalize the gradient and use a step size that's a fraction of the simplex "radius".

Actually, let me think about this more carefully. The function we're minimizing is `f(a) = max_k (a*a)[k]` (or its LSE approximation). This function is bounded below by 0 and the values of `(a*a)[k]` are at most ~S^2 (where S = sum(a) = sqrt(2n)).

The gradient of the LSE has norm roughly `2 * sum(w_k) * ||a|| = 2 * ||a||` (since sum w_k = 1).

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization starting from known-good solution:
1. FISTA with adaptive LSE smoothing at original size
2. Multi-scale upsample to 2x for more degrees of freedom
3. Active-set focused polishing with exact subgradient
4. Final intensive high-beta polish
Key improvements over baseline: focused multi-scale strategy, adaptive
top-k gradient, LP-inspired active set identification, and backtracking.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(y, c, beta):
    """Compute LSE gradient w.r.t. the sequence."""
    n = len(y)
    cm = c.max()
    shifted = beta * (c - cm)
    shifted = shifted - shifted.max()
    e = np.exp(shifted)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(y, c, k, power=1.0):
    """Top-k gradient with power weighting."""
    n = len(y)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    if power == 1.0:
        w = np.ones(k) / k
    else:
        w = np.power(np.maximum(vals, 1e-30), power - 1.0)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient(y, c, tol_frac=0.0):
    """Subgradient from lags within tolerance of max."""
    n = len(y)
    max_val = c.max()
    if tol_frac > 0:
        active = np.where(c >= max_val * (1.0 - tol_frac))[0]
    else:
        active = np.array([int(np.argmax(c))])
    w_full = np.zeros(len(c))
    w_full[active] = 1.0 / len(active)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(y, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, max_time, rng, beta_init=10.0, beta_rate=0.006,
              beta_max=5000.0, mode="lse", step_scale=1.0):
    """
    FISTA with adaptive LSE smoothing.
    mode: 'lse', 'hybrid', 'topk'
    Returns (best_a, best_conv_max).
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = step_scale / beta_init
    stuck_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        step = step_scale / max(beta, 1.0)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it
            stuck_count = 0
        else:
            stuck_count += 1

        # Gradient computation
        if mode == "lse":
            grad = lse_gradient(y, c, beta)
        elif mode == "hybrid":
            if stuck_count > 500 and (it % 5 == 0 or rng.random() < 0.2):
                grad = exact_subgradient(y, c, tol_frac=0.01)
            else:
                grad = lse_gradient(y, c, beta)
        elif mode == "topk":
            k = max(4, min(256, n // 8))
            if stuck_count > 1000:
                k = max(1, k // 4)
            grad = topk_gradient(y, c, k, power=1.5)
        else:
            grad = lse_gradient(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step: reduce when stuck
        if stuck_count > 3000:
            step *= 0.3
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-stuck_count / 5000.0)
            if noise_scale > 1e-6:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stuck_count = 0
                last_improve = it

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic restart
        if it % 8000 == 0:
            t_mom = 1.0
            decay = np.exp(-it / 50000.0)
            if decay > 0.05:
                noise = rng.normal(0, 0.002 * S / np.sqrt(n) * decay, n)
                y = project_simplex(y + noise, S)

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def run_active_polish(a_init, max_time, rng):
    """
    Intensive polishing using exact subgradient with adaptive top-k.
    Focuses on the active constraints.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0, 1000).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_conv = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_conv - 1e-14:
            best_conv = cur_max
            best_a = y.copy()
            last_improve = it

        stuck = it - last_improve

        # Adaptive k: start with 1, increase when stuck
        if stuck < 200:
            k = 1
            tol_frac = 0.0
        elif stuck < 1000:
            k = 3
            tol_frac = 0.005
        elif stuck < 5000:
            k = min(16, 4 + stuck // 500)
            tol_frac = 0.01
        else:
            k = min(64, 16 + stuck // 1000)
            tol_frac = 0.02

        if k == 1:
            grad = exact_subgradient(y, c, tol_frac=0.0)
        else:
            grad = topk_gradient(y, c, k, power=1.0)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Decreasing step size
        step = S / (gnorm * (1.0 + 0.005 * it))
        step = min(step, S / (0.5 * gnorm))

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise injection when stuck
        if stuck > 5000 and it % 500 == 0:
            noise_scale = max(1e-5, 0.001 * S / np.sqrt(n) * np.exp(-stuck / 10000.0))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best_conv:
        best_conv = c_final.max()
    return best_a, best_conv


def upsample(a, factor=2):
    """Upsample using cubic-like interpolation."""
    n = len(a)
    n_new = n * factor
    t_old = np.linspace(0, 1, n, endpoint=False)
    t_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    n_new = n // factor
    a_trunc = a[:n_new * factor]
    return a_trunc.reshape(n_new, factor).mean(axis=1)


def compute_obj(a):
    """Compute the actual objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)

    Search for non-negative sequence minimizing 2*n*max_conv/sum^2.
    Multi-phase: FISTA exploration, multi-scale upsample, active-set polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    global_best_obj = np.inf
    global_best_a = None

    # Get initial sequence

        a_start = np.clip(a_start, 0, 1000)
        if a_start.sum() < 0.01:
            a_start = None
    except Exception:
        a_start = None

    if a_start is None:
        n = 8192
        tt = np.linspace(0, 1, n, endpoint=False)
        a_start = 0.54 + 0.46 * np.cos(2 * np.pi * tt)

    n0 = len(a_start)
    obj0 = compute_obj(a_start)
    print("Initial: n=%d, obj=%.6f" % (n0, obj0))

    if obj0 < global_best_obj:
        global_best_obj = obj0
        global_best_a = a_start.copy()

    # === Phase 1: FISTA at original size (25% time) ===
    p1_budget = (deadline - time.monotonic()) * 0.25
    print("\n=== Phase 1: FISTA at n=%d (budget %.0fs) ===" % (n0, p1_budget))
    t1 = time.monotonic()

    # Run 2 FISTA instances with different parameters
    a1a, conv1a = run_fista(a_start, p1_budget * 0.5, rng,
                             beta_init=10.0, beta_rate=0.008,
                             beta_max=5000.0, mode="lse", step_scale=1.0)
    obj1a = compute_obj(a1a)
    print("  FISTA-lse: obj=%.6f (t=%.0fs)" % (obj1a, time.monotonic() - t1))
    if obj1a < global_best_obj:
        global_best_obj = obj1a
        global_best_a = a1a.copy()

    remaining1 = p1_budget - (time.monotonic() - t1)
    if remaining1 > 10:
        a1b, conv1b = run_fista(global_best_a, remaining1, rng,
                                 beta_init=50.0, beta_rate=0.005,
                                 beta_max=10000.0, mode="hybrid", step_scale=0.5)
        obj1b = compute_obj(a1b)
        print("  FISTA-hybrid: obj=%.6f (t=%.0fs)" % (obj1b, time.monotonic() - t1))
        if obj1b < global_best_obj:
            global_best_obj = obj1b
            global_best_a = a1b.copy()

    # === Phase 2: Upsample to 2n (30% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 150 and n0 <= 16384:
        p2_budget = remaining * 0.30
        n2 = n0 * 2
        print("\n=== Phase 2: Upsample to n=%d (budget %.0fs) ===" % (n2, p2_budget))
        t2 = time.monotonic()

        a2_init = upsample(global_best_a, factor=2)
        # Normalize
        S2 = np.sqrt(2.0 * n2)
        a2_init = a2_init * (S2 / a2_init.sum())

        a2, conv2 = run_fista(a2_init, p2_budget * 0.6, rng,
                               beta_init=5.0, beta_rate=0.006,
                               beta_max=5000.0, mode="lse", step_scale=1.0)
        obj2 = compute_obj(a2)
        print("  FISTA at 2n: obj=%.6f (t=%.0fs)" % (obj2, time.monotonic() - t2))
        if obj2 < global_best_obj:
            global_best_obj = obj2
            global_best_a = a2.copy()

        # Continue with hybrid mode
        remaining2 = p2_budget - (time.monotonic() - t2)
        if remaining2 > 10:
            a2b, conv2b = run_fista(global_best_a, remaining2, rng,
                                     beta_init=30.0, beta_rate=0.004,
                                     beta_max=8000.0, mode="hybrid", step_scale=0.7)
            obj2b = compute_obj(a2b)
            print("  Hybrid at 2n: obj=%.6f (t=%.0fs)" % (obj2b, time.monotonic() - t2))
            if obj2b < global_best_obj:
                global_best_obj = obj2b
                global_best_a = a2b.copy()

    # === Phase 3: Active-set polish (35% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        p3_budget = remaining * 0.35
        n_best = len(global_best_a)
        print("\n=== Phase 3: Active-set polish at n=%d (budget %.0fs) ===" % (n_best, p3_budget))
        t3 = time.monotonic()

        a3, conv3 = run_active_polish(global_best_a, p3_budget * 0.5, rng)
        obj3 = compute_obj(a3)
        print("  Active polish: obj=%.6f (t=%.0fs)" % (obj3, time.monotonic() - t3))
        if obj3 < global_best_obj:
            global_best_obj = obj3
            global_best_a = a3.copy()

        # Continue with topk mode
        remaining3 = p3_budget - (time.monotonic() - t3)
        if remaining3 > 10:
            a3b, conv3b = run_fista(global_best_a, remaining3, rng,
                                     beta_init=100.0, beta_rate=0.003,
                                     beta_max=20000.0, mode="topk", step_scale=0.3)
            obj3b = compute_obj(a3b)
            print("  Topk polish: obj=%.6f (t=%.0fs)" % (obj3b, time.monotonic() - t3))
            if obj3b < global_best_obj:
                global_best_obj = obj3b
                global_best_a = a3b.copy()

    # === Phase 4: Final intensive polish (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        p4_budget = remaining * 0.85
        n_best = len(global_best_a)
        print("\n=== Phase 4: Final polish (budget %.0fs) ===" % p4_budget)
        t4 = time.monotonic()

        # Very high beta FISTA
        a4, conv4 = run_fista(global_best_a, p4_budget * 0.5, rng,
                               beta_init=200.0, beta_rate=0.002,
                               beta_max=50000.0, mode="hybrid", step_scale=0.2)
        obj4 = compute_obj(a4)
        print("  High-beta: obj=%.6f (t=%.0fs)" % (obj4, time.monotonic() - t4))
        if obj4 < global_best_obj:
            global_best_obj = obj4
            global_best_a = a4.copy()

        remaining4 = p4_budget - (time.monotonic() - t4)
        if remaining4 > 5:
            a4b, conv4b = run_active_polish(global_best_a, remaining4, rng)
            obj4b = compute_obj(a4b)
            print("  Final active: obj=%.6f (t=%.0fs)" % (obj4b, time.monotonic() - t4))
            if obj4b < global_best_obj:
                global_best_obj = obj4b
                global_best_a = a4b.copy()

    # === Phase 5 (bonus): Try downsampling the best to see if it helps ===
    remaining = deadline - time.monotonic()
    if remaining > 30 and len(global_best_a) > 4096:
        n_ds = len(global_best_a) // 2
        a_ds = downsample(global_best_a, factor=2)
        S_ds = np.sqrt(2.0 * n_ds)
        a_ds = a_ds * (S_ds / a_ds.sum())
        obj_ds = compute_obj(a_ds)
        print("\n=== Phase 5: Downsampled n=%d: obj=%.6f ===" % (n_ds, obj_ds))
        if obj_ds < global_best_obj:
            global_best_obj = obj_ds
            global_best_a = a_ds.copy()
            # Polish the downsampled version
            a_ds2, _ = run_active_polish(a_ds, remaining * 0.7, rng)
            obj_ds2 = compute_obj(a_ds2)
            if obj_ds2 < global_best_obj:
                global_best_obj = obj_ds2
                global_best_a = a_ds2.copy()
                print("  After polish: obj=%.6f" % obj_ds2)

    # Final verification
    if global_best_a is not None:
        final_list = [float(x) for x in np.clip(global_best_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nFinal verified objective: %.6f (n=%d)" % (ver, len(final_list)))

            global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
            final_list = [float(x) for x in global_best_a]

        n_fb = 8192
        global_best_a = np.ones(n_fb) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_best_a]

    return final_list
