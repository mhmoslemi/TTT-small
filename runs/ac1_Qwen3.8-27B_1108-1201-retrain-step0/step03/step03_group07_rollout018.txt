import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def lp_direction(a, M, n_active=500, solver='ECOS'):
    """Solve LP to find best direction g."""
    n = len(a)
    c = fft_convolve_self(a)
    
    # Find active constraints
    thr = M * 0.99  # 99% of max
    active_k = np.where(c >= thr)[0]
    if len(active_k) > n_active:
        # Take the most active
        active_k = active_k[np.argsort(c[active_k])[::-1][:n_active]]
    
    # Build constraint matrix for active constraints
    # C[k_idx, j] = a[k_idx - j] if 0 <= k_idx - j < n
    C = np.zeros((len(active_k), n))
    for idx, k in enumerate(active_k):
        for j in range(n):
            if 0 <= k - j <= n - 1:
                C[idx, j] = a[k - j]
    
    # Solve LP
    g = cp.Variable(n, nonneg=True)
    constraints = [C @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if g.value is not None:
            g_val = np.asarray(g.value).flatten()
            g_val = np.maximum(g_val, 0)
            return g_val
    except:
        pass
    return None

    cols_list = []

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

    B = fft(b, fsize)

def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def build_active_matrix(a, active_k):
    """Build sparse constraint matrix for convolution with a at active indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_k):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([idx] * len(js))
            cols.extend(js.tolist())
            vals.extend(a[k - js].tolist())
    if not rows:
        return None
    return csr_matrix((np.array(vals), (np.array(rows), np.array(cols))),
                      shape=(len(active_k), n))

def lp_direction(a, M, max_active=400, threshold_frac=0.99, solver=cp.SCS):
    """Find best direction g: maximize sum(g) s.t. g>=0, (a*g)[k]<=M for active k."""
    n = len(a)
    c = fft_convolve_self(a)
    
    # Find active constraints
    thr = threshold_frac * M
    active_k = np.where(c >= thr)[0]
    if len(active_k) == 0:
        active_k = np.array([np.argmax(c)])
    if len(active_k) > max_active:
        order = np.argsort(c[active_k])[::-1]
        active_k = active_k[order[:max_active]]
    
    C = build_active_matrix(a, active_k)
    if C is None:
        return None
    
    g = cp.Variable(n, nonneg=True)
    constraints = [C @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if g.value is not None:
            g_val = np.maximum(np.asarray(g.value).flatten(), 0.0)
            if g_val.sum() > 1e-10:
                return g_val
    except Exception as e:
        pass
    
    return None

# ============ Line Search ============

def line_search(a, g, S):
    """Find t in [0,1] minimizing max of ((1-t)a+tg)*((1-t)a+tg)."""
    c_a = fft_convolve_self(a)
    c_g = fft_convolve_self(g)
    c_ag = fft_convolve_ab(a, g)
    
    A = c_a - 2.0 * c_ag + c_g
    B = 2.0 * (c_ag - c_a)
    
    # Grid search
    best_t = 0.0
    best_val = np.inf
    
    # Coarse grid
    for t in np.linspace(0, 1, 200):
        val = ((1-t)**2 * c_a + 2*t*(1-t) * c_ag + t**2 * c_g).max()
        if val < best_val:
            best_val = val
            best_t = t
    
    # Fine grid around best
    dt = 1.0 / 200
    t_lo = max(0, best_t - dt)
    t_hi = min(1, best_t + dt)
    for t in np.linspace(t_lo, t_hi, 200):
        val = ((1-t)**2 * c_a + 2*t*(1-t) * c_ag + t**2 * c_g).max()
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t, best_val

def lp_descent(a0, time_budget, max_active=400, threshold_frac=0.99,
               stagnant_limit=30, perturb_scale=0.08, solver=cp.SCS,
               verbose=True, seed=0):

    a = a0.copy().astype(np.float64)
    n = len(a)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    best_iter = 0
    iter_no = 0
    start = time.monotonic()
    ps = perturb_scale
    
    while time.monotonic() - start < time_budget:
        M = c.max()
        
        g = lp_direction(a, M, max_active=max_active,
                        threshold_frac=threshold_frac, solver=solver)
        
        if g is not None and g.sum() > 1e-10:
            g = g * (S / g.sum())
            t, _ = line_search(a, g, S)
            if t > 0:
                a = (1 - t) * a + t * g
                a = np.maximum(a, 0.0)
                S_new = a.sum()
                if S_new > 1e-10:
                    a = a * (S / S_new)
            else:
                # No improvement along this direction, try small step
                a = (1 - 0.01) * a + 0.01 * g
                a = np.maximum(a, 0.0)
                a = a * (S / a.sum())
        else:
            # LP failed, perturb
            a = a + ps * S / np.sqrt(n) * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
        
        c = fft_convolve_self(a)
        cur_obj = obj_value(a, c)
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
            if verbose and iter_no % 5 == 0:
                print(f"    iter={iter_no}, obj={cur_obj:.6f}")
        
        # Stagnation
        if iter_no - best_iter > stagnant_limit:
            a = a + ps * S / np.sqrt(n) * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            ps = max(ps * 0.6, 0.01)
            if verbose:
                print(f"    [perturb] iter={iter_no}, obj={obj_value(a,c):.6f}, best={best_obj:.6f}")
        
        iter_no += 1
        if verbose and iter_no % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter={iter_no}, obj={obj_value(a,c):.6f}, best={best_obj:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_obj

# ============ Peak Shaving Heuristic ============

def peak_shave(a, S, n_steps=100, verbose=False):
    """Directly reduce peaks by decreasing contributing a values."""
    a = a.copy()
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    
    for step in range(n_steps):
        k_max = np.argmax(c)
        # Find which a values contribute most to c[k_max]
        j_start = max(0, k_max - n + 1)
        j_end = min(n - 1, k_max)
        contributions = a[j_start:j_end+1] * a[k_max - np.arange(j_start, j_end+1)]
        
        # Reduce the largest contributors
        top = np.argmax(contributions)
        j_idx = j_start + top
        a[j_idx] *= 0.9
        a = np.maximum(a, 0)
        a = a * (S / a.sum())
        c = fft_convolve_self(a)
        obj = obj_value(a, c)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        else:
            a[j_idx] /= 0.9  # revert
            break
    
    return best_a, best_obj

"""

            best_iter = iter_no
            if verbose:
                print(f"    [perturb] iter={iter_no}, obj={obj_value(a, c):.6f}, best={best_obj:.6f}, ps={ps:.4f}")

        iter_no += 1
        if verbose and iter_no % 50 == 0:
            el = time.monotonic() - start
            print(f"    iter={iter_no}, obj={obj_value(a, c):.6f}, best={best_obj:.6f}, t={el:.1f}s")

    return best_a, best_obj


def gradient_polish(a0, S, time_budget, lr=0.003, top_k=5, verbose=False):
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    v = np.zeros(n)
    start = time.monotonic()
    best_iter = 0
    iter_no = 0

    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        top_idx = np.argsort(c)[::-1][:top_k]
        c_max = c[top_idx[0]]
        w = np.exp(3.0 * (c[top_idx] - c_max))
        w /= w.sum()

        w_full = np.zeros(len(c))
        w_full[top_idx] = w
        a_rev = a[::-1]
        fsize = 1 << (len(c) + n - 1 - 1).bit_length()
        W = fft(w_full, fsize)
        R = fft(a_rev, fsize)
        full_conv = np.real(ifft(W * R))
        grad = 2.0 * full_conv[n - 1:2 * n - 1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        v = 0.9 * v + 0.1 * grad

        a = a * np.exp(-lr * v)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())

        cur_obj = obj_value(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
        elif iter_no - best_iter > 200:
            a = best_a + 0.05 * S / np.sqrt(n) * np.random.standard_normal(n)
            a = np.maximum(a, 0)
            a = a * (S / a.sum())
            v *= 0.3
            best_iter = iter_no

        iter_no += 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    global_best = np.inf
    global_a = None

    # Load given sequence
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            n_given = len(given_raw)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_raw * (S_given / given_raw.sum())
            ver = evaluate_sequence(given_a.tolist())
            print(f"Given: n={n_given}, obj={ver:.6f}")
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print(f"Load error: {e}")

    # Phase 1: Quick exploration (12% of time)
    explore_time = 0.12 * (deadline - time.monotonic())
    print(f"\nPhase 1: Exploration ({explore_time:.0f}s)")

    candidates = []
    if global_a is not None:
        candidates.append(('given', global_a))

    for n_try in [4096, 8192]:
        S_try = np.sqrt(2.0 * n_try)
        i = np.arange(n_try)
        a1 = 1.0 + 0.4 * np.cos(2 * np.pi * 5 * i / n_try) + 0.2 * np.cos(2 * np.pi * 13 * i / n_try)
        a1 = np.maximum(a1, 0.01)
        a1 *= S_try / a1.sum()
        candidates.append((f'cos_{n_try}', a1))

        a2 = np.exp(-2.0 * ((i / n_try - 0.5) ** 2) / 0.05)
        a2 = np.maximum(a2, 0.01)
        a2 *= S_try / a2.sum()
        candidates.append((f'gauss_{n_try}', a2))

    per_t = explore_time / len(candidates)
    for name, a_c in candidates:
        if time.monotonic() > deadline - 180:
            break
        t0 = time.monotonic()
        try:
            ab, ob = lp_descent(a_c, per_t, max_active=150, threshold_frac=0.995,
                                stagnant_limit=12, perturb_scale=0.1,
                                verbose=False, seed=seed)
        except Exception:
            ab, ob = a_c.copy(), obj_value(a_c)
        el = time.monotonic() - t0
        print(f"  {name}: {ob:.6f} ({el:.1f}s)")
        if ob < global_best and np.isfinite(ob):
            global_best = ob
            global_a = ab.copy()

    # Phase 2: Main LP optimization (63% of time)
    main_time = 0.63 * (deadline - time.monotonic())
    print(f"\nPhase 2: Main LP descent ({main_time:.0f}s)")

    if global_a is not None and main_time > 30:
        # 2a: Broad exploration with fewer active constraints
        t_sub = main_time * 0.35
        print(f"  2a: Broad (active=200, thr=0.99), budget={t_sub:.0f}s")
        ab, ob = lp_descent(global_a, t_sub, max_active=200, threshold_frac=0.99,
                            stagnant_limit=30, perturb_scale=0.08,
                            verbose=True, seed=seed + 1)
        if ob < global_best and np.isfinite(ob):
            global_best = ob
            global_a = ab.copy()
            print(f"    -> {ob:.6f}")

        # 2b: Medium precision
        t_sub = main_time * 0.35
        print(f"  2b: Medium (active=500, thr=0.985), budget={t_sub:.0f}s")
        ab, ob = lp_descent(global_a, t_sub, max_active=500, threshold_frac=0.985,
                            stagnant_limit=40, perturb_scale=0.06,
                            verbose=True, seed=seed + 2)
        if ob < global_best and np.isfinite(ob):
            global_best = ob
            global_a = ab.copy()
            print(f"    -> {ob:.6f}")

        # 2c: High precision
        t_sub = main_time * 0.30
        print(f"  2c: Tight (active=1000, thr=0.97), budget={t_sub:.0f}s")
        ab, ob = lp_descent(global_a, t_sub, max_active=1000, threshold_frac=0.97,
                            stagnant_limit=50, perturb_scale=0.04,
                            verbose=True, seed=seed + 3)
        if ob < global_best and np.isfinite(ob):
            global_best = ob
            global_a = ab.copy()
            print(f"    -> {ob:.6f}")

    # Phase 3: Polish (25% of time)
    polish_time = deadline - time.monotonic() - 15.0
    print(f"\nPhase 3: Polish ({polish_time:.0f}s)")

    if global_a is not None and polish_time > 10:
        n = len(global_a)
        S = global_a.sum()

        # 3a: Very tight LP
        t_3a = polish_time * 0.45
        print(f"  3a: Ultra-tight LP (active=2000, thr=0.95), budget={t_3a:.0f}s")
        ab, ob = lp_descent(global_a, t_3a, max_active=2000, threshold_frac=0.95,
                            stagnant_limit=60, perturb_scale=0.02,
                            verbose=True, seed=seed + 4)
        if ob < global_best and np.isfinite(ob):
            global_best = ob
            global_a = ab.copy()
            print(f"    -> {ob:.6f}")

        # 3b: Gradient-based fine polish
        t_3b = polish_time * 0.35
        print(f"  3b: Gradient polish, budget={t_3b:.0f}s")
        ab, ob = gradient_polish(global_a, S, t_3b, lr=0.002, top_k=3, verbose=False)
        if ob < global_best and np.isfinite(ob):
            global_best = ob
            global_a = ab.copy()
            print(f"    -> {ob:.6f}")

        # 3c: Final LP micro-steps
        t_3c = polish_time * 0.20
        print(f"  3c: Final LP, budget={t_3c:.0f}s")
        ab, ob = lp_descent(global_a, t_3c, max_active=3000, threshold_frac=0.94,
                            stagnant_limit=40, perturb_scale=0.01,
                            verbose=True, seed=seed + 5)
        if ob < global_best and np.isfinite(ob):
            global_best = ob
            global_a = ab.copy()
            print(f"    -> {ob:.6f}")

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: local_best={global_best:.6f}, verified={ver:.6f}")

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: {ver:.6f}")

    return final_a
