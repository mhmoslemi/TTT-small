This is indeed a banded Toeplitz-like matrix. For k < n, row k has nonzeros in columns 0 to k. For k >= n, row k has nonzeros in columns k-n+1 to n-1.

import numpy as np
import time
import scipy.sparse as sp
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize, linprog


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) * S / n
    # Need to project onto {x>=0, sum=S}
    v = np.maximum(v, 0.0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z (softmax params)."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    # Compute grad_a = 2 * cross_corr(w, a) truncated to n
    n_c = len(w)
    fsize = 1 << (n_c + n - 1).bit_length()
    W = fft(w, fsize)
    A_conj = np.conj(fft(a, fsize))
    grad_a = 2.0 * np.real(ifft(W * A_conj))[:n]

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z

    """L-BFGS with log-sum-exp continuation. Returns (best_a, best_score)."""

        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()

        if time.monotonic() - start > max_time - 3:

        exp_z = np.exp(z_shifted)
        a_cur = S * exp_z / exp_z.sum()
        c_cur = fft_convolve_self(a_cur)
        cur_score = c_cur.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_cur.copy()

        def fun(z_val):
            return softmax_obj_grad(z_val, beta, S)[0]

        def jac(z_val):
            return softmax_obj_grad(z_val, beta, S)[1]

        t0 = time.monotonic()
        try:

def build_sparse_toeplitz_rows(a, indices, n):
    """Build sparse rows of the Toeplitz-like matrix for given convolution indices.
    T[row, j] = a[indices[row] - j] if 0 <= indices[row] - j < n
    """
    m = len(indices)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(indices):
        k = int(k)
        # j ranges such that 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            j_range = np.arange(j_lo, j_hi + 1)
            a_idx = k - j_range
            rows.extend([idx] * len(j_range))
            cols.extend(j_range.tolist())
            vals.extend(a[a_idx].tolist())
    if len(rows) == 0:
        return sp.csr_matrix((m, n))
    return sp.csr_matrix((vals, (rows, cols)), shape=(m, n))


def solve_lp_direction(a, n, S, M, top_m_indices, max_cut_rounds=3):
    """Solve LP: maximize sum(g) s.t. (a*g)_k <= M for k in active set, g >= 0.
    Uses constraint generation if initial solution violates other constraints.
    """
    active = list(top_m_indices)
    c = fft_convolve_self(a)

    g = None
    for round_idx in range(max_cut_rounds):
        # Build sparse matrix
        T = build_sparse_toeplitz_rows(a, active, n)
        m_act = len(active)

        # LP: minimize -sum(g) subject to T@g <= M*1, g >= 0
        c_obj = -np.ones(n)
        A_ub = T
        b_ub = np.full(m_act, M)
        bounds = [(0.0, None)] * n

        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                             method='highs', options={'presolve': True, 'time_limit': 10})
            if result.success:
                g = result.x
            else:
                if g is None:
                    return None
                break
        except Exception:
            if g is None:
                return None
            break

        if g is None:
            break

        # Check for violations of other constraints
        fg = fft_cross_convolve(a, g)
        violations = fg > M + 1e-8
        if not np.any(violations):
            break

        # Add worst violators
        viol_indices = np.where(violations)[0]
        worst = viol_indices[np.argsort(fg[viol_indices])[-5:]]
        for w in worst:
            if w not in active:
                active.append(int(w))
        # Cap active set size
        if len(active) > 500:
            active = sorted(active, key=lambda k: -c[k])[:500]

    return g


def run_lp_refinement(a_init, n, S, max_time, rng):
    """LP-based direction finding refinement."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = normalize_to_sum(a, S)
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    iteration = 0

    while time.monotonic() - start < max_time:
        iteration += 1
        c = fft_convolve_self(a)
        M = c.max()

        # Select top-m indices for LP constraints
        top_m = min(100, 2 * n - 1)
        top_indices = np.argsort(c)[::-1][:top_m]

        # Solve LP
        g = solve_lp_direction(a, n, S, M, top_indices, max_cut_rounds=2)

        if g is None:
            # Fallback: use subgradient direction
            grad = np.zeros(n)
            for k in top_indices[:10]:
                k = int(k)
                j_lo = max(0, k - n + 1)
                j_hi = min(n - 1, k)
                if j_lo <= j_hi:
                    grad[j_lo:j_hi+1] += 2.0 * a[k - (j_lo:j_hi+1)]
            g = np.maximum(0.0, -grad)
            gs = g.sum()
            if gs < 1e-15:
                break
            g = g / gs * S

        g_sum = g.sum()
        if g_sum < 1e-15:
            break

        g_rescaled = (S / g_sum) * g

        # Backtracking line search on TRUE objective
        best_t = 0.0
        best_new_score = best_score
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - t) * a + t * g_rescaled
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial * S / s
            score = fft_convolve_self(a_trial).max()
            if score < best_new_score - 1e-12:
                best_new_score = score
                best_t = t

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g_rescaled
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a * S / s
            if best_new_score < best_score:
                best_score = best_new_score
                best_a = a.copy()
                if iteration % 10 == 1:
                    elapsed = time.monotonic() - start
                    print("  LP iter %d: score=%.10f, t=%.2f (%.0fs elapsed)" %
                          (iteration, best_score, best_t, elapsed))
        else:
            # No improvement; try smaller top_m or perturbation
            if iteration % 20 == 0:
                # Perturb to escape
                noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
                a_pert = a + noise
                a_pert = normalize_to_sum(a_pert, S)
                # Keep going with perturbed a
                a = a_pert
            else:
                # Reduce top_m to focus on fewer constraints
                top_m = max(10, top_m // 2)
                if top_m < 10:
                    break

    return best_a, best_score


def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k=5):
    """Nesterov accelerated top-k subgradient method."""
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_val = 1.0
    start = time.monotonic()
    c = fft_convolve_self(y)
    best_score = c.max()
    best_local = y.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Top-k subgradient
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        grad = np.zeros(n)
        for i, k in enumerate(top_indices):
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += weights[i] * 2.0 * a[k - (j_lo:j_hi+1)]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * S / s_y
        else:
            y = a.copy()
        a = x_new
        t_val = t_new

        if it % 2000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t_val = 1.0

    return best_local, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for non-negative sequence minimizing 2*n*max(conv)/sum^2."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    total_deadline = time.monotonic() + budget_s - 20.0

    # Determine n and get initial sequence
    inits = []
    n_main = 8192

    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0.0, 1000.0)))
            n_main = len(given)
    except Exception:
        pass

    n = n_main
    S = np.sqrt(2.0 * float(n))
    tt = np.linspace(0, 1, n, endpoint=False)

    # Additional initializations
    inits.append(("rand", rng.random(n) + 0.1))
    inits.append(("multi_sin", 1.0 + 0.3*np.sin(2*np.pi*5*tt) + 0.2*np.cos(2*np.pi*11*tt) +
                  0.15*np.sin(2*np.pi*17*tt) + 0.1*np.cos(2*np.pi*29*tt) + 0.05))
    inits.append(("bump", np.exp(-0.5*((tt-0.3)*8.0)**2) + 0.5*np.exp(-0.5*((tt-0.7)*8.0)**2) + 0.05))

    print("=== Search: n=%d, %d inits, budget=%.0fs ===" % (n, len(inits), budget_s))
    print("Target: 1.5027, Previous best: 1.508805")

    global_best = np.inf
    global_a = None

    def update_global(a_candidate, score):
        nonlocal global_best, global_a
        if score < global_best:
            global_best = score
            global_a = a_candidate.copy()

    # === PHASE 1: L-BFGS Continuation (40% of time) ===
    phase1_time = (total_deadline - time.monotonic()) * 0.4
    print("\n=== Phase 1: L-BFGS Continuation (%.0fs) ===" % phase1_time)

    per_init_time = max(phase1_time / len(inits), 15.0)

    for name, arr in inits:
        if time.monotonic() > total_deadline - 60:
            break
        t0 = time.monotonic()
        a_norm = normalize_to_sum(arr, S)
        c = fft_convolve_self(a_norm)
        init_score = c.max()
        update_global(a_norm, init_score)

        beta_sched = [5, 10, 20, 50, 100, 200, 500, 1000, 2000]
        a_new, score = run_lbfgs(a_norm, n, S, per_init_time, rng, beta_sched, maxiter_per_beta=150)
        a_new = normalize_to_sum(a_new, S)
        c_new = fft_convolve_self(a_new)
        final_score = c_new.max()
        update_global(a_new, final_score)

        print("  %s: init=%.8f -> %.8f (%.1fs)" % (name, init_score, final_score, time.monotonic()-t0))
        if time.monotonic() - t0 > per_init_time * 1.5:
            break

    print("After Phase 1: best=%.10f" % global_best)

    # === PHASE 2: LP-based Refinement (40% of time) ===
    if global_a is not None:
        remaining = total_deadline - time.monotonic()
        phase2_time = remaining * 0.5
        print("\n=== Phase 2: LP Refinement (%.0fs) ===" % phase2_time)

        if phase2_time > 10:
            a_lp, score_lp = run_lp_refinement(global_a, n, S, phase2_time, rng)
            a_lp = normalize_to_sum(a_lp, S)
            c_lp = fft_convolve_self(a_lp)
            final_lp = c_lp.max()
            update_global(a_lp, final_lp)
            print("  LP refinement: %.10f" % final_lp)

    # === PHASE 3: Nesterov Polish (15% of time) ===
    if global_a is not None:
        remaining = total_deadline - time.monotonic()
        phase3_time = remaining * 0.4
        print("\n=== Phase 3: Nesterov Polish (%.0fs) ===" % phase3_time)

        if phase3_time > 10:
            for top_k in [3, 5, 8]:
                if time.monotonic() > total_deadline - 15:
                    break
                seg = min(phase3_time / 3, 120)
                a_nes, score_nes = run_nesterov_toppk(global_a, n, S, seg, rng, top_k=top_k)
                a_nes = normalize_to_sum(a_nes, S)
                c_nes = fft_convolve_self(a_nes)
                update_global(a_nes, c_nes.max())
                print("  Nesterov top_k=%d: %.10f" % (top_k, global_best))

    # === PHASE 4: Final L-BFGS polish ===
    if global_a is not None:
        remaining = total_deadline - time.monotonic()
        print("\n=== Phase 4: Final L-BFGS Polish (%.0fs) ===" % remaining)

        if remaining > 15:
            a_fin, score_fin = run_lbfgs(global_a, n, S, remaining, rng,
                                          beta_schedule=[500, 1000, 2000, 5000],
                                          maxiter_per_beta=100)
            a_fin = normalize_to_sum(a_fin, S)
            c_fin = fft_convolve_self(a_fin)
            update_global(a_fin, c_fin.max())
            print("  Final L-BFGS: %.10f" % global_best)

    # === PHASE 5: One more LP round on best ===
    if global_a is not None:
        remaining = total_deadline - time.monotonic()
        if remaining > 10:
            print("\n=== Phase 5: Final LP Round (%.0fs) ===" % remaining)
            a_lp2, score_lp2 = run_lp_refinement(global_a, n, S, remaining, rng)
            a_lp2 = normalize_to_sum(a_lp2, S)
            c_lp2 = fft_convolve_self(a_lp2)
            update_global(a_lp2, c_lp2.max())
            print("  Final LP: %.10f" % global_best)

    # === Verification ===
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\n=== Verification ===")
        print("evaluate_sequence returned: %.10f" % ver if np.isfinite(ver) else "INF")
        if not np.isfinite(ver):
            print("  Falling back to uniform")
            global_a = np.ones(256, dtype=np.float64)

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("\nFinal: length=%d, score=%.10f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
