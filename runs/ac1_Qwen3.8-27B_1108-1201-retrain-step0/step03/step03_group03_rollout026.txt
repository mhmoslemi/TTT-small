import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

import numpy as np

def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

        if len(active_indices) > 100:
            top_vals = c[active_indices]
            top_order = np.argsort(top_vals)[::-1][:100]
            active_indices = active_indices[top_order]
        
        K = len(active_indices)
        
        # Compute gradient: average of shifted a over active lags
        # For lag k, grad[i] = a[k-i] (if valid) + a[i-k+n-1] (if valid, for symmetry)
        # Actually for full convolution at lag k:
        # c[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i]*a[k-i]
        # dc[k]/da[j] = a[k-j] + a[j-k] (if both valid and k-j != j-k, i.e., 2j != k)
        #              = 2*a[k-j] (if k-j == j-k, i.e., 2j = k... wait that's j=k/2)
        # Let me just compute it numerically via FFT
        w_full = np.zeros(2 * n - 1)
        w_full[active_indices] = 1.0 / K
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14 * S:
            break
        
        # Step size: bisection-based
        # We want to find the largest step such that max doesn't increase
        lo, hi = 0.0, 2.0 * S / gnorm
        for _ in range(10):
            mid = (lo + hi) / 2
            a_trial = project_simplex(a - mid * grad, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_max:
                lo = mid
            else:
                hi = mid
        step = lo * 0.9
        
        if step < 1e-12:
            # Try smaller active set
            active_indices = np.argsort(c)[::-1][:5]
            K = len(active_indices)
            w_full = np.zeros(2 * n - 1)
            w_full[active_indices] = 1.0 / K
            W = fft(w_full, fsize)
            A = fft(a, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 2.0 * S / gnorm * 0.1
        
        a = project_simplex(a - step * grad, S)
        
        # Periodic spectral step
        if it % 50 == 0:
            a = spectral_waterfill(a, S, alpha=0.9, blend=0.2)
        
        # Perturbation if stalled
        if it % 200 == 0 and best_score > 0 and it > 100:
            if rng.random() < 0.1:
                noise = 0.01 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)

    return best_local, best_score


def run_block_cd(a_init, n, S, max_time, rng, block_size=256):
    """
    Block coordinate descent: optimize one block at a time.
    For each block, use bisection on a scalar parameter to reduce the max.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    n_blocks = n // block_size
    remainder = n % block_size
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()
        
        # Process each block
        for b in range(n_blocks):
            if time.monotonic() - start > max_time:
                break
            lo = b * block_size
            hi = lo + block_size
            block = a[lo:hi].copy()
            
            # Try to optimally adjust this block
            # Strategy: try shifting mass within the block to reduce max
            c_block_max = c.max()
            
            # Compute local gradient for this block
            # For each element a[j] in the block, the gradient of max_b is:
            # sum over active lags k of a[k-j] (contribution to c[k])
            top_lags = np.argsort(c)[::-1][:min(30, len(c))]
            
            grad_block = np.zeros(block_size)
            for k in top_lags:
                k = int(k)
                for local_i in range(block_size):
                    j = lo + local_i
                    # Contribution of a[j] to c[k]: a[k-j] if valid
                    partner = k - j
                    if 0 <= partner < n and partner != j:
                        grad_block[local_i] += a[partner]
            
            gnorm = np.linalg.norm(grad_block)
            if gnorm < 1e-15:
                continue
            
            # Bisection for step size
            scale = S / n
            lo_step, hi_step = 0.0, 5.0 * scale / max(gnorm, 1e-15)
            for _ in range(8):
                mid = (lo_step + hi_step) / 2
                a_trial = a.copy()
                new_block = np.maximum(block - mid * grad_block, 0)
                # Rescale block to maintain total sum
                block_sum = a[lo:hi].sum()
                new_block_sum = new_block.sum()
                if new_block_sum > 1e-15:
                    new_block = new_block * (block_sum / new_block_sum)
                a_trial[lo:hi] = new_block
                # Rescale entire array
                total = a_trial.sum()
                if total > 1e-15:
                    a_trial = a_trial * (S / total)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < c_block_max:
                    lo_step = mid
                else:
                    hi_step = mid
            
            if lo_step > 1e-12:
                new_block = np.maximum(block - lo_step * grad_block, 0)
                block_sum = a[lo:hi].sum()
                new_block_sum = new_block.sum()
                if new_block_sum > 1e-15:
                    new_block = new_block * (block_sum / new_block_sum)
                a[lo:hi] = new_block
                total = a.sum()
                if total > 1e-15:
                    a = a * (S / total)

    return best_local, best_score


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = mu."""
    half = n // 2
    h = rng.random(half) * 0.8 + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum()) if a.sum() > 0 else np.ones(n) * S / n
    return a


def make_spectral_init(n, rng, S, target_flatness=0.7):
    """
    Create initialization with a target spectral shape.
    Start with a flat-ish spectrum and invert.
    """
    fsize = 1 << (2 * n - 2).bit_length()
    # Create a spectrum that's "almost flat" with some random perturbation
    freqs = np.arange(fsize)
    # Base: flat
    H_mag = np.ones(fsize)
    # Add random modulation
    n_mod = rng.integers(3, 8)
    for _ in range(n_mod):
        center = rng.integers(0, fsize // 2)
        width = rng.integers(fsize // 100, fsize // 10)
        amp = rng.random() * (1 - target_flatness)
        # Gaussian modulation
        mod = 1 + amp * np.exp(-0.5 * ((freqs - center) / width) ** 2)
        H_mag *= mod
    
    # Ensure non-negative time domain
    H_phase = rng.standard_normal(fsize) * np.pi
    # Make H Hermitian-symmetric for real output
    H = H_mag * np.exp(1j * H_phase)
    H[fsize // 2 + 1:] = np.conj(H[fsize // 2 - 1:0:-1])
    H[0] = H_mag[0]  # DC is real
    H[fsize // 2] = H_mag[fsize // 2]  # Nyquist is real
    
    a = np.real(ifft(H, fsize))[:n]
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a * (S / a.sum())
    return a


def make_cosine_init(n, rng, S):
    """Cosine-based initialization with random frequency and phase."""
    tt = np.arange(n, dtype=np.float64) / n
    n_freqs = rng.integers(2, 6)
    a = np.ones(n)
    for _ in range(n_freqs):
        freq = rng.integers(1, 15)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.5
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    S = np.sqrt(2.0 * n)
    inits = []
    
    if parent is not None and len(parent) >= 64:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if p.sum() > 0:
            inits.append(("parent", p * (S / p.sum())))
            # Also add perturbed parent
            p_pert = p + rng.normal(0, 0.01 * S / np.sqrt(n), n)
            p_pert = np.maximum(p_pert, 0)
            inits.append(("parent_pert", p_pert * (S / p_pert.sum())))
    
    for i in range(3):
        inits.append(("anti_pal_%d" % i, make_anti_palindromic_init(n, rng, S)))
    
    for i in range(3):
        inits.append(("spectral_%d" % i, make_spectral_init(n, rng, S)))
    
    for i in range(3):
        inits.append(("cosine_%d" % i, make_cosine_init(n, rng, S)))
    
    # Random
    r = rng.random(n) + 0.05
    inits.append(("random", r * (S / r.sum())))
    
    return inits


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def run_full_optimization(a_init, n, S, total_time, rng):
    """
    Run a full optimization pipeline on a single initialization.
    Phase A: Smoothed subgradient (70% of time)
    Phase B: Active-set exact (25% of time)
    Phase C: Block CD (5% of time)
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = safe_normalize(a_init, n)
    
    best_score = np.inf
    best_a = a.copy()
    
    # Phase A: Smoothed subgradient
    tA = total_time * 0.65
    aA, sA = run_smoothed_subgrad(a, n, S, tA, rng,
                                   beta_start=5.0, beta_end=3000.0,
                                   use_spectral=True, spectral_freq=15,
                                   spectral_alpha=0.85)
    aA = safe_normalize(aA, n)
    cA = fft_convolve_self(aA)
    if cA.max() < best_score:
        best_score = cA.max()
        best_a = aA.copy()
    
    # Phase B: Active-set
    rem = total_time - (time.monotonic() - time.monotonic())  # will use passed time
    tB = total_time * 0.25
    aB, sB = run_active_set_exact(aA, n, S, tB, rng)
    aB = safe_normalize(aB, n)
    cB = fft_convolve_self(aB)
    if cB.max() < best_score:
        best_score = cB.max()
        best_a = aB.copy()
    
    # Phase C: Block CD
    tC = total_time * 0.10
    aC, sC = run_block_cd(aB, n, S, tC, rng, block_size=512)
    aC = safe_normalize(aC, n)
    cC = fft_convolve_self(aC)
    if cC.max() < best_score:
        best_score = cC.max()
        best_a = aC.copy()
    
    return best_a, best_score
