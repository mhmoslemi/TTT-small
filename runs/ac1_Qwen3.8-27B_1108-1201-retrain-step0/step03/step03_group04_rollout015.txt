    w = np.zeros(len(c))

    return w

def peak_shave(a, S, T, rng, n_shave=5):

                pairs.append((i, j, a[i]*a[j]))

        pairs.sort(key=lambda x: -x[2])

        if not cold_indices:
            cold_indices = [i for i in range(n) if a[i] > 0]

        moved = 0
        for i, j, prod in pairs[:n_shave]:

            if a[i] >= a[j]:

2. Targeted peak-shaving via pressure-based mass redistribution

5. Final ultra-fine coordinate polish

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def grad_via_fft(a, weights, n):
    """Gradient of sum_k weights[k] * (a*a)[k] with respect to a."""
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def softmax_weights(c, beta):
    """Normalized softmax weights on convolution values."""
    cm = c.max()
    e = np.exp(beta * (c - cm))
    s = e.sum()
    if s < 1e-300:
        return np.full(len(c), 1.0 / len(c))
    return e / s


def toppk_weights(c, k):
    """Normalized top-k weights (proportional to values)."""
    k = min(k, len(c))
    idx = np.argpartition(c, -k)[-k:]
    v = c[idx]
    s = v.sum()
    if s < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    w = np.zeros(len(c))
    w[idx] = v / s
    return w

        cur = c.max()
        if cur < best_score:

def peak_shave(a0, S, T, rng, batch_size=3):
    """
    Direct peak shaving: identify hottest elements at peak lags,
    redistribute mass to cold elements.
    """
    n = len(a0)
    a = a0.copy()
    t_start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()

    while time.monotonic() - t_start < T:
        c = fft_conv_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # Find top peak lags
        n_peaks = min(batch_size, len(c))
        peak_lags = np.argpartition(c, -n_peaks)[-n_peaks:]

        # Compute pressure: how much each element contributes to peak lags
        pressure = np.zeros(n)
        for ki in peak_lags:
            ki = int(ki)
            for i in range(n):
                j = ki - i
                if 0 <= j < n:
                    pressure[i] += a[j]

        # Identify hot (high pressure) and cold (low pressure) elements
        # Only consider elements with non-zero value
        mask = a > 1e-10
        if mask.sum() < 2:
            break

        hot_thresh = np.percentile(pressure[mask], 80)
        cold_thresh = np.percentile(pressure[mask], 20)

        hot_idx = np.where(mask & (pressure >= hot_thresh))[0]
        cold_idx = np.where(mask & (pressure <= cold_thresh))[0]

        if len(hot_idx) == 0 or len(cold_idx) == 0:
            # Fallback: use random pairs
            hot_idx = np.where(mask)[0]
            cold_idx = np.where(mask)[0]

        # Redistribute mass from hot to cold
        n_moves = min(20, len(hot_idx), len(cold_idx))
        for _ in range(n_moves):
            hi = int(rng.integers(len(hot_idx)))
            ci = int(rng.integers(len(cold_idx)))
            h = hot_idx[hi]
            c_idx = cold_idx[ci]
            if h == c_idx:
                continue
            delta = 0.005 * a[h]
            if delta < 1e-15:
                continue
            a[h] -= delta
            a[c_idx] += delta

        a = project_simplex(a, S)

    return best_a, best_score


def basin_hop(a0, S, T, rng, inner_T=2.0, jump_scale=0.03):
    """
    Basin hopping: alternate between local optimization and random jumps.
    """
    n = len(a0)
    a = a0.copy()
    t_start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()

    while time.monotonic() - t_start < T:
        remaining = T - (time.monotonic() - t_start)
        if remaining < 3:
            break
        inner_T_use = min(inner_T, remaining * 0.5)

        # Local optimization
        a_opt, s_opt = fista_core(a, S, inner_T_use, rng, beta0=50, betaT=500)
        a_opt = np.clip(a_opt, 0, 1000)
        s_opt = fft_conv_self(a_opt).max()
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()

        # Check time
        if time.monotonic() - t_start >= T:
            break

        # Random jump
        scale = jump_scale * S / np.sqrt(n)
        a_new = a_opt + rng.normal(0, scale, n)
        a_new = project_simplex(a_new, S)
        a = a_new

    return best_a, best_score

        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # Find peak lag
        k_star = int(np.argmax(c))

        # Find elements with highest "pressure" at peak
        pressure = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                pressure[i] = a[j]

        # Get top n_hot elements by pressure
        top = np.argpartition(pressure, -n_hot)[-n_hot:]
        rng.shuffle(top)

        # Try small perturbations on each
        for i in top:
            if time.monotonic() - t_start >= T:
                break
            old = a[i]
            if old < 1e-15:
                continue
            # Try a few multiplicative adjustments
            for factor in [0.5, 0.8, 1.2]:
                a[i] = old * factor
                a = project_simplex(a, S)
                c_test = fft_conv_self(a)
                if c_test.max() < cur:
                    cur = c_test.max()
                    if cur < best_score:
                        best_score = cur
                        best_a = a.copy()
                else:
                    a[i] = old
                    a = project_simplex(a, S)
                    break

    return best_a, best_score


def safe_normalize(a, n):
    """Clip to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2385):
    """
    Multi-phase optimizer for minimizing 2*n*max(a*a)/sum(a)^2.
    Uses adaptive FISTA, peak shaving, basin hopping, and coordinate polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 40.0

    # Get initial sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size < 10:
            raise ValueError("too short")
        n = len(given)
    except Exception:
        n = 8192
        given = np.ones(n) * 0.5

    S = np.sqrt(2.0 * float(n))
    a_init = safe_normalize(given, n)
    c0 = fft_conv_self(a_init)
    best_score = c0.max()
    best_a = a_init.copy()
    obj0 = 2.0 * n * best_score / (S * S)
    print("n=%d, initial max_conv=%.6f, objective=%.6f" % (n, best_score, obj0))

    # === Phase 1: Adaptive FISTA (40% of available time) ===
    avail = deadline - time.monotonic()
    t1 = 0.35 * avail
    if t1 > 10:
        print("Phase 1: Adaptive FISTA (%.0f sec)" % t1)
        t0 = time.monotonic()
        a1, s1 = fista_core(a_init, S, t1, rng, beta0=15.0, betaT=800.0)
        a1 = safe_normalize(a1, n)
        s1 = fft_conv_self(a1).max()
        if s1 < best_score:
            best_score = s1
            best_a = a1.copy()
        print("  FISTA: max_conv=%.6f obj=%.6f [%.0fs]" % (s1, 2*n*s1/S**2, time.monotonic()-t0))

    # === Phase 2: Top-k FISTA (20% of remaining) ===
    avail = deadline - time.monotonic()
    t2 = 0.20 * avail
    if t2 > 10:
        print("Phase 2: Top-k FISTA (%.0f sec)" % t2)
        t0 = time.monotonic()
        a2, s2 = fista_toppk(best_a, S, t2, rng, k_start=1, k_end=30, k_rate=50)

        if s2 < best_score:
            best_score = s2
            best_a = a2.copy()
        print("  TopK: max_conv=%.6f obj=%.6f [%.0fs]" % (s2, 2*n*s2/S**2, time.monotonic()-t0))

    # === Phase 3: Peak shaving (15% of remaining) ===
    avail = deadline - time.monotonic()
    t3 = 0.15 * avail
    if t3 > 5:
        print("Phase 3: Peak shaving (%.0f sec)" % t3)
        t0 = time.monotonic()
        a3, s3 = peak_shave(best_a, S, t3, rng, batch_size=3)
        a3 = safe_normalize(a3, n)
        s3 = fft_conv_self(a3).max()
        if s3 < best_score:
            best_score = s3
            best_a = a3.copy()
        # Follow up with quick FISTA
        t3b = min(5.0, deadline - time.monotonic() - 20)
        if t3b > 2:
            a3b, s3b = fista_core(best_a, S, t3b, rng, beta0=200, betaT=1000)
            a3b = safe_normalize(a3b, n)
            s3b = fft_conv_self(a3b).max()
            if s3b < best_score:
                best_score = s3b
                best_a = a3b.copy()
        print("  PeakShave: max_conv=%.6f obj=%.6f [%.0fs]" % (best_score, 2*n*best_score/S**2, time.monotonic()-t0))

    # === Phase 4: Basin hopping (15% of remaining) ===
    avail = deadline - time.monotonic()
    t4 = 0.15 * avail
    if t4 > 10:
        print("Phase 4: Basin hopping (%.0f sec)" % t4)
        t0 = time.monotonic()
        a4, s4 = basin_hop(best_a, S, t4, rng, inner_T=3.0, jump_scale=0.02)

        if s4 < best_score:

        print("  BasinHop: max_conv=%.6f obj=%.6f [%.0fs]" % (s4, 2*n*s4/S**2, time.monotonic()-t0))

    # === Phase 5: Coordinate polish + final FISTA (15% of remaining) ===
    avail = deadline - time.monotonic()
    t5a = 0.4 * avail
    t5b = 0.6 * avail
    if t5a > 3:
        print("Phase 5a: Coordinate polish (%.0f sec)" % t5a)
        t0 = time.monotonic()
        a5, s5 = coordinate_polish(best_a, S, t5a, rng, n_hot=80)

        if s5 < best_score:

            best_a = a5.copy()
        print("  CoordPolish: max_conv=%.6f obj=%.6f [%.0fs]" % (s5, 2*n*s5/S**2, time.monotonic()-t0))

        t5b = deadline - time.monotonic() - 10
        if t5b > 2:

            a5b = safe_normalize(a5b, n)

                best_score = s5b
                best_a = a5b.copy()
            print("  FinalFISTA: max_conv=%.6f obj=%.6f" % (s5b, 2*n*s5b/S**2))

    # === Final verification ===
    final_obj = 2.0 * n * best_score / (S * S)
    print("FINAL: n=%d, max_conv=%.6f, objective=%.6f" % (n, best_score, final_obj))

    final_list = [float(x) for x in best_a]
    ver = evaluate_sequence(final_list)

        print("Verified: %.6f" % ver)
    else:
        print("Verification FAILED, using fallback")
        final_list = [0.5] * 256

    return final_list
