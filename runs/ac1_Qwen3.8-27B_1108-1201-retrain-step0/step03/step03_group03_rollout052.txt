def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        return np.zeros(n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def lse_grad(a, c, beta):
    """Gradient of LSE of c."""
    n = len(a)
    cm = c.max()
    eb = np.exp(beta * (c - cm))
    w = eb / eb.sum()
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def make_anti_pal(n, rng, S, peakedness=0.5):
    """Anti-palindromic init with controlled peakedness."""
    half = n // 2
    # peakedness: 0 = uniform, 1 = maximally peaked
    h = rng.random(half)
    h = h ** (1.0 / (1.0 + 5.0 * peakedness))  # More peaked = more extreme
    h = h / h.sum() * (half * S / n)  # Normalize so sum(h) = half*S/n... 
    # Actually, for anti-palindromic: sum(a) = half * mu where mu = S/half
    # a[i] for i < half: h[i], a[n-1-i] = mu - h[i]
    # sum = sum(h) + sum(mu - h) = half * mu = S

    eps = 0.01 * mu

    a[:half] = h

    if n % 2 == 1:

    a = np.maximum(a, 0.0)
    return a * (S / a.sum())

def run_lse_fista(a0, n, S, max_time, rng, beta_start=5.0, beta_end=200.0):
    """LSE-smoothed FISTA with adaptive beta."""
    a = np.maximum(a0, 0.0).copy()
    a = a * (S / a.sum())
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = a.copy()
    best_score = np.inf
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1.0, max_time * 2.0))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        c = fft_conv(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best = y.copy()
        
        grad = lse_grad(y, c, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        # Adaptive step size
        step = 2.0 / (beta * np.linalg.norm(c - c.min()) + gn + 1e-10)
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic perturbation
        if it % 200 == 0 and rng.random() < 0.2:
            noise = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t = 1.0
    
    return best, best_score

def run_peak_target(a0, n, S, max_time, rng):

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        return np.zeros(n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
