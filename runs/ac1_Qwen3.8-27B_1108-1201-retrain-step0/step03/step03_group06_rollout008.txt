import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_subgrad_weighted(c, a, T, K=None):
    """Compute weighted subgradient using exponential or top-K peak weighting."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)

    if K is not None and K < len(c):
        # Top-K selection with geometric decay weights
        topk_idx = np.argsort(c)[::-1][:K]
        w = np.zeros(len(c))
        for i, idx in enumerate(topk_idx):
            w[idx] = np.exp(-0.3 * i)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
    else:
        # Exponential temperature weighting
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def run_adam_adaptive_k(a_init, n, S, time_budget, T_init=10.0, T_final=2000.0,

    Adam optimizer with adaptive K (top-K peak selection).

            'increase' = monotonically increase K,
            'adaptive' = adjust K based on improvement rate

    a = np.maximum(a, 1e-12)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())

    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t = 0
    eps = 1e-8

    c = fft_convolve_self(a)

    best_a = a.copy()

            K = int(1 + 199 * progress)

            K = max(1, K_adaptive)

        prev_best = best_obj
        if cur_obj < best_obj - 1e-13:

            best_a = a.copy()
            no_improve = 0

            no_improve += 1
            improvement_rate = 0.9 * improvement_rate + 0.1 * (0.0)

def run_exact_peak_descent(a_init, n, S, time_budget, verbose=False):
    """
    Direct subgradient descent on the exact peak (K=1) with backtracking.
    This is the sharpest possible descent direction.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    S_target = S

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    step = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        if no_improve > 80:
            # Restart from best with small perturbation
            rng = np.random.default_rng(step)
            a = best_a + 0.003 * rng.standard_normal(n)
            a = normalize_nonneg(a, S_target)
            a_prev = a.copy()
            no_improve = 0
            step += 1
            continue

        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Exact peak gradient
        k_star = int(np.argmax(c))
        # g[j] = 2 * a[k_star - j] if 0 <= k_star - j < n, else 0
        lo = max(0, k_star - n + 1)
        hi = min(k_star + 1, n)
        g = np.zeros(n)
        if hi > lo:
            g[lo:hi] = 2.0 * a[k_star - (hi - 1):k_star - lo + 1][::-1]

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g_dir = g / gnorm

        # Backtracking line search
        lr = 0.08 * (1.0 - 0.5 * min(1.0, (time.monotonic() - start) / time_budget))
        a_prev = a.copy()
        improved = False
        for i in range(15):
            a_trial = a - lr * g_dir
            a_trial = normalize_nonneg(a_trial, S_target)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial < cur_obj - 1e-11 * max(1, cur_obj):
                a = a_trial
                improved = True
                break
            lr *= 0.5

        if not improved:
            no_improve += 5

        step += 1
        if verbose and step % 500 == 0:
            print("    PeakDesc step=%d: obj=%.7f" % (step, best_obj))

    return best_a, best_obj


def pair_coordinate_descent(a_init, n, S, time_budget, verbose=False):
    """
    Pair coordinate descent: update pairs of coordinates simultaneously.
    For each pair (i,j), find the optimal scaling that minimizes the peak.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(99999)

    # Precompute pair lists for efficiency
    sweep = 0
    while time.monotonic() - start < time_budget:
        improved = False
        # Random pairs
        n_pairs = min(n // 2, 5000)
        pairs = rng.integers(0, n, size=(n_pairs, 2))

        for i, j in pairs:
            if time.monotonic() - start > time_budget:
                break
            if i == j:
                continue

            orig_i = a[i]
            orig_j = a[j]

            # Try multiplicative adjustments to the pair
            best_local_obj = best_obj
            best_fi = 1.0
            best_fj = 1.0

            for fi in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
                for fj in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
                    if fi == 1.0 and fj == 1.0:
                        continue
                    a[i] = orig_i * fi
                    a[j] = orig_j * fj
                    a = normalize_nonneg(a, S)
                    c_check = fft_convolve_self(a)
                    obj_check = c_check.max()
                    if obj_check < best_local_obj - 1e-13:
                        best_local_obj = obj_check
                        best_fi = fi
                        best_fj = fj

            a[i] = orig_i * best_fi
            a[j] = orig_j * best_fj
            a = normalize_nonneg(a, S)

            if best_local_obj < best_obj - 1e-13:
                best_obj = best_local_obj
                best_a = a.copy()
                improved = True

            # Update c only if improved
            if improved:
                c = fft_convolve_self(a)

        sweep += 1
        if not improved and sweep >= 5:
            break
        if verbose and sweep % 10 == 0:
            print("    PCD sweep=%d: obj=%.7f" % (sweep, best_obj))

    return best_a, best_obj


def single_coordinate_descent(a_init, n, S, time_budget, verbose=False):

    a = a_init.copy().astype(np.float64)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    sweep = 0

    while time.monotonic() - start < time_budget:
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue

            best_local = orig
            best_local_obj = best_obj

            # Geometric search around current value
            for factor in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0]:
                a[i] = orig * factor
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-13:

def spectral_equalization(a_init, n, S, time_budget, verbose=False):
    """
    Try to equalize the power spectrum |FFT(a)|^2 to reduce convolution peaks.
    Uses iterative Fourier transform algorithm (IFTA).
    """

    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    target = S / n  # target level for flat spectrum

    while time.monotonic() - start < time_budget:
        A = fft(a, n)
        # Target: |A|^2 should be as flat as possible
        # But A[0] = sum(a) = S is fixed
        # For k > 0, try to make |A[k]| as equal as possible
        mags = np.abs(A)
        # Desired magnitude: DC stays, others target a common level
        non_dc = mags[1:]
        target_mag = np.sqrt(np.sum(non_dc ** 2) / (n - 1)) if n > 1 else 0

        new_mags = np.copy(mags)
        new_mags[0] = mags[0]  # keep DC
        # Blend toward target
        blend = 0.1
        new_mags[1:] = (1 - blend) * mags[1:] + blend * target_mag

        # Keep phase
        phases = np.angle(A)
        A_new = new_mags * np.exp(1j * phases)
        a_new = np.real(ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new * (S / a_new.sum()) if a_new.sum() > 0 else a

        c_new = fft_convolve_self(a_new)
        if c_new.max() < best_obj - 1e-13:
            best_obj = c_new.max()
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def get_spectral_init(n, S, rng):
    """Generate initialization with near-flat spectrum."""
    # Start with random phases, target flat magnitude
    n_fft = 1 << (2 * n - 1).bit_length()
    phases = rng.random(n_fft) * 2 * np.pi
    mags = np.ones(n_fft)
    mags[0] = S  # DC component
    # Add some structure
    for k in range(1, min(20, n_fft // 2)):
        mags[k] = mags[-k] = S / (n + k * 0.1)

    A = mags * np.exp(1j * phases)
    # Make it conjugate symmetric for real output
    for k in range(1, n_fft // 2 + 1):
        A[-k] = np.conj(A[k])
    a = np.real(ifft(A))[:n]
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.full(n, S / n)
    else:
        a = a * (S / a.sum())
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 30.0)
    deadline = time.monotonic() + total_budget

    global_best = np.inf
    global_a = None

    # Load given sequence
    n_given = 4096
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            global_a = given_raw
            n_given = len(global_a)
            S = np.sqrt(2.0 * n_given)
            global_a = global_a * (S / global_a.sum())
            ver = evaluate_sequence(global_a.tolist())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver):
                global_best = ver
    except Exception as e:
        print("Could not load given: " + str(e))
        global_a = np.ones(4096) * (np.sqrt(2 * 4096) / 4096)
        global_best = evaluate_sequence(global_a.tolist())

    n = len(global_a)
    S = np.sqrt(2.0 * float(n))

    # Verify and print
    print("Starting optimization: n=%d, S=%.4f, best=%.7f" % (n, S, global_best))

    a = global_a.copy()
    a = normalize_nonneg(a, S)

    # ============ PHASE 1: Intensive Adam with adaptive K (50% time) ============
    p1_budget = 0.50 * total_budget
    print("\nPhase 1: Adam adaptive-K (budget=%.0fs)" % p1_budget)

    # Sub-phase 1a: Low-to-mid temperature
    t1 = 0.35 * p1_budget
    a_best, _ = run_adam_adaptive_k(
        a, n, S, t1,
        T_init=5.0, T_final=200.0, lr_init=0.04, K_mode="oscillate"
    )
    a = normalize_nonneg(a_best, S)
    obj = evaluate_sequence(a.tolist())
    if np.isfinite(obj) and obj < global_best:
        global_best = obj
        global_a = a.copy()
        print("  P1a Adam osc K5-100: obj=%.7f" % obj)

    # Sub-phase 1b: Mid-to-high temperature
    t2 = 0.35 * p1_budget
    a_best, _ = run_adam_adaptive_k(
        a, n, S, t2,
        T_init=50.0, T_final=800.0, lr_init=0.03, K_mode="adaptive"
    )
    a = normalize_nonneg(a_best, S)
    obj = evaluate_sequence(a.tolist())
    if np.isfinite(obj) and obj < global_best:
        global_best = obj
        global_a = a.copy()
        print("  P1b Adam adaptive K: obj=%.7f" % obj)

    # Sub-phase 1c: High temperature, focus on exact peak
    t3 = 0.30 * p1_budget
    a_best, _ = run_exact_peak_descent(
        a, n, S, t3
    )
    a = normalize_nonneg(a_best, S)
    obj = evaluate_sequence(a.tolist())
    if np.isfinite(obj) and obj < global_best:
        global_best = obj
        global_a = a.copy()
        print("  P1c ExactPeak: obj=%.7f" % obj)

    # ============ PHASE 2: Pair + Single Coordinate Descent (25% time) ============
    p2_budget = 0.25 * total_budget
    print("\nPhase 2: Coordinate Descent (budget=%.0fs)" % p2_budget)
    a = global_a.copy()

    # Pair CD
    t_cd1 = 0.50 * p2_budget
    a_best, _ = pair_coordinate_descent(a, n, S, t_cd1)
    a = normalize_nonneg(a_best, S)
    obj = evaluate_sequence(a.tolist())

        global_best = obj
        global_a = a.copy()
        print("  P2 PairCD: obj=%.7f" % obj)

    # Single CD
    t_cd2 = 0.50 * p2_budget
    a_best, _ = single_coordinate_descent(a, n, S, t_cd2)
    a = normalize_nonneg(a_best, S)
    obj = evaluate_sequence(a.tolist())
    if np.isfinite(obj) and obj < global_best:
        global_best = obj
        global_a = a.copy()
        print("  P2 SingleCD: obj=%.7f" % obj)

    # ============ PHASE 3: Spectral + Final Polish (15% time) ============
    p3_budget = 0.15 * total_budget
    print("\nPhase 3: Spectral + Polish (budget=%.0fs)" % p3_budget)
    a = global_a.copy()

    # Spectral equalization
    t_spec = 0.30 * p3_budget
    a_best, _ = spectral_equalization(a, n, S, t_spec)
    a = normalize_nonneg(a_best, S)
    obj = evaluate_sequence(a.tolist())
    if np.isfinite(obj) and obj < global_best:

        global_a = a.copy()
        print("  P3 Spectral: obj=%.7f" % obj)

    # Final Adam with very high T
    t_final_adam = 0.35 * p3_budget
    a_best, _ = run_adam_adaptive_k(
        a, n, S, t_final_adam,
        T_init=500.0, T_final=5000.0, lr_init=0.015, K_mode="oscillate"
    )
    a = normalize_nonneg(a_best, S)
    obj = evaluate_sequence(a.tolist())
    if np.isfinite(obj) and obj < global_best:
        global_best = obj
        global_a = a.copy()
        print("  P3 FinalAdam: obj=%.7f" % obj)

    # Final CD polish
    t_final_cd = p3_budget - t_spec - t_final_adam
    if t_final_cd > 3:
        a_best, _ = pair_coordinate_descent(a, n, S, t_final_cd)
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  P3 FinalCD: obj=%.7f" % obj)

    # ============ PHASE 4: Random Restart Exploration (10% time) ============
    p4_budget = deadline - time.monotonic()
    if global_a is not None and p4_budget > 20:
        print("\nPhase 4: Restarts (budget=%.0fs)" % p4_budget)
        a = global_a.copy()
        n_rounds = max(3, int(p4_budget / 30))
        per_round = p4_budget / (n_rounds + 1)

        for r in range(n_rounds):
            if time.monotonic() > deadline - 5:
                break
            # Structured perturbation
            perturb_scale = 0.003 + 0.002 * r
            pert_type = r % 4
            if pert_type == 0:
                freq = (r + 1) * 3
                phase = rng.random() * 2 * np.PI
                a = a * (1.0 + perturb_scale * np.sin(2 * np.pi * freq * np.arange(n) / n + phase))
            elif pert_type == 1:
                a = a + perturb_scale * rng.standard_normal(n)
            elif pert_type == 2:
                # Anti-peak perturbation: boost where convolution is low
                c = fft_convolve_self(a)
                low_regions = c < np.median(c)
                # Map low regions in convolution back to time domain indices
                # Simple approach: boost coordinates that contribute least to peaks
                contrib = np.zeros(n)
                k_star = int(np.argmax(c))
                lo = max(0, k_star - n + 1)
                hi = min(k_star + 1, n)
                if hi > lo:
                    contrib[lo:hi] = a[max(0, k_star - hi + 1):min(n, k_star - lo + 1)][::-1]
                a = a - perturb_scale * contrib / (contrib.max() + 1e-10)
            else:
                # Mix with a fresh spectral init
                fresh = get_spectral_init(n, S, rng)
                a = 0.9 * a + 0.1 * fresh

            a = normalize_nonneg(a, S)

            a_best, _ = run_adam_adaptive_k(
                a, n, S, per_round,
                T_init=200.0, T_final=3000.0, lr_init=0.02, K_mode="adaptive"
            )
            a_best = normalize_nonneg(a_best, S)
            obj = evaluate_sequence(a_best.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_best.copy()
                print("  Restart %d: obj=%.7f" % (r, obj))

    # ============ FINAL: Ultra-fine polish ============
    remaining = deadline - time.monotonic()
    if remaining > 5 and global_a is not None:
        a = global_a.copy()
        a_best, _ = pair_coordinate_descent(a, n, S, remaining * 0.6)
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  UltraPolish: obj=%.7f" % obj)

        remaining2 = deadline - time.monotonic()
        if remaining2 > 3:
            a_best, _ = single_coordinate_descent(a, n, S, remaining2)
            a = normalize_nonneg(a_best, S)
            obj = evaluate_sequence(a.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a.copy()
                print("  FinalPolish: obj=%.7f" % obj)

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f (verified=%s)" % (
        global_best, "%.7f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)

    return final_a
