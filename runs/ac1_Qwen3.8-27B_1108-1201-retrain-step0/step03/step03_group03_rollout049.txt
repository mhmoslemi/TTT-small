Δc[k] = 2δ * (a[k-i] - a[k-j]) for k in [0, 2n-2]

where a[x] = 0 for x < 0 or x ≥ n.

So Δc is the difference of two shifted copies of a, scaled by 2δ.

The new max is: max(c + 2δ * (shift_a_i - shift_a_j))

where shift_a_i[k] = a[k-i] (zero-padded) and similarly for j.

This can be computed efficiently with numpy array operations.

Minimize 2*n*max(conv(a,a))/sum(a)^2 using greedy pair swapping with exact 1D optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def get_shift(a, i, conv_len):
    """Get the shifted version of a for convolution purposes.
    result[k] = a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    result = np.zeros(conv_len)
    lo = i
    hi = min(i + n, conv_len)
    if lo < hi:
        result[lo:hi] = a[:hi-lo]
    return result

def optimize_delta(a, c, i, j, shift_i, shift_j):
    """Find optimal delta for transferring mass from j to i (a[i] += delta, a[j] -= delta).
    Minimize max(c + 2*delta*(shift_i - shift_j)).
    """
    d = shift_i - shift_j  # d[k] = a[k-i] - a[k-j]
    
    # Feasible range: a[i] + delta >= 0 and a[j] - delta >= 0
    delta_min = max(0 - a[i], -a[j])  # = -min(a[i], a[j])... wait
    # a[i] + delta >= 0 => delta >= -a[i]
    # a[j] - delta >= 0 => delta <= a[j]
    lo = -a[i]
    hi = a[j]
    
    if lo >= hi:
        return 0.0
    
    # Ternary search on [lo, hi]
    # f(delta) = max(c + 2*delta*d)
    for _ in range(50):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(c + 2 * m1 * d)
        f2 = np.max(c + 2 * m2 * d)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    return (lo + hi) / 2
