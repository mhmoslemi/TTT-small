```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses LP-based direction finding (solving max sum(b) s.t. b>=0, conv(f,b)_k <= M for top-k peaks)
combined with gradient-based polishing.
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_row(f, k, n):
    """Build row k of the convolution matrix. Row k has entries: A[k,i] = f[k-i] if 0<=k-i<n."""
    row = np.zeros(n)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    if i_start <= i_end:
        # row[i] = f[k-i] for i in [i_start, i_end]
        # Let j = k - i, so j goes from k-i_end to k-i_start
        # row[i_start:i_end+1] = f[k-i_end : k-i_start+1][::-1]
        row[i_start:i_end+1] = f[k-i_end:k-i_start+1][::-1]
    return row


def lp_direction(f, M, top_k, n):
    """
    Solve LP: max sum(b) s.t. b >= 0, conv(f,b)_k <= M for k in top_k peaks.
    Returns (direction, improvement) or (None, None) if failed.
    """
    c = fft_convolve_self(f)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_indices = sorted(set(int(k) for k in top_indices))
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(top_indices):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start <= i_end:
            for i in range(i_start, i_end + 1):
                rows.append(idx)
                cols.append(i)
                vals.append(f[k - i])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(len(top_indices), n))
    
    # LP: max 1^T b s.t. A b <= M, b >= 0
    # linprog: min -1^T b s.t. A b <= M, -I b <= 0
    c_obj = -np.ones(n)
    A_ub = sp.vstack([A, -sp.eye(n)]).tocsr()
    b_ub = np.concatenate([np.full(len(top_indices), M), np.zeros(n)])
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs',
                     options={'presolve': True, 'time_limit': 30})
    
    if result.success:
        b = np.maximum(result.x, 0)
        return b
    return None


def lp_based_optimize(a_init, n, S, max_time, rng, top_k=50):
    """
    Optimize using LP-based direction finding.
    At each step: solve LP to get direction b, then update a = (1-t)*a + t*b.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    # First compute initial objective
    c = fft_convolve_self(a)
    cur_max = c.max()
    best_score = cur_max
    best_local = a.copy()
    
    it = 0
    t = 0.1  # mixing parameter
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()
        
        if M < best_score:
            best_score = M
            best_local = a.copy()
        
        # Solve LP
        b = lp_direction(a, M, top_k, n)
        
        if b is None:
            # Fallback: use gradient
            break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        # Rescale b to match normalization
        b = b * (S / b_sum)
        
        # Update
        t_new = t  # Could decay t over time
        a_new = (1 - t_new) * a + t_new * b
        a_new = np.maximum(a_new, 0)
        a_new = project_simplex(a_new, S)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() <= M * 1.01:
            a = a_new
        else:
            # Try smaller step
            t_new = t_new * 0.5
            a_new = (1 - t_new) * a + t_new * b
            a_new = np.maximum(a_new, 0)
            a_new = project_simplex(a_new, S)
            a = a_new
        
        # Decay t slowly

        if it % 50 == 0:
            c_check = fft_convolve_self(a)
            print("  LP iter %d: max_c=%.6f, best=%.6f, t=%.4f" % (it, c_check.max(), best_score, t))
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses LP-based direction finding (state-of-the-art approach) combined with 
gradient polishing and multiple initializations.
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.asarray(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0]
    if len(rho) < 1:
        return np.zeros(n)
    hat = rho[-1]
    theta = (cssv[hat] - S) / (hat + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_toppk(a, c, top_k):
    """
    Compute gradient of max(c) using top-k subgradient.
    grad[i] = 2 * sum_{k in top} a[k-i] (with a[m]=0 for m out of range)
    Using FFT: grad = 2 * ifft(conj(fft(a)) * fft(w))[:n]
    where w is the indicator of top peaks.
    """
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    
    w = np.zeros(2 * n - 1)
    for k in top_indices:
        w[int(k)] = 1.0
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad, top_indices


def compute_grad_weighted(a, c, top_k):
    """
    Weighted top-k gradient: weight by the peak values for better conditioning.
    """
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    
    # Normalize weights
    w = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w[int(k)] = top_vals[i]
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad


def solve_lp_direction(f, M, top_k, n, time_limit=30):
    """
    Solve LP: max sum(b) s.t. b >= 0, conv(f,b)_k <= M for k in top_k peaks.
    Returns b or None if failed.
    """
    c = fft_convolve_self(f)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_indices_sorted = sorted(set(int(k) for k in top_indices))
    nk = len(top_indices_sorted)
    
    # Build sparse constraint matrix A (nk x n)
    # A[idx, i] = f[k - i] if 0 <= k - i < n, where k = top_indices_sorted[idx]
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(top_indices_sorted):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start > i_end:
            continue
        i_range = range(i_start, i_end + 1)
        # A[idx, i] = f[k-i]
        f_vals = f[k - i_end : k - i_start + 1][::-1]
        rows.extend([idx] * len(i_range))
        cols.extend(list(i_range))
        vals.extend(f_vals.tolist())
    
    if not rows:
        return None
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(nk, n))
    
    # LP: max 1^T b s.t. A b <= M*1, b >= 0
    # linprog: min -1^T b s.t. [A; -I] b <= [M*1; 0]
    c_obj = -np.ones(n)
    A_ub = sp.vstack([A, -sp.eye(n)]).tocsr()
    b_ub = np.concatenate([np.full(nk, M), np.zeros(n)])
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs',
                        options={'presolve': True, 'time_limit': time_limit})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    
    return None

def generate_inits(n, rng):
    """Generate diverse asymmetric initializations."""
    tt = np.linspace(0, 1, n, endpoint=False)
    inits = []
    
    # Exponential ramps
    for alpha in [1.0, 2.0, 3.0, 5.0, 7.0]:
        inits.append(np.exp(alpha * tt))
        inits.append(np.exp(alpha * (1.0 - tt)))
    
    # Power-law
    for p in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append((tt + 0.01) ** p)
        inits.append((1.0 - tt + 0.01) ** p)
    
    # Gaussian at different positions
    for center in [0.2, 0.3, 0.7, 0.8]:
        inits.append(np.exp(-((tt - center) ** 2) / 0.05))
    
    # Mixed
    inits.append(np.exp(2 * tt) * (1.0 + 0.5 * np.sin(4 * np.pi * tt)))
    inits.append(np.exp(-2 * tt) * (1.0 + 0.5 * np.cos(6 * np.pi * tt)))
    
    # Linear
    inits.append(tt + 0.1)
    inits.append(1.0 - tt + 0.1)
    
    # Random
    for _ in range(3):
        inits.append(rng.random(n) * 0.5 + 0.5 * np.exp(rng.random() * 3 * tt))
    
    return inits
```
