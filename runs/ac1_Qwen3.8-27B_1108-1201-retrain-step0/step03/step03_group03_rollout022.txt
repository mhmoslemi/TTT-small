"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Gerchberg-Saxton spectral flattening for initialization
2. FISTA with log-sum-exp smoothing
3. Subgradient with top-k active lags
4. Coordinate descent polish
5. Multiple restarts with diverse initializations

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def spectral_flatten(a_init, n, S, n_iters=200, alpha=0.5, rng=None):
    """
    Gerchberg-Saxton style spectral flattening.
    Iteratively modifies the magnitude spectrum to be flatter.
    """
    a = np.clip(a_init, 0, 1000).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    fsize = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = A / np.where(mag > 0, mag, 1)
        
        # Target: flat magnitude spectrum
        # |A[0]| = S (fixed), target for others
        # Average power: sum(|A|^2) / fsize = sum(a^2) (by Parseval)
        # For flat: |A[j]|^2 = sum(a^2) * fsize / fsize = sum(a^2) for all j
        # But |A[0]| = S is fixed, so target for j>0:
        # (S^2 + (fsize-1)*target^2) / fsize = sum(a^2)
        # target^2 = (fsize * sum(a^2) - S^2) / (fsize - 1)
        
        a2_sum = np.sum(a * a)
        target_power = (fsize * a2_sum - S * S) / max(fsize - 1, 1)
        target_mag = np.sqrt(max(target_power, 0))
        
        # Blend current and target
        new_mag = np.ones_like(mag)
        new_mag[0] = S  # DC is fixed
        new_mag[1:] = (1 - alpha) * mag[1:] + alpha * target_mag
        
        # Actually, let's use a multiplicative blend (geometric mean) which is more stable
        new_mag[1:] = np.sqrt(mag[1:] * target_mag + 1e-30)
        
        A_new = phase * new_mag
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new * (S / s_new)
        else:
            a = a_init * (S / s) if s > 0 else np.ones(n) * S / n
    
    return a


def lse_subgradient(a, c, beta):
    """Subgradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (2 * n - 1 + n - 1).bit_length()
    # Wait, we need to convolve w (length 2n-1) with a (length n)
    # Result length: 2n-1 + n - 1 = 3n-2
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A), fsize))[:n]
    return grad


def topk_subgradient(a, c, top_k, temperature=1.0):
    """Subgradient using top-k lags with softmax weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices].copy()
    tv_max = top_vals.max()
    inv_temp = 1.0 / max(temperature, 1e-10)
    weights = np.exp(inv_temp * (top_vals - tv_max))
    weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A), fsize))[:n]
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=200.0):
    """FISTA with log-sum-exp smoothing and adaptive beta."""
    a = np.clip(a_init, 0, 1000).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = 10.0
    stall_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init + 0.5 * it)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stall_count = 0
        elif it % 20 == 0:
            stall_count += 1

        grad = lse_subgradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size with backtracking
        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)

        # Adagrad-like adaptive step (simpler and more robust)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Adaptive L
        if it % 10 == 0:
            c_a = fft_convolve_self(a)
            if c_a.max() > cur:
                L *= 1.2
            else:
                L = max(L * 0.98, 5.0)

        # Periodic perturbation
        if stall_count > 50 and rng.random() < 0.2:
            noise = 0.01 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_momentum = 1.0
            stall_count = 0

    return best_local, best_score


def run_subgradient(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=64):
    """Subgradient with top-k active lags and Polyak step."""
    a = np.clip(a_init, 0, 1000).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / 10.0

    while time.monotonic() - start < max_time:
        it += 1
        k = min(top_k_max, top_k_init + it // 100)
        temp = max(0.5, 3.0 * np.exp(-it / 5000))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = topk_subgradient(y, c, k, temperature=temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Polyak-like step
        f_star_est = (S * S) / (2 * n - 1)  # lower bound on max conv
        gap = cur - f_star_est
        step = gap / max(gnorm * gnorm, 1e-20)
        step = min(step, S)

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        if it % 2000 == 0 and rng.random() < 0.1:
            noise = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_momentum = 1.0

    return best_local, best_score


def run_coord_descent(a_init, n, S, max_time, rng):
    """Coordinate descent: transfer mass between coordinates to reduce peak."""
    a = np.clip(a_init, 0, 1000).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Find top 3 lags
        top_lags = np.argsort(c)[::-1][:3]
        
        # Compute pressure on each coordinate
        pressure = np.zeros(n)
        for lag in top_lags:
            lag = int(lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            indices = np.arange(i_min, i_max + 1)
            partner = lag - indices
            pressure[indices] += a[partner]
        
        # Transfer from high-pressure to low-pressure coordinates
        p_median = np.median(pressure)
        high_idx = np.where(pressure > p_median * 1.5)[0]
        low_idx = np.where(pressure < p_median * 0.5)[0]
        
        if len(high_idx) == 0 or len(low_idx) == 0:
            # Try smaller threshold
            high_idx = np.where(pressure > pressure.max() * 0.7)[0]
            low_idx = np.where(pressure < pressure.min() + 0.3 * (pressure.max() - pressure.min()))[0]
        
        if len(high_idx) == 0 or len(low_idx) == 0:
            break
        
        # Choose source and target
        src = high_idx[rng.integers(len(high_idx))]
        dst = low_idx[rng.integers(len(low_idx))]
        
        # Binary search for optimal transfer amount
        max_transfer = a[src] * 0.5
        lo, hi = 0.0, max_transfer
        best_delta = 0.0
        best_val = cur
        
        for _ in range(8):
            mid = (lo + hi) / 2
            a_trial = a.copy()
            a_trial[src] -= mid
            a_trial[dst] += mid
            c_trial = fft_convolve_self(a_trial)
            val = c_trial.max()
            if val < best_val:
                best_val = val
                best_delta = mid
                lo = mid
            else:
                hi = mid
        
        if best_delta > 0:
            a[src] -= best_delta
            a[dst] += best_delta
            if a[src] < 0:
                a[src] = 0
                a[dst] -= a[src]  # this shouldn't happen
        
    return best_local, best_score

def make_init_periodic(n, rng, S):

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Gerchberg-Saxton spectral flattening for initialization
2. FISTA with log-sum-exp smoothing and adaptive parameters
3. Subgradient with top-k active lags and Polyak step
4. Coordinate descent polish targeting peak lags
5. Multiple diverse initializations with strategic restarts
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def subgradient_fft(w_full, a):
    """Compute 2 * cross_correlation(w_full, a)[:n] via FFT.
    w_full has length 2n-1, a has length n.
    Result[i] = 2 * sum_k w_full[k] * a[k-i] (with a[j]=0 outside [0,n-1]).
    """
    n = len(a)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A), fsize))[:n]
    return grad


def lse_subgradient(a, c, beta):
    """Subgradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    return subgradient_fft(w, a)


def topk_subgradient(a, c, top_k, temperature=1.0):
    """Subgradient using top-k lags with softmax weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices].copy()
    tv_max = top_vals.max()
    inv_temp = 1.0 / max(temperature, 1e-10)
    weights = np.exp(inv_temp * (top_vals - tv_max))
    weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    return subgradient_fft(w_full, a)


def spectral_flatten(a_init, n, S, n_iters=150, alpha=0.4):
    """
    Gerchberg-Saxton style spectral flattening.
    Iteratively flattens the magnitude spectrum to reduce peak convolution.
    """
    a = np.clip(a_init, 0, 1000).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    fsize = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = np.inf

    for it in range(n_iters):
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = A / np.where(mag > 1e-30, mag, 1.0)

        # Target: flatten |A| for j > 0
        # Parseval: sum(|A[j]|^2) = fsize * sum(a^2)
        a2_sum = np.sum(a * a)
        total_power = fsize * a2_sum
        # |A[0]|^2 = S^2, so remaining power for fsize-1 bins:
        remaining = total_power - S * S
        target_power_each = remaining / (fsize - 1)
        target_mag = np.sqrt(max(target_power_each, 0.0))

        # Geometric mean blend (more stable than arithmetic)
        new_mag = np.copy(mag)
        new_mag[0] = S
        new_mag[1:] = np.sqrt(mag[1:] * target_mag + 1e-30)

        A_new = phase * new_mag
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new * (S / s_new)
        else:
            break

        if it % 20 == 0:
            c = fft_convolve_self(a)
            score = c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    c = fft_convolve_self(a)
    if c.max() < best_score:
        best_a = a.copy()

    return best_a


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=150.0):
    """FISTA with log-sum-exp smoothing and adaptive step."""
    a = np.clip(a_init, 0, 1000).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = 5.0
    stall_count = 0
    last_best = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init + 0.3 * it)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stall_count = 0
            last_best = cur
        elif it % 30 == 0:
            if cur < last_best - 1e-12:
                last_best = cur
                stall_count = 0
            else:
                stall_count += 1

        grad = lse_subgradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Adaptive L
        if it % 15 == 0:
            c_a = fft_convolve_self(a)
            if c_a.max() > cur * 1.001:
                L *= 1.3
            else:
                L = max(L * 0.97, 2.0)

        # Perturbation to escape stagnation
        if stall_count > 40 and rng.random() < 0.25:
            noise_scale = 0.008 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_momentum = 1.0
            stall_count = 0

    return best_local, best_score


def run_subgradient(a_init, n, S, max_time, rng, top_k_init=3, top_k_max=48):
    """Subgradient with top-k active lags and adaptive step."""
    a = np.clip(a_init, 0, 1000).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    # Lower bound estimate for gap-based step
    f_lb = (S * S) / (2.0 * n)

    while time.monotonic() - start < max_time:
        it += 1
        k = min(top_k_max, top_k_init + it // 150)
        temp = max(0.3, 2.0 * np.exp(-it / 3000.0))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        grad = topk_subgradient(y, c, k, temperature=temp)
        gnorm2 = np.dot(grad, grad)
        if gnorm2 < 1e-30:
            break

        # Gap-based step size (Polyak)
        gap = max(cur - f_lb, 1e-10)
        step = gap / gnorm2
        step = min(step, S * 0.5)
        step = max(step, 1e-10)

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Occasional restart nudge
        if it % 3000 == 0 and rng.random() < 0.08:
            noise = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_momentum = 1.0

    return best_local, best_score


def run_coord_descent(a_init, n, S, max_time, rng, top_lags=3):
    """Coordinate descent: transfer mass to reduce peak convolution values."""
    a = np.clip(a_init, 0, 1000).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    iters = 0

    while time.monotonic() - start < max_time:
        iters += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Find top lags
        tl = min(top_lags, len(c))
        top_idx = np.argsort(c)[::-1][:tl]

        # Compute pressure: for each coordinate i, sum of a[partner] over top lags
        pressure = np.zeros(n)
        for lag in top_idx:
            lag = int(lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            indices = np.arange(i_min, i_max + 1)
            partner = lag - indices
            pressure[indices] += a[partner]

        # Find high and low pressure coordinates
        p_max = pressure.max()
        p_min = pressure.min()
        if p_max - p_min < 1e-15:
            break

        # Select source (high pressure) and target (low pressure)
        high_thresh = p_min + 0.7 * (p_max - p_min)
        low_thresh = p_min + 0.3 * (p_max - p_min)
        high_idx = np.where(pressure > high_thresh)[0]
        low_idx = np.where(pressure < low_thresh)[0]

        if len(high_idx) == 0:
            high_idx = np.where(pressure > np.median(pressure))[0]
        if len(low_idx) == 0:
            low_idx = np.where(pressure < np.median(pressure))[0]
        if len(high_idx) == 0 or len(low_idx) == 0:
            break

        src = int(high_idx[rng.integers(len(high_idx))])
        dst = int(low_idx[rng.integers(len(low_idx))])

        # Try multiple transfer amounts via ternary-like search
        max_transfer = min(a[src], S * 0.1) * 0.3
        if max_transfer < 1e-15:
            continue

        # Quick binary search for best delta
        lo = 0.0
        hi = max_transfer
        best_delta = 0.0
        best_val = cur

        for _ in range(10):
            mid = (lo + hi) / 2.0
            a_trial = a.copy()
            a_trial[src] -= mid
            a_trial[dst] += mid
            # Only check the top lags for speed
            c_trial = fft_convolve_self(a_trial)
            val = c_trial.max()
            if val < best_val:
                best_val = val
                best_delta = mid
                lo = mid
            else:
                hi = mid

        if best_delta > 0:
            a[src] -= best_delta
            a[dst] += best_delta
        else:
            # Try small random pair
            if rng.random() < 0.5:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    delta = a[i] * 0.01
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s > 0:
                        a = a * (S / s)

    return best_local, best_score


def make_init_anti_pal(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = mu."""
    half = n // 2
    h = rng.random(half) + 0.05
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2.0
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-15:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_init_periodic(n, rng, S, n_freqs=3):
    """Sum of sinusoids with random frequencies."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(n_freqs):
        freq = int(rng.integers(2, 25))
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.35
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    return a * (S / a.sum())


def make_init_wshape(n, rng, S):
    """W/M shaped: |sin(k*pi*x)| pattern."""
    tt = np.arange(n, dtype=np.float64) / n
    k = int(rng.integers(2, 7))
    a = np.abs(np.sin(np.pi * k * tt)) + 0.03
    return a * (S / a.sum())


def make_init_ramp(n, rng, S):
    """Power-law ramp."""
    tt = np.arange(n, dtype=np.float64) / n
    p = 0.5 + rng.random() * 2.0
    if rng.random() < 0.5:
        a = tt ** p + 0.03
    else:
        a = (1.0 - tt) ** p + 0.03
    return a * (S / a.sum())


def make_init_bimodal(n, rng, S):
    """Two peaks at random positions."""
    tt = np.arange(n, dtype=np.float64) / n
    c1 = rng.random() * 0.8 + 0.1
    c2 = 1.0 - c1
    w = 0.02 + rng.random() * 0.05
    a = np.exp(-((tt - c1) / w) ** 2) + np.exp(-((tt - c2) / w) ** 2) + 0.01
    return a * (S / a.sum())


def generate_inits(n, rng, S, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and len(parent) >= 64:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if p.sum() > 0:
            inits.append(("parent", p * (S / p.sum())))
    for i in range(3):
        inits.append(("anti_pal_%d" % i, make_init_anti_pal(n, rng, S)))
    for i in range(3):
        inits.append(("periodic_%d" % i, make_init_periodic(n, rng, S)))
    for i in range(2):
        inits.append(("wshape_%d" % i, make_init_wshape(n, rng, S)))
    inits.append(("ramp", make_init_ramp(n, rng, S)))
    inits.append(("bimodal", make_init_bimodal(n, rng, S)))
    rand_a = rng.random(n) + 0.05
    inits.append(("random", rand_a * (S / rand_a.sum())))
    return inits


def safe_normalize(a, n):
    """Clamp and normalize to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None
    primary_n = 8192

    # Try to get parent
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
            primary_n = len(parent)
    except Exception:
        pass

    S = np.sqrt(2.0 * primary_n)
    print("n=%d, parent_loaded=%s" % (primary_n, parent is not None))

    # === Phase 1: Spectral flattening exploration ===
    print("=== Phase 1: Spectral flattening ===")
    inits = generate_inits(primary_n, rng, S, parent)
    time_per_init = max(2.0, (deadline - time.monotonic() - 600) / max(len(inits), 1))

    candidates = []
    for name, arr in inits:
        if time.monotonic() > deadline - 600:
            break
        t0 = time.monotonic()
        a_flat = spectral_flatten(arr, primary_n, S, n_iters=80, alpha=0.4)
        c = fft_convolve_self(a_flat)
        score = c.max()
        obj = 2.0 * primary_n * score / (S * S)
        candidates.append((obj, a_flat, name))
        print("  %s: obj=%.6f (%.1fs)" % (name, obj, time.monotonic() - t0))

    # Sort by objective
    candidates.sort(key=lambda x: x[0])
    print("  Best init: %s (obj=%.6f)" % (candidates[0][2], candidates[0][0]))

    # Use best spectral flatten result as starting point
    global_a = candidates[0][1].copy()
    global_best = candidates[0][1].sum()  # placeholder, will recompute
    c_check = fft_convolve_self(global_a)
    global_best = c_check.max()

    # === Phase 2: FISTA refinement on top candidates ===
    print("=== Phase 2: FISTA refinement ===")
    remaining = deadline - time.monotonic()
    top_n_candidates = min(3, len(candidates))
    time_per_refine = remaining * 0.35 / top_n_candidates

    for rank in range(top_n_candidates):
        if time.monotonic() > deadline - 300:
            break
        obj, a_cand, name = candidates[rank]
        t0 = time.monotonic()
        alloc = min(time_per_refine, remaining * 0.15)
        a_f, _ = run_fista(a_cand, primary_n, S, alloc, rng,
                           beta_init=3.0, beta_max=80.0)
        a_f = safe_normalize(a_f, primary_n)
        c_f = fft_convolve_self(a_f)
        score = c_f.max()
        if score < global_best:
            global_best = score
            global_a = a_f.copy()
            print("  FISTA[%s]: obj=%.6f" % (name, 2.0 * primary_n * score / (S * S)))

    # === Phase 3: Subgradient refinement ===
    if global_a is not None:
        print("=== Phase 3: Subgradient ===")
        remaining = deadline - time.monotonic()
        if remaining > 60:
            alloc = min(remaining * 0.3, 200)
            a_sg, _ = run_subgradient(global_a, primary_n, S, alloc, rng,
                                      top_k_init=2, top_k_max=32)
            a_sg = safe_normalize(a_sg, primary_n)
            c_sg = fft_convolve_self(a_sg)
            if c_sg.max() < global_best:
                global_best = c_sg.max()
                global_a = a_sg.copy()
                print("  Subgrad: obj=%.6f" % (2.0 * primary_n * global_best / (S * S)))

    # === Phase 4: Coordinate descent polish ===
    if global_a is not None:
        print("=== Phase 4: Coordinate descent ===")
        remaining = deadline - time.monotonic()
        if remaining > 30:
            alloc = min(remaining * 0.35, 250)
            a_cd, _ = run_coord_descent(global_a, primary_n, S, alloc, rng, top_lags=3)
            a_cd = safe_normalize(a_cd, primary_n)
            c_cd = fft_convolve_self(a_cd)
            if c_cd.max() < global_best:
                global_best = c_cd.max()
                global_a = a_cd.copy()
                print("  CoordDesc: obj=%.6f" % (2.0 * primary_n * global_best / (S * S)))

    # === Phase 5: Final high-beta FISTA polish ===
    if global_a is not None:
        print("=== Phase 5: Final polish ===")
        remaining = deadline - time.monotonic()
        if remaining > 15:
            alloc = min(remaining * 0.7, 120)
            a_p, _ = run_fista(global_a, primary_n, S, alloc, rng,
                               beta_init=30.0, beta_max=150.0)
            a_p = safe_normalize(a_p, primary_n)
            c_p = fft_convolve_self(a_p)
            if c_p.max() < global_best:
                global_best = c_p.max()
                global_a = a_p.copy()
                print("  FinalFISTA: obj=%.6f" % (2.0 * primary_n * global_best / (S * S)))

            # Quick coord descent after
            remaining2 = deadline - time.monotonic()
            if remaining2 > 15:
                a_cd2, _ = run_coord_descent(global_a, primary_n, S, min(remaining2 * 0.6, 80), rng, top_lags=2)
                a_cd2 = safe_normalize(a_cd2, primary_n)
                c_cd2 = fft_convolve_self(a_cd2)
                if c_cd2.max() < global_best:
                    global_best = c_cd2.max()
                    global_a = a_cd2.copy()
                    print("  FinalCoord: obj=%.6f" % (2.0 * primary_n * global_best / (S * S)))

    # === Verification ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    n_final = len(global_a)
    print("=== Result: eval=%.6f, internal_obj=%.6f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'),
           2.0 * n_final * global_best / (S * S) if global_a is not None else float('inf'),
           n_final))

    if not np.isfinite(ver):
        print("  WARNING: falling back to uniform")
        global_a = np.ones(1024, dtype=np.float64) * 1.0
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)

    return final_list
