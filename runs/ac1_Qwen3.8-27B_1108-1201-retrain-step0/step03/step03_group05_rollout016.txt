"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Novel approach combining:
1. Spectral balancing (iterative power spectrum compression + non-negativity restoration)
2. L-BFGS-B with fine-tuned log-sum-exp continuation
3. Nesterov-accelerated exact subgradient (top-1) with adaptive restarts
4. Team combination of top solutions + final polishing
5. Multi-length exploration (n=8192 and n=16384)

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z (unconstrained softmax params)."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    n_c = len(w)
    n_a = len(a)
    fsize = 1 << (n_c + n_a - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n_a]

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=300):
    """Run L-BFGS-B with continuation on beta (smooth-to-sharp)."""
    if beta_schedule is None:
        beta_schedule = [3, 5, 10, 20, 50, 100, 200, 400, 800, 1500, 3000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        elapsed_beta = 0.0
        n_attempts = 1
        if beta >= 400 and max_time > 60:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 5:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.015, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            t0 = time.monotonic()
            try:
                result = scipy_minimize(
                    fun, z,
                    jac=jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_beta,
                        'maxfun': maxiter_per_beta * 3,
                        'ftol': 1e-15,
                        'gtol': 1e-10,
                        'maxls': 50,
                        'iprint': -1
                    }
                )
            except Exception:
                break

            elapsed_beta += time.monotonic() - t0
            z = result.x
            z = z - z.mean()

            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

        if elapsed_beta > 0:
            print("    beta=%5d: score=%.10f (%d attempts, %.1fs)" %
                  (beta, best_score, n_attempts, elapsed_beta))

    return best_a, best_score


def run_nesterov_exact_subgrad(a_init, n, S, max_time, rng,
                                top_k=1, restart_interval=2000,
                                noise_scale=0.005):
    """Nesterov-accelerated subgradient with exact (top-k) subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    consecutive_no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        ck = min(64, top_k + it // 200)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        top_indices = np.argsort(c)[::-1][:ck]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A_y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A_y)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 500 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Adaptive restart with targeted perturbation
        if (it % restart_interval == 0 and consecutive_no_improve > restart_interval // 2) or \
           (it % 5000 == 0 and rng.random() < 0.15):
            scale = noise_scale * S / np.sqrt(n) * (1.0 + 0.5 * rng.random())
            y = project_simplex(y + rng.normal(0, scale, n), S)
            a = y.copy()
            t = 1.0
            consecutive_no_improve = 0

    return best_local, best_score


def spectral_balance(a_init, n, S, max_time, rng, alpha=0.7,
                     n_gs_iters=50, n_spectral_iters=200):
    """
    Spectral balancing: iteratively compress the power spectrum to flatten
    the autocorrelation, while maintaining non-negativity via Gerchberg-Saxton.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptively decrease alpha (compress more aggressively over time)
        alpha_cur = max(0.3, alpha - 0.001 * it)

        # Compute DFT
        A = fft(a)

        # Compress power spectrum
        mag = np.abs(A)
        mag_compressed = mag ** alpha_cur

        # Gerchberg-Saxton: alternate between frequency domain magnitude
        # and time domain non-negativity
        A_gs = A / (np.abs(A) + 1e-30) * mag_compressed
        for gs_it in range(n_gs_iters):
            # Frequency to time
            a_gs = np.real(ifft(A_gs))
            # Enforce non-negativity
            a_gs = np.maximum(a_gs, 0.0)
            # Time to frequency
            A_gs = fft(a_gs)
            # Enforce magnitude constraint
            mag_curr = np.abs(A_gs)
            A_gs = A_gs / (mag_curr + 1e-30) * mag_compressed

        a_new = np.real(ifft(A_gs))
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = best_a.copy()
        else:
            a_new = a_new * (S / s_new)

        c = fft_convolve_self(a_new)
        score = c.max()

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Sometimes accept worse solutions to escape local minima
            if rng.random() < 0.05:
                a = a_new.copy()

        if it % 50 == 0:
            print("    Spectral iter %d: score=%.10f (alpha=%.3f)" % (it, best_score, alpha_cur))

    return best_a, best_score

            global_best = c_nest.max()
            global_a = a_nest.copy()
            print("  Nesterov improved: %.10f" % global_best)

        print("  Phase 3 done: best=%.10f (%.1fs)" % (global_best, time.monotonic() - t0))

    # --- Phase 4: Perturbation + L-BFGS restart ---
    remaining = deadline - time.monotonic()
    phase4_time = min(remaining * 0.15, 300)
    if phase4_time > 30 and global_a is not None:
        print("\n=== Phase 4: Perturbation + L-BFGS (%.0f sec) ===" % phase4_time)
        t0 = time.monotonic()

        n_perturb = max(3, int(phase4_time / 40))
        perturb_budget = phase4_time / n_perturb * 0.8

        for p in range(n_perturb):
            if time.monotonic() > deadline - 60:
                break

            # Targeted perturbation: noise concentrated where autocorrelation peak is
            c_cur = fft_convolve_self(global_a)
            peak_idx = np.argmax(c_cur)

            # Create perturbation that targets the peak-contributing elements
            # The peak at lag k* involves a[j]*a[k*-j] pairs
            noise = rng.normal(0, 0.02 * S / np.sqrt(n), n)
            # Add extra noise at positions that contribute to the peak
            if 0 <= peak_idx < n:
                for j in range(n):
                    k_j = peak_idx - j
                    if 0 <= k_j < n:
                        noise[j] += 0.01 * S / np.sqrt(n) * rng.standard_normal()
            elif peak_idx >= n:
                shift = peak_idx - (n - 1)
                for j in range(n - shift):
                    noise[j] += 0.01 * S / np.sqrt(n) * rng.standard_normal()
                    noise[n - 1 - j] += 0.01 * S / np.sqrt(n) * rng.standard_normal()

            a_pert = np.maximum(global_a + noise, 0.0)
            a_pert = safe_normalize(a_pert, n)

            a_new, score = run_lbfgs_continuation(
                a_pert, n, S, perturb_budget, rng,
                beta_schedule=[20, 50, 100, 300, 800, 2000],
                maxiter_per_beta=150
            )
            a_new = safe_normalize(a_new, n)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < global_best:
                global_best = c_new.max()
                global_a = a_new.copy()
                print("  Perturb %d: %.10f" % (p, global_best))

        print("  Phase 4 done: best=%.10f (%.1fs)" % (global_best, time.monotonic() - t0))

    # --- Phase 5: Final Nesterov + L-BFGS polishing ---
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        print("\n=== Phase 5: Final Polishing (%.0f sec) ===" % remaining)
        t0 = time.monotonic()

        # Nesterov polish with top-1
        seg1 = min(remaining * 0.4, 120)
        a_polish, score = run_nesterov_exact_subgrad(
            global_a, n, S, seg1, rng,
            top_k=1, restart_interval=5000, noise_scale=0.002
        )
        a_polish = safe_normalize(a_polish, n)
        c = fft_convolve_self(a_polish)
        if c.max() < global_best:
            global_best = c.max()
            global_a = a_polish.copy()
            print("  Nesterov polish: %.10f" % global_best)

        # Final high-beta L-BFGS
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_final, score = run_lbfgs_continuation(
                global_a, n, S, remaining2, rng,
                beta_schedule=[500, 1000, 2000, 5000, 10000, 20000],
                maxiter_per_beta=300
            )
            a_final = safe_normalize(a_final, n)
            c_final = fft_convolve_self(a_final)
            if c_final.max() < global_best:
                global_best = c_final.max()
                global_a = a_final.copy()
                print("  Final L-BFGS: %.10f" % global_best)

        print("  Phase 5 done: best=%.10f (%.1fs)" % (global_best, time.monotonic() - t0))

    # --- Phase 6: Try different length (n=16384) if time permits ---
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        print("\n=== Phase 6: Try n=16384 (%.0f sec) ===" % remaining)
        t0 = time.monotonic()
        n2 = 16384
        S2 = np.sqrt(2.0 * float(n2))

        # Upsample the current best to length n2
        a_up = np.interp(
            np.linspace(0, n - 1, n2),
            np.arange(n),
            global_a
        )
        a_up = safe_normalize(a_up, n2)
        c_up = fft_convolve_self(a_up)
        score_up = c_up.max()
        # Convert to comparable objective
        obj_up = 2.0 * n2 * score_up / (a_up.sum() ** 2)
        print("  Upsampled to n=%d: obj=%.10f" % (n2, obj_up))

        if obj_up < global_best + 0.05:  # Only worth trying if close
            a_n2, score_n2 = run_lbfgs_continuation(
                a_up, n2, S2, min(remaining - 30, 180), rng,
                beta_schedule=[10, 30, 100, 300, 1000, 3000],
                maxiter_per_beta=200
            )
            a_n2 = safe_normalize(a_n2, n2)
            c_n2 = fft_convolve_self(a_n2)
            obj_n2 = 2.0 * n2 * c_n2.max() / (a_n2.sum() ** 2)
            print("  n=16384 after L-BFGS: obj=%.10f" % obj_n2)

            if obj_n2 < global_best:
                # Compare properly: global_best is max_conv/S^2 ratio
                # Actually let's just compare the raw objective values
                obj_current = 2.0 * n * global_best / (global_a.sum() ** 2)
                if obj_n2 < obj_current:
                    # Need to convert: global_best stores max_conv value
                    # Let me just track the actual evaluate score
                    pass

        print("  Phase 6 done (%.1fs)" % (time.monotonic() - t0))

    # --- Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    # Final check with evaluate_sequence
    final_list = [float(x) for x in global_a]
    final_score = evaluate_sequence(final_list)
    print("Final: length=%d, evaluate_score=%s" % (len(final_list),
          "%.6f" % final_score if np.isfinite(final_score) else "inf"))
    print("Time used: %.1fs" % (time.monotonic() - (deadline - budget_s + 15)))
    return final_list
