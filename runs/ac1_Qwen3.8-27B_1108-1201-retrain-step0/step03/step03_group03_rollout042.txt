Actually, one more thought: the QP approach. If I want to minimize `a^T Q a` where Q is the sum of top-K shift matrices, I can use the following projected gradient method:

```
for iter:
    grad = 2 * Q @ a  (computed efficiently using the shift structure)
    a_new = project_simplex(a - step * grad, S)
    # Adaptive step via backtracking
```

The key advantage over the LSE approach is that the QP subproblem has a well-defined global minimum (Q is PSD), so we're not fighting with the approximation error of LSE.

But wait, there's a subtlety: when we solve the QP for top-K lags, we might increase other lags. So we need to periodically re-evaluate which lags are the top ones.

This is essentially an SDP/active-set method for the original min-max problem:
- Minimize t
- Subject to a^T H_k a <= t for all k
- sum(a) = S, a >= 0

The active-set method would:
1. Solve the QP for the current active set of lags
2. Check if any non-active lag now exceeds t
3. If yes, add it to the active set
4. Repeat

This is exactly what I'll implement!

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using active-set QP + FISTA hybrid.

Key innovations over previous approach:
1. Active-set QP: directly minimize sum of top-K lag quadratics using
   the shift-matrix structure for efficient Hessian-vector products.
2. Adaptive active-set management: dynamically add/remove lags from
   the active set based on current peak values.
3. Multi-scale warm start: optimize short sequences first, then upscale.
4. Better FISTA with cosine-annealed beta schedule.
5. Coordinate descent polish on peak-contributing index pairs.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def qv_product(v, lags, n):
    """Compute Q @ v where Q = sum of shift matrices for given lags.
    
    Q is defined by: (Q @ v)[j] = sum_{k in lags} (v[j+k] + v[j-k])
    with boundary handling (out-of-range terms are 0).
    For k=0: contribution is 2*v[j].
    """
    result = np.zeros(n)
    for k in lags:
        k = int(k)
        if k == 0:
            result += 2.0 * v
        else:
            # v[j+k] for j = 0, ..., n-k-1
            result[:n-k] += v[k:]
            # v[j-k] for j = k, ..., n-1
            result[k:] += v[:n-k]
    return result


def q_form(v, lags, n):
    """Compute v^T Q v where Q = sum of shift matrices for given lags."""
    return v @ qv_product(v, lags, n)


def solve_qp_active_set(a, S, lags, n, max_iters=100, tol=1e-8):
    """Solve min a^T Q a / 2 s.t. sum(a) = S, a >= 0, using projected gradient.
    
    Q = sum of shift matrices for the given lags.
    Returns the solution.
    """
    x = a.copy().astype(np.float64)
    x = project_simplex(x, S)
    
    best_obj = q_form(x, lags, n)
    best_x = x.copy()
    
    step = 1.0
    for it in range(max_iters):
        grad = qv_product(x, lags, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < tol:
            break
        
        # Backtracking line search
        for bt in range(20):
            x_trial = project_simplex(x - step * grad, S)
            obj_trial = q_form(x_trial, lags, n)
            if obj_trial <= best_obj - 1e-10 * step * gnorm**2:
                break
            step *= 0.5
        else:
            step = 1e-10
            x_trial = project_simplex(x - step * grad, S)
        
        if obj_trial < best_obj:
            best_obj = obj_trial
            best_x = x_trial.copy()
            x = x_trial
        else:
            # Restart with smaller step
            step *= 0.5
        
        # Reset step periodically
        if it % 20 == 19:
            step = 2.0 / max(qv_product(x, lags, n + int(max(lags)) + 1).max(), 1e-10) if max(lags) > 0 else 1.0
    
    return best_x


def run_fista_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=2000.0,
                   qp_every=200, qp_lags=5):
    """FISTA with LSE approximation + periodic QP active-set steps."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = beta_init * 4.0  # Lipschitz estimate (conservative)
    stall_count = 0
    last_improve = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive beta: increasing schedule
        frac = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_init * (beta_max / beta_init) ** frac
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stall_count = 0
            last_improve = cur
        elif it % 100 == 0:
            if cur < last_improve - 1e-8:
                last_improve = cur
                stall_count = 0
            else:
                stall_count += 1

        # --- QP Active Set Step (periodic) ---
        if it % qp_every == 0:
            top_lags = np.argsort(c)[::-1][:qp_lags]
            # Solve QP for these lags
            a_qp = solve_qp_active_set(y, S, top_lags.tolist(), n, max_iters=50)
            c_qp = fft_convolve_self(a_qp)
            if c_qp.max() < cur:
                y = a_qp
                x_qp = a_qp
                a = x_qp
                t_momentum = 1.0
                # Update best
                if c_qp.max() < best_score:
                    best_score = c_qp.max()
                    best_local = a_qp.copy()
                continue

        # --- Standard FISTA Step ---
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / L
        f_y = c.max()
        x_new = None
        for bt in range(5):
            x_candidate = project_simplex(y - step * grad, S)
            if bt > 0:
                c_cand = fft_convolve_self(x_candidate)
                f_cand = c_cand.max()
                if f_cand <= f_y - 1e-4 * step * gnorm * gnorm:
                    x_new = x_candidate
                    break
                step *= 0.5
            else:
                x_new = x_candidate

        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        # Adaptive L
        c_x = fft_convolve_self(x_new)
        if c_x.max() > f_y:
            L *= 1.5
        elif it % 50 == 0:
            L = max(L * 0.9, beta * 2.0)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Restart on stall
        if stall_count > 50 and rng.random() < 0.2:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_momentum = 1.0
            stall_count = 0

    return best_local, best_score


def run_qp_dedicated(a_init, n, S, max_time, rng):
    """Dedicated QP active-set optimization: focus entirely on top lags."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Get top lags (adaptive number)
        n_lags = min(10, 3 + int(7 * (time.monotonic() - start) / max_time))
        top_lags = np.argsort(c)[::-1][:n_lags].tolist()
        
        # Check if the solution is already optimal for this active set
        # (i.e., all non-active lags <= max active lag)
        
        # Solve QP
        a_new = solve_qp_active_set(a, S, top_lags, n, max_iters=200)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() < cur:
            a = a_new
        else:
            # Try with fewer lags
            if n_lags > 2:
                top_lags_few = np.argsort(c)[::-1][:max(2, n_lags//2)].tolist()
                a_new2 = solve_qp_active_set(a, S, top_lags_few, n, max_iters=100)
                c_new2 = fft_convolve_self(a_new2)
                if c_new2.max() < cur:
                    a = a_new2
                else:
                    # Perturb and continue
                    noise = 0.002 * S / np.sqrt(n)
                    a = project_simplex(a + rng.normal(0, noise, n), S)
                    
        # Periodic global check
        if int(time.monotonic() - start) % 30 < 1 and rng.random() < 0.1:
            # Try a slightly different active set
            alt_lags = np.argsort(c)[::-1][:3 + rng.integers(5)].tolist()
            a_alt = solve_qp_active_set(a, S, alt_lags, n, max_iters=50)
            c_alt = fft_convolve_self(a_alt)
            if c_alt.max() < best_score:
                best_score = c_alt.max()
                best_local = a_alt.copy()
                a = a_alt

    return best_local, best_score


def coordinate_descent_polish(a, n, S, max_time, rng):
    """Coordinate descent: move mass between pairs to reduce peak convolution."""
    a = np.clip(a.copy(), 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Find the peak lag and its contributing pairs
        peak_lag = int(np.argmax(c))
        
        # Find pairs contributing to this lag
        i_min = max(0, peak_lag - (n - 1))
        i_max = min(peak_lag, n - 1)
        indices = np.arange(i_min, i_max + 1)
        partners = peak_lag - indices
        products = a[indices] * a[partners]
        
        # Try moving mass from high-product pairs to low-mass positions
        top_pair_count = min(20, len(indices))
        top_pairs = np.argsort(products)[::-1][:top_pair_count]
        
        improved = False
        for pi in top_pairs:
            i = indices[pi]
            j = partners[pi]
            
            # Try transferring a small amount from the larger to the smaller
            if a[i] > a[j]:
                donor = i
                receiver = j
            else:
                donor = j
                receiver = i
            
            # Try a few transfer amounts
            for frac in [0.01, 0.05, 0.1, 0.2]:
                amt = frac * a[donor]
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[donor] -= amt
                a_trial[receiver] += amt
                a_trial = project_simplex(a_trial, S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur:
                    a = a_trial
                    cur = c_trial.max()
                    improved = True
                    break
        
        if not improved and rng.random() < 0.3:
            # Try random coordinate swaps
            for _ in range(10):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                amt = 0.01 * S / n
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < 0:
                    continue
                a_trial = project_simplex(a_trial, S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur:
                    a = a_trial
                    break

    return best_local, best_score


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.05
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_v_shape_init(n, rng, S):
    """V-shaped: minimum in the middle, high at edges."""
    tt = np.linspace(0, 1, n)
    mid = n // 2
    depth = rng.random() * 0.5 + 0.2
    a = 1.0 + depth * np.abs(2 * tt - 1)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_inverse_v_init(n, rng, S):
    """Inverse-V: peak in the middle."""
    tt = np.linspace(0, 1, n)
    depth = rng.random() * 0.5 + 0.2
    a = 1.0 + depth * np.exp(-((tt - 0.5) ** 2) / (2 * 0.1 ** 2))
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_periodic_init(n, rng, S, num_freqs=3):
    """Periodic with multiple frequencies."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 30)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.3
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_exponential_init(n, rng, S):
    """Exponential decay from both ends (high at edges)."""
    tt = np.linspace(0, 1, n)
    rate = rng.random() * 5 + 1
    a = np.exp(rate * (np.abs(tt - 0.5) - 0.5))
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def safe_normalize(a, n):
    """Clamp and rescale."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def get_objective(a):
    """Compute the objective value 2*n*max_b/sum_a^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)
```

```python
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def hvp_shift(v, k, n):
    """Compute (E_k @ v) where E_k[i,j] = 1 iff i+j=k.
    
    (E_k @ v)[j] = v[k-j] if 0 <= k-j < n, else 0.
    Valid j range: j_min = max(0, k-(n-1)), j_max = min(k, n-1).
    """
    j_min = max(0, k - (n - 1))
    j_max = min(k, n - 1)
    if j_min > j_max:
        return np.zeros(n)
    lo = k - j_max  # = max(0, k-n+1)
    hi = k - j_min  # = min(k, n-1)
    result = np.zeros(n)
    result[j_min:j_max + 1] = v[lo:hi + 1][::-1]
    return result


def qv_product(v, lags, n):
    """Compute (sum_{k in lags} E_k) @ v efficiently."""
    result = np.zeros(n)
    for k in lags:
        k = int(k)
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        if j_min <= j_max:
            lo = k - j_max
            hi = k - j_min
            result[j_min:j_max + 1] += v[lo:hi + 1][::-1]
    return result


def q_form(v, lags, n):
    """Compute v^T Q v where Q = sum of shift matrices."""
    return v @ qv_product(v, lags, n)


def solve_qp_projected(a, S, lags, n, max_iters=200, verbose=False):
    """Solve min a^T Q a s.t. sum(a)=S, a>=0 via projected gradient with backtracking.
    
    Q = sum of shift matrices for given lags.
    Gradient of objective = 2 * Q @ a.
    """
    x = np.clip(a.copy(), 0.0, None)
    s = x.sum()
    if s < 1e-15:
        x = np.ones(n) * S / n
    else:
        x = x * (S / s)
    
    best_obj = q_form(x, lags, n)
    best_x = x.copy()
    step = 1.0
    stall = 0
    
    for it in range(max_iters):
        grad = 2.0 * qv_product(x, lags, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        accepted = False
        for bt in range(30):
            x_trial = project_simplex(x - step * grad, S)
            obj_trial = q_form(x_trial, lags, n)
            if obj_trial <= best_obj - 1e-12 * step * gnorm * gnorm:
                x = x_trial
                if obj_trial < best_obj:
                    best_obj = obj_trial
                    best_x = x_trial.copy()
                    stall = 0
                else:
                    stall += 1
                accepted = True
                break
            step *= 0.5
        
        if not accepted:
            stall += 1
            step = max(step, 1e-15)
    
        if stall > 30:
            step = 2.0 / max(gnorm, 1e-10) * 0.1
            stall = 0
        
        if verbose and it % 50 == 0:
            print("    QP it %d: obj=%.8f, gnorm=%.4e" % (it, best_obj, gnorm))
    
    return best_x


def run_fista_hybrid(a_init, n, S, max_time, rng,
                     beta_init=5.0, beta_end=800.0,
                     qp_interval=150, qp_k=5, qp_iters=80):
    """FISTA with LSE + periodic QP active-set steps.
    
    Beta follows a cosine annealing schedule from beta_init to beta_end.
    Every qp_interval iterations, solve a QP for the top qp_k lags.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = beta_init * 4.0
    stall_count = 0
    last_improve = np.inf

    total_beta_range = beta_end - beta_init

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_init + total_beta_range * (1 - np.cos(np.pi * frac)) / 2.0

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stall_count = 0
            last_improve = cur
        elif it % 50 == 0:
            if cur < last_improve - 1e-8:
                last_improve = cur
                stall_count = 0
            else:
                stall_count += 1

        # Periodic QP step
        if it % qp_interval == 0:
            top_lags = np.argsort(c)[::-1][:qp_k].tolist()
            a_qp = solve_qp_projected(y, S, top_lags, n, max_iters=qp_iters)
            c_qp = fft_convolve_self(a_qp)
            if c_qp.max() < cur:
                y = a_qp
                a = a_qp
                t_momentum = 1.0
                if c_qp.max() < best_score:
                    best_score = c_qp.max()
                    best_local = a_qp.copy()
                continue

        # FISTA step with LSE gradient
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / L
        f_y = c.max()
        x_new = None
        for bt in range(6):
            x_candidate = project_simplex(y - step * grad, S)
            if bt > 0:
                c_cand = fft_convolve_self(x_candidate)
                f_cand = c_cand.max()
                if f_cand <= f_y - 1e-4 * step * gnorm * gnorm:
                    x_new = x_candidate
                    break
                step *= 0.5
            else:
                x_new = x_candidate

        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        # Adaptive L
        c_x = fft_convolve_self(x_new)
        if c_x.max() > f_y:
            L *= 1.5
        elif it % 30 == 0:
            L = max(L * 0.92, beta * 2.0)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Restart on prolonged stall
        if stall_count > 40 and rng.random() < 0.25:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_momentum = 1.0
            stall_count = 0

    return best_local, best_score


def run_qp_iterative(a_init, n, S, max_time, rng):
    """Iterative active-set QP: solve QP for top-K, check all lags, update set."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Adaptive K: start small, grow over time
        frac = min(1.0, (time.monotonic() - start) / max_time)
        K = 3 + int(7 * frac)

        top_lags = np.argsort(c)[::-1][:K].tolist()

        # Solve QP for current active set
        a_new = solve_qp_projected(a, S, top_lags, n, max_iters=150)
        c_new = fft_convolve_self(a_new)

        if c_new.max() < cur:
            a = a_new
        else:
            # Try with top-3 only (more focused)
            top3 = np.argsort(c)[::-1][:3].tolist()
            a_new3 = solve_qp_projected(a, S, top3, n, max_iters=100)
            c_new3 = fft_convolve_self(a_new3)
            if c_new3.max() < cur:
                a = a_new3
            else:
                # Add the lag that increased the most
                c_old = c.copy()
                increased = c_new - c_old
                worst_new_lag = int(np.argmax(increased[increased > 0])) if np.any(increased > 0) else int(np.argmax(c_new))
                expanded_lags = top_lags + [worst_new_lag]
                # Remove duplicates
                expanded_lags = list(set(expanded_lags))[:K + 2]
                a_exp = solve_qp_projected(a, S, expanded_lags, n, max_iters=80)
                c_exp = fft_convolve_self(a_exp)
                if c_exp.max() < cur:
                    a = a_exp
                else:
                    # Small perturbation to escape
                    noise = 0.002 * S / np.sqrt(n)
                    a = project_simplex(a + rng.normal(0, noise, n), S)

    return best_local, best_score


def coordinate_polish(a_init, n, S, max_time, rng):
    """Coordinate descent: transfer mass between pairs to reduce peak convolution."""
    a = np.clip(a_init.copy(), 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    eval_count = 0

    while time.monotonic() - start < max_time:
        if eval_count % 5 == 0:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = a.copy()
            peak_lag = int(np.argmax(c))
        eval_count += 1

        # Find contributing pairs for peak lag
        i_min = max(0, peak_lag - (n - 1))
        i_max = min(peak_lag, n - 1)
        indices = np.arange(i_min, i_max + 1)
        partners = peak_lag - indices
        products = a[indices] * a[partners]

        # Target top-product pairs
        n_pairs = min(30, len(indices))
        top_pairs = np.argsort(products)[::-1][:n_pairs]

        improved = False
        for pi in top_pairs:
            i = int(indices[pi])
            j = int(partners[pi])
            if i == j:
                continue

            # Try transferring from larger to smaller
            if a[i] >= a[j]:
                donor, receiver = i, j
            else:
                donor, receiver = j, i

            for frac in [0.02, 0.08, 0.15]:
                amt = frac * a[donor]
                if amt < 1e-15 or a[donor] - amt < 0:
                    continue
                a_trial = a.copy()
                a_trial[donor] -= amt
                a_trial[receiver] += amt
                a_trial = project_simplex(a_trial, S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur:
                    a = a_trial
                    cur = c_trial.max()
                    improved = True
                    break

        # Also try random perturbations
        if not improved and rng.random() < 0.5:
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                amt = 0.005 * S / n
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial.min() < -1e-10:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = project_simplex(a_trial, S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur:
                    a = a_trial
                    cur = c_trial.max()
                    break

        # Time check
        if time.monotonic() - start > max_time:
            break

    return best_local, best_score


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.05
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_v_shape_init(n, rng, S):
    """V-shaped: high at edges, low in middle."""
    tt = np.linspace(0, 1, n)
    depth = rng.random() * 0.6 + 0.3
    a = 1.0 + depth * np.abs(2 * tt - 1)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_gaussian_bump_init(n, rng, S):
    """Gaussian bump in the center."""
    tt = np.linspace(0, 1, n)
    width = rng.random() * 0.15 + 0.05
    amp = rng.random() * 0.5 + 0.3
    a = 1.0 + amp * np.exp(-((tt - 0.5) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_periodic_init(n, rng, S, num_freqs=4):
    """Multi-frequency periodic."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = int(rng.integers(2, 40))
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.25
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_exponential_edge_init(n, rng, S):
    """Exponential decay from edges (high at boundaries)."""
    tt = np.linspace(0, 1, n)
    rate = rng.random() * 4 + 1
    a = np.exp(rate * (np.abs(tt - 0.5) - 0.5))
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def get_objective(a):
    """Compute 2*n*max_b/sum_a^2."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimize 2*n*max(conv(a,a))/sum(a)^2 using hybrid FISTA + QP active-set.
    
    Multi-phase strategy:
    Phase 1: Explore diverse initializations with FISTA + QP (55% time)
    Phase 2: Dedicated iterative QP on top results (25% time)
    Phase 3: Intensive polish - high-beta FISTA, QP, coordinate descent (20% time)
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best_obj = np.inf
    global_a = None

    # Try parent
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    n = 8192
    S = np.sqrt(2.0 * n)

    if parent is not None and len(parent) == n:
        print("Using parent of length %d" % n)
    elif parent is not None and len(parent) > n:
        # Downsample
        idx = np.linspace(0, len(parent) - 1, n).astype(int)
        parent = parent[idx]
        print("Downsampled parent to length %d" % n)
    elif parent is not None:
        # Upsample
        idx = np.linspace(0, len(parent) - 1, n).astype(int)
        parent = parent[idx]
        print("Upsampled parent to length %d" % n)
    else:
        print("No parent, using n=%d" % n)

    print("=== n=%d, S=%.4f, budget=%.0fs ===" % (n, S, budget_s))

    # Generate diverse initializations
    inits = []
    if parent is not None:
        p = parent * (S / parent.sum())
        inits.append(("parent", p))

    for i in range(3):
        inits.append(("anti_pal_%d" % i, make_anti_palindromic_init(n, rng, S)))
    for i in range(2):
        inits.append(("vshape_%d" % i, make_v_shape_init(n, rng, S)))
    for i in range(2):
        inits.append(("gauss_%d" % i, make_gaussian_bump_init(n, rng, S)))
    for i in range(2):
        inits.append(("periodic_%d" % i, make_periodic_init(n, rng, S)))
    for i in range(2):
        inits.append(("exp_edge_%d" % i, make_exponential_edge_init(n, rng, S)))
    rand_a = rng.random(n) + 0.05
    inits.append(("random", rand_a * (S / rand_a.sum())))

    n_inits = len(inits)
    usable_time = max(10.0, deadline - time.monotonic())

    # === Phase 1: Exploration ===
    phase1_time = usable_time * 0.50
    per_init_time = phase1_time / n_inits
    print("=== Phase 1: %d inits x %.1fs ===" % (n_inits, per_init_time))

    results = []
    for name, arr in inits:
        if time.monotonic() > deadline - 120:
            break
        t0 = time.monotonic()
        a_f, _ = run_fista_hybrid(arr, n, S, per_init_time, rng,
                                   beta_init=5.0, beta_end=400.0,
                                   qp_interval=100, qp_k=5, qp_iters=60)
        a_f = safe_normalize(a_f, n)
        obj_f = get_objective(a_f)
        results.append((obj_f, name, a_f.copy()))
        if obj_f < global_best_obj:
            global_best_obj = obj_f
            global_a = a_f.copy()
        print("  P1 [%s]: %.6f (%.0fs)" % (name, obj_f, time.monotonic() - t0))

    results.sort(key=lambda x: x[0])
    print("Phase 1 best: %.6f" % global_best_obj)

    # === Phase 2: Dedicated QP on top-3 ===
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        top3 = results[:3]
        per_qp = remaining * 0.30 / max(1, len(top3))
        print("=== Phase 2: QP on top-3, %.1fs each ===" % per_qp)
        for obj_r, name_r, a_r in top3:
            if time.monotonic() > deadline - 60:
                break
            t0 = time.monotonic()
            a_q, _ = run_qp_iterative(a_r, n, S, per_qp, rng)
            a_q = safe_normalize(a_q, n)
            obj_q = get_objective(a_q)
            results.append((obj_q, "qp_" + name_r, a_q.copy()))
            if obj_q < global_best_obj:
                global_best_obj = obj_q
                global_a = a_q.copy()
                print("  P2 QP [%s]: %.6f (%.0fs)" % (name_r, obj_q, time.monotonic() - t0))

    # === Phase 3: Intensive polish ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        print("=== Phase 3: Polish (%.0fs remaining) ===" % remaining)

        # 3a: High-beta FISTA with frequent QP
        seg = max(5.0, remaining * 0.25)
        a_p, _ = run_fista_hybrid(global_a, n, S, seg, rng,
                                   beta_init=200.0, beta_end=3000.0,
                                   qp_interval=50, qp_k=3, qp_iters=100)
        a_p = safe_normalize(a_p, n)
        obj_p = get_objective(a_p)
        if obj_p < global_best_obj:
            global_best_obj = obj_p
            global_a = a_p.copy()
            print("  3a FISTA hi-beta: %.6f" % obj_p)

        # 3b: Dedicated QP
        remaining2 = deadline - time.monotonic()
        seg2 = max(5.0, remaining2 * 0.25)
        a_q, _ = run_qp_iterative(global_a, n, S, seg2, rng)
        a_q = safe_normalize(a_q, n)
        obj_q = get_objective(a_q)
        if obj_q < global_best_obj:
            global_best_obj = obj_q
            global_a = a_q.copy()
            print("  3b QP: %.6f" % obj_q)

        # 3c: Coordinate descent polish
        remaining3 = deadline - time.monotonic()
        seg3 = max(5.0, remaining3 * 0.20)
        a_c, _ = coordinate_polish(global_a, n, S, seg3, rng)
        a_c = safe_normalize(a_c, n)
        obj_c = get_objective(a_c)
        if obj_c < global_best_obj:
            global_best_obj = obj_c
            global_a = a_c.copy()
            print("  3c CoordDesc: %.6f" % obj_c)

        # 3d: Final high-beta FISTA
        remaining4 = deadline - time.monotonic()
        if remaining4 > 10:
            a_f2, _ = run_fista_hybrid(global_a, n, S, remaining4 * 0.85, rng,
                                        beta_init=1000.0, beta_end=5000.0,
                                        qp_interval=30, qp_k=3, qp_iters=150)
            a_f2 = safe_normalize(a_f2, n)
            obj_f2 = get_objective(a_f2)
            if obj_f2 < global_best_obj:
                global_best_obj = obj_f2
                global_a = a_f2.copy()
                print("  3d Final FISTA: %.6f" % obj_f2)

    # Fallback
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    # Final validation
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL: evaluator=%.6f, internal=%.6f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best_obj, len(global_a)))

    if not np.isfinite(ver):
        print("WARNING: evaluator rejected! Using fallback.")
        final_list = [1.0] * 1024
        ver = evaluate_sequence(final_list)
        print("Fallback: %.6f" % ver)

    return final_list
```
