def project_simplex(v, S):
    n = len(v)

    v = np.clip(v, 0.0, 1000.0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):

    theta = (cumsum[rho - 1] - S) / rho

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization combining FFT-based LSE gradient FISTA,
top-k subgradient with FFT-accelerated combined gradients, single-lag
fast subgradient polishing, and Polyak averaging. Multiple n values explored.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.maximum(v, 0.0)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Ensure exact sum (numerical fix)
    diff = S - result.sum()
    if abs(diff) > 1e-12:
        # Distribute among positive entries
        pos = result > 0
        if pos.any():
            result[pos] += diff / pos.sum()
            result = np.maximum(result, 0.0)
    return result


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_weights(c, beta, top_k=None):
    """Compute LSE weights for convolution values."""
    cm = c.max()
    if top_k is not None and top_k < len(c):
        # Focus only on top-k
        thresh = np.sort(c)[::-1][min(top_k, len(c))-1]
        c_masked = np.where(c >= thresh, c, -np.inf)
    else:
        c_masked = c.copy()
    e = np.exp(beta * (c_masked - cm))
    tot = e.sum()
    if tot < 1e-300:
        # Fallback: one-hot at argmax
        e = np.zeros(len(c))
        e[int(np.argmax(c))] = 1.0
    else:
        e = e / tot
    return e


def fft_cross_correlate(a, h):
    """Compute cross-correlation: result[i] = sum_j a[j] * h[i+j].
    a has length n, h has length 2n-1.
    result[i] = sum_{j=0}^{n-1} a[j] * h[i+j] (where h[i+j] is valid).
    """
    n = len(a)
    hlen = len(h)
    # result[i] = sum_j a[j] * h[i+j] for i+j in [0, hlen-1]
    # This is a cross-correlation of a and h
    # Cross-correlation: (a ⋆ h)[i] = sum_j a[j] * conj(h[j-i])
    # But we want: sum_j a[j] * h[i+j] = sum_j a[j] * h[i+j]
    # Let's define: for each i, sum over j where 0<=j<n and 0<=i+j<hlen
    # This is like a "sliding dot product"
    # = (a flipped) * h, i.e., convolution of flip(a) with h
    # flip(a)[j] = a[n-1-j]
    # conv(flip(a), h)[i] = sum_j flip(a)[j] * h[i-j] = sum_j a[n-1-j] * h[i-j]
    # Let j' = n-1-j: = sum_{j'} a[j'] * h[i-n+1+j']
    # So conv(flip(a), h)[i] = sum_{j'} a[j'] * h[(i-n+1)+j'] = result[i-n+1]
    # Therefore: result[i] = conv(flip(a), h)[i+n-1]
    
    fsize = 1 << (n + hlen - 1 - 1).bit_length()
    a_flip = a[::-1]
    A = fft(a_flip, fsize)
    H = fft(h, fsize)
    conv = np.real(ifft(A * H))
    return conv[n-1:n-1+n]  # result[i] for i=0,...,n-1


def run_lse_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_rate=0.005,
                  beta_max=1000.0, top_k=None, use_averaging=True, restart_interval=3000):
    """FISTA with LSE gradient and optional Polyak averaging."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    # Polyak averaging
    avg_a = a.copy()
    avg_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
        
        # Update Polyak average
        if use_averaging:
            avg_count += 1
            avg_a = avg_a + (y - avg_a) / avg_count
            c_avg = fft_convolve_self(avg_a)
            if c_avg.max() < best:
                best = c_avg.max()
                best_a = avg_a.copy()
        
        # Compute LSE weights
        w = lse_weights(c, beta, top_k=top_k)
        
        # Compute gradient via cross-correlation
        grad = fft_cross_correlate(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size
        step = 1.0 / (2.0 * beta * gnorm)  # Normalized step
        # Actually, for FISTA we need a Lipschitz-based step. Let me use a simpler approach.
        # The LSE function has Lipschitz constant ~ beta * ||w * a||_inf (roughly)
        # Let's use a more conservative step:
        step = 1.0 / (beta * gnorm)
        
        x_new = project_simplex(y - step * grad, S)
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-project to ensure constraint
        y = project_simplex(y, S)
        a = x_new
        t = t_new
        
        # Periodic restart
        if it % restart_interval == 0:
            decay = np.exp(-it / (10 * restart_interval))
            noise_scale = 0.005 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                noise = rng.normal(0, noise_scale, n)
                y = project_simplex(y + noise, S)
                t = 1.0
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_single_lag_subgrad(a_init, n, S, max_time, rng, step_init=0.1, step_decay=0.9999):
    """Fast subgradient method focusing on the single max lag.
    Exploits the structure: gradient at max lag k* is 2*a[k*-i] (shifted/reversed a)."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    step = step_init
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
            last_improve = it
        
        if it - last_improve > 5000:
            # Stuck, add noise
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            last_improve = it
            step = step_init * 0.5
        
        k_star = int(np.argmax(c))
        # Gradient: g[i] = 2*a[k_star - i] for 0 <= k_star - i < n
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        idx = np.arange(lo, hi + 1)
        g = np.zeros(n)
        g[idx] = 2.0 * a[k_star - idx]
        
        # Take step
        a_new = a - step * g / np.linalg.norm(g)
        a = project_simplex(a_new, S)
        step *= step_decay
    
    return best_a, best


def run_topk_subgrad(a_init, n, S, max_time, rng, k_top=100, step_init=0.05):
    """Subgradient method using top-k lags with FFT-accelerated combined gradient."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    step = step_init
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
            last_improve = it
        
        if it - last_improve > 3000:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            last_improve = it
        
        # Get top-k lags
        k_top_actual = min(k_top, len(c))
        top_idx = np.argsort(c)[::-1][:k_top_actual]
        
        # Binary indicator of top k
        h = np.zeros(len(c))
        h[top_idx] = 1.0
        
        # Combined gradient via cross-correlation
        grad = fft_cross_correlate(a, h)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        a_new = a - step * grad / gnorm
        a = project_simplex(a_new, S)
        step *= 0.99995
    
    return best_a, best

    if given is not None and 1024 <= len(given) <= 16384:

    else:

    if n_main > 4096:
