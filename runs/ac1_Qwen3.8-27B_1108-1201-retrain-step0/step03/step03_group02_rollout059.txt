        if it % 100 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
    
    c_final = fft_convolve_self(best_local)
    return best_local, c_final.max()


def coordinate_descent(a_init, n, S, max_time, rng):
    """
    Coordinate descent: pick pairs (j, m), optimize 1D sub-problem.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    
    it = 0
    stagnant = 0
    max_stagnant = 200
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Find peak position
        k_star = int(np.argmax(c))
        
        # Choose pair (j, m)
        # Strategy: pick j such that it contributes to the peak
        # conv[k_star] = sum_i a[i]*a[k_star-i]
        # The "biggest" terms are where a[i]*a[k_star-i] is large
        
        # Method: pick j from indices with small a[j] (room to grow)
        # and set m = k_star - j (if valid)
        
        found = False
        for _ in range(20):
            j = rng.integers(0, n)
            m = k_star - j
            if 0 <= m < n and m != j:
                found = True
                break
        
        if not found:
            j = rng.integers(0, n)
            m = rng.integers(0, n)
            if m == j:
                m = (m + 1) % n
        
        T = a[j] + a[m]
        if T < 1e-15:
            continue
        
        # 1D optimization: find best a[j] in [0, T]
        # Use coarse grid + golden section refinement
        best_aj = a[j]
        best_obj = best_score
        
        # Coarse grid
        n_coarse = 20
        grid = np.linspace(0, T, n_coarse)
        # Add current value
        grid = np.unique(np.sort(np.concatenate([grid, [a[j]]])))
        
        for aj in grid:
            old_j = a[j]
            old_m = a[m]
            a[j] = aj
            a[m] = T - aj
            
            # Compute new max (quick: use delta)
            # Full conv for safety
            c_new = fft_convolve_self(a)
            obj = c_new.max()
            
            if obj < best_obj:
                best_obj = obj
                best_aj = aj
        
        # Apply best
        a[j] = best_aj
        a[m] = T - best_aj
        
        # Recompute
        c = fft_convolve_self(a)
        if c.max() < best_score:
            best_score = c.max()
            best_local = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Escape from local min
        if stagnant > max_stagnant:
            # Random perturbation
            n_pert = max(5, n // 100)
            idx = rng.integers(0, n, size=n_pert)
            a[idx] += rng.normal(0, 0.05 * S / np.sqrt(n), size=n_pert)
            a = np.maximum(a, 0.0)
            s2 = a.sum()
            if s2 > 1e-15:
                a = a * (S / s2)
            c = fft_convolve_self(a)
            stagnant = 0
            max_stagnant = min(max_stagnant * 2, 1000)
    
    return best_local, best_score

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing
    2*n*max(conv(a,a)) / (sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)
    
    global_best = np.inf
    global_a = None
    
    # --- Try to use parent sequence ---
    parent_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 512 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            parent_a = np.clip(given, 0, 1000)
            # Normalize to sum = sqrt(2n)
            n_p = len(parent_a)
            S_p = np.sqrt(2.0 * n_p)
            parent_a = parent_a * (S_p / parent_a.sum())
            obj_p = compute_objective_normalized(parent_a)
            print(f"Parent: n={n_p}, obj={obj_p:.6f}")
            if obj_p < global_best:
                global_best = obj_p
                global_a = parent_a.copy()
    except Exception:
        pass
    
    # --- Determine n and generate initializations ---
    if parent_a is not None:
        n_default = len(parent_a)
    else:
        n_default = 8192
    
    # Try multiple n values
    n_values = [4096, 8192, 16384]
    if parent_a is not None and n_default not in n_values:
        n_values.insert(0, n_default)
    
    # For each n, create geometric initializations
    all_inits = []
    
    for n in n_values:
        tt = np.linspace(0, 1, n, endpoint=False)
        
        # Geometric with various rates (the key family)
        for c in [0.8, 1.0, 1.2, 1.25, 1.3, 1.5, 2.0, 3.0]:
            r = np.exp(c / n)
            all_inits.append((f"geom_c{c}_n{n}", np.power(r, np.arange(n))))
            # Reverse
            all_inits.append((f"geomr_c{c}_n{n}", np.power(r, n - 1 - np.arange(n))))
        
        # Power-law
        for p in [0.5, 1.0, 1.5, 2.0]:
            all_inits.append((f"pow_{p}_n{n}", (tt + 0.01) ** p))
            all_inits.append((f"powr_{p}_n{n}", (1 - tt + 0.01) ** p))
        
        # Exponential (different parameterization)
        for alpha in [1.0, 2.0, 4.0, 8.0]:
            all_inits.append((f"exp_{alpha}_n{n}", np.exp(alpha * tt)))
            all_inits.append((f"expr_{alpha}_n{n}", np.exp(alpha * (1 - tt))))
    
    # Add parent if available
    if parent_a is not None:
        all_inits.insert(0, ("parent", parent_a))
    
    n_inits = len(all_inits)
    print(f"Total initializations: {n_inits}")
    
    # --- Phase 1: Quick screening (evaluate objective of each init) ---
    print("=== Phase 1: Screening initializations ===")
    scored_inits = []
    for name, arr in all_inits:
        n_i = len(arr)
        S_i = np.sqrt(2.0 * n_i)
        arr_norm = np.clip(arr, 0, 1000) * (S_i / max(arr.sum(), 1e-15))
        obj = compute_objective_normalized(arr_norm)
        scored_inits.append((obj, name, arr_norm))
        if obj < global_best:
            global_best = obj
            global_a = arr_norm.copy()
    
    # Sort by objective
    scored_inits.sort(key=lambda x: x[0])
    print(f"  Top 5: {[f'{s[1]}={s[0]:.4f}' for s in scored_inits[:5]]}")
    print(f"  Best after screening: {global_best:.6f}")
    
    # --- Phase 2: FISTA on top candidates ---
    usable_time = max(10.0, deadline - time.monotonic())
    
    # Select top candidates (by diversity, not just score)
    selected = []
    selected_ns = set()
    for obj, name, arr in scored_inits:
        n_i = len(arr)
        if n_i in selected_ns:
            continue
        selected.append((obj, name, arr))
        selected_ns.add(n_i)
        if len(selected) >= 6:
            break
    
    # Also add the overall best
    if len(selected) > 0:
        best_entry = scored_inits[0]
        if best_entry not in selected:
            selected.append(best_entry)
    
    n_selected = len(selected)
    fista_time_each = usable_time * 0
