"""

Strategy:

- Targeted peak reduction via coordinate-level redistribution
- Multi-scale exploration (n=8192 primary, n=16384 secondary)

- KKT-inspired active set tracking for focused optimization

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def lse_value(c, beta):
    """Smooth approximation to max(c)."""
    c_max = c.max()
    return c_max + np.log(np.sum(np.exp(beta * (c - c_max)))) / beta


def topk_subgradient(a, c, top_k):
    """Subgradient from top-k lags with uniform weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    w_full = np.zeros(2 * n - 1)
    w_full[top_indices] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_subgradient(a, c, temp):
    """Subgradient using softmax weighting over all lags."""
    n = len(a)
    c_max = c.max()
    w = np.exp((c - c_max) / temp)
    w = w / w.sum()
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = mu."""
    half = n // 2
    h = rng.random(half) + 0.05
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())
    return a

    elif mode == 'bimodal':

    elif mode == 'random':

def run_lse_fista(a_init, n, S, max_time, rng, beta_init=1.0, beta_final=2000.0,
                  n_beta_steps=8, momentum_decay=False):
    """
    LSE-FISTA with staged beta annealing.
    Each stage: run FISTA for a fixed beta, then increase beta.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    # Create beta schedule
    betas = np.geomspace(beta_init, beta_final, n_beta_steps)

    for stage, beta in enumerate(betas):
        stage_time = max_time * (0.4 + 0.15 * stage / n_beta_steps) / n_beta_steps
        stage_deadline = start + stage_time
        # Last stages get more time
        if stage >= n_beta_steps - 3:
            stage_deadline = start + max_time * 0.9

        y = a.copy()
        t_mom = 1.0
        L = 2.0 * beta * S / n * 1.5  # Initial Lipschitz estimate
        stage_it = 0
        stall = 0
        best_stage_score = np.inf

        while time.monotonic() < stage_deadline and time.monotonic() < start + max_time:
            stage_it += 1

            c = fft_convolve_self(y)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = y.copy()
            if cur < best_stage_score:
                best_stage_score = cur
                stall = 0
            else:
                stall += 1

            grad = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break

            # Backtracking line search
            step = 1.0 / L
            f_y = lse_value(c, beta)
            x_new = None
            for bt in range(4):
                x_cand = project_simplex(y - step * grad, S)
                if bt == 0:
                    x_new = x_cand
                    break
                else:
                    c_cand = fft_convolve_self(x_cand)
                    f_cand = lse_value(c_cand, beta)
                    if f_cand <= f_y - 1e-4 * step * gnorm * gnorm:
                        x_new = x_cand
                        break
                    step *= 0.5

            if x_new is None:
                x_new = project_simplex(y - step * grad, S)

            # Update Lipschitz
            c_x = fft_convolve_self(x_new)
            if c_x.max() > c.max() * 1.01:
                L *= 1.3
            elif stage_it % 30 == 0:
                L = max(L * 0.97, 2.0 * beta * S / n)

            # FISTA momentum
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t_mom = t_new

            # Stall detection: add perturbation
            if stall > 40 and rng.random() < 0.2:
                noise_scale = 0.005 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                stall = 0

        # End of stage
        if best_score < np.inf:
            pass  # Track globally

    return best_local, best_score


def run_peaking_reducer(a_init, n, S, max_time, rng):
    """
    Targeted peak reduction: identify contributing pairs for the top lag(s)
    and redistribute mass to reduce them.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() < start + max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Find top active lags
        top_lags = np.argsort(c)[::-1][:5]
        threshold = c[top_lags[-1]] * 0.95
        active_lags = top_lags[c[top_lags] >= threshold]

        # Build combined subgradient from active lags
        n_active = len(active_lags)
        if n_active == 0:
            break

        # Weight lags by how close they are to max
        lag_weights = np.exp((c[active_lags] - c[active_lags[0]]) * 5.0)
        lag_weights = lag_weights / lag_weights.sum()

        # Compute weighted subgradient
        w_full = np.zeros(2 * n - 1)
        for i, lag in enumerate(active_lags):
            w_full[int(lag)] = lag_weights[i]
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step
        step = 1.0 / (np.max(np.abs(grad)) * 0.5 + 1e-10)
        step = min(step, S * 0.1 / (n * 0.01 + gnorm))

        x_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(x_new)
        if c_new.max() < cur:
            a = x_new
        else:
            # Try smaller step
            x_new2 = project_simplex(a - step * 0.5 * grad, S)
            c_new2 = fft_convolve_self(x_new2)
            if c_new2.max() < cur:
                a = x_new2
            else:
                # Try coordinate-level fix
                # Find the coordinate with highest gradient and reduce it
                max_grad_idx = np.argmax(grad)
                min_grad_idx = np.argmin(grad)
                delta = 0.01 * S / n
                if a[max_grad_idx] > delta:
                    a[max_grad_idx] -= delta
                    a[min_grad_idx] += delta
                    a = np.maximum(a, 0.0)
                    a = a * (S / a.sum())
                    c_chk = fft_convolve_self(a)
                    if c_chk.max() >= cur * 1.01:
                        # Revert
                        a[max_grad_idx] += delta
                        a[min_grad_idx] -= delta
                        a = np.maximum(a, 0.0)
                        a = a * (S / a.sum())

        # Periodic structured perturbation
        if it % 200 == 0 and rng.random() < 0.1:
            freq = rng.integers(3, 20)
            tt = np.arange(n) / n
            pert = 0.003 * S * np.sin(2 * np.pi * freq * tt + rng.random() * 6.28)
            a = project_simplex(a + pert, S)

    return best_local, best_score


def run_alternating_blocks(a_init, n, S, max_time, rng, block_size=256):
    """
    Alternating block optimization: divide sequence into blocks,
    optimize each block while holding others fixed.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    n_blocks = n // block_size
    if n_blocks < 2:
        block_size = max(n // 16, 16)
        n_blocks = n // block_size

    it = 0
    while time.monotonic() < start + max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Pick the block most involved in the peak lag
        peak_lag = int(c.argmax())
        # Which block indices contribute to this lag?
        i_min = max(0, peak_lag - (n - 1))
        i_max = min(peak_lag, n - 1)
        # Find the block with most contributing pairs
        block_contrib = np.zeros(n_blocks)
        for b in range(n_blocks):
            b_start = b * block_size
            b_end = min(b_start + block_size, n)
            # Count how many indices in this block are in [i_min, i_max]
            overlap_start = max(b_start, i_min)
            overlap_end = min(b_end, i_max + 1)
            if overlap_start < overlap_end:
                block_contrib[b] = overlap_end - overlap_start

        # Optimize the most-contributing block
        target_block = int(np.argmax(block_contrib))
        b_start = target_block * block_size
        b_end = min(b_start + block_size, n)

        # Do LSE optimization on this block only
        block_a = a[b_start:b_end].copy()
        rest_sum = S - block_a.sum()

        for _ in range(20):
            c_full = fft_convolve_self(a)
            grad_full = lse_gradient(a, c_full, beta=50.0)
            grad_block = grad_full[b_start:b_end]

            if np.linalg.norm(grad_block) < 1e-10:
                break

            step = 0.5 / (np.max(np.abs(grad_block)) + 1e-10)
            block_new = project_simplex(block_a - step * grad_block,
                                        max(block_a.sum(), 1e-10))
            # Re-normalize block to maintain total sum
            if block_new.sum() > 1e-10:
                block_new = block_new * (block_a.sum() / block_new.sum())
            a[b_start:b_end] = block_new

        a = np.maximum(a, 0.0)
        a = a * (S / a.sum())

    return best_local, best_score


def run_spectral_nudge(a_init, n, S, max_time, rng):
    """
    Use frequency domain insight: the peak of autocorrelation relates to
    concentration of energy in frequency domain. Nudge the spectrum to
    be more uniform (which reduces time-domain peak).
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    # FFT size for circular approximation
    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() < start + max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Compute power spectrum
        A = fft(a, fsize)
        power = np.abs(A) ** 2

        # The DC component (index 0) determines the sum, don't touch it
        # Nudge non-DC components toward uniform
        target_power = power[0] / fsize  # Approximate uniform level
        alpha = 0.01  # Mixing parameter

        # Create nudged spectrum: blend current toward flat
        A_nudged = A.copy()
        for i in range(1, fsize // 2 + 1):
            if i < fsize - i:  # Avoid double-counting conjugate
                # Move toward target
                current_mag = np.abs(A[i])
                target_mag = np.sqrt(max(target_power, 1e-20))
                phase = np.angle(A[i]) if current_mag > 1e-15 else 0.0
                new_mag = (1 - alpha) * current_mag + alpha * target_mag
                A_nudged[i] = new_mag * np.exp(1j * phase)
                conj_idx = fsize - i
                if conj_idx < fsize:
                    A_nudged[conj_idx] = np.conj(A_nudged[i])

        # Inverse FFT to get time domain
        a_nudged = np.real(ifft(A_nudged, fsize))[:n]
        a_nudged = np.maximum(a_nudged, 0.0)
        if a_nudged.sum() < 1e-10:
            continue
        a_nudged = a_nudged * (S / a_nudged.sum())

        c_nudged = fft_convolve_self(a_nudged)
        if c_nudged.max() < cur:
            a = a_nudged
        else:
            # Try with different alpha
            for alpha2 in [0.005, 0.02, 0.05]:
                A_nudged2 = A.copy()
                for i in range(1, fsize // 2 + 1):
                    if i < fsize - i:
                        current_mag = np.abs(A[i])
                        target_mag = np.sqrt(max(target_power, 1e-20))
                        phase = np.angle(A[i]) if current_mag > 1e-15 else 0.0
                        new_mag = (1 - alpha2) * current_mag + alpha2 * target_mag
                        A_nudged2[i] = new_mag * np.exp(1j * phase)
                        conj_idx = fsize - i
                        if conj_idx < fsize:
                            A_nudged2[conj_idx] = np.conj(A_nudged2[i])
                a_try = np.real(ifft(A_nudged2, fsize))[:n]
                a_try = np.maximum(a_try, 0.0)
                if a_try.sum() < 1e-10:
                    continue
                a_try = a_try * (S / a_try.sum())
                c_try = fft_convolve_self(a_try)
                if c_try.max() < cur:
                    a = a_try
                    break

    return best_local, best_score


def objective_from_max_b(max_b, n):
    """Compute objective value from max_b and n (with sum = sqrt(2n))."""
    S = np.sqrt(2.0 * n)
    return 2.0 * n * max_b / (S * S)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None
    global_n = 0

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Parent available: %s, length: %s" % (parent is not None,
          len(parent) if parent is not None else 'N/A'))

    # Determine primary length
    if parent is not None and len(parent) >= 64:
        primary_n = len(parent)
    else:
        primary_n = 8192

    print("Primary n = %d" % primary_n)
    S_primary = np.sqrt(2.0 * primary_n)

    def update_best(arr, name):
        nonlocal global_best, global_a, global_n
        n = len(arr)
        S = np.sqrt(2.0 * n)
        arr_norm = safe_normalize(arr, n)
        c = fft_convolve_self(arr_norm)
        max_b = c.max()
        obj = objective_from_max_b(max_b, n)
        if obj < global_best:
            global_best = obj
            global_a = arr_norm.copy()
            global_n = n
            print("  NEW BEST [%s]: obj=%.6f (n=%d, max_b=%.4f)" % (name, obj, n, max_b))
        return obj

    # === Phase 1: LSE-FISTA with beta annealing on parent ===
    if time.monotonic() < deadline - 60:
        print("\n=== Phase 1: LSE-FISTA beta annealing (n=%d) ===" % primary_n)
        a_start = parent.copy() if parent is not None else make_anti_palindromic_init(
            primary_n, rng, S_primary)
        seg_time = min(400.0, (deadline - time.monotonic()) * 0.35)
        a1, s1 = run_lse_fista(a_start, primary_n, S_primary, seg_time, rng,
                                beta_init=1.0, beta_final=3000.0, n_beta_steps=10)
        update_best(a1, "FISTA_primary")

    # === Phase 2: Peak reducer ===
    if global_a is not None and time.monotonic() < deadline - 60:
        print("\n=== Phase 2: Peak reducer ===")
        seg_time = min(300.0, (deadline - time.monotonic()) * 0.20)
        a2, s2 = run_peaking_reducer(global_a, global_n, np.sqrt(2.0 * global_n),
                                      seg_time, rng)
        update_best(a2, "PeakReducer")

    # === Phase 3: Spectral nudge ===
    if global_a is not None and time.monotonic() < deadline - 60:
        print("\n=== Phase 3: Spectral nudge ===")
        seg_time = min(200.0, (deadline - time.monotonic()) * 0.15)
        a3, s3 = run_spectral_nudge(global_a, global_n, np.sqrt(2.0 * global_n),
                                     seg_time, rng)
        update_best(a3, "Spectral")

    # === Phase 4: Try different initializations ===
    if time.monotonic() < deadline - 120:
        print("\n=== Phase 4: Diverse initializations ===")
        modes = ['sin_multi', 'w_shape', 'bimodal', 'anti_pal', 'random', 'ramp']
        n_trials = len(modes)
        seg_each = min(80.0, (deadline - time.monotonic()) * 0.20 / n_trials)

        for mode in modes:
            if time.monotonic() > deadline - 60:
                break
            if mode == 'anti_pal':
                a_init = make_anti_palindromic_init(primary_n, rng, S_primary)
            else:
                a_init = make_structured_init(primary_n, rng, S_primary, mode)

            a_t, s_t = run_lse_fista(a_init, primary_n, S_primary, seg_each, rng,
                                      beta_init=1.0, beta_final=1000.0, n_beta_steps=5)
            update_best(a_t, "init_%s" % mode)

    # === Phase 5: Try longer sequence ===
    if time.monotonic() < deadline - 180:
        n_long = 16384
        if n_long > 32768:
            n_long = 16384
        print("\n=== Phase 5: Longer sequence n=%d ===" % n_long)
        S_long = np.sqrt(2.0 * n_long)

        if parent is not None and len(parent) >= 64:
            # Upsample
            tt_old = np.linspace(0, 1, len(parent))
            tt_new = np.linspace(0, 1, n_long)
            a_long = np.interp(tt_new, tt_old, parent)
            a_long = np.maximum(a_long, 0.0)
        else:
            a_long = make_anti_palindromic_init(n_long, rng, S_long)

        seg_long = min(250.0, max(30.0, (deadline - time.monotonic()) * 0.25))
        a_l, s_l = run_lse_fista(a_long, n_long, S_long, seg_long, rng,
                                  beta_init=1.0, beta_final=2000.0, n_beta_steps=8)
        update_best(a_l, "long_n%d" % n_long)

    # === Phase 6: Intensive polishing of best ===
    if global_a is not None and time.monotonic() < deadline - 30:
        print("\n=== Phase 6: Intensive polishing ===")
        n = global_n
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()

        # 6a: High-beta FISTA
        seg = max(10.0, remaining * 0.30)
        a_p, s_p = run_lse_fista(global_a, n, S, seg, rng,
                                  beta_init=200.0, beta_final=5000.0, n_beta_steps=6)
        update_best(a_p, "Polish_HiBeta")

        # 6b: Peak reducer again
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            seg = max(10.0, remaining2 * 0.25)
            a_p2, s_p2 = run_peaking_reducer(global_a, n, S, seg, rng)
            update_best(a_p2, "Polish_Peak")

        # 6c: Block optimization
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            seg = max(10.0, remaining3 * 0.25)
            a_p3, s_p3 = run_alternating_blocks(global_a, n, S, seg, rng,
                                                  block_size=max(128, n // 64))
            update_best(a_p3, "Polish_Blocks")

        # 6d: Final spectral nudge
        remaining4 = deadline - time.monotonic()
        if remaining4 > 10:
            seg = max(5.0, remaining4 * 0.25)
            a_p4, s_p4 = run_spectral_nudge(global_a, n, S, seg, rng)
            update_best(a_p4, "Polish_Spectral")

    # === Phase 7: Final micro-polish with very high beta ===
    if global_a is not None and time.monotonic() < deadline - 5:
        remaining5 = deadline - time.monotonic()
        if remaining5 > 5:
            n = global_n
            S = np.sqrt(2.0 * n)
            # Quick high-beta pass
            a_final, s_final = run_lse_fista(global_a, n, S, remaining5 * 0.9, rng,
                                              beta_init=1000.0, beta_final=10000.0,
                                              n_beta_steps=3)
            update_best(a_final, "FinalMicro")

    # === Verification ===
    if global_a is None:
        print("WARNING: No valid sequence found, using fallback")
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\n=== VERIFICATION ===")
    print("Internal best objective: %.6f" % global_best)
    print("evaluate_sequence result: %.6f" % (ver if np.isfinite(ver) else float('inf')))
    print("Sequence length: %d" % len(final_list))

    if not np.isfinite(ver):
        print("WARNING: evaluate_sequence rejected, trying fallback")
        global_a = np.ones(2048, dtype=np.float64) * 1.0
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback result: %.6f" % (ver if np.isfinite(ver) else float('inf')))

    print("=== FINAL ANSWER: %.6f ===" % (ver if np.isfinite(ver) else float('inf')))
    return final_list
