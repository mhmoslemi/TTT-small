"""

Uses uniform sequences as anchor (theoretical optimum for non-negative seqs),

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score_fast(a):
    """Fast computation of 2*n*max_conv/sum^2 using symmetry (max at lag 0 for non-neg)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    # For non-negative, max is at lag 0 = sum(a^2)
    max_b = np.sum(a * a)
    return 2.0 * n * max_b / (s * s)


def nesterov_subgradient(a_init, n, S, max_time, rng, top_k_init=20, top_k_max=512):
    """Nesterov-accelerated subgradient on top-k active lags."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = compute_score_fast(y)
    best_local = y.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)

        c = fft_convolve_self(y)
        cur_score = compute_score_fast(y)
        if cur_score < best_score:
            best_score = cur_score
            best_local = y.copy()

        # Subgradient of max conv: use top-k lags weighted by proximity to max
        c_max = c.max()
        # Weight by how close each lag is to the max
        threshold = c_max * 0.8
        active = c > threshold
        if not np.any(active):
            active[np.argmax(c)] = True
        top_idx = np.where(active)[0]
        if len(top_idx) > top_k:
            top_idx = top_idx[np.argsort(c[top_idx])[::-1][:top_k]]

        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = c[top_idx]
        w_sum = w_full.sum()
        if w_sum < 1e-30:
            break
        w_full /= w_sum

        # Cross-correlation gradient
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 5 == 0:
            sc_x = compute_score_fast(x_new)
            if sc_x < best_score:
                best_score = sc_x
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 500 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t_mom = 1.0

    return best_local, best_score


def spectral_shaping(a_init, N, n, S, max_time, rng, top_k=10, alpha=0.2):
    """Spectral shaping: modify |FFT|^2 to flatten autocorrelation peak."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = compute_score_fast(a)
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = c.max()

        cur_score = compute_score_fast(a)
        if cur_score < best_score:
            best_score = cur_score
            best_local = a.copy()

        # Adaptive alpha
        excess = max(0.0, cur_score - 2.0) / max(cur_score, 1e-10)
        alpha_curr = alpha * (0.2 + 0.8 * min(1.0, excess * 10))

        # Build cosine effective from top lags
        m_arr = np.arange(N)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m_arr * int(k) / N)

        P_new = np.maximum(P * (1.0 - alpha_curr * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)

        if it % 200 == 100 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)

    return best_local, best_score


def fista_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=200.0):
    """FISTA with log-sum-exp smooth approximation of the max."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = compute_score_fast(y)
    best_local = y.copy()
    step = 1.0 / (beta_init * 2.0)
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.003 * it))

        c = fft_convolve_self(y)
        cur_score = compute_score_fast(y)
        if cur_score < best_score:
            best_score = cur_score
            best_local = y.copy()

        c_max = c.max()
        # LSE smoothing
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            sc_x = compute_score_fast(x_new)
            if sc_x < best_score:
                best_score = sc_x
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_mom

        step = 1.0 / (beta * 2.0 * (1.0 + 0.0003 * it))
        step = max(step, 1e-8)

        if it % 1000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.004 * S / np.sqrt(n), n), S)
            t_mom = 1.0

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and normalize to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    print("n=%d, S=%.4f, N(FFT)=%d" % (n, S, N))

    def eval_and_update(a, label):
        nonlocal global_best, global_a
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        fast_sc = compute_score_fast(a)
        # Verify with actual evaluation periodically
        if fast_sc < global_best + 0.01:
            test_list = [float(x) for x in a]
            try:
                real_sc = evaluate_sequence(test_list)
            except Exception:
                real_sc = fast_sc
            if np.isfinite(real_sc) and real_sc < global_best:
                global_best = real_sc
                global_a = a.copy()
                print("  [%s] score=%.6f" % (label, global_best))

    # --- Phase 1: Quick evaluation of diverse inits ---
    print("=== Phase 1: Diverse initialization ===")
    tt = np.linspace(0, 1, n, endpoint=False)

    candidates = []

    # Uniform (theoretical optimum for non-negative)
    candidates.append(("uniform", np.ones(n) * S / n))

    # Given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            candidates.append(("given", np.clip(given, 0.0, 1000.0)))
    except Exception:
        pass

    # Random
    candidates.append(("rand", rng.random(n) * S / n))

    # Chirp
    candidates.append(("chirp", (0.5 + 0.3 * np.sin(2 * np.pi * (3 * tt + 0.5 * tt * tt))) * S / n))

    # Multi-sine
    candidates.append(("multisine", (1.0 + 0.2 * np.sin(2 * np.pi * 5 * tt) + 0.1 * np.sin(2 * np.pi * 11 * tt)) * S / n))

    # Bump
    candidates.append(("bump", (np.exp(-0.5 * ((tt - 0.5) * 4.0) ** 2) + 0.2) * S / n))

    # Binary balanced
    bin_seq = (rng.random(n) < 0.5).astype(float) * S / n
    candidates.append(("binary", bin_seq))

    # Hamming-like: blocks of alternating high/low
    hamming = np.zeros(n)
    blk = n // 16
    for i in range(min(16, n // max(blk, 1))):
        hamming[i * blk:(i + 1) * blk] = 1.0 if i % 2 == 0 else 0.3
    candidates.append(("hamming", hamming * S / max(hamming.sum(), 1e-10) * n / n))

    for name, arr in candidates:
        if time.monotonic() > deadline - 60:
            break
        a_norm = safe_normalize(arr, n)
        eval_and_update(a_norm, name)

    print("  Phase 1 best: %.6f" % global_best)

    # --- Phase 2: Local optimization from best candidates ---
    print("=== Phase 2: Local optimization ===")
    t_p2 = time.monotonic()
    phase2_budget = (deadline - t_p2) * 0.5

    # Sort candidates by fast score, take top few
    scored = []
    for name, arr in candidates:
        a_norm = safe_normalize(arr, n)
        sc = compute_score_fast(a_norm)
        scored.append((sc, name, a_norm))
    scored.sort(key=lambda x: x[0])

    n_opt = min(len(scored), 4)
    per_opt = phase2_budget / max(n_opt, 1)

    for rank in range(n_opt):
        if time.monotonic() > deadline - 30:
            break
        sc, name, a_init = scored[rank]
        print("  Optimizing [%s] (init score=%.6f)... " % (name, sc), end="")

        a_curr = a_init.copy()
        t0 = time.monotonic()

        # Cycle through optimizers
        for method in ['ng', 'ss', 'fista', 'ng', 'ss']:
            rem = time.monotonic()
            rem_budget = min(per_opt, (deadline - 30) - time.monotonic())
            if rem_budget < 2:
                break

            if method == 'ng':
                a_curr, _ = nesterov_subgradient(a_curr, n, S, rem_budget, rng,
                                                  top_k_init=20, top_k_max=256)
            elif method == 'ss':
                a_curr, _ = spectral_shaping(a_curr, N, n, S, rem_budget, rng,
                                             top_k=10, alpha=0.15)
            elif method == 'fista':
                a_curr, _ = fista_lse(a_curr, n, S, rem_budget, rng,
                                      beta_init=5.0, beta_max=100.0)

            a_curr = safe_normalize(a_curr, n)
            eval_and_update(a_curr, "%s_%s" % (name, method))

        print(" done. Best so far: %.6f" % global_best)

    # --- Phase 3: Focused polishing on global best ---
    print("=== Phase 3: Focused polish ===")
    t_p3 = time.monotonic()
    remaining = deadline - t_p3

    if global_a is not None and remaining > 10:
        a_curr = global_a.copy()

        # High-beta FISTA
        seg1 = min(remaining * 0.25, 120.0)
        if seg1 > 3:
            a_curr, _ = fista_lse(a_curr, n, S, seg1, rng, beta_init=50.0, beta_max=500.0)
            a_curr = safe_normalize(a_curr, n)
            eval_and_update(a_curr, "polish_fista")

        # Nesterov with high top-k
        rem2 = deadline - time.monotonic()
        if rem2 > 5:
            seg2 = min(rem2 * 0.3, 120.0)
            a_curr, _ = nesterov_subgradient(a_curr, n, S, seg2, rng,
                                             top_k_init=50, top_k_max=512)
            a_curr = safe_normalize(a_curr, n)
            eval_and_update(a_curr, "polish_ng")

        # Spectral fine-tuning
        rem3 = deadline - time.monotonic()
        if rem3 > 3:
            seg3 = min(rem3 * 0.4, 60.0)
            a_curr, _ = spectral_shaping(a_curr, N, n, S, seg3, rng,
                                         top_k=20, alpha=0.05)
            a_curr = safe_normalize(a_curr, n)
            eval_and_update(a_curr, "polish_ss")

        # Final Nesterov
        rem4 = deadline - time.monotonic()
        if rem4 > 3:
            a_curr, _ = nesterov_subgradient(a_curr, n, S, rem4, rng,
                                             top_k_init=100, top_k_max=1024)
            a_curr = safe_normalize(a_curr, n)
            eval_and_update(a_curr, "final_ng")

    # --- Phase 4: Try different lengths if time remains ---
    if time.monotonic() < deadline - 30:
        print("=== Phase 4: Length exploration ===")
        for test_n in [1024, 2048, 8192]:
            if time.monotonic() > deadline - 15:
                break
            if test_n == n:
                continue
            a_test = np.ones(test_n) * np.sqrt(2.0 * test_n) / test_n
            eval_and_update(a_test, "uniform_%d" % test_n)

    # --- Final verification and return ---
    print("=== Final ===")
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Final verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using uniform fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        print("  No valid candidate found, using uniform fallback")
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d" % len(global_a))
    return [float(x) for x in global_a]
