def adam_update(params, m, v, t, grad, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Standard Adam update."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    params = params - lr * m_hat / (np.sqrt(v_hat) + eps)
    return params, m, v, m_hat, v_hat


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def freq_domain_shaping(a, n, S, alpha=0.7):
    """Apply amplitude shaping in frequency domain."""
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    amp = np.abs(A)
    phase = np.angle(A)
    
    # Shape the amplitude
    mean_amp = amp[amp > 0].mean() if np.any(amp > 0) else 1.0
    # Power compression: amp_new = amp^alpha
    amp_new = np.power(amp, alpha)
    # Or linear interpolation toward mean:
    # amp_new = (1-t)*amp + t*mean_amp
    # amp_new = amp**alpha  # use power compression
    
    # Reconstruct
    A_new = amp_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = normalize_nonneg(a_new, S)
    return a_new

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using continuation method with
log-sum-exp + variance penalty, frequency-domain amplitude shaping,
Adam optimizer in log-parameter space, and coordinate descent polish.

Key improvements over previous approach:
1. Variance penalty to encourage flat autocorrelation (not just low peak)
2. Frequency-domain amplitude shaping for local minimum escape
3. Refined continuation schedule with two-phase T/lam scheduling
4. Efficient gradient computation using FFT cross-correlation structure
5. Adaptive frequency shaping with acceptance criterion
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient(a, c, T, lam):
    """
    Compute gradient of LSE(c, T) + lam * var(c) with respect to a.
    Uses FFT-based cross-correlation for efficiency.
    
    grad[m] = 2 * (w cross a)[m] + (4*lam/size) * (c_dev cross a)[m]
    where (w cross a)[m] = sum_j w[m+j] * a[j]
    """
    n = len(a)
    size = 2 * n - 1
    
    # Softmax weights for LSE
    c_max = c.max()
    scaled = T * (c - c_max)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(size)
        w[np.argmax(c)] = 1.0
    else:
        w = w / w_sum
    
    # FFT size for cross-correlation of w (len size) with a (len n)
    fsize = 1 << (size + n - 1).bit_length()
    
    A = fft(a, fsize)
    
    # LSE gradient: 2 * (w cross a)
    W = fft(w, fsize)
    corr_w = np.real(ifft(W * np.conj(A)))[:n]
    grad = 2.0 * corr_w
    
    # Variance gradient: (4*lam/size) * (c_dev cross a)
    if lam > 1e-10:
        c_dev = c - c.mean()
        W2 = fft(c_dev, fsize)
        corr_cd = np.real(ifft(W2 * np.conj(A)))[:n]
        grad += (4.0 * lam / size) * corr_cd
    
    return grad


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def freq_domain_shaping(a, n, S, alpha=0.85):
    """Apply power compression to amplitude spectrum in zero-padded FFT domain."""
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    amp = np.abs(A)
    phase = np.angle(A)
    
    # Power compression
    amp_new = np.power(amp, alpha)
    
    # Reconstruct
    A_new = amp_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = normalize_nonneg(a_new, S)
    return a_new


def freq_domain_linear_mix(a, n, S, t=0.3):
    """Linearly mix amplitude toward its mean in zero-padded FFT domain."""
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    amp = np.abs(A)
    phase = np.angle(A)
    
    mean_amp = amp.mean()
    amp_new = (1.0 - t) * amp + t * mean_amp
    
    A_new = amp_new * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = normalize_nonneg(a_new, S)
    return a_new


def run_cont_opt(a_init, n, S, time_budget,
                 T_start=1.0, T_end=300.0,
                 lam_start=0.5, lam_end=0.0,
                 lr=0.04, freq_every=200,
                 verbose=True):
    """
    Continuation optimization with log-sum-exp + variance penalty.
    Uses Adam in log-parameter space.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    # Log parameters
    a_safe = np.maximum(a, 1e-15)
    theta = np.log(a_safe / a_safe.sum())
    
    n_par = n
    m_adam = np.zeros(n_par)
    v_adam = np.zeros(n_par)
    t_step = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t_step += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Continuation schedules
        # T increases (focus more on peak), lam decreases (less variance penalty)
        T = T_start * (T_end / T_start) ** (progress ** 0.8)
        lam = lam_start * ((1.0 - progress) ** 2)
        
        # Convert theta to a
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(np.clip(theta_shifted, -50, 50))
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n_par, S / n_par)
        else:
            a = S * exp_theta / total
        
        # Compute autocorrelation
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Compute gradient in a-space
        grad_a = compute_gradient(a, c, T, lam)
        
        # Convert to theta-space gradient
        dot = np.dot(grad_a, a)
        grad_theta = a * (grad_a - dot / S)
        
        # Adam update
        lr_t = lr * (0.4 + 0.6 * (1.0 - progress))
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad_theta
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_theta ** 2)
        m_hat = m_adam / (1.0 - beta1 ** t_step)
        v_hat = v_adam / (1.0 - beta2 ** t_step)
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic frequency-domain shaping
        if t_step % freq_every == 0 and progress > 0.05:
            if (t_step // freq_every) % 2 == 0:
                a_shaped = freq_domain_shaping(a, n, S, alpha=0.9)
            else:
                a_shaped = freq_domain_linear_mix(a, n, S, t=0.15)
            c_shaped = fft_convolve_self(a_shaped)
            if c_shaped.max() < cur_obj * 1.02:
                a = a_shaped
                a_safe = np.maximum(a, 1e-15)
                theta = np.log(a_safe / a_safe.sum())
                # Also update best if improved
                if c_shaped.max() < best_obj - 1e-14:
                    best_obj = c_shaped.max()
                    best_a = a_shaped.copy()
        
        if verbose and t_step % 1000 == 0:
            print("    ContOpt: step=%d, obj=%.7f, T=%.1f, lam=%.4f, time=%.0fs" 
                  % (t_step, best_obj, T, lam, elapsed))
    
    return best_a, best_obj

def coordinate_descent(a_init, n, S, time_budget, max_sweeps=30, verbose=True):
    """Coordinate descent: try multiplicative changes to individual coordinates."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(999)
    start = time.monotonic()
    
    factors = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            original = a[i]
            if original < 1e-12:
                continue
            
            best_local = original
            best_local_obj = best_obj
            
            for factor in factors:
                a[i] = original * factor
                a_norm = a.sum()
                if a_norm < 1e-15:
                    break
                a_trial = a * (S / a_norm)
                c_check = fft_convolve_self(a_trial)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-13:
                    best_local_obj = obj_check
                    best_local = original * factor
            
            a[i] = best_local
            a = a * (S / a.sum()) if a.sum() > 1e-15 else a
            c = fft_convolve_self(a)
            
            if c.max() < best_obj - 1e-13:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        if not improved and sweep >= 2:
            if verbose:
                print("    CD: converged at sweep %d, obj=%.7f" % (sweep, best_obj))
            break
    
    return best_a, best_obj


def get_initialization(name, n, S, rng):
    """Generate various initializations."""
    i = np.arange(n, dtype=np.float64)
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "hann_window":
        a = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1)))
        a = np.maximum(a, 1e-6)
        return a * (S / a.sum())
    elif name == "gaussian":
        a = np.exp(-0.5 * ((i - n / 2) / (n / 8)) ** 2)
        a = np.maximum(a, 1e-6)
        return a * (S / a.sum())
    elif name == "cosine_mix":
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * i / n + rng.random() * 2) + \
            0.1 * np.cos(2 * np.pi * 7 * i / n + rng.random() * 2) + \
            0.05 * np.cos(2 * np.pi * 17 * i / n + rng.random() * 2)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "flat_plus_noise":
        a = 1.0 + 0.05 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses continuation method with variance penalty, frequency-domain shaping,
    and coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 30.0)
    deadline = time.monotonic() + total_budget
    
    global_best = np.inf
    global_a = None
    
    # Load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.8f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given sequence: " + str(e))
    
    # Determine working length
    if given_a is not None:
        n = len(given_a)
    else:
        n = 4096
    
    S = np.sqrt(2.0 * float(n))
    print("Working with n=%d, S=%.4f" % (n, S))
    print("Target: 1.5027, Current best: %.8f" % global_best)
    
    # ============ PHASE 1: Quick exploration of initializations ============
    explore_budget = 0.12 * total_budget
    explore_deadline = time.monotonic() + explore_budget
    
    print("\n=== Phase 1: Exploration (%.0fs) ===" % explore_budget)
    
    init_names = ["hann_window", "gaussian", "cosine_mix", "flat_plus_noise", "random"]
    per_init = explore_budget / (len(init_names) + 1)
    
    # Also try from given
    if given_a is not None:
        candidates = [("given", given_a.copy())]
    else:
        candidates = []
    
    for name in init_names:
        a_init = get_initialization(name, n, S, rng)
        candidates.append((name, a_init))
    
    for name, a_init in candidates:
        if time.monotonic() > explore_deadline - 5:
            break
        t0 = time.monotonic()
        a_best, obj_best = run_cont_opt(
            a_init, n, S, per_init,
            T_start=1.0, T_end=50.0,
            lam_start=0.3, lam_end=0.0,
            lr=0.05, freq_every=100,
            verbose=False
        )
        a_check = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_check.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_check.copy()
            print("  Explore [%s]: obj=%.8f (%.0fs)" % (name, ver, time.monotonic() - t0))
        else:
            print("  Explore [%s]: obj=%.4f (no improvement)" % (name, ver if np.isfinite(ver) else 999))
    
    # ============ PHASE 2: Main continuation optimization ============
    main_budget = 0.45 * total_budget
    print("\n=== Phase 2: Main optimization (%.0fs) ===" % main_budget)
    
    if global_a is not None:
        a = global_a.copy()
        
        # Round 1: Broad continuation with variance penalty
        r1_budget = 0.30 * main_budget
        a_best, _ = run_cont_opt(
            a, n, S, r1_budget,
            T_start=1.0, T_end=100.0,
            lam_start=0.5, lam_end=0.0,
            lr=0.04, freq_every=200,
            verbose=True
        )
        a = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            print("  R1 (T:1-100, lam:0.5-0): obj=%.8f" % ver)
        
        # Round 2: Peak-focused with high T
        r2_budget = 0.35 * main_budget
        a_best, _ = run_subgrad_exp(
            a, n, S, r2_budget,
            T=100, lr=0.025,
            verbose=True
        )
        a = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            print("  R2 (SubGrad T=100+): obj=%.8f" % ver)
        
        # Round 3: Very high T, small steps
        r3_budget = 0.20 * main_budget
        a_best, _ = run_subgrad_exp(
            a, n, S, r3_budget,
            T=500, lr=0.015,
            verbose=True
        )
        a = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            print("  R3 (SubGrad T=500+): obj=%.8f" % ver)
        
        # Round 4: Mixed - continuation with frequency kicks
        r4_budget = 0.15 * main_budget
        a_best, _ = run_cont_opt(
            a, n, S, r4_budget,
            T_start=50.0, T_end=500.0,
            lam_start=0.1, lam_end=0.0,
            lr=0.02, freq_every=150,
            verbose=True
        )
        a = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            print("  R4 (ContOpt T:50-500 + freq): obj=%.8f" % ver)
    
    # ============ PHASE 3: Coordinate descent polish ============
    polish_budget = 0.25 * total_budget
    print("\n=== Phase 3: Polish (%.0fs) ===" % polish_budget)
    
    if global_a is not None:
        a = global_a.copy()
        
        # CD round 1
        cd1_budget = 0.4 * polish_budget
        a_best, _ = coordinate_descent(a, n, S, cd1_budget, max_sweeps=20)
        a = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            print("  CD1: obj=%.8f" % ver)
        
        # High-T subgradient between CD rounds
        sg_budget = 0.3 * polish_budget
        a_best, _ = run_subgrad_exp(
            a, n, S, sg_budget,
            T=1000, lr=0.01,
            verbose=False
        )
        a = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            print("  SG high-T: obj=%.8f" % ver)
        
        # CD round 2
        cd2_budget = 0.3 * polish_budget
        a_best, _ = coordinate_descent(a, n, S, cd2_budget, max_sweeps=15)
        a = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            print("  CD2: obj=%.8f" % ver)
    
    # ============ PHASE 4: Random restarts ============
    restart_budget = deadline - time.monotonic()
    print("\n=== Phase 4: Restarts (%.0fs) ===" % restart_budget)
    
    if global_a is not None and restart_budget > 20:
        n_rounds = max(2, int(restart_budget / 40))
        per_round = restart_budget / (n_rounds + 1)
        
        for r in range(n_rounds):
            if time.monotonic() > deadline - 10:
                break
            
            # Different noise patterns for diversity
            noise_scale = 0.003 * (1 + 0.5 * r)
            if r % 3 == 0:
                a_noisy = global_a + noise_scale * rng.standard_normal(n)
            elif r % 3 == 1:
                # Local perturbation
                idx = rng.choice(n, size=n // 4, replace=False)
                a_noisy = global_a.copy()
                a_noisy[idx] += noise_scale * rng.standard_normal(len(idx))
            else:
                # Frequency-domain perturbation
                a_noisy = freq_domain_shaping(global_a, n, S, alpha=0.95)
            
            a_noisy = normalize_nonneg(a_noisy, S)
            
            a_best, _ = run_cont_opt(
                a_noisy, n, S, per_round,
                T_start=100.0, T_end=2000.0,
                lam_start=0.05, lam_end=0.0,
                lr=0.015, freq_every=100,
                verbose=False
            )
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  Restart %d: obj=%.8f" % (r, ver))
            else:
                print("  Restart %d: obj=%.8f (no improve)" % (r, ver if np.isfinite(ver) else 999))
    
    # ============ Final: one more CD pass if time remains ============
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 10:
        print("\n=== Final CD (%.0fs) ===" % remaining)
        a_best, _ = coordinate_descent(global_a, n, S, remaining * 0.8, max_sweeps=10)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  Final CD: obj=%.8f" % ver)
    
    # Final verification
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.8f (verified=%s)" % (
        global_best, "%.8f" % ver if np.isfinite(ver) else "inf"))
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.8f" % ver)
    
    return final_a
