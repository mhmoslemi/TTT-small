"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.
Uses L-BFGS-B with log-sum-exp continuation, simulated annealing wrapper,
and multi-length exploration. FFT-accelerated throughout.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_a(a, w):
    """Compute grad_a[j] = 2 * sum_k w[k] * a[j-k] (cross-correlation via FFT)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """
    Compute smoothed objective and gradient w.r.t. z (unconstrained softmax params).
    a = S * softmax(z), objective = (1/beta)*log(sum exp(beta*c_k))
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_grad_a(a, w)

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=300,
                           verbose=False):
    """Run L-BFGS-B with continuation on beta."""
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        # At high beta, try multiple runs
        n_attempts = 1
        if beta >= 500 and max_time > 30:
            n_attempts = 2
        if beta >= 2000 and max_time > 60:
            n_attempts = 3

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 3:
                break
            if attempt > 0:
                # Perturb in z-space
                noise_scale = 0.01 / np.sqrt(beta / 100.0)
                z = z + rng.normal(0, noise_scale, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            t0 = time.monotonic()
            try:
                result = scipy_minimize(
                    fun, z,
                    jac=jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_beta,
                        'maxfun': maxiter_per_beta * 3,
                        'ftol': 1e-16,
                        'gtol': 1e-12,
                        'maxls': 50,
                        'iprint': -1
                    }
                )
            except Exception:
                break

            z = result.x
            z = z - z.mean()

            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()
                if verbose:
                    print("    beta=%d attempt=%d: score=%.10f" % (beta, attempt, score))

def run_peak_reduction(a_init, n, S, max_time, rng, 
                       top_k_init=10, top_k_max=256,
                       step_scale=1.0, verbose=False):
    """
    Simulated annealing with top-k peak reduction.
    Adaptively adjusts step size based on improvement.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    current_score = best_score
    it = 0
    no_improve_count = 0
    step_multiplier = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 200)
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            no_improve_count = 0
            # Increase step slightly on improvement
            step_multiplier = min(step_multiplier * 1.01, 5.0)
        else:
            no_improve_count += 1
            if no_improve_count > 200:
                step_multiplier = max(step_multiplier * 0.95, 0.1)

        # Get top-k peaks
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals ** 2 / (top_vals ** 2).sum()  # Weight more toward the highest peak

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        base_step = S * step_scale * step_multiplier / np.sqrt(n * it + 1)
        x_new = project_simplex(a - base_step * grad, S)

        a = x_new

        # Random perturbation (simulated annealing)
        if it % 500 == 0 and rng.random() < 0.15:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def get_objective_value(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_a(a, w):
    """Compute grad_a[j] = 2 * sum_k w[k] * a[j-k] via FFT."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. softmax params z."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_grad_a(a, w)

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z

        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break

        n_attempts = 1
        if beta >= 500 and max_time > 30:
            n_attempts = 2
        if beta >= 2000 and max_time > 60:
            n_attempts = 3

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 2:
                break
            if attempt > 0:
                noise_scale = 0.015 / max(1.0, np.sqrt(beta / 100.0))
                z = z + rng.normal(0, noise_scale, n)
                z = z - z.mean()

            obj_fun = ObjFun(beta, S)
            obj_jac = ObjJac(beta, S)

            try:
                result = scipy_minimize(
                    obj_fun, z,
                    jac=obj_jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_beta,
                        'maxfun': maxiter_per_beta * 3,
                        'ftol': 1e-16,
                        'gtol': 1e-12,
                        'maxls': 50,
                        'iprint': -1
                    }
                )
            except Exception:
                break

            z = result.x
            z = z - z.mean()

            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def run_peak_reduction(a_init, n, S, max_time, rng,
                       top_k_init=5, top_k_max=200, step_scale=1.0):
    """Simulated annealing with top-k peak reduction and adaptive step size."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    no_improve = 0
    step_mult = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 150)

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            no_improve = 0
            step_mult = min(step_mult * 1.005, 3.0)
        else:
            no_improve += 1
            if no_improve > 300:
                step_mult = max(step_mult * 0.97, 0.1)
                if no_improve > 1000:
                    # Restart from best
                    a = best_a.copy()
                    no_improve = 0
                    step_mult = 1.0

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        # Weight towards highest peaks
        weights = top_vals ** 3 / (top_vals ** 3).sum()

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S * step_scale * step_mult / np.sqrt(n * (it + 1) * 0.01 + 1)
        a = project_simplex(a - step * grad, S)

        # Random annealing perturbation
        if it % 300 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def get_objective_value(a):
    """Compute the actual objective value 2n*max_conv/sum^2."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def perturb_gaussian(a, n, S, rng, scale):
    """Gaussian perturbation."""
    a_pert = a + rng.normal(0, scale * S / np.sqrt(n), n)
    return safe_normalize(np.maximum(a_pert, 0.0), n)


def perturb_swap(a, n, S, rng, n_swaps):
    """Element swap perturbation: move mass between random pairs."""
    a_pert = a.copy()
    for _ in range(n_swaps):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        amount = min(a_pert[i], a_pert[j]) * (0.2 + 0.3 * rng.random())
        a_pert[i] -= amount
        a_pert[j] += amount
    return safe_normalize(np.maximum(a_pert, 0.0), n)


def perturb_block(a, n, S, rng):
    """Block perturbation: modify a contiguous segment."""
    a_pert = a.copy()
    block_size = max(8, n // 512)
    start = rng.integers(0, n - block_size)
    multiplier = 0.3 + 1.4 * rng.random()
    a_pert[start:start + block_size] *= multiplier
    return safe_normalize(np.maximum(a_pert, 0.0), n)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    
    Multi-phase approach:
    1. Diverse initialization (given + structured + random)
    2. L-BFGS-B with log-sum-exp continuation (high beta)
    3. Peak reduction via simulated annealing
    4. Cross-pollination and perturbation-based restarts
    5. Final ultra-high-beta polishing
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    global_best = np.inf
    global_a = None
    global_n = 0

    # --- Get initial sequence ---
    inits = {}
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits['given'] = np.clip(given, 0.0, 1000.0)
    except Exception:
        pass

    if 'given' in inits:
        primary_len = len(inits['given'])
    else:
        primary_len = 8192

    # Build list of (name, n, normalized_array) initializations
    all_inits = []

    # Primary: given sequence
    if 'given' in inits:
        a_g = safe_normalize(inits['given'], primary_len)
        all_inits.append(("given", primary_len, a_g))

    # Multi-frequency initializations at primary length
    n = primary_len
    S = np.sqrt(2.0 * float(n))
    tt = np.linspace(0, 1, n, endpoint=False)

    freq_sets = [
        [1, 3, 7, 15, 31],
        [2, 5, 11, 23, 47],
        [3, 8, 17, 35, 71],
        [5, 12, 29, 61, 127],
    ]
    for fi, freq_set in enumerate(freq_sets):
        sig = np.ones(n, dtype=np.float64)
        for f in freq_set:
            if 4 * f < n:
                sig += 0.2 * np.sin(2 * np.pi * f * tt + rng.random() * 2 * np.pi)
                sig += 0.15 * np.cos(2 * np.pi * f * tt + rng.random() * 2 * np.pi)
        sig = np.abs(sig) + 0.01
        all_inits.append(("freq%d" % fi, n, safe_normalize(sig, n)))

    # Random
    all_inits.append(("rand", n, safe_normalize(rng.random(n) + 0.05, n)))

    # Logarithmic chirp
    chirp = np.ones(n) + 0.3 * np.sin(2 * np.pi * tt * tt * n * 0.5) + 0.1 * np.cos(2 * np.pi * tt * n * 0.3)
    all_inits.append(("chirp", n, safe_normalize(np.abs(chirp) + 0.01, n)))

    # Bump structure
    bumps = np.ones(n) * 0.1
    for _ in range(5):
        center = rng.random()
        width = 0.02 + 0.05 * rng.random()
        bumps += 0.5 * np.exp(-0.5 * ((tt - center) / width) ** 2)
    all_inits.append(("bumps", n, safe_normalize(bumps, n)))

    # Also try a different length if budget allows
    alt_n = n * 2 if n <= 8192 else n // 2
    if alt_n >= 1024 and alt_n != n:
        S_alt = np.sqrt(2.0 * float(alt_n))
        alt_tt = np.linspace(0, 1, alt_n, endpoint=False)
        alt_sig = np.ones(alt_n) + 0.25 * np.sin(2 * np.pi * 7 * alt_tt) + \
                  0.2 * np.cos(2 * np.pi * 19 * alt_tt) + \
                  0.15 * np.sin(2 * np.pi * 43 * alt_tt) + 0.05
        all_inits.append(("alt_freq", alt_n, safe_normalize(np.abs(alt_sig) + 0.01, alt_n)))
        all_inits.append(("alt_rand", alt_n, safe_normalize(rng.random(alt_n) + 0.05, alt_n)))

    print("=== Optimization: budget=%.0fs, %d inits, primary n=%d ===" %
          (budget_s, len(all_inits), primary_len))
    print("Target: 1.5027, Previous best: 1.508805")

    # --- Phase 1: L-BFGS continuation on each init ---
    phase1_time = (deadline - time.monotonic()) * 0.50
    per_init = max(5.0, phase1_time / len(all_inits))

    print("\n=== Phase 1: L-BFGS Continuation (total ~%.0fs, ~%.1fs each) ===" %
          (phase1_time, per_init))

    phase1_results = []

    for i, (name, ni, arr) in enumerate(all_inits):
        if time.monotonic() > deadline - 100:
            break

        t0 = time.monotonic()
        Si = np.sqrt(2.0 * float(ni))
        obj_val = get_objective_value(arr)

        if obj_val < global_best:
            global_best = obj_val
            global_a = arr.copy()
            global_n = ni
            tag = " [NEW BEST]"
        else:
            tag = ""

        remaining_for_this = min(per_init, deadline - time.monotonic() - 80)
        if remaining_for_this < 5:
            print("  %-10s (n=%5d): %.8f%s (skipped opt)" % (name, ni, obj_val, tag))
            phase1_results.append((obj_val, arr, ni, name))
            continue

        # Adaptive schedule
        if remaining_for_this > 120:
            beta_sched = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
            maxit = 300
        elif remaining_for_this > 60:
            beta_sched = [10, 20, 50, 100, 300, 800, 2000, 5000]
            maxit = 250
        elif remaining_for_this > 30:
            beta_sched = [20, 50, 100, 300, 1000, 3000]
            maxit = 200
        else:
            beta_sched = [50, 100, 500, 2000]
            maxit = 150

        a_new, _ = run_lbfgs_continuation(
            arr, ni, Si, remaining_for_this, rng,
            beta_schedule=beta_sched, maxiter_per_beta=maxit
        )
        a_new = safe_normalize(a_new, ni)
        final_obj = get_objective_value(a_new)

        phase1_results.append((final_obj, a_new, ni, name))

        if final_obj < global_best:
            global_best = final_obj
            global_a = a_new.copy()
            global_n = ni
            print("  %-10s (n=%5d): %.8f [NEW BEST] (%.1fs)" %
                  (name, ni, global_best, time.monotonic() - t0))
        else:
            print("  %-10s (n=%5d): %.8f (%.1fs)" %
                  (name, ni, final_obj, time.monotonic() - t0))

    print("\n=== After Phase 1: global_best=%.8f (n=%d) ===" % (global_best, global_n))

    # --- Phase 2: Peak reduction on top candidates ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 50:
            print("\n=== Phase 2: Peak Reduction (~%.0fs) ===" % (remaining * 0.3))

            phase1_results.sort(key=lambda x: x[0])
            n_pr = min(3, len(phase1_results))
            pr_budget = remaining * 0.3 / max(1, n_pr)

            for j in range(n_pr):
                if time.monotonic() > deadline - 80:
                    break
                _, a_j, n_j, name_j = phase1_results[j]
                S_j = np.sqrt(2.0 * float(n_j))

                t0 = time.monotonic()
                a_new, _ = run_peak_reduction(
                    a_j, n_j, S_j, min(pr_budget, deadline - time.monotonic() - 60),
                    rng, top_k_init=5, top_k_max=128, step_scale=1.5
                )
                a_new = safe_normalize(a_new, n_j)
                obj_val = get_objective_value(a_new)

                if obj_val < global_best:
                    global_best = obj_val
                    global_a = a_new.copy()
                    global_n = n_j
                    print("  [NEW BEST] PeakRed %s (n=%d): %.8f (%.1fs)" %
                          (name_j, n_j, global_best, time.monotonic() - t0))
                else:
                    print("  PeakRed %s (n=%d): %.8f (%.1fs)" %
                          (name_j, n_j, obj_val, time.monotonic() - t0))

    print("\n=== After Phase 2: global_best=%.8f ===" % global_best)

    # --- Phase 3: Perturbation + L-BFGS (main exploration phase) ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            print("\n=== Phase 3: Perturbation + L-BFGS (~%.0fs) ===" % remaining)

            S_g = np.sqrt(2.0 * float(global_n))
            n_perturb = max(4, int(remaining / 50))
            perturb_budget = remaining * 0.7 / n_perturb

            for p in range(n_perturb):
                if time.monotonic() > deadline - 40:
                    break

                strategy = p % 4
                if strategy == 0:
                    scale = 0.03 / (1.0 + 0.3 * p)
                    a_pert = perturb_gaussian(global_a, global_n, S_g, rng, scale)
                elif strategy == 1:
                    n_swaps = max(20, global_n // 80)
                    a_pert = perturb_swap(global_a, global_n, S_g, rng, n_swaps)
                elif strategy == 2:
                    a_pert = perturb_block(global_a, global_n, S_g, rng)
                else:
                    # Hybrid: small Gaussian + swap
                    a_tmp = perturb_gaussian(global_a, global_n, S_g, rng, 0.02)
                    a_pert = perturb_swap(a_tmp, global_n, S_g, rng, global_n // 100)

                a_new, _ = run_lbfgs_continuation(
                    a_pert, global_n, S_g, perturb_budget, rng,
                    beta_schedule=[20, 50, 100, 300, 1000, 3000, 10000],
                    maxiter_per_beta=250
                )
                a_new = safe_normalize(a_new, global_n)
                obj_val = get_objective_value(a_new)

                if obj_val < global_best:
                    global_best = obj_val
                    global_a = a_new.copy()
                    print("  [NEW BEST] Pert %d (strat=%d): %.8f" % (p, strategy, global_best))
                else:
                    print("  Pert %d (strat=%d): %.8f" % (p, strategy, obj_val))

    print("\n=== After Phase 3: global_best=%.8f ===" % global_best)

    # --- Phase 4: Cross-pollination ---
    if global_a is not None and len(phase1_results) >= 2:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("\n=== Phase 4: Cross-pollination (~%.0fs) ===" % remaining)

            S_g = np.sqrt(2.0 * float(global_n))
            phase1_results.sort(key=lambda x: x[0])

            n_xp = min(3, len(phase1_results) - 1)
            xp_budget = remaining * 0.6 / max(1, n_xp)

            for j in range(1, n_xp + 1):
                if time.monotonic() > deadline - 20:
                    break
                _, a_j, n_j, _ = phase1_results[j]
                if n_j != global_n:
                    continue

                for blend in [0.3, 0.5, 0.7]:
                    if time.monotonic() > deadline - 10:
                        break
                    a_blend = blend * global_a + (1 - blend) * a_j
                    a_blend = safe_normalize(a_blend, global_n)

                    a_new, _ = run_lbfgs_continuation(
                        a_blend, global_n, S_g, min(xp_budget / 3, 40), rng,
                        beta_schedule=[50, 200, 800, 3000, 10000],
                        maxiter_per_beta=200
                    )
                    a_new = safe_normalize(a_new, global_n)
                    obj_val = get_objective_value(a_new)

                    if obj_val < global_best:
                        global_best = obj_val
                        global_a = a_new.copy()
                        print("  [NEW BEST] XP poll %d blend=%.1f: %.8f" % (j, blend, global_best))

    print("\n=== After Phase 4: global_best=%.8f ===" % global_best)

    # --- Phase 5: Final ultra-high-beta polishing ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print("\n=== Phase 5: Final Polish (~%.0fs) ===" % remaining)

            S_g = np.sqrt(2.0 * float(global_n))
            a_final, _ = run_lbfgs_continuation(
                global_a, global_n, S_g, remaining * 0.9, rng,
                beta_schedule=[500, 1000, 2000, 5000, 10000, 20000, 50000],
                maxiter_per_beta=400
            )
            a_final = safe_normalize(a_final, global_n)
            obj_val = get_objective_value(a_final)

            if obj_val < global_best:
                global_best = obj_val
                global_a = a_final.copy()
                print("  Final polish: %.8f [NEW BEST]" % global_best)
            else:
                print("  Final polish: %.8f" % obj_val)

    # --- Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.8f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, best_score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
