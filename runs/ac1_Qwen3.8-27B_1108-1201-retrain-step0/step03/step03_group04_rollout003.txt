multi-scale refinement, and adaptive subgradient polishing.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_convolution_gradient(a, c_weights):
    """
    Compute gradient of sum_k c_weights[k] * (a*a)[k] with respect to a.
    Uses FFT cross-correlation.
    """
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(c_weights, fsize)
    A = fft(a, fsize)
    # grad[i] = sum_k c_weights[k] * d(a*a)[k]/da[i]
    # d(a*a)[k]/da[i] = a[k-i] + a[i-k] (with appropriate bounds)
    # This is equivalent to: grad = 2 * cross_correlation(c_weights, a) (circular, padded)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Softmax weights."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total

This is the "flat-top" spectrum problem. For real non-negative sequences, this is related to the theory of "minimum peak-to-average power ratio" (PAPR) sequences.

One known approach: Use the "Graeme" or "successive overrelaxation" method:
1. Start with an initial sequence a
2. Compute its DFT A
3. Modify |A| to be more flat (while keeping A[0] = S)
4. Inverse DFT to get new a
5. Project to non-negative (clip negative values)
6. Repeat

This is similar to the "Gerchberg-Saxton" or "error diffusion" algorithm used in optics and signal processing.

def flat_spectrum_iteration(a, n_iterations, target_flat=1.0):
    """
    Iterate: FFT -> flatten magnitude -> IFFT -> clip to non-negative -> normalize
    """
    a = a.copy()
    n = len(a)
    fsize = 1 << (2*n - 2).bit_length()
    
    for _ in range(n_iterations):
        A = fft(a, fsize)
        # Target: |A[k]| = S/sqrt(2n-1) for k != 0, |A[0]| = S
        S = a.sum()
        target_mag = S / np.sqrt(2*n - 1)
        
        # New magnitude: interpolate between current and target
        new_mag = np.abs(A)
        new_mag[0] = S  # Keep DC
        for k in range(1, len(new_mag)):
            if k <= n - 1 or k >= fsize - n + 1:
                # These are the "valid" frequency bins
                new_mag[k] = (1 - target_flat) * new_mag[k] + target_flat * target_mag
        
        # Apply with original phases
        phases = np.angle(A)
        A_new = new_mag * np.exp(1j * phases)
        
        # Ensure Hermitian symmetry for real output
        A_new[:n] = (A_new[:n] + np.conj(A_new[fsize-n+1:-1][::-1])) / 2
        
        a = np.real(ifft(A_new, fsize)[:2*n-1])
        # Take the linear convolution region - actually we want the sequence a
        # The IFFT gives us the padded sequence
        a = np.real(ifft(A_new))[:n]
        
        # Clip to non-negative
        a = np.maximum(a, 0)
        # Normalize
        s = a.sum()
        if s > 0:
            a = a * (S / s)  # or normalize to sum = S
    
    return a

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_grad_weighted(a, weights):
    """Gradient of sum_k weights[k]*c[k] where c = a*a."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def softmax_weights(c, beta):
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total

def toppk_weights(c, top_k):
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights

def normalize(a, S):
    a = np.clip(a, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.ones(n) * S / n
    return a * (S / s)

def adam_smooth_optimize(a_init, S, max_time, rng, beta_start=10.0, beta_end=500.0, lr=0.01):
    """
    Adam optimization of the soft-max of the convolution.
    Minimizes (1/beta)*log(sum exp(beta*c[k])) which approximates max(c).
    """
    n = len(a_init)
    a = normalize(a_init, S).copy()
    c_full = 2 * n - 1
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        # Anneal beta
        frac = min(1.0, t / (max_time * 0.7))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute convolution and gradient
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score - 1e-10:
            best_score = cur_max
            best_a = a.copy()
        
        # Soft-max weights (proportional to gradient of log-sum-exp)
        weights = softmax_weights(c, beta)
        grad = compute_grad_weighted(a, weights)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive learning rate
        lr_t = lr / (1 + 0.001 * t)
        step = lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex
        a_new = project_simplex(a - step, S)
        a = a_new
        
        # Periodic check and perturbation
        if t % 5000 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() < best_score:
                best_score = c_check.max()
                best_a = a.copy()
            # Small perturbation to escape local minima
            if rng.random() < 0.3:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
    
    return best_a, best_score

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase with diverse starts, soft-max Adam optimization,
spectral flattening steps, and adaptive subgradient polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_weighted(a, weights):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

    total = exp_bc.sum()
    if total < 1e-300:

    return exp_bc / total

    if top_k >= len(c):
        s = c.sum()
        if s < 1e-30:
            return np.full(len(c), 1.0 / len(c))
        return c / s
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:

    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights

    s = a.sum()
    n = len(a)
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def spectral_flatten(a, n_iters=50, blend=0.5):
    """
    Iteratively flatten the magnitude spectrum.
    This helps spread energy across all lags of the convolution.
    """
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    S = a.sum()
    
    for _ in range(n_iters):
        A = fft(a, fsize)
        mag = np.abs(A)
        angle = np.angle(A)
        
        # Target: flat magnitude (excluding DC)
        # A[0] = S, target for others: S / sqrt(2n-1)
        target = S / np.sqrt(2 * n - 1)
        
        # Blend current and target magnitudes
        new_mag = mag.copy()
        new_mag[0] = S  # Keep DC fixed
        # For the "useful" frequency bins (those that affect the linear convolution)
        # In a padded FFT of size fsize, the linear convolution corresponds to
        # the first 2n-1 time-domain samples, which is the full IFFT output
        # All frequency bins contribute, so we flatten all except DC
        for k in range(1, fsize):
            new_mag[k] = (1 - blend) * mag[k] + blend * target
        
        # Reconstruct with original phases
        A_new = new_mag * np.exp(1j * angle)
        
        # Ensure Hermitian symmetry
        # For real signal: A[k] = conj(A[fsize-k])
        for k in range(1, fsize // 2 + 1):
            if k == fsize - k:
                continue
            # Average to enforce symmetry
            avg = (A_new[k] + np.conj(A_new[fsize - k])) / 2
            A_new[k] = avg
            A_new[fsize - k] = np.conj(avg)
        
        a = np.real(ifft(A_new, fsize))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * (S / s)
        else:
            a = np.ones(n) * S / n
    
    return a

    Adam optimization with soft-max smoothing.
    Uses exp-parameterization to handle non-negativity.
    """

    z = np.log(np.maximum(a, 1e-15))

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-10
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    t = 0
    stall = 0

        t += 1
        
        # Anneal beta (increase sharpness over time)
        frac = min(1.0, t / max(1, int(max_time * 0.6)))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute current a from z
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum() * S
        
        # Compute convolution and gradient
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score - 1e-10:
            best_score = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        w = softmax_weights(c, beta)
        grad_a = compute_grad_weighted(a, w)
        
        # Convert to z-gradient
        # dF/dz_i = a_i * (grad_a_i - sum_j grad_a_j * a_j / S)
        weighted_mean = np.dot(grad_a, a) / S
        grad_z = a * (grad_a - weighted_mean)

        v = b2 * v + (1 - b2) * (grad_z ** 2)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic spectral flattening
        if t % 2000 == 0 and stall > 200:
            a_curr = normalize_seq(a, S)
            a_flattened = spectral_flatten(a_curr, n_iters=20, blend=0.3)
            z = np.log(np.maximum(a_flattened, 1e-15))
            z -= z.max()
            m = np.zeros(n)
            v = np.zeros(n)
            stall = 0
        
        # Random restart from perturbation
        if stall > 5000 and rng.random() < 0.1:
            noise = rng.normal(0, 0.05, n)
            z = z + noise
            m = np.zeros(n)
            v = np.zeros(n)
            stall = 0
    
    return best_a, best_score


def run_subgradient_polish(a_init, n, S, max_time, rng, top_k_start=1, top_k_end=50, step_scale=1.0):
    """
    Subgradient method with top-k weighting for final polishing.
    """
    a = normalize_seq(a_init, S).copy()
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    t = 0
    step = step_scale * S / n
    stall = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        
        # Progressive top-k
        progress = min(1.0, t / max(1, int(max_time * 0.5)))
        top_k = max(1, int(top_k_start + (top_k_end - top_k_start) * progress))
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        if cur_max < best_score - 1e-10:
            best_score = cur_max
            best_a = a.copy()
            stall = 0
            step = min(step * 1.02, 2 * S / n)
        else:
            stall += 1
            step *= 0.995
        
        w = toppk_weights(c, top_k)
        grad = compute_grad_weighted(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Subgradient step with projection
        a_new = project_simplex(a - step * grad / gnorm, S)
        
        # Check if step helped
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller step
            a_half = project_simplex(a - 0.5 * step * grad / gnorm, S)
            c_half = fft_convolve_self(a_half)
            if c_half.max() < cur_max:
                a = a_half
                step *= 0.8
            else:
                step *= 0.9
                # Small random perturbation
                if rng.random() < 0.3:
                    noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                    a = project_simplex(a + noise, S)
        
        if stall > 3000:
            # Bigger perturbation
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            step = step_scale * S / n
            stall = 0
    
    return best_a, best_score


def run_nesterov_refined(a_init, n, S, max_time, rng, beta_init=30.0, beta_max=1000.0):
    """
    Nesterov accelerated gradient with softmax weights.
    Different from previous: uses backtracking and adaptive momentum.
    """
    a = normalize_seq(a_init, S).copy()
    y = a.copy()
    t_mom = 1.0
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    step = 1.0 / (beta_init * 3)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.02 * it))
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score - 1e-10:
            best_score = cur_max
            best_a = y.copy()
        
        w = softmax_weights(c, beta)
        grad = compute_grad_weighted(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Nesterov step
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-project to maintain sum constraint
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        a = x_new
        t_mom = t_new
        
        # Adaptive step
        c_x = fft_convolve_self(x_new)
        if c_x.max() < cur_max:
            step = min(step * 1.05, 1.0 / (beta * 0.5))
        else:
            step *= 0.95
        
        # Perturbation
        if it % 3000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
    
    return best_a, best_score


def generate_diverse_starts(n, S, given_a, rng):
    """Generate diverse starting points."""
    starts = []
    centers = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    
    # 1. Given sequence
    if given_a is not None:
        starts.append(normalize_seq(given_a[:n], S) if len(given_a) >= n else normalize_seq(given_a, S))
    
    # 2. Gaussian with various widths
    for sigma_frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
        sigma = sigma_frac * n
        g = np.exp(-centers**2 / (2 * sigma**2))
        starts.append(normalize_seq(g, S))
    
    # 3. Raised cosine
    for width_frac in [0.3, 0.5, 0.7, 1.0]:
        width = width_frac * n
        rc = np.ones(n)
        half = width / 2
        mask = np.abs(centers) > half
        rc[mask] = 0.5 * (1 + np.cos(np.pi * (centers[mask] - np.sign(centers[mask]) * half) / (n/2 - half)))
        rc = np.maximum(rc, 0)
        starts.append(normalize_seq(rc, S))
    
    # 4. Random smoothed
    for scale in [0.5, 1.0, 2.0]:
        r = rng.random(n)
        # Smooth with a kernel
        kernel_size = max(3, int(n * 0.05))
        kernel = np.exp(np.linspace(-3, 3, kernel_size))
        kernel /= kernel.sum()
        r_smooth = np.convolve(r, kernel, mode='same')
        starts.append(normalize_seq(r_smooth, S))
    
    # 5. Cosine-weighted (like a Hamming window)
    for alpha in [0.5, 0.7, 1.0]:
        hw = 0.54 - alpha * 0.46 * np.cos(2 * np.pi * centers / n)
        starts.append(normalize_seq(hw, S))
    
    # 6. "Anti-peak" initialization: reduce mass where it contributes most to peak
    # Start uniform, then reduce center (which contributes most to center lag)
    uni = np.ones(n)
    center_idx = n // 2
    reduction = np.exp(-centers**2 / (2 * (n*0.1)**2))
    anti = uni * (1 - 0.5 * reduction)
    starts.append(normalize_seq(anti, S))
    
    return starts


def propose_candidate(seed=42, budget_s=2385):
    """
    Multi-phase optimization for minimizing 2*n*max_conv/sum^2.
    Uses diverse initializations, soft-max Adam, spectral flattening,
    Nesterov acceleration, and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    
    # Get initial sequence
    given_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given_a = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    if given_a is not None:
        n = len(given_a)
    else:
        n = 8192
        given_a = np.ones(n) * 0.5
    
    S = np.sqrt(2.0 * float(n))
    
    # Generate diverse starts
    starts = generate_diverse_starts(n, S, given_a, rng)
    
    # Evaluate all starts
    best_overall = np.inf
    best_a = None
    for i, s in enumerate(starts):
        c = fft_convolve_self(s)
        score = c.max()
        obj = 2.0 * n * score / (S * S)
        if score < best_overall:
            best_overall = score
            best_a = s.copy()
        if i < 10:
            print("Start %d: score=%.6f obj=%.6f" % (i, score, obj))
    
    print("Best start: score=%.6f obj=%.6f" % (best_overall, 2.0*n*best_overall/S**2))
    
    # Phase 1: Soft-max Adam on best start (30% time)
    time_budget = deadline - time.monotonic()
    t1 = 0.25 * time_budget
    if t1 > 10:
        print("Phase 1: SoftMax Adam (%.0f sec)" % t1)
        t0 = time.monotonic()
        a1, _ = run_softmax_adam(best_a, n, S, t1, rng, beta_start=5, beta_end=200, lr_scale=0.3)
        a1 = normalize_seq(a1, S)
        c1 = fft_convolve_self(a1)
        if c1.max() < best_overall:
            best_overall = c1.max()
            best_a = a1.copy()
            print("  SoftMax Adam: %.6f (obj %.6f) [%.0fs]" % (best_overall, 2*n*best_overall/S**2, time.monotonic()-t0))
    
    # Phase 2: Nesterov (20% time)
    remaining = deadline - time.monotonic()
    t2 = 0.20 * remaining
    if t2 > 10:
        print("Phase 2: Nesterov (%.0f sec)" % t2)
        t0 = time.monotonic()
        a2, _ = run_nesterov_refined(best_a, n, S, t2, rng, beta_init=30, beta_max=800)
        a2 = normalize_seq(a2, S)
        c2 = fft_convolve_self(a2)
        if c2.max() < best_overall:
            best_overall = c2.max()
            best_a = a2.copy()
            print("  Nesterov: %.6f (obj %.6f) [%.0fs]" % (best_overall, 2*n*best_overall/S**2, time.monotonic()-t0))
    
    # Phase 3: Subgradient polish (15% time)
    remaining = deadline - time.monotonic()
    t3 = 0.15 * remaining
    if t3 > 5:
        print("Phase 3: Subgradient polish (%.0f sec)" % t3)
        t0 = time.monotonic()

        a3 = normalize_seq(a3, S)
        c3 = fft_convolve_self(a3)
        if c3.max() < best_overall:
            best_overall = c3.max()
            best_a = a3.copy()
            print("  Subgrad: %.6f (obj %.6f) [%.0fs]" % (best_overall, 2*n*best_overall/S**2, time.monotonic()-t0))
    
    # Phase 4: Try other diverse starts with quick optimization (20% time)
    remaining = deadline - time.monotonic()

    if t4 > 10:
