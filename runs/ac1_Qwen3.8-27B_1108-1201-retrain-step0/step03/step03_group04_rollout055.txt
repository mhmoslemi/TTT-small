```python
from scipy.optimize import linprog

def lp_direction(f, M, n):
    """Find g that maximizes sum(g) subject to g >= 0, (f*g) <= M."""
    # c = [1, 1, ..., 1] (maximize sum)
    c = -np.ones(n)  # minimize -sum(g)
    
    # A_ub * g <= b_ub
    # (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_j f[j] * g[k-j] where g is defined on [0, n-1]
    # For lag k: sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]
    # Let i = k-j, so j = k-i, i ranges from max(0, k-n+1) to min(n-1, k)
    # sum_{i} f[k-i] * g[i]
    
    # Build A_ub matrix (2n-1 x n)
    A_ub = np.zeros((2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    for i in range(n):  # g[i]
        # g[i] appears in (f*g)[k] for k where j = k-i is in [0, n-1]
        # i.e., k-i in [0, n-1], so k in [i, i+n-1]
        for k in range(i, min(i+n, 2*n-1)):
            j = k - i
            A_ub[k, i] = f[j]
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        return g / g.sum() * np.sqrt(2*n)  # normalize
    return None
```

```python
        i_end = min(n, k + 1)

        j_end = k - i_start

        rows[idx, i_start:i_end] = f[j_start:j_end]
```

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: LP-based direction finding (constraint generation) as the core
optimization, combined with Nesterov-accelerated gradient polishing,
multi-scale exploration (n=8192 and n=16384), and safe backtracking.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv_at_lags(f, g, lags, n):
    """Compute (f*g)[k] for specific lags k."""
    fsize = 1 << (2 * n - 1).bit_length()
    F = fft(f, fsize)
    G = fft(g, fsize)
    conv = np.real(ifft(F * G))[:2*n-1]
    return conv[lags]


def lp_direction(f, M, n, lags, timeout=30):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in lags."""
    # Build constraint matrix
    num_constraints = len(lags)
    A = np.zeros((num_constraints, n), dtype=np.float64)
    
    for idx, k in enumerate(lags):
        # (f*g)[k] = sum_i f[k-i] * g[i] for valid i
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                A[idx, i] = f[j]
    
    c = -np.ones(n)
    b = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', 
                      options={'time_limit': timeout})
        if res.success:
            g = res.x
            if g.sum() > 1e-15:
                return g
    except Exception as e:
        print("  LP error: %s" % str(e))
    return None

    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_a = a.copy()
    best_M = fft_convolve_self(a).max()
    it = 0
    stall = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        M = c.max()

            best_M = M
            best_a = a.copy()
            stall = 0
```

```python
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_cross(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:size]


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, lags, n):
    num_constraints = len(lags)
    A = np.zeros((num_constraints, n), dtype=np.float64)
    for idx, k in enumerate(lags):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_end >= i_start:
            j_start = k - i_end
            j_end = k - i_start
            A[idx, i_start:i_end + 1] = f[j_start:j_end + 1]
    return A


def lp_direction(f, M, n, lags, timeout=25):
    A = build_lp_matrix(f, lags, n)
    c = -np.ones(n)
    b = np.full(len(lags), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            g = res.x
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def gradient_step(a, S, n, top_k=5, step_frac=0.25):
    c = fft_convolve_self(a)
    top_k = min(top_k, len(c))
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_idx]
    w = top_vals / top_vals.sum()
    weights = np.zeros(2 * n - 1)
    weights[top_idx] = w
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return a, c.max()
    step = step_frac * S / np.sqrt(n)
    a_new = project_simplex(a - step * grad / gnorm, S)
    return a_new, c.max()


def run_lp_optimization(a_init, n, S, max_time, rng, num_lags_start=50, num_lags_max=200):
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_a = a.copy()
    best_M = fft_convolve_self(a).max()
    it = 0
    stall = 0
    lp_fail_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        remaining = max_time - (time.monotonic() - start)
        if remaining < 3:
            break

        c = fft_convolve_self(a)
        M = c.max()

        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            stall = 0
            if it % 30 == 1:
                obj = 2.0 * n * best_M / (S * S)
                print("    LP iter %d: M=%.6f obj=%.6f [%.0fs elapsed]" % (it, best_M, obj, time.monotonic() - start))
        else:
            stall += 1

        progress = min(1.0, it / 300.0)
        num_lags = int(num_lags_start + (num_lags_max - num_lags_start) * progress)
        num_lags = min(num_lags, 2 * n - 1)

        top_lags = np.argsort(c)[-num_lags:]
        n_random = min(15, max(1, num_lags // 10))
        random_lags = rng.integers(0, 2 * n - 1, size=n_random)
        all_lags = np.unique(np.concatenate([top_lags, random_lags]))
        all_lags = all_lags[:min(num_lags + n_random, 2 * n - 1)]

        g = lp_direction(a, M, n, all_lags, timeout=min(20, remaining * 0.3))

        if g is None:
            lp_fail_count += 1
            a_new, _ = gradient_step(a, S, n, top_k=min(5, n), step_frac=0.2)
            M_new = fft_convolve_self(a_new).max()
            if M_new < M - 1e-14:
                a = a_new
            else:
                noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
            continue

        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            a_new, _ = gradient_step(a, S, n, top_k=3, step_frac=0.15)
            M_new = fft_convolve_self(a_new).max()
            if M_new < M - 1e-14:
                a = a_new
            continue
        g = g / gs * S

        g_conv = fft_convolve_self(g)
        G = g_conv.max()
        fg_conv = fft_convolve_cross(a, g)
        M_fg = fg_conv.max()

        first_improve = False
        t = min(0.5, 0.9 * M / max(G, M_fg, M * 0.5, 1e-15))
        t = max(t, 1e-4)

        for _ in range(12):
            if t < 1e-7:
                break
            a_new = (1.0 - t) * a + t * g
            a_new = project_simplex(a_new, S)
            M_new = fft_convolve_self(a_new).max()
            if M_new < M - 1e-14:
                a = a_new
                first_improve = True
                break
            t *= 0.5

        if not first_improve:
            stall += 2
            if stall > 15:
                noise_scale = 0.015 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
                stall = 0
                lp_fail_count = 0

        if it % 150 == 0 and rng.random() < 0.25:
            noise_scale = 0.003 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best_M


def run_gradient_polish(a_init, n, S, max_time, rng, top_k_init=1, top_k_max=32, top_k_rate=100):
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_M = fft_convolve_self(a).max()
    it = 0
    step = S / (n * 3.0)

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // top_k_rate)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_M - 1e-12:
            best_M = cur
            best_a = y.copy()
            if it % 50 == 1:
                obj = 2.0 * n * best_M / (S * S)
                print("    Grad iter %d: M=%.6f obj=%.6f [%.0fs]" % (it, best_M, obj, time.monotonic() - start))

        top_idx = np.argpartition(c, -top_k)[-top_k:]
        top_vals = c[top_idx]
        w = top_vals / max(top_vals.sum(), 1e-15)
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = w

        fsize = 1 << (2 * n - 1).bit_length()
        W = fft(weights, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step_decay = 1.0 / (1.0 + 0.002 * it)
        a_new = project_simplex(y - step * step_decay * grad / gnorm, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        a = a_new
        t_mom = t_new

        if it % 200 == 0 and rng.random() < 0.15:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0

    return best_a, best_M


def run_safe_polish(a_init, n, S, max_time, rng):
    """Very careful coordinate-level polish using direct peak reduction."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_a = a.copy()
    best_M = fft_convolve_self(a).max()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        M = c.max()

        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            if it % 20 == 1:
                obj = 2.0 * n * best_M / (S * S)
                print("    Safe iter %d: M=%.6f obj=%.6f [%.0fs]" % (it, best_M, obj, time.monotonic() - start))

        peak_lag = np.argmax(c)
        # Find which a[j] contribute most to the peak
        # (a*a)[peak_lag] = sum_j a[j] * a[peak_lag - j]
        i_start = max(0, peak_lag - n + 1)
        i_end = min(n - 1, peak_lag)
        contributions = np.array([a[j] * a[peak_lag - j] if 0 <= peak_lag - j < n else 0.0
                                  for j in range(i_start, i_end + 1)])
        top_contrib_idx = np.argsort(contributions)[-min(8, len(contributions)):]
        top_j = np.array(range(i_start, i_end + 1))[top_contrib_idx]

        # Try reducing the top contributors and redistributing
        for j in top_j:
            if a[j] < 1e-6:
                continue
            # Try moving mass from a[j] to the least-contributing position
            a_test = a.copy()
            delta = 0.05 * a[j]
            a_test[j] -= delta
            # Redistribute to a random position
            target = rng.integers(0, n)
            a_test[target] += delta
            a_test = project_simplex(a_test, S)
            M_test = fft_convolve_self(a_test).max()
            if M_test < M - 1e-14:
                a = a_test
                break
        else:
            # No coordinate move helped; try small random perturbation
            if it % 50 == 0:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)

    return best_a, best_M


def upsample_sequence(a, new_n):
    old_n = len(a)
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    a_resized = np.interp(new_x, old_x, a, period=1.0)
    return np.maximum(a_resized, 0.0)


def propose_candidate(seed=42, budget_s=2385, **kwargs):
    """
    Minimize 2*n*max_conv/sum^2 using LP-based direction finding,
    gradient polishing, and multi-scale exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0

    global_best_obj = np.inf
    global_a = None
    global_n = 0

    # Get parent sequence
    parent_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            parent_a = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Determine candidate configurations to try
    candidates = []

    if parent_a is not None:
        n_parent = len(parent_a)
        S_parent = np.sqrt(2.0 * float(n_parent))
        a_parent = np.clip(parent_a, 0, 1000) * (S_parent / max(a_parent.sum(), 1e-15))
        c_parent = fft_convolve_self(a_parent)
        obj_parent = 2.0 * n_parent * c_parent.max() / (S_parent * S_parent)
        print("Parent: n=%d, obj=%.6f" % (n_parent, obj_parent))

        candidates.append((n_parent, a_parent, "parent"))

        # Upsampled version
        if n_parent < 16384:
            n_up = max(16384, n_parent * 2)
            if n_up <= 32768:
                a_up = upsample_sequence(a_parent, n_up)
                S_up = np.sqrt(2.0 * float(n_up))
                a_up = a_up * (S_up / max(a_up.sum(), 1e-15))
                candidates.append((n_up, a_up, "upsampled_%d" % n_up))

        # Smaller version
        if n_parent > 4096:
            n_sm = 4096
            a_sm = a_parent[:n_sm]
            S_sm = np.sqrt(2.0 * float(n_sm))
            a_sm = a_sm * (S_sm / max(a_sm.sum(), 1e-15))
            candidates.append((n_sm, a_sm, "smaller_%d" % n_sm))
    else:
        n_parent = 8192
        a_parent = np.ones(n_parent) * 0.5
        S_parent = np.sqrt(2.0 * float(n_parent))
        a_parent = a_parent * (S_parent / a_parent.sum())
        candidates.append((n_parent, a_parent, "default"))

    # Add a random candidate for diversity
    n_rand = 8192
    a_rand = rng.random(n_rand)
    S_rand = np.sqrt(2.0 * float(n_rand))
    a_rand = a_rand * (S_rand / a_rand.sum())
    candidates.append((n_rand, a_rand, "random"))

    # Select top candidates by quick evaluation (keep at most 4)
    scored = []
    for n_c, a_c, name in candidates:
        S_c = np.sqrt(2.0 * float(n_c))
        c_conv = fft_convolve_self(a_c)
        obj_c = 2.0 * n_c * c_conv.max() / (S_c * S_c)
        scored.append((obj_c, n_c, a_c, name))
        print("  Candidate '%s': n=%d, obj=%.6f" % (name, n_c, obj_c))

    scored.sort(key=lambda x: x[0])
    scored = scored[:4]

    # Allocate time proportionally: more time to better candidates
    total_time = deadline - time.monotonic()
    weights = np.array([1.0 / (s[0] + 0.1) for s in scored])
    weights = weights / weights.sum()
    # Ensure minimum 100s per candidate
    time_alloc = np.maximum(weights * total_time, 100.0)
    time_alloc = time_alloc / time_alloc.sum() * total_time

    for idx, (obj_c, n_c, a_c, name) in enumerate(scored):
        if time.monotonic() > deadline - 60:
            break

        S_c = np.sqrt(2.0 * float(n_c))
        allocated = time_alloc[idx]
        print("\n=== Optimizing '%s' (n=%d, initial obj=%.6f) with %.0fs ===" % (name, n_c, obj_c, allocated))

        # Sub-pipeline: LP optimization -> gradient polish -> safe polish
        t_lp = allocated * 0.55
        t_grad = allocated * 0.25
        t_safe = allocated * 0.20

        # Phase A: LP-based optimization
        if t_lp > 30:
            a_lp, M_lp = run_lp_optimization(a_c, n_c, S_c, t_lp, rng,
                                             num_lags_start=40, num_lags_max=180)
            obj_lp = 2.0 * n_c * M_lp / (S_c * S_c)
            print("  After LP: obj=%.6f" % obj_lp)
            if obj_lp < global_best_obj:
                global_best_obj = obj_lp
                global_a = a_lp.copy()
                global_n = n_c
        else:
            a_lp = a_c
            M_lp = fft_convolve_self(a_lp).max()

        # Phase B: Gradient polish
        if t_grad > 15:
            a_gp, M_gp = run_gradient_polish(a_lp, n_c, S_c, t_grad, rng,
                                             top_k_init=1, top_k_max=32, top_k_rate=80)
            obj_gp = 2.0 * n_c * M_gp / (S_c * S_c)
            print("  After gradient polish: obj=%.6f" % obj_gp)
            if obj_gp < global_best_obj:
                global_best_obj = obj_gp
                global_a = a_gp.copy()
                global_n = n_c
        else:
            a_gp = a_lp

        # Phase C: Safe coordinate polish
        if t_safe > 10 and time.monotonic() < deadline - 30:
            a_sp, M_sp = run_safe_polish(a_gp, n_c, S_c, t_safe, rng)
            obj_sp = 2.0 * n_c * M_sp / (S_c * S_c)
            print("  After safe polish: obj=%.6f" % obj_sp)
            if obj_sp < global_best_obj:
                global_best_obj = obj_sp
                global_a = a_sp.copy()
                global_n = n_c

    # Final exploration: try perturbing the best and re-optimizing
    remaining = deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        print("\n=== Final exploration (%.0fs remaining) ===" % remaining)
        n_best = global_n
        S_best = np.sqrt(2.0 * float(n_best))

        for trial in range(3):
            if time.monotonic() > deadline - 30:
                break
            t_trial = (remaining - 30 - trial * 40) / 3.0
            if t_trial < 30:
                break

            # Different perturbation strategies
            if trial == 0:
                perturbed = global_a + rng.normal(0, 0.01 * S_best / np.sqrt(n_best), n_best)
            elif trial == 1:
                smooth = np.convolve(rng.normal(0, 1, n_best), np.ones(128) / 128, mode='same')
                perturbed = global_a + 0.008 * S_best / np.sqrt(n_best) * smooth
            else:
                perturbed = global_a * (1.0 + rng.normal(0, 0.02, n_best))

            perturbed = np.maximum(perturbed, 0.0)
            perturbed = perturbed * (S_best / max(perturbed.sum(), 1e-15))

            a_tr, M_tr = run_lp_optimization(perturbed, n_best, S_best, t_trial, rng,
                                             num_lags_start=50, num_lags_max=150)
            obj_tr = 2.0 * n_best * M_tr / (S_best * S_best)
            print("  Trial %d: obj=%.6f" % (trial, obj_tr))
            if obj_tr < global_best_obj:
                global_best_obj = obj_tr
                global_a = a_tr.copy()
                global_n = n_best

    # Final answer
    if global_a is None:
        # Fallback
        n_fb = 8192
        a_fb = np.ones(n_fb) * 0.5
        S_fb = np.sqrt(2.0 * n_fb)
        a_fb = a_fb * (S_fb / a_fb.sum())
        final_list = [float(x) for x in np.clip(a_fb, 0, 1000)]
        final_obj = 2.0 * n_fb * fft_convolve_self(a_fb).max() / (S_fb * S_fb)
        print("FALLBACK: obj=%.6f" % final_obj)
        return final_list

    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    print("\nFINAL: n=%d, obj=%.6f" % (global_n, global_best_obj))

    # Verify
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: %.6f" % ver)
        if ver < global_best_obj:
            print("  (verified value is lower, using it)")
    else:
        print("Verification failed!")
        # Try with small epsilon floor
        final_list = [max(0.0, float(x)) for x in global_a]
        ver2 = evaluate_sequence(final_list)
        if np.isfinite(ver2):
            print("  Retry verified: %.6f" % ver2)
        else:
            n_fb = 256
            a_fb = np.ones(n_fb) * 0.5
            S_fb = np.sqrt(2.0 * n_fb)
            a_fb = a_fb * (S_fb / a_fb.sum())
            final_list = [float(x) for x in a_fb]
            print("  Using ultimate fallback, obj=%.6f" % (2.0 * n_fb * fft_convolve_self(a_fb).max() / (S_fb * S_fb)))

    return final_list
```
