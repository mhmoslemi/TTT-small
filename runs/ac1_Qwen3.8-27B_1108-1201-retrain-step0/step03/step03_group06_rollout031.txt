"""
LP-based iterative peak reduction for self-convolution minimization.

Algorithm:
1. Start with initial non-negative sequence normalized to sum = sqrt(2n)
2. Iteratively:
   a. Compute current peak M = max(a*a)
   b. Solve LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0
   c. Normalize b to have same sum as a
   d. Line search: find largest t such that peak of ((1-t)*a + t*b)^2 <= M
   e. Update a
3. Final polish with coordinate descent
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def build_conv_matrix_values(a, rows, cols):
    """Compute values for the convolution matrix given sparsity structure."""
    k = rows.astype(np.int64)
    j = cols.astype(np.int64)
    return a[k - j]


def solve_lp_peak(a, M, rows, cols, n):
    """Solve the LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0."""
    vals = build_conv_matrix_values(a, rows, cols)
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_time=30)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_time=30)  # SCS might not be available
        except:
            try:
                prob.solve(solver=cp.CLARABEL, max_time=30)
            except:
                return None
    
    if b.value is None:
        return None
    return np.maximum(np.asarray(b.value).flatten(), 0.0)


def lp_iteration(a, S, n, rows, cols, time_budget=30):
    """One LP-based iteration."""
    c = fft_convolve_self(a)
    M = c.max()
    
    b_val = solve_lp_peak(a, M, rows, cols, n)
    if b_val is None or b_val.sum() < 1e-10:
        return a, M, 0.0
    
    # Normalize b to have sum S
    b_val = b_val * (S / b_val.sum())
    
    # Line search for step size
    best_t = 0.0
    best_a = a.copy()
    best_obj = M
    
    for t in [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005]:
        a_new = (1 - t) * a + t * b_val
        a_new = normalize_nonneg(a_new, S)
        c_new = fft_convolve_self(a_new)
        obj_new = c_new.max()
        if obj_new <= M * (1 + 1e-9):
            best_t = t
            best_a = a_new
            best_obj = obj_new
            break
    
    if best_t == 0.0:
        # Use tiny step
        t = 0.001
        a_new = (1 - t) * a + t * b_val
        best_a = normalize_nonneg(a_new, S)
        best_obj = fft_convolve_self(best_a).max()
    
    return best_a, best_obj, best_t


def run_lp_method(a_init, S, n, rows, cols, time_budget, verbose=True):
    """Run the LP-based iterative method."""
    a = normalize_nonneg(a_init.copy(), S)
    c = fft_convolve_self(a)
    best_M = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    iteration = 0
    no_improve = 0
    min_improve = 1e-10
    
    while time.monotonic() - start < time_budget:
        a_new, M_new, t = lp_iteration(a, S, n, rows, cols)
        
        if M_new < best_M - min_improve:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        iteration += 1
        
        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"  LP iter {iteration}: M={best_M:.10f}, t={t:.4f}, t_elapsed={elapsed:.1f}s, no_improve={no_improve}")
        
        if no_improve > 30:
            if verbose:
                print("  LP: no improvement for 30 iterations, stopping")
            break
    
    return best_a, best_M


def coordinate_descent_polish(a, n, S, time_budget, verbose=True):

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(42)
    sweep = 0
    no_improve_sweeps = 0
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                    break
            
            orig = a[i]
            if orig < 1e-14:
                continue
            
            best_local = orig
            best_local_obj = best_obj
            
            for factor in [0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-15:
                    a[i] = orig

                a_temp = a * (S / s)
                c_temp = fft_convolve_self(a_temp)
                obj_temp = c_temp.max()
                if obj_temp < best_local_obj - 1e-12:
                    best_local_obj = obj_temp
                    best_local = orig * factor
            
            if best_local != orig:
                a[i] = best_local
                a = normalize_nonneg(a, S)
                c = fft_convolve_self(a)
                if c.max() < best_obj - 1e-12:
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
            else:
                a[i] = orig
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"  CD sweep {sweep}: obj={best_obj:.10f}, t_elapsed={elapsed:.1f}s")
        
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:
                break
        else:
            no_improve_sweeps = 0
    
    return best_a, best_obj


def run_subgradient(a_init, n, S, time_budget, T_init=50.0, T_final=2000.0,
                    lr_init=0.05, momentum=0.85, verbose=True):
    """Weighted subgradient descent with Nesterov momentum."""
    a = normalize_nonneg(a_init.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    a_prev = a.copy()
    beta = momentum
    start = time.monotonic()
    step = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.7)
        
        # Nesterov lookahead
        y = a + (beta - 1.0) / (1.0 + beta) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        else:
            y = a.copy()
        
        c_y = fft_convolve_self(y)
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            # Restart with noise
            noise = 0.01 * max(0.1, 1.0 - progress)
            a = best_a + noise * np.random.default_rng(step).standard_normal(n)
            a = normalize_nonneg(a, S)
            a_prev = a.copy()
            no_improve = 0
            step += 1
            continue
        
        # Compute weighted subgradient
        c_max = c.max()
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        
        # g[m] = 2 * sum_k w[k] * a[k-m] (cross-correlation)
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g = 2.0 * corr[:n]
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g_dir = g / gnorm
        
        # Backtracking line search
        lr = lr_init * (1.0 - 0.7 * progress)
        best_step_a = a.copy()
        
        for i in range(15):
            a_trial = a - lr * g_dir
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            
            if obj_trial < cur_obj * (1 - 1e-10):
                best_step_a = a_trial
                break
            lr *= 0.5
        
        a_prev = a.copy()
        a = best_step_a
        step += 1
        
        if verbose and step % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"  SG step {step}: obj={best_obj:.10f}, T={T:.1f}, t_elapsed={elapsed:.1f}s")
    
    return best_a, best_obj

"""
LP-based iterative peak reduction for self-convolution minimization.

Algorithm:
1. Start with initial non-negative sequence normalized to sum = sqrt(2n)
2. Iteratively solve LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0
   This finds the optimal direction to redistribute mass while keeping peak bounded
3. Line search to find safe step size
4. Polish with subgradient and coordinate descent
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def build_conv_matrix_sparsity(n):
    """Build sparsity structure for the convolution matrix A where A[k,j] = a[k-j]."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        rows_list.extend([k] * len(js))
        cols_list.extend(js)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def make_conv_matrix(a, rows, cols, n):
    """Create the sparse convolution matrix."""
    vals = a[rows.astype(np.int64) - cols.astype(np.int64)]
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M, rows, cols, n, time_limit=20):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0."""
    A = make_conv_matrix(a, rows, cols, n)
    
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    
    solvers_to_try = [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver, max_time=time_limit)
            if b.value is not None:
                return np.maximum(np.asarray(b.value).flatten(), 0.0)
        except Exception:
            continue
    return None


def line_search_step(a, b, S, M, n):
    """Find best step size t minimizing peak of ((1-t)*a + t*b)."""
    best_t = 0.0
    best_obj = M
    best_a = a.copy()
    
    for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
        a_new = (1.0 - t) * a + t * b
        a_new = normalize_nonneg(a_new, S)
        c_new = fft_convolve_self(a_new)
        obj_new = c_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_t = t
            best_a = a_new.copy()
    
    return best_a, best_obj, best_t


def run_lp_method(a_init, S, n, rows, cols, time_budget, verbose=True):
    """Run the LP-based iterative method."""
    a = normalize_nonneg(a_init.copy(), S)
    c = fft_convolve_self(a)
    best_M = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    iteration = 0
    no_improve = 0
    min_improve = 1e-9
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 5:
            break
        
        # Determine time limit for LP (leave room for line search)
        lp_time = min(25.0, remaining * 0.7)
        
        a_new, M_new, t_used = lp_single_iteration(a, S, n, rows, cols, lp_time)
        
        if M_new < best_M - min_improve:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        iteration += 1
        
        if verbose and iteration % 3 == 0:
            print(f"  LP iter {iteration}: M={best_M:.10f}, t={t_used:.4f}, "
                  f"elapsed={elapsed:.1f}s, no_improve={no_improve}")
        
        if no_improve > 40:
            if verbose:
                print(f"  LP: stall at iter {iteration}, M={best_M:.10f}")
            break
    
    return best_a, best_M


def lp_single_iteration(a, S, n, rows, cols, lp_time):
    """Single LP iteration: solve LP, line search, return updated a."""
    c = fft_convolve_self(a)
    M = c.max()
    
    if M < 1e-15:
        return a.copy(), M, 0.0
    
    b_val = solve_lp(a, M, rows, cols, n, time_limit=lp_time)
    
    if b_val is None or b_val.sum() < 1e-10:
        return a.copy(), M, 0.0
    
    # Normalize b to have sum S
    b_val = b_val * (S / b_val.sum())
    
    # Line search
    best_a, best_M, best_t = line_search_step(a, b_val, S, M, n)
    
    return best_a, best_M, best_t


def run_subgradient(a_init, n, S, time_budget, T_init=100.0, T_final=5000.0,
                    lr_init=0.04, momentum=0.85, verbose=True):
    """Weighted subgradient descent with Nesterov momentum."""
    a = normalize_nonneg(a_init.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    a_prev = a.copy()
    beta = momentum
    start = time.monotonic()
    step = 0
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.6)
        
        # Nesterov lookahead
        y = a + (beta - 1.0) / (1.0 + beta) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        else:
            y = a.copy()
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 80:
            noise = 0.008 * max(0.1, 1.0 - progress)
            a = best_a + noise * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            a_prev = a.copy()
            no_improve = 0
            step += 1
            continue
        
        # Compute weighted subgradient
        c_max = c.max()
        if c_max < 1e-15:
            break
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        
        # g[m] = 2 * cross_corr(w, a)[m]
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g = 2.0 * corr[:n]
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g_dir = g / gnorm
        
        # Backtracking line search
        lr = lr_init * (1.0 - 0.6 * progress)
        best_step_a = a.copy()
        
        for i in range(15):
            a_trial = a - lr * g_dir
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial < cur_obj * (1.0 - 1e-10):
                best_step_a = a_trial
                break
            lr *= 0.5
        
        a_prev = a.copy()
        a = best_step_a
        step += 1
        
        if verbose and step % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"  SG step {step}: obj={best_obj:.10f}, T={T:.1f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def coordinate_descent_polish(a, n, S, time_budget, verbose=True):
    """Coordinate descent: try adjusting individual coordinates."""
    a = normalize_nonneg(a.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(777)
    sweep = 0
    no_improve_sweeps = 0
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            orig = a[i]
            if orig < 1e-14:
                continue
            
            best_local = orig
            best_local_obj = best_obj
            
            for factor in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                new_val = orig * factor
                a[i] = new_val
                s = a.sum()
                if s < 1e-15:
                    a[i] = orig
                    continue
                a_temp = a * (S / s)
                c_temp = fft_convolve_self(a_temp)
                obj_temp = c_temp.max()
                if obj_temp < best_local_obj - 1e-12:
                    best_local_obj = obj_temp
                    best_local = new_val
            
            if best_local != orig:
                a[i] = best_local
                a = normalize_nonneg(a, S)
                c = fft_convolve_self(a)
                if c.max() < best_obj - 1e-12:
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
            else:
                a[i] = orig
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"  CD sweep {sweep}: obj={best_obj:.10f}, elapsed={elapsed:.1f}s")
        
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 4:
                break
        else:
            no_improve_sweeps = 0
    
    return best_a, best_obj


def run_adam_high_T(a_init, n, S, time_budget, T_init=500.0, T_final=5000.0,
                    lr_init=0.02, verbose=True):
    """Adam optimizer on log-parameters with very high temperature (peak-focused)."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-14)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    eps = 1e-8
    beta1, beta2 = 0.9, 0.999
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t_count += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.5)
        
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
        else:
            a = S * exp_theta / total
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        c_max = c.max()
        if c_max < 1e-15:
            break
        
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g_a = 2.0 * corr[:n]
        
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot
        
        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break
        
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)
        
        lr = lr_init * (0.3 + 0.7 * (1.0 - progress))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if verbose and t_count % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"  Adam step {t_count}: obj={best_obj:.10f}, T={T:.1f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    
    Uses LP-based peak reduction as the main algorithm, combined with
    subgradient descent and coordinate descent for polishing.
    """
    total_budget = max(1.0, float(budget_s) - 30.0)
    deadline = time.monotonic() + total_budget
    
    n = 4096
    S = np.sqrt(2.0 * n)
    global_best = np.inf
    global_a = None
    
    # Load given sequence
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            n_given = len(given_raw)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_raw * (S_given / given_raw.sum())
            ver = evaluate_sequence(given_a.tolist())
            print(f"Given: n={n_given}, obj={ver:.7f}")
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print(f"Could not load given: {e}")
    
    # Precompute convolution matrix sparsity
    print(f"Building convolution matrix sparsity for n={n}...")
    t0 = time.monotonic()
    rows, cols = build_conv_matrix_sparsity(n)
    print(f"  Done in {time.monotonic()-t0:.2f}s, {len(rows)} non-zeros")
    
    # Prepare starting point
    if global_a is not None and len(global_a) == n:
        a_start = global_a.copy()
    elif global_a is not None:
        # Interpolate to length n
        a_start = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(global_a)), global_a)
        a_start = normalize_nonneg(a_start, S)
    else:
        i = np.arange(n)
        a_start = (1.0 + 0.2 * np.cos(2 * np.pi * 3 * i / n) +
                   0.1 * np.cos(2 * np.pi * 7 * i / n + 0.7) +
                   0.05 * np.cos(2 * np.pi * 13 * i / n + 1.3))
        a_start = np.maximum(a_start, 0.01)
        a_start = a_start * (S / a_start.sum())
    
    # ============ PHASE 1: LP-based optimization (50% budget) ============
    lp_budget = 0.50 * total_budget
    print(f"\nPhase 1: LP optimization (budget={lp_budget:.0f}s)")
    
    a_best, M_best = run_lp_method(a_start, S, n, rows, cols, lp_budget, verbose=True)
    a_best = normalize_nonneg(a_best, S)
    obj = evaluate_sequence(a_best.tolist())
    if np.isfinite(obj) and obj < global_best:
        global_best = obj
        global_a = a_best.copy()
        print(f"  Phase 1 result: obj={obj:.7f}")
    
    # ============ PHASE 2: Subgradient high-T (20% budget) ============
    remaining = deadline - time.monotonic()
    sg_budget = min(0.20 * total_budget, remaining * 0.5)
    
    if global_a is not None and sg_budget > 30:
        n_cur = len(global_a)
        S_cur = np.sqrt(2.0 * n_cur)
        print(f"\nPhase 2: Subgradient (budget={sg_budget:.0f}s)")
        
        a_best, obj_best = run_subgradient(
            global_a, n_cur, S_cur, sg_budget,
            T_init=200, T_final=5000, lr_init=0.03, momentum=0.85
        )
        a_best = normalize_nonneg(a_best, S_cur)
        obj = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_best.copy()
            print(f"  Phase 2 result: obj={obj:.7f}")
    
    # ============ PHASE 3: Adam high-T (15% budget) ============
    remaining = deadline - time.monotonic()
    adam_budget = min(0.15 * total_budget, remaining * 0.6)
    
    if global_a is not None and adam_budget > 30:
        n_cur = len(global_a)
        S_cur = np.sqrt(2.0 * n_cur)
        print(f"\nPhase 3: Adam high-T (budget={adam_budget:.0f}s)")
        
        a_best, obj_best = run_adam_high_T(
            global_a, n_cur, S_cur, adam_budget,
            T_init=500, T_final=10000, lr_init=0.015
        )
        a_best = normalize_nonneg(a_best, S_cur)
        obj = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_best.copy()
            print(f"  Phase 3 result: obj={obj:.7f}")
    
    # ============ PHASE 4: Coordinate descent polish (10% budget) ============
    remaining = deadline - time.monotonic()
    cd_budget = min(0.10 * total_budget, remaining * 0.7)
    
    if global_a is not None and cd_budget > 20:
        n_cur = len(global_a)
        S_cur = np.sqrt(2.0 * n_cur)
        print(f"\nPhase 4: CD polish (budget={cd_budget:.0f}s)")
        
        a_best, obj_best = coordinate_descent_polish(global_a, n_cur, S_cur, cd_budget)
        a_best = normalize_nonneg(a_best, S_cur)
        obj = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_best.copy()
            print(f"  Phase 4 result: obj={obj:.7f}")
    
    # ============ PHASE 5: Final LP round if time remains ============
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 40 and len(global_a) == n:
        S_cur = np.sqrt(2.0 * n)
        lp2_budget = remaining - 10
        print(f"\nPhase 5: Final LP (budget={lp2_budget:.0f}s)")
        
        a_best, M_best = run_lp_method(global_a, S_cur, n, rows, cols, lp2_budget, verbose=True)
        a_best = normalize_nonneg(a_best, S_cur)
        obj = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_best.copy()
            print(f"  Phase 5 result: obj={obj:.7f}")
    
    # Final CD if any time left
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 10:
        n_cur = len(global_a)
        S_cur = np.sqrt(2.0 * n_cur)
        a_best, _ = coordinate_descent_polish(global_a, n_cur, S_cur, remaining - 5, verbose=False)
        a_best = normalize_nonneg(a_best, S_cur)
        obj = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_best.copy()
    
    # Final verification
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: obj={ver:.7f}, n={len(final_a)}")
    
    return final_a
