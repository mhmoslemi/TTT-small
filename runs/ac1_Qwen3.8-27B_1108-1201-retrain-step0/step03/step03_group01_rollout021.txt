    n = len(v)

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using adaptive subgradient + spectral shaping + coordinate descent.
Key improvements: FFT-based gradient with backtracking line search, multi-scale optimization, 
chirp-based initialization, and coordinate descent polish.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a, fsize=None):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    if fsize is None:
        fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_gradient(a, c, top_idx, top_weights, N):

    Compute subgradient using FFT.
    a: sequence of length n
    c: convolution array (length 2n-1)
    top_idx: indices of top lags
    top_weights: weights for each top lag (normalized to sum to 1)
    N: FFT size (>= 3n-2 to avoid wrap-around)
    Returns gradient of weighted sum of top lags with respect to a.
    """
    n = len(a)
    # Build the "effective cosine" in frequency domain
    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += top_weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
    
    # gradient = ifft(fft(a_padded) * 2 * cos_eff) [:n]
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)

    return grad_full[:n]


def backtrack_line_search(a, grad, S, fsize, max_step, min_step, c0, max_b0):
    """
    Backtracking line search for the subgradient step.
    a: current point
    grad: subgradient
    S: target sum
    fsize: FFT size
    max_step: maximum step size
    min_step: minimum step size
    c0: current convolution
    max_b0: current max of convolution
    Returns the best step size and new point.
    """
    n = len(a)
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return 0.0, a.copy()
    
    step = max_step
    best_step = 0.0
    best_a = a.copy()
    best_max_b = max_b0
    
    for _ in range(10):
        a_try = project_simplex(a - step * grad / grad_norm, S)
        c_try = compute_conv(a_try, fsize)
        max_b_try = c_try.max()
        
        if max_b_try < best_max_b:
            best_max_b = max_b_try
            best_a = a_try.copy()
            best_step = step
        
        if max_b_try < max_b0:
            return step, a_try
        
        step *= 0.5
        if step < min_step:
            break
    
    return best_step, best_a

            best_a = a.copy()

            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
        
        # Deflate
        P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new, S)
        
        # Occasional noise
        if it % 100 == 50 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
    
    return best_a, best_max_b


def subgradient_descent(a_init, n, S, N, max_time, rng, 
                        top_k_init=5, top_k_max=512, 
                        max_step_init=0.01, noise_interval=500, noise_prob=0.1):
    """
    Subgradient descent with FFT gradient, backtracking line search, and Nesterov acceleration.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    a_prev = a.copy()
    t = 1.0
    y = a.copy()  # Nesterov momentum variable
    start = time.monotonic()
    best_a = a.copy()
    best_max_b = np.inf
    it = 0
    fsize = N  # Use the same FFT size
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)
        
        # Compute convolution at y (Nesterov point)
        c = compute_conv(y, fsize)
        max_b = c.max()
        
        if max_b < best_max_b:
            best_max_b = max_b
            best_a = y.copy()
        
        # Get top-k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        
        weights = top_vals / total
        
        # Compute subgradient
        grad = fft_gradient(y, c, top_idx, weights, fsize)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        # Adaptive step size
        # Step size should be proportional to S / (n * something)
        # Use a geometrically decreasing schedule
        step_init = max_step_init * S / (1.0 + 0.005 * it)
        
        # Backtracking line search
        # Try a few step sizes and pick the best
        best_step = 0.0
        best_a_new = a.copy()
        step = step_init
        for ls in range(8):
            a_new = project_simplex(y - step * grad / grad_norm, S)
            c_new = compute_conv(a_new, fsize)
            max_b_new = c_new.max()
            
            if max_b_new < best_max_b:
                best_max_b = max_b_new
                best_a = a_new.copy()
            
            if max_b_new < max_b:
                best_step = step
                best_a_new = a_new
                break
            
            step *= 0.5
            if step < 1e-8:
                break
        
        if best_step == 0.0:
            # No improvement found, use a very small step
            best_a_new = project_simplex(y - 1e-6 *egrad / max(grad_norm, 1e-15), S)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = best_a_new + ((t - 1.0) / t_new) * (best_a_new - a_prev)
        y = np.maximum(y, 0.0)
        # Renormalize y
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        a_prev = a
        a = best_a_new
        t = t_new
        
        # Noise injection
        if it % noise_interval == 0 and rng.random() < noise_prob:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0  # Reset momentum
    
    return best_a, best_max_b


def coordinate_descent_polish(a_init, n, S, N, max_time, rng, n_passes=None):
    """
    Coordinate descent: try adjusting pairs of elements to reduce max autocorrelation.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_a = a.copy()
    best_max_b = compute_conv(a, N).max()
    it_pass = 0
    
    while time.monotonic() - start < max_time:
        it_pass += 1
        improved = False
        
        # Random subset of positions to try (for speed)
        n_try = min(n, 5000)
        positions = rng.choice(n, size=n_try, replace=False)
        
        for pos in positions:
            if time.monotonic() - start > max_time:
                break
            
            j = int(positions)
            # Try moving "mass" from position j to position k
            # or from k to j
            
            # Find a partner position
            # Try the position that's most "dissimilar" (for diversity)
            k = int(rng.integers(0, n))
            if k == j:
                k = (j + 1) % n
            
            # Try a small transfer
            delta = 0.01 * S / n  # Small transfer amount
            
            # Option 1: move from j to k
            if a[j] > delta:
                a_try = a.copy()
                a_try[j] -= delta
                a_try[k] += delta
                c_try = compute_conv(a_try, N)
                if c_try.max() < best_max_b:
                    best_max_b = c_try.max()
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
            
            # Option 2: move from k to j
            if a[k] > delta:
                a_try = a.copy()
                a_try[k] -= delta
                a_try[j] += delta
                c_try = compute_conv(a_try, N)
                if c_try.max() < best_max_b:
                    best_max_b = c_try.max()
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
        
        if not improved and it_pass > 5:
            break
    
    return best_a, best_max_b


def chirp_init(n, rng):
    """Create a chirp-based initialization."""
    t = np.arange(n) / n
    # Quadratic chirp
    freq = 3.0 + rng.random() * 5.0
    beta = 10.0 + rng.random() * 20.0
    signal = np.sin(2 * np.pi * (freq * t + beta * t * t))
    a = np.abs(signal) + 0.05
    return a


def multi_sinusoid_init(n, rng):
    """Create a multi-sinusoid initialization."""
    t = np.arange(n) / n
    a = np.zeros(n)
    for _ in range(3 + int(rng.integers(0, 4))):
        f = 1.0 + rng.random() * 15.0
        phi = rng.random() * 2 * np.pi
        amp = 0.3 + rng.random() * 0.7
        a += amp * np.abs(np.sin(2 * np.pi * f * t + phi))
    a += 0.05
    return a


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    global_a = None

    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (3 * n - 1).bit_length()  # Large enough FFT size to avoid wrap-around

    inits.append(("msin2", multi_sinusoid_init(n, rng)))

    inits.append(("rand", rng.random(n) + 0.05))
    
    # Edge-weighted (heavier at edges to compensate for triangular bias)
