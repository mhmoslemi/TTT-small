"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive subgradient with 
Nesterov momentum, diverse restarts, spectral-domain perturbations,
and multi-length exploration. Novel features: adaptive lag weighting,
frequency-domain flat-top adjustments, and paired coordinate swaps.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def subgradient_topk(a, c, K, lag_weights=None):
    """Compute subgradient using top-K lags with given weights.
    If lag_weights is None, uses uniform weights."""
    n = len(a)
    k = min(K, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    if lag_weights is None:
        lag_weights = np.ones(k) / k
    else:
        lag_weights = lag_weights[:k]
        lag_weights = lag_weights / lag_weights.sum()
    
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = lag_weights[i]
    
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def adaptive_lag_weights(c, K, alpha=1.0):
    """Compute weights for top-K lags based on how much they exceed the mean."""
    top_vals = np.sort(c)[::-1][:K]
    remaining = np.sort(c)[K:]
    if len(remaining) > 0:
        threshold = remaining.mean() + alpha * remaining.std()
    else:
        threshold = top_vals.min()
    excess = np.maximum(top_vals - threshold, 0.01)
    weights = excess ** 2
    return weights / weights.sum()


def run_subgradient_nesterov(a_init, n, S, max_time, rng, 
                              K_init=5, K_max=200, 
                              lr_init=None, momentum=0.9):
    """Subgradient Nesterov method with adaptive K and step size."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stall = 0
    last_improve = np.inf
    step = lr_init if lr_init else 2.0 * S / (n * 10.0)

    while time.monotonic() - start < max_time:
        it += 1
        K = min(K_max, K_init + it // 100)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stall = 0
            last_improve = cur
        elif it % 30 == 0:
            if cur < last_improve - 1e-12:
                last_improve = cur
                stall = 0
            else:
                stall += 1

        # Compute subgradient
        if K <= 3:
            grad = subgradient_topk(y, c, K)
        else:
            weights = adaptive_lag_weights(c, K, alpha=0.5)
            grad = subgradient_topk(y, c, K, lag_weights=weights)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Adaptive step size
        step = 2.0 * S / (n * (1.0 + 0.001 * it) * max(gnorm / (S / n), 0.05))
        step = min(step, 0.1 * S / np.sqrt(n))

        x_new = project_simplex(y - step * grad, S)

        # Periodic verification
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
            if c_x.max() > cur * 1.05:
                step *= 0.7

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)

        t_momentum = t_new

            t_momentum = 1.0

            step = 2.0 * S / (n * 10.0)

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = 10.0 / S
    stall = 0
    last_improve = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        progress = (time.monotonic() - start) / max_time
        beta = beta_start * (beta_end / beta_start) ** progress

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stall = 0
            last_improve = cur
        elif it % 10 == 0:
            if cur < last_improve - 1e-12:
                last_improve = cur
                stall = 0
            else:
                stall += 1

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = 1.0 / L
        c_y = c
        f_y = c_y.max()
        x_new = None
        for bt in range(4):
            x_candidate = project_simplex(y - step * grad, S)
            if bt > 0:
                c_cand = fft_convolve_self(x_candidate)
                f_cand = c_cand.max()
                if f_cand <= f_y + 1e-10:
                    x_new = x_candidate
                    break
                step *= 0.5
            else:
                x_new = x_candidate

        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        # Adaptive L
        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() > f_y * 1.01:
                L *= 1.3
            elif c_x.max() < f_y * 0.99:
                L = max(L * 0.97, beta * 0.1 / S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        if stall > 40:
            noise = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_momentum = 1.0
            stall = 0

    return best_local, best_score


def spectral_flat_perturb(a, n, S, rng, strength=0.05):
    """Perturb in frequency domain to flatten the spectral magnitude.
    This targets reducing peak in autocorrelation."""
    fsize = 1 << (2 * n - 2).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = np.angle(A)
    
    # Identify frequency bins with high magnitude (correspond to peak autocorr)
    mean_mag = mag.mean()
    # Attenuate high bins, boost low bins
    target = np.full_like(mag, mean_mag)
    new_mag = mag + strength * (target - mag)
    new_mag = np.maximum(new_mag, 0.01 * mean_mag)
    
    # Small random perturbation
    new_mag *= (1.0 + strength * 0.1 * rng.standard_normal(fsize))
    new_mag = np.maximum(new_mag, 0.0)
    
    A_new = new_mag * np.exp(1j * phase)
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() > 1e-15:
        a_new = a_new * (S / a_new.sum())
    else:
        a_new = np.ones(n) * S / n
    return a_new


def paired_swap_optimize(a, n, S, max_time, rng):
    """Paired coordinate swap: find pairs (i,j) where swapping small amounts
    reduces the max convolution lag."""
    a = np.clip(a_init := a, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Find top 5 lags
        top_lags = np.argsort(c)[::-1][:5]
        
        # For each top lag, find the pair of indices contributing most
        improved = False
        for lag in top_lags:
            lag = int(lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue
            indices = np.arange(i_min, i_max + 1)
            partner = lag - indices
            products = a[indices] * a[partner]
            
            # Find the pair with largest product
            top_idx = np.argmax(products)
            i = int(indices[top_idx])
            j = int(partner[top_idx])
            
            # Try to reduce the product by transferring from the larger to a small element
            if a[i] > a[j]:
                big, small = i, j
            else:
                big, small = j, i
            
            # Find a small element to receive the transfer
            small_elements = np.argsort(a)[:20]
            for target in small_elements:
                if target == big or target == small:
                    continue
                # Try transfer
                delta = 0.01 * a[big]
                a_trial = a.copy()
                a_trial[big] -= delta
                a_trial[target] += delta
                if a_trial[big] < 0:
                    continue
                a_trial = project_simplex(a_trial, S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur:
                    a = a_trial
                    cur = c_trial.max()
                    improved = True
                    break
                if improved:
                    break
        
        if not improved:
            # If no single swap helps, try a small random perturbation
            noise = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)
    
    return best_local, best_score


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = mu."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-15:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_periodic_init(n, rng, S, num_freqs=3):
    """Periodic initialization with sinusoidal components."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 20)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.4
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a * (S / a.sum())
    return a


def make_w_shape_init(n, rng, S):
    """W-shaped initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    k = rng.integers(2, 6)
    a = np.abs(np.sin(2 * np.pi * k * tt)) + 0.05
    a = a * (S / a.sum())
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    S = np.sqrt(2.0 * n)
    inits = []
    
    if parent is not None and len(parent) >= 64:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if p.sum() > 0:
            inits.append(("parent", p.copy()))
            # Scaled versions
            inits.append(("parent_scaled", p * (S / p.sum())))
    
    for i in range(2):
        inits.append(("anti_pal_%d" % i, make_anti_palindromic_init(n, rng, S)))
    for i in range(2):
        inits.append(("periodic_%d" % i, make_periodic_init(n, rng, S)))
    inits.append(("wshape", make_w_shape_init(n, rng, S)))
    
    tt = np.linspace(0, 1, n, endpoint=False)
    for freq in [5, 11]:
        sin_a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * tt)
        sin_a = np.maximum(sin_a, 0.05)
        inits.append(("sin_%d" % freq, sin_a * (S / sin_a.sum())))
    
    rand_a = rng.random(n) + 0.05
    inits.append(("random", rand_a * (S / rand_a.sum())))
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None
    global_n = 0

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    primary_n = 8192
    if parent is not None and len(parent) >= 64:
        primary_n = len(parent)

    print("Primary length: %d, parent available: %s" % (primary_n, parent is not None))

    def update_best(a, name=""):
        nonlocal global_best, global_a, global_n
        n = len(a)
        S = np.sqrt(2.0 * n)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a * (S / s)
        c = fft_convolve_self(a_norm)
        obj = 2.0 * n * c.max() / S**2
        if obj < global_best:
            global_best = obj
            global_a = a_norm.copy()
            global_n = n
            print("  %s: obj=%.6f (n=%d)" % (name, global_best, n))

    # === Phase 1: Multi-start exploration ===
    inits = generate_initializations(primary_n, rng, parent)
    n_inits = len(inits)
    usable_time = max(10.0, deadline - time.monotonic())
    per_init_time = usable_time * 0.55 / n_inits

    print("=== Phase 1: %d inits, %.1f sec each ===" % (n_inits, per_init_time))

    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break
        n = len(arr)
        if n < 64:
            continue
        S = np.sqrt(2.0 * n)
        t0 = time.monotonic()

        # Subgradient Nesterov
        seg = per_init_time * 0.55
        a_sub, s_sub = run_subgradient_nesterov(arr, n, S, seg, rng,
                                                 K_init=3, K_max=150)
        update_best(a_sub, "SubGrad[%s]" % name)

        # LSE FISTA
        rem = per_init_time - (time.monotonic() - t0)
        if rem > 3:
            a_f, s_f = run_lse_fista(a_sub, n, S, rem, rng,
                                      beta_start=5.0, beta_end=200.0)
            update_best(a_f, "FISTA[%s]" % name)

    print("=== After Phase 1: best=%.6f, n=%d ===" % (global_best, global_n))

    # === Phase 2: Longer sequence exploration ===
    if time.monotonic() < deadline - 200:
        for n_try in [primary_n * 2, primary_n // 2]:
            if n_try < 1024 or n_try > 32768:
                continue
            if time.monotonic() > deadline - 100:
                break
            S_try = np.sqrt(2.0 * n_try)
            print("=== Phase 2: Trying n=%d ===" % n_try)
            
            if parent is not None and len(parent) >= 64:
                from numpy import interp
                tt_old = np.linspace(0, 1, len(parent))
                tt_new = np.linspace(0, 1, n_try)
                init_try = interp(tt_new, tt_old, parent)
                init_try = np.maximum(init_try, 0.0)
                if init_try.sum() > 0:
                    init_try = init_try * (S_try / init_try.sum())
            else:
                init_try = make_anti_palindromic_init(n_try, rng, S_try)

            seg = min(120.0, max(20.0, (deadline - time.monotonic()) * 0.15))
            a_try, s_try = run_subgradient_nesterov(init_try, n_try, S_try, seg, rng,
                                                     K_init=5, K_max=100)
            update_best(a_try, "LongSeq(n=%d)" % n_try)
            
            rem = max(10.0, 120.0 - (time.monotonic() - (deadline - time.monotonic() + seg)))
            if rem > 10 and time.monotonic() < deadline - 60:
                a_try2, _ = run_lse_fista(a_try, n_try, S_try, min(rem, 60.0), rng,
                                           beta_start=20.0, beta_end=300.0)
                update_best(a_try2, "FISTA_LongSeq")

    print("=== After Phase 2: best=%.6f, n=%d ===" % (global_best, global_n))

    # === Phase 3: Intensive polishing of best solution ===
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        print("=== Phase 3: Polishing (remaining %.0f sec) ===" % remaining)

        # 3a: High-K subgradient with adaptive weights
        seg3a = max(10.0, remaining * 0.25)
        a_p, s_p = run_subgradient_nesterov(global_a, n, S, seg3a, rng,
                                             K_init=10, K_max=300)
        update_best(a_p, "Polish_SubGrad_hiK")

        # 3b: LSE FISTA with high beta
        remaining2 = deadline - time.monotonic()
        seg3b = max(10.0, remaining2 * 0.25)
        a_p2, s_p2 = run_lse_fista(global_a if global_a is not None else a_p, n, S,
                                    seg3b, rng, beta_start=50.0, beta_end=800.0)
        update_best(a_p2, "Polish_FISTA_hiBeta")

        # 3c: Spectral perturbation + re-optimize
        remaining3 = deadline - time.monotonic()
        if remaining3 > 30:
            for _ in range(3):
                if time.monotonic() > deadline - 30:
                    break
                a_spec = spectral_flat_perturb(global_a, n, S, rng, strength=0.03)
                seg_spec = max(10.0, min(60.0, (deadline - time.monotonic()) * 0.1))
                a_sp, _ = run_subgradient_nesterov(a_spec, n, S, seg_spec, rng,
                                                    K_init=5, K_max=100)
                update_best(a_sp, "Spectral_perturb")

        # 3d: Paired swap optimization
        remaining4 = deadline - time.monotonic()
        if remaining4 > 20:
            a_swap, s_swap = paired_swap_optimize(global_a, n, S,
                                                   min(60.0, remaining4 * 0.3), rng)
            update_best(a_swap, "PairedSwap")

        # 3e: Final intensive polish
        remaining5 = deadline - time.monotonic()
        if remaining5 > 10:
            a_final, _ = run_subgradient_nesterov(global_a, n, S, remaining5 * 0.8, rng,
                                                   K_init=20, K_max=500)
            update_best(a_final, "Final_Polish")

    # === Phase 4: Novel exploration with different structure ===
    if time.monotonic() < deadline - 30:
        print("=== Phase 4: Novel exploration ===")
        # Try a sequence with a specific "chirp" structure
        n_chirp = primary_n
        S_chirp = np.sqrt(2.0 * n_chirp)
        tt = np.arange(n_chirerp := n_chirp) / n_chirp
        chirp = np.exp(1j * 2 * np.pi * (3 * tt**2))
        a_chirp = np.abs(chirp) + 0.1  # This is just 1.1 everywhere... 
        # Better: use the real part pattern
        a_chirp = 1.0 + 0.3 * np.cos(2 * np.pi * (5 * tt + 2 * tt**2))
        a_chirp = np.maximum(a_chirp, 0.05)
        a_chirp = a_chirp * (S_chirp / a_chirp.sum())
        
        seg_chirp = min(60.0, max(10.0, (deadline - time.monotonic()) * 0.15))
        a_c, _ = run_subgradient_nesterov(a_chirp, n_chirp, S_chirp, seg_chirp, rng,
                                           K_init=5, K_max=100)
        update_best(a_c, "Chirp_struct")

        # Try "block" structure: alternating high/low blocks
        n_block = primary_n
        S_block = np.sqrt(2.0 * n_block)
        block_size = max(4, n_block // 64)
        a_block = np.zeros(n_block)
        for i in range(0, n_block, block_size):
            val = 1.0 + 0.5 * rng.random()
            a_block[i:i+block_size] = val
        a_block = a_block * (S_block / a_block.sum())
        
        remaining_blk = deadline - time.monotonic()
        if remaining_blk > 20:
            seg_blk = min(40.0, remaining_blk * 0.3)
            a_b, _ = run_subgradient_nesterov(a_block, n_block, S_block, seg_blk, rng,
                                               K_init=3, K_max=80)
            update_best(a_b, "Block_struct")

    # === Final verification ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Verification: %.6f, internal=%.6f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: evaluate_sequence rejected, using fallback")
        global_a = np.ones(1024, dtype=np.float64) * 1.0
        final_list = [float(x) for x in global_a]
        ver2 = evaluate_sequence(final_list)
        print("  Fallback: %.6f" % (ver2 if np.isfinite(ver2) else float('inf')))

    return final_list
