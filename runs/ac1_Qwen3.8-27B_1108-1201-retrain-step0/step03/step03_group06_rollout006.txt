        return a * (S / a.sum())
    elif name == "gaussian":
        a = np.exp(-0.5 * ((i - n/2) / (n/8)) ** 2)
        a = a + 0.001
        return a * (S / a.sum())
    elif name == "cosine":
        freqs = rng.integers(1, 20, 3)
        phases = rng.random(3) * 2 * np.pi
        a = 1.0
        for f, p in zip(freqs, phases):
            a += 0.2 * np.cos(2 * np.pi * f * i / n + p)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "bipolar_smoothed":
        # Start from a bipolar-like pattern and smooth
        a = np.where(np.arange(n) % 2 == 0, 1.0, 0.1)
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.001)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses adaptive smooth gradient with Adam, L-BFGS polishing, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 30.0)
    deadline = time.monotonic() + total_budget
    
    global_best = np.inf
    global_a = None
    
    # Try to load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given sequence: " + str(e))
    
    # Determine n
    n = 4096
    S = np.sqrt(2.0 * float(n))

            global_best = obj

            print("  Start from given: obj=%.7f" % obj)
    
    for init_name in init_names:
        if time.monotonic() > phase1_deadline - 5:
            break
        remaining = phase1_deadline - time.monotonic()
        per_init_time = min(per_init_time, remaining / max(1, len(init_names)))
        if per_init_time < 3:
            break
