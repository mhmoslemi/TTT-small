    a = np.maximum(a, 1e-300)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = compute_obj(a, c)
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)
    eta = eta0
    iter_no = 0
    start = time.monotonic()
    stagnation = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        if use_softmax:
            beta = beta0 + (beta_max - beta0) * min(1.0, progress * 1.5)
            grad = softmax_subgrad(c, a, beta)
        else:
            grad = topk_subgrad(c, a, top_k)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        a_new = a * np.exp(-eta * v)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new * (S / a_new.sum())

        c_new = fft_convolve_self(a_new)
        obj_new = compute_obj(a_new, c_new)

        if obj_new < best_obj:
            a = a_new
            c = c_new
            best_obj = obj_new
            best_a = a.copy()
            best_iter = iter_no
            stagnation = 0
            eta = min(eta * 1.015, eta0 * 3.0)
        else:
            # Accept with probability that decreases with stagnation
            if stagnation < 50 or np.random.random() < 0.3:
                a = a_new
                c = c_new
            stagnation += 1
            if stagnation > 80:
                eta = max(eta * 0.92, eta0 * 0.05)

        # Structured annealing
        if iter_no > 0 and iter_no % anneal_every == 0:
            temp = anneal_scale * (1.0 - 0.7 * progress)
            if temp > 0.01:
                # Gradient-direction perturbation with randomization
                perturbation = v * (temp * 0.5) + np.random.randn(n) * (temp * S / np.sqrt(n))
                a = a + perturbation
                a = np.maximum(a, 1e-300)
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                v *= 0.2
                eta = eta0 * 0.8
                stagnation = 0

        iter_no += 1
        if verbose and iter_no % 2000 == 0:
            cur_obj = compute_obj(a, c)
            print(f"    iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"eta={eta:.4f}, stag={stagnation}, t={elapsed:.1f}s")

    return best_a, best_obj


def peak_pair_reduction(a0, S, time_budget, verbose=False):
    """Directly reduce the worst peak by adjusting specific coordinate pairs."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = compute_obj(a, c)
    best_a = a.copy()
    start = time.monotonic()
    iters = 0

    while time.monotonic() - start < time_budget:
        k_star = np.argmax(c)
        c_max = c[k_star]
        c_mean = (S * S) / (2 * n - 1)

        # Find contributing pairs
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        indices = np.arange(lo, hi + 1)
        products = a[indices] * a[k_star - indices]

        # Reduce the top contributing pairs
        top_pairs = np.argsort(products)[::-1][:min(20, len(indices))]

        improved = False
        for tp in top_pairs:
            idx = indices[tp]
            partner = k_star - idx
            orig = a[idx]
            if orig < 1e-12:
                continue
            # Try reducing a[idx]
            for factor in [0.6, 0.8, 0.9]:
                a[idx] = orig * factor
                total = a.sum()
                if total < 1e-10:
                    a[idx] = orig
                    break
                a_norm = a * (S / total)
                c_new = fft_convolve_self(a_norm)
                obj_new = compute_obj(a_new, c_new)
                if obj_new < best_obj:
                    a = a_norm.copy()
                    c = c_new
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    break
            a[idx] = np.maximum(a[idx], 0)
            total = a.sum()
            if total > 0:
                a = a * (S / total)

        iters += 1
        if verbose and iters % 20 == 0:
            print(f"    peak_pair iter={iters}, obj={best_obj:.6f}, t={time.monotonic()-start:.1f}s")

        if not improved and iters > 10:
            break

    return best_a, best_obj


def coordinate_polish(a0, S, time_budget, verbose=False):
    """Coordinate descent polishing."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = compute_obj(a, c)
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(123)

    sweep = 0
    while time.monotonic() - start < time_budget:
        improved = False
        indices = rng.permutation(n)
        for i in indices:
            if time.monotonic() - start >= time_budget:
                break
            orig = a[i]
            if orig < 1e-15:
                continue
            for factor in [0.6, 0.8, 1.2, 1.4]:
                a[i] = orig * factor
                total = a.sum()
                if total < 1e-10:
                    a[i] = orig
                    break
                a_norm = a * (S / total)
                c_new = fft_convolve_self(a_norm)
                max_new = c_new.max()
                max_old = c.max()
                if max_new < max_old * 0.999:
                    a = a_norm.copy()
                    c = c_new
                    best_obj = compute_obj(a, c)
                    best_a = a.copy()
                    improved = True
                    break
            a[i] = np.maximum(a[i], 0)
            total = a.sum()
            if total > 0:
                a = a * (S / total)

        sweep += 1
        if verbose and sweep % 10 == 0:
            print(f"    coord sweep={sweep}, obj={best_obj:.6f}, t={time.monotonic()-start:.1f}s")
        if not improved and sweep > 5:
            break

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400.0):
    """Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 12.0)

    global_best = np.inf
    global_a = None

    # --- Load given sequence ---
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            n_g = len(given_raw)
            S_g = np.sqrt(2.0 * n_g)
            given_a = given_raw * (S_g / given_raw.sum())
            val = evaluate_sequence(given_a.tolist())
            print(f"Given: n={n_g}, obj={val:.6f}")
            if np.isfinite(val) and val < global_best:
                global_best = val
                global_a = given_a.copy()
    except Exception as e:
        print(f"No given sequence: {e}")

    # --- Generate initializations ---
    n_main = 4096
    S_main = np.sqrt(2.0 * n_main)
    idx = np.arange(n_main)

    initializations = []
    if given_a is not None and len(given_a) == n_main:
        initializations.append(('given', given_a.copy()))

    # Cosine with various frequencies and amplitudes
    for freq in [5, 6, 7, 8, 9]:
        for amp in [0.3, 0.5, 0.7]:
            a_cos = 1.0 + amp * np.cos(2 * np.pi * freq * idx / n_main)
            a_cos = np.maximum(a_cos, 0.01)
            a_cos = a_cos * S_main / a_cos.sum()
            initializations.append((f'c{freq}a{amp:.1f}', a_cos))

    # Random with structure
    a_r1 = np.maximum(rng.random(n_main), 0.01)
    a_r1 = a_r1 * S_main / a_r1.sum()
    initializations.append(('rand1', a_r1))

    # Two humps
    a_2h = (np.exp(-0.5 * ((idx - n_main * 0.25) / (n_main * 0.08)) ** 2) +
            np.exp(-0.5 * ((idx - n_main * 0.75) / (n_main * 0.08)) ** 2))
    a_2h = np.maximum(a_2h, 0.01)
    a_2h = a_2h * S_main / a_2h.sum()
    initializations.append(('2hump', a_2h))

    # Three humps
    a_3h = (np.exp(-0.5 * ((idx - n_main * 0.2) / (n_main * 0.07)) ** 2) +
            np.exp(-0.5 * ((idx - n_main * 0.5) / (n_main * 0.07)) ** 2) +
            np.exp(-0.5 * ((idx - n_main * 0.8) / (n_main * 0.07)) ** 2))
    a_3h = np.maximum(a_3h, 0.01)
    a_3h = a_3h * S_main / a_3h.sum()
    initializations.append(('3hump', a_3h))

    # --- Phase 1: Quick screening (8% time) ---
    screen_time = 0.08 * (deadline - time.monotonic())
    per_screen = max(1.0, screen_time / len(initializations))
    print(f"\nPhase 1: Screening {len(initializations)} inits ({screen_time:.0f}s)")

    screen_results = []
    for name, a_init in initializations:
        if time.monotonic() > deadline - 120:
            break
        a_best, obj_best = mirror_descent_v2(
            a_init, min(per_screen, deadline - time.monotonic() - 120),
            beta0=5, beta_max=30, eta0=0.35, momentum=0.88,
            anneal_every=40, anneal_scale=0.15, verbose=False, use_softmax=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
        screen_results.append((obj_best, a_best))
        print(f"  {name}: {obj_best:.6f}")

    # Keep top 3 from screening
    screen_results.sort(key=lambda x: x[0])
    top_candidates = [x[1] for x in screen_results[:3]]
    if global_a is not None and global_best < screen_results[0][0]:
        top_candidates.insert(0, global_a.copy())

    # --- Phase 2: Intensive optimization (55% time) ---
    main_time = 0.55 * (deadline - time.monotonic())
    print(f"\nPhase 2: Intensive optimization ({main_time:.0f}s)")

    if global_a is not None and main_time > 20:
        # Run on the best candidate with multiple parameter sets
        passes = [
            (5, 50, 0.3, 0.90, 200, 0.12, True, 0, 'broad_softmax'),
            (10, 70, 0.25, 0.92, 150, 0.08, True, 0, 'med_softmax'),
            (20, 100, 0.15, 0.94, 120, 0.05, True, 0, 'tight_softmax'),
            (0, 0, 0.2, 0.91, 180, 0.10, False, 100, 'topk_hard'),
        ]

        sub_time = main_time / len(passes)
        for beta0, beta_max, eta0, mom, anneal_e, anneal_s, use_sm, tk, label in passes:
            if time.monotonic() > deadline - 60:
                break
            print(f"  [{label}]")
            a_best, obj_best = mirror_descent_v2(
                global_a, sub_time, beta0=beta0, beta_max=beta_max,
                eta0=eta0, momentum=mom, anneal_every=anneal_e,
                anneal_scale=anneal_s, verbose=True,
                use_softmax=use_sm, top_k=tk
            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print(f"    -> {obj_best:.6f} (NEW BEST)")
            else:
                print(f"    -> {obj_best:.6f}")

        # Try second-best candidate briefly
        if len(top_candidates) > 1 and time.monotonic() < deadline - 60:
            print(f"  [try candidate 2]")
            a_best, obj_best = mirror_descent_v2(
                top_candidates[1], 30.0,
                beta0=10, beta_max=60, eta0=0.25, momentum=0.92,
                anneal_every=150, anneal_scale=0.08, verbose=False, use_softmax=True
            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print(f"    -> {obj_best:.6f} (NEW BEST)")
            else:
                print(f"    -> {obj_best:.6f}")

    # --- Phase 3: Fine polishing (25% time) ---
    polish_time = deadline - time.monotonic() - 8.0
    print(f"\nPhase 3: Polish ({polish_time:.0f}s)")

    if global_a is not None and polish_time > 5:
        n = len(global_a)
        S = global_a.sum()

        # 3a: Very fine mirror descent
        t_3a = polish_time * 0.40
        print(f"  3a: Fine MD ({t_3a:.0f}s)")
        a_best, obj_best = mirror_descent_v2(
            global_a, t_3a, beta0=30, beta_max=100,
            eta0=0.08, momentum=0.95, anneal_every=300,
            anneal_scale=0.02, verbose=True, use_softmax=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")
        else:
            print(f"    -> {obj_best:.6f}")

        # 3b: Peak pair reduction
        t_3b = polish_time * 0.25
        print(f"  3b: Peak pairs ({t_3b:.0f}s)")
        a_best, obj_best = peak_pair_reduction(global_a, S, t_3b, verbose=True)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")
        else:
            print(f"    -> {obj_best:.6f}")

        # 3c: Coordinate descent
        t_3c = polish_time * 0.20
        print(f"  3c: Coord descent ({t_3c:.0f}s)")
        a_best, obj_best = coordinate_polish(global_a, S, t_3c, verbose=True)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")
        else:
            print(f"    -> {obj_best:.6f}")

        # 3d: Final ultra-fine mirror descent
        t_3d = polish_time * 0.15
        print(f"  3d: Ultra-fine MD ({t_3d:.0f}s)")
        a_best, obj_best = mirror_descent_v2(
            global_a, t_3d, beta0=50, beta_max=100,
            eta0=0.05, momentum=0.96, anneal_every=500,
            anneal_scale=0.01, verbose=False, use_softmax=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")
        else:
            print(f"    -> {obj_best:.6f}")

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: local_obj={global_best:.6f}, verified={ver:.6f}")

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        print(f"Fallback: {evaluate_sequence(final_a):.6f}")

    return final_a
