"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
Uses accelerated proximal gradient with backtracking line search,
adaptive smooth-max approximation, Polyak subgradient method,
and multi-scale n exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_gradient(a, c, beta):

    Gradient of the smooth approximation (1/beta)*log(sum exp(beta*c_k)) w.r.t. a.
    Uses FFT for efficiency.

    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # grad[m] = 2 * sum_k w[k] * a[k-m] (cross-correlation)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def toppk_gradient(a, c, top_k):
    """Gradient using top-k active lags with uniform weights."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total

    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def obj_from_conv(c, n):
    """Compute objective from convolution, assuming a is normalized to sum=sqrt(2n)."""
    return c.max()


def evaluate_internal(a):
    """Compute objective value for a (assumed valid)."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def normalize_to_target(a, n):
    """Normalize a to have sum = sqrt(2n)."""
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def fista_backtrack(a_init, n, S, max_time, rng, beta_init=10.0, beta_rate=0.01,
                     beta_max=2000.0, noise_scale=0.002):
    """
    FISTA with backtracking line search for the smooth max approximation.
    Returns (best_a, best_score) where best_score is max_conv (with sum=S).
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (n * 10.0)  # initial step guess

    # Lower bound on max_conv given sum=S: S^2/(2n-1)
    lb = S * S / (2 * n - 1)

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init + beta_rate * it)

        c_y = fft_convolve_self(y)
        fy = c_y.max()
        if fy < best_score:
            best_score = fy
            best_local = y.copy()

        grad = softmax_gradient(y, c_y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step_local = step
        improved = False
        for _ in range(20):
            x_new = project_simplex(y - step_local * grad, S)
            c_new = fft_convolve_self(x_new)
            fx_new = c_new.max()
            if fx_new < fy:
                improved = True
                break
            step_local *= 0.5
        if not improved:
            x_new = project_simplex(y - step_local * grad, S)

        # Update step for next iteration
        step = max(step_local * 1.5, 1e-10)
        step = min(step, S / (gnorm + 1e-15))

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Ensure y is on the simplex approximately
        sy = y.sum()
        if sy > 0:
            y = project_simplex(y, S)
        a = x_new
        t = t_new

        # Diversification
        if it % 5000 == 0 and rng.random() < 0.1:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
            step = S / (n * 10.0)

    # Final check
    c_final = fft_convolve_self(best_local)
    if c_final.max() < best_score:
        best_score = c_final.max()
    return best_local, best_score


def polyak_subgradient(a_init, n, S, max_time, rng, top_k_init=8, top_k_max=256):
    """
    Polyak subgradient method with backtracking.
    Uses exact max (top-k approximation) with Polyak step size.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    # Lower bound: S^2 / (2n-1)
    lb = S * S / (2.0 * n - 1.0)

    # Running average for step size (Polyak with Nesterov acceleration)
    y = a.copy()
    t = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        k = min(top_k_max, top_k_init + it // 20)

        c_y = fft_convolve_self(y)
        fy = c_y.max()
        if fy < best_score:
            best_score = fy
            best_local = y.copy()

        grad = toppk_gradient(y, c_y, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Project gradient onto tangent space of simplex
        # Subtract mean (tangent to sum constraint)
        grad_t = grad - np.mean(grad)
        gt_norm = np.linalg.norm(grad_t)
        if gt_norm < 1e-15:
            grad_use = grad
            gt_norm = gnorm
        else:
            grad_use = grad_t

        # Polyak step size: (f - f*) / ||g||^2, with backtracking
        gap = fy - lb
        if gap < 1e-10:
            step = 1e-6
        else:
            step = gap / (gt_norm * gt_norm + 1e-30)
            step = min(step, S / (gt_norm + 1e-15) * 2.0)

        # Backtracking
        improved = False
        for _ in range(15):
            x_new = project_simplex(y - step * grad_use, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < fy:
                improved = True
                break
            step *= 0.5

        if not improved:
            x_new = project_simplex(y - step * grad_use, S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = project_simplex(np.maximum(y, 0.0), S)
        a = x_new
        t = t_new

        # Diversification
        if it % 3000 == 0 and rng.random() < 0.08:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0

    c_final = fft_convolve_self(best_local)
    if c_final.max() < best_score:
        best_score = c_final.max()
    return best_local, best_score


def active_set_newton(a_init, n, S, max_time, rng, k_active=16):
    """
    Active-set inspired method: identify top k constraints,
    solve approximate KKT system, take projected step.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    y = a.copy()
    t = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        k = min(512, k_active + it // 30)

        c_y = fft_convolve_self(y)
        fy = c_y.max()
        if fy < best_score:
            best_score = fy
            best_local = y.copy()

        # Get top-k with smooth weights
        top_indices = np.argsort(c_y)[::-1][:k]
        top_vals = c_y[top_indices]
        # Softmax weights (gentle)
        tv_scaled = (top_vals - top_vals[0]) * 10.0
        tv_scaled = np.clip(tv_scaled, -500, 500)
        w_top = np.exp(tv_scaled) / np.exp(tv_scaled).sum()

        w_full = np.zeros(2 * n - 1)
        for i, idx in enumerate(top_indices):
            w_full[int(idx)] = w_top[i]

        # Compute gradient
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Project to tangent space
        grad_t = grad - np.mean(grad)
        gt_norm = np.linalg.norm(grad_t)
        if gt_norm < 1e-15:
            continue

        # Step with backtracking
        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        improved = False
        for _ in range(10):
            x_new = project_simplex(y - step * grad_t / gt_norm * 2.0, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < fy:
                improved = True
                break
            step *= 0.5

        if not improved:
            x_new = project_simplex(y - step * grad_t / gt_norm * 2.0, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = project_simplex(np.maximum(y, 0.0), S)
        a = x_new
        t = t_new

    c_final = fft_convolve_self(best_local)
    if c_final.max() < best_score:
        best_score = c_final.max()
    return best_local, best_score


def fine_polish(a_init, n, S, max_time, rng):
    """
    High-precision polishing: very small steps, many active constraints.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    y = a.copy()
    t = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        # Use many active constraints for precision
        k = min(1024, 64 + it // 10)

        c_y = fft_convolve_self(y)
        fy = c_y.max()
        if fy < best_score:
            best_score = fy
            best_local = y.copy()

        grad = softmax_gradient(y, c_y, beta=50.0 + 0.5 * it)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        grad_t = grad - np.mean(grad)
        gt_norm = np.linalg.norm(grad_t)
        if gt_norm < 1e-15:
            continue

        # Small step with backtracking
        step = 1.0 / (gt_norm * (1.0 + 0.0005 * it))
        improved = False
        for _ in range(10):
            x_new = project_simplex(y - step * grad_t, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < fy - 1e-12:
                improved = True
                break
            step *= 0.5

        if not improved:
            # Accept anyway if it's not worse
            x_new = project_simplex(y - step * grad_t, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() > fy:
                x_new = y  # Don't accept worsening moves
                continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = project_simplex(np.maximum(y, 0.0), S)
        a = x_new
        t = t_new

        # Very occasional perturbation
        if it % 10000 == 0 and rng.random() < 0.05:
            idx = rng.random(n) < 0.1
            noise = rng.normal(0, 1e-4 * S / np.sqrt(n), n)
            noise[~idx] = 0
            y = project_simplex(y + noise, S)
            t = 1.0

    c_final = fft_convolve_self(best_local)
    if c_final.max() < best_score:
        best_score = c_final.max()
    return best_local, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for optimal non-negative sequence minimizing 2n*max(conv)/sum^2.
    Strategy:
    1. Start from parent if available, else use structured inits.
    2. Multi-phase optimization: FISTA -> Polyak subgradient -> Active set -> Fine polish.
    3. Multiple n values for robustness.
    4. Diversification via random restarts.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None
    global_n = 0

    def try_update(a, n):
        nonlocal global_best, global_a, global_n
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        a = normalize_to_target(a, n)
        c = fft_convolve_self(a)
        score = c.max()
        if score < global_best:
            global_best = score
            global_a = a.copy()
            global_n = n
            print("  NEW BEST: n=%d, score=%.8f" % (n, score))

    # --- Get parent sequence ---
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            parent = np.clip(given, 0.0, 1000.0)
    except Exception:
        pass

    # --- Determine n values to try ---
    if parent is not None:
        n_parent = len(parent)
        n_values = sorted(set([512, 1024, 2048, 4096, n_parent, n_parent * 2]))
        n_values = [n for n in n_values if 256 <= n <= 65536]
    else:
        n_values = [512, 1024, 2048, 4096, 8192]

    print("=== Starting optimization, n_values=%s ===" % n_values)
    print("Parent available: %s, length: %s" % (parent is not None, len(parent) if parent is not None else 0))

    # Time allocation: 70% for main optimization, 30% for polish
    main_time = (deadline - time.monotonic()) * 0.7
    per_n = main_time / len(n_values)

    # --- Phase 1: Multi-n exploration ---
    for n in n_values:
        if time.monotonic() > deadline - 120:
            break
        S = np.sqrt(2.0 * float(n))
        t0 = time.monotonic()
        alloc = min(per_n, deadline - time.monotonic() - 120)
        if alloc < 5:
            continue

        print("  n=%d, time_alloc=%.1fs" % (n, alloc))

        # Initialize
        if parent is not None and abs(n - len(parent)) < len(parent) * 0.5:
            # Interpolate or truncate parent to length n
            p = parent
            if n == len(p):
                a_init = p.copy()
            elif n < len(p):
                # Subsample
                indices = np.linspace(0, len(p) - 1, n).astype(int)
                a_init = p[indices].copy()
            else:
                # Upsample
                indices = np.linspace(0, len(p) - 1, n)
                a_init = np.interp(indices, np.arange(len(p)), p).copy()
        else:
            # Generate structured initialization
            tt = np.linspace(0, 1, n, endpoint=False)
            # Try a few and pick the best
            candidates = []
            for alpha in [1.0, 2.0, 3.0]:
                candidates.append(np.exp(-alpha * tt))
                candidates.append(np.exp(-alpha * (1.0 - tt)))
            candidates.append(tt + 0.1)
            candidates.append(1.0 - tt + 0.1)
            candidates.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.5 * tt)
            candidates.append(np.random.default_rng(0).random(n) + 0.1)

            # Pick the one with best initial objective
            best_init = None
            best_init_score = np.inf
            for cand in candidates:
                cand = np.clip(cand, 0.01, 100.0)
                cand = normalize_to_target(cand, n)
                c = fft_convolve_self(cand)
                sc = c.max()
                if sc < best_init_score:
                    best_init_score = sc
                    best_init = cand.copy()
            a_init = best_init

        a_init = normalize_to_target(a_init, n)

        # Multi-phase optimization
        # Phase 1a: FISTA (40% of allocated time)
        t1 = alloc * 0.4
        a1, s1 = fista_backtrack(a_init, n, S, t1, rng, beta_init=5.0, beta_rate=0.05, beta_max=1000.0)
        try_update(a1, n)

        # Phase 1b: Polyak subgradient (30%)
        remaining = min(alloc - t1, deadline - time.monotonic() - 120)
        if remaining > 5:
            t2 = remaining * 0.5
            a2, s2 = polyak_subgradient(a1, n, S, t2, rng, top_k_init=8, top_k_max=128)
            try_update(a2, n)
            best_here = a2 if s2 < s1 else a1
        else:
            best_here = a1

        # Phase 1c: Active set (remaining)
        remaining2 = min(alloc - t1 - (t2 if remaining > 5 else 0), deadline - time.monotonic() - 120)
        if remaining2 > 5:
            a3, s3 = active_set_newton(best_here, n, S, remaining2, rng, k_active=16)
            try_update(a3, n)

    print("=== After Phase 1 (exploration): best=%.8f, n=%d ===" % (global_best, global_n))

    # --- Phase 2: Deep polish on best ---
    if global_a is not None:
        n_best = global_n
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = deadline - time.monotonic()

        if remaining > 20:
            print("=== Phase 2: Polish (remaining %.1f sec) ===" % remaining)
            a_cur = global_a.copy()

            # Polish 1: High-beta FISTA
            seg = remaining * 0.2
            a_p1, s1 = fista_backtrack(a_cur, n_best, S_best, seg, rng,
                                       beta_init=50.0, beta_rate=0.2, beta_max=5000.0,
                                       noise_scale=0.001)
            try_update(a_p1, n_best)
            if s1 < (fft_convolve_self(a_cur).max()):
                a_cur = a_p1

            # Polish 2: Fine polish
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                seg2 = remaining2 * 0.3
                a_p2, s2 = fine_polish(a_cur, n_best, S_best, seg2, rng)
                try_update(a_p2, n_best)
                if s2 < (fft_convolve_self(a_cur).max()):
                    a_cur = a_p2

            # Polish 3: Polyak with high k
            remaining3 = deadline - time.monotonic()
            if remaining3 > 10:
                seg3 = remaining3 * 0.3
                a_p3, s3 = polyak_subgradient(a_cur, n_best, S_best, seg3, rng,
                                              top_k_init=32, top_k_max=512)
                try_update(a_p3, n_best)
                if s3 < (fft_convolve_self(a_cur).max()):
                    a_cur = a_p3

            # Polish 4: Final fine polish
            remaining4 = deadline - time.monotonic()
            if remaining4 > 5:
                a_p4, s4 = fine_polish(a_cur, n_best, S_best, remaining4, rng)
                try_update(a_p4, n_best)

    # --- Phase 3: Try a few random restarts from best ---
    if global_a is not None:
        n_best = global_n
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("=== Phase 3: Random restarts (remaining %.1f sec) ===" % remaining)
            n_restarts = min(5, int(remaining // 30))
            for r in range(n_restarts):
                if time.monotonic() > deadline - 10:
                    break
                # Perturb best
                scale = 0.05 + 0.03 * r
                noise = rng.normal(0, scale * S_best / np.sqrt(n_best), n_best)
                a_pert = global_a + noise
                a_pert = np.clip(a_pert, 0, 1000.0)
                a_pert = normalize_to_target(a_pert, n_best)

                alloc_r = min(25.0, deadline - time.monotonic() - 10)
                if alloc_r < 5:
                    break

                a_r, s_r = fista_backtrack(a_pert, n_best, S_best, alloc_r * 0.5, rng,
                                           beta_init=20.0, beta_rate=0.1, beta_max=2000.0)
                try_update(a_r, n_best)

                remaining_r = min(alloc_r * 0.5, deadline - time.monotonic() - 10)
                if remaining_r > 3:
                    a_r2, s_r2 = fine_polish(a_r, n_best, S_best, remaining_r, rng)
                    try_update(a_r2, n_best)

    # --- Final verification and return ---
    if global_a is None:
        # Fallback
        n_fb = 8192
        global_a = np.ones(n_fb) * np.sqrt(2.0 * n_fb) / n_fb
        global_n = n_fb
        global_best = fft_convolve_self(global_a).max()
        print("  Fallback: uniform sequence, score=%.6f" % global_best)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192) * 1.0
        global_n = 8192

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Final: n=%d, internal=%.8f, verified=%s ===" %
          (global_n, global_best, "%.8f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver2 = evaluate_sequence(final_list)
        print("  Fallback verified: %s" % ("%.6f" % ver2 if np.isfinite(ver2) else "inf"))

    return final_list
