import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    # Bisection on tau
    v_max = v.max()
    if v_max <= 0:
        return np.full(n, S / n)
    lo, hi = 0.0, v_max
    for _ in range(80):
        mid = (lo + hi) / 2
        s_mid = np.maximum(v - mid, 0).sum()
        if s_mid > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.maximum(v - tau, 0)

def compute_gradient(c, a, T):
    """Compute gradient of soft_max(conv(a,a), T) w.r.t. a."""
    # p[j] = exp(c[j]/T) / sum(exp(c[j]/T))
    c_shifted = c - c.max()  # numerical stability
    p = np.exp(c_shifted / T)
    p /= p.sum()

    # = 2 * conv(a, p_full)[n-1+m] ... 
    # Actually: grad[m] = 2 * sum_{j} p[j] * a[j-m] * 1[0 <= j-m < n]
    # Let k = j-m, then grad[m] = 2 * sum_k p[m+k] * a[k] * 1[0<=m+k<2n-1]

    # This is cross-correlation of a with p, or equivalently:
    # = conv(p, a_rev)[n-1+m] where a_rev = a[::-1]
    
    a_rev = a[::-1]
    fsize = 1 << (len(p) + n - 1 - 1).bit_length()
    P = fft(p, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(P * R))
    grad = 2.0 * full_conv[n-1:2*n-1]
    return grad

def smooth_optimize(a0, time_budget, T_start=10.0, T_end=0.1, n_anneal=500, lr=0.01, verbose=True):
    """Smooth optimization with soft-max and temperature annealing."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0)
    a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()

    elapsed = 0

            break
        
        # Anneal temperature
        progress = it / max(1, n_anneal - 1)
        T = T_start * (T_end / T_start) ** progress
        
        # Multiple iterations at each temperature
        n_iters = max(1, int(n_anneal / 10))
        
        for sub in range(n_iters):
            if time.monotonic() - start > time_budget:
                break
            
            grad = compute_gradient(c, a, T)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad = grad / gnorm

            # Check improvement
            c_new = fft_convolve_self(a_new)
            obj_new = 2.0 * n * c_new.max() / (S * S)
            obj_cur = 2.0 * n * c.max() / (S * S)
            
            if obj_new < obj_cur:

            else:

                lr = max(lr * 0.5, 1e-6)

            lr = min(lr * 1.1, 0.1)
            
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
        
        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"    T={T:.4f}, obj={best_obj:.6f}, t={elapsed:.1f}s")

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using smooth optimization with temperature annealing,
Frank-Wolfe conditional gradient, and peak-shaving polish.
Key innovations: differentiable soft-max objective, additive+FW hybrid updates,
adaptive temperature annealing, multi-phase optimization with structured restarts.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_crossconv(a, b):
    """Full linear convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:out_len]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S} using bisection."""
    n = len(v)
    if n == 0:
        return v
    vmax = v.max()
    if vmax <= 0:
        return np.full(n, S / n)
    if v.sum() <= S and v.min() >= 0:
        return v + (S - v.sum()) / n
    lo, hi = 0.0, vmax
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_mid = np.maximum(v - mid, 0.0).sum()
        if s_mid > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.maximum(v - tau, 0.0)


def softmax_gradient(c, a, T):
    """
    Compute gradient of soft_max(conv(a,a), T) w.r.t. a.
    grad[m] = 2 * sum_k a[k] * p[m+k] where p = softmax(c/T)
    """
    n = len(a)
    c_shifted = c - c.max()
    p = np.exp(c_shifted / T)
    p /= p.sum()
    
    # grad[m] = 2 * sum_{k=0}^{n-1} a[k] * p[m+k]
    # = 2 * conv(a, p_rev)[2n-2-m]
    p_rev = p[::-1]
    fsize = 1 << (n + 2 * n - 1 - 1).bit_length()
    A = fft(a, fsize)
    P = fft(p_rev, fsize)
    C = np.real(ifft(A * P))
    # We want C[2n-2-m] for m=0..n-1, i.e., C[2n-2], C[2n-3], ..., C[n-1]
    grad = 2.0 * C[2 * n - 2 : n - 2 : -1]
    return grad, p


def peak_gradient(c, a, top_k=5, T=0.0):
    """
    Compute a focused gradient targeting only the top-k peaks.
    This is cheaper and more targeted than the full soft-max gradient.
    """
    n = len(a)
    c_shifted = c - c.max()
    if T > 0:
        p = np.exp(c_shifted / T)
        p /= p.sum()
        # Focus on top-k by zeroing out the rest
        threshold_idx = np.argpartition(c, -top_k)[top_k - 1]
        p[c < c[threshold_idx]] = 0.0
        p /= p.sum() + 1e-30
    else:
        # Hard top-k
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        p = np.zeros_like(c)
        p[top_idx] = 1.0 / top_k
    
    p_rev = p[::-1]
    fsize = 1 << (n + 2 * n - 1 - 1).bit_length()
    A = fft(a, fsize)
    P = fft(p_rev, fsize)
    C = np.real(ifft(A * P))
    grad = 2.0 * C[2 * n - 2 : n - 2 : -1]
    return grad


def obj_val(a, c=None):
    """Compute objective: 2n * max(c) / S^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    if c is None:
        c = fft_conv_self(a)
    return 2.0 * n * c.max() / (S * S)


def smooth_optimize(a0, time_budget, T_start=5.0, T_end=0.05, lr_init=0.5,
                    verbose=True, label="smooth"):
    """
    Smooth optimization using soft-max gradient + simplex projection.
    Temperature anneals from T_start to T_end.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())
    
    c = fft_conv_self(a)
    best_obj = obj_val(a, c)
    best_a = a.copy()
    lr = lr_init
    start = time.monotonic()
    iter_no = 0
    best_iter = 0
    
    while time.monotonic() - start < time_budget:
        elapsed_frac = (time.monotonic() - start) / time_budget
        # Log-linear temperature anneal
        T = T_start * (T_end / T_start) ** elapsed_frac
        
        grad, p = softmax_gradient(c, a, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm * S  # scale gradient by S for meaningful step
        
        # Try the step
        a_trial = project_simplex(a - lr * grad, S)
        c_trial = fft_conv_self(a_trial)
        obj_trial = obj_val(a_trial, c_trial)
        obj_cur = obj_val(a, c)
        
        if obj_trial < obj_cur:
            a = a_trial
            c = c_trial
            lr = min(lr * 1.05, 2.0 * S / np.sqrt(n))
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a = a.copy()
                best_iter = iter_no
        else:
            lr *= 0.7
            if lr < 1e-8 * S:
                # Restart with perturbation
                noise = np.random.randn(n) * 0.01 * S / np.sqrt(n)
                a = project_simplex(a + noise, S)
                c = fft_conv_self(a)
                lr = lr_init * 0.5
                best_iter = iter_no
        
        iter_no += 1
        if verbose and iter_no % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"  [{label}] iter={iter_no}, T={T:.4f}, obj={obj_val(a,c):.6f}, "
                  f"best={best_obj:.6f}, lr={lr:.4f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def frank_wolfe_optimize(a0, time_budget, T_start=3.0, T_end=0.05, verbose=True, label="fw"):
    """
    Frank-Wolfe (conditional gradient) method with soft-max objective.
    Very cheap per iteration: just find argmin of gradient and interpolate.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())
    
    c = fft_conv_self(a)
    best_obj = obj_val(a, c)
    best_a = a.copy()
    start = time.monotonic()
    iter_no = 0
    best_iter = 0
    
    while time.monotonic() - start < time_budget:
        elapsed_frac = (time.monotonic() - start) / time_budget
        T = T_start * (T_end / T_start) ** elapsed_frac
        
        grad, p = softmax_gradient(c, a, T)
        
        # FW step: find the simplex vertex that most decreases the objective
        min_idx = np.argmin(grad)
        v = np.zeros(n)
        v[min_idx] = S
        
        # Step size: use 2/(t+2) schedule or backtracking
        t = 2.0 / (iter_no + 2)
        
        # Also try a greedy step size (0.5 * t) and the full FW step
        best_trial = None
        best_trial_obj = obj_val(a, c)
        for gamma in [t, 0.5 * t, 2.0 * t / (iter_no + 2), min(1.0, 0.1)]:
            a_trial = (1.0 - gamma) * a + gamma * v
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial * (S / s)
            c_trial = fft_conv_self(a_trial)
            obj_trial = obj_val(a_trial, c_trial)
            if obj_trial < best_trial_obj:
                best_trial_obj = obj_trial
                best_trial = a_trial
        
        if best_trial is not None:
            a = best_trial
            c = fft_conv_self(a)
            if obj_val(a, c) < best_obj:
                best_obj = obj_val(a, c)
                best_a = a.copy()
                best_iter = iter_no
        else:
            # No improvement; try perturbation
            if iter_no - best_iter > 500:
                idx = np.random.randint(n)
                a[idx] += 0.01 * S / np.sqrt(n)
                a = project_simplex(a, S)
                c = fft_conv_self(a)
                best_iter = iter_no
        
        iter_no += 1
        if verbose and iter_no % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"  [{label}] iter={iter_no}, T={T:.4f}, obj={obj_val(a,c):.6f}, "
                  f"best={best_obj:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_obj

    S = a.sum()
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())
    
    c = fft_conv_self(a)
    best_obj = obj_val(a, c)
    best_a = a.copy()
    start = time.monotonic()
    lr = 0.1
    iter_no = 0
    best_iter = 0
    v = np.zeros(n)  # momentum
    
    while time.monotonic() - start < time_budget:
        # Find the worst peak
        k_max = int(np.argmax(c))
        
        # Gradient of c[k_max] w.r.t. a[m] = 2 * a[k_max - m] (for valid m)
        grad = np.zeros(n)
        for m in range(n):
            j = k_max - m
            if 0 <= j < n:
                grad[m] = 2.0 * a[j]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm * S
        
        # Momentum
        v = 0.85 * v + 0.15 * grad
        
        # Update
        a_trial = project_simplex(a - lr * v, S)
        c_trial = fft_conv_self(a_trial)
        obj_trial = obj_val(a_trial, c_trial)
        obj_cur = obj_val(a, c)
        
        if obj_trial < obj_cur:
            a = a_trial
            c = c_trial
            lr = min(lr * 1.03, S / (np.sqrt(n) * 0.5))
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a = a.copy()
                best_iter = iter_no
        else:
            lr *= 0.7
            v *= 0.5
            if iter_no - best_iter > 300:
                # Diversify: target a different peak
                top_peaks = np.argsort(c)[::-1][:10]
                k_max = int(top_peaks[np.random.randint(min(10, len(top_peaks)) )])
                noise = np.random.randn(n) * 0.005 * S / np.sqrt(n)
                a = project_simplex(a + noise, S)
                c = fft_conv_self(a)
                best_iter = iter_no
                lr = 0.1
        
        iter_no += 1
        if verbose and iter_no % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"  [{label}] iter={iter_no}, obj={obj_val(a,c):.6f}, "
                  f"best={best_obj:.6f}, lr={lr:.4f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def multi_peak_optimize(a0, time_budget, n_peaks=3, T=0.5, verbose=True, label="multi"):
    """
    Optimize targeting the top n_peaks simultaneously with a weighted gradient.
    Weights are proportional to how much each peak exceeds the median.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())
    
    c = fft_conv_self(a)
    best_obj = obj_val(a, c)
    best_a = a.copy()
    start = time.monotonic()
    lr = 0.3
    iter_no = 0
    best_iter = 0
    v = np.zeros(n)
    
    while time.monotonic() - start < time_budget:
        # Weight peaks by excess over median
        c_med = np.median(c)
        excess = np.maximum(c - c_med, 0)
        # Focus on top peaks
        top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        w = np.zeros_like(c)
        for k in top_idx:
            w[k] = excess[k] if excess[k] > 0 else 1e-6
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros_like(c)
            w[top_idx] = 1.0 / n_peaks
        else:
            w /= w_sum
        
        # Compute weighted gradient
        w_rev = w[::-1]
        fsize = 1 << (n + 2 * n - 1 - 1).bit_length()
        A = fft(a, fsize)
        W = fft(w_rev, fsize)
        C = np.real(ifft(A * W))
        grad = 2.0 * C[2 * n - 2 : n - 2 : -1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm * S
        
        # Momentum
        v = 0.9 * v + 0.1 * grad
        vnorm = np.linalg.norm(v)
        if vnorm > 0:
            v = v / vnorm * S
        
        a_trial = project_simplex(a - lr * v, S)
        c_trial = fft_conv_self(a_trial)
        obj_trial = obj_val(a_trial, c_trial)
        obj_cur = obj_val(a, c)
        
        if obj_trial < obj_cur:
            a = a_trial
            c = c_trial
            lr = min(lr * 1.02, S)
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a = a.copy()
                best_iter = iter_no
        else:
            lr *= 0.8
            v *= 0.7
            if iter_no - best_iter > 500:
                # Reset with perturbation
                noise = np.random.randn(n) * 0.02 * S / np.sqrt(n)
                a = project_simplex(a + noise, S)
                c = fft_conv_self(a)
                v *= 0.1
                best_iter = iter_no
                lr = 0.3
        
        iter_no += 1
        if verbose and iter_no % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"  [{label}] iter={iter_no}, obj={obj_val(a,c):.6f}, "
                  f"best={best_obj:.6f}, lr={lr:.3f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def structured_perturb(a, S, rng, mode="gaussian"):
    """Apply a structured perturbation to escape local minima."""
    n = len(a)
    if mode == "gaussian":
        noise = rng.standard_normal(n) * 0.05 * S / np.sqrt(n)
    elif mode == "smooth_lowfreq":
        # Low-frequency perturbation (preserves smooth structure)
        freqs = 2 * np.pi * np.arange(n) / n
        coeffs = rng.standard_normal(10) * 0.1 * S / n
        noise = np.zeros(n)
        for i in range(min(10, n)):
            noise += coeffs[i % len(coeffs)] * np.cos(freqs * (i + 1) + rng.uniform(0, 2 * np.pi))
    elif mode == "block":
        # Block perturbation
        block_size = max(1, n // 32)
        block_noise = rng.standard_normal(n // block_size + 1) * 0.02 * S / np.sqrt(n)
        noise = np.repeat(block_noise, block_size)[:n]
    else:
        noise = rng.standard_normal(n) * 0.03 * S / np.sqrt(n)
    
    a_new = a + noise
    return project_simplex(a_new, S)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase approach: explore -> smooth optimize -> FW optimize -> polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    global_best = np.inf
    global_a = None
    
    # --- Load given sequence ---
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw.copy()
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print(f"Given sequence: n={n_given}, obj={ver:.6f}")
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print(f"Could not load given sequence: {e}")
    
    # --- Determine working length ---
    n_work = 4096
    S_work = np.sqrt(2.0 * n_work)
    
    # --- Phase 1: Quick exploration with different initializations (15% time) ---
    explore_time = 0.15 * (deadline - time.monotonic())
    print(f"\nPhase 1: Exploration ({explore_time:.0f}s)")
    
    candidates = []
    
    if given_a is not None:
        # Resample to n_work if different length
        if len(given_a) == n_work:
            candidates.append(('given', given_a.copy()))
        else:
            # Interpolate to n_work
            from numpy import interp
            x_old = np.linspace(0, 1, len(given_a))
            x_new = np.linspace(0, 1, n_work)
            a_interp = interp(x_new, x_old, given_a)
            S_i = a_interp.sum()
            a_interp = np.maximum(a_interp, 0) * (S_work / a_interp.sum())
            candidates.append(('given_interp', a_interp))
    
    # Cosine-modulated
    i = np.arange(n_work)
    a_cos = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * i / n_work) + 0.2 * np.cos(2 * np.pi * 7 * i / n_work)
    a_cos = np.maximum(a_cos, 0.01)
    a_cos *= S_work / a_cos.sum()
    candidates.append(('cosine', a_cos))
    
    # Raised cosine window
    a_rise = 0.5 * (1 + np.cos(2 * np.pi * i / n_work))
    a_rise = np.maximum(a_rise, 0.01)
    a_rise *= S_work / a_rise.sum()
    candidates.append(('raised_cos', a_rise))
    
    # Sum of Gaussians
    a_gauss = np.zeros(n_work)
    for center_frac, width_frac, amp in [(0.3, 0.1, 1.0), (0.5, 0.08, 1.2), (0.7, 0.1, 1.0)]:
        a_gauss += amp * np.exp(-0.5 * ((i - n_work * center_frac) / (n_work * width_frac)) ** 2)
    a_gauss = np.maximum(a_gauss, 0.01)
    a_gauss *= S_work / a_gauss.sum()
    candidates.append(('gauss', a_gauss))
    
    # Random
    a_rand = rng.random(n_work)
    a_rand = np.maximum(a_rand, 0.01)
    a_rand *= S_work / a_rand.sum()
    candidates.append(('rand', a_rand))
    
    per_cand_time = explore_time / len(candidates)
    for name, a_cand in candidates:
        if time.monotonic() > deadline - 60:
            break
        t0 = time.monotonic()
        a_best, obj_best = smooth_optimize(
            a_cand, per_cand_time,
            T_start=3.0, T_end=0.5, lr_init=0.3,
            verbose=False, label=f"explore_{name}"
        )
        elapsed = time.monotonic() - t0
        print(f"  {name}: obj={obj_best:.6f} ({elapsed:.1f}s)")
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
    
    # --- Phase 2: Main smooth optimization (40% time) ---
    main_time = 0.40 * (deadline - time.monotonic())
    print(f"\nPhase 2: Main smooth optimization ({main_time:.0f}s)")
    
    if global_a is not None and main_time > 30:
        # 2a: Wide temperature range (exploratory)
        t2a = main_time * 0.4
        print(f"  2a: Wide anneal ({t2a:.0f}s)")
        a_best, obj_best = smooth_optimize(
            global_a, t2a,
            T_start=10.0, T_end=0.5, lr_init=0.5,
            verbose=True, label="wide"
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")
        
        # 2b: Narrow temperature range (exploitative)
        t2b = main_time * 0.6
        print(f"  2b: Narrow anneal ({t2b:.0f}s)")
        a_best, obj_best = smooth_optimize(
            global_a, t2b,
            T_start=2.0, T_end=0.05, lr_init=0.3,
            verbose=True, label="narrow"
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")
    
    # --- Phase 3: Frank-Wolfe optimization (20% time) ---
    fw_time = 0.20 * (deadline - time.monotonic())
    print(f"\nPhase 3: Frank-Wolfe ({fw_time:.0f}s)")
    
    if global_a is not None and fw_time > 20:
        a_best, obj_best = frank_wolfe_optimize(
            global_a, fw_time,
            T_start=3.0, T_end=0.05,
            verbose=True, label="fw"
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  -> {obj_best:.6f}")
    
    # --- Phase 4: Peak shaving polish (15% time) ---
    polish_time = 0.15 * (deadline - time.monotonic())
    print(f"\nPhase 4: Peak shaving polish ({polish_time:.0f}s)")
    
    if global_a is not None and polish_time > 10:
        # 4a: Single peak focus
        t4a = polish_time * 0.4
        print(f"  4a: Single peak ({t4a:.0f}s)")
        a_best, obj_best = peak_shave_polish(global_a, t4a, verbose=True, label="peak1")
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")
        
        # 4b: Multi-peak focus
        t4b = polish_time * 0.6
        print(f"  4b: Multi-peak ({t4b:.0f}s)")
        a_best, obj_best = multi_peak_optimize(global_a, t4b, n_peaks=3, verbose=True, label="multi")
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")
    
    # --- Phase 5: Fine-tuning with restarts (10% time) ---
    fine_time = deadline - time.monotonic() - 10.0
    print(f"\nPhase 5: Fine-tuning with restarts ({fine_time:.0f}s)")
    
    if global_a is not None and fine_time > 10:
        n_restarts = max(2, int(fine_time / 60))
        per_restart = fine_time / n_restarts
        
        for r in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            # Perturb the current best
            mode = ["gaussian", "smooth_lowfreq", "block"][r % 3]
            a_pert = structured_perturb(global_a, global_a.sum(), rng, mode=mode)
            
            # Quick smooth optimization from perturbed point
            a_best, obj_best = smooth_optimize(
                a_pert, per_restart,
                T_start=1.0, T_end=0.02, lr_init=0.2,
                verbose=False, label=f"restart{r}"
            )
            print(f"  restart {r} ({mode}): obj={obj_best:.6f}")
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print(f"    -> NEW BEST: {obj_best:.6f}")
    
    # --- Final verification ---
    print(f"\nBest found (local): {global_best:.6f}")
    
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"FINAL: local_obj={global_best:.6f}, verified={ver:.6f}")
    
    if not np.isfinite(ver):
        # Fallback
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.6f}")
    
    return final_a
