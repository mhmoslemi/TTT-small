        active_idx = np.arange(len(c))
    else:
        active_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    
    K = len(active_idx)
    
    # Build constraint matrix
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for i in range(K):
        k = active_idx[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad_c[k][m] = 2*a[k-m] for m in [lo, hi]
        A_ub[i, lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
        A_ub[i, -1] = -1.0
        b_ub[i] = c[k]
    
    # Equality: sum(a) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        a_lp = result.x[:n].copy()
        a_lp = np.maximum(a_lp, 0.0)  # clip tiny negatives
        t_lp = result.x[-1]
        return a_lp, t_lp
    else:
        return a.copy(), None


def run_kelley(a0, S, time_budget, top_k=500, verbose=True):
    """Run Kelley cutting-plane method for a time budget."""
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    cur_obj = obj_value(a, c)
    best_obj = cur_obj
    best_a = a.copy()
    start = time.monotonic()
    iters = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Solve LP
        a_lp, t_lp = kelley_lp(a, c, top_k=top_k, S=S)
        
        if t_lp is None:
            # LP failed, try with fewer constraints
            a_lp, t_lp = kelley_lp(a, c, top_k=min(100, top_k), S=S)
            if t_lp is None:
                break
        
        # Line search
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_try = (1.0 - alpha) * a + alpha * a_lp
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try * (S / s)
            
            c_try = fft_convolve_self(a_try)
            obj_try = obj_value(a_try, c_try)
            
            if obj_try < cur_obj * (1.0 - 1e-12):
                a = a_try
                c = c_try
                cur_obj = obj_try
                improved = True
                break
        
        if improved:
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                # Perturb to escape
                noise = np.random.randn(len(a)) * 0.01 * S / np.sqrt(len(a))
                a = a + noise
                a = np.maximum(a, 0.0)
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                cur_obj = obj_value(a, c)
                no_improve = 0
                top_k = max(50, top_k // 2)  # reduce constraints
        
        iters += 1
        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - start
            print(f"    kelley iter={iters}, obj={cur_obj:.8f}, best={best_obj:.8f}, "
                  f"t_lp={t_lp:.8f}, top_k={top_k}, t={elapsed:.1f}s")
    
    return best_a, best_obj

Hmm, wait. I need to think about this more carefully. The Kelley LP method solves a relaxation of the original problem. The LP solution `a_lp` minimizes `t` subject to the linearized constraints. The true objective at `a_lp` is `2n * max_c(at a_lp)` which is >= `t_lp` (since the linearization underestimates the true function outside the current point).

So the LP gives a lower bound. If `t_lp` is close to the current objective, we're near optimal. If `t_lp` is much smaller, there's room for improvement.

The key question is: will the LP solution actually improve the true objective? In general, for a non-convex problem, the Kelley method might not converge to the global optimum, but it often converges to a local optimum that's better than the starting point.

Let me also think about whether there's a better way to use the LP. Instead of just one step, I could:
1. Solve the LP.
2. Move to the LP solution.
3. Recompute the true objective.
4. Add the new linearization to the constraint set.
5. Repeat.

This is the standard Kelley cutting plane method with a growing constraint set. Let me implement this properly:

def run_kelley_v2(a0, S, time_budget, verbose=True):
    """Kelley cutting plane with growing constraint set."""
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())
    n = len(a)
    
    c = fft_convolve_self(a)
    cur_obj = obj_value(a, c)
    best_obj = cur_obj
    best_a = a.copy()
    start = time.monotonic()

    # Each cut from peak k at point a_k: grad_c[k](a_k)^T a <= t + c_k[k]
    cuts_g = []  # list of gradient vectors
    cuts_h = []  # list of RHS values

    no_improve = 0

        active_idx = np.where(c >= active_threshold)[0]

            top_local = np.argsort(c_at_active, kind='stable')[::-1][:500]

            b_old = np.array(cuts_h)
