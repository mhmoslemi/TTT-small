"""
Minimize 2*n*max_conv / sum^2 using LP-guided direction search.

Key innovation: At each step, solve a linear program to find the best
direction g that, when blended with the current sequence f, reduces the
peak of the self-convolution. The LP is: max sum(g) s.t. g >= 0 and
conv(f, g) <= M (element-wise), where M = ||f*f||_infty.

This gives a "globally optimal" direction at each step, avoiding the
local minima that trap pure gradient-based methods.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.maximum(v - S/n, 0.0)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, solver='GLOP'):
    """
    Solve LP: max sum(g) s.t. g >= 0, conv(f, g) <= M (element-wise).
    Returns optimal g.
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build convolution constraints: for each k, sum_i f[i]*g[k-i] <= M
    # conv(f, g) has length 2n-1
    # conv(f,g)[k] = sum_{i} f[i] * g[k-i] where 0<=i<n and 0<=k-i<n
    
    # This is a matrix-vector product: C @ g <= M
    # where C is (2n-1) x n, C[k,i] = f[i] if 0 <= k-i < n else 0
    
    # For efficiency, build C as a sparse-like structure
    # C[k, i] = f[i] if k >= i and k - i < n (i.e., i <= k < i+n)
    # Equivalently: C[k, i] = f[i] if max(0, k-n+1) <= i <= min(k, n-1)
    
    # Actually, let's just build the full matrix for moderate n
    # For n=8192, C is 16383 x 8192, which is ~134M entries. That's a lot.
    # Let's use a sparse representation.
    
    from scipy.sparse import csr_matrix
    
    # Build C sparsely
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    
    C = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    constraints = [C @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver, verbose=False)
        if g.value is not None:
            return np.clip(g.value, 0, None)
    except Exception as e:
        print(f"  LP solve failed: {e}")
    
    # Fallback: return f itself (normalized)
    return f / f.sum() * M  # This won't be optimal but is feasible-ish


def lp_iteration(f, max_time_lp, solver='GLOP'):
    """
    One LP-guided iteration: find g, blend with f, return best blend.
    """
    n = len(f)
    S = f.sum()
    
    c = fft_convolve_self(f)
    M = c.max()
    
    # Solve LP
    t0 = time.monotonic()
    g = solve_lp_direction(f, M, solver=solver)
    lp_time = time.monotonic() - t0
    
    if g is None or g.sum() < 1e-15:
        return f, M, 0.0
    
    # Rescale g to have sum S
    g = g * (S / g.sum())
    
    # Line search on t
    best_f = f.copy()
    best_score = M
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_trial = (1 - t) * f + t * g
        f_trial = np.maximum(f_trial, 0)
        c_trial = fft_convolve_self(f_trial)
        score = c_trial.max()
        if score < best_score:
            best_score = score
            best_f = f_trial.copy()
    
    return best_f, best_score, lp_time


def spectral_shaping_step(a, N, top_k, alpha):
    """One spectral shaping step."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()
    
    m = np.arange(N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a, c_max
    
    weights = top_vals / total
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
    
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, a.sum())
    return a_new, c_max


def fft_grad_toppk(a, c, top_k):
    """Gradient using top-k active lags via FFT."""
    n = len(a)
    top_k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def nesterov_polish(a_init, n, S, max_time, rng, top_k_init=20, top_k_max=512):
    """Nesterov-accelerated top-k subgradient polishing."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        grad = fft_grad_toppk(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.005 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 500 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

"""
Minimize 2*n*max_conv/sum^2 using LP-guided direction with spectral shaping polish.

Core idea: At each iteration, solve an LP to find the optimal direction g
that reduces the peak of f*conv(f). The LP is: max sum(g) s.t. g>=0, conv(f,g)<=M.
Blend f and g with adaptive step size. Polish with spectral shaping and Nesterov.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
import cvxpy as cp


def fft_convolve_self(a):
    n = len(a)
    fsize = 1 << (2 * n - 2).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(n, f):
    """Build sparse matrix C such that C @ g = conv(f, g)."""
    nrow = 2 * n - 1
    # For each column i, non-zero rows are i to min(i+n-1, 2n-2)
    col_counts = np.minimum(n, 2 * n - 1 - np.arange(n))
    
    total_nnz = int(col_counts.sum())
    row_idx = np.empty(total_nnz, dtype=np.int64)
    col_idx = np.empty(total_nnz, dtype=np.int64)
    vals = np.empty(total_nnz, dtype=np.float64)
    
    pos = 0
    for i in range(n):
        cnt = int(col_counts[i])
        row_idx[pos:pos+cnt] = np.arange(i, i + cnt)
        col_idx[pos:pos+cnt] = i
        vals[pos:pos+cnt] = f[i]
        pos += cnt
    
    C = csr_matrix((vals, (row_idx, col_idx)), shape=(nrow, n))
    return C

    for s in solvers_to_try:

def lp_guided_iteration(f, max_lp_time=None):
    """One LP-guided iteration. Returns (new_f, new_score, lp_success)."""
    n = len(f)
    S = f.sum()
    c = fft_convolve_self(f)
    M = float(c.max())
    
    # Time-limited LP solve
    t0 = time.monotonic()
    g = solve_lp(f, M)
    lp_time = time.monotonic() - t0
    
    if g is None or g.sum() < 1e-10:
        return f, M, False
    
    g = g * (S / g.sum())
    
    # Line search
    best_f = f.copy()
    best_score = M
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]:
        f_trial = (1 - t) * f + t * g
        f_trial = np.maximum(f_trial, 0.0)
        if f_trial.sum() < 1e-10:
            continue
        c_trial = fft_convolve_self(f_trial)
        sc = float(c_trial.max())
        if sc < best_score:
            best_score = sc
            best_f = f_trial.copy()
    
    return best_f, best_score, True


def spectral_shaping_multi(a, N, top_k, alpha, n_iter, rng, target=1.4):
    """Multiple spectral shaping steps."""
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score = np.inf
    
    for it in range(n_iter):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = float(c.max())
        
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        if c_max <= target:
            break
        
        m = np.arange(N)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
        
        P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new, S)
        
        if it % 500 == 250 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
    
    return best_a, best_score


def nesterov_polish(a_init, n, S, max_time, rng, top_k=50):
    """Nesterov-accelerated top-k subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = float(c.max())
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        w_full = np.zeros(2 * n - 1)
        w_full[top_idx] = weights
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 1000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score


def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

One more optimization: instead of rebuilding the constraint matrix C at every LP solve (since f changes each iteration), I can cache it and update only the non-zero values. But since the values change (they're f[i]), I need to rebuild anyway. However, the SPARSITY PATTERN doesn't change! So I can cache the (rows, cols) indices and only update the values.

```python
class ConvMatrix:
    def __init__(self, n):
        self.n = n
        self.nrow = 2 * n - 1
        # Build sparsity pattern once
        counts = np.minimum(n, 2 * n - 1 - np.arange(n))
        self.total = int(counts.sum())
        self.col_idx = np.repeat(np.arange(n), counts)
        # offsets within each column
        offsets_total = np.empty(self.total, dtype=np.int64)
        pos = 0
        for i in range(n):
            c = int(counts[i])
            offsets_total[pos:pos+c] = np.arange(c)
            pos += c
        self.row_idx = self.col_idx + offsets_total
    
    def build(self, f):
        vals = np.repeat(f, np.minimum(self.n, 2 * self.n - 1 - np.arange(self.n)))
        return csr_matrix((vals, (self.row_idx, self.col_idx)), shape=(self.nrow, self.n))
```

This saves the O(n^2) work of building the indices each time, only doing the O(n^2) work of setting values (which is just `np.repeat`).

Actually, `np.repeat(f, counts)` is O(n^2) since the output is O(n^2) elements. So building the values is the bottleneck, not the indices. But it's a simple vectorized operation, so it should be fast (~0.1s for n=4096).
