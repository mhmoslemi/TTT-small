Key insight: By Cauchy-Schwarz, for any non-negative sequence a of length n:
  max(conv(a,a)) >= conv(a,a)[0] = sum(a_i^2) >= (sum(a_i))^2 / n
Therefore: 2*n*max_b/(sum_a)^2 >= 2, with equality iff a is constant.

The optimal solution is a uniform (constant) sequence, achieving exactly 2.0.
We include a verification step and a fallback optimizer just in case.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def run_fista(a_init, n, S, max_time, rng, beta_init=50.0, beta_max=500.0):
    """FISTA for polishing, using soft-max approximation of max."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total
        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (beta * 2.0)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    return best_local, best_score

        # Verify with internal computation

        max_b = c.max()
        sum_a = a.sum()
        score = 2.0 * n * max_b / (sum_a ** 2)
        
        if score < best_score:
            best_score = score
            best_seq = a.copy()
    
    print("Uniform sequence internal score: %.10f (n=%d)" % (best_score, len(best_seq)))
    
    # Verify with actual evaluate_sequence
    final_list = [float(x) for x in best_seq]
    ver_score = evaluate_sequence(final_list)
    print("evaluate_sequence(uniform n=%d): %.10f" % (len(best_seq), ver_score))
    
    if np.isfinite(ver_score) and ver_score <= 2.0 + 1e-9:
        print("Achieved theoretical minimum of 2.0!")
        return final_list
    
    # Fallback: if something unexpected happened, run optimization
    print("Unexpected result, running optimization fallback...")
    
    # Try to use parent
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    if parent is not None:
        n = len(parent)
        S = np.sqrt(2.0 * n)
        a_init = parent * (S / parent.sum())
        remaining = deadline - time.monotonic()
        if remaining > 10:
            a_opt, s_opt = run_fista(a_init, n, S, remaining * 0.8, rng)
            a_opt = np.clip(a_opt, 0, 1000)
            # Scale doesn't matter for the ratio, but let's normalize
            s = a_opt.sum()
            if s > 0:
                a_opt = a_opt * (S / s)
            final_list_opt = [float(x) for x in a_opt]
            ver_opt = evaluate_sequence(final_list_opt)
            if np.isfinite(ver_opt) and ver_opt < ver_score:
                print("Optimized: %.10f" % ver_opt)
                return final_list_opt
    
    # Final fallback: just return uniform
    print("Returning uniform sequence as final answer.")
    final_list = [float(x) for x in best_seq]
    ver = evaluate_sequence(final_list)
    print("Final verification: %.10f" % ver)
    return final_list
