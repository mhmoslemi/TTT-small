Then we take a step: a_new = a + alpha * delta (for small alpha), project onto {a_i >= 0, sum a_i = S}, and check if the max decreased.

This is a Newton-like step for the active set. It should converge quadratically when the active set is correct.

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS-B quasi-Newton optimization
on smoothed max objective, combined with active-set lag equalization and
anti-palindromic structured initializations.

Key differences from previous approach:
- L-BFGS-B (quasi-Newton) instead of FISTA (first-order) for faster convergence
- Active-set equalization: directly solve linear system to equalize top lags
- Multi-scale beta annealing with restarts
- Anti-palindromic + structured initializations
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

# Precompute FFT size
def get_fft_size(n):
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    fsize = get_fft_size(n)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:2*n-1]

def lse_and_grad(a, beta):
    """Compute LSE(c; beta) and its gradient w.r.t. a."""
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta
    
    # Gradient of LSE w.r.t. a
    fsize = get_fft_size(len(a))
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:len(a)]
    
    return lse_val, c_max, w, grad

def objective_and_grad(a, beta):
    """Objective g_beta(a) = 2n * LSE / S^2 and its gradient."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return 1e10, np.zeros(n)
    
    lse_val, c_max, w, grad_lse = lse_and_grad(a, beta)
    
    # g = 2n * lse_val / S^2
    g = 2.0 * n * lse_val / (S * S)
    
    # dg/da = 2n/S^3 * (S * grad_lse - 2 * lse_val * 1)
    grad = (2.0 * n / (S**3)) * (S * grad_lse - 2.0 * lse_val)
    
    return g, grad

def lbfgsb_optimize(a_init, n, beta, max_time, tol=1e-10):
    """Run L-BFGS-B optimization on the smoothed objective."""
    a_init = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    if a_init.sum() < 1e-15:
        a_init = np.ones(n) / n
    
    S_init = a_init.sum()
    
    def func_and_grad(a):
        # Ensure non-negativity (L-BFGS-B should handle this via bounds,
        # but clip as safety)
        a = np.maximum(a, 0)
        val, grad = objective_and_grad(a, beta)
        return val, grad
    
    bounds = [(0, 1000)] * n
    start = time.monotonic()
    
    best_a = a_init.copy()
    best_val = func_and_grad(a_init)[0]
    
    # Use scipy L-BFGS-B
    # We'll run in chunks to check time
    it_count = [0]
    
    def callback(xk):
        it_count[0] += 1
        if time.monotonic() - start > max_time - 5:
            return True  # Stop
        if it_count[0] % 100 == 0:
            val, _ = objective_and_grad(xk, beta)
            print("  L-BFGS iter %d: obj=%.8f" % (it_count[0], val))
        return False
    
    result = minimize(
        func_and_grad,
        a_init,
        jac=True,
        method='L-BFGS-B',
        bounds=bounds,
        options={
            'maxiter': 50000,
            'maxfun': 50000,
            'ftol': tol,
            'gtol': 1e-8,
            'maxls': 50,
            'callback': callback,
        }
    )
    
    if result.x is not None and result.fun < best_val:
        best_a = np.clip(result.x, 0, 1000)
        best_val = result.fun
    
    # Also check intermediate best
    val_final, _ = objective_and_grad(best_a, beta)
    if val_final < best_val:
        best_val = val_final
    
    return best_a, best_val

def active_set_equalize(a, n, S, top_K, max_iter=1000, max_time=30):
    """Iteratively equalize the top K convolution lags."""
    a = np.clip(a, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_a = a.copy()
    best_max = fft_convolve_self(a).max()
    
    for it in range(max_iter):
        if time.monotonic() - start > max_time:
            break
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
        
        # Find top K lags
        top_idx = np.argsort(c)[::-1][:top_K]
        top_vals = c[top_idx]
        
        # Check if already well-equalized
        if top_vals.max() - top_vals.min() < 1e-10 * cur_max:
            # Try to reduce all of them
            pass
        
        # Build the linear system
        # For each top lag k, the gradient of c_k w.r.t. a is:
        # dc_k/da_j = a_{k-j} + a_{j-k} (for valid indices)
        # In FFT terms: 2 * ifft(fft(e_k) * conj(fft(a)))[:n] where e_k is one-hot at position k
        
        # More efficiently: the gradient vector for lag k is 2 * (a shifted)
        # c_k = sum_i a_i * a_{k-i}, so dc_k/da_j = a_{k-j} + a_{j-k}
        # For k < n: dc_k/da_j = a_{k-j} (for j <= k) + a_{j-k} (for j >= k) ... 
        # Actually: c_k = sum_{i: 0<=i<n, 0<=k-i<n} a_i * a_{k-i}
        # dc_k/da_j: j appears as a_j * a_{k-j} (when 0 <= k-j < n) and a_{k-j} * a_j (when 0 <= j-k < n)
        # = a_{k-j} * [1 if 0<=k-j<n] + a_{j-k} * [1 if 0<=j-k<n]
        # If j = k-j, i.e., j = k/2, then it's 2*a_{k/2}
        
        # Simpler: just compute gradients via FFT
        fsize = get_fft_size(n)
        A = fft(a, fsize)
        
        G = np.zeros((top_K, n))
        for ii, k in enumerate(top_idx):
            e = np.zeros(2*n-1)
            e[int(k)] = 1.0
            E = fft(e, fsize)
            G[ii] = 2.0 * np.real(ifft(E * np.conj(A)))[:n]
        
        # We want: G @ delta = -eps * ones (reduce all top lags by eps)
        # With constraint: sum(delta) = 0
        # Minimum norm solution with sum constraint:
        # delta = -eps * (G^T @ inv(G @ G^T) @ 1 - mean correction)
        
        GGT = G @ G.T
        # Add small regularization for numerical stability
        GGT += 1e-10 * np.eye(top_K)
        
        try:
            sol = np.linalg.solve(GGT, np.ones(top_K))
            delta = -G.T @ sol
        except np.linalg.LinAlgError:
            break
        
        # Enforce sum(delta) = 0
        delta -= delta.mean()
        
        # Scale delta to have reasonable step size
        delta_norm = np.linalg.norm(delta)
        if delta_norm < 1e-15:
            break
        
        # Ensure a + alpha*delta >= 0
        # Find max alpha such that a + alpha*delta >= 0
        neg_mask = delta < -1e-15
        if np.any(neg_mask):
            alpha_max = (a[neg_mask] / (-delta[neg_mask])).min()
            alpha_max *= 0.9  # Safety factor
        else:
            alpha_max = 1.0
        
        alpha = min(alpha_max, S / (n * delta_norm) * 0.1)
        
        a_new = a + alpha * delta
        a_new = np.maximum(a_new, 0)
        a_new = a_new * (S / a_new.sum()) if a_new.sum() > 0 else a
        
        # Check if it improved
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max * (1 - 1e-12):
            a = a_new
        else:
            # Reduce step size
            alpha *= 0.5
            a_new = a + alpha * delta
            a_new = np.maximum(a_new, 0)
            a_new = a_new * (S / a_new.sum()) if a_new.sum() > 0 else a
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max:
                a = a_new
            else:
                # Try opposite direction
                a_new = a - 0.5 * alpha * delta
                a_new = np.maximum(a_new, 0)
                a_new = a_new * (S / a_new.sum()) if a_new.sum() > 0 else a
                c_new = fft_convolve_self(a_new)
                if c_new.max() < cur_max:
                    a = a_new
                else:
                    break
    
    return best_a, best_max

def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def make_anti_palindromic(n, rng, S):
    """Anti-palindromic initialization."""
    half = n // 2
    h = rng.random(half) + 0.01
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())
    return a

def make_cosine_init(n, rng, S):
    """Cosine window initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.cos(np.pi * tt) ** 2 + 0.01
    a = a * (S / a.sum())
    return a

def make_dual_peak(n, rng, S):
    """Two Gaussian peaks."""
    tt = np.arange(n, dtype=np.float64) / n
    c1 = rng.random() * 0.3 + 0.1
    c2 = 1.0 - c1
    w = rng.random() * 0.1 + 0.02
    a = np.exp(-0.5 * ((tt - c1) / w) ** 2) + np.exp(-0.5 * ((tt - c2) / w) ** 2)
    a = a * (S / a.sum())
    return a

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS-B quasi-Newton optimization
on a smoothed max-convolution objective, combined with active-set lag equalization,
anti-palindromic structured initializations, and coordinate-level polishing.

Key improvements over previous FISTA-based approach:
- L-BFGS-B (quasi-Newton) for superlinear convergence near optimum
- Active-set equalization: solve linear system to equalize top lags directly
- Multi-scale beta annealing (50 -> 500 -> 2000)
- Anti-palindromic + diverse structured initializations
- Coordinate-level fine polish at the end
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def get_fft_size(n):
    """Next power of 2 >= 2n-1."""
    size = 2 * n - 1
    return 1 << (size - 1).bit_length()


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    fsize = get_fft_size(n)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def lse_value_and_grad(a, beta):
    """Compute LSE(c; beta) and its gradient w.r.t. a using FFT."""
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, np.zeros(len(a))
    w = exp_bc / total
    lse_val = c_max + np.log(total) / beta

    fsize = get_fft_size(len(a))
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:len(a)]
    return lse_val, grad


def packed_obj_grad(a, beta):
    """Return (objective, gradient) for g_beta(a) = 2n * LSE / S^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return 1e10, np.zeros(n)
    lse_val, grad_lse = lse_value_and_grad(a, beta)
    g = 2.0 * n * lse_val / (S * S)
    grad = (2.0 * n / (S ** 3)) * (S * grad_lse - 2.0 * lse_val)
    return g, grad


def lbfgsb_run(a_init, n, beta, max_time, ftol=1e-12, verbose=False):
    """Run L-BFGS-B on the smoothed objective. Returns (best_a, best_obj)."""
    a_init = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    if a_init.sum() < 1e-15:
        a_init = np.ones(n) / n

    start = time.monotonic()
    best_a = a_init.copy()
    best_obj = packed_obj_grad(a_init, beta)[0]

    def func(a):
        a = np.maximum(a, 0.0)
        val, grad = packed_obj_grad(a, beta)
        return val, grad

    bounds = [(0.0, 1000.0)] * n
    it_count = [0]
    stop_flag = [False]

    def callback(xk):
        it_count[0] += 1
        if time.monotonic() - start > max_time - 2.0:
            stop_flag[0] = True
            return True
        if verbose and it_count[0] % 200 == 0:
            val, _ = packed_obj_grad(np.maximum(xk, 0), beta)
            print("    L-BFGS iter=%d obj=%.10f" % (it_count[0], val))
        return stop_flag[0]

    try:
        result = minimize(
            func, a_init, jac=True, method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 30000, 'maxfun': 50000,
                     'ftol': ftol, 'gtol': 1e-10, 'maxls': 40,
                     'callback': callback}
        )
        if result.x is not None:
            rx = np.clip(result.x, 0, 1000)
            rv, _ = packed_obj_grad(rx, beta)
            if rv < best_obj:
                best_obj = rv
                best_a = rx.copy()
    except Exception as e:
        if verbose:
            print("    L-BFGS error: %s" % str(e))

    return best_a, best_obj


def active_set_equalize(a, n, S, top_K, max_iter=2000, max_time=60.0):
    """Iteratively equalize top K convolution lags via linearized Newton steps."""
    a = np.clip(a, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_a = a.copy()
    c = fft_convolve_self(a)
    best_max = c.max()
    fsize = get_fft_size(n)

    for it in range(max_iter):
        if time.monotonic() - start > max_time - 1.0:
            break

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        top_idx = np.argsort(c)[::-1][:top_K]
        top_vals = c[top_idx]

        # If top lags are well-equalized and we're converging, stop
        if top_K > 1 and (top_vals[0] - top_vals[-1]) < 1e-12 * max(cur_max, 1e-30):
            if it > 50:
                break

        # Build gradient matrix G: row j = gradient of c[top_idx[j]] w.r.t. a
        A = fft(a, fsize)
        G = np.zeros((top_K, n))
        for ii in range(top_K):
            k = int(top_idx[ii])
            e = np.zeros(2 * n - 1)
            e[k] = 1.0
            E = fft(e, fsize)
            G[ii] = 2.0 * np.real(ifft(E * np.conj(A)))[:n]

        # Solve: find delta with sum(delta)=0 minimizing max_j (c_j + G_j @ delta)
        # Equivalent to: G @ delta = -eps * 1 (equalize all to cur_max - eps)
        # Min-norm solution: delta = -G^T (G G^T)^{-1} 1, then center
        GGT = G @ G.T
        GGT += 1e-8 * np.diag(np.abs(GGT.diagonal()) + 1e-30)
        try:
            sol = np.linalg.solve(GGT, np.ones(top_K))
            delta = -G.T @ sol
        except np.linalg.LinAlgError:
            try:
                sol, _ = np.linalg.lstsq(GGT, np.ones(top_K), rcond=None)
                delta = -G.T @ sol
            except Exception:
                break

        delta -= delta.mean()
        dnorm = np.linalg.norm(delta)
        if dnorm < 1e-18:
            break

        # Determine step size
        neg_mask = delta < -1e-15
        if np.any(neg_mask):
            alpha_max = (a[neg_mask] / (-delta[neg_mask])).min() * 0.95
        else:
            alpha_max = S / (0.1 * dnorm)
        alpha = min(alpha_max, 0.5 * S / (n * dnorm + 1e-30))

        a_new = a + alpha * delta
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new * (S / sn)

        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max:
            a = a_new
        else:
            # Try smaller step
            alpha *= 0.4
            a_new = a + alpha * delta
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new * (S / sn)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < cur_max:
                    a = a_new
                else:
                    # Try opposite
                    a_new = a - 0.3 * alpha * delta
                    a_new = np.maximum(a_new, 0.0)
                    sn = a_new.sum()
                    if sn > 1e-15:

                            elif it > 100:

                    else:

            else:
                break

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def make_anti_palindromic(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = mu."""
    half = n // 2
    h = rng.random(half) + 0.005
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2.0
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())
    return a


def make_cosine_window(n, rng, S):
    """Cosine window with random modulation."""
    tt = np.arange(n, dtype=np.float64) / n
    amp = 0.2 + 0.3 * rng.random()
    a = 1.0 + amp * np.cos(np.pi * tt)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_two_gaussian(n, rng, S):
    """Two Gaussian peaks."""
    tt = np.arange(n, dtype=np.float64) / n
    c1 = 0.15 + 0.2 * rng.random()
    c2 = 0.65 + 0.2 * rng.random()
    w1 = 0.03 + 0.05 * rng.random()
    w2 = 0.03 + 0.05 * rng.random()
    a = np.exp(-0.5 * ((tt - c1) / w1) ** 2) + np.exp(-0.5 * ((tt - c2) / w2) ** 2)
    a = a * (S / a.sum())
    return a


def make_sin_mod(n, rng, S):

        best_a = best_a * (S / best_a.sum())
    return best_a, best_obj


def run_phase2(n, a_start, max_time, rng, verbose=True):
    """Phase 2: Iterative refinement with increasing beta and active-set."""
    S = np.sqrt(2.0 * n)
    start = time.monotonic()
    best_a = a_start.copy()
    best_obj = 2.0 * n * fft_convolve_self(best_a).max() / (best_a.sum() ** 2)

    betas = [200, 500, 1000, 2000, 5000]
    for beta in betas:
        rem = max_time - (time.monotonic() - start)
        if rem < 10:
            break
        t_alloc = min(60.0, rem * 0.25)

        a_new, _ = lbfgsb_run(best_a, n, beta=beta, max_time=t_alloc,
                              ftol=1e-12, verbose=False)
        obj_new = 2.0 * n * fft_convolve_self(a_new).max() / (a_new.sum() ** 2)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print("    P2 beta=%d: obj=%.9f" % (beta, best_obj))

        # Active-set after each beta increase
        rem2 = max_time - (time.monotonic() - start)
        if rem2 > 5:
            K = min(10 + beta // 100, 40)
            a_as, v_as = active_set_equalize(best_a, n, S, top_K=K,
                                              max_iter=1000, max_time=min(20.0, rem2 * 0.15))
            obj_as = 2.0 * n * v_as / (S ** 2)
            if obj_as < best_obj:
                best_obj = obj_as
                best_a = a_as.copy()
                if verbose:
                    print("    P2 AS K=%d: obj=%.9f" % (K, best_obj))

    return best_a, best_obj


def run_phase3(n, a_start, max_time, rng, verbose=True):
    """Phase 3: Final intensive polish - coordinate descent + micro active-set."""
    S = np.sqrt(2.0 * n)
    start = time.monotonic()
    a = a_start.copy()
    s = a.sum()
    if s > 0:
        a = a * (S / s)
    best_obj = 2.0 * n * fft_convolve_self(a).max() / (S ** 2)

    # 3a: Very high beta L-BFGS
    fsize = get_fft_size(n)
    rem = max_time - (time.monotonic() - start)
    if rem > 10:
        a_new, _ = lbfgsb_run(a, n, beta=10000, max_time=min(30.0, rem * 0.3),
                              ftol=1e-13, verbose=False)
        obj_new = 2.0 * n * fft_convolve_self(a_new).max() / (a_new.sum() ** 2)
        if obj_new < best_obj:
            best_obj = obj_new
            a = a_new
            if verbose:
                print("    P3 L-BFGS b=10000: obj=%.9f" % best_obj)

    # 3b: Active-set with many lags
    rem = max_time - (time.monotonic() - start)
    if rem > 5:
        a_as, v_as = active_set_equalize(a, n, S, top_K=50,
                                          max_iter=5000, max_time=min(30.0, rem * 0.3))
        obj_as = 2.0 * n * v_as / (S ** 2)
        if obj_as < best_obj:
            best_obj = obj_as
            a = a_as
            if verbose:
                print("    P3 AS K=50: obj=%.9f" % best_obj)

    # 3c: Coordinate-level perturbations
    rem = max_time - (time.monotonic() - start)
    if rem > 5:
        c = fft_convolve_self(a)
        cur_max = c.max()
        # Compute gradient concentrated on top lags
        top_lags = np.argsort(c)[::-1][:20]
        w = np.zeros(2 * n - 1)
        top_vals = c[top_lags]
        tv_max = top_vals.max()
        weights = np.exp((top_vals - tv_max) / 0.01)
        weights = weights / weights.sum()
        for ii, k in enumerate(top_lags):
            w[int(k)] = weights[ii]
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Try perturbations on top-gradient coordinates
        coord_order = np.argsort(np.abs(grad))[::-1]
        step = S / n * 0.05
        improved = 0
        for ci in range(min(200, n)):
            if time.monotonic() - start > max_time - 3:
                break
            idx = coord_order[ci]
            for direction in [-1, 1]:
                a_try = a.copy()
                a_try[idx] += direction * step
                if a_try[idx] < 0:
                    continue
                a_try = np.maximum(a_try, 0)
                sn = a_try.sum()
                if sn < 1e-15:
                    continue
                a_try = a_try * (S / sn)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < cur_max - 1e-13:
                    a = a_try
                    cur_max = c_try.max()
                    best_obj = 2.0 * n * cur_max / (S ** 2)
                    improved += 1
                    step *= 0.85
                    break
        if verbose and improved > 0:
            print("    P3 coord: obj=%.9f (%d imp.)" % (best_obj, improved))

    a = a * (S / a.sum())
    return a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function.

    Uses L-BFGS-B quasi-Newton on smoothed max-convolution + active-set
    lag equalization + coordinate polish. Multi-phase with beta annealing.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(1.0, budget_s - 20.0)

    print("Budget: %.0fs, phases within %.0fs" % (budget_s, budget_s - 20.0))

    # Get parent
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
            print("Parent: length=%d, sum=%.4f" % (len(parent), parent.sum()))
    except Exception:
        pass

    global_best = np.inf
    global_a = None

    # Determine primary length
    if parent is not None:
        n_primary = len(parent)
    else:
        n_primary = 8192

    print("=== Phase 1: Exploration (n=%d) ===" % n_primary)
    t0 = time.monotonic()
    t1 = min(700.0, max(60.0, (deadline - t0) * 0.35))
    a1, obj1 = run_phase1(n_primary, parent, t1, rng, verbose=True)
    if a1 is not None and obj1 < global_best:
        global_best = obj1
        global_a = a1
    print("Phase 1 done: obj=%.9f, elapsed=%.0fs" % (global_best, time.monotonic() - t0))

    # Phase 2: Refinement
    if global_a is not None and time.monotonic() < deadline - 120:
        n = len(global_a)
        print("=== Phase 2: Refinement (n=%d) ===" % n)
        t0 = time.monotonic()
        rem = min(600.0, max(30.0, (deadline - t0) * 0.3))
        a2, obj2 = run_phase2(n, global_a, rem, rng, verbose=True)
        if obj2 < global_best:
            global_best = obj2
            global_a = a2
        print("Phase 2 done: obj=%.9f, elapsed=%.0fs" % (global_best, time.monotonic() - t0))

    # Phase 3: Final polish
    if global_a is not None and time.monotonic() < deadline - 60:
        n = len(global_a)
        print("=== Phase 3: Final Polish (n=%d) ===" % n)
        t0 = time.monotonic()
        rem = min(300.0, max(20.0, (deadline - t0) * 0.3))
        a3, obj3 = run_phase3(n, global_a, rem, rng, verbose=True)
        if obj3 < global_best:
            global_best = obj3
            global_a = a3
        print("Phase 3 done: obj=%.9f, elapsed=%.0fs" % (global_best, time.monotonic() - t0))

    # Phase 4: Restart exploration (different structure) if time remains
    if time.monotonic() < deadline - 120:
        n_try = n_primary
        print("=== Phase 4: Restart (n=%d) ===" % n_try)
        t0 = time.monotonic()
        rem = min(200.0, max(15.0, (deadline - t0) * 0.25))

        # New random seed for diversity
        rng2 = np.random.default_rng(seed + 9999)
        S = np.sqrt(2.0 * n_try)
        a_new_init = make_anti_palindromic(n_try, rng2, S)
        # Perturb the best found
        if global_a is not None and len(global_a) == n_try:
            noise = 0.02 * S / np.sqrt(n_try)
            a_new_init = proj = project_simplex(global_a + rng2.normal(0, noise, n_try), S)

        a4, obj4 = lbfgsb_run(a_new_init, n_try, beta=200.0, max_time=rem * 0.4,
                              ftol=1e-11, verbose=False)
        obj4 = 2.0 * n_try * fft_convolve_self(a4).max() / (a4.sum() ** 2)
        if obj4 < global_best:
            global_best = obj4
            global_a = a4
            print("  Phase 4 improved: obj=%.9f" % global_best)

        # Quick active-set
        rem2 = (deadline - time.monotonic()) - 30
        if rem2 > 5:
            a4b, v4b = active_set_equalize(global_a, n_try, S, top_K=20,
                                            max_iter=2000, max_time=min(20.0, rem2 * 0.5))
            obj4b = 2.0 * n_try * v4b / (S ** 2)
            if obj4b < global_best:
                global_best = obj4b
                global_a = a4b
                print("  Phase 4 AS: obj=%.9f" % global_best)

    # Final verification
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64) * 1.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== VERIFICATION: eval=%.9f, internal=%.9f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(final_list)))

    if not np.isfinite(ver):
        print("  WARNING: evaluate rejected! Using fallback.")
        global_a = np.ones(1024, dtype=np.float64) * 1.0
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback eval: %.9f" % ver)

    print("=== FINAL ANSWER: %.9f (n=%d) ===" % (ver, len(final_list)))
    return final_list
