    # Build constraint matrix

    C[K, :] = 1.0  # sum(g) <= S

    # Minimize -sum(g)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=C, b_ub=b, bounds=[(0, None)] * n, method='highs')

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key innovation: LP-based direction finding using the fact that conv(f,g) is
linear in g when f is fixed. At each step, solve an LP to find a direction g
that reduces the top peaks of the convolution, then take a projected step.

Algorithm:
1. Generate diverse initializations (frequency-domain flat-spectrum, ramps, parent)
2. LP-based optimization with adaptive active set (top K peaks)
3. FISTA polish with log-sum-exp for final refinement
4. Backtracking/line search for robust step selection
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sequence(a, S):
    """Normalize a to have sum = S, clip to [0, 1000]."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * S / len(a)
    a = a * (S / s)
    return np.clip(a, 0.0, 1000.0)


def build_lp_matrix(f, peak_indices):
    """Build the LP constraint matrix for conv(f, g) <= M at given peaks.
    Returns matrix C of shape (len(peak_indices), n) where C[i,j] is the
    coefficient of g[j] in the constraint for peak peak_indices[i].
    """
    n = len(f)
    K = len(peak_indices)
    C = np.zeros((K, n))
    for i, k in enumerate(peak_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k) + 1
        if j_start < j_end:
            js = np.arange(j_start, j_end)
            C[i, js] = f[k - js]
    return C


def solve_lp_direction(f, peak_indices, M, S):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for peaks, sum(g) <= S, g >= 0.
    Returns g (not yet normalized) or None if infeasible.
    """
    n = len(f)
    K = len(peak_indices)
    if K == 0:
        return None

    C = build_lp_matrix(f, peak_indices)

    # Full constraint matrix: K peak rows + 1 sum row
    A_ub = np.vstack([C, np.ones((1, n))])
    b_ub = np.concatenate([np.full(K, M), [S]])

    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None

    if res.status != 0:
        return None

    g = res.x
    if g is None or not np.all(np.isfinite(g)):
        return None
    return np.maximum(g, 0.0)


def lp_optimize(a_init, S, max_time, rng, K_start=30, K_end=300, eps=0.0):
    """LP-based optimization with adaptive active set."""
    a = normalize_sequence(a_init, S)
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()

    c = fft_convolve_self(a)
    M = c.max()
    best_M = M

    t = 0.3  # step size
    no_improve_count = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = elapsed / max_time

        # Adaptive K
        K = int(K_start + (K_end - K_start) * min(1.0, progress * 1.5))
        K = min(K, 2 * n - 1, 500)

        c = fft_convolve_self(a)
        M = c.max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # If stuck too long, try to restart with perturbation
        if no_improve_count > 200:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            t = 0.3
            no_improve_count = 0
            continue

        # Find top K peaks
        peak_indices = np.argsort(c)[::-1][:K]
        max_peak_val = c[peak_indices[0]]

        if max_peak_val < 1e-15:
            break

        # Solve LP
        target_M = max_peak_val * (1.0 + eps)
        g = solve_lp_direction(a, peak_indices, target_M, S)

        if g is None or g.sum() < 1e-15:
            # Fallback: use negative gradient direction
            top_val = c.max()
            top_k = int(np.argmax(c))
            grad = np.zeros(n)
            for m in range(n):
                j1 = top_k - m
                if 0 <= j1 < n:
                    grad[m] += a[j1]
                j2 = m - top_k
                if 0 <= j2 < n:
                    grad[m] += a[j2]
            a_new = project_simplex(a - 0.01 * S * grad / max(np.linalg.norm(grad), 1e-15), S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < M:
                a = a_new
            continue

        # Normalize g
        g = g * (S / g.sum())

        # Line search for step size
        best_t = t
        best_M_new = M
        for t_try in [t, t * 0.5, t * 2, t * 0.25, 0.01, 0.1, 0.5]:
            t_try = min(1.0, max(0.001, t_try))
            a_new = (1.0 - t_try) * a + t_try * g
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn < 1e-15:
                continue
            a_new = a_new * (S / sn)
            c_new = fft_convolve_self(a_new)
            M_new = c_new.max()
            if M_new < best_M_new:
                best_M_new = M_new
                best_t = t_try

        if best_M_new < M:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            a_new = a_new * (S / sn)
            a = a_new
            t = best_t
            # Gradually increase step size
            if it % 100 == 0:
                t = min(t * 1.2, 0.8)
        else:
            # No improvement; reduce step and slightly increase K
            t = max(0.01, t * 0.5)
            if it % 50 == 0:
                # Perturb to escape
                noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)

    return best_a, best_M


def freq_flat_init(n, S, rng, n_iter=50):
    """Frequency-domain flat-spectrum initialization."""
    a = rng.random(n) * 0.5 + 0.1
    a = normalize_sequence(a, S)

    fsize = 1 << (2 * n - 1).bit_length()

    for _ in range(n_iter):
        A = fft(a, fsize)[:n]
        # Flatten magnitude
        mags = np.abs(A)
        target_mag = mags.mean()
        if target_mag < 1e-15:
            break
        phases = np.angle(A)
        A_new = target_mag * np.exp(1j * phases)
        a_new = np.real(ifft(A_new, fsize))[:n]
        a = np.maximum(a_new, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a * (S / s)

    return a


def gen_ramp_init(n, S, rng, alpha=2.0, noise_frac=0.1):
    """Generate ramp-like initialization with noise."""
    t = np.linspace(0, 1, n, endpoint=False)
    shape = np.exp(-alpha * t)
    noise = rng.random(n) * noise_frac
    a = shape + noise
    return normalize_sequence(a, S)


def gen_bump_init(n, S, rng, center=0.3, width=0.2, height=1.0):
    """Generate a bump (Gaussian-like) initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = height * np.exp(-((t - center) / width) ** 2)
    a += rng.random(n) * 0.05
    return normalize_sequence(a, S)


def gen_two_bump_init(n, S, rng, c1=0.2, c2=0.7, w1=0.15, w2=0.1, h1=1.0, h2=0.5):
    """Two-bump asymmetric initialization."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = h1 * np.exp(-((t - c1) / w1) ** 2) + h2 * np.exp(-((t - c2) / w2) ** 2)
    a += rng.random(n) * 0.05
    return normalize_sequence(a, S)


def fista_polish(a_init, S, max_time, rng, beta=100.0):
    """FISTA polish with log-sum-exp gradient."""
    n = len(a_init)
    a = normalize_sequence(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_M = fft_convolve_self(a).max()
    it = 0

    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur_M = c.max()
        if cur_M < best_M:
            best_M = cur_M
            best_a = y.copy()

        # Log-sum-exp gradient
        c_max = c.max()
        beta_eff = min(beta, 500.0)
        exp_bc = np.exp(beta_eff * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = exp_bc / total

        # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m] (cross-correlation)
        conv_size = 2 * n - 1
        fsize2 = 1 << (conv_size + n - 1).bit_length()
        W = fft(w, fsize2)
        A = fft(y, fsize2)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta_eff + 10.0)
        step = max(step, 1e-6)

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic noise
        if it % 500 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0

    return best_a, best_M


def objective_internal(a):
    """Compute objective for internally-normalized sequence (sum = sqrt(2n))."""
    n = len(a)
    c = fft_convolve_self(a)
    return c.max()  # Since sum = sqrt(2n), objective = max_c

        global_a = np.ones(4096)
        global_a = normalize_sequence(global_a, np.sqrt(2 * 4096))

    # --- Validation ---
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(4096)
        global_a = normalize_sequence(global_a, np.sqrt(2 * 4096))

    # Round to avoid floating point issues
    global_a = np.round(global_a, 10)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification: %.6f (internal: %.6f)" % (ver if np.isfinite(ver) else float('inf'), global_best))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using safe fallback")
        global_a = np.ones(4096)
        S_safe = np.sqrt(2 * 4096)
        global_a = normalize_sequence(global_a, S_safe)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)

    print("Returning sequence of length %d, verified=%.6f" % (len(final_list), ver))
    return final_list
