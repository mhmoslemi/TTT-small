"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Softmax-smoothed gradient descent with adaptive temperature
2. Frequency-domain spectral equalization  
3. Peak-specific mass redistribution
4. Multi-start with diverse initializations

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def cross_corr_gradient(a, w):
    """
    Compute grad[m] = 2 * sum_j a[j] * w[j+m] for m=0..n-1
    using FFT.
    """
    n = len(a)
    # Need N >= n + (2n-1) - 1 = 3n-2
    N = 1 << (3 * n - 2 + 1).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    corr = np.real(ifft(np.conj(A) * W))
    return 2.0 * corr[:n]


def softmax_objective(a, c, beta):
    """Compute softmax-approximated max of c and its gradient w.r.t. a."""
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    log_sum = np.log(exp_c.sum())
    obj_smooth = c_max + log_sum / beta  # This is the logsumexp
    
    # Weights for gradient
    w = exp_c / exp_c.sum()
    
    # Gradient
    grad = cross_corr_gradient(a, w)
    
    return obj_smooth, w, grad


def spectral_equalize(a, n_iter=100, nonneg=True):
    """
    Iteratively equalize the magnitude spectrum of a.
    This pushes toward flat autocorrelation.
    """
    n = len(a)
    S = a.sum()
    a_work = a.copy()
    
    for _ in range(n_iter):
        A = fft(a_work, 2 * n)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: constant magnitude
        target_mag = np.sqrt(np.mean(mag ** 2))
        
        # Blend: partially move toward target
        A_new = (0.5 * mag + 0.5 * target_mag) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, 2 * n))[:n]
        
        if nonneg:
            a_new = np.maximum(a_new, 0.0)
        
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new * (S / s)
        
        a_work = a_new
    
    return a_work


def mass_redistribute(a, c, top_peaks=5, num_moves=100):
    """
    For top peaks, identify contributing indices and try to redistribute mass
    from high-contributing to low-contributing indices.
    """
    n = len(a)
    S = a.sum()
    a_work = a.copy()
    c_work = c.copy()
    best_obj = obj_value(a_work, c_work)
    
    # Get top peak indices
    top_idx = np.argsort(c_work)[::-1][:top_peaks]
    
    for _ in range(num_moves):
        # Pick a random peak
        k = top_idx[np.random.randint(top_peaks)]
        
        # Find contributing pairs
        valid_i = np.arange(max(0, k - n + 1), min(k + 1, n))
        pairs = [(i, k - i) for i in valid_i]
        
        if not pairs:
            continue
        
        # Find the pair with maximum contribution
        max_contrib = 0
        max_i = None
        max_j = None
        for i, j in pairs:
            contrib = a_work[i] * a_work[j]
            if contrib > max_contrib:
                max_contrib = contrib
                max_i = i
                max_j = j
        
        if max_i is None:
            continue
        
        # Try to reduce a[max_i] or a[max_j] and add to a random other index
        # Pick the larger one to reduce
        if a_work[max_i] >= a_work[max_j]:
            reduce_idx = max_i
        else:
            reduce_idx = max_j
        
        # Find a good target to increase: an index that contributes to few peaks
        # Simple heuristic: pick a random index
        target_idx = np.random.randint(n)
        if target_idx == reduce_idx:
            target_idx = (target_idx + 1) % n
        
        # Transfer small amount
        transfer = 0.01 * a_work[reduce_idx]
        a_work[reduce_idx] -= transfer
        a_work[target_idx] += transfer
        
        # Check if improved
        c_check = fft_convolve_self(a_work)
        obj_check = c_check.max()  # We compare max(c) since sum is fixed
        
        if obj_check < best_obj:
            best_obj = obj_check
            c_work = c_check
        else:
            # Revert
            a_work[reduce_idx] += transfer
            a_work[target_idx] -= transfer
    
    return a_work


def optimize_softmax(a0, time_budget, beta_start=50.0, beta_end=500.0,
                     lr_start=0.1, lr_end=0.001, verbose=True,
                     use_multiplicative=True, cool_rate=0.9999):
    """
    Main optimization loop using softmax-smoothed objective.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    
    beta = beta_start
    lr = lr_start
    iter_no = 0
    start = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - start < time_budget:

        _, w, grad = softmax_objective(a, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Update

            a_new = a * np.exp(-lr * grad)
            a_new = np.maximum(a_new, 1e-30)
        else:
            a_new = a - lr * grad * S / n
            a_new = np.maximum(a_new, 0.0)
        
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s)
        
        c_new = fft_convolve_self(a_new)
        obj_new = obj_value(a_new, c_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
            no_improve = 0
        else:
            no_improve += 1
            # Adapt step size
            if no_improve % 50 == 0:
                lr *= 0.9
            # Try with current a (don't update)
        
        # Cool beta (increase for sharper focus)
        beta = min(beta * cool_rate ** (-1), beta_end)  # Actually cool means increase beta
        # Let me use a different schedule

def optimize_multiplicative_subgrad(a0, time_budget, top_k=10, eta=0.3,
                                     momentum=0.85, verbose=True):
    """
    Multiplicative (mirror) descent with top-k subgradient.
    Similar to previous program but in a standalone function.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    
    v = np.zeros(n)
    iter_no = 0
    start = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        # Top-k subgradient
        top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
        w_full = np.zeros(len(c))
        w_full[top_idx] = 1.0 / top_k
        
        grad = cross_corr_gradient(a, w_full)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        v = momentum * v + (1 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm
        
        # Multiplicative update
        a_new = a * np.exp(-eta * v)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new * (S / a_new.sum())
        
        c_new = fft_convolve_self(a_new)
        obj_new = obj_value(a_new, c_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
            no_improve = 0
            eta = min(eta * 1.01, 1.0)
        else:
            no_improve += 1
            eta = max(eta * 0.95, 0.01)
            # Don't update a
        
        if no_improve > 1000:
            # Perturb
            noise = np.random.randn(n) * 0.02 * S / np.sqrt(n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.1
            no_improve = 0
        
        # Decay eta slightly
        eta *= 0.99999
        
        iter_no += 1
        if verbose and iter_no % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"      [subgrad] iter={iter_no}, obj={obj_value(a,c):.6f}, "
                  f"best={best_obj:.6f}, eta={eta:.4f}, top_k={top_k}, t={elapsed:.1f}s")
    
    return best_a, best_obj
