import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def objective_from_conv(a, c):
    """Compute objective given a and its convolution c."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def weighted_subgradient(a, c, top_k, temperature):
    """
    Compute weighted subgradient of max(c) approximation.
    Uses exponential weighting of the top-k lags with given temperature.
    
    grad[m] = 2 * sum_k w[k] * a[k-m] (cross-correlation of weights with a)
    """
    n = len(a)
    c_max = c.max()
    
    # Compute weights: exponential of (c - c_max) / temperature over top-k
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    
    # Temperature-scaled weights
    shifted = (top_vals - c_max) / temperature
    exp_w = np.exp(shifted)
    total = exp_w.sum()
    if total < 1e-300:
        weights = np.ones(top_k) / top_k
    else:
        weights = exp_w / total
    
    # Build full weight vector
    c_len = len(c)
    w_full = np.zeros(c_len)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    
    # Cross-correlation: grad[m] = 2 * sum_k w_full[k] * a[k-m]
    fsize = 1 << (c_len + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_subgradient(a, c, beta):
    """Compute gradient of logsumexp(beta*c)/beta w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""

    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

        global_a = safe_normalize(global_a, n_main)
    
    n_best = len(global_a)
    remaining = deadline - time.monotonic()
    
    if remaining > 30:
        print("=== Phase 2: Deep Optimization (%.0f sec, n=%d) ===" % (remaining, n_best))
        
        # Split time between different methods
        method1_time = remaining * 0.4
        method2_time = remaining * 0.3
        method3_time = remaining * 0.3
        
        # Method 1: Subgradient with backtracking (adaptive K)
        a1, s1 = optimize_sequence(global_a, method1_time, rng, n_phase="main")
        a1 = safe_normalize(a1, n_best)
        c1 = fft_convolve_self(a1)
        if c1.max() < global_best:
            global_best = c1.max()
            global_a = a1.copy()
            print("  Phase 2a (subgrad+bt): %.6f" % global_best)
        
        # Method 2: FISTA with high beta (sharp softmax)
        remaining2 = deadline - time.monotonic()
        if remaining2 > 5:
            a2, s2 = optimize_fista_soft(global_a, method2_time, rng,
                                          beta_init=50.0, beta_max=2000.0)
            a2 = safe_normalize(a2, n_best)
            c2 = fft_convolve_self(a2)
            if c2.max() < global_best:
                global_best = c2.max()
                global_a = a2.copy()
                print("  Phase 2b (FISTA high-beta): %.6f" % global_best)
        
        # Save state
        print("  After Phase 2: %.6f" % global_best)
    
    # Phase 3: Micro-optimization
    remaining3 = deadline - time.monotonic()
    if remaining3 > 5 and global_a is not None:
        print("=== Phase 3: Micro Optimization (%.0f sec) ===" % remaining3)
        
        # Use the remaining time for very targeted optimization
        a3, s3 = optimize_micro(global_a, remaining3, rng)
        a3 = safe_normalize(a3, n_best)
        c3 = fft_convolve_self(a3)
        if c3.max() < global_best:
            global_best = c3.max()
            global_a = a3.copy()
            print("  Phase 3 (micro): %.6f" % global_best)
    
    # Final verification
    if global_a is None:
        global_a = np.exp(-3.0 * np.linspace(0, 1, 8192, endpoint=False))
        global_a = safe_normalize(global_a, 8192)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    
    print("=== FINAL ===")
    print("  Length: %d" % len(final_list))
    print("  Sum: %.4f" % sum(final_list))
    print("  Internal best: %.6f" % global_best)
    print("  Verified: %.6f" % ver if np.isfinite(ver) else "  Verified: inf")
    
    if not np.isfinite(ver):
        print("  Fallback to simple sequence")
        n_fb = 8192
        tt = np.linspace(0, 1, n_fb, endpoint=False)
        global_a = np.exp(-3.0 * tt)
        S_fb = np.sqrt(2.0 * n_fb)
        global_a = global_a * (S_fb / global_a.sum())
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback verified: %.6f" % ver)
    
    return final_list
