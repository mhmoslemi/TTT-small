"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def topk_lse_gradient(a, c, beta, k):
    """LSE gradient only on top-k lags."""
    n = len(a)
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    e = np.exp(beta * (c_top - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w_top = e / tot
    w_full = np.zeros(len(c))
    w_full[idx] = w_top
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def exact_subgradient(a, c, topk=1):
    """Exact subgradient using top-k lags."""
    n = len(a)
    idx = np.argsort(c)[::-1][:topk]
    grad = np.zeros(n)
    for k in idx:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        idx2 = np.arange(lo, hi + 1)
        grad[idx2] += 2.0 * a[k - idx2] / topk
    return grad

def spectral_init(n, rng, alpha=1.0, p=1.0):
    """Initialize using spectral shaping: |F|^2 ~ 1/(1 + alpha*|f|^p)."""
    f = np.fft.fftfreq(n, d=1.0/n)
    f_norm = np.abs(f) * n  # normalize to [0, n/2]
    F_mag = 1.0 / (1.0 + alpha * np.power(f_norm, p))
    # Random phase
    phase = rng.uniform(0, 2*np.pi, n)
    phase[0] = 0  # DC has zero phase
    F = F_mag * np.exp(1j * phase)
    a = np.real(ifft(F))
    a = np.maximum(a, 0.0)
    return a

def downsample_upsample(a, n_small):
    """Downsample by averaging blocks, then linearly interpolate back."""
    n = len(a)
    if n_small >= n:
        return a.copy()
    nblocks = n // n_small
    a_small = a[:nblocks * n_small].reshape(n_small, nblocks).mean(axis=1)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n)
    a_up = np.interp(t_new, t_old, a_small)
    return a_up

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Adaptive top-k LSE FISTA with multi-scale warm starts,
spectral shaping initializations, progressive beta annealing,
and aggressive restarts to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Rescale to sum = sqrt(2n) and clip to [0, 1000]."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_lse_gradient(a, c, beta, k):
    """LSE gradient restricted to top-k lags only."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    c_top = c[idx]
    cm = c_top.max()
    e = np.exp(beta * (c_top - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w_top = e / tot
    w_full = np.zeros(len(c))
    w_full[idx] = w_top
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exact_subgradient_topk(a, c, k):
    """Exact subgradient averaged over top-k argmax lags."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for kk in idx:
        lo = max(0, kk - n + 1)
        hi = min(kk, n - 1)
        idx2 = np.arange(lo, hi + 1)
        grad[idx2] += 2.0 * a[kk - idx2] / k
    return grad


def run_fista_adaptive(a_init, n, max_time, rng,
                       beta_init=5.0, beta_rate=0.003, beta_max=3000.0,
                       topk_start=4, topk_end=256, use_exact=False):
    """FISTA with adaptive top-k LSE and optional exact subgradient steps."""
    S = np.sqrt(2.0 * n)
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: start moderate, grow to high
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        # Adaptive topk: start small, grow over iterations
        frac = min(1.0, it / (max_time * 10.0))
        topk = int(topk_start + (topk_end - topk_start) * frac)
        topk = max(4, min(topk, 2 * n - 1))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        # Choose gradient type
        use_exact_step = use_exact and it > 200 and (it % 50 == 0) and (rng.random() < 0.3)
        if use_exact_step:
            k_exact = min(8, max(1, it // 200))
            grad = exact_subgradient_topk(y, c, k_exact)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            # Step size for subgradient: use ratio of S to gradient norm
            step = S / (gnorm * np.sqrt(n) * (1.0 + 0.005 * np.log1p(it / 100.0)))
            step = min(step, 2.0 / (n * np.sqrt(cur + 1e-10)))
        else:
            grad = topk_lse_gradient(y, c, beta, topk)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta)

        # Stagnation handling
        stagnant = it - last_improve_it
        if stagnant > 2000:
            # Inject noise and reset momentum
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve_it = it
        elif stagnant > 1000:
            step *= 0.7

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)

        a = x_new
        t_mom = t_new

        # Periodic larger restart

                t_mom = 1.0

    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()

        # Get top lags
        topk = min(8, 1 + it // 500)
        idx = np.argsort(c)[::-1][:topk]

        # Gradient from top lags
        grad = np.zeros(n)
        for kk in idx:
            lo = max(0, kk - n + 1)
            hi = min(kk, n - 1)
            idx2 = np.arange(lo, hi + 1)
            grad[idx2] += 2.0 * a[kk - idx2]
        grad /= topk

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / (n * 0.3)
        for _ in range(5):
            a_trial = project_simplex(a - step * grad / max(gnorm, 1e-15), S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur:
                a = a_trial
                break
            step *= 0.5
        else:
            # No improvement even with tiny step
            # Try adding small noise
            noise_scale = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_a, best


def spectral_init(n, rng, alpha=1.0, p=1.0):
    """Initialize using spectral shaping."""
    f = np.fft.fftfreq(n, d=1.0 / n)
    f_norm = np.abs(f) * n
    F_mag = 1.0 / (1.0 + alpha * np.power(f_norm, p))
    phase = rng.uniform(0, 2 * np.pi, n)
    phase[0] = 0.0
    F = F_mag * np.exp(1j * phase)
    a = np.real(ifft(F))
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-4.0 * ((tt - 0.5) / 0.3) ** 2))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(np.sin(np.pi * tt) ** 6)
    inits.append(1.0 - 2.0 * np.abs(tt - 0.5))

    # Multi-sinusoidal
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 3 * tt))
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * 1.5 * tt) + 0.15 * np.cos(2 * np.pi * 4 * tt))

    # Spectral
    for alpha in [0.5, 1.0, 2.0]:
        inits.append(spectral_init(n, rng, alpha=alpha, p=1.0))
    inits.append(spectral_init(n, rng, alpha=1.0, p=2.0))

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.3, n)))

    return [np.maximum(x, 0.001) for x in inits]


def downsample_upsample(a, n_small):
    """Downsample by block averaging, upsample by interpolation."""
    n = len(a)
    if n_small >= n:
        return a.copy()
    nblocks = n // n_small
    a_small = a[:nblocks * n_small].reshape(n_small, nblocks).mean(axis=1)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n)
    a_up = np.interp(t_new, t_old, a_small)
    return a_up


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing 2*n*max_conv/sum^2."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)

    global_best = np.inf
    global_a = None

    # Try to use the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Determine main length
    if given is not None and 1024 <= len(given) <= 16384:
        main_len = len(given)
    else:
        main_len = 8192

    n = main_len
    print("Main n=%d, budget=%.0fs" % (n, budget_s))

    # === Build initializations ===
    inits = []
    if given is not None:
        inits.append(("given_full", given))
        if n >= 4096:
            inits.append(("given_c4", downsample_upsample(given, n // 4)))
            inits.append(("given_c8", downsample_upsample(given, n // 8)))
        if n <= 8192:
            # Upsample to 2x
            t_old = np.linspace(0, 1, len(given))
            t_new = np.linspace(0, 1, 2 * len(given))
            inits.append(("given_up2", np.interp(t_new, t_old, given)))

    gen_inits = generate_initializations(n, rng)
    for i, g in enumerate(gen_inits[:10]):
        inits.append(("gen_%d" % i, g))

    print("Total initializations: %d" % len(inits))

    # === Phase 1: Exploration ===
    exploration_frac = 0.55
    exploration_time = (deadline - time.monotonic()) * exploration_frac
    exploration_time = max(60.0, exploration_time)
    n_inits = len(inits)
    per_init = exploration_time / max(1, n_inits)

    # Weight: give "given" inits more time
    time_weights = np.ones(n_inits)
    gi = 0
    if given is not None:
        time_weights[0] = 3.0
        time_weights[1] = 1.5
        if n_inits > 2:
            time_weights[2] = 1.5
        gi = 3
    time_weights[gi + 0] = 1.2  # first gen init
    time_weights /= time_weights.sum()
    per_init_times = exploration_time * time_weights

    print("=== Phase 1: Exploration (%d inits, %.0fs) ===" % (n_inits, exploration_time))
    p1_start = time.monotonic()

    for i, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - (deadline - p1_start) * (1.0 - exploration_frac) - 40:
            break
        if len(arr) < 64:
            continue
        t_alloc = per_init_times[i]
        if t_alloc < 8:
            continue

        arr_n = len(arr)
        # Vary parameters for diversity
        beta_inits = [5.0, 8.0, 3.0, 10.0, 5.0, 15.0, 5.0, 8.0, 12.0, 6.0]
        bi = beta_inits[i % len(beta_inits)]
        use_exact = (i % 3 == 0)

        a_opt, score = run_fista_adaptive(arr, arr_n, t_alloc, rng,
                                            beta_init=bi, beta_rate=0.003,
                                            beta_max=3000.0,
                                            topk_start=2, topk_end=200,
                                            use_exact=use_exact)
        a_norm = normalize_sum(a_opt, arr_n)
        c_chk = fft_convolve_self(a_norm)
        real_score = c_chk.max()

        if real_score < global_best:
            global_best = real_score
            global_a = a_norm.copy()
            print("  [%s] n=%d: %.6f (elapsed %.0fs)" % (
                name, arr_n, global_best, time.monotonic() - p1_start))

    print("Phase 1 done: best=%.6f" % global_best)

    # === Phase 2: Polish ===
    if global_a is not None:
        n_best = len(global_a)
        remaining = deadline - time.monotonic()
        print("=== Phase 2: Polish (remaining %.0fs) ===" % remaining)

        if remaining > 20:
            # Step 1: High-beta FISTA
            t1 = remaining * 0.35
            a_p1, s1 = run_fista_adaptive(global_a, n_best, t1, rng,
                                            beta_init=30.0, beta_rate=0.004,
                                            beta_max=5000.0,
                                            topk_start=4, topk_end=300,
                                            use_exact=True)
            a_p1 = normalize_sum(a_p1, n_best)
            c_p1 = fft_convolve_self(a_p1)
            if c_p1.max() < global_best:
                global_best = c_p1.max()
                global_a = a_p1.copy()
                print("  After FISTA polish: %.6f" % global_best)

            # Step 2: Very high beta polish
            remaining2 = deadline - time.monotonic()
            if remaining2 > 15:
                t2 = remaining2 * 0.35
                a_p2, s2 = polish_high_beta(global_a, n_best, t2, rng)
                a_p2 = normalize_sum(a_p2, n_best)
                c_p2 = fft_convolve_self(a_p2)
                if c_p2.max() < global_best:
                    global_best = c_p2.max()
                    global_a = a_p2.copy()
                    print("  After high-beta polish: %.6f" % global_best)

            # Step 3: Peak shaving
            remaining3 = deadline - time.monotonic()
            if remaining3 > 10:
                a_p3, s3 = peak_shave_direct(global_a, n_best, remaining3 * 0.6, rng)
                a_p3 = normalize_sum(a_p3, n_best)
                c_p3 = fft_convolve_self(a_p3)
                if c_p3.max() < global_best:
                    global_best = c_p3.max()
                    global_a = a_p3.copy()
                    print("  After peak shave: %.6f" % global_best)

    # === Phase 3: Try alternate length ===
    remaining = deadline - time.monotonic()
    if remaining > 80:
        alt_n = n * 2 if n <= 10240 else n // 2
        if 2048 <= alt_n <= 32768 and alt_n != n:
            print("=== Phase 3: Alt n=%d (remaining %.0fs) ===" % (alt_n, remaining))
            alt_inits = generate_initializations(alt_n, rng)[:4]
            for ai, arr2 in enumerate(alt_inits[:2]):
                if time.monotonic() > deadline - 30:
                    break
                a_alt, s_alt = run_fista_adaptive(arr2, alt_n, remaining * 0.2, rng,
                                                    beta_init=8.0, beta_rate=0.004,
                                                    beta_max=3000.0,
                                                    topk_start=2, topk_end=200,
                                                    use_exact=True)
                a_alt = normalize_sum(a_alt, alt_n)
                c_alt = fft_convolve_self(a_alt)
                if c_alt.max() < global_best:
                    global_best = c_alt.max()
                    global_a = a_alt.copy()
                    print("  Alt n=%d: %.6f" % (alt_n, global_best))

    # === Final verification ===
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("FINAL SCORE: %.6f (length %d)" % (ver, len(global_a)))
        else:
            print("Verification failed, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    return [float(x) for x in global_a]
