import time
from scipy.fft import fft, ifft

def compute_autocorr(a):
    """Compute full autocorrelation (self-convolution) of a."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Binary search for threshold
    lo = v.min() - S
    hi = v.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.maximum(v - mid, 0)
        s = proj.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    return np.maximum(v - theta, 0.0)

    Iteratively flatten the spectrum to reduce autocorrelation peak.
    a: input sequence (non-negative, sum = S)
    n_iters: number of iterations
    lambda_schedule: array of lambda values (flattening strength)
    """
    a = a.copy().astype(np.float64)
    S = a.sum()
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    
    for it in range(n_iters):
        lam = lambda_schedule[min(it, len(lambda_schedule) - 1)]
        
        A = fft(a, fsize)[:2*n-1]  # Actually need full size
        # Hmm, I need to be careful with the FFT size
        
        # Compute |A|^2
        mag2 = np.abs(A) ** 2
        
        # Target flat level
        flat_level = (S ** 2) / (2 * n - 1)  # Total energy / number of lags
        
        # Blend current and flat
        mag2_new = (mag2 + lam * flat_level) / (1 + lam)
        
        # Update magnitude, preserve phase
        phase = np.angle(A)
        mag_new = np.sqrt(mag2_new)
        A_new = mag_new * np.exp(1j * phase)

        a_new = np.real(ifft(A_new, fsize))[:n]

        a_new = np.maximum(a_new, 0)

            a_new = np.ones(n) * S / n

        c = fft_convolve_self(a_new)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_weighted(a, w):
    """
    Compute gradient of sum_k w[k]*c[k] w.r.t. a, where c = a*a (convolution).
    grad[m] = 2 * sum_k w[k] * a[k-m]
    Uses FFT for efficiency.
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, S):
    """Clamp to non-negative and rescale to sum = S."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        n = len(a)
        return np.ones(n) * S / n
    return a * (S / s)


def spectral_flattening(a, n_iters=500, rng=None):
    """
    POCS-type spectral flattening to reduce autocorrelation peak.
    Iteratively: (1) flatten |A|^2, (2) project to a >= 0, (3) renormalize.
    """

        rng = np.random.default_rng(0)
    
    a = a.copy().astype(np.float64)
    S = a.sum()
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    
    # Adaptive lambda schedule: start aggressive, then gentle
    for it in range(n_iters):
        # Lambda decreases over time (less aggressive flattening)
        lam = 0.5 * (1.0 + 0.5 * np.cos(np.pi * it / max(n_iters, 1)))
        
        A = fft(a, fsize)
        mag2 = np.abs(A) ** 2
        phase = np.angle(A)
        
        # Target: flat spectrum at level S^2 / fsize
        flat_level = S * S / fsize
        
        # Blend: mag2_new = (mag2 + lam * flat_level) / (1 + lam)
        mag2_new = (mag2 + lam * flat_level) / (1.0 + lam)
        
        # Update magnitude, preserve phase
        mag_new = np.sqrt(mag2_new)
        A_new = mag_new * np.exp(1j * phase)
        
        # Inverse FFT, project to non-negative
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        
        # Renormalize
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) * S / n
        else:
            a_new = a_new * (S / s)
        
        # Check score (every 10 iterations to save time)
        if it % 10 == 0 or it == n_iters - 1:
            c = fft_convolve_self(a_new)
            score = c.max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score


def fista_toppk(a_init, S, max_time, rng, top_k_init=5, top_k_max=100):
    """

    """

        it += 1

            best_local = y.copy()

        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            total = 1.0

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]
        
        grad = compute_grad_weighted(y, w_full)

        step = 2.0 * S / (n * np.sqrt(1.0 + 0.01 * it))

        y = project_simplex(y, S)
        
        a = x_new
        t = t_new

            y = project_simplex(y + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score

        stuck = 0

                c = fft_convolve_self(a)
                current_score = c.max()
    
    return best_a, best_score

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_weighted(a, w):
    """Compute gradient of sum_k w[k]*c[k] w.r.t. a where c=a*a."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad
