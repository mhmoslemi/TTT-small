from scipy.fft import fft, ifft

def fft_convolve_self(a):

    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:2*n-1]


def log_sum_exp(values, T):
    """Compute T * log(sum(exp(values/T))) safely."""
    v = values / T
    vmax = v.max()
    return T * (vmax + np.log(np.sum(np.exp(v - vmax))))

    a = np.exp(x_shifted)

    c = fft_convolve_self(a)
    obj = log_sum_exp(c, T)
    return obj

        phase_time = time_budget / 20.0 - (time.monotonic() - start_time)
        iter_budget = max(1, int(min(n_iters_max, phase_time * 100)))
        
        try:
            result = minimize(
                func, x, jac=jac, method='L-BFGS-B',
                options={'maxiter': iter_budget, 'maxfun': iter_budget * 2,

            x = result.x

def subgrad_optimize(a0, time_budget, eta0=0.5, momentum=0.9):
    """Subgradient method with peak chasing (improved version)."""
    n = len(a0)
    S = a0.sum()
    a = a0.copy().astype(np.float64)
    
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    
    v = np.zeros(n)
    eta = eta0
    start_time = time.monotonic()
    iter_no = 0
    best_iter = 0
    
    while time.monotonic() - start_time < time_budget:
        # Find active peaks (top by value)
        c_max = c.max()
        active_mask = c >= c_max * 0.99
        n_active = active_mask.sum()
        
        # Subgradient: weighted average of gradients at active peaks
        # grad[i] = sum_k w_k * a[k-i] for valid (k,i)
        a_rev = a[::-1]
        fsize = 1 << (len(c) + n - 2).bit_length()
        
        if n_active <= 10:
            # Direct computation
            w = np.full(n_active, 1.0 / n_active)
            active_idx = np.where(active_mask)[0]
            W = np.zeros(len(c))
            W[active_idx] = w
            Wf = fft(W, fsize)
            Rf = fft(a_rev, fsize)
            full_conv = np.real(ifft(Wf * Rf))
            grad = 2.0 * full_conv[n-1:2*n-1]
        else:
            # Use the top few
            top_k = min(50, len(c))
            top_idx = np.argsort(c)[-top_k:]
            W = np.zeros(len(c))
            W[top_idx] = 1.0 / top_k
            Wf = fft(W, fsize)
            Rf = fft(a_rev, fsize)
            full_conv = np.real(ifft(Wf * Rf))
            grad = 2.0 * full_conv[n-1:2*n-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Momentum
        v = momentum * v + (1 - momentum) * grad
        
        # Multiplicative update (mirror descent)
        a = a * np.exp(-eta * v)
        a = np.maximum(a, 1e-30)
        a = a * (S / a.sum())
        
        c = fft_convolve_self(a)
        cur_obj = 2.0 * n * c.max() / (S * S)
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
            eta = min(eta * 1.01, eta0 * 3)
        else:
            eta = max(eta * 0.97, eta0 * 0.1)
        
        # Stagnation: perturb
        if iter_no - best_iter > 2000:
            noise = np.random.randn(n) * 0.05 * S / np.sqrt(n)
            a = a + noise
            a = np.maximum(a, 1e-10)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.5
            eta = eta0
            best_iter = iter_no
        
        eta *= 0.99999
        iter_no += 1
    
    return best_a, best_obj


def coordinate_polish(a0, time_budget):
    """Local polish: try small perturbations around the current solution."""
    n = len(a0)
    S = a0.sum()
    a = a0.copy().astype(np.float64)
    
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    
    start_time = time.monotonic()
    
    # Identify the peak location
    peak_idx = np.argmax(c)
    
    # Try local modifications around the peak's contributing elements
    while time.monotonic() - start_time < time_budget:
        c = fft_convolve_self(a)
        peak_idx = np.argmax(c)
        c_max = c[peak_idx]
        
        # Find contributing elements
        # conv[k] = sum_i a[i]*a[k-i]
        # For peak at k, contributing pairs are (i, k-i) for valid i
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        
        improved = False
        for i in range(lo, hi + 1):
            j = peak_idx - i
            if j < 0 or j >= n:
                continue
            if a[i] < 1e-10 or a[j] < 1e-10:
                continue
            
            # Try reducing the product a[i]*a[j] by shifting mass
            # Option 1: reduce a[i], add to a[i±1]
            for delta in [0.5, 1.0, 2.0]:
                orig_i = a[i]
                orig_neighbor = a[i+1] if i+1 < n else a[i-1]
                
                # Move some mass from a[i] to a neighbor
                shift = delta * orig_i / n
                a[i] -= shift
                if i+1 < n:
                    a[i+1] += shift
                else:
                    a[i-1] += shift
                
                a = np.maximum(a, 0)
                a = a * (S / a.sum())
                c_new = fft_convolve_self(a)
                obj_new = 2.0 * n * c_new.max() / (S * S)
                
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    # Revert
                    a[i] += shift
                    if i+1 < n:
                        a[i+1] -= shift
                    else:
                        a[i-1] -= shift
                    a = np.maximum(a, 0)
                    a = a * (S / a.sum())
            
            if improved:
                break
        
        if not improved:
            # Try random local perturbation
            idx = np.random.randint(0, n)
            shift = 0.01 * S / n
            target = np.random.randint(0, n)
            a[idx] += shift
            a[target] -= shift
            a = np.maximum(a, 0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            obj = 2.0 * n * c.max() / (S * S)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using multi-phase optimization:
1. Diversified initialization with structured candidates
2. L-BFGS with log-sum-exp smooth approximation (annealing temperature)
3. Subgradient peak-chasing with adaptive active set
4. Block coordinate descent polish
Key insight: np.convolve(a,a) is NOT autocorrelation; for non-negative a,
the peak can be anywhere in the 2n-1 length convolution, not just at lag 0.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def obj_from_conv(c, n, S):
    """Compute objective from precomputed convolution."""
    return 2.0 * n * c.max() / (S * S)


def softmax_from_x(x, S):
    """Convert unconstrained x to a non-negative vector with sum S."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.full(len(x), S / len(x))
    return a / s * S


def lse_obj_and_grad(x, T, S):
    """Compute LSE objective and gradient w.r.t. x (softmax parameterization)."""
    a = softmax_from_x(x, S)
    n = len(a)

    c = fft_convolve_self(a)

    # LSE: T * log(sum(exp(c/T)))
    c_scaled = c / T
    c_max = c_max_val = c_scaled.max()
    exp_sum = np.sum(np.exp(c_scaled - c_max_val))
    obj = T * (c_max_val + np.log(exp_sum))

    # Softmax weights p_k = exp(c[k]/T) / sum(exp(c[j]/T))
    p = np.exp(c_scaled - c_max_val) / exp_sum

    # Gradient of c[k] w.r.t. a[i]: d(c[k])/da[i] = a[k-i] (for valid range)
    # d(LSE)/da[i] = sum_k p[k] * a[k-i] = conv(p, a_rev)[n-1+i]
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 2).bit_length()
    P = fft(p, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(P * R))
    grad_a = full_conv[n - 1:2 * n - 1]

    # Chain rule through softmax:
    # da[i]/dx[j] = a[i]*(delta[i,j] - a[j]/S)
    # d_obj/dx[j] = a[j]*grad_a[j] - (a[j]/S)*sum_i(grad_a[i]*a[i])
    dot = np.dot(grad_a, a)
    grad_x = a * grad_a - a * (dot / S)

    return obj, grad_x


def lse_optimize(a0, time_budget, T_start=None, T_end=None, verbose=False):
    """
    L-BFGS optimization with log-sum-exp smoothing and temperature annealing.
    """
    n = len(a0)
    S = a0.sum()

    # Initialize x
    eps = 1e-10
    x = np.log(np.maximum(a0, eps))

    # Auto-set temperature range
    c0 = fft_convolve_self(a0)
    c_peak = c0.max()
    if T_start is None:
        T_start = c_peak * 0.1
    if T_end is None:
        T_end = c_peak * 0.001

    best_a = a0.copy()
    best_obj = obj_from_conv(c0, n, S)

    start_time = time.monotonic()
    n_phases = 15

    for phase in range(n_phases):
        elapsed = time.monotonic() - start_time
        remaining = time_budget - elapsed
        if remaining < 2:
            break

        # Anneal temperature
        frac = phase / max(1, n_phases - 1)
        T = T_start * (T_end / T_start) ** frac
        T = max(T, 1e-8)

        phase_budget = remaining / (n_phases - phase)

        def func(x_):
            obj, _ = lse_obj_and_grad(x_, T, S)
            return obj

        def jac(x_):
            _, g = lse_obj_and_grad(x_, T, S)
            return g

        iter_budget = max(10, int(min(2000, phase_budget * 50)))

        try:
            result = minimize(
                func, x, jac=jac, method='L-BFGS-B',
                options={'maxiter': iter_budget, 'maxfun': iter_budget * 3,
                         'ftol': 1e-14, 'gtol': 1e-10, 'disp': False}
            )
            x = result.x

            # Evaluate actual objective
            a_new = softmax_from_x(x, S)
            c_new = fft_convolve_self(a_new)
            obj_new = obj_from_conv(c_new, n, S)

            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

            if verbose:
                print(f"    LSE phase {phase+1}/{n_phases}: T={T:.4f}, obj={obj_new:.6f}, best={best_obj:.6f}")

        except Exception as e:
            if verbose:
                print(f"    LSE phase {phase+1} error: {e}")
            break

    return best_a, best_obj


def subgrad_optimize(a0, time_budget, eta0=0.5, momentum=0.88, top_k=20, verbose=False):
    """
    Subgradient/mirror descent with adaptive top-k peak chasing.
    """
    n = len(a0)
    S = a0.sum()
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = obj_from_conv(c, n, S)
    best_a = a.copy()

    v = np.zeros(n)
    eta = eta0
    start_time = time.monotonic()
    iter_no = 0
    best_iter = 0
    stagnation_limit = 1500

    while time.monotonic() - start_time < time_budget:
        # Find top-k peaks
        c_max = c.max()
        if c_max < 1e-15:
            break

        k = min(top_k, len(c))
        top_idx = np.argpartition(c, -k)[-k:]
        # Weight by proximity to max
        c_top = c[top_idx]
        w = np.exp(5.0 * (c_top - c_max))
        w /= w.sum()

        # Subgradient
        W_full = np.zeros(len(c))
        W_full[top_idx] = w
        a_rev = a[::-1]
        fsize = 1 << (len(c) + n - 2).bit_length()
        Wf = fft(W_full, fsize)
        Rf = fft(a_rev, fsize)
        full_conv = np.real(ifft(Wf * Rf))
        grad = 2.0 * full_conv[n - 1:2 * n - 1]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        # Mirror descent (multiplicative) update with backtracking
        accepted = False
        for bt in range(5):
            eta_try = eta * (0.7 ** bt)
            a_trial = a * np.exp(-eta_try * v)
            a_trial = np.maximum(a_trial, 1e-30)
            a_trial = a_trial * (S / a_trial.sum())

            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_from_conv(c_trial, n, S)

            if obj_trial <= obj_from_conv(c, n, S) * (1.0 + 1e-10):
                a = a_trial
                c = c_trial
                accepted = True
                break

        if not accepted:
            # Accept with tiny step
            a = a * np.exp(-eta * 0.05 * v)
            a = np.maximum(a, 1e-30)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)

        cur_obj = obj_from_conv(c, n, S)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
            eta = min(eta * 1.02, eta0 * 3.0)
        else:
            eta = max(eta * 0.96, eta0 * 0.05)

        # Stagnation detection
        if iter_no - best_iter > stagnation_limit:
            noise_scale = 0.03 * S / np.sqrt(n)
            noise = np.random.randn(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 1e-10)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.3
            eta = eta0
            best_iter = iter_no
            if verbose:
                print(f"    [perturb] iter={iter_no}, cur={cur_obj:.6f}, best={best_obj:.6f}")

        eta *= 0.99998
        iter_no += 1

        if verbose and iter_no % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    subgrad iter={iter_no}, cur={cur_obj:.6f}, best={best_obj:.6f}, eta={eta:.4f}, t={elapsed:.1f}s")

    return best_a, best_obj


def block_polish(a0, time_budget, block_size=128, verbose=False):
    """
    Block coordinate descent: optimize blocks of coordinates using L-BFGS.
    """
    n = len(a0)
    S = a0.sum()
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = obj_from_conv(c, n, S)
    best_a = a.copy()

    n_blocks = n // block_size
    if n_blocks < 1:
        return best_a, best_obj

    start_time = time.monotonic()
    sweep = 0

    while time.monotonic() - start_time < time_budget:
        improved_this_sweep = False

        for b in range(n_blocks):
            if time.monotonic() - start_time > time_budget:
                break

            lo = b * block_size
            hi = lo + block_size
            if hi > n:
                hi = n

            # Save current block
            orig_block = a[lo:hi].copy()

            # Define sub-objective: optimize block only
            # Use LSE with small T for local optimization
            c_peak = c.max()
            T_block = max(c_peak * 0.01, 1e-6)

            # Create local problem: fix all except block
            a_fix = a.copy()

            def block_func(x_b):
                a_trial = a_fix.copy()
                a_trial[lo:hi] = np.exp(x_b - x_b.max())
                a_trial[lo:hi] = np.maximum(a_trial[lo:hi], 1e-30)
                # Renormalize to keep sum = S
                a_trial = a_trial * (S / a_trial.sum())
                c_trial = fft_convolve_self(a_trial)
                # LSE objective
                cs = c_trial / T_block
                return T_block * (cs.max() + np.log(np.sum(np.exp(cs - cs.max()))))

            # Initialize x_b from current block
            x_b_init = np.log(np.maximum(orig_block, 1e-10))

            try:
                res = minimize(
                    block_func, x_b_init, method='L-BFGS-B',
                    options={'maxiter': 50, 'ftol': 1e-14, 'gtol': 1e-8}
                )
                new_block = np.exp(res.x - res.x.max())
                new_block = np.maximum(new_block, 1e-30)

                # Check if it improves
                a_trial = a.copy()
                a_trial[lo:hi] = new_block
                a_trial = a_trial * (S / a_trial.sum())
                c_trial = fft_convolve_self(a_trial)
                obj_trial = obj_from_conv(c_trial, n, S)

                if obj_trial < best_obj:
                    a = a_trial.copy()
                    c = c_trial
                    best_obj = obj_trial
                    best_a = a.copy()
                    improved_this_sweep = True
            except Exception:
                pass

        sweep += 1
        if not improved_this_sweep and sweep > 2:
            break

    return best_a, best_obj


def random_restart_polish(a0, time_budget, n_restarts=5, verbose=False):
    """
    Try small random perturbations and keep improvements.
    """
    n = len(a0)
    S = a0.sum()
    a = a0.copy().astype(np.float64)
    c = fft_convolve_self(a)
    best_obj = obj_from_conv(c, n, S)
    best_a = a.copy()

    start_time = time.monotonic()
    per_restart = time_budget / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() - start_time > time_budget:
            break

        # Generate perturbation
        noise_scale = 0.01 * S / np.sqrt(n)
        noise = np.random.randn(n) * noise_scale
        a_pert = a + noise
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert * (S / a_pert.sum())

        # Quick subgradient refinement from perturbed start
        c_pert = fft_convolve_self(a_pert)
        obj_pert = obj_from_conv(c_pert, n, S)

        if obj_pert < best_obj:
            # Keep and refine
            a_best, obj_best = subgrad_optimize(a_pert, per_restart, eta0=0.2, momentum=0.9, top_k=5)
            if obj_best < best_obj:
                best_obj = obj_best
                best_a = a_best.copy()
                if verbose:
                    print(f"    restart {r+1}: improved to {obj_best:.6f}")

    return best_a, best_obj


def propose_candidate(budget_s=2400):
    """
    Multi-phase search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    """
    rng = np.random.default_rng(99999)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    global_best = np.inf
    global_a = None

    n = 4096
    S_target = np.sqrt(2.0 * n)

    # --- Load given sequence ---
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw.astype(np.float64).copy()
            if len(given_a) != n:
                # Resample to length n
                old_idx = np.linspace(0, len(given_a) - 1, n)
                idx_int = old_idx.astype(int)
                idx_frac = old_idx - idx_int
                given_a = given_a[idx_int] * (1 - idx_frac) + given_a[np.minimum(idx_int + 1, len(given_a) - 1)] * idx_frac
            given_a = given_a * (S_target / given_a.sum())
            c = fft_convolve_self(given_a)
            obj = obj_from_conv(c, n, S_target)
            print(f"Given sequence: n={n}, obj={obj:.6f}")
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = given_a.copy()
    except Exception as e:
        print(f"Could not load given sequence: {e}")

    # --- Generate diverse candidates ---
    candidates = []
    i = np.arange(n, dtype=np.float64)

    if given_a is not None:
        candidates.append(('given', given_a.copy()))

    # U-shaped (high at edges)
    a_u = 1.0 / (1.0 + 30.0 * np.sin(np.pi * i / n) ** 2)
    a_u = np.maximum(a_u, 0.005)
    a_u *= S_target / a_u.sum()
    candidates.append(('u_shape', a_u))

    # Various cosine modulations
    for freq, amp in [(3, 0.6), (5, 0.5), (7, 0.7), (11, 0.4), (13, 0.5)]:
        a_c = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n)
        a_c = np.maximum(a_c, 0.005)
        a_c *= S_target / a_c.sum()
        candidates.append((f'cos_{freq}', a_c))

    # Gaussian at edges (bimodal)
    a_bimod = (np.exp(-0.5 * (i / (0.15 * n)) ** 2) +
               np.exp(-0.5 * ((i - n) / (0.15 * n)) ** 2))
    a_bimod = np.maximum(a_bimod, 0.005)
    a_bimod *= S_target / a_bimod.sum()
    candidates.append(('bimodal', a_bimod))

    # Power law
    a_pow = 1.0 / (1.0 + i / (n * 0.3))
    a_pow += 1.0 / (1.0 + (n - 1 - i) / (n * 0.3))
    a_pow = np.maximum(a_pow, 0.005)
    a_pow *= S_target / a_pow.sum()
    candidates.append(('power', a_pow))

    # Random
    a_rand = rng.random(n)
    a_rand = np.maximum(a_rand, 0.01)
    a_rand *= S_target / a_rand.sum()
    candidates.append(('random', a_rand))

    # Evaluate all candidates
    print(f"\nPhase 0: Evaluating {len(candidates)} candidates")
    cand_results = []
    for name, a_cand in candidates:
        c = fft_convolve_self(a_cand)
        obj = obj_from_conv(c, n, S_target)
        cand_results.append((obj, name, a_cand))
        print(f"  {name}: obj={obj:.6f}")
        if obj < global_best:
            global_best = obj
            global_a = a_cand.copy()

    cand_results.sort(key=lambda x: x[0])

    # --- Phase 1: L-BFGS on top candidates ---
    remaining = deadline - time.monotonic()
    phase1_time = 0.30 * remaining
    print(f"\nPhase 1: L-BFGS optimization ({phase1_time:.0f}s)")

    top_k_cands = min(3, len(cand_results))
    per_cand_time = phase1_time / top_k_cands

    for idx in range(top_k_cands):
        if time.monotonic() > deadline - 60:
            break
        obj_init, name, a_cand = cand_results[idx]
        print(f"  L-BFGS on '{name}' (init={obj_init:.6f}, budget={per_cand_time:.0f}s)")
        a_best, obj_best = lse_optimize(
            a_cand, per_cand_time,
            T_start=obj_init * S_target * S_target / (2.0 * n) * 0.1,
            T_end=obj_init * S_target * S_target / (2.0 * n) * 0.002,
            verbose=True
        )
        print(f"    -> obj={obj_best:.6f}")
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

    # --- Phase 2: Subgradient peak chasing ---
    remaining = deadline - time.monotonic()
    phase2_time = 0.30 * remaining
    print(f"\nPhase 2: Subgradient peak chasing ({phase2_time:.0f}s)")

    if global_a is not None and phase2_time > 10:
        # Split into two runs with different parameters
        half = phase2_time / 2
        a_best, obj_best = subgrad_optimize(global_a, half, eta0=0.4, momentum=0.85, top_k=30, verbose=True)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
        a_best, obj_best = subgrad_optimize(global_a, half, eta0=0.2, momentum=0.92, top_k=5, verbose=True)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

    # --- Phase 3: L-BFGS refinement ---
    remaining = deadline - time.monotonic()
    phase3_time = 0.20 * remaining
    print(f"\nPhase 3: L-BFGS refinement ({phase3_time:.0f}s)")

    if global_a is not None and phase3_time > 10:
        c_cur = fft_convolve_self(global_a)
        c_peak = c_cur.max()
        a_best, obj_best = lse_optimize(
            global_a, phase3_time,
            T_start=c_peak * 0.05,
            T_end=c_peak * 0.001,
            verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

    # --- Phase 4: Block polish + random restarts ---
    remaining = deadline - time.monotonic()
    phase4_time = 0.15 * remaining
    print(f"\nPhase 4: Block polish + restarts ({phase4_time:.0f}s)")

    if global_a is not None and phase4_time > 5:
        # Block coordinate descent
        block_time = phase4_time * 0.5
        print(f"  Block polish ({block_time:.0f}s)")
        a_best, obj_best = block_polish(global_a, block_time, block_size=128, verbose=True)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

        # Random restarts
        restart_time = phase4_time * 0.5
        if restart_time > 5:
            print(f"  Random restarts ({restart_time:.0f}s)")
            a_best, obj_best = random_restart_polish(global_a, restart_time, n_restarts=4, verbose=True)
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()

    # --- Final: quick subgradient polish ---
    remaining = deadline - time.monotonic() - 10.0
    if global_a is not None and remaining > 5:
        print(f"\nFinal polish ({remaining:.0f}s)")
        a_best, obj_best = subgrad_optimize(global_a, remaining, eta0=0.1, momentum=0.95, top_k=3, verbose=True)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(n) * (S_target / n)

    global_a = np.clip(global_a.astype(np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(n)

    # Verify with evaluate_sequence
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: local_best={global_best:.6f}, verified={ver:.6f}")

    if not np.isfinite(ver):
        print("WARNING: verified value is not finite, using fallback")
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.6f}")

    return final_a
