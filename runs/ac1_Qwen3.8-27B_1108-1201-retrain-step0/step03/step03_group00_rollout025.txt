"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: LP-based cutting plane optimization (finding the optimal direction
to reduce max autocorrelation) combined with FISTA LSE polishing.
Starts from the previous best sequence and iteratively improves.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, active_lags, n):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0.
    Returns the optimal g vector, or None if infeasible.
    """
    K = len(active_lags)
    if K == 0:
        return None
    
    # Build sparse constraint matrix A_ub (K x n)
    # A_ub[i, j] = f[active_lags[i] - j] if 0 <= active_lags[i] - j < n
    rows = []
    cols = []
    vals = []
    
    for i, k in enumerate(active_lags):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            ks = k - js  # f[k-j] for each j
            rows.extend([i] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[ks].tolist())
    
    if len(vals) == 0:
        return None
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(K, n))
    b_ub = np.full(K, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                     method='highs', options={'presolve': True, 'time_limit': 30})
    
    if result.status != 0:
        return None
    return result.x

    M = c.max()
    best = M
    best_a = f.copy()
    
    # Initial active set: top 20 lags
    active_lags = np.argsort(c)[::-1][:20].tolist()

        iteration += 1
        
        # Solve LP
        g_raw = solve_lp_direction(f, M, active_lags, n)

def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_rate=0.005, beta_max=1000.0):
    """FISTA with LSE smoothing."""
    a = project_simplex(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    c = fft_convolve_self(y)
    best = c.max()
    best_a = y.copy()
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)
        
        # Adaptive: if stuck, inject noise
        if it - last_improve > 2000:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return best_a, best
