Let me think about the frequency domain perturbation more carefully.

The idea: given a current sequence a, we can modify its "effective spectrum" to reduce peaks in the convolution.

Step 1: Compute A = fft(a_padded) where a_padded is a zero-padded to length 2n-1 (or a power of 2 >= 2n-1).
Step 2: The periodogram is |A|^2.
Step 3: Modify |A|^2 to be flatter: replace the top p% of |A|^2 values with their average (or apply a soft cap).
Step 4: We need to find a non-negative sequence a' such that |fft(a'_padded)|^2 ≈ modified |A|^2 and sum(a') = 1.

Step 4 is a phase retrieval problem. We can solve it with IFTA (Iterative Fourier Transform Algorithm):
- Initialize a' (e.g., the current a).
- Iterate:
  - Compute B = fft(a'_padded).
  - Set |B| = sqrt(modified |A|^2) (keep the phase of B).
  - a'_time = real(ifft(B)).
  - Enforce a' >= 0 (set negative values to 0) and sum(a') = 1.
- Repeat until convergence or max iterations.

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using improved mirror descent.
Key improvements over baseline:
- Full softmax subgradient (all peaks weighted by exp(beta*c[k]))
- Adaptive step size (Plackett-Hunter style)
- Nesterov momentum in log-space
- Frequency-domain perturbation for escaping local minima
- Multi-restart strategy with different initializations
- Cosine annealing for temperature
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = fft(a, L)
    c = np.real(ifft(A * A))
    return c[:size]


def obj(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_conv_self(a)
    return 2.0 * n * c.max() / (S * S)


def softmax_subgrad(c, a, beta):
    """
    Compute subgradient of max c[k] approximated by softmax-weighted average.
    grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
             = 2 * cross_corr(a, w)[j]
    """
    n = len(a)
    L = 1 << (3 * n - 2 - 1).bit_length() if 3*n-2 > 1 else 4
    # Actually need L >= (2n-1) + n - 1 = 3n-2 for linear cross-correlation
    L = 1 << max(0, (3 * n - 2 - 1).bit_length())
    
    c_max = c.max()
    w = np.exp(beta * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    W = fft(w, L)
    A = fft(a, L)
    # Cross-correlation: ifft(W * conj(A)) gives sum_k w[k]*a[k-j] at lag j
    cross = np.real(ifft(W * np.conj(A)))
    grad = 2.0 * cross[:n]
    return grad


def mirror_descent_v2(a0, time_budget, 
                       beta_start=3.0, beta_end=50.0,
                       eta_start=0.5,
                       momentum=0.85,
                       stagnation_limit=200,
                       verbose=True, seed=42):
    """
    Improved mirror descent with full softmax subgradient and adaptive step size.
    """
    rng = np.random.default_rng(seed)
    a = a0.copy().astype(np.float64)
    n = len(a)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S  # normalize to sum=1 (objective is scale-invariant)
    
    c = fft_conv_self(a)
    best_obj = obj(a, c)
    best_a = a.copy()
    best_iter = 0
    
    v = np.zeros(n)  # momentum
    eta = eta_start
    
    iter_no = 0
    start = time.monotonic()
    total_iters_planned = time_budget * 500  # estimate ~500 iters/sec
    
    while time.monotonic() - start < time_budget:
        # Animate beta (cosine schedule)
        progress = min(1.0, iter_no / max(1, int(total_iters_planned * 0.8)))
        beta = beta_start + (beta_end - beta_start) * (1 - np.cos(np.pi * progress)) / 2
        
        # Compute subgradient
        grad = softmax_subgrad(c, a, beta)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm
        
        # Multiplicative update
        a_trial = a * np.exp(-eta * v)
        a_trial = np.maximum(a_trial, 1e-300)
        a_trial = a_trial / a_trial.sum()
        
        c_trial = fft_conv_self(a_trial)
        obj_trial = obj(a_trial, c_trial)
        
        # Adaptive step size
        if obj_trial < best_obj * (1 + 1e-9):
            a = a_trial
            c = c_trial
            cur_obj = obj_trial
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                best_iter = iter_no
            eta = min(eta * 1.05, 3.0)
        else:
            # Reject step, reduce eta
            eta = max(eta * 0.8, 0.01)
            cur_obj = best_obj  # stay where we are
        
        # Stagnation
        if iter_no - best_iter > stagnation_limit:
            # Frequency domain perturbation
            L_pert = 1 << (2 * n - 1 - 1).bit_length()
            A = fft(a, L_pert)
            phase = np.angle(A)
            mag = np.abs(A)
            noise = 0.1 * rng.standard_normal(L_pert) * mag
            A_new = (mag + noise) * np.exp(1j * phase)
            a_pert = np.real(ifft(A_new))[:n]
            a_pert = np.maximum(a_pert, 0.0)
            s = a_pert.sum()
            if s > 1e-10:
                a = a_pert / s
            else:
                a = best_a.copy()
            c = fft_conv_self(a)
            v *= 0.1
            eta = eta_start
            best_iter = iter_no
            if verbose:
                print(f"    [perturb] iter={iter_no}, best={best_obj:.6f}, t={time.monotonic()-start:.1f}s")
        
        iter_no += 1
        if verbose and iter_no % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"eta={eta:.3f}, beta={beta:.1f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def coordinate_polish(a0, time_budget, verbose=True):

    Coordinate descent: for each coordinate, try to reduce the max peak.
    """

    S = a.sum()

    c = fft_conv_self(a)
    best_obj = obj(a, c)
    best_a = a.copy()
    start = time.monotonic()
    sweep_count = 0
    
    while time.monotonic() - start < time_budget:
        sweep_count += 1
        improved = False
        # Find the worst peak
        k_max = np.argmax(c)
        
        # Process coordinates in order of their contribution to the worst peak
        # Coordinates that contribute most to c[k_max] are those where a[k_max - j] is large
        # grad[j] for peak k_max: 2*a[k_max-j]
        order = np.argsort(-np.abs(np.roll(a, k_max)))[:min(n, 500)]
        # Actually, the relevant indices are j where k_max-j is in [0, n-1]
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        relevant = np.arange(lo, hi + 1)
        if len(relevant) > 200:
            # Pick the ones with largest a values
            vals = a[relevant]
            top_idx = np.argsort(-vals)[:200]
            relevant = relevant[top_idx]
        
        for j in relevant:
            if time.monotonic() - start > time_budget:
                break
            orig = a[j]
            if orig < 1e-10:
                continue
            # Try reducing a[j] (this reduces c[k_max] since a[j] contributes to it)
            for factor in [0.5, 0.7, 0.85]:
                a[j] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[j] = orig
                    break
                a_norm = a / s
                c_new = fft_conv_self(a_norm)
                obj_new = obj(a_norm, c_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_norm.copy()
                    a = a_norm
                    c = c_new
                    improved = True
                    break
            a[j] = a[j]  # (already updated or restored)
            # Renormalize
            a = a / a.sum()
            c = fft_conv_self(a)
        
        if not improved:
            # Try increasing some coordinates (to spread mass away from peaks)
            # Increase coordinates that DON'T contribute to the peak
            for j in range(n):
                if time.monotonic() - start > time_budget:
                    break
                # Check if j contributes to k_max
                if lo <= j <= hi:
                    continue  # skip, this contributes to the peak
                orig = a[j]
                for factor in [1.5, 2.0, 3.0]:
                    a[j] = orig * factor
                    s = a.sum()
                    a_norm = a / s
                    c_new = fft_conv_self(a_norm)
                    obj_new = obj(a_norm, c_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_norm.copy()
                        a = a_norm
                        c = c_new
                        improved = True
                        break
                a[j] = a[j]
                a = a / a.sum()
                c = fft_conv_self(a)
        
        if not improved:
            break
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using improved mirror descent with:
- Full softmax subgradient over all convolution peaks (no top-k truncation)
- Adaptive Plackett-Hunter step size control
- Nesterov momentum in log-space
- Cosine-annealed temperature schedule
- Frequency-domain perturbation for escaping local minima
- Multi-phase strategy: explore -> exploit -> polish
- Incremental convolution updates for coordinate descent polish
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = fft(a, L)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_fn(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_conv_self(a)
    return float(2.0 * n * c.max() / (S * S))


def softmax_subgrad(c, a, beta):
    """
    Compute subgradient using full softmax weighting over all 2n-1 convolution peaks.
    grad[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation of a and w)
    """
    n = len(a)
    # Need L >= 3n-2 for linear cross-correlation of w (len 2n-1) and a (len n)
    L = 1 << (3 * n - 3).bit_length()

    c_max = c.max()
    scaled = beta * (c - c_max)
    # Numerical stability: clip to avoid underflow
    scaled = np.maximum(scaled, -700.0)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    W = fft(w, L)
    A = fft(a, L)
    # Cross-correlation: r[j] = sum_k w[k] * a[k-j-j] at lag j
    # ifft(W * conj(A))[j] = sum_k w[k] * a[(k-j) mod L]
    cross = np.real(ifft(W * np.conj(A)))
    grad = 2.0 * cross[:n]
    return grad


def mirror_descent_v2(a0, time_budget,
                      beta_start=3.0, beta_end=50.0,
                      eta_start=0.4, eta_min=0.005, eta_max=2.5,
                      momentum=0.85,
                      stagnation_limit=250,
                      verbose=True, seed=42):
    """
    Mirror descent with full softmax subgradient, adaptive step size,
    Nesterov momentum, and frequency-domain perturbation on stagnation.
    """
    rng = np.random.default_rng(seed)
    a = a0.copy().astype(np.float64)
    n = len(a)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S  # normalize to sum=1

    c = fft_conv_self(a)
    best_obj = obj_fn(a, c)
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)
    eta = eta_start
    iter_no = 0
    start = time.monotonic()

    # Estimated total iterations for schedule
    est_iters_per_sec = 400
    total_est = int(time_budget * est_iters_per_sec)

    while time.monotonic() - start < time_budget:
        # Cosine annealing for beta
        progress = min(1.0, iter_no / max(1, int(total_est * 0.85)))
        beta = beta_start + (beta_end - beta_start) * (1.0 - np.cos(np.pi * progress)) / 2.0

        # Compute subgradient
        grad = softmax_subgrad(c, a, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        # Multiplicative (mirror) update in log-space
        a_trial = a * np.exp(-eta * v)
        a_trial = np.maximum(a_trial, 1e-300)
        a_trial_s = a_trial.sum()
        if a_trial_s < 1e-10:
            eta = max(eta * 0.5, eta_min)
            iter_no += 1
            continue
        a_trial = a_trial / a_trial_s

        c_trial = fft_conv_self(a_trial)
        obj_trial = obj_fn(a_trial, c_trial)

        # Adaptive step size (Plackett-Hunter)
        if obj_trial < best_obj + 1e-12:
            a = a_trial
            c = c_trial
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a = a.copy()
                best_iter = iter_no
            eta = min(eta * 1.03, eta_max)
        else:
            eta = max(eta * 0.85, eta_min)

        # Stagnation handling: frequency-domain perturbation
        if iter_no - best_iter > stagnation_limit:
            L_p = 1 << (2 * n - 2).bit_length()
            A = fft(a, L_p)
            mag = np.abs(A)
            phase = np.angle(A)
            # Add relative noise
            noise_frac = 0.15
            noise = noise_frac * rng.standard_normal(L_p) * mag
            A_new = (mag + noise) * np.exp(1j * phase)
            a_pert = np.real(ifft(A_new))[:n]
            a_pert = np.maximum(a_pert, 0.0)
            s_pert = a_pert.sum()
            if s_pert > 1e-10:
                a = a_pert / s_pert
            else:
                # Fallback: random perturbation
                a = a + 0.05 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
            c = fft_conv_self(a)
            v *= 0.1
            eta = eta_start
            best_iter = iter_no
            if verbose:
                print(f"    [perturb] iter={iter_no}, best={best_obj:.6f}, "
                      f"t={time.monotonic() - start:.1f}s")

        iter_no += 1
        if verbose and iter_no % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter={iter_no}, obj={obj_fn(a, c):.6f}, best={best_obj:.6f}, "
                  f"eta={eta:.3f}, beta={beta:.1f}, t={elapsed:.1f}s")

    return best_a, best_obj


def incremental_conv_update(c, a, j, new_val, old_val):
    """
    Update convolution c in-place after changing a[j] from old_val to new_val.
    Uses the formula: dc[k] = 2*(new_val - old_val)*a[k-j] for valid k-j.
    Returns updated c (in-place).
    """
    n = len(a)
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return c
    # c[k] gets 2*delta*a[k-j] added for k in [j, j+n-1]
    k_lo = j
    k_hi = min(2 * n - 2, j + n - 1)
    c[k_lo:k_hi + 1] += 2.0 * delta * a[0:k_hi + 1 - k_lo]
    return c


def coordinate_polish_v2(a0, time_budget, verbose=True):

        if len(relevant) == 0:

        contrib = a[k_max - relevant]
