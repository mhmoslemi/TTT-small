    c = np.real(ifft(A * A))
    return c[:size]


def compute_conv_grad(a, k_peak):
    """
    Compute gradient of c[k_peak] w.r.t. a.
    c[k] = sum_j a[j]*a[k-j], so dc[k]/da[i] = 2*a[k-i] (with a[m]=0 outside [0,n-1]).
    """
    n = len(a)
    grad = np.zeros(n)
    # grad[i] = 2 * a[k_peak - i] if 0 <= k_peak - i <= n-1, else 0
    lo = max(0, k_peak - (n - 1))
    hi = min(n - 1, k_peak)
    if lo <= hi:
        grad[lo:hi+1] = 2.0 * a[k_peak - hi:k_peak - lo + 1][::-1]
    return grad


def compute_smoothed_grad(a, weights, full_conv_len):
    """
    Compute smoothed gradient: 2 * sum_k weights[k] * a[k-i] (via FFT).
    weights is a probability vector over the full convolution indices.
    """
    n = len(a)
    fsize = 1 << (full_conv_len + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    a = S * softmax(z), objective = smoothed_max(conv(a,a))

    obj = (c_max + np.log(sum_exp_bc)) / beta

    return obj, grad_z

    S = np.sqrt(2.0 * float(n))
    tt = np.linspace(0, 1, n, endpoint=False)

    # Add structured initializations
    inits.append(("rand", rng.random(n) + 0.05))
    inits.append(("anti_sym", 1.0 + 0.4 * np.sin(np.pi * tt) + 0.2 * np.sin(2 * np.pi * tt)))
    inits.append(("multi_freq", 1.0 + 0.3 * np.sin(2*np.pi*2*tt) + 0.25*np.sin(2*np.pi*5*tt) +
                  0.2*np.sin(2*np.pi*9*tt) + 0.15*np.sin(2*np.pi*13*tt)))

    n_inits = len(inits)
    print("=== Optimization: n=%d, %d inits, budget=%.0fs ===" % (n, n_inits, budget_s))
    print("Target: 1.5027, Previous best: 1.508805")

    # Evaluate initial sequences
    for name, arr in inits:
        a_norm = safe_normalize(arr, n)
        c = fft_convolve_self(a_norm)
        score = c.max()
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()

    print("Initial best: %.8f" % global_best)

    # Phase 1: Nesterov subgradient on each initialization
    remaining = deadline - time.monotonic()
    per_init_time = remaining * 0.50 / n_inits
    per_init_time = max(per_init_time, 15.0)

    print("\n=== Phase 1: Nesterov Subgradient ===")
    phase1_results = []

    for name, arr in inits:
        if time.monotonic() > deadline - 180:
            break
        t0 = time.monotonic()
        a_norm = safe_normalize(arr, n)
        a_new, score = run_nesterov_direct(a_norm, n, S, per_init_time, rng)
        a_new = safe_normalize(a_new, n)
        c_new = fft_convolve_self(a_new)
        final_score = c_new.max()
        phase1_results.append((a_new, final_score))

        if final_score < global_best:
            global_best = final_score
            global_a = a_new.copy()
            print("  [IMPROVE] %s: %.8f (%.1fs)" % (name, global_best, time.monotonic() - t0))
        else:
            print("  [%s: %.8f, best=%.6f, %.1fs]" % (name, final_score, global_best, time.monotonic() - t0))

    print("\nAfter Phase 1: global_best=%.8f" % global_best)

    # Phase 2: L-BFGS polishing on best result
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            lbgfs_budget = remaining * 0.35
            print("\n=== Phase 2: L-BFGS Polishing (%.0fs) ===" % lbgfs_budget)

            a_new, score = run_lbgfs_log(
                global_a, n, S, lbgfs_budget, rng,
                beta_schedule=[50, 200, 500, 1500, 4000, 10000],
                maxiter=300
            )
            a_new = safe_normalize(a_new, n)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < global_best:
                global_best = c_new.max()
                global_a = a_new.copy()
                print("  L-BFGS: %.8f" % global_best)
            else:
                print("  L-BFGS: %.8f (no improvement)" % c_new.max())

    # Phase 3: Multiplicative direction search
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            mp_budget = remaining * 0.25
            print("\n=== Phase 3: Multiplicative Direction (%.0fs) ===" % mp_budget)

            a_new, score = run_multiplicative_direction(
                global_a, n, S, mp_budget, rng, n_inner=100
            )
            a_new = safe_normalize(a_new, n)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < global_best:
                global_best = c_new.max()
                global_a = a_new.copy()
                print("  MultDir: %.8f" % global_best)
            else:
                print("  MultDir: %.8f (no improvement)" % c_new.max())

    # Phase 4: Perturbation + short L-BFGS restarts
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("\n=== Phase 4: Perturbation + L-BFGS (%.0fs) ===" % remaining)
            n_pert = max(3, int(remaining / 30))
            pert_budget = remaining * 0.6 / n_pert

            for p in range(n_pert):
                if time.monotonic() > deadline - 20:
                    break
                noise_scale = 0.02 / (1.0 + 0.2 * p)
                a_pert = global_a + rng.normal(0, noise_scale * S / np.sqrt(n), n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = safe_normalize(a_pert, n)

                a_new, score = run_lbgfs_log(
                    a_pert, n, S, pert_budget, rng,
                    beta_schedule=[100, 500, 2000],
                    maxiter=150
                )
                a_new = safe_normalize(a_new, n)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < global_best:
                    global_best = c_new.max()
                    global_a = a_new.copy()
                    print("  Perturb %d: %.8f" % (p, global_best))

    # Phase 5: Final Nesterov polish
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            print("\n=== Phase 5: Final Nesterov Polish (%.0fs) ===" % remaining)
            a_new, score = run_nesterov_direct(global_a, n, S, remaining, rng)
            a_new = safe_normalize(a_new, n)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < global_best:
                global_best = c_new.max()
                global_a = a_new.copy()
                print("  Final: %.8f" % global_best)

    # Verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using uniform fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.8f" % (len(global_a), global_best))
    print("Time used: %.1fs" % (time.monotonic() - (time.monotonic() - (budget_s - 20) + (budget_s - 20))))
    return [float(x) for x in global_a]
