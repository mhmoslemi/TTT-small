    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_gradient_fft(a, weights):
    """Gradient of sum_k weights[k] * (a*a)_k w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def softmax_weights(c, beta):
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total

def toppk_weights(c, top_k):
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights

def solve_lp_best_response(f, M, active_lags, n):
    """
    Solve max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0.
    Returns the optimal g vector.
    """
    K = len(active_lags)
    if K == 0:
        return np.ones(n)
    
    # Build constraint matrix: K x n
    # Row i (for lag k = active_lags[i]): T[i, j] = f[k-j] if 0 <= k-j < n else 0
    rows = []
    cols = []
    vals = []
    
    for i, k in enumerate(active_lags):
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        fs = f[k - js]  # f[k - j] for j in js
        rows.extend([i] * len(js))
        cols.extend(js.tolist())
        vals.extend(fs.tolist())
    
    if len(rows) == 0:
        return np.ones(n)
    
    T = csr_matrix((vals, (rows, cols)), shape=(K, n))
    
    # Maximize sum(g) = minimize -sum(g)
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    
    res = linprog(c_obj, A_ub=T, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        g = res.x
        g = np.maximum(g, 0)
        return g
    else:
        # Fallback: uniform
        return np.ones(n)

def run_lp_cutting_plane(f_init, n, S, max_time, rng, K_init=50, K_max=500):
    """
    LP-based best response with cutting plane.
    Iteratively: solve LP with active lags, check violations, add violated lags.
    """
    f = np.clip(f_init, 0.0, 1000.0).copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) * S / n
    else:
        f = f * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = f.copy()
    
    # Initial active lags: the top K by convolution value
    c = fft_convolve_self(f)
    M = c.max()
    active_lags = np.argpartition(c, -K_init)[-K_init:]
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        # Solve LP
        g = solve_lp_best_response(f, M, active_lags, n)
        
        # Rescale g to match sum S
        g_sum = g.sum()
        if g_sum < 1e-15:
            g = np.ones(n) * S / n
        else:
            g = g * (S / g_sum)
        
        # Check if g is better
        c_g = fft_convolve_self(g)
        g_score = c_g.max()
        
        # Blend
        t_blend = 0.3
        f_new = (1 - t_blend) * f + t_blend * g
        f_new = np.maximum(f_new, 0)
        f_new_sum = f_new.sum()
        if f_new_sum > 1e-15:
            f_new = f_new * (S / f_new_sum)

            best_local = f_new.copy()
        
        f = f_new
        M = c_new.max()

        # A lag k is "violated" if (f*g upcoming) might exceed M
        # More precisely: check if any lag has convolution value close to M
        top_lags = np.argpartition(c_new, -min(K_max, len(c_new)))[-min(K_max, len(c_new)):]
        active_lags = np.unique(np.concatenate([active_lags, top_lags]))
        active_lags = active_lags[:K_max]  # Cap the number
        
        # Also periodically re-solve with all top lags

            active_lags = np.argpartition(c_new, -K_max)[-K_max:]

            # Try more aggressive blending
            t_blend = 0.5
            active_lags = np.argpartition(c_new, -K_max)[-K_max:]
    
    return best_local, best_score

def run_fista_adaptive(a_init, n, S, max_time, rng, beta_init=30.0, beta_max=3000.0):
    """FISTA with adaptive beta and dual variable tracking."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 1.0 / (beta_init * 2.0)
    stall_counter = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            stall_counter = 0
        else:
            stall_counter += 1
        
        # Use a mix of softmax and top-k for robustness
        if it < 1000:
            w = toppk_weights(c, min(10, it // 100 + 1))
        else:
            w = softmax_weights(c, beta)
        
        grad = compute_gradient_fft(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Adaptive step
        if stall_counter > 300:
            step *= 0.7
            stall_counter = 0
        elif stall_counter == 0:
            step = min(step * 1.02, 1.0 / (beta * 0.5))
        
        # Periodic perturbation
        if it % 2000 == 0 and rng.random() < 0.2:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
    
    return best_local, best_score

def run_coordinate_descent(a_init, n, S, max_time, rng):
    """
    Coordinate descent: update one coordinate at a time to reduce max convolution.
    Focus on the coordinates that most affect the peak lags.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()

    best_score = c.max()
    
    # Focus on the top few lags
    top_k = 5
    step_size = 0.01 * S / n
    
    while time.monotonic() - start < max_time:
        it = 0
        c = fft_convolve_self(a)
        cur = c.max()
        
        # Get the top lag
        peak_lag = np.argmax(c)
        
        # Try random coordinate updates
        for _ in range(1000):
            if time.monotonic() - start > max_time:
                break
            j = rng.integers(0, n)
            # Try increasing and decreasing a[j]
            for delta in [step_size, -step_size, 2*step_size, -2*step_size]:
                a[j] = max(0, a[j] + delta)
                # Renormalize
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = max(0, a[j] - delta)
                    continue
                a_temp = a * (S / s_new)
                c_temp = fft_convolve_self(a_temp)
                if c_temp.max() < cur:
                    a = a_temp
                    cur = c_temp.max()
                    if cur < best_score:
                        best_score = cur
                        best_local = a.copy()
                    step_size = min(step_size * 1.1, 0.1 * S / n)
                    break
                else:
                    a[j] = max(0, a[j] - delta)
            else:
                step_size *= 0.95
        
        if cur < best_score + 1e-10:
            best_score = cur
            best_local = a.copy()
    
    return best_local, best_score

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Combines LP-based best response (cutting plane) with adaptive FISTA,
dual ascent (exponentiated gradient on dual variables), and random restarts.
Key innovations: (1) LP best response for exact equilibrium-seeking,
(2) principled dual variable updates from game theory,
(3) multi-phase hybrid with adaptive time allocation.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Gradient of sum_k weights[k] * (a*a)_k w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Compute normalized top-k weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights


def build_constraint_rows(f, active_lags, n):
    """Build sparse constraint matrix rows for given active lags."""
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(active_lags):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        fs = f[k - js]
        rows.extend([i] * len(js))
        cols.extend(js.tolist())
        vals.extend(fs.tolist())
    return rows, cols, vals


def solve_lp_best_response(f, M, active_lags, n):
    """
    Solve max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0.
    Returns optimal g or None on failure.
    """
    K = len(active_lags)
    if K == 0:
        return None

    rows, cols, vals = build_constraint_rows(f, active_lags, n)
    if len(rows) == 0:
        return None

    T = csr_matrix((vals, (rows, cols)), shape=(K, n))

    c_obj = -np.ones(n)
    b_ub = np.full(K, M)

    try:
        res = linprog(c_obj, A_ub=T, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': 30})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g
    except Exception:
        pass
    return None


def run_lp_cutting_plane(f_init, n, S, max_time, rng, K_init=50, K_max=300, alpha=0.3):
    """
    LP-based best response with cutting plane.
    Iteratively solves LP with active lag constraints, blends solution.
    """
    f = np.clip(f_init, 0.0, 1000.0).copy().astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) * S / n
    else:
        f = f * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = f.copy()
    lp_fail_count = 0

    c = fft_convolve_self(f)
    M = c.max()
    active_lags = np.argpartition(c, -K_init)[-K_init:]

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        if time.monotonic() - start > max_time - 5:
            break

        c = fft_convolve_self(f)
        M = c.max()
        if M < best_score - 1e-12:
            best_score = M
            best_local = f.copy()

        # Solve LP
        g = solve_lp_best_response(f, M, active_lags, n)
        if g is None:
            lp_fail_count += 1
            if lp_fail_count > 5:
                break
            continue

        lp_fail_count = 0
        g_sum = g.sum()
        if g_sum < 1e-15:
            continue
        g = g * (S / g_sum)

        # Blend
        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0)
        f_new_sum = f_new.sum()
        if f_new_sum > 1e-15:
            f_new = f_new * (S / f_new_sum)

        c_new = fft_convolve_self(f_new)
        new_score = c_new.max()

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_local = f_new.copy()

        f = f_new

        # Update active lags
        K_use = min(K_max, len(c_new))
        top_lags = np.argpartition(c_new, -K_use)[-K_use:]
        # Keep union of previous and new top lags, capped
        combined = np.unique(np.concatenate([active_lags[:K_max // 2], top_lags]))
        active_lags = combined[:K_max]

        # Occasionally reset to pure top lags
        if it % 15 == 0:
            active_lags = np.argpartition(c_new, -K_max)[-K_max:]

        # Adaptive alpha
        if new_score > M * 1.001:
            alpha = max(0.05, alpha * 0.5)
        elif new_score < M * 0.999:
            alpha = min(0.8, alpha * 1.1)

    return best_local, best_score


def run_fista_refined(a_init, n, S, max_time, rng, beta_init=40.0, beta_max=5000.0):
    """FISTA with adaptive beta, top-k to softmax transition, and micro-structure."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 1.0 / (beta_init * 2.5)
    stall_counter = 0
    best_stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: starts low (smooth), increases (focus on peaks)
        progress = min(1.0, it / 5000.0)
        beta = beta_init + (beta_max - beta_init) * progress

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            stall_counter = 0
            best_stall = max(best_stall, stall_counter)
        else:
            stall_counter += 1

        # Weight strategy: top-k for early iterations, softmax for later
        if it < 2000:
            k = min(50, 1 + it // 200)
            w = toppk_weights(c, k)
        else:
            w = softmax_weights(c, beta)

        grad = compute_gradient_fft(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum with reset on stall
        if stall_counter > 2000:
            t = 1.0
            y = x_new
            a = x_new
            step = 1.0 / (beta * 2.0)
            stall_counter = 0
        else:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = x_new + ((t - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t = t_new

        # Adaptive step size
        if stall_counter > 500:
            step *= 0.6
            stall_counter = 0
        else:
            step = min(step * 1.005, 2.0 / beta)

        # Periodic perturbation to escape local minima
        if it % 3000 == 0 and rng.random() < 0.2:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0

            low_lags = np.argpartition(c, 100)[:100]
            w_flat = np.zeros_like(c)

                c_flat = fft_convolve_self(x_flat)
                if c_flat.max() < cur:
                    y = x_flat
                    a = x_flat
                    t = 1.0

    return best_local, best_score


def run_dual_ascent(a_init, n, S, max_time, rng, T_expected=5000):
    """
    Exponentiated gradient on dual variables (principled step size from game theory).
    The dual variables represent "weights" on each lag, updated multiplicatively.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    # Initialize dual variables uniformly
    n_lags = 2 * n - 1
    lam = np.ones(n_lags) / n_lags

    # Principled step size from game theory: eta = sqrt(2*ln(K)/T)
    eta = np.sqrt(2.0 * np.log(n_lags) / max(T_expected, 100))
    eta = min(eta, 0.5)  # Cap for stability

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        # Decay eta slowly
        eta_t = eta / (1.0 + 0.001 * it)

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()

        # Use lam as weights for the gradient
        grad = compute_gradient_fft(a, lam)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Gradient step
        step = S / (n * gnorm) * eta_t
        a_new = project_simplex(a - step * grad, S)

        # Update dual variables: exponential tilt toward high-convolution lags
        # lam_k *= exp(eta_t * c_k / max(c))
        c_norm = c / max(c.max(), 1e-15)
        lam = lam * np.exp(eta_t * c_norm)
        lam_sum = lam.sum()
        if lam_sum < 1e-300:
            lam = np.ones(n_lags) / n_lags
        else:
            lam = lam / lam_sum

        # Periodic regularization on lam to prevent concentration
        if it % 500 == 0:
            lam = 0.95 * lam + 0.05 * (np.ones(n_lags) / n_lags)

        a = a_new

    return best_local, best_score

                # Block shuffle
                n_blocks = 16
                block_size = n // n_blocks
                perm = rng.permutation(n_blocks)
                new_a = np.empty_like(global_a)
                for i in range(n_blocks):
                    new_a[i*block_size:(i+1)*block_size] = global_a[perm[i]*block_size:(perm[i]+1)*block_size]
                perturbed = new_a + rng.normal(0, 0.002 * S / np.sqrt(n), n)
            else:
                # Multiplicative perturbation
                perturbed = global_a * (1.0 + rng.normal(0, 0.015, n))

            perturbed = safe_normalize(perturbed, n)
            c_pert = fft_convolve_self(perturbed)
            pert_score = c_pert.max()

            # Quick optimization from perturbed start
            a_r, _ = run_fista_refined(perturbed, n, S, per_restart, rng,
                                        beta_init=60.0, beta_max=3000.0)
            a_r = safe_normalize(a_r, n)
            c_r = fft_convolve_self(a_r)
            if c_r.max() < global_best:
                global_best = c_r.max()
                global_a = a_r.copy()
                print("  Restart %d: %.8f (obj %.8f)" % (r, global_best, 2 * n * global_best / S**2))

    # === Phase 6: LP from best (8% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        phase6_time = min(remaining * 0.4, 0.08 * (budget_s - 40))
        if phase6_time > 10:
            print("Phase 6: LP Polish (%.0f sec)" % phase6_time)
            t0 = time.monotonic()
            a_p6, _ = run_lp_cutting_plane(global_a, n, S, phase6_time, rng,
                                            K_init=100, K_max=400, alpha=0.2)
            a_p6 = safe_normalize(a_p6, n)
            c_p6 = fft_convolve_self(a_p6)
            if c_p6.max() < global_best:
                global_best = c_p6.max()
                global_a = a_p6.copy()
                print("  LP2: %.8f (obj %.8f) [%.0fs]" % (global_best, 2 * n * global_best / S**2, time.monotonic() - t0))

    # === Phase 7: Final micro-polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 5:
        print("Phase 7: Final micro-polish (%.0f sec)" % remaining)
        t0 = time.monotonic()
        a_f, _ = run_hybrid_polish(global_a, n, S, remaining * 0.7, rng)
        a_f = safe_normalize(a_f, n)
        c_f = fft_convolve_self(a_f)
        if c_f.max() < global_best:
            global_best = c_f.max()
            global_a = a_f.copy()
            print("  Final: %.8f (obj %.8f) [%.0fs]" % (global_best, 2 * n * global_best / S**2, time.monotonic() - t0))

        # One more LP attempt if time allows
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            a_f2, _ = run_lp_cutting_plane(global_a, n, S, remaining2, rng,
                                            K_init=200, K_max=500, alpha=0.15)
            a_f2 = safe_normalize(a_f2, n)
            c_f2 = fft_convolve_self(a_f2)
            if c_f2.max() < global_best:
                global_best = c_f2.max()
                global_a = a_f2.copy()
                print("  Final LP: %.8f (obj %.8f)" % (global_best, 2 * n * global_best / S**2))

    # === Final output ===
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    final_obj = 2.0 * n * global_best / (S * S)
    print("FINAL: length=%d, score=%.8f, objective=%.8f" % (len(final_list), global_best, final_obj))

    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: %.8f" % ver)
    else:
        print("Verification failed, using fallback")
        final_list = [0.5] * 256

    return final_list
