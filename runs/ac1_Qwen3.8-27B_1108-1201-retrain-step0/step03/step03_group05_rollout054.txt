    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve_f(a, b):
    """Compute (a*b)_k = sum_j a_{k-j} b_j for k=0..2n-2 via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. softmax params z."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    # grad_a[k] = 2 * sum_j w[j] * a[k-j] (cross-correlation)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def solve_lp_direction(f, M, n, solver='GLPK'):
    """
    Solve the Frank-Wolfe LP:
    Maximize sum(g_j) subject to g >= 0 and (f*g)_k <= M for all k.
    Returns the optimal g vector.
    """
    g = cp.Variable(n, non_negative=True)
    
    # Compute the constraint matrix sparsely
    # (f*g)_k = sum_j f_{k-j} * g_j
    # We need this for k = 0, .., 2n-2
    
    # Build the constraint: F @ g <= M * ones(2n-1)
    # F[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # This is a convolution matrix. Let's build it.
    # For efficiency, build as a sparse matrix.
    from scipy.sparse import csr_matrix
    import scipy.sparse as sp
    
    # Row k has non-zeros at columns j where 0 <= k-j < n
    # i.e., max(0, k-n+1) <= j <= min(k, n-1)
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    F = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [F @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if g.value is None:
            return None
        g_val = np.asarray(g.value).flatten()
        g_val = np.maximum(g_val, 0.0)
        s = g_val.sum()
        if s < 1e-10:
            return None
        return g_val
    except Exception as e:
        print("      LP solve failed: %s" % str(e))
        return None


def run_lp_frank_wolfe(a_init, n, S, max_time, rng, lp_every=5, fw_steps=100):
    """
    Frank-Wolfe with LP oracle, combined with subgradient steps
    between LP calls.
    """
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        M = c.max()
        S = a.sum()
        score = 2.0 * n * M / (S * S)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if it <= 20 or it % 50 == 0:
                print("    FW iter %d: score=%.8f, M=%.6f" % (it, best_score, M))
        
        # Every lp_every iterations, solve the LP
        if it % lp_every == 1:
            t0 = time.monotonic()
            g = solve_lp_direction(a / S, M, n)
            lp_time = time.monotonic() - t0
            
            if g is not None:
                g_sum = g.sum()
                if g_sum > 1e-10:
                    # Frank-Wolfe update: a_new = (1 - lr) * a + lr * S * g / g_sum
                    lr = 2.0 / (it + 2)
                    a_new = (1 - lr) * a + lr * S * g / g_sum
                    a_new = np.maximum(a_new, 0.0)
                    s = a_new.sum()
                    if s > 1e-10:
                        a_new = a_new * S / s
                        a = a_new
                    else:
                        a = np.ones(n) * S / n
            if lp_time < 30:
                print("      LP time: %.2fs, iter %d" % (lp_time, it))
        else:
            # Subgradient step
            # Use top-k peaks
            top_k = min(50, n // 10)
            peak_indices = np.argsort(c)[::-1][:top_k]
            peak_vals = c[peak_indices]
            total = peak_vals.sum()
            if total < 1e-30:
                break
            weights = peak_vals / total
            
            # Subgradient: grad_j = 2 * sum_k w_k * a_{k-j}
            w_full = np.zeros(2 * n - 1)
            for i, k in enumerate(peak_indices):
                w_full[int(k)] = weights[i]
            
            fsize = 1 << (len(w_full) + n - 2).bit_length()
            W = fft(w_full, fsize)
            A = fft(a, fsize)
            grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
            
            # Projected gradient step
            lr = 2.0 * S / (n * (it + 2))
            a_new = project_simplex(a - lr * grad, S)
            a = a_new
    
    return best_a, best_score


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=300):
    """Run L-BFGS-B with continuation on beta."""
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        def fun(z_val):
            return softmax_obj_grad(z_val, beta, S)[0]

        def jac(z_val):
            return softmax_obj_grad(z_val, beta, S)[1]

        t0 = time.monotonic()
        try:
            result = scipy_minimize(
                fun, z,
                jac=jac,
                method='L-BFGS-B',
                options={
                    'maxiter': maxiter_per_beta,
                    'maxfun': maxiter_per_beta * 3,
                    'ftol': 1e-15,
                    'gtol': 1e-10,
                    'maxls': 50,
                    'iprint': -1
                }
            )
        except Exception as e:
            print("      L-BFGS error: %s" % str(e))
            break

        elapsed = time.monotonic() - t0
        z = result.x
        z = z - z.mean()

        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = S * exp_z / exp_z.sum()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * S / s

        c = fft_convolve_self(a)
        score = c.max()

        if score < best_score:
            best_score = score
            best_a = a.copy()

        print("    beta=%4d: score=%.8f (%.1fs)" % (beta, best_score, elapsed))

    return best_a, best_score


def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=256):
    """Nesterov-accelerated top-k subgradient."""
    a = safe_normalize(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    return best_local, best_score

Actually, I realize there might be an issue with my LP formulation. Let me double-check.

The primal LP:
- Maximize $\mathbf{1}^T g$
- Subject to $Fg \leq M \mathbf{1}$ where $F_{k,j} = f_{k-j}$ for valid $k,j$
- Subject to $g \geq 0$

The dual LP:
- Minimize $M \mathbf{1}^T y$
- Subject to $F^T y \geq \mathbf{1}$
- Subject to $y \geq 0$

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Uses a multi-scale optimization strategy:
1. L-BFGS with log-sum-exp continuation for smooth local optimization
2. LP-based Frank-Wolfe (conditional gradient) for globally optimal directions
3. Nesterov-accelerated top-k subgradient for fast convergence
4. Restricted LP (top-K peaks) for efficient global direction finding
5. Multi-length exploration and final high-precision polishing
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize
from scipy.sparse import csr_matrix
import cvxpy as cp


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_grad(a, w):
    """Compute grad[j] = 2 * sum_k w[k] * a[k-j] via FFT (cross-correlation)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a, n):
    """Compute 2*n*max_conv/sum^2."""
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def softmax_obj_grad(z, beta, S):
    """Compute smoothed log-sum-exp objective and gradient w.r.t. z."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = fft_cross_grad(a, w)
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def solve_restricted_lp(f_normalized, M, peak_indices, B_bound):
    """
    Solve restricted Frank-Wolfe LP using only top-K peak constraints.
    Maximize sum(g) s.t. g >= 0, (f*g)_k <= M for k in peak_indices.
    f_normalized should sum to 1.
    Returns optimal g or None.
    """
    n = len(f_normalized)
    g = cp.Variable(n, non_negative=True)

    rows = []
    cols = []
    vals = []

    for k in peak_indices:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(len(rows))
            cols.append(j)
            vals.append(f_normalized[k - j])

    if not rows:
        return None

    F = csr_matrix((vals, (rows, cols)), shape=(len(peak_indices), n))
    rhs = M * np.ones(len(peak_indices))

    constraints = [F @ g <= rhs, cp.sum(g) <= B_bound]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.GLOP)
        if g.value is None:
            prob.solve(solver=cp.GLPK)
            if g.value is None:
                return None
        g_val = np.asarray(g.value).flatten()
        g_val = np.maximum(g_val, 0.0)
        if g_val.sum() < 1e-15:
            return None
        return g_val
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
            if g.value is None:
                return None
            g_val = np.asarray(g.value).flatten()
            g_val = np.maximum(g_val, 0.0)
            if g_val.sum() < 1e-15:
                return None
            return g_val
        except Exception:
            return None


def solve_full_lp(f_normalized, M, n):
    """
    Solve full Frank-Wolfe LP:
    Maximize sum(g) s.t. g >= 0, (f*g)_k <= M for all k = 0..2n-2.
    """
    g = cp.Variable(n, non_negative=True)

    total_rows = 2 * n - 1
    rows = []
    cols = []
    vals = []

    for k in range(total_rows):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f_normalized[k - j])

    F = csr_matrix((vals, (rows, cols)), shape=(total_rows, n))
    rhs = M * np.ones(total_rows)

    B_bound = 4.0 * n
    constraints = [F @ g <= rhs, cp.sum(g) <= B_bound]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.GLOP, verbose=False)
        if g.value is None:
            prob.solve(solver=cp.GLPK, verbose=False)
            if g.value is None:
                return None
        g_val = np.asarray(g.value).flatten()
        g_val = np.maximum(g_val, 0.0)
        if g_val.sum() < 1e-15:
            return None
        return g_val
    except Exception:
        try:
            prob.solve(solver=cp.CBC, verbose=False)
            if g.value is None:
                return None
            g_val = np.asarray(g.value).flatten()
            g_val = np.maximum(g_val, 0.0)
            if g_val.sum() < 1e-15:
                return None
            return g_val
        except Exception:
            return None


def run_lbfgs(a_init, n, S, max_time, rng, beta_schedule=None, maxiter_per_beta=200):
    """Run L-BFGS-B with log-sum-exp continuation."""
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break

        def fun(z_val):
            return softmax_obj_grad(z_val, beta, S)[0]

        def jac(z_val):
            return softmax_obj_grad(z_val, beta, S)[1]

        t0 = time.monotonic()
        try:
            result = scipy_minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_beta, 'maxfun': maxiter_per_beta * 3,
                         'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50, 'iprint': -1}
            )
        except Exception:
            break

        elapsed = time.monotonic() - t0
        z = result.x
        z = z - z.mean()

        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = S * exp_z / exp_z.sum()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * S / s

        c = fft_convolve_self(a)
        score = c.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if time.monotonic() - start > max_time - 3:
            break

    return best_a, best_score


def run_nesterov(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=256):
    """Nesterov-accelerated top-k subgradient method."""
    a = safe_normalize(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 80)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        grad = fft_cross_grad(a, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    return best_local, best_score
