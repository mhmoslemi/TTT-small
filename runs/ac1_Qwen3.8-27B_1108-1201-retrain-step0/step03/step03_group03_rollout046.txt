```python
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def fista_smoothed(a_init, max_time, beta_start, beta_end, rng):
    """FISTA with LSE smoothing, adaptive beta."""
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a))
        S = 1.0
    n = len(a)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    elapsed_total = max_time
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: linearly interpolate from beta_start to beta_end
        frac = min(1.0, (time.monotonic() - start) / elapsed_total)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = y.copy()
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: 1/L where L is the Lipschitz constant of the gradient
        # For LSE with parameter beta, L ≈ beta * ||a||_1^2 (roughly)
        step = 1.0 / (beta * S * 2.0)
        
        x_new = project_simplex(y - step * grad, S)
        
        # Momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return best_a, best_score

def active_set_optimize(a_init, max_time, top_k, rng):
    """Active set: minimize max of top-k lags using subgradient."""
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    S = a.sum()
    n = len(a)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    step = 2.0 / (n * S)
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = y.copy()
        
        # Get top-k lags
        k = min(top_k, len(c))
        top_idx = np.partition(c, -k)[-k:]
        # Subgradient: average of gradients at top-k lags
        # grad[j] = sum over top lags l of (a[l-j] + a[j-l]) / k
        grad = np.zeros(n)
        for l in range(len(c)):
            if c[l] >= top_idx[0]:  # this is the minimum of top-k
                # Add gradient contribution
                jm = l - np.arange(n)  # l - j for j = 0..n-1
                for j in range(n):
                    k_minus_j = l - j
                    j_minus_k = j - l
                    if 0 <= k_minus_j < n:
                        grad[j] += a[k_minus_j]
                    if 0 <= j_minus_k < n:
                        grad[j] += a[j_minus_k]
        grad /= np.sum(c >= top_idx[0])
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step
        step = 2.0 * S / (n * gnorm * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return best_a, best_score
```
