```python
"""
Search for nonnegative sequences minimizing
    2*n*max(convolve(a,a)) / sum(a)**2.

Algorithm:
- Normalize the sequence to sum(sqrt(2n)), so the objective is max(convolve(a,a)).
- Use several diverse initializations (parentheses, smooth randoms, Fourier modes,
  Gaussian bumps, powers, cosines, cross-overs).
- Run a population of mirror-descent / subgradient runs.  Each run uses adaptive
  peak weights (top-k or full softmax), momentum in log-space, backtracking
  line search on the max coefficient, and adaptive perturbations/random restarts
  to escape stagnation.
- Final long polished run on the global best, with conservative parameters if the
  current best is already close to the target.
"""

import time
import math
import numpy as np

try:
    from scipy.fft import fft, ifft
except Exception:
    from numpy.fft import fft, ifft


EPS = 1e-12
TARGET_VAL = 1.5027


def p2(size):
    """Smallest power of two >= size."""
    if size <= 1:
        return 1
    return 1 << (int(size) - 1).bit_length()


def fft_conv_self(a):
    """Full linear convolution a * a, clipped to [0, 2n-1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    out_len = 2 * n - 1
    m = p2(out_len)
    A = fft(a, m)
    c = ifft(A * A).real
    c = c[:out_len]
    np.maximum(c, 0.0, out=c)
    return c


def normalize_nonneg(a, S):
    """Clip to nonnegative finite and rescale to sum S."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, None)
    if a.size == 0:
        return a
    s = float(a.sum())
    if not np.isfinite(s) or s <= EPS:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    a *= S / s

    if a.size and float(a.max()) > 1000.0:
        a = np.clip(a, 0.0, 1000.0)
        s2 = float(a.sum())
        if s2 > EPS:
            a *= S / s2
        else:
            a = np.ones(a.size, dtype=np.float64)
            a *= S / a.size
    return a


def compute_h_sparse(a, idx, w):
    """
    h_i = sum_k w_k * a[k-i] for active peak indices idx.
    This is the gradient (up to factor 2) of weighted sum of convolution peaks.
    """
    n = len(a)
    L = c_len = len(idx)
    h = np.zeros(n, dtype=np.float64)
    for k, weight in zip(idx, w):
        if weight <= 0.0:
            continue
        k = int(k)
        lo = k - (n - 1)
        if lo < 0:
            lo = 0
        hi = k
        if hi >= n:
            hi = n - 1
        if lo > hi:
            continue
        j_start = k - hi
        j_end = k - lo
        h[lo:hi + 1] += weight * a[j_start:j_end + 1][::-1]
    return h


def compute_h_fft(a, w_full):
    """
    Same h_i = sum_k w_full[k] a[k-i], computed by FFT for dense weights.
    """
    n = len(a)
    L = len(w_full)
    out_len = L + n - 1
    m = p2(out_len)
    W = fft(w_full, m)
    R = fft(a[::-1], m)
    conv = ifft(W * R).real
    return conv[n - 1:n - 1 + n]


def random_candidate(n, S, rng):
    """Random smoothed positive candidate."""
    x = rng.random(n) + 0.1
    if n > 16:
        w = int(rng.integers(1, max(2, n // 8)))
        if w > 1:

            x = np.convolve(x, kernel, mode="same")
    x = np.maximum(x, 1e-6)
    if rng.random() < 0.35:
        x = x * np.exp(rng.normal(0.0, 0.25, n))
    return normalize_nonneg(x, S)

    i = np.arange(n, dtype=np.float64)

        cands.append(x)

    # Gaussian bump mixtures.
    for nb in (2, 3, 5, 8):
        x = 0.02 * np.ones(n, dtype=np.float64)
        for _ in range(nb):

            width = max(2.0, float(rng.uniform(n / 60.0, n / 6.0)))
            x += np.exp(-0.5 * ((i - center) / width) ** 2)
        cands.append(x)

    # One-sided power curves and sums of power curves.
    t = i / float(max(1, n - 1))

        cands.append((1.0 - t) ** p + 0.05)

    """Keep best `keep` (objective, array) tuples."""
    if not results:

    if keep >= len(results):

    order = np.argsort(objs, kind="stable")
    out = []
    for idx in order[:keep]:
        out.append(results[idx])
    return out

            dict(top_k=4, beta=20.0, lr=0.10, mom=0.85, full=False, clip=1.0,
                 noise=0.0, stall_limit=2000, p_scale=0.02),
            dict(top_k=16, beta=6.0, lr=0.15, mom=0.80, full=False, clip=1.5,
                 noise=0.0, stall_limit=1500, p_scale=0.03),
        ]
    return [
        dict(top_k=16, beta=8.0, lr=0.35, mom=0.70, full=False, clip=2.0,
             noise=0.0, stall_limit=800, p_scale=0.08),
        dict(top_k=2, beta=20.0, lr=0.35, mom=0.85, full=False, clip=2.0,
             noise=0.0, stall_limit=800, p_scale=0.10),
        dict(top_k=64, beta=3.0, lr=0.30, mom=0.60, full=True, clip=1.5,
             noise=0.0, stall_limit=1000, p_scale=0.08),
    ]


def run_directional(a0, S, time_budget, rng, params, verbose=False):
    """
    Mirror-descent / subgradient optimization on a fixed sum simplex.

    Objective is max coefficient of a*a.  The direction is derived from a
    weighted sum of currently large convolution peaks, with adaptive weights,
    momentum, and line search.
    """
    a = normalize_nonneg(a0, S)
    n = a.size
    if n == 0:
        return a, np.inf

    c = fft_conv_self(a)
    cur_M = float(c.max())
    best_M = cur_M
    best_a = a.copy()

    v = np.zeros(n, dtype=np.float64)

    lr = float(params.get("lr", 0.3))
    lr0 = lr
    max_lr = 3.0
    min_lr = 0.02
    mom = float(params.get("mom", 0.7))
    top_k = int(params.get("top_k", 32))
    beta = float(params.get("beta", 10.0))
    full = bool(params.get("full", False))
    max_clip = float(params.get("clip", 2.0))
    noise = float(params.get("noise", 0.0))
    stall_limit = int(params.get("stall_limit", 800))
    p_scale = float(params.get("p_scale", 0.1))

    start = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - start < time_budget:
        if not np.isfinite(cur_M) or cur_M <= 0.0:
            break

        cmax = float(c.max())
        if cmax > cur_M:
            cur_M = cmax

        if full:
            temp = max(cmax / (beta + 1.0), 1e-15)
            w_full = np.exp((c - cmax) / temp)
            ws = float(w_full.sum())
            if not np.isfinite(ws) or ws <= 1e-300:
                w_full = np.zeros(c.size, dtype=np.float64)
                w_full[int(np.argmax(c))] = 1.0
                ws = 1.0
            else:
                w_full /= ws
            h = compute_h_fft(a, w_full)
            W = float(np.dot(w_full, c))
            active_idx = None
            active_w = w_full
        else:
            K = min(max(1, top_k), c.size)
            if K >= c.size:
                idx = np.arange(c.size, dtype=np.int64)
            else:
                # Tiny noise breaks symmetric plateaus, e.g. uniform sequence.
                noise_std = max(cmax * 1e-8, 1e-12)
                csel = c + rng.normal(0.0, noise_std, c.size)
                idx = np.argpartition(csel, -K)[-K:]

            ctop = c[idx]
            cmax_top = float(ctop.max())
            if beta <= 0.0 or K == 1:
                w = np.full(K, 1.0 / K, dtype=np.float64)
            else:
                temp = max(cmax / (beta + 1.0), 1e-15)
                w = np.exp((ctop - cmax_top) / temp)
                ws = float(w.sum())
                if not np.isfinite(ws) or ws <= 1e-300:
                    w = np.zeros(K, dtype=np.float64)
                    w[int(np.argmax(ctop))] = 1.0
                else:
                    w /= ws

            if K > 64:
                w_full = np.zeros(c.size, dtype=np.float64)
                w_full[idx] = w
                h = compute_h_fft(a, w_full)
                W = float(np.dot(w_full, c))
                active_idx = None
                active_w = w_full
            else:
                h = compute_h_sparse(a, idx, w)
                W = float(np.dot(w, ctop))
                active_idx = idx
                active_w = w

        d = h - h.mean()
        dscale = float(np.max(np.abs(d))) if d.size else 0.0
        if not np.isfinite(dscale) or dscale < 1e-14:
            # De-subgradient direction to break symmetric flat directions.
            d = np.sin((np.pi * (np.arange(n) + 0.5)) / max(1, n) * 2.0)
            d = d - d.mean()
            dscale = float(np.max(np.abs(d)))
            if not np.isfinite(dscale) or dscale < 1e-14:
                d = rng.normal(0.0, 1.0, n)
                d = d - d.mean()
                dscale = float(np.max(np.abs(d)))
                if not np.isfinite(dscale) or dscale < 1e-14:
                    break
        d = d / dscale

        v = mom * v + (1.0 - mom) * d
        d = v - v.mean()
        dscale = float(np.max(np.abs(d))) if d.size else 0.0
        if not np.isfinite(dscale) or dscale < 1e-14:
            v = np.zeros(n, dtype=np.float64)
            d = rng.normal(0.0, 1.0, n)
            d = d - d.mean()
            dscale = float(np.max(np.abs(d)))
            if not np.isfinite(dscale) or dscale < 1e-14:
                break
        d = d / dscale
        d = np.clip(d, -3.0, 3.0)

        accepted = False
        cap = max(cur_M, best_M) * 1.02 + 1e-8
        amps = (lr, lr * 0.5, lr * 0.25)

        for amp in amps:
            if amp <= 0.0:
                continue
            logfac = -amp * d
            logfac = np.clip(logfac, -max_clip, max_clip)
            cand = a * np.exp(logfac)
            cand = np.maximum(cand, 1e-12)
            cand = normalize_nonneg(cand, S)

            c2 = fft_conv_self(cand)
            M2 = float(c2.max())

            if active_idx is None:
                W2 = float(np.dot(active_w, c2))
            else:
                W2 = float(np.dot(active_w, c2[active_idx]))

            if not np.isfinite(M2) or not np.isfinite(W2):
                continue

            if M2 < cur_M - 1e-11:
                accepted = True
            elif W2 < W - 1e-9 and M2 < cap:
                accepted = True

            if accepted:
                a = cand
                c = c2
                cur_M = M2
                lr = min(max_lr, lr * 1.15)
                break

        if accepted:
            stall = 0
            if cur_M < best_M:
                best_M = cur_M
                best_a = a.copy()
        else:
            # Try one very small step; if it does not improve max too much, take it.
            amp = lr * 0.1
            if amp > 0.0:
                logfac = -amp * d
                logfac = np.clip(logfac, -0.5 * max_clip, 0.5 * max_clip)
                cand = a * np.exp(logfac)
                cand = np.maximum(cand, 1e-12)
                cand = normalize_nonneg(cand, S)
                c2 = fft_conv_self(cand)
                M2 = float(c2.max())
                if np.isfinite(M2) and M2 < cur_M * 1.02 + 1e-8:
                    a = cand
                    c = c2
                    cur_M = M2
                    if cur_M < best_M:
                        best_M = cur_M
                        best_a = a.copy()
            lr = max(min_lr, lr * 0.7)
            stall += 1

            if stall > stall_limit:
                # Randomized restart / perturbation.
                if rng.random() < 0.3:
                    base = random_candidate(n, S, rng)
                else:
                    base = a
                sc = p_scale * (1.0 + 0.2 * min(3, stall // stall_limit))
                ax = base * np.exp(rng.normal(0.0, min(0.6, sc), n))
                a = normalize_nonneg(ax, S)
                c = fft_conv_self(a)
                cur_M = float(c.max())
                v = np.zeros(n, dtype=np.float64)
                lr = lr0 * 0.7
                stall = 0
                if cur_M < best_M:
                    best_M = cur_M
                    best_a = a.copy()

        if noise > 0.0 and (it % 100 == 0):
            ax = a * np.exp(rng.normal(0.0, noise, n))
            a = normalize_nonneg(ax, S)
            c = fft_conv_self(a)
            cur_M = float(c.max())
            if cur_M < best_M:
                best_M = cur_M
                best_a = a.copy()
                stall = 0

        if verbose and it % 5000 == 0:
            print(f"        dir it={it} cur={cur_M:.7f} best={best_M:.7f}")

        it += 1

    return best_a, best_M


def sort_population(pop):
    """Sort population ascending by objective."""
    if not pop:
        return []
    objs = np.array([float(m["M"]) for m in pop], dtype=np.float64)
    order = np.argsort(objs, kind="stable")
    return [pop[i] for i in order]


def make_mutant(top_pop, global_a, n, S, rng, strategies):
    """Create one mutated population member from good parents."""
    base = None
    if top_pop and len(top_pop) > 0 and rng.random() < 0.7:
        base = top_pop[int(rng.integers(0, min(3, len(top_pop))))]["a"]
    if base is None:
        if global_a is not None:
            base = global_a
        else:
            base = random_candidate(n, S, rng)

    r = rng.random()
    if r < 0.45:
        sig = float(rng.uniform(0.03, 0.20))
        ax = base * np.exp(rng.normal(0.0, sig, n))
    elif r < 0.80 and len(top_pop) >= 2:
        a1 = base
        a2 = top_pop[int(rng.integers(0, len(top_pop)))["a"] if False else len(top_pop)]]
        # The above indexing line has a typo risk; use safe version below.
        a2 = top_pop[int(rng.integers(0, len(top_pop)))]["a"]
        th = float(rng.uniform(0.3, 0.7))
        eps = 1e-12
        ax = np.exp((1.0 - th) * np.log(np.maximum(a1, eps)) +
                    th * np.log(np.maximum(a2, eps)))
    else:
        ax = random_candidate(n, S, rng)

    ax = normalize_nonneg(ax, S)
    return {
        "a": ax,
        "M": np.inf,
        "strat": int(rng.integers(0, len(strategies)))
    }


def ensure_population_objectives(pop, global_a_ref=None):
    """Fill missing objectives and return global best update."""
    best_a = global_a_ref
    best_m = np.inf if global_a_ref is None else float("inf")
    if global_a_ref is not None:
        # Caller passes current global separately; this function just updates mem objectives.
        pass

    for mem in pop:
        if not np.isfinite(mem["M"]):
            c = fft_conv_self(mem["a"])
            mem["M"] = float(c.max())
    return pop, best_a, best_m


def propose_candidate(seed=0, time_budget=2400):
    """Return a nonnegative list minimizing the evaluation objective."""
    deadline = time.monotonic() + max(5.0, float(time_budget) - 20.0)
    rng = np.random.default_rng(seed)

    parent = load_parent()
    if parent is not None and 64 <= parent.size <= 8192:
        n = parent.size
    else:
        n = 4096

    S = math.sqrt(2.0 * float(n))

    global_a = None
    global_M = np.inf

    # Parent as primary seed.
    if parent is not None:
        try:
            pp = normalize_nonneg(parent, S)
            c = fft_conv_self(pp)
            M = float(c.max())
            if np.isfinite(M):
                global_a = pp
                global_M = M
                print(f"parent n={n}, obj={M:.8f}")
        except Exception as e:
            print(f"parent init failed: {e}")

    # Initial candidate pool.
    inits = generate_initial_candidates(n, S, rng, parent)
    results = []
    for a in inits:
        try:
            c = fft_conv_self(a)
            M = float(c.max())
            if np.isfinite(M):
                results.append((M, a))
        except Exception:
            pass

    results = extract_top(results, 12)

    # Build population.
    population = []
    for M, a in results:
        if M < global_M:
            global_M = M
            global_a = a.copy()
        population.append({"a": a.copy(), "M": M, "strat": 0})

    if global_a is not None:
        population.append({"a": global_a.copy(), "M": global_M, "strat": 0})

    strategies = make_strategies()
    if not population:
        a0 = normalize_nonneg(np.ones(n, dtype=np.float64), S)
        c0 = fft_conv_self(a0)
        M0 = float(c0.max())
        if global_a is None:
            global_a = a0.copy()
            global_M = M0
        population.append({"a": a0.copy(), "M": M0, "strat": 0})

    pop_size = 6
    while len(population) < pop_size:
        a = random_candidate(n, S, rng)
        c = fft_conv_self(a)
        M = float(c.max())
        if np.isfinite(M) and M < global_M:
            global_M = M
            global_a = a.copy()
        population.append({"a": a, "M": M, "strat": len(population) % len(strategies)})

    population = sort_population(population)[:pop_size]
    for i, mem in enumerate(population):
        mem["strat"] = i % len(strategies)

    if global_M < np.inf:
        print(f"initial best obj={global_M:.8f}")

    # Main population search.
    final_reserve = min(600.0, 0.25 * float(time_budget))
    cycle_count = 0

    while True:
        rem = deadline - time.monotonic() - final_reserve
        if rem < 5.0 or cycle_count > 300:
            break

        slice_t = min(60.0, max(5.0, rem / (pop_size * 4.0)))

        for mem in population:
            if time.monotonic() > deadline - final_reserve - 5.0:
                break
            params = strategies[mem["strat"] % len(strategies)]
            try:
                a_out, M_out = run_directional(mem["a"], S, slice_t, rng, params,
                                               verbose=False)
                if np.isfinite(M_out):
                    mem["a"] = a_out
                    mem["M"] = M_out
                if np.isfinite(M_out) and M_out < global_M:
                    global_M = M_out
                    global_a = a_out.copy()
            except Exception:
                pass

        if time.monotonic() > deadline - final_reserve - 5.0:
            break

        # Selection and breeding.
        population = sort_population(population)
        keep = max(2, pop_size // 2)
        top = population[:keep]

        new_members = []
        while len(top) + len(new_members) < pop_size:
            nm = make_mutant(top, global_a, n, S, rng, strategies)
            new_members.append(nm)

        population = top + new_members

        for mem in population:
            if not np.isfinite(mem["M"]):
                c = fft_conv_self(mem["a"])
                mem["M"] = float(c.max())
            if np.isfinite(mem["M"]) and mem["M"] < global_M:
                global_M = mem["M"]
                global_a = mem["a"].copy()

        population = sort_population(population)[:pop_size]
        for i, mem in enumerate(population):
            mem["strat"] = (i + cycle_count) % len(strategies)

        cycle_count += 1
        if cycle_count % 3 == 0:
            print(f"cycle={cycle_count}, population best={population[0]['M']:.7f}, "
                  f"global={global_M:.7f}, rem={rem:.0f}s")

    # Final polished phase.
    final_time = deadline - time.monotonic()
    if final_time > 30.0 and global_a is not None:
        conservative = bool(global_M < TARGET_VAL + 0.005)
        profiles = get_polish_profiles(conservative)

        fracs = [0.40, 0.35, 0.25]
        if len(profiles) == 2:
            fracs = [0.55, 0.45]
        elif len(profiles) >= 3:
            fracs = [0.38, 0.34, 0.28]

        for prof, frac in zip(profiles, fracs):
            if time.monotonic() > deadline - 10.0:
                break
            t = max(10.0, frac * final_time * 0.9)
            prof = dict(prof)
            if conservative:
                prof["lr"] = min(float(prof.get("lr", 0.15)), 0.20)
            try:
                a_out, M_out = run_directional(global_a, S, t, rng, prof, verbose=False)
            except Exception:
                continue
            if np.isfinite(M_out) and M_out < global_M:
                global_M = M_out
                global_a = a_out.copy()
                print(f"polish improved to {M_out:.8f}")

    # Final validation and safe return.
    if global_a is None:
        global_a = normalize_nonneg(np.ones(n, dtype=np.float64), S)

    global_a = np.asarray(global_a, dtype=np.float64)
    global_a = np.where(np.isfinite(global_a), global_a, 0.0)
    global_a = np.clip(global_a, 0.0, None)
    s = float(global_a.sum())
    if not np.isfinite(s) or s <= 0.01:
        global_a = np.ones(n, dtype=np.float64) / float(n)
    else:
        # Return on unit-sum scale; objective is scale-invariant and values stay safe.
        global_a = global_a / s
        global_a = np.clip(global_a, 0.0, 1000.0)

    out = [float(x) for x in global_a]

    try:
        ver = evaluate_sequence(out)
        if np.isfinite(ver):
            print(f"FINAL local_obj={global_M:.8f}, verified={ver:.8f}, n={n}")
        else:
            print(f"FINAL local_obj={global_M:.8f}, verified_nonfinite, n={n}")
    except Exception as e:
        print(f"Final verify failed: {e}")

    try:
        ver = evaluate_sequence(out)
    except Exception:
        ver = np.inf

    if not np.isfinite(ver):
        fallback = [1.0 / max(1, n)] * n
        try:
            ver = evaluate_sequence(fallback)
        except Exception:
            ver = np.inf
        if np.isfinite(ver):
            return fallback

    return out
```
