```python
def lp_solve(a, M, active_lags):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active_lags, b >= 0"""
    n = len(a)
    # Build constraint matrix for active lags only
    A = np.zeros((len(active_lags), n))
    for i, k in enumerate(active_lags):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[i, j] = a[idx]
    # Solve
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    res = linprog(c, A_ub=A, b_ub=M * np.ones(len(active_lags)), bounds=[(0, None)]*n, method='highs')
    return res.x if res.success else None
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LP-based cutting-plane optimization
combined with subgradient polishing. Key innovation: iterative LP relaxation that
finds optimal "valley-filling" directions, followed by adaptive subgradient refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def cross_convolve(a, b):
    """Full cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def compute_subgradient(a, c, top_k):
    """Compute subgradient of max(c) using top-k active lags."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total

    # Build weight vector of length 2n-1
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]

    # Gradient: grad[m] = 2 * sum_k w_full[k] * a[k-m] (cross-correlation)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    # cross-correlation: sum_k w[k] * a[k-m] = ifft(fft(w) * conj(fft(a)))
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def objective_val(a):
    """Compute objective 2n*max_conv/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def solve_lp_direction(a, M, active_lags, n):
    """
    Solve LP: maximize sum(b) subject to (a*b)[k] <= M for k in active_lags, b >= 0.
    Returns optimal b or None if LP fails.
    """
    num_constraints = len(active_lags)
    if num_constraints == 0:
        return None

    # Build constraint matrix A where A[i,j] = a[active_lags[i] - j] if valid
    A = np.zeros((num_constraints, n), dtype=np.float64)
    for i, k in enumerate(active_lags):
        k = int(k)
        # For column j, A[i,j] = a[k-j] if 0 <= k-j < n
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            j_range = np.arange(j_min, j_max + 1)
            a_indices = k - j_range
            A[i, j_range] = a[a_indices]

    # Maximize sum(b) -> minimize -sum(b)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(num_constraints)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 5.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_based_optimize(a_init, n, S, max_time, rng):
    """
    LP-based cutting plane optimization.
    Iteratively finds the best direction b via LP, then takes steps toward b.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0

    # Initial active set
    c = fft_convolve_self(a)
    initial_k = min(100, n)
    active_lags = np.argsort(c)[::-1][:initial_k]

    while time.monotonic() - start < max_time:
        it += 1
        remaining = max_time - (time.monotonic() - start)
        if remaining < 2.0:
            break

        c = fft_convolve_self(a)
        M = c.max()
        cur_obj = 2.0 * n * M / (S * S)

        if cur_obj < best_score:
            best_score = cur_obj
            best_a = a.copy()

        # Gradually increase active set
        frac = (time.monotonic() - start) / max_time
        num_active = min(n, int(50 + 200 * frac))
        active_lags = np.argsort(c)[::-1][:num_active]

        # Solve LP for direction b
        # Use a slightly relaxed constraint to allow more movement
        relaxation = 1.0 + 0.01 * (1.0 - frac)
        b = solve_lp_direction(a, M * relaxation, active_lags, n)

        if b is None or b.sum() < 1e-15:
            # Fallback: use subgradient direction
            grad = compute_subgradient(a, c, min(50, n))
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            b = project_simplex(a - 0.01 * grad / gnorm * S, S)
        else:
            # Rescale b to have same sum
            b = b * (S / b.sum())
            b = np.clip(b, 0, 1000)

        # Line search for step size
        best_t = 0.0
        best_val = cur_obj
        trial_ts = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
        for trial_t in trial_ts:
            a_trial = (1.0 - trial_t) * a + trial_t * b
            a_trial = np.clip(a_trial, 0, 1000)
            st = a_trial.sum()
            if st < 0.01:
                continue
            c_trial = fft_convolve_self(a_trial)
            val_trial = 2.0 * n * c_trial.max() / (st * st)
            if val_trial < best_val:
                best_val = val_trial
                best_t = trial_t

        if best_t == 0.0:
            # Try very small steps
            for trial_t in [0.01, 0.02, 0.03]:
                a_trial = (1.0 - trial_t) * a + trial_t * b
                a_trial = np.clip(a_trial, 0, 1000)
                st = a_trial.sum()
                if st < 0.01:
                    continue
                c_trial = fft_convolve_self(a_trial)
                val_trial = 2.0 * n * c_trial.max() / (st * st)
                if val_trial < best_val:
                    best_val = val_trial
                    best_t = trial_t

        if best_t == 0.0:
            # No improvement - try perturbation
            if it % 10 == 0:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
            else:
                break
        else:
            a = (1.0 - best_t) * a + best_t * b
            a = np.clip(a, 0, 1000)
            # Renormalize
            st = a.sum()
            if st > 1e-15:
                a = a * (S / st)

        # Periodic perturbation to escape local minima
        if it % 50 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    # Final evaluation
    c_final = fft_convolve_self(a)
    final_obj = 2.0 * n * c_final.max() / (S * S)
    if final_obj < best_score:
        best_score = final_obj
        best_a = a.copy()

    return best_a, best_score


def subgradient_polish(a_init, n, S, max_time, rng, top_k_init=20, top_k_max=256):
    """Subgradient method with Nesterov momentum for polishing."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    y = a.copy()
    t_mom = 1.0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        remaining = max_time - (time.monotonic() - start)
        if remaining < 1.0:
            break

        c = fft_convolve_self(y)
        M = c.max()
        cur_obj = 2.0 * n * M / (S * S)

        if cur_obj < best_score:
            best_score = cur_obj
            best_a = y.copy()

        # Adaptive top-k
        frac = (time.monotonic() - start) / max_time
        top_k = min(top_k_max, top_k_init + int(200 * frac))

        grad = compute_subgradient(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = S / (n * (0.5 + frac))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic noise
        if it % 200 == 0 and rng.random() < 0.15:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0

    # Final check
    c_final = fft_convolve_self(a)
    final_obj = 2.0 * n * c_final.max() / (S * S)
    if final_obj < best_score:
        best_score = final_obj
        best_a = a.copy()

    return best_a, best_score


def soft_max_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=300.0):
    """FISTA with adaptive log-sum-exp smoothing."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        if time.monotonic() - start > max_time - 1.0:
            break

        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        cur_obj = 2.0 * n * cur / (S * S)
        if cur_obj < best_score:
            best_score = cur_obj
            best_a = y.copy()

        # Soft gradient
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = S / (n * beta * (1.0 + 0.001 * it))
        step = max(step, 1e-9)
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    c_final = fft_convolve_self(a)
    final_obj = 2.0 * n * c_final.max() / (S * S)
    if final_obj < best_score:
        best_score = final_obj
        best_a = a.copy()

    return best_a, best_score


def generate_initializations(n, rng):
    """Generate diverse asymmetric initializations."""
    tt = np.linspace(0, 1, n, endpoint=False)
    inits = []

    # Exponential ramps
    for alpha in [1.0, 2.0, 3.0, 5.0, 8.0]:
        inits.append(np.exp(alpha * tt) + 0.01)
        inits.append(np.exp(alpha * (1.0 - tt)) + 0.01)

    # Power-law
    for p in [0.3, 0.5, 1.0, 2.0, 3.0]:
        inits.append((tt + 0.01) ** p + 0.01)
        inits.append(((1.0 - tt) + 0.01) ** p + 0.01)

    # Linear ramps
    inits.append(tt + 0.1)
    inits.append(1.0 - tt + 0.1)

    # Quadratic
    inits.append(tt * tt + 0.01)
    inits.append((1.0 - tt) ** 2 + 0.01)

    # Sigmoid
    for c in [0.3, 0.4, 0.6, 0.7]:
        inits.append(1.0 / (1.0 + np.exp(-10 * (tt - c))) + 0.05)

    # Multi-scale
    inits.append(1.0 + 0.5 * np.sin(2 * np.pi * 3 * tt) + tt)
    inits.append(1.0 + 0.3 * np.sin(4 * np.pi * tt) + 0.5 * tt)

    # Gaussian bump
    for center in [0.2, 0.3, 0.7, 0.8]:
        sigma = 0.15
        inits.append(np.exp(-((tt - center) ** 2) / (2 * sigma * sigma)) + 0.05)

    # Random asymmetric
    for _ in range(3):
        base = rng.random(n)
        ramp = np.linspace(0.5, 1.5, n)
        inits.append(base * ramp + 0.01)

    return inits

    return a * (S / s)

                global_a = a_norm.copy()
            print("  After Phase 3: best=%.6f" % global_best)

    # Phase 4: Final refinement
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            # Alternate between LP and subgradient
            print("=== Phase 4: Final refinement (%.1fs) ===" % remaining)
            seg = remaining / 3
            for round_num in range(3):
                if time.monotonic() > deadline - 3:
                    break
                if round_num % 2 == 0:
                    a_opt, score = lp_based_optimize(global_a, n, S, seg, rng)
                else:
                    a_opt, score = subgradient_polish(global_a, n, S, seg, rng,
                                                      top_k_init=50, top_k_max=256)
                a_norm = safe_normalize(a_opt, n)
                c = fft_convolve_self(a_norm)
                obj = 2.0 * n * c.max() / (S * S)
                if obj < global_best:
                    global_best = obj
                    global_a = a_norm.copy()
                print("  Round %d: best=%.6f" % (round_num, global_best))

    # Try a different n for diversity
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            # Try n2 = 2048 or 2048
            n2 = 2048 if n != 2048 else 2048
            S2 = np.sqrt(2.0 * float(n2))
            # Interpolate global_a to n2
            from numpy import interp as np_interp
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, n2)
            a_interp = np_interp(t_new, t_old, global_a)
            a_interp = a_interp * (S2 / a_interp.sum())

            prog_time = remaining * 0.5
            print("=== Extra: Try n2=%d (%.1fs) ===" % (n2, prog_time))
            a_opt, score = lp_based_optimize(a_interp, n2, S2, prog_time, rng)
            a_norm = safe_normalize(a_opt, n2)
            c = fft_convolve_self(a_norm)
            obj = 2.0 * n2 * c.max() / (S2 * S2)
            if obj < global_best:
                global_best = obj
                global_a = a_norm.copy()
                print("  n2 result: %.6f (new best!)" % obj)
            else:
                print("  n2 result: %.6f (not better)" % obj)

    # Final verification
    if global_a is None:
        # Fallback
        n_fb = 2048
        t_fb = np.linspace(0, 1, n_fb)
        global_a = np.exp(3.0 * t_fb) + 0.01
        global_a = safe_normalize(global_a, n_fb)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024) * np.sqrt(2 * 1024) / 1024

    # Verify with evaluate_sequence
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Verification: %.6f ===" % ver if np.isfinite(ver) else "=== Verification: INVALID ===")

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        n_fb = 2048
        t_fb = np.linspace(0, 1, n_fb)
        global_a = np.exp(2.0 * t_fb) + 0.01
        global_a = safe_normalize(global_a, n_fb)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)

    print("=== FINAL: length=%d, verified=%.6f ===" % (len(final_list), ver))
    return final_list
```
