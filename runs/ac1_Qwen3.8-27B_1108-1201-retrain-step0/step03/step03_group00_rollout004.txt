"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. FISTA with adaptive LSE smoothing and spectral shaping perturbations
2. LP-based refinement on downsampled sequences

import numpy as np
import time
from scipy.fft import fft, ifft
import scipy.optimize

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    """Gradient of LSE approximation."""
    n = len(a)
    cm = c.max()
    e = np.exp(np.clip(beta * (c - cm), -700, 700))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), np.zeros(len(c))
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, w

def spectral_shaping(a, n, S, rng, num_freqs=10):
    """Apply spectral shaping to flatten the spectrum."""
    fsize = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize)
    P = np.abs(A) ** 2
    # Find top frequency bins (excluding DC)
    idx = np.argsort(P)[::-1][:num_freqs]
    idx = idx[idx != 0]  # exclude DC
    if len(idx) == 0:
        return a
    # Reduce the magnitude at these bins
    for i in idx:
        phase = np.angle(A[i])
        A[i] *= 0.9
        A[fsize - i] = np.conj(A[i])  # maintain Hermitian symmetry
    a_new = np.real(ifft(A, fsize))[:n]
    a_new = project_simplex(a_new, S)
    return a_new

def run_fista(a_init, n, S, max_time, rng, mode="lse",
              beta_init=3.0, beta_rate=0.003, beta_max=500.0,
              use_spectral=False, spectral_every=500):
    """FISTA with adaptive smoothing and optional spectral shaping."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0
    
    # Precompute FFT size
    csize = 2 * n - 1
    csize_pad = 1 << (csize - 1).bit_length()
    gsize = 1 << (csize + n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it
        
        # Compute gradient
        grad, w = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Step size: 1/(L) where L ~ 2*beta*max|a| (rough Lipschitz constant)
        L = 2.0 * beta * np.max(np.abs(y)) * 2.0  # conservative
        step = 1.0 / max(L, 1e-10)
        
        # Backtracking line search (limited)
        for bt in range(5):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            # Check sufficient decrease in LSE
            cm_new = c_new.max()
            e_new = np.exp(np.clip(beta * (c_new - cm_new), -700, 700))
            lse_new = cm_new + np.log(e_new.sum()) / beta
            cm_old = c.max()
            e_old = np.exp(np.clip(beta * (c - cm_old), -700, 700))
            lse_old = cm_old + np.log(e_old.sum()) / beta
            
            if lse_new <= lse_old - 0.5 * step * step * gnorm * gnorm + 1e-12:
                break
            step *= 0.5
        
        # Accelerated update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Ensure sum constraint approximately
        y_sum = y.sum()
        if y_sum < 1e-10:
            y = project_simplex(y, S)
        elif abs(y_sum - S) / S > 0.1:
            y = project_simplex(y, S)
        a = x_new
        t = t_new
        
        # Spectral shaping perturbation
        if use_spectral and it % spectral_every == 0 and it > 100:
            decay = min(1.0, 100.0 / it)
            if rng.random() < 0.3 * decay:
                n_freqs = max(3, 20 - it // 500)
                y = spectral_shaping(y, n, S, rng, num_freqs=n_freqs)
                t = 1.0
        
        # Stuck detection and restart
        if it - last_improve_it > 2000:
            noise_scale = 0.005 * S / np.sqrt(n) * max(0.1, 1.0 - it / 50000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
                last_improve_it = it
            else:
                break
        
        # Periodic restart
        if it % 5000 == 0:
            decay = np.exp(-it / 30000.0)
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
    
    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

def run_lp_refinement(a_init, n_lp, S_lp, max_time, rng, num_iters=200):
    """LP-based refinement using the best-response iteration."""
    a = normalize_sum(a_init[:n_lp] if len(a_init) >= n_lp else a_init, n_lp)
    S = np.sqrt(2.0 * n_lp)
    a = normalize_sum(a, n_lp)
    
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    for it in range(num_iters):
        if time.monotonic() - start > max_time:
            break
        
        c = fft_convolve_self(a)
        M = c.max()
        if M < best:
            best = M
            best_a = a.copy()
        
        # Build the LP constraint matrix
        # A[k, j] = a[|k-j|] if |k-j| < n_lp, else 0
        # We need: A @ g <= M for all k
        # Maximize sum(g) subject to A @ g <= M, g >= 0
        
        # Build sparse matrix
        from scipy.sparse import csr_matrix
        
        rows = np.zeros(n_lp * n_lp, dtype=np.int32)
        cols = np.zeros(n_lp * n_lp, dtype=np.int32)
        vals = np.zeros(n_lp * n_lp, dtype=np.float64)
        nnz = 0
        
        for j in range(n_lp):
            for k in range(j, j + n_lp):
                if k < 2 * n_lp - 1:
                    rows[nnz] = k
                    cols[nnz] = j
                    vals[nnz] = a[k - j] if k >= j else a[j - k]
                    # Wait, this is wrong. Let me reconsider.
                    # (a * g)[k] = sum_j a[j] * g[k-j] where 0 <= j < n_lp and 0 <= k-j < n_lp
                    # So for a given j, k ranges from j to j+n_lp-1
                    # And the coefficient of g[j] in row k is a[k-j]
                    # But we need k-j to be a valid index of a, i.e., 0 <= k-j <= n_lp-1
                    # Which is satisfied since k >= j and k <= j+n_lp-1
                    nnz += 1
        
        # Actually, let me just build it directly
        # (a*g)[k] = sum_{j=max(0,k-n_lp+1)}^{min(k, n_lp-1)} a[k-j] * g[j]
        # So the coefficient of g[j] in row k is a[k-j] (when valid)
        
        # Let me rebuild
        rows_list = []
        cols_list = []
        vals_list = []
        
        for j in range(n_lp):
            for k in range(j, min(j + n_lp, 2 * n_lp - 1)):
                coeff = a[k - j]
                if coeff > 1e-15:
                    rows_list.append(k)
                    cols_list.append(j)
                    vals_list.append(coeff)
        
        A_mat = csr_matrix((vals_list, (rows_list, cols_list)), shape=(2 * n_lp - 1, n_lp))
        
        # Solve LP: max 1^T g s.t. A_mat @ g <= M, g >= 0
        c_obj = -np.ones(n_lp)  # minimize -sum(g)
        b_ub = np.full(2 * n_lp - 1, M)
        
        try:
            result = scipy.optimize.linprog(
                c_obj,
                A_ub=A_mat,
                b_ub=b_ub,
                bounds=[(0, None)] * n_lp,
                method='highs'
            )
            if not result.success:
                break
            
            g = result.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            
            # Rescale g to have sum S
            g = g * (S / g_sum)
            
            # Update
            t = 0.3  # mixing parameter
            a_new = (1 - t) * a + t * g
            a_new = project_simplex(a_new, S)
            a = a_new
            
        except Exception:
            break
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

def downsample_and_upsample(a, n_small):
    """Downsample a to n_small by averaging, then back to n by interpolation."""
    n = len(a)
    if n_small >= n:
        return a.copy()
    nblocks = n // n_small
    a_small = a[:nblocks * n_small].reshape(n_small, nblocks).mean(axis=1)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n)
    a_up = np.interp(t_new, t_old, a_small)
    return np.maximum(a_up, 0.01)

def run_topk_fista(a_init, n, S, max_time, rng, k_init=16, k_growth=0.5, k_max=256):
    """FISTA using top-k subgradient instead of LSE."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        k = min(k_max, int(k_init + k_growth * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        # Top-k gradient
        idx = np.argsort(c)[::-1][:k]
        vals = c[idx]
        w = vals / vals.sum()
        w_full = np.zeros(len(c))
        w_full[idx] = w
        
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Step size
        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it - last_improve > 3000:
            noise = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t = 1.0
            last_improve = it
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best
