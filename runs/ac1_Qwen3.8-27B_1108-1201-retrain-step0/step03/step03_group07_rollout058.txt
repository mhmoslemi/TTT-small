def soft_weights(c, temperature, top_k=None):

        w /= w.sum()
    return w

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def compute_grad_from_weights(w, a):
    """grad[m] = 2 * sum_k w[k] * a[k-m] (linear cross-correlation)."""
    n = len(a)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def get_weights(c, temperature, top_k=None):
    """Compute peak weights using soft-max."""
    if top_k is not None and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        threshold = c[idx].min()
        mask = c >= threshold
    else:
        mask = np.ones_like(c, dtype=bool)
    
    w = np.zeros_like(c)
    cm = c[mask]
    w[mask] = np.exp(temperature * (cm - cm.max()))
    total = w.sum()
    if total > 0:
        w /= total
    return w

def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def peak_equalize_polish(a0, time_budget, n_peaks=3):
    """Polish by trying to equalize the top n_peaks peaks."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    fsize = 1 << (3 * n - 2).bit_length()
    
    lr = 0.01
    
    while time.monotonic() - start < time_budget:
        # Find top peaks
        top_idx = np.argsort(c, kind='stable')[::-1][:n_peaks]
        
        # Compute combined gradient from top peaks
        w = np.zeros_like(c)
        weights = np.ones(n_peaks) / n_peaks
        for i, idx in enumerate(top_idx):
            w[idx] += weights[i]
        
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        grad = 2.0 * corr[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Multiplicative update
        a = a * np.exp(-lr * grad)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())
        
        c = fft_convolve_self(a)
        cur_obj = obj_value(a, c)
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            lr = min(lr * 1.05, 0.1)
        else:
            lr = max(lr * 0.8, 0.001)
    
    return best_a, best_obj

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def compute_grad(w, a, fsize=None):
    """grad[m] = 2 * sum_k w[k] * a[k-m] (linear cross-correlation)."""
    n = len(a)
    if fsize is None:
        fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def get_soft_weights(c, temperature, top_k=None):
    """Compute soft-max weights over peaks."""
    n_c = len(c)
    if top_k is not None and top_k < n_c:

        thresh = c[idx].min()
        mask = c >= thresh
    else:

    w = np.zeros(n_c)

    s = w.sum()
    if s > 0:
