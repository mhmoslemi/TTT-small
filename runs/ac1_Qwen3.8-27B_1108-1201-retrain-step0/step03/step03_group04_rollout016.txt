Strategy: Multi-n exploration, peak-focused subgradient with hard top-k,

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_for_lag(a, k):
    """Compute gradient of b[k] with respect to a.
    grad[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    # For each i in [0, n-1], grad[i] = 2*a[k-i] if 0 <= k-i < n
    # i.e., k-i in [0, n-1] means i in [k-n+1, k]
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        # grad[i] = 2*a[k-i] for i in [lo, hi]
        # Let j = k-i, then j ranges from k-hi to k-lo
        # grad[k-j] = 2*a[j] for j in [k-hi, k-lo]
        j_lo = k - hi
        j_hi = k - lo
        grad[lo:hi+1] = 2.0 * a[j_lo:j_hi+1][::-1]
    return grad


def compute_grad_topk(a, c, top_k):
    """Compute gradient for top-k lags (hard selection)."""
    n = len(a)
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    # Sort by value descending
    top_vals = c[top_indices]
    top_order = np.argsort(top_vals)[::-1]
    top_indices = top_indices[top_order]
    top_vals = top_vals[top_order]
    
    # Weight: focus more on the highest
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    
    # Use the max value's weight for the top-1, and spread for others
    grad = np.zeros(n)
    for idx in range(len(top_indices)):
        weight = top_vals[idx] / total
        grad += weight * compute_grad_for_lag(a, top_indices[idx])
    return grad


def peak_subgradient(a, S, max_iter, step_init, step_decay, backtracking=True, 
                     top_k=1, step_down=0.5, step_max=None):
    """Peak-focused subgradient with backtracking line search."""
    n = len(a)
    a = np.clip(a, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    best_score = np.inf
    best_local = a.copy()
    step = step_init
    stall = 0
    
    for it in range(max_iter):
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute gradient for top-k lags
        grad = compute_grad_topk(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Try the step
        if backtracking:
            # Backtracking: try step, step/2, step/4, ...
            accepted = False
            try_step = step
            for _ in range(10):
                a_new = project_simplex(a - try_step * grad, S)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < cur - 1e-14:
                    a = a_new
                    accepted = True
                    break
                try_step *= 0.5
            if not accepted:
                # No improvement even with tiny step
                if try_step < 1e-10 * S / n:
                    break
                a = project_simplex(a - try_step * grad, S)
        else:
            a = project_simplex(a - step * grad, S)
        
        # Decay step
        step *= step_decay
        
    return best_local, best_score


def fista_toppk(a_init, n, S, max_iter, top_k_init=1, top_k_rate=100, beta=100.0):
    """FISTA with progressive top-k."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    best_score = np.inf
    best_local = a.copy()
    
    for it in range(max_iter):
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = y.copy()
        
        top_k = min(50, top_k_init + it // top_k_rate)
        grad = compute_grad_topk(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = 1.0 / (beta * gnorm)
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return best_local, best_score

def make_tapered_init(n, taper_type='hann'):
    """Create a tapered initialization."""
    t = np.linspace(0, 1, n)
    if taper_type == 'hann':
        w = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif taper_type == 'blackman':
        w = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    elif taper_type == 'raised_cosine':
        w = np.sin(np.pi * t)
    elif taper_type == 'parabolic':
        w = 4 * t * (1 - t)
    else:
        w = np.ones(n)
    return np.maximum(w, 1e-10)

"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.
Strategy: Multi-n exploration with diverse initializations, aggressive peak-1
subgradient with backtracking, progressive top-k FISTA, and ultra-fine polish.
Key ideas: exploit that gradient of peak lag k is 2*a[reversed+shifted by k],
enabling very fast O(n) gradient steps with O(n log n) FFT convolution checks.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex_fast(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def grad_single_lag(a, k):
    """Gradient of b[k] w.r.t. a: grad[i] = 2*a[k-i] for valid i."""
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return np.zeros(n)
    # grad[lo..hi] = 2*a[k-hi .. k-lo] reversed
    j_lo = k - hi
    j_hi = k - lo
    result = np.zeros(n)
    result[lo:hi + 1] = 2.0 * a[j_lo:j_hi + 1][::-1]
    return result


def grad_topk(a, c, top_k):
    """Weighted gradient for top-k peak lags."""
    n = len(a)
    top_k = min(top_k, len(c))
    if top_k == 1:
        return grad_single_lag(a, int(c.argmax()))
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    grad = np.zeros(n)
    for idx, val in zip(top_idx, top_vals):
        w = val / total
        grad += w * grad_single_lag(a, int(idx))
    return grad


def normalize(a, S):
    """Clip to [0,1000] and rescale to sum S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def make_init(n, S, kind, rng=None):
    """Generate diverse initializations."""
    t = np.linspace(0, 1, n)
    if kind == 'const':
        a = np.ones(n)
    elif kind == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif kind == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    elif kind == 'parab':
        a = 4 * t * (1 - t)
    elif kind == 'sinc':
        a = np.sinc(t - 0.5) ** 2
        if a.min() < 0:

        a = rng.random(n)
    elif kind == 'bimodal':
        a = np.exp(-8 * (t - 0.25) ** 2) + np.exp(-8 * (t - 0.75) ** 2)
    else:
        a = np.ones(n)
    return normalize(a, S)
