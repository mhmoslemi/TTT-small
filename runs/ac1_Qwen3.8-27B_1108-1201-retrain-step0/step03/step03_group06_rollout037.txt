def build_toeplitz_matrix(f):
    n = len(f)
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    from scipy.sparse import csr_matrix
    return csr_matrix((vals, (rows, cols)), shape=(m, n))

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based best-response iterations.

The key insight is that given a current sequence f, we can solve a linear program
to find the "best response" g: maximize sum(g) subject to conv(f,g) <= M and g >= 0,
where M = max(conv(f,f)). Moving from f towards g (normalized) produces a sequence
with non-increasing peak autocorrelation.

We combine this with gradient-based polishing and coordinate descent for final refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def build_constraint_matrix(f):
    """Build the sparse Toeplitz constraint matrix for conv(f, g) <= M.
    
    Returns a (2n-1) x n sparse matrix F such that (F @ g)[k] = conv(f,g)[k].
    """
    n = len(f)
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(m, n))


def solve_lp_best_response(f, M, solver=None):
    """Solve the LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    
    Returns the optimal g (numpy array) or None if infeasible/failed.
    """
    n = len(f)
    F = build_constraint_matrix(f)
    
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [F @ g <= M, g >= 0]
    prob = cp.Problem(objective, constraints)
    
    try:
        if solver is not None:
            prob.solve(solver=solver)
        else:
            prob.solve()
    except Exception:
        # Try different solvers
        for s in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=s)
                break
            except Exception:
                continue
    
    if g.value is None or prob.status not in ['optimal', 'optimal_inaccurate']:
        return None
    
    g_val = np.asarray(g.value).flatten()
    g_val = np.maximum(g_val, 0)
    return g_val


def get_initialization(name, n, S, rng):
    """Generate various initializations."""
    i = np.arange(n)
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "cosine_mix":
        a = 1.0 + 0.25 * np.cos(2 * np.pi * 5 * i / n + rng.random() * 2) + \
            0.15 * np.cos(2 * np.pi * 11 * i / n + rng.random() * 2) + \
            0.1 * np.cos(2 * np.pi * 23 * i / n + rng.random() * 2)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "triangle":
        t = i / n
        a = 1.0 - 2.0 * np.abs(t - 0.5)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def run_lp_iteration(a, S, t_step=0.5, solver=None):
    """Perform one LP-based best-response iteration.
    
    Returns (new_a, old_obj, new_obj).
    """
    n = len(a)
    c = fft_convolve_self(a)
    M = c.max()
    old_obj = M
    
    g = solve_lp_best_response(a, M, solver=solver)
    if g is None or g.sum() < 1e-15:
        return a.copy(), old_obj, old_obj
    
    # Normalize g to have sum S
    g_sum = g.sum()
    g = g * (S / g_sum)
    
    # Update
    a_new = (1 - t_step) * a + t_step * g
    a_new = normalize_nonneg(a_new, S)
    
    c_new = fft_convolve_self(a_new)
    new_obj = c_new.max()
    
    return a_new, old_obj, new_obj


def run_subgradient(a, n, S, time_budget, T=100.0, lr=0.05, momentum=0.9):
    """Weighted subgradient descent with momentum."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    S_target = S
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    a_prev = a.copy()
    
    start = time.monotonic()
    step = 0
    
    while time.monotonic() - start < time_budget:
        # Lookahead
        y = a + (momentum - 1.0) / (1.0 + momentum) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S_target / s_y)
        else:
            y = a.copy()
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
        
        c_max = c.max()
        if c_max < 1e-30:
            break
        
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-30:
            w = w / w_sum
        else:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        
        # Subgradient: g[m] = 2 * sum_k w[k] * a[k-m]
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        grad = 2.0 * corr[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Line search
        lr_current = lr
        a_trial = a
        trial_obj = cur_obj
        for i in range(10):
            a_trial = a - lr_current * grad / gnorm
            a_trial = normalize_nonneg(a_trial, S_target)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_obj - 1e-12 * cur_obj:
                break
            lr_current *= 0.5
        
        a_prev = a.copy()
        a = a_trial
        step += 1
    
    return best_a, best_obj


def coordinate_descent(a, n, S, time_budget, max_sweeps=20):
    """Coordinate descent with random ordering."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(999)
    start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            original = a[i]
            if original < 1e-12:
                continue
            
            best_local = original
            best_local_obj = best_obj
            
            for factor in [0.5, 0.75, 1.33, 1.5, 0.25, 2.0, 0.6, 1.4]:
                a[i] = original * factor
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-14:
                    best_local_obj = obj_check
                    best_local = original * factor
            
            a[i] = best_local
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            
            if c.max() < best_obj - 1e-14:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        if not improved and sweep >= 3:
            break
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based best-response iterations
combined with gradient refinement and coordinate descent polish.

Key innovation: Solve a linear program at each iteration to find the best response
g that maximizes sum(g) subject to conv(f,g) <= M and g >= 0. Moving towards g
(re normalized) produces monotone improvement in the peak autocorrelation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def build_constraint_matrix(f):
    """Build sparse matrix F where (F @ g) = convolve(f, g) for len(g)=n."""
    n = len(f)
    m = 2 * n - 1
    # Column j has f[0..n-1] at rows j..j+n-1
    base = np.repeat(np.arange(n), n)
    offset = np.tile(np.arange(n), n)
    rows_idx = base + offset
    cols_idx = np.repeat(np.arange(n), n)
    vals = np.tile(f, n)
    return csr_matrix((vals, (rows_idx, cols_idx)), shape=(m, n))


def solve_lp(f, M, solver=None, timeout=60):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    F = build_constraint_matrix(f)
    g = cp.Variable(n)
    constraints = [F @ g <= M, g >= 0]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    solvers_to_try = []
    if solver is not None:
        solvers_to_try.append(solver)
    solvers_to_try.extend([cp.CLARABEL, cp.SCS, cp.ECOS, cp.PYTHON_MIP])
    
    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                g_val = np.asarray(g.value).flatten().copy()
                g_val = np.maximum(g_val, 0.0)
                if g_val.sum() > 0:
                    return g_val
        except Exception:
            continue
    return None


def run_subgradient(a, n, S, time_budget, T=200.0, lr=0.04, momentum=0.9):
    """Weighted subgradient descent with Nesterov momentum and backtracking."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    a_prev = a.copy()
    
    start = time.monotonic()
    step = 0
    no_improve = 0
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        
        # Nesterov lookahead
        y = a + (momentum - 1.0) / (1.0 + momentum) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * (S / s_y)
        else:
            y = a.copy()
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            scale = 0.01 * max(0.1, 1.0 - progress)
            rng = np.random.default_rng(step * 7 + 3)
            a = best_a + scale * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            a_prev = a.copy()
            no_improve = 0
            step += 1
            continue
        
        c_max = c.max()
        if c_max < 1e-30:
            break
        
        # Exponential peak weighting
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-30:
            w = w / w_sum
        else:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        
        # Subgradient via FFT cross-correlation
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        grad = 2.0 * corr[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        lr_cur = lr * (1.0 - 0.5 * progress)
        found_step = False
        for i in range(12):
            a_trial = a - lr_cur * grad / gnorm
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_obj - 1e-13 * cur_obj:
                a = a_trial
                found_step = True
                break
            lr_cur *= 0.5
        
        if not found_step:
            a = a - lr_cur * grad / gnorm
            a = normalize_nonneg(a, S)
        
        a_prev = best_a if cur_obj >= best_obj else a_prev
        a_prev = a.copy() if not found_step else a_prev
        a_prev = a.copy()
        step += 1
    
    return best_a, best_obj


def coordinate_descent(a, n, S, time_budget, max_sweeps=20):
    """Randomized coordinate descent with multiplicative updates."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(777)
    start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            original = a[i]
            if original < 1e-12:
                continue
            best_local = original
            best_local_obj = best_obj
            for factor in [0.5, 0.7, 1.4, 1.6, 0.3, 0.8, 1.2]:
                a[i] = original * factor
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-14:
                    best_local_obj = obj_check
                    best_local = original * factor
            a[i] = best_local
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            if c.max() < best_obj - 1e-14:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        if not improved and sweep >= 4:
            break
    return best_a, best_obj
