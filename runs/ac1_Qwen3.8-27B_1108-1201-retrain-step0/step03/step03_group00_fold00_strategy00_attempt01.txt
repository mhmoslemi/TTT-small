def normalize_sum(seq, n):
    """Rescale seq to have sum exactly √(2n)"""
    current_sum = np.sum(np.asarray(seq))
    if current_sum < 1e-15:
        return np.ones(n) * np.sqrt(2.0 * n) / n if n > 0 else np.array([])
    target = np.sqrt(2.0 * n)
    norm_factor = target / current_sum
    new_seq = (np.asarray(seq) * norm_factor).tolist()
    return new_seq


def evaluate_sequence(seq):
    if not isinstance(seq, list) or len(seq) == 0:
        return np.inf
    total_sum = sum(seq)
    if total_sum < 0.01:
        return np.inf
        
    seq_np = np.asarray(seq, dtype=np.float64)
    conv = np.convolve(seq_np, seq_np, mode="full")
    max_conv_value = conv.max()
    
    return float(2.0 * len(seq) * max_conv_value / (total_sum ** 2))


def lse_gradient(seq, c_conv, beta):
    n = len(seq)
    cm = c_conv.max()
    if cm == 0:
        return np.zeros(n)
    exponentiated_diffs = np.exp(beta * (c_conv - cm))
    total = exponentiated_diffs.sum()
    if total < 1e-300:
        return np.zeros(n)
        
    weights = exponentiated_diffs / total
    
    fsize = 1 << (n * 2 - 1 - 9).bit_length() + 9
    try:
        A = fft(seq, fsize)
        W = fft(weights, fsize)
        conj_A = np.conj(A)
        return 2.0 * np.real(ifft(W * conj_A))[:n]
    except Exception as e:
        print(f"FFT Error in LSE Gradient: {e}")
        return np.zeros(n)  # fallback if FFT fails


def topk_subgradient(seq, c, k, power=2):
    seq_np = np.expand_dims(np.asarray(seq), axis=0)
    c_vals = c.reshape(1, -1)

        grad[idx] = seq_np[idx] * c_vals[idx]((c_vals[idx]-np.mean(c_vals))**2 < np.std(c_vals)**2)
        # Apply gradient pruning to ensure performance
    return grad
