2. FISTA with adaptive LSE smoothing
3. Peak-targeted coordinate descent polishing
4. Simulated annealing restarts

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of smoothed max via LSE."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fista_optimize(a_init, n, S, max_time, rng, beta_start=5.0, beta_end=300.0):
    """FISTA with adaptive LSE."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = c_max_init = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    L = 1.0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta
        progress = min(1.0, it / max(1, int(max_time * 50)))
        beta = beta_start + (beta_end - beta_start) * progress ** 1.5
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = y.copy()
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Step size with backtracking
        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)
        
        # Simple check: did we improve?
        if it % 10 == 0:
            c_new = fft_convolve_self(x_new)
            if c_new.max() > cur_max * 1.01:
                L *= 1.5
                x_new = project_simplex(y - (step/2) * grad, S)
            else:
                L = max(L * 0.98, 1.0)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return best_a, best_score

        return a, np.inf

        it += 1
        
        if it % 100 == 0:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                stall = 0
            else:
                stall += 1

                break

        j = int(partners[top_idx])

                    c_new = fft_convolve_self(a_new)

                    else:

            keep_idx = j
        else:

            if np.any(small_mask):

                a_new[reduce_idx] -= delta
                a_new[add_idx] += delta

def spectral_init(n, rng, S):
    """Initialize with flat-ish spectrum."""
    fsize = 1 << (2 * n - 2).bit_length()
    # Flat magnitude with slight variation
    mags = np.ones(fsize)
    mags[0] = S
    # Add some structure
    for m in range(1, fsize // 2):
        mags[m] = mags[fsize - m] = 1.0 + 0.3 * rng.random()
    
    phases = rng.random(fsize) * 2 * np.pi
    phases[0] = 0
    for m in range(1, fsize // 2):
        phases[fsize - m] = -phases[m]
    
    A = mags * np.exp(1j * phases)
    a = np.real(ifft(A))[:n]
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a * (S / a.sum())
    return a


def cosine_init(n, rng, S):
    """Cosine-tapered initialization."""
    tt = np.arange(n) / (n - 1)
    p = rng.random() * 2 + 0.5
    a = (0.5 + 0.5 * np.cos(np.pi * tt)) ** p + 0.01
    a = a * (S / a.sum())
    return a

    tt = np.arange(n) / n
    center1 = 0.2 + 0.3 * rng.random()
    center2 = 0.5 + 0.3 * rng.random()

    width2 = 0.05 + 0.1 * rng.random()

def anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = mu."""
    half = n // 2
    h = rng.random(half) + 0.05
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = max(0, mu - h[i])
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())
    return a

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. Diverse initializations (spectral, cosine-tapered, bimodal, anti-palindromic, periodic)
2. FISTA with adaptive LSE smoothing and careful backtracking
3. Subgradient averaging for robust convergence
4. Peak-targeted coordinate descent for final polishing
5. Simulated annealing perturbations to escape local minima
6. Multi-length exploration (n=8192 primary, n=16384 secondary)

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_subgradient(a, c, k=1):
    """Subgradient using top-k active lags with uniform weights."""
    n = len(a)
    k = min(k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[top_indices] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def weighted_subgradient(a, c, beta_weights=20.0):
    """Subgradient with weights proportional to exp(beta*(c[k]-c_max))."""
    n = len(a)
    c_max = c.max()
    weights = np.exp(beta_weights * (c - c_max))
    total = weights.sum()
    if total < 1e-300:
        return np.zeros(n)
    weights = weights / total
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fista_optimize(a_init, n, S, max_time, rng, beta_start=5.0, beta_end=200.0,
                   use_momentum=True, momentum_decay=0.0):
    """FISTA with adaptive LSE and backtracking line search."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c_init = fft_convolve_self(a)
    best_score = c_init.max()
    best_a = a.copy()
    it = 0
    L = 2.0
    stall = 0
    last_best = best_score
    iter_since_best = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_start + (beta_end - beta_start) * (progress ** 1.2)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = y.copy()
            iter_since_best = 0
        else:
            iter_since_best += 1

        # Restart momentum if stalled
        if iter_since_best > 200:
            stall += 1
            if stall > 5:
                # Perturb and restart
                noise = 0.005 * S / np.sqrt(n) * rng.standard_normal(n)
                y = project_simplex(y + noise, S)
                a = y.copy()
                t_mom = 1.0
                stall = 0
                iter_since_best = 0
                continue

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)

        # Backtracking check every few iterations
        if it % 5 == 0:
            c_new = fft_convolve_self(x_new)
            if c_new.max() > cur_max + 1e-8:
                L *= 2.0
                step = 1.0 / L
                x_new = project_simplex(y - step * grad, S)
            elif it % 50 == 0 and c_new.max() < last_best:
                L = max(L * 0.9, 2.0)
                last_best = c_new.max()

        # Nesterov momentum
        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            t_mom = t_new
        else:
            y = x_new.copy()

        a = x_new

    return best_a, best_score

    count = 0

    while time.monotonic() - start < max_time:

def coordinate_polish(a_init, n, S, max_time, rng):
    """Coordinate descent targeting the worst lags."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    it = 0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Periodic full recheck
        if it % 50 == 0:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                stall = 0
            else:
                stall += 1
            if stall > 30:
                # Try a perturbation
                noise = 0.002 * S / np.sqrt(n) * rng.standard_normal(n)
                a_try = project_simplex(a + noise, S)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < best_score:
                    a = a_try
                    best_score = c_try.max()
                    c = c_try
                    stall = 0
                else:
                    break
            continue

        # Find top 3 lags
        top_lags = np.argsort(c)[::-1][:3]

        for lag in top_lags:
            lag = int(lag)
            i_min = max(0, lag - n + 1)
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue

            indices = np.arange(i_min, i_max + 1)
            partners = lag - indices
            products = a[indices] * a[partners]

            # Find the top contributing pair
            top_p = np.argmax(products)
            i = int(indices[top_p])
            j = int(partners[top_p])

            if i == j:
                # Reduce a[i]^2 contribution
                delta = 0.03 * a[i]
                if delta < 1e-15:
                    continue
                # Add to a low-contributing element
                med = np.median(a)
                candidates = np.where(a < med)[0]
                if len(candidates) == 0:
                    candidates = np.where(a == a.min())[0]
                add_idx = int(candidates[rng.integers(len(candidates))])

                a_try = a.copy()
                a_try[i] = max(0.0, a_try[i] - delta)
                a_try[add_idx] += delta
                a_try = a_try * (S / a_try.sum())
                c_try = fft_convolve_self(a_try)
                if c_try.max() < best_score:
                    a = a_try
                    c = c_try
                    best_score = c.max()
            else:
                # Reduce the larger element of the pair
                if a[i] >= a[j]:
                    ridx = i
                else:
                    ridx = j

                delta = 0.03 * a[ridx]
                if delta < 1e-15:
                    continue

                # Find a safe place to add
                # Avoid indices that contribute to top lags
                avoid = set()
                for tl in top_lags:
                    tl = int(tl)
                    tl_min = max(0, tl - n + 1)
                    tl_max = min(tl, n - 1)
                    for idx in range(tl_min, tl_max + 1):
                        avoid.add(idx)
                        avoid.add(tl - idx)

                candidates = np.array([x for x in range(n) if x not in avoid and a[x] < np.median(a)])
                if len(candidates) == 0:
                    candidates = np.where(a == a.min())[0]
                add_idx = int(candidates[rng.integers(len(candidates))])

                a_try = a.copy()
                a_try[ridx] = max(0.0, a_try[ridx] - delta)
                a_try[add_idx] += delta
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try * (S / s_try)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < best_score:
                    a = a_try
                    c = c_try
                    best_score = c.max()

    return a, best_score


def simulated_annealing_polish(a_init, n, S, max_time, rng, T_start=0.01, T_end=0.0001):
    """Simulated annealing on the sequence."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    cur_score = c.max()
    it = 0
    total_iters = max(1, int(max_time * 100))  # estimate ~100 iters/sec

    while time.monotonic() - start < max_time:
        it += 1
        progress = it / total_iters
        T = T_start * (T_end / T_start) ** progress

        # Random perturbation: pick 2-5 indices and redistribute
        n_perturb = rng.integers(2, 6)
        indices = rng.integers(0, n, size=n_perturb)
        deltas = rng.standard_normal(n_perturb) * (0.01 * S / n)
        deltas -= deltas.mean()  # keep sum approximately zero

        a_try = a.copy()
        for idx, d in zip(indices, deltas):
            a_try[idx] += d
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try * (S / s_try)

        c_try = fft_convolve_self(a_try)
        try_score = c_try.max()
        delta = try_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_try
            c = c_try
            cur_score = try_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score


def spectral_init(n, rng, S):
    """Initialize with near-flat spectrum."""
    fsize = 1 << (2 * n - 2).bit_length()
    mags = np.ones(fsize)
    mags[0] = S
    # Add mild random variation
    for _ in range(3):
        center = rng.integers(1, fsize // 2)
        width = rng.integers(5, fsize // 10)
        depth = rng.random() * 0.3
        for m in range(max(1, center - width), min(fsize // 2, center + width)):
            attenuation = depth * np.exp(-0.5 * ((m - center) / width) ** 2)
            mags[m] *= (1.0 - attenuation)
            mags[fsize - m] *= (1.0 - attenuation)

    phases = rng.random(fsize) * 2 * np.pi
    phases[0] = 0.0
    for m in range(1, fsize // 2 + 1):
        phases[fsize - m] = -phases[m]

    A = mags * np.exp(1j * phases)
    a = np.real(ifft(A))[:n]
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a * (S / a.sum())
    return a


def cosine_taper_init(n, rng, S):
    """Cosine-tapered (window-function-like) initialization."""
    tt = np.arange(n, dtype=np.float64) / (n - 1)
    p = 0.5 + rng.random() * 2.0
    a = (0.5 + 0.5 * np.cos(np.pi * tt)) ** p + 0.01
    a = a * (S / a.sum())
    return a


def bimodal_init(n, rng, S):
    """Two separated Gaussian bumps."""
    tt = np.arange(n, dtype=np.float64) / n
    c1 = 0.15 + 0.25 * rng.random()
    c2 = 0.55 + 0.25 * rng.random()
    w1 = 0.04 + 0.08 * rng.random()
    w2 = 0.04 + 0.08 * rng.random()
    a = np.exp(-0.5 * ((tt - c1) / w1) ** 2) + np.exp(-0.5 * ((tt - c2) / w2) ** 2)
    a += 0.005
    a = a * (S / a.sum())
    return a


def anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = mu."""
    half = n // 2
    h = 0.1 + 0.8 * rng.random(half)
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = max(0.0, mu - h[i])
    if n % 2 == 1:
        a[half] = mu / 2.0
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())
    return a


def periodic_init(n, rng, S, n_freqs=3):
    """Multi-frequency periodic initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n) * 0.8
    for _ in range(n_freqs):
        freq = int(rng.integers(2, 20))
        phase = rng.random() * 2 * np.pi
        amp = 0.15 + 0.25 * rng.random()
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def random_init(n, S, rng):
    """Random positive initialization."""
    a = rng.random(n) + 0.05
    a = a * (S / a.sum())
    return a


def w_shape_init(n, rng, S):
    """W or M shaped initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    k = int(rng.integers(2, 6))
    a = np.abs(np.sin(np.pi * k * tt)) + 0.02
    a = a * (S / a.sum())
    return a


def generate_init_batch(n, rng, S, parent=None):
    """Generate a diverse batch of initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(("parent", parent.copy()))
    inits.append(("spectral1", spectral_init(n, rng, S)))
    inits.append(("spectral2", spectral_init(n, rng, S)))
    inits.append(("cosine1", cosine_taper_init(n, rng, S)))
    inits.append(("cosine2", cosine_taper_init(n, rng, S)))
    inits.append(("bimodal1", bimodal_init(n, rng, S)))
    inits.append(("bimodal2", bimodal_init(n, rng, S)))
    inits.append(("anti_pal1", anti_palindromic_init(n, rng, S)))
    inits.append(("anti_pal2", anti_palindromic_init(n, rng, S)))
    inits.append(("periodic1", periodic_init(n, rng, S)))
    inits.append(("wshape1", w_shape_init(n, rng, S)))
    inits.append(("random1", random_init(n, S, rng)))
    return inits


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """Main search function to minimize 2*n*max(conv(a,a))/sum(a)^2."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    global_best_score = np.inf
    global_best_a = None

    # Try parent
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000).astype(np.float64)
    except Exception:
        pass

    # Primary length
    n = 8192
    S = np.sqrt(2.0 * n)

    if parent is not None:
        # Use parent's length if reasonable
        if 1024 <= len(parent) <= 32768:
            n = len(parent)
            S = np.sqrt(2.0 * n)

    print("=== Starting: n=%d, S=%.2f, parent=%s, budget=%.0fs ===" %
          (n, S, parent is not None, budget_s))

    # === Phase 1: Exploratory ===
    inits = generate_init_batch(n, rng, S, parent)
    n_inits = len(inits)
    phase1_time = (deadline - time.monotonic()) * 0.30
    per_init = max(5.0, phase1_time / n_inits)

    print("Phase 1: %d inits, ~%.1f s each" % (n_inits, per_init))

    results = []
    for name, init_a in inits:
        if time.monotonic() > deadline - 300:
            break
        t0 = time.monotonic()
        a_res, score = fista_optimize(init_a, n, S, per_init, rng,
                                       beta_start=3.0, beta_end=80.0)
        c_chk = fft_convolve_self(a_res)
        actual = c_chk.max()
        results.append((actual, name, a_res))
        if actual < global_best_score:
            global_best_score = actual
            global_best_a = a_res.copy()
            print("  [%s] %.6f (%.1fs)" % (name, actual, time.monotonic() - t0))

    results.sort(key=lambda x: x[0])
    print("Phase 1 best: %.6f (%s)" % (results[0][0], results[0][1]))

    # === Phase 2: Exploitation on top candidates ===
    remaining = deadline - time.monotonic() - 400  # reserve 400s
    top_k = min(3, len(results))
    per_ext = max(20.0, remaining / top_k)

    print("Phase 2: Extended on top %d, ~%.0f s each" % (top_k, per_ext))

    for score, name, a_cur in results[:top_k]:
        if time.monotonic() > deadline - 420:
            break
        t0 = time.monotonic()
        a_res, score2 = fista_optimize(a_cur, n, S, per_ext, rng,
                                        beta_start=20.0, beta_end=400.0,
                                        use_momentum=True)
        c_chk = fft_convolve_self(a_res)
        actual = c_chk.max()
        if actual < global_best_score:
            global_best_score = actual
            global_best_a = a_res.copy()
            print("  [%s_ext] %.6f (%.1fs)" % (name, actual, time.monotonic() - t0))

        # Also try subgradient averaging on the result
        rem2 = per_ext - (time.monotonic() - t0)
        if rem2 > 10:
            a_sg, score_sg = subgradient_average(a_res, n, S, rem2, rng, k=5)
            c_sg = fft_convolve_self(a_sg)
            if c_sg.max() < global_best_score:
                global_best_score = c_sg.max()
                global_best_a = a_sg.copy()
                print("  [%s_sg] %.6f" % (name, c_sg.max()))

    # === Phase 3: Polishing ===
    if global_best_a is not None:
        n_best = len(global_best_a)
        S_best = np.sqrt(2.0 * n_best)
        remaining = deadline - time.monotonic()
        print("Phase 3: Polishing (remaining %.0f s), current best=%.6f" %
              (remaining, global_best_score))

        # 3a: Coordinate descent
        t_polish1 = max(10.0, remaining * 0.30)
        a_pol, score_pol = coordinate_polish(global_best_a, n_best, S_best, t_polish1, rng)
        c_pol = fft_convolve_self(a_pol)
        if c_pol.max() < global_best_score:
            global_best_score = c_pol.max()
            global_best_a = a_pol.copy()
            print("  CoordDescent: %.6f" % global_best_score)

        # 3b: Simulated annealing
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            t_sa = max(10.0, remaining2 * 0.25)
            a_sa, score_sa = simulated_annealing_polish(global_best_a, n_best, S_best,
                                                         t_sa, rng)
            c_sa = fft_convolve_self(a_sa)
            if c_sa.max() < global_best_score:
                global_best_score = c_sa.max()
                global_best_a = a_sa.copy()
                print("  SimAnneal: %.6f" % global_best_score)

        # 3c: Final FISTA with very high beta
        remaining3 = deadline - time.monotonic()
        if remaining3 > 15:
            t_final = max(5.0, remaining3 * 0.5)
            a_fin, score_fin = fista_optimize(global_best_a, n_best, S_best, t_final, rng,
                                               beta_start=100.0, beta_end=800.0,
                                               use_momentum=False)
            c_fin = fft_convolve_self(a_fin)
            if c_fin.max() < global_best_score:
                global_best_score = c_fin.max()
                global_best_a = a_fin.copy()
                print("  FinalFISTA: %.6f" % global_best_score)

        # 3d: Another coordinate polish
        remaining4 = deadline - time.monotonic()
        if remaining4 > 10:
            a_pol2, score_pol2 = coordinate_polish(global_best_a, n_best, S_best,
                                                    remaining4 * 0.7, rng)
            c_pol2 = fft_convolve_self(a_pol2)
            if c_pol2.max() < global_best_score:
                global_best_score = c_pol2.max()
                global_best_a = a_pol2.copy()
                print("  CoordDescent2: %.6f" % global_best_score)

    # === Phase 4: Try longer sequence ===
    if time.monotonic() < deadline - 150:
        n_long = 16384
        S_long = np.sqrt(2.0 * n_long)
        t_long = min(120.0, max(20.0, (deadline - time.monotonic()) - 80))
        print("Phase 4: n=%d for %.0f s" % (n_long, t_long))

        a_long_init = spectral_init(n_long, rng, S_long)
        a_long, score_long = fista_optimize(a_long_init, n_long, S_long, t_long, rng,
                                             beta_start=3.0, beta_end=150.0)
        c_long = fft_convolve_self(a_long)
        obj_long = 2.0 * n_long * c_long.max() / (a_long.sum() ** 2)
        n_cur = len(global_best_a) if global_best_a is not None else n
        obj_cur = 2.0 * n_cur * global_best_score / S ** 2
        print("  n=%d obj=%.6f vs current obj=%.6f" % (n_long, obj_long, obj_cur))

        if obj_long < obj_cur:
            global_best_a = a_long
            global_best_score = c_long.max()
            # Quick polish
            if time.monotonic() < deadline - 30:
                S_l2 = np.sqrt(2.0 * n_long)
                a_lp, sp = coordinate_polish(a_long, n_long, S_l2, 30, rng)
                c_lp = fft_convolve_self(a_lp)
                if c_lp.max() < global_best_score:
                    global_best_score = c_lp.max()
                    global_best_a = a_lp.copy()
                    print("  Long polished: obj=%.6f" % (2.0 * n_long * global_best_score / S_l2 ** 2))

    # === Final verification ===
    if global_best_a is None:
        global_best_a = np.ones(1024, dtype=np.float64)

    global_best_a = np.clip(global_best_a, 0.0, 1000.0)
    if global_best_a.sum() < 0.01:
        global_best_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_best_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL: evaluate=%.6f, n=%d, sum=%.2f ===" %
          (ver if np.isfinite(ver) else float('inf'), len(final_list), sum(final_list)))

    if not np.isfinite(ver):
        print("  Fallback to uniform")
        final_list = [1.0] * 1024
        ver = evaluate_sequence(final_list)
        print("  Fallback score: %.6f" % ver)

    return final_list
