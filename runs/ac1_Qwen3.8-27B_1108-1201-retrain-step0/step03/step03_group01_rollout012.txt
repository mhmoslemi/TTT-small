Key innovations:

   then solve a linearized system to equalize them. This directly targets the "bottleneck" lags.

3. Adaptive spectral shaping: modify frequency domain to reduce peaks.
4. Coordinate descent polish: exact 1D optimization per coordinate.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta < w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def compute_lse_grad(a, c, beta):
    """Compute gradient of LSE approximation to max(c). O(n log n) via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max
    w = exp_bc / total
    # Gradient of sum_k w[k] * c[k] w.r.t. a[i]
    # = sum_k w[k] * 2*a[k-i] (for valid k-i)
    # = 2 * (w_reversed * a) evaluated appropriately
    # Actually: sum_k w[k] * 2 * a[k-i] = 2 * (flip(w) * a)[i] where flip(w) is w reversed
    # This is a cross-correlation, which is a convolution with w reversed.
    w_rev = w[::-1]
    fsize = 1 << (len(w_rev) + n - 1).bit_length()
    W = fft(w_rev, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * A))[:n]
    return grad, c_max


def mirror_descent(a, N, max_time, rng, eta_init=0.1, eta_decay=0.9999):
    """
    Mirror descent with multiplicative updates for minimizing max(conv(a,a)).
    O(n) per step (using the simple max gradient).
    """
    n = len(a)
    S = a.sum()
    a = a.copy()
    best_score = np.inf
    best_a = a.copy()
    eta = eta_init
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        
        # Gradient: 2 * a[k* - i] where k* = argmax c
        k_star = int(np.argmax(c))
        # grad[i] = 2 * a[k_star - i] if 0 <= k_star - i < n, else 0
        # This is essentially a reversed and shifted copy of a
        grad = np.zeros(n)
        # a[k_star - i] is valid when 0 <= k_star - i < n, i.e., k_star - n + 1 <= i <= k_star
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            # grad[lo:hi+1] = 2 * a[k_star - i] for i in [lo, hi]
            # = 2 * a[k_star - lo : k_star - hi + 1][::-1]
            idx = k_star - np.arange(lo, hi + 1)
            grad[lo:hi+1] = 2.0 * a[idx]
        
        # Multiplicative update: a[i] = a[i] * exp(-eta * grad[i])
        # To avoid underflow, use log domain
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= eta * grad
        # Renormalize: sum(exp(log_a)) = S
        max_log = log_a.max()
        exp_la = np.exp(log_a - max_log)
        a = S * exp_la / exp_la.sum()
        
        eta *= eta_decay
        
        if c_max < 1.5030:
            break
    
    return best_a, best_score


def lse_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=500.0, N=None):
    """FISTA with LSE smooth approximation."""
    if N is None:
        N = 1 << (2*n - 1).bit_length()
    
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        # Compute LSE gradient
        c_max = cur
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total
        
        # grad = 2 * cross_correlation(w, a)
        # grad[i] = 2 * sum_k w[k] * a[k-i] (for valid)
        # = 2 * (flip(w) * a) [i]  (convolution)
        w_rev = w[::-1]
        fsize = 1 << (2 * max(len(w_rev), n) - 1).bit_length()
        W = fft(w_rev, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * A))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size for FISTA: 1/(L*t) where L is Lipschitz constant
        # L ≈ 2 * beta * max(c) * 2 (rough estimate)
        step = 1.0 / (beta * 4.0 * c_max + 1e-10)
        
        x_new = project_simplex(y - step * grad, S)
        
        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 5000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score


def active_set_step(a, c, top_k=20, N=None):
    """
    One active-set water-filling step.
    Identifies top-k lags and solves a linearized system to equalize them.
    """
    n = len(a)
    S = a.sum()
    
    k_indices = np.argsort(c)[::-1][:top_k]
    target = c[k_indices[-1]]  # Target: reduce all top-k to this level
    
    # We want to find delta such that c[k] + 2*(conv(a, delta))[k] = target for k in k_indices
    # i.e., 2*(conv(a, delta))[k] = target - c[k] for k in k_indices
    rhs = target - c[k_indices]
    
    # Solve: minimize ||2*P_K conv(a, delta) - rhs||^2 + lambda*||delta||^2
    # where P_K extracts components at k_indices
    # 
    # Use iterative method (a few CG iterations on the normal equations)
    # Normal equations: (4*A^T P_K^T P_K A + lambda*I) delta = 4*A^T P_K^T (target_vec - P_K conv(a, 0))
    # where A is the "convolution with a" operator
    
    # Simpler approach: use a few steps of gradient descent on delta
    lam = 0.01 * S / n  # regularization
    delta = np.zeros(n)
    lr = S / (top_k * c[k_indices[0]] * n + 1.0)
    
    for _ in range(20):
        # Compute residual: r[k] = 2*conv(a, delta)[k] - rhs[k] for k in k_indices
        conv_ad = fft_cross_convolve(a, delta)
        r_full = 2.0 * conv_ad[k_indices] - rhs
        # Gradient of ||r||^2 w.r.t. delta:
        # = 2 * A^T * P_K^T * r = 2 * (P_K r) convolved with a (reversed)
        r_embedded = np.zeros(2*n-1)
        r_embedded[k_indices] = r_full
        a_rev = a[::-1]
        grad_delta = 2.0 * fft_cross_convolve(a_rev, r_embedded)[:n]
        delta -= lr * (grad_delta + 2 * lam * delta)
    
    # Ensure sum(delta) = 0 (to preserve normalization)
    delta -= delta.mean()
    
    # Find max step size t such that a + t*delta >= 0
    neg = delta < 0
    if np.any(neg):
        t_max = (a[neg] / (-delta[neg])).min()
    else:
        t_max = 1.0
    t = min(1.0, 0.5 * t_max)  # Use half of max step for safety
    
    a_new = a + t * delta
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new * (S / a_new.sum())
    
    return a_new


def coordinate_descent_sweep(a, n, S, n_sweeps=5, N=None):

            best_a = a.copy()

            t_old = a[i]

            # coefficient[k] = 2*a[k-i] for 0 <= k-i < n, k != 2i
            #                 = 0 otherwise
            #                 = 1 for k = 2i (the t^2 term)

            if lo <= hi:
                j_range = np.arange(lo, hi+1)
                coeff[j_range] = 2.0 * a[j_range - i]
            # Special case: k = 2i
            if 2*i < 2*n-1:
                coeff[2*i] = 1.0  # The t^2 coefficient (we'll handle this separately)
            
            # base[k] = c[k] - t_old^2 * 1_{k=2i} - 2*t_old*a[k-i] * 1_{valid, k!=2i}
            base = c.copy()
            # Subtract the contribution of the current a[i]
            # For k where 0 <= k-i < n and k != 2i: base[k] -= 2*t_old*a[k-i]
            # For k = 2i: base[k] -= t_old^2
            if 2*i < 2*n-1:
                base[2*i] -= t_old * t_old
            # For other valid k:
            valid_k = np.arange(lo, hi+1)
            valid_k = valid_k[valid_k != 2*i]
            if len(valid_k) > 0:
                base[valid_k] -= 2.0 * t_old * a[valid_k - i]
            
            # Now find t >= 0 minimizing max_k [base[k] + t*coeff[k] + (t^2 - t_old^2)*1_{k=2i}]
            # = max_k [base[k] + t*coeff[k]] (for k != 2i) and [base[2i] + t^2] (for k = 2i)
            
            # This is a 1D convex minimization. Use ternary search or bisection on subgradient.
            # The function is convex in t (max of convex functions: linear and quadratic).
            
            # Find the range: t in [0, t_upper]
            # t_upper: at most 2*S (since a[i] can be at most S)
            t_lo = 0.0
            t_hi = 2.0 * S  # generous upper bound
            
            # Subgradient method: find t where 0 is in subdifferential
            # Subgradient at t: convex hull of {coeff[k] for active k != 2i} union {2t for k=2i if active}
            
            # Use bisection on the "derivative" (subgradient)
            for _ in range(30):  # 30 iterations of bisection
                t_mid = 0.5 * (t_lo + t_hi)
                # Compute c values at t_mid
                c_at_mid = base + t_mid * coeff
                if 2*i < 2*n-1:
                    c_at_mid[2*i] = base[2*i] + t_mid * t_mid  # quadratic term
                # Find max
                c_max_mid = c_at_mid.max()
                # Find which lags are active (within tolerance)
                tol = 0.01 * max(1.0, c_max_mid)
                active = c_at_mid >= c_max_mid - tol
                # Subgradient: for active k != 2i: coeff[k]; for active k = 2i: 2*t_mid
                subgrads = []
                active_other = active.copy()
                if 2*i < 2*n-1:
                    active_other[2*i] = False
                    if active[2*i]:
                        subgrads.append(2.0 * t_mid)
                if np.any(active_other):
                    subgrads.extend(coeff[active_other].tolist())
                
                if not subgrads:
                    break
                
                sg_min = min(subgrads)
                sg_max = max(subgrads)
                
                if sg_min > 0:
                    # All subgradients positive, decrease t
                    t_hi = t_mid
                elif sg_max < 0:
                    # All subgradients negative, increase t
                    t_lo = t_mid
                else:
                    # 0 is in subdifferential, optimal found
                    t_lo = t_mid
                    t_hi = t_mid
                    break
            
            t_opt = 0.5 * (t_lo + t_hi)
            a[i] = t_opt
        
        # Renormalize after each sweep
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a * (S / s)
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best_score:
        best_score = c_final.max()
        best_a = a.copy()
    
    return best_a, best_score


def spectral_shaping(a, N, top_k=10, alpha=0.3, n_iter=1000, rng=None):
    """Spectral shaping to reduce peak convolution."""
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score = np.inf
    
    for it in range(n_iter):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2*n-1]
        c_max = c.max()
        
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        
        m = np.arange(N)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        
        weights = top_vals / total
        cos_eff = np.zeros(N)
        for i_k in range(len(top_idx)):
            cos_eff += weights[i_k] * np.cos(2.0 * np.pi * m * int(top_idx[i_k]) / N)
        
        P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)
        
        if c_max < 1.503:
            break
    
    return best_a, best_score


def make_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Uniform
    inits.append(("uniform", np.ones(n) * 0.5))
    
    # Random
    inits.append(("random", rng.random(n) + 0.1))
    
    # V-shaped (high at edges, low in middle)
    v_shape = np.abs(tt - 0.5) * 2.0  # 0.5 at edges, 0 at middle
    inits.append(("v_shape", v_shape + 0.1))
    
    # Inverted V (low at edges, high in middle)
    inv_v = 1.0 - np.abs(tt - 0.5) * 2.0
    inits.append(("inv_v", inv_v + 0.1))
    
    # Chirp
    inits.append(("chirp", 0.5 + 0.4 * np.sin(2*np.pi*(5*tt + tt*tt))))
    
    # Multi-sine
    inits.append(("multi_sin", 0.5 + 0.3*np.sin(2*np.pi*3*tt) + 0.2*np.sin(2*np.pi*7*tt) + 0.1*np.sin(2*np.pi*13*tt)))
    
    # Edge-concentrated
    edge = np.exp(-4.0 * np.minimum(tt, 1-tt)**2 / 0.01) + 0.1
    inits.append(("edge", edge))
    
    # Two-peak
    two_peak = np.exp(-50*(tt-0.2)**2) + np.exp(-50*(tt-0.8)**2) + 0.05
    inits.append(("two_peak", two_peak))
    
    # Alternating (high-low pattern)
    alt = np.where(np.arange(n) % 2 == 0, 0.8, 0.2)
    inits.append(("alternating", alt.astype(float)))
    
    return inits


def safe_score(a):
    """Compute the score 2*n*max_c/sum^2 safely."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return float(2 * len(a) * c.max() / (s * s))
