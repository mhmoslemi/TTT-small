import numpy as np

from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def power_sum_obj_grad(a, p):
    """
    Compute (sum_k b[k]^p)^(1/p) and its gradient w.r.t. a.
    b = a * a (linear convolution)
    """
    n = len(a)
    b = fft_convolve_self(a)
    
    # Clip to avoid issues with negative values (shouldn't happen for real sequences)
    b = np.maximum(b, 0.0)
    
    bp = b ** p
    s = bp.sum()
    obj = s ** (1.0 / p)
    
    # grad_a[j] = (1/p) * s^((1/p)-1) * sum_k p * b[k]^(p-1) * 2*a[k-j]
    #          = s^((1/p)-1) * sum_k 2 * b[k]^(p-1) * a[k-j]
    # This is 2 * s^((1/p)-1) * cross_correlation(a, b^(p-1))
    
    weights = b ** (p - 1)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    grad = grad * (s ** ((1.0/p) - 1.0))
    
    return obj, grad, b

def support_thinning(a, n, rng, max_time):
    """

    best_a = a.copy()

            break

    return best_a, best_score * (2 * n)

        temp = ce_temp_init * (ce_temp_final / ce_temp_init) ** progress

        grad = grad - grad.mean()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr_t = lr / (1.0 + 0.0005 * t)
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new, 1.0)
        
        # Check score
        b_new = fft_convolve_self(a_new)
        score_new = b_new.max()
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Periodic perturbation
        if step % 500 == 0 and step > 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.01, n), 1.0)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
    
    return best_a, best_score

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Uses Adam optimizer with multiple smooth approximations to max,
temperature annealing, and peak-directed refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_weighted_grad(a, weights):
    """Compute 2 * cross_correlation(a, weights) via FFT.
    weights has length 2n-1, result has length n.
    """
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    """

    while time.monotonic() - start < max_time:
        t += 1

        progress = min(1.0, t / max(int(max_time * 10), 1))  # rough estimate

        elapsed = time.monotonic() - start

    t = 0
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.99
    lr = 0.05
    
    while time.monotonic() - start < max_time:
        t += 1
        b = fft_convolve_self(a)
        b_pos = np.maximum(b, 1e-30)
        
        # Select top-k peaks
        k = min(top_k, len(b))
        peak_idx = np.argsort(b_pos)[::-1][:k]
        
        # Create sparse weight vector
        weights = np.zeros_like(b)
        peak_vals = b_pos[peak_idx]
        peak_weights = peak_vals / peak_vals.sum()
        for i, pi in enumerate(peak_idx):
            weights[int(pi)] = peak_weights[i]
        
        grad = compute_weighted_grad(a, weights)
        grad = grad - grad.mean()
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr_t = lr / (1.0 + 0.001 * t)
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = project_simplex(a_new, 1.0)
        
        b_new = fft_convolve_self(a_new)
        if b_new.max() < best_score:
            best_score = b_new.max()
            best_a = a_new.copy()
        
        a = a_new
        
        # Increase top_k over time
        if t % 5000 == 0:
            top_k = min(256, top_k + 5)
    
    return best_a, best_score


def safe_normalize(a, S=1.0):
    """Clamp and rescale."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def score_of(a):
    """Compute 2*n*max_b/sum_a^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return float(2.0 * n * b.max() / (S * S))
