import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_toppk(a, c, top_k):
    """Gradient using top-k active lags, computed via FFT cross-correlation."""
    n = len(a)
    if top_k > len(c):
        top_k = len(c)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_shaping_step(a, N, top_k, alpha, rng=None):
    """
    One step of spectral shaping: modify |FFT(a)|^2 to reduce peak convolution.
    Returns (new_a, c, c_max, improvement).
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    # Compute effective cosine for top-k lags
    m = np.arange(N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a, c, c_max, 0.0
    weights = top_vals / total

    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    # Scale alpha by how far above target we are
    alpha_eff = alpha * min(1.0, c_max / 1.0)
    P_new = np.maximum(P * (1.0 - alpha_eff * cos_eff), 0.0)

    # Preserve phase, modify magnitude
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]

    # Project to simplex
    a_new = project_simplex(a_new_full, a.sum())
    return a_new, c, c_max, 0.0

def nesterov_toppk_polish(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=512):

        a = a * (S / s)

    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:

        cur = c.max()

            best_local = y.copy()

def fista_lse(a_init, n, S, max_time, rng, beta_init=3.0, beta_max=300.0):

            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        step = 1.0 / (beta * 2.0 * (1.0 + 0.0005 * it))

def multi_segment_init(n, k, rng):
    """Create a multi-segment initialization with optimized heights."""
    # Use equal-width segments for simplicity
    seg_len = n // k
    remainder = n % k
    boundaries = [0]
    for i in range(k):
        boundaries.append(boundaries[-1] + seg_len + (1 if i < remainder else 0))

    # Try random heights and keep the best
    best_score = np.inf
    best_heights = None
    for trial in range(200):
        heights = rng.random(k) + 0.1
        a = np.zeros(n)
        for i in range(k):
            a[boundaries[i]:boundaries[i + 1]] = heights[i]
        c = fft_convolve_self(a)
        score = c.max() / (a.sum() ** 2 / (2 * n))
        if score < best_score:
            best_score = score
            best_heights = heights.copy()

    # Build final sequence
    a = np.zeros(n)
    for i in range(k):
        a[boundaries[i]:boundaries[i + 1]] = best_heights[i]
    return a, best_score


def make_msequence_init(n, rng):
    """Create an m-sequence inspired initialization (binary with good autocorrelation)."""
    # Generate a pseudo-random binary sequence with balanced 0s and 1s
    a = rng.random(n)
    # Apply a "chirp" modulation to create structure
    t = np.arange(n) / n
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (3 * t + 0.5 * t * t)) + 0.1 * rng.standard_normal(n)
    a = np.clip(a, 0.05, 1.0)
    return a


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
