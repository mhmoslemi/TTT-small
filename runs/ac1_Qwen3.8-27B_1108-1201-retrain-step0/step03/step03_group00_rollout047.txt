"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-resolution FISTA with adaptive beta scheduling,
lag-aware perturbations, multi-start with diverse initializations,
and final polishing with exact subgradient and L-BFGS.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_value_and_grad(c, a, beta):
    """Compute LSE value and gradient with respect to a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    lse_val = cm + np.log(e.sum()) / beta
    tot = e.sum()
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad


def exact_subgradient(a, c):
    """Exact subgradient of max(c) at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def run_fista_adaptive(a_init, n, S, max_time, rng, beta_init=10.0, mode="adaptive"):
    """
    FISTA with adaptive beta and lag-aware restarts.
    Returns (best_a, best_score).
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    beta = beta_init
    stagnation_count = 0
    
    while time.monotonic() - start < max_time - 2:
        it += 1
        
        c = fft_convolve_self(y)
        cur_max = c.max()

            best = cur_max
            best_a = y.copy()
            last_improve = it
            stagnation_count = 0
        else:
            stagnation_count += 1

                beta = min(5000.0, beta * 1.02)
            elif stagnation_count < 500:

                pass
            else:

                k_max = int(np.argmax(c))
                lo = max(0, k_max - n + 1)
                hi = min(k_max, n - 1)
                # Add targeted noise to elements contributing to the max lag
                noise = np.zeros(n)
                contrib_mask = np.zeros(n)
                contrib_mask[lo:hi+1] = 1.0

                else:
                    noise_weight = np.ones(hi - lo + 1) / (hi - lo + 1)

                a = y.copy()
                t_mom = 1.0

                last_improve = it
                stagnation_count = 0
