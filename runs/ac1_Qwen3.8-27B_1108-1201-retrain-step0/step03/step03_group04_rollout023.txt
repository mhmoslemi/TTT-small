def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)

        return v

    """Subgradient of max_k b_k w.r.t. a, using active lags."""
    n = len(a)

    return grad

    S = np.sqrt(2.0 * float(n))

    return a * (S / s)

    best_score = np.inf
    best_local = a.copy()
    it = 0
    last_best = best_score

    while time.monotonic() - start < time_budget:
        it += 1
        remaining_frac = 1.0 - (time.monotonic() - start) / max(time_budget, 1e-6)
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_obj = 2.0 * n * cur_max / (S * S)
        if cur_obj < best_score - 1e-10:
            best_score = cur_obj
            best_local = y.copy()

        top_k = min(top_k_max, top_k_start + it // top_k_growth)
        active = toppk_lags(c, top_k)
        rms = np.sqrt(np.mean(c ** 2))
        rms_w = rms_start * (rms_end / rms_start) ** ((1.0 - remaining_frac) / max(rms_decay * time_budget, 1e-6)) if rms_start > 0 else 0.0
        if rms_w < 0.001:
            rms_w = 0.0
        if rms_w > 0.01:
            rms_lag_idx = (2 * n - 1) // 2
            active = np.unique(np.concatenate([active, [rms_lag_idx]]))

        grad = compute_subgradient(y, active)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        step = 1.0 / (1.0 + decay * it)
        x_new = project_simplex(y - step * grad / gn, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum ** 2))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        if it % perturb_every == 0 and rng.random() < perturb_prob:
            scale = 0.004 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, scale, n), S)
            t_momentum = 1.0

        if it % 500 == 0 and time.monotonic() - start < time_budget - 1.0:
            check_c = fft_convolve_self(x_new)
            check_obj = 2.0 * n * check_c.max() / (S * S)
            if check_obj < best_score:
                best_score = check_obj
                best_local = x_new.copy()

    return best_local, best_score


def make_profile(shape, n):
    """Generate initial profile of given shape."""
    idx = np.arange(n, dtype=np.float64)
    t = idx / max(n - 1, 1)
    if shape == 'uniform':
        a = np.ones(n)
    elif shape == 'ramp':
        a = 0.5 + idx / (n - 1)
    elif shape == 'ramp2':
        a = 0.3 + 1.7 * (idx / (n - 1)) ** 2
    elif shape == 'cosine':
        a = 0.3 + 0.7 * np.cos(np.pi * t) ** 2
    elif shape == 'beta':
        a = (t + 0.1) * (1 - t + 0.1) + 0.1
    elif shape == 'sigmoid':
        a = 0.3 + 0.7 / (1.0 + np.exp(-10.0 * (t - 0.6)))
    elif shape == 'quad_center':
        a = 1.0 - 0.5 * (2 * t - 1) ** 2
    elif shape == 'power':
        a = (t + 0.05) ** 0.5 + 0.2
    else:
        a = np.ones(n)
    return np.maximum(a, 0.01)


def propose_candidate():
    """Main entry: search for best sequence minimizing the objective."""
    if hasattr(__builtins__, '__dict__') and isinstance(__builtins__, dict):
        given = __builtins__.get('height_sequence_1', None)
    else:
        given = globals().get('height_sequence_1', None)
    if given is None:
        try:
            given = height_sequence_1  # noqa
        except NameError:
            given = None

    rng = np.random.default_rng(12345)
    total_budget = 2380.0
    deadline = time.monotonic() + total_budget
    global_best = np.inf
    global_a = None
    global_n = 0

    def update_best(a, n):
        nonlocal global_best, global_a, global_n
        c = fft_convolve_self(a)
        obj = objective_from_conv(a, c)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            global_n = n
            return True
        return False

    if given is not None:
        given = np.asarray(given, dtype=np.float64).ravel()
        given = np.clip(given, 0, 1000)
        n_given = len(given)
        if n_given >= 64:
            a_try = normalize_seq(given, n_given)
            c_try = fft_convolve_self(a_try)
            obj_try = objective_from_conv(a_try, c_try)
            print("Given sequence: n=%d, obj=%.6f" % (n_given, obj_try))
            global_best = obj_try
            global_a = a_try.copy()
            global_n = n_given
        else:
            n_given = 8192
    else:
        n_given = 8192

    # Phase 1: Coarse-to-fine at n=512
    n_coarse = 512
    print("\n=== Phase 1: Coarse search n=%d ===" % n_coarse)
    t0 = time.monotonic()
    best_coarse = None
    best_coarse_obj = np.inf
    shapes = ['ramp', 'ramp2', 'cosine', 'beta', 'sigmoid', 'power', 'quad_center', 'uniform']
    per_shape = min(25.0, (deadline - time.monotonic()) * 0.15 / len(shapes))

    for shape in shapes:
        if time.monotonic() > deadline - 60:
            break
        a_c = make_profile(shape, n_coarse)
        a_c = normalize_seq(a_c, n_coarse)
        a_res, obj = nesterov_optimize(a_c, n_coarse, per_shape, rng,
                                       top_k_start=1, top_k_max=30,
                                       top_k_growth=20, decay=0.005,
                                       rms_start=0.3, rms_end=0.0, rms_decay=0.002)
        c_res = fft_convolve_self(a_res)
        obj_res = objective_from_conv(a_res, c_res)
        print("  %s: obj=%.6f" % (shape, obj_res))
        if obj_res < best_coarse_obj:
            best_coarse_obj = obj_res
            best_coarse = a_res.copy()
    print("  Phase 1 done: best=%.6f (%.1fs)" % (
        best_coarse_obj, time.monotonic() - t0))

    # Phase 2: Transfer to n=8192
    n_main = 8192
    print("\n=== Phase 2: Main optimization n=%d ===" % n_main)
    t0 = time.monotonic()

    # Upsample coarse profile
    if best_coarse is not None:
        idx_coarse = np.linspace(0, 1, n_coarse)
        idx_main = np.linspace(0, 1, n_main)
        profile_big = np.interp(idx_main, idx_coarse, best_coarse)
    else:
        profile_big = np.ones(n_main)

    # Multiple restarts with perturbations at n_main
    restarts = 5
    per_restart = min(300.0, (deadline - time.monotonic()) * 0.4 / restarts)

    for r in range(restarts):
        if time.monotonic() > deadline - 120:
            break
        if r == 0:
            a_start = profile_big.copy()
        elif r == 1:
            a_start = make_profile('uniform', n_main)
        elif r == 2:
            a_start = profile_big + 0.01 * S / np.sqrt(n_main) * rng.normal(0, 1, n_main)
        elif r == 3:
            a_start = make_profile('ramp2', n_main)
        else:
            # Interpolate between best coarse and uniform
            w = 0.5 + 0.3 * r
            a_start = w * profile_big + (1 - w) * np.ones(n_main)

        a_start = normalize_seq(a_start, n_main)
        a_res, obj = nesterov_optimize(a_start, n_main, per_restart, rng,
                                       top_k_start=1, top_k_max=100,
                                       top_k_growth=40, decay=0.002,
                                       rms_start=0.3, rms_end=0.0, rms_decay=0.001,
                                       perturb_every=2000, perturb_prob=0.2)
        c_res = fft_convolve_self(a_res)
        obj_res = objective_from_conv(a_res, c_res)
        improved = False
        if obj_res < global_best:
            global_best = obj_res
            global_a = a_res.copy()
            global_n = n_main
            improved = True
        print("  Restart %d: obj=%.6f%s (%.0fs)" % (
            r, obj_res, " *" if improved else "", time.monotonic() - t0))

    # Phase 3: Polish the best
    print("\n=== Phase 3: Polishing ===")
    t0 = time.monotonic()
    remaining = deadline - time.monotonic() - 30

    if remaining > 30 and global_a is not None:
        # High-precision polish with large top-k
        a_p1, obj = nesterov_optimize(global_a, n_main, remaining * 0.4, rng,
                                       top_k_start=50, top_k_max=500,
                                       top_k_growth=100, decay=0.001,
                                       rms_start=0.0, persiturb_prob=0.0)
        # Fix: use correct params
        a_p1, obj1 = nesterov_optimize(global_a, n_main, remaining * 0.4, rng,
                                       top_k_start=20, top_k_max=300,
                                       top_k_growth=200, decay=0.0005,
                                       rms_start=0.0, rms_end=0.0,
                                       rms_decay=0.01,
                                       perturb_every=5000, perturb_prob=0.05)
        c_p1 = fft_convolve_self(a_p1)
        obj_p1 = objective_from_conv(a_p1, c_p1)
        if obj_p1 < global_best:
            global_best = obj_p1
            global_a = a_p1.copy()
            print("  Polish 1: %.6f" % obj_p1)

        # Second pass with different params
        remaining2 = deadline - time.monotonic() - 30
        if remaining2 > 15:
            a_p2, obj2 = nesterov_optimize(global_a, n_main, remaining2, rng,
                                           top_k_start=1, top_k_max=50,
                                           top_k_growth=50, decay=0.0005,
                                           rms_start=0.1, rms_end=0.0,
                                           rms_decay=0.005,
                                           perturb_every=3000, perturb_prob=0.1)
            c_p2 = fft_convolve_self(a_p2)
            obj_p2 = objective_from_conv(a_p2, c_p2)
            if obj_p2 < global_best:
                global_best = obj_p2
                global_a = a_p2.copy()
                print("  Polish 2: %.6f" % obj_p2)

    # Phase 4: Try smaller n (4096) if time allows
    if time.monotonic() < deadline - 120:
        print("\n=== Phase 4: Try n=4096 ===")
        n_alt = 4096
        t0 = time.monotonic()
        a_alt = make_profile('ramp', n_alt)
        a_alt = normalize_seq(a_alt, n_alt)
        # Also try the global best downsampled
        if global_a is not None:
            idx_big = np.linspace(0, 1, len(global_a))
            idx_alt = np.linspace(0, 1, n_alt)
            a_alt_best = np.interp(idx_alt, idx_big, global_a)
            a_alt_best = normalize_seq(a_alt_best, n_alt)
        else:
            a_alt_best = a_alt

        for trial, a_init in enumerate([a_alt, a_alt_best]):
            if time.monotonic() > deadline - 60:
                break
            per_trial = min(90.0, (deadline - time.monotonic()) * 0.15)
            a_res, obj = nesterov_optimize(a_init, n_alt, per_trial, rng,
                                           top_k_start=1, top_k_max=80,
                                           top_k_growth=35, decay=0.002,
                                           rms_start=0.2, rms_end=0.0,
                                           rms_decay=0.002)
            c_res = fft_convolve_self(a_res)
            obj_res = objective_from_conv(a_res, c_res)
            if obj_res < global_best:
                global_best = obj_res
                global_a = a_res.copy()
                global_n = n_alt
                print("  n=4096 trial %d: %.6f *" % (trial, obj_res))
            else:
                print("  n=4096 trial %d: %.6f" % (trial, obj_res))

    # Final verification and output
    if global_a is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in global_a]
        # Verify
        try:
            ver = evaluate_sequence(final_list)
            if np.isfinite(ver):
                print("\nFINAL: n=%d, obj=%.6f (verified: %.6f)" % (
                    len(final_list), global_best, ver))
                return final_list
        except Exception as e:
            print("Verification error: %s" % str(e))

        print("\nFINAL: n=%d, obj=%.6f" % (len(final_list), global_best))
        return final_list

    # Fallback: uniform sequence
    print("FALLBACK: uniform sequence")
    fallback = [1.0] * 256
    return fallback
