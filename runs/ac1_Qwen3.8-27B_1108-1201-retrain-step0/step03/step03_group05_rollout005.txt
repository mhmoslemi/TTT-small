"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- Multi-scale L-BFGS-B with log-sum-exp continuation
- LP-based direction finding for escaping local minima
- Nesterov-accelerated subgradient polishing

- Cross-pollination and perturbation strategies

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize, linprog
import scipy.sparse as sp


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=200):
    """Run L-BFGS-B with continuation on beta."""
    if beta_schedule is None:
        beta_schedule = [3, 5, 10, 20, 50, 100, 200, 400, 800, 1500, 3000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break

        for attempt in range(2 if beta >= 200 else 1):
            if time.monotonic() - start > max_time - 3:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.02, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            t0 = time.monotonic()
            try:
                result = scipy_minimize(fun, z, jac=jac,
                    method='L-BFGS-B',
                    options={'maxiter': maxiter_per_beta, 'maxfun': maxiter_per_beta * 2,
                             'ftol': 1e-14, 'gtol': 1e-9, 'maxls': 40, 'iprint': -1})
            except Exception:
                break

            z = result.x - result.x.mean()
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def solve_lp_direction(a, active_indices, M, n):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in active_indices, g >= 0.
    Returns normalized g (sum = 1) or None.
    """
    K = len(active_indices)
    if K == 0:
        return np.ones(n) / n

    # Build sparse constraint matrix
    row_list = []
    col_list = []
    val_list = []

    for idx, k in enumerate(active_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k) + 1
        if j_start < j_end:
            j_vals = np.arange(j_start, j_end)
            row_list.extend([idx] * len(j_vals))
            col_list.extend(j_vals)
            val_list.extend(a[k - j_vals])

    if not row_list:
        return np.ones(n) / n

    A = sp.csr_matrix((val_list, (row_list, col_list)), shape=(K, n))

    c_obj = -np.ones(n)
    b_ub = np.full(K, M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs',
                         options={'presolve': True, 'time_limit': 30})
    except Exception as e:
        print("  LP error: %s" % str(e))
        return None

    if result.success:
        g = result.x
        s = g.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return g / s
    return None

                grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
                step = 0.01 * S / n
                a_candidate = project_simplex(a - step * grad, S)
                a = a_candidate

        if it % 10 == 0:
            print("    LP iter %d: score=%.8f" % (it, best_score))

    return best_a, best_score


def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=256):
    """Nesterov-accelerated top-k subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 80)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 5000 == 0 and rng.random() < 0.08:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t_mom = 1.0

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_score(a):
    """Compute the objective value 2*n*max_b/sum_a^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)
