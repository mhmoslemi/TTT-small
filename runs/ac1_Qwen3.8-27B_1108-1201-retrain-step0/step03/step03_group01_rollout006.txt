"""

Key improvements over previous:

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights_full):
    """Compute gradient using FFT cross-correlation with given weights."""
    n = len(a)
    fsize = 1 << (len(weights_full) + n - 2).bit_length()
    W = fft(weights_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_shaping_exp(a, N, top_k, alpha, weight_momentum=0.9):
    """
    One step of spectral shaping with exponential damping.
    Returns (new_a, c, c_max).
    """
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()
    if c_max < 1e-30:
        return a, c, c_max

    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a, c, c_max
    weights = top_vals / total

    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    # Exponential damping: P_new = P * exp(-alpha * max(cos_eff, 0))
    # Only damp where cos_eff is positive (contributing to peaks)
    damping = np.exp(-alpha * np.maximum(cos_eff, 0.0))
    P_new = P * damping

    # Ensure non-negative spectrum
    P_new = np.maximum(P_new, 1e-30)

    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new_full, S)
    return a_new, c, c_max


def fista_power_sum(a_init, n, S, max_time, rng, p_init=8, p_max=128,
                    lr_init=None, target_max=None):
    """
    FISTA with power-sum approximation: minimize (sum c_i^p)^(1/p).
    Adaptive p increases over iterations for sharper max approximation.
    Backtracking line search for step size.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    fsize = 1 << (2 * n - 1).bit_length()

    if lr_init is None:
        lr_init = S / (n * 2.0)

    lr = lr_init

    while time.monotonic() - start < max_time:
        it += 1
        p = min(p_max, p_init * (1.0 + 0.01 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        if target_max is not None and cur <= target_max:
            break

        # Power-sum gradient
        c_max = c.max()
        # Use shifted and scaled for numerical stability
        c_shifted = c / max(c_max, 1e-10)
        c_pow = np.maximum(c_shifted, 0.0) ** (p - 1)
        total = c_pow.sum()
        if total < 1e-300:
            break
        w = c_pow / total

        # Gradient via FFT
        grad = compute_gradient_fft(y, w * (p / (p * c_max)) * c_max)
        # Simplify: grad = 2 * p / p * (c^(p-1)/sum c^p) cross-correlated with a
        # Actually let's just use: grad_i = 2 * sum_k c[k]^(p-1)/sum(c^p) * a[k-i] (with bounds)
        # Which is what compute_gradient_fft does with the right weights

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search (simplified: just use fixed step with decay)
        step = lr / (1.0 + 0.005 * it)
        step = max(step, 1e-8)

        # Direction: normalize gradient to unit norm for consistent step size
        direction = -grad / max(gnorm, 1e-15)
        x_new = project_simplex(y + step * direction * gnorm, S)
        # Actually let's use: x_new = project(y - step * grad)
        x_new = project_simplex(y - step * grad, S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Random perturbation to escape local minima
        if it % 1500 == 0 and rng.random() < 0.15:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def subgradient_decreasing_k(a_init, n, S, max_time, rng, k_start=256, k_end=4,
                             lr_init=None):
    """
    Subgradient method with decreasing top-k.
    Starts with broad focus (large k) to flatten overall shape,
    then narrows to focus on the most important peaks.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    fsize = 1 << (2 * n - 1).bit_length()

    if lr_init is None:
        lr_init = S / (n * 4.0)

    lr = lr_init
    total_iters_estimate = 5000  # rough estimate for scheduling k

    while time.monotonic() - start < max_time:
        it += 1
        # Decrease k over time
        frac = min(1.0, it / max(total_iters_estimate, 1))
        k_curr = int(k_start * (1 - frac) + k_end * frac)
        k_curr = max(k_curr, 2)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        top_idx = np.argsort(c)[::-1][:k_curr]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            w_full[int(k)] = weights[i]

        grad = compute_gradient_fft(y, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = lr / (1.0 + 0.003 * it)
        step = max(step, 1e-9)

        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 2000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def lse_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=1000.0):
    """FISTA with log-sum-exp smooth approximation, high beta for final polish."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:

    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.02 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        c_max = c.max()
        # Numerically stable softmax
        c_shifted = c - c_max
        exp_bc = np.exp(beta * c_shifted)
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        grad = compute_gradient_fft(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta * (1.0 + 0.001 * it))
        step = max(step, 1e-8)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

            stall_count = 0
            # Also try a fresh spectral burst from perturbed state
            for _ in range(200):
                a_curr, _, _ = spectral_shaping_exp(a_curr, N, 16, 0.1)
            a_curr = safe_normalize(a_curr, n)

        if outer_it % 10 == 0 and time.monotonic() - last_print > 30:
            print("  [Phase2 outer %d] Best: %.6f" % (outer_it, global_best))
            last_print = time.monotonic()

    print("=== Phase 2 done. Best: %.6f ===" % global_best)

    # --- Phase 3: Decreasing-k subgradient + LSE polish ---
    print("=== Phase 3: Focused polish ===")
    t_phase3 = time.monotonic()
    remaining3 = deadline - t_phase3

    if global_a is not None and remaining3 > 10:
        a_curr = global_a.copy()

        # Decreasing-k subgradient (broad to narrow)
        seg1 = min(remaining3 * 0.35, 200.0)
        if seg1 > 5:
            a_sg, sc_sg = subgradient_decreasing_k(a_curr, n, S, seg1, rng,
                                                    k_start=256, k_end=4)
            a_sg = safe_normalize(a_sg, n)
            c_sg = fft_convolve_self(a_sg)
            if c_sg.max() < global_best:
                global_best = c_sg.max()
                global_a = a_sg.copy()
                print("  Subgradient dec-k: %.6f" % global_best)
            a_curr = global_a.copy()

        # High-beta LSE for final peak reduction
        rem4 = deadline - time.monotonic()
        if rem4 > 5:
            seg2 = min(rem4 * 0.3, 100.0)
            a_l2, sc_l2 = lse_fista(a_curr, n, S, seg2, rng,
                                     beta_init=100.0, beta_max=2000.0)
            a_l2 = safe_normalize(a_l2, n)
            c_l2 = fft_convolve_self(a_l2)
            if c_l2.max() < global_best:
                global_best = c_l2.max()
                global_a = a_l2.copy()
                print("  LSE high-beta: %.6f" % global_best)
            a_curr = global_a.copy()

        # Final spectral shaping with very small alpha (fine tuning)
        rem5 = deadline - time.monotonic()
        if rem5 > 5:
            n_final = min(int(rem5 * 0.5 / 0.003), 5000)
            a_final = a_curr.copy()
            for _ in range(n_final):
                if time.monotonic() > deadline - 10:
                    break
                a_final, c_fin, c_max_fin = spectral_shaping_exp(a_final, N, 32, 0.03)
                if c_max_fin < global_best:
                    global_best = c_max_fin
                    global_a = a_final.copy()
            a_final = safe_normalize(a_final, n)
            c_final2 = fft_convolve_self(a_final)
            if c_final2.max() < global_best:
                global_best = c_final2.max()
                global_a = a_final.copy()
                print("  Final SS fine: %.6f" % global_best)

        # One more LSE with very high beta
        rem6 = deadline - time.monotonic()
        if rem6 > 5:
            a_l3, _ = lse_fista(global_a, n, S, min(rem6, 60.0), rng,
                                 beta_init=500.0, beta_max=5000.0)
            a_l3 = safe_normalize(a_l3, n)
            c_l3 = fft_convolve_self(a_l3)
            if c_l3.max() < global_best:
                global_best = c_l3.max()
                global_a = a_l3.copy()
                print("  Final LSE: %.6f" % global_best)

    print("=== Phase 3 done. Best: %.6f ===" % global_best)

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("Verification: inf, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
