"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key insight: ramp a_i = i/(n-1) gives ~4/3. We start from this and optimize carefully.
Uses LSE smooth approximation with high beta, safe updates, and coordinate descent polish.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, n=fsize)
    c = np.real(ifft(A * A, n=fsize))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, n=fsize)
    A = fft(a, n=fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A), n=fsize))[:n]
    return grad


def topk_subgradient(a, c, top_k):
    """Subgradient using top-k active lags with uniform weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    w_full = np.zeros(2 * n - 1)
    w_full[top_indices] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, n=fsize)
    A = fft(a, n=fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A), n=fsize))[:n]
    return grad


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def safe_normalize(a, S_target=None):
    """Clamp to [0, 1000], normalize to sum = sqrt(2n) if S_target is None."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    if S_target is None:
        S_target = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S_target / n
    return a * (S_target / s)


def make_ramp(n):
    """Simple ramp: a_i = i/(n-1). Gives objective ~4/3."""
    a = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    return a


def make_power_law(n, p=1.0):
    """Power law: a_i = (i+1)^p, normalized."""
    a = (np.arange(n, dtype=np.float64) + 1.0) ** p
    a = a / a.sum()
    return a


def make_ramp_oscillation(n, alpha=0.1, freq=1.0):
    """Ramp with small oscillation: a_i = i/(n-1) + alpha*sin(2*pi*freq*i/n)/n."""
    base = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    tt = np.arange(n, dtype=np.float64) / n
    osc = alpha * np.sin(2 * np.pi * freq * tt) / n
    a = base + osc
    a = np.maximum(a, 0.0)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a

    best_obj = 2.0 * n * best_max_b / (S * S)

        beta = beta_start + (beta_end - beta_start) * frac

            break

        # Adaptive step size
        step = 2.0 * S / (n * (1.0 + 0.001 * it) * max(gnorm / (S / n), 0.1))
        x_new = project_simplex(a - step * grad, S)

        # Only accept if it improves
        c_new = fft_convolve_self(x_new)
        if c_new.max() < cur_max:
            a = x_new
            if c_new.max() < best_max_b:
                best_max_b = c_new.max()
                best_a = a.copy()
                best_obj = 2.0 * n * best_max_b / (S * S)
        else:
            # Halve step
            step *= 0.7
            x_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur_max:
                a = x_new
                if c_new.max() < best_max_b:
                    best_max_b = c_new.max()
                    best_a = a.copy()
            else:
                step *= 0.5

    return best_a, best_obj


def run_topk_optimize(a_init, n, S, max_time, rng, top_k=5):
    """Top-k subgradient with safe updates."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = make_ramp(n) * (S / make_ramp(n).sum())
    else:
        a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_max_b = c.max()
    best_a = a.copy()

    step = 2.0 * S / (n * 20.0)
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_max_b - 1e-12:
            best_max_b = cur_max
            best_a = a.copy()

        grad = topk_subgradient(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(x_new)
        if c_new.max() < cur_max:
            a = x_new
            step *= 1.05
        else:
            step *= 0.8

        # Periodic perturbation
        if it % 1000 == 0 and rng.random() < 0.2:
            noise_scale = 0.003 * S / np.sqrt(n)
            a_try = project_simplex(a + rng.normal(0, noise_scale, n), S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < best_max_b:
                best_a = a_try.copy()
                best_max_b = c_try.max()
                a = a_try

    return best_a, 2.0 * n * best_max_b / (S * S)


def coordinate_descent_polish(a, n, S, max_time, rng, num_passes=10):
    """Coordinate descent: try perturbing individual coordinates."""
    a = np.clip(a, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_max_b = c.max()
    best_a = a.copy()

    # Precompute useful info
    delta_candidates = [0.001, 0.005, 0.01, 0.02, 0.05, -0.001, -0.005, -0.01, -0.02, -0.05]
    delta_candidates = [d * S / n for d in delta_candidates]

    pass_count = 0
    while time.monotonic() - start < max_time and pass_count < num_passes:
        pass_count += 1
        # Random subset of coordinates to try
        if n > 2048:
            subset_size = min(n, 512 * pass_count)
            indices = rng.choice(n, size=subset_size, replace=False)
        else:
            indices = np.arange(n)

        improved = False
        for i in indices:
            if time.monotonic() - start > max_time - 5:
                break
            for delta in delta_candidates:
                a_try = a.copy()
                a_try[i] = max(0.0, a_try[i] + delta)
                # Renormalize
                a_try = a_try * (S / a_try.sum())
                c_try = fft_convolve_self(a_try)
                if c_try.max() < best_max_b - 1e-10:
                    a = a_try
                    best_max_b = c_try.max()
                    best_a = a.copy()
                    improved = True
                    break
        if not improved and pass_count > 3:
            # Try larger perturbations
            large_deltas = [d * S / n for d in [0.1, 0.2, 0.5, -0.1, -0.2, -0.5]]
            for i in indices[:128]:
                if time.monotonic() - start > max_time - 5:
                    break
                for delta in large_deltas:
                    a_try = a.copy()
                    a_try[i] = max(0.0, a_try[i] + delta)
                    a_try = a_try * (S / a_try.sum())
                    c_try = fft_convolve_self(a_try)
                    if c_try.max() < best_max_b - 1e-10:
                        a = a_try
                        best_max_b = c_try.max()
                        best_a = a.copy()
                        break

    return best_a, 2.0 * n * best_max_b / (S * S)


def pair_exchange_polish(a, n, S, max_time, rng):
    """Try exchanging mass between pairs of coordinates."""
    a = np.clip(a, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_max_b = c.max()
    best_a = a.copy()

    # Find the peak lag
    peak_lag = int(np.argmax(c))

    # For the peak lag, find pairs that contribute most
    i_min = max(0, peak_lag - (n - 1))
    i_max = min(peak_lag, n - 1)
    if i_min > i_max:
        return best_a, 2.0 * n * best_max_b / (S * S)

    indices = np.arange(i_min, i_max + 1)
    partners = peak_lag - indices
    products = a[indices] * a[partners]

    # Try moving mass from high-product pairs to low-product pairs
    top_prod = np.argsort(products)[::-1][:32]
    bot_prod = np.argsort(products)[:32]

    for ti in top_prod:
        if time.monotonic() - start > max_time - 5:
            break
        for bi in bot_prod:
            if time.monotonic() - start > max_time - 5:
                break
            si = indices[ti]
            sp = partners[ti]
            bi_idx = indices[bi]
            bp = partners[bi]

            # Move small amount from si to bi_idx
            for frac in [0.01, 0.05, 0.1, 0.2]:
                amt = frac * a[si]
                if amt > a[si] * 0.5:
                    continue
                a_try = a.copy()
                a_try[si] -= amt
                a_try[bi_idx] += amt
                a_try = np.maximum(a_try, 0.0)
                a_try = a_try * (S / a_try.sum())
                c_try = fft_convolve_self(a_try)
                if c_try.max() < best_max_b - 1e-10:
                    a = a_try
                    best_max_b = c_try.max()
                    best_a = a.copy()
                    break

    return best_a, 2.0 * n * best_max_b / (S * S)
