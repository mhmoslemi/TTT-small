from scipy.sparse import csr_matrix

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_convolve_ab(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]

def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

    n = len(a)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_ab(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]

    n = len(a)

    n = len(a)

    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    for j in range(lo, hi + 1):
        v[j] = a[k - j]
    return v

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_ab(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def get_conv_row(a, k):
    """Get the k-th row of the convolution matrix: row_j = a[k-j] if valid."""
    n = len(a)
    row = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:

    return row

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding with
cutting plane separation, followed by mirror descent polishing.
The LP finds b maximizing sum(b) subject to conv(a,b) <= current_max, providing
a principled update direction that simultaneously addresses all peak constraints.
Cutting plane keeps LP tractable by only including active (binding) peak constraints.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_ab(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def get_conv_row(a, k):
    """Get the k-th row of the convolution constraint matrix.
    Returns row[j] = a[k-j] if 0<=k-j<n, else 0."""
    n = len(a)
    row = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        row[lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return row


def lp_direction(a, M, max_rounds=25, max_active=250, solver_name=None):
    """
    Find b >= 0 maximizing sum(b) subject to conv(a,b)[k] <= M for all k.
    Uses cutting plane: start with top peaks, iteratively add violated constraints.
    """
    n = len(a)
    c_aa = fft_convolve_self(a)

    # Initial active set: top peaks
    n_init = min(15, 2 * n - 1)
    active_k = list(np.argsort(c_aa)[::-1][:n_init])

    b_val = None
    for round_num in range(max_rounds):
        b = cp.Variable(n, nonneg=True)
        constraints = []
        for k in active_k:
            row = get_conv_row(a, int(k))
            if np.any(row != 0):
                constraints.append(row @ b <= M)

        if not constraints:
            return a.copy()

        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        solved = False
        if solver_name is not None:
            try:
                prob.solve(solver=solver_name, verbose=False)
                solved = True
            except Exception:
                pass
        if not solved:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
                solved = True
            except Exception:
                pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                return None

        if b.value is None:
            return None

        b_val = np.maximum(b.value, 0).copy()

        # Separation: check all constraints via FFT
        conv_ab = fft_convolve_ab(a, b_val)
        violations = conv_ab - M
        max_viol = violations.max()

        if max_viol < 1e-3 * M + 1e-12:
            return b_val

        # Add violated constraints
        violated_mask = violations > 1e-6 * M
        if not np.any(violated_mask):
            return b_val

        violated_idx = np.where(violated_mask)[0]
        sorted_viol = violated_idx[np.argsort(violations[violated_idx])[::-1]]

        added = 0
        for k in sorted_viol:
            if k not in active_k:
                active_k.append(int(k))
                added += 1
            if added >= 20:
                break

        if len(active_k) > max_active:
            # Keep only the most violated active constraints
            active_k = active_k[:max_active]

    return b_val


def line_search_blend(a, b, S):
    """Find optimal blend parameter t for a_new = (1-t)*a + t*b to minimize max(self-conv)."""
    best_t = 0.0
    best_max_conv = np.inf

    # Coarse search
    for t in np.linspace(0.0, 1.0, 21):
        a_new = (1 - t) * a + t * b
        c_new = fft_convolve_self(a_new)
        mc = c_new.max()
        if mc < best_max_conv:
            best_max_conv = mc
            best_t = t

    # Fine search around best
    if 0 < best_t < 1:
        for t in np.linspace(max(0.0, best_t - 0.05), min(1.0, best_t + 0.05), 21):
            a_new = (1 - t) * a + t * b
            c_new = fft_convolve_self(a_new)
            mc = c_new.max()
            if mc < best_max_conv:
                best_max_conv = mc
                best_t = t

    return best_t


def compute_subgradient(c, a, top_k):
    """Compute subgradient of max of top-k peaks of c with respect to a."""
    n = len(a)
    top_k = max(1, min(top_k, len(c)))
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]

    w_full = np.zeros(len(c))
    w_full[top_idx] = 1.0 / top_k

    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1 - 1).bit_length()
    W = fft(w_full, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def mirror_descent_polish(a0, time_budget, eta0=0.12, top_k_start=25, top_k_end=1,
                          momentum=0.92, verbose=False):
    """Mirror descent (multiplicative updates) for final refinement."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()

    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    best_iter = 0
    v = np.zeros(n)
    eta = eta0
    iter_no = 0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        progress = min(1.0, iter_no / 5000.0)
        top_k_cur = max(1, int(top_k_start + (top_k_end - top_k_start) * progress))

        grad = compute_subgradient(c, a, top_k_cur)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        v = momentum * v + (1 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        # Multiplicative update with backtracking
        accepted = False
        for bt in range(7):
            eta_try = eta * (0.7 ** bt)
            a_trial = a * np.exp(-eta_try * v)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < c.max() * (1 - 1e-12):
                a = a_trial
                c = c_trial
                accepted = True
                break

        if not accepted:
            a = a * np.exp(-eta * 0.05 * v)
            a = np.maximum(a, 1e-300)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)

        cur_obj = obj_value(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
            eta = min(eta * 1.003, eta0 * 3.0)
        else:
            eta = max(eta * 0.97, eta0 * 0.1)

        if iter_no - best_iter > 1000:
            rng_tmp = np.random.default_rng(iter_no)
            noise = rng_tmp.randn(n) * 0.008 * S / np.sqrt(n)
            a = np.maximum(a + noise, 0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.4
            eta = eta0
            best_iter = iter_no

        iter_no += 1
        eta *= 0.99998

        if verbose and iter_no % 300 == 0:
            elapsed = time.monotonic() - start
            print(f"      MD iter={iter_no}, obj={cur_obj:.7f}, best={best_obj:.7f}, t={elapsed:.1f}s")

    return best_a, best_obj


def single_peak_polish(a0, time_budget, lr=0.003):
    """Focus on reducing the single worst peak using multiplicative updates."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    v = np.zeros(n)
    start = time.monotonic()
    iter_no = 0

    while time.monotonic() - start < time_budget:
        k_max = int(np.argmax(c))

        # Gradient of c[k_max] w.r.t. a[m] = 2*a[k_max - m] for valid m
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[k_max - m]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        v = 0.88 * v + 0.12 * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        a = a * np.exp(-lr * v)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())
        c = fft_convolve_self(a)

        cur_obj = obj_value(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        iter_no += 1

    return best_a, best_obj


def coordinate_descent_polish(a0, S, time_budget, max_sweeps=50):
    """Coordinate descent: try reducing peaks by adjusting individual elements."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        # Focus on positions that contribute to the top peaks
        top_peaks = np.argsort(c)[::-1][:20]
        # Find which a[i] contribute most to these peaks
        contributing = set()
        for k in top_peaks:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1):
                if a[m] > 0.01 * S / n:
                    contributing.add(m)

        idx_list = list(contributing)
        rng_sweep = np.random.default_rng(sweep)
        rng_sweep.shuffle(idx_list)

        for i in idx_list:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            for factor in [0.6, 0.75, 0.9, 1.1, 1.3]:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_norm = a * (S / s)
                c_new = fft_convolve_self(a_norm)
                if c_new.max() < c.max() * (1 - 1e-10):
                    a = a_norm
                    c = c_new
                    obj_new = obj_value(a, c)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a.copy()
                    improved = True
                    break
            a[i] = a[i]  # keep current (already updated if improved)

        if not improved:
            break

    return best_a, best_obj

        best_state['a'] = np.ones(256)

    final_a = np.clip(best_state['a'].astype(np.float64), 0.0, 1000.0)
    if final_a.sum() < 0.01:
        final_a = np.ones(256)

    final_list = final_a.tolist()
    ver = evaluate_sequence(final_list)
    print(f"\nFINAL: tracked={best_state['obj']:.7f}, verified={ver:.7f}, n={len(final_list)}")

    if not np.isfinite(ver):
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
        print(f"FALLBACK: obj={ver:.6f}")

    return final_list
