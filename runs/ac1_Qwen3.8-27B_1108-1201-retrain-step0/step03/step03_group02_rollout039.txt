```python
"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
Improved approach: adaptive FISTA + LP dual + structured perturbations + multi-n search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def objective(a):
    """Compute 2n*max(c)/S^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S*S)

def grad_max_smooth(a, c, beta):
    """Gradient of smooth max approximation."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def grad_toppk(a, c, k):
    """Gradient using top-k peaks."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2*n - 1)
    for i, idx in enumerate(top_idx):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad
```

```python
"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
Uses adaptive FISTA, LP-dual refinement (multiplicative weights), structured
perturbations, and multi-n search. Builds on parent sequence with improved
local optimization and escape mechanisms.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full cross-convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def grad_smooth_max(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a.
    Uses the identity: d/da [sum_k w_k * c[k]] = 2 * (w_reversed * a) cross-correlation.
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def grad_toppk(a, c, k):
    """Gradient using top-k active lags."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_objective(a):
    """Compute 2n*max(c)/S^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def run_fista_adaptive(a_init, n, S, max_time, rng, beta_init=2.0, beta_max=500.0):
    """FISTA with adaptive beta and projected gradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_coeff = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stagnation_counter = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: increase steadily
        beta = min(beta_max, beta_init * np.exp(0.005 * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        grad = grad_smooth_max(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Adaptive step size with backtracking
        step = 1.0 / (beta + 1.0)
        x_new = project_simplex(y - step * grad, S)

        # Verify improvement with backtracking
        for _ in range(5):
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max * 1.001 + 1e-10:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_coeff * t_coeff))
        y = x_new + ((t_coeff - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_coeff = t_new

        # Structured perturbation to escape local minima
        if stagnation_counter > 500 and rng.random() < 0.3:
            freq = rng.choice([1, 2, 3, 5, 7, 11])
            tt = np.linspace(0, 1, n)
            perturb = 0.02 * S / np.sqrt(n) * np.sin(2 * np.pi * freq * tt + rng.random() * 2 * np.pi)
            y = project_simplex(y + perturb, S)
            t_coeff = 1.0
            stagnation_counter = 0

    return best_local, best_score

def run_lp_dual_refinement(a_init, n, S, max_time, rng):
    """
    LP-dual refinement using multiplicative weights.
    Finds direction b that maximizes sum subject to (a*b)[k] <= M.
    Uses b as a mixing direction: a_new = (1-t)*a + t*rescaled_b.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    M_current = fft_convolve_self(a).max()

    # Initialize b
    b = np.ones(n) / n * S

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        # Compute a*b convolution
        ab = fft_cross_convolve(a, b)
        # Find violations
        violations = np.maximum(0, ab - M_current)
        max_violation = violations.max()

        if max_violation < M_current * 1e-6:
            # b is nearly feasible, use it as direction
            b_sum = b.sum()
            if b_sum > 1e-15:
                b_rescaled = b * (S / b_sum)
                # Mix
                t_mix = 0.3
                a_new = (1 - t_mix) * a + t_mix * b_rescaled
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new * (S / s_new)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < best_score:
                    best_score = c_new.max()
                    best_local = a_new.copy()
                a = a_new
                M_current = c_new.max()
            break

        # Multiplicative update on b to reduce violations
        # For each i, compute how much b[i] contributes to violations
        # contribution[i] = sum_k a[k-i] * violations[k] (for valid k)
        v_full = np.zeros(2 * n - 1)
        v_full[:] = violations
        fsize = 1 << (len(v_full) + n - 1).bit_length()
        V = fft(v_full, fsize)
        A = fft(a, fsize)
        contrib = np.real(ifft(V * np.conj(A)))[:n]

        # Update b multiplicatively
        eta = 0.1
        b = b * np.exp(-eta * contrib / max(contrib.max(), 1e-30))
        b = np.maximum(b, 1e-30)

        # Every 50 iterations, try mixing b into a
        if it % 50 == 0:
            b_sum = b.sum()
            if b_sum > 1e-15:
                b_rescaled = b * (S / b_sum)
                t_mix = 0.1
                a_trial = (1 - t_mix) * a + t_mix * b_rescaled
                a_trial = np.maximum(a_trial, 0.0)
                s_trial = a_trial.sum()
                if s_trial > 1e-15:
                    a_trial = a_trial * (S / s_trial)
                    c_trial = fft_convolve_self(a_trial)
                    if c_trial.max() < best_score:
                        best_score = c_trial.max()
                        best_local = a_trial.copy()
                        a = a_trial
                        M_current = c_trial.max()

    # Final: ensure best_local has correct sum
    bl_sum = best_local.sum()
    if bl_sum > 1e-15:
        best_local = best_local * (S / bl_sum)

    return best_local, best_score

                break
            step *= 0.5
            x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if not improved:
            # Try opposite: maybe we need to increase some elements
            # Use a small random perturbation
            if rng.random() < 0.5:
                tt = np.linspace(0, 1, n)
                freq = rng.integers(1, 20)
                perturb = 0.001 * S / np.sqrt(n) * np.sin(2 * np.pi * freq * tt)
                y = project_simplex(y + perturb, S)
            continue

        y = x_new
        a = x_new
        # No momentum for this polish phase (t=1 always)

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def generate_initializations(n, rng):
    """Generate diverse asymmetric initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Exponential ramps
    for alpha in [1.0, 2.0, 3.0, 5.0, 8.0]:
        inits.append(np.exp(alpha * tt))
        inits.append(np.exp(alpha * (1.0 - tt)))

    # Power-law
    for p in [0.3, 0.5, 1.0, 2.0, 3.0]:
        inits.append((tt + 0.01) ** p)
        inits.append((1.0 - tt + 0.01) ** p)

    # U-shaped (high at ends, low in middle)
    inits.append(1.0 + 0.5 * np.cos(np.pi * tt))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * tt))
    inits.append(1.0 + 0.3 * np.cos(np.pi * tt) + 0.2 * np.cos(2 * np.pi * tt))

    # Asymmetric with multiple features
    inits.append(1.0 + 0.5 * tt + 0.3 * np.sin(4 * np.pi * tt))
    inits.append(np.exp(2.0 * tt) * (1.0 + 0.3 * np.sin(3 * np.pi * tt)))

    # Step-like asymmetric
    arr = np.ones(n)
    arr[:n // 3] = 3.0
    arr[2 * n // 3:] = 0.3
    inits.append(arr)

    # Random smooth
    from scipy.ndimage import gaussian_filter1d
    rough = rng.random(n)
    smooth = gaussian_filter1d(rough, sigma=n // 100)
    inits.append(smooth + 0.1)

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: exploration, FISTA, LP-dual, peak polish, single-peak final.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Try to use parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Decide on n values to try
    n_values = []
    if parent is not None:
        n_values.append(len(parent))
    # Add some alternatives
    for n_try in [4096, 8192, 16384]:
        if n_try not in n_values and n_try != (len(parent) if parent is not None else 0):
            n_values.append(n_try)

    # If no parent, default to 8192
    if not n_values:
        n_values = [8192]

    # Sort: put parent n first, then others
    n_values = sorted(n_values, key=lambda x: (x != (len(parent) if parent is not None else -1), x))

    print("N values to try: %s" % n_values)
    print("Parent available: %s" % (parent is not None))

    # Phase 1: Quick exploration across n values and initializations
    # Allocate 30% of time for exploration
    explore_time = max(30.0, (deadline - time.monotonic()) * 0.30)
    per_candidate = explore_time / (len(n_values) * 5)  # 5 inits per n

    print("=== Phase 1: Exploration (%.1f sec) ===" % explore_time)

    candidates = []  # (score, a, n)

    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        S = np.sqrt(2.0 * float(n))

        # Generate initializations
        if n == (len(parent) if parent is not None else -1):
            inits = [parent.copy()]
        else:
            inits = generate_initializations(n, rng)[:4]

        for arr in inits:
            if time.monotonic() > deadline - 60:
                break
            arr = np.asarray(arr, dtype=np.float64)
            if len(arr) != n:
                continue
            arr = safe_normalize(arr, n)

            # Quick FISTA (short time)
            t0 = time.monotonic()
            a_opt, score = run_fista_adaptive(arr, n, S, per_candidate, rng,
                                              beta_init=5.0, beta_max=200.0)
            a_opt = safe_normalize(a_opt, n)
            c = fft_convolve_self(a_opt)
            obj = 2.0 * n * c.max() / (S * S)

            candidates.append((obj, a_opt, n))
            if obj < global_best:
                global_best = obj
                global_a = a_opt.copy()
                print("  n=%d: obj=%.6f" % (n, obj))

    print("After exploration: best=%.6f" % global_best)

    # Phase 2: Deep optimization on best candidate
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = deadline - time.monotonic()

        if remaining > 30:
            print("=== Phase 2: Deep optimization (%.1f sec remaining) ===" % remaining)

            # 2a: FISTA with high beta (30% of remaining)
            seg = remaining * 0.30
            a_p2a, s2a = run_fista_adaptive(global_a, n_best, S_best, seg, rng,
                                            beta_init=50.0, beta_max=800.0)
            a_p2a = safe_normalize(a_p2a, n_best)
            c2a = fft_convolve_self(a_p2a)
            obj2a = 2.0 * n_best * c2a.max() / (S_best * S_best)
            if obj2a < global_best:
                global_best = obj2a
                global_a = a_p2a.copy()
                print("  FISTA high-beta: %.6f" % global_best)

            # 2b: Peak pincer (25% of remaining)
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                seg2 = remaining2 * 0.25
                a_p2b, s2b = run_peak_pincer(global_a, n_best, S_best, seg2, rng,
                                             top_k_start=3, top_k_end=64)
                a_p2b = safe_normalize(a_p2b, n_best)
                c2b = fft_convolve_self(a_p2b)
                obj2b = 2.0 * n_best * c2b.max() / (S_best * S_best)
                if obj2b < global_best:
                    global_best = obj2b
                    global_a = a_p2b.copy()
                    print("  Peak pincer: %.6f" % global_best)

            # 2c: LP dual refinement (20% of remaining)
            remaining3 = deadline - time.monotonic()
            if remaining3 > 20:
                seg3 = remaining3 * 0.20
                a_p2c, s2c = run_lp_dual_refinement(global_a, n_best, S_best, seg3, rng)
                a_p2c = safe_normalize(a_p2c, n_best)
                c2c = fft_convolve_self(a_p2c)
                obj2c = 2.0 * n_best * c2c.max() / (S_best * S_best)
                if obj2c < global_best:
                    global_best = obj2c
                    global_a = a_p2c.copy()
                    print("  LP dual: %.6f" % global_best)

            # 2d: Single peak polish (remaining)
            remaining4 = deadline - time.monotonic()
            if remaining4 > 10:
                a_p2d, s2d = run_single_peak_polish(global_a, n_best, S_best, remaining4 * 0.7, rng)
                a_p2d = safe_normalize(a_p2d, n_best)
                c2d = fft_convolve_self(a_p2d)
                obj2d = 2.0 * n_best * c2d.max() / (S_best * S_best)
                if obj2d < global_best:
                    global_best = obj2d
                    global_a = a_p2d.copy()
                    print("  Single peak polish: %.6f" % global_best)

                # Final micro-polish
                remaining5 = deadline - time.monotonic()
                if remaining5 > 5:
                    a_p2e, s2e = run_single_peak_polish(global_a, n_best, S_best, remaining5, rng)
                    a_p2e = safe_normalize(a_p2e, n_best)
                    c2e = fft_convolve_self(a_p2e)
                    obj2e = 2.0 * n_best * c2e.max() / (S_best * S_best)
                    if obj2e < global_best:
                        global_best = obj2e
                        global_a = a_p2e.copy()
                        print("  Final micro-polish: %.6f" % global_best)

    # Phase 3: Try a different n if time remains and we haven't beaten target
    if global_best > 1.5027:
        remaining = deadline - time.monotonic()
        if remaining > 120:
            print("=== Phase 3: Alternative n (%.1f sec) ===" % remaining)
            # Try a larger n
            n_alt = 16384 if (global_a is not None and len(global_a) == 8192) else 8192
            S_alt = np.sqrt(2.0 * float(n_alt))
            tt = np.linspace(0, 1, n_alt)
            # Use the best sequence as a template (interpolate)
            if global_a is not None and len(global_a) != n_alt:
                from scipy.interpolate import interp1d
                old_t = np.linspace(0, 1, len(global_a))
                f = interp1d(old_t, global_a / global_a.sum(), kind='cubic', fill_value='extrapolate')
                a_alt = np.maximum(f(tt), 0) * S_alt
            else:
                # Generate fresh
                a_alt = np.exp(3.0 * tt) * S_alt / np.exp(3.0 * tt).sum()

            a_alt = safe_normalize(a_alt, n_alt)
            seg_alt = remaining * 0.6
            a_alt_opt, _ = run_fista_adaptive(a_alt, n_alt, S_alt, seg_alt, rng,
                                              beta_init=10.0, beta_max=400.0)
            a_alt_opt = safe_normalize(a_alt_opt, n_alt)
            c_alt = fft_convolve_self(a_alt_opt)
            obj_alt = 2.0 * n_alt * c_alt.max() / (S_alt * S_alt)
            print("  Alt n=%d: obj=%.6f" % (n_alt, obj_alt))
            if obj_alt < global_best:
                global_best = obj_alt
                global_a = a_alt_opt.copy()

            # Polish the alt
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_alt_pol, _ = run_single_peak_polish(global_a, n_alt, S_alt, remaining2, rng)
                a_alt_pol = safe_normalize(a_alt_pol, n_alt)
                c_pol = fft_convolve_self(a_alt_pol)
                obj_pol = 2.0 * n_alt * c_pol.max() / (S_alt * S_alt)
                if obj_pol < global_best:
                    global_best = obj_pol
                    global_a = a_alt_pol.copy()
                    print("  Alt polished: %.6f" % global_best)

    # Final verification
    if global_a is None:
        global_a = np.ones(8192, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        global_a = np.ones(8192, dtype=np.float64) * (np.sqrt(2.0 * 8192) / 8192)
        final_list = [float(x) for x in global_a]

    print("Final: length=%d, internal=%.6f, verified=%.6f" %
          (len(final_list), global_best, ver if np.isfinite(ver) else -1))
    return final_list
```
