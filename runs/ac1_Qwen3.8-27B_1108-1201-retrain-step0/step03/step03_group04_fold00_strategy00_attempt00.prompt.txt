        if np.isnan(x) or np.isinf(x):

    sequence = [float(x) for x in sequence]

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)

    valid = theta_candidates <= w

        rho = n
    else:

    theta = (cumsum[rho - 1] - S) / rho

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient given weights on convolution lags using FFT cross-correlation."""

    if total < 1e-30:

    weights = np.zeros(len(c))
