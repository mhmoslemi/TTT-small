def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_cross(a, b):
    """Full cross-convolution of a and b via FFT. Both length n."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def hard_topk_gradient(a, c, top_k):
    """Gradient using exactly top-k active lags (uniform weighting)."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    w_full = np.zeros(2 * n - 1)
    w_full[top_indices] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_topk_gradient(a, c, top_k, temperature=1.0):
    """Gradient using top-k lags with softmax weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices].copy()
    tv_max = top_vals.max()
    weights = np.exp(top_vals / max(temperature, 1e-10) - tv_max / max(temperature, 1e-10))
    weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

    L = 2.0
    stall_count = 0
    last_improve = np.inf

    while time.monotonic() - start < max_time:

        beta = min(beta_max, beta_init * np.exp(0.005 * it))

        c = fft_convolve_self(y)

                stall_count = 0
            else:

        grad = lse_gradient(y, c, beta)

        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)

def solve_lp_direction(a, n, S, top_K=100):
    """Solve LP to find direction g that maximizes sum(g) subject to
    conv(a, g)[l] <= M for top K lags, g >= 0.
    Returns normalized g with sum = S, or None if LP fails.
    """
    c = fft_convolve_self(a)
    M = c.max()
    top_lags = np.argsort(c)[::-1][:top_K]

    # Build constraint matrix: for each top lag l, conv(a,g)[l] = sum_i a[i]*g[l-i] <= M
    K = len(top_lags)
    rows = []
    cols = []
    vals = []
    for row_idx, l in enumerate(top_lags):
        l = int(l)
        i_min = max(0, l - (n - 1))
        i_max = min(l, n - 1)
        for i in range(i_min, i_max + 1):
            j = l - i
            if 0 <= j < n and a[i] > 1e-12:
                rows.append(row_idx)
                cols.append(j)
                vals.append(a[i])

    A_csr = csr_matrix((vals, (rows, cols)), shape=(K, n))

    # Objective: maximize sum(g) -> minimize -sum(g)
    c_obj = -np.ones(n)

    # Constraints: A_csr @ g <= M * ones(K), and -sum(g) <= -S (i.e., sum(g) >= S)
    # Actually, let's just do: A_csr @ g <= M*1, g >= 0
    # And add sum(g) = S as equality (to fix the scale)

    # Let's add: sum(g) >= S as an inequality: -sum(g) <= -S

    A_ineq = A_csr
    b_ineq = np.full(K, M)

    # Add sum constraint: we want sum(g) to be as large as possible,
    # so actually we don't need to constrain it from below.
    # The LP naturally maximizes it. But to get a bounded problem,

        top_K = min(top_K, 2 * n - 1)

        a_new = np.maximum(a_new, 0.0)

def run_coordinate_descent(a_init, n, S, max_time, rng, top_K=20):
    """Coordinate descent targeting specific high-convolution pairs."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Get top lags
        top_lags = np.argsort(c)[::-1][:top_K]

        improved = False
        for lag in top_lags:
            if time.monotonic() - start > max_time:
                break
            lag = int(lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue

            indices = np.arange(i_min, i_max + 1)
            partner = lag - indices
            products = a[indices] * a[partner]

            # Find the pair with the largest product
            max_prod_idx = np.argmax(products)
            idx = indices[max_prod_idx]
            pidx = partner[max_prod_idx]

            # Try reducing the larger element and adding to a small element elsewhere
            if a[idx] < 1e-10 and a[pidx] < 1e-10:
                continue

            # Find a good place to add mass (where it contributes least to top lags)
            # Simple heuristic: add to a random small element
            small_idx = np.where(a < np.median(a) * 0.5)[0]
            if len(small_idx) == 0:
                small_idx = np.where(a == a.min())[0]

            add_pos = small_idx[rng.integers(len(small_idx))]

            # Transfer mass from larger to smaller
            if a[idx] >= a[pidx]:
                src, dst = idx, pidx
            else:
                src, dst = pidx, idx

            for frac in [0.5, 0.3, 0.1, 0.05]:
                delta = frac * a[src]
                if delta < 1e-15:
                    break
                a_trial = a.copy()
                a_trial[src] -= delta
                a_trial[add_pos] += delta
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial * (S / a_trial.sum())

                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur - 1e-12:
                    a = a_trial
                    improved = True
                    cur = c_trial.max()
                    break

        if not improved and it % 20 == 0:
            # Small random perturbation
            noise = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)

    return best_local, best_score


def make_window_init(n, rng, S, window_type='raised_cosine'):
    """Create a window-function initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    if window_type == 'raised_cosine':
        a = 0.5 * (1 + np.cos(2 * np.pi * tt))
    elif window_type == 'sin_sq':
        a = np.sin(np.pi * tt) ** 2
    elif window_type == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * (tt - 0.5)))
    elif window_type == 'gaussian':
        center = n / 2
        sigma = n / (2 * (1.5 + rng.random() * 2.0))
        a = np.exp(-0.5 * ((tt * n - center) / sigma) ** 2)
    elif window_type == 'parabolic':
        x = 2 * tt - 1
        a = np.maximum(1 - x ** 2, 0.0)
    elif window_type == 'power':
        p = 0.5 + rng.random() * 1.5
        x = np.abs(2 * tt - 1)
        a = (1 - x ** p)
    else:
        a = 0.5 * (1 + np.cos(2 * np.pi * tt))

    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    return a


def make_anti_pal_init(n, rng, S):
    """Anti-palindromic initialization."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_periodic_init(n, rng, S, num_freqs=3):
    """Periodic initialization with multiple frequencies."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 25)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.3
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_concentrated_init(n, rng, S):
    """Concentrated (narrow) initialization."""
    center = n // 2
    width = int(n * (0.1 + rng.random() * 0.3))
    a = np.zeros(n)
    lo = max(0, center - width // 2)
    hi = min(n, center + width // 2 + 1)
    a[lo:hi] = 1.0
    # Smooth edges
    if lo > 0:
        a[lo-1] = 0.5
    if hi < n:
        a[hi] = 0.5
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum()) if a.sum() > 0 else np.ones(n) * S / n
    return a


def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a):
    """Compute the objective 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.

    Strategy:
    Phase 1: Exploratory FISTA on diverse initializations (200s)
    Phase 2: LP-based iterative refinement (1000s)
    Phase 3: High-precision polishing with multiple methods (700s)
    Phase 4: Final verification and safety checks (300s)
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose primary length
    primary_n = 8192
    if parent is not None and len(parent) >= 1024:
        primary_n = len(parent)

    print("Primary n=%d, parent available: %s" % (primary_n, parent is not None))

    # === Phase 1: Exploratory FISTA ===
    S = np.sqrt(2.0 * primary_n)
    phase1_time = min(250.0, max(100.0, (deadline - time.monotonic()) * 0.12))

    # Generate diverse initializations
    inits = []
    inits.append(("sin_sq", make_window_init(primary_n, rng, S, 'sin_sq')))
    inits.append(("raised_cos", make_window_init(primary_n, rng, S, 'raised_cosine')))
    inits.append(("hann", make_window_init(primary_n, rng, S, 'hann')))
    inits.append(("gauss", make_window_init(primary_n, rng, S, 'gaussian')))
    inits.append(("parabolic", make_window_init(primary_n, rng, S, 'parabolic')))
    inits.append(("power", make_window_init(primary_n, rng, S, 'power')))
    inits.append(("anti_pal_1", make_anti_pal_init(primary_n, rng, S)))
    inits.append(("anti_pal_2", make_anti_pal_init(primary_n, rng, S)))
    inits.append(("periodic_1", make_periodic_init(primary_n, rng, S, 3)))
    inits.append(("periodic_2", make_periodic_init(primary_n, rng, S, 5)))
    inits.append(("concentrated", make_concentrated_init(primary_n, rng, S)))

    if parent is not None and len(parent) == primary_n:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if p.sum() > 0.01:
            inits.append(("parent", p * (S / p.sum())))
    elif parent is not None and len(parent) >= 64:
        # Interpolate parent to primary_n
        tt_old = np.linspace(0, 1, len(parent))
        tt_new = np.linspace(0, 1, primary_n)
        p_interp = np.interp(tt_new, tt_old, parent)
        p_interp = np.maximum(p_interp, 0.0)
        if p_interp.sum() > 0.01:
            inits.append(("parent_interp", p_interp * (S / p_interp.sum())))

    n_inits = len(inits)
    per_init_time = phase1_time / n_inits
    print("=== Phase 1: %d inits, %.1f s each, total %.0f s ===" % (n_inits, per_init_time, phase1_time))

    for name, arr in inits:
        if time.monotonic() > deadline - 600:
            break
        n = len(arr)
        S_n = np.sqrt(2.0 * n)
        arr = safe_normalize(arr, n)

        # Quick FISTA
        seg = min(per_init_time, max(5.0, per_init_time * 0.9))
        a_f, s_f = run_fista(arr, n, S_n, seg, rng, beta_init=1.0, beta_max=200.0)
        a_f = safe_normalize(a_f, n)
        obj_f = compute_objective(a_f)
        if obj_f < global_best:
            global_best = obj_f
            global_a = a_f.copy()
            print("  FISTA [%s]: obj=%.6f (max_c=%.4f)" % (name, obj_f, s_f))

    print("=== After Phase 1: best_obj=%.6f, n=%d ===" %
          (global_best, len(global_a) if global_a is not None else 0))

    if global_a is None:
        global_a = make_window_init(primary_n, rng, S, 'sin_sq')
        global_best = compute_objective(global_a)

    # === Phase 2: LP-based refinement ===
    remaining2 = deadline - time.monotonic()
    phase2_time = min(1000.0, max(200.0, remaining2 * 0.45))
    n = len(global_a)
    S = np.sqrt(2.0 * n)
    print("=== Phase 2: LP refinement, n=%d, time=%.0f s ===" % (n, phase2_time))

    # Split into sub-phases with increasing K
    sub_phases = [
        (phase2_time * 0.3, 10),
        (phase2_time * 0.3, 50),
        (phase2_time * 0.4, 150),
    ]

    for i, (sub_time, top_K) in enumerate(sub_phases):
        if time.monotonic() > deadline - 300:
            break
        print("  LP sub-phase %d: K=%d, time=%.0f s" % (i + 1, top_K, sub_time))
        a_lp, s_lp = run_lp_refine(global_a, n, S, sub_time, rng,
                                    top_K_start=max(5, top_K // 2),
                                    top_K_end=top_K)
        a_lp = safe_normalize(a_lp, n)
        obj_lp = compute_objective(a_lp)
        if obj_lp < global_best:
            global_best = obj_lp
            global_a = a_lp.copy()
            print("  After LP sub-phase %d: obj=%.6f" % (i + 1, obj_lp))

    print("=== After Phase 2: best_obj=%.6f ===" % global_best)

    # === Phase 3: High-precision polishing ===
    remaining3 = deadline - time.monotonic()
    phase3_time = min(700.0, max(100.0, remaining3 * 0.5))
    n = len(global_a)
    S = np.sqrt(2.0 * n)
    print("=== Phase 3: Polishing, remaining=%.0f s ===" % phase3_time)

    # 3a: High-beta FISTA
    seg3a = max(20.0, phase3_time * 0.25)
    a_p1, s_p1 = run_fista(global_a, n, S, seg3a, rng, beta_init=50.0, beta_max=800.0)
    a_p1 = safe_normalize(a_p1, n)
    obj_p1 = compute_objective(a_p1)
    if obj_p1 < global_best:
        global_best = obj_p1
        global_a = a_p1.copy()
        print("  Polish FISTA hi-beta: obj=%.6f" % global_best)

    # 3b: Coordinate descent
    remaining3b = deadline - time.monotonic()
    if remaining3b > 60:
        seg3b = max(20.0, min(200.0, remaining3b * 0.25))
        a_p2, s_p2 = run_coordinate_descent(global_a, n, S, seg3b, rng, top_K=10)
        a_p2 = safe_normalize(a_p2, n)
        obj_p2 = compute_objective(a_p2)
        if obj_p2 < global_best:
            global_best = obj_p2
            global_a = a_p2.copy()
            print("  Polish CoordDesc: obj=%.6f" % global_best)

    # 3c: Another round of LP with very high K
    remaining3c = deadline - time.monotonic()
    if remaining3c > 60:
        seg3c = max(20.0, min(200.0, remaining3c * 0.3))
        a_p3, s_p3 = run_lp_refine(global_a, n, S, seg3c, rng,
                                    top_K_start=100, top_K_end=500)
        a_p3 = safe_normalize(a_p3, n)
        obj_p3 = compute_objective(a_p3)
        if obj_p3 < global_best:
            global_best = obj_p3
            global_a = a_p3.copy()
            print("  Polish LP hi-K: obj=%.6f" % global_best)

    # 3d: Final FISTA with very high beta
    remaining3d = deadline - time.monotonic()
    if remaining3d > 30:
        seg3d = max(10.0, remaining3d * 0.6)
        a_p4, s_p4 = run_fista(global_a, n, S, seg3d, rng, beta_init=200.0, beta_max=2000.0)
        a_p4 = safe_normalize(a_p4, n)
        obj_p4 = compute_objective(a_p4)
        if obj_p4 < global_best:
            global_best = obj_p4
            global_a = a_p4.copy()
            print("  Final FISTA: obj=%.6f" % global_best)

    print("=== After Phase 3: best_obj=%.6f ===" % global_best)

    # === Phase 4: Try different length for more attack surface ===
    remaining4 = deadline - time.monotonic()
    if remaining4 > 200 and primary_n <= 8192:
        n_long = primary_n * 2
        if n_long > 32768:
            n_long = 16384
        S_long = np.sqrt(2.0 * n_long)
        print("=== Phase 4: Trying n=%d ===" % n_long)

        # Interpolate or create new
        if len(global_a) == primary_n and primary_n * 2 == n_long:
            # Upsample
            tt_old = np.linspace(0, 1, primary_n)
            tt_new = np.linspace(0, 1, n_long)
            a_long_init = np.interp(tt_new, tt_old, global_a)
            a_long_init = np.maximum(a_long_init, 0.0)
        else:
            a_long_init = make_window_init(n_long, rng, S_long, 'sin_sq')

        a_long_init = safe_normalize(a_long_init, n_long)

        seg_long = min(150.0, max(30.0, remaining4 * 0.4))
        a_l, s_l = run_fista(a_long_init, n_long, S_long, seg_long, rng,
                             beta_init=2.0, beta_max=300.0)
        a_l = safe_normalize(a_l, n_long)
        obj_l = compute_objective(a_l)

        if obj_l < global_best:
            global_best = obj_l
            global_a = a_l.copy()
            print("  Long seq n=%d: obj=%.6f" % (n_long, obj_l))

        # Quick LP on the long sequence
        remaining4b = deadline - time.monotonic()
        if remaining4b > 60 and obj_l < global_best * 1.01:
            seg_lp = min(100.0, remaining4b * 0.4)
            a_l2, s_l2 = run_lp_refine(a_l, n_long, S_long, seg_lp, rng,
                                        top_K_start=20, top_K_end=100)
            a_l2 = safe_normalize(a_l2, n_long)
            obj_l2 = compute_objective(a_l2)
            if obj_l2 < global_best:
                global_best = obj_l2
                global_a = a_l2.copy()
                print("  Long seq LP: obj=%.6f" % obj_l2)

    print("=== After Phase 4: best_obj=%.6f ===" % global_best)

    # === Phase 5: Final verification ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Verification: eval=%.6f, internal_obj=%.6f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: evaluate_sequence rejected! Using fallback.")
        global_a = make_window_init(4096, rng, np.sqrt(8192.0), 'sin_sq')
        final_list = [float(x) for x in global_a]
        ver2 = evaluate_sequence(final_list)
        print("  Fallback eval: %.6f" % (ver2 if np.isfinite(ver2) else float('inf')))

    print("=== FINAL: %.6f ===" % (ver if np.isfinite(ver) else float('inf')))
    return final_list
