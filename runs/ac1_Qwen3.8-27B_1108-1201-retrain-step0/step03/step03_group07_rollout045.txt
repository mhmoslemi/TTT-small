"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive mirror descent with
frequency-domain analysis, multi-mode peak selection, and structural
perturbations. Key improvements over baseline: (1) Gaussian-smoothed
subgradient for more stable updates, (2) frequency-domain peak analysis
to identify problematic components, (3) adaptive step size with
Armijo backtracking, (4) multi-frequency restart strategy, (5)
two-scale optimization (coarse then fine).
"""
import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_from_conv(a, c):
    """Compute objective from precomputed convolution."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def gaussian_smoothed_subgrad(c, a, sigma, peak_fraction=0.1):
    """
    Compute a smoothed subgradient using Gaussian weighting over peaks.
    Peaks are identified as the top `peak_fraction` of convolution values.
    """
    n = len(a)
    c_max = c.max()
    c_min = c.min()
    c_range = c_max - c_min
    if c_range < 1e-15:
        return np.zeros(n)
    
    # Gaussian weight: w[k] = exp(-(c[k] - c_max)^2 / (2*sigma^2))
    # This smoothly weights all peaks, not just the top k
    sigma = max(sigma, 0.01 * c_range)
    w = np.exp(-(c - c_max) ** 2 / (2 * sigma ** 2))
    
    # Ensure we're focusing on peaks (not valleys)
    threshold = c_max - sigma * 2
    w[c < threshold] *= 0.1
    
    w_sum = w.sum()
    if w_sum < 1e-15:
        # Fall back to top-1
        k_peak = np.argmax(c)
        w = np.zeros(len(c))
        w[k_peak] = 1.0
    else:
        w /= w_sum
    
    # grad[m] = 2 * sum_k w[k] * a[k-m] for valid indices
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 2).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def compute_topk_subgrad(c, a, top_k):
    """Compute subgradient as average over top-k peaks."""
    n = len(a)
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / top_k
    
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 2).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad

    Adaptive mirror descent with multiple subgradient modes and

    best_iter = 0

    mode_cycle = [0, 0, 0, 1, 1, 2]
    mode_idx = 0
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = elapsed / max(time_budget, 1e-10)

            sigma = 0.05 * c_range * (1.0 + progress)  # increase smoothing over time
            grad = gaussian_smoothed_subgrad(c, a, sigma)
        elif mode == 1:
            top_k = max(20, int(100 * (1 - progress)))
            grad = compute_topk_subgrad(c, a, top_k)
        else:
            grad = compute_topk_subgrad(c, a, 1)
        
        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # In frequency domain: c[k_peak] = sum_m P[m] * exp(2*pi*i*m*k_peak/fsize) / fsize

        top_freqs = np.argsort(real_contrib)[::-1][:50]

def multi_freq_generate(n, freqs, amplitudes):
    """Generate sequences with cosine modulation at given frequencies."""
    i = np.arange(n)
    S = np.sqrt(2.0 * n)
    candidates = []
    for f, amp in zip(freqs, amplitudes):
        a = 1.0 + amp * np.cos(2 * np.pi * f * i / n)
        a = np.maximum(a, 0.01)
        a = a * S / a.sum()
        candidates.append(a)
    return candidates
