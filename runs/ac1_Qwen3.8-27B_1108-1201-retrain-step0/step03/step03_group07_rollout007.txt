import numpy as np
import time
from scipy.fft import fft, ifft

def next_fast_len(n):
    """Return the next 'fast' FFT length (power of 2 for simplicity)."""
    return 1 << (n - 1).bit_length()

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = next_fast_len(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_soft_subgrad(c, a, T):
    """
    Compute the subgradient of T*log(sum(exp(c/T))) w.r.t. a.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Softmax weights (numerically stable)
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / T)
    w = exp_c / exp_c.sum()
    
    # grad[m] = 2T * sum_k w[k] * a[k-m]
    # = 2T * conv(w, a_rev)[n-1+m]
    a_rev = a[::-1]
    fsize = next_fast_len(len(c) + n - 1)
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * T * full_conv[n-1:2*n-1]
    return grad

def mirror_descent_improved(a0, time_budget, T_start=1.0, T_end=0.01,
                            eta0=0.3, momentum=0.9, verbose=True):

    Improved mirror descent with soft-max subgradient and Nesterov acceleration.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 1e-300)
    a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    
    v = np.zeros(n)  # Nesterov velocity
    y = a.copy()  # Nesterov lookahead
    T = T_start
    eta = eta0
    iter_no = 0
    start = time.monotonic()
    stagnant = 0
    
    while time.monotonic() - start < time_budget:
        # Nesterov: y = a + momentum * v
        y = a + momentum * v
        
        # Compute subgradient at y

        grad = compute_soft_subgrad(c_y, y, T)
        
        # Normalize

        if gnorm > 1e-15:
            grad = grad / gnorm

            a = a_new
            c = c_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:

            eta *= 0.7
            stagnant += 1
        
        # Decay temperature
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start * (T_end / T_start) ** progress

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using mirror descent with soft-max subgradient,
adaptive temperature annealing, spectral reshaping for peak suppression, and
multi-phase optimization (explore -> exploit -> polish).
Key innovations: dense smooth subgradient via log-sum-exp, frequency-domain
deflation to flatten convolution, and strategic restarts with diverse perturbations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def _next_fast_len(n):
    return 1 << (n - 1).bit_length()


def _fft_conv_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = _next_fast_len(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def _fft_conv_cross(a, b):
    out_len = len(a) + len(b) - 1
    fsize = _next_fast_len(out_len)
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:out_len]


def _obj(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    if c is None:
        c = _fft_conv_self(a)
    return 2.0 * n * c.max() / (S * S)


def _soft_subgrad(c, a, T):
    """Compute gradient of T*logsumexp(c/T) w.r.t. a. Returns vector of len(a)."""
    n = len(a)
    c_shift = c - c.max()
    exp_c = np.exp(c_shift / T)
    w = exp_c / exp_c.sum()
    # grad[m] = 2T * sum_k w[k] * a[k-m] = 2T * (w * a_rev)[n-1+m]
    a_rev = a[::-1]
    fsize = _next_fast_len(len(c) + n - 1)
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full = np.real(ifft(W * R))
    return 2.0 * T * full[n - 1:2 * n - 1]


def _spectral_deflate(a, S, amount=0.1):
    """
    Spectral reshaping: reduce the dynamic range of |FFT(a)| to flatten convolution.
    a_new has flatter power spectrum, which tends to flatten the self-convolution.
    """
    n = len(a)
    fsize = _next_fast_len(2 * n)
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = A / np.maximum(mag, 1e-300)
    # Compress magnitude: mag_new = mag^{1-amount} * mag_mean^{amount}
    mag_mean = mag.mean()
    mag_new = mag ** (1.0 - amount) * (mag_mean ** amount)
    A_new = mag_new * phase
    a_new = np.real(ifft(A_new, fsize))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        return a.copy()
    a_new = a_new * (S / s)
    return a_new


def _mirror_descent_soft(a0, time_budget, T_start=0.5, T_end=0.005,

    c = _fft_conv_self(a)
    cur_obj = _obj(a, c)
    best_obj = cur_obj
    best_a = a.copy()

    T = T_start

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:

        grad = grad / gnorm

        vnorm = np.linalg.norm(v)

            cur_obj = _obj(a, c)

                a = a_spec
                c = c_spec
                cur_obj = obj_spec
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()

        # Anneal temperature
        progress = min(1.0, (time.monotonic() - start) / total_time)
        T = T_start * (T_end / T_start) ** progress

        # Stagnation: structured restart
        if stagnant > 800:
            mode = stagnant % 3
            if mode == 0:
                # Random perturbation
                frac = 0.03 + 0.02 * (stagnant % 5)
                noise = np.random.randn(n) * frac * S / np.sqrt(n)
                a = a + noise
            elif mode == 1:
                # Spectral boost (more aggressive deflation)
                a = _spectral_deflate(a, S, amount=0.15)
            else:
                # Shift-based perturbation
                shift = np.random.randint(1, max(2, n // 64))
                a = np.roll(a, shift)
            a = np.maximum(a, 1e-300)
            a = a * (S / a.sum())
            c = _fft_conv_self(a)
            cur_obj = _obj(a, c)
            v *= 0.3
            eta = eta0
            stagnant = 0
            if verbose:
                print(f"      [restart #{stagnant}] mode={mode}, obj={cur_obj:.6f}, best={best_obj:.6f}")

        # Slow eta decay
        eta *= 0.999995

        iter_no += 1
        if verbose and iter_no % verbose_every == 0:
            elapsed = time.monotonic() - start
            print(f"      iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"T={T:.5f}, eta={eta:.3f}, t={elapsed:.0f}s")

    return best_a, best_obj


def _coordinate_polish(a0, S, time_budget, top_k=5):
    """
    Coordinate descent: for each top-k peak, try reducing contributing elements.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    c = _fft_conv_self(a)
    best_obj = _obj(a, c)
    best_a = a.copy()
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        # Find top-k peaks
        peak_idx = np.argsort(c)[::-1][:top_k]
        improved = False

        for k in peak_idx:
            if time.monotonic() - start > time_budget:
                break
            # Elements contributing to c[k]: a[i] * a[k-i] for valid i
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                orig_i = a[i]
                orig_j = a[j]
                if orig_i < 1e-15 and orig_j < 1e-15:
                    continue
                # Try reducing the larger element
                if orig_i >= orig_j:
                    a[i] = orig_i * 0.85
                else:
                    a[j] = orig_j * 0.85
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-12:
                    a[i] = orig_i
                    a[j] = orig_j
                    continue
                a_norm = a * (S / s)
                c_new = _fft_conv_self(a_norm)
                obj_new = _obj(a_norm, c_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_norm.copy()
                    a = a_norm
                    c = c_new
                    improved = True
                else:
                    a[i] = orig_i
                    a[j] = orig_j
                    a = a * (S / a.sum())

        if not improved:
            # Try increasing elements at valleys
            valley_idx = np.argsort(c)[:top_k]
            for k in valley_idx:
                if time.monotonic() - start > time_budget:
                    break
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                for i in range(lo, hi + 1):
                    j = k - i
                    orig_i = a[i]
                    orig_j = a[j]
                    # Try increasing the smaller element
                    if orig_i <= orig_j:
                        a[i] = orig_i * 1.15
                    else:
                        a[j] = orig_j * 1.15
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s < 1e-12:
                        a[i] = orig_i
                        a[j] = orig_j
                        continue
                    a_norm = a * (S / s)
                    c_new = _fft_conv_self(a_norm)
                    obj_new = _obj(a_norm, c_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_norm.copy()
                        a = a_norm
                        c = c_new
                        improved = True
                    else:
                        a[i] = orig_i
                        a[j] = orig_j
                        a = a * (S / a.sum())

        if not improved:
            break

    return best_a, best_obj


def _directed_peak_descent(a0, S, time_budget, verbose=True):
    """
    Focus exclusively on the single worst peak, using analytic gradient.
    More aggressive than soft-max approach.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    n = len(a)
    c = _fft_conv_self(a)
    best_obj = _obj(a, c)
    best_a = a.copy()
    start = time.monotonic()
    eta = 0.5
    stagnant = 0
    iter_no = 0

    while time.monotonic() - start < time_budget:
        k_max = np.argmax(c)
        # c[k_max] = sum_i a[i]*a[k_max - i]
        # d(c[k_max])/da[m] = 2*a[k_max - m] for 0 <= k_max - m < n
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        grad = np.zeros(n)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[k_max - m]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        grad = grad / gnorm

        # Try multiplicative reduction
        a_trial = a * np.exp(-eta * grad)
        a_trial = np.maximum(a_trial, 1e-300)
        a_trial = a_trial * (S / a_trial.sum())
        c_trial = _fft_conv_self(a_trial)
        obj_trial = _obj(a_trial, c_trial)

        if obj_trial < best_obj:
            a = a_trial
            c = c_trial
            best_obj = obj_trial
            best_a = a.copy()
            stagnant = 0
            eta = min(eta * 1.02, 5.0)
        else:
            eta = max(eta * 0.7, 0.01)
            stagnant += 1

        if stagnant > 200:
            # Perturb and reset
            idx = np.random.choice(n, size=n // 16, replace=False)
            a[idx] *= np.exp(np.random.randn(len(idx)) * 0.3)
            a = np.maximum(a, 1e-300)
            a = a * (S / a.sum())
            c = _fft_conv_self(a)
            eta = 0.5
            stagnant = 0

        iter_no += 1
        if verbose and iter_no % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"      [peak] iter={iter_no}, obj={best_obj:.6f}, peak={c[k_max]:.4f}, t={elapsed:.0f}s")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses multi-phase strategy: explore diverse initializations, then exploit
    the best with soft-max mirror descent, directed peak descent, and polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    global_best = np.inf
    global_a = None

    # --- Load starting sequence ---
    start_a = None
    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        raw = np.clip(raw, 0.0, 1000.0)
        if raw.size >= 16 and raw.sum() > 0.01:
            start_a = raw.copy()
            n_g = len(start_a)
            S_g = np.sqrt(2.0 * n_g)
            start_a = start_a * (S_g / start_a.sum())
            v = evaluate_sequence(start_a.tolist())
            print(f"Starting sequence: n={n_g}, obj={v:.6f}")
            if np.isfinite(v) and v < global_best:
                global_best = v
                global_a = start_a.copy()
    except Exception as e:
        print(f"Could not load given sequence: {e}")

    # --- Generate alternative initializations ---
    n = 4096
    S = np.sqrt(2.0 * n)
    i = np.arange(n)

    initializations = []
    if start_a is not None and len(start_a) == n:
        initializations.append(('given', start_a.copy()))

    # Cosine with frequency 7
    a_cos7 = 1.0 + 0.7 * np.cos(2 * np.pi * 7 * i / n)
    a_cos7 = np.maximum(a_cos7, 0.01)
    a_cos7 *= S / a_cos7.sum()
    initializations.append(('cos7', a_cos7))

    # Cosine with frequency 3
    a_cos3 = 1.0 + 0.7 * np.cos(2 * np.pi * 3 * i / n)
    a_cos3 = np.maximum(a_cos3, 0.01)
    a_cos3 *= S / a_cos3.sum()
    initializations.append(('cos3', a_cos3))

    # U-shaped (more mass at edges)
    a_u = 0.5 + 0.5 * (2 * i / (n - 1) - 1.0) ** 2
    a_u = np.maximum(a_u, 0.01)
    a_u *= S / a_u.sum()
    initializations.append(('u_shape', a_u))

    # Two humps
    a_2h = (np.exp(-0.5 * ((i - n * 0.2) / (n * 0.06)) ** 2) +
            np.exp(-0.5 * ((i - n * 0.8) / (n * 0.06)) ** 2))
    a_2h = np.maximum(a_2h, 0.01)
    a_2h *= S / a_2h.sum()
    initializations.append(('2hump', a_2h))

    # Random
    a_rand = rng.random(n)
    a_rand = np.maximum(a_rand, 0.01)
    a_rand *= S / a_rand.sum()
    initializations.append(('random', a_rand))

    # Evaluate all initializations
    print("\n--- Initializations ---")
    for name, a_init in initializations:
        v = _obj(a_init)
        print(f"  {name}: obj={v:.6f}")
        if np.isfinite(v) and v < global_best:
            global_best = v
            global_a = a_init.copy()

    # --- Phase 1: Quick exploration on top candidates (15% time) ---
    explore_time = 0.15 * (deadline - time.monotonic())
    print(f"\n--- Phase 1: Exploration ({explore_time:.0f}s) ---")

    # Sort by initial objective
    scored = []
    for name, a_init in initializations:
        v = _obj(a_init)
        scored.append((v, name, a_init))
    scored.sort(key=lambda x: x[0])

    # Run mirror descent on top 3 initializations
    top_count = min(3, len(scored))
    per_time = explore_time / top_count
    for rank in range(top_count):
        if time.monotonic() > deadline - 120:
            break
        v0, name, a0 = scored[rank]
        print(f"  Exploring '{name}' (init obj={v0:.6f})...")
        t0 = time.monotonic()
        a_best, obj_best = _mirror_descent_soft(
            a0, per_time,
            T_start=1.0, T_end=0.05,
            eta0=1.5, momentum=0.8,
            spectral_every=300, verbose=False
        )
        elapsed = time.monotonic() - t0
        print(f"    -> {obj_best:.6f} ({elapsed:.1f}s)")
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

    # --- Phase 2: Main optimization (60% time) ---
    main_time = 0.60 * (deadline - time.monotonic())
    print(f"\n--- Phase 2: Main optimization ({main_time:.0f}s) ---")

    if global_a is not None and main_time > 30:
        # 2a: Soft-max mirror descent (broad)
        t_2a = main_time * 0.4
        print(f"  2a: Soft-max MD (broad, {t_2a:.0f}s)")
        a_best, obj_best = _mirror_descent_soft(
            global_a, t_2a,
            T_start=0.3, T_end=0.01,
            eta0=1.0, momentum=0.85,
            spectral_every=400, verbose=True, verbose_every=5000
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

        # 2b: Soft-max mirror descent (focused)
        t_2b = main_time * 0.3
        print(f"  2b: Soft-max MD (focused, {t_2b:.0f}s)")
        a_best, obj_best = _mirror_descent_soft(
            global_a, t_2b,
            T_start=0.05, T_end=0.002,
            eta0=0.8, momentum=0.9,
            spectral_every=0, verbose=True, verbose_every=5000
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

        # 2c: Directed peak descent
        t_2c = main_time * 0.3
        print(f"  2c: Directed peak descent ({t_2c:.0f}s)")
        a_best, obj_best = _directed_peak_descent(
            global_a, global_a.sum(), t_2c, verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

    # --- Phase 3: Polish (20% time) ---
    polish_time = deadline - time.monotonic() - 15.0
    print(f"\n--- Phase 3: Polish ({polish_time:.0f}s) ---")

    if global_a is not None and polish_time > 10:
        S_cur = global_a.sum()

        # 3a: Coordinate polish
        t_3a = polish_time * 0.4
        print(f"  3a: Coordinate polish ({t_3a:.0f}s)")
        a_best, obj_best = _coordinate_polish(global_a, S_cur, t_3a, top_k=3)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

        # 3b: Final fine-tuning with very focused soft-max
        t_3b = polish_time * 0.35
        print(f"  3b: Fine mirror descent ({t_3b:.0f}s)")
        a_best, obj_best = _mirror_descent_soft(
            global_a, t_3b,
            T_start=0.01, T_end=0.001,
            eta0=0.3, momentum=0.95,
            spectral_every=0, verbose=False
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

        # 3c: Another directed peak pass
        t_3c = polish_time * 0.25
        print(f"  3c: Final peak descent ({t_3c:.0f}s)")
        a_best, obj_best = _directed_peak_descent(
            global_a, global_a.sum(), t_3c, verbose=False
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(256) * (np.sqrt(2.0 * 256) / 256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * (np.sqrt(2.0 * 256) / 256)

    final_list = global_a.tolist()
    ver = evaluate_sequence(final_list)
    print(f"\nFINAL: local_obj={global_best:.6f}, verified={ver:.6f}")

    if not np.isfinite(ver):
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
        print(f"Fallback: obj={ver:.6f}")

    return final_list
