def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using smoothed subgradient descent
with adaptive step sizes, multi-scale initialization, and coordinate descent polishing.

Key ideas:
- Log-sum-exp smoothing of the max for smooth gradients
- Nesterov momentum with backtracking line search
- Multi-scale: optimize low-res then upscale and refine
- Coordinate descent polishing for final improvement
- Diverse initializations including anti-palindromic and periodic structures

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.array([], dtype=np.float64)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def smooth_gradient(a, c, beta):
    """
    Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a.
    This approximates the subgradient of max(c) as beta -> inf.
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    # grad[j] = 2 * sum_k w[k] * a[k-j] (with a[m]=0 outside [0,n-1])
    # This is 2 * cross_correlation(w, a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # cross-correlation: ifft(conj(W) * A)[j] = sum_m w[m+j] * a[m]
    # We want: sum_k w[k] * a[k-j] = ifft(conj(W)*A)[N-j] for j>0, [0] for j=0
    G = np.real(ifft(np.conj(W) * A))
    grad = np.zeros(n)
    grad[0] = 2.0 * G[0]
    if n > 1:
        grad[1:] = 2.0 * G[fsize - np.arange(1, n)]
    return grad


def hard_subgradient(a, c, num_active=1):
    """
    Hard subgradient using top-num_active lags with uniform weighting.
    """
    n = len(a)
    k = min(num_active, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    
    grad = np.zeros(n)
    for idx in top_indices:
        idx = int(idx)
        # subgrad[j] = 2*a[idx-j] for valid j
        j_lo = max(0, idx - n + 1)
        j_hi = min(idx, n - 1)
        if j_lo <= j_hi:
            j_range = np.arange(j_lo, j_hi + 1)
            grad[j_range] += 2.0 * a[idx - j_range]
    grad /= k
    return grad


def run_smoothed_descent(a_init, n, S, max_time, rng,
                         beta_start=5.0, beta_end=200.0,
                         use_momentum=True):
    """
    Smoothed gradient descent with Nesterov momentum and adaptive step size.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    step = S / n  # initial step
    patience = 0
    last_best = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = np.clip(y, 0, 1000)
            best_a = best_a * (S / best_a.sum()) if best_a.sum() > 1e-15 else best_a
            patience = 0
        else:
            patience += 1

        # Early exit if too stuck
        if patience > 2000:
            # Restart with perturbation
            noise = 0.02 * S / np.sqrt(n) * rng.standard_normal(n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
            patience = 0
            continue

        # Compute gradient
        grad = smooth_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size: aim for ~1% reduction in objective per step
        # f(y + alpha*grad) ~ f(y) + alpha*|grad|^2,y (approximately for convex)
        # We want: alpha * gnorm^2 / ||y - x_new|| <= curvature
        # Use: step = 2 * (best_score - cur_max + 1e-10) / (gnorm^2 + 1e-14)
        # But also cap to avoid too-large steps
        target_reduction = 0.01 * cur_max
        step_candidate = min(2.0 * target_reduction / (gnorm * gnorm + 1e-14), 2.0 * S / (n * gnorm + 1e-14))
        # Blend with previous step for stability
        step = 0.7 * step + 0.3 * step_candidate
        step = max(step, 1e-6 * S / n)

        # Gradient step with projection
        x_new = project_simplex(y - step * grad, S)

        # Check if we improved; if not, halve step and retry
        c_x = fft_convolve_self(x_new)
        if c_x.max() > cur_max * (1 + 0.01):
            # Backtracking
            for _ in range(3):
                step *= 0.5
                x_new = project_simplex(y - step * grad, S)
                c_x = fft_convolve_self(x_new)
                if c_x.max() <= cur_max:
                    break

        # Nesterov momentum update
        if use_momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            # Renormalize y to stay near the simplex
            y_s = y.sum()
            if y_s > 1e-15:

            a = x_new
            t_mom = t_new
        else:
            a = x_new
            y = x_new.copy()

        # Periodic perturbation for exploration

            y = project_simplex(y + noise, S)
            t_mom = 1.0
            patience = 0

    # Final normalization
    best_a = np.clip(best_a, 0.0, 1000.0)
    s_final = best_a.sum()
    if s_final > 1e-15:
        best_a = best_a * (S / s_final)
    else:
        best_a = np.ones(n) * S / n

    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def run_subgradient_descent(a_init, n, S, max_time, rng,
                            top_k_start=1, top_k_end=8):
    """
    Hard subgradient descent with Polyak step size.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    avg_step = S / (n * 2.0)
    momentum_buf = 0.0

    while time.monotonic() - start < max_time:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max_time)
        top_k = int(top_k_start + (top_k_end - top_k_start) * progress)

        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()

        num_active = min(top_k, len(c))
        grad = hard_subgradient(a, c, num_active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Polyak step: step = (f(x) - f*) / ||g||^2, use best_score as f* estimate
        step = (cur_max - best_score * 0.999 + 1e-10) / (gnorm * gnorm + 1e-14)
        step = min(step, 2.0 * S / (n * gnorm + 1e-14))
        step = max(step, 1e-8 * S / n)

        # Momentum
        momentum_buf = 0.9 * momentum_buf + 0.1 * grad
        x_new = project_simplex(a - step * momentum_buf, S)

        # Accept if improved
        c_x = fft_convolve_self(x_new)
        if c_x.max() < cur_max:
            a = x_new
        else:
            # Try smaller step
            x_new2 = project_simplex(a - 0.5 * step * momentum_buf, S)
            c_x2 = fft_convolve_self(x_new2)
            if c_x2.max() < cur_max:
                a = x_new2
                step *= 0.5

        # Periodic restart
        if it % 2000 == 0:
            noise = 0.01 * S / np.sqrt(n) * rng.standard_normal(n)
            a = project_simplex(a + noise, S)

    best_a = np.clip(best_a, 0.0, 1000.0)
    s_final = best_a.sum()
    if s_final > 1e-15:
        best_a = best_a * (S / s_final)
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def coordinate_descent(a_init, n, S, max_time, rng, max_iter_per_coord=3):
    """
    Coordinate descent: try to improve by adjusting pairs of coordinates
    that contribute to the peak lag.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best_score = c.max()

    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        # Find top 5 lags
        top_lags = np.argsort(c)[::-1][:5]

        improved = False
        for lag in top_lags:
            lag = int(lag)
            if time.monotonic() - start > max_time:
                break

            # Pairs (i, lag-i) contributing to b[lag]
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue

            indices = np.arange(i_min, i_max + 1)
            partners = lag - indices
            products = a[indices] * a[partners]

            # Find the top contributing pair
            top_idx = np.argmax(products)
            i_top = int(indices[top_idx])
            p_top = int(partners[top_idx])

            if i_top == p_top:
                continue

            # Try to reduce the pair: move mass from the larger to a smaller element
            if a[i_top] > a[p_top]:
                src = i_top
                dst = p_top
            else:
                src = p_top
                dst = i_top

            # Find a good target to add mass: a small element that doesn't
            # contribute much to the top lags
            median_val = np.median(a)
            small_mask = a < median_val
            if not np.any(small_mask):
                small_mask = np.argsort(a)[:n // 4] == np.arange(n)

            candidates = np.where(small_mask)[0]
            if len(candidates) == 0:
                continue

            # Try a few random targets
            for _ in range(min(3, len(candidates))):
                target = int(candidates[rng.integers(len(candidates))])
                if target == src or target == dst:
                    continue

                # Transfer a small amount of mass
                for frac in [0.01, 0.005, 0.001]:
                    delta = frac * a[src]
                    a_try = a.copy()
                    a_try[src] -= delta
                    a_try[target] += delta
                    if a_try[src] < 0:
                        continue

                    c_try = fft_convolve_self(a_try)
                    if c_try.max() < best_score - 1e-12:
                        a = a_try
                        best_score = c_try.max()
                        best_a = a.copy()
                        c = c_try
                        improved = True
                        break

        if not improved and it % 50 == 0:
            # Small random perturbation
            noise = 0.001 * S / np.sqrt(n) * rng.standard_normal(n)
            a = project_simplex(a + noise, S)
            c = fft_convolve_self(a)

    best_a = np.clip(best_a, 0.0, 1000.0)
    s_final = best_a.sum()
    if s_final > 1e-15:
        best_a = best_a * (S / s_final)
    c_final = fft_convolve_self(best_a)
    return best_a, c_final.max()


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic initialization: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total > 1e-15:
        a = a * (S / total)
    else:
        a = np.ones(n) * S / n
    return a


def make_periodic_init(n, rng, S, num_freqs=3):
    """Periodic initialization with multiple tones."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 30)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.5
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.05)
    a = a * (S / a.sum())
    return a


def make_dual_peak_init(n, rng, S):
    """Dual peak (bimodal) initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    center1 = 0.25 + 0.1 * rng.random()
    center2 = 0.75 - 0.1 * rng.random()
    width = 0.1 + 0.1 * rng.random()
    a = np.exp(-((tt - center1) / width) ** 2) + np.exp(-((tt - center2) / width) ** 2)
    a = a * (S / a.sum())
    return a


def make_fractal_init(n, rng, S):
    """Thue-Morse-like fractal initialization."""
    # Generate Thue-Morse sequence
    tm = np.zeros(n)
    for i in range(n):
        cnt = 0
        m = i
        while m > 0:
            cnt += m & 1
            m >>= 1
        tm[i] = cnt % 2

    # Convert to positive values with smoothing
    a = 1.0 + 0.3 * (2.0 * tm - 1.0)
    # Apply a lowpass filter to smooth
    kernel_size = max(3, n // 100)
    if kernel_size % 2 == 0:
        kernel_size += 1
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.05)
    a = a * (S / a.sum())
    return a


def downsample(a, factor):
    """Downsample by averaging blocks."""
    n = len(a)
    new_n = n // factor
    a_trimmed = a[:new_n * factor]
    return a_trimmed.reshape(new_n, factor).mean(axis=1)


def upsample(a, factor):
    """Upsample by linear interpolation."""
    n = len(a)
    new_n = n * factor
    tt_old = np.linspace(0, 1, n)
    tt_new = np.linspace(0, 1, new_n)
    return np.interp(tt_new, tt_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses smoothed gradient descent, subgradient methods, and coordinate descent
    with multi-scale initialization and adaptive restarts.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(10.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose primary length
    if parent is not None and len(parent) >= 64:
        n_primary = len(parent)
    else:
        n_primary = 8192

    print("Starting: n=%d, parent=%s" % (n_primary, "yes" if parent is not None else "no"))

    S = np.sqrt(2.0 * n_primary)

    # Generate initializations
    inits = []
    if parent is not None:
        p = parent.copy()
        p_s = p.sum()
        if p_s > 0.01:
            inits.append(("parent", p * (S / p_s)))

    inits.append(("anti_pal_1", make_anti_palindromic_init(n_primary, rng, S)))
    inits.append(("anti_pal_2", make_anti_palindromic_init(n_primary, rng, S)))
    inits.append(("periodic_1", make_periodic_init(n_primary, rng, S)))
    inits.append(("dual_peak", make_dual_peak_init(n_primary, rng, S)))
    inits.append(("fractal", make_fractal_init(n_primary, rng, S)))

    inits.append(("random", rand_a * (S / rand_a.sum())))

    print("=== Phase 1: Multi-scale optimization ===")

    n_low = n_primary // 4

        t0 = time.monotonic()
