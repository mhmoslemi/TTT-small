```
def corr_weights_direct(a, w, threshold):
    n = int(a.size)
    out = np.zeros(n, dtype=np.float64)
    active = np.nonzero(w > threshold)[0]
    for l in active:
        l = int(l)
        start = l - n + 1
        if start < 0: start = 0
        end = l
        if end > n - 1: end = n - 1
        if start <= end:
            idx_start = l - start
            idx_stop = l - end
            out[start:end+1] += w[l] * a[idx_stop:idx_start+1][::-1]
    return out
```

```python
"""
Minimize the peak of the full self-convolution of a nonnegative sequence.

Method:
- Softmax parameterization of nonnegative arrays with fixed L1 norm.
- Fast FFT evaluations and FFT/correlation gradients.
- Continuation on a log-sum-exp smoothed maximum.
- Additional thresholded-excess objective phases to push the max downward.
- LP-like multiplicative refinement, perturbations, and local polishing.
"""

import time
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def next_power_of_two(m):
    """Smallest power of two >= m."""
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fft_convolve_self(a):
    """Full linear self-convolution of a."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    nf = next_power_of_two(size)
    A = fft(a, nf)
    c = np.real(ifft(A * A))
    return c[:size].astype(np.float64, copy=False)


def corr_weights_fft(a, w):
    """
    Compute q_j = sum_l w_l * a_{l-j}, where valid indices are used.
    This is the un-doubling of the gradient of a weighted convolution.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.zeros(n, dtype=np.float64)
    nf = next_power_of_two(len(w) + n - 1)
    c = np.real(ifft(fft(w, nf) * fft(a[::-1], nf)))
    return c[n - 1:n - 1 + n].astype(np.float64, copy=False)


def corr_weights_direct(a, w, threshold):
    """Direct computation of q_j for sparse weights."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    out = np.zeros(n, dtype=np.float64)
    active = np.nonzero(w > threshold)[0]
    for l in active:
        l = int(l)
        start = l - n + 1
        if start < 0:
            start = 0
        end = l
        if end > n - 1:
            end = n - 1
        if start <= end:
            idx_start = l - start
            idx_stop = l - end
            out[start:end + 1] += w[l] * a[idx_stop:idx_start + 1][::-1]
    return out


def corr_weights(a, w, use_direct=True):
    """Hybrid direct/FFT implementation of q_j."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.zeros(n, dtype=np.float64)
    if use_direct:
        threshold = 1e-18
        active_count = int(np.count_nonzero(w > threshold))
        if active_count <= 48:
            return corr_weights_direct(a, w, threshold)
    return corr_weights_fft(a, w)


def fast_score(a):
    """Fast surrogate score: max coefficient of a*a."""
    try:
        c = fft_convolve_self(np.asarray(a, dtype=np.float64))
        if c.size == 0:
            return np.inf
        v = float(np.max(c))
        if not np.isfinite(v):
            return np.inf
        return v
    except Exception:
        return np.inf


def normalize_to_sum(a, S):
    """Clip to [0, 1000] and scale to sum S."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.clip(a, 0.0, 1000.0)
    n = int(a.size)
    if n == 0:
        return a
    s = float(a.sum())
    if s <= 1e-18 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) * (float(S) / float(n))
    return a * (float(S) / s)


def z_to_a(z, S):
    """Convert unconstrained parameters to positive vector of sum S."""
    z = np.asarray(z, dtype=np.float64)
    n = int(z.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return np.ones(n, dtype=np.float64) * (float(S) / float(n))
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) * (float(S) / float(n))
    return float(S) * e / s


def make_z(a, S):
    """Logarithmic parameterization of a positive vector."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(a.size)
    if n == 0:
        return np.array([], dtype=np.float64)
    a = np.clip(a, 1e-300, 1000.0)
    S = max(float(S), 1e-300)
    z = np.log(a / S)
    return z - z.mean()


def soft_fun_fg(x, beta, S, reg, use_direct):
    """
    Return (objective, gradient) for smoothed max of self-convolution.
    a = S * softmax(x), objective approximates max_j (a*a)_j.
    """
    x = np.asarray(x, dtype=np.float64)
    n = int(x.size)
    if n == 0 or not np.all(np.isfinite(x)):
        return 1e100, np.zeros(n, dtype=np.float64)

    beta = max(float(beta), 1e-3)
    S = float(S)
    reg = float(reg)

    a = z_to_a(x, S)
    c = fft_convolve_self(a)
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return 1e100, np.zeros(n, dtype=np.float64)

    delta = beta * (c - cmax)
    e = np.exp(delta)
    se = float(e.sum())
    if se <= 1e-300 or not np.isfinite(se):
        return 1e100, np.zeros(n, dye=np.float64)

    obj = cmax + np.log(se) / beta
    w = e / se

    q = corr_weights(a, w, use_direct=use_direct)
    grad_a = 2.0 * q

    dot = float(np.dot(grad_a, a))
    if S > 0:
        grad_z = a * (grad_a - (dot / S))
    else:
        grad_z = np.zeros_like(grad_a)

    if reg != 0.0:
        obj += reg * float(np.dot(x, x))
        grad_z += 2.0 * reg * x

    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros(n, dtype=np.float64)

    if not np.isfinite(obj):
        obj = 1e100

    return float(obj), grad_z.astype(np.float64, copy=False)


def soft_fun_only(x, beta, S, reg, use_direct):
    """Scalar wrapper for soft_fun_fg."""
    return soft_fun_fg(x, beta, S, reg, use_direct)[0]


def soft_jac_only(x, beta, S, reg, use_direct):
    """Jacobian wrapper for soft_fun_fg."""
    return soft_fun_fg(x, beta, S, reg, use_direct)[1]


def excess_fun_fg(x, T, S, reg, use_direct):
    """
    Return (objective, gradient) for minimizing sum of squared excess
    over a threshold T on convolution coefficients.
    """
    x = np.asarray(x, dtype=np.float64)
    n = int(x.size)
    if n == 0 or not np.all(np.isfinite(x)):
        return 1e100, np.zeros(n, dtype=np.float64)

    T = float(T)
    S = float(S)
    reg = float(reg)

    a = z_to_a(x, S)
    c = fft_convolve_self(a)
    if not np.all(np.isfinite(c)):
        return 1e100, np.zeros(n, dtype=np.float64)

    excess = np.maximum(c - T, 0.0)
    obj = 0.5 * float(np.dot(excess, excess))

    ex_sum = float(excess.sum())
    if ex_sum > 1e-300:
        q = corr_weights(a, excess, use_direct=use_direct)
        grad_a = 4.0 * q
    else:
        grad_a = np.zeros(n, dtype=np.float64)

    dot = float(np.dot(grad_a, a))
    if S > 0:
        grad_z = a * (grad_a - (dot / S))
    else:
        grad_z = np.zeros_like(grad_a)

    if reg != 0.0:
        obj += reg * float(np.dot(x, x))
        grad_z += 2.0 * reg * x

    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros(n, dtype=np.float64)

    if not np.isfinite(obj):
        obj = 1e100

    return float(obj), grad_z.astype(np.float64, copy=False)


def excess_fun_only(x, T, S, reg, use_direct):
    """Scalar wrapper for excess_fun_fg."""
    return excess_fun_fg(x, T, S, reg, use_direct)[0]


def excess_jac_only(x, T, S, reg, use_direct):
    """Jacobian wrapper for excess_fun_fg."""
    return excess_fun_fg(x, T, S, reg, use_direct)[1]


def run_lbfgs_continuation(a0, S, max_time, betas, maxiter, rng,
                           use_direct=True, reg=1e-10, label=""):
    """L-BFGS continuation on the log-sum-exp smoothed maximum."""
    S = float(S)
    a0 = np.asarray(a0, dtype=np.float64).reshape(-1)
    a = normalize_to_sum(a0, S)
    n = int(a.size)
    if n == 0:
        return a, np.inf

    best_score = fast_score(a)
    best_a = a.copy()
    z_current = make_z(a, S)
    best_z = z_current.copy()

    start = time.monotonic()
    max_time = max(0.0, float(max_time))
    if label:
        print("  LSE continuation: %s betas=%d budget=%.1fs" %
              (label, len(betas), max_time))

    maxfun = 500 if n <= 8192 else 300

    for beta in betas:
        if time.monotonic() > start + max_time - 5.0:
            break

        n_attempts = 1
        if beta >= 800 and max_time > 90.0:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() > start + max_time - 5.0:
                break

            if attempt == 0:
                z0 = z_current.copy()
            else:
                noise = 0.04 if beta >= 2000 else 0.08
                z0 = z_current + rng.normal(0.0, noise, n)
                z0 = z0 - z0.mean()

            args = (float(beta), S, float(reg), bool(use_direct))
            options = {
                "maxiter": int(maxiter),
                "maxfun": int(maxfun),
                "ftol": 1e-14,
                "gtol": 1e-12,
                "maxls": 30,
                "iprint": -1,
            }

            res = None
            try:
                res = scipy_minimize(
                    soft_fun_fg,
                    z0,
                    jac=True,
                    args=args,
                    method="L-BFGS-B",
                    options=options,
                )
            except Exception:
                try:
                    res = scipy_minimize(
                        soft_fun_only,
                        z0,
                        jac=soft_jac_only,
                        args=args,
                        method="L-BFGS-B",
                        options=options,
                    )
                except Exception:
                    res = None

            if res is None or not hasattr(res, "x"):
                continue

            z_new = np.asarray(res.x, dtype=np.float64)
            if z_new.size != n or not np.all(np.isfinite(z_new)):
                continue

            z_new = z_new - z_new.mean()
            a_new = z_to_a(z_new, S)
            a_new = normalize_to_sum(a_new, S)
            sc = fast_score(a_new)

            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_a = a_new.copy()
                best_z = z_new.copy()
                z_current = best_z.copy()
                print("      improved LSE -> %.8f" % best_score)
            elif np.isfinite(sc) and sc <= best_score + 0.05:
                z_current = z_new.copy()

    return best_a, float(best_score)

def lp_refine(a0, S, max_time, rng):
    """
    LP-like multiplicative refinement.
    Given weights on near-max coefficients, propose g_i = 1 / (A^T u)_i,
    then line-search a convex combination with the current sequence.
    """
    S = float(S)
    a = normalize_to_sum(np.asarray(a0, dtype=np.float64), S)
    n = int(a.size)
    if n == 0:
        return a, np.inf

    best_a = a.copy()
    best_score = fast_score(a)
    start = time.monotonic()
    max_time = max(0.0, float(max_time))

    it = 0
    hard_variant = 0

    while time.monotonic() < start + max_time - 1.0:
        c = fft_convolve_self(a)
        if c.size == 0:
            break
        m = float(np.max(c))
        if not np.isfinite(m):
            break

        if hard_variant % 2 == 0:
            gamma = 0.02 + 0.02 * (it % 4)
            diff = m - c
            u = np.exp(-diff / max(gamma, 1e-12))
            u[diff > 8.0 * gamma] = 0.0
            su = float(u.sum())
            if su <= 1e-300:
                mask = c > max(0.0, m - 0.05)
                if float(np.sum(mask)) == 0:
                    u = np.ones_like(c) * (1.0 / max(c.size, 1))
                    su = 1.0
                else:
                    u = mask.astype(np.float64)
                    su = float(u.sum())
            u = u / su
        else:
            topk = min(c.size, 16 + 4 * (it % 12))
            idx = np.argsort(c)[::-1][:topk]
            vals = c[idx]
            base = max(0.0, float(np.max(vals)) - 0.05)
            wv = np.maximum(vals - base, 1e-10)
            u = np.zeros_like(c)
            u[idx] = wv / max(float(wv.sum()), 1e-300)

        q = corr_weights(a, u, use_direct=True)
        qmax = float(np.max(q))
        if qmax <= 1e-300 or not np.isfinite(qmax):
            break

        tiny = 1e-12 * qmax
        g = 1.0 / np.maximum(q, tiny)
        med = float(np.median(g))
        if med <= 0.0 or not np.isfinite(med):
            med = 1e-12
        g = np.minimum(g, 50.0 * med)
        sg = float(g.sum())
        if sg <= 1e-300 or not np.isfinite(sg):
            continue
        g = g * (S / sg)

        accepted = False
        for lam in (1.0, 0.75, 0.5, 0.3, 0.15, 0.05):
            cand = (1.0 - lam) * a + lam * g
            cand = normalize_to_sum(cand, S)
            sc = fast_score(cand)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = float(sc)
                best_a = cand.copy()
                accepted = True
                break

        if accepted:
            a = best_a.copy()
            it = 0
            hard_variant += 1
            print("      LP refine: %.8f" % best_score)
        else:
            it += 1
            if it % 5 == 0:
                noise = 0.02 * S / np.sqrt(float(n))
                a = normalize_to_sum(a + rng.normal(0.0, noise, n), S)
                sc = fast_score(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = float(sc)
                    best_a = a.copy()
            if it > 70:
                a = normalize_to_sum(0.9 * best_a + 0.1 * rng.random(n), S)
                hard_variant = 0

    return best_a, float(best_score)


def upsample_sequence(a, n2):
    """Linear interpolation between grid points (periodic grid endpoint=False)."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(a.size)
    n2 = int(n2)
    if n2 <= 0:
        return np.array([], dtype=np.float64)
    if n == 0:
        return np.zeros(n2, dtype=np.float64)
    if n == n2:
        return a.copy()
    if n == 1:
        return np.full(n2, float(a[0]), dtype=np.float64)

    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n2, endpoint=False)
    return np.interp(x_new, x_old, a).astype(np.float64, copy=False)


def build_inits(n, S, seed):
    """Construct several diverse nonnegative initializations."""
    rng = np.random.default_rng(seed)
    inits = []
    n = int(n)
    S = float(S)

    inits.append(("uniform", np.ones(n, dtype=np.float64) * (S / float(n))))

    t = np.linspace(0.0, 1.0, n, endpoint=False)

    arr = (1.0
           + 0.30 * np.sin(2.0 * np.pi * 2.0 * t)
           + 0.20 * np.cos(2.0 * np.pi * 5.0 * t)
           + 0.14 * np.sin(2.0 * np.pi * 11.0 * t)
           + 0.08 * np.cos(2.0 * np.pi * 23.0 * t))
    arr = np.maximum(arr, 1e-6)
    inits.append(("sine", normalize_to_sum(arr, S)))

    arr = (0.9 + 0.55 * np.cos(2.0 * np.pi * 3.0 * t) ** 2
           + 0.18 * np.sin(2.0 * np.pi * 8.0 * t) ** 2)
    arr = np.maximum(arr, 1e-6)
    inits.append(("cos2", normalize_to_sum(arr, S)))

    arr = (np.exp(-6.0 * (t - 0.25) ** 2)
           + 0.6 * np.exp(-6.0 * (t - 0.65) ** 2)
           + 0.15)
    inits.append(("bump", normalize_to_sum(arr, S)))

    r = rng.integers(0, 2, size=n).astype(np.float64)
    r = 2.0 * r - 1.0
    arr = 1.0 + 0.55 * r
    arr = np.maximum(arr, 1e-6)
    inits.append(("sign", normalize_to_sum(arr, S)))

    arr = rng.random(n, dtype=np.float64) + 0.1
    inits.append(("rand", normalize_to_sum(arr, S)))

    return inits


def score_list_safe(candidate):
    """Use evaluate_sequence if available; otherwise use fast surrogate."""
    try:
        val = evaluate_sequence(candidate)
        val = float(val)
        if np.isfinite(val):
            return val
    except Exception:
        pass

    try:
        arr = np.asarray(candidate, dtype=np.float64)
        val = fast_score(normalize_to_sum(arr, np.sqrt(2.0 * max(int(arr.size), 1))))
        if np.isfinite(val):
            return val
    except Exception:
        pass

    return np.inf


def propose_candidate(seed=42, budget_s=2360, **kwargs):
    """
    Search for a nonnegative sequence minimizing the given evaluation function.
    Uses the previous construction when available, then performs multiple
    optimization phases.
    """
    start_all = time.monotonic()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2360.0
    if not np.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 2360.0
    deadline = start_all + max(0.0, budget_s - 25.0)

    rng = np.random.default_rng(seed)

    given = None
    try:
        raw = height_sequence_1
        arr = np.asarray(raw, dtype=np.float64).reshape(-1)
        if arr.size >= 8:
            g = np.clip(arr, 0.0, 1000.0)
            if float(g.sum()) > 1e-12:
                given = g
    except Exception:
        given = None

    if given is not None:
        n = int(given.size)
        if not (512 <= n <= 8192):
            n = 8192
    else:
        n = 8192

    S = np.sqrt(2.0 * float(n))
    global_a = None
    global_best = np.inf
    fallback_a = None

    if given is not None:
        g = upsample_sequence(given, n)
        g = normalize_to_sum(g, S)
        fallback_a = g.copy()
        global_a = g.copy()
        global_best = fast_score(g)
        print("Initial construction length=%d fast_score=%.8f" % (n, global_best))
    else:
        global_a = np.ones(n, dtype=np.float64) * (S / float(n))
        fallback_a = global_a.copy()
        global_best = fast_score(global_a)
        print("No previous construction; using uniform length=%d fast_score=%.8f"
              % (n, global_best))

    # ---------- Phase 1: main LSE continuation ----------
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 45.0:
        init_sc = global_best
        if init_sc < 1.53:
            betas = [20, 50, 100, 200, 400, 800, 1600, 3200, 6400, 10000]
        elif init_sc < 1.80:
            betas = [10, 20, 50, 100, 200, 400, 800, 1600, 3200, 6400]
        else:
            betas = [3, 5, 10, 20, 50, 100, 200, 400, 800, 1600, 3200, 6400]
        b1 = min(remaining * 0.38, 650.0)
        a_new, sc = run_lbfgs_continuation(
            global_a, S, b1, betas,
            maxiter=120 if n <= 8192 else 80,
            rng=rng,
            use_direct=True,
            reg=1e-10,
            label="main"
        )
        if np.isfinite(sc) and sc < global_best:
            global_best = sc
            global_a = a_new.copy()
        print("After phase 1: %.8f" % global_best)

    # ---------- Phase 2: thresholded excess polishing ----------
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 55.0 and global_best > 1.005:
        drops = (0.0005, 0.0015, 0.0025, 0.0040, 0.0055, 0.0070, 0.0085, 0.0100)
        T_list = [max(1.005, global_best - d) for d in drops]
        b2 = min(remaining * 0.16, 240.0)
        a_new, sc = run_excess_continuation(
            global_a, S, b2, T_list,
            maxiter=90,
            rng=rng,
            use_direct=True,
            reg=1e-12,
            label="excess"
        )
        if np.isfinite(sc) and sc < global_best:
            global_best = sc
            global_a = a_new.copy()
        print("After phase 2: %.8f" % global_best)

    # ---------- Phase 3: LP-like multiplicative refinement ----------
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 45.0:
        b3 = min(remaining * 0.12, 150.0)
        a_new, sc = lp_refine(global_a, S, b3, rng)
        if np.isfinite(sc) and sc < global_best:
            global_best = sc
            global_a = a_new.copy()
        print("After phase 3: %.8f" % global_best)

    # ---------- Phase 4: perturb and restart ----------
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 140.0:
        cur_n = int(global_a.size)
        cur_S = np.sqrt(2.0 * float(cur_n))
        perturb_budget = min(remaining * 0.28, 380.0)
        per_perturb = max(18.0, perturb_budget / 6.0)
        pert_specs = [
            (0.02, 0.00, "noise"),
            (0.05, 0.00, "noise"),
            (0.10, 0.00, "noise"),
            (0.05, 0.15, "blend"),
            (0.03, 0.20, "blend"),
            (0.08, 0.00, "noise"),
        ]
        print("Phase 4: perturb/restart, total=%.1fs" % perturb_budget)
        for pidx, (scf, blf, kind) in enumerate(pert_specs):
            if time.monotonic() > deadline - 70.0:
                break
            scale = scf * cur_S / np.sqrt(float(cur_n))
            if kind == "noise":
                a_p = normalize_to_sum(global_a + rng.normal(0.0, scale, cur_n), cur_S)
            else:
                rnd = rng.random(cur_n, dtype=np.float64)
                a_p = normalize_to_sum((1.0 - blf) * global_a + blf * rnd, cur_S)

            sp = fast_score(a_p)
            if not np.isfinite(sp) or (sp > global_best + 0.05 and pidx > 1):
                continue

            betas_p = [200, 500, 1000, 3000, 8000]
            a_new, sc = run_lbfgs_continuation(
                a_p, cur_S, per_perturb, betas_p,
                maxiter=70,
                rng=rng,
                use_direct=True,
                reg=1e-10,
                label="perturb%d" % pidx
            )
            if np.isfinite(sc) and sc < global_best:
                global_best = sc
                global_a = a_new.copy()
                print("    perturb %d improved to %.8f" % (pidx, global_best))

                short_lp = min(12.0, max(0.0, deadline - time.monotonic() - 50.0))
                if short_lp > 8.0:
                    a_lp, sc_lp = lp_refine(global_a, cur_S, short_lp, rng)
                    if np.isfinite(sc_lp) and sc_lp < global_best:
                        global_best = sc_lp
                        global_a = a_lp.copy()

        print("After phase 4: %.8f" % global_best)

    # ---------- Phase 5: diverse starts, if time remains ----------
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 130.0:
        cur_n = int(global_a.size)
        cur_S = np.sqrt(2.0 * float(cur_n))
        inits = build_inits(cur_n, cur_S, int(seed) + 1)
        print("Phase 5: diverse starts, time=%.1fs" % remaining)
        for name, arr in inits:
            if time.monotonic() > deadline - 70.0:
                break
            if name == "uniform" and given is not None:
                continue
            per_div = min(80.0, (deadline - time.monotonic()) * 0.18)
            if per_div <= 12.0:
                break
            betas_d = [1, 3, 5, 10, 20, 50, 100, 200, 500, 1000, 3000]
            a_new, sc = run_lbfgs_continuation(
                arr, cur_S, per_div, betas_d,
                maxiter=60,
                rng=rng,
                use_direct=True,
                reg=1e-10,
                label="diverse_" + name
            )
            if np.isfinite(sc) and sc < global_best:
                global_best = sc
                global_a = a_new.copy()
                print("    diverse %s improved to %.8f" % (name, global_best))

        print("After phase 5: %.8f" % global_best)

    # ---------- Phase 6: optional longer-length experiment ----------
    remaining = max(0.0, deadline - time.monotonic())
    cur_n = int(global_a.size)
    if remaining > 260.0 and 1024 <= cur_n < 16384:
        n2 = min(16384, cur_n * 2)
        S2 = np.sqrt(2.0 * float(n2))
        a2 = upsample_sequence(global_a, n2)
        a2 = normalize_to_sum(a2, S2)
        b6 = min(220.0, (deadline - time.monotonic()) * 0.35)
        betas_l = [20, 50, 100, 200, 500, 1000, 3000, 8000, 16000]
        print("Phase 6: longer experiment n=%d, budget=%.1fs" % (n2, b6))
        a_new, sc = run_lbfgs_continuation(
            a2, S2, b6, betas_l,
            maxiter=70,
            rng=rng,
            use_direct=True,
            reg=1e-10,
            label="longer"
        )
        if np.isfinite(sc) and sc < global_best - 5e-5:
            global_best = sc
            global_a = a_new.copy()
            print("    longer improved to %.8f (n=%d)" % (global_best, n2))
        else:
            print("    longer no advisory improvement: %.8f" % sc)

        print("After phase 6: %.8f" % global_best)

    # ---------- Phase 7: final high-beta polish ----------
    remaining = max(0.0, deadline - time.monotonic())
    cur_n = int(global_a.size)
    cur_S = np.sqrt(2.0 * float(cur_n))
    if remaining > 50.0:
        sc_now = fast_score(global_a)
        if sc_now < 1.53:
            betas_f = [200, 500, 1000, 2000, 4000, 8000, 15000]
            maxiter_f = 100
        else:
            betas_f = [100, 200, 500, 1000, 2000, 4000, 8000]
            maxiter_f = 110
        b7 = min(remaining * 0.62, 320.0)
        a_new, sc = run_lbfgs_continuation(
            global_a, cur_S, b7, betas_f,
            maxiter=maxiter_f,
            rng=rng,
            use_direct=True,
            reg=1e-10,
            label="final_lse"
        )
        if np.isfinite(sc) and sc < global_best:
            global_best = sc
            global_a = a_new.copy()

        remaining = max(0.0, deadline - time.monotonic())
        if remaining > 25.0:
            a_lp, sc_lp = lp_refine(global_a, cur_S, min(remaining * 0.45, 60.0), rng)
            if np.isfinite(sc_lp) and sc_lp < global_best:
                global_best = sc_lp
                global_a = a_lp.copy()

        remaining = max(0.0, deadline - time.monotonic())
        cur_S = np.sqrt(2.0 * float(global_a.size))
        if remaining > 25.0 and global_best > 1.005:
            Tlist = [
                max(1.005, global_best - 0.0010),
                max(1.005, global_best - 0.0030),
                max(1.005, global_best - 0.0050),
                max(1.005, global_best - 0.0070),
            ]
            a_new, sc = run_excess_continuation(
                global_a, cur_S, min(remaining * 0.5, 80.0), Tlist,
                maxiter=70,
                rng=rng,
                use_direct=True,
                reg=1e-12,
                label="final_excess"
            )
            if np.isfinite(sc) and sc < global_best:
                global_best = sc
                global_a = a_new.copy()

        print("After phase 7: %.8f" % global_best)

    # ---------- Final validation and return ----------
    final_S = np.sqrt(2.0 * float(max(int(global_a.size), 1)))
    final_a = normalize_to_sum(global_a, final_S)
    final_list = [float(x) for x in final_a]

    candidates = [final_list]
    if fallback_a is not None:
        fallback_list = [float(x) for x in np.clip(fallback_a, 0.0, 1000.0)]
        candidates.append(fallback_list)
    if len(final_list) >= 8:
        uniform_fallback = [float(v) for v in np.ones(len(final_list), dtype=np.float64) *
                            (1.0 / float(len(final_list)))]
        candidates.append(uniform_fallback)

    best_val = np.inf
    best_list = final_list
    for cand in candidates:
        try:
            val = score_list_safe(cand)
        except Exception:
            val = np.inf
        if np.isfinite(val) and val < best_val:
            best_val = val
            best_list = cand

    if not np.isfinite(best_val):
        best_list = final_list

    fin_arr = np.asarray(best_list, dtype=np.float64)
    if fin_arr.size == 0 or float(np.sum(np.clip(fin_arr, 0.0, 1000.0))) <= 0.01:
        L = max(8192, int(fin_arr.size))
        best_list = [float(v) for v in np.ones(L, dtype=np.float64) * (1.0 / float(L))]
    else:
        best_list = [float(max(0.0, min(1000.0, x))) for x in fin_arr]

    elapsed = time.monotonic() - start_all
    print("Final fast_score=%.8f length=%d elapsed=%.1fs" %
          (floating:0.000000 % 0 if False else global_best, int(len(best_list)), elapsed))
    print("Returning sequence of length %d" % len(best_list))
    return best_list
```
