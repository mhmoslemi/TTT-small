"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: LP-guided direction finding (cutting plane) combined with 
Nesterov-accelerated top-k subgradient polishing.
Key novelty: Uses linear programming to find optimal update directions,
which provides globally better directions than pure subgradient methods.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_correlate(a, b):
    """Full cross-correlation/convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient given weights on convolution lags using FFT."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Compute normalized top-k weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights


def construct_constraint_row(a, k):
    """Construct row k of the cross-correlation constraint matrix.
    Returns coeff vector v such that v @ b = (a*b)(k)."""
    n = len(a)
    v = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        v[lo:hi+1] = a[k-hi:k-lo+1][::-1]
    return v


def solve_cutting_plane_lp(a, M, max_constraints=200, max_iters=20, tol=1e-6):
    """
    Solve: max sum(b) s.t. (a*b)(k) <= M for all k, b >= 0
    Using cutting plane method.
    """
    n = len(a)
    # Start with top lags
    r = fft_convolve_self(a)  # Actually for the LP we need (a*b), but we use r to pick initial lags
    # For initial active set, use lags where (a*a) is highest
    active_lags = np.argpartition(r, -50)[-50:]
    
    for it in range(max_iters):
        m = len(active_lags)
        # Build constraint matrix
        A_ub = np.zeros((m, n))
        for i, k in enumerate(active_lags):
            A_ub[i] = construct_constraint_row(a, int(k))
        b_ub = np.full(m, M)
        c_obj = -np.ones(n)  # minimize -sum(b)
        
        # Solve LP
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs', 
                      options={'presolve': True, 'time_limit': 30})
        
        if not res.success:
            return None, active_lags
        
        b_star = res.x
        
        # Check for violated constraints
        # Compute (a*b_star) for all lags
        cross = fft_cross_correlate(a, b_star)
        violations = np.where(cross > M + tol)[0]
        
        if len(violations) == 0:
            return b_star, active_lags
        
        # Add violated constraints (up to max_constraints total)
        available_slots = max_constraints - len(active_lags)
        if available_slots <= 0:
            # Already at max, just return current solution
            return b_star, active_lags
        
        # Prioritize most violated
        violation_amounts = cross[violations] - M
        top_violated = violations[np.argsort(violation_amounts)[::-1][:min(available_slots, len(violations))]]
        active_lags = np.unique(np.concatenate([active_lags, top_violated]))
        active_lags = active_lags[:max_constraints]
    
    return b_star, active_lags


def lp_guided_optimize(a_init, n, S, max_time, rng):
    """LP-guided optimization with cutting planes."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    # Initial peak
    c = fft_convolve_self(a)
    best_score = c.max()
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()

            best_score = cur
            best_local = a.copy()

            if gnorm < 1e-14:
                break

        else:

                best_t = 0
                best_new_score = cur
                for t in [1.0, 0.7, 0.5, 0.3, 0.1]:

                    a_candidate = np.maximum(a_candidate, 0.0)
                    s_c = a_candidate.sum()
                    if s_c < 1e-15:
                        continue

                    c_cand = fft_convolve_self(a_candidate)

    return best_local, best_score


def nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=1, top_k_max=50, top_k_rate=100, 
                    step_init=None):
    """Nesterov top-k subgradient method."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    c0 = fft_convolve_self(a)
    best_score = c0.max()
    
    if step_init is None:
        step = 2.0 * S / (n * 3.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // top_k_rate)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
        
        weights = toppk_weights(c, top_k)
        grad = compute_gradient_fft(y, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step with decay
        step = step_init / (1.0 + 0.01 * it) if step_init else 2.0 * S / (n * 3.0 * (1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic perturbation
        if it % 1500 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
    
    return best_local, best_score

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: LP-guided direction finding (cutting plane) + Nesterov-accelerated
top-k subgradient + spectral shaping + multi-restart refinement.
Key novelty: Uses linear programming to find globally better update directions
than pure subgradient methods, combined with spectral domain shaping.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(a, b):
    """Full convolution of a and b via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient given weights on convolution lags using FFT."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Compute normalized top-k weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights


def construct_constraint_row(a, k):
    """Construct row k of the cross-correlation constraint matrix.
    Returns coeff vector v such that v @ b = (a*b)(k)."""
    n = len(a)
    v = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        v[lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return v


def solve_cutting_plane_lp(a, M, max_constraints=150, max_iters=15, tol=1e-8):
    """
    Solve: max sum(b) s.t. (a*b)(k) <= M for all k, b >= 0
    Using cutting plane method.
    Returns (b_star, success_flag).
    """
    n = len(a)
    r = fft_convolve_self(a)
    active_lags = np.argpartition(r, -min(30, 2 * n - 1))[-min(30, 2 * n - 1):]

    b_star = None
    for it in range(max_iters):
        m = len(active_lags)
        A_ub = np.zeros((m, n))
        for i, k in enumerate(active_lags):
            A_ub[i] = construct_constraint_row(a, int(k))
        b_ub = np.full(m, M)
        c_obj = -np.ones(n)

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs',
                      options={'presolve': True})

        if not res.success:
            if b_star is not None:
                return b_star, True
            return None, False

        b_star = res.x

        cross = fft_convolve(a, b_star)
        violations = np.where(cross > M + tol)[0]

        if len(violations) == 0:
            return b_star, True

        available = max_constraints - len(active_lags)
        if available <= 0:
            return b_star, True

        viol_amounts = cross[violations] - M
        n_add = min(available, len(violations), 50)
        top_violated = violations[np.argsort(viol_amounts)[::-1][:n_add]]
        active_lags = np.unique(np.concatenate([active_lags, top_violated]))
        active_lags = active_lags[:max_constraints]

    return b_star, True


def lp_guided_optimize(a_init, n, S, max_time, rng, max_constraints=150):
    """LP-guided optimization with cutting planes and line search."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    it = 0
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 50:
            # Perturb and reset
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            stall = 0
            continue

        M = cur
        b_star, success = solve_cutting_plane_lp(a, M, max_constraints=max_constraints, max_iters=10)

        if not success or b_star is None:
            weights = toppk_weights(c, 5)
            grad = compute_gradient_fft(a, weights)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = S / (n * 4.0)
            a = project_simplex(a - step * grad / gnorm, S)
            continue

        sum_b = b_star.sum()
        if sum_b < 1e-15 or abs(sum_b - S) < 1e-8 * S:
            weights = toppk_weights(c, 3)
            grad = compute_gradient_fft(a, weights)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = S / (n * 4.0)
            a = project_simplex(a - step * grad / gnorm, S)
            continue

        b_norm = b_star * (S / sum_b)

        best_t = 0.0
        best_new_score = cur
        for t_val in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            a_cand = a + t_val * (b_norm - a)
            a_cand = np.maximum(a_cand, 0.0)
            s_cand = a_cand.sum()
            if s_cand < 1e-15:
                continue
            a_cand = a_cand * (S / s_cand)
            c_cand = fft_convolve_self(a_cand)
            if c_cand.max() < best_new_score:
                best_new_score = c_cand.max()
                best_t = t_val

        if best_t > 0:
            a = a + best_t * (b_norm - a)
            a = np.maximum(a, 0.0)
            s_new = a.sum()
            if s_new > 1e-15:
                a = a * (S / s_new)
        else:
            weights = toppk_weights(c, 3)
            grad = compute_gradient_fft(a, weights)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = S / (n * 4.0)
            a = project_simplex(a - step * grad / gnorm, S)

    return best_local, best_score


def spectral_shaping(a_init, n, S, max_time, rng, gamma=0.5):
    """Spectral shaping: flatten the power spectrum to reduce autocorrelation peaks."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()

        A = fft(a, n)
        mag = np.abs(A)
        phase = np.angle(A)
        mean_mag = mag.mean()

        # Flatten magnitude
        new_mag = mag * (1.0 - gamma) + gamma * mean_mag
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, n))
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * (S / s_new)
        a = a_new

    return best_local, best_score


def nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=1, top_k_max=50,
                    top_k_rate=100, step_init=None):
    """Nesterov top-k subgradient method with adaptive step."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    c0 = fft_convolve_self(a)
    best_score = c0.max()
    best_local = a.copy()
    it = 0

    if step_init is None:
        step_base = 2.0 * S / (n * 3.0)
    else:
        step_base = step_init

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // top_k_rate)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = y.copy()

        weights = toppk_weights(c, top_k)
        grad = compute_gradient_fft(y, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = step_base / (1.0 + 0.005 * it)
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 2000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def generate_flat_spectrum_init(n, rng):
    """Generate initialization with flat spectrum target."""
    phase = rng.uniform(0, 2 * np.pi, n)
    magnitude = np.ones(n)
    A = magnitude * np.exp(1j * phase)
    a = np.real(ifft(A, n))
    a = np.abs(a)
    a = np.maximum(a, 0.0)
    return a


def generate_smooth_random_init(n, rng, kernel_size=64):
    """Generate smooth random non-negative initialization."""
    noise = rng.normal(0, 1, n)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(noise, kernel, mode='same')
    a = np.abs(smooth)
    a = np.maximum(a, 0.0)
    return a

    except Exception:
        pass

    if given_a is not None:
        n = len(given_a)
    else:
        n = 8192
        given_a = np.ones(n) * 0.5

    S = np.sqrt(2.0 * float(n))

    # === Phase 0: Multiple diverse initializations ===
    print("Phase 0: Generating diverse initializations...")
    inits = []
    a0 = safe_normalize(given_a, n)
    inits.append(("given", a0))
    a1 = safe_normalize(generate_flat_spectrum_init(n, rng), n)
    inits.append(("flat_spec", a1))
    a2 = safe_normalize(generate_smooth_random_init(n, rng, 32), n)
    inits.append(("smooth_rand", a2))
    a3 = safe_normalize(generate_smooth_random_init(n, rng, 128), n)
    inits.append(("smooth_rand2", a3))

    best_init = None
    best_init_score = np.inf
    for name, a_i in inits:
        c_i = fft_convolve_self(a_i)
        score_i = c_i.max()
        obj_i = 2.0 * n * score_i / (S * S)
        print("  %s: score=%.6f obj=%.6f" % (name, score_i, obj_i))
        if score_i < best_init_score:
            best_init_score = score_i
            best_init = a_i.copy()

    a_start = best_init
    global_best = best_init_score
    global_a = best_init.copy()
    print("Best init objective: %.6f" % (2.0 * n * global_best / (S * S)))

    # === Phase 1: Nesterov warm-up (12% time) ===
    remaining = deadline - time.monotonic()
    phase1_time = 0.12 * remaining

        c_p1 = fft_convolve_self(a_p1)
        if c_p1.max() < global_best:
            global_best = c_p1.max()

            print("  After warm-up: %.6f (obj %.6f) [%.0fs]" % (

    # === Phase 2: LP-guided optimization (30% time) ===
    remaining = deadline - time.monotonic()
