import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def logsumexp_smooth_max(c, T):
    """Compute log-sum-exp smooth max and its weights."""
    c_max = c.max()
    if c_max < 1e-30:
        return 0.0, np.zeros(len(c))
    scaled = T * (c - c_max)
    scaled = scaled - scaled.max()  # numerical stability
    exp_scaled = np.exp(scaled)
    w_sum = exp_scaled.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = exp_scaled / w_sum
    smooth_max = c_max + np.log(w_sum) / T
    return smooth_max, w


def compute_gradient(c, a, w):
    """Compute gradient of weighted sum: g[p] = 2 * sum_m w[m] * a[m-p]"""
    n = len(a)
    # Cross-correlation of w with a
    fsize = 1 << (n + len(w) - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    """Compute the actual objective: 2*n*max(conv)/sum^2"""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def optimize_logsumexp_adam(a_init, n, S, time_budget, T_schedule=None, verbose=False):
    """
    Optimize log-sum-exp smooth max using Adam optimizer.
    Directly optimizes in the original space with projection.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    n_par = n
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t_step = 0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t_step += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Temperature schedule
        if T_schedule is not None:
            T = T_schedule(progress)
        else:
            # Default: logarithmic increase with periodic resets
            base_T = 5.0 * (1.0 + 20.0 * progress ** 1.5)
            # Periodic reset every 20% of time
            cycle = (progress * 5.0) % 1.0
            T = base_T * (0.5 + 0.5 * cycle)
        
        # Compute smooth max and weights
        smooth_val, w = logsumexp_smooth_max(c, T)
        
        # Compute gradient
        g = compute_gradient(c, a, w)
        
        # Normalize gradient (for Adam)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t_step)
        v_hat = v / (1.0 - beta2 ** t_step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = normalize_nonneg(a_new, S)
        
        c = fft_convolve_self(a_new)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a_new.copy()
            stagnant_steps = 0
        else:
            stagnant_steps += 1
        
        # If stagnant, add perturbation
        if stagnant_steps > max_stagnant:
            noise_scale = 0.01 * (1.0 + stagnant_steps / max_stagnant)
            a_new = best_a + noise_scale * np.random.default_rng(t_step).standard_normal(n_par)
            a_new = normalize_nonneg(a_new, S)
            c = fft_convolve_self(a_new)
            stagnant_steps = 0
            m = np.zeros(n_par)
            v = np.zeros(n_par)
            t_step = 0  # Reset Adam moments
        
        a = a_new
        
        if verbose and t_step % 1000 == 0:
            print("    AdamV2 step %d: obj=%.7f, T=%.1f, stagnant=%d" % (
                t_step, best_obj, T, stagnant_steps))
    
    return best_a, best_obj


def smart_coordinate_descent(a_init, n, S, time_budget, verbose=False):
    """
    Coordinate descent that targets the peak of the convolution.
    Instead of random coordinates, picks coordinates that most affect the peak lag.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    iteration = 0
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - start < time_budget:
        iteration += 1
        
        if no_improve > max_no_improve:
            # Reset with small perturbation
            a = best_a + 0.005 * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            no_improve = 0
            continue
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find the peak lag
        peak_lag = np.argmax(c)
        
        # The peak convolution value c[peak_lag] = sum_k a[k]*a[peak_lag-k]
        # To reduce it, we can:
        # 1. Reduce a[k] for k that contribute most to the peak
        # 2. Or increase a[k] for k that DON'T contribute to the peak (to shift mass)
        
        # Find top contributors to the peak
        products = np.zeros(n)
        for k in range(n):
            j = peak_lag - k
            if 0 <= j < n:
                products[k] = a[k] * a[j]
        
        # Get top 10% contributors
        n_top = max(1, n // 10)
        top_idx = np.argsort(products)[-n_top:]
        
        # Pick a random coordinate from the top contributors
        # and try to reduce it (or a random other coordinate to increase)
        if rng.random() < 0.5:
            # Reduce a top contributor
            i = top_idx[rng.integers(len(top_idx))]
        else:
            # Increase a non-top coordinate
            non_top = np.setdiff1d(np.arange(n), top_idx)
            i = non_top[rng.integers(len(non_top))]
        
        # Try several step sizes
        original = a[i]
        best_local = original
        best_local_obj = cur_obj
        
        for factor in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]:
            a[i] = original * factor
            a = normalize_nonneg(a, S)
            c_check = fft_convolve_self(a)
            obj_check = c_check.max()
            if obj_check < best_local_obj - 1e-12:
                best_local_obj = obj_check
                best_local = original * factor
        
        a[i] = best_local
        a = normalize_nonneg(a, S)
    
    return best_a, best_obj


def multi_peak_gradient(a, n, S, T, k_peaks=3):
    """
    Compute gradient targeting the top-k peaks of the convolution.
    This helps balance multiple competing peaks.
    """
    c = fft_convolve_self(a)
    
    # Find top-k peaks
    peak_indices = np.argsort(c)[-k_peaks:]
    
    # Weight: equal weight on top-k peaks
    w = np.zeros(len(c))
    for idx in peak_indices:
        w[idx] = 1.0
    w /= w.sum()
    
    return compute_gradient(c, a, w), c


def optimize_multi_peak(a_init, n, S, time_budget, n_peaks=3, verbose=False):
    """
    Optimize targeting the top-n peaks simultaneously using Adam.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    n_par = n
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t_step = 0
    beta1, beta2 = 0.85, 0.999
    eps = 1e-8
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t_step += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Gradient based on top-k peaks
        g, c = multi_peak_gradient(a, n, S, T=0, k_peaks=n_peaks)
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t_step)
        v_hat = v / (1.0 - beta2 ** t_step)
        
        lr = 0.03 * (1.0 - 0.5 * progress)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = normalize_nonneg(a_new, S)
        
        c_new = fft_convolve_self(a_new)
        cur_obj = c_new.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a_new.copy()
        
        a = a_new
        c = c_new
    
    return best_a, best_obj


def spectral_init(n, S, rng, method="random_phase"):
    """
    Generate initialization using frequency domain methods.
    """
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    if method == "random_phase":
        # Random magnitude, random phase -> real non-negative after projection
        mag = rng.chisquare(df=n, size=fsize)  # Random positive magnitudes
        phase = rng.uniform(0, 2 * np.pi, fsize)
        # Ensure Hermitian symmetry for real signal
        A = mag * np.exp(1j * phase)
        A[0] = mag[0]  # DC is real
        for k in range(1, fsize // 2 + 1):
            A[fsize - k] = np.conj(A[k])
        a = np.real(ifft(A))[:n]
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = np.full(n, S / n)
        return a * (S / a.sum())
    
    elif method == "low_papr_spectrum":
        # Try to create a spectrum that gives low peak in time domain convolution
        # Flat magnitude spectrum
        mag = np.ones(fsize)
        phase = rng.uniform(0, 2 * np.pi, fsize)
        phase[0] = 0
        A = mag * np.exp(1j * phase)
        for k in range(1, fsize // 2 + 1):
            A[fsize - k] = np.conj(A[k])
        a = np.real(ifft(A))[:n]
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = np.full(n, S / n)
        return a * (S / a.sum())
    
    elif method == "msequence_like":
        # Try to create an m-sequence-like pattern
        a = rng.choice([1.0, 0.3], size=n, p=[0.5, 0.5])
        # Smooth it a bit
        kernel = np.ones(3) / 3
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    
    else:
        return np.full(n, S / n)


def run_fienup_corrected(a_init, n, S, time_budget, verbose=False):
    """
    Correct Fienup-style algorithm:
    1. Time domain: enforce a >= 0, sum(a) = S
    2. Frequency domain: modify spectrum to flatten the convolution
    
    Key insight: The convolution c = a*a has spectrum C = A^2.
    We want c to be flat. In the frequency domain, this means C should be 
    concentrated at DC (for a perfectly flat c). But C = A^2, so we need
    A to have most of its energy at DC, which means a should be constant.
    But constant a gives max/mean ratio = n/(n) = 1 for the triangle... 
    wait, that's not right either.
    
    Let me reconsider. For a constant a, c is a triangle with peak = n*a^2 and 
    mean = S^2/(2n-1). The peak/mean ratio is about n/(1) = n... no.
    
    Actually for constant a_k = c:
    c_conv[m] = (min(m+1, n, 2n-1-m)) * c^2
    peak = n * c^2 = n * (S/n)^2 = S^2/n
    mean = S^2/(2n-1) ≈ S^2/(2n)
    peak/mean ≈ 2
    
    So we need peak/mean < 1.5027 * (2n-1)/(2n) ≈ 1.5027
    
    Wait, the objective is 2n * max_c / S^2. For constant: 2n * (S^2/n) / S^2 = 2.
    We need 2n * max_c / S^2 < 1.5027, so max_c < 1.5027 * S^2 / (2n) = 1.5027/2 * S^2/n... 
    
    With S = sqrt(2n): max_c < 1.5027 * 2n / (2n) = 1.5027.
    
    So we need the peak of the convolution to be < 1.5027 when S = sqrt(2n).
    
    For constant: peak = 2. Need to get it to 1.5027.
    
    The Fienup approach:
    1. Compute c = a*a
    2. "Flatten" c: c_flat = c / max(c) * target_peak where target_peak < current max
    3. Find a' such that a'*a' ≈ c_flat
    4. This requires taking a "convolution square root"
    5. In frequency domain: A' = sqrt(C_flat) (elementwise square root of spectrum)
    6. a' = IDFT(A')
    7. Enforce a' >= 0 and sum = S
    
    This is the key insight! Let me implement this.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    relax = 0.5  # Relaxation parameter for Fienup
    
    while time.monotonic() - start < time_budget:
        it += 1
        
        # Compute current convolution
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Fienup step:
        # 1. Compute spectrum of convolution
        A = fft(a, fsize)
        C = A * A  # Spectrum of c
        
        # 2. "Flatten" the convolution: reduce peaks
        # c_desired = c with peaks compressed
        c_max = c.max()
        if c_max < 1e-10:
            break
        # Compress: c_new = c / (c_max)^(alpha) * mean(c)
        # This reduces peaks while preserving mean approximately
        alpha = 0.5  # Compression strength
        c_mean = c.sum() / size
        c_new = (c / c_max) ** alpha * c_mean
        # Ensure the DC component is preserved (sum must be S^2)
        c_new_sum = c_new.sum()
        c_new *= (S * S) / c_new_sum
        
        # 3. Compute desired spectrum
        C_new = fft(c_new, fsize)
        
        # 4. Take square root to get new A
        # A_new = sqrt(C_new) - but this is complex square root
        # We want to preserve the phase structure of A as much as possible
        # Use: A_new = A * (C_new / C)^(1/2) = A * sqrt(C_new/C)
        # This modifies |A| and phase
        with np.errstate(divide='ignore', invalid='ignore'):
            ratio = C_new / C
            ratio = np.where(np.abs(C) > 1e-30, ratio, 1.0)
        
        # Complex square root
        sqrt_ratio = np.sqrt(ratio)
        A_new = A * sqrt_ratio
        
        # 5. Get new time domain
        a_new = np.real(ifft(A_new))[:n]
        
        # 6. Enforce non-negativity and sum constraint
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s)
        
        # 7. Relaxation
        a = (1 - relax) * a + relax * a_new
        
        if verbose and it % 200 == 0:
            c_check = fft_convolve_self(a)
            print("    Fienup iter %d: obj=%.7f, best=%.7f" % (it, c_check.max(), best_obj))
    
    return best_a, best_obj

Wait, I need to reconsider. The strategy mentions the Fienup approach with "convolution square root" in the frequency domain. Let me make sure this is correct.

If c = a * a (convolution), then in frequency domain: C(ω) = A(ω)² where A(ω) = FFT(a).

If I want to modify c to be flatter (say c_new), then I need:

This is valid! The "convolution square root" is just the elementwise square root of the FFT of the desired convolution.

The issue is that:
1. C_new might not be a perfect square of a valid FFT (it might not correspond to a real signal)
2. The resulting a_new might not be non-negative

But as a Fienup-style iteration, we just project back to the feasible set after each step.

Let me also think about whether there are other approaches I'm missing:

**Alternative: Use the fact that we're minimizing a quasi-convex function**

The function f(a) = max_m (a*a)[m] is convex in a (since each (a*a)[m] is... wait, is it convex?

(a*a)[m] = sum_k a[k]*a[m-k]

This is a quadratic form. For m = n-1 (center): sum_k a[k]*a[n-1-k]. This is NOT necessarily convex in a.

Actually, (a*a)[m] as a function of a is a quadratic form with matrix Q_m where (Q_m)_{ij} = 1 if i+j=m. This matrix is symmetric. The Hessian of (a*a)[m] is 2*Q_m (since it's a quadratic form). Q_m is the "anti-diagonal" of a matrix, which has eigenvalues ±1. So (a*a)[m] is indefinite (neither convex nor concave).

This means the problem is non-convex, and we need global optimization techniques.

**Another idea: Use the relationship to the L2 norm of a**

We have sum_m (a*a)[m] = (sum a)^2 = S^2.
And max_m (a*a)[m] ≥ S^2/(2n-1) by the pigeonhole principle.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def logsumexp_weights(c, T):
    """Compute softmax weights for exponential peak weighting."""
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(len(c))
    scaled = T * (c - c_max)
    scaled = scaled - scaled.max()
    exp_scaled = np.exp(scaled)
    w_sum = exp_scaled.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = exp_scaled / w_sum
    return w


def compute_gradient(c, a, w):
    """Compute gradient: g[p] = 2 * sum_m w[m] * a[m-p] (with a[j]=0 outside [0,n-1])."""
    n = len(a)
    fsize = 1 << (n + len(w) - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def fienup_convolution_sqrt(a_init, n, S, time_budget, alpha=0.5, relax=0.5, verbose=False):
    """
    Fienup-style iteration using convolution square root.
    Key idea: c = a*a has spectrum C = A^2.
    To flatten c, compute c_new (flatter), then A_new = sqrt(FFT(c_new)), a_new = IDFT(A_new).
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    it = 0
    stagnation = 0
    max_stagnation = 300
    
    while time.monotonic() - start < time_budget:
        it += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Adaptive alpha: start gentle, become more aggressive
        alpha_eff = alpha * (0.5 + 0.5 * progress)
        
        # Adaptive relaxation
        relax_eff = relax * (0.5 + 0.5 * (1 - progress))
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # If stagnant, restart from best with perturbation
        if stagnation > max_stagnation:
            noise = 0.01 * (1 + stagnation / max_stagnation)
            a = best_a + noise * np.random.default_rng(it).standard_normal(n)
            a = normalize_nonneg(a, S)
            stagnation = 0
            continue
        
        # Fienup step
        A = fft(a, fsize)
        C = A * A
        
        # Flatten the convolution
        c_max = c.max()
        if c_max < 1e-10:
            break
        c_mean = c.sum() / size
        # Compress peaks: c_new = (c/c_max)^alpha * c_mean, then rescale to preserve sum
        c_new = (np.maximum(c, 0) / c_max) ** alpha_eff * c_mean
        c_new_sum = c_new.sum()
        if c_new_sum > 1e-10:
            c_new *= (S * S) / c_new_sum
        
        # Get desired spectrum and take square root
        C_new = fft(c_new, fsize)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            ratio = C_new / C
            ratio = np.where(np.abs(C) > 1e-30, ratio, 1.0)
        
        sqrt_ratio = np.sqrt(ratio)
        A_new = A * sqrt_ratio
        
        a_new_full = ifft(A_new)
        a_new = np.real(a_new_full[:n])
        
        # Enforce constraints
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new * (S / s)
        
        # Relaxation
        a = (1 - relax_eff) * a + relax_eff * a_new
        
        if verbose and it % 500 == 0:
            c_check = fft_convolve_self(a)
            print("    Fienup iter %d: cur=%.7f, best=%.7f, stagn=%d" % (it, c_check.max(), best_obj, stagnation))
    
    return best_a, best_obj


def optimize_adam_lse(a_init, n, S, time_budget, T_start=10.0, T_end=3000.0,
                      lr_start=0.04, lr_end=0.005, verbose=False):
    """
    Adam optimizer with log-sum-exp smooth max objective.
    Uses adaptive temperature with sawtooth resets.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    n_par = n
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t_step = 0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    stagnant = 0
    max_stagnant = 400
    
    while time.monotonic() - start < time_budget:
        t_step += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Temperature: log ramp with periodic sawtooth
        base_T = T_start * (T_end / T_start) ** progress
        sawtooth = (t_step * 0.003) % 1.0
        T = base_T * (0.4 + 0.6 * sawtooth)
        T = max(T, 5.0)
        
        # Learning rate
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Compute weights and gradient
        w = logsumexp_weights(c, T)
        g = compute_gradient(c, a, w)
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t_step)
        v_hat = v / (1.0 - beta2 ** t_step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = normalize_nonneg(a_new, S)
        
        c = fft_convolve_self(a_new)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Restart if stagnant
        if stagnant > max_stagnant:
            noise = 0.008 * (1.0 + stagnant / max_stagnant)
            a_new = best_a + noise * np.random.default_rng(t_step).standard_normal(n_par)
            a_new = normalize_nonneg(a_new, S)
            c = fft_convolve_self(a_new)
            stagnant = 0
            m = np.zeros(n_par)
            v = np.zeros(n_par)
            t_step = 0
        
        a = a_new
        
        if verbose and t_step % 2000 == 0:
            print("    AdamLSE step %d: obj=%.7f, T=%.1f, stag=%d" % (t_step, best_obj, T, stagnant))
    
    return best_a, best_obj


def optimize_multi_peak(a_init, n, S, time_budget, n_peaks=3, verbose=False):
    """
    Target the top-k peaks simultaneously to balance them.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    n_par = n
    m = np.zeros(n_par)
    v = np.zeros(n_par)
    t_step = 0
    beta1, beta2 = 0.85, 0.998
    eps = 1e-8
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    stagnant = 0
    
    while time.monotonic() - start < time_budget:
        t_step += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Weight on top-k peaks with soft weighting
        c_sorted = np.sort(c)[::-1]
        # Use top-k peaks with geometric decay weighting
        w = np.zeros(len(c))
        c_max = c.max()
        for k in range(n_peaks):
            threshold = c_sorted[k]
            mask = np.isclose(c, threshold, atol=c_max * 0.01)
            w[mask] = (0.7 ** k)
        w_sum = w.sum()
        if w_sum > 1e-10:
            w /= w_sum
        else:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        
        g = compute_gradient(c, a, w)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t_step)
        v_hat = v / (1.0 - beta2 ** t_step)
        
        lr = 0.03 * (1.0 - 0.5 * progress)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = normalize_nonneg(a_new, S)
        
        c = fft_convolve_self(a_new)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 300:
            noise = 0.006
            a_new = best_a + noise * np.random.default_rng(t_step).standard_normal(n_par)
            a_new = normalize_nonneg(a_new, S)
            c = fft_convolve_self(a_new)
            stagnant = 0
            m = np.zeros(n_par)
            v = np.zeros(n_par)
            t_step = 0
        
        a = a_new
    
    return best_a, best_obj


def smart_coordinate_descent(a_init, n, S, time_budget, verbose=False):
    """
    Coordinate descent that targets peak-contributing coordinates.
    Uses the peak lag to determine which coordinates to adjust.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(42)
    stagnant = 0
    max_stagnant = 100
    
    while time.monotonic() - start < time_budget:
        if stagnant > max_stagnant:
            a = best_a + 0.005 * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            stagnant = 0
            continue
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Find top 3 peak lags
        top_lags = np.argsort(c)[-3:]
        
        # For each peak lag, find contributing coordinates
        for lag in top_lags:
            # c[lag] = sum_k a[k]*a[lag-k]
            # Try adjusting a few random contributing coordinates
            products = np.zeros(n)
            for k in range(n):
                j = lag - k
                if 0 <= j < n:
                    products[k] = a[k] * a[j]
            
            total = products.sum()
            if total < 1e-10:
                continue
            
            # Pick a coordinate proportional to its contribution
            probs = products / total
            i = rng.choice(n, p=probs)
            
            original = a[i]
            if original < 1e-10:
                # Try increasing a zero
                a[i] = a.mean() * 0.1
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                if c_check.max() < best_obj - 1e-10:
                    best_obj = c_check.max()
                    best_a = a.copy()
                    stagnant = 0
                else:
                    a[i] = 0
                    a = normalize_nonneg(a, S)
                continue
            
            best_local = original
            best_local_obj = c.max()
            
            for factor in [0.8, 0.9, 1.1, 1.2, 0.7, 0.95, 1.05]:
                a[i] = original * factor
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-12:
                    best_local_obj = obj_check
                    best_local = original * factor
            
            a[i] = best_local
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
    
    return best_a, best_obj


def spectral_init(n, S, rng, method="random_phase"):
    """Generate initialization using frequency domain methods."""
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    if method == "random_phase":
        mag = rng.chisquare(df=n // 4 + 1, size=fsize)
        phase = rng.uniform(0, 2 * np.pi, fsize)
        phase[0] = 0.0
        A = mag * np.exp(1j * phase)
        for k in range(1, fsize // 2 + 1):
            A[fsize - k] = np.conj(A[k])
        a = np.real(ifft(A))[:n]
        a = np.maximum(a, 0)
        if a.sum() < 1e-10:
            a = np.abs(a) + 0.1
        return a * (S / a.sum())
    
    elif method == "flat_spectrum":
        mag = np.ones(fsize)
        phase = rng.uniform(0, 2 * np.pi, fsize)
        phase[0] = 0.0
        A = mag * np.exp(1j * phase)
        for k in range(1, fsize // 2 + 1):
            A[fsize - k] = np.conj(A[k])
        a = np.real(ifft(A))[:n]
        a = np.maximum(np.abs(a), 0.01)
        return a * (S / a.sum())
    
    elif method == "bell_shape":
        # Start with a bell shape and add random perturbation
        i = np.arange(n)
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.35)
        a = np.exp(-0.5 * ((i - center) / width) ** 2)
        a += 0.1 * rng.random(n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    
    elif method == "bimodal":
        i = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.15)
        a = np.exp(-0.5 * ((i - c1) / w) ** 2) + np.exp(-0.5 * ((i - c2) / w) ** 2)
        a += 0.05 * rng.random(n)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    
    else:
        return np.full(n, S / n)


def gradient_descent_momentum(a_init, n, S, time_budget, T=100.0, lr=0.05,

    t = 0
    stagnant = 0

        c = fft_convolve_self(a_new)

            a_new = normalize_nonneg(a_new, S)
            c = fft_convolve_self(a_new)
            v = np.zeros(n)
            stagnant = 0
        
        a = a_new
    
    return best_a, best_obj

        a_f, obj_f = fienup_convolution_sqrt(given_a, n, S, p1_budget * 0.35, alpha=0.5, relax=0.6)
        a_f = normalize_nonneg(a_f, S)
        ver = evaluate_sequence(a_f.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_f.copy()
            print("  Fienup(given): obj=%.7f" % ver)
    
    # Fienup from spectral initializations
    remaining = p1_deadline - time.monotonic()
    methods = ["random_phase", "flat_spectrum", "bell_shape", "bimodal"]
    n_trials = min(len(methods), max(2, int(remaining / 20)))
    per_trial = remaining / (n_trials + 1)
    
    for trial in range(n_trials):
        if time.monotonic() > p1_deadline - 5:
            break
        method = methods[trial % len(methods)]
        a_init = spectral_init(n, S, rng, method)
        a_f, _ = fienup_convolution_sqrt(a_init, n, S, per_trial, alpha=0.4 + 0.2 * trial, relax=0.5)
        a_f = normalize_nonneg(a_f, S)
        ver = evaluate_sequence(a_f.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_f.copy()
            print("  Fienup(%s): obj=%.7f" % (method, ver))
    
    # ============ PHASE 2: Adam LSE refinement (35% time) ============
    p2_budget = min(0.35 * total_budget, deadline - time.monotonic())
    p2_deadline = time.monotonic() + p2_budget
    
    if global_a is not None and p2_budget > 15:
        print("\n=== Phase 2: Adam LSE (%.0fs) ===" % p2_budget)
        
        # Round 1: Moderate T
        r1 = 0.35 * p2_budget
        a_b, _ = optimize_adam_lse(global_a, n, S, r1, T_start=10, T_end=500, lr_start=0.04, lr_end=0.01)
        a_b = normalize_nonneg(a_b, S)
        ver = evaluate_sequence(a_b.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_b.copy()
            print("  AdamLSE R1 (T10-500): obj=%.7f" % ver)
        
        # Round 2: High T (focus on dominant peak)
        r2 = 0.30 * p2_budget
        a_b, _ = optimize_adam_lse(global_a, n, S, r2, T_start=100, T_end=5000, lr_start=0.02, lr_end=0.003)
        a_b = normalize_nonneg(a_b, S)
        ver = evaluate_sequence(a_b.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_b.copy()
            print("  AdamLSE R2 (T100-5000): obj=%.7f" % ver)
        
        # Round 3: Low T (broad smoothing) followed by high T
        r3 = 0.35 * p2_budget
        a_b, _ = optimize_adam_lse(global_a, n, S, r3 * 0.4, T_start=5, T_end=50, lr_start=0.05, lr_end=0.02)
        a_b = normalize_nonneg(a_b, S)
        a_b, _ = optimize_adam_lse(a_b, n, S, r3 * 0.6, T_start=50, T_end=3000, lr_start=0.02, lr_end=0.003)
        a_b = normalize_nonneg(a_b, S)
        ver = evaluate_sequence(a_b.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_b.copy()
            print("  AdamLSE R3 (low->high T): obj=%.7f" % ver)
    
    # ============ PHASE 3: Multi-peak + Momentum (20% time) ============
    p3_budget = min(0.20 * total_budget, deadline - time.monotonic())
    p3_deadline = time.monotonic() + p3_budget
    
    if global_a is not None and p3_budget > 10:
        print("\n=== Phase 3: Multi-peak + Momentum (%.0fs) ===" % p3_budget)
        
        # Multi-peak with 3 peaks
        r1 = 0.4 * p3_budget
        a_b, _ = optimize_multi_peak(global_a, n, S, r1, n_peaks=3)
        a_b = normalize_nonneg(a_b, S)
        ver = evaluate_sequence(a_b.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_b.copy()
            print("  MultiPeak(3): obj=%.7f" % ver)
        
        # Momentum GD
        r2 = 0.3 * p3_budget
        a_b, _ = gradient_descent_momentum(global_a, n, S, r2, T=50.0, lr=0.03, momentum=0.92)
        a_b = normalize_nonneg(a_b, S)
        ver = evaluate_sequence(a_b.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_b.copy()
            print("  MomGD: obj=%.7f" % ver)
        
        # Multi-peak with 5 peaks
        r3 = 0.3 * p3_budget
        a_b, _ = optimize_multi_peak(global_a, n, S, r3, n_peaks=5)
        a_b = normalize_nonneg(a_b, S)
        ver = evaluate_sequence(a_b.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_b.copy()
            print("  MultiPeak(5): obj=%.7f" % ver)
    
    # ============ PHASE 4: Polish (20% time) ============
    p4_budget = min(0.20 * total_budget, deadline - time.monotonic())
    p4_deadline = time.monotonic() + p4_budget
    
    if global_a is not None and p4_budget > 10:
        print("\n=== Phase 4: Polish (%.0fs) ===" % p4_budget)
        
        # Coordinate descent
        cd_budget = 0.4 * p4_budget
        a_b, _ = smart_coordinate_descent(global_a, n, S, cd_budget)
        a_b = normalize_nonneg(a_b, S)
        ver = evaluate_sequence(a_b.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_b.copy()
            print("  CoordDesc: obj=%.7f" % ver)
        
        # Fienup-Adam cycle
        fa_budget = 0.35 * p4_budget
        a_f, _ = fienup_convolution_sqrt(global_a, n, S, fa_budget * 0.4, alpha=0.6, relax=0.4)
        a_f = normalize_nonneg(a_f, S)
        a_b, _ = optimize_adam_lse(a_f, n, S, fa_budget * 0.6, T_start=200, T_end=8000, lr_start=0.015, lr_end=0.002)
        a_b = normalize_nonneg(a_b, S)
        ver = evaluate_sequence(a_b.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_b.copy()
            print("  Fienup+Adam: obj=%.7f" % ver)
        
        # Final coordinate descent
        remaining = p4_deadline - time.monotonic()
        if remaining > 5:
            a_b, _ = smart_coordinate_descent(global_a, n, S, remaining)
            a_b = normalize_nonneg(a_b, S)
            ver = evaluate_sequence(a_b.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_b.copy()
                print("  Final CD: obj=%.7f" % ver)
    
    # ============ FINAL CHECK ============
    remaining = deadline - time.monotonic()
    if global_a is not None and remaining > 3:
        # One last coordinate descent pass
        a_b, _ = smart_coordinate_descent(global_a, n, S, remaining)
        a_b = normalize_nonneg(a_b, S)
        ver = evaluate_sequence(a_b.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_b.copy()
            print("  Last CD: obj=%.7f" % ver)
    
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: tracked=%.7f, verified=%.7f" % (global_best, ver))
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        print("Fallback: obj=%.7f" % evaluate_sequence(final_a))
    
    return final_a
