# Helper functions
def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def objective_smoothed(a, c, beta):
    """Log-sum-exp approximation to max(c)."""
    c_max = c.max()
    return c_max + np.log(np.sum(np.exp((c - c_max) * beta)) / len(c)) / beta

    W = fft(w, fsize)
    A = fft(a, fsize)

        if mode == "lse":
            grad = lse_gradient(y, c, beta)
        else:
            temp = 1.0 + 3.0 * (1.0 - progress)
            grad = topk_soft_gradient(y, c, top_k, temp=temp)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Adaptive step size
        if mode == "lse":
            step = 1.0 / max(L, beta * c.max() * 0.5)
        else:
            step = 2.0 * S / (top_k * max(c.max(), 1e-10) * n) * n
            step = min(step, 2.0 * S / (n * gnorm + 1e-15))

        # Backtracking
        x_new = project_simplex(y - step * grad, S)
        for bt in range(4):
            if bt == 0:
                break
            c_cand = fft_convolve_self(x_new)
            if c_cand.max() <= cur_max + 1e-10:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        # Update L
        if c_cand.max() if 'c_cand' in dir() else cur_max > cur_max * 1.001:
            L *= 1.3
        elif it % 50 == 0:
            L = max(L * 0.97, beta * c.max() * 0.3 if mode == "lse" else L * 0.97)

        # Nesterov update with decaying momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        mom_coeff = (t_mom - 1.0) / t_new
        # Decay momentum in later stages
        if progress > 0.7:
            mom_coeff *= 0.5
        y = x_new + mom_coeff * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic structured restart
        if stall > 30 and rng.random() < 0.25:
            # Anti-palindromic snap: blend with anti-palindromic version
            half = n // 2
            a_anti = np.zeros(n)
            for i in range(half):
                avg = (a[i] + a[n - 1 - i]) / 2.0
                a_anti[i] = avg
                a_anti[n - 1 - i] = avg
            if n % 2 == 1:
                a_anti[half] = a[half]
            blend = rng.uniform(0.3, 0.7)
            y = a * (1 - blend) + a_anti * blend
            y = project_simplex(y, S)
            # Add small noise
            noise = 0.02 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0
            stall = 0
            L = max(L, beta * c.max() * 0.5) if mode == "lse" else L

    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, top_k=10, eta_init=0.3):
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        k = min(top_k, len(c))
        grad = hard_topk_gradient(a, c, k)
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break

        eta = eta_init / (1.0 + 0.02 * it) ** 0.5
        log_update = -eta * grad / gmax
        log_update = np.clip(log_update, -4.0, 4.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-25)
        a = a * (S / a.sum())

        # Periodic noise injection
        if it % 2000 == 0 and rng.random() < 0.1:
            noise_scale = 0.01 * S / np.sqrt(n)
            a = np.maximum(a + rng.normal(0, noise_scale, n), 0.0)
            a = a * (S / a.sum())

    return best_local, best_score


def run_coordinate_polish(a_init, n, S, max_time, rng, top_k=20):
    """Coordinate descent: try small perturbations to individual elements."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    c = fft_convolve_self(a)
    best_score = c.max()
    it = 0

    # Precompute which elements contribute most to top lags
    top_lags = np.argsort(c)[::-1][:top_k]

    while time.monotonic() - start < max_time:
        it += 1
        # Update top lags periodically
        if it % 50 == 0:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = a.copy()
            top_lags = np.argsort(c)[::-1][:top_k]

        # Pick a random element to perturb
        j = rng.integers(n)
        # Try a few perturbation sizes
        for delta_frac in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
            delta = delta_frac * S / n
            a[j] = max(0.0, a[j] + delta)
            # Redistribute to maintain sum
            diff = S - a.sum()
            if abs(diff) > 1e-15:
                # Distribute difference to a random set of elements
                others = np.delete(np.arange(n), j)
                target = others[rng.integers(0, len(others), 10)]
                a[target] += diff / 10.0
                a = np.maximum(a, 0.0)
                a = a * (S / a.sum())

            c_new = fft_convolve_self(a)
            if c_new.max() < best_score:
                best_score = c_new.max()
                best_local = a.copy()
                break
            else:
                # Revert
                a = best_local.copy()
                break

    return best_local, best_score


def make_anti_palindromic_init(n, rng, S):
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    a[n - 1 - np.arange(half)] = mu - h
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_broadband_init(n, rng, S):
    """Near-uniform with small random perturbation (broadband spectrum)."""
    a = np.ones(n) + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a * (S / a.sum())
    return a


def make_periodic_init(n, rng, S, num_freqs=3):
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 20)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.4
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a * (S / a.sum())
    return a


def make_tapered_init(n, rng, S):
    """Tapered ends, higher middle - reduces endpoint convolution contributions."""
    tt = np.arange(n, dtype=np.float64) / n
    # Raised cosine taper
    taper = 0.3 + 0.7 * np.sin(np.pi * tt)
    a = taper + 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a * (S / a.sum())
    return a


def make_w_shape_init(n, rng, S):
    tt = np.arange(n, dtype=np.float64) / n
    k = rng.integers(2, 6)
    a = 0.5 + 0.5 * np.abs(np.sin(np.pi * k * tt))
    a += 0.03 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a * (S / a.sum())
    return a


def generate_inits(n, rng, S, parent=None):
    inits = []
    if parent is not None and len(parent) >= 64:
        p = np.clip(parent.astype(np.float64), 0, 1000)
        if p.sum() > 0:
            inits.append(("parent", p * (S / p.sum())))
    
    for i in range(3):
        inits.append(("anti%d" % i, make_anti_palindromic_init(n, rng, S)))
    for i in range(2):
        inits.append(("broad%d" % i, make_broadband_init(n, rng, S)))
    for i in range(2):
        inits.append(("peri%d" % i, make_periodic_init(n, rng, S)))
    for i in range(2):
        inits.append(("taper%d" % i, make_tapered_init(n, rng, S)))
    for i in range(2):
        inits.append(("w%d" % i, make_w_shape_init(n, rng, S)))
    
    # Random
    rand_a = rng.random(n) + 0.05
    inits.append(("rand", rand_a * (S / rand_a.sum())))
    
    return inits


def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a):
    n = len(a)
    S = np.sqrt(2.0 * n)
    a_norm = a * (S / a.sum()) if a.sum() > 0 else a
    c = fft_convolve_self(a_norm)
    return 2.0 * n * c.max() / (S ** 2)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(10.0, budget_s - 20.0)

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    global_best_obj = np.inf
    global_best_a = None

    print("Parent available: %s" % (parent is not None))

    # Determine which lengths to try
    lengths_to_try = []
    if parent is not None:
        lengths_to_try.append(len(parent))
    for n_try in [8192, 16384]:
        if n_try not in lengths_to_try:
            lengths_to_try.append(n_try)

    # Allocate time to each length
    n_lengths = len(lengths_to_try)
    # Give more time to the primary length
    time_weights = [1.0] * n_lengths
    # Primary gets 60%, rest split
    if n_lengths > 1:
        time_weights[0] = 0.6
        remaining_w = 0.4 / (n_lengths - 1)
        for i in range(1, n_lengths):
            time_weights[i] = remaining_w

    for ni, n in enumerate(lengths_to_try):
        if time.monotonic() > deadline - 60:
            break

        S = np.sqrt(2.0 * n)
        alloc_time = (deadline - time.monotonic()) - 30.0
        phase1_time = alloc_time * time_weights[ni] * 0.65
        phase2_time = alloc_time * time_weights[ni] * 0.25
        polish_time = alloc_time * time_weights[ni] * 0.10

        print("=== n=%d, phase1=%.0fs, phase2=%.0fs, polish=%.0fs ===" % (n, phase1_time, phase2_time, polish_time))

        # Phase 1: Multi-start FISTA
        inits = generate_inits(n, rng, S, parent if len(parent) == n else None)
        n_inits = len(inits)
        per_init = phase1_time / n_inits

        local_best = np.inf
        local_best_a = None

        for name, arr in inits:
            if time.monotonic() > deadline - 60:
                break
            t0 = time.monotonic()

            # Run FISTA with LSE mode
            seg = per_init * 0.55
            a_f, s_f = run_fista(arr, n, S, seg, rng,
                                  beta_start=1.0, beta_end=600.0, mode="lse")
            a_f = safe_normalize(a_f, n)
            obj_f = compute_objective(a_f)
            if obj_f < local_best:
                local_best = obj_f
                local_best_a = a_f.copy()
                print("  FISTA-LSE [%s]: obj=%.6f (t=%.0fs)" % (name, obj_f, time.monotonic() - t0))

            # Run FISTA with top-k mode from the same start
            rem = per_init - (time.monotonic() - t0)
            if rem > 3:
                a_t, s_t = run_fista(arr, n, S, rem, rng,
                                      mode="topk", top_k_start=2, top_k_end=40)
                a_t = safe_normalize(a_t, n)
                obj_t = compute_objective(a_t)
                if obj_t < local_best:
                    local_best = obj_t
                    local_best_a = a_t.copy()
                    print("  FISTA-TK [%s]: obj=%.6f" % (name, obj_t))

        print("  Phase1 best n=%d: obj=%.6f" % (n, local_best))

        if local_best_a is None:
            continue

        # Check against global best
        if local_best < global_best_obj:
            global_best_obj = local_best
            global_best_a = local_best_a.copy()
            print("  ** New global best: %.6f (n=%d)" % (global_best_obj, n))

        # Phase 2: Continue optimizing the local best
        if time.monotonic() < deadline - 60 and phase2_time > 5:
            S2 = np.sqrt(2.0 * n)
            a_cont, _ = run_fista(local_best_a, n, S2, phase2_time, rng,
                                   beta_start=50.0, beta_end=1000.0, mode="lse")
            a_cont = safe_normalize(a_cont, n)
            obj_cont = compute_objective(a_cont)
            if obj_cont < local_best:
                local_best = obj_cont
                local_best_a = a_cont.copy()
                print("  Phase2 FISTA: obj=%.6f" % obj_cont)

            # Mirror descent
            rem2 = max(5.0, phase2_time - (time.monotonic() - (deadline - phase2_time - phase1_time - 30)))
            if rem2 > 5 and time.monotonic() < deadline - 60:
                a_md, _ = run_mirror_descent(local_best_a, n, S2, min(rem2, phase2_time * 0.4),
                                              rng, top_k=8, eta_init=0.2)
                a_md = safe_normalize(a_md, n)
                obj_md = compute_objective(a_md)
                if obj_md < local_best:
                    local_best = obj_md
                    local_best_a = a_md.copy()
                    print("  Phase2 Mirror: obj=%.6f" % obj_md)

            if local_best < global_best_obj:
                global_best_obj = local_best
                global_best_a = local_best_a.copy()

        # Polish
        if time.monotonic() < deadline - 30 and polish_time > 5:
            a_pol, _ = run_coordinate_polish(local_best_a, n, S, polish_time, rng, top_k=15)
            a_pol = safe_normalize(a_pol, n)
            obj_pol = compute_objective(a_pol)
            if obj_pol < local_best:
                local_best = obj_pol
                local_best_a = a_pol.copy()
                print("  Polish: obj=%.6f" % obj_pol)

            if local_best < global_best_obj:
                global_best_obj = local_best
                global_best_a = local_best_a.copy()

    # Final intensive polish on the global best
    if global_best_a is not None and time.monotonic() < deadline - 20:
        n = len(global_best_a)
        S = np.sqrt(2.0 * n)
        remaining = max(5.0, deadline - time.monotonic() - 10)
        print("=== Final polish (remaining=%.0fs) ===" % remaining)

        # High-beta FISTA
        seg = remaining * 0.4
        a_hb, _ = run_fista(global_best_a, n, S, seg, rng,
                             beta_start=100.0, beta_end=2000.0, mode="lse")
        a_hb = safe_normalize(a_hb, n)
        obj_hb = compute_objective(a_hb)
        if obj_hb < global_best_obj:
            global_best_obj = obj_hb
            global_best_a = a_hb.copy()
            print("  Final FISTA hi-beta: %.6f" % obj_hb)

        # Top-k focus
        rem = max(5.0, deadline - time.monotonic() - 10)
        seg2 = rem * 0.3
        a_tf, _ = run_fista(global_best_a, n, S, seg2, rng,
                             mode="topk", top_k_start=5, top_k_end=30)
        a_tf = safe_normalize(a_tf, n)
        obj_tf = compute_objective(a_tf)
        if obj_tf < global_best_obj:
            global_best_obj = obj_tf
            global_best_a = a_tf.copy()
            print("  Final top-k: %.6f" % obj_tf)

        # Mirror descent final
        rem2 = max(5.0, deadline - time.monotonic() - 10)
        if rem2 > 5:
            a_mf, _ = run_mirror_descent(global_best_a, n, S, rem2 * 0.5,
                                          rng, top_k=5, eta_init=0.1)
            a_mf = safe_normalize(a_mf, n)
            obj_mf = compute_objective(a_mf)
            if obj_mf < global_best_obj:
                global_best_obj = obj_mf
                global_best_a = a_mf.copy()
                print("  Final mirror: %.6f" % obj_mf)

        # Coordinate polish
        rem3 = max(3.0, deadline - time.monotonic() - 10)
        if rem3 > 3:
            a_cp, _ = run_coordinate_polish(global_best_a, n, S, rem3 * 0.8,
                                             rng, top_k=10)
            a_cp = safe_normalize(a_cp, n)
            obj_cp = compute_objective(a_cp)
            if obj_cp < global_best_obj:
                global_best_obj = obj_cp
                global_best_a = a_cp.copy()
                print("  Final coord polish: %.6f" % obj_cp)

    # Fallback
    if global_best_a is None:
        global_best_a = np.ones(1024, dtype=np.float64)
        print("WARNING: No valid solution found, using fallback")

    # Final safety checks
    global_best_a = np.clip(global_best_a, 0.0, 1000.0)
    if global_best_a.sum() < 0.01:
        global_best_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_best_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL: evaluate=%.6f, internal=%.6f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best_obj, len(global_best_a)))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using uniform fallback")
        final_list = [1.0] * 1024
        ver = evaluate_sequence(final_list)
        print("  Fallback evaluate: %.6f" % ver)

    return final_list
