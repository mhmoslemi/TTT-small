"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Strategy: Multi-initialization approach combining spectral shaping

and stochastic local search.

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def spectral_shape(a, target_flatness=0.5, n_iter=100):
    """
    Spectral shaping: try to flatten the power spectrum of a.
    This reduces the peak of the autocorrelation.
    """
    n = len(a)
    fsize = 1 << (2 * n - 1).bit_length()
    a_curr = a.copy().astype(np.float64)
    
    for it in range(n_iter):
        A = fft(a_curr, fsize)
        power = np.abs(A) ** 2
        # Target: interpolated between flat and current
        flat = np.sum(power) / fsize
        target_power = target_flatness * flat + (1 - target_flatness) * power
        # Ensure positivity and normalization
        target_power = np.maximum(target_power, 1e-10)
        target_power *= power.sum() / target_power.sum()
        
        # Take square root to get target amplitude
        target_amp = np.sqrt(target_power)
        
        # Keep original phase, use target amplitude
        phase = np.angle(A)
        A_new = target_amp * np.exp(1j * phase)
        
        a_new = np.real(ifft(A_new, fsize))[:n]
        
        # Project to non-negative with same sum
        old_sum = a_curr.sum()
        a_new = np.maximum(a_new, 0)
        if a_new.sum() > 1e-15:
            a_new = a_new * old_sum / a_new.sum()
        else:
            a_new = a_curr.copy()
        
        a_curr = a_new
    
    return a_curr

def subgradient_descent(a_init, max_iters, step_size, beta, rng):
    """Subgradient descent with softmax weights on convolution peaks."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    S = a.sum()
    S_fixed = S  # Keep sum fixed
    
    best_score = np.inf
    best_a = a.copy()
    
    for it in range(max_iters):
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        
        # Softmax weights
        c_centered = c - c.mean()
        weights = np.exp(beta * c_centered / c_centered.std())
        weights /= weights.sum()
        
        # Gradient: d(max conv)/d(a_i) approximated by weighted gradient
        # grad[i] = sum_k weights[k] * d(conv[k])/d(a[i])
        # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (for valid indices)
        # This is almost the cross-correlation of a with the weighted conv
        fsize = 1 << (len(weights) + n - 2).bit_length()
        W = fft(weights, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = step_size / (1.0 + 0.01 * it)
        a_new = project_simplex(a - step * grad / gnorm, S_fixed)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() <= cur_max * 1.001:  # Allow tiny increase
            a = a_new
        else:
            # Half step
            a_new = project_simplex(a - 0.5 * step * grad / gnorm, S_fixed)
            c_new = fft_convolve_self(a_new)
            if c_new.max() <= cur_max:
                a = a_new
            else:
                # Quarter step or random perturbation
                if rng.random() < 0.5:
                    a = project_simplex(a - 0.25 * step * grad / gnorm, S_fixed)
                else:
                    noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                    a = project_simplex(a + noise, S_fixed)
    
    return best_a, best_score

def stochastic_local_search(a_init, max_iters, rng):
    """Random perturbations with acceptance criterion."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    S = a.sum()
    
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    
    temperature = 0.01 * S / np.sqrt(n)
    
    for it in range(max_iters):
        # Random perturbation
        if rng.random() < 0.7:
            # Gaussian noise
            noise = rng.normal(0, temperature, n)
            a_new = project_simplex(a + noise, S)
        else:
            # Swap-based: move mass from one position to another
            i = rng.integers(n)
            j = rng.integers(n)
            delta = temperature * rng.uniform(-1, 1)
            a_new = a.copy()
            a_new[i] = max(0, a_new[i] - delta)
            a_new[j] = a_new[j] + delta
            # Re-normalize
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new * S / s
        
        c_new = fft_convolve_self(a_new)
        score_new = c_new.max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            temperature *= 0.999  # Cool down slightly
        elif rng.random() < np.exp(-(score_new - best_score) / (temperature * 10)):
            a = a_new
            temperature = max(temperature * 0.999, 0.001 * S / np.sqrt(n))
        else:
            temperature *= 1.001  # Slight reheat
    
    return best_a, best_score

def safe_normalize(a, n):

    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))

        return np.ones(n) * S / n
    return a * (S / s)
