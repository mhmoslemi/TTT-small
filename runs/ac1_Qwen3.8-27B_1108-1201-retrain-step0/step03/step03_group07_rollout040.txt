import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_ab(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def fft_crosscorr(w, a):
    """Compute sum_k w[k] * a[k-i] for i = 0..n-1."""
    n = len(a)
    a_rev = a[::-1]
    out_len = len(w) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full = np.real(ifft(W * R))
    return full[n - 1:2 * n - 1]


def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def softmax_temp(c, current_M):

    c_sorted = np.sort(c)[::-1]

    gap = current_M - c_sorted[1]

        # All peaks are equal, use moderate temperature
        t = 10.0 / max(current_M, 1e-10)

def lp_direction(a, M, n_constraints):
    """Solve reduced LP to find a good direction."""
    n = len(a)
    c = fft_convolve_self(a)
    top_idx = np.argsort(c, kind='stable')[::-1][:n_constraints]
    
    m = len(top_idx)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_idx):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[j])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(m, n))
    c_obj = -np.ones(n)
    b_ub = np.full(m, M)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': 30})
    
    if res.success and res.x is not None:
        return res.x
    return None


def coordinate_polish(a0, time_budget, max_rounds=50):

    a = a0.copy().astype(np.float64)

            break
        improved = False
        # Process in random order
        order = np.random.permutation(n)
        for i in order:

                break
            orig = a[i]
            if orig < 1e-12:

            for factor in factors:
                a[i] = orig * factor

                if s < 1e-10:
                    a[i] = orig
                    break
                a_norm = a * (S / s)

                    a = a_norm
                    c = c_new
                    improved = True
                    break

            # Renormalize after each coordinate change
            s = a.sum()
            if s > 1e-10:
                a = a * (S / s)

            break
    
    return best_a, best_obj


def single_peak_chase(a0, time_budget, lr=0.003):
    """Focus exclusively on reducing the single worst peak."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    v = np.zeros(n)
    momentum = 0.9
    
    while time.monotonic() - start < time_budget:
        k_max = np.argmax(c)
        
        # Gradient of c[k_max] w.r.t. a
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[k_max - m]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        v = momentum * v + (1 - momentum) * grad

        a = a * (S / a.sum())

        obj_new = obj_value(a, c)
        if obj_new < best_obj:

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
- Softmax mirror descent with Nesterov momentum (main optimizer)
- LP-based direction finding (periodic, for escaping local minima)
- Multi-phase strategy: explore -> exploit -> polish -> restart
- Adaptive temperature scheduling for the softmax smoothing

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_crosscorr(w, a):
    """Compute sum_k w[k]*a[k-i] for i=0..n-1 via FFT."""
    n = len(a)
    a_rev = a[::-1]
    out_len = len(w) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full = np.real(ifft(W * R))
    return full[n - 1:2 * n - 1]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def softmax_weights(c, t):
    """Compute softmax(t * c) with numerical stability."""
    c_shifted = c - c.max()
    exp_c = np.exp(t * c_shifted)
    s = exp_c.sum()
    if s < 1e-300:
        return np.full(len(c), 1.0 / len(c))
    return exp_c / s


def compute_softmax_grad(a, c, t):
    """Compute gradient of softmax-smoothed max of conv(a,a)."""
    w = softmax_weights(c, t)
    grad = 2.0 * fft_crosscorr(w, a)
    return grad


def mirror_descent_softmax(a, time_budget, eta0=0.3, t_init=None,
                           t_growth=1.003, t_max=None,
                           momentum=0.88, nesterov=True,
                           verbose=False, perturb_every=1000,
                           perturb_frac=0.03, lp_every=5000,
                           lp_n_constraints=30, lp_time_limit=20):
    """
    Mirror descent with softmax-smoothed gradient, Nesterov momentum,
    LP guidance, and adaptive step size.
    """
    a = a.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-10:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) / n
        S = 1.0

    c = fft_convolve_self(a)
    M = c.max()
    best_obj = obj_value(a, c)
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)
    if t_init is None:
        t_init = 5.0 / max(M, 1e-10)
    t = t_init
    if t_max is None:
        t_max = 500.0 / max(M, 1e-10)
    eta = eta0

    iter_no = 0
    start = time.monotonic()
    stagnant_count = 0

    while time.monotonic() - start < time_budget:
        # Compute gradient
        grad = compute_softmax_grad(a, c, t)

        # Nesterov momentum
        if nesterov and iter_no > 0:
            v = momentum * v + (1.0 - momentum) * grad
        else:
            v = momentum * v + (1.0 - momentum) * grad

        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm

        # Mirror descent step with backtracking
        accepted = False
        for bt in range(6):
            a_trial = a * np.exp(-eta * v)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)

            if obj_trial <= best_obj * (1.0 + 1e-8):
                a = a_trial
                c = c_trial
                M = c.max()
                accepted = True
                if obj_trial < best_obj:
                    best_obj = obj_trial
                    best_a = a.copy()
                    best_iter = iter_no
                    stagnant_count = 0
                    eta = min(eta * 1.015, eta0 * 4.0)
                break

            eta *= 0.6

        if not accepted:
            # Take a very small step
            a = a * np.exp(-eta * 0.05 * v)
            a = np.maximum(a, 1e-300)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            M = c.max()

        # Grow temperature
        t = min(t * t_growth, t_max)

        # LP direction (periodically)
        if lp_every > 0 and iter_no % lp_every == 0 and n <= 8192:
            lp_start = time.monotonic()
            try:
                b = lp_direction(a, c.max(), lp_n_constraints, lp_time_limit)
                if b is not None and b.sum() > 1e-10:
                    b = b * (S / b.sum())
                    for alpha in [0.5, 0.3, 0.7, 0.2, 1.0]:
                        a_try = (1.0 - alpha) * a + alpha * b
                        a_try = np.maximum(a_try, 0.0)
                        if a_try.sum() < 1e-10:
                            continue
                        a_try = a_try * (S / a_try.sum())
                        c_try = fft_convolve_self(a_try)
                        obj_try = obj_value(a_try, c_try)
                        if obj_try < obj_value(a, c) * (1.0 - 1e-10):
                            a = a_try
                            c = c_try
                            M = c.max()
                            if obj_try < best_obj:
                                best_obj = obj_try
                                best_a = a.copy()
                                best_iter = iter_no
                                stagnant_count = 0
                            break
            except Exception:
                pass
            lp_elapsed = time.monotonic() - lp_start
            if verbose:
                print(f"    [LP] iter={iter_no}, time={lp_elapsed:.1f}s, obj={obj_value(a,c):.6f}")

        # Stagnation detection and perturbation
        stagnant_count += 1
        if stagnant_count > perturb_every:
            noise_scale = perturb_frac * S / np.sqrt(n)
            noise = np.random.randn(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            M = c.max()
            v *= 0.2
            eta = eta0
            stagnant_count = 0
            best_iter = iter_no
            if verbose:
                print(f"    [perturb] iter={iter_no}, obj={obj_value(a,c):.6f}")

        iter_no += 1
        if verbose and iter_no % 3000 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter={iter_no}, obj={obj_value(a,c):.6f}, best={best_obj:.6f}, "
                  f"t={t:.1f}, eta={eta:.4f}, t_elapsed={elapsed:.1f}s")

    return best_a, best_obj


def lp_direction(a, M, n_constraints, time_limit=20):
    """Solve reduced LP: max sum(b) s.t. (a*b)[k] <= M for top peaks, b >= 0."""
    n = len(a)
    c = fft_convolve_self(a)
    top_idx = np.argsort(c, kind='stable')[::-1][:n_constraints]
    m = len(top_idx)

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(top_idx):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(a[j])

    if len(vals) == 0:
        return None

    A_ub = csr_matrix((vals, (rows, cols)), shape=(m, n))
    c_obj = -np.ones(n)
    b_ub = np.full(m, M)

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': time_limit})

    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def coordinate_polish(a0, time_budget, max_rounds=100):
    """Coordinate descent polish: try scaling each element to reduce peak."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()

    factors = [0.7, 0.85, 1.15, 1.3, 0.6, 1.4, 0.5, 1.6]

    for rnd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - start > time_budget * 0.95:
                break
            orig = a[i]
            if orig < 1e-12:
                continue
            best_local = None
            best_local_obj = obj_value(a, c)
            for factor in factors:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_norm = a * (S / s)
                c_new = fft_convolve_self(a_norm)
                obj_new = c_new.max()
                if obj_new < best_local_obj:
                    best_local_obj = obj_new
                    best_local = a_norm.copy()
            if best_local is not None:
                a = best_local
                c = fft_convolve_self(a)
                if obj_value(a, c) < best_obj:
                    best_obj = obj_value(a, c)
                    best_a = a.copy()
                    improved = True
            else:
                a[i] = orig
            s = a.sum()
            if s > 1e-10:
                a = a * (S / s)

        if not improved:
            break

    return best_a, best_obj


def single_peak_chase(a0, time_budget, lr=0.004):
    """Focus on reducing the single worst peak."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    v = np.zeros(n)
    momentum = 0.92

    while time.monotonic() - start < time_budget:
        k_max = np.argmax(c)
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[k_max - m]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        v = momentum * v + (1.0 - momentum) * grad
        a = a * np.exp(-lr * v)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())

        c = fft_convolve_self(a)
        obj_new = obj_value(a, c)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()

    return best_a, best_obj


def multi_peak_chase(a0, time_budget, lr=0.003, n_peaks=5):
    """Focus on reducing the top n_peaks simultaneously."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    v = np.zeros(n)
    momentum = 0.9

    while time.monotonic() - start < time_budget:
        top_idx = np.argsort(c, kind='stable')[::-1][:n_peaks]
        c_max = c[top_idx[0]]
        # Weight peaks by their excess over second peak
        if len(top_idx) > 1:
            gap = c_max - c[top_idx[1]]
            weights = np.exp(3.0 * (c[top_idx] - c[top_idx[1]]))
        else:
            weights = np.ones(1)
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for wi, k in zip(weights, top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1):
                grad[m] += 2.0 * wi * a[k - m]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        v = momentum * v + (1.0 - momentum) * grad
        a = a * np.exp(-lr * v)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())

        c = fft_convolve_self(a)
        obj_new = obj_value(a, c)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()

    return best_a, best_obj

        print(f"    -> {global_best:.6f}")

        # 3c: Coordinate descent
        t3c = p3_time * 0.25
        print(f"  3c: Coordinate descent ({t3c:.0f}s)")
        a_best, obj_best = coordinate_polish(global_a, t3c)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
        print(f"    -> {global_best:.6f}")

        # 3d: Fine softmax mirror descent
        t3d = p3_time - t3a - t3b - t3c
        if t3d > 5:
            print(f"  3d: Fine mirror descent ({t3d:.0f}s)")
            c_cur = fft_convolve_self(global_a)
            M_cur = c_cur.max()
            a_best, obj_best = mirror_descent_softmax(
                global_a, t3d,
                eta0=0.05, t_init=100.0 / max(M_cur, 1e-10),
                t_growth=1.015, t_max=1000.0 / max(M_cur, 1e-10),
                momentum=0.95, nesterov=False,
                perturb_every=300, perturb_frac=0.005,
                lp_every=0, verbose=False
            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
            print(f"    -> {global_best:.6f}")

    # ============ Phase 4: Random restarts (15%) ============
    remaining = deadline - time.monotonic()
    p4_time = max(5.0, remaining - 20.0)
    print(f"\n{'='*60}")
    print(f"Phase 4: Random restarts ({p4_time:.0f}s)")
    print(f"{'='*60}")

    if p4_time > 20:
        restart_time = 18.0
        n_restarts = int(p4_time / restart_time)
        print(f"  Max {n_restarts} restarts")

        for r in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break

            # Various perturbation strategies
            strategy = r % 4
            if strategy == 0:
                # Small Gaussian noise
                noise_scale = 0.01 * S / np.sqrt(n)
                a_pert = global_a + np.random.randn(n) * noise_scale
            elif strategy == 1:
                # Multiplicative noise
                a_pert = global_a * np.exp(0.01 * np.random.randn(n))
            elif strategy == 2:
                # Random subset perturbation
                a_pert = global_a.copy()
                n_pert = n // 10
                idx = np.random.choice(n, n_pert, replace=False)
                a_pert[idx] *= np.exp(0.05 * np.random.randn(n_pert))
            else:
                # LP-based restart
                try:
                    c_cur = fft_convolve_self(global_a)
                    b = lp_direction(global_a, c_cur.max(), 50, 10)
                    if b is not None and b.sum() > 1e-10:
                        a_pert = 0.5 * global_a + 0.5 * b * (S / b.sum())
                    else:
                        a_pert = global_a + 0.01 * S / np.sqrt(n) * np.random.randn(n)
                except Exception:
                    a_pert = global_a + 0.01 * S / np.sqrt(n) * np.random.randn(n)

            a_pert = np.maximum(a_pert, 0.0)
            if a_pert.sum() < 1e-10:
                a_pert = global_a.copy()
            a_pert = a_pert * (S / a_pert.sum())

            c_cur = fft_convolve_self(a_pert)
            M_cur = c_cur.max()
            a_best, obj_best = mirror_descent_softmax(
                a_pert, restart_time,
                eta0=0.15, t_init=20.0 / max(M_cur, 1e-10),
                t_growth=1.006, t_max=200.0 / max(M_cur, 1e-10),
                momentum=0.9, nesterov=True,
                perturb_every=200, perturb_frac=0.01,
                lp_every=500, lp_n_constraints=15, lp_time_limit=5,
                verbose=False
            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print(f"  Restart {r} (strat={strategy}): improved to {global_best:.6f}")

    # ============ Final: One more polish pass ============
    final_time = max(5.0, deadline - time.monotonic() - 15.0)
    if final_time > 10:
        print(f"\nFinal polish ({final_time:.0f}s)")
        c_cur = fft_convolve_self(global_a)
        M_cur = c_cur.max()
        a_best, obj_best = mirror_descent_softmax(
            global_a, final_time,
            eta0=0.08, t_init=50.0 / max(M_cur, 1e-10),
            t_growth=1.005, t_max=500.0 / max(M_cur, 1e-10),
            momentum=0.92, nesterov=True,
            perturb_every=200, perturb_frac=0.005,
            lp_every=500, lp_n_constraints=10, lp_time_limit=5,
            verbose=False
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
        print(f"  Final: {global_best:.6f}")

    # ============ Verification ============
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: local_obj={global_best:.6f}, verified={ver:.6f}")
    print(f"{'='*60}")

    if not np.isfinite(ver):
        print("  WARNING: verification failed, using fallback")
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"  Fallback obj={ver:.6f}")

    return final_a
