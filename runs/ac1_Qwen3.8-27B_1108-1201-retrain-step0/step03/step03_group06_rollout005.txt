```python
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def normalize_nonneg(a, S):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix C where C[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    K, J = np.meshgrid(k, j, indexing='ij')
    C = np.zeros((2 * n - 1, n))
    valid = (K - J >= 0) & (K - J < n)
    C[valid] = a[K[valid] - J[valid]]
    return C

def lp_solve_direction(a, M):
    """Solve the LP to find the optimal direction g."""
    n = len(a)
    g = cp.Variable(n, non_neg=True)
    C = build_conv_matrix(a)
    constraints = [C @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.GLOP)
    if g.value is None:
        return a.copy()
    return np.maximum(g.value, 0)

def lp_iteration(a, S, t_init=0.1, max_backtrack=20):
    """One LP-based iteration: find direction, line search, update."""
    n = len(a)
    c = fft_convolve_self(a)
    M = c.max()
    
    g = lp_solve_direction(a, M)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return a, M
    g = g * (S / g_sum)
    
    # Backtracking line search
    t = t_init
    for _ in range(max_backtrack):
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new = a_new * (S / a_new.sum())
        c_new = fft_convolve_self(a_new)
        if c_new.max() < M:
            return a_new, c_new.max()
        t *= 0.5
    
    return a, M
```

    c_new = fft_convolve_self(a_new)

            row_list.append(np.full(len(js), k))
            col_list.append(js)
            data_list.append(a[k - js])
    rows = np.concatenate(row_list)
    cols = np.concatenate(col_list)
    data = np.concatenate(data_list)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

def lp_direction(a, M):
    n = len(a)
    C = build_conv_matrix_sparse(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success or res.x is None:
        return a.copy()
    return np.maximum(res.x, 0.0)

def compute_weighted_subgrad(c, a, T):
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]

def subgradient_descent(a, S, time_budget, T_init, T_final, lr_init, rng):
    n = len(a)
    a = a.copy()
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    start = time.monotonic()
    T = T_init
    lr = lr_init
    step = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        progress = (time.monotonic() - start) / time_budget
        T = T_init * (T_final / T_init) ** (progress ** 0.7)
        lr = lr_init * (0.3 + 0.7 * (1.0 - progress))

        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            scale = 0.005 * max(0.1, 1.0 - progress)
            a = best_a + scale * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            no_improve = 0
            continue

        g = compute_weighted_subgrad(c, a, T)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        a_new = a - lr * g / gnorm
        a_new = normalize_nonneg(a_new, S)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_obj:
            a = a_new
        else:
            lr *= 0.9

        step += 1

    return best_a, best_obj

def coordinate_descent(a, S, time_budget, rng):
    n = len(a)
    a = a.copy()
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        i = rng.integers(0, n)
        orig = a[i]
        if orig < 1e-14:
            continue
        best_local = orig
        best_local_obj = best_obj
        for factor in [0.5, 0.7, 1.4, 1.6, 0.3, 2.0, 0.8, 1.2]:
            a[i] = orig * factor
            a = normalize_nonneg(a, S)
            c_check = fft_convolve_self(a)
            obj = c_check.max()
            if obj < best_local_obj - 1e-14:
                best_local_obj = obj
                best_local = orig * factor
        a[i] = best_local
        a = normalize_nonneg(a, S)
        c = fft_convolve_self(a)
        if c.max() < best_obj - 1e-14:
            best_obj = c.max()
            best_a = a.copy()

    return best_a, best_obj

def lp_iteration(a, S, t_init=0.3, max_backtrack=20):
    n = len(a)
    c = fft_convolve_self(a)
    M = c.max()
    g = lp_direction(a, M)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return a, M, False
    g = g * (S / g_sum)
    t = t_init
    for _ in range(max_backtrack):
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new * (S / s)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < M - 1e-13:
            return a_new, c_new.max(), True
        t *= 0.5
    return a, M, False

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 30.0)
    deadline = time.monotonic() + total_budget

    global_best = np.inf
    global_a = None

    # Load initial solution
    n = 4096
    S = np.sqrt(2.0 * float(n))

    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            n = len(given_raw)
            S = np.sqrt(2.0 * float(n))
            a = given_raw * (S / given_raw.sum())
            obj = evaluate_sequence(a.tolist())
            print("Initial: n=%d, obj=%.7f" % (n, obj))
            if np.isfinite(obj):
                global_best = obj
                global_a = a.copy()
    except Exception as e:
        print("Could not load initial: " + str(e))

    if global_a is None:
        a = np.full(n, S / n)
        global_best = 2.0
        global_a = a.copy()

    # ============ PHASE 1: LP-based optimization (55% budget) ============
    lp_budget = 0.55 * total_budget
    lp_deadline = time.monotonic() + lp_budget
    print("\nPhase 1: LP optimization (budget=%.0fs)" % lp_budget)

    a = global_a.copy()
    t_lp = 0.3
    no_improve = 0
    max_no_improve = 50
    lp_count = 0
    lp_failed = False

    while time.monotonic() < lp_deadline - 5:
        c = fft_convolve_self(a)
        M = c.max()

        if M < global_best - 1e-12:
            global_best = M
            global_a = a.copy()
            no_improve = 0
            print("  LP iter %d: obj=%.7f" % (lp_count, M))
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            # Perturb from best
            scale = 0.003 + 0.002 * (no_improve - max_no_improve) / max_no_improve
            a = global_a + scale * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            no_improve = 0
            lp_count += 1
            continue

        t0 = time.monotonic()
        a_new, M_new, improved = lp_iteration(a, S, t_init=t_lp)
        lp_time = time.monotonic() - t0
        lp_count += 1

        if not improved:
            # Try much smaller step
            c = fft_convolve_self(a)
            M = c.max()
            g = lp_direction(a, M)
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g * (S / g_sum)
                a_new = 0.99 * a + 0.01 * g
                a_new = normalize_nonneg(a_new, S)
                a = a_new
            else:
                a = new_a
        else:
            a = a_new

        if lp_time > 60:
            print("  LP too slow (%.1fs), switching to subgradient" % lp_time)
            lp_failed = True
            break

        # Decrease step size slowly
        if lp_count % 20 == 0:
            t_lp *= 0.9

    print("  LP phase done: %d iterations, best=%.7f" % (lp_count, global_best))

    # ============ PHASE 2: Subgradient fine-tuning (25% budget) ============
    sg_budget = 0.25 * total_budget
    sg_deadline = time.monotonic() + sg_budget
    print("\nPhase 2: Subgradient (budget=%.0fs, current=%.7f)" % (sg_budget, global_best))

    a = global_a.copy()

    # Multiple subgradient runs with different T schedules
    sg_configs = [
        (50, 500, 0.04),
        (200, 2000, 0.02),
        (10, 200, 0.06),
    ]

    per_config = sg_budget / len(sg_configs)
    for idx, (T_init, T_final, lr_init) in enumerate(sg_configs):
        if time.monotonic() > sg_deadline - 5:
            break
        a_best, obj_best = subgradient_descent(
            global_a, S, per_config, T_init, T_final, lr_init, rng
        )
        a_best = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_best.copy()
            print("  SG config %d: obj=%.7f" % (idx, obj))

    # ============ PHASE 3: CD polish (15% budget) ============
    cd_budget = deadline - time.monotonic()
    print("\nPhase 3: CD polish (budget=%.0fs)" % cd_budget)

    a = global_a.copy()
    if cd_budget > 5:
        a_best, obj_best = coordinate_descent(a, S, cd_budget * 0.7, rng)
        a_best = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_best.copy()
            print("  CD: obj=%.7f" % obj)

    # Final subgradient squeeze on best
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_best, obj_best = subgradient_descent(
            global_a, S, remaining, 200, 3000, 0.01, rng
        )
        a_best = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a_best.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_best.copy()
            print("  Final SG: obj=%.7f" % obj)

    # Final verification
    final_a = np.clip(global_a, 0.0, 1000.0)
    if final_a.sum() < 0.01:
        final_a = np.ones(256)
    ver = evaluate_sequence(final_a.tolist())
    print("\nFINAL: obj=%.7f (verified=%.7f)" % (global_best, ver))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a.tolist())
        print("Fallback: obj=%.7f" % ver)

    return final_a.tolist()
