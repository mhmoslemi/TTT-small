Minimize 2*n*max_conv/sum^2 using LP-based alternating optimization.

Key innovations:
1. LP-based direction finding: At each step, solve a linear program to find
   the optimal non-negative direction g that maximizes mass while keeping
   cross-correlation with current a bounded. This provides a strong "global"
   pull towards the optimum.
2. Reduced-constraint LP: Only enforce the top-k most critical convolution
   constraints, making the LP tractable for large n.
3. Adaptive line search: Find the best step size along the LP direction.
4. Gradient polishing: Use Nesterov-accelerated subgradient for local refinement.
5. Multi-start with diverse initializations including structured sequences.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]

    top_indices = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix A (top_k x n)
    # (a*g)(k) = sum_j a[j] * g[k-j] for valid j
    A_rows = []
    for k in top_indices:
        k = int(k)
        row = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            row[j_start:j_end+1] = a[j_start:j_end+1]
        A_rows.append(row)
    
    A = np.array(A_rows)
    b_ub = M * np.ones(top_k)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs',
                     options={'presolve': True})
    
    if result.status == 0:
        g = np.maximum(result.x, 0)
        return g
    else:
        # Fallback: return a itself
        return a.copy()


def lp_step(a, S, top_k, max_lp_time=5.0):
    """
    One LP-based step. Returns (new_a, score_improvement).
    """
    n = len(a)
    c = fft_convolve_self(a)
    M = c.max()
    
    # Solve LP
    t0 = time.monotonic()
    g = solve_lp_direction(a, M, top_k)
    lp_time = time.monotonic() - t0
    
    # Rescale g to match sum
    g_sum = g.sum()
    if g_sum < 1e-15:
        return a, 0.0
    g = g * (S / g_sum)
    
    # Line search: find best t in [0, 1]
    best_t = 0.0
    best_peak = M
    ts = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    for t in ts:
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        # Rescale to maintain sum
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            continue
        a_new = a_new * (S / a_new_sum)
        c_new = fft_convolve_self(a_new)
        peak_new = c_new.max()
        if peak_new < best_peak:
            best_peak = peak_new
            best_t = t
    
    if best_t == 0.0:
        return a, 0.0
    
    a_new = (1 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 0)
    a_new_sum = a_new.sum()
    a_new = a_new * (S / a_new_sum)
    
    return a_new, M - best_peak


def nesterov_subgradient(a_init, n, S, max_time, rng, top_k_init=20, top_k_max=256):
    """Nesterov-accelerated top-k subgradient method."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        # Subgradient: for the max of quad forms, subgradient is 2*(conv of y with delta_k)*a_k
        # where k is the argmax lag
        # More practically: use top-k weighted subgradient
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Gradient: 2 * sum_k w_k * (a convolved with e_k)
        # (a conv e_k)(j) = a[k-j] (if valid)
        # This is equivalent to: grad = 2 * sum_k w_k * shift(a, k)
        # Where shift(a, k)[j] = a[k-j] if 0<=k-j<n else 0
        # Which is the cross-correlation of a with a "comb" at position k
        
        # Efficient computation via FFT:
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]
        
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size with backtracking-like decay
        step = 2.0 * S / (n * (1.0 + 0.005 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 500 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score


def fista_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=300.0):
    """FISTA with log-sum-exp smooth approximation."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.002 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        c_max = c.max()
        # LSE approximation of max
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total
        
        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / (beta * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad, S)
        
        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 3000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.004 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score


def spectral_shaping(a, N, top_k, alpha, rng=None):
    """One spectral shaping step."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()
    
    m = np.arange(N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    
    weights = top_vals / total
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
    
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new_full, a.sum())
    return a_new


def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for non-negative sequence minimizing evaluate_sequence.
    Uses LP-based alternating optimization combined with gradient methods.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)
    
    global_best = np.inf
    global_a = None
    
    # Determine main length
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            main_len = len(given)
        else:
            main_len = 8192
    except Exception:
        main_len = 8192
    
    n = main_len
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()
    
    print("n=%d, S=%.4f, N(FFT)=%d" % (n, S, N))
    
    # ============ PHASE 1: LP-based exploration ============
    print("=== Phase 1: LP-based exploration ===")
    t_start = time.monotonic()
    phase1_budget = budget_s * 0.4
    
    # Initializations
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
    except Exception:
        pass
    
    tt = np.linspace(0, 1, n, endpoint=False)
    inits.append(("rand", rng.random(n) + 0.05))
    inits.append(("chirp", 0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n)))
    inits.append(("sin_multi", 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt)))
    inits.append(("bump", np.exp(-0.5 * ((tt - 0.5) * 5.0) ** 2) + 0.1))
    inits.append(("trig", 0.5 + 0.3 * np.cos(2 * np.pi * 2 * tt) * np.cos(2 * np.pi * 5 * tt) + 0.2 * np.sin(2 * np.pi * 11 * tt)))
    
    # Evaluate and sort inits by score
    init_scores = []
    for name, arr in inits:
        a_norm = safe_normalize(arr, n)
        c = fft_convolve_self(a_norm)
        score = c.max()
        init_scores.append((score, name, a_norm))
        print("  Init [%s]: peak=%.6f" % (name, score))
    
    init_scores.sort()
    
    # Run LP-based optimization on each init (time-limited)
    per_init = phase1_budget / len(inits_scores)
    
    for idx, (score, name, a_init) in enumerate(init_scores):
        if time.monotonic() > deadline - 60:
            break
        
        t0 = time.monotonic()
        a_curr = a_init.copy()
        
        # LP iterations
        n_lp_iters = 0
        while time.monotonic() - t0 < per_init * 0.6:
            n_lp_iters += 1
            a_next, improvement = lp_step(a_curr, S, top_k=min(50 + 50 * n_lp_iters, 200))
            if improvement > 1e-8:
                a_curr = a_next
                c_curr = fft_convolve_self(a_curr)
                if c_curr.max() < global_best:
                    global_best = c_curr.max()
                    global_a = a_curr.copy()
            else:
                break
        
        if global_a is not None and name == init_scores[0][1]:
            print("  [%s] After %d LP iters: %.6f" % (name, n_lp_iters, global_best))
        
        # Gradient polish
        rem = per_init - (time.monotonic() - t0)
        if rem > 3:
            a_g, _ = nesterov_subgradient(a_curr, n, S, min(rem, 60.0), rng,
                                           top_k_init=20, top_k_max=128)
            a_g = safe_normalize(a_g, n)
            c_g = fft_convolve_self(a_g)
            if c_g.max() < global_best:
                global_best = c_g.max()
                global_a = a_g.copy()
                print("  [%s] After grad: %.6f" % (name, global_best))
    
    print("Phase 1 done. Best: %.6f (time: %.1fs)" % (global_best, time.monotonic() - t_start))
    
    # ============ PHASE 2: Focused optimization ============
    print("=== Phase 2: Focused optimization ===")
    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = fft_convolve_self(global_a).max()
    
    t_phase2 = time.monotonic()
    phase2_budget = budget_s * 0.35
    
    a_curr = global_a.copy()

    last_best = global_best

    while time.monotonic() < t_phase2 + phase2_budget:
        outer_it += 1
        rem = (t_phase2 + phase2_budget) - time.monotonic()
        if rem < 5:
            break
        
        # LP step
        lp_time = min(rem * 0.3, 30.0)
        if lp_time > 2:

            if imp > 1e-10:
                a_curr = a_lp
                c_lp = fft_convolve_self(a_curr)
                if c_lp.max() < global_best:
                    global_best = c_lp.max()
