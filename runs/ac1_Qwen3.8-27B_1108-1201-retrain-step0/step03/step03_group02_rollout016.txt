"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LP-based peak reduction.

The algorithm iteratively solves a linear program to find the best direction
to reduce the peak of the self-convolution, using the "state of the art"
approach: maximize sum(g) subject to conv(f,g) <= M and g >= 0.
Combined with diverse asymmetric initializations and gradient-based polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(f, g):
    """Full cross-convolution of f and g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    F = fft(f, fsize)
    G = fft(g, fsize)
    c = np.real(ifft(F * G))
    return c[:size]


def build_conv_matrix(f, n):
    """Build sparse matrix A such that A @ g = convolve(f, g) for any g of length n."""
    # A[k, j] = f[j] for k in [j, j+n-1] (clamped to [0, 2n-2])
    num_entries = np.minimum(n, 2 * n - 1 - np.arange(n))
    total = int(num_entries.sum())
    
    data = np.repeat(f, num_entries)
    row_offsets = np.concatenate([np.arange(ne) for ne in num_entries]) if n > 0 else np.array([])
    col_starts = np.repeat(np.arange(n), num_entries)
    rows = (col_starts + row_offsets).astype(np.int32)
    indptr = np.concatenate([[0], np.cumsum(num_entries)]).astype(np.int32)
    
    return csr_matrix((data, rows, indptr), shape=(2*n-1, n))


def solve_lp_direction(f, M, n):
    """
    Solve: maximize sum(g)
    subject to: conv(f, g)[k] <= M for all k
                0 <= g[j] <= 1000
    Returns g or None if failed.
    """
    A = build_conv_matrix(f, n)
    c = -np.ones(n)  # minimize -sum(g)
    b = np.full(2*n - 1, M)
    bounds = [(0, 1000)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception as e:
        print(f"  LP error: {e}")
    
    return None

def fft_grad_soft(a, c, beta):
    """Gradient of soft-max approximation of max(c)."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

        it += 1

        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()

        top_indices = np.argsort(c)[::-1][:k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        w_full = np.zeros(2 * n - 1)
        for i, idx in enumerate(top_indices):
            w_full[int(idx)] = weights[i]
        
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.005 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        # Simple backtracking
        c_new = fft_convolve_self(x_new)
        if c_new.max() > cur * 1.01:
            x_new = project_simplex(0.5 * y + 0.5 * x_new, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)

    return best_a, best_score
