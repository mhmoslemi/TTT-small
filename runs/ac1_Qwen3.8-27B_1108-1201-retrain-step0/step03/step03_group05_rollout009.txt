"""
Minimize 2*n*max_conv/sum^2 exploiting convexity of the normalized problem.

After normalizing sum(a)=S=sqrt(2n), minimizing max(a*a) is CONVEX (max of convex quadratics).
Strategy:
1. Nesterov-accelerated projected subgradient with soft active-set weighting (Phase 1)
2. L-BFGS on LogSumExp surrogate with aggressive continuation (Phase 2)
3. Final Nesterov polish with sharpened weights + coordinate descent (Phase 3)
4. Multi-start with perturbation to escape local minima in the smoothed landscape
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def nesterov_subgrad(a_init, S, max_time, rng, weight_beta=20.0,
                     weight_beta_max=500.0, step_scale=1.0):
    """
    Nesterov-accelerated projected subgradient for convex problem:
    minimize max_k (x*x)_k subject to x >= 0, sum(x) = S.
    
    Uses soft active-set weighting for smoother gradients.
    Step size based on Lipschitz constant L=2 of the individual quadratics.
    """
    n = len(a_init)
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    wb = weight_beta

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        c_max = c.max()

        if c_max < best_score:
            best_score = c_max
            best_a = y.copy()

        # Soft weighting around the peak
        margin = c_max - c
        # Clip margin to avoid underflow
        margin_clipped = np.minimum(margin, 50.0 / max(wb, 1e-6))
        w = np.exp(-wb * margin_clipped)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum

        # Gradient: 2 * sum_k w[k] * P_k(a)
        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Step size: for L-smooth convex, optimal Nesterov step = 1/L.
        # L = 2 (operator norm of Hessian 2*P_k). 
        # On the simplex with sum=S, effective step in a-space is S/L * something.
        # Use: step = step_scale / (1.0 + 0.0005 * it) for gradual decay
        step = step_scale / (1.0 + 0.0005 * it)
        
        # Projected step
        x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-normalize to stay on simplex
        y_sum = y.sum()
        if y_sum > 0:
            y = y * (S / y_sum)
        a = x_new
        t = t_new

        # Gradually sharpen the weight
        if it % 2000 == 0:
            wb = min(weight_beta_max, wb * 1.3)

        # Periodic restart of momentum
        if it % 10000 == 0:
            t = 1.0
            y = a.copy()

        # Check time
        if it % 10000 == 0:
            elapsed = time.monotonic() - start
            print("    Nesterov it=%d, score=%.8f, wb=%.1f, t=%.1fs" %
                  (it, best_score, wb, elapsed))

    return best_a, best_score


def softmax_obj_grad(z, beta, S):
    """
    Compute LogSumExp-smoothed objective and gradient w.r.t. z (unconstrained).
    a = S * softmax(z), objective = (1/beta) * log(sum exp(beta*c_k))
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    bc = beta * (c - c_max)
    exp_bc = np.exp(bc)
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=250):
    """
    L-BFGS-B with continuation on beta (smooth-to-sharp) on softmax parameters.
    """
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 40, 80, 150, 300, 500, 1000, 2000, 4000, 8000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() - start > max_time - 5:
            break

        # At high beta, try perturbation restarts
        n_attempts = 1
        if beta >= 500:
            n_attempts = 2
        if beta >= 2000:
            n_attempts = 3

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 5:
                    break
            if attempt > 0:
                # Perturb to escape local minima in z-space
                noise_scale = 0.01 / (1 + 0.5 * bi)
                z = z + rng.normal(0, noise_scale, n)
                z = z - z.mean()

            t0 = time.monotonic()
            try:
                result = scipy_minimize(
                    lambda zv: softmax_obj_grad(zv, beta, S)[0],
                    z,
                    jac=lambda zv: softmax_obj_grad(zv, beta, S)[1],
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_beta,
                        'maxfun': maxiter_per_beta * 3,
                        'ftol': 1e-15,
                        'gtol': 1e-10,
                        'maxls': 50,
                        'iprint': -1
                    }
                )
            except Exception as e:
                print("      L-BFGS error: %s" % str(e))
                break

            elapsed_beta = time.monotonic() - t0
            z = result.x
            z = z - z.mean()

            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

        print("    beta=%5d: score=%.8f (%d att, %.1fs)" %
              (beta, best_score, n_attempts, time.monotonic() - t0 if 't0' in dir() else 0))

    return best_a, best_score


def coordinate_descent_polish(a_init, S, max_time, rng, n_outer=50, n_inner=200):
    """
    Coordinate descent on the active set: repeatedly identify the peak lag,
    then do a line search on reducing that specific lag's value.
    """
    n = len(a_init)
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for outer in range(n_outer):
        if time.monotonic() - start > max_time:
            break

        c = fft_convolve_self(a)
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        # Find top few lags
        top_k = min(10, 2 * n - 1)
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Find the index with maximum gradient (most contributing to peak)
        # Reduce that element and increase others proportionally
        idx_max = np.argmax(grad)
        
        # Line search: reduce a[idx_max] by delta, redistribute
        for inner in range(n_inner):
            if time.monotonic() - start > max_time:
                break
            # Try reducing a[idx_max]
            delta = a[idx_max] * 0.1
            if delta < 1e-15:
                break
            
            a_trial = a.copy()
            a_trial[idx_max] = max(0, a_trial[idx_max] - delta)
            # Redistribute to minimize the peak
            deficit = delta
            a_rest = a_trial.copy()
            a_rest[idx_max] = 0
            rest_sum = a_rest.sum()
            if rest_sum > 0:
                a_trial += deficit * (a_rest / rest_sum)
            a_trial[idx_max] = max(0, a_trial[idx_max])
            
            # Normalize
            a_trial_sum = a_trial.sum()
            if a_trial_sum > 0:
                a_trial = a_trial * (S / a_trial_sum)
            
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < c_max * (1 - 1e-6):
                a = a_trial
                c = c_trial
                c_max = c.max()
                if c_max < best_score:
                    best_score = c_max
                    best_a = a.copy()
            else:
                # Try small increase instead
                delta = a[idx_max] * 0.05
                a_trial = a.copy()
                a_trial[idx_max] += delta
                # Reduce others
                deficit = delta
                a_rest = a_trial.copy()
                a_rest[idx_max] = 0
                rest_sum = a_rest.sum()
                if rest_sum > 0:
                    a_trial -= deficit * (a_rest / rest_sum)
                a_trial = np.maximum(a_trial, 0)
                a_trial_sum = a_trial.sum()
                if a_trial_sum > 0:
                    a_trial = a_trial * (S / a_trial_sum)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < c_max * (1 - 1e-6):
                    a = a_trial
                    c = c_trial
                    c_max = c.max()
                    if c_max < best_score:
                        best_score = c_max
                        best_a = a.copy()
                else:
                    break  # If neither direction helps, move on

        # Update gradient direction
        if outer % 10 == 0:
            print("    CD outer=%d: score=%.8f (%.1fs)" %
                  (outer, best_score, time.monotonic() - start))

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
