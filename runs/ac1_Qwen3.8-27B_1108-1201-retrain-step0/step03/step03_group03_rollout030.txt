**Vectorized coordinate descent:**

**New idea: Use the structure of the optimum**

At the optimum (KKT conditions), for the non-smooth problem:
minimize max_k c[k]
subject to sum(a) = S, a >= 0

The KKT conditions say:
- There exists lambda (for sum constraint) and mu_k >= 0 (for c[k] <= t) such that:
  sum_{k: mu_k > 0} mu_k * grad(c[k]) + lambda * 1 = 0 (for coordinates with a[i] > 0)
  sum_{k: mu_k > 0} mu_k * grad(c[k]) + lambda * 1 >= 0 (for coordinates with a[i] = 0)

This means that at the optimum, the gradient of the "active" lags (weighted by the KKT multipliers) is constant (= -lambda) on the support of a.

This gives us a way to check if we're at the optimum: the weighted sum of gradients of the active lags should be constant on the support.

**Practical algorithm based on KKT:**

1. Start with a good initialization
2. Compute c and find the top K lags
3. Solve for the KKT multipliers mu_k (non-negative) such that the weighted gradient is as flat as possible on the support
4. Update a in the direction that makes the weighted gradient flatter
5. Repeat

This is essentially what the "active set" method does, but with an explicit KKT analysis.

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy:
1. Anti-palindromic + periodic initializations
2. FISTA with log-sum-exp smoothing and adaptive parameters
3. L-BFGS polish on the smoothed objective
4. Coordinate descent peak shaving
5. KKT-based active set refinement
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_value_and_grad(a, c, beta):
    """Log-sum-exp value and gradient with respect to a."""
    n = len(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -50, 50)
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        return c_max + np.log(len(c)) / beta, np.zeros(n)
    log_lse = c_max + np.log(total) / beta
    w = exp_scaled / total
    # Gradient of LSE w.r.t. a: sum_k w_k * grad(c[k])
    # grad(c[k])[i] = 2*a[k-i] if 0 <= k-i < n else 0
    # This is 2 * (w * a_flipped) truncated
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return log_lse, grad


def objective_and_grad(a, beta):
    """Objective (smoothed) and gradient for scipy optimizer."""
    c = fft_convolve_self(a)
    val, grad = lse_value_and_grad(a, c, beta)
    return val, grad


def topk_smoothed_obj(a, c, k, temperature):
    """Top-k smoothed objective: sum of top-k c values with softmax weighting."""
    k = min(k, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    tv_max = top_vals.max()
    w = np.exp((top_vals - tv_max) / temperature)
    w = w / w.sum()
    # Objective: weighted sum of top-k
    obj = np.dot(w, top_vals)
    return obj, top_idx, w


def safe_normalize(a, S):
    """Clip to [0, 1000] and normalize to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.05
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_periodic_init(n, rng, S, num_freqs=3):
    """Periodic initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 20)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.5
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a * (S / a.sum())


def make_bump_init(n, rng, S):
    """Gaussian bump initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    center = rng.random()
    width = rng.random() * 0.2 + 0.1
    a = np.exp(-0.5 * ((tt - center) / width) ** 2) + 0.01
    return a * (S / a.sum())


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=200.0):
    """FISTA with adaptive beta."""
    a = safe_normalize(a_init, S)
    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stall_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stall_count = 0
        elif it % 100 == 0:
            stall_count += 1

        val, grad = lse_value_and_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step = 2.0 / (beta * 4.0 * np.linalg.norm(a, 2) ** 2 / n)
        step = min(step, S / (n * 10))
        
        x_new = project_simplex(y - step * grad, S)
        
        # Simple backtracking
        for _ in range(3):
            c_x = fft_convolve_self(x_new)
            if c_x.max() <= cur + 1e-10:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum ** 2))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Periodic perturbation
        if stall_count > 50 and rng.random() < 0.2:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_momentum = 1.0
            stall_count = 0

    return best_local, best_score


def run_lbfgs(a_init, n, S, max_time, rng, beta=50.0):
    """L-BFGS on the smoothed objective with simplex constraint via reparameterization."""
    a = safe_normalize(a_init, S)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    # Use log-sum-exp parameterization: a_i = S * exp(x_i) / sum(exp(x_j))
    # This automatically satisfies a >= 0 and sum(a) = S
    x0 = np.log(np.maximum(a, 1e-15))
    x0 = x0 - x0.mean()  # Center for numerical stability

    def soft_objective(x):
        if time.monotonic() - start > max_time:
            return np.inf, np.zeros_like(x)
        exp_x = np.exp(x - x.max())
        a_param = S * exp_x / exp_x.sum()
        c = fft_convolve_self(a_param)
        val, grad_c = lse_value_and_grad(a_param, c, beta)
        # Chain rule: d(a_param)/d(x_j) = a_param[j] * (delta_ij - a_param[i]/S)
        # d(val)/d(x_j) = sum_i grad_c[i] * d(a_param[i])/d(x_j)
        # = sum_i grad_c[i] * a_param[i] * (delta_ij - a_param[i]... 
        # Actually: a_param[i] = S * exp(x_i) / sum(exp(x))
        # d(a_param[i])/d(x_j) = a_param[i] * (delta_ij - exp(x_j)/sum(exp(x)))
        # = a_param[i] * (delta_ij - a_param[j] / S)
        # d(val)/d(x_j) = sum_i grad_c[i] * a_param[i] * (delta_ij - a_param[j]/S)
        # = grad_c[j] * a_param[j] - (a_param[j]/S) * sum_i grad_c[i] * a_param[i]
        ga = grad_c * a_param
        total = ga.sum()
        grad_x = ga - (a_param / S) * total
        return val, grad_x

    # Run L-BFGS with time limit
    result = scipy_minimize(soft_objective, x0, method='L-BFGS-B',
                            options={'maxiter': 5000, 'maxfun': 2000,
                                    'max_time': max(1, max_time - 1),
                                    'factr': 1e7})
    
    exp_x = np.exp(result.x - result.x.max())
    a_result = S * exp_x / exp_x.sum()
    a_result = safe_normalize(a_result, S)
    c_result = fft_convolve_self(a_result)
    score = c_result.max()
    
    if score < best_score:
        best_score = score
        best_local = a_result.copy()

    return best_local, best_score


def run_peak_shaving(a_init, n, S, max_time, rng):
    """Iteratively reduce the peak convolution value."""
    a = safe_normalize(a_init, S)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Find top 5 lags
        top_k = min(5, len(c))
        top_indices = np.argsort(c)[::-1][:top_k]
        
        # Compute gradient direction (sum of gradients of top lags)
        w_full = np.zeros(2 * n - 1)
        weights = np.ones(top_k) / top_k
        w_full[top_indices] = weights
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        # Adaptive step
        step = 0.1 * S / (n * np.linalg.norm(grad) + 1e-15)
        step = min(step, 0.5 * S / n)

        x_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(x_new)
        
        if c_new.max() < cur:
            a = x_new
        else:
            # Try smaller step
            step *= 0.5
            x_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur:
                a = x_new
            else:
                # Add small noise to escape
                noise = 0.001 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)

    return best_local, best_score


def run_kkt_refine(a_init, n, S, max_time, rng):
    """KKT-based refinement: equalize top lags."""
    a = safe_normalize(a_init, S)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Identify active set: lags within 5% of max
        threshold = cur * 0.95
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.argsort(c)[::-1][:10]
        elif len(active) > 100:
            active = np.argsort(c)[::-1][:100]

        # We want to minimize the variance of c[k] over active k
        # Gradient of variance: d/d(a) [sum_{k in A} (c[k] - mean)^2]
        # = 2 * sum_{k in A} (c[k] - mean) * grad(c[k])
        active_c = c[active]
        mean_c = active_c.mean()
        weights = (active_c - mean_c)  # Positive for above-mean, negative for below-mean
        
        # Only use lags that are above the mean (we want to reduce peaks)
        positive_mask = weights > 0
        if not np.any(positive_mask):
            # All below mean, just reduce the max
            active_subset = active[np.argmax(c[active])]
        else:
            active_subset = active[positive_mask]
        
        w_full = np.zeros(2 * n - 1)
        w_full[active_subset] = 1.0 / len(active_subset)
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        step = 0.05 * S / (n * (np.linalg.norm(grad) + 1e-15))
        step = min(step, 0.2 * S / n)

        x_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(x_new)
        if c_new.max() < cur:
            a = x_new
        else:
            step *= 0.5
            x_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur:
                a = x_new

    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, top_k=10, eta=0.1):
    """Mirror descent (multiplicative weights) with top-k subgradient."""
    a = safe_normalize(a_init, S)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        w_full = np.zeros(2 * n - 1)
        w_full[top_indices] = 1.0 / top_k
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        eta_t = eta / (1.0 + 0.01 * it)
        log_update = -eta_t * grad / gmax
        log_update = np.clip(log_update, -3.0, 3.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-20)
        a = a_new * (S / a_new.sum())

    return best_local, best_score

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy:
1. Diverse initializations (anti-palindromic, periodic, parent)
2. FISTA with log-sum-exp smoothing and adaptive beta schedule
3. L-BFGS polish using softmax parameterization (handles constraints exactly)
4. Peak shaving with adaptive step
5. KKT-based active set leveling
6. Mirror descent (multiplicative weights) as alternative view
7. Final high-beta FISTA convergence push
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_value_and_grad(a, c, beta):
    """Log-sum-exp of c with parameter beta, value and gradient w.r.t. a."""
    n = len(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -60, 60)
    exp_scaled = np.exp(scaled)
    total = exp_scaled.sum()
    if total < 1e-300:
        return c_max + np.log(len(c)) / beta, np.zeros(n)
    log_lse = c_max + np.log(total) / beta
    w = exp_scaled / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return log_lse, grad


def topk_gradient(a, c, top_k, temperature=1.0):
    """Gradient using top-k lags with softmax weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices].copy()
    tv_max = top_vals.max()
    weights = np.exp(top_vals / max(temperature, 1e-10) - tv_max / max(temperature, 1e-10))
    weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def safe_normalize(a, S):
    """Clip to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.05
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-15:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_periodic_init(n, rng, S, num_freqs=3):
    """Periodic with multiple sinusoidal components."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 25)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.5
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a * (S / a.sum())


def make_w_shape_init(n, rng, S):
    """W/M shaped initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    k = rng.integers(2, 6)
    a = np.abs(np.sin(2 * np.pi * k * tt)) + 0.05
    a = a * (S / a.sum())
    return a


def make_bump_init(n, rng, S):
    """Gaussian bump."""
    tt = np.arange(n, dtype=np.float64) / n
    center = rng.random()
    width = rng.random() * 0.2 + 0.08
    a = np.exp(-0.5 * ((tt - center) / width) ** 2) + 0.01
    return a * (S / a.sum())


def run_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=2000.0):
    """FISTA with adaptive beta, backtracking line search, and restarts."""
    a = safe_normalize(a_init, S)
    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stall = 0
    L = 4.0 * beta_init  # Initial Lipschitz estimate

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stall = 0
        elif it % 80 == 0:
            stall += 1

        val, grad = lse_value_and_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)

        f_y = val
        accepted = False
        for bt in range(6):
            c_x = fft_convolve_self(x_new)
            f_x = c_x.max()
            if f_x <= f_y - 1e-5 * step * gnorm * gnorm or bt == 5:
                accepted = True
                break
            step *= 0.4
            x_new = project_simplex(y - step * grad, S)

        if not accepted:
            x_new = project_simplex(y - step * grad, S)

        # Update Lipschitz
        c_x = fft_convolve_self(x_new)
        if c_x.max() > f_y + 1e-3:
            L *= 1.5
        elif it % 50 == 0:
            L = max(L * 0.97, 4.0 * beta)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Periodic perturbation to escape local minima
        if stall > 40 and rng.random() < 0.25:
            noise_scale = 0.008 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_momentum = 1.0
            stall = 0
            L = 4.0 * beta

    return best_local, best_score


def run_lbfgs_polish(a_init, n, S, max_time, rng, beta=80.0):
    """L-BFGS polish using softmax parameterization (exact constraint handling)."""
    a = safe_normalize(a_init, S)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    timeout = [False]

    x0 = np.log(np.maximum(a, 1e-15))
    x0 = x0 - x0.mean()

    def soft_obj(x):
        if time.monotonic() - start > max_time * 0.95:
            timeout[0] = True
            return np.inf, np.zeros_like(x)
        exp_x = np.exp(x - x.max())
        a_p = S * exp_x / exp_x.sum()
        c = fft_convolve_self(a_p)
        val, grad_c = lse_value_and_grad(a_p, c, beta)
        # Chain rule through softmax
        ga = grad_c * a_p
        total = ga.sum()
        grad_x = ga - (a_p / S) * total
        return val, grad_x

    max_iter = int(max(50, max_time * 8))
    result = scipy_minimize(soft_obj, x0, method='L-BFGS-B',
                            options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                    'maxls': 20, 'ftol': 1e-12, 'gtol': 1e-8})

    if result.fun < np.inf:
        exp_x = np.exp(result.x - result.x.max())
        a_result = S * exp_x / exp_x.sum()
        a_result = safe_normalize(a_result, S)
        c_result = fft_convolve_self(a_result)
        score = c_result.max()
        if score < best_score:
            best_score = score
            best_local = a_result.copy()

    return best_local, best_score


def run_peak_shaving(a_init, n, S, max_time, rng):
    """Iteratively reduce the peak by computing descent direction on top lags."""
    a = safe_normalize(a_init, S)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 0.1 * S / n
    top_k = 5

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(50, 5 + it // 200)

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        grad = topk_gradient(a, c, top_k, temperature=max(1.0, 10.0 - 0.01 * it))
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step
        step = step * 0.999
        step_new = 2.0 * step
        x_new = project_simplex(a - step_new * grad / (gnorm + 1e-15), S)
        c_new = fft_convolve_self(x_new)

        if c_new.max() < cur * (1 - 1e-8):
            a = x_new
            step = step_new
        else:
            # Halve step
            step_new *= 0.5
            x_new = project_simplex(a - step_new * grad / (gnorm + 1e-15), S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur:
                a = x_new
                step = step_new
            elif it % 200 == 0 and rng.random() < 0.3:
                # Escape local minimum
                noise = 0.003 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise, n), S)
                step = 0.05 * S / n

    return best_local, best_score


def run_kkt_level(a_init, n, S, max_time, rng):
    """KKT-based refinement: find multipliers that flatten the active set."""
    a = safe_normalize(a_init, S)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Active set: lags within 2% of max
        threshold = cur * 0.98
        active_mask = c >= threshold
        n_active = active_mask.sum()

        if n_active < 3:
            active_indices = np.argsort(c)[::-1][:10]
        elif n_active > 200:
            active_indices = np.argsort(c)[::-1][:200]
        else:
            active_indices = np.where(active_mask)[0]

        # Compute weighted gradient where weights emphasize the highest lags
        active_c = c[active_indices]
        ac_max = active_c.max()
        ac_min = active_c.min()
        if ac_max - ac_min > 1e-15:
            # Inverse weighting: give more weight to higher lags
            w = (active_c - ac_min + 1e-10)
        else:
            w = np.ones(len(active_indices))
        w = w / w.sum()

        w_full = np.zeros(2 * n - 1)
        w_full[active_indices] = w
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        step = 0.03 * S / (n * (np.linalg.norm(grad) / (S / n) + 0.1))
        step = min(step, 0.3 * S / n)

        x_new = project_simplex(a - step * grad, S)
        c_new = fft_convolve_self(x_new)
        if c_new.max() < cur:
            a = x_new
        else:
            step *= 0.5
            x_new = project_simplex(a - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() < cur:
                a = x_new

    return best_local, best_score


def run_mirror(a_init, n, S, max_time, rng, top_k=15, eta_init=0.15):
    """Mirror descent (multiplicative weights update)."""
    a = safe_normalize(a_init, S)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        eta = eta_init / (1.0 + 0.005 * it)
        grad = topk_gradient(a, c, top_k, temperature=1.0)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        log_update = -eta * grad / gmax
        log_update = np.clip(log_update, -4.0, 4.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-20)
        a = a_new * (S / a_new.sum())

    return best_local, best_score


def run_coordinate_descent(a_init, n, S, max_time, rng):
    """Coordinate descent: update one coordinate at a time to minimize max c."""
    a = safe_normalize(a_init, S)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    order = rng.permutation(n)

    while time.monotonic() - start < max_time:
        it += 1
        if it % 10 == 1:
            c = fft_convolve_self(a)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_local = a.copy()

        # Update one coordinate
        i = order[it % n]
        i = int(i)

        # a[i] affects c[k] for k in [i, i+n-1]
        k_range = np.arange(i, min(i + n, 2 * n - 1))
        # For each k, c[k] = base_k + 2*a[i]*a[k-i] (or a[i]^2 if k=2i)
        # We want to minimize max over k of (base_k + 2*a[i]*a[k-i])
        # subject to a[i] >= 0 and adjusting for sum constraint

        # Compute the "base" convolution (without a[i]'s contribution)
        # This is expensive to do exactly, so use an approximation:
        # Current c[k] includes a[i]'s contribution. If we set a[i] = 0,
        # c[k] would decrease by 2*a[i]*a[k-i] (for valid k-i).
        # So base_k = c[k] - 2*a[i]*a[k-i] (for valid k-i)

        # For simplicity, just try a few candidate values of a[i]
        # and pick the best one
        current_ia = a[i]
        candidates = [0.0, current_ia * 0.5, current_ia * 1.5,
                      current_ia * 0.8, current_ia * 1.2]

        best_val = cur
        best_ia = current_ia
        for cand in candidates:
            cand = max(0.0, min(1000.0, cand))
            if abs(cand - current_ia) < 1e-15:
                continue
            a[i] = cand
            # Adjust sum: redistribute the difference
            diff = current_ia - cand
            if abs(diff) > 1e-15:
                # Add diff to a random other coordinate (or spread)
                j = (i + n // 2) % n
                a[j] = max(0.0, a[j] + diff)
            c_check = fft_convolve_self(a)
            if c_check.max() < best_val:
                best_val = c_check.max()
                best_ia = cand
        a[i] = best_ia
        # Re-normalize
        a = safe_normalize(a, S)

    # Final
    c = fft_convolve_self(a)
    if c.max() < best_score:
        best_score = c.max()
        best_local = a.copy()
    return best_local, best_score


def propose_candidate(seed=42, budget_s=2350, **kwargs):
    """Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2."""
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + max(10.0, total_budget - 30.0)

    global_best = np.inf
    global_a = None

    # Get parent sequence if available
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    primary_n = 8192
    if parent is not None and len(parent) >= 64:
        primary_n = len(parent)

    S = np.sqrt(2.0 * primary_n)
    print("n=%d, parent available: %s" % (primary_n, parent is not None))

    # === Phase 1: Multi-start FISTA (40% of time) ===
    inits = []
    if parent is not None:
        inits.append(("parent", safe_normalize(parent, S)))
    for i in range(5):
        inits.append(("anti_%d" % i, make_anti_palindromic_init(primary_n, rng, S)))
    for i in range(2):
        inits.append(("peri_%d" % i, make_periodic_init(primary_n, rng, S)))
    inits.append(("wshape", make_w_shape_init(primary_n, rng, S)))
    inits.append(("bump", make_bump_init(primary_n, rng, S)))

    phase1_time = max(30.0, (deadline - time.monotonic()) * 0.40)
    per_init = phase1_time / len(inits)

    print("=== Phase 1: FISTA x%d, %.1f sec each ===" % (len(inits), per_init))

    best_p1 = None
    best_p1_score = np.inf

    for name, arr in inits:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        alloc = min(per_init, rem - 30)
        if alloc < 5:
            break
        t0 = time.monotonic()
        a_out, s_out = run_fista(arr, primary_n, S, alloc, rng,
                                  beta_init=10.0, beta_max=1500.0)
        a_out = safe_normalize(a_out, S)
        c_out = fft_convolve_self(a_out)
        score = c_out.max()
        if score < best_p1_score:
            best_p1_score = score
            best_p1 = a_out.copy()
        if score < global_best:
            global_best = score
            global_a = a_out.copy()
        print("  P1 [%s]: %.6f (%.1fs)" % (name, score, time.monotonic() - t0))

    print("After P1: best=%.6f" % global_best)

    # === Phase 2: L-BFGS polish (10% of remaining) ===
    if global_a is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        seg = min(120.0, max(15.0, (deadline - time.monotonic()) * 0.10))
        print("=== Phase 2: L-BFGS (%.0fs) ===" % seg)
        a_lb, _ = run_lbfgs_polish(global_a, primary_n, S, seg, rng, beta=80.0)
        a_lb = safe_normalize(a_lb, S)
        c_lb = fft_convolve_self(a_lb)
        if c_lb.max() < global_best:
            global_best = c_lb.max()
            global_a = a_lb.copy()
            print("  L-BFGS: %.6f" % global_best)
        else:
            print("  L-BFGS: no improvement (%.6f)" % c_lb.max())

    # === Phase 3: Peak shaving (15%) ===
    if global_a is not None and time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        seg = min(200.0, max(20.0, (deadline - time.monotonic()) * 0.15))
        print("=== Phase 3: Peak shaving (%.0fs) ===" % seg)
        a_ps, _ = run_peak_shaving(global_a, primary_n, S, seg, rng)
        a_ps = safe_normalize(a_ps, S)
        c_ps = fft_convolve_self(a_ps)
        if c_ps.max() < global_best:
            global_best = c_ps.max()
            global_a = a_ps.copy()
            print("  PeakShave: %.6f" % global_best)
        else:
            print("  PeakShave: no improvement (%.6f)" % c_ps.max())

    # === Phase 4: KKT leveling (10%) ===
    if global_a is not None and time.monotonic() < deadline - 30:
        t0 = time.monotonic()
        seg = min(150.0, max(15.0, (deadline - time.monotonic()) * 0.10))
        print("=== Phase 4: KKT level (%.0fs) ===" % seg)
        a_kkt, _ = run_kkt_level(global_a, primary_n, S, seg, rng)
        a_kkt = safe_normalize(a_kkt, S)
        c_kkt = fft_convolve_self(a_kkt)
        if c_kkt.max() < global_best:
            global_best = c_kkt.max()
            global_a = a_kkt.copy()
            print("  KKT: %.6f" % global_best)
        else:
            print("  KKT: no improvement (%.6f)" % c_kkt.max())

    # === Phase 5: Mirror descent (8%) ===
    if global_a is not None and time.monotonic() < deadline - 25:
        t0 = time.monotonic()
        seg = min(100.0, max(10.0, (deadline - time.monotonic()) * 0.08))
        print("=== Phase 5: Mirror (%.0fs) ===" % seg)
        a_md, _ = run_mirror(global_a, primary_n, S, seg, rng, top_k=20, eta_init=0.10)
        a_md = safe_normalize(a_md, S)
        c_md = fft_convolve_self(a_md)
        if c_md.max() < global_best:
            global_best = c_md.max()
            global_a = a_md.copy()
            print("  Mirror: %.6f" % global_best)
        else:
            print("  Mirror: no improvement (%.6f)" % c_md.max())

    # === Phase 6: Final high-beta FISTA (remaining time) ===
    if global_a is not None and time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        seg = max(10.0, (deadline - time.monotonic()) * 0.85)
        print("=== Phase 6: Final FISTA hi-beta (%.0fs) ===" % seg)
        a_fin, _ = run_fista(global_a, primary_n, S, seg, rng,
                              beta_init=200.0, beta_max=5000.0)
        a_fin = safe_normalize(a_fin, S)
        c_fin = fft_convolve_self(a_fin)
        if c_fin.max() < global_best:
            global_best = c_fin.max()
            global_a = a_fin.copy()
            print("  Final FISTA: %.6f" % global_best)
        else:
            print("  Final FISTA: no improvement (%.6f)" % c_fin.max())

    # === Phase 7: One more L-BFGS if time remains ===
    if global_a is not None and time.monotonic() < deadline - 15:
        seg = max(10.0, (deadline - time.monotonic()) * 0.5)
        print("=== Phase 7: Final L-BFGS (%.0fs) ===" % seg)
        a_fl, _ = run_lbfgs_polish(global_a, primary_n, S, seg, rng, beta=200.0)
        a_fl = safe_normalize(a_fl, S)
        c_fl = fft_convolve_self(a_fl)
        if c_fl.max() < global_best:
            global_best = c_fl.max()
            global_a = a_fl.copy()
            print("  Final L-BFGS: %.6f" % global_best)

    # === Verification and output ===
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * 1.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Verification: %.6f (internal: %.6f), n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        print("WARNING: evaluate_sequence rejected. Using fallback.")
        global_a = np.ones(1024, dtype=np.float64) * 1.0
        final_list = [1.0] * 1024
        ver = evaluate_sequence(final_list)
        print("Fallback: %.6f" % ver)

    print("=== FINAL ANSWER: %.6f ===" % ver)
    return final_list
