import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_power_weighted_grad(c, a, p):
    """Compute gradient of sum_k (c[k]/c_max)^p w.r.t. a.
    dc[k]/da[j] = 2*sum_i [a[i]*d(k-i, j)] which simplifies to
    dc[k]/da[j] = a[k-j] + a[j-k] (with bounds, a[m]=0 if m not in [0,n-1])
    Gradient: g[j] = p * (c[k*]/c_max)^(p-1) / c_max * (a[k*-j] + a[j-k*])
    More generally with weighting w[k]: g[j] = 2 * sum_k w[k] * a[k-j]
    where the sum is over valid k.
    """
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)

    # Power weighting
    w = (c / c_max) ** p
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = w / w_sum * len(c)  # normalize to have mean 1

    # g[j] = 2 * sum_k w[k] * a[k-j] (valid when 0<=k-j<n)
    # This is 2 * cross-correlation of w and a
    # Using FFT: corr[j] = sum_k w[k] * a[k-j]
    # = sum_m w[j+m] * a[m] (reindexing: k=j+m)
    # This is the cross-correlation (w * a)_j in a specific sense
    fsize = 1 << (3 * n).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # corr[j] = sum_k w[k] * a[k-j] = ifft(W * conj(A))[j]
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def run_adam_power_weighted(a_init, n, S, time_budget, p_init=4.0, p_final=256.0,
                            lr_init=0.05, beta1=0.9, beta2=0.999,
                            cyclical=False, verbose=False):
    """Adam optimizer in log-space with adaptive power weighting."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        # Adaptive power: based on peak-to-mean ratio
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()

        c_mean = c.mean()
        c_max = c.max()
        ratio = c_max / max(c_mean, 1e-30)

        # Adaptive p: higher when peaks are more pronounced
        if cyclical:
            # Cyclical schedule: p goes up and down to escape local minima
            base_p = p_init * (p_final / p_init) ** (progress ** 0.5)
            p = base_p * (1.0 + 0.5 * np.sin(2 * np.pi * elapsed / 60.0))
            p = max(2.0, min(p, 1000.0))
        else:
            p = p_init * (p_final / p_init) ** (progress ** 0.7)

        # If ratio is very high, use higher p
        if ratio > 3.0:
            p = max(p, 100.0)
        elif ratio < 1.3:
            p = min(p, 8.0)

        # Convert theta to a
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
            theta = np.full(n, np.log(1.0 / n))
        else:
            a = S * exp_theta / total

        # Recompute c at current a
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()

        # Compute weighted gradient
        g_a = compute_power_weighted_grad(c, a, p)

        # Chain rule for log-parameters
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot

        # Adam update
        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break

        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Learning rate with warmup and decay
        warmup = min(1.0, t / 100.0)
        lr = lr_init * warmup * (0.3 + 0.7 * (1.0 - progress ** 0.5))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if verbose and t % 500 == 0:
            print("    AdamPW step %d: obj=%.7f, p=%.1f, ratio=%.3f" % (t, best_obj, p, ratio))

    return best_a, best_obj

            theta = np.full(n, np.log(1.0 / n))
        else:
            a = S * exp_theta / total

        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()

        # Exponential weighting
        c_max = c.max()
        if c_max < 1e-30:
            break
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum * len(c)

        # Gradient
        fsize = 1 << (3 * n).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g_a = 2.0 * corr[:n]

        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot

        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break

        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        warmup = min(1.0, t / 100.0)
        lr = lr_init * warmup * (0.2 + 0.8 * (1.0 - progress ** 0.4))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if verbose and t % 500 == 0:
            print("    ExpT step %d: obj=%.7f, T=%.0f" % (t, best_obj, T))

    return best_a, best_obj


def coordinate_descent_bisect(a, n, S, time_budget, max_sweeps=100, verbose=False):
    """Coordinate descent with binary search for optimal coordinate value."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(777)

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            original = a[i]
            if original < 1e-14:
                continue

            # Binary search for optimal value of a[i]
            # The objective as a function of a[i] (with others fixed, after normalization)
            # is not necessarily unimodal, so we try several values
            best_val = original
            best_local_obj = best_obj

            # Try a range of values: from 0 to 3*original
            candidates = [0.0, original * 0.1, original * 0.3, original * 0.5,
                          original * 0.7, original, original * 1.3, original * 1.5,
                          original * 2.0, original * 3.0, original * 5.0]

            for candidate in candidates:
                a[i] = candidate
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-13:
                    best_local_obj = obj_check
                    best_val = candidate

            a[i] = best_val
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)

            if c.max() < best_obj - 1e-13:
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        if not improved and sweep >= 5:
            if verbose:
                print("    CD bisect converged at sweep %d" % sweep)
            break

    return best_a, best_obj


def greedy_support_reduction(a, n, S, time_budget, max_zeros=200, verbose=False):
    """Greedily try to zero out coordinates that improve the objective."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    start = time.monotonic()

    n_zeroed = 0
    max_iterations = max_zeros * 3

    for iteration in range(max_iterations):
        if time.monotonic() - start > time_budget:
            break
        if n_zeroed >= max_zeros:
            break

        # Find the smallest non-zero coordinate
        nonzero_mask = a > 1e-12
        if not nonzero_mask.any():
            break

        sorted_idx = np.argsort(a)
        found = False
        for idx in sorted_idx:
            if a[idx] < 1e-12:
                continue
            saved = a[idx]
            a[idx] = 0.0
            a = normalize_nonneg(a, S)
            c_check = fft_convolve_self(a)
            obj_check = c_check.max()
            if obj_check < best_obj - 1e-13:
                best_obj = obj_check
                best_a = a.copy()
                n_zeroed += 1
                found = True
                break
            else:
                a[idx] = saved
                a = normalize_nonneg(a, S)

        if not found:
            break

    if verbose:
        print("    Support reduction: zeroed %d coordinates, obj=%.7f" % (n_zeroed, best_obj))

    return best_a, best_obj


def multi_scale_optimize(target_n, time_budget, seed=42):
    """Multi-scale optimization: start small, grow to target_n."""
    rng = np.random.default_rng(seed)
    scales = [64, 128, 256, 512, 1024, 2048, 4096, 8192, 16384]
    # Filter to relevant scales
    scales = [s for s in scales if s <= target_n]
    if target_n not in scales:
        scales.append(target_n)
    scales.sort()

    # Allocate time: more time at larger scales
    weights = [s ** 0.5 for s in scales]
    total_w = sum(weights)
    times = [time_budget * w / total_w for w in weights]
    times[-1] = time_budget - sum(times[:-1])  # Give remaining to largest
    times = [max(1.0, t) for t in times]

    a_current = None
    best_a = None
    best_obj = np.inf

    for scale_idx, n in enumerate(scales):
        S = np.sqrt(2.0 * float(n))
        t_budget = times[scale_idx]

        if a_current is None or len(a_current) != n:
            # Initialize
            if a_current is None:
                # Start with a structured initialization
                i = np.arange(n)
                # Use a "bump" function that's not constant
                a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * i / n + rng.random() * 2) + \
                    0.15 * np.cos(2 * np.pi * 7 * i / n + rng.random() * 2)
                a = np.maximum(a, 0.01)
                a = a * (S / a.sum())
            else:
                # Upscale by interpolation
                old_n = len(a_current)
                # Simple linear upsampling
                old_idx = np.linspace(0, old_n - 1, n)
                a = np.interp(np.arange(n), np.arange(old_n), a_current)
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum > 1e-15:
                    a = a * (S / a_sum)
                else:
                    a = np.full(n, S / n)

        # Optimize at this scale
        if n <= 256:
            # Small: use exponential T (fast convergence)
            a_best, obj = run_exponential_t(a, n, S, t_budget, T_init=5, T_final=200, lr_init=0.06)
        elif n <= 2048:
            # Medium: use power weighting
            a_best, obj = run_adam_power_weighted(a, n, S, t_budget, p_init=4, p_final=128, lr_init=0.04)
        else:
            # Large: use adaptive power weighting with cyclical schedule
            a_best, obj = run_adam_power_weighted(a, n, S, t_budget, p_init=8, p_final=512, lr_init=0.03, cyclical=True)

        a_best = normalize_nonneg(a_best, S)

        if obj < best_obj:
            best_obj = obj
            best_a = a_best.copy()

        a_current = a_best.copy()
        if scale_idx < len(scales) - 1:
            print("  Multi-scale n=%d: obj=%.7f (budget=%.1fs)" % (n, obj, t_budget))

    return best_a, best_obj


def direct_subgradient_restart(a_init, n, S, time_budget, T_vals=None, lr_init=0.05):
    """Direct subgradient with restarts at different T values."""
    if T_vals is None:
        T_vals = [20, 100, 500, 1000, 3000]

    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    best_a = a.copy()
    best_obj = fft_convolve_self(a).max()

    n_T = len(T_vals)
    per_T_budget = time_budget / n_T

    for T in T_vals:
        t0 = time.monotonic()
        beta = 0.85
        a_prev = a.copy()

        while time.monotonic() - t0 < per_T_budget:
            c = fft_convolve_self(a)
            cur_obj = c.max()
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()

            c_max = c.max()
            if c_max < 1e-30:
                break

            scaled = T * c / c_max
            scaled = scaled - scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                break
            w = w / w_sum

            fsize = 1 << (3 * n).bit_length()
            W = fft(w, fsize)
            A = fft(a, fsize)
            corr = np.real(ifft(W * np.conj(A)))
            g = 2.0 * corr[:n]

            gnorm = np.linalg.norm(g)
            if gnorm < 1e-15:
                break

            # Nesterov
            y = a + (beta - 1.0) / (1.0 + beta) * (a - a_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy > 1e-15:
                y = y * (S / sy)

            # Line search
            lr = lr_init
            a_trial = a
            best_trial_obj = cur_obj
            for _ in range(8):
                a_candidate = a - lr * g / gnorm
                a_candidate = normalize_nonneg(a_candidate, S)
                c_cand = fft_convolve_self(a_candidate)
                obj_cand = c_cand.max()
                if obj_cand < best_trial_obj - 1e-12:
                    best_trial_obj = obj_cand
                    a_trial = a_candidate.copy()
                    break
                lr *= 0.5

            a_prev = a.copy()
            a = a_trial

    return best_a, best_obj


def try_snap_to_rationals(a, n, S, time_budget, verbose=False):
    """Try snapping values to nearby rational multiples of the mean."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    start = time.monotonic()

    mean_val = S / n
    # Try snapping to multiples of mean_val / k for various k
    for k in [2, 3, 4, 5, 8, 10]:
        if time.monotonic() - start > time_budget:
            break
        step = mean_val / k
        a_snapped = np.round(a / step) * step
        a_snapped = np.maximum(a_snapped, 0.0)
        a_snapped = normalize_nonneg(a_snapped, S)
        c_s = fft_convolve_self(a_snapped)
        obj_s = c_s.max()
        if obj_s < best_obj - 1e-13:
            best_obj = obj_s
            best_a = a_snapped.copy()
            if verbose:
                print("    Snap k=%d: obj=%.7f" % (k, obj_s))

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 30.0)
    deadline = time.monotonic() + total_budget

    global_best = np.inf
    global_a = None

    # Load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: " + str(e))

    # ============ PHASE 1: Multi-scale optimization (40% time) ============
    ms_budget = 0.40 * total_budget
    print("\n=== Phase 1: Multi-scale optimization (budget=%.0fs) ===" % ms_budget)

    # Try target length 4096 (same as given) and 8192
    for target_n in [4096, 8192]:
        if time.monotonic() > deadline - 60:
            break
        per_target = min(ms_budget * 0.6, deadline - time.monotonic() - 30)
        if per_target < 30:
            break

        a_ms, obj_ms = multi_scale_optimize(target_n, per_target, seed=seed)
        a_ms = normalize_nonneg(a_ms, np.sqrt(2.0 * target_n))
        ver = evaluate_sequence(a_ms.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_ms.copy()
            print("  Multi-scale n=%d: obj=%.7f" % (target_n, ver))

        # Also try starting from given if available
        if given_a is not None and target_n == 4096:
            S4096 = np.sqrt(2.0 * 4096)
            a_given_zero = np.zeros(4096)
            a_given_zero[:len(given_a)] = given_a
            a_given_zero = normalize_nonneg(a_given_zero, S4096)
            remaining = per_target * 0.4
            if remaining > 20:
                a_g, obj_g = run_adam_power_weighted(a_given_zero, 4096, S4096, remaining,
                                                      p_init=8, p_final=256, lr_init=0.03)
                a_g = normalize_nonneg(a_g, S4096)
                ver_g = evaluate_sequence(a_g.tolist())
                if np.isfinite(ver_g) and ver_g < global_best:
                    global_best = ver_g
                    global_a = a_g.copy()
                    print("  From given n=4096: obj=%.7f" % ver_g)

    # ============ PHASE 2: Deep exploitation (35% time) ============
    exploit_budget = 0.35 * total_budget
    exploit_deadline = time.monotonic() + exploit_budget
    print("\n=== Phase 2: Exploitation (budget=%.0fs) ===" % exploit_budget)

    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a = global_a.copy()
        a = normalize_nonneg(a, S)

        # Round 1: High T exponential weighting
        r1_time = 0.30 * exploit_budget
        if r1_time > 15:
            a_best, _ = run_exponential_t(a, n, S, r1_time, T_init=100, T_final=5000, lr_init=0.02)
            a = normalize_nonneg(a_best, S)
            obj = evaluate_sequence(a.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a.copy()
                print("  R1 ExpT high: obj=%.7f" % obj)

        # Round 2: Power weighted with cyclical schedule
        r2_time = 0.25 * exploit_budget
        if r2_time > 15:
            a_best, _ = run_adam_power_weighted(a, n, S, r2_time,
                                                 p_init=32, p_final=1024, lr_init=0.02, cyclical=True)
            a = normalize_nonneg(a_best, S)
            obj = evaluate_sequence(a.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a.copy()
                print("  R2 PowerW cyclical: obj=%.7f" % obj)

        # Round 3: Direct subgradient with restarts
        r3_time = 0.25 * exploit_budget
        if r3_time > 15:
            a_best, _ = direct_subgradient_restart(a, n, S, r3_time,
                                                    T_vals=[50, 200, 1000, 5000], lr_init=0.04)
            a = normalize_nonneg(a_best, S)
            obj = evaluate_sequence(a.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a.copy()
                print("  R3 Subgrad restart: obj=%.7f" % obj)

        # Round 4: Low T for overall flattening
        r4_time = exploit_deadline - time.monotonic()
        if r4_time > 10:
            a_best, _ = run_adam_power_weighted(a, n, S, r4_time,
                                                 p_init=2, p_final=32, lr_init=0.05)
            a = normalize_nonneg(a_best, S)
            obj = evaluate_sequence(a.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a.copy()
                print("  R4 PowerW low-T: obj=%.7f" % obj)

    # ============ PHASE 3: Polish (25% time) ============
    polish_budget = deadline - time.monotonic()
    print("\n=== Phase 3: Polish (budget=%.0fs) ===" % polish_budget)

    if global_a is not None and polish_budget > 10:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a = global_a.copy()
        a = normalize_nonneg(a, S)

        # Coordinate descent
        cd_budget = 0.35 * polish_budget
        if cd_budget > 10:
            a_best, _ = coordinate_descent_bisect(a, n, S, cd_budget, max_sweeps=80)
            a = normalize_nonneg(a_best, S)
            obj = evaluate_sequence(a.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a.copy()
                print("  CD bisect: obj=%.7f" % obj)

        # Support reduction
        sr_budget = 0.25 * polish_budget
        if sr_budget > 5:
            a_best, _ = greedy_support_reduction(a, n, S, sr_budget, max_zeros=100)
            a = normalize_nonneg(a_best, S)
            obj = evaluate_sequence(a.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a.copy()
                print("  Support reduction: obj=%.7f" % obj)

        # Snap to rationals
        snap_budget = 0.15 * polish_budget
        if snap_budget > 3:
            a_best, _ = try_snap_to_rationals(a, n, S, snap_budget)
            a = normalize_nonneg(a_best, S)
            obj = evaluate_sequence(a.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a.copy()
                print("  Snap: obj=%.7f" % obj)

        # Final CD
        remaining = deadline - time.monotonic() - 5
        if remaining > 5:
            a_best, _ = coordinate_descent_bisect(a, n, S, remaining, max_sweeps=50)
            a = normalize_nonneg(a_best, S)
            obj = evaluate_sequence(a.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a.copy()
                print("  Final CD: obj=%.7f" % obj)

    # ============ PHASE 4: Random restarts if time remains ============
    restart_budget = deadline - time.monotonic() - 5
    if global_a is not None and restart_budget > 20:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        print("\n=== Phase 4: Restarts (budget=%.0fs) ===" % restart_budget)
        n_rounds = max(1, int(restart_budget / 30))
        per_round = restart_budget / (n_rounds + 1)

        for r in range(n_rounds):
            if time.monotonic() > deadline - 5:
                break
            for noise_scale in [0.005, 0.02, 0.05]:
                a_noisy = global_a + noise_scale * rng.standard_normal(n)
                a_noisy = normalize_nonneg(a_noisy, S)
                t0 = time.monotonic()
                a_best, _ = run_adam_power_weighted(a_noisy, n, S, per_round * 0.5,
                                                     p_init=16, p_final=512, lr_init=0.03)
                a_best = normalize_nonneg(a_best, S)
                obj = evaluate_sequence(a_best.tolist())
                if np.isfinite(obj) and obj < global_best:
                    global_best = obj
                    global_a = a_best.copy()
                    print("  Restart r=%d noise=%.3f: obj=%.7f" % (r, noise_scale, obj))
                    break

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f (verified=%s)" % (
        global_best, "%.7f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)

    return final_a
