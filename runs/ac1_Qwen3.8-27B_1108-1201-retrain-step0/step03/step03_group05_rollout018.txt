"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: L-BFGS continuation with very high beta (log-sum-exp),
coordinate descent polishing, active-set subgradient refinement,
and structured perturbation restarts.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT, length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_conv(a, weights):
    """grad[m] = 2*sum_k weights[k]*a[k-m] via FFT."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """Smoothed objective and gradient w.r.t. softmax params z."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_grad_conv(a, w)
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=300,
                           ftol=1e-14, gtol=1e-10):
    """L-BFGS with continuation on beta."""
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        n_attempts = 1
        if beta >= 5000 and max_time > 30:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 5:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.005, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            t0 = time.monotonic()
            try:
                result = scipy_minimize(
                    fun, z, jac=jac, method='L-BFGS-B',
                    options={'maxiter': maxiter_per_beta,
                             'maxfun': maxiter_per_beta * 3,
                             'ftol': ftol, 'gtol': gtol,
                             'maxls': 50, 'iprint': -1}
                )
            except Exception:
                break

            z = result.x
            z = z - z.mean()
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if (time.monotonic() - start) % 10 < 2:
            pass  # keep quiet

    return best_a, best_score


def coordinate_descent(a, n, S, max_time, rng, n_rounds=5):
    """Coordinate descent: optimize one element at a time."""
    a = np.array(a, dtype=np.float64)
    s = a.sum()
    a = a * S / s
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    improved = True

    while improved and time.monotonic() - start < max_time:
        improved = False
        for _ in range(n_rounds):
            if time.monotonic() - start > max_time - 2:
                break
            perm = np.arange(n)
            rng.shuffle(perm)
            for i in perm:
                if time.monotonic() - start > max_time - 2:
                    break
                # Try a range of values for a[i]
                old_val = a[i]
                a_rest_sum = S - old_val
                if old_val < 1e-15:
                    # Try turning it on
                    candidates = [0.0, S / (n + 1), S / (2 * n), 2 * S / (2 * n)]
                else:
                    spread = old_val * 0.3
                    candidates = [old_val + rng.uniform(-spread, spread) for _ in range(8)]
                    candidates.append(0.0)
                    candidates.append(old_val)

                for cand in candidates:
                    if cand < 0 or cand > a_rest_sum:
                        continue
                    a[i] = cand
                    c = fft_convolve_self(a)
                    sc = c.max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                    else:
                        a[i] = old_val

            a = best_a.copy()

    return best_a, best_score


def nesterov_subgrad(a_init, n, S, max_time, rng, n_topk_max=256):
    """Nesterov accelerated subgradient with top-k weights."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(n_topk_max, 5 + it // 100)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Use all positions within 95% of max for smoother subgradient
        threshold = 0.95 * cur
        active_mask = c >= threshold
        if active_mask.sum() == 0:
            active_mask = c >= 0.9 * cur
        n_active = active_mask.sum()
        n_use = min(top_k, max(5, n_active // 2))
        top_indices = np.argsort(c)[::-1][:n_use]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for idx, k in enumerate(top_indices):
            w_full[int(k)] = weights[idx]

        grad = compute_grad_conv(a, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = 2.0 * S / (n * np.sqrt(1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def local_polish(a_init, n, S, max_time, rng):
    """Final polishing: high-beta L-BFGS + coordinate descent."""
    start = time.monotonic()
    best_a = a_init.copy()
    best_score = fft_convolve_self(a_init).max()

    # High-beta L-BFGS
    t1 = min(max_time * 0.6, time.monotonic() - start + max_time * 0.6)
    a1, s1 = run_lbfgs_continuation(
        best_a, n, S, min(max_time * 0.6, max_time - 5), rng,
        beta_schedule=[100, 500, 2000, 5000, 10000, 20000, 50000, 100000],
        maxiter_per_beta=500, ftol=1e-16, gtol=1e-12
    )
    a1 = safe_normalize(a1, n)
    c1 = fft_convolve_self(a1)
    if c1.max() < best_score:
        best_score = c1.max()
        best_a = a1.copy()

    remaining = max(1.0, max_time - (time.monotonic() - start))
    if remaining > 5:
        a2, s2 = coordinate_descent(best_a, n, S, remaining, rng, n_rounds=3)
        a2 = safe_normalize(a2, n)
        c2 = fft_convolve_self(a2)
        if c2.max() < best_score:
            best_score = c2.max()
            best_a = a2.copy()

    return best_a, best_score
