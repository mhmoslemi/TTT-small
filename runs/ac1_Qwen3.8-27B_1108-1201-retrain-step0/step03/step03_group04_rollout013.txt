"""
Minimize 2*n*max_conv/sum^2 using adaptive proximal gradient with backtracking,
log-sum-exp annealing, and multi-restart strategy.
Key improvements: Armijo backtracking line search, geometric beta annealing,
structured perturbations, and multi-n exploration.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient of sum(weights * convolution) w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Compute normalized top-k weights (uniform over top-k)."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    weights = np.zeros(len(c))
    weights[top_indices] = 1.0 / top_k
    return weights


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def run_logsumexp_fista(a_init, n, S, max_time, rng, beta_init=20.0, beta_final=2000.0, decay=0.002):
    """
    FISTA with log-sum-exp smooth approximation and geometric beta annealing.
    Uses Armijo backtracking for step size.
    """
    a = safe_normalize(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 2.0 * S / (n * beta_init)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_final, beta_init * np.exp(decay * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
        
        w = softmax_weights(c, beta)
        grad = compute_gradient_fft(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Armijo backtracking
        alpha = step
        for _ in range(10):
            x_new = project_simplex(y - alpha * grad, S)
            c_new = fft_convolve_self(x_new)
            # Check sufficient decrease in the log-sum-exp objective
            lse_before = np.log(np.sum(np.exp(beta * (c - c.max())))) / beta
            lse_after = np.log(np.sum(np.exp(beta * (c_new - c_new.max())))) / beta
            if lse_after <= lse_before - 0.5 * alpha * np.dot(grad, x_new - y):
                break
            alpha *= 0.5
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Adaptive step
        if it % 500 == 0:
            # Re-estimate step size
            step = alpha * 1.1
        
        # Periodic perturbation
        if it % 5000 == 0 and rng.random() < 0.3:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
    
    return best_local, best_score


def run_toppk_backtracking(a_init, n, S, max_time, rng, top_k_init=1, top_k_max=50):
    """
    Nesterov with top-k (uniform) weights and backtracking line search.
    """
    a = safe_normalize(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 2.0 * S / n
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
        
        weights = toppk_weights(c, top_k)
        grad = compute_gradient_fft(y, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Normalized gradient direction
        direction = -grad / gnorm
        
        # Backtracking line search with Armijo condition
        alpha = step
        c_y = fft_convolve_self(y)
        for _ in range(8):
            x_new = project_simplex(y + alpha * direction, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= c_y.max() + 1e-10:
                break
            alpha *= 0.5
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Update step
        step = alpha * 1.05
        
        # Perturb if stalled
        if it % 3000 == 0 and rng.random() < 0.2:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
    
    return best_local, best_score


def run_coordinate_descent_polish(a_init, n, S, max_time, rng, window_size=32):
    """
    Coordinate descent: pick a random window of elements, optimize them
    while keeping others fixed. This is like a block coordinate method.
    """
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Pick a random window
        start_idx = rng.integers(0, n - window_size)
        end_idx = start_idx + window_size
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()
        
        # Optimize the window using a few gradient steps
        # Compute gradient for the window
        w = toppk_weights(c, 1)
        grad = compute_gradient_fft(a, w)
        
        # Try a few step sizes
        best_in_window = cur
        for ls in [0.01, 0.05, 0.1, 0.5, 1.0]:
            a_trial = a.copy()
            a_trial[start_idx:end_idx] -= ls * grad[start_idx:end_idx]
            a_trial = safe_normalize(a_trial, n)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < best_in_window:
                best_in_window = c_trial.max()
                a = a_trial
                break
    
    return best_local, best_score


def run_structured_perturbation(a_init, n, S, max_time, rng):
    """
    Structured perturbation based on convolution structure.
    Identify positions that contribute most to the peak lag and
    try to reduce them by shifting mass to less critical positions.
    """
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()
        
        peak_lag = np.argmax(c)
        
        # The peak lag k means: sum_i a[i] * a[k-i] is maximized
        # Find the positions that contribute most to this peak
        # Contribution of position i to lag peak_lag: a[i] * a[peak_lag - i]
        
        # Get the contribution vector
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_lag - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        # Top contributors
        top_contrib = np.argsort(contrib)[::-1][:min(64, n // 4)]
        
        # Try to reduce mass at top contributors and add to random others
        n_move = min(8, len(top_contrib))
        if n_move == 0:
            continue
        
        source_indices = top_contrib[:n_move]
        target_indices = rng.choice(set(range(n)) - set(source_indices), size=n_move, replace=False)
        
        # Amount to move
        total_mass = sum(a[i] for i in source_indices)
        move_amount = 0.01 * total_mass / n_move
        
        a_trial = a.copy()
        for i, j in zip(source_indices, target_indices):
            move = min(move_amount, a[i] * 0.5)
            a_trial[i] -= move
            a_trial[j] += move
        
        a_trial = safe_normalize(a_trial, n)
        c_trial = fft_convolve_self(a_trial)
        
        if c_trial.max() < cur:
            a = a_trial
        else:
            # Try smaller move
            a_trial2 = a.copy()
            for i, j in zip(source_indices, target_indices):
                move = min(0.001 * a[i], a[i] * 0.1)
                a_trial2[i] -= move
                a_trial2[j] += move
            a_trial2 = safe_normalize(a_trial2, n)
            c_trial2 = fft_convolve_self(a_trial2)
            if c_trial2.max() < cur:
                a = a_trial2
    
    return best_local, best_score

import numpy as np

from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Gradient of sum(weights * conv) w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Softmax weights."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights_uniform(c, top_k):
    """Uniform weights on top-k entries."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    weights = np.zeros(len(c))
    weights[top_indices] = 1.0 / top_k
    return weights


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    """Gaussian window initialization."""
    x = np.arange(n) - n / 2.0

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)

    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)

    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]
