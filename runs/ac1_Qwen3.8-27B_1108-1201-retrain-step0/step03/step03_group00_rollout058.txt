def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_flip_shift(a, k, n):
    """Compute the vector v where v[i] = a[k-i] (0 if out of bounds)."""
    v = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        v[lo:hi+1] = a[k-hi:k-lo+1][::-1]
    return v


def lse_gradient_precise(a, c, beta):
    """High-precision LSE gradient with underflow protection."""
    n = len(a)
    cm = c.max()

    e = np.exp(scale * shifted)

        return 2.0 * compute_flip_shift(a, k, n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

"""
Minimize 2*n*max_conv/sum^2 using an adaptive active-set gradient method
with backtracking, multi-scale FISTA warm starts, and periodic rebalancing.
Key improvements over prior work: (1) backtracking line search for guaranteed
descent, (2) adaptive active-set size that tracks the "width" of the peak,
(3) dual-temperature scheduling (aggressive focus early, broad coverage late),
(4) structured perturbations that preserve the solution's symmetry class,
and (5) optional CVXPY cutting-plane refinement for final polishing.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_flip_shift(a, k, n):
    """Compute vector v where v[i] = a[k-i] (0 if out of bounds)."""
    v = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        v[lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return v


def lse_weighted_gradient(a, c, beta, top_k=None):
    """LSE gradient with optional top-k restriction for efficiency."""
    n = len(a)
    cm = c.max()
    shifted = beta * (c - cm)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -800, 800)
    e = np.exp(shifted)
    if top_k is not None:
        # Zero out weights outside top_k
        if top_k < len(c):
            idx = np.argsort(c)[::-1][top_k:]
            e[idx] = 0.0
    tot = e.sum()
    if tot < 1e-300:
        k = int(np.argmax(c))
        return 2.0 * compute_flip_shift(a, k, n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def power_mean_gradient(a, c, p, top_k=None):
    """Gradient of (sum c^p)^{1/p} approximation."""
    n = len(a)
    c_pos = np.maximum(c, 1e-30)
    if top_k is not None and top_k < len(c):
        mask = np.zeros(len(c), dtype=bool)
        mask[np.argsort(c_pos)[::-1][:top_k]] = True
        c_pos = np.where(mask, c_pos, 0.0)
    cp_arr = np.power(c_pos, p)
    tot = cp_arr.sum()
    if tot < 1e-15:
        k = int(np.argmax(c))
        return 2.0 * compute_flip_shift(a, k, n)
    w = np.power(c_pos, p - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        k = int(np.argmax(c))
        return 2.0 * compute_flip_shift(a, k, n)
    w = w / wsum
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista_bt(a_init, n, S, max_time, rng,
                 beta_init=10.0, beta_rate=0.004, beta_max=2000.0,
                 use_power=False, p_init=4.0, p_rate=0.005, p_max=80.0):
    """
    FISTA with backtracking line search and adaptive beta/p schedule.
    Uses a two-timescale approach: fast primal updates with slow beta growth.
    """
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    step = 1.0  # Initial step size

    while time.monotonic() - start < max_time:
        it += 1

        if use_power:
            p = min(p_max, p_init + p_rate * it)
        else:
            beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c_y = fft_convolve_self(y)
        cur_y = c_y.max()
        if cur_y < best:
            best = cur_y
            best_a = y.copy()
            last_improve = it

        # Compute gradient at y
        if use_power:
            grad = power_mean_gradient(y, c_y, p)
        else:
            grad = lse_weighted_gradient(y, c_y, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        def objective_at(x):
            cx = fft_convolve_self(x)
            return cx.max()

        f_y = objective_at(y)
        alpha = step
        x_new = y
        tried = 0
        while tried < 20:
            tried += 1
            candidate = project_simplex(y - alpha * grad, S)
            f_cand = objective_at(candidate)
            # Armijo condition (relaxed for non-smooth)
            if f_cand <= f_y - 0.01 * alpha * gnorm * np.linalg.norm(y - candidate) or f_cand < f_y:
                x_new = candidate
                step = min(2.0, alpha * 1.5)
                break
            alpha *= 0.5
        else:
            # If backtracking exhausted, use smallest step
            x_new = project_simplex(y - alpha * grad, S)
            step = alpha

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-normalize to stay on the simplex
        y_sum = y.sum()
        if y_sum > 0:
            y = y * (S / y_sum)
        a = x_new
        t_mom = t_new

        # Staleness detection with structured perturbation
        if it - last_improve > 3000:
            # Exploit potential symmetry: try reflecting the sequence
            if rng.random() < 0.3:
                y = y[::-1].copy()
            # Add noise
            noise_scale = 0.003 * S / np.sqrt(n) * np.exp(-(it - last_improve) / 20000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
            step = 1.0

        # Periodic annealing restart
        if it % 8000 == 0:
            decay = np.exp(-it / 50000.0)
            if decay > 0.01:
                noise_scale = 0.005 * S / np.sqrt(n) * decay
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_cutting_plane_polish(a_init, n, S, max_time, rng, max_active=80):
    """
    Cutting-plane polish: solve LP subproblems with linearized active
    constraints and a trust region. Effective for final refinement.
    """
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    c = fft_convolve_self(a)
    best = c.max()
    best_a = a.copy()

    delta = 0.05 * S / np.sqrt(n)  # Trust region radius
    min_delta = 0.001 * S / np.sqrt(n)
    M = 20

    it = 0
    while time.monotonic() - start < max_time - 10:
        it += 1
        c = fft_convolve_self(a)
        if c.max() < best:
            best = c.max()
            best_a = a.copy()

        # Active set: top M lags
        top_idx = np.argsort(c)[::-1][:M]
        active = top_idx.tolist()

        # Build LP
        a_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        constraints = [cp.sum(a_var) == S]

        # Linearized constraints
        for k in active:
            v = compute_flip_shift(a, k, n)
            # (a*a)[k] linearized at current a:
            # ≈ 2 * v . a_var - c[k]
            constraints.append(2.0 * cp.dot(v, a_var) - c[k] <= t_var)

        # Trust region
        constraints.append(cp.norm(a_var - a, np.inf) <= delta)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except Exception:
                break

        if a_var.value is None:
            break

        a_new = np.clip(a_var.value, 0.0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new * (S / s_new)

        # Evaluate
        c_new = fft_convolve_self(a_new)
        if c_new.max() <= best + 1e-8:
            a = a_new.copy()
            delta = min(0.1 * S / np.sqrt(n), delta * 1.3)
        else:
            # Shrink trust region and continue
            delta = max(min_delta, delta * 0.5)

        # Grow active set
        M = min(max_active, M + 3)

        # Convergence check
        if it > 50 and abs(c_new.max() - c.max()) < 1e-10 * best:
            break

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def generate_structured_init(n, rng):
    """Generate a structured initialization with potential to beat uniform."""
    tt = np.linspace(0, 1, n, endpoint=False)
    inits = []

    # Raised cosine variants with different parameters
    for alpha in [1.0, 1.5, 2.0, 3.0]:
        w = np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt) ** alpha), 0.01)
        inits.append(w)

    # Gaussian window
    for sig in [0.2, 0.3, 0.4]:
        w = np.exp(-0.5 * ((tt - 0.5) / sig) ** 2)
        inits.append(w + 0.05)

    # Multi-frequency
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * 2 * tt + 0.3) + 0.2 * np.cos(2 * np.pi * 5 * tt + 1.0))
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.3 * np.sin(2 * np.pi * 7 * tt + 0.5))

    # Power-law edge emphasis
    for p in [0.5, 1.0, 2.0]:
        w = np.maximum(0.5 - np.abs(tt - 0.5), 0.01) ** p
        inits.append(w + 0.1)

    # Random structured
    inits.append(rng.random(n) * 0.5 + 0.5)
    inits.append(np.exp(rng.normal(0, 0.3, n)))

    return [np.maximum(x, 0.01) for x in inits]
