Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS on softmax parameterization
with adaptive logsumexp temperature, combined with coordinate descent polishing.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def obj_grad_softmax(x, T):
    """
    Objective and gradient for softmax parameterization.
    a = softmax(x), objective = 2n * logsumexp(T*c)/T where c = conv(a,a)
    """
    n = len(x)
    # Compute softmax
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a /= a.sum()
    
    # Compute convolution
    c = fft_convolve_self(a)
    
    # Smoothed max via logsumexp
    c_max = c.max()
    exp_c = np.exp(T * (c - c_max))
    log_sum = np.log(exp_c.sum())
    soft_max = c_max + log_sum / T
    
    obj = 2.0 * n * soft_max
    
    # Gradient of soft_max w.r.t. c
    w = exp_c / exp_c.sum()
    
    # Gradient of c w.r.t. a:
    # d c[k] / d a[i] = 2*a[k-i] if 0 <= k-i < n, else 0
    # grad_a[i] = 2n * sum_k w[k] * 2*a[k-i] * 1{0<=k-i<n}
    #           = 4n * sum_j a[j] * w[i+j]  (where j = k-i, k = i+j)
    
    # Compute via FFT:
    a_rev = a[::-1]
    out_len = n + len(w) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A_rev = fft(a_rev, fsize)
    W = fft(w, fsize)
    full_conv = np.real(ifft(A_rev * W))
    grad_a = 4.0 * n * full_conv[n-1:2*n-1]
    
    # Gradient w.r.t. x (softmax Jacobian):
    # d obj / d x[m] = a[m] * (grad_a[m] - dot(grad_a, a))
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x


def l_bfgs_optimize(a0, time_budget, T_start=50.0, T_end=5000.0, 
                    n_temperature_steps=5, verbose=True):
    """
    Optimize using L-BFGS with increasing temperature.
    a0: initial point on the simplex (sum=1, all >= 0)
    """
    n = len(a0)
    a = a0.copy()
    best_obj = compute_obj(a)
    best_a = a.copy()
    start = time.monotonic()
    
    # Convert to log-space
    a_safe = np.maximum(a, 1e-30)
    x = np.log(a_safe / a_safe.sum())
    
    T_values = np.geomspace(T_start, T_end, n_temperature_steps)
    
    for step_idx, T in enumerate(T_values):
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break
        
        # Time per L-BFGS call
        max_time = remaining / (n_temperature_steps - step_idx) - 2
        
        def fun_and_grad(x_eval):
            return obj_grad_softmax(x_eval, T)
        
        result = minimize(
            fun_and_grad, x, method='L-BFGS-B',
            jac=True,  # fun_and_grad returns (f, g)
            options={
                'maxiter': int(max(10, max_time * 50)),  # rough estimate
                'maxfun': int(max(100, max_time * 500)),
                'ftol': 1e-12,
                'gtol': 1e-8,
                'disp': False,
                'maxls': 50,
                'maxcor': 20,
            }
        )
        
        # Get new a
        x_new = result.x
        a_new = np.exp(x_new - x_new.max())
        a_new /= a_new.sum()
        
        obj_new = compute_obj(a_new)
        
        if verbose:
            elapsed = time.monotonic() - start
            print(f"    T={T:.0f}: obj={obj_new:.8f}, best={best_obj:.8f}, "
                  f"iters={result.niter}, t={elapsed:.1f}s")
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        x = x_new
    
    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS on softmax parameterization
with adaptive logsumexp temperature for smooth max approximation, followed by
stochastic and coordinate polishing. Key innovation: smooth the non-differentiable
max via logsumexp, enabling efficient quasi-Newton optimization on the simplex.
"""
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_value(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return a / s


def obj_grad_softmax(x, T):
    """
    Objective and gradient for softmax parameterization.
    a = softmax(x), objective = 2n * logsumexp(T*c)/T where c = conv(a,a).
    Since sum(a)=1, this equals 2n * smoothed_max(c).
    """
    n = len(x)
    a = softmax(x)
    
    c = fft_convolve_self(a)
    
    # Smoothed max via logsumexp
    c_max = c.max()
    exp_c = np.exp(T * (c - c_max))
    exp_sum = exp_c.sum()
    soft_max = c_max + np.log(exp_sum) / T
    
    obj = 2.0 * n * soft_max
    
    # Weight for each convolution coefficient
    w = exp_c / exp_sum
    
    # Compute grad_a: grad_a[i] = 4n * sum_j a[j] * w[i+j]
    a_rev = a[::-1]
    out_len = n + len(w) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A_rev = fft(a_rev, fsize)
    W = fft(w, fsize)
    full_conv = np.real(ifft(A_rev * W))
    grad_a = 4.0 * n * full_conv[n-1:2*n-1]
    
    # Softmax Jacobian chain rule:
    # d obj / d x[m] = a[m] * (grad_a[m] - sum_i grad_a[i]*a[i])
    dot = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x


def l_bfgs_phase(a0, time_budget, T_list, verbose=True):
    """
    Run L-BFGS with a sequence of temperatures (increasing sharpness).
    a0: initial point on simplex (sum=1, all >= 0).
    Returns best (a, obj) found.
    """
    n = len(a0)
    a = a0.copy()
    best_obj = 2.0 * n * fft_convolve_self(a).max()
    best_a = a.copy()
    
    a_safe = np.maximum(a, 1e-30)
    x = np.log(a_safe / a_safe.sum())
    
    start = time.monotonic()
    n_temps = len(T_list)
    
    for ti, T in enumerate(T_list):
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining < 3.0:
            break
        
        alloc_time = remaining / (n_temps - ti)
        
        def fun_and_grad(x_eval):
            o, g = obj_grad_softmax(x_eval, T)
            return o, g
        
        # Estimate max iterations: each eval ~0.5ms for n=4096, L-BFGS uses ~10 evals/iter
        max_iters = max(20, int(alloc_time * 800))
        max_fun = max(200, int(alloc_time * 8000))
        
        try:
            result = minimize(
                fun_and_grad, x, method='L-BFGS-B',
                jac=True,
                options={
                    'maxiter': max_iters,
                    'maxfun': max_fun,
                    'ftol': 1e-14,
                    'gtol': 1e-10,
                    'maxls': 30,
                    'maxcor': 15,
                    'disp': False,
                }
            )
            x = result.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error at T={T}: {e}")
            break
        
        a_new = softmax(x)
        obj_new = 2.0 * n * fft_convolve_self(a_new).max()
        
        if verbose:
            elapsed = time.monotonic() - start
            status = " *" if obj_new < best_obj else ""
            print(f"    T={T:.0f}: obj={obj_new:.8f}, best={best_obj:.8f}, "
                  f"niter={result.niter}, t={elapsed:.1f}s{status}")
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj


def focused_coord_descent(a0, time_budget, verbose=True):
    """
    Coordinate descent focusing on coordinates that contribute to the current max peak.
    For each coordinate, try a range of multiplicative factors.
    """
    a = a0.copy()
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    start = time.monotonic()
    
    factors = np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 3.0])
    
    sweep = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        improved = False
        c_max = c.max()
        k_peak = np.argmax(c)
        
        # Determine which indices contribute to the peak
        # c[k_peak] = sum_j a[j]*a[k_peak-j]
        # Indices j where both j and k_peak-j are valid
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        
        # Sort by contribution to the peak (descending)
        contribs = np.zeros(n)
        for j in range(lo, hi + 1):
            contribs[j] = a[j] * a[k_peak - j]
        
        order = np.argsort(-contribs)
        
        # Process top contributors first
        for _pass in range(3):  # 3 passes over the ordered coordinates
            for idx in order:
                if time.monotonic() - start > time_budget:
                    break
                if contribs[idx] < 1e-15 and _pass > 0:
                    continue  # Only process zero-contributing coords in first pass
                
                orig = a[idx]
                if orig < 1e-15:
                    continue
                
                local_best_obj = 2.0 * n * c_max / (S * S)
                local_best_f = 1.0
                
                for f in factors:
                    a[idx] = orig * f
                    s_new = a.sum()
                    if s_new < 1e-10:
                        a[idx] = orig
                        continue
                    a_norm = a * (S / s_new)
                    c_try = fft_convolve_self(a_norm)
                    obj_try = 2.0 * n * c_try.max() / (S * S)
                    if obj_try < local_best_obj:
                        local_best_obj = obj_try
                        local_best_f = f
                
                if local_best_f != 1.0:
                    a[idx] = orig * local_best_f
                    a = a * (S / a.sum())
                    c = fft_convolve_self(a)
                    cur_obj = 2.0 * n * c.max() / (S * S)
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                        improved = True
        
        sweep += 1
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    coord sweep {sweep}: obj={best_obj:.8f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def stochastic_hill_climb(a0, time_budget, n_perturb=15, verbose=True):
