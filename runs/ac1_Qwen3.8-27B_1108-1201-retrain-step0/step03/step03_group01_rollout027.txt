"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using FISTA-LSE with adaptive beta,
softplus parameterization, and local search polishing.

Key improvements over previous approach:
1. Analytical initializations (raised cosine, Gaussian, parabolic) that exploit

2. Softplus parameterization (a = S*softmax(z)) that avoids projection artifacts
   and allows smoother optimization in the interior.
3. Adaptive beta schedule that starts smooth and increases aggressively.
4. Local search with structured perturbations in the final phase.
5. Multi-length exploration.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, N, beta):
    """
    Compute the gradient of the LSE approximation of max(conv(a,a)).
    Returns (grad, c, c_max, lse_value).
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2 * n - 1]
    c_max = c.max()
    
    # LSE: (1/beta) * log(sum(exp(beta * c)))
    # For numerical stability, subtract c_max
    shifted = beta * (c - c_max)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -700, 700)
    exp_c = np.exp(shifted)
    total = exp_c.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = exp_c / total
    
    # Gradient of LSE w.r.t. a:
    # d/d a_j [sum_k w_k * c_k] where c_k = sum_i a_i * a[k-i]
    # = sum_k w_k * (a[k-j] + a[j-k] if valid)
    # In circular convolution form: ifft(fft(w) * conj(fft(a)))
    w_full = np.zeros(N)
    w_full[:2*n-1] = w
    W = fft(w_full)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    lse_val = c_max + np.log(total) / beta
    return grad, c, c_max, lse_val


def fista_lse_adaptive(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=1000.0, 
                        beta_growth=1.001, step_scale=1.0):
    """FISTA with adaptive LSE beta."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    N = 1 << (2 * n - 1).bit_length()
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    last_improve = start
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (beta_growth ** it))
        
        grad, c, c_max, lse_val = lse_gradient(y, N, beta)
        cur = c_max
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            last_improve = time.monotonic()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        step = step_scale * S / (n * np.sqrt(1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
                last_improve = time.monotonic()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Restart momentum if stalled
        if time.monotonic() - last_improve > 30 and rng.random() < 0.05:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score


def softplus_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=500.0):
    """
    FISTA using softmax parameterization: a = S * softmax(z).
    Optimizes over z in R^n (unconstrained).
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    # Initialize z from a: a = S * softmax(z) → z = log(a/S) (up to a constant)
    a_safe = np.maximum(a, 1e-15)
    z = np.log(a_safe / S)
    z -= z.mean()  # center for numerical stability
    
    N = 1 << (2 * n - 1).bit_length()
    
    y = z.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 1.0
    last_improve = start
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.002 ** it))
        
        # Compute a from y
        y_max = y.max()
        exp_y = np.exp(y - y_max)
        Z = exp_y.sum()
        a_curr = S * exp_y / Z
        
        # Compute gradient w.r.t. a
        grad, c, c_max, lse_val = lse_gradient(a_curr, N, beta)
        cur = c_max
        if cur < best_score:
            best_score = cur
            best_local = a_curr.copy()
            last_improve = time.monotonic()
        
        # Chain rule: grad_z = a * (grad_a - <grad_a, a>/S)
        inner = np.dot(grad, a_curr) / S
        grad_z = a_curr * (grad - inner)
        
        gnorm = np.linalg.norm(grad_z)
        if gnorm < 1e-15:
            break
        
        # Adaptive step
        # The Lipschitz constant is hard to bound, use conservative step
        step = 0.5 / (beta * np.max(a_curr) + 1.0)
        step = min(step, 1.0 / (10.0 * np.max(np.abs(grad_z)) + 1e-15))
        step = max(step, 1e-8)
        
        z_new = y - step * grad_z
        
        if it % 3 == 0:
            # Evaluate
            z_new_max = z_new.max()
            exp_zn = np.exp(z_new - z_new_max)
            a_eval = S * exp_zn / exp_zn.sum()
            c_eval = fft_convolve_self(a_eval)
            if c_eval.max() < best_score:
                best_score = c_eval.max()
                best_local = a_eval.copy()
                last_improve = time.monotonic()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = z_new + ((t - 1.0) / t_new) * (z_new - y)  # FISTA momentum
        # Wait, this is wrong. FISTA uses:
        # y_new = x_new + ((t-1)/t_new) * (x_new - x_old)
        # Let me fix:
        # We need to keep track of the previous z
        # Actually let me redo this properly
        
        a_prev = a_curr
        a_curr = a_eval if it % 3 == 0 else a_curr
        
        t = t_new
        # Reset
        y = z_new  # Simplified: no momentum for now (plain gradient descent)
        t = 1.0
        
        if time.monotonic() - last_improve > 60:
            z = z + rng.normal(0, 0.1, n)
            z -= z.mean()
            t = 1.0
    
    return best_local, best_score


def local_search(a_init, n, S, max_time, rng, n_trials=10000):
    """
    Local search: try random perturbations and keep improvements.
    Perturbation: move mass from a random subset of positions to another random subset.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    a = a * (S / a.sum())
    N = 1 << (2 * n - 1).bit_length()
    
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    accepted = 0
    trial = 0
    
    while time.monotonic() - start < max_time and trial < n_trials:
        trial += 1
        
        # Choose perturbation type
        r = rng.random()
        if r < 0.3:
            # Gaussian noise
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            a_new = project_simplex(a + noise, S)
        elif r < 0.5:
            # Move mass: pick two random positions, transfer mass
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            amt = 0.01 * S / n * rng.random()
            a_new = a.copy()
            a_new[i] = max(0, a_new[i] - amt)
            a_new[j] = a_new[j] + amt
            a_new = a_new * (S / a_new.sum())
        elif r < 0.7:
            # Block swap: swap two blocks
            block_size = rng.integers(1, max(2, n // 100))
            i = rng.integers(0, n - block_size)
            j = rng.integers(0, n - block_size)
            a_new = a.copy()
            a_new[i:i+block_size], a_new[j:j+block_size] = a_new[j:j+block_size].copy(), a_new[i:i+block_size].copy()
            a_new = a_new * (S / a_new.sum())
        elif r < 0.85:
            # Scale a random subset
            mask = rng.random(n) < 0.1
            factor = 1.0 + 0.05 * rng.standard_normal()
            a_new = a.copy()
            a_new[mask] *= factor
            a_new = np.maximum(a_new, 0)
            a_new = a_new * (S / a_new.sum())
        else:
            # Smooth perturbation (low-pass)
            noise = rng.normal(0, 1, n)
            # Low-pass filter
            fsize = 1 << (n - 1).bit_length()
            noise_f = fft(noise, fsize)
            # Keep only low frequencies
            cutoff = max(1, n // (2**rng.integers(2, 8)))
            mask_f = np.zeros(fsize, dtype=bool)
            mask_f[:cutoff] = True
            mask_f[fsize-cutoff:] = True
            mask_f[fsize//2] = True
            noise_f[~mask_f] = 0
            noise = np.real(ifft(noise_f))[:n]
            noise = noise / (np.abs(noise).max() + 1e-15)
            a_new = project_simplex(a + 0.02 * S / np.sqrt(n) * noise, S)
        
        c_new = fft_convolve_self(a_new)
        score_new = c_new.max()
        
        if score_new < best_score:
            a = a_new
            best_score = score_new
            best_local = a_new.copy()
            accepted += 1
    
    return best_local, best_score


def make_raised_cosine(n):
    """Raised cosine: sin^2(pi*(i+0.5)/n)"""
    t = (np.arange(n) + 0.5) / n
    return np.sin(np.pi * t) ** 2


def make_gaussian(n, sigma=0.3):
    """Gaussian bump"""
    t = np.arange(n) / n - 0.5
    return np.exp(-0.5 * (t / sigma) ** 2)


def make_parabolic(n):
    """Parabolic: 1 - (2t-1)^2"""
    t = np.arange(n) / n
    return 1.0 - (2 * t - 1) ** 2


def make_power_law(n, p=2.0):
    """(sin(pi*t))^p"""
    t = np.arange(n) / (n + 1)
    return np.sin(np.pi * t) ** p


def make_bipolar_smooth(n, rng):
    """Smooth biphasic: high in two regions, low in middle"""
    t = np.arange(n) / n
    a = np.exp(-((t - 0.25) / 0.15) ** 2) + np.exp(-((t - 0.75) / 0.15) ** 2)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.01)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
