def fft_grad_toppk(a, c, top_k):
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    weights = top_vals / total
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_conv(a):
    """Compute full self-convolution using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad(a, w, N_fft=None):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    w has length 2n-1 (the convolution length).
    Uses FFT cross-correlation: grad = 2*ifft(fft(w,N)*conj(fft(a,N)))[:n]
    """
    n = len(a)
    if N_fft is None:
        N_fft = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, N_fft)
    A = fft(a, N_fft)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def lp_optimize(a_init, n, S, max_time, rng, p_start=4, p_end=512,
                p_growth=1.08, top_k=None):
    """
    Adaptive Lp optimization with backtracking line search.
    Minimizes (sum(c^p)/p)^(1/p) as p increases toward infinity.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    N_fft = 1 << (2 * n - 1 + n - 2).bit_length()
    start_t = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.inf
    p = p_start
    it = 0
    step = 1.0  # will be adjusted by backtracking

    while time.monotonic() - start_t < max_time:
        it += 1
        c = compute_conv(a)
        c_max = c.max()
        if c_max < best_conv_max:
            best_conv_max = c_max
            best_a = a.copy()

        # Compute weights for Lp gradient
        if top_k is not None and top_k < len(c):
            idx = np.argsort(c)[::-1][:top_k]
            w = np.zeros(len(c))
            c_top = c[idx]
            w[idx] = c_top ** (p - 1)
        else:
            w = c ** (p - 1)
            # Clip to avoid overflow
            w_max = w.max()
            if w_max > 1e100:
                w = w / w_max

        w_sum = w.sum()
        if w_sum < 1e-30:
            break

        grad = compute_grad(a, w / w_sum, N_fft)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search (Armijo condition)
        f_curr = np.sum(c ** p) / p
        alpha = step
        max_backtracks = 12
        for _ in range(max_backtracks):
            a_trial = project_simplex(a - alpha * grad, S)
            c_trial = compute_conv(a_trial)
            if top_k is not None and top_k < len(c_trial):
                idx_t = np.argsort(c_trial)[::-1][:top_k]
                f_trial = np.sum(c_trial[idx_t] ** p) / p
            else:
                f_trial = np.sum(c_trial ** p) / p

            if f_trial <= f_curr - 1e-4 * alpha * np.dot(grad, a - a_trial):
                break
            alpha *= 0.5

        a = a_trial
        step = alpha / 0.5 if alpha > 0 else 0.5

        # Progressively increase p
        p = min(p_end, p * p_growth)

        # Occasional perturbation to escape local minima
        if it % 500 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_a, best_conv_max


def lse_optimize(a_init, n, S, max_time, rng, beta_start=5.0, beta_end=500.0,
                 top_k=None):
    """
    Log-sum-exp optimization with Nesterov acceleration.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    N_fft = 1 << (2 * n - 1 + n - 2).bit_length()
    y = a.copy()
    t_prev = 1.0
    start_t = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.inf
    it = 0
    beta = beta_start

    while time.monotonic() - start_t < max_time:
        it += 1
        beta = min(beta_end, beta_start * (1.0 + 0.005 * it))

        c = compute_conv(y)
        c_max = c.max()
        if c_max < best_conv_max:
            best_conv_max = c_max
            best_a = y.copy()

        # LSE gradient
        c_shifted = c - c_max
        exp_c = np.exp(beta * c_shifted)
        exp_sum = exp_c.sum()
        if exp_sum < 1e-300:
            break
        w = exp_c / exp_sum

        if top_k is not None and top_k < len(c):
            top_idx = np.argsort(c)[::-1][:top_k]
            w_sparse = np.zeros(len(c))
            w_sparse[top_idx] = w[top_idx]
            w = w_sparse

        grad = compute_grad(y, w, N_fft)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: 1/(beta * L) where L is Lipschitz constant ~ 2*max(a)
        L = 2.0 * np.max(a) + 1.0
        step = 1.0 / (beta * L * 0.5)

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev ** 2))
        y = x_new + ((t_prev - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_prev = t_new

        # Perturbation
        if it % 1000 == 0 and rng.random() < 0.15:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t_prev = 1.0

    return best_a, best_conv_max


def spectral_phase_optimize(a_init, n, S, max_time, rng, n_iter=500,
                            top_k=16, alpha=0.4):
    """
    Spectral optimization: modify phase of FFT to reduce convolution peak.
    Since conv(a,a) = ifft(A^2), we can modify phase of A to flatten the result.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    N_fft = 1 << (2 * n - 1 + 1).bit_length()
    start_t = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.inf

    for it in range(n_iter):
        if time.monotonic() - start_t > max_time:
            break

        a_pad = np.zeros(N_fft)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        phase = np.angle(A)

        # Compute current convolution
        c_full = np.real(ifft(A * A))[:2 * n - 1]
        c_max = c_full.max()
        if c_max < best_conv_max:
            best_conv_max = c_max
            best_a = a.copy()

        if c_max < 1.0:
            break

        # Compute effective gradient in frequency domain
        top_idx = np.argsort(c_full)[::-1][:top_k]
        top_vals = c_full[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # cos(2*phase) weighted by peak contributions
        m = np.arange(N_fft)
        cos_eff = np.zeros(N_fft)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * phase[int(np.min(k, N_fft - k))] + 2.0 * np.pi * m * int(k) / N_fft)

        # Modify magnitude to suppress peaks
        alpha_eff = alpha * min(1.0, max(0.0, c_max - 0.5))
        P_new = np.maximum(P * (1.0 - alpha_eff * cos_eff), 0.0)

        # Also try phase adjustment
        # The key: conv = ifft(A^2) = ifft(|A|^2 * exp(2i*phase))
        # To flatten, we want to adjust phase so that |A|^2 * exp(2i*phase)
        # has a flatter inverse FFT
        # Gradient of max wrt phase: use the top peaks
        A_new = np.sqrt(P_new) * np.exp(1j * phase)

        # Project back
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)

        # Occasional random phase perturbation
        if it % 200 == 100 and rng.random() < 0.3:
            a_pad = np.zeros(N_fft)
            a_pad[:n] = a
            A = fft(a_pad)
            P = np.abs(A) ** 2
            phase = np.angle(A) + rng.normal(0, 0.05, N_fft)
            A_new = np.sqrt(P) * np.exp(1j * phase)
            a = project_simplex(np.real(ifft(A_new))[:n], S)

    return best_a, best_conv_max


def nesterov_subgrad(a_init, n, S, max_time, rng, top_k_init=16, top_k_max=256):
    """Nesterov-accelerated top-k subgradient method."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_prev = 1.0
    start_t = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.inf
    it = 0

    N_fft = 1 << (2 * n - 1 + n - 2).bit_length()

    while time.monotonic() - start_t < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)

        c = compute_conv(y)
        cur = c.max()
        if cur < best_conv_max:
            best_conv_max = cur
            best_a = y.copy()

        # Top-k subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        w = np.zeros(2 * n - 1)
        w[top_idx] = top_vals / total

        grad = compute_grad(y, w, N_fft)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev ** 2))
        y = x_new + ((t_prev - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_prev = t_new

        if it % 500 == 0 and rng.random() < 0.12:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t_prev = 1.0

    return best_a, best_conv_max


def make_initialization(name, n, rng, S):
    """Create diverse initializations."""
    tt = np.linspace(0, 1, n, endpoint=False)

    if name == "uniform":
        return np.ones(n) * S / n
    elif name == "random":
        a = rng.random(n) + 0.1
        return a * (S / a.sum())
    elif name == "chirp":
        a = 0.5 + 0.4 * np.sin(2 * np.pi * (3 * tt + 0.5 * tt ** 2)) + 0.1 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
        return a * (S / a.sum())
    elif name == "sine_multi":
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.sin(2 * np.pi * 5 * tt) + 0.1 * np.sin(2 * np.pi * 11 * tt)
        a = np.clip(a, 0.01, None)
        return a * (S / a.sum())
    elif name == "cosine_bell":
        a = 0.5 * (1 + np.cos(2 * np.pi * tt)) + 0.1
        return a * (S / a.sum())
    elif name == "two_valued":
        # Random binary pattern
        a = (rng.random(n) > 0.5).astype(float) * 0.8 + 0.1
        return a * (S / a.sum())
    elif name == "segmented":
        k = rng.integers(3, 8)
        seg_len = n // k
        boundaries = np.arange(0, n + 1, seg_len)
        boundaries[-1] = n
        heights = rng.random(len(boundaries) - 1) + 0.1
        a = np.zeros(n)
        for i in range(len(boundaries) - 1):
            a[boundaries[i]:boundaries[i + 1]] = heights[i]
        return a * (S / a.sum())
    elif name == "msequence":
        # Mersenne-twister inspired pattern
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 7 * tt) * np.cos(2 * np.pi * 3 * tt)
        a += 0.1 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
        return a * (S / a.sum())
    else:
        return np.ones(n) * S / n


def safe_normalize(a, n, S):
    """Clamp to valid range and normalize."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing evaluate_sequence.
    Uses adaptive Lp optimization, spectral phase methods, and Nesterov subgradient.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Determine length - try to use previous if available
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 128:
            n_primary = len(given)
        else:
            n_primary = 8192
    except Exception:
        n_primary = 8192

    # We'll try the primary length and possibly one other
    n_list = [n_primary]
    if n_primary == 8192:
        n_list.append(4096)
    elif n_primary == 4096:
        n_list.append(8192)
    else:
        n_list.append(8192)

    print("Primary n=%d, alternatives=%s" % (n_primary, n_list))

    for n in n_list:
        if time.monotonic() > deadline - 120:
            break

        S = np.sqrt(2.0 * float(n))
        print("\n=== Optimizing n=%d, S=%.4f ===" % (n, S))

        # Generate initializations
        init_names = ["uniform", "random", "chirp", "sine_multi",
                      "cosine_bell", "two_valued", "segmented", "msequence"]

        # Add previous best as initialization
        if n == n_primary:
            try:
                given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
                if given.size == n and np.sum(np.clip(given, 0, 1000)) > 1:
                    init_names.append("previous_best")
            except Exception:
                pass

        # Quick evaluation of each init
        candidates = []
        for name in init_names:
            if time.monotonic() > deadline - 60:
                break
            if name == "previous_best":
                try:
                    given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
                    a = safe_normalize(given, n, S)
                except Exception:
                    continue
            else:
                a = make_initialization(name, n, rng, S)

            c = compute_conv(a)
            score = c.max()
            candidates.append((name, a, score))
            print("  Init [%s]: conv_max=%.6f, obj=%.6f" % (name, score, 2 * n * score / S ** 2))

        # Sort by score
        candidates.sort(key=lambda x: x[2])

        # Optimizer budget allocation
        n_cands = min(5, len(candidates))  # Only optimize top candidates
        total_time = deadline - time.monotonic()
        per_cand = total_time / n_cands

        for ci in range(n_cands):
            if time.monotonic() > deadline - 30:
                break
            name, a_init, init_score = candidates[ci]
            time_left = min(per_cand, deadline - time.monotonic() - 10)
            if time_left < 5:
                break

            print("  Optimizing [%s] (init score=%.6f), budget=%.0fs" % (name, init_score, time_left))

            # Sub-allocate time among methods
            t_lp = time_left * 0.4
            t_spec = time_left * 0.25
            t_lse = time_left * 0.15
            t_nest = time_left * 0.2

            # Method 1: Adaptive Lp with backtracking
            if t_lp > 3:
                a_lp, sc_lp = lp_optimize(a_init, n, S, t_lp, rng,
                                          p_start=4, p_end=256, p_growth=1.1,
                                          top_k=32)
                a_lp = safe_normalize(a_lp, n, S)
                c_lp = compute_conv(a_lp)
                if c_lp.max() < global_best:
                    global_best = c_lp.max()
                    global_a = a_lp.copy()
                    print("    Lp: %.6f (obj=%.6f)" % (c_lp.max(), 2 * n * c_lp.max() / S ** 2))
                best_so_far = a_lp if c_lp.max() < init_score else a_init
            else:
                best_so_far = a_init

            # Method 2: Spectral phase optimization
            if t_spec > 3:
                a_sp, sc_sp = spectral_phase_optimize(best_so_far, n, S, t_spec, rng,
                                                      n_iter=1000, top_k=16, alpha=0.35)
                a_sp = safe_normalize(a_sp, n, S)
                c_sp = compute_conv(a_sp)
                if c_sp.max() < global_best:
                    global_best = c_sp.max()
                    global_a = a_sp.copy()
                    print("    Spectral: %.6f (obj=%.6f)" % (c_sp.max(), 2 * n * c_sp.max() / S ** 2))
                best_so_far = a_sp if c_sp.max() < (c_lp.max() if t_lp > 3 else init_score) else best_so_far

            # Method 3: LSE with Nesterov
            if t_lse > 3:
                a_lse, sc_lse = lse_optimize(best_so_far, n, S, t_lse, rng,
                                             beta_start=10.0, beta_end=300.0,
                                             top_k=32)
                a_lse = safe_normalize(a_lse, n, S)
                c_lse = compute_conv(a_lse)
                if c_lse.max() < global_best:
                    global_best = c_lse.max()
                    global_a = a_lse.copy()
                    print("    LSE: %.6f (obj=%.6f)" % (c_lse.max(), 2 * n * c_lse.max() / S ** 2))
                best_so_far = a_lse if c_lse.max() < global_best else (global_a if global_a is not None else best_so_far)

            # Method 4: Nesterov subgradient polish
            if t_nest > 3:
                a_ng, sc_ng = nesterov_subgrad(best_so_far, n, S, t_nest, rng,
                                               top_k_init=16, top_k_max=200)
                a_ng = safe_normalize(a_ng, n, S)
                c_ng = compute_conv(a_ng)
                if c_ng.max() < global_best:
                    global_best = c_ng.max()
                    global_a = a_ng.copy()
                    print("    Nesterov: %.6f (obj=%.6f)" % (c_ng.max(), 2 * n * c_ng.max() / S ** 2))

    # Final polish phase
    if global_a is not None and time.monotonic() < deadline - 10:
        print("\n=== Final Polish ===")
        n_final = len(global_a)
        S_final = np.sqrt(2.0 * float(n_final))

        # Very high p Lp
        t1 = min(60.0, deadline - time.monotonic() - 5)
        if t1 > 5:
            a_p1, sc_p1 = lp_optimize(global_a, n_final, S_final, t1, rng,
                                      p_start=64, p_end=1024, p_growth=1.15,
                                      top_k=16)
            a_p1 = safe_normalize(a_p1, n_final, S_final)
            c1 = compute_conv(a_p1)
            if c1.max() < global_best:
                global_best = c1.max()
                global_a = a_p1.copy()
                print("  Polish Lp-high-p: %.6f (obj=%.6f)" % (c1.max(), 2 * n_final * c1.max() / S_final ** 2))

        # High-k Nesterov
        t2 = min(60.0, deadline - time.monotonic() - 5)
        if t2 > 5:
            a_p2, sc_p2 = nesterov_subgrad(global_a, n_final, S_final, t2, rng,
                                           top_k_init=32, top_k_max=512)
            a_p2 = safe_normalize(a_p2, n_final, S_final)
            c2 = compute_conv(a_p2)
            if c2.max() < global_best:
                global_best = c2.max()
                global_a = a_p2.copy()
                print("  Polish Nesterov: %.6f (obj=%.6f)" % (c2.max(), 2 * n_final * c2.max() / S_final ** 2))

        # Very fine spectral
        t3 = deadline - time.monotonic() - 5
        if t3 > 5:
            a_p3, sc_p3 = spectral_phase_optimize(global_a, n_final, S_final, t3, rng,
                                                  n_iter=2000, top_k=8, alpha=0.1)
            a_p3 = safe_normalize(a_p3, n_final, S_final)
            c3 = compute_conv(a_p3)
            if c3.max() < global_best:
                global_best = c3.max()
                global_a = a_p3.copy()
                print("  Polish Spectral: %.6f (obj=%.6f)" % (c3.max(), 2 * n_final * c3.max() / S_final ** 2))

    # Verification and return
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            print("\nVerification: %.6f" % ver if np.isfinite(ver) else "\nVerification: inf")
            if not np.isfinite(ver):
                print("  evaluate_sequence rejected, using fallback")
                global_a = np.ones(256, dtype=np.float64)
        except Exception as e:
            print("  Verification error: %s" % str(e))
    else:
        global_a = np.ones(256, dtype=np.float64)
        print("No valid candidate found, using fallback")

    final_obj = 2 * len(global_a) * compute_conv(global_a).max() / (global_a.sum() ** 2)
    print("Final: length=%d, internal_obj=%.6f" % (len(global_a), final_obj))
    return [float(x) for x in global_a]
