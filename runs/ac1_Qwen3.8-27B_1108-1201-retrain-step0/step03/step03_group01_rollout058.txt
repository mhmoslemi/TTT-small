The problem is: min max(c_k) s.t. a >= 0, sum(a) = S, where c = a*a.

This is equivalent to: min t s.t. c_k <= t for all k, a >= 0, sum(a) = S.

The constraint c_k <= t is: sum_i a_i * a_{k-i} <= t, which is a quadratic constraint. This makes the problem non-convex in general.

However, if we relax a >= 0 to just sum(a) = S (allowing negative values), the problem becomes:
min t s.t. sum_i a_i * a_{k-i} <= t for all k, sum(a) = S.

This is still non-convex (quadratic constraints). But it's a "quatically constrained quadratic program" (QCQP), which can be approximated by semidefinite programming (SDP).

The SDP relaxation would be:
min t
s.t. [t*I - C_k] >= 0 (PSD) for all k, where C_k is the matrix of the quadratic form
     sum(a) = S

But this is too expensive for n=8192.

An alternative: use the "sampled" approach where we only enforce the constraint at a subset of lags. This gives a (still non-convex) problem that's easier to solve.

One more idea: **Use the "proportional" allocation principle.** At the optimum, the "budget" S should be allocated among the a_i in a way that balances all the c_k values. This is like a "water-filling" problem.

Specifically, at the optimum (assuming all c_k are equal to t), we have:
sum_i a_i * a_{k-i} = t for all k.

This is a system of 2n-1 equations in n unknowns (overdetermined). The solution (if it exists) would give the optimal sequence.

For the uniform sequence, c_k = (k+1)*S^2/n^2 for k=0..n-1, which is not constant. So the uniform sequence is not at the optimum (for large n).

The optimal sequence should have a shape such that the autocorrelation is as flat as possible. This is a "flat autocorrelation" design problem.

For complex sequences, perfect flat autocorrelation is achievable (e.g., Zadoff-Chu, m-sequences). For real non-negative sequences, it's not achievable, but we can get close.

"""
Minimize 2*n*max_conv/sum^2 using complex spectral peak reduction with Nesterov acceleration.
Key innovations: complex-valued spectral gradient (uses conj(A) for exact gradient),
adaptive backtracking, multi-resolution initialization, and diverse multi-start.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a, N):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def compute_score(a):
    """Compute the objective score for a sequence."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2 * n * c.max() / (S * S))


def spectral_peak_reduction_step(a, N, top_k, step, threshold_frac=0.7):
    """
    One step of complex spectral peak reduction.
    Uses the exact gradient: in freq domain, grad c_k = 2*exp(-2*pi*i*m*k/N)*conj(A).
    Combined gradient for weighted peaks: 2*conj(A)*W where W = sum w_k*exp(-2*pi*i*m*k/N).
    """
    n = len(a)
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2 * n - 1]

    # Find active peaks
    c_max = c.max()
    threshold = threshold_frac * c_max
    active_mask = c > threshold
    active_indices = np.where(active_mask)[0]

    if len(active_indices) == 0:
        active_indices = np.argsort(c)[::-1][:top_k]
    else:
        if len(active_indices) > top_k:
            active_order = np.argsort(c[active_indices])[::-1][:top_k]
            active_indices = active_indices[active_order]

    active_vals = c[active_indices]
    total = active_vals.sum()
    if total < 1e-30:
        return a, c_max

    weights = active_vals / total

    # Compute W = DFT of weight kernel
    # W_m = sum_k w_k * exp(-2*pi*i*m*k/N)
    # Use FFT for efficiency
    w_kernel = np.zeros(N)
    for i, k in enumerate(active_indices):
        w_kernel[int(k)] = weights[i]
    W = fft(w_kernel)

    # Gradient in frequency domain: G = 2 * conj(A) * W
    # a_new = a - step * IFFT(G)
    G = 2.0 * np.conj(A) * W
    a_candidate = a - step * np.real(ifft(G))[:n]

    # Project to simplex
    S = a.sum()
    a_new = project_simplex(a_candidate, S)
    return a_new, c_max


def nesterov_spectral_optimize(a_init, n, S, N, max_time, rng,
                                top_k_init=20, top_k_max=256,
                                step_init=0.01, threshold_frac=0.7):
    """Nesterov-accelerated spectral peak reduction."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_param = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stall_count = 0
    last_best = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)
        step = step_init / (1.0 + 0.01 * t_param)

        # Compute gradient at y (Nesterov extrapolated point)
        A_y = fft(y, N)
        c_y = np.real(ifft(A_y * A_y))[:2 * n - 1]
        c_max_y = c_y.max()

        if c_max_y < best_score:
            best_score = c_max_y
            best_local = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Find active peaks
        threshold = threshold_frac * c_max_y
        active_mask = c_y > threshold
        active_indices = np.where(active_mask)[0]
        if len(active_indices) == 0:
            active_indices = np.argsort(c_y)[::-1][:top_k]
        else:
            if len(active_indices) > top_k:
                active_order = np.argsort(c_y[active_indices])[::-1][:top_k]
                active_indices = active_indices[active_order]
        active_vals = c_y[active_indices]
        total = active_vals.sum()
        if total < 1e-30:
            break
        weights = active_vals / total

        # Build W and compute step
        w_kernel = np.zeros(N)
        for i, k in enumerate(active_indices):
            w_kernel[int(k)] = weights[i]
        W = fft(w_kernel)

        # Gradient direction
        G = 2.0 * np.conj(A_y) * W
        grad_time = np.real(ifft(G))[:n]

        # Step
        x_new = y - step * grad_time

        # Project
        x_new = project_simplex(x_new, S)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        y = x_new + ((t_param - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Fix sum
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        a = x_new
        t_param = t_new

        # Periodic evaluation at x (not just y)
        if it % 50 == 0:
            c_x = fft_convolve_self(x_new, N)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
                stall_count = 0

        # Restart if stalled
        if stall_count >= 500:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t_param = 1.0
            stall_count = 0

        # Larger perturbation every 2000 iterations
        if it % 2000 == 0 and rng.random() < 0.2:
            y = project_simplex(y + rng.normal(0, 0.02 * S / np.sqrt(n), n), S)
            t_param = 1.0
            stall_count = 0

    return best_local, best_score


def fista_lse_optimize(a_init, n, S, N, max_time, rng, beta_init=5.0, beta_max=500.0):
    """FISTA with log-sum-exp smooth approximation of max."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_param = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.002 * it))

        A_y = fft(y, N)
        c_y = np.real(ifft(A_y * A_y))[:2 * n - 1]
        c_max = c_y.max()

        if c_max < best_score:
            best_score = c_max
            best_local = y.copy()

        # LSE gradient
        c_shifted = c_y - c_max
        exp_c = np.exp(beta * c_shifted)
        total = exp_c.sum()
        if total < 1e-300:
            break
        w = exp_c / total

        # Gradient = 2 * IFFT(FFT(w) * conj(FFT(y)))
        W = fft(w, N)
        G = 2.0 * np.conj(A_y) * W
        grad = np.real(ifft(G))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta * 2.0)
        x_new = project_simplex(y - step * grad, S)

        if it % 20 == 0:
            c_x = fft_convolve_self(x_new, N)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        y = x_new + ((t_param - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t_param = t_new

            t_param = 1.0

    return best_local, best_score

def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
