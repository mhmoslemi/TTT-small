import numpy as np

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_topk(a, w):
    """Compute 2 * cross-correlation of w and a via FFT.
    w has length 2n-1, a has length n.
    Returns gradient array of length n.
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def evaluate_conv_score(a):
    """Compute the objective value (before the 2n/sum^2 normalization)."""
    c = fft_convolve_self(a)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

        t += 1

        frac = min(1.0, t / 5000.0)
        K = max(1, int(top_k_start + frac * (top_k_end - top_k_start)))
        
        # Learning rate decay
        lr = lr_init / (1.0 + 0.001 * t)
        
        # Compute convolution
        c = fft_convolve_self(a)
        cur_max = c.max()
        cur_score = 2.0 * n * cur_max / (S * S)
        
        if cur_score < best_score:

            best_a = a.copy()
        
        # Find top-K positions
        if K >= len(c):
            w = np.ones(len(c)) / len(c)
        else:

            w = np.zeros(len(c))
            w[top_idx] = 1.0 / K

        grad = compute_grad_topk(a, w)
