"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. High-beta FISTA for quick improvement
2. LP-guided direction finding (reduced LP on top-K lags)
3. Multi-restart with diverse perturbations
4. Final polish with very high beta

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def topk_subgradient(a, c, k):
    """Subgradient using uniform weighting over top-k lags."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:k]
    w_full = np.zeros(2 * n - 1)
    w_full[top_indices] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def weighted_subgradient(a, c, k, alpha=1.0):
    """Subgradient using exponential weighting over top-k lags."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices]
    # Weight by how close to max
    vrange = top_vals[0] - top_vals[-1]
    if vrange < 1e-15:
        weights = np.ones(k) / k
    else:
        normalized = (top_vals - top_vals[-1]) / vrange
        weights = np.exp(alpha * normalized)
        weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-guided optimization,
FISTA with adaptive smoothing, spectral shaping, and multi-restart.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cand = (cumsum - S) / idx
    valid = theta_cand <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def smooth_max_grad(a, c, beta):
    """Gradient of smooth approximation to max(c) via log-sum-exp."""
    n = len(a)
    cm = c.max()
    exp_bc = np.exp(beta * (c - cm))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def subgrad_topk(a, c, k):
    """Subgradient: uniform weight on top-k lags."""
    n = len(a)
    k = min(k, len(c))
    top = np.argsort(c)[::-1][:k]
    w = np.zeros(2 * n - 1)
    w[top] = 1.0 / k
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = a.copy()
    best_score = np.inf
    it = 0
    L = 2.0
    
    while time.monotonic() - start < max_time:
        it += 1

        P = np.abs(A) ** 2
        P_mean = P.mean()

        strength = max(0.01, 0.1 * (1 - it / 1000))
        P_flat = P_mean * np.ones_like(P)
        P_new = P * (1 - strength) + P_flat * strength
        
        # Update A preserving phase
        phase = A / np.where(np.abs(A) > 1e-30, np.abs(A), 1.0)
        A_new = phase * np.sqrt(np.maximum(P_new, 1e-30))
        
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)

        if s_new < 1e-15:
            break
        a_new = a_new * (S / s_new)
        
        c = fft_convolve_self(a_new)
        score = c.max()
        if score < best_score:
            best_score = score
            best = a_new.copy()
        
        a = a_new
    
    return best, best_score

def lp_guided_step(a, c, top_k=50):
    """
    Solve reduced LP to find best descent direction.
    Returns (direction, t_value) or None.
    """
    try:
        import cvxpy as cp
    except ImportError:
        return None
    
    n = len(a)
    d = cp.Variable(n)
    t = cp.Variable()
    
    top_indices = np.argsort(c)[::-1][:top_k]
    
    constraints = [cp.sum(d) == 0, d >= -a]
    
    for k in top_indices:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min > j_max:
            constraints.append(-t <= -c[k])
            continue
        i_min = k - j_max
        i_max = k - j_min
        coeffs = a[k - i_min : k - i_max + 1][::-1]
        ad_k = coeffs @ d[i_min:i_max + 1]
        constraints.append(2 * ad_k - t <= -c[k])
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    solvers_to_try = [cp.PDLP, cp.ECOS, cp.GLPK]
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver, timeout=30)
            break
        except Exception:
            continue
    
    if d.value is None:
        return None
    return np.asarray(d.value, dtype=np.float64), float(t.value)

def coordinate_polish(a0, n, S, max_time, rng):

            best_score = cur

        peak_lag = int(np.argmax(c))
        
        # Find contributing pairs
        i_min = max(0, peak_lag - n + 1)
        i_max = min(peak_lag, n - 1)
        if i_min > i_max:
            break
        
        indices = np.arange(i_min, i_max + 1)
        partners = peak_lag - indices
        products = a[indices] * a[partners]
        
        # Find the pair with the largest product
        max_idx = np.argmax(products)
        i = indices[max_idx]
        j = partners[max_idx]
        
        if i == j:
            # Single element contribution: a[i]^2
            # Reduce a[i] and increase a random other element
            src = i
            # Pick a destination that doesn't increase the peak
            dest_candidates = np.where(a < a.mean())[0]
            if len(dest_candidates) == 0:
                break
            dest = dest_candidates[rng.integers(len(dest_candidates))]
        else:
            # Pair contribution: a[i] * a[j]
            # Transfer from the larger to a random small element
            if a[i] >= a[j]:
                src = i
                partner = j
            else:
                src = j
                partner = i

        delta = 0.01 * a[src]
        a[src] -= delta
        a[dest] += delta  # or a[partner] -= delta/2, a[other] += delta/2
        
        a = np.maximum(a, 0)
        a = a * (S / a.sum())
    
    return best, best_score

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cand = (cumsum - S) / idx
    valid = theta_cand <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a via FFT."""
    n = len(a)
    cm = c.max()
    exp_bc = np.exp(beta * (c - cm))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgrad_topk(a, c, k):
    """Subgradient: uniform weight on top-k lags."""
    n = len(a)
    k = min(k, len(c))
    top = np.argsort(c)[::-1][:k]
    w = np.zeros(2 * n - 1)
    w[top] = 1.0 / k
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def exp_weighted_subgrad(a, c, k, temperature=1.0):
    """Subgradient with exponential weighting on top-k lags."""
    n = len(a)
    k = min(k, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    top_vals = c[top_idx]
    tv_spread = top_vals[0] - top_vals[-1]
    if tv_spread < 1e-15:
        weights = np.ones(k) / k
    else:
        normalized = (top_vals - top_vals[-1]) / tv_spread
        weights = np.exp(temperature * normalized)
        weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_idx):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def safe_normalize(a, n):

    if s <= 1e-15:

    return a * (S / s)

def solve_reduced_lp(a, c, top_k, timeout=20):
    """Solve reduced LP for best descent direction. Returns (d, t) or (None, None)."""
    try:
        import cvxpy as cp
    except ImportError:
        return None, None

    n = len(a)
    d = cp.Variable(n)
    t = cp.Variable()

    top_indices = np.argsort(c)[::-1][:top_k]

    constraints = [cp.sum(d) == 0, d >= -a, d <= S_upper_bound(a)]

    for k in top_indices:
        k = int(k)
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_min > i_max:
            continue
        start_idx = k - i_max
        end_idx = k - i_min + 1
        coeffs = a[start_idx:end_idx][::-1]
        ad_k = coeffs @ d[i_min:i_max + 1]
        constraints.append(2.0 * ad_k - t <= -c[k])

    prob = cp.Problem(MyMinimize(t), constraints)

    solvers = [cp.PDLP, cp.ECOS, cp.GLPK]
    for solver in solvers:
        try:
            prob.solve(solver=solver, timeout=timeout)
            if d.value is not None:
                return np.asarray(d.value, dtype=np.float64), float(t.value)
        except Exception:
            continue

    return None, None

def spectral_shaping(a0, n, S, max_time, rng):

        P_new = P * (1.0 - strength) + P_flat * strength

        phase = np.where(np.abs(A) > 1e-30, A / np.abs(A), 1.0 + 0j)
        A_new = phase * np.sqrt(np.maximum(P_new, 1e-30))

        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new * (S / s_new)

        c = fft_convolve_self(a_new)
        score = c.max()
        if score < best_score:
            best_score = score
            best = a_new.copy()

        a = a_new

    return best, best_score


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-15:
        return np.ones(n) * S / n
    return a * (S / a_sum)


def make_periodic_init(n, rng, S, num_freqs=3):
    """Periodic initialization with multiple sinusoidal components."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 25)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.4
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a * (S / a.sum())


def make_w_shape_init(n, rng, S):
    """W-shaped initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    k = rng.integers(2, 6)
    phase = rng.random() * np.pi / k
    a = np.abs(np.sin(2 * np.pi * k * tt + phase)) + 0.05
    return a * (S / a.sum())


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    S = np.sqrt(2.0 * n)
    inits = []

    if parent is not None and len(parent) >= 64:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if p.sum() > 0:
            inits.append(("parent", p * (S / p.sum())))

    for i in range(3):
        inits.append(("anti_pal_%d" % i, make_anti_palindromic_init(n, rng, S)))
    for i in range(2):
        inits.append(("periodic_%d" % i, make_periodic_init(n, rng, S)))
    for i in range(2):
        inits.append(("wshape_%d" % i, make_w_shape_init(n, rng, S)))

    tt = np.linspace(0, 1, n, endpoint=False)
    for freq in [3, 7, 13]:
        sin_a = 1.0 + 0.35 * np.sin(2 * np.pi * freq * tt)
        sin_a = np.maximum(sin_a, 0.05)
        inits.append(("sin_%d" % freq, sin_a * (S / sin_a.sum())))

    rand_a = rng.random(n) + 0.05
    inits.append(("random", rand_a * (S / rand_a.sum())))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(10.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    # Get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    primary_n = len(parent) if parent is not None else 8192
    print("Primary length: %d, parent: %s" % (primary_n, "yes" if parent is not None else "no"))

    # === Phase 1: Quick FISTA on multiple initializations ===
    inits = generate_initializations(primary_n, rng, parent)
    usable = max(20.0, deadline - time.monotonic())
    per_init_time = usable * 0.35 / len(inits)

    print("=== Phase 1: %d inits, %.1f sec each ===" % (len(inits), per_init_time))

    for name, arr in inits:
        if time.monotonic() > deadline - 180:
            break
        n = len(arr)
        if n < 64:
            continue
        S = np.sqrt(2.0 * n)
        t0 = time.monotonic()

        seg = min(per_init_time, max(5.0, (deadline - time.monotonic()) * 0.3))
        a_f, s_f = fista_optimize(arr, n, S, seg, rng,
                                   beta_start=5.0, beta_end=200.0)
        a_f = safe_normalize(a_f, n)
        c_f = fft_convolve_self(a_f)
        score_f = c_f.max()
        if score_f < global_best:
            global_best = score_f
            global_a = a_f.copy()
            print("  Phase1 [%s]: %.6f (%.0fs)" % (name, score_f, time.monotonic() - t0))

    print("=== Phase 1 done: best=%.6f ===" % global_best)

    if global_a is None:
        global_a = np.ones(primary_n) * np.sqrt(2.0 * primary_n) / primary_n
        global_best = fft_convolve_self(global_a).max()

    # === Phase 2: LP-guided optimization on best ===
    if time.monotonic() < deadline - 300:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        alloc2 = min(500.0, max(60.0, (deadline - time.monotonic()) * 0.25))
        print("=== Phase 2: LP-guided (%.0f sec) ===" % alloc2)
        t0 = time.monotonic()

        a_lp, s_lp = lse_fista_with_lp(global_a, n, S, alloc2, rng,
                                        top_k_lp=50, lp_interval=200,
                                        beta_start=20.0, beta_end=800.0)
        a_lp = safe_normalize(a_lp, n)
        c_lp = fft_convolve_self(a_lp)
        if c_lp.max() < global_best:
            global_best = c_lp.max()
            global_a = a_lp.copy()
            print("  LP-guided: %.6f" % global_best)
        print("  Phase 2 done in %.1fs" % (time.monotonic() - t0))

    # === Phase 3: Spectral shaping ===
    if time.monotonic() < deadline - 200:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        alloc3 = min(180.0, max(30.0, (deadline - time.monotonic()) * 0.12))
        print("=== Phase 3: Spectral shaping (%.0f sec) ===" % alloc3)
        t0 = time.monotonic()

        a_sp, s_sp = spectral_shaping(global_a, n, S, alloc3, rng)
        a_sp = safe_normalize(a_sp, n)
        c_sp = fft_convolve_self(a_sp)
        if c_sp.max() < global_best:
            global_best = c_sp.max()
            global_a = a_sp.copy()
            print("  Spectral: %.6f" % global_best)
        print("  Phase 3 done in %.1fs" % (time.monotonic() - t0))

    # === Phase 4: High-beta FISTA polish ===
    if time.monotonic() < deadline - 200:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        alloc4 = min(300.0, max(30.0, (deadline - time.monotonic()) * 0.20))
        print("=== Phase 4: High-beta FISTA (%.0f sec) ===" % alloc4)
        t0 = time.monotonic()

        a_hb, s_hb = fista_optimize(global_a, n, S, alloc4, rng,
                                     beta_start=100.0, beta_end=3000.0)
        a_hb = safe_normalize(a_hb, n)
        c_hb = fft_convolve_self(a_hb)
        if c_hb.max() < global_best:
            global_best = c_hb.max()
            global_a = a_hb.copy()
            print("  High-beta: %.6f" % global_best)

        # Also try top-k subgradient variant
        rem = alloc4 - (time.monotonic() - t0)
        if rem > 30:
            a_tk, s_tk = fista_optimize(global_a, n, S, rem, rng,
                                         use_topk=True, topk_val=8,
                                         beta_start=50.0, beta_end=500.0)
            a_tk = safe_normalize(a_tk, n)
            c_tk = fft_convolve_self(a_tk)
            if c_tk.max() < global_best:
                global_best = c_tk.max()
                global_a = a_tk.copy()
                print("  TopK FISTA: %.6f" % global_best)
        print("  Phase 4 done in %.1fs" % (time.monotonic() - t0))

    # === Phase 5: Multi-restart with diverse perturbations ===
    if time.monotonic() < deadline - 100:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        num_restarts = min(12, max(3, int(remaining / 80)))
        per_restart = max(20.0, remaining * 0.55 / num_restarts)
        print("=== Phase 5: %d restarts, %.1f sec each ===" % (num_restarts, per_restart))

        for ri in range(num_restarts):
            if time.monotonic() > deadline - 60:
                break

            # Create diverse perturbation
            ptype = ri % 4
            if ptype == 0:
                noise = 0.01 * S / np.sqrt(n)
                a_try = global_a + rng.normal(0, noise, n)
            elif ptype == 1:
                # Frequency-domain rotation
                fsize = 1 << (2 * n - 1).bit_length()
                A = fft(global_a, fsize)
                angle = rng.random() * 0.1
                A_rot = A * np.exp(1j * angle * np.arange(fsize))
                a_try = np.real(ifft(A_rot))[:n]
                a_try = np.maximum(a_try, 0)
            elif ptype == 2:
                # Swap-based perturbation
                a_try = global_a.copy()
                idx1 = rng.integers(0, n)
                idx2 = rng.integers(0, n)
                swap_amt = 0.05 * a_try[idx1]
                a_try[idx1] -= swap_amt
                a_try[idx2] += swap_amt
                a_try = np.maximum(a_try, 0)
            else:
                # Anti-palindromic blend
                a_pal = (global_a + global_a[::-1]) / 2.0
                a_try = (1 - 0.1) * global_a + 0.1 * a_pal

            a_try = safe_normalize(a_try, n)
            t0 = time.monotonic()
            a_r, s_r = fista_optimize(a_try, n, S, per_restart, rng,
                                       beta_start=50.0, beta_end=1000.0)
            a_r = safe_normalize(a_r, n)
            c_r = fft_convolve_self(a_r)
            if c_r.max() < global_best:
                global_best = c_r.max()
                global_a = a_r.copy()
                print("  Restart %d: %.6f" % (ri, global_best))

    # === Phase 6: Final very-high-beta polish ===
    if time.monotonic() < deadline - 30:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        alloc6 = max(10.0, deadline - time.monotonic() - 20.0)
        print("=== Phase 6: Final polish (%.0f sec) ===" % alloc6)
        t0 = time.monotonic()

        a_fin, s_fin = fista_optimize(global_a, n, S, alloc6 * 0.6, rng,
                                       beta_start=500.0, beta_end=10000.0,
                                       perturb_prob=0.05)
        a_fin = safe_normalize(a_fin, n)
        c_fin = fft_convolve_self(a_fin)
        if c_fin.max() < global_best:
            global_best = c_fin.max()
            global_a = a_fin.copy()
            print("  Final FISTA: %.6f" % global_best)

        rem = alloc6 - (time.monotonic() - t0)
        if rem > 10:
            a_fin2, s_fin2 = fista_optimize(global_a, n, S, rem, rng,
                                             use_topk=True, topk_val=5,
                                             beta_start=200.0, beta_end=2000.0)
            a_fin2 = safe_normalize(a_fin2, n)
            c_fin2 = fft_convolve_self(a_fin2)
            if c_fin2.max() < global_best:
                global_best = c_fin2.max()
                global_a = a_fin2.copy()
                print("  Final TopK: %.6f" % global_best)

    # === Phase 7: Try slightly different length if time permits ===
    if time.monotonic() < deadline - 60:
        n_alt = primary_n + 1 if primary_n % 2 == 0 else primary_n - 1
        if n_alt >= 64 and n_alt != primary_n:
            S_alt = np.sqrt(2.0 * n_alt)
            a_alt_init = np.interp(np.linspace(0, 1, n_alt),
                                    np.linspace(0, 1, len(global_a)), global_a)
            a_alt_init = safe_normalize(a_alt_init, n_alt)
            alloc7 = max(10.0, deadline - time.monotonic() - 20.0)
            a_alt, s_alt = fista_optimize(a_alt_init, n_alt, S_alt, min(60.0, alloc7),
                                           rng, beta_start=50.0, beta_end=500.0)
            a_alt = safe_normalize(a_alt, n_alt)
            c_alt = fft_convolve_self(a_alt)
            obj_alt = 2.0 * n_alt * c_alt.max() / S_alt ** 2
            obj_cur = 2.0 * len(global_a) * global_best / (np.sqrt(2.0 * len(global_a))) ** 2
            if obj_alt < obj_cur:
                global_best = c_alt.max()
                global_a = a_alt.copy()
                print("  Alt length %d: obj=%.6f" % (n_alt, obj_alt))

    # === Verification ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Verification: %.6f (internal: %.6f), n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: rejected, using fallback")
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver2 = evaluate_sequence(final_list)
        print("  Fallback: %.6f" % (ver2 if np.isfinite(ver2) else float('inf')))
    else:
        print("=== FINAL: %.6f ===" % ver)

    return final_list
