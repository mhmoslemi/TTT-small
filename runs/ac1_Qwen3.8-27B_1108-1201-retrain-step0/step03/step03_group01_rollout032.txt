"""
Minimize 2*n*max_conv/sum^2 using adaptive LSE-FISTA + frequency-domain annealing + coordinate polishing.

Key improvements over previous approach:
1. Adaptive LSE temperature: β increases over time, focusing optimization on the worst lags
2. Frequency-domain simulated annealing: perturb |FFT| at random frequencies to escape local minima
3. Pair-swap coordinate descent: local refinement by moving mass between pairs of positions
4. Better initializations using the previous best + structured perturbations
5. Multi-start with diversity: different random seeds, different strategies per start

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(c, n):
    """Compute the objective: 2*n*max(c)/S^2 where S = sum(a)."""
    # Since we normalize S = sqrt(2n), score = max(c)
    # Actually let's just return max(c) for now, we'll compute the real score at the end
    return c.max()


def lse_fista(a_init, n, S, N, max_time, rng, beta_init=10.0, beta_rate=0.01, momentum=0.9):
    """
    FISTA with log-sum-exp smoothing and adaptive beta.
    N is the FFT size (power of 2).
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    beta = beta_init
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = beta_init + beta_rate * it
        
        # Compute convolution
        A = fft(y, N)
        c = np.real(ifft(A * A))[:2*n-1]
        c_max = c.max()
        
        if c_max < best_score:
            best_score = c_max
            best_a = y.copy()
        
        if c_max <= S * S / (2*n) * 1.5027:  # target
            break
        
        # LSE gradient
        c_shifted = c - c_max
        exp_bc = np.exp(np.minimum(beta * c_shifted, 50.0))  # clip for numerical stability
        total = exp_bc.sum()
        if total < 1e-300:
            w = np.full(2*n-1, 1.0/(2*n-1))
        else:
            w = exp_bc / total
        
        # Gradient: 2 * cross-correlation of w with y
        fsize = N
        W = fft(w, fsize)
        Y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: use 2/(L) where L is the Lipschitz constant
        # L ≈ 2 * max(w) * 2 * max(|y|) * 2 ≈ 8 * max(w) * max(y)
        # For safety, use a conservative step
        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        
        # Gradient step
        x_new = project_simplex(y - step * grad, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Renormalize
        s_y = y.sum()
        if s_y > 0:
            y = y * S / s_y
        a = x_new
        t = t_new
        
        # Periodic restart to avoid getting stuck
        if it % 5000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
    
    return best_a, best_score


def freq_annealing(a_init, n, S, N, max_time, rng, temp_init=0.1, temp_min=0.001):
    """
    Frequency-domain simulated annealing.
    Perturb |FFT(a)| at random frequencies, project back, accept/reject.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    current_score = None
    
    # Initial evaluation
    A = fft(a, N)
    c = np.real(ifft(A * A))[:2*n-1]
    current_score = c.max()
    best_score = current_score
    
    temp = temp_init
    it = 0
    n_perturb = 10  # number of frequencies to perturb per step
    
    while time.monotonic() - start < max_time:
        it += 1
        # Cool down
        temp = max(temp_min, temp_init * np.exp(-0.001 * it))
        
        # Perturb magnitude at random frequencies
        A_new = A.copy()
        freqs = rng.integers(0, N, size=n_perturb)
        for f in freqs:
            mag_change = rng.normal(0, temp * np.abs(A_new[f]))
            A_new[f] *= (1.0 + mag_change)
            # Mirror for real signal
            mirror = (N - f) % N
            A_new[mirror] = np.conj(A_new[f])
        
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new, S)
        
        # Evaluate
        A_new = fft(a_new, N)
        c_new = np.real(ifft(A_new * A_new))[:2*n-1]
        score_new = c_new.max()
        
        # Accept/reject
        delta = score_new - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(temp * S, 1e-10)):
            a = a_new
            A = A_new
            current_score = score_new
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
        
        # Update A after projection (since projection changes the signal)
        A = fft(a, N)
    
    return best_a, best_score


def pair_swap_cd(a_init, n, S, N, max_time, rng, n_trials=50):
    """
    Pair-swap coordinate descent.
    For each position i, try moving a small amount to position j.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_a = a.copy()
    
    # Compute initial convolution
    c = fft_convolve(a)
    best_score = c.max()
    
    while time.monotonic() - start < max_time:
        # Get top lags
        top_k = 20
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        
        # For each source position i (random order), try to move mass to best target
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if a[i] < S / n * 0.01:  # skip very small positions
                continue
            
            # The "cost" of position i: its contribution to top lags
            cost = 0.0
            for k in top_idx[:10]:
                j = int(k) - i
                if 0 <= j < n:
                    cost += a[j]
            
            if cost <= 0:
                continue
            
            # Find best target: minimize the "damage" at top lags
            # Try a few random targets
            best_delta = 0.0
            best_j = -1
            
            # The amount to move: try a fraction of a[i]
            delta_frac = min(0.05, 1.0 / cost)
            delta = a[i] * delta_frac
            
            # Quick check: for the top lag k*, change is 2*delta*(a_{k*-j} - a_{k*-i})
            # We want a_{k*-j} < a_{k*-i}
            
            # Try random targets

                rj = int(k_star) - j

                if change < -0.001 * c[k_star]:  # significant reduction
                    best_delta = delta
                    best_j = j
                    break
            
            if best_j >= 0:
                a[i] -= best_delta
