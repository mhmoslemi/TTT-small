Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
Uses multi-phase optimization: coarse FISTA, targeted peak reduction, fine polish.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_toppk(a, c, top_k):
    """Gradient of top-k weighted sum of convolution values."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_soft(a, c, beta):
    """Soft-max gradient."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

    """Compute 2n*max_conv/S^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

    return grad


def obj_from_maxc(a):
    """Objective = 2n * maxc / S^2, assuming normalized S."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_conv_self(a)
    return 2.0 * n * c.max() / (S * S)


def norm_to(a, S):
    """Clip to [0,1000] and rescale to sum S."""
    a = np.clip(a, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a), dtype=np.float64) * S / len(a)
    return a * (S / s)


def optimize_coarse(a, n, S, max_time, rng, beta_init=5.0, beta_max=500.0):
    """FISTA with log-sum-exp for coarse optimization."""
    a = norm_to(a, S)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.0015 * it))
        c = fft_conv_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        grad = fft_grad_soft(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (beta * 2.0 * (1.0 + 0.0005 * it))
        step = max(step, 1e-9)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
    return best_local, best_score


def optimize_topk(a, n, S, max_time, rng, top_k_base=5, top_k_max=64):
    """Targeted top-k peak reduction with adaptive step and backtracking."""
    a = norm_to(a, S)
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        k = min(top_k_max, top_k_base + it // 200)
        c = fft_conv_self(a)
        cur = c.max()
        obj = 2.0 * n * cur / S**2
        if obj < best_score:
            best_score = obj
            best_local = a.copy()
        grad = fft_grad_topk(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        # Adaptive step size
        base_step = S / (gnorm * np.sqrt(it + 1))
        step = base_step
        best_x = a.copy()
        best_c = cur
        for _ in range(8):
            x_try = project_simplex(a - step * grad, S)
            c_try = fft_conv_self(x_try)
            if c_try.max() < best_c:
                best_c = c_try.max()
                best_x = x_try
            step *= 0.6
        a = best_x
        # Occasionally add noise to escape local minima
        if it % 2000 == 0:
            noise_scale = S * 0.001 / np.sqrt(n)
            noise = rng.normal(0, noise_scale, n)
            a = project_simplex(a + noise, S)
    return best_local, best_score


def optimize_peak_shave(a, n, S, max_time, rng):
    """
    Direct peak shaving: identify the max lag, find contributing pairs,
    and transfer mass from high-product pairs to low-impact positions.
    """
    a = norm_to(a, S)
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_conv_self(a)
        cur = c.max()
        obj = 2.0 * n * cur / S**2
        if obj < best_score:
            best_score = obj
            best_local = a.copy()

        # Top peaks
        top_peaks = np.argsort(c)[::-1][:5]
        improved = False
        for k_star in top_peaks:
            k_star = int(k_star)
            # Pairs contributing to lag k_star
            j_min = max(0, k_star - n + 1)
            j_max = min(n - 1, k_star)
            if j_min > j_max:
                continue
            js = np.arange(j_min, j_max + 1)
            ps = k_star - js
            products = a[js] * a[ps]
            # Find the top contributing pair
            order = np.argsort(products)[::-1][:20]
            for oi in order:
                j = int(js[oi])
                p = int(ps[oi])
                if j == p:
                    continue
                prod = products[oi]
                if prod < 1e-10:
                    break
                # Try reducing the larger element
                if a[j] >= a[p]:
                    src = j
                else:
                    src = p
                # Find best target: position q where adding mass barely affects peaks
                # The effect on lag k is a[q]*a[k-q] (new term) + existing changes
                # Simplified: add to position with smallest a[k_star - q]
                q_ks = k_star - np.arange(n)
                valid_q = (q_ks >= 0) & (q_ks < n)
                scores = np.full(n, 1e10)
                if np.any(valid_q):
                    scores[valid_q] = a[q_ks[valid_q]]
                # Positions where k_star - q is out of range have zero effect on this lag
                scores[~valid_q] = 0.0
                scores[src] = 1e10  # don't add back to source
                q = int(np.argmin(scores))
                # Determine step size
                delta = 0.02 * a[src]
                if delta < 1e-12:
                    continue
                a_new = a.copy()
                a_new[src] -= delta
                a_new[q] += delta
                a_new = np.maximum(a_new, 0.0)
                s_new = a_new.sum()
                if s_new < 1e-15:
                    continue
                a_new = a_new * (S / s_new)
                c_new = fft_conv_self(a_new)
                if c_new.max() < cur * (1 - 1e-8):
                    a = a_new
                    cur = c_new.max()
                    improved = True
                    if 2.0 * n * cur / S**2 < best_score:
                        best_score = 2.0 * n * cur / S**2
                        best_local = a.copy()
                    break
        if not improved and it > 100:
            # Stuck: perturb
            if rng.random() < 0.3:
                idx = rng.integers(0, n)
                delta = 0.01 * S / n
                a[idx] = max(0, a[idx] - delta)
                a[rng.integers(0, n)] += delta
                a = norm_to(a, S)
    return best_local, best_score


def optimize_fine(a, n, S, max_time, rng, top_k=3):
    """Very fine optimization with tiny steps and many iterations."""
    a = norm_to(a, S)
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_conv_self(a)
        cur = c.max()
        obj = 2.0 * n * cur / S**2
        if obj < best_score:
            best_score = obj
            best_local = a.copy()
        grad = fft_grad_topk(a, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        # Very small step
        step = S * 0.01 / (gnorm * (1.0 + 0.001 * it))
        x_new = project_simplex(a - step * grad, S)
        c_new = fft_conv_self(x_new)
        if c_new.max() >= cur:
            # Backtrack
            for _ in range(5):
                step *= 0.5
                x_new = project_simplex(a - step * grad, S)
                c_new = fft_conv_self(x_new)
                if c_new.max() < cur:
                    break
            else:
                break
        a = x_new
    return best_local, best_score


def make_initializations(n, rng):
    """Create diverse asymmetric initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Geometric sequences (front-loaded)
    for r in [0.9, 0.95, 0.99, 0.999, 0.9999]:
        inits.append(np.power(r, np.arange(n, dtype=np.float64)))

    # Back-loaded geometric
    for r in [0.9, 0.95, 0.99, 0.999]:
        inits.append(np.power(r, np.arange(n, dtype=np.float64)[::-1]))

    # Exponential
    for alpha in [2, 5, 10, 20, 50]:
        inits.append(np.exp(-alpha * t))
        inits.append(np.exp(-alpha * (1 - t)))

    # Power law
    for p in [2, 4, 8, 16]:
        inits.append(np.power(t + 0.001, p))
        inits.append(np.power(1 - t + 0.001, p))

    # Step functions
    for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        arr = np.ones(n)
        arr[:int(n * frac)] = 5.0
        inits.append(arr)
        arr2 = np.ones(n)
        arr2[int(n * (1 - frac)):] = 5.0
        inits.append(arr2)

    # Two-hump asymmetric
    for pos in [0.2, 0.3, 0.7]:
        center = int(n * pos)
        width = n // 8
        arr = np.ones(n) * 0.1
        arr[max(0, center - width):min(n, center + width)] = 5.0
        inits.append(arr)
        # Single hump
        arr2 = np.ones(n) * 0.1
        arr2[:center] = 3.0
        inits.append(arr2)

    # Random asymmetric
    for _ in range(4):
        arr = rng.random(n) * np.linspace(0.1, 3.0, n)
        inits.append(arr)

    # Sawtooth
    period = n // 4
    for _ in range(period):
        pass
    saw = np.tile(np.linspace(0.1, 2.0, period), n // period + 1)[:n]
    inits.append(saw)

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv)/sum^2.
    Multi-phase: diverse init -> coarse FISTA -> top-k -> peak shave -> fine polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Try to use parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 256 and np.sum(np.clip(given, 0, 1000)) > 1.0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Decide sequence length
    if parent is not None:
        n = len(parent)
    else:
        n = 8192

    S = np.sqrt(2.0 * float(n))

    print("n=%d, S=%.2f" % (n, S))

    # Build initializations
    inits = []
    if parent is not None:
        inits.append(("parent", parent.copy()))
        # Perturbed parent
        for scale in [0.5, 1.0, 2.0]:
            pert = parent * rng.uniform(scale * 0.5, scale * 1.5, n)
            inits.append(("parent_pert_%s" % scale, pert))

    # Add diverse inits (limit to keep time manageable)
    diverse = make_initializations(n, rng)
    for i, d in enumerate(diverse[:20]):
        inits.append(("diverse_%d" % i, d.astype(np.float64)))

    n_inits = len(inits)
    print("Total initializations: %d" % n_inits)

    # Phase 1: Coarse optimization on each init
    explore_time = (deadline - time.monotonic()) * 0.4
    per_init = explore_time / max(1, n_inits)
    per_init = max(per_init, 3.0)  # minimum 3 seconds per init

    results = []
    for name, arr in inits:
        if time.monotonic() > deadline - 120:
            break
        t0 = time.monotonic()
        a_opt, score = optimize_coarse(arr, n, S, per_init, rng, beta_init=10.0, beta_max=200.0)
        a_opt = norm_to(a_opt, S)
        obj = obj_from_maxc(a_opt)
        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
            print("  [%s] obj=%.6f" % (name, obj))
        results.append((obj, a_opt))
        elapsed = time.monotonic() - t0

    print("After Phase 1 (coarse): best=%.6f" % global_best)

    if global_a is None:
        global_a = norm_to(inits[0][1], S)
        global_best = obj_from_maxc(global_a)

    # Phase 2: Top-k optimization on best
    remaining = deadline - time.monotonic() - 60
    if remaining > 10:
        seg = remaining * 0.35
        a2, s2 = optimize_topk(global_a, n, S, seg, rng, top_k_base=3, top_k_max=32)
        a2 = norm_to(a2, S)
        obj2 = obj_from_maxc(a2)
        if obj2 < global_best:
            global_best = obj2
            global_a = a2.copy()
            print("Phase 2 (topk): %.6f" % global_best)

    # Phase 3: Peak shaving
    remaining = deadline - time.monotonic() - 60
    if remaining > 10:
        seg = remaining * 0.30
        a3, s3 = optimize_peak_shave(global_a, n, S, seg, rng)
        a3 = norm_to(a3, S)
        obj3 = obj_from_maxc(a3)
        if obj3 < global_best:
            global_best = obj3
            global_a = a3.copy()
            print("Phase 3 (peak_shave): %.6f" % global_best)

    # Phase 4: Fine polish
    remaining = deadline - time.monotonic() - 30
    if remaining > 5:
        a4, s4 = optimize_fine(global_a, n, S, remaining, rng, top_k=2)
        a4 = norm_to(a4, S)
        obj4 = obj_from_maxc(a4)
        if obj4 < global_best:
            global_best = obj4
            global_a = a4.copy()
            print("Phase 4 (fine): %.6f" % global_best)

    # Phase 5: Final peak shave on best
    remaining = deadline - time.monotonic() - 15
    if remaining > 5:
        a5, s5 = optimize_peak_shave(global_a, n, S, remaining, rng)
        a5 = norm_to(a5, S)
        obj5 = obj_from_maxc(a5)
        if obj5 < global_best:
            global_best = obj5
            global_a = a5.copy()
            print("Phase 5 (final shave): %.6f" % global_best)

    # Final verification
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Internal: %.6f | Verified: %s" % (global_best, "%.6f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback verified: %.6f" % ver)

    print("Returned length %d, score %.6f" % (len(final_list), ver))
    return final_list
