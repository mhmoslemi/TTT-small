```python
    # This is equivalent to minimizing -sum(g) subject to the constraints

    # (f*g)[k] = sum_i f[i] * g[k-i] (with valid indices)

    # For efficiency, we'll use the structure: A is the "convolution matrix" of f
    # A[k] = [0...0 f[k] f[k-1] ... f[0] 0 ... 0] (shifted)

        hi = min(k, n-1)
        for j in range(lo, hi+1):

    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
```

```python
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tau = (cumsum - S) / idx
    valid = tau <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def topk_subgradient(a, c, k):
    """Subgradient using top-k active lags with uniform weights."""
    n = len(a)
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def upsample(a, n_new):
    """Upsample a to length n_new using linear interpolation."""
    n_old = len(a)
    t_old = np.linspace(0, 1, n_old)
    t_new = np.linspace(0, 1, n_new)
    return np.interp(t_new, t_old, a)

def downsample(a, n_new):
    """Downsample a to length n_new by block averaging."""
    n_old = len(a)
    nblocks = n_old // n_new
    if nblocks * n_new < n_old:
        # Handle the remainder by appending to the last block
        a_pad = np.concatenate([a, np.zeros(nblocks * n_new - n_old)])
    else:
        a_pad = a[:nblocks * n_new]
    return a_pad.reshape(n_new, nblocks).mean(axis=1)
```

    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def run_fista(a_init, S, max_time, rng, mode="lse", beta_init=5.0, beta_rate=0.004, beta_max=800.0):
    """FISTA with LSE or topk subgradient."""
    n = len(a_init)
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    a = a * (S / max(a.sum(), 1e-15))
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        if mode == "topk":
            k = min(512, 16 + it // 10)
            grad = topk_subgradient(y, c, k, power=2.0)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = S / (gnorm * (1.0 + 0.01 * np.log1p(it)))
        else:
            grad = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = 1.0 / (2.0 * beta)
        
        # Stuck detection
        if it - last_improve > 3000:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            last_improve = it
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic restart
        if it % 5000 == 0:
            decay = np.exp(-it / 30000.0)
            if decay > 0.1:
                noise_scale = 0.003 * S / np.sqrt(n) * decay
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

        M = c.max()

            best = M
            best_a = a.copy()

            direction = lp_direction(a, M)
            if direction is not None:
                direction = np.clip(direction, 0, 1000)
                dsum = direction.sum()
                if dsum > 1e-10:
                    direction = direction * (S / dsum)
                    # Blend with current
                    t_blend = 0.1  # Small blend factor
                    a_new = (1 - t_blend) * a + t_blend * direction
                    a_new = project_simplex(a_new, S)
                    c_new = fft_convolve_self(a_new)
