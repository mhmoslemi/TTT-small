    Directly chase the worst peak using exact gradient with Nesterov momentum.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = project_nn_sum(a, S)

    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)
    lr = lr_start
    start = time.monotonic()
    iter_no = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        # Exact gradient of the worst peak
        grad = compute_peak_grad(c, a)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        # Multiplicative update with backtracking
        cur_obj = obj_value(a, c)
        accepted = False
        lr_cur = lr

        for bt in range(15):
            a_trial = a * np.exp(-lr_cur * v)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial * (S / a_trial.sum())

            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)

            if obj_trial < cur_obj * (1.0 + 1e-12):
                a = a_trial
                c = c_trial
                accepted = True
                lr = lr_cur
                break
            lr_cur *= 0.5

        if not accepted:
            a = a * np.exp(-lr_cur * 0.01 * v)
            a = np.maximum(a, 1e-300)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            cur_obj = obj_value(a, c)

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
            stagnant = 0
            lr = min(lr * 1.005, lr_start * 2.0)
        else:
            stagnant += 1
            lr = max(lr * 0.95, lr_start * 0.1)

        if stagnant > 500:
            # Targeted perturbation: add noise concentrated near the peak
            k_max = np.argmax(c)
            sigma = n * 0.05
            idx = np.arange(n)
            weight = np.exp(-0.5 * ((idx - k_max) / sigma) ** 2)
            noise = np.random.randn(n) * 0.02 * S / np.sqrt(n) * weight
            noise /= max(weight.sum() / n, 1e-10)
            a = a + noise
            a = project_nn_sum(a, S)
            c = fft_convolve_self(a)
            v *= 0.3
            lr = lr_start
            stagnant = 0

        iter_no += 1
        if verbose and iter_no % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    Peak iter={iter_no}, obj={obj_value(a,c):.6f}, "
                  f"best={best_obj:.6f}, lr={lr:.6f}, t={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_polish(a0, time_budget, verbose=True):
    """
    Aggressive coordinate descent: for each coordinate, use bisection to find
    the best value that minimizes the objective.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = project_nn_sum(a, S)

    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()

    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        # Process indices in order of contribution to the worst peak
        k_max = np.argmax(c)
        # Indices that contribute most to c[k_max]
        contrib = np.zeros(n)
        lo1 = max(0, k_max - n + 1)
        hi1 = min(n - 1, k_max)
        if lo1 <= hi1:
            contrib[lo1:hi1 + 1] += a[k_max - hi1:k_max - lo1 + 1][::-1]
        lo2 = max(0, k_max)
        hi2 = min(n - 1, k_max + n - 1)
        if lo2 <= hi2:
            contrib[lo2:hi2 + 1] += a[lo2 - k_max:hi2 - k_max + 1]

        order = np.argsort(-contrib)

        improved_this_sweep = 0
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-14:
                continue

            # Try several multiplicative factors
            best_factor = 1.0
            best_local_obj = best_obj
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_norm = a * (S / s)
                c_new = fft_convolve_self(a_norm)
                obj_new = c_new.max()  # Proportional to objective since S fixed
                if obj_new < best_local_obj:
                    best_local_obj = obj_new
                    best_factor = factor

            if best_factor != 1.0:
                a[i] = orig * best_factor
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                cur_obj = obj_value(a, c)
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
                    improved_this_sweep += 1
            else:
                a[i] = orig  # Restore

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    Coord sweep={sweep}, improvements={improved_this_sweep}, "
                  f"obj={best_obj:.6f}, t={elapsed:.1f}s")

        if improved_this_sweep == 0:
            break

    return best_a, best_obj


def multi_scale_refine(a0, time_budget, verbose=True):
    """
    Coarse-to-fine refinement: downsample, optimize, upsample, refine.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()

    # Try a few downsampling factors
    for factor in [2, 3]:
        if time.monotonic() > time_budget:
            break
        n_coarse = n // factor
        if n_coarse < 64:
            continue

        # Downsample by averaging
        a_coarse = np.zeros(n_coarse)
        for i in range(n_coarse):
            a_coarse[i] = a[i * factor:(i + 1) * factor].mean()
        S_coarse = a_coarse.sum()
        a_coarse = project_nn_sum(a_coarse, S_coarse)

        # Optimize the coarse version
        coarse_time = min(30.0, (time_budget - 10.0) * 0.3)
        a_opt, obj_coarse = soft_max_anneal(
            a_coarse, coarse_time,
            T_start=max(fft_convolve_self(a_coarse).max() * 0.3, 0.5),
            T_end=0.05,
            lr_start=0.1, lr_end=0.01,
            momentum=0.8, verbose=False
        )

        # Upsample by replication and blend with original
        a_up = np.zeros(n)
        for i in range(n_coarse):
            a_up[i * factor:(i + 1) * factor] = a_opt[i]
        S_up = a_up.sum()
        a_up = project_nn_sum(a_up, S_up * (S / max(S_up, 1e-10)))

        # Blend
        t_blend = 0.5
        a_new = (1 - t_blend) * a + t_blend * a_up
        a_new = project_nn_sum(a_new, S)

        # Short refinement
        refine_time = min(20.0, (time_budget - 10.0) * 0.1)
        a_refined, obj_refined = peak_chase(a_new, refine_time, lr_start=0.02, verbose=False)

        if obj_refined < obj_value(a0):
            if verbose:
                print(f"    Multi-scale (factor={factor}): {obj_value(a0):.6f} -> {obj_refined:.6f}")
            a = a_refined

    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses multi-phase approach: diverse init -> soft-max anneal -> peak chase -> coord polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    global_best = np.inf
    global_a = None

    # --- Load given sequence ---
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw.copy()
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print(f"Given sequence: n={n_given}, obj={ver:.6f}")
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print(f"Could not load given sequence: {e}")

    if given_a is None:
        # Fallback initialization
        n = 4096
        S = np.sqrt(2.0 * n)
        i = np.arange(n)
        given_a = 1.0 + 0.3 * np.cos(2 * np.pi * 7 * i / n)
        given_a = np.maximum(given_a, 0.01)
        given_a *= S / given_a.sum()
        ver = evaluate_sequence(given_a.tolist())
        print(f"Generated init: n={n}, obj={ver:.6f}")
        global_best = ver
        global_a = given_a.copy()

    n = len(global_a)
    S = global_a.sum()

    # --- Phase 1: Quick diverse exploration (12% time) ---
    total_time = deadline - time.monotonic()
    phase1_time = 0.12 * total_time
    print(f"\nPhase 1: Diverse exploration ({phase1_time:.0f}s)")

    candidates = [('given', global_a.copy())]

    # Generate diverse candidates
    for freq in [5, 11, 13, 23, 37]:
        i = np.arange(n)
        a_c = 1.0 + 0.4 * np.cos(2 * np.pi * freq * i / n) + 0.2 * np.sin(2 * np.pi * freq * 2 * i / n)
        a_c = np.maximum(a_c, 0.01)
        a_c *= S / a_c.sum()
        candidates.append((f'cos_f{freq}', a_c))

    # Band-pass structure
    i = np.arange(n)
    a_bp = np.zeros(n)
    for f in [3, 7, 11]:
        a_bp += np.cos(2 * np.pi * f * i / n + rng.random() * 2 * np.pi)
    a_bp = 1.0 + 0.3 * a_bp
    a_bp = np.maximum(a_bp, 0.01)
    a_bp *= S / a_bp.sum()
    candidates.append(('bandpass', a_bp))

    # Random with structure
    a_rand = np.abs(rng.standard_normal(n)) + 0.1
    a_rand *= S / a_rand.sum()
    candidates.append(('random', a_rand))

    # Evaluate candidates quickly
    per_cand_time = phase1_time / len(candidates)
    for name, a_cand in candidates:
        if time.monotonic() > deadline - 120:
            break
        t0 = time.monotonic()
        if name == 'given':
            a_best, obj_best = global_a.copy(), global_best
        else:
            a_best, obj_best = soft_max_anneal(
                a_cand, per_cand_time,
                T_start=max(fft_convolve_self(a_cand).max() * 0.4, 0.5),
                T_end=0.1,
                lr_start=0.1, lr_end=0.02,
                momentum=0.8, perturb_every=500,
                verbose=False
            )
        elapsed = time.monotonic() - t0
        print(f"  {name}: obj={obj_best:.6f} ({elapsed:.1f}s)")
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

    # --- Phase 2: Main soft-max annealing (45% time) ---
    remaining = deadline - time.monotonic()
    phase2_time = 0.45 * remaining
    print(f"\nPhase 2: Main soft-max annealing ({phase2_time:.0f}s)")

    if phase2_time > 10:
        c_init = fft_convolve_self(global_a)
        T_start = c_init.max() * 0.3
        T_end = c_init.max() * 0.005

        a_best, obj_best = soft_max_anneal(
            global_a, phase2_time,
            T_start=T_start, T_end=T_end,
            lr_start=0.08, lr_end=0.003,
            momentum=0.88, perturb_every=3000,
            perturb_strength=0.04,
            verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  Phase 2 result: {obj_best:.6f}")

    # --- Phase 3: Peak chasing (25% time) ---
    remaining = deadline - time.monotonic()
    phase3_time = 0.25 * remaining
    print(f"\nPhase 3: Peak chasing ({phase3_time:.0f}s)")

    if phase3_time > 10:
        a_best, obj_best = peak_chase(
            global_a, phase3_time,
            lr_start=0.04, momentum=0.9,
            verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  Phase 3 result: {obj_best:.6f}")

    # --- Phase 4: Coordinate descent polish (13% time) ---
    remaining = deadline - time.monotonic()
    phase4_time = 0.13 * remaining
    print(f"\nPhase 4: Coordinate polish ({phase4_time:.0f}s)")

    if phase4_time > 5:
        a_best, obj_best = coordinate_polish(
            global_a, phase4_time, verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  Phase 4 result: {obj_best:.6f}")

    # --- Phase 5: Final fine-tuning with remaining time ---
    remaining = deadline - time.monotonic()
    print(f"\nPhase 5: Final refinement ({remaining:.0f}s)")

    if remaining > 10:
        # Alternate between peak chase and coordinate polish
        half = remaining / 2.0
        a_best, obj_best = peak_chase(
            global_a, half * 0.6,
            lr_start=0.015, momentum=0.92,
            verbose=False
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

        a_best, obj_best = coordinate_polish(
            global_a, half * 0.4, verbose=False
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()

        # One more peak chase
        rem2 = deadline - time.monotonic() - 5.0
        if rem2 > 5:
            a_best, obj_best = peak_chase(
                global_a, rem2,
                lr_start=0.01, momentum=0.93,
                verbose=False
            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: local_obj={global_best:.6f}, verified={ver:.6f}, n={len(final_a)}")

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.6f}")

    return final_a
