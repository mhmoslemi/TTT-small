"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Algorithm: Hybrid optimization combining:
1. LSE-smoothed FISTA with adaptive beta
2. Subgradient method with Polyak step on top-k lags
3. Targeted coordinate descent on worst-contributing pairs
4. Multi-frequency anti-palindromic initializations

Key insight: The subgradient of c[k] w.r.t. a is g[j] = 2*a[k-j] (sparse!),
enabling very efficient gradient computations for the subgradient method.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient_fft(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def subgradient_topk(a, c, k, temperature=1.0):
    """Compute smoothed subgradient using top-k lags with softmax weights.
    Uses the analytic formula: d(c[lg])/d(a[j]) = 2*a[lg-j] for valid j."""
    n = len(a)
    k = min(k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices]
    tv_max = top_vals.max()
    weights = np.exp(top_vals / max(temperature, 1e-10) - tv_max / max(temperature, 1e-10))
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for ii in range(k):
        lg = int(top_indices[ii])
        w = weights[ii]
        if w < 1e-15:
            continue
        # g[j] += 2*w*a[lg-j] for j in [max(0, lg-n+1), min(lg, n-1)]
        j_min = max(0, lg - n + 1)
        j_max = min(lg, n - 1)
        if j_min <= j_max:
            # a[lg-j] for j in [j_min, j_max] is a[lg-j_max : lg-j_min+1][::-1]
            seg = a[lg - j_max : lg - j_min + 1][::-1]
            grad[j_min:j_max+1] += 2.0 * w * seg
    return grad


def hard_subgradient(a, c, k):
    """Hard subgradient: uniform weight on top-k lags."""
    n = len(a)
    k = min(k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    w = 1.0 / k
    grad = np.zeros(n)
    for ii in range(k):
        lg = int(top_indices[ii])
        j_min = max(0, lg - n + 1)
        j_max = min(lg, n - 1)
        if j_min <= j_max:
            seg = a[lg - j_max : lg - j_min + 1][::-1]
            grad[j_min:j_max+1] += 2.0 * w * seg
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=2000.0):
    """FISTA on LSE-smoothed max-convolution with adaptive beta."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = beta_init * 2.0 * S
    stall = 0
    last_best = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stall = 0
        elif it % 100 == 0:
            if cur < last_best - 1e-12:
                last_best = cur
                stall = 0
            else:
                stall += 1

        grad = lse_gradient_fft(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / L
        f_y = fft_convolve_self(y).max()
        x_new = project_simplex(y - step * grad, S)

        # Backtracking
        for bt in range(4):
            c_cand = fft_convolve_self(x_new)
            f_cand = c_cand.max()
            if f_cand <= f_y + 1e-10:
                break
            step *= 0.5
            x_new = project_simplex(y - step * grad, S)

        if f_y > f_cand * 1.001:
            L *= 1.5
        elif it % 50 == 0:
            L = max(L * 0.97, beta * S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if stall > 50 and rng.random() < 0.2:
            noise = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0
            stall = 0

    return best_local, best_score

        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    temp = 5.0

        it += 1
        k = min(k_max, k_init + it // 300)

            best_score = cur

        grad = subgradient_topk(y, c, k, temperature=temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak step
        gap = max(cur - f_lb, 1e-10)
        step = gap / (gnorm * gnorm + 1e-20)
        # Cap step to avoid instability
        step = min(step, 2.0 * S / (n * gnorm + 1e-10))

        x_new = project_simplex(y - step * grad, S)

        # Check improvement
        if it % 5 == 0:
            c_new = fft_convolve_self(x_new)
            if c_new.max() > cur * 1.05:
                step *= 0.5
                x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic restart
        if it % 2000 == 0 and rng.random() < 0.1:
            noise = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0

    return best_local, best_score


def run_coord_descent(a_init, n, S, max_time, rng, n_top=5, n_pairs=20):
    """Coordinate descent: reduce worst-contributing pairs in top lags."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    scale = 0.01 * S / n

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Find top lags
        top_lags = np.argsort(c)[::-1][:n_top]

        # For each top lag, find worst pairs and adjust
        improved = False
        for lag in top_lags:
            lag = int(lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue
            indices = np.arange(i_min, i_max + 1)
            partner = lag - indices
            products = a[indices] * a[partner]
            
            # Get top contributing pairs
            top_pair_idx = np.argsort(products)[::-1][:n_pairs]
            
            for pi in top_pair_idx:
                if time.monotonic() - start > max_time:
                    break
                idx = indices[pi]
                pidx = partner[pi]
                
                # Reduce the larger of the pair
                if a[idx] >= a[pidx]:
                    reduce_idx = idx
                else:
                    reduce_idx = pidx
                
                # Find a small element to increase
                median_val = np.median(a)
                small_mask = a < median_val * 0.5
                if not np.any(small_mask):
                    small_mask = a < median_val
                candidates = np.where(small_mask)[0]
                if len(candidates) == 0:
                    candidates = np.where(a == a.min())[0]
                add_idx = candidates[rng.integers(len(candidates))]
                
                if reduce_idx == add_idx:
                    continue
                
                # Try the perturbation
                a_trial = a.copy()
                delta = scale * min(a[reduce_idx], S / n)
                a_trial[reduce_idx] -= delta
                a_trial[add_idx] += delta
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial * (S / a_trial.sum())
                
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur:
                    a = a_trial
                    cur = c_trial.max()
                    improved = True
                    if cur < best_score:
                        best_score = cur
                        best_local = a.copy()

    return best_local, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, eta_init=0.1, k=10):
    """Mirror descent (multiplicative weights) with top-k subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        eta = eta_init / (1.0 + 0.005 * it)
        grad = hard_subgradient(a, c, k)
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break

        log_update = -eta * grad / gmax
        log_update = np.clip(log_update, -3.0, 3.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_new * (S / s_new)

    return best_local, best_score


def make_anti_palindromic_cos(n, S, alpha=0.6, freq=1):
    """Anti-palindromic with cosine perturbation."""
    half = n // 2
    mu = 2.0 * S / n
    # b[i] is anti-palindromic: b[i] = -b[n-1-i]
    # Use: b[i] = alpha * sin(2*pi*freq*i/n)  (this is anti-palindromic for integer freq)
    tt = np.arange(n, dtype=np.float64)
    b = alpha * np.sin(2.0 * np.pi * freq * tt / n)
    a = mu / 2.0 * (1.0 + b)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())
    return a


def make_multi_freq_init(n, S, rng, n_freqs=3):
    """Multi-frequency anti-palindromic initialization."""
    mu = 2.0 * S / n
    tt = np.arange(n, dtype=np.float64)
    b = np.zeros(n)
    for _ in range(n_freqs):
        freq = rng.integers(1, 10)
        amp = rng.random() * 0.3
        phase = rng.random() * np.pi
        # Use sin to maintain anti-palindromic property
        b += amp * np.sin(2.0 * np.pi * freq * tt / n + phase)
    a = mu / 2.0 * (1.0 + b)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())
    return a


def make_bump_init(n, S, rng):
    """Gaussian bump initialization (two bumps for anti-palindromic)."""
    mu = 2.0 * S / n
    center1 = n * (0.2 + 0.1 * rng.random())
    center2 = n - center1
    width = n * (0.05 + 0.05 * rng.random())
    tt = np.arange(n, dtype=np.float64)
    b = np.exp(-0.5 * ((tt - center1) / width) ** 2) - np.exp(-0.5 * ((tt - center2) / width) ** 2)
    b_max = np.abs(b).max()
    if b_max > 0:
        b = b / b_max * 0.8
    a = mu / 2.0 * (1.0 + b)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())
    return a


def safe_normalize(a, n):
    """Clamp and normalize to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a):
    """Compute the actual objective value."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(1.0, budget_s - 20.0)

    global_best_obj = np.inf
    global_a = None

    # Get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    n = 8192
    S = np.sqrt(2.0 * n)

    # Normalize parent
    if parent is not None:
        if len(parent) != n:
            # Resample to n
            tt_old = np.linspace(0, 1, len(parent))
            tt_new = np.linspace(0, 1, n)
            parent = np.interp(tt_new, tt_old, parent)
            parent = np.maximum(parent, 0.0)
        parent = safe_normalize(parent, n)
        print("Parent loaded, n=%d, initial obj=%.6f" % (n, compute_objective(parent)))

    def update_best(a, tag):
        nonlocal global_best_obj, global_a
        a = safe_normalize(a, len(a))
        obj = compute_objective(a)
        if obj < global_best_obj:
            global_best_obj = obj
            global_a = a.copy()
            print("  %s: obj=%.6f (elapsed %.0fs)" % (tag, obj, time.monotonic() - (start_total)))
            return True
        return False

    start_total = time.monotonic()

    # === Generate initializations ===
    inits = []
    if parent is not None:
        inits.append(("parent", parent.copy()))
    
    for i in range(4):
        freq = rng.integers(1, 8)
        alpha = 0.4 + 0.3 * rng.random()
        inits.append(("apal_cos_%d" % i, make_anti_palindromic_cos(n, S, alpha, freq)))
    
    for i in range(3):
        inits.append(("mft_%d" % i, make_multi_freq_init(n, S, rng, n_freqs=rng.integers(2, 5))))
    
    for i in range(2):
        inits.append(("bump_%d" % i, make_bump_init(n, S, rng)))
    
    # Random
    rand_a = rng.random(n) + 0.1
    inits.append(("rand", rand_a * (S / rand_a.sum())))
    
    # Sparse-ish
    sparse = np.zeros(n)
    n_spikes = rng.integers(4, 16)
    spike_positions = rng.choice(n, n_spikes, replace=False)
    for pos in spike_positions:
        spread = rng.integers(5, 50)
        start = max(0, pos - spread)
        end = min(n, pos + spread + 1)
        tt = np.arange(start, end) - pos
        sparse[start:end] += np.exp(-0.5 * (tt / (spread * 0.3)) ** 2)
    sparse = np.maximum(sparse, 0.0)
    inits.append(("sparse", sparse * (S / sparse.sum())))

    print("=== Phase 1: LSE-FISTA on %d initializations ===" % len(inits))
    
    # Phase 1: FISTA on each initialization
    phase1_time = min(400.0, (deadline - time.monotonic()) * 0.25)
    per_init = max(5.0, phase1_time / len(inits))
    
    best_after_p1 = None
    best_p1_score = np.inf
    
    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break
        t0 = time.monotonic()
        a_f, s_f = run_fista(arr, n, S, per_init, rng, beta_init=10.0, beta_max=1000.0)
        a_f = safe_normalize(a_f, n)
        obj_f = compute_objective(a_f)
        if obj_f < best_p1_score:
            best_p1_score = obj_f
            best_after_p1 = a_f.copy()
        update_best(a_f, "FISTAp1[%s]" % name)
        print("  Phase1 %s: obj=%.6f (%.1fs)" % (name, obj_f, time.monotonic() - t0))

    if best_after_p1 is None:
        best_after_p1 = parent if parent is not None else make_anti_palindromic_cos(n, S, 0.6, 1)
        global_a = safe_normalize(best_after_p1, n)
        global_best_obj = compute_objective(global_a)
    
    print("=== After Phase 1: best obj=%.6f ===" % global_best_obj)

    # === Phase 2: Subgradient with Polyak step ===
    print("=== Phase 2: Subgradient method ===")
    phase2_time = min(800.0, (deadline - time.monotonic()) * 0.35)

        a_sg, s_sg = run_subgradient(global_a, n, S, seg, rng, k_init=k_init, k_max=k_max)
        a_sg = safe_normalize(a_sg, n)

        a_cd = safe_normalize(a_cd, n)

    # 4b: Mirror descent
