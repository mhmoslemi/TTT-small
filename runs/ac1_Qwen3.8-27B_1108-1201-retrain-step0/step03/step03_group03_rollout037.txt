"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Strategy: Smooth parametric initialization → FISTA (adaptive beta) → 
Top-pair coordinate descent → Active-set refinement.
Uses n=16384 primary, with FFT-accelerated convolutions.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize_scalar


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_subgradient(a, c, top_k):
    """Subgradient using top-k lags uniformly weighted."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    w_full = np.zeros(2 * n - 1)
    w_full[top_indices] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_topk_subgradient(a, c, top_k, temperature=1.0):
    """Subgradient using top-k lags with softmax weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices]
    tv_max = top_vals.max()
    weights = np.exp(top_vals / max(temperature, 1e-10) - tv_max / max(temperature, 1e-10))
    weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def cosine_init(n, S, freq=1.0, amp=1.0, phase=0.0):
    """Cosine-shaped initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    a = 1.0 + amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def raised_cosine_init(n, S, freq=1.0):
    """Raised cosine (always non-negative)."""
    tt = np.arange(n, dtype=np.float64) / n
    a = 0.5 * (1 + np.cos(2 * np.pi * freq * tt))
    a = a + 0.05
    a = a * (S / a.sum())
    return a


def anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def gaussian_bump_init(n, S, center=0.5, width=0.3):
    """Gaussian bump initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.exp(-0.5 * ((tt - center) / width) ** 2)
    a = a + 0.05
    a = a * (S / a.sum())
    return a


def mixed_smooth_init(n, rng, S):
    """Mixture of smooth functions."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(3):
        freq = rng.integers(1, 8)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.3
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = a + 0.1
    a = a * (S / a.sum())
    return a


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=500.0,
              beta_growth=0.005, use_lse=True):
    """FISTA with adaptive beta and backtracking."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = 2.0  # Initial Lipschitz estimate (will be updated)
    stall_count = 0
    last_improve = np.inf
    no_improve_thresh = 0.001

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init + beta_growth * it)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            stall_count = 0
            last_improve = cur
        elif it % 30 == 0:
            if cur < last_improve - no_improve_thresh:
                last_improve = cur
                stall_count = 0
            else:
                stall_count += 1

        # Compute gradient
        if use_lse:
            grad = lse_gradient(y, c, beta)
        else:
            top_k = min(50, 5 + it // 100)
            grad = topk_subgradient(y, c, top_k)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step with backtracking
        step = 1.0 / L
        x_candidate = project_simplex(y - step * grad, S)

        # Simple backtracking (every few iterations to save time)
        if it % 10 == 0:
            c_cand = fft_convolve_self(x_candidate)
            if c_cand.max() > cur * 1.001:
                L *= 1.3
                step = 1.0 / L
                x_candidate = project_simplex(y - step * grad, S)
            elif it % 50 == 0 and it > 100:
                L = max(L * 0.97, 1.0)

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_candidate + ((t_momentum - 1.0) / t_new) * (x_candidate - a)
        y = np.maximum(y, 0.0)
        a = x_candidate
        t_momentum = t_new

        # Perturbation to escape plateaus
        if stall_count > 50 and rng.random() < 0.2:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_momentum = 1.0
            stall_count = 0
            L = max(L * 0.8, 1.0)

    return best_local, best_score


def run_top_pair_cd(a_init, n, S, max_time, rng, top_lags=10, top_pairs_per_lag=10,
                    max_pairs=200, ternary_iters=40):
    """Top-pair coordinate descent: optimize pairs contributing to top lags."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Find top lags
        top_lag_indices = np.argsort(c)[::-1][:top_lags]

        # Collect pairs contributing to top lags
        pairs = set()
        for lag in top_lag_indices:
            lag = int(lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue
            indices = np.arange(i_min, i_max + 1)
            partners = lag - indices
            products = a[indices] * a[partners]
            # Get top pairs by product
            top_p = np.argsort(products)[::-1][:top_pairs_per_lag]
            for pi in top_p:
                i1 = int(indices[pi])
                i2 = int(partners[pi])
                if i1 != i2:
                    pair = (min(i1, i2), max(i1, i2))
                    pairs.add(pair)

        # Limit number of pairs
        pair_list = list(pairs)
        if len(pair_list) > max_pairs:
            rng.shuffle(pair_list)
            pair_list = pair_list[:max_pairs]

        # Optimize each pair
        improved = False
        for (i1, i2) in pair_list:
            C = a[i1] + a[i2]
            if C < 1e-15:
                continue
            a1_old = a[i1]

            # Define objective: max of b_k over top lags (and a few others)
            # b_k(t) for t in [0, C] where a[i1] = t, a[i2] = C - t
            # Change in b_k: 2*(t-a1_old)*a[k-i1] + 2*(a1_old-t)*a[k-i2]
            #   + (t-a1_old)^2*[k==2*i1] + 2*(t-a1_old)*(a1_old-t)*[k==i1+i2]
            #   + (a1_old-t)^2*[k==2*i2]

            # Pre-compute coefficients for affected lags
            affected_lags = set()
            for lag in top_lag_indices:
                lag = int(lag)
                if 0 <= lag < len(c):
                    affected_lags.add(lag)
            # Also include lags where the quadratic terms matter
            if 2 * i1 < 2 * n - 1:
                affected_lags.add(2 * i1)
            if 2 * i2 < 2 * n - 1:
                affected_lags.add(2 * i2)
            if i1 + i2 < 2 * n - 1:
                affected_lags.add(i1 + i2)

            affected_lags = [l for l in affected_lags if 0 <= l < len(c)]

            # Precompute base values and coefficients
            base_vals = np.array([c[l] for l in affected_lags])
            # alpha_l = 2*a[l-i1] - 2*a[l-i2] (with validity)
            alphas = np.zeros(len(affected_lags))
            betas = np.zeros(len(affected_lags))
            for idx, l in enumerate(affected_lags):
                a_ki1 = a[l - i1] if 0 <= l - i1 < n else 0.0
                a_ki2 = a[l - i2] if 0 <= l - i2 < n else 0.0
                alphas[idx] = 2.0 * a_ki1 - 2.0 * a_ki2
                beta_k = 0.0
                if l == 2 * i1:
                    beta_k += 1.0
                if l == i1 + i2:
                    beta_k += -2.0
                if l == 2 * i2:
                    beta_k += 1.0
                betas[idx] = beta_k

            # Minimize max(base + alpha*d + beta*d^2) for d in [-a1_old, C - a1_old]
            lo = 0.0
            hi = C

            def obj_1d(t):
                d = t - a1_old
                vals = base_vals + alphas * d + betas * d * d
                return vals.max()

            # Ternary search
            for _ in range(ternary_iters):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if obj_1d(m1) < obj_1d(m2):
                    hi = m2
                else:
                    lo = m1

            t_opt = (lo + hi) / 2.0
            # Check boundary
            t_opt = min(max(t_opt, 0.0), C)

            # Apply if improved
            old_max = obj_1d(a1_old)
            new_max = obj_1d(t_opt)
            if new_max < old_max - 1e-15:
                a[i1] = t_opt
                a[i2] = C - t_opt
                improved = True

        if it % 50 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() < best_score:
                best_score = c_check.max()
                best_local = a.copy()

        # Periodic full re-evaluation
        if it % 100 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() < best_score:
                best_score = c_check.max()
                best_local = a.copy()

        # Noise injection to escape local minima
        if it % 200 == 0 and rng.random() < 0.3:
            noise_scale = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_local, best_score


def run_active_set_refine(a_init, n, S, max_time, rng, top_k_start=5, top_k_end=100):
    """Active-set subgradient with increasing top_k and momentum."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step_size = S / (n * 5.0)

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / max(1, int(max_time * 10)))
        top_k = int(top_k_start + (top_k_end - top_k_start) * progress)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Temperature for softmax weighting
        temp = max(0.5, 10.0 * (1.0 - progress))
        grad = softmax_topk_subgradient(y, c, top_k, temperature=temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step
        step = 2.0 * S / (n * max(gnorm / (S / n), 0.1) * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad, S)

        # Backtrack check (periodic)
        if it % 20 == 0:
            c_new = fft_convolve_self(x_new)
            if c_new.max() < best_score:
                best_score = c_new.max()
                best_local = x_new.copy()
            if c_new.max() > cur * 1.02:
                step *= 0.7

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Noise
        if it % 1000 == 0 and rng.random() < 0.1:
            noise = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_momentum = 1.0

    return best_local, best_score

    inits.append(("cos2", cosine_init(primary_n, S, freq=2, amp=0.4)))
    inits.append(("cos3", cosine_init(primary_n, S, freq=3, amp=0.3)))
    inits.append(("rcos1", raised_cosine_init(primary_n, S, freq=1)))
    inits.append(("rcos2", raised_cosine_init(primary_n, S, freq=2)))
    inits.append(("gauss_center", gaussian_bump_init(primary_n, S, 0.5, 0.4)))
    inits.append(("gauss_wide", gaussian_bump_init(primary_n, S, 0.5, 0.6)))

    for i in range(3):
        inits.append(("anti_pal_%d" % i, anti_palindromic_init(primary_n, rng, S)))
    for i in range(3):
        inits.append(("mixed_%d" % i, mixed_smooth_init(primary_n, rng, S)))

    # Uniform
    inits.append(("uniform", np.ones(primary_n) * (S / primary_n)))

    n_inits = len(inits)
    print("Number of initializations: %d" % n_inits)

    # === Phase 1: FISTA on each initialization (60% of time) ===
    usable_time = max(30.0, deadline - time.monotonic())
    phase1_time = usable_time * 0.60
    per_init_time = phase1_time / n_inits

    print("=== Phase 1: FISTA, %.1f sec per init ===" % per_init_time)

    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break
        n = len(arr)
        S_n = np.sqrt(2.0 * n)
        arr = np.clip(arr, 0, 1000)
        if arr.sum() < 1e-10:
            continue
        arr = arr * (S_n / arr.sum())

        t0 = time.monotonic()
        remaining = max(5.0, per_init_time - (time.monotonic() - t0))

        # FISTA with medium beta
        a_f, s_f = run_fista(arr, n, S_n, remaining * 0.7, rng,
                             beta_init=5.0, beta_max=200.0, beta_growth=0.003)
        a_f = np.clip(a_f, 0, 1000)
        obj_f = safe_compute_objective(a_f)
        if obj_f < global_best:
            global_best = obj_f
            global_a = a_f.copy()
            print("  FISTA [%s]: obj=%.6f (elapsed %.0fs)" %
                  (name, global_best, time.monotonic() - t0))

        # High-beta polish
        rem = per_init_time - (time.monotonic() - t0)
        if rem > 5:
            a_p, s_p = run_fista(a_f, n, S_n, rem, rng,
                                 beta_init=50.0, beta_max=800.0, beta_growth=0.01)
            a_p = np.clip(a_p, 0, 1000)
            obj_p = safe_compute_objective(a_p)
            if obj_p < global_best:
                global_best = obj_p
                global_a = a_p.copy()
                print("  FISTA-hi [%s]: obj=%.6f" % (name, global_best))

    print("=== After Phase 1: best=%s, n=%d ===" %
          ("%.6f" % global_best if np.isfinite(global_best) else "inf",
           len(global_a) if global_a is not None else 0))

    if global_a is None:
        global_a = cosine_init(primary_n, S, freq=1, amp=0.5)
        global_best = safe_compute_objective(global_a)

    # === Phase 2: Top-pair coordinate descent (25% of time) ===
    remaining_total = deadline - time.monotonic()
    if remaining_total > 60:
        n = len(global_a)
        S_n = np.sqrt(2.0 * n)
        phase2_time = min(remaining_total * 0.35, 600.0)
        print("=== Phase 2: Top-pair CD (%.0f sec) ===" % phase2_time)

        # Split into two runs with different parameters
        half_time = phase2_time * 0.5

        a_cd1, s_cd1 = run_top_pair_cd(global_a, n, S_n, half_time, rng,
                                       top_lags=10, top_pairs_per_lag=10,
                                       max_pairs=300, ternary_iters=35)
        obj_cd1 = safe_compute_objective(a_cd1)
        if obj_cd1 < global_best:
            global_best = obj_cd1
            global_a = a_cd1.copy()
            print("  CD1: obj=%.6f" % global_best)

        half_time2 = phase2_time - (time.monotonic() - (deadline - remaining_total)) 
        half_time2 = max(10.0, phase2_time * 0.5 - (time.monotonic() - (deadline - remaining_total)))
        # Just use remaining time
        elapsed_p2 = (deadline - remaining_total) - time.monotonic() + phase2_time
        # Simpler: just use half
        a_cd2, s_cd2 = run_top_pair_cd(global_a, n, S_n, phase2_time * 0.4, rng,
                                       top_lags=20, top_pairs_per_lag=5,
                                       max_pairs=500, ternary_iters=30)
        obj_cd2 = safe_compute_objective(a_cd2)
        if obj_cd2 < global_best:
            global_best = obj_cd2
            global_a = a_cd2.copy()
            print("  CD2: obj=%.6f" % global_best)

    # === Phase 3: Active-set refinement (10% of time) ===
    remaining_total = deadline - time.monotonic()
    if remaining_total > 30:
        n = len(global_a)
        S_n = np.sqrt(2.0 * n)
        phase3_time = min(remaining_total * 0.30, 300.0)
        print("=== Phase 3: Active-set refine (%.0f sec) ===" % phase3_time)

        a_as, s_as = run_active_set_refine(global_a, n, S_n, phase3_time, rng,
                                           top_k_start=10, top_k_end=150)
        obj_as = safe_compute_objective(a_as)
        if obj_as < global_best:
            global_best = obj_as
            global_a = a_as.copy()
            print("  ActiveSet: obj=%.6f" % global_best)

    # === Phase 4: Final high-beta FISTA polish ===
    remaining_total = deadline - time.monotonic()
    if remaining_total > 20:
        n = len(global_a)
        S_n = np.sqrt(2.0 * n)
        print("=== Phase 4: Final FISTA polish (%.0f sec) ===" % remaining_total)

        a_final, s_final = run_fista(global_a, n, S_n, remaining_total * 0.8, rng,
                                     beta_init=100.0, beta_max=2000.0, beta_growth=0.02)
        obj_final = safe_compute_objective(a_final)
        if obj_final < global_best:
            global_best = obj_final
            global_a = a_final.copy()
            print("  Final FISTA: obj=%.6f" % global_best)

    # === Phase 5: One more CD pass with very tight parameters ===
    remaining_total = deadline - time.monotonic()
    if remaining_total > 30:
        n = len(global_a)
        S_n = np.sqrt(2.0 * n)
        print("=== Phase 5: Final CD (%.0f sec) ===" % remaining_total)

        a_cd_final, _ = run_top_pair_cd(global_a, n, S_n, remaining_total * 0.7, rng,
                                        top_lags=30, top_pairs_per_lag=5,
                                        max_pairs=1000, ternary_iters=40)
        obj_cd_final = safe_compute_objective(a_cd_final)
        if obj_cd_final < global_best:
            global_best = obj_cd_final
            global_a = a_cd_final.copy()
            print("  Final CD: obj=%.6f" % global_best)

    # === Verification ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Verification: %.6f (internal: %.6f), length=%d ===" %
          (ver if np.isfinite(ver) else float('inf'),
           global_best, len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: evaluate_sequence rejected, using fallback")
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback: %.6f" % ver)

    return final_list
