def project_simplex(v,S):
    v = np.asarray(v, dtype=float).copy()
    n = v.size
    if n == 0: return v
    w = np.sort(v)[::-1]
    cum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=float)
    theta_cand = (cum - S) / idx
    ok = theta_cand < w  # or <=
    if not ok.any():
       rho = n
       theta = (cum[rho-1]-S)/rho
    else:
       rho = int(np.where(ok)[0][-1]) + 1
       theta = (cum[rho-1]-S)/rho
    out = v - theta
    np.maximum(out,0,out)
    # Due to numerical, enforce sum S
    s = out.sum()
    if s <= 1e-14:
       out.fill(S/n)
    else:
       out *= S/s
    return out
```

"""
Minimize 2*n*max(a*a)/sum(a)^2 over non-negative sequences.

Key improvements over the previous approach:
1. Use the correct gradient for the full self-convolution a*a, not autocorrelation.
2. Combine annealed log-sum-exp smoothing with a target-excess penalty that tracks
   the current best and pushes the maximum coefficient below it.
3. Use FISTA-style acceleration with careful restarts, multi-start initializations,
   and optional higher-resolution extension of the best candidate.
4. Keep the final length bounded (<=16384) to keep the evaluator's direct convolve
   reasonably fast.
"""
import numpy as np
import time
import math

try:
    from scipy.fft import fft, ifft
except Exception:
    from numpy.fft import fft, ifft

_ENV_CACHE = {}


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_sum(a, n):
    v = np.asarray(a, dtype=np.float64).reshape(-1)
    if v.size != n:
        v = adapt_length(v, n)
    v = np.nan_to_num(v, nan=0.0, posinf=1000.0, neginf=0.0)
    v = np.clip(v, 0.0, 1000.0)
    S = math.sqrt(2.0 * n)
    s = float(v.sum())
    if s <= 1e-15:
        return np.full(n, S / n, dtype=np.float64)
    v *= (S / s)
    return v


def clip_rescale(v, S):
    v = np.asarray(v, dtype=np.float64)
    out = np.maximum(v, 0.0)
    s = float(out.sum())
    n = out.size
    if n == 0:
        return out
    if s <= 1e-14:
        out.fill(S / n)
    else:
        out *= (S / s)
    return out


def project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cum - S) / idx
    ok = theta <= w
    if not np.any(ok):
        rho = n
    else:
        rho = int(np.where(ok)[0][-1]) + 1
    theta = (cum[rho - 1] - S) / rho
    out = v - theta
    out = np.maximum(out, 0.0)
    s = float(out.sum())
    if s <= 1e-14:
        out.fill(S / n)
    else:
        out *= (S / s)
    return out


def adapt_length(seq, n):
    s = np.asarray(seq, dtype=np.float64).reshape(-1)
    if s.size == 0:
        s = np.ones(n, dtype=np.float64)
    s = np.nan_to_num(s, nan=0.0, posinf=1000.0, neginf=0.0)
    s = np.clip(s, 0.0, 1000.0)
    if s.size == n:
        return normalize_sum(s, n)
    if s.size == 1:
        out = np.full(n, s[0], dtype=np.float64)
    else:
        old = np.linspace(0.0, 1.0, s.size)
        new = np.linspace(0.0, 1.0, n)
        out = np.interp(new, old, s)
    return normalize_sum(out, n)


def read_given_sequence():
    try:
        seq = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if seq.size == 0:
        return None
    seq = np.nan_to_num(seq, nan=0.0, posinf=1000.0, neginf=0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    return seq


def get_env(n):
    if n in _ENV_CACHE:
        return _ENV_CACHE[n]
    m = 2 * n - 1
    M = m - 1
    L = next_pow2(max(1, 3 * n - 2))
    k = np.arange(L, dtype=np.float64)
    E = np.exp((-2j * math.pi * M / float(L)) * k)
    env = {
        "n": n,
        "m": m,
        "M": M,
        "L": L,
        "E": E,
        "a_buf": np.zeros(L, dtype=np.float64),
        "w_buf": np.zeros(L, dtype=np.float64),
    }
    if len(_ENV_CACHE) >= 3:
        _ENV_CACHE.clear()
    _ENV_CACHE[n] = env
    return env


def fft_a(env, y):
    n = env["n"]
    buf = env["a_buf"]
    buf.fill(0.0)
    buf[:n] = y
    return fft(buf)


def conv_from_A(env, A):
    A2 = A * A
    c = ifft(A2).real[:env["m"]]
    return c


def fft_grad_a(env, A, weights):
    n = env["n"]
    M = env["M"]
    m = env["m"]
    wbuf = env["w_buf"]
    wbuf.fill(0.0)
    wbuf[:m] = weights
    W = fft(wbuf)
    Y = env["E"] * A
    Y *= W
    T = ifft(Y).real
    return 2.0 * T[M - n + 1:M + 1][::-1]


def objective_fft(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n < 1:
        return math.inf
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return math.inf
    L = next_pow2(2 * n - 1)
    A = fft(a, L)
    c = ifft(A * A).real[:2 * n - 1]
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return math.inf
    return float(2.0 * n * maxc / (s * s))


def make_weights_lse(c, beta):
    m = c.size
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        e = np.zeros(m, dtype=np.float64)
        e[0] = 1.0
        return e
    if beta <= 0.0 or not np.isfinite(beta):
        return np.full(m, 1.0 / m, dtype=np.float64)
    e = np.exp(beta * (c - cmax))
    se = float(np.sum(e))
    if se <= 1e-300:
        e.fill(0.0)
        idx = int(np.argmax(c))
        e[idx] = 1.0
    else:
        e /= se
    return e


def make_weights_excess(c, T, power):
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= T:
        return make_weights_lse(c, 10.0)
    e = c - T
    np.maximum(e, 0.0, out=e)
    if power == 2:
        e *= e
    elif power != 1:
        np.power(e, power, out=e)
    se = float(np.sum(e))
    if se <= 1e-300:
        e.fill(0.0)
        idx = int(np.argmax(c))
        e[idx] = 1.0
    else:
        e /= se
    return e


def run_one(a0, n, limit, seed=0, mode="anneal", beta_init=5.0, beta_rate=0.001,
            beta_stop=80.0, beta_max=5000.0, target_gap=0.004, target_lower=1.02,
            excess_power=2.0, step_lse_factor=0.5, step_target_factor=0.12,
            step_exact_factor=0.02, momentum=True, exact_projection=True,
            restart_gap=0.02, restart_iters=1500, noise_scale=0.01, noise_interval=5000):
    n = int(n)
    S = math.sqrt(2.0 * n)
    a = normalize_sum(a0, n)
    limit = float(max(0.0, limit))
    start = time.monotonic()
    deadline = start + limit

    env = get_env(n)
    A = fft_a(env, a)
    c = conv_from_A(env, A)
    best_val = float(np.max(c))
    best_a = a.copy()
    y = a.copy()
    t = 1.0
    prev_cmax = best_val
    last_best_it = 0
    it = 0
    rng = np.random.default_rng(seed)
    a_norm = S / math.sqrt(n)

    while time.monotonic() < deadline:
        it += 1
        A = fft_a(env, y)
        c = conv_from_A(env, A)
        cmax = float(np.max(c))

        if cmax < best_val:
            best_val = cmax
            best_a = y.copy()
            last_best_it = it

        if prev_cmax < math.inf and cmax > prev_cmax + restart_gap and it - last_best_it > 30:
            y = best_a.copy()
            t = 1.0
            prev_cmax = cmax
            continue

        if mode == "exact":
            idx = int(np.argmax(c))
            weights = np.zeros(c.size, dtype=np.float64)
            weights[idx] = 1.0
            use_target = True
            step_cap = 0.02
        elif mode == "lse":
            beta = min(beta_max, beta_init * math.exp(beta_rate * it))
            weights = make_weights_lse(c, beta)
            use_target = False
            step_cap = 1.0 / (2.0 * beta + 1e-12)
        elif mode == "target":
            T = target_lower
            if best_val < math.inf:
                T = max(target_lower, best_val - target_gap)
            if cmax > T:
                weights = make_weights_excess(c, T, excess_power)
                use_target = True
                step_cap = 0.25
            else:
                weights = make_weights_lse(c, 20.0)
                use_target = False
                step_cap = 0.2
        else:
            beta = min(beta_max, beta_init * math.exp(beta_rate * it))
            if beta >= beta_stop:
                T = target_lower
                if best_val < math.inf:
                    T = max(target_lower, best_val - target_gap)
                if cmax > T:
                    weights = make_weights_excess(c, T, excess_power)
                    use_target = True
                    step_cap = 0.25
                else:
                    beta_fb = min(beta, 100.0)
                    weights = make_weights_lse(c, beta_fb)
                    use_target = False
                    step_cap = 1.0 / (2.0 * beta_fb + 1e-12)
            else:
                weights = make_weights_lse(c, beta)
                use_target = False
                step_cap = 1.0 / (2.0 * beta + 1e-12)

        grad = fft_grad_a(env, A, weights)
        gnorm = float(np.linalg.norm(grad))

        if gnorm < 1e-13:
            if best_a is not None:
                y = best_a.copy()
                if noise_scale > 0.0:
                    y = clip_rescale(y + rng.normal(0.0, noise_scale * a_norm, n), S)
                t = 1.0
            break

        if mode == "exact":
            step = step_exact_factor * a_norm / gnorm
            if step > step_cap:
                step = step_cap
            if step < 1e-7:
                step = 1e-7
        elif use_target:
            step = step_target_factor * a_norm / gnorm
            if step > step_cap:
                step = step_cap
            if step < 1e-8:
                step = 1e-8
        else:
            step = step_lse_factor * a_norm / gnorm
            if step > step_cap:
                step = step_cap
            if step < 1e-8:
                step = 1e-8

        x = y - step * grad
        if exact_projection:
            x = project_simplex(x, S)
        else:
            x = clip_rescale(x, S)

        if momentum and mode != "exact":
            tn = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t * t))
            yn = x + ((t - 1.0) / tn) * (x - y)
            yn = np.maximum(yn, 0.0)
            sy = float(yn.sum())
            if sy <= 1e-14:
                yn = x.copy()
            else:
                yn *= (S / sy)
            t = tn
        else:
            yn = x
            t = 1.0

        if not np.all(np.isfinite(yn)):
            yn = best_a.copy()
            t = 1.0

        if noise_scale > 0.0 and noise_interval > 0 and it % noise_interval == 0:
            yn = clip_rescale(yn + rng.normal(0.0, noise_scale * a_norm, n), S)
            t = 1.0

        if it - last_best_it > restart_iters or (t > 500.0 and it - last_best_it > 200):
            yn = best_a.copy()
            if noise_scale > 0.0:
                yn = clip_rescale(yn + rng.normal(0.0, noise_scale * a_norm, n), S)
            t = 1.0

        y = yn
        prev_cmax = cmax

    for cand in (best_a, y, a):
        if cand is None:
            continue
        A = fft_a(env, cand)
        c = conv_from_A(env, A)
        val = float(np.max(c))
        if val < best_val:
            best_val = val
            best_a = cand.copy()

    best_a = normalize_sum(best_a, n)
    return best_a, best_val


def make_inits(n, rng, given=None, max_count=10):
    inits = []
    if given is not None:
        inits.append(("given", normalize_sum(adapt_length(given, n), n)))

    t = np.linspace(0.0, 1.0, n, endpoint=False)

    if len(inits) < max_count:
        inits.append(("ramp035", np.maximum(t ** 0.35, 1e-6)))
    if len(inits) < max_count:
        inits.append(("inv035", np.maximum((1.0 - t) ** 0.35, 1e-6)))
    if len(inits) < max_count:
        inits.append(("sum05", np.maximum(t ** 0.5 + (1.0 - t) ** 0.5, 1e-5)))
    if len(inits) < max_count:
        inits.append(("exp2", np.exp(-2.0 * t) + 1e-4))
    if len(inits) < max_count:
        inits.append(("inexp2", np.exp(-2.0 * (1.0 - t)) + 1e-4))
    if len(inits) < max_count:
        inits.append(("log03", 1.0 / (1.0 + np.exp(-20.0 * (t - 0.30))) + 1e-4))
    if len(inits) < max_count:
        inits.append(("log07", 1.0 / (1.0 + np.exp(-20.0 * (t - 0.70))) + 1e-4))
    if len(inits) < max_count:
        c0 = 0.35
        w0 = 0.12
        inits.append(("gau035", np.exp(-((t - c0) / w0) ** 2) + 1e-4))
    if len(inits) < max_count:
        inits.append(("randexp", rng.exponential(1.0, n) + 1e-3))
    if len(inits) < max_count:
        inits.append(("randnorm", np.exp(rng.normal(0.0, 0.6, n)) + 1e-3))

    out = []
    for name, arr in inits:
        if len(out) >= max_count:
            break
        arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
        arr = np.maximum(arr, 1e-8)
        out.append((name, normalize_sum(arr, n)))
    return out


def remaining_time(state):
    return max(0.0, state["deadline"] - time.monotonic())


def try_run_candidate(a0, limit, seed, state, mode="anneal", **opts):
    if a0 is None:
        return
    rem = remaining_time(state)
    limit = min(float(limit), rem)
    if limit < 2.0:
        return

    arr = np.asarray(a0, dtype=np.float64).reshape(-1)
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.size < 32:
        return
    if float(arr.sum()) < 1e-12:
        arr = np.ones(arr.size, dtype=np.float64)

    a_opt, val = run_one(arr, arr.size, limit, seed, mode=mode, **opts)
    if a_opt is None:
        return

    a_opt = np.asarray(a_opt, dtype=np.float64).reshape(-1)
    if not np.all(np.isfinite(a_opt)):
        return
    a_opt = np.clip(a_opt, 0.0, 1000.0)

    val2 = objective_fft(a_opt)
    if not np.isfinite(val2):
        return

    if val2 < state["best_val"]:
        state["best_val"] = float(val2)
        state["best_a"] = a_opt.copy()
        print("new best %.6f len=%d elapsed=%.1fs" %
              (val2, a_opt.size, time.monotonic() - state["start"]))


def fallback_sequence(n):
    S = math.sqrt(2.0 * n)
    return [float(S / n)] * n
