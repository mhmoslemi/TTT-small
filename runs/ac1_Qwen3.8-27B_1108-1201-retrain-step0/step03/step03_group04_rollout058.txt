"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using hard-threshold peak shaving
with exact gradient structure, backtracking line search, multi-peak handling,
adaptive strategies, and periodic restarts. Key innovation: exploit the exact
gradient structure db[k]/da[j] = 2*a[k-j] for direct, efficient peak reduction.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def peak_gradient_exact(a, k):
    """Compute db[k]/da exactly: grad[j] = 2*a[k-j] if 0<=k-j<n."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k + 1, n)
    if lo < hi:
        j_range = np.arange(lo, hi)
        indices = k - j_range
        valid = (indices >= 0) & (indices < n)
        if np.any(valid):
            grad[j_range[valid]] = 2.0 * a[indices[valid]]
    return grad


def soft_threshold_dir(g, c, sigma):
    """Soft-threshold direction: d[j] = -c*clip((g[j]-lam)/sigma, -1, 1)."""
    lam = np.median(g)
    raw = (g - lam) / max(sigma, 1e-30)
    d = -c * np.clip(raw, -1.0, 1.0)
    d -= d.mean()
    return d


def hard_threshold_dir(g, c):
    """Hard-threshold direction (L-infinity optimal)."""
    lam = np.median(g)
    d = np.where(g > lam, -c, np.where(g < lam, c, 0.0))
    d -= d.mean()
    return d


def compute_objective(a, n):
    """Compute the objective value."""
    S = np.sqrt(2.0 * n)
    conv = fft_convolve_self(a)
    return 2.0 * n * conv.max() / (S * S)


def run_peak_shaving(a_init, n, S, max_time, rng, top_M_init=1, top_M_max=10,
                     sigma_init=0.1, use_soft=True):
    """Hard/soft-threshold peak shaving with backtracking line search."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best = a.copy()
    best_max = np.inf
    it = 0
    c = S / n
    sigma = sigma_init
    stall = 0
    last_best = best_max

    while time.monotonic() - start < max_time:
        it += 1
        conv = fft_convolve_self(a)
        cur_max = conv.max()

        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive top-M
        M = min(top_M_max, top_M_init + it // 2000)

        # Get top-M lags
        top_indices = np.argpartition(conv, -M)[-M:]
        top_vals = conv[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # Combined gradient
        grad = np.zeros(n)
        for k_idx, w in zip(top_indices, weights):
            grad += w * peak_gradient_exact(a, int(k_idx))

        # Direction
        if use_soft:
            sigma_cur = max(0.01, sigma_init * (1.0 + 0.001 * it))
            d = soft_threshold_dir(grad, c, sigma_cur)
        else:
            d = hard_threshold_dir(grad, c)

        # Line search
        max_t = 1e6

        improved = False
        for _ in range(12):

                break
            a_new = a + t * d
            if (a_new >= 0).all() and abs(a_new.sum() - S) < 1e-8:
                new_conv = fft_convolve_self(a_new)
                if new_conv.max() < cur_max - 1e-14:
                    a = a_new
                    improved = True
                    break
            t *= 0.5

            c = min(c * 1.005, 2.0 * S / n)

        else:
            c *= 0.95
            sigma *= 1.01

        # Periodic perturbation to escape local minima

        if it % 2000 == 0:
            obj = 2.0 * n * best_max / (S * S)
            print(f"    PeakShave it={it}, max={best_max:.6f}, obj={obj:.6f}, stall={stall}")

    return best, best_max


def run_smooth_nesterov(a_init, n, S, max_time, rng, beta=200.0):
    """Smooth subgradient with Nesterov acceleration for fine-tuning."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best = a.copy()
    best_max = np.inf
    it = 0
    step = 2.0 * S / (n * 10.0)

    fsize = 1 << (2 * n).bit_length()

    while time.monotonic() - start < max_time:
        it += 1

        # Compute convolution and softmax weights
        conv = fft_convolve_self(y)
        cur_max = conv.max()
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best = y.copy()

        # Softmax weights
        c_max = conv.max()
        exp_bc = np.exp(beta * (conv - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            weights = np.full_like(conv, 1.0 / len(conv))
        else:
            weights = exp_bc / total

        # Gradient via FFT
        W = fft(weights, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step with decay
        step_cur = step / (1.0 + 0.001 * it)
        x_new = project_simplex(y - step_cur * grad / max(gnorm, 1e-15), S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-normalize
        y = y * (S / max(y.sum(), 1e-15))
        a = x_new
        t_mom = t_new

        if it % 3000 == 0 and rng.random() < 0.1:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0

    return best, best_max

def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    time_left = deadline - time.monotonic()
    if time_left > 20 and global_best_seq is not None:
        n_best = len(global_best_seq)
        S_best = np.sqrt(2.0 * n_best)
        print(f"\n=== Final polish (budget={time_left:.0f}s) ===")

        # Phase A: Ultra-fine peak shaving
        t_polish_a = 0.4 * time_left
        a_pa, ma = run_peak_shaving(global_best_seq, n_best, S_best, t_polish_a, rng,
                                     top_M_init=1, top_M_max=3,
                                     sigma_init=0.01, use_soft=False)
        a_pa = safe_normalize(a_pa, n_best)
        conv_pa = fft_convolve_self(a_pa)
        obj_pa = 2.0 * n_best * conv_pa.max() / (S_best * S_best)
        if obj_pa < global_best_obj:
            global_best_obj = obj_pa
            global_best_seq = a_pa.copy()
            print(f"  Polish A: obj={global_best_obj:.6f}")

        # Phase B: Smooth refinement
        time_left2 = deadline - time.monotonic()
        if time_left2 > 10:
            a_pb, mb = run_smooth_nesterov(global_best_seq, n_best, S_best,
                                            0.5 * time_left2, rng, beta=500.0)
            a_pb = safe_normalize(a_pb, n_best)
            conv_pb = fft_convolve_self(a_pb)
            obj_pb = 2.0 * n_best * conv_pb.max() / (S_best * S_best)
            if obj_pb < global_best_obj:
                global_best_obj = obj_pb
                global_best_seq = a_pb.copy()
                print(f"  Polish B: obj={global_best_obj:.6f}")

        # Phase C: Micro perturbations + re-optimize
        time_left3 = deadline - time.monotonic()
        if time_left3 > 15:
            n_micro = max(1, int(time_left3 / 8))
            per_micro = time_left3 / n_micro
            for mi in range(n_micro):
                if time.monotonic() > deadline - 5:
                    break
                scaling = [0.001, 0.002, 0.003, 0.005, 0.008][mi % 5]
                pert = global_best_seq + rng.normal(0, scaling * S_best / np.sqrt(n_best), n_best)
                pert = safe_normalize(pert, n_best)
                a_mc, mc = run_peak_shaving(pert, n_best, S_best, per_micro, rng,
                                             top_M_init=1, top_M_max=2,
                                             sigma_init=0.01, use_soft=False)
                a_mc = safe_normalize(a_mc, n_best)
                conv_mc = fft_convolve_self(a_mc)
                obj_mc = 2.0 * n_best * conv_mc.max() / (S_best * S_best)
                if obj_mc < global_best_obj:
                    global_best_obj = obj_mc
                    global_best_seq = a_mc.copy()
                    print(f"  Micro {mi}: obj={global_best_obj:.6f}")

    # Final output
    if global_best_seq is None:
        global_best_seq = given_a
        global_best_obj = np.inf

    final_list = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    final_obj = 2.0 * len(final_list) * fft_convolve_self(np.array(final_list)).max() / (np.sum(final_list)**2)
    print(f"\nFINAL: length={len(final_list)}, objective={final_obj:.6f}")

    # Verify
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print(f"Verified: {ver:.6f}")
    else:
        print("Verification failed! Using fallback.")
        final_list = [0.5] * 256

    return final_list
