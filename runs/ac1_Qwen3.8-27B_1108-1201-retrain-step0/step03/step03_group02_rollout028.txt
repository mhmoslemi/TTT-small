    y = a.copy()
    t_mom = 1.0
    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()

        # Adaptive top-k
        k = min(64, max(4, int(4 + it // 10)))
        grad = compute_subgradient_topk(y, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: decrease over time
        step = S / (gnorm * (1.0 + 0.005 * it))
        step = min(step, 2.0 * S / n)

        x_new = project_simplex(y - step * grad, S)

        # Momentum (FISTA-style)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

    return best_local, best_score


def hybrid_optimize(a_init, n, S, max_time, rng):
    """
    Combined optimizer: multiplicative phase followed by gradient polish.
    """
    start = time.monotonic()

    # Phase 1: Multiplicative (60% of time)
    t1 = max_time * 0.6
    a1, s1 = multiplicative_optimize(a_init, n, S, t1, rng, eta_init=0.4)

    # Phase 2: Gradient polish (30% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 2:
        t2 = remaining * 0.5
        a2, s2 = gradient_polish(a1, n, S, t2, rng)
        if s2 < s1:
            a1, s1 = a2, s2

    # Phase 3: Final multiplicative fine-tune (10% of time)
    remaining2 = max_time - (time.monotonic() - start)
    if remaining2 > 2:
        t3 = remaining2
        a3, s3 = multiplicative_optimize(a1, n, S, t3, rng, eta_init=0.1)
        if s3 < s1:
            a1, s1 = a3, s3

    return a1, s1


def make_initializations(n, rng, parent=None):
    """Generate diverse asymmetric initializations of length n."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # From parent if available
    if parent is not None:
        p = np.clip(parent, 1e-6, 1000.0)
        if len(p) == n:
            inits.append(("parent", p.copy()))
        else:
            # Resample to length n
            from scipy.interpolate import interp1d
            old_tt = np.linspace(0, 1, len(p), endpoint=False)
            f = interp1d(old_tt, p, kind='cubic', fill_value='extrapolate')
            inits.append(("parent_resampled", np.clip(f(tt), 1e-6, 1000.0)))

    # Exponential ramps
    for alpha in [0.8, 1.2, 1.5, 2.0, 3.0]:
        inits.append(("exp_f_%1.1f" % alpha, np.exp(alpha * tt)))
        inits.append(("exp_b_%1.1f" % alpha, np.exp(alpha * (1.0 - tt))))

    # Power law
    for p in [0.5, 1.0, 2.0, 4.0]:
        inits.append(("pow_f_%1.1f" % p, (tt + 0.01) ** p))
        inits.append(("pow_b_%1.1f" % p, (1.0 - tt + 0.01) ** p))

    # Cosine-modulated exponential (asymmetric)
    for alpha in [1.0, 2.0]:
        for freq in [2, 4, 8]:
            mod = 1.0 + 0.3 * np.cos(2 * np.pi * freq * tt)
            inits.append(("cosmod_f_a%1.0f_f%d" % (alpha, freq), np.exp(alpha * tt) * mod))
            inits.append(("cosmod_b_a%1.0f_f%d" % (alpha, freq), np.exp(alpha * (1 - tt)) * mod))

    # Two-bump asymmetric
    for mu1, mu2, w in [(0.2, 0.7, 0.3), (0.3, 0.8, 0.25), (0.1, 0.6, 0.35)]:
        bump = np.exp(-((tt - mu1) ** 2) / (2 * 0.05 ** 2)) + \
               w * np.exp(-((tt - mu2) ** 2) / (2 * 0.08 ** 2))
        inits.append(("bump_%1.1f_%1.1f" % (mu1, mu2), bump + 0.01))

    # Step functions (asymmetric)
    for frac in [0.3, 0.4, 0.6, 0.7]:
        step_arr = np.ones(n)
        step_arr[:int(n * frac)] = 3.0
        inits.append(("step_f_%1.1f" % frac, step_arr))
        step_arr2 = np.ones(n)
        step_arr2[int(n * (1 - frac)):] = 3.0
        inits.append(("step_b_%1.1f" % frac, step_arr2))

    # Random with trend
    for seed_offset in [0, 1, 2]:
        r = np.random.default_rng(12345 + seed_offset * 777)
        rand = r.random(n)
        ramps = [tt + 0.1, 1 - tt + 0.1, 0.5 + 0.5 * tt]
        inits.append(("rand_trend_%d" % seed_offset, rand * ramps[seed_offset]))

    # Pure linear
    inits.append(("lin_f", tt + 0.05))
    inits.append(("lin_b", 1.0 - tt + 0.05))

    # Quadratic asymmetric
    inits.append(("quad_f", (tt ** 2 + 0.01)))
    inits.append(("quad_b", ((1 - tt) ** 2 + 0.01)))

    # Hyperbolic
    inits.append(("hyp_f", 1.0 / (tt + 0.1)))
    inits.append(("hyp_b", 1.0 / (1.0 - tt + 0.1)))

    return inits


def evaluate_internal(a):
    """Compute objective without calling evaluate_sequence (faster)."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses hybrid multiplicative + gradient optimization with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Get parent sequence if available
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            parent = np.clip(given, 1e-6, 1000.0)
    except Exception:
        pass

    # Determine which lengths to try
    # Allocate time: more time to the most promising lengths
    # Start with parent length, then try nearby
    lengths_to_try = []
    if parent is not None:
        lengths_to_try.append(len(parent))
    # Add other candidates
    for n_try in [4096, 8192, 16384, 32768]:
        if n_try not in lengths_to_try:
            lengths_to_try.append(n_try)

    # Sort by priority: parent length first, then by size (larger = potentially better)
    lengths_to_try = sorted(set(lengths_to_try), key=lambda x: (0 if parent is not None and x == len(parent) else 1, -x))

    # Time allocation:
    # Phase 1: Quick exploration of multiple lengths (30% of time)
    # Phase 2: Deep optimization on best found (50% of time)
    # Phase 3: Final polishing (20% of time)

    total_time = max(60.0, deadline - time.monotonic())
    phase1_time = total_time * 0.35
    phase2_time = total_time * 0.45
    phase3_time = total_time * 0.20

    print("=== Phase 1: Exploration (budget %.0f s) ===" % phase1_time)

    # Phase 1: Quick exploration
    results = []
    time_per_length = phase1_time / len(lengths_to_try)

    for n in lengths_to_try:
        if time.monotonic() > deadline - 120:
            break
        if n < 512 or n > 65536:
            continue

        S = np.sqrt(2.0 * float(n))
        t0 = time.monotonic()
        time_budget = min(time_per_length, deadline - time.monotonic() - 60)
        if time_budget < 10:
            break

        inits = make_initializations(n, rng, parent)

        # Also add perturbed parent if parent has different length
        if parent is not None and len(parent) != n:
            from scipy.interpolate import interp1d
            old_tt = np.linspace(0, 1, len(parent), endpoint=False)
            tt_new = np.linspace(0, 1, n, endpoint=False)
            f = interp1d(old_tt, parent, kind='cubic', fill_value='extrapolate')
            inits.append(("parent_interp", np.clip(f(tt_new), 1e-6, 1000.0)))

        # For quick exploration, use fewer initializations and shorter runs
        n_inits_to_try = min(len(inits), max(3, int(time_budget / 15)))
        per_init_time = time_budget / n_inits_to_try

        for i in range(n_inits_to_try):
            if time.monotonic() > deadline - 120:
                break
            name, arr = inits[i]
            arr = np.asarray(arr, dtype=np.float64)
            if len(arr) != n:
                continue
            arr = np.clip(arr, 1e-8, 1000.0)

            a_opt, score = hybrid_optimize(arr, n, S, per_init_time, rng)
            obj = evaluate_internal(a_opt)

            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_opt.copy()
                print("  [n=%d] [%s]: obj=%.6f" % (n, name, obj))

            results.append((obj, a_opt))

        elapsed = time.monotonic() - t0
        print("  n=%d: best so far=%.6f (time %.0f s)" % (n, global_best, elapsed))

    print("=== After Phase 1: global_best=%.6f ===" % global_best)

    # Phase 2: Deep optimization on best candidate
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = deadline - time.monotonic()
        phase2_actual = min(phase2_time, remaining)

        print("=== Phase 2: Deep optimization (budget %.0f s, n=%d) ===" % (phase2_actual, n_best))

        # Multiple rounds of hybrid optimization with perturbations
        for round_i in range(5):
            if time.monotonic() > deadline - 30:
                break
            round_time = min(phase2_actual / max(1, 5 - round_i), deadline - time.monotonic() - 25)
            if round_time < 5:
                break

            # Perturb the current best slightly
            if round_i > 0:
                a_perturbed = global_a.copy()
                n_perturb = max(1, n_best // 100)
                idx = rng.integers(0, n_best, size=n_perturb)
                a_perturbed[idx] *= np.exp(rng.normal(0, 0.2, size=n_perturb))
                a_perturbed = np.clip(a_perturbed, 1e-8, 1000.0)
                s = a_perturbed.sum()
                if s > 0:
                    a_perturbed = a_perturbed * (S_best / s)
            else:
                a_perturbed = global_a.copy()

            a_opt, score = hybrid_optimize(a_perturbed, n_best, S_best, round_time, rng)
            obj = evaluate_internal(a_opt)

            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_opt.copy()
                print("  Round %d: obj=%.6f" % (round_i, global_best))
            else:
                print("  Round %d: no improvement (score=%.6f)" % (round_i, obj))

    # Phase 3: Final polishing
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = deadline - time.monotonic()
        phase3_actual = min(phase3_time, remaining)

        print("=== Phase 3: Final polish (budget %.0f s) ===" % phase3_actual)

        # Gradient polish for precision
        if phase3_actual > 10:
            a_pol, s_pol = gradient_polish(global_a, n_best, S_best, phase3_actual * 0.5, rng)
            obj = evaluate_internal(a_pol)
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_pol.copy()
                print("  Gradient polish: obj=%.6f" % global_best)

        # Very fine multiplicative
        remaining2 = deadline - time.monotonic()
        if remaining2 > 5:
            a_fin, s_fin = multiplicative_optimize(global_a, n_best, S_best, remaining2, rng, eta_init=0.05)
            obj = evaluate_internal(a_fin)
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_fin.copy()
                print("  Fine multiplicative: obj=%.6f" % global_best)

    # Also try a completely different approach in remaining time:
    # Direct coordinate descent on the best peak
    if global_a is not None and time.monotonic() < deadline - 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        a_cd = global_a.copy()
        remaining = deadline - time.monotonic() - 5
        t0 = time.monotonic()

        while time.monotonic() - t0 < remaining:
            c = fft_convolve_self(a_cd)
            p_star = int(np.argmax(c))
            cur = c[p_star]

            # Try reducing a[i] for the top contributors to c[p_star]
            i_lo = max(0, p_star - n_best + 1)
            i_hi = min(p_star, n_best - 1)
            contribs = a_cd[i_lo:i_hi + 1] * a_cd[p_star - i_hi:p_star - i_lo + 1][::-1]
            top_contrib_idx = np.argsort(contribs)[::-1][:min(32, len(contribs))]

            improved = False
            for tc in top_contrib_idx:
                i = i_lo + int(tc)
                # Try reducing a[i] by small amounts
                for frac in [0.9, 0.8, 0.7, 0.5]:
                    a_try = a_cd.copy()
                    a_try[i] *= frac
                    s_try = a_try.sum()
                    if s_try > 0:
                        a_try = a_try * (S_best / s_try)
                    c_try = fft_convolve_self(a_try)
                    if c_try.max() < cur:
                        a_cd = a_try
                        cur = c_try.max()
                        improved = True
                        break
                if improved:
                    break

            if not improved:
                # Try increasing non-contributing elements
                mask = np.ones(n_best, dtype=bool)
                mask[i_lo:i_hi + 1] = False
                candidates = np.where(mask)[0]
                if len(candidates) > 0:
                    # Pick a few random candidates
                    trial_idx = rng.choice(candidates, size=min(10, len(candidates)), replace=False)
                    for ti in trial_idx:
                        a_try = a_cd.copy()
                        a_try[ti] += S_best / n_best * 0.1
                        s_try = a_try.sum()
                        if s_try > 0:
                            a_try = a_try * (S_best / s_try)
                        c_try = fft_convolve_self(a_try)
                        if c_try.max() < cur:
                            a_cd = a_try
                            cur = c_try.max()
                            break

            if not improved:
                break

        obj = evaluate_internal(a_cd)
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_cd.copy()
            print("  Coordinate descent: obj=%.6f" % global_best)

    # Final validation and return
    if global_a is None:
        global_a = np.ones(8192, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Final: internal=%.6f, verified=%s, n=%d ===" %
          (global_best, "%.6f" % ver if np.isfinite(ver) else "inf", len(final_list)))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        global_a = np.exp(1.5 * np.linspace(0, 1, 8192, endpoint=False))
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback verified: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))

    return final_list
