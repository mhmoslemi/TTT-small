def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()

        rho = n

    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    """Convert logits x to a non-negative vector summing to S."""
    xm = x - x.max()

    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    step = S / (n * 4.0)
    stall = 0

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = a.copy()
            stall = 0
        else:
            stall += 1

        k = min(top_k, len(c))
        top_idx = np.argpartition(c, -k)[-k:]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / k

        fsize = 1 << (len(c) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        x_new = project_simplex(a - step * grad / gnorm, S)
        c_new = fft_convolve_self(x_new)

        if c_new.max() < cur:
            a = x_new
            step = min(step * 1.01, S / (n * 0.5))
        else:
            x_half = project_simplex(a - 0.5 * step * grad / gnorm, S)
            c_half = fft_convolve_self(x_half)
            if c_half.max() < cur:
                a = x_half
            else:
                a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            step *= 0.97

        if stall > 2000:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            stall = 0
            step = S / (n * 4.0)

    return best_local, best_score


def make_init(n, dtype, rng):
    """Create structured initialization."""
    t = np.arange(n)
    if dtype == 'uniform':
        a = np.ones(n)
    elif dtype == 'cosine1':
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t / n)
    elif dtype == 'cosine2':
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * t / n) + 0.2 * np.cos(2 * np.pi * 5 * t / n)
    elif dtype == 'parabolic':
        tn = t / (n - 1)
        a = 1.0 - (2 * tn - 1) ** 2
    elif dtype == 'gaussian':
        center = n / 2
        sigma = n / 5
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif dtype == 'multi_cos':
        a = np.ones(n) * 0.4
        for m in [1, 3, 7, 13]:
            a += 0.15 * np.cos(2 * np.pi * m * t / n + rng.random() * 2 * np.pi)
        a = np.maximum(a, 0.01)
    elif dtype == 'random':
        a = rng.random(n) + 0.1
    elif dtype == 'comb':
        a = 0.3 + 0.7 * (0.5 + 0.5 * np.cos(2 * np.pi * t / n))
        a *= (0.7 + 0.3 * np.cos(2 * np.pi * 7 * t / n))
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    a = a + rng.normal(0, 0.005, n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def resample_to_length(a, n):
    """Resample a sequence to a new length."""
    from scipy.interpolate import interp1d
    old_n = len(a)
    if old_n == n:
        return a.copy()
    old_t = np.linspace(0, 1, old_n)
    new_t = np.linspace(0, 1, n)
    f = interp1d(old_t, a, kind='cubic', fill_value='extrapolate')
    result = np.maximum(f(new_t), 0)
    s = result.sum()
    if s > 0:
        result = result / s
    else:
        result = np.ones(n) / n
    return result


def propose_candidate(seed=42, budget_s=2385, **kwargs):
    """
    Minimize 2*n*max_conv/sum^2 using L-BFGS-B on log-sum-exp smoothed
    objective with multiple sequence lengths and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s - 25.0)
    S = 1.0

    global_best = np.inf
    global_a = None
    global_n = 0

    # Get parent
    parent_a = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 64:
            parent_a = np.clip(parent, 0, 1000)
            ps = parent_a.sum()
            if ps > 0:
                parent_a = parent_a / ps
    except Exception:
        pass

    print("Starting optimization, budget=%.0fs" % budget_s)

    # Define optimization tasks: (n, init_type, beta, time_fraction)
    tasks = []

    if parent_a is not None:
        pn = len(parent_a)
        tasks.append((pn, 'parent', 30.0, 0.12))
        tasks.append((pn, 'parent', 100.0, 0.08))
        # Also try resampled to larger n
        tasks.append((2 * pn, 'parent', 30.0, 0.10))

    # n = 8192
    tasks.append((8192, 'cosine1', 25.0, 0.10))
    tasks.append((8192, 'cosine1', 80.0, 0.06))
    tasks.append((8192, 'multi_cos', 25.0, 0.08))
    tasks.append((8192, 'parabolic', 25.0, 0.06))

    # n = 16384
    tasks.append((16384, 'cosine1', 25.0, 0.10))
    tasks.append((16384, 'multi_cos', 25.0, 0.08))
    tasks.append((16384, 'comb', 25.0, 0.06))
    tasks.append((16384, 'parabolic', 25.0, 0.06))

    # Fine-tuning tasks (will be assigned remaining time)
    total_frac = sum(t[3] for t in tasks)
    fine_frac = 1.0 - total_frac
    if fine_frac < 0.05:
        fine_frac = 0.05
        total_frac = 1.0 - fine_frac
        for i in range(len(tasks)):
            tasks[i] = (tasks[i][0], tasks[i][1], tasks[i][2], tasks[i][3] * total_frac / sum(t[3] for t in tasks))

    total_time = budget_s - 25.0

    for ti, (n, init_type, beta, frac) in enumerate(tasks):
        if time.monotonic() > deadline - 20:
            break

        t_alloc = frac * total_time
        if t_alloc < 5:
            continue

        print("Task %d/%d: n=%d, init=%s, beta=%.0f, t=%.0fs" % (ti + 1, len(tasks), n, init_type, beta, t_alloc))

        # Create initialization
        if init_type == 'parent' and parent_a is not None:
            if len(parent_a) == n:
                a_init = parent_a.copy()
            else:
                a_init = resample_to_length(parent_a, n)
        else:
            a_init = make_init(n, init_type, rng)

        # Split time: 65% L-BFGS, 35% fine-tune
        lbfgs_time = 0.65 * t_alloc
        fine_time = 0.35 * t_alloc

        # L-BFGS phase
        a_l, score_l = run_lbfgs(a_init, n, S, beta, lbfgs_time)
        obj_l = 2.0 * n * score_l
        print("  L-BFGS: max_conv=%.8f, obj=%.6f" % (score_l, obj_l))

        if score_l < global_best:
            global_best = score_l
            global_a = a_l.copy()
            global_n = n
            print("  * New global best: obj=%.6f" % obj_l)

        # Fine-tune phase
        if fine_time > 2 and time.monotonic() < deadline - 15:
            a_f, score_f = run_exact_fine(a_l, n, S, fine_time, rng, top_k=1)
            obj_f = 2.0 * n * score_f
            print("  Fine:   max_conv=%.8f, obj=%.6f" % (score_f, obj_f))

            if score_f < global_best:
                global_best = score_f
                global_a = a_f.copy()
                global_n = n
                print("  * New global best: obj=%.6f" % obj_f)

    # Final polish on best solution
    if global_a is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 8
        if remaining > 5:
            print("Final polish: %.0fs on n=%d" % (remaining, global_n))
            a_p, score_p = run_exact_fine(global_a, global_n, S, remaining, rng, top_k=1)
            obj_p = 2.0 * global_n * score_p
            print("  Polished: obj=%.6f" % obj_p)
            if score_p < global_best:
                global_best = score_p
                global_a = a_p.copy()

    # Output
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        final_obj = 2.0 * global_n * global_best
        print("FINAL: n=%d, max_conv=%.8f, objective=%.6f" % (global_n, global_best, final_obj))

        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified: %.6f" % ver)
        else:
            print("Verification FAILED, using fallback")
            final_list = [0.5] * 256
    else:
        print("No valid solution found, using fallback")
        final_list = [0.5] * 256

    return final_list
