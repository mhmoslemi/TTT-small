"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Novel approach combining:
1. Gerchberg-Saxton style frequency-domain iterations (flatten power spectrum)
2. FISTA with log-sum-exp smoothing and adaptive parameters

4. Multi-phase time allocation with periodic diversification

Use FFT-based operations throughout for O(n log n) per iteration.

import numpy as np
import time
from scipy.fft import fft, ifft


def next_pow2(n):
    return 1 << (n - 1).bit_length()


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = next_pow2(size)
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta, fsize):
    """Gradient of (1/beta)*log(sum(ema(beta*c))) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_subgradient(a, c, top_k, fsize):
    """Subgradient using top-k active lags with uniform weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argpartition(c, -k)[-k:]
    w_full = np.zeros(fsize)
    w_full[top_indices] = 1.0 / k
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_objective(a):
    """Compute the full evaluation objective."""
    n = len(a)
    c = fft_convolve_self(a)
    max_b = c.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)

        elif flatten_mode == 'mag_pow':
            # Power compression: |A|^alpha where alpha < 1
            gamma = 0.5 + 0.5 * (1.0 - alpha)  # from 1.0 to 0.5

        else:
            new_mag = mag
        
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]

        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    N = next_pow2(2 * n - 1)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = init_L if init_L is not None else beta_init * 4.0
    stall = 0
    last_best = np.inf
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = y.copy()
            stall = 0
            last_best = cur
        elif it % 100 == 0 and cur >= last_best:
            stall += 1
        
        grad = lse_gradient(y, c, beta, N)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        step = 1.0 / L
        f_y = c.max()
        x_new = None
        for bt in range(4):
            x_cand = project_simplex(y - step * grad, S)
            if bt > 0:
                c_cand = fft_convolve_self(x_cand)
                if c_cand.max() <= f_y - 0.5e-4 * step * gnorm * gnorm:
                    x_new = x_cand
                    break
                step *= 0.5
            else:
                x_new = x_cand
        
        if x_new is None:
            x_new = project_simplex(y - step * grad, S)
        
        # Update Lipschitz estimate
        c_x = fft_convolve_self(x_new)
        if c_x.max() > f_y + 1e-10:
            L *= 1.8
        elif it % 50 == 0:
            L = max(L * 0.97, beta * 3.0)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Perturbation to escape stalls
        if stall > 20 and rng.random() < 0.2:
            noise = 0.02 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0
            stall = 0
    
    return best_local, best_score


def peak_shave(a_init, n, S, max_time, rng, n_top_lags=5, n_top_pairs=30):
    """
    Directly target the peak convolution lags by redistributing mass.
    For each hot lag k, find pairs (i, k-i) with large product and
    move mass from the larger element to a "cold" position.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_a = a.copy()
    best_score = np.inf
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Restart with perturbation
            noise = 0.01 * S / np.sqrt(n)
            a = project_simplex(best_a + rng.normal(0, noise, n), S)
            no_improve = 0
            continue
        
        # Find top lags
        top_lags = np.argpartition(c, -n_top_lags)[-n_top_lags:]
        top_lags = top_lags[np.argsort(c[top_lags])[::-1]]
        
        improved = False
        for lag in top_lags:
            lag = int(lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue
            
            indices = np.arange(i_min, i_max + 1)
            partner = lag - indices
            products = a[indices] * a[partner]
            
            # Get top pairs by product
            n_pairs = min(n_top_pairs, len(indices))
            top_pair_local = np.argsort(products)[::-1][:n_pairs]
            
            for pi in top_pair_local:
                idx = int(indices[pi])
                pidx = int(partner[pi])
                if idx == pidx:
                    continue  # Diagonal term, can't redistribute between same element
                
                # Find a "cold" position to add mass
                # Prefer positions that don't contribute much to any top lag
                # Simple heuristic: pick a random index with small value
                cold_candidates = np.where(a < 0.5 * a.max())[0]
                if len(cold_candidates) == 0:
                    cold_candidates = np.where(a < a.mean())[0]
                if len(cold_candidates) == 0:
                    continue
                
                add_idx = cold_candidates[rng.integers(len(cold_candidates))]
                if add_idx == idx or add_idx == pidx:
                    continue
                
                # Move a small amount from the larger of the pair to the cold position
                if a[idx] >= a[pidx]:
                    donor = idx
                else:
                    donor = pidx
                
                # Adaptive step: proportional to the element value
                delta = 0.01 * a[donor]
                a[donor] -= delta
                a[add_idx] += delta
                
                # Check if it helped (spot check the affected lags)
                new_val_lag = 0.0
                ii_min = max(0, lag - (n - 1))
                ii_max = min(lag, n - 1)
                for i2 in range(ii_min, ii_max + 1):
                    j2 = lag - i2
                    if 0 <= j2 < n:
                        new_val_lag += a[i2] * a[j2]
                
                if new_val_lag < products[pi] - 1e-12:
                    improved = True
                    pass  # Keep the change
                else:
                    # Revert
                    a[donor] += delta
                    a[add_idx] -= delta
        
        # Renormalize
        s_cur = a.sum()
        if s_cur < 1e-15:
            a = best_a.copy()
        else:
            a = a * (S / s_cur)
            a = np.clip(a, 0.0, 1000.0)
        
        if it % 1000 == 0:
            obj = 2.0 * n * best_score / (S * S)
            print("    PeakShave iter %d: max_b=%.6f, obj=%.6f (elapsed %.0fs)" % 
                  (it, best_score, obj, time.monotonic() - start))
    
    return best_a, best_score


def mirror_descent(a_init, n, S, max_time, rng, top_k=10, eta_init=0.1):
    """Mirror descent with multiplicative updates for non-negativity."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    N = next_pow2(2 * n - 1)
    start = time.monotonic()
    best_a = a.copy()
    best_score = np.inf
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_a = a.copy()
        
        eta = eta_init / (1.0 + 0.005 * it)
        grad = topk_subgradient(a, c, top_k, N)
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        
        log_update = np.clip(-eta * grad / gmax, -3.0, 3.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_new * (S / s_new)
            a = np.minimum(a, 1000.0)
        
        if it % 5000 == 0:
            obj = 2.0 * n * best_score / (S * S)
            print("    Mirror iter %d: max_b=%.6f, obj=%.6f (elapsed %.0fs)" % 
                  (it, best_score, obj, time.monotonic() - start))
    
    return best_a, best_score


def make_anti_palindromic(n, rng, S):
    """Anti-palindromic initialization: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())
    return a


def make_periodic(n, rng, S, n_freqs=4):
    """Periodic initialization with multiple frequency components."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(n_freqs):
        freq = rng.integers(2, 25)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.4
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_w_shape(n, rng, S):
    """W/M-shaped initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    k = rng.integers(2, 6)
    a = np.abs(np.sin(2 * np.pi * k * tt)) + 0.05
    a = a * (S / a.sum())
    return a


def make_ramp(n, rng, S):
    """Ramp/power-law initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    p = rng.random() * 1.5 + 0.3
    a = tt ** p + 0.05
    if rng.random() < 0.5:
        a = (1.0 - tt) ** p + 0.05
    a = a * (S / a.sum())
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    
    Multi-phase strategy:
    Phase 1: Gerchberg-Saxton exploration (40% time)
    Phase 2: FISTA local optimization (30% time)  
    Phase 3: Peak shaving + mirror descent (20% time)
    Phase 4: Final high-beta FISTA polish (10% time)
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)
    
    # Get parent sequence if available
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    # Choose primary sequence length
    primary_n = 8192
    if parent is not None and len(parent) >= 64:
        primary_n = len(parent)
    
    S = np.sqrt(2.0 * primary_n)
    print("Length: %d, S=%.4f, parent: %s" % (primary_n, S, parent is not None))
    
    global_best = np.inf
    global_a = None
    
    def update_best(a, label):
        nonlocal global_best, global_a
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return
        c = fft_convolve_self(a)
        score = c.max()
        obj = 2.0 * len(a) * score / (s * s)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  NEW BEST [%s]: obj=%.6f (max_b=%.4f, n=%d)" % (label, obj, score, len(a)))
    
    # ============ Phase 1: GS Exploration ============
    t0 = time.monotonic()
    phase1_budget = (deadline - t0) * 0.35
    print("=== Phase 1: GS Exploration (%.0f sec) ===" % phase1_budget)
    
    # Generate initializations
    inits = []
    if parent is not None and len(parent) >= 64:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if p.sum() > 0:
            inits.append(("parent", p * (S / p.sum())))
    
    for i in range(3):
        inits.append(("anti_pal_%d" % i, make_anti_palindromic(primary_n, rng, S)))
    for i in range(3):
        inits.append(("periodic_%d" % i, make_periodic(primary_n, rng, S)))
    for i in range(2):
        inits.append(("wshape_%d" % i, make_w_shape(primary_n, rng, S)))
    for i in range(2):
        inits.append(("ramp_%d" % i, make_ramp(primary_n, rng, S)))
    
    # Add a few random
    for i in range(2):
        ra = rng.random(primary_n) + 0.05
        inits.append(("rand_%d" % i, ra * (S / ra.sum())))
    
    n_inits = len(inits)
    per_init_time = phase1_budget / n_inits
    print("  %d initializations, %.1f sec each" % (n_inits, per_init_time))
    
    # Run GS on each initialization
    gs_best_a = None
    gs_best_score = np.inf
    
    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break
        rem = min(per_init_time, deadline - time.monotonic() - 300)
        if rem < 2:
            continue
        t1 = time.monotonic()
        a_gs, s_gs = gs_optimize(arr, primary_n, S, rem, rng,
                                  flatten_mode='mag',
                                  alpha_start=0.03, alpha_end=0.6,
                                  polish_interval=10000)
        update_best(a_gs, "GS_" + name)
        if s_gs < gs_best_score:
            gs_best_score = s_gs
            gs_best_a = a_gs.copy()
        if time.monotonic() - t1 > 0:
            pass  # timing already in gs_optimize
    
    # Also try a longer sequence
    if time.monotonic() < deadline - 400:
        n_long = 16384
        S_long = np.sqrt(2.0 * n_long)
        rem_long = min(120.0, (deadline - time.monotonic()) * 0.1)
        if rem_long > 20:
            print("  Trying n=%d..." % n_long)
            a_long_init = make_anti_palindromic(n_long, rng, S_long)
            a_l, s_l = gs_optimize(a_long_init, n_long, S_long, rem_long, rng,
                                    flatten_mode='mag_pow',
                                    alpha_start=0.03, alpha_end=0.5)
            # Compare objectives properly
            obj_l = 2.0 * n_long * s_l / (S_long * S_long)
            if global_a is not None:
                obj_g = 2.0 * len(global_a) * 0  # we'll use global_best directly
            if obj_l < global_best:
                global_best = obj_l
                global_a = a_l.copy()
                print("  NEW BEST [GS_long]: obj=%.6f" % global_best)
    
    phase1_time = time.monotonic() - t0
    print("  Phase 1 done in %.0f sec. Best so far: %.6f" % (phase1_time, global_best))
    
    # ============ Phase 2: FISTA Local Optimization ============
    t0 = time.monotonic()
    phase2_budget = (deadline - t0) * 0.30
    print("=== Phase 2: FISTA Optimization (%.0f seg) ===" % phase2_budget)
    
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        
        # Multiple FISTA runs with different beta ranges
        fista_configs = [
            {"beta_init": 5.0, "beta_max": 100.0, "frac": 0.35},
            {"beta_init": 50.0, "beta_max": 300.0, "frac": 0.35},
            {"beta_init": 200.0, "beta_max": 800.0, "frac": 0.30},
        ]
        
        for cfg in fista_configs:
            if time.monotonic() > deadline - 150:
                break
            rem = phase2_budget * cfg["frac"]
            rem = min(rem, deadline - time.monotonic() - 150)
            if rem < 3:
                continue
            print("  FISTA beta=[%d, %d] for %.0f sec" % (
                cfg["beta_init"], cfg["beta_max"], rem))
            a_f, s_f = fista_lse(global_a, n, S, rem, rng,
                                  beta_init=cfg["beta_init"],
                                  beta_max=cfg["beta_max"])
            update_best(a_f, "FISTA_b%d" % int(cfg["beta_init"]))
    else:
        # Fallback: use GS best
        if gs_best_a is not None:
            global_a = gs_best_a
            n = len(global_a)
            S = np.sqrt(2.0 * n)
            a_f, s_f = fista_lse(global_a, n, S, phase2_budget * 0.5, rng)
            update_best(a_f, "FISTA_fallback")
    
    print("  Phase 2 done. Best: %.6f" % global_best)
    
    # ============ Phase 3: Peak Shaving + Mirror Descent ============
    t0 = time.monotonic()
    phase3_budget = (deadline - t0) * 0.25
    print("=== Phase 3: Peak Shaving + Mirror (%.0f seg) ===" % phase3_budget)
    
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        
        # Peak shaving
        rem3a = phase3_budget * 0.5
        rem3a = min(rem3a, deadline - time.monotonic() - 60)
        if rem3a > 3:
            print("  Peak shaving for %.0f sec" % rem3a)
            a_ps, s_ps = peak_shave(global_a, n, S, rem3a, rng,
                                     n_top_lags=5, n_top_pairs=20)
            update_best(a_ps, "PeakShave")
        
        # Mirror descent
        rem3b = min(phase3_budget * 0.3, deadline - time.monotonic() - 60)
        if rem3b > 3:
            print("  Mirror descent for %.0f sec" % rem3b)
            a_md, s_md = mirror_descent(global_a, n, S, rem3b, rng,
                                         top_k=15, eta_init=0.05)
            update_best(a_md, "Mirror")
        
        # Another peak shave round
        rem3c = min(phase3_budget * 0.2, deadline - time.monotonic() - 30)
        if rem3c > 3 and global_a is not None:
            n = len(global_a)
            S = np.sqrt(2.0 * n)
            print("  Peak shave round 2 for %.0f sec" % rem3c)
            a_ps2, s_ps2 = peak_shave(global_a, n, S, rem3c, rng,
                                       n_top_lags=8, n_top_pairs=50)
            update_best(a_ps2, "PeakShave2")
    
    print("  Phase 3 done. Best: %.6f" % global_best)
    
    # ============ Phase 4: Final Polish ============
    t0 = time.monotonic()
    phase4_budget = max(10.0, (deadline - t0) * 0.85)
    print("=== Phase 4: Final Polish (%.0f sec) ===" % phase4_budget)
    
    if global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        
        # Very high beta FISTA for final polish
        a_final, s_final = fista_lse(global_a, n, S, phase4_budget * 0.6, rng,
                                      beta_init=500.0, beta_max=2000.0)
        update_best(a_final, "FinalFISTA_hi")
        
        # One more peak shave
        rem4 = deadline - time.monotonic() - 10
        if rem4 > 5 and global_a is not None:
            n = len(global_a)
            S = np.sqrt(2.0 * n)
            a_ps3, _ = peak_shave(global_a, n, S, rem4 * 0.7, rng,
                                   n_top_lags=3, n_top_pairs=100)
            update_best(a_ps3, "FinalPS")
    
    # ============ Verification ============
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== Final verification: %.6f (internal: %.6f), length=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))
    
    if not np.isfinite(ver):
        print("  WARNING: evaluate_sequence rejected, using fallback")
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback: %.6f" % ver)
    
    total_time = time.monotonic() - (deadline - budget_s + 15.0)
    print("=== DONE. Final score: %.6f, time used: %.0f sec ===" % 
          (ver if np.isfinite(ver) else float('inf'), time.monotonic() - (deadline - budget_s + 15.0)))
    
    return final_list
