```python
def multiplicative_update(a, T, eta, peak_weight=1.0):
    """One step of multiplicative update to reduce peaks above T."""
    c = fft_convolve_self(a)
    # Find violations
    violation = np.maximum(c - T, 0)
    if violation.max() < 1e-15:
        return a, True  # Feasible!
    
    # Weight by violation amount (raised to peak_weight)
    w = violation ** peak_weight
    w_sum = w.sum()
    if w_sum < 1e-30:
        return a, True
    
    w = w / w_sum
    
    # Compute gradient-like quantity: for each i, sum over violated k of w[k] * 2*a[k-i]
    # This is 2 * cross_corr(w, a)
    n = len(a)
    fsize = 1 << (3*n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    grad = 2.0 * corr[:n]
    
    # Multiplicative update
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new * (a.sum() / a_new.sum())  # Renormalize to keep sum constant
    
    return a_new, False
```

```python
def greedy_peak_reduction(a, n_steps=1000):
    for step in range(n_steps):
        c = fft_convolve_self(a)
        k_star = np.argmax(c)
        max_val = c[k_star]
        
        # Find the index i that most contributes to this peak
        # c[k_star] = sum_i a[i]*a[k_star-i]
        products = np.zeros(len(a))
        for i in range(len(a)):
            j = k_star - i
            if 0 <= j < len(a):
                products[i] = a[i] * a[j]
        
        i_star = np.argmax(products)
        
        # Find a target index j to move mass to
        # We want j such that moving mass there doesn't increase other peaks
        # Heuristic: move to an index where a is already large (to dilute the contribution)
        # Or: move to an index that minimizes the max resulting peak
        
        # Simple heuristic: try a few candidate targets
        best_delta = 0
        best_obj = max_val
        
        # Try moving a small amount from i_star to several candidate indices
        candidates = np.argsort(a)[-10:]  # Top 10 indices by value
        for j in candidates:
            if j == i_star:
                continue
            # Try moving 1% of a[i_star] to j
            for frac in [0.01, 0.05, 0.1, 0.2]:

                a_trial[i_star] -= delta

                if obj_trial < best_obj:

                    best_delta = delta

            a[i_star] -= best_delta
            a[best_j] += best_delta
```

```python
def compute_top_conv(a, k_positions):
    """Compute conv(a,a) at specific positions."""
    n = len(a)
    result = np.zeros(len(k_positions))
    for idx, k in enumerate(k_positions):
        # c[k] = sum_i a[i]*a[k-i] for valid i
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_min <= i_max:
            result[idx] = np.dot(a[i_min:i_max+1], a[k-i_max:k-i_min+1][::-1])
    return result
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using fast 2-coordinate exchange 
with peak-tracking and binary search for optimal step size.

Key innovations:
1. O(K) evaluation of peak changes for 2-coordinate moves (no FFT in inner loop)
2. Binary search on target peak level for exact optimal delta
3. Periodic full FFT recompute to update peak positions
4. Responsibility-based source selection and score-based target selection
5. Multi-phase strategy: fast local search → subgradient refinement → polish
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_peak_changes(peak_positions, peak_values, a, i, j, delta):
    """Compute new peak values after 2-coordinate exchange (i loses delta, j gains delta)."""
    n = len(a)
    K = len(peak_positions)
    new_vals = np.zeros(K)
    
    for m in range(K):
        km = int(peak_positions[m])
        # Change: 2*delta*(a[km-j] - a[km-i]) - delta^2 * 1_{km=i+j}
        aj = a[km - j] if 0 <= km - j < n else 0.0
        ai = a[km - i] if 0 <= km - i < n else 0.0
        quad = 1.0 if km == i + j else 0.0
        new_vals[m] = peak_values[m] + 2.0 * delta * (aj - ai) - delta * delta * quad
    
    return new_vals


def find_optimal_delta(peak_values, peak_positions, a, i, j):
    """
    Find delta in [0, a[i]] minimizing max_m new_peak[m]
    where new_peak[m] = pv[m] + 2*delta*(a[km-j] - a[km-i]) - delta^2 * 1_{km=i+j}
    """
    n = len(a)
    K = len(peak_positions)
    max_delta = a[i]
    
    if max_delta < 1e-30:
        return 0.0
    
    # Compute slopes and quadratic terms
    slopes = np.zeros(K)
    quad = np.zeros(K)
    for m in range(K):
        km = int(peak_positions[m])
        slopes[m] = 2.0 * (
            (a[km - j] if 0 <= km - j < n else 0.0) -
            (a[km - i] if 0 <= km - i < n else 0.0)
        )
        quad[m] = 1.0 if km == i + j else 0.0
    
    # For small delta, ignore quadratic term and solve linear problem
    # min_T max_m (pv[m] + slopes[m] * delta)
    # s.t. delta in [0, max_delta]

    # At delta=0: T = max(pv)
    T_high = float(peak_values.max())
    T_low = 0.0

        vals = peak_values + slopes * d - quad * d * d
        T_low = max(T_low, 0.0)

        # Let me just use bisection properly

    # f(T) = "is there a delta in [0, max_delta] such that pv[m] + slopes[m]*delta - quad[m]*delta^2 <= T for all m?"
    # For the linear approximation (ignoring quad):
    # pv[m] + slopes[m]*delta <= T for all m
    # If slopes[m] > 0: delta <= (T - pv[m]) / slopes[m]
    # If slopes[m] < 0: delta >= (T - pv[m]) / slopes[m]

    def feasible(T):

        delta_high = max_delta

                ub = (T - peak_values[m]) / slopes[m]
                delta_high = min(delta_high, ub)

                lb = (T - peak_values[m]) / slopes[m]
                delta_low = max(delta_low, lb)
            else:
                if peak_values[m] > T + 1e-15:
                    return False
        return delta_low <= delta_high + 1e-15
    
    # Check if T = max(pv) is feasible (it should be, at delta=0)
    if not feasible(T_high):
        return 0.0
    
    # Bisection
    lo = 0.0
    hi = T_high
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if feasible(mid):
            hi = mid
        else:
            lo = mid
    
    T_opt = hi
    
    # Now find the optimal delta: it's the midpoint of [delta_low, delta_high] at T = T_opt
    # Or more precisely, any delta in the feasible range works
    delta_low = 0.0
    delta_high = max_delta
    for m in range(K):
        if slopes[m] > 1e-30:
            ub = (T_opt - peak_values[m]) / slopes[m]
            delta_high = min(delta_high, ub)
        elif slopes[m] < -1e-30:
            lb = (T_opt - peak_values[m]) / slopes[m]
            delta_low = max(delta_low, lb)

    Fast 2-coordinate exchange optimization.

    S_target = np.sqrt(2.0 * n)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

            peak_positions = top_idx.astype(np.int64)
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using fast 2-coordinate exchange with
O(K) peak tracking, binary search for optimal step, and multi-phase strategy.

Key innovations over previous approach:
1. O(K) evaluation of peak changes for 2-coordinate moves (no FFT in inner loop)
2. Exact binary search on target peak level for optimal delta
3. Responsibility-based source selection and weighted score-based target selection
4. Periodic full FFT recompute to maintain peak tracking accuracy
5. Multi-phase: fast 2-coord exchange → subgradient refinement → coordinate polish
6. Exploration of different sequence lengths

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_peak_changes(peak_positions, peak_values, a, i, j, delta):
    """Compute new peak values after 2-coordinate exchange using OLD a values."""
    n = len(a)
    K = len(peak_positions)
    new_vals = np.zeros(K)
    for m in range(K):
        km = int(peak_positions[m])
        aj = a[km - j] if 0 <= km - j < n else 0.0
        ai = a[km - i] if 0 <= km - i < n else 0.0
        quad = 1.0 if km == i + j else 0.0
        new_vals[m] = peak_values[m] + 2.0 * delta * (aj - ai) - delta * delta * quad
    return new_vals


def find_optimal_delta(peak_values, peak_positions, a, i, j):
    """
    Find delta in [0, a[i]] minimizing max_m new_peak[m].
    Uses binary search on target level T.
    """
    n = len(a)
    K = len(peak_positions)
    max_delta = a[i]

    if max_delta < 1e-30:
        return 0.0

    # Compute slopes: new_peak[m] ≈ pv[m] + 2*delta*(a[km-j] - a[km-i])
    slopes = np.zeros(K)
    quad = np.zeros(K)
    for m in range(K):
        km = int(peak_positions[m])
        slopes[m] = 2.0 * (
            (a[km - j] if 0 <= km - j < n else 0.0) -
            (a[km - i] if 0 <= km - i < n else 0.0)
        )
        quad[m] = 1.0 if km == i + j else 0.0

    # Bisection on T
    T_high = float(peak_values.max())

    def feasible(T):
        delta_low = 0.0
        delta_high = max_delta
        for m in range(K):
            sm = slopes[m]
            pvm = peak_values[m]
            if sm > 1e-15:
                ub = (T - pvm) / sm
                if ub < delta_high:
                    delta_high = ub
            elif sm < -1e-15:
                lb = (T - pvm) / sm
                if lb > delta_low:
                    delta_low = lb
            else:
                if pvm > T + 1e-12:
                    return False
        return delta_low <= delta_high + 1e-12

    if not feasible(T_high):
        return 0.0

    lo = 0.0
    hi = T_high
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if feasible(mid):
            hi = mid
        else:
            lo = mid

    T_opt = hi

    # Find a good delta at T_opt
    delta_low = 0.0
    delta_high = max_delta
    for m in range(K):
        sm = slopes[m]
        pvm = peak_values[m]
        if sm > 1e-15:
            ub = (T_opt - pvm) / sm
            if ub < delta_high:
                delta_high = ub
        elif sm < -1e-15:
            lb = (T_opt - pvm) / sm
            if lb > delta_low:
                delta_low = lb

    delta_low = max(0.0, delta_low)
    delta_high = min(max_delta, delta_high)

    if delta_low > delta_high + 1e-15:
        delta_low = 0.0
        delta_high = max_delta

    # Evaluate a few candidates
    candidates = [delta_low, delta_high, (delta_low + delta_high) / 2.0, 0.5 * delta_high]
    best_delta = 0.0
    best_obj = np.inf

        vals = peak_values + slopes * d - quad * d * d
        obj = float(vals.max())
        if obj < best_obj:
            best_obj = obj
            best_delta = d

    return best_delta


def fast_2coord_exchange(a_init, time_budget, K=80, recompute_every=50, verbose=False):

    Fast 2-coordinate exchange optimization with O(K) peak tracking.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    S_target = np.sqrt(2.0 * n)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.full(n, S_target / n)
    else:
        a = a * (S_target / a.sum())

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    # Initial peak tracking
    if K > len(c):
        K = len(c)
    top_idx = np.argsort(c)[-K]
    peak_positions = top_idx.astype(np.int64).copy()
    peak_values = c[peak_positions].copy()

    step = 0
    no_improve = 0
    rng = np.random.default_rng(777)
    consecutive_no_delta = 0

            if cur_max < best_obj - 1e-12:

                no_improve = 0

            else:

                a = a * (S_target / a.sum())
                no_improve = 0
                consecutive_no_delta = 0
                step = 0

            top_idx = np.argsort(c)[-K]
            peak_positions = top_idx.astype(np.int64).copy()

        worst_m = int(np.argmax(peak_values))
        k_worst = int(peak_positions[worst_m])

        i_valid_min = max(0, k_worst - n + 1)
        i_valid_max = min(k_worst, n - 1)

        # Source candidates: top responsibility

def run_adam_log(a_init, n, S, time_budget, T_init=20.0, T_final=500.0,

    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    a = a * (S / a.sum())
    theta = np.log(a / a.sum())

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8

    best_obj = c.max()

    fsize = 1 << (3 * n - 2).bit_length()

    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.5)

        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
        else:
            a = S * exp_theta / total

        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()

        c_max = c.max()
        if c_max < 1e-30:
            break
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

        W = fft(w, fsize)
        A = fft(a, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        g_a = 2.0 * corr[:n]

        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot

        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break

        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_init * (0.3 + 0.7 * (1.0 - progress))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_obj


def coordinate_descent_polish(a, n, S, time_budget, max_sweeps=20):
    """Coordinate descent: try multiplicative adjustments to individual coordinates."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.full(n, S / n)
    else:
        a = a * (S / a.sum())

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(99999)
    factors = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - start > time_budget:
                break
            original = a[i]
            if original < 1e-12:
                continue

            best_local = original
            best_local_obj = best_obj

            for factor in factors:
                a[i] = original * factor
                a = a * (S / a.sum())
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-12:
                    best_local_obj = obj_check
                    best_local = original * factor

            a[i] = best_local
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            if c.max() < best_obj - 1e-12:
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        if not improved and sweep >= 2:
            break

    return best_a, best_obj


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 30.0)
    deadline = time.monotonic() + total_budget

    global_best = np.inf
    global_a = None

    # Load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw.copy()
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: " + str(e))

    if given_a is None:
        n_given = 4096
        given_a = np.ones(n_given)
        S_given = np.sqrt(2.0 * n_given)
        given_a = given_a * (S_given / given_a.sum())
        ver = evaluate_sequence(given_a.tolist())
        print("Default: n=%d, obj=%.7f" % (n_given, ver))
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = given_a.copy()

    n = len(global_a)
    S = np.sqrt(2.0 * float(n))
    print("Starting n=%d, obj=%.7f" % (n, global_best))

    # ============ PHASE 1: Fast 2-coordinate exchange (45% time) ============
    phase1_budget = 0.45 * total_budget
    print("\nPhase 1: Fast 2-coordinate exchange (budget=%.0fs, n=%d)" % (phase1_budget, n))

    a_best, obj_best = fast_2coord_exchange(

    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  Phase 1 result: obj=%.7f" % ver)
    else:
        print("  Phase 1: no improvement (obj=%.7f vs best=%.7f)" % (ver, global_best))

    # ============ PHASE 2: Adam log-space refinement (25% time) ============
    phase2_budget = 0.25 * total_budget
    print("\nPhase 2: Adam log refinement (budget=%.0fs)" % phase2_budget)

    # Round 1: moderate T
    r1_budget = 0.4 * phase2_budget
    a_best, _ = run_adam_log(
        global_a, n, S, r1_budget,
        T_init=50, T_final=300, lr_init=0.02
    )
    a_best = normalize_nonneg(a_best, S)
    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  R1 Adam T50-300: obj=%.7f" % ver)

    # Round 2: high T (focus on max peak)
    r2_budget = 0.35 * phase2_budget
    a_best, _ = run_adam_log(
        global_a, n, S, r2_budget,
        T_init=200, T_final=1000, lr_init=0.015
    )
    a_best = normalize_nonneg(a_best, S)
    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  R2 Adam T200-1000: obj=%.7f" % ver)

    # Round 3: Very high T
    r3_budget = 0.25 * phase2_budget
    a_best, _ = run_adam_log(
        global_a, n, S, r3_budget,

    )
    a_best = normalize_nonneg(a_best, S)
    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  R3 Adam T500-3000: obj=%.7f" % ver)

    # ============ PHASE 3: Coordinate descent polish (15% time) ============
    phase3_budget = 0.15 * total_budget
    print("\nPhase 3: Coordinate descent polish (budget=%.0fs)" % phase3_budget)

    a_best, _ = coordinate_descent_polish(global_a, n, S, phase3_budget, max_sweeps=15)
    a_best = normalize_nonneg(a_best, S)
    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  CD polish: obj=%.7f" % ver)

    # ============ PHASE 4: Second round of 2-coord exchange (10% time) ============
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("\nPhase 4: Second 2CE round (budget=%.0fs)" % remaining)
        a_best, obj_best = fast_2coord_exchange(
            global_a, remaining, K=100, recompute_every=30, verbose=False
        )
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  Phase 4: obj=%.7f" % ver)

    # ============ PHASE 5: Try n=8192 if time remains (5% time) ============
    remaining = deadline - time.monotonic()
    if remaining > 60 and n < 8000:
        n2 = 8192
        S2 = np.sqrt(2.0 * float(n2))
        # Initialize from upsampling or random
        a2 = rng.random(n2)
        # Mix in structure from best solution
        if global_a is not None:
            # Upsample by repetition with interpolation
            a_interp = np.interp(np.linspace(0, 1, n2), np.linspace(0, 1, n), global_a / global_a.sum())
            a2 = 0.5 * a2 / a2.sum() + 0.5 * a_interp / a_interp.sum()
        a2 = a2 * (S2 / a2.sum())

        print("\nPhase 5: Try n=%d (budget=%.0fs)" % (n2, remaining * 0.7))
        a_best2, _ = fast_2coord_exchange(a2, remaining * 0.7, K=60, recompute_every=100, verbose=False)
        a_best2 = normalize_nonneg(a_best2, S2)
        ver2 = evaluate_sequence(a_best2.tolist())
        if np.isfinite(ver2) and ver2 < global_best:
            global_best = ver2
            global_a = a_best2.copy()
            n = n2
            S = S2
            print("  n=%d: obj=%.7f" % (n2, ver2))

    # ============ Final polish ============
    remaining = deadline - time.monotonic()
    if remaining > 10 and global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a_best, _ = coordinate_descent_polish(global_a, n, S, remaining, max_sweeps=5)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  Final CD: obj=%.7f" % ver)

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f (verified=%s)" % (
        global_best, "%.7f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)

    return final_a
```
