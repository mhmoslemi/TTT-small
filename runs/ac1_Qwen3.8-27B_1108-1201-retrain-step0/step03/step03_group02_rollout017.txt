import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def grad_topk(a, c, k, alpha=1.0):
    """Subgradient using top-k peaks with exponential weighting."""
    n = len(a)
    c_max = c.max()
    # Exponential weighting on top-k
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices]
    # Use exponential weighting: w_i = exp(-alpha * (c_max - c_i))
    weights = np.exp(-alpha * (c_max - top_vals))
    weights /= weights.sum()
    
    w_full = np.zeros(2 * n - 1)
    for i, k_idx in enumerate(top_indices):
        w_full[int(k_idx)] = weights[i]

    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if v.sum() >= S:
        w = np.sort(v)[::-1]
        cumsum = np.cumsum(w)
        idx = np.arange(1, n + 1, dtype=np.float64)
        theta_candidates = (cumsum - S) / idx
        valid = theta_candidates <= w
        if not np.any(valid):
            rho = n
        else:
            rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
        return np.maximum(v - theta, 0.0)
    else:
        # Sum is less than S, just scale up
        return v * (S / v.sum())


def compute_grad_softmax(a, c, beta):
    """Gradient using soft-max weighting over all lags."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_grad_topk(a, c, k, sharpness=1.0):
    """Gradient using top-k peaks with exponential sharpness weighting."""
    n = len(a)
    c_max = c.max()
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices]
    weights = np.exp(sharpness * (top_vals - c_max))
    total = weights.sum()
    if total < 1e-300:
        weights = np.ones(k) / k
    else:
        weights = weights / total
    w_full = np.zeros(2 * n - 1)
    for i in range(k):
        w_full[int(top_indices[i])] = weights[i]
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

            best_local = y.copy()

        # Compute gradient
        grad = compute_grad_softmax(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Barzilai-Borwein step size
        if bb_step and prev_grad is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            s_norm2 = np.dot(s_vec, s_vec)
            y_norm2 = np.dot(y_vec, y_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            if s_norm2 > 1e-20 and y_norm2 > 1e-20:
                # BB1: gamma = s^T s / s^T y
                gamma1 = s_norm2 / max(s_dot_y, 1e-20)
                # BB2: gamma = s^T y / y^T y
                gamma2 = s_dot_y / y_norm2 if y_norm2 > 1e-20 else step
                # Alternate or average
                if it % 2 == 0:
                    step_candidate = max(1e-8, min(gamma1, S * 2))
                else:
                    step_candidate = max(1e-8, min(gamma2, S * 2))
                # Blend with current step for stability
                step = 0.5 * step + 0.5 * step_candidate
            else:
                step *= 0.9

        prev_a = a.copy()
        prev_grad = grad.copy()

        # Nesterov step
        x_new = project_simplex(y - step * grad, S)

        # Check improvement, backtrack if needed
        if it % 10 == 0:
            c_check = fft_convolve_self(x_new)
            if c_check.max() > cur * 1.05:
                # Backtrack: move halfway
                x_new = 0.5 * y + 0.5 * x_new
                x_new = project_simplex(x_new, S)
                step *= 0.5

        # Update momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        cap = 50.0  # Cap momentum to avoid oscillation
        t_eff = min(t_mom, cap)
        t_new_eff = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_eff * t_eff))
        y = x_new + ((t_eff - 1.0) / t_new_eff) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic restart
        if it % restart_interval == 0:
            t_mom = 1.0
            # Small perturbation
            if rng.random() < 0.3:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                y = project_simplex(y + noise, S)

    return best_local, best_score


def optimize_topk_deflation(a_init, n, S, max_time, rng,
                            k_start=4, k_end=64):
    """
    Top-k subgradient optimization with deflation schedule.
    Gradually increases k to target more peaks.
    """
    a = normalize_seq(a_init, S)
    y = a.copy()
    t_mom = 1.0
    start_time = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (4.0 * np.sqrt(n))

    while time.monotonic() - start_time < max_time:
        it += 1
        if it % 300 == 1:
            elapsed = time.monotonic() - start_time
            print("    topk-deflat iter %d, t=%.1f/%.1f, best=%.6f" %
                  (it, elapsed, max_time, best_score))

        # Deflation: increase k over time
        progress = min(1.0, it / max(1, int(max_time * 20)))
        k = int(k_start + (k_end - k_start) * progress)
        k = min(k, 128)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Sharpness increases over time
        sharpness = 2.0 + 5.0 * progress

        grad = compute_grad_topk(y, c, k, sharpness)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Adaptive step
        step = step * (1.0 / (1.0 + 0.002 * (it / 100.0)))
        step = max(step, 1e-7 * S)

        x_new = project_simplex(y - step * grad, S)

        # Backtracking check
        c_new = fft_convolve_self(x_new)
        if c_new.max() > cur * 1.02:
            step *= 0.5
            x_new = 0.5 * y + 0.5 * x_new
            x_new = project_simplex(x_new, S)

        # Momentum update
        t_eff = min(t_mom, 30.0)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_eff * t_eff))
        y = x_new + ((t_eff - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 400 == 0 and rng.random() < 0.2:
            t_mom = 1.0
            noise = rng.normal(0, 0.0005 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)

    return best_local, best_score


def coordinate_polish(a, n, S, max_time, rng):
    """
    Coordinate descent polish: for each top peak, reduce the contributing
    coordinate with the largest product term.
    """
    a = normalize_seq(a, S)
    start_time = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    # Precompute the FFT size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    while time.monotonic() - start_time < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Find top peaks (use vectorized gradient for speed)
        n_peaks = min(16, size)
        top_peaks = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        for m in top_peaks:
            m = int(m)
            # Subgradient for this peak: grad[j] = 2*a[m-j] for valid j
            j_start = max(0, m - n + 1)
            j_end = min(n, m + 1)
            
            if j_end <= j_start:
                continue
            
            # Contribution at each coordinate j: 2 * a[j] * a[m-j]
            js = np.arange(j_start, j_end)
            mjs = m - js
            valid = (mjs >= 0) & (mjs < n)
            js = js[valid]
            mjs = mjs[valid]
            
            if len(js) == 0:
                continue
            
            # Products
            products = a[js] * a[mjs]
            total_prod = products.sum()
            if total_prod < 1e-15:
                continue
            
            # Try reducing the coordinate with largest a[j]*a[m-j]
            idx_max = np.argmax(products)
            j_target = int(js[idx_max])
            
            # Try a small reduction
            for frac in [0.05, 0.02, 0.01]:
                a_trial = a.copy()
                a_trial[j_target] *= (1.0 - frac)
                a_trial = normalize_seq(a_trial, S)
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur * (1.0 - 1e-5):
                    a = a_trial
                    improved = True
                    break
        
        if not improved:
            # Try increasing coordinates instead (to redistribute)
            # Find the coordinate whose increase most reduces the peak
            m = int(top_peaks[0])
            j_start = max(0, m - n + 1)
            j_end = min(n, m + 1)
            js = np.arange(j_start, j_end)
            mjs = m - js
            valid = (mjs >= 0) & (mjs < n)
            js = js[valid]
            mjs = mjs[valid]
            
            if len(js) > 0:
                # Try increasing the coordinate with smallest a[j]+a[m-j]
                # (to reduce the product a[j]*a[m-j] by making one factor smaller
                #  relative to the other - actually this increases the product)
                # Instead, try decreasing the coordinate with largest a[j]
                for frac in [0.03, 0.01]:
                    for j_try in np.random.choice(js, min(5, len(js)), replace=False):
                        a_trial = a.copy()
                        a_trial[int(j_try)] *= (1.0 - frac)
                        a_trial = normalize_seq(a_trial, S)
                        c_trial = fft_convolve_self(a_trial)
                        if c_trial.max() < cur * (1.0 - 1e-5):
                            a = a_trial
                            improved = True
                            break
                    if improved:
                        break
    
    return best_local, best_score


def make_initializations(n, rng):
    """Create diverse asymmetric initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Exponential ramps
    for alpha in [2.0, 4.0, 6.0]:
        inits.append(np.exp(alpha * tt))
        inits.append(np.exp(alpha * (1.0 - tt)))
    
    # Power law
    for p in [1.0, 2.0, 4.0]:
        inits.append((tt + 0.05) ** p)
        inits.append((1.0 - tt + 0.05) ** p)
    
    # Raised cosine (asymmetric)
    inits.append(0.5 + 0.5 * np.cos(np.pi * tt * 2 - 0.5))
    
    # Linear + noise
    inits.append(tt + 0.1 + 0.1 * rng.random(n))
    inits.append((1.0 - tt) + 0.1 + 0.1 * rng.random(n))
    
    # Bump in middle (asymmetric)
    center = 0.4
    inits.append(np.exp(-5.0 * (tt - center) ** 2))
    
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    
    Uses multi-phase optimization:
    1. Quick exploration of initializations
    2. Deep Nesterov-accelerated optimization
    3. Top-k deflation refinement
    4. Coordinate descent polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 25.0)
    
    global_best = np.inf
    global_a = None
    
    # Get parent if available
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given_clipped = np.clip(given, 0.0, 1000.0)
            if given_clipped.sum() > 0.01:
                parent = given_clipped
    except Exception:
        pass
    
    print("Parent available: %s (len=%s)" % (parent is not None,
          len(parent) if parent is not None else 0))
    
    # Determine candidate n values
    if parent is not None:
        n_parent = len(parent)
        n_candidates = [n_parent]
        if n_parent >= 8192:
            n_candidates.append(n_parent // 2)
        n_candidates.append(min(16384, n_parent * 2))
    else:
        n_candidates = [8192]
    
    # === Phase 1: Quick exploration (15% of time) ===
    phase1_budget = min(300.0, (deadline - time.monotonic()) * 0.15)
    print("\n=== Phase 1: Exploration (%.0f sec) ===" % phase1_budget)
    
    candidates = []
    
    # First, try parent if available
    if parent is not None:
        n_p = len(parent)
        S_p = np.sqrt(2.0 * float(n_p))
        alloc_p = min(phase1_budget * 0.4, (deadline - time.monotonic()) / 3)
        print("  Optimizing parent (n=%d, %.0f sec)..." % (n_p, alloc_p))
        a_opt, score = optimize_nesterov(parent, n_p, S_p, alloc_p, rng,
                                          beta_start=10.0, beta_end=60.0)
        a_norm = normalize_seq(a_opt, S_p)
        c = fft_convolve_self(a_norm)
        obj = c.max()
        candidates.append((obj, a_norm))
        print("  Parent after quick opt: %.6f" % obj)
        if obj < global_best:
            global_best = obj
            global_a = a_norm.copy()
    
    # Try a few other initializations with remaining time
    remaining_p1 = deadline - time.monotonic() - (phase1_budget * 0.3)
    if remaining_p1 > 30:
        n_explore = n_candidates[0]
        S_exp = np.sqrt(2.0 * float(n_explore))
        inits = make_initializations(n_explore, rng)
        per_init = max(5.0, remaining_p1 / (len(inits) + 1))
        
        for i, init in enumerate(inits[:6]):  # Limit to 6 inits
            if time.monotonic() > deadline - 300:
                break
            alloc = min(per_init, (deadline - time.monotonic()) / 2)
            a_opt, score = optimize_nesterov(init, n_explore, S_exp, alloc, rng,
                                              beta_start=5.0, beta_end=50.0)
            a_norm = normalize_seq(a_opt, S_exp)
            c = fft_convolve_self(a_norm)
            obj = c.max()
            candidates.append((obj, a_norm))
            if obj < global_best:
                global_best = obj
                global_a = a_norm.copy()
                print("  Init %d: %.6f" % (i, obj))
    
    # Try different n if time allows
    for n_try in n_candidates[1:]:
        if time.monotonic() > deadline - 200:
            break
        S_try = np.sqrt(2.0 * float(n_try))
        alloc_try = min(60.0, (deadline - time.monotonic()) / 3)
        if alloc_try < 10:
            continue
        print("  Trying n=%d (%.0f sec)..." % (n_try, alloc_try))
        
        # Use exponential init for different n
        tt = np.linspace(0, 1, n_try, endpoint=False)
        init = np.exp(3.0 * tt) + 0.05
        a_opt, score = optimize_nesterov(init, n_try, S_try, alloc_try, rng,
                                          beta_start=5.0, beta_end=40.0)
        a_norm = normalize_seq(a_opt, S_try)
        c = fft_convolve_self(a_norm)
        obj = c.max()
        if obj < global_best:
            global_best = obj
            global_a = a_norm.copy()
            print("  n=%d: %.6f (NEW BEST)" % (n_try, obj))
        else:
            print("  n=%d: %.6f" % (n_try, obj))
    
    print("\nAfter Phase 1: best=%.6f (n=%d)" % (global_best,
          len(global_a) if global_a is not None else 0))
    
    # === Phase 2: Deep Nesterov optimization (50% of remaining) ===
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = max(0.0, deadline - time.monotonic())
        phase2_budget = remaining * 0.50
        
        print("\n=== Phase 2: Deep Nesterov (%.0f sec) ===" % phase2_budget)
        
        # Multiple passes with different hyperparameters
        trials = [
            {'beta_start': 20.0, 'beta_end': 100.0, 'restart': 300},
            {'beta_start': 50.0, 'beta_end': 200.0, 'restart': 200},
            {'beta_start': 100.0, 'beta_end': 500.0, 'restart': 150},
        ]
        
        per_trial = phase2_budget / len(trials)
        
        for ti, trial_params in enumerate(trials):
            if time.monotonic() > deadline - 60:
                break
            alloc = min(per_trial, deadline - time.monotonic() - 30)
            if alloc < 10:
                break
            
            print("  Trial %d (beta=%d->%d, restart=%d, %.0f sec)" %
                  (ti, trial_params['beta_start'], trial_params['beta_end'],
                   trial_params['restart'], alloc))
            
            a_opt, score = optimize_nesterov(
                global_a, n_best, S_best, alloc, rng,
                beta_start=trial_params['beta_start'],
                beta_end=trial_params['beta_end'],
                restart_interval=trial_params['restart'])
            
            a_norm = normalize_seq(a_opt, S_best)
            c = fft_convolve_self(a_norm)
            obj = c.max()
            if obj < global_best:
                global_best = obj
                global_a = a_norm.copy()
                print("  -> NEW BEST: %.6f" % obj)
            else:
                print("  -> %.6f (no improvement)" % obj)
    
    # === Phase 3: Top-k deflation (25% of remaining) ===
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = max(0.0, deadline - time.monotonic())
        phase3_budget = remaining * 0.25
        
        if phase3_budget > 20:
            print("\n=== Phase 3: Top-k Deflation (%.0f sec) ===" % phase3_budget)
            
            a_opt, score = optimize_topk_deflation(
                global_a, n_best, S_best, phase3_budget, rng,
                k_start=2, k_end=32)
            a_norm = normalize_seq(a_opt, S_best)
            c = fft_convolve_self(a_norm)
            obj = c.max()
            if obj < global_best:
                global_best = obj
                global_a = a_norm.copy()
                print("  NEW BEST: %.6f" % obj)
            else:
                print("  %.6f" % obj)
    
    # === Phase 4: Final polish (20% of remaining) ===
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = max(0.0, deadline - time.monotonic())
        
        if remaining > 10:
            print("\n=== Phase 4: Final Polish (%.0f sec) ===" % remaining)
            
            # Quick Nesterov with very high beta
            alloc = remaining * 0.5
            a_opt, score = optimize_nesterov(
                global_a, n_best, S_best, alloc, rng,
                beta_start=200.0, beta_end=1000.0,
                restart_interval=100)
            a_norm = normalize_seq(a_opt, S_best)
            c = fft_convolve_self(a_norm)
            obj = c.max()
            if obj < global_best:
                global_best = obj
                global_a = a_norm.copy()
                print("  After high-beta: %.6f" % obj)
            
            # Coordinate descent on remaining time
            remaining2 = max(0.0, deadline - time.monotonic())
            if remaining2 > 5:
                a_pol, score_p = coordinate_polish(
                    global_a, n_best, S_best, remaining2, rng)
                a_norm = normalize_seq(a_pol, S_best)
                c = fft_convolve_self(a_norm)
                obj = c.max()
                if obj < global_best:
                    global_best = obj
                    global_a = a_norm.copy()
                    print("  After CD: %.6f" % obj)
    
    # === Final verification ===
    if global_a is None:
        print("WARNING: No valid candidate found, using fallback")
        global_a = np.ones(1024, dtype=np.float64)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    
    if not np.isfinite(ver):
        print("ERROR: evaluate_sequence rejected! Trying fallback...")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
    
    print("\n=== FINAL ===")
    print("Length: %d" % len(final_list))
    print("Internal score: %.6f" % global_best)
    print("Verified score: %.6f" % ver)
    
    return final_list
