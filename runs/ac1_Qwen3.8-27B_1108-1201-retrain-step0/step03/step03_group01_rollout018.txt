import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_ramp_init(n, rng, asymmetry=1.0):

def spectral_step(a, N, top_k, alpha, asym_penalty, rng, target_max=1.5):
    """
    One spectral shaping step with asymmetry preservation.
    Returns (new_a, c, c_max).
    """
    n = len(a)
    S = a.sum()
    
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    if c_max < target_max:
        return a.copy(), c, c_max

    # Find top-k lags contributing most to the peak
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a.copy(), c, c_max
    weights = top_vals / total

    # Compute effective cosine to dampen
    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        lk = int(k)
        # Handle the symmetric equivalent lag
        sym_lag = min(lk, 2 * n - 2 - lk)
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * sym_lag / N)

    # Adaptive alpha
    excess = max(0.0, c_max - target_max) / max(c_max, 1e-10)
    alpha_curr = alpha * (0.2 + 0.8 * excess)
    alpha_curr = min(alpha_curr, 0.8)

    P_new = np.maximum(P * (1.0 - alpha_curr * cos_eff), 0.0)

    # Preserve phase, modify magnitude
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]

    # Asymmetry penalty: add small ramp to prevent symmetrization
    if asym_penalty > 0:
        ramp = np.arange(1, n + 1, dtype=np.float64) / n
        ramp = ramp - ramp.mean()
        a_new_full += asym_penalty * S * ramp

    a_new = project_simplex(a_new_full, S)
    return a_new, c, c_max


def spectral_optimize(a, N, n_steps, top_k, alpha, asym_penalty, rng, target_max=1.5,
                      noise_interval=200, noise_prob=0.3):
    """Run multiple spectral shaping steps. Returns best found."""
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_max = fft_convolve_self(a).max()
    t_start = time.monotonic()

    for it in range(n_steps):
        a_new, c, c_max = spectral_step(a, N, top_k, alpha, asym_penalty, rng, target_max)
        
        if c_max < best_max:
            best_max = c_max
            best_a = a_new.copy()

        a = a_new

        # Periodic noise injection
        if it % noise_interval == noise_interval // 2 and rng.random() < noise_prob:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

        # Early stop if target reached
        if best_max <= target_max * 1.001:
            break

    return best_a, best_max


def nesterov_subgradient(a_init, n, S, max_time, rng, top_k_init=20, top_k_max=512,
                         step_scale=1.0):
    """Nesterov-accelerated top-k subgradient method."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(y).max()
    best_local = y.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Compute top-k subgradient
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # Gradient: grad_j = 2 * sum_k w_k * a[k] where the lag structure is used
        # c_k = sum_i a_i * a_{k-i}, so dc_k/da_j = sum_i (delta_{j,i} * a_{k-i} + a_i * delta_{j,k-i})
        # = a_{k-j} + a_{k-j} (by symmetry) = 2*a_{k-j} when valid
        # Total grad_j = 2 * sum_k w_k * (a_{k-j} + a_{k-j}) / 2 ... 
        # Actually: grad_j = 2 * sum_k w_k * (a_{(k-j) mod valid} + a_{(k-j) mod valid})
        # Let's use cross-correlation via FFT for efficiency
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]
        
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = step_scale * 2.0 * S / (n * (1.0 + 0.005 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        # Check improvement
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic restart
        if it % 2000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def lse_optimize(a_init, n, S, max_time, rng, beta_init=10.0, beta_max=1000.0):
    """FISTA with log-sum-exp smooth approximation of max."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(y).max()
    best_local = y.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # LSE gradient
        c_max = c.max()
        shifted = beta * (c - c_max)
        # Clip for numerical stability
        shifted = np.clip(shifted, -500, 500)
        exp_bc = np.exp(shifted)
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        fsize = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta * 2.0 * (1.0 + 0.001 * it))
        step = max(step, 1e-8)
        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic noise
        if it % 3000 == 0 and rng.random() < 0.08:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def optimize_single(a_init, n, max_time, rng, target=1.5027):
    """Full optimization pipeline for a single sequence. Returns (best_a, best_score)."""
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()
    
    a = safe_normalize(a_init, n)
    best_a = a.copy()
    best_score = compute_score(a)

    t_start = time.monotonic()
    remaining = max_time

    # Stage 1: Spectral shaping (40% of time)
    ss_time = remaining * 0.40
    n_ss = max(100, int(ss_time / 0.003))
    top_k_ss = min(20, n // 8)
    a, sc = spectral_optimize(a, N, n_ss, top_k_ss, 0.35, 0.02, rng, target_max=target)
    a = safe_normalize(a, n)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    # Stage 2: Nesterov subgradient (30% of time)
    ng_time = remaining * 0.30
    if time.monotonic() - t_start < max_time - 5:
        a2, sc2 = nesterov_subgradient(a, n, S, min(ng_time, max_time - (time.monotonic() - t_start) - 5),
                                        rng, top_k_init=15, top_k_max=200)
        a2 = safe_normalize(a2, n)
        if sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()
            a = a2.copy()

    # Stage 3: LSE optimization (30% of time)
    lse_time = remaining * 0.30
    if time.monotonic() - t_start < max_time - 3:
        rem = max_time - (time.monotonic() - t_start) - 2
        a3, sc3 = lse_optimize(a, n, S, min(lse_time, rem), rng,
                                beta_init=5.0, beta_max=500.0)
        a3 = safe_normalize(a3, n)
        if sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()

    return best_a, best_score


def coarse_to_fine(base_a, target_n, n_base, rng):
    """Expand a base sequence to target length using interpolation + refinement."""
    # Interpolate base to target length
    x_base = np.linspace(0, 1, n_base)
    x_target = np.linspace(0, 1, target_n)
    expanded = np.interp(x_target, x_base, base_a)
    return expanded

    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 10.0

    global_best = np.inf
    global_a = None

    # Determine main length - try 8192 primarily
    main_len = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            main_len = len(given)
    except Exception:
        pass

    n = main_len
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    # --- Phase 1: Diverse exploration (35% of budget) ---
    print("=== Phase 1: Exploration ===")
    t_phase1_start = time.monotonic()
    phase1_budget = (deadline - t_phase1_start) * 0.35

    inits = []
    
    # Try the given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
    except Exception:
        pass

    # Ramped inits (key for asymmetry)
    inits.append(("ramp1", make_ramp_init(n, rng, asymmetry=1.0)))
    inits.append(("ramp2", make_ramp_init(n, rng, asymmetry=2.0)))
    inits.append(("ramp05", make_ramp_init(n, rng, asymmetry=0.5)))
    inits.append(("quad", make_quadratic_init(n, rng, asymmetry=1.0)))
    inits.append(("quad2", make_quadratic_init(n, rng, asymmetry=2.0)))
    inits.append(("cos_ramp", make_cosine_ramp_init(n, rng)))
    inits.append(("multi_freq", make_multi_freq_init(n, rng)))
    inits.append(("bin_smooth", make_binary_smooth_init(n, rng)))
    inits.append(("rand", rng.random(n) + 0.1))
    
    # Exponential ramp
    t = np.arange(n) / n
    inits.append(("exp_ramp", np.exp(2.0 * t) + 0.1 * rng.standard_normal(n)))
    # S-curve
    inits.append(("sigmoid", 1.0 / (1.0 + np.exp(-8.0 * (t - 0.4))) + 0.05 * rng.standard_normal(n)))

    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d initializations, ~%.1fs each" % (n_inits, per_init))

    for idx, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - 120:
            break
        a_norm = safe_normalize(arr, n)
        
        rem_for_init = phase1_budget - (time.monotonic() - t_phase1_start)
        if rem_for_init < 3:
            break
        # Use remaining time divided by remaining inits
        actual_time = min(per_init, rem_for_init / (n_inits - idx))
        
        t0 = time.monotonic()
        a_opt, sc = optimize_single(a_norm, n, actual_time, rng, target=1.5027)
        a_opt = safe_normalize(a_opt, n)
        sc_final = compute_score(a_opt)
        
        if sc_final < global_best:
            global_best = sc_final
            global_a = a_opt.copy()
            print("  [%s] score=%.6f (t=%.1fs)" % (name, global_best, time.monotonic() - t0))
        else:
            print("  [%s] score=%.6f (no improvement)" % (name, sc_final))

    print("=== Phase 1 done. Best: %.6f ===" % global_best)

    # --- Phase 2: Focused refinement on best (45% of budget) ---
    print("=== Phase 2: Focused refinement ===")
    t_phase2_start = time.monotonic()
    phase2_budget = (deadline - t_phase2_start) * 0.45

    if global_a is None:
        global_a = make_ramp_init(n, rng, asymmetry=1.0)
        global_a = safe_normalize(global_a, n)
        global_best = compute_score(global_a)

    a_curr = global_a.copy()
    stall_rounds = 0
    last_best = global_best

    # Alternating spectral + subgradient with progressive parameters
    n_rounds = 8
    per_round = phase2_budget / n_rounds

    for rnd in range(n_rounds):
        if time.monotonic() > deadline - 60:
            break
        
        rem = phase2_budget - (time.monotonic() - t_phase2_start)
        if rem < 5:
            break
        actual_round_time = min(per_round, rem - 5)

        # Progressive parameters
        top_k = 10 + 5 * rnd
        alpha = 0.35 / (1.0 + 0.15 * rnd)
        asym = max(0.005, 0.03 / (1.0 + 0.5 * rnd))
        n_steps = max(50, int(actual_round_time * 0.4 / 0.003))

        # Spectral shaping
        a_ss, sc_ss = spectral_optimize(a_curr, N, n_steps, top_k, alpha, asym,
                                         rng, target_max=1.50)
        a_ss = safe_normalize(a_ss, n)
        
        improved = False
        if sc_ss < global_best:
            global_best = sc_ss
            global_a = a_ss.copy()
            a_curr = a_ss.copy()
            stall_rounds = 0
            improved = True
            print("  [R%d-S] %.6f" % (rnd, global_best))

        # Subgradient polish
        rem2 = actual_round_time - (time.monotonic() - t_phase2_start) - (n_steps * 0.003)
        if rem2 > 3:
            ng_time = min(rem2, 60.0)
            a_ng, sc_ng = nesterov_subgradient(a_curr, n, S, ng_time, rng,
                                                top_k_init=20 + 10 * rnd,
                                                top_k_max=min(512, 50 + 20 * rnd))
            a_ng = safe_normalize(a_ng, n)
            if sc_ng < global_best:
                global_best = sc_ng
                global_a = a_ng.copy()
                a_curr = a_ng.copy()
                stall_rounds = 0
                improved = True
                print("  [R%d-N] %.6f" % (rnd, global_best))

        if not improved:
            stall_rounds += 1
            if stall_rounds >= 3:
                print("  [R%d] Stalled, structured restart" % rnd)
                # Structured restart: keep the "shape" but re-perturb
                asymmetry_level = rng.uniform(0.5, 2.5)
                ramp = np.arange(1, n + 1, dtype=np.float64) / n
                a_curr = a_curr * (1.0 - 0.3) + 0.3 * asymmetry_level * ramp
                a_curr = project_simplex(a_curr, S)
                # Add noise
                a_curr = project_simplex(a_curr + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
                stall_rounds = 0

    print("=== Phase 2 done. Best: %.6f ===" % global_best)

    # --- Phase 3: Fine polishing (20% of budget) ---
    print("=== Phase 3: Fine polishing ===")
    t_phase3_start = time.monotonic()
    remaining = deadline - t_phase3_start

    if global_a is not None and remaining > 10:
        # LSE with high beta
        seg1 = min(remaining * 0.35, 400.0)
        a_p1, sc_p1 = lse_optimize(global_a, n, S, seg1, rng,
                                    beta_init=50.0, beta_max=2000.0)
        a_p1 = safe_normalize(a_p1, n)
        sc1 = compute_score(a_p1)
        if sc1 < global_best:
            global_best = sc1
            global_a = a_p1.copy()
            print("  [LSE-high] %.6f" % global_best)

        # Very fine spectral with tiny alpha
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            n_fine = max(100, int(rem2 * 0.3 / 0.002))
            a_f, sc_f = spectral_optimize(global_a, N, n_fine, top_k=30, alpha=0.05,
                                           asym_penalty=0.002, rng, target_max=1.49)
            a_f = safe_normalize(a_f, n)
            sc_f2 = compute_score(a_f)
            if sc_f2 < global_best:
                global_best = sc_f2
                global_a = a_f.copy()
                print("  [SS-fine] %.6f" % global_best)

        # Final Nesterov with very high k
        rem3 = deadline - time.monotonic()
        if rem3 > 5:
            a_final, sc_final = nesterov_subgradient(global_a, n, S, rem3 - 2, rng,
                                                      top_k_init=100, top_k_max=1024)
            a_final = safe_normalize(a_final, n)
            sc_final2 = compute_score(a_final)
            if sc_final2 < global_best:
                global_best = sc_final2
                global_a = a_final.copy()
                print("  [NG-final] %.6f" % global_best)

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback verification: %.6f" % ver if np.isfinite(ver) else "  Fallback: inf")

    print("Returning sequence of length %d, score=%.6f" % (len(final_list), global_best))
    return final_list
