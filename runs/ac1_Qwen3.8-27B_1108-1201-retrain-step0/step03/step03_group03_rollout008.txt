Minimize 2*n*max(autocorr)/sum^2 using multi-peak balancing + FISTA + SA.
Key innovations over previous: (1) multi-lag equalization via water-filling,
(2) adaptive beta with much higher peaks, (3) pair-swap local search,

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_smoothed_max(c, beta):
    """Compute the LSE-smoothed max and its gradient w.r.t. input."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, np.zeros_like(c)
    w = exp_bc / total
    return c_max + np.log(total) / beta, w

def lse_gradient_wrt_a(a, c, beta):
    """Gradient of LSE-smoothed max of autocorr w.r.t. a."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(len(a))
    w = exp_bc / total
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def multi_lag_gradient(a, c, M, top_lags=None):
    """Gradient that targets top-M lags equally (water-filling style)."""
    n = len(a)
    if top_lags is None:
        top_lags = np.argsort(c)[::-1][:M]
    # Weights: softmax over the top M values
    vals = c[top_lags].copy()
    v_max = vals.max()
    weights = np.exp(vals - v_max)
    weights = weights / weights.sum()
    
    # Build full weight vector
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_lags):
        w_full[int(idx)] = weights[i]
    
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def pair_swap_search(a, c, S, rng, n_swaps=100):
    """Try random pair swaps and accept if they improve the max autocorrelation."""
    n = len(a)
    best_max = c.max()
    a_local = a.copy()
    c_local = c.copy()
    
    for _ in range(n_swaps):
        # Pick two random positions
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Try to move mass from larger to smaller
        if a_local[i] < a_local[j]:
            i, j = j, i
        # Move a small fraction
        delta = rng.random() * 0.5 * a_local[i]
        a_trial = a_local.copy()
        a_trial[i] -= delta
        a_trial[j] += delta
        if a_trial[i] < 0:
            delta = a_local[i]
            a_trial[i] = 0
            a_trial[j] += delta
        
        # Recompute autocorrelation (only the changed parts)
        # For efficiency, use FFT
        c_trial = fft_convolve_self(a_trial)
        if c_trial.max() < best_max:
            best_max = c_trial.max()
            a_local = a_trial
            c_local = c_trial
    
    return a_local, best_max

def anti_palindromic_init(n, rng, S):
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a * (S / a.sum())
    return a

def oscillatory_init(n, rng, S, K=3):
    """Initialize with a sum of K sinusoids (like a Fourier series)."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(K):
        freq = rng.integers(2, max(3, n // 4))
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.5
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    return a * (S / a.sum())

def gradient_descent_fista(a_init, n, S, max_time, rng, beta_schedule):
    """FISTA with adaptive beta from a table."""
    a = np.clip(a_init, 0, 1000).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    L = 10.0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Get current beta from schedule
        frac = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_schedule(frac)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()
        
        grad = lse_gradient_wrt_a(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)
        
        # Backtracking (simplified: check every 10 iterations)
        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() > cur * 1.05:
                L *= 2.0
            elif c_x.max() < cur * 0.99:
                L *= 0.9
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic perturbation
        if it % 500 == 0 and rng.random() < 0.2:
            noise = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)

def coordinate_swap_optimize(a, n, S, max_time, rng):

                src = j

                    a = a_trial
                    c = c_trial
                    best_max = c_trial.max()
