"""
Minimize 2*n*max_conv/sum^2 using LP-based optimization with constraint selection,
combined with subgradient polishing and multiple initializations.
Key idea: At each step, solve a small LP to find the best direction that keeps
the convolution bounded, then take a step in that direction.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def get_active_lags(b, M, eps=0.01, max_lags=50):
    """Get lags where b[k] is close to M."""
    threshold = M * (1.0 - eps)
    active = np.where(b >= threshold)[0]
    if len(active) > max_lags:
        # Keep only the top max_lags
        top_idx = np.argpartition(b, -max_lags)[-max_lags:]
        active = top_idx
    return list(active)


def solve_lp_direction(a, M, active_lags, solver=cp.ECOS):
    """Solve the LP to find optimal direction g."""
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    
    for k in active_lags:
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        i_range = np.arange(i_min, i_max + 1)
        coeffs = np.zeros(n)
        coeffs[i_range] = a[k - i_range]
        constraints.append(coeffs @ g <= M)
    
    # Add a bounding constraint to prevent unboundedness
    constraints.append(cp.sum(g) <= 10.0 * M / max(a.max(), 1e-10))
    
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if g.value is not None:
            return np.asarray(g.value, dtype=np.float64)
    except Exception as e:
        print(f"  LP solve failed: {e}")
    
    return None


def lp_optimize(a_init, n, S, max_time, rng, verbose=True):
    """LP-based optimization."""
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        b = fft_convolve_self(a)
        M = b.max()
        
        if M < best_score - 1e-12:
            best_score = M
            best_local = a.copy()
            if verbose and it % 50 == 0:
                obj = 2.0 * n * best_score / (S * S)
                print(f"  LP iter {it}: M={best_score:.6f}, obj={obj:.6f} [{time.monotonic()-start:.0f}s]")
        
        # Get active lags
        active_lags = get_active_lags(b, M, eps=0.01, max_lags=30)
        
        # Solve LP
        g = solve_lp_direction(a, M, active_lags)
        
        if g is None or g.sum() < 1e-15:
            # Fallback: use subgradient step
            k_star = np.argmax(b)
            i_min = max(0, k_star - n + 1)
            i_max = min(n - 1, k_star)
            subgrad = np.zeros(n)
            subgrad[i_min:i_max+1] = 2.0 * a[k_star - np.arange(i_min, i_max+1)]
            step = S / (n * 10.0)
            a_new = project_simplex(a - step * subgrad, S)
        else:
            # Rescale g to match sum S
            g = g * (S / g.sum())
            
            # Line search for step size
            best_t = 0.0
            best_M_new = M
            for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
                a_candidate = (1 - t) * a + t * g
                a_candidate = np.maximum(a_candidate, 0)
                b_cand = fft_convolve_self(a_candidate)
                M_cand = b_cand.max()
                if M_cand < best_M_new:
                    best_M_new = M_cand
                    best_t = t
                if M_cand < M * 0.999:
                    break
            
            if best_t > 0:
                a = (1 - best_t) * a + best_t * g
                a = np.maximum(a, 0)
                # Renormalize
                sv = a.sum()
                if sv > 1e-15:
                    a = a * (S / sv)
            else:
                # No improvement, try a smaller step or perturb
                if rng.random() < 0.3:
                    noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                    a = project_simplex(a + noise, S)
    
    return best_local, best_score


def subgradient_polish(a_init, n, S, max_time, rng, top_k=3):
    """Subgradient polishing with top-k weights."""
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        b = fft_convolve_self(a)
        M = b.max()
        
        if M < best_score - 1e-12:
            best_score = M
            best_local = a.copy()
        
        # Top-k subgradient
        top_k_actual = min(top_k, 2*n-1)
        top_indices = np.argpartition(b, -top_k_actual)[-top_k_actual:]
        top_vals = b[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        
        weights = np.zeros(2*n-1)
        weights[top_indices] = top_vals / total
        
        # Compute gradient via FFT
        fsize = 1 << (2*n - 1).bit_length()
        W = fft(weights, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = S / (n * 5.0) / (1.0 + 0.001 * it)
        a_new = project_simplex(a - step * grad / max(gnorm, 1e-15), S)
        
        # Verify improvement
        b_new = fft_convolve_self(a_new)
        if b_new.max() < M:
            a = a_new
        elif rng.random() < 0.2:
            noise = rng.normal(0, 0.0005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
    
    return best_local, best_score

import numpy as np

from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def get_active_lags(b, M, eps=0.02, max_lags=40):
    """Get lags where b[k] is within (1-eps) of M."""
    threshold = M * (1.0 - eps)
    active = np.where(b >= threshold)[0]
    if len(active) > max_lags:
        top_idx = np.argpartition(b, -max_lags)[-max_lags:]
        active = np.sort(top_idx)
    return list(active)


def solve_lp_direction(a, M, active_lags, tighten=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*tighten for active k, g >= 0."""
    n = len(a)
    M_eff = M * tighten
    g = cp.Variable(n, nonneg=True)
    constraints = []

    for k in active_lags:
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min > i_max:
            continue
        i_range = np.arange(i_min, i_max + 1)
        coeffs = np.zeros(n)
        coeffs[i_range] = a[k - i_range]
        constraints.append(cp.dot(coeffs, g) <= M_eff)

    if not constraints:
        return None

    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)

    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                val = np.asarray(g.value, dtype=np.float64)
                val = np.maximum(val, 0.0)
                if val.sum() > 1e-15:
                    return val
        except Exception:
            pass

    return None

        g = solve_lp_direction(a, M, active_lags, tighten=tighten)

def subgradient_polish(a_init, n, S, max_time, rng, top_k_start=1, top_k_max=20):
    """Nesterov-accelerated subgradient with top-k weights."""
    a = safe_normalize(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_start + it // 200)

        b = fft_convolve_self(y)
        M = b.max()

        if M < best_score - 1e-12:
            best_score = M
            best_local = y.copy()

        # Top-k weights
        top_k = min(top_k, 2 * n - 1)
        top_indices = np.argpartition(b, -top_k)[-top_k:]
        top_vals = b[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break

        weights = np.zeros(2 * n - 1)
        weights[top_indices] = top_vals / total

        # Gradient via FFT
        W = fft(weights, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = (2.0 * S) / (n * gnorm * (1.0 + 0.002 * it))
        x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic perturbation
        if it % 500 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0

    return best_local, best_score


def spectral_init(n, rng):
    """Create initial sequences using spectral methods."""
    inits = []

    # Cosine (Hann) window
    t = np.arange(n) / n
    inits.append(("hann", 0.5 * (1.0 + np.cos(2.0 * np.pi * t))))

    # Raised cosine variant
    inits.append(("raised_cos", 0.5 * (1.0 + np.cos(np.pi * (2.0 * t - 1.0)))))

    # Tent / triangular
    inits.append(("tent", 1.0 - np.abs(2.0 * t - 1.0)))

    # Flat-top window approximation
    inits.append(("flat_top", np.ones(n) * (1.0 - 0.3 * np.cos(2.0 * np.pi * t))))

    # Bimodal: two peaks
    peak_width = n // 8
    bimodal = np.zeros(n)
    bimodal[:peak_width] = np.linspace(1, 0, peak_width)
    bimodal[n - peak_width:] = np.linspace(1, 0, peak_width)
    inits.append(("bimodal", bimodal))

    # U-shape (inverse tent)
    inits.append(("u_shape", np.abs(2.0 * t - 1.0)))

    # Random
    inits.append(("random", np.abs(rng.normal(0, 1, n))))

    # Smoothed random
    raw = np.abs(rng.normal(0, 1, n))
    kernel = np.ones(16) / 16.0
    smooth_raw = np.convolve(raw, kernel, mode='same')
    inits.append(("smooth_rand", smooth_raw))

    return inits

            obj_f2 = 2.0 * global_best_n * M_f2 / (S_f * S_f)
            print(f"  Final subgrad: obj = {obj_f2:.6f} [{time.monotonic()-t0:.0f}s]")
            if obj_f2 < global_best_obj:
                global_best_obj = obj_f2
                global_best_a = a_f2.copy()

        # One more LP round on the best
        remaining3 = deadline - time.monotonic()
        if remaining3 > 10:
            t0 = time.monotonic()
            a_f3, M_f3 = lp_optimize(global_best_a, global_best_n, S_f, remaining3, rng, verbose=False)
            a_f3 = safe_normalize(a_f3, global_best_n)
            obj_f3 = 2.0 * global_best_n * M_f3 / (S_f * S_f)
            print(f"  Final LP2: obj = {obj_f3:.6f} [{time.monotonic()-t0:.0f}s]")
            if obj_f3 < global_best_obj:
                global_best_obj = obj_f3
                global_best_a = a_f3.copy()

    # === Output ===
    if global_best_a is None:
        # Fallback
        n_fb = 1024
        a_fb = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n_fb) / n_fb))
        S_fb = np.sqrt(2.0 * n_fb)
        a_fb = a_fb * (S_fb / a_fb.sum())
        global_best_a = a_fb
        global_best_n = n_fb

    final_list = [float(max(0.0, min(1000.0, x))) for x in global_best_a]
    final_obj = evaluate_sequence(final_list)
    print(f"\nFINAL: length={len(final_list)}, objective={final_obj:.6f}")

    if not np.isfinite(final_obj):
        print("  Verification failed, using fallback")
        final_list = [0.5] * 256
        final_obj = evaluate_sequence(final_list)
        print(f"  Fallback objective: {final_obj:.6f}")

    return final_list
