    c = np.real(ifft(A * A))

    n = len(a)

    w /= w.sum()

    R = fft(a_rev, fsize)

    return grad, w

        w[c.argmax()] = 1.0
    else:
        w /= w_sum
    
    # grad[m] = 2 * sum_k w[k] * a[k-m] for valid (k-m)
    # = 2 * (w * a_rev)[n-1+m] where a_rev = a[::-1]
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad, w


def mirror_descent_soft(a0, time_budget, T_start=1.0, T_end=100.0,
                        eta0=0.1, min_eta=0.001, verbose=False):

    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a * (S / a.sum())

    best_obj = obj_value(a, c)
    best_a = a.copy()

    T = T_start
    iter_no = 0
    start = time.monotonic()
    patience = 0
    
    while time.monotonic() - start < time_budget:
        elapsed_frac = min(1.0, (time.monotonic() - start) / time_budget)
        # Anneal temperature
        T = T_start + (T_end - T_start) * elapsed_frac ** 0.5

        # Normalize gradient for stability
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Momentum
        mu = 0.9
        v = mu * v + (1 - mu) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm
        
        # Multiplicative update with line search
        accepted = False
        for bt in range(10):
            eta_try = eta * (0.7 ** bt)
            a_trial = a * np.exp(-eta_try * v)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial * (S / a_trial.sum())
            
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)
            
            cur_obj = obj_value(a, c)
            if obj_trial < cur_obj * (1.0 + 1e-8):
                a = a_trial
                c = c_trial
                accepted = True
                break
        
        if not accepted:
            # Small step
            a = a * np.exp(-eta * 0.05 * v)
            a = np.maximum(a, 1e-300)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
        
        cur_obj = obj_value(a, c)
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()

            eta = min(eta * 1.05, eta0 * 3.0)

            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.5
            patience = 0
            eta = eta0
        
        iter_no += 1
        if verbose and iter_no % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, T={T:.2f}, eta={eta:.5f}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def freq_domain_polish(a0, time_budget, alpha=0.1):
    """Polish by adding frequency-domain regularization.
    Encourages |A|^2 to be flat, which makes the convolution more uniform."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    
    start = time.monotonic()
    eta = 0.05
    iter_no = 0
    
    while time.monotonic() - start < time_budget:
        # Compute soft-max gradient of max(c)
        grad_max, _ = compute_softgrad(c, a, T=50)
        
        # Compute frequency-domain penalty gradient
        # We want to minimize variance of |A|^2
        A = fft(a, 2 * n - 1)
        power = np.abs(A) ** 2
        power_mean = power.mean()
        # Gradient of ||power - power_mean||^2 w.r.t. a
        # This is complex, so use a simpler proxy:
        # The gradient of sum_k (c[k] - c.mean())^2 w.r.t. a[m]
        # = 2 * sum_k (c[k] - c.mean()) * 2*a[k-m]
        c_dev = c - c.mean()
        a_rev = a[::-1]
        fsize = 1 << (len(c) + n - 1).bit_length()
        C = fft(c_dev, fsize)
        R = fft(a_rev, fsize)
        full_conv = np.real(ifft(C * R))
        grad_var = 4.0 * full_conv[n - 1:2 * n - 1]
        
        # Combined gradient
        grad = grad_max + alpha * grad_var
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Multiplicative update
        a_trial = a * np.exp(-eta * grad)
        a_trial = np.maximum(a_trial, 1e-300)
        a_trial = a_trial * (S / a_trial.sum())
        
        c_trial = fft_convolve_self(a_trial)
        obj_trial = obj_value(a_trial, c_trial)
        
        if obj_trial < best_obj:
            best_obj = obj_trial
            best_a = a_trial.copy()
        a = a_trial
        c = c_trial
        
        iter_no += 1
    
    return best_a, best_obj

"""
Minimize 2n*max(conv(a,a))/sum(a)^2 using softmax-smoothed multiplicative gradient
descent with temperature annealing, frequency-domain variance penalty, diverse
initializations, and adaptive perturbations to escape local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.ndimage import gaussian_filter1d


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def compute_softgrad(c, a, T):
    n = len(a)
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -60, 60)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[int(c.argmax())] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad, w


def compute_var_grad(c, a):
    n = len(a)
    c_dev = c - c.mean()
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1).bit_length()
    C = fft(c_dev, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(C * R))
    return 4.0 * full_conv[n - 1:2 * n - 1]


def soft_mirror_descent(a0, time_budget, T_start, T_end, eta0, alpha_var,
                        verbose=False):
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a * (S / a.sum())
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    v = np.zeros(n)
    eta = eta0
    iter_no = 0
    start = time.monotonic()
    patience = 0
    best_iter = 0
    while time.monotonic() - start < time_budget:
        elapsed_frac = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_start + (T_end - T_start) * elapsed_frac ** 0.7
        grad_max, _ = compute_softgrad(c, a, T)
        grad = grad_max.copy()
        if alpha_var > 0.0:
            grad_var = compute_var_grad(c, a)
            grad += alpha_var * grad_var
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        mu = 0.88
        v = mu * v + (1.0 - mu) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm
        cur_obj = obj_value(a, c)
        accepted = False
        for bt in range(8):
            eta_try = eta * (0.75 ** bt)
            a_trial = a * np.exp(-eta_try * v)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)
            if obj_trial < cur_obj * (1.0 + 1e-9):
                a = a_trial
                c = c_trial
                accepted = True
                break
        if not accepted:
            a = a * np.exp(-eta * 0.05 * v)
            a = np.maximum(a, 1e-300)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
        cur_obj = obj_value(a, c)
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
            patience = 0
            eta = min(eta * 1.03, eta0 * 4.0)
        else:
            patience += 1
            if patience % 30 == 0:
                eta = max(eta * 0.92, eta0 * 0.1)
        if patience > 800:
            noise_scale = 0.04 * S / np.sqrt(n)
            a = a + np.random.randn(n) * noise_scale
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.3
            patience = 0
            best_iter = iter_no
            eta = eta0
        iter_no += 1
        if verbose and iter_no % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"      iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, T={T:.1f}, eta={eta:.5f}, t={elapsed:.1f}s")
    return best_a, best_obj


def sharp_peak_descent(a0, time_budget, eta0=0.05):
    """Focus on the single worst peak (T -> infinity limit)."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    v = np.zeros(n)
    eta = eta0
    start = time.monotonic()
    patience = 0
    while time.monotonic() - start < time_budget:
        k_max = int(np.argmax(c))
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for m in range(lo, hi + 1):
            grad[m] += 2.0 * a[k_max - m]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        v = 0.92 * v + 0.08 * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm
        cur_obj = obj_value(a, c)
        a_trial = a * np.exp(-eta * v)
        a_trial = np.maximum(a_trial, 1e-300)
        a_trial = a_trial * (S / a_trial.sum())
        c_trial = fft_convolve_self(a_trial)
        obj_trial = obj_value(a_trial, c_trial)
        if obj_trial < cur_obj:
            a = a_trial
            c = c_trial
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_a = a.copy()
            patience = 0
            eta = min(eta * 1.02, eta0 * 3.0)
        else:
            patience += 1
            if patience > 200:
                a = a + np.random.randn(n) * 0.02 * S / np.sqrt(n)
                a = np.maximum(a, 0.0)
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                v *= 0.5
                patience = 0
                eta = eta0
    return best_a, best_obj


def single_point_polish(a0, time_budget):
    """Try reducing individual large coefficients to lower peaks."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    order = np.argsort(a)[::-1]
    idx = 0
    while time.monotonic() - start < time_budget:
        i = order[idx % n]
        idx += 1
        orig = a[i]
        if orig < 1e-12:
            continue
        for factor in [0.8, 0.9, 1.1, 1.25]:
            a[i] = orig * factor
            s = a.sum()
            if s < 1e-10:
                a[i] = orig
                break
            a_norm = a * (S / s)
            c_new = fft_convolve_self(a_norm)
            obj_new = obj_value(a_norm, c_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_norm.copy()
                a = a_norm
                c = c_new
                break
            a[i] = orig
    return best_a, best_obj


def make_diverse_candidates(rng, given_a):
    cands = []
    if given_a is not None:
        cands.append(('given', given_a.copy()))
    for n in [4096, 8192]:
        S = np.sqrt(2.0 * n)
        i = np.arange(n)
        for freq in [3, 7, 11, 17, 29]:
            amp = rng.uniform(0.3, 0.7)
            phase = rng.uniform(0, 2 * np.pi)
            a = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
            a = np.maximum(a, 0.01)
            a *= S / a.sum()
            cands.append((f'cos{freq}_n{n}', a.copy()))
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * i / n) + 0.25 * np.cos(2 * np.pi * 13 * i / n + 1.0)
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
        cands.append((f'multifreq_n{n}', a.copy()))
        a = rng.random(n)
        a = gaussian_filter1d(a, sigma=8.0)
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
        cands.append((f'smoothrand_n{n}', a.copy()))
        period = 256
        a = 1.0 + 0.4 * np.sin(2 * np.pi * (np.arange(n) % period) / period)
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
        cands.append((f'periodic_n{n}', a.copy()))
    return cands
