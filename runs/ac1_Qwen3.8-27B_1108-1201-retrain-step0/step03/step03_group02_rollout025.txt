Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
Improved approach: exact top-k subgradient + FISTA with adaptive restart + backtracking.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def subgradient_toppk(a, c, k):
    """Exact subgradient using top-k active lags (uniform weights)."""
    n = len(a)
    c_len = len(c)
    top_idx = np.argsort(c)[::-1][:k]
    
    # For each top lag k_lag, the gradient of (a*a)[k_lag] w.r.t. a is:
    # grad[m] = 2 * a[k_lag - m] (for valid k_lag - m)
    # This is a reversed and shifted copy of a.
    
    grad = np.zeros(n)
    for k_lag in top_idx:
        # grad += 2 * a shifted: grad[m] += 2 * a[k_lag - m]
        # a[k_lag - m] is valid when 0 <= k_lag - m < n, i.e., k_lag - n + 1 <= m <= k_lag
        lo = max(0, k_lag - n + 1)
        hi = min(n - 1, k_lag)
        if lo <= hi:
            # m ranges from lo to hi
            # a[k_lag - m] for m in [lo, hi] is a[k_lag - lo : k_lag - hi] (reversed)
            idx = np.arange(hi, lo - 1, -1)  # a indices: k_lag - lo, k_lag - lo - 1, ..., k_lag - hi
            grad[lo:hi+1] += 2.0 * a[idx]
    
    return grad / k  # Average over top-k

Alternative: Use the FFT-based gradient computation.

The gradient of `(a*a)[k]` is `2 * a reversed and shifted by k`. In the frequency domain:
- FFT of `a` reversed: `A_conj = conj(A)` (if a is real)
- Shift by k: multiply by `exp(-2πi*k*j/fsize)`
- So the FFT of the gradient at lag k is: `2 * conj(A) * exp(-2πi*k*j/fsize)`

For a weighted sum of top-k lags:
`FFT(grad) = 2 * conj(A) * sum_{k in top} w_k * exp(-2πi*k*j/fsize)`

This is efficient if I precompute the phases.

    # weighted_phase[j] = sum_k w_k * exp(-2πi * k * j / fsize)
    # where w_k = 1/k for top k lags
    
    # For efficiency, compute directly
    j = np.arange(fsize)
    # top_idx as float
    k_float = top_idx.astype(np.float64)
    # Phase matrix: fsize x top_k (might be too large)
    # Better: use matrix-vector product
    
    # phase_top[t, j] = exp(-2πi * top_idx[t] * j / fsize)
    # weighted_sum[j] = (1/k) * sum_t phase_top[t, j]
    
    # Actually, let's just compute it:
    exponents = -2j * np.pi * np.outer(k_float, j) / fsize
    weight_sum = np.exp(exponents).mean(axis=0)  # (1/k) * sum of phase factors
    
    # grad_fft = 2 * conj(A) * weight_sum

    grad = np.real(ifft(grad_fft))[:n]

**Backtracking line search (Armijo):**
- Start with step = step_max
- Compute new point: x_new = project(y - step * grad, S)
- Evaluate objective at x_new
- If f(x_new) > f(y) - c * step * ||grad||^2 (Armijo condition), halve step and retry

- Only do backtracking every few iterations
- Use a simpler criterion: if f(x_new) > f(y), halve step (no Armijo condition)

- If f(x_new) > f(y), reject the step and try a smaller step
- If f(x_new) <= f(y), accept the step

- Adaptive restart: if ||x - x_prev|| < ||x_prev - x_prevprev||, reset momentum (set t = 1)

    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, c, top_k, top_indices):
    """Compute exact subgradient using top-k active lags (uniform weights)."""
    n = len(a)
    grad = np.zeros(n)
    for k_lag in top_indices[:top_k]:
        k_lag = int(k_lag)
        lo = max(0, k_lag - n + 1)
        hi = min(n - 1, k_lag)
        if lo <= hi:
            m_vals = np.arange(lo, hi + 1)
            a_idx = k_lag - m_vals
            grad[lo:hi + 1] += 2.0 * a[a_idx]
    return grad / float(top_k)
