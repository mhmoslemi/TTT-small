   ```python
   """
   Minimize 2*n*max_conv/sum^2 using L-BFGS-B with log-sum-exp smoothing and adaptive beta annealing.
   
   Key innovations:
   1. Smooth optimization via log-sum-exp approximation of the max function, with adaptive
      temperature annealing (increasing beta).
   2. L-BFGS-B solver which handles non-negativity constraints directly and uses quasi-Newton
      information for much faster convergence than subgradient methods.
   3. Multi-start with diverse initializations including the previous best, chirp, and binary.
   4. MPD-style peak shaving as a post-processing step.
   5. Length exploration (try multiple sequence lengths).
   """
   import numpy as np
   import time
   from scipy.fft import fft, ifft
   from scipy.optimize import minimize
   
   def project_simplex(v, S):
       """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
       n = len(v)
       if n == 0:
           return v
       v = np.asarray(v, dtype=np.float64)
       w = np.sort(v)[::-1]
       cumsum = np.cumsum(w)
       idx = np.arange(1, n + 1, dtype=np.float64)
       theta_candidates = (cumsum - S) / idx
       valid = theta_candidates <= w
       if not np.any(valid):
           rho = n
       else:
           rho = int(np.max(np.where(valid)[0]) + 1)
       theta = (cumsum[rho - 1] - S) / rho
       return np.maximum(v - theta, 0.0)
   
   def fft_convolve_self(a):
       """Full self-convolution of a via FFT. Returns array of length 2n-1."""
       n = len(a)
       size = 2 * n - 1
       fsize = 1 << (size - 1).bit_length()
       A = fft(a, fsize)
       c = np.real(ifft(A * A))
       return c[:size]
   
   def compute_lse_and_grad(a, beta):
       """Compute LSE objective value and gradient."""
       n = len(a)
       c = fft_convolve_self(a)
       c_max = c.max()
       shifted = beta * (c - c_max)
       # Numerical stability
       shifted_max = shifted.max()
       exp_shifted = np.exp(shifted - shifted_max)
       log_sum = shifted_max + np.log(exp_shifted.sum())
       obj = (c_max + log_sum / beta)
       w = exp_shifted / exp_shifted.sum()
       
       # Gradient: 2 * cross_correlation(a, w)
       N = 1 << (n + (2 * n - 1) - 2).bit_length()
       A = fft(a, N)
       W = fft(w, N)
       grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
       return obj, grad, c, c_max
   
   def lse_objective_wrapper(a, beta):
       """Objective function for scipy optimizer."""
       obj, _, _, _ = compute_lse_and_grad(a, beta)
       return obj
   
   def lse_gradient_wrapper(a, beta):
       """Gradient function for scipy optimizer."""
       _, grad, _, _ = compute_lse_and_grad(a, beta)
       return grad
   
   def run_lbfgs(a_init, n, S, beta, maxiter, ftol):
       """Run L-BFGS-B optimization at a given beta. Returns (best_a, best_score)."""
       a = np.clip(a_init, 0.0, 1000.0).copy()
       s = a.sum()
       if s < 1e-15:
           a = np.ones(n) * S / n
       else:
           a = a * (S / s)
       
       best_a = a.copy()
       best_score = np.inf
       
       # Track best during optimization using callback
       def callback(xk):
           nonlocal best_score, best_a
           c = fft_convolve_self(xk)
           score = c.max()
           if score < best_score:
               best_score = score
               best_a = xk.copy()
           return False  # Don't stop
       
       try:
           result = minimize(
               lse_objective_wrapper,
               a,
               args=(beta,),
               jac=lse_gradient_wrapper,
               method='L-BFGS-B',
               bounds=[(0.0, 1000.0)] * n,
               options={
                   'maxiter': maxiter,
                   'ftol': ftol,
                   'gtol': 1e-8,
                   'maxls': 50,
                   'maxfun': maxiter * 10,
                   'callback': callback
               }
           )
       except Exception as e:
           print(f"  L-BFGS error: {e}")
       
       # Check final result
       c = fft_convolve_self(result.x)
       if c.max() < best_score:
           best_score = c.max()
           best_a = result.x.copy()
       
       return best_a, best_score
   
   def beta_annealing_lbfgs(a_init, n, S, betas, maxiter_per_beta, ftol, time_limit):
       """Run L-BFGS-B with increasing beta. Returns best (a, score)."""
       a = a_init.copy()
       best_a = a.copy()
       best_score = fft_convolve_self(a).max()
       start = time.monotonic()
       
       for beta in betas:
           if time.monotonic() - start > time_limit:
               break
           a, score = run_lbfgs(a, n, S, beta, maxiter_per_beta, ftol)
           a = np.clip(a, 0.0, 1000.0)
           s = a.sum()
           if s < 1e-15:
               a = np.ones(n) * S / n
           else:
               a = a * (S / s)
           c = fft_convolve_self(a)
           if c.max() < best_score:
               best_score = c.max()
               best_a = a.copy()
           print(f"    beta={beta:.0f}: score={best_score:.6f}")
       
       return best_a, best_score
   
   def mpd_peak_shaving(a, n, S, n_iter, time_limit):
       """MPD-style peak shaving: iteratively reduce the peak of the convolution."""
       a = a.copy()
       a = a * (S / a.sum())
       best_a = a.copy()
       best_score = fft_convolve_self(a).max()
       start = time.monotonic()
       
       for it in range(n_iter):
           if time.monotonic() - start > time_limit:
               break
           c = fft_convolve_self(a)
           c_max = c.max()
           if c_max < best_score:
               best_score = c_max
               best_a = a.copy()
           
           # Find the peak location
           k_star = np.argmax(c)
           
           # The peak is c[k_star] = sum_i a[i]*a[k_star-i]
           # Find which pairs contribute most
           valid_i = np.arange(max(0, k_star - n + 1), min(k_star + 1, n))
           pairs = [(i, k_star - i) for i in valid_i]
           contributions = np.array([a[i] * a[k_star - i] for i, _ in pairs])
           
           # Reduce the top contributing pairs
           top_frac = min(0.1, 5.0 / len(pairs))
           n_top = max(1, int(len(pairs) * top_frac))
           top_idx = np.argsort(contributions)[::-1][:n_top]
           
           # For each top pair, reduce the larger element
           reduction = 0.01 * S / n
           for idx in top_idx:
               i = pairs[idx][0]
               j = pairs[idx][1]
               if a[i] > a[j]:
                   a[i] = max(0, a[i] - reduction)
               else:
                   a[j] = max(0, a[j] - reduction)
           
           # Redistribute to positions that don't contribute to the peak
           deficit = S - a.sum()
           if deficit > 1e-10:
               # Find positions that don't contribute to c[k_star]
               peak_positions = set()
               for i, j in pairs:
                   peak_positions.add(i)
                   peak_positions.add(j)
               free_positions = [p for p in range(n) if p not in peak_positions]
               if free_positions:
                   # Distribute evenly to free positions
                   per_val = deficit / len(free_positions)
                   for p in free_positions:
                       a[p] += per_val
                   a = np.clip(a, 0.0, 1000.0)
               else:
                   # If no free positions, use projection
                   a = project_simplex(a, S)
       
       return best_a, best_score
   
   def safe_normalize(a, n):
       """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
       a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
       s = a.sum()
       S = np.sqrt(2.0 * float(n))
       if s <= 1e-15:
           return np.ones(n) * S / n
       return a * (S / s)
   
   def make_chirp_init(n, rng):
       """Chirp initialization with good frequency spread."""
       t = np.arange(n) / n
       # Quadratic chirp (Gabor-like)
       a = 0.5 + 0.35 * np.sin(2 * np.pi * (3 * t + 0.7 * t * t)) + 0.15 * rng.standard_normal(n) * 0.1
       a = np.clip(a, 0.01, 1.0)
       # Apply a Gaussian window to reduce edge effects
       window = np.exp(-0.5 * ((t - 0.5) * 3.0) ** 2)
       a = a * (0.3 + 0.7 * window)
       return a
   
   def make_binary_init(n, rng):
       """Binary-inspired initialization."""
       a = (rng.random(n) > 0.5).astype(np.float64)
       # Add small perturbation to break symmetries
       a = a + 0.05 * rng.standard_normal(n)
       a = np.clip(a, 0.01, 1.0)
       return a
   
   def make_walsh_init(n, rng):
       """Walsh-Hadamard inspired initialization."""
       # Use a random row of the Walsh-Hadamard matrix (approximated)
       a = np.zeros(n)
       # Use multiple random frequencies
       n_freqs = 3
       for k in range(n_freqs):
           freq = rng.integers(1, n)
           phase = rng.random() * np.pi
           a += np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
       a = 0.5 + 0.3 * a / (n_freqs + 0.1)
       a = np.clip(a, 0.05, 1.0)
       return a
   
   def score_from_conv(c):
       """Compute normalized score from convolution (assuming sum=1)."""
       return c.max()
   ```

```python
"""
Minimize 2*n*max_conv/sum^2 using L-BFGS-B with log-sum-exp smoothing and adaptive beta annealing.

Key innovations over previous approach:
1. Smooth optimization via log-sum-exp (LSE) approximation of the max function, enabling use of
   quasi-Newton methods with proper line search (L-BFGS-B) instead of noisy subgradient methods.
2. Adaptive beta annealing: start with smooth (low beta) and progressively sharpen (high beta),
   allowing the optimizer to navigate the landscape efficiently at each scale.
3. L-BFGS-B handles non-negativity constraints directly via box constraints, ensuring feasibility.
4. MPD-style (Minimum Peak Distortion) peak shaving as a complementary post-processing step.
5. Multi-start with diverse initializations and length exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_lse_and_grad(a, beta):
    """Compute LSE objective value and gradient for the max-convolution problem."""
    n = len(a)
    c = fft_convolve_self(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    # Extra numerical stability
    shifted_max = shifted.max()
    exp_shifted = np.exp(shifted - shifted_max)
    log_sum = shifted_max + np.log(exp_shifted.sum())
    obj = c_max + log_sum / beta
    w = exp_shifted / exp_shifted.sum()

    # Gradient: 2 * cross_correlation(a, w)
    # (a convolve-correlate w)[j] = sum_i a[i] * w[i+j]
    # Need FFT size >= (n-1) + (2n-1) - 1 + 1 = 3n - 2
    N = 1 << (3 * n - 2).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return obj, grad, c, c_max


def lse_objective(a, beta):
    """Objective function wrapper for scipy optimizer."""
    obj, _, _, _ = compute_lse_and_grad(a, beta)
    return obj


def lse_gradient(a, beta):
    """Gradient function wrapper for scipy optimizer."""
    _, grad, _, _ = compute_lse_and_grad(a, beta)
    return grad


def run_lbfgs_at_beta(a_init, n, S, beta, maxiter, ftol, gtol):
    """Run L-BFGS-B at a fixed beta. Returns (best_a, best_maxconv)."""
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_state = {'a': a.copy(), 'score': fft_convolve_self(a).max()}
    call_count = [0]

    def callback(xk):
        call_count[0] += 1
        if call_count[0] % 50 == 1:
            c = fft_convolve_self(xk)
            sc = c.max()
            if sc < best_state['score']:
                best_state['score'] = sc
                best_state['a'] = xk.copy()
        return False

    try:
        result = minimize(
            lse_objective,
            a,
            args=(beta,),
            jac=lse_gradient,
            method='L-BFGS-B',
            bounds=[(0.0, 1000.0)] * n,
            options={
                'maxiter': maxiter,
                'ftol': ftol,
                'gtol': gtol,
                'maxls': 50,
                'maxfun': maxiter * 20,
                'callback': callback
            }
        )
    except Exception as e:
        print(f"    L-BFGS exception at beta={beta}: {e}")
        result = None

    if result is not None:
        c = fft_convolve_self(result.x)
        if c.max() < best_state['score']:
            best_state['score'] = c.max()
            best_state['a'] = result.x.copy()

    return best_state['a'], best_state['score']

def mpd_peak_shaving(a, n, S, n_iter, time_limit, aggr=0.005):
    """
    MPD-style peak shaving: iteratively reduce the peak of the convolution
    by reducing the largest contributing pairs and redistributing mass.
    """
    a = safe_normalize(a, n)
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    start = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - start > time_limit:
            break

        c = fft_convolve_self(a)
        c_max = c.max()
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        # Find peak location(s)
        k_star = np.argmax(c)

        # Find contributing pairs
        i_min = max(0, k_star - n + 1)
        i_max = min(k_star, n - 1)
        valid_i = np.arange(i_min, i_max + 1)
        if len(valid_i) == 0:
            break

        j_vals = k_star - valid_i
        contributions = a[valid_i] * a[j_vals]
        total = contributions.sum()
        if total < 1e-30:
            break

        # Identify top contributing pairs
        n_top = max(1, min(len(valid_i), int(0.05 * len(valid_i))))
        top_local = np.argsort(contributions)[::-1][:n_top]

        # Determine which positions are "peak positions"
        peak_mask = np.zeros(n, dtype=bool)
        for idx in top_local:
            i = valid_i[idx]
            j = j_vals[idx]
            peak_mask[i] = True
            peak_mask[j] = True

        # Find free positions (not contributing to peak)
        free = np.where(~peak_mask)[0]
        if len(free) == 0:
            free = np.where(~peak_mask | (a < np.median(a)))[0]
        if len(free) == 0:
            free = np.array([rng_free(n)])

        # Reduce top contributing elements
        reduction = aggr * S / n
        deficit = 0.0
        for idx in top_local:
            i = valid_i[idx]
            j = j_vals[idx]
            if a[i] >= a[j]:
                red = min(reduction, a[i] * 0.3)
                a[i] -= red
                deficit += red
            else:
                red = min(reduction, a[j] * 0.3)
                a[j] -= red
                deficit += red

        # Redistribute to free positions
        if deficit > 1e-15 and len(free) > 0:
            per_val = deficit / len(free)
            a[free] += per_val
            a = np.maximum(a, 0.0)

    return best_a, best_score


def rng_free(n):
    """Return a random position (top-level to avoid closure)."""
    return int(np.random.randint(0, n))


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def get_score(a):
    """Compute the actual score for a normalized sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)

def make_fibonacci_init(n, rng):
    """Fibonacci-like sequence (has interesting autocorrelation properties)."""
    a = np.zeros(n)

def gradient_check(a, beta, n_check=5):
    """Verify gradient correctness with finite differences."""
    obj, grad, _, _ = compute_lse_and_grad(a, beta)
    eps = 1e-5
    for i in range(min(n_check, len(a))):
        a_plus = a.copy()
        a_minus = a.copy()
        a_plus[i] += eps
        a_minus[i] -= eps
        f_plus, _, _, _ = compute_lse_and_grad(a_plus, beta)
        f_minus, _, _, _ = compute_lse_and_grad(a_minus, beta)
        fd = (f_plus - f_minus) / (2 * eps)
        if abs(fd - grad[i]) > 1e-3 * max(1.0, abs(grad[i])):
            return False, i, fd, grad[i]
    return True, -1, 0, 0

    # Determine sequence length
    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 128:
            n = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))

    print(f"=" * 60)
    print(f"n={n}, S={S:.4f}, budget={budget_s}s")
    print(f"=" * 60)

    # --- Phase 0: Verify gradient (quick check) ---
    a_test = safe_normalize(np.ones(n) * 0.5 + 0.1 * rng.standard_normal(n), n)
    ok, idx, fd, gg = gradient_check(a_test, 10.0, n_check=3)
    print(f"Gradient check: {'PASS' if ok else 'FAIL'} (idx={idx}, fd={fd:.6f}, gg={gg:.6f})")
    if not ok:
        print("  WARNING: Gradient mismatch, continuing anyway")

    # --- Phase 1: Multi-start optimization ---
    print(f"\n{'=' * 60}")
    print(f"PHASE 1: Multi-start L-BFGS-B + LSE optimization")
    print(f"{'=' * 60}")

    inits = []

    # From previous best
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("prev_best", np.clip(given, 0, 1000).astype(np.float64)))
    except Exception:
        pass

    # Chirp variants
    inits.append(("chirp1", make_chirp_init(n, rng)))
    inits.append(("chirp2", make_chirp_init(n, rng)))

    # Binary
    inits.append(("binary1", make_binary_init(n, rng)))
    inits.append(("binary2", make_binary_init(n, rng)))

    # Walsh
    inits.append(("walsh", make_walsh_init(n, rng)))

    # Segments
    inits.append(("seg5", make_segment_init(n, 5, rng)))
    inits.append(("seg7", make_segment_init(n, 7, rng)))

    # Fibonacci-like
    inits.append(("fib", make_fibonacci_init(n, rng)))

    # Random smooth (multi-sine)
    t = np.arange(n) / n
    a_rand = 0.5 + 0.2 * np.sin(2 * np.pi * 2 * t) + 0.15 * np.sin(2 * np.pi * 5 * t + 0.7) \
        + 0.1 * np.sin(2 * np.pi * 11 * t + 1.3) + 0.05 * rng.standard_normal(n)
    inits.append(("rand_sin", np.clip(a_rand, 0.01, 1.0)))

    n_inits = len(inits)
    phase1_time = (deadline - time.monotonic()) * 0.55
    time_per_init = phase1_time / n_inits
    print(f"  {n_inits} initializations, ~{time_per_init:.0f}s each (total {phase1_time:.0f}s)")

    # Beta schedule for main optimization
    betas_main = [1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]

    for name, arr in inits:
        if time.monotonic() > deadline - 60:
            break

        a_norm = safe_normalize(arr, n)
        c0 = fft_convolve_self(a_norm)
        init_score = c0.max()

        # Quick check of actual score
        actual_init_score = get_score(a_norm)
        print(f"  [{name}] init: maxconv={init_score:.4f}, score={actual_init_score:.6f}")

        if actual_init_score < global_best:
            global_best = actual_init_score
            global_a = a_norm.copy()
            global_n = n

        # L-BFGS-B with beta annealing
        maxiter = max(200, int(time_per_init * 0.6 / 0.0005))
        maxiter = min(maxiter, 30000)

        a_opt, score_opt = beta_annealing(
            a_norm, n, S, betas_main,
            maxiter_per_beta=maxiter,
            ftol=1e-10, gtol=1e-7,
            time_limit=time_per_init * 0.6
        )

        a_opt = safe_normalize(a_opt, n)
        actual_score = get_score(a_opt)
        print(f"  [{name}] L-BFGS: score={actual_score:.6f}")

        if actual_score < global_best:
            global_best = actual_score
            global_a = a_opt.copy()
            global_n = n

        # MPD peak shaving on the result
        rem = time_per_init * 0.25
        if rem > 2:
            a_mpd, score_mpd = mpd_peak_shaving(a_opt, n, S,
                                                 n_iter=50000, time_limit=rem,
                                                 aggr=0.003)
            a_mpd = safe_normalize(a_mpd, n)
            mpd_actual = get_score(a_mpd)
            if mpd_actual < global_best:
                global_best = mpd_actual
                global_a = a_mpd.copy()
                global_n = n
                print(f"  [{name}] MPD: score={mpd_actual:.6f}")

    print(f"\n  Phase 1 best: {global_best:.6f}")

    # --- Phase 2: Focused optimization on the best ---
    print(f"\n{'=' * 60}")
    print(f"PHASE 2: Focused optimization")
    print(f"{'=' * 60}")

    if global_a is not None:
        phase2_time = (deadline - time.monotonic()) * 0.25
        if phase2_time > 10:
            # Higher beta schedule
            betas_high = [50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]
            maxiter2 = max(500, int(phase2_time / (len(betas_high) * 0.0005)))
            maxiter2 = min(maxiter2, 50000)

            a_f, score_f = beta_annealing(
                global_a, n, S, betas_high,
                maxiter_per_beta=maxiter2,
                ftol=1e-12, gtol=1e-9,
                time_limit=phase2_time * 0.7
            )
            a_f = safe_normalize(a_f, n)
            actual_f = get_score(a_f)
            if actual_f < global_best:
                global_best = actual_f
                global_a = a_f.copy()
                global_n = n
                print(f"  Phase 2 L-BFGS high-beta: score={global_best:.6f}")

            # MPD with smaller step
            rem2 = phase2_time * 0.3
            if rem2 > 3:
                a_mpd2, _ = mpd_peak_shaving(global_a, n, S,
                                              n_iter=100000, time_limit=rem2,
                                              aggr=0.001)
                a_mpd2 = safe_normalize(a_mpd2, n)
                mpd2_score = get_score(a_mpd2)
                if mpd2_score < global_best:
                    global_best = mpd2_score
                    global_a = a_mpd2.copy()
                    global_n = n
                    print(f"  Phase 2 MPD fine: score={global_best:.6f}")

    # --- Phase 3: Diversification restarts ---
    print(f"\n{'=' * 60}")
    print(f"PHASE 3: Diversification")
    print(f"{'=' * 60}")

    if global_a is not None:
        phase3_time = (deadline - time.monotonic()) * 0.15
        if phase3_time > 10:
            n_restarts = min(5, int(phase3_time / 15))
            time_per_restart = phase3_time / max(1, n_restarts)

            for r in range(n_restarts):
                if time.monotonic() > deadline - 30:
                    break

                # Create perturbed version
                pert_scale = 0.02 + 0.01 * r
                a_pert = global_a + pert_scale * S / np.sqrt(n) * rng.standard_normal(n)
                a_pert = safe_normalize(a_pert, n)

                # Quick L-BFGS with moderate beta
                betas_r = [5.0, 20.0, 100.0, 500.0]
                maxiter_r = max(100, int(time_per_restart * 0.6 / 0.0005))
                maxiter_r = min(maxiter_r, 10000)

                a_r, score_r = beta_annealing(
                    a_pert, n, S, betas_r,
                    maxiter_per_beta=maxiter_r,
                    ftol=1e-10, gtol=1e-7,
                    time_limit=time_per_restart * 0.7
                )
                a_r = safe_normalize(a_r, n)
                actual_r = get_score(a_r)
                if actual_r < global_best:
                    global_best = actual_r
                    global_a = a_r.copy()
                    global_n = n
                    print(f"  Restart {r}: score={global_best:.6f}")
                else:
                    print(f"  Restart {r}: score={actual_r:.6f} (no improvement)")

    # --- Phase 4: Final polish ---
    print(f"\n{'=' * 60}")
    print(f"PHASE 4: Final polish")
    print(f"{'=' * 60}")

    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            # Very high beta L-BFGS
            rem1 = remaining * 0.4
            betas_final = [200.0, 500.0, 1000.0, 3000.0]
            maxiter_f = max(200, int(rem1 / (len(betas_final) * 0.0005)))
            maxiter_f = min(maxiter_f, 30000)

            a_fin, _ = beta_annealing(
                global_a, n, S, betas_final,
                maxiter_per_beta=maxiter_f,
                ftol=1e-13, gtol=1e-10,
                time_limit=rem1
            )
            a_fin = safe_normalize(a_fin, n)
            fin_score = get_score(a_fin)
            if fin_score < global_best:
                global_best = fin_score
                global_a = a_fin.copy()
                global_n = n
                print(f"  Final L-BFGS: score={global_best:.6f}")

            # Very fine MPD
            rem2 = deadline - time.monotonic()
            if rem2 > 5:
                a_fm, _ = mpd_peak_shaving(global_a, n, S,
                                            n_iter=200000, time_limit=rem2,
                                            aggr=0.0005)
                a_fm = safe_normalize(a_fm, n)
                fm_score = get_score(a_fm)
                if fm_score < global_best:
                    global_best = fm_score
                    global_a = a_fm.copy()
                    global_n = n
                    print(f"  Final MPD: score={global_best:.6f}")

    # --- Verification ---
    print(f"\n{'=' * 60}")
    print(f"VERIFICATION")
    print(f"{'=' * 60}")

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)
        global_n = 256

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
        global_n = 256

    # Ensure all values are valid floats
    global_a = np.nan_to_num(global_a, nan=0.0, posinf=1000.0, neginf=0.0)
    global_a = np.maximum(global_a, 0.0)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)

    if np.isfinite(ver):
        print(f"  Verified score: {ver:.6f} (n={len(global_a)})")
        print(f"  Internal estimate: {global_best:.6f}")
        if ver > global_best * 1.01:
            print(f"  WARNING: Verified score differs from internal estimate!")
    else:
        print(f"  VERIFICATION FAILED (got inf)")
        print(f"  Fallback to uniform sequence of length 256")
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print(f"  Fallback score: {ver:.6f}")

    total_time = time.monotonic() - (deadline - budget_s + 25.0)
    print(f"\n  Total runtime: {total_time:.1f}s")
    print(f"  Returning sequence of length {len(final_list)}")

    return final_list
```
