import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def apply_A_p(v, p, n):
    """Apply A_p = sum_k p_k S_k to vector v.
    p is a probability vector over lags 0..n-1.
    """
    # (A_p v)[i] = p_0 * v[i] + sum_{k=1} p_k * (v[i+k] + v[i-k])
    # This is a convolution with kernel h where h[0]=p[0], h[k]=p[k], h[-k]=p[k]
    # Using FFT:
    # Create h of length 2n-1: h[0]=p[0], h[k]=p[k] for k=1..n-1, h[2n-1-k]=p[k] for k=1..n-1
    h = np.zeros(2 * n - 1)
    h[0] = p[0]
    if n > 1:
        h[1:n] = p[1:n]
        h[n:] = p[1:n][::-1]
    
    fsize = 1 << (2 * n - 2).bit_length()
    H = fft(h, fsize)
    V = fft(v, fsize)
    result = np.real(ifft(H * V))[:n]
    return result

def solve_qp_simplex(p, n, S, max_iter=500, tol=1e-10):
    """Solve min a^T A_p a s.t. a >= 0, sum(a) = S using FISTA.

    a = np.ones(n) * S / n
    y = a.copy()
    t = 1.0
    step = 0.9 / 2.0  # Slightly less than 1/L for safety
    
    for it in range(max_iter):
        grad = 2.0 * apply_A_p(y, p, n)
        x_new = project_simplex(y - step * grad, S)

            diff = np.linalg.norm(x_new - a)

        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return a

```python
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_cross_corr(r, a):
    """Compute cross-correlation (r ⋆ a)[j] = sum_m r[j+m]*a[m] for j=0..len(a)-1"""
    n = len(a)
    # Need N >= len(r) + n - 1
    N = 1 << (len(r) + n - 2).bit_length()
    R = fft(r, N)
    A = fft(a, N)
    cc = np.real(ifft(np.conj(R) * A))
    return cc[:n]

def compute_qp_gradient(a, p):
    """Gradient of sum_k p[k]*c[k] where c = convolve(a,a)."""
    return 2.0 * fft_cross_corr(p, a)

def solve_qp_fista(p, n, S, a_init, max_iter=2000, tol=1e-9):
    """Solve min sum_k p[k]*c[k] s.t. a>=0, sum(a)=S using FISTA."""
    a = np.clip(a_init, 0, 1000).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    step = 0.4  # 1/L where L <= 2, use 0.4 for safety
    
    prev_obj = np.inf
    for it in range(max_iter):
        grad = compute_qp_gradient(y, p)
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Early stopping
        if it % 500 == 0:
            obj = np.dot(p, fft_convolve_self(a))
            if abs(prev_obj - obj) < tol:
                break
            prev_obj = obj
    
    return a
```

        grad = compute_grad(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        alpha = step
        accepted = False
        for ls in range(line_search_depth):
            x_trial = project_simplex(y - alpha * grad, S)
            c_trial = fft_convolve_self(x_trial)
            if c_trial.max() < cur_max - 1e-10 * alpha * gnorm * gnorm / max(n, 1):
                accepted = True
                break
            alpha *= 0.5

        if not accepted:
            x_new = project_simplex(y - alpha * grad, S)
            c_trial_max = fft_convolve_self(x_new).max()
            if c_trial_max >= cur_max:
                # Accept anyway with tiny step to avoid stalling
                alpha *= 0.5
                x_new = project_simplex(y - alpha * grad, S)

        # Nesterov update
        if momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y_candidate = x_new + ((t - 1.0) / t_new) * (x_new - a)
            y_candidate = np.maximum(y_candidate, 0.0)
            # Restart if momentum hurts
            c_y = fft_convolve_self(y_candidate)
            if c_y.max() > cur_max * 1.005:
                y = x_new
                t = 1.0
            else:
                y = y_candidate
                t = t_new
        else:
            y = x_new
            t = 1.0

        a = x_new
        step = alpha * 1.01  # Slightly increase for next iter

        # Periodic perturbation
        if it % perturb_interval == 0 and rng.random() < perturb_prob:
            noise = rng.normal(0, perturb_scale * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
            step = step_init * 0.7

        # Safety: ensure y stays on simplex
        y = project_simplex(y, S)

    return best_a, best_obj


def run_top1_polish(a_init, n, S, max_time, rng, label=""):
    """Very focused top-1 subgradient polishing with tiny steps."""
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    it = 0
    step = S / (4.0 * n)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_obj - 1e-13:
            best_obj = cur_max
            best_a = a.copy()
            step = min(step * 1.01, S / (2.0 * n))
        else:
            step *= 0.98

        # Focus on top 1-2 lags
        top_k = 1 if it > 500 else 2
        w = toppk_w(c, top_k)
        grad = compute_grad(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        x_new = project_simplex(a - step * grad / max(gnorm, 1e-15) * S / n, S)
        c_new = fft_convolve_self(x_new)

        if c_new.max() < cur_max:
            a = x_new
        elif rng.random() < 0.1:
            # Tiny random nudge
            noise = rng.normal(0, 0.0001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=2385, **kwargs):
    """
    Minimize 2*n*max_conv/sum^2 over non-negative sequences.
    Multi-phase: explore -> exploit -> polish, with adaptive beta,
    backtracking line search, and Nesterov momentum with restart.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s - 25.0)
    n = 8192
    S = np.sqrt(2.0 * n)

    # Get starting sequence
    given_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given_a = np.clip(given, 0, 1000)
            n = len(given_a)
            S = np.sqrt(2.0 * n)
    except Exception:
        pass

    if given_a is None:
        given_a = np.ones(n) * 0.5

    a_start = normalize_seq(given_a, n)
    c_start = fft_convolve_self(a_start)
    initial_score = c_start.max()
    obj_start = 2.0 * n * initial_score / (S * S)
    print("Initial: n=%d, max_conv=%.6f, objective=%.6f" % (n, initial_score, obj_start))

    global_best = initial_score
    global_a = a_start.copy()

    # === Phase 1: Explore (35% of time) ===
    remaining = deadline - time.monotonic()
    phase1_time = 0.35 * remaining
    if phase1_time > 10:
        print("Phase 1: Explore (beta 20->200) [%.0f sec]" % phase1_time)
        t0 = time.monotonic()
        a_p1, obj_p1 = run_optimize_phase(
            a_start, n, S, phase1_time, rng,
            beta_start=20, beta_end=200,
            step_init=0.35, momentum=True,
            perturb_interval=2000, perturb_prob=0.25, perturb_scale=0.008,
            line_search_depth=4, label="P1"
        )
        a_p1 = normalize_seq(a_p1, n)
        c_p1 = fft_convolve_self(a_p1)
        s1 = c_p1.max()
        if s1 < global_best:
            global_best = s1
            global_a = a_p1.copy()
            print("  P1: max_conv=%.6f obj=%.6f [%.0fs]" % (s1, 2*n*s1/S**2, time.monotonic()-t0))
        else:
            print("  P1: no improvement (best=%.6f) [%.0fs]" % (global_best, time.monotonic()-t0))

    # === Phase 2: Exploit (35% of time) ===
    remaining = deadline - time.monotonic()
    phase2_time = 0.35 * remaining
    if phase2_time > 10:
        print("Phase 2: Exploit (beta 200->1000) [%.0f sec]" % phase2_time)
        t0 = time.monotonic()
        a_p2, obj_p2 = run_optimize_phase(
            global_a, n, S, phase2_time, rng,
            beta_start=200, beta_end=1000,
            step_init=0.15, momentum=True,
            perturb_interval=4000, perturb_prob=0.15, perturb_scale=0.003,
            line_search_depth=3, label="P2"
        )
        a_p2 = normalize_seq(a_p2, n)
        c_p2 = fft_convolve_self(a_p2)
        s2 = c_p2.max()
        if s2 < global_best:
            global_best = s2
            global_a = a_p2.copy()
            print("  P2: max_conv=%.6f obj=%.6f [%.0fs]" % (s2, 2*n*s2/S**2, time.monotonic()-t0))
        else:
            print("  P2: no improvement (best=%.6f) [%.0fs]" % (global_best, time.monotonic()-t0))

    # === Phase 3: Polish (20% of time) ===
    remaining = deadline - time.monotonic()
    phase3_time = 0.20 * remaining
    if phase3_time > 5:
        print("Phase 3: Top-1 Polish [%.0f sec]" % phase3_time)
        t0 = time.monotonic()
        a_p3, obj_p3 = run_top1_polish(global_a, n, S, phase3_time, rng, label="P3")
        a_p3 = normalize_seq(a_p3, n)
        c_p3 = fft_convolve_self(a_p3)
        s3 = c_p3.max()
        if s3 < global_best:
            global_best = s3
            global_a = a_p3.copy()
            print("  P3: max_conv=%.6f obj=%.6f [%.0fs]" % (s3, 2*n*s3/S**2, time.monotonic()-t0))
        else:
            print("  P3: no improvement (best=%.6f) [%.0fs]" % (global_best, time.monotonic()-t0))

    # === Phase 4: Diversified restarts (10% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("Phase 4: Diversified exploration [%.0f sec]" % remaining)
        n_variants = 4
        per_variant = remaining * 0.8 / n_variants

        for v in range(n_variants):
            if time.monotonic() > deadline - 30:
                break
            if v == 0:
                perturbed = global_a + rng.normal(0, 0.004 * S / np.sqrt(n), n)
            elif v == 1:
                # Smooth low-frequency perturbation
                freq = np.random.choice([3, 7, 11, 13], size=3, replace=False)
                noise = np.zeros(n)
                for f in freq:
                    noise += 0.5 * np.sin(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 2*np.pi))
                perturbed = global_a + 0.003 * S / np.sqrt(n) * noise
            elif v == 2:
                # Multiplicative perturbation
                perturbed = global_a * (1.0 + rng.normal(0, 0.008, n))
            else:
                # Block perturbation: add constant to random blocks
                perturbed = global_a.copy()
                for _ in range(5):
                    block_start = rng.integers(0, n - 256)
                    block_len = rng.integers(64, 512)
                    perturbed[block_start:block_start+block_len] += rng.normal(0, 0.005 * S / np.sqrt(n))

            perturbed = normalize_seq(perturbed, n)
            c_v = fft_convolve_self(perturbed)
            base_obj = c_v.max()
            print("  Variant %d start: obj=%.6f" % (v, 2*n*base_obj/S**2))

            a_v, obj_v = run_optimize_phase(
                perturbed, n, S, per_variant, rng,
                beta_start=100, beta_end=800,
                step_init=0.2, momentum=True,
                perturb_interval=3000, perturb_prob=0.1, perturb_scale=0.002,
                line_search_depth=3, label="V%d" % v
            )
            a_v = normalize_seq(a_v, n)
            c_v2 = fft_convolve_self(a_v)
            sv = c_v2.max()
            if sv < global_best:
                global_best = sv
                global_a = a_v.copy()
                print("  Variant %d: max_conv=%.6f obj=%.6f" % (v, sv, 2*n*sv/S**2))

    # === Final ultra-fine polish ===
    remaining = deadline - time.monotonic()
    if remaining > 5:
        print("Final polish [%.0f sec]" % remaining)
        t0 = time.monotonic()
        a_f, obj_f = run_top1_polish(global_a, n, S, remaining, rng, label="Final")
        a_f = normalize_seq(a_f, n)
        c_f = fft_convolve_self(a_f)
        sf = c_f.max()
        if sf < global_best:
            global_best = sf
            global_a = a_f.copy()
            print("  Final: max_conv=%.6f obj=%.6f [%.0fs]" % (sf, 2*n*sf/S**2, time.monotonic()-t0))

    # === Output ===
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    final_obj = 2.0 * n * global_best / (S * S)
    print("FINAL: n=%d, max_conv=%.6f, objective=%.6f" % (len(final_list), global_best, final_obj))

    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified objective: %.6f" % ver)
    else:
        print("Verification failed, using fallback")
        final_list = [0.5] * 256

    return final_list
