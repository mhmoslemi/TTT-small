Specifically, if we linearize the autocorrelation around the current point a:
c(a + delta) ≈ c(a) + J(a) delta + (1/2) delta^T H(a) delta

where J(a) is the Jacobian of c with respect to a, and H(a) is the Hessian.

The first-order approximation gives:
c(a + delta)[k] ≈ c(a)[k] + sum_i (dc[ k]/da[i]) delta[i]

The gradient of c[k] with respect to a[i] is:

"""
Minimize 2*n*max_conv/sum^2 using adaptive FISTA-LSE + spectral shaping + multi-start.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient(y, w_vec, n):
    """Compute gradient = 2 * cross_correlation(w_vec, y)."""
    fsize = 1 << (len(w_vec) + n - 2).bit_length()
    W = fft(w_vec, fsize)
    A = fft(y, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fista_lse(a_init, n, S, max_time, rng, beta_init=30.0, beta_max=400.0,
              top_k_init=5, top_k_max=300, top_k_boost=4.0,
              restart_interval=500, restart_prob=0.15):
    """FISTA with adaptive LSE + top-k boost."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive parameters
        beta = min(beta_max, beta_init + (beta_max - beta_init) * min(1.0, it / 5000.0))
        top_k = min(top_k_max, top_k_init + it // 30)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        # Compute LSE weights with top-k boost
        c_max = c.max()
        scaled = np.clip(beta * (c - c_max), -500, 0.0)
        exp_bc = np.exp(scaled)
        
        if top_k > 1:
            top_idx = np.argpartition(c, -top_k)[-top_k:]
            exp_bc[top_idx] *= top_k_boost
        
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w_vec = exp_bc / total
        
        # Gradient
        grad = compute_gradient(y, w_vec, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size
        L = 2.0 * beta * S * w_vec.max()
        step = 1.0 / L
        step = max(step, 1e-10)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)

                best_score = c_x.max()
                best_local = x_new.copy()

        momentum = (t - 1.0) / t_new
        y = np.maximum(x_new + momentum * (x_new - a), 0.0)
        a = x_new
        t = t_new

            y = project_simplex(best_local + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t = 1.0
    
    return best_local, best_score


def spectral_balance(a, N, n, S, target, n_iters, alpha, rng, top_k=20):
    """Spectral balancing: iteratively flatten the power spectrum."""
    a_curr = a.copy()
    best = a_curr.copy()
    best_score = fft_convolve_self(a_curr).max()
    
    for it in range(n_iters):
        c = fft_convolve_self(a_curr)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best = a_curr.copy()
            if best_score <= target:
                break
        
        a_pad = np.zeros(N)
        a_pad[:n] = a_curr
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c_circ = np.real(ifft(P))[:2*n-1]
        
        # Compute effective attenuation
        top_idx = np.argpartition(c_circ, -top_k)[-top_k:]
        top_vals = c_circ[top_idx]
        tot = top_vals.sum()
        if tot < 1e-10:
            break
        weights = top_vals / tot
        
        m = np.arange(N)
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
        
        alpha_eff = alpha * (0.5 + 0.5 * min(1.0, cur / max(target, 1e-10)))
        P_new = np.maximum(P * (1.0 - alpha_eff * cos_eff), 0.0)
        
        # Phase: use current phase most of the time, randomize occasionally
        if it % 200 == 100 and rng.random() < 0.3:
            phase = rng.standard_normal(N)
            phase[0] = 0.0  # Keep DC real
        else:
            phase = np.angle(A)
        
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_curr = project_simplex(a_new, S)
    
    return best, best_score
