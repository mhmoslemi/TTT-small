        g = res.x
        
        # Check feasibility
        conv_ag = fft_convolve_self_combined(a, g)  # Need efficient computation
        violations = conv_ag - M
        if violations.max() < 1e-10:
            return g
        
        # Add most violated constraint
        new_idx = np.argmax(violations)
        if new_idx not in active:
            active = np.append(active, new_idx)
    
    return g

2. Iteratively: clip the time-domain signal to reduce peaks, then project back (like the Gilbert-Showalter algorithm)

But in our case, the "time domain" is the convolution (frequency domain of a), and the "frequency domain" is a itself. So we'd:
1. Compute conv = autocorrelation of a
2. "Clip" conv: conv_clipped = min(conv, M) for some M
3. Invert: a_new = ifft(fft(conv_clipped) / fft(a)) ... no this doesn't work directly.

Actually, the Gilbert-Showalter algorithm works like this:
- Given x in frequency domain, compute time domain t = ifft(x)
- Modify t to reduce peaks (e.g., t_clipped = t * exp(-lambda * Re(t)))
- Update frequency: x_new = fft(t_clipped)
- The constraint in frequency domain is maintained

In our case:
- a is in "frequency domain" (time domain of the signal)
- conv = autocorrelation = ifft(|fft(a)|^2) is in "time domain"
- We want to minimize max(conv) subject to a >= 0 and sum(a) = S

The analogous algorithm would be:
1. Compute conv = ifft(|FFT(a)|^2) / n (normalization)
2. Modify conv to reduce peaks: conv_mod = conv * exp(-lambda * Re(conv)) or conv_mod = min(conv, threshold)
3. Invert: we need to find a_new such that autocorrelation(a_new) = conv_mod
   This means |FFT(a_new)|^2 = n * FFT(conv_mod)
   So FFT(a_new) = sqrt(n * FFT(conv_mod)) * phase(FFT(a))
4. a_new = ifft(FFT(a_new))
5. Project a_new to {a >= 0, sum(a) = S}

Wait, step 3 is tricky. We have |FFT(a_new)|^2 = n * FFT(conv_mod). The right side must be non-negative for this to work, which requires conv_mod to have a non-negative Fourier transform. This is not guaranteed.

The "temperature" parameter controls how "hard" the max is. At temperature=0, it's the hard max (subgradient at argmax). As temperature increases, it becomes softer (closer to the log-sum-exp gradient).

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- L-BFGS-B on softmax parameterization with log-sum-exp continuation
- Peak-targeting subgradient with line search for final polish
- Multiplicative weights update for natural non-negativity
- FFT-accelerated gradient computation

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Convolution of a and b via FFT. Returns array of length len(a)+len(b)-1."""
    size = len(a) + len(b) - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def compute_grad_from_weights(a, w):
    """
    Compute grad[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation via FFT).
    w is defined on the convolution indices [0, 2n-2].
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax_obj_grad(z, beta, S):
    """
    Compute smoothed objective and gradient w.r.t. z.
    a = S * softmax(z), objective = (1/beta)*log(sum exp(beta*c_k))
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_bc
    # Wait, that's a typo. Let me fix.
    w = exp_bc / sum_exp_bc
    grad_a = compute_grad_from_weights(a, w)

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def true_subgradient(a, top_k=10):
    """Compute a subgradient of max(conv) using top-k weighted average."""
    c = fft_convolve_self(a)
    # Use top-k with weights proportional to values
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(len(a)), c.max()
    weights = top_vals / total
    
    w_full = np.zeros(2 * len(a) - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    
    grad = compute_grad_from_weights(a, w_full)
    return grad, c.max()


def run_peak_targeting(a_init, S, max_time, rng, top_k_list=None):
    """
    Peak-targeting subgradient method with line search.
    Directly minimizes the true max by targeting the active peaks.
    """
    if top_k_list is None:
        top_k_list = [3, 5, 10, 20, 50]
    
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    
    start = time.monotonic()
    t = 1.0
    y = a.copy()  # Nesterov momentum
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        # Cycle through different top_k values
        top_k = top_k_list[it % len(top_k_list)]
        
        grad, cur_max = true_subgradient(y, top_k)
        
        if cur_max < best_score:
            best_score = cur_max
            best_a = y.copy()
        
        # Nesterov step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size: balance between progress and stability
        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic restart to escape local minima
        if it % 500 == 0 and rng.random() < 0.1:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
        
        # Check if we should do a multiplicative update instead
        if it % 100 == 0:
            # Multiplicative update
            scale = grad / max(gnorm, 1e-15)
            eta_mult = 0.1 / (1.0 + 0.001 * it)
            a_mult = a * np.exp(-eta_mult * scale)
            a_mult = np.maximum(a_mult, 1e-30)
            a_mult = a_mult * (S / a_mult.sum())
            c_mult = fft_convolve_self(a_mult)
            if c_mult.max() < best_score:
                best_score = c_mult.max()
                best_a = a_mult.copy()
                a = a_mult.copy()
                y = a_mult.copy()
    
    return best_a, best_score


def run_frank_wolfe(a_init, S, max_time, rng):
    """
    Frank-Wolfe (conditional gradient) method.
    At each step, solve the linearization: min grad^T g s.t. g in simplex
    The solution is g = S * e_j where j = argmin(grad).
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        grad, cur_max = true_subgradient(a, top_k=10)
        
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # FW step: s = S * e_{argmin(grad)}
        j_min = np.argmin(grad)
        s = np.zeros(n)
        s[j_min] = S
        
        # Step size: gamma_t = 2/(t+2) or line search
        gamma = 2.0 / (it + 2.0)
        
        a_new = (1 - gamma) * a + gamma * s
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new * (S / a_new.sum())
        
        # Line search (simple bisection)
        # f(a + t*(s-a)) for t in [0,1]
        d = s - a
        a_candidate = a_new
        c_candidate = fft_convolve_self(a_candidate).max()
        
        if c_candidate > cur_max and it > 10:
            # Try smaller step
            for t_frac in [0.5, 0.25, 0.1]:
                a_trial = a + t_frac * gamma * d
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial * (S / a_trial.sum())
                c_trial = fft_convolve_self(a_trial).max()
                if c_trial < cur_max:
                    a_candidate = a_trial
                    break
        
        a = a_candidate
        
        if it % 200 == 0:
            c_check = fft_convolve_self(a).max()
            if c_check < best_score:
                best_score = c_check
                best_a = a.copy()
    
    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def run_lbfgs_smoothed(a_init, n, S, max_time, beta_schedule=None, maxiter_per_beta=200):
    """Run L-BFGS-B with log-sum-exp continuation."""
    if beta_schedule is None:
        beta_schedule = [10, 20, 50, 100, 200, 500, 1000, 2000]
    
    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()
    
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    
    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break
        
        def fun(z_val):
            return softmax_obj_grad(z_val, beta, S)[0]
        
        def jac(z_val):
            return softmax_obj_grad(z_val, beta, S)[1]
        
        t0 = time.monotonic()
        try:
            result = scipy_minimize(
                fun, z,
                jac=jac,
                method='L-BFGS-B',
                options={
                    'maxiter': maxiter_per_beta,
                    'maxfun': maxiter_per_beta * 3,
                    'ftol': 1e-15,
                    'gtol': 1e-10,
                    'maxls': 50,
                    'iprint': -1
                }
            )
        except Exception:
            break
        
        z = result.x
        z = z - z.mean()
        
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = S * exp_z / exp_z.sum()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * S / s
        
        c = fft_convolve_self(a)
        score = c.max()
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        elapsed = time.monotonic() - t0
        print("    beta=%5d: score=%.10f (%.1fs)" % (beta, best_score, elapsed))
    
    return best_a, best_score

def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_from_weights(a, w):
    """
    Compute grad[j] = 2 * sum_k w[k] * a[k-j] via FFT cross-correlation.
    w is defined on convolution indices [0, 2n-2].
    """

    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax_obj_grad(z, beta, S):
