"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Key improvements over previous approach:
1. Direct non-negative optimization (allows zeros, unlike softmax)
2. Mirror descent / multiplicative updates for different geometry
3. Aggressive top-k subgradient (including top-1 for sharper descent)
4. Multiple L-BFGS passes with different smoothing levels
5. Adaptive coordinate refinement
6. Try multiple sequence lengths
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgrad_topk(a, b, top_k, weights=None):
    """Compute top-k weighted subgradient of max(b) w.r.t. a."""
    n = len(a)
    top_indices = np.argsort(b)[::-1][:top_k]
    top_vals = np.maximum(b[top_indices], 0)
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    if weights is None:
        weights = top_vals / total
    else:
        weights = weights[:top_k] / max(weights[:top_k].sum(), 1e-30)

    w_full = np.zeros(len(b))
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]

    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return g


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_seq(a, S):
    """Clamp to [0,1000] and scale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def smooth_max_obj(a, S, beta):
    """Smoothed max of convolution using log-sum-exp."""
    n = len(a)
    b = fft_convolve_self(a)
    b_max = b.max()
    exp_beta = np.exp(beta * (b - b_max))
    return (b_max + np.log(np.sum(exp_beta))) / beta


def smooth_max_grad_z(a, S, beta):
    """Gradient of smoothed max w.r.t. a (on the simplex sum=a=S)."""
    n = len(a)
    b = fft_convolve_self(a)
    b_max = b.max()
    exp_beta = np.exp(beta * (b - b_max))
    sum_exp = np.sum(exp_beta)
    w = exp_beta / sum_exp

    # grad_a = 2 * cross_corr(a, w)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    return grad_a


def run_lbfgs_direct(a_init, n, S, max_time, rng, beta_list=None, maxiter=200):
    """L-BFGS-B directly on a with bounds [0,1000], log-sum-exp smoothing."""
    if beta_list is None:
        beta_list = [10, 30, 100, 300, 1000, 3000, 10000]

    a = normalize_seq(a_init, S)
    b = fft_convolve_self(a)
    best_score = b.max()
    best_a = a.copy()

    start = time.monotonic()

    for beta in beta_list:
        if time.monotonic() - start > max_time - 5:
            break

        def fun(a_val):
            a_clipped = np.clip(a_val, 0, 1000)
            s = a_clipped.sum()
            if s < 1e-10:
                return 1e10
            a_norm = a_clipped * (S / s)
            return smooth_max_obj(a_norm, S, beta)

        def jac(a_val):
            a_clipped = np.clip(a_val, 0, 1000)
            s = a_clipped.sum()
            if s < 1e-10:
                return np.zeros(len(a_val))
            a_norm = a_clipped * (S / s)
            grad_a = smooth_max_grad_z(a_norm, S, beta)
            # Chain rule: d/da_val of objective where a_norm = a_clipped * S/sum(a_clipped)
            # This is complex, so use the simpler approach:
            # grad w.r.t. a_val ≈ grad_a (since the normalization is nearly identity near optimum)
            return grad_a

        try:
            result = scipy_minimize(
                fun, a,
                jac=jac,
                method='L-BFGS-B',
                bounds=[(0, 1000)] * n,
                options={
                    'maxiter': maxiter,
                    'maxfun': maxiter * 3,
                    'ftol': 1e-15,
                    'gtol': 1e-8,
                    'maxls': 50,
                    'iprint': -1
                }
            )
            a_candidate = np.clip(result.x, 0, 1000)
            a_candidate = normalize_seq(a_candidate, S)
        except Exception:
            break

        b_new = fft_convolve_self(a_candidate)
        if b_new.max() < best_score:
            best_score = b_new.max()
            best_a = a_candidate.copy()
        a = a_candidate

    return best_a, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, top_k_init=1, top_k_max=64):
    """
    Multiplicative update (mirror descent) with top-k subgradient.
    a[j] <- a[j] * exp(-eta * g[j] / S)
    Then project to simplex.
    """
    a = normalize_seq(a_init, S)
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    start = time.monotonic()
    it = 0
    momentum = np.zeros(n)
    beta_mom = 0.9

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 200)

        b = fft_convolve_self(a)
        cur = b.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        g = compute_subgrad_topk(a, b, top_k)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-16:
            break

        # Adaptive step size
        eta = 2.0 / (g_norm * (1.0 + 0.01 * it))
        eta = min(eta, 0.5)

        # Multiplicative update (mirror descent with entropy)
        # a[j] * exp(-eta * g[j])
        update = -eta * g / max(g_norm, 1e-15)
        update = np.clip(update, -5.0, 5.0)  # prevent overflow
        a_new = a * np.exp(update)

        # Ensure positivity and normalize
        a_new = np.maximum(a_new, 1e-15)
        a_new = normalize_seq(a_new, S)

        # Nesterov-like momentum
        a_new = a_new + beta_mom * momentum
        momentum = a_new - a
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)

        a = a_new

        # Periodic perturbation
        if it % 1000 == 0 and rng.random() < 0.1:
            a = a + rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = np.maximum(a, 0.0)
            a = normalize_seq(a, S)
            momentum = np.zeros(n)

    return best_a, best_score


def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=1, top_k_max=256):
    """Nesterov-accelerated top-k subgradient with adaptive step size."""
    a = normalize_seq(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    grad_ema = np.zeros(n)
    grad_sq_ema = np.zeros(n)
    beta1, beta2 = 0.9, 0.999

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 150)

        b = fft_convolve_self(y)
        cur = b.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()

        g = compute_subgrad_topk(y, b, top_k)

        # Adam-style adaptive step
        grad_ema = beta1 * grad_ema + (1 - beta1) * g
        grad_sq_ema = beta2 * grad_sq_ema + (1 - beta2) * g ** 2
        grad_hat = grad_ema / (1 - beta1 ** it)
        sq_hat = grad_sq_ema / (1 - beta2 ** it)
        g_adapted = grad_hat / (np.sqrt(sq_hat) + 1e-8)

        gnorm = np.linalg.norm(g_adapted)
        if gnorm < 1e-14:
            break

        step = 0.5 * S / (np.sqrt(n) * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * g_adapted / gnorm, S)

        if it % 50 == 0:
            b_x = fft_convolve_self(x_new)
            if b_x.max() < best_score:
                best_score = b_x.max()
                best_a = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - y)
        y = np.maximum(y, 0.0)
        y = normalize_seq(y, S)
        t = t_new

        if it % 2000 == 0 and rng.random() < 0.05:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0
            grad_ema = np.zeros(n)
            grad_sq_ema = np.zeros(n)

    return best_a, best_score


def run_coordinate_refine(a_init, n, S, max_time, rng):
    """
    Coordinate descent: for each coordinate, find optimal value via 1D search.
    Uses the fact that changing a[j] only affects b[k] for k in [j, j+n-1].
    """
    a = normalize_seq(a_init, S)
    b = fft_convolve_self(a)
    best_score = b.max()
    best_a = a.copy()
    start = time.monotonic()

    n_sweps = 0
    while time.monotonic() - start < max_time:
        n_sweps += 1
        improved = False

        # Randomize coordinate order
        order = rng.permutation(n)

        for j in order:
            old = a[j]
            S_prime = S - old  # sum of all other coordinates

            # Affected b[k] for k in [j, min(j+n-1, 2n-2)]
            k_min = j
            k_max = min(j + n - 1, 2 * n - 2)

            # c[k] = b[k] - old * a[k-j] for affected k
            # New b[k] = new_val * a[k-j] + c[k]

            # Find optimal new_val in [0, upper_bound]
            # Objective: max(max_affected(new_val), max_unaffected) / (S_prime + new_val)^2

            # max_unaffected
            if k_min > 0:
                unaff_left = b[:k_min].max()
            else:
                unaff_left = -np.inf
            if k_max < 2 * n - 2:
                unaff_right = b[k_max + 1:].max()
            else:
                unaff_right = -np.inf
            max_unaff = max(unaff_left, unaff_right, 0.0)

            # For the affected part, compute the best candidate
            # Try a few values: 0, old, and values derived from active constraints
            candidates = [0.0, old, old * 0.5, old * 1.5]

            # Add candidates from the top affected b values
            affected_range = b[k_min:k_max + 1]
            a_range = a[0:min(n, k_max + 1 - k_min)]  # a[k-j] for k in range

            # For each k in affected range where a[k-j] > 0:
            # Optimal new = S_prime - 2*c[k]/a[k-j] (if positive)
            c_val = b[k_min:k_max + 1] - old * a_range
            valid_mask = a_range > 1e-10
            if np.any(valid_mask):
                # c[k] = b[k] - old*a[k-j]
                # new_k = S_prime - 2*c[k]/a[k-j]
                new_candidates = S_prime - 2.0 * c_val[valid_mask] / a_range[valid_mask]
                new_candidates = new_candidates[new_candidates > 0]
                # Take top few
                if len(new_candidates) > 0:
                    # Pick ones that minimize estimated max
                    for nc in new_candidates[:5]:
                        candidates.append(min(nc, 1000.0))

            # Also try the point where affected_max = max_unaff
            if max_unaff > 0 and np.any(valid_mask):
                # Find new where max_k(new*a[k-j]+c[k]) = max_unaff
                # This is: new*a_max_affected + c_max_at_that = max_unaff
                # Approximate: new = (max_unaff - min(c_val)) / max(a_range)
                approx_new = (max_unaff - np.minimum(c_val[valid_mask], 0)) / max(a_range[valid_mask], 1e-10)
                if 0 < approx_new < 1000:
                    candidates.append(approx_new)

            # Evaluate objective at each candidate
            best_candidate_val = old
            best_candidate_score = max(b.max(), max_unaff) / (S ** 2)

            for cv in candidates:
                cv = max(0.0, min(cv, 1000.0))
                # Compute max of affected part
                new_affected = cv * a_range + c_val
                aff_max = np.max(new_affected) if len(new_affected) > 0 else 0.0
                total_max = max(aff_max, max_unaff)
                denom = (S_prime + cv) ** 2
                if denom > 1e-30:
                    score = total_max / denom
                    if score < best_candidate_score:
                        best_candidate_score = score
                        best_candidate_val = cv

            # Apply update
            if abs(best_candidate_val - old) > 1e-12:
                delta = best_candidate_val - old
                a[j] = best_candidate_val
                # Update b
                for k in range(k_min, k_max + 1):
                    idx = k - j
                    if 0 <= idx < n:
                        b[k] += delta * a[idx]

                new_bmax = b.max()
                if new_bmax < best_score:
                    best_score = new_bmax
                    best_a = a.copy()
                    improved = True

        if not improved and n_sweps > 3:
            break

    return best_a, best_score

    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0.0, 1000.0)))
    except Exception:
        pass

    # Choose main n
    if inits:
        main_len = len(inits[0][1])
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * float(n))
    tt = np.linspace(0, 1, n, endpoint=False)

    # Add diverse initializations
    inits.append(("rand", rng.random(n) + 0.05))
    inits.append(("sin_multi", 1.0 + 0.3*np.sin(2*np.pi*2*tt) +
                  0.2*np.cos(2*np.pi*5*tt) + 0.15*np.sin(2*np.pi*11*tt)))
    inits.append(("gaussian_shift", np.exp(-0.5*((tt-0.35)*8.0)**2) + 0.3 +
                  0.1*np.exp(-0.5*((tt-0.65)*8.0)**2)))
    inits.append(("ramp", np.linspace(0.1, 2.0, n) + 0.05))

    n_inits = len(inits)
    print("=== Optimization: n=%d, %d inits, budget=%.0fs ===" % (n, n_inits, budget_s))
    print("Target: 1.5027, Previous best: 1.508805")

    # === Phase 1: L-BFGS direct on each init ===
    usable = deadline - time.monotonic()
    per_init = max(usable * 0.35 / n_inits, 15.0)

    print("\n=== Phase 1: L-BFGS Direct (%.1f sec each) ===" % per_init)
    phase1_results = {}

    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break

        t0 = time.monotonic()
        a_norm = normalize_seq(arr, S)
        c = fft_convolve_self(a_norm)
        init_score = c.max()
        init_obj = 2.0 * n * init_score / S**2

        if init_obj < global_best:
            global_best = init_obj
            global_a = a_norm.copy()

        a_new, b_score = run_lbfgs_direct(
            a_norm, n, S, per_init, rng,
            beta_list=[10, 30, 100, 300, 1000, 3000],
            maxiter=150
        )
        final_obj = 2.0 * n * b_score / S**2
        phase1_results[name] = (a_new, b_score)

        if final_obj < global_best:
            global_best = final_obj
            global_a = a_new.copy()
            print("  [OK] %s: %.8f (init=%.4f, %.1fs)" % (name, global_best, init_obj, time.monotonic() - t0))
        else:
            print("  [%s: %.8f (best=%.6f, %.1fs)]" % (name, final_obj, global_best, time.monotonic() - t0))

    print("\n=== After Phase 1: global_best=%.8f ===" % global_best)

    # === Phase 2: Mirror descent on best ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            seg = min(remaining * 0.2, 180)
            print("\n=== Phase 2: Mirror Descent (%.1f sec) ===" % seg)
            a_m, score = run_mirror_descent(global_a, n, S, seg, rng,
                                            top_k_init=1, top_k_max=32)
            obj = 2.0 * n * score / S**2
            if obj < global_best:
                global_best = obj
                global_a = a_m.copy()
                print("  Mirror: %.8f" % global_best)
            else:
                print("  Mirror: %.8f (no improvement)" % obj)

    # === Phase 3: Nesterov top-k ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            seg = min(remaining * 0.2, 180)
            print("\n=== Phase 3: Nesterov Top-K (%.1f sec) ===" % seg)
            a_n, score = run_nesterov_toppk(global_a, n, S, seg, rng,
                                            top_k_init=1, top_k_max=128)
            obj = 2.0 * n * score / S**2
            if obj < global_best:
                global_best = obj
                global_a = a_n.copy()
                print("  Nesterov: %.8f" % global_best)
            else:
                print("  Nesterov: %.8f (no improvement)" % obj)

    # === Phase 4: Coordinate refinement ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            seg = min(remaining * 0.15, 120)
            print("\n=== Phase 4: Coordinate Refine (%.1f sec) ===" % seg)
            a_c, score = run_coordinate_refine(global_a, n, S, seg, rng)
            obj = 2.0 * n * score / S**2
            if obj < global_best:
                global_best = obj
                global_a = a_c.copy()
                print("  Coord: %.8f" % global_best)
            else:
                print("  Coord: %.8f (no improvement)" % obj)

    # === Phase 5: Perturbation + L-BFGS restart ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("\n=== Phase 5: Perturbation + L-BFGS (%.1f sec) ===" % remaining)

            noise_scales = [0.02, 0.01, 0.005, 0.002, 0.001]
            n_perturb = max(3, int(remaining / 40))
            per_pert = remaining * 0.7 / max(n_perturb, 1)

            for p in range(n_perturb):
                if time.monotonic() > deadline - 60:
                    break

                scale = noise_scales[p % len(noise_scales)]
                a_pert = global_a + rng.normal(0, scale * S / np.sqrt(n), n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = normalize_seq(a_pert, S)

                a_new, score = run_lbfgs_direct(
                    a_pert, n, S, per_pert, rng,
                    beta_list=[100, 500, 2000, 10000],
                    maxiter=100
                )
                obj = 2.0 * n * score / S**2
                if obj < global_best:
                    global_best = obj
                    global_a = a_new.copy()
                    print("  Perturb %d (scale=%.4f): %.8f" % (p, scale, global_best))

    # === Phase 6: Final high-beta L-BFGS polish ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print("\n=== Phase 6: Final Polish (%.1f sec) ===" % remaining)
            a_f, score = run_lbfgs_direct(
                global_a, n, S, remaining, rng,
                beta_list=[500, 2000, 10000, 50000],
                maxiter=200
            )
            obj = 2.0 * n * score / S**2
            if obj < global_best:
                global_best = obj
                global_a = a_f.copy()
                print("  Final: %.8f" % global_best)

    # === Phase 7: Try smaller n ===
    if time.monotonic() < deadline - 60:
        for n_try in [4096, 2048]:
            if time.monotonic() > deadline - 30:
                break
            S_try = np.sqrt(2.0 * float(n_try))
            # Resize global_a by interpolation
            if global_a is not None:
                from scipy.interpolate import interp1d
                f = interp1d(np.linspace(0, 1, n), global_a, kind='cubic', fill_value='extrapolate')
                tt_try = np.linspace(0, 1, n_try, endpoint=False)
                a_try = np.maximum(f(tt_try), 0)
            else:
                a_try = np.ones(n_try) * S_try / n_try
            a_try = normalize_seq(a_try, S_try)

            a_try2, score_try = run_lbfgs_direct(a_try, n_try, S_try, 60, rng,
                                                  beta_list=[50, 200, 1000], maxiter=100)
            obj_try = 2.0 * n_try * score_try / S_try**2
            if obj_try < global_best:
                global_best = obj_try
                global_a = a_try2.copy()
                print("  n=%d: %.8f" % (n_try, global_best))
            else:
                print("  n=%d: %.8f (no improvement)" % (n_try, obj_try))

    # === Verification ===
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.8f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_obj = 2.0 * len(global_a) * fft_convolve_self(global_a).max() / global_a.sum()**2
    print("\nFinal: length=%d, score=%.8f" % (len(global_a), final_obj))
    return [float(x) for x in global_a]
