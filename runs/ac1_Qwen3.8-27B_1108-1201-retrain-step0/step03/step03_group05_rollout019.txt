def fft_convolve_self(a):

def fft_convolve_self(a):
    """Full self-convolution via FFT, length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_from_weights(a, w_full):
    """Gradient of sum_j w_j*(f*f)(j) w.r.t. f via FFT cross-correlation."""
    n = len(a)
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """Smoothed objective and gradient w.r.t. softmax params z."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    exp_bc = np.exp(beta * (c - c_max))

    c_max = c.max()

    # Active set: positions within threshold of max
    active_thresh = (1.0 - threshold_frac) * c_max
    active_mask = c >= active_thresh
    n_active = active_mask.sum()

    if n_active < 1:
        top_idx = np.argmax(c)
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0
    else:
        # Weight active positions proportionally to their value
        w = np.where(active_mask, c, 0.0)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0
        else:
            w = w / w_sum

    grad = compute_grad_from_weights(a, w)

    # Multiplicative (mirror) update on the simplex
    # Use clipping to avoid overflow
    log_update = -eta * grad
    log_update = np.clip(log_update, -50.0, 50.0)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-30)

    # Normalize to maintain sum = S
    S_target = a.sum()
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new = a_new * (S_target / s)

    return a_new


def run_mirror_descent(a_init, S, max_time, rng, initial_eta=0.1,
                       decay=0.9995, threshold_frac=0.01, patience=3000):
    """Mirror descent (multiplicative weights) with adaptive step size."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) * S / len(a)
    else:
        a = a * (S / s)

    n = len(a)
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    eta = initial_eta
    no_improve = 0
    it = 0
    n_active_report = 0

    while time.monotonic() - start < max_time:
        it += 1

        # Occasionally refresh the active set with a slightly larger threshold
        if it % 5000 == 0:
            threshold_frac = max(0.001, threshold_frac * 0.9)

        a_new = mirror_descent_step(a, eta, threshold_frac)

        # Evaluate
        c_new = fft_convolve_self(a_new)
        cur_new = c_new.max()

        if cur_new < best_score - 1e-12:
            best_score = cur_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                # Reduce step size and try again
                eta *= 0.5
                no_improve = 0
                if eta < 1e-10:
                    # Try a small random restart
                    a_new = a + rng.normal(0, 0.005 * S / np.sqrt(n), n)
                    a_new = np.maximum(a_new, 0.0)
                    a_new = a_new * (S / a_new.sum())
                    eta = initial_eta * 0.5

        a = a_new
        eta = max(eta * decay, 1e-10)

        if it % 20000 == 0:
            print("    MD it=%d: best=%.8f, eta=%.2e, no_improve=%d" %
                  (it, best_score, eta, no_improve))

    return best_a, best_score


def run_lbfgs_continuation(a_init, n, S, max_time, rng, beta_schedule=None,
                           maxiter_per_beta=250):
    """L-BFGS-B with continuation on log-sum-exp magnitude beta."""
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break

        attempts = 1
        if beta >= 500 and max_time > 60:
            attempts = 2
        if beta >= 2000 and max_time > 120:
            attempts = 3

        for attempt in range(attempts):
            if time.monotonic() - start > max_time - 2:
                break
            if attempt > 0:
                noise_scale = 0.005 * attempt
                z = z + rng.normal(0, noise_scale, n)
                z = z - z.mean()

            def _fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def _jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            try:
                result = scipy_minimize(
                    _fun, z, jac=_jac, method='L-BFGS-B',
                    options={'maxiter': maxiter_per_beta,
                             'maxfun': maxiter_per_beta * 3,
                             'ftol': 1e-15, 'gtol': 1e-11,
                             'maxls': 60, 'iprint': -1}
                )
                z = result.x
            except Exception:
                break

            z = z - z.mean()
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def safe_normalize(a, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    n = len(a)
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def generate_diverse_perturbation(a_base, S, n, rng, strategy_idx):
    """Generate a perturbed version of a_base using different strategies."""
    if strategy_idx % 5 == 0:
        # Dense Gaussian noise (decaying scale)
        noise_scale = 0.04 / (1.0 + 0.3 * (strategy_idx // 5))
        a_pert = a_base + rng.normal(0, noise_scale * S / np.sqrt(n), n)
    elif strategy_idx % 5 == 1:
        # Sparse perturbations
        a_pert = a_base.copy()
        n_sparse = max(64, n // 16)
        idx = rng.choice(n, n_sparse, replace=False)
        scale = 0.08 * S / np.sqrt(n_sparse)
        a_pert[idx] += rng.normal(0, scale, n_sparse)
    elif strategy_idx % 5 == 2:
        # Multiplicative log-normal perturbation
        log_noise = rng.normal(0, 0.04, n)
        a_pert = a_base * np.exp(log_noise)
    elif strategy_idx % 5 == 3:
        # Blend with uniform (encourages flattening)
        alpha = 0.1 + 0.05 * (strategy_idx % 4)
        uniform = np.ones(n) * S / n
        a_pert = (1 - alpha) * a_base + alpha * uniform
    else:
        # Low-frequency perturbation (smooth bump)
        a_pert = a_base.copy()
        center = rng.integers(n // 4, 3 * n // 4)
        width = n // 8
        bump = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        a_pert += 0.03 * S * bump / bump.sum() * n
    return safe_normalize(a_pert, S)


def propose_candidate(seed=42, budget_s=1800):
    """Search for non-negative sequence minimizing 2n*max(conv)/sum^2.

    Multi-phase approach:
    - Phase 1: L-BFGS continuation with aggressive beta schedule
    - Phase 2: Mirror descent (multiplicative weights) with exact subgradient
    - Phase 3: Diverse restarts (Gaussian, sparse, multiplicative, blend, bump)
    - Phase 4: Final high-beta L-BFGS polish + mirror descent
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)
    t_global_start = time.monotonic()

    # --- Get initial sequence ---
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 128 and np.sum(np.clip(given, 0, 1000)) > 0:
            n = len(given)
            a_init = np.clip(given, 0.0, 1000.0)
        else:
            raise ValueError("too small or zero")
    except Exception:
        n = 4096
        a_init = np.ones(n)

    S = np.sqrt(2.0 * float(n))
    a_init = safe_normalize(a_init, S)

    global_best = fft_convolve_self(a_init).max()
    global_a = a_init.copy()
    print("=== Init: n=%d, score=%.8f ===" % (n, global_best))
    print("Target: 1.5027 | Previous best: 1.508805")

    # ========== Phase 1: L-BFGS Continuation (40% of budget) ==========
    remaining = deadline - time.monotonic()
    phase1_time = remaining * 0.40
    if phase1_time > 15:
        print("\n=== Phase 1: L-BFGS Continuation (%.0fs) ===" % phase1_time)
        a1, s1 = run_lbfgs_continuation(
            a_init, n, S, phase1_time, rng,
            beta_schedule=[5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000],
            maxiter_per_beta=300
        )
        a1 = safe_normalize(a1, S)
        s1 = fft_convolve_self(a1).max()
        if s1 < global_best:
            global_best = s1
            global_a = a1.copy()
            print("  Phase 1 result: %.8f (improved)" % global_best)
        else:
            print("  Phase 1: no improvement (%.8f vs %.8f)" % (s1, global_best))

    # ========== Phase 2: Mirror Descent (20% of budget) ==========
    remaining = deadline - time.monotonic()
    if remaining > 20:
        phase2_time = remaining * 0.20
        if phase2_time > 10:
            print("\n=== Phase 2: Mirror Descent (%.0fs) ===" % phase2_time)

            # Try multiple threshold settings
            for idx, thresh in enumerate([0.05, 0.02, 0.005]):
                if time.monotonic() > deadline - 60:
                    break
                seg = phase2_time / 3
                if idx == 2:
                    seg = min(seg, (deadline - time.monotonic()) - 30)
                    if seg < 5:
                        break
                a2, s2 = run_mirror_descent(
                    global_a, S, seg, rng,
                    initial_eta=0.08, decay=0.9997,
                    threshold_frac=thresh, patience=2000
                )
                a2 = safe_normalize(a2, S)
                s2 = fft_convolve_self(a2).max()
                if s2 < global_best:
                    global_best = s2
                    global_a = a2.copy()
                    print("  MD (thresh=%.3f): %.8f" % (thresh, global_best))
                else:
                    print("  MD (thresh=%.3f): %.8f (no improvement)" % (thresh, s2))

    # ========== Phase 3: Diverse Restarts (35% of remaining) ==========
    remaining = deadline - time.monotonic()
    if remaining > 40:
        phase3_time = remaining * 0.35
        n_restarts = max(4, int(phase3_time / 45))
        per_restart = phase3_time / n_restarts

        print("\n=== Phase 3: Diverse Restarts (%d x %.0fs) ===" % (n_restarts, per_restart))

        for r in range(n_restarts):
            if time.monotonic() > deadline - 120:
                break

            a_pert = generate_diverse_perturbation(global_a, S, n, rng, r)

            # Short L-BFGS on perturbed point
            a_new, s_new = run_lbfgs_continuation(
                a_pert, n, S, per_restart * 0.7, rng,
                beta_schedule=[20, 50, 100, 300, 1000, 3000],
                maxiter_per_beta=200
            )

            # Brief mirror descent on the result
            remaining_for_md = per_restart * 0.3
            if remaining_for_md > 3:
                a_new, s_new2 = run_mirror_descent(
                    a_new, S, remaining_for_md, rng,
                    initial_eta=0.02, decay=0.9999,
                    threshold_frac=0.005, patience=1000
                )
                s_new2 = fft_convolve_self(a_new).max()
                if s_new2 < s_new:
                    s_new = s_new2

            a_new = safe_normalize(a_new, S)
            s_final = fft_convolve_self(a_new).max()

            if s_final < global_best:
                global_best = s_final
                global_a = a_new.copy()
                print("  Restart %d (strat=%d): %.8f ★" % (r, r % 5, global_best))
            else:
                print("  Restart %d (strat=%d): %.8f" % (r, r % 5, s_final))

    # ========== Phase 4: Final Polish (remaining time) ==========
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("\n=== Phase 4: Final Polish (%.0fs) ===" % remaining)

        # Mirror descent with very small steps
        md_time = min(remaining * 0.35, 80.0)
        if md_time > 10:
            a_polish, s_polish = run_mirror_descent(
                global_a, S, md_time, rng,
                initial_eta=0.005, decay=0.99995,
                threshold_frac=0.001, patience=5000
            )
            a_polish = safe_normalize(a_polish, S)
            s_polish = fft_convolve_self(a_polish).max()
            if s_polish < global_best:
                global_best = s_polish
                global_a = a_polish.copy()
                print("  MD polish: %.8f" % global_best)

        # Very high beta L-BFGS
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a_final, s_final = run_lbfgs_continuation(
                global_a, n, S, remaining2, rng,
                beta_schedule=[2000, 5000, 10000, 20000, 50000],
                maxiter_per_beta=400
            )
            a_final = safe_normalize(a_final, S)
            s_final = fft_convolve_self(a_final).max()
            if s_final < global_best:
                global_best = s_final
                global_a = a_final.copy()
                print("  Final L-BFGS: %.8f" % global_best)

        # One last mirror descent nudge
        remaining3 = deadline - time.monotonic()
        if remaining3 > 10:
            a_last, s_last = run_mirror_descent(
                global_a, S, remaining3, rng,
                initial_eta=0.002, decay=0.99998,
                threshold_frac=0.0005, patience=10000
            )
            a_last = safe_normalize(a_last, S)
            s_last = fft_convolve_self(a_last).max()
            if s_last < global_best:
                global_best = s_last
                global_a = a_last.copy()
                print("  Last MD: %.8f" % global_best)

    # --- Verification ---
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    ver = evaluate_sequence(final_list)
    print("\n=== FINAL: internal_score=%.8f, verified=%s ===" %
          (global_best, "%.6f" % ver if np.isfinite(ver) else "REJECTED"))
    print("Total time: %.1fs" % (time.monotonic() - t_global_start))

    if not np.isfinite(ver) or global_a.sum() < 0.01:
        print("  Fallback to uniform")
        return [1.0] * 256

    return final_list
