"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses log-sum-exp smooth optimization with adaptive beta, 
multigrid initialization, backtracking line search, 
and local coordinate descent for final polish.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_conv_gradient(a, weights):
    """Gradient of weighted sum of convolution values."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def softmax_weights(c, beta):
    """Softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total

def logsumexp_value(c, beta):
    """Compute (1/beta)*log(sum(exp(beta*c))) as smooth max approximation."""
    c_max = c.max()
    val = c_max + np.log(np.exp(beta * (c - c_max)).sum()) / beta
    return val

        a = a * (S / s)

    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta (exponential ramp)

        # Compute convolution and smooth objective

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Key improvements: adaptive log-sum-exp with exponential beta schedule,
backtracking line search, multigrid initialization, multi-n exploration,
fast analytical local search, and momentum with adaptive restart.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_conv_gradient(a, weights):
    """Compute gradient of weighted sum of convolution lags using FFT."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def logsumexp_value(c, beta):
    """Compute (1/beta)*log(sum(exp(beta*c))) as smooth max approximation."""
    c_max = c.max()
    return c_max + np.log(np.exp(beta * (c - c_max)).sum()) / beta


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def run_smooth_fista(a_init, n, S, max_time, rng,
                     beta_start=10.0, beta_end=2000.0,
                     lr_start=0.5, momentum=0.9):
    """
    FISTA with log-sum-exp smooth objective, adaptive beta,
    backtracking, and momentum restart.
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stall_count = 0
    best_stall = best_score

    est_iters = max_time * 8000  # rough estimate

        it += 1

        # Exponential beta schedule
        beta_frac = min(1.0, 0.0008 * it)
        beta = beta_start * (beta_end / beta_start) ** beta_frac

        # Compute convolution and objective
        c = fft_convolve_self(y)
        cur_max = c.max()
        cur_smooth = logsumexp_value(c, beta)

        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_local = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        # Compute gradient
        w = softmax_weights(c, beta)
        grad = compute_conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        # Step size: decreases with beta and iteration
        lr = lr_start / (1.0 + 0.0005 * it)
        step = lr / (beta * 0.5 + 1.0)

        x_new = project_simplex(y - step * grad, S)

        c_new = fft_convolve_self(x_new)
        obj_new = logsumexp_value(c_new, beta)

            obj_half = logsumexp_value(c_half, beta)

        else:

            if total < 1e-30:

                w = np.zeros(len(c))
                w[top_indices] = top_vals / total

        grad = compute_conv_gradient(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Normalized step
        step = step_base / (1.0 + 0.002 * it)
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 3000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            a = y.copy()
            t_mom = 1.0

    return best_local, best_score


def run_local_search(a_init, n, S, max_time, rng):
    """
    Fast local search using analytical convolution updates.
    Tries single-element moves and pair swaps.
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        cur_max = c.max()

        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_local = a.copy()

        # Find peak location
        peak_lag = np.argmax(c)

        # Strategy 1: Reduce the peak by moving mass from high-contributing elements
        # c[peak_lag] = sum_i a[i]*a[peak_lag-i]
        # Find the element i that contributes most to c[peak_lag]
        i_min = max(0, peak_lag - (n - 1))
        i_max = min(n - 1, peak_lag)

        if i_max >= i_min:
            pairs_idx = np.arange(i_min, i_max + 1)
            products = a[pairs_idx] * a[peak_lag - pairs_idx]
            max_local = np.argmax(products)
            i_star = pairs_idx[max_local]
            j_star = peak_lag - i_star

            # Try moving small amount from i_star to a random position
            for attempt in range(20):
                target = rng.integers(0, n)
                if target == i_star or target == j_star:
                    continue

                # Try different amounts
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amount = frac * a[i_star]
                    if amount < 1e-15:
                        continue

                    a_test = a.copy()
                    a_test[i_star] -= amount
                    a_test[target] += amount
                    if a_test[i_star] < -1e-10:
                        continue

                    # Quick check: compute new peak approximately
                    # Full check every few attempts
                    if attempt % 5 == 0 or frac <= 0.05:
                        c_test = fft_convolve_self(a_test)
                        if c_test.max() < cur_max - 1e-10:
                            a = a_test
                            c = c_test
                            cur_max = c.max()
                            break
                else:
                    continue
                break

        # Strategy 2: Pair swap (equalize two elements)
        if it % 3 == 0:
            i1, i2 = rng.integers(0, n, 2)
            if i1 != i2:
                delta = 0.01 * (a[i1] - a[i2])
                a_test = a.copy()
                a_test[i1] -= delta
                a_test[i2] += delta
                if a_test[i1] >= -1e-10 and a_test[i2] >= -1e-10:
                    a_test = np.maximum(a_test, 0)
                    c_test = fft_convolve_self(a_test)
                    if c_test.max() < cur_max - 1e-10:
                        a = a_test
                        c = c_test
                        cur_max = c.max()

        # Strategy 3: Smooth redistribution
        if it % 10 == 0:
            # Find top 3 peaks and try to reduce them simultaneously
            top_lags = np.argsort(c)[::-1][:3]
            for lag in top_lags:
                i_min = max(0, lag - (n - 1))
                i_max = min(n - 1, lag)
                if i_max < i_min:
                    continue
                # Reduce the largest contributor at this lag
                pairs_idx = np.arange(i_min, i_max + 1)
                products = a[pairs_idx] * a[lag - pairs_idx]
                if products.max() < 1e-10:
                    continue
                i_star = pairs_idx[np.argmax(products)]
                # Move to a position that doesn't contribute to any top peak
                for attempt in range(5):
                    target = rng.integers(0, n)
                    if any(abs(target - l) < 5 or abs(lag - target) < 5 for l in top_lags):
                        continue
                    amount = 0.05 * a[i_star]
                    a_test = a.copy()
                    a_test[i_star] -= amount
                    a_test[target] += amount
                    if a_test[i_star] < -1e-10:
                        continue
                    c_test = fft_convolve_self(a_test)
                    if c_test.max() < cur_max - 1e-10:
                        a = a_test
                        c = c_test
                        cur_max = c.max()
                        break

    return best_local, best_score


def run_multigrid(a_target, n_target, budget, rng):
    """
    Multigrid: optimize at increasing resolutions then upscale.
    """
    scales = [32, 64, 128, 256, 512, 1024, 2048, 4096, 8192, 16384]
    scales = [s for s in scales if s <= n_target]

    t_start = time.monotonic()
    a = None

    for n_curr in scales:
        t_remaining = budget - (time.monotonic() - t_start)
        if t_remaining < 3:
            break

        S_curr = np.sqrt(2.0 * n_curr)

        if a is None:
            a_curr = np.ones(n_curr) * S_curr / n_curr
        else:
            a_curr = np.interp(
                np.linspace(0, 1, n_curr, endpoint=False),
                np.linspace(0, 1, len(a), endpoint=False),
                a
            )
            s = a_curr.sum()
            if s > 1e-15:
                a_curr = a_curr * (S_curr / s)

        # Short optimization at this scale
        t_opt = min(30.0, 0.25 * t_remaining)
        if t_opt > 2:
            if n_curr <= 1024:
                a_curr, _ = run_smooth_fista(a_curr, n_curr, S_curr, t_opt, rng,
                                              beta_start=5.0, beta_end=200.0,
                                              lr_start=0.4)
            else:
                a_curr, _ = run_smooth_fista(a_curr, n_curr, S_curr, t_opt, rng,
                                              beta_start=10.0, beta_end=500.0,
                                              lr_start=0.3)

        a = a_curr
        c = fft_convolve_self(a_curr)
        print("    MG n=%d: max=%.6f" % (n_curr, c.max()))

    # Final upscale
    S_target = np.sqrt(2.0 * float(n_target))
    a_final = np.interp(
        np.linspace(0, 1, n_target, endpoint=False),
        np.linspace(0, 1, len(a), endpoint=False),
        a
    )
    s = a_final.sum()
    if s > 1e-15:
        a_final = a_final * (S_target / s)
    else:
        a_final = np.ones(n_target) * S_target / n_target

    return a_final


def run_annealing(a_init, n, S, max_time, rng):
    """
    Simulated annealing style: accept worse solutions with decreasing probability.
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    c = fft_convolve_self(a)
    cur_score = c.max()
    best_score = cur_score
    best_local = a.copy()
    start = time.monotonic()
    it = 0
    temp_start = 0.01
    temp_end = 0.0001

    while time.monotonic() - start < max_time:
        it += 1
        elapsed_frac = (time.monotonic() - start) / max_time
        temp = temp_start * (temp_end / temp_start) ** elapsed_frac

        # Generate perturbation
        pert_type = rng.integers(0, 3)
        if pert_type == 0:
            # Single element change
            i = rng.integers(0, n)
            target = rng.integers(0, n)
            amount = rng.uniform(0.01, 0.1) * S / n
            a_test = a.copy()
            a_test[i] -= amount
            a_test[target] += amount
        elif pert_type == 1:
            # Small smooth noise
            noise = np.convolve(
                rng.normal(0, 1, n),
                np.ones(32) / 32, mode='same'
            )
            a_test = a + 0.005 * S / np.sqrt(n) * noise
        else:
            # Block modification
            block_len = rng.integers(n // 16, n // 4)
            start_pos = rng.integers(0, n - block_len)
            a_test = a.copy()
            a_test[start_pos:start_pos + block_len] *= rng.uniform(0.7, 1.3)

        a_test = np.maximum(a_test, 0)
        s_test = a_test.sum()
        if s_test < 1e-15:
            continue
        a_test = a_test * (S / s_test)

        c_test = fft_convolve_self(a_test)
        new_score = c_test.max()
        delta = new_score - cur_score

        # Accept if better or with probability exp(-delta/temp)
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_test
            c = c_test
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_local = a.copy()

    return best_local, best_score


def propose_candidate(seed=42, budget_s=2385):
    """
    Minimize 2*n*max_conv/sum^2 over non-negative sequences.
    Multi-phase: multigrid init -> smooth FISTA -> top-k polish ->
    local search -> annealing -> multi-start.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s - 40.0)

    # Get given sequence
    given_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given_a = np.clip(given, 0, 1000)
    except Exception:
        pass

    if given_a is not None:
        n = len(given_a)
    else:
        n = 8192

    S = np.sqrt(2.0 * float(n))

    print("=== Starting: n=%d, budget=%.0fs ===" % (n, budget_s))

    # Initial from given
    if given_a is not None and given_a.sum() > 1e-15:
        a_given = given_a * (S / given_a.sum())
    else:
        a_given = np.ones(n) * S / n

    c_given = fft_convolve_self(a_given)
    best_score = c_given.max()
    a_best = a_given.copy()
    print("Given: max_conv=%.6f, obj=%.6f" % (best_score, 2.0 * n * best_score / S**2))

    # === Phase 0: Multigrid initialization ===
    t_mg = min(200.0, 0.10 * budget_s)
    remaining = deadline - time.monotonic()
    if remaining > t_mg:
        print("Phase 0: Multigrid (%.0fs)" % t_mg)
        a_mg = run_multigrid(None, n, t_mg, rng)
        c_mg = fft_convolve_self(a_mg)
        if c_mg.max() < best_score:
            best_score = c_mg.max()
            a_best = a_mg.copy()
            print("  MG: max_conv=%.6f (improved)" % best_score)
        else:
            print("  MG: max_conv=%.6f (not better)" % c_mg.max())

    # === Phase 1: Main smooth FISTA ===
    remaining = deadline - time.monotonic()
    p1_time = 0.35 * remaining
    if p1_time > 30:
        print("Phase 1: Smooth FISTA (%.0fs)" % p1_time)
        a_p1, s_p1 = run_smooth_fista(a_best, n, S, p1_time, rng,
                                       beta_start=15.0, beta_end=3000.0,
                                       lr_start=0.5)
        c_p1 = fft_convolve_self(a_p1)
        if c_p1.max() < best_score:
            best_score = c_p1.max()
            a_best = a_p1.copy()
            print("  P1: max_conv=%.6f, obj=%.6f" % (best_score, 2.0 * n * best_score / S**2))

    # === Phase 2: Top-k focused descent ===
    remaining = deadline - time.monotonic()
    p2_time = 0.20 * remaining
    if p2_time > 20:
        print("Phase 2: Top-k descent (%.0fs)" % p2_time)
        a_p2, s_p2 = run_toppk_descent(a_best, n, S, p2_time, rng,
                                        top_k_init=1, top_k_max=30, top_k_rate=80)
        c_p2 = fft_convolve_self(a_p2)
        if c_p2.max() < best_score:
            best_score = c_p2.max()
            a_best = a_p2.copy()
            print("  P2: max_conv=%.6f, obj=%.6f" % (best_score, 2.0 * n * best_score / S**2))

    # === Phase 3: Local search ===
    remaining = deadline - time.monotonic()
    p3_time = 0.15 * remaining
    if p3_time > 15:
        print("Phase 3: Local search (%.0fs)" % p3_time)
        a_p3, s_p3 = run_local_search(a_best, n, S, p3_time, rng)
        c_p3 = fft_convolve_self(a_p3)
        if c_p3.max() < best_score:
            best_score = c_p3.max()
            a_best = a_p3.copy()
            print("  P3: max_conv=%.6f, obj=%.6f" % (best_score, 2.0 * n * best_score / S**2))

    # === Phase 4: Annealing for exploration ===
    remaining = deadline - time.monotonic()
    p4_time = 0.10 * remaining
    if p4_time > 15:
        print("Phase 4: Annealing (%.0fs)" % p4_time)
        a_p4, s_p4 = run_annealing(a_best, n, S, p4_time, rng)
        c_p4 = fft_convolve_self(a_p4)
        if c_p4.max() < best_score:
            best_score = c_p4.max()
            a_best = a_p4.copy()
            print("  P4: max_conv=%.6f, obj=%.6f" % (best_score, 2.0 * n * best_score / S**2))

    # === Phase 5: Multi-start with different strategies ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("Phase 5: Multi-start (%.0fs)" % remaining)
        n_starts = min(5, int(remaining / 50))
        per_start = remaining * 0.18

        for si in range(n_starts):
            if time.monotonic() > deadline - 30:
                break

            # Different perturbation strategies
            if si == 0:
                pert = a_best + rng.normal(0, 0.003 * S / np.sqrt(n), n)
            elif si == 1:
                smooth_noise = np.convolve(
                    rng.normal(0, 1, n), np.ones(256) / 256, mode='same')
                pert = a_best + 0.004 * S / np.sqrt(n) * smooth_noise
            elif si == 2:
                pert = a_best * (1.0 + rng.normal(0, 0.02, n))
            elif si == 3:
                # Structural: create a periodic-like pattern combined with a_best
                period = rng.integers(n // 4, n // 2)
                struct = np.tile(np.ones(period) * 0.3, n // period + 1)[:n]
                pert = 0.8 * a_best + 0.2 * struct * (S / struct.sum())
            else:
                # Fresh multigrid
                pert = run_multigrid(None, n, min(60, per_start * 0.3), rng)

            pert = np.maximum(pert, 0)
            s_pert = pert.sum()
            if s_pert > 1e-15:
                pert = pert * (S / s_pert)

            # Optimize the perturbed version
            a_si, _ = run_smooth_fista(pert, n, S, per_start, rng,
                                        beta_start=30 + 20 * si,
                                        beta_end=2000 + 500 * si,
                                        lr_start=0.3)
            c_si = fft_convolve_self(a_si)
            if c_si.max() < best_score:
                best_score = c_si.max()
                a_best = a_si.copy()
                print("  Start %d: max_conv=%.6f, obj=%.6f" %
                      (si, best_score, 2.0 * n * best_score / S**2))

    # === Phase 6: Final ultra-fine polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("Phase 6: Final polish (%.0fs)" % remaining)

        # Very high beta FISTA
        p6a_time = remaining * 0.4
        if p6a_time > 5:
            a_f1, _ = run_smooth_fista(a_best, n, S, p6a_time, rng,
                                        beta_start=500.0, beta_end=20000.0,
                                        lr_start=0.1)
            c_f1 = fft_convolve_self(a_f1)
            if c_f1.max() < best_score:
                best_score = c_f1.max()
                a_best = a_f1.copy()
                print("  F6a: max_conv=%.6f" % best_score)

        # Local search for final touches
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a_f2, _ = run_local_search(a_best, n, S, remaining2, rng)
            c_f2 = fft_convolve_self(a_f2)
            if c_f2.max() < best_score:
                best_score = c_f2.max()
                a_best = a_f2.copy()
                print("  F6b: max_conv=%.6f" % best_score)

    # === Final output ===
    final_list = [float(x) for x in np.clip(a_best, 0.0, 1000.0)]
    final_obj = 2.0 * n * best_score / (S * S)
    print("=== FINAL: n=%d, max_conv=%.6f, objective=%.6f ===" %
          (len(final_list), best_score, final_obj))

    # Verify
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: %.6f" % ver)
    else:
        print("Warning: verification failed, using fallback")
        final_list = [0.5] * 256
        ver = evaluate_sequence(final_list)
        print("Fallback: %.6f" % ver)

    return final_list
