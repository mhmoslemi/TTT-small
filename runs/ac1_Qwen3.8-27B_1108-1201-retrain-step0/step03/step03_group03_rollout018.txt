1. Drop the sum constraint and handle it separately (e.g., project onto the simplex)
2. Do a paired coordinate update (change `a[i]` and `a[j]` simultaneously to maintain the sum)

Option 2 is better. For a pair `(i, j)`, we change `a[i] += delta` and `a[j] -= delta` (maintaining the sum). Then we check if the max convolution is reduced.

This is essentially a "swap" operation. We can search over all pairs (i, j) and find the one that gives the best improvement.

But with n = 8192, there are ~33 million pairs, which is too many to check exhaustively.

Instead, we can use a greedy approach: for the top lag `k*`, find the pair (i, j) that gives the best improvement when we move mass from i to j, and perform that swap.

For a swap `a[i] += delta, a[j] -= delta`:
- Change in `b[k]` for affected k:
  - k involving i: `b[k] += delta * a[k-i]`
  - k involving j: `b[k] -= delta * a[k-j]`

For the top lag `k*`:
- If i is paired with k*: `b[k*] += delta * a[k*-i]`
- If j is paired with k*: `b[k*] -= delta * a[k*-j]`

So to reduce `b[k*]`, we want `a[k*-j] > a[k*-i]`, i.e., j is "less coupled" to k* than i is.

This gives a clear direction: move mass from elements strongly coupled to the top lag to elements weakly coupled.

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy:
- Anti-palindromic structured initializations (key structural insight)
- FISTA with adaptive softmax gradient (coarse to fine)
- Pressure-based mirror descent for fine polishing
- Novel pair-swap local search targeting top convolution lags
- Multi-length exploration (4096, 8192, 16384)
- Aggressive restarts with diverse perturbations

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_objective(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def softmax_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a."""
    n = len(a)
    c_shifted = c - c.max()
    exp_bc = np.exp(beta * c_shifted)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << ((2 * n - 1) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def hard_gradient(a, c, top_k):
    """Subgradient using top-k lags with softmax weighting."""
    n = len(a)
    k = min(top_k, len(c))
    # Sort lags by value, use softmax over top k
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices].copy()
    tv_shift = top_vals - top_vals.max()
    weights = np.exp(10.0 * tv_shift)
    weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]
    fsize = 1 << ((2 * n - 1) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista(a_init, S, time_budget, beta_start=1.0, beta_end=200.0, seed=0):
    """FISTA with adaptive beta scheduling and backtracking."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    L = 1.0  # Lipschitz estimate
    stall = 0
    last_best_score = np.inf
    best_improve_count = 0

    while time.monotonic() - start < time_budget:
        it += 1
        # Adaptive beta: exponential ramp
        frac = min(1.0, it / max(1, int(time_budget * 10)))
        beta = beta_start * (beta_end / beta_start) ** frac

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = y.copy()
            stall = 0
            best_improve_count += 1
        elif it % 30 == 0:
            stall += 1
        if best_improve_count > 200:
            # Converged, increase beta
            beta = min(beta * 2.0, beta_end * 5)

        grad = softmax_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Gradient step with backtracking
        step = 1.0 / L
        f_y = cur_max
        x_new = None
        for bt in range(8):
            x_cand = project_simplex(y - step * grad, S)
            if bt == 0:
                x_new = x_cand
            else:
                c_cand = fft_convolve_self(x_cand)
                f_cand = c_cand.max()
                if f_cand <= f_y + 1e-4 * step * gnorm * gnorm:
                    x_new = x_cand
                    break
                step *= 0.5
        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        # Update L
        c_x = fft_convolve_self(x_new)
        if c_x.max() > f_y + 1e-3:
            L *= 1.3
        elif it % 10 == 0:
            L = max(L * 0.97, 0.5)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic restart
        if stall > 50 and rng.random() < 0.4:
            noise_scale = 0.02 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            stall = 0

    return best_a, best_score


def run_mirror_descent(a_init, S, time_budget, top_k=20, eta_init=0.1, seed=0):
    """Mirror descent (multiplicative weights) with pressure-based updates."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    eta = eta_init

    while time.monotonic() - start < time_budget:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()

        # Compute pressure: weighted sum of coupling to top lags
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        tv_shift = top_vals - top_vals.max()
        w = np.exp(15.0 * tv_shift)
        w = w / w.sum()

        # Pressure on each element
        pressure = np.zeros(n)
        for weight, lag in zip(w, top_indices):
            lag = int(lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue
            idx = np.arange(i_min, i_max + 1)
            partner = lag - idx
            # Element i is coupled to partner in this lag
            contributions = a[partner]
            pressure[idx] += weight * contributions
            # Also the symmetric: element partner is coupled to idx
            pressure[partner] += weight * a[idx]

        # Normalize pressure
        p_mean = pressure.mean()
        p_std = pressure.std() + 1e-10
        pressure_norm = (pressure - p_mean) / p_std

        # Multiplicative update
        eta_decayed = eta_init / (1.0 + 0.005 * it)
        log_update = -eta_decayed * pressure_norm
        log_update = np.clip(log_update, -3.0, 3.0)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new * (S / s_new)

        # Occasional random restart
        if it % 2000 == 0 and rng.random() < 0.1:
            noise = 0.01 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)

    return best_a, best_score


def run_pair_local_search(a_init, S, time_budget, top_lags=5, seed=0):
    """
    Pair-swap local search: find element pairs (i,j) where moving mass
    from i to j reduces the max convolution.
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score - 1e-10:
            best_score = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            # Perturb to escape
            noise_scale = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            no_improve = 0
            continue

        # Get top lags
        top_idx = np.argsort(c)[::-1][:top_lags]

        # For each top lag, compute pressure on each element
        pressure = np.zeros(n)
        for lag in top_idx:
            lag = int(lag)
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min > i_max:
                continue
            idx = np.arange(i_min, i_max + 1)
            partner = lag - idx
            pressure[idx] += a[partner]
            pressure[partner] += a[idx]

        # Find high-pressure and low-pressure elements
        p_thresh_high = np.percentile(pressure, 90)
        p_thresh_low = np.percentile(pressure, 10)

        high_mask = pressure > p_thresh_high
        low_mask = pressure < p_thresh_low
        high_indices = np.where(high_mask)[0]
        low_indices = np.where(low_mask)[0]

        if len(high_indices) == 0 or len(low_indices) == 0:
            continue

        # Try several random swaps
        n_try = min(50, len(high_indices) * len(low_indices))
        best_swap_gain = 0
        best_i, best_j, best_delta = -1, -1, 0.0

        for _ in range(n_try):
            i = high_indices[rng.integers(len(high_indices))]
            j = low_indices[rng.integers(len(low_indices))]
            if i == j:
                continue
            # Try moving a small fraction of a[i] to a[j]
            delta = a[i] * 0.1
            if delta < 1e-15:
                continue

            a_trial = a.copy()
            a_trial[i] -= delta
            a_trial[j] += delta
            if a_trial[i] < 0:
                a_trial[i] = 0
                a_trial[j] += a[i]
                continue

            c_trial = fft_convolve_self(a_trial)
            gain = cur_max - c_trial.max()
            if gain > best_swap_gain:
                best_swap_gain = gain
                best_i, best_j, best_delta = int(i), int(j), float(delta)

        if best_i >= 0 and best_swap_gain > 0:
            a[best_i] -= best_delta
            a[best_j] += best_delta
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a * (S / s)

    return best_a, best_score

def make_anti_palindromic(n, rng, S, concentration=1.0):
    """
    Anti-palindromic init: a[i] + a[n-1-i] = const.
    With concentration parameter controlling how peaked the first half is.
    """
    half = n // 2
    # Create first half with power-law concentration
    t = np.arange(half, dtype=np.float64) / half
    h = (1.0 - t) ** concentration + 0.05
    h = h / h.sum() * (S / 2.0)

    mu = S / half  # Each pair sums to mu
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_periodic(n, rng, S, num_freqs=3):
    """Periodic initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 16)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.4
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.05)
    a = a * (S / a.sum())
    return a


def make_sparse_spike(n, rng, S, n_spikes=20):
    """Sparse spike initialization: a few large values."""
    a = np.ones(n) * 0.1
    spike_positions = rng.choice(n, size=n_spikes, replace=False)
    spike_magnitudes = rng.dirichlet(np.ones(n_spikes) * 0.5) * 0.8
    a[spike_positions] += spike_magnitudes
    a = a * (S / a.sum())
    return a


def make_bimodal(n, rng, S):
    """Bimodal: mass concentrated near both ends (U-shape)."""
    tt = np.arange(n, dtype=np.float64) / n
    sigma = rng.random() * 0.1 + 0.05
    a = np.exp(-0.5 * ((tt - 0.0) / sigma) ** 2) + np.exp(-0.5 * ((tt - 1.0) / sigma) ** 2)
    a = a + 0.01
    a = a * (S / a.sum())
    return a


def generate_diverse_inits(n, rng, S, parent=None):
    """Generate diverse initializations."""
    inits = []
    labels = []

    if parent is not None and len(parent) == n:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if p.sum() > 0:
            inits.append(p * (S / p.sum()))
            labels.append("parent")

    for conc in [0.5, 1.0, 2.0, 3.0]:
        inits.append(make_anti_palindromic(n, rng, S, concentration=conc))
        labels.append("anti_pal_c%d" % int(conc * 10))

    for i in range(2):
        inits.append(make_periodic(n, rng, S, num_freqs=rng.integers(2, 5)))
        labels.append("periodic%d" % i)

    inits.append(make_sparse_spike(n, rng, S, n_spikes=n // 100))
    labels.append("sparse")

    inits.append(make_bimodal(n, rng, S))
    labels.append("bimodal")

    # Random
    rand_a = rng.random(n) + 0.01
    inits.append(rand_a * (S / rand_a.sum()))
    labels.append("random")

    # Anti-palindromic with noise
    ap = make_anti_palindromic(n, rng, S, concentration=1.5)
    noise = rng.normal(0, 0.05 * S / np.sqrt(n), n)
    ap_noisy = project_simplex(ap + noise, S)
    inits.append(ap_noisy)
    labels.append("anti_pal_noisy")

    return list(zip(labels, inits))


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv)/sum^2.
    Uses multi-phase optimization with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best_obj = np.inf
    global_a = None
    global_n = 0

    # Try to get parent
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Parent available: %s (len=%d)" % (parent is not None, len(parent) if parent is not None else 0))

    # Phase 1: Multi-start at primary length
    primary_n = 8192
    if parent is not None and len(parent) >= 64:
        primary_n = len(parent)
        if primary_n > 16384:
            primary_n = 16384
        if primary_n < 1024:
            primary_n = 8192

    print("=== Phase 1: Multi-start at n=%d ===" % primary_n)
    S = np.sqrt(2.0 * primary_n)
    inits = generate_diverse_inits(primary_n, rng, S, parent)
    n_inits = len(inits)

    # Time allocation for Phase 1
    phase1_time = min(900.0, max(60.0, (deadline - time.monotonic()) * 0.55))
    per_init_time = phase1_time / n_inits

    for idx, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - 300:
            break
        n = len(arr)
        S_i = np.sqrt(2.0 * n)

        # Staggered: earlier inits get more time
        time_frac = 1.0 / (1.0 + 0.1 * idx)
        t_alloc = per_init_time * time_frac * n_inits / (n_inits * 0.5 + sum(1.0/(1.0+0.1*j) for j in range(n_inits)))
        t_alloc = max(10.0, min(t_alloc, per_init_time * 1.5))

        # Coarse FISTA
        seg_coarse = t_alloc * 0.5
        a_c, s_c = run_fista(arr, S_i, seg_coarse, beta_start=1.0, beta_end=50.0, seed=seed + idx)
        a_c = safe_normalize(a_c, n)
        c_c = fft_convolve_self(a_c)
        obj_c = 2.0 * n * c_c.max() / (S_i * S_i)
        if obj_c < global_best_obj:
            global_best_obj = obj_c
            global_a = a_c.copy()
            global_n = n
            print("  [%s] coarse FISTA: obj=%.6f (t=%.0fs)" % (name, obj_c, time.monotonic() - (deadline - phase1_time)))

        # Fine FISTA
        rem = t_alloc - (time.monotonic() - (deadline - phase1_time) + (phase1_time - per_init_time * n_inits / n_inits))
        # Simpler: just check remaining time
        if time.monotonic() < deadline - 300:
            seg_fine = min(t_alloc * 0.3, max(5.0, (deadline - time.monotonic()) * 0.02))
            a_f, s_f = run_hard_fista(a_c, S_i, seg_fine, top_k_start=3, top_k_end=32, seed=seed + 100 + idx)
            a_f = safe_normalize(a_f, n)
            c_f = fft_convolve_self(a_f)
            obj_f = 2.0 * n * c_f.max() / (S_i * S_i)
            if obj_f < global_best_obj:
                global_best_obj = obj_f
                global_a = a_f.copy()
                global_n = n
                print("  [%s] fine FISTA: obj=%.6f" % (name, obj_f))

    print("=== Phase 1 done: best=%.6f at n=%d ===" % (global_best_obj, global_n))

    # Phase 2: Polish the best found
    if global_a is not None and time.monotonic() < deadline - 120:
        n = len(global_a)
        S_i = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()

        print("=== Phase 2: Polishing (remaining %.0fs) ===" % remaining)

        # 2a: Mirror descent
        seg_md = max(10.0, remaining * 0.15)
        a_md, s_md = run_mirror_descent(global_a, S_i, seg_md, top_k=30, eta_init=0.08, seed=seed + 500)
        a_md = safe_normalize(a_md, n)
        c_md = fft_convolve_self(a_md)
        obj_md = 2.0 * n * c_md.max() / (S_i * S_i)
        if obj_md < global_best_obj:
            global_best_obj = obj_md
            global_a = a_md.copy()
            print("  Mirror descent: obj=%.6f" % obj_md)

        # 2b: Hard FISTA with high k
        remaining2 = deadline - time.monotonic()
        seg_hf = max(10.0, remaining2 * 0.20)
        a_hf, s_hf = run_hard_fista(global_a, S_i, seg_hf, top_k_start=10, top_k_end=128, seed=seed + 600)
        a_hf = safe_normalize(a_hf, n)
        c_hf = fft_convolve_self(a_hf)
        obj_hf = 2.0 * n * c_hf.max() / (S_i * S_i)
        if obj_hf < global_best_obj:
            global_best_obj = obj_hf
            global_a = a_hf.copy()
            print("  Hard FISTA: obj=%.6f" % obj_hf)

        # 2c: Pair local search
        remaining3 = deadline - time.monotonic()
        seg_pls = max(10.0, remaining3 * 0.25)
        a_pls, s_pls = run_pair_local_search(global_a, S_i, seg_pls, top_lags=5, seed=seed + 700)
        a_pls = safe_normalize(a_pls, n)
        c_pls = fft_convolve_self(a_pls)
        obj_pls = 2.0 * n * c_pls.max() / (S_i * S_i)
        if obj_pls < global_best_obj:
            global_best_obj = obj_pls
            global_a = a_pls.copy()
            print("  Pair local search: obj=%.6f" % obj_pls)

        # 2d: Final FISTA with very high beta
        remaining4 = deadline - time.monotonic()
        if remaining4 > 20:
            seg_final = remaining4 * 0.6
            a_fin, s_fin = run_fista(global_a, S_i, seg_final, beta_start=50.0, beta_end=500.0, seed=seed + 800)
            a_fin = safe_normalize(a_fin, n)
            c_fin = fft_convolve_self(a_fin)
            obj_fin = 2.0 * n * c_fin.max() / (S_i * S_i)
            if obj_fin < global_best_obj:
                global_best_obj = obj_fin
                global_a = a_fin.copy()
                print("  Final FISTA: obj=%.6f" % obj_fin)

    # Phase 3: Try different length if time permits
    if time.monotonic() < deadline - 200:
        alt_n = 4096 if global_n > 4096 else 16384
        if alt_n != global_n:
            S_alt = np.sqrt(2.0 * alt_n)
            print("=== Phase 3: Trying n=%d ===" % alt_n)
            inits_alt = generate_diverse_inits(alt_n, rng, S_alt, parent=None)
            # Quick FISTA on best 3 initializations
            for name, arr in inits_alt[:4]:
                if time.monotonic() > deadline - 100:
                    break
                S_a = np.sqrt(2.0 * len(arr))
                seg = max(5.0, (deadline - time.monotonic()) * 0.05)
                a_alt, s_alt = run_fista(arr, S_a, seg, beta_start=1.0, beta_end=100.0, seed=seed + 900)
                a_alt = safe_normalize(a_alt, len(arr))
                c_alt = fft_convolve_self(a_alt)
                obj_alt = 2.0 * len(arr) * c_alt.max() / (S_a * S_a)
                if obj_alt < global_best_obj:
                    global_best_obj = obj_alt
                    global_a = a_alt.copy()
                    global_n = len(arr)
                    print("  Alt n=%d [%s]: obj=%.6f" % (alt_n, name, obj_alt))

    # Phase 4: Final polish loop
    if global_a is not None and time.monotonic() < deadline - 30:
        n = len(global_a)
        S_i = np.sqrt(2.0 * n)
        # Rapid successive polish iterations
        remaining = deadline - time.monotonic()
        n_rounds = max(1, int(remaining / 15))
        for r in range(n_rounds):
            if time.monotonic() > deadline - 15:
                break
            seg = max(5.0, (deadline - time.monotonic()) / max(1, n_rounds - r))
            # Alternate between methods
            if r % 2 == 0:
                a_r, s_r = run_fista(global_a, S_i, seg, beta_start=100.0, beta_end=1000.0, seed=seed + 1000 + r)
            else:
                a_r, s_r = run_pair_local_search(global_a, S_i, seg, top_lags=3, seed=seed + 2000 + r)
            a_r = safe_normalize(a_r, n)
            c_r = fft_convolve_self(a_r)
            obj_r = 2.0 * n * c_r.max() / (S_i * S_i)
            if obj_r < global_best_obj:
                global_best_obj = obj_r
                global_a = a_r.copy()
                print("  Polish round %d: obj=%.6f" % (r, obj_r))

    # Final verification
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL: evaluate=%.6f, internal=%.6f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best_obj, len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: evaluator rejected, using fallback")
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver2 = evaluate_sequence(final_list)
        print("  Fallback: %.6f" % (ver2 if np.isfinite(ver2) else float('inf')))

    return final_list
