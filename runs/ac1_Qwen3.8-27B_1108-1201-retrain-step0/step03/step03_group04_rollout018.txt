def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization combining adaptive FISTA with geometric beta growth,

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient of weighted sum of convolution values w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k, temperature=0.0):
    """Compute normalized top-k weights with optional temperature smoothing."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    if temperature > 0:
        tv_max = top_vals.max()
        exp_vals = np.exp((top_vals - tv_max) / temperature)
        weights = np.zeros(len(c))
        weights[top_indices] = exp_vals / exp_vals.sum()
    else:
        total = top_vals.sum()
        if total < 1e-30:
            return np.full(len(c), 1.0 / len(c))
        weights = np.zeros(len(c))
        weights[top_indices] = top_vals / total
    return weights


def spectral_flatten(a, n, S, alpha=0.3, peak_fraction=0.05):
    """
    Spectral-domain peak flattening: reduce peaks of conv(a,a) by modifying in frequency domain.
    Returns a new (projected) sequence.
    """
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    A = fft(a, fsize)
    C = np.real(ifft(A * A))[:size]
    
    # Identify the peak level to flatten
    c_max = C.max()
    c_mean = C.mean()
    threshold = c_mean + peak_fraction * (c_max - c_mean)
    
    # Softly reduce peaks above threshold
    excess = np.maximum(C - threshold, 0.0)
    C_target = C - alpha * excess
    
    # Recover from frequency domain
    F_target = fft(C_target, fsize)
    
    # X^2 = F_target, so X = sqrt(F_target)
    # Complex square root
    F_safe = F_target.copy()
    # Handle near-zero values
    magnitude = np.abs(F_safe)
    phase = np.angle(F_safe)
    sqrt_mag = np.sqrt(magnitude)
    sqrt_phase = phase / 2.0
    X_new = sqrt_mag * np.exp(1j * sqrt_phase)
    
    # Enforce conjugate symmetry for real output
    X_new[0] = np.real(X_new[0]) + 0j
    for k in range(1, fsize // 2 + 1):
        X_new[k] = (X_new[k] + np.conj(X_new[fsize - k])) / 2.0
    if fsize % 2 == 0:
        X_new[fsize // 2] = np.real(X_new[fsize // 2]) + 0j
    
    a_new = np.real(ifft(X_new))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def run_fista_adaptive(a_init, n, S, max_time, rng, beta_init=80.0, beta_max=5000.0,

    best_local = a.copy()
    it = 0
    step = 2.0 * S / (n * beta_init)
    stall_counter = 0
    consecutive_no_improve = 0
    last_best = np.inf

    while time.monotonic() - start < max_time:

        beta = min(beta_max, beta_init * (geometric_rate ** it))

            best_local = y.copy()
            stall_counter = 0
            consecutive_no_improve = 0

        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        a = x_new
        t = t_new

            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            consecutive_no_improve = 0

        # Stronger perturbation when very stalled
        if stall_counter > stall_factor:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            step = 2.0 * S / (n * beta)
            stall_counter = 0
            consecutive_no_improve = 0

    return best_local, best_score

        print("Phase 2: Spectral iteration (%.0f sec)" % phase2_time)
        t0 = time.monotonic()
        a_p2, _ = run_spectral_iteration(global_a, n, S, phase2_time, rng, alpha=0.15)
        a_p2 = safe_normalize(a_p2, n)
        c_p2 = fft_convolve_self(a_p2)
        if c_p2.max() < global_best:
            global_best = c_p2.max()
            global_a = a_p2.copy()
            print("  Spectral: obj=%.6f [%.0fs]" % (2*n*global_best/S**2, time.monotonic()-t0))
        else:
            print("  Spectral: no improvement, best=%.6f [%.0fs]" % (2*n*global_best/S**2, time.monotonic()-t0))

    # === Phase 3: Top-k Nesterov with temperature (20% time) ===
    remaining = deadline - time.monotonic()
    phase3_time = 0.20 * remaining
    if phase3_time > 20:
        print("Phase 3: Top-k Nesterov (%.0f sec)" % phase3_time)
        t0 = time.monotonic()
        a_p3, _ = run_toppk_nesterov(global_a, n, S, phase3_time, rng,
                                      top_k_init=1, top_k_max=30,
                                      top_k_rate=150, temp_schedule=True)
        a_p3 = safe_normalize(a_p3, n)
        c_p3 = fft_convolve_self(a_p3)
        if c_p3.max() < global_best:
            global_best = c_p3.max()
            global_a = a_p3.copy()
            print("  Nesterov: obj=%.6f [%.0fs]" % (2*n*global_best/S**2, time.monotonic()-t0))
        else:
            print("  Nesterov: no improvement, best=%.6f [%.0fs]" % (2*n*global_best/S**2, time.monotonic()-t0))

    # === Phase 4: Ultra-focused polish (15% time) ===
    remaining = deadline - time.monotonic()
    phase4_time = 0.15 * remaining
    if phase4_time > 10:
        print("Phase 4: Ultra-focus polish (%.0f sec)" % phase4_time)
        t0 = time.monotonic()
        a_p4, _ = run_ustar_polish(global_a, n, S, phase4_time, rng)
        a_p4 = safe_normalize(a_p4, n)
        c_p4 = fft_convolve_self(a_p4)
        if c_p4.max() < global_best:
            global_best = c_p4.max()
            global_a = a_p4.copy()
            print("  Polish: obj=%.6f [%.0fs]" % (2*n*global_best/S**2, time.monotonic()-t0))
        else:
            print("  Polish: no improvement, best=%.6f [%.0fs]" % (2*n*global_best/S**2, time.monotonic()-t0))

    # === Phase 5: Diverse restarts (10% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("Phase 5: Diverse restarts (%.0f sec)" % remaining)
        n_variants = 4
        per_variant = remaining * 0.8 / n_variants
        for v in range(n_variants):
            if time.monotonic() > deadline - 30:
                break

            if v == 0:
                # Gaussian perturbation
                perturbed = global_a + rng.normal(0, 0.004 * S / np.sqrt(n), n)
            elif v == 1:
                # Smooth (low-frequency) perturbation
                noise = rng.normal(0, 1, n)
                smooth = np.convolve(noise, np.ones(128) / 128, mode='same')
                smooth = smooth / (np.abs(smooth).max() + 1e-15)
                perturbed = global_a + 0.003 * S / np.sqrt(n) * smooth
            elif v == 2:
                # Multiplicative perturbation
                perturbed = global_a * (1.0 + rng.normal(0, 0.008, n))
            else:
                # Structured perturbation (sinusoidal)
                tt = np.linspace(0, 2 * np.pi, n, endpoint=False)
                freq = rng.integers(3, 20)
                pert = 0.004 * S / np.sqrt(n) * np.sin(freq * tt + rng.uniform(0, 2*np.pi))
                perturbed = global_a + pert

            perturbed = safe_normalize(perturbed, n)

            # Quick FISTA on perturbed
            a_v, _ = run_fista_adaptive(perturbed, n, S, per_variant, rng,
                                         beta_init=100.0, beta_max=3000.0,
                                         geometric_rate=1.006, stall_factor=150)
            a_v = safe_normalize(a_v, n)
            c_v = fft_convolve_self(a_v)
            if c_v.max() < global_best:
                global_best = c_v.max()
                global_a = a_v.copy()
                print("  Variant %d: obj=%.6f" % (v, 2*n*global_best/S**2))

    # === Phase 6: Final refinement (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("Phase 6: Final refinement (%.0f sec)" % remaining)

        # Spectral + subgradient alternation
        a_f1, _ = run_spectral_iteration(global_a, n, S, remaining * 0.35, rng, alpha=0.1)
        a_f1 = safe_normalize(a_f1, n)
        c_f1 = fft_convolve_self(a_f1)
        if c_f1.max() < global_best:
            global_best = c_f1.max()
            global_a = a_f1.copy()
            print("  Refine1: obj=%.6f" % (2*n*global_best/S**2))

        remaining2 = deadline - time.monotonic()
        if remaining2 > 5:
            a_f2, _ = run_ustar_polish(global_a, n, S, remaining2, rng)
            a_f2 = safe_normalize(a_f2, n)
            c_f2 = fft_convolve_self(a_f2)
            if c_f2.max() < global_best:
                global_best = c_f2.max()
                global_a = a_f2.copy()
                print("  Refine2: obj=%.6f" % (2*n*global_best/S**2))

    # === Final output ===
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    final_obj = 2.0 * n * global_best / (S * S)
    print("FINAL: length=%d, max_conv=%.6f, objective=%.6f" % (len(final_list), global_best, final_obj))

    # Verify
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: %.6f" % ver)
    else:
        print("Verification failed, using fallback")
        final_list = [0.5] * 256

    return final_list
