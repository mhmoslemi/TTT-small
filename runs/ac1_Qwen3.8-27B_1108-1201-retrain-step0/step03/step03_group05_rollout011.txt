"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Key insight: The middle of the convolution is sum_j a_j*a_{n-1-j} (NOT sum a_j^2
for non-symmetric sequences). Asymmetry allows beating the Cauchy-Schwarz bound.

Algorithm:
- L-BFGS-B on softmax parameterization with log-sum-exp continuation
- Direct subgradient polishing with adaptive top-k peak selection using "excess over mean"
- Spectral rebalancing: project DFT toward flat magnitude
- Multiple asymmetric initializations
- Adaptive time allocation across phases

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_grad_a(a, w):
    """Compute grad_a[j] = 2 * sum_k w[k] * a[j-k] (cross-correlation via FFT)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """
    Compute smoothed objective and gradient w.r.t. z (unconstrained softmax params).
    a = S * softmax(z), objective = (1/beta)*log(sum exp(beta*c_k))
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_grad_a(a, w)

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def softmax_obj_grad_excess(z, beta, S, excess_thresh=0.0):
    """
    Smoothed objective using 'excess over mean' for peak selection.
    This focuses optimization on lags that are significantly above average.
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_mean = c.mean()
    c_excess = np.maximum(c - c_mean, 0.0)
    c_max = c.max()
    
    # Use log-sum-exp on excess values
    if c_excess.max() < 1e-15:
        obj = c_max
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        grad_a = compute_grad_a(a, w)
        dot_ga_a = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot_ga_a / S)
        return obj, grad_z

    exp_bc = np.exp(beta * (c_excess - c_excess.max()))
    sum_exp_bc = exp_bc.sum()
    obj = (c_mean + c_excess.max() + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    # Zero out weights where excess is zero
    w[c_excess < 1e-15] = 0.0
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum

    grad_a = compute_grad_a(a, w)
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=300,
                           use_excess=False):
    """
    Run L-BFGS-B with continuation on beta (smooth-to-sharp).
    Returns best (a, score) found.
    """
    if beta_schedule is None:
        beta_schedule = [3, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break

        n_attempts = 1
        if beta >= 500 and max_time > 60:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 3:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.01, n)
                z = z - z.mean()

            if use_excess:
                def fun(z_val):
                    return softmax_obj_grad_excess(z_val, beta, S)[0]
                def jac(z_val):
                    return softmax_obj_grad_excess(z_val, beta, S)[1]
            else:
                def fun(z_val):
                    return softmax_obj_grad(z_val, beta, S)[0]
                def jac(z_val):
                    return softmax_obj_grad(z_val, beta, S)[1]

            try:
                result = scipy_minimize(
                    fun, z,
                    jac=jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_beta,
                        'maxfun': maxiter_per_beta * 3,
                        'ftol': 1e-15,
                        'gtol': 1e-10,
                        'maxls': 50,
                        'iprint': -1
                    }
                )
            except Exception:
                break

            z = result.x
            z = z - z.mean()

            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)

            else:
                a = a * (S / s_val)

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def spectral_rebalance(a, n, S, n_iter=50, rng=None):
    """
    Push the DFT magnitude toward flatness, then project back to non-negative simplex.
    This exploits the relationship: flat autocorrelation <-> specific spectral structure.
    """
    a = a.copy().astype(np.float64)
    for _ in range(n_iter):
        N = 1 << (2 * n - 1 - 1).bit_length()
        A = fft(a, N)
        mag = np.abs(A)
        # Target: make |A| as flat as possible (but keep DC component)
        target_mag = np.full(N, mag[N // 2] / np.sqrt(N - 1))
        # Blend: move 10% toward target
        new_mag = mag + 0.1 * (target_mag - mag)
        new_mag = np.maximum(new_mag, 0)
        # Keep phase
        phase = np.angle(A)
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(new_A, N))[:n]
        a_new = np.maximum(a_new, 0)
        s_val = a_new.sum()
        if s_val > 1e-10:
            a_new = a_new * S / s_val
        else:
            break
        a = a_new
    return a


def run_subgradient_polish(a_init, n, S, max_time, rng,
                           top_k_start=3, top_k_max=128):
    """
    Subgradient method with adaptive top-k and backtracking line search.
    Uses 'excess over mean' for peak selection.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_start + it // 100)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            # Restart with perturbation
            y = project_simplex(y + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            t = 1.0
            no_improve = 0
            continue

        # Select peaks based on excess over mean
        c_mean = c.mean()
        c_excess = c - c_mean
        top_indices = np.argsort(c_excess)[::-1][:top_k]
        top_vals = np.maximum(c_excess[top_indices], c[top_indices].min())
        total = top_vals.sum()
        if total < 1e-30:
            top_indices = np.argsort(c)[::-1][:top_k]
            top_vals = c[top_indices]
            total = top_vals.sum()
            if total < 1e-30:
                break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        grad = compute_grad_a(a, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step_base = 2.0 * S / (n * (1.0 + 0.001 * it))
        step = step_base
        x_candidate = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        # Check if this improves
        c_x = fft_convolve_self(x_candidate)
        if c_x.max() > cur:
            # Reduce step
            step *= 0.5
            x_candidate = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_candidate + ((t - 1.0) / t_new) * (x_candidate - a)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        a = x_candidate
        t = t_new

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""

    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

                    global_a = a_new.copy()
                    print("  Perturb %d (strat=%d): %.8f" % (p, strategy, global_best))

    # --- Phase 5: Cross-pollination between best phase 1 results ---
    if len(phase1_best) >= 2:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("\n=== Phase 5: Cross-pollination (%.1f sec) ===" % remaining * 0.2)
            
            sorted_results = sorted(phase1_best.items(), key=lambda x: x[1][1])
            n_xpoll = max(2, int(remaining * 0.2 / 30))
            xpoll_budget = remaining * 0.2 / n_xpoll

            for xp in range(n_xpoll):
                if time.monotonic() > deadline - 30:
                    break

                idx1 = xp % len(sorted_results)
                idx2 = (xp + 1) % len(sorted_results)
                if idx2 == idx1:
                    idx2 = (idx2 + 1) % len(sorted_results)

                a1 = sorted_results[idx1][1][0]
                a2 = sorted_results[idx2][1][0]

                for blend in [0.3, 0.5, 0.7]:
                    if time.monotonic() > deadline - 20:
                        break
                    a_blend = blend * a1 + (1 - blend) * a2
                    a_blend = safe_normalize(a_blend, n)

                    a_new, score = run_lbfgs_continuation(
                        a_blend, n, S, xpoll_budget / 3, rng,
                        beta_schedule=[50, 200, 500, 1500],
                        maxiter_per_beta=100
                    )
                    a_new = safe_normalize(a_new, n)
                    c_new = fft_convolve_self(a_new)
                    if c_new.max() < global_best:
                        global_best = c_new.max()
                        global_a = a_new.copy()
                        print("  XPoll %d (blend=%.1f): %.8f" % (xp, blend, global_best))

    # --- Phase 6: Final high-precision polishing ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        print("\n=== Phase 6: Final Polish (%.1f sec) ===" % remaining)

        if remaining > 20:
            # Final Nesterov polish with aggressive parameters
            seg1 = min(remaining * 0.4, 120)
            a_polish, score = run_subgradient_polish(
                global_a, n, S, seg1, rng,
                top_k_start=3, top_k_max=64
            )
            a_polish = safe_normalize(a_polish, n)
            c = fft_convolve_self(a_polish)
            if c.max() < global_best:
                global_best = c.max()
                global_a = a_polish.copy()
                print("  Nesterov final: %.8f" % global_best)

        # Final L-BFGS with very high beta
        remaining2 = deadline - time.monotonic()
        if remaining2 > 15:
            a_final, score = run_lbfgs_continuation(
                global_a, n, S, remaining2, rng,
                beta_schedule=[500, 1000, 2000, 5000, 10000, 20000],
                maxiter_per_beta=300
            )
            a_final = safe_normalize(a_final, n)
            c_final = fft_convolve_self(a_final)
            if c_final.max() < global_best:
                global_best = c_final.max()
                global_a = a_final.copy()
                print("  Final L-BFGS: %.8f" % global_best)

    # --- Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.8f" % (len(global_a), global_best))
    print("Time used: %.1fs" % (time.monotonic() - (deadline - budget_s + 15)))
    return [float(x) for x in global_a]
