```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: LP-based majorization-minimization (best-response iteration) 
combined with FISTA polishing. The core idea is to repeatedly solve a 
linear program that finds the best response direction (maximizing the 
sum of the new sequence subject to convolution bounds with the current 
sequence), then blend towards it. This is a direct linearization of 
the quadratic minimax problem and should converge faster than 
pure gradient methods for the final polishing stage.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c) with temperature beta."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, n, S, max_time, rng, beta_init=50.0, beta_rate=0.003, beta_max=2000.0):
    """FISTA with LSE approximation for polishing."""
    a = project_simplex(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it - last_improve > 3000:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best
```

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: LP-based best-response iteration combined with FISTA polishing.
At each step, solve an LP to find the direction that maximizes the sum
of the new sequence while keeping the cross-convolution with the current
sequence bounded. This provides a more directed update than gradient
methods, especially near the optimum. FISTA with high-temperature LSE
is used for final polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import coo_matrix

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def build_conv_matrix_sparse(f):
    """Build the (2n-1) x n convolution matrix."""
    n = len(f)
    # Use COO format
    cols = []
    rows = []
    data = []
    for j in range(n):
        r_start = j
        r_end = j + n
        actual_end = min(r_end, 2*n - 1)
        for i in range(actual_end - r_start):
            rows.append(r_start + i)
            cols.append(j)
            data.append(f[i])
    T = coo_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return T.tocsr()

def solve_lp_best_response(f, M):
    """Solve LP: max 1^T g s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    T = build_conv_matrix_sparse(f)
    
    import cvxpy as cp
    g = cp.Variable(n, nonneg=True)
    obj = cp.Maximize(cp.sum(g))
    constraints = [T @ g <= M]
    prob = cp.Problem(obj, constraints)
    
    for solver in ['GUROBI', 'MOSEK', 'CLARABEL', 'PDLP', 'GLOP', 'HIGHS']:
        try:
            prob.solve(solver=solver)
            if g.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
                return np.maximum(np.asarray(g.value).flatten(), 0)
        except Exception:
            continue
    
    # Fallback to scipy linprog
    from scipy.optimize import linprog
    # Max 1^T g  =>  Min -1^T g
    # s.t. T @ g <= M  (equality form: T @ g <= M*ones)
    #      g >= 0
    res = linprog(c=-np.ones(n), A_ub=T, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)]*n, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    
    return None

def run_fista(a_init, n, S, max_time, rng, beta_init=50.0, beta_rate=0.003, beta_max=2000.0):
    a = project_simplex(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (2.0 * beta)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        if it - last_improve > 3000:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best
```

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: LP-based best-response majorization-minimization combined with
adaptive FISTA polishing. The LP step finds the optimal direction by
solving max(1^T g) s.t. (f*g)_k <= M, g >= 0, which provides a more
directed update than gradient methods. FISTA with exponentially increasing
LSE temperature handles final convergence. Multiple diverse initializations
and adaptive step-size control ensure robustness.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.sparse import coo_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of beta * logsumexp(c) - used as smooth approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def build_conv_matrix(f):
    """Build sparse (2n-1) x n convolution matrix T where T[k,j] = f[k-j]."""
    n = len(f)
    n_flag = n
    # Build in COO format efficiently
    # Column j has nonzeros at rows j, j+1, ..., j+n-1 (clipped to [0, 2n-2])
    # with values f[0], f[1], ..., f[n-1]
    nnz = n * n  # Each column has exactly n nonzeros (for n >= 1, 2n-1 >= n)
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)
    pos = 0
    for j in range(n):
        end = min(j + n, 2 * n - 1)
        cnt = end - j
        rows[pos:pos+cnt] = np.arange(j, end, dtype=np.int32)
        cols[pos:pos+cnt] = j
        data[pos:pos+cnt] = f[:cnt]
        pos += cnt
    T = coo_matrix((data[:pos], (rows[:pos], cols[:pos])), shape=(2*n-1, n))
    return T.tocsr()


def solve_lp_best_response(f, M, timeout=None):
    """Solve LP: max 1^T g s.t. conv_matrix(f) @ g <= M*1, g >= 0."""
    n = len(f)
    
    # Try cvxpy first
    try:
        import cvxpy as cp
        T = build_conv_matrix(f)
        g = cp.Variable(n, nonneg=True)
        obj = cp.Maximize(cp.sum(g))
        constraints = [T @ g <= M]
        prob = cp.Problem(obj, constraints)
        
        for solver in ['GUROBI', 'MOSEK', 'CLARABEL', 'PDLP', 'GLOP']:
            try:
                if timeout is not None:
                    prob.solve(solver=solver, time_limit=timeout)
                else:
                    prob.solve(solver=solver)
                if g.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
                    return np.maximum(np.asarray(g.value).flatten(), 0.0)
            except Exception:
                continue
        del T
    except Exception:
        pass
    
    # Fallback: scipy linprog with HiGHS
    try:
        from scipy.optimize import linprog
        T = build_conv_matrix(f)
        res = linprog(c=-np.ones(n),
                      A_ub=T, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': 10.0})
        if res.success:
            return np.maximum(res.x, 0.0)
        del T
    except Exception:
        pass
    
    return None


def run_lp_iteration(a, n, S, rng, lp_n=None):
    """
    One LP-based best-response iteration.
    Returns (new_a, improvement_flag).
    """
    if lp_n is None or lp_n >= n:
        # Work at full size
        c = fft_convolve_self(a)
        M = c.max()
        if M < 1e-10:
            return a.copy(), False
        g = solve_lp_best_response(a, M, timeout=30)
        if g is None or g.sum() < 1e-10:
            return a.copy(), False
        g = g * (S / g.sum())
        g = project_simplex(g, S)
        # Line search for best blend
        best_t = 0.1
        best_score = M
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            candidate = (1 - t) * a + t * g
            candidate = project_simplex(candidate, S)
            cc = fft_convolve_self(candidate)
            sc = cc.max()
            if sc < best_score:
                best_score = sc
                best_t = t
        new_a = (1 - best_t) * a + best_t * g
        new_a = project_simplex(new_a, S)
        improved = new_a.max() >= 0 and best_score < M
        return new_a, improved
    else:
        # Downsample, solve LP, upsample
        n_small = lp_n
        nblocks = n // n_small
        if nblocks < 2:
            return a.copy(), False
        a_small = a[:nblocks * nsmall].reshape(n_small, nblocks).mean(axis=1)
        S_small = np.sqrt(2.0 * n_small)
        a_small = project_simplex(a_small, S_small)
        
        c_small = fft_convolve_self(a_small)
        M_small = c_small.max()
        g_small = solve_lp_best_response(a_small, M_small, timeout=15)
        if g_small is None or g_small.sum() < 1e-10:
            return a.copy(), False
        g_small = g_small * (S_small / g_small.sum())
        
        # Upsample: repeat each element nblocks times
        g_full = np.repeat(g_small, nblocks)
        if len(g_full) < n:
            g_full = np.concatenate([g_full, np.full(n - len(g_full), g_full[-1])])
        g_full = g_full[:n]
        g_full = project_simplex(g_full, S)
        
        # Blend
        new_a = 0.5 * a + 0.5 * g_full
        new_a = project_simplex(new_a, S)
        return new_a, True


def run_lp_phase(a_init, n, S, max_time, rng, lp_n=None):
    """Run LP-based iterations until time runs out or convergence."""
    a = project_simplex(a_init, S)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    c = fft_convolve_self(a)
    best = c.max()
    
    no_improve_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        a_new, improved = run_lp_iteration(a, n, S, rng, lp_n=lp_n)
        
        c_new = fft_convolve_self(a_new)
        score = c_new.max()
        
        if score < best - 1e-10:

            best_a = a_new.copy()
            a = a_new
            no_improve_count = 0
        else:

                a_pert = project_simplex(a + noise, S)

                if score < best - 1e-10:

    return best_a, best


def run_fista(a_init, n, S, max_time, rng, beta_init=50.0, beta_rate=0.003, beta_max=3000.0):
    """FISTA with LSE for polishing."""
    a = project_simplex(a_init, S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        # Adaptive step: LSE is beta-Lipschitz, so step = 1/(2*beta) is safe
        # But we can use a larger step initially
        step = 1.0 / (2.0 * beta)
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

        if it - last_improve > 2000:
            # Stuck - inject noise and restart momentum
            decay = max(0.001, 0.01 * np.exp(-it / 5000.0))
            noise_scale = decay * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_k = 1.0
            last_improve = it

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_subgradient_polish(a_init, n, S, max_time, rng):
    """Polish using exact subgradient at peak lag with adaptive step."""
    a = project_simplex(a_init, S)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
        
        # Find all lags near the peak (top-k)
        thresh = cur * 0.999
        active = np.where(c >= thresh)[0]
        
        # Build subgradient: average of subgradients at active lags
        grad = np.zeros(n)
        for k in active[:64]:  # Limit to 64 active constraints
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(active) if len(active) > 0 else 1
        
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        # Polyak-like step: we know the gap is at least (cur - best_known_lower)
        # Use a simple decreasing step
        step = 2.0 * S / (n * np.sqrt(1.0 + it * 0.01))
        step = min(step, 1.0)
        
        x_new = project_simplex(y - step * grad / gn, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

    return best_a, best


def generate_sinc_init(n, rng):
    """Generate a non-negative sinc-like initialization."""
    center = n / 2.0
    # Try different widths
    best = None
    best_score = np.inf
    for width in [n * 0.1, n * 0.2, n * 0.3, n * 0.4, n * 0.5]:
        x = np.arange(n) - center
        a = np.sinc(x / width)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a * (np.sqrt(2.0 * n) / s)
        c = fft_convolve_self(a)
        sc = c.max()
        if sc < best_score:
            best_score = sc
            best = a.copy()
    if best is None:
        best = np.ones(n) * np.sqrt(2.0 * n) / n
    return best


def generate_diverse_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(np.exp(-8.0 * (tt - 0.5) ** 2))
    
    # Multi-sinusoidal
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt + 0.5))
    inits.append(1.0 + 0.4 * np.cos(2 * np.pi * 5 * tt) + 0.2 * np.sin(2 * np.pi * 2 * tt + 1.0))
    
    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.8, n)))
    
    # Block structure
    for nb in [4, 8, 16]:
        blen = n // nb
        heights = rng.uniform(0.2, 2.0, nb)
        block = np.repeat(heights, blen)
        if len(block) < n:
            block = np.concatenate([block, np.full(n - len(block), heights[-1])])
        inits.append(block[:n])
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.8
    alt[1::2] = 0.4
    inits.append(alt)
    
    return [np.maximum(x, 0.01) for x in inits]

            global_a = project_simplex(a_fin, S)
            print("  Final FISTA improved to: %.8f" % global_best)
    
    # Final verification
    if global_a is None:
        global_a = np.ones(n) * S / n
    
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    
    # Verify with evaluate_sequence
    try:
        ver = evaluate_sequence(final_list)
        print("Verification score: %.8f" % ver)
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected! Using fallback.")
            global_a = np.ones(2048, dtype=np.float64) * (np.sqrt(2 * 2048) / 2048)
            final_list = [float(x) for x in global_a]
    except Exception as e:
        print("  Verification error: %s" % str(e))
    
    # Ensure validity
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(2048, dtype=np.float64) * (np.sqrt(2 * 2048) / 2048)
    
    result = [float(x) for x in global_a]
    print("Returning sequence of length %d, internal score=%.8f" % (len(result), global_best))
    return result
```
