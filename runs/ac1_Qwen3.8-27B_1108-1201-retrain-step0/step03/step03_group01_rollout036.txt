```python
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Water-filling algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def self_convolve(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    A = fft(a, L)
    c = np.real(ifft(A * A))
    return c[:size]

def lse_objective_and_grad(a, beta):
    """Compute LSE objective and gradient."""
    n = len(a)
    b = self_convolve(a)
    # Numerical stability: shift by max
    b_shift = b - b.max()
    exp_b = np.exp(beta * b_shift)
    Z = exp_b.sum()
    w = exp_b / Z
    obj = b.max() + np.log(Z) / beta
    
    # Gradient: 2 * (w cross-correlate a)
    L = 1 << (3 * n - 2 - 1).bit_length()  # >= 3n-2
    W = fft(w, L)
    A = fft(a, L)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    return obj, grad, b, w

def lse_nesterov(a0, n, S, max_time, rng, beta_init=5.0, beta_max=5000.0, 
                  beta_growth=1.001, step_init=1.0, step_decay=0.999):
    """Nesterov-accelerated LSE optimization with adaptive beta."""
    a = project_simplex(a0.copy(), S)
    y = a.copy()
    t = 1.0
    beta = beta_init
    step = step_init
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    stale = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        obj, grad, b, w = lse_objective_and_grad(y, beta)
        bmax = b.max()
        if bmax < best_score:
            best_score = bmax
            best_a = y.copy()
            stale = 0
        else:
            stale += 1
        
        # Update beta (increase over time for sharper approximation)
        beta = min(beta_max, beta * beta_growth)
        step = max(0.01, step * step_decay)
        
        # Nesterov step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_normalized = grad / gnorm
        x_new = project_simplex(y - step * grad_normalized, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        # Re-project to maintain sum constraint
        y = project_simplex(y, S)
        a = x_new
        t = t_new
        
        # Periodic restart to escape local minima
        if stale > 500 and rng.random() < 0.05:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t = 1.0
            beta = beta_init
            stale = 0
    
    return best_a, best_score
```
