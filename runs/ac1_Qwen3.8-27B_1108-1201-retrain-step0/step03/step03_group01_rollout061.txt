def project_simplex(v, S):

    n = v.size
    if n == 0: return v
    if S <= 0: return np.zeros(n, dtype=np.float64)
    v = v.copy()
    mn = v.min()
    if mn >= -1e-12:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 1e-300:
            return np.full(n, S/n, dtype=np.float64)
        return v * (S/s)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = u - (css - S)/idx > 0.0

"""
Search for non-negative step-function coefficients minimizing

    2*n*max(convolve(a,a))/(sum(a)^2)

by working with the scale-invariant normalization sum(a) = sqrt(2n), under which
the score is exactly max(convolve(a,a)).

The main idea is to optimize the square spectrum F(omega)^2 directly, not the
power spectrum |F(omega)|^2.  A global spectral move multiplies F by sqrt(D),
where D is chosen to suppress the currently largest convolution coefficients.
This is alternated with relative/additive gradient moves using active-lag
cross-correlations.  Multi-start, symmetrization, and random restarts are used
to avoid local minima.
"""

import os

os.environ.setdefault("OMP_NUM_THREADS", "1")
os.environ.setdefault("OPENBLAS_NUM_THREADS", "1")
os.environ.setdefault("MKL_NUM_THREADS", "1")

import numpy as np
import time
import operator
from scipy.fft import fft, ifft


def _fft(x, N):
    return fft(x, n=N)


def _ifft(x, N):
    return ifft(x, n=N)


def make_final_list_global(fallback):
    try:
        cutter = float(os.environ.get("SSCUTTER", "0.0"))
    except Exception:
        cutter = 0.0
    out = list(map(float, list(fallback)))
    return out


def propose_candidate(seed=42, budget_s=2350.0, **kwargs):
    start_wall = time.monotonic()

    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2350.0
    seed = int(seed)

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Length choice.  Prefer the inherited construction if it is in a
    # reasonable range; otherwise use 8192.
    # ------------------------------------------------------------------
    gaveq = None
    try:
        gaveq = globals().get("height_sequence_1", None)
    except Exception:
        gaveq = None

    n = 8192
    if gaveq is not None:
        try:
            gaveq_arr = np.asarray(gaveq, dtype=np.float64).ravel()
            glen = int(gaveq_arr.size)
            if 512 <= glen <= 16384:
                n = glen
        except Exception:
            pass

    N = 1 << (2 * n - 2).bit_length()
    if N < 2 * n - 1:
        N <<= 1

    L = 2 * n - 1
    S = float(np.sqrt(2.0 * float(n)))
    mean_scale = S / float(n)

    phase = (2.0 * np.pi / float(N)) * np.arange(N, dtype=np.float64)

    # ------------------------------------------------------------------
    # Helper functions are intentionally defined inside propose_candidate
    # only to see n, N, L, S, phase.  They contain no closures over mutable
    # state in the forbidden sense; no lambdas are used.
    # ------------------------------------------------------------------

    def norm_to_sum(a, S_target):
        a = np.asarray(a, dtype=np.float64).ravel().copy()
        if a.size == 0:
            return a
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s <= 1e-300:
            return np.full(a.shape, S_target / float(a.size), dtype=np.float64)
        return a * (S_target / s)

    def project_simplex(v, S_target):
        v = np.asarray(v, dtype=np.float64).ravel().copy()
        nv = v.size
        if nv == 0:
            return v
        if S_target <= 0.0:
            return np.zeros(nv, dtype=np.float64)

        # Fast exact path when already non-negative.
        if v.min() >= -1e-12:
            v = np.maximum(v, 0.0)
            s = float(v.sum())
            if s <= 1e-300:
                return np.full(nv, S_target / float(nv), dtype=np.float64)
            return v * (S_target / s)

        u = np.sort(v)[::-1]
        css = np.cumsum(u)
        idx = np.arange(1, nv + 1, dtype=np.float64)
        cond = u - (css - S_target) / idx > 0.0
        if np.any(cond):
            i = int(np.max(np.where(cond)[0]))
            theta = (css[i] - S_target) / float(i + 1)
        else:
            theta = S_target / float(nv)

        out = v - theta
        out[out < 0.0] = 0.0
        s = float(out.sum())
        if s <= 1e-300:
            return np.full(nv, S_target / float(nv), dtype=np.float64)
        if abs(s - S_target) > 1e-9 * max(1.0, S_target):
            out *= (S_target / s)
        return out

    def interpolate_sequence(src, target_len):
        src = np.asarray(src, dtype=np.float64).ravel()
        src = np.clip(src, 0.0, 1000.0)
        m = int(src.size)
        if m == 0:
            return np.zeros(target_len, dtype=np.float64)
        if m == target_len:
            return src.copy()
        if m == 1:
            return np.full(target_len, float(src[0]), dtype=np.float64)
        xs = np.linspace(0.0, 1.0, m, endpoint=False, dtype=np.float64)
        xt = np.linspace(0.0, 1.0, target_len, endpoint=False, dtype=np.float64)
        return np.interp(xt, xs, src)

    def conv_from_fft(F, N_used, n_used):
        L_used = 2 * n_used - 1
        c = _ifft(F * F, N_used)[:L_used].real
        c = 0.5 * (c + c[::-1])
        c[c < 0.0] = 0.0
        return c

    def conv_coeffs(a, N_used, n_used):
        F = _fft(a, N_used)
        return conv_from_fft(F, N_used, n_used)

    def score_from_coeffs(c, a_sum, n_used):
        if a_sum <= 1e-300:
            return np.inf
        return float(2.0 * float(n_used) * float(c.max()) / (a_sum * a_sum))

    def score_fast(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        s = float(a.sum())
        if s <= 1e-300:
            return np.inf
        c = conv_coeffs(a, N, n)
        return score_from_coeffs(c, s, n)

        base_vals = np.maximum(np.asarray(base_vals, dtype=np.float64), 0.0)
        if top_k is None:
            p[:] = np.power(base_vals, float(power))
            p[p < 1e-300] = 0.0
            return p

        top_k = int(top_k)
        if top_k <= 0:
            return p
        if top_k >= n:
            p[:] = np.power(base_vals, float(power))
        else:
            idx = np.argpartition(base_vals, n - top_k)[-top_k:]
            p[idx] = np.power(base_vals[idx], float(power))
        p[p < 1e-300] = 0.0
        return p

    def make_soft_p(base_vals, cmax_val, q_val):
        p = np.zeros(n, dtype=np.float64)
        base_vals = np.maximum(np.asarray(base_vals, dtype=np.float64), 0.0)
        if cmax_val <= 1e-300:
            return p
        if q_val is None or float(q_val) <= 0.0:
            tau = max(0.03 * float(cmax_val), 1e-12)
            p = np.exp((base_vals - float(cmax_val)) / tau)
        else:
            q_val = float(q_val)
            p = np.power(base_vals / max(float(cmax_val), 1e-300), q_val)
        p[p < 1e-300] = 0.0
        return p

    def full_weights_from_p(p):
        p = np.maximum(np.asarray(p, dtype=np.float64), 0.0)
        center = n - 1
        total = 2.0 * float(p.sum()) - float(p[center])
        if total <= 1e-300:
            return np.zeros(L, dtype=np.float64), 0.0

        out = np.zeros(L, dtype=np.float64)
        scaled = p / total
        out[:n] = scaled
        if n > 1:
            out[n:L] = p[:n - 1][::-1] / total
        return out, total

    def spectral_lag_weights_from_p(p):
        p = np.maximum(np.asarray(p, dtype=np.float64), 0.0)
        if n <= 1:
            return None, None
        noncenter = p.copy()
        noncenter[n - 1] = 0.0
        total_non = 2.0 * float(noncenter.sum())
        if total_non <= 1e-300:
            return None, None
        nz = np.nonzero(noncenter[:n - 1])[0]
        if nz.size == 0:
            return None, None
        weights = 2.0 * noncenter[nz] / total_non
        return nz, weights

    def compute_cos_eff(lags, weights):
        out = np.zeros(N, dtype=np.float64)
        for i in range(lags.size):
            w = float(weights[i])
            if w > 1e-18:
                ell = int(lags[i])
                out += w * np.cos(phase * ell)
        return out

    def apply_spectral_square(F, cos_eff, alpha):
        D = 1.0 - float(alpha) * cos_eff
        D = np.maximum(D, 1e-14)
        F_new = F * np.sqrt(D)
        a_new = _ifft(F_new, N)[:n].real
        return project_simplex(a_new, S)

    def gradient_from_weights(F, full_w):
        W = _fft(full_w, N)
        d = _ifft(W * np.conj(F), N)[:n].real
        d[d < 0.0] = 0.0
        return d

    def multiplicative_update(a, d, scale, beta):
        if scale <= 1e-18:
            return a.copy()
        x = d / scale
        x = np.clip(x, 0.0, 100.0)
        factor = 1.0 / (1.0 + float(beta) * x)
        return project_simplex(a * factor, S)

    def overlap_penalty(a0, a1, k=32):
        # Small overlap penalty to diversify restarts.
        a0 = np.asarray(a0, dtype=np.float64).ravel()
        a1 = np.asarray(a1, dtype=np.float64).ravel()
        m = min(k, a0.size, a1.size)
        if m <= 0:
            return 0.0
        return float(np.sum(a0[:m] * a1[:m])) / (float(a0[:m].sum()) * float(a1[:m].sum()) + 1e-300)

    def hybrid_block(a_init, block_seconds, target_score):
        a_local = norm_to_sum(a_init, S)
        best_local = a_local.copy()
        best_score = score_fast(a_local)
        if not np.isfinite(best_score):
            best_score = np.inf

        deadline_local = time.monotonic() + max(1.0, float(block_seconds))
        no_improve = 0
        kc = 0
        qc = 0

        # Cycles of active-set sizes and filter orders.
        k_cycle = [32, 64, 128, 256, 512, 256, 128, 64]
        q_cycle = [1.0, 2.0, 3.0, 4.0]

        while time.monotonic() < deadline_local - 0.25:
            F = _fft(a_local, N)
            c = conv_from_fft(F, N, n)
            a_sum = float(a_local.sum())
            sc = score_from_coeffs(c, a_sum, n)

            if sc < best_score:
                best_score = sc
                best_local = a_local.copy()

            if not np.isfinite(sc):
                break
            if sc <= target_score:
                break

            candidates = []

            # ---------- spectral square shaping ----------
            k = min(n, k_cycle[kc % len(k_cycle)])
            p_top = make_top_p(c[:n], k, 2.0 if (kc % 3) < 2 else 1.0)

            lags, lag_w = spectral_lag_weights_from_p(p_top)
            if lags is not None and lags.size > 0:
                cos_eff = compute_cos_eff(lags, lag_w)
                max_abs = float(np.max(np.abs(cos_eff)))
                if max_abs > 1e-4:
                    if sc > 1.55:
                        base_alpha = 0.45
                    elif sc > 1.5027:
                        base_alpha = 0.28
                    else:
                        base_alpha = 0.11
                    mod = 0.75 + 0.25 * float(kc % 2)
                    alpha1 = min(0.9, base_alpha * mod)
                    alpha2 = min(0.9, 0.55 * alpha1)

                    a2 = apply_spectral_square(F, cos_eff, alpha1)
                    if a2.size == n:
                        candidates.append(a2)
                    if kc % 2 == 0:
                        a3 = apply_spectral_square(F, cos_eff, alpha2)

                            candidates.append(a3)

                if mean_d > 1e-18:

                sd = float(np.std(dz))

                    if kc % 5 == 4:

                    step_value = eta_use * mean_scale
                    a5 = project_simplex(a_local - step_value * (dz / sd), S)
                    if a5.size == n:
                        candidates.append(a5)

            # ---------- evaluate a few candidates ----------
            best_cand = None
            best_cand_score = np.inf

            for idx in range(min(4, len(candidates))):
                a2 = candidates[idx]
                if a2.size != n:
                    continue
                if not np.all(np.isfinite(a2)):
                    continue
                sc2 = score_fast(a2)
                if np.isfinite(sc2) and sc2 < best_cand_score:
                    best_cand_score = sc2
                    best_cand = a2

            if best_cand is not None:
                if best_cand_score < best_score:
                    best_score = best_cand_score
                    best_local = best_cand.copy()
                    a_local = best_cand.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                    accept_tol = 0.006 if best_score < 1.515 else 0.02
                    if no_improve >= 2 and best_cand_score <= best_score + accept_tol:

                            a_local = best_cand.copy()
                    if no_improve >= 6:

                        a_local = project_simplex(
                            best_local + rng.normal(0.0, sigma, n), S

            else:

                if no_improve >= 3:

                    a_local = project_simplex(
                        best_local + rng.normal(0.0, sigma, n), S
                    )
                    no_improve = 0

            kc += 1
            qc += 1

        return best_local, best_score

    # ------------------------------------------------------------------
    # Initialization
    # ------------------------------------------------------------------

    if gaveq is not None:
        try:
            gaveq_arr = np.asarray(gaveq, dtype=np.float64).ravel()
            if gaveq_arr.size > 0:
                g = interpolate_sequence(gaveq_arr, n)
                g = norm_to_sum(g, S)
                inits.append(g)
        except Exception:
            pass

    tt = np.linspace(0.0, 1.0, n, endpoint=False, dtype=np.float64)

    q = (2.0 * tt - 1.0) ** 2

        if np.isfinite(a).all() and a.min() > 0.0:

        a = np.exp(float(beta) * (tt - 0.5) ** 2)

    seen = 0

            if a.size != n:

            s = float(a.sum())
            if s <= 0.01:

            if np.isfinite(sc):

                seen += 1
    scored.sort(key=operator.itemgetter(0))

    if len(scored) == 0:

        global_score = score_fast(global_a)
        final_list = [float(x) for x in global_a]
        return final_list

    global_a = scored[0][1].copy()
    global_score = float(scored[0][0])

            state = a_new.copy()

                    sig = 2.0e-4 * mean_scale
                    state = project_simplex(base_alt + rng.normal(0.0, sig, n), S)
                else:

                        global_a + rng.normal(0.0, 2.0e-4 * mean_scale, n), S

            else:

    # ------------------------------------------------------------------

        if sc_final < global_score - 1e-12:
            global_score = float(sc_final)
            global_a = a_final.copy()
            print("  final polish: %.8f" % global_score, flush=True)

            global_a + rng.normal(0.0, 1.0e-4 * mean_scale, n), S

        "Returning length=%d, internal bound=%.8f after %.3f s"
        % (len(final_list), global_score, time.monotonic() - start_wall),
        flush=True,
    )
    return final_list
