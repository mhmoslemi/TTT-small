def active_set_lp(a, c, n, K, step_size):
    """Solve LP to find optimal perturbation direction."""
    # c = conv(a,a), already computed
    S = a.sum()
    
    # Top K lags by convolution value
    top_k = np.argsort(c)[::-1][:K]
    
    # (a*d)[k] = sum_i a_i * d_{k-i} for valid i
    # For linear convolution of a (length n) with d (length n):
    # (a*d)[k] = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i] * d[k-i]
    
    # Build constraint matrix A_ub for t_max >= (a*d)[k]
    # which is: -(a*d)[k] <= -t_max
    # i.e., sum_i a_i * d_{k-i} - t_max <= 0
    
    # The full constraint matrix has K rows (one per top lag) + 1 (sum constraint is equality)
    # Variables: d_0,...,d_{n-1}, t_max (n+1 total)
    
    # For each top lag k:
    #   sum_{i} a_i * d_{k-i} - t_max <= 0
    #   = sum_j a_{k-j} * d_j - t_max <= 0 (reindexing with j = k-i)
    #   Valid j: max(0, k-n+1) <= j <= min(k, n-1)
    
    A_rows = []
    b_ub = []
    for k in top_k:
        row = np.zeros(n + 1)
        # Coefficients for d_j: a_{k-j} for valid j
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            row[j] = a[k - j]
        row[n] = -1.0  # coefficient for t_max
        A_rows.append(row)
        b_ub.append(0.0)
    
    A_ub = np.array(A_rows)
    
    # Equality constraint: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: d_i >= -a_i / step_size (approximate non-negativity)
    # t_max >= -inf (or some reasonable lower bound)
    bounds = [(-a[i] / step_size, np.inf) for i in range(n)]
    bounds.append((-np.inf, np.inf))
    
    # Solve LP
    res = linprog(c=np.concatenate([np.zeros(n), [1.0]]),
                  A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds,
                  method='highs')
    
    if res.status != 0:
        return None
    
    d = res.x[:n]
    return d

"""
Minimize 2*n*max_conv/sum^2 using active-set LP + spectral shaping + FISTA LSE.

Key innovation: Active-set linear programming to find optimal perturbation
directions, combined with spectral shaping for global moves and FISTA LSE
for local polishing. The LP directly optimizes the first-order change in
the convolution peak, providing more principled steps than pure subgradient
methods.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fft_convolve_cross(a, b):
    """Full cross-convolution of a and b via FFT. len(a)=len(b)=n."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]

def spectral_shaping(a, N, top_k, alpha):
    """One spectral shaping step."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()
    
    m = np.arange(N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a.copy(), c, c_max
    
    weights = top_vals / total
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
    
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new_full, a.sum())
    return a_new, c, c_max

def active_set_lp(a, c, n, K, max_iter=1000):
    """
    Solve LP to find optimal perturbation direction.
    Returns direction d (sum(d)=0, ||d||_inf <= 1) or None.
    """
    top_k = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix: for each top lag k,
    # conv(a,d)[k] - t <= 0
    # conv(a,d)[k] = sum_j a[k-j] * d[j] for valid j
    
    A_rows = []
    for k in top_k:
        row = np.zeros(n + 1)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            row[j] = a[k - j]
        row[n] = -1.0
        A_rows.append(row)
    
    A_ub = np.array(A_rows)
    b_ub = np.zeros(K)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Objective: min t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    # Add tiny regularization on d to avoid degenerate solutions
    c_obj[:n] = 1e-10
    
    # Bounds
    bounds = [(-1.0, 1.0)] * n + [(-np.inf, np.inf)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                     A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None
    
    d = result.x[:n].copy()
    # Ensure sum is exactly 0
    d -= d.mean()
    return d

"""
Minimize 2*n*max_conv/sum^2 using active-set LP + spectral shaping + FISTA LSE.

Key innovation: Active-set linear programming solves for the optimal first-order
perturbation direction by formulating the reduction of top-K convolution peaks
as a linear program. This provides more principled local steps than pure
subgradient methods, combined with spectral shaping for global moves and FISTA
LSE for polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def score_of(a):
    """Compute objective: 2*n*max(conv(a,a))/sum(a)^2."""
    c = fft_convolve_self(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    return 2.0 * len(a) * c.max() / (S * S)


def spectral_shaping(a, N, top_k, alpha):
    """One spectral shaping step. Returns (new_a, c, c_max)."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    m = np.arange(N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a.copy(), c, c_max

    weights = top_vals / total
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new_full, a.sum())
    return a_new, c, c_max


def active_set_lp(a, c, n, K, rng):
    """
    Solve LP to find optimal perturbation direction d (sum=0, bounded).
    Minimizes max of conv(a,d)[k] over top-K lags.
    Returns direction d or None.
    """
    # Choose top K lags (with some randomization for diversity)
    top_2K = np.argsort(c)[::-1][:min(2 * K, 2 * n - 1)]
    if len(top_2K) >= K:
        if rng.random() < 0.3:
            # Random subset of top 2K
            chosen = rng.choice(len(top_2K), size=K, replace=False)
            top_k = top_2K[chosen]
        else:
            top_k = top_2K[:K]
    else:
        top_k = top_2K

    # Build constraint matrix: conv(a,d)[k] - t <= 0
    # conv(a,d)[k] = sum_j a[k-j]*d[j] for valid j
    A_rows = np.zeros((len(top_k), n + 1))
    for row_idx, k in enumerate(top_k):
        j_min = max(0, int(k) - n + 1)
        j_max = min(int(k), n - 1)
        for j in range(j_min, j_max + 1):
            A_rows[row_idx, j] = a[int(k) - j]
        A_rows[row_idx, n] = -1.0

    b_ub = np.zeros(len(top_k))

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    # Objective: min t + small_reg * sum(|d_i|) approximated as min t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    c_obj[:n] = 1e-8  # tiny regularization

    # Bounds on d
    bound = 1.0
    bounds = [(-bound, bound)] * n + [(-np.inf, np.inf)]

    try:
        result = linprog(c_obj, A_ub=A_rows, b_ub=b_ub,
                         A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
    except Exception:
        return None

    if result.status != 0:
        return None

    d = result.x[:n].copy()
    d -= d.mean()  # Ensure sum is exactly 0
    return d


def line_search(a, d, max_evals=8):
    """Backtracking line search for step a + t*d. Returns (best_a, best_score)."""
    # Max feasible step
    neg = d < -1e-15
    if np.any(neg):
        t_max = np.min(a[neg] / (-d[neg]))
    else:
        t_max = 1.0
    t_max = max(t_max, 1e-12)
    t_max = min(t_max, 1.0)  # Cap step size

    best_score = np.inf
    best_a = a.copy()

    for frac in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
        t = t_max * frac
        a_new = a + t * d
        a_new = np.clip(a_new, 0.0, None)
        c_new = fft_convolve_self(a_new)
        S_new = a_new.sum()
        if S_new < 1e-10:
            continue
        sc = 2.0 * len(a_new) * c_new.max() / (S_new * S_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

    return best_a, best_score


def fista_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=500.0):
    """FISTA with log-sum-exp smooth approximation of max(c)."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    S = a.sum()

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.001 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        c_max = c.max()
        shifted = c - c_max
        exp_bc = np.exp(beta * shifted)
        total = exp_bc.sum()
        if total < 1e-300 or not np.isfinite(total):
            break
        w = exp_bc / total

        # Gradient computation via FFT
        fsize = 1 << (2 * n - 1 + n - 2).bit_length()
        W = fft(w, fsize)
        Y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 / (beta * gnorm + 1e-10)
        step = min(step, 0.5 / beta)
        x_new = project_simplex(y - step * grad, S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            Sx = x_new.sum()
            if Sx > 1e-10:
                sc_x = 2.0 * n * c_x.max() / (Sx * Sx)
                if sc_x < best_score:
                    best_score = sc_x
                    best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 3000 == 0 and rng.random() < 0.08:
            y = project_simplex(y + rng.normal(0, 0.004 * S / np.sqrt(n), n), S)
            t_mom = 1.0

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    return inits


def propose_candidate(seed=42, budget_s=2300, **kwargs):
    """
    Search for non-negative sequence minimizing evaluate_sequence.
    Combines active-set LP (principled local steps), spectral shaping
    (global moves), and FISTA LSE (polishing) with adaptive switching.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + budget_s - 45.0  # 45s buffer for cleanup

    global_best = np.inf
    global_a = None

    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()  # 16384

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    # --- Phase 1: Diverse exploration with spectral shaping ---
    print("=== Phase 1: Exploration ===")
    t_p1 = time.monotonic()
    p1_budget = min(400.0, (deadline - t_p1) * 0.25)

    inits = make_initializations(n, rng)
    n_inits = len(inits)
    per_init = p1_budget / max(n_inits, 1)
    print("  %d initializations, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 120:
            break
        a = safe_normalize(arr, n)

        # Spectral shaping burst
        t0 = time.monotonic()
        n_ss = min(5000, int(per_init * 0.7 / 0.0025))
        for _ in range(n_ss):
            if time.monotonic() - t0 > per_init * 0.7:
                break
            a, c_ss, _ = spectral_shaping(a, N, top_k=12, alpha=0.25)

        a = safe_normalize(a, n)
        sc = score_of(a)
        if sc < global_best:
            global_best = sc
            global_a = a.copy()
            print("  [%s] %.6f" % (name, global_best))

        # Quick FISTA on best from SS
        rem = per_init - (time.monotonic() - t0)
        if rem > 5 and global_a is not None and name != "given":
            a_f, _ = fista_lse(a, n, S, min(rem, 60.0), rng, beta_init=10.0, beta_max=100.0)
            a_f = safe_normalize(a_f, n)
            sc_f = score_of(a_f)
            if sc_f < global_best:
                global_best = sc_f
                global_a = a_f.copy()
                print("  [%s+FISTA] %.6f" % (name, global_best))

    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = 2.0

    print("Phase 1 done. Best: %.6f" % global_best)

    # --- Phase 2: Active-set LP + Spectral shaping ---
    print("=== Phase 2: Active-set LP optimization ===")
    t_p2 = time.monotonic()
    p2_deadline = min(deadline - 300.0, t_p2 + (deadline - t_p2) * 0.55)

    a_curr = global_a.copy()
    stall = 0
    lp_total = 0
    ss_total = 0
    last_print = time.monotonic()

    while time.monotonic() < p2_deadline:
        # --- Spectral shaping block ---
        ss_start = time.monotonic()
        n_ss_steps = 300
        alpha_ss = 0.15 + 0.1 * min(stall / 4.0, 1.0)  # More aggressive when stalled
        for _ in range(n_ss_steps):
            if time.monotonic() - ss_start > 4.0:
                break
            a_curr, c_tmp, _ = spectral_shaping(a_curr, N, top_k=15, alpha=alpha_ss)
        a_curr = safe_normalize(a_curr, n)
        ss_total += 1

        sc = score_of(a_curr)
        if sc < global_best:
            global_best = sc
            global_a = a_curr.copy()
            stall = 0
            if time.monotonic() - last_print > 30:
                print("  [SS] %.6f (t=%.0fs)" % (global_best, time.monotonic() - t_p2))
                last_print = time.monotonic()
        else:
            stall += 1

        # --- Active-set LP block ---
        if time.monotonic() < p2_deadline - 15:
            c_curr = fft_convolve_self(a_curr)
            K = 20 + 5 * min(lp_total // 200, 10)  # Adaptive K
            K = min(K, 60)

            n_lps_this_block = 4
            for _ in range(n_lps_this_block):
                if time.monotonic() > p2_deadline - 10:
                    break

                # Re-compute c if needed
                if lp_total % 4 == 0:
                    c_curr = fft_convolve_self(a_curr)

                d = active_set_lp(a_curr, c_curr, n, K, rng)
                lp_total += 1

                if d is None:
                    continue

                # Check if the LP found a useful direction
                t_opt = d @ c_curr[:n]  # Rough check (not exact)
                # Better: compute actual conv(a,d) at top lags
                # For now, just do the line search

                a_new, sc_new = line_search(a_curr, d, max_evals=6)
                a_new = safe_normalize(a_new, n)
                sc_new = score_of(a_new)

                if sc_new < global_best - 1e-9:
                    global_best = sc_new
                    global_a = a_new.copy()
                    a_curr = a_new.copy()
                    stall = 0
                    if time.monotonic() - last_print > 15:
                        print("  [LP #%d] %.6f (t=%.0fs)" % (lp_total, global_best,
                                                             time.monotonic() - t_p2))
                        last_print = time.monotonic()
                else:
                    stall += 1

        # --- Restart if stalled ---
        if stall >= 8:
            # Larger perturbation
            noise_scale = 0.005 + 0.003 * min(stall / 8.0, 3.0)
            a_curr = project_simplex(a_curr + rng.normal(0, noise_scale * S / np.sqrt(n), n), S)
            stall = 0
            print("  [Restart] at %.6f, noise=%.4f" % (global_best, noise_scale))

    print("Phase 2 done. Best: %.6f (LPs=%d, SS blocks=%d)" % (global_best, lp_total, ss_total))

    # --- Phase 3: FISTA polishing ---
    print("=== Phase 3: FISTA Polish ===")
    t_p3 = time.monotonic()

    if global_a is not None:
        for beta_pair in [(5.0, 50.0), (20.0, 100.0), (50.0, 200.0), (100.0, 400.0), (200.0, 800.0)]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            seg_time = min(rem * 0.3, 150.0)
            a_f, _ = fista_lse(global_a, n, S, seg_time, rng,
                               beta_init=beta_pair[0], beta_max=beta_pair[1])
            a_f = safe_normalize(a_f, n)
            sc_f = score_of(a_f)
            if sc_f < global_best:
                global_best = sc_f
                global_a = a_f.copy()
                print("  [FISTA b=%d-%d] %.6f" % (beta_pair[0], beta_pair[1], global_best))

    # --- Phase 4: Final LP polish ---
    print("=== Phase 4: Final LP polish ===")
    rem = deadline - time.monotonic()
    if rem > 20 and global_a is not None:
        a_polish = global_a.copy()
        for i in range(30):
            if time.monotonic() > deadline - 10:
                break
            c_p = fft_convolve_self(a_polish)
            d = active_set_lp(a_polish, c_p, n, K=15, rng=rng)
            if d is None:
                # Try smaller K
                d = active_set_lp(a_polish, c_p, n, K=8, rng=rng)
            if d is None:
                continue
            a_new, sc_new = line_search(a_polish, d, max_evals=5)
            a_new = safe_normalize(a_new, n)
            sc_new = score_of(a_new)
            if sc_new < global_best - 1e-10:
                global_best = sc_new
                global_a = a_new.copy()
                a_polish = a_new.copy()

    # --- Final verification ---
    print("=== Final ===")
    final_a = np.clip(global_a, 0.0, 1000.0)
    final_a = safe_normalize(final_a, n)

    final_list = [float(x) for x in final_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Internal: %.6f | Verified: %.6f | Length: %d" % (global_best, ver, len(final_list)))
    else:
        print("Verification FAILED, using fallback")
        final_a = np.ones(256, dtype=np.float64) * (S / 256)
        final_list = [float(x) for x in final_a]
        ver = evaluate_sequence(final_list)
        print("Fallback score: %.6f" % ver)

    return final_list
