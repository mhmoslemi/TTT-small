import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def grad_toppk_weighted(a, c, top_k, target):
    """
    Weighted top-K subgradient.
    For each of the top peaks, weight by how much it exceeds the target.
    """
    n = len(a)
    c_sorted = np.sort(c)[::-1]
    top_indices = np.argsort(c)[::-1][:top_k]
    
    # Weights: how much each peak exceeds the target
    excess = np.maximum(c[top_indices] - target, 0)
    total_excess = excess.sum()
    if total_excess < 1e-30:
        # If no peak exceeds target, use uniform weights on top peaks
        excess = np.ones(top_k)
        total_excess = top_k
    
    weights = excess / total_excess
    
    # Build the weight array in full size
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    
    # Gradient: grad[m] = 2 * sum_k w[k] * a[k-m] (cross-correlation)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def project_simplex(v, S):

"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 using adaptive top-K subgradient 
optimization with backtracking and multi-resolution strategy.

Key improvements:
1. Start from parent solution (strong warm start)
2. Adaptive K with excess-based weighting for peak targeting
3. Backtracking line search for robust step selection
4. Continuous parameter adaptation (no hard phase boundaries)
5. Periodic restarts with diminishing perturbations
6. Final fine-polish with very large effective K

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.array([], dtype=np.float64)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient(a, weights):
    """
    Compute gradient of sum_k weights[k]*c[k] w.r.t. a.
    grad[m] = 2 * sum_k weights[k] * a[k-m] (cross-correlation).
    """
    n = len(a)
    fsize = 1 << (len(weights) + n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def get_topk_weights(c, K):
    """Compute weights for top-K peaks using excess-over-mean weighting."""
    n_total = len(c)
    mean_c = c.sum() / n_total
    K = min(K, n_total)
    top_indices = np.argsort(c)[::-1][:K]
    excess = np.maximum(c[top_indices] - mean_c, 0.0)
    total = excess.sum()
    if total < 1e-30:
        excess = np.ones(K)
        total = K
    weights = excess / total
    w_arr = np.zeros(n_total)
    for i in range(K):
        w_arr[int(top_indices[i])] = weights[i]
    return w_arr


def get_softmax_weights(c, beta):
    """Compute softmax weights over all convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
        return w
    return exp_bc / total


def obj_from_conv(n, c, S):
    """Compute objective from convolution and known sum."""
    if S < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (S * S)

            given = np.clip(given, 0.0, 1000.0)
            if given.sum() > 0.01:
                n = len(given)
                S = np.sqrt(2.0 * float(n))
                parent = given * (S / given.sum())
    except Exception:
        pass
    
    if parent is not None:
        c_p = fft_convolve_self(parent)
        global_best = obj_from_conv(n, c_p, S)
        global_a = parent.copy()
        print("Parent loaded: n=%d, obj=%.6f" % (n, global_best))
    else:
        # Fallback initialization
        tt = np.linspace(0, 1, n, endpoint=False)
        a_init = np.exp(2.0 * (1.0 - tt))
        global_a = a_init * (S / a_init.sum())
        c = fft_convolve_self(global_a)
        global_best = obj_from_conv(n, c, S)
        print("Fallback init: n=%d, obj=%.6f" % (n, global_best))
    
    # --- Phase 1: Main adaptive optimization from best start ---
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print("Phase 1: Adaptive optimization (%.0f sec)" % remaining)
        t0 = time.monotonic()
        a_opt, obj_opt = run_adaptive_opt(global_a, n, S, remaining, rng)
        a_opt = a_opt * (S / a_opt.sum())
        c = fft_convolve_self(a_opt)
        obj = obj_from_conv(n, c, S)
        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
            print("  Phase 1 result: %.6f (improvement: %.6f)" % (global_best, obj_opt - global_best))
        elapsed = time.monotonic() - t0
        print("  Phase 1 done in %.1f sec, best=%.6f" % (elapsed, global_best))
    
    # --- Phase 2: Try diverse initializations ---
    remaining = deadline - time.monotonic()
    if remaining > 90:
        print("Phase 2: Diversity exploration (%.0f sec)" % remaining)
        inits = diversity_init(n, rng, S)
        # Sort by initial objective (quick screen)
        scored_inits = []
        for arr in inits:
            c = fft_convolve_self(arr)
            obj = obj_from_conv(n, c, S)
            scored_inits.append((obj, arr))
        scored_inits.sort(key=lambda x: x[0])
        
        # Take top 4 diverse inits
        per_init = remaining / 5.0
        for idx, (obj_init, arr) in enumerate(scored_inits[:4]):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            print("  Init %d: initial obj=%.4f, optimizing..." % (idx, obj_init))
            a_t, obj_t = run_adaptive_opt(arr, n, S, min(per_init, rem - 10), rng)
            a_t = a_t * (S / a_t.sum())
            c_t = fft_convolve_self(a_t)
            obj_t = obj_from_conv(n, c_t, S)
            if obj_t < global_best:
                global_best = obj_t
                global_a = a_t.copy()
                print("  ** Improvement! %.6f" % global_best)
            else:
                print("  No improvement (best so far=%.4f)" % global_best)
    
    # --- Phase 3: Fine polish ---
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("Phase 3: Fine polish (%.0f sec)" % remaining)
        t0 = time.monotonic()
        a_pol, obj_pol = run_fine_polish(global_a, n, S, remaining, rng)
        a_pol = a_pol * (S / a_pol.sum())
        c_pol = fft_convolve_self(a_pol)
        obj_pol = obj_from_conv(n, c_pol, S)
        if obj_pol < global_best:
            global_best = obj_pol
            global_a = a_pol.copy()
            print("  ** Polish improvement! %.6f" % global_best)
        else:
            print("  Polish: no improvement, %.6f" % global_best)
        print("  Phase 3 done in %.1f sec" % (time.monotonic() - t0))
    
    # --- Final verification ---
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("FINAL: len=%d, internal=%.6f, verified=%s" %
          (len(final_list), global_best,
           "%.6f" % ver if np.isfinite(ver) else "inf"))
    
    if not np.isfinite(ver):
        print("evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback verified: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
    
    return final_list
