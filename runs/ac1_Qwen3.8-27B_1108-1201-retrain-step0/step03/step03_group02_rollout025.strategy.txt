      def tournament_selection(pop, tournament_size=3):
          # Select tournament_size candidates and choose the best (lowest objective)
          candidates = random.sample(pop, tournament_size)
          best_candidate = None
          best_score = np.inf
          for cand in candidates:
              s = evaluate_sequence(cand)  # cand is a list
              if s < best_score:
                  best_score = s

      def crossover(a, b, crossover_rate, n):

          if crossover_rate > 0:
              idx = np.random.randint(0, n)
              return np.concatenate([a[:idx], b[idx:]])
          else:
              # if we don't want to crossover
              return a
