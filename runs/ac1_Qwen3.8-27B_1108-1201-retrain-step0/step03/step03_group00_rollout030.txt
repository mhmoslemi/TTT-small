"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: LP-based best response iteration combined with FISTA polishing.
The LP finds the optimal direction to move in at each step,
exploiting the convex structure of the problem.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return float(2 * n * c.max() / (S ** 2))


def solve_best_response_lp(f, M, solver='SCS'):
    """
    Solve the LP: max sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    Returns the optimal g (not normalized).
    """
    n = len(f)
    g = cp.Variable(n)
    
    # Build the constraint matrix implicitly using CVXPY expressions
    # (f*g)(k) = sum_j f(j) * g(k-j) for valid j
    # We need to express this efficiently
    
    # For efficiency, use the fact that (f*g)(k) can be computed as a matrix-vector product
    # A_{k,j} = f(k-j) if 0 <= k-j < n, else 0
    
    # Build A as a sparse matrix
    import scipy.sparse as sp
    
    # A is (2n-1) x n
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            # A[k, j] = f(k - j)
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [A @ g <= M, g >= 0]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if g.value is not None:
            return np.maximum(g.value, 0)
    except Exception as e:
        print("LP solve failed: %s" % str(e))
    
    return None


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_rate=0.004, beta_max=800.0):
    """FISTA with LSE smoothing."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)

        if it - last_improve_it > 2000:
            step *= 0.5
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve_it = it

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 4000 == 0:
            decay = np.exp(-it / 20000.0)
            noise_scale = 0.003 * S / np.sqrt(n) * decay
            if noise_scale > 1e-6:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

import numpy as np

import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_rate=0.003, beta_max=1000.0):
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)

        if it - last_improve_it > 3000:
            step *= 0.5
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve_it = it

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 5000 == 0:

            if noise_scale > 1e-6:

                t = 1.0

    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best

    One LP-based best response iteration.

"""
Minimize 2*n*max(f*f)/sum(f)^2 over non-negative sequences.
Strategy: Hybrid FISTA + LP best-response iterations with reduced constraints,
combined with multiplicative repulsion updates and multi-scale exploration.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def lse_value(c, beta):
    """Value of LSE approximation."""
    cm = c.max()
    return cm + np.log(np.exp(beta * (c - cm)).sum()) / beta


def run_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_rate=0.002, beta_max=800.0,
              use_repulsion=False, repulsion_freq=500, repulsion_lr=0.05):

    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve_it = it

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)

            step *= 0.5
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve_it = it
            # Reset beta to escape
            it = max(0, int(np.log(max(beta_init, 1) / beta_init) / beta_rate) if beta_rate > 0 else it // 2)

        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic repulsion update
        if use_repulsion and it % repulsion_freq == 0:
            c_rep = fft_convolve_self(y)
            cm_rep = c_rep.max()
            w_rep = np.exp(30.0 * (c_rep - cm_rep))
            tot_rep = w_rep.sum()
            if tot_rep > 0:
                w_rep /= tot_rep
                fsize_r = 1 << (len(w_rep) + n - 1).bit_length()
                W_r = fft(w_rep, fsize_r)
                A_r = fft(y, fsize_r)
                rep = np.real(ifft(W_r * np.conj(A_r)))[:n]
                y = y * np.exp(-repulsion_lr * rep / max(y.max(), 1e-10))
                y = np.maximum(y, 0)

    if c_final.max() < best:

        best_a = a.copy()

def solve_lp_best_response(f, n, S, top_k=150, solver='SCS', time_limit=10.0):
    """
    Solve the LP: max sum(g) s.t. (f*g)(k) <= M for top-k lags, g >= 0.
    Returns normalized g or None if failed.
    """
    c = fft_convolve_self(f)
    M = c.max()

    # Find top-k most binding lags
    idx = np.argsort(c)[::-1][:top_k]

    # Build reduced constraint matrix (top_k x n)
    rows = np.zeros((top_k, n))
    for i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        js = np.arange(lo, hi + 1)
        rows[i, js] = f[k - js]

    # Solve LP
    g_var = cp.Variable(n)
    constraints = [rows @ g_var <= M, g_var >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    start = time.monotonic()
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None

    if time.monotonic() - start > time_limit * 3:
        print("  LP took too long, skipping")
        return None

    if g_var.value is not None:
        g = np.maximum(g_var.value, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            return None
        g = g * (S / g_sum)
        return g

    return None


def lp_iteration(f, n, S, top_k=150, solver='SCS'):
    """
    One full LP-based iteration: solve LP, find best step size via bisection.
    Returns updated f.
    """
    c_old = fft_convolve_self(f)
    M_old = c_old.max()

    g = solve_lp_best_response(f, n, S, top_k=top_k, solver=solver)
    if g is None:
        return f.copy(), M_old

    # Find best step size t in [0, 1] via bisection
    def eval_step(t):
        f_test = (1 - t) * f + t * g
        f_test = np.maximum(f_test, 0)
        f_test = project_simplex(f_test, S)
        c_test = fft_convolve_self(f_test)
        return c_test.max()

    # Try t=1 first
    val1 = eval_step(1.0)
    if val1 < M_old * 1.001:
        f_new = (1.0) * f + 1.0 * g
        f_new = np.maximum(f_new, 0)
        f_new = project_simplex(f_new, S)
        return f_new, val1

    # Bisection
    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_val = M_old
    for _ in range(8):
        mid = (lo + hi) / 2
        val_mid = eval_step(mid)
        if val_mid < best_val:
            best_val = val_mid
            best_t = mid
            lo = mid
        else:
            hi = mid

    if best_t > 0.01:
        f_new = (1 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0)
        f_new = project_simplex(f_new, S)
        return f_new, best_val

    return f.copy(), M_old


def multiplicative_repulsion(f, n, S, beta=50.0, lr=0.1):
    """Fast multiplicative repulsion update."""
    c = fft_convolve_self(f)
    cm = c.max()
    w = np.exp(beta * (c - cm))
    tot = w.sum()
    if tot < 1e-300:
        return f.copy()
    w /= tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(f, fsize)
    rep = np.real(ifft(W * np.conj(A)))[:n]
    # Multiplicative update
    f_new = f * np.exp(-lr * rep / max(f.max(), 1e-10))
    f_new = np.maximum(f_new, 1e-15)
    f_new = project_simplex(f_new, S)
    return f_new

def interpolate_sequence(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)

        a1, s1 = run_fista(given, n, S, usable_time * 0.15, rng,
                           beta_init=20.0, beta_rate=0.002, beta_max=400.0,
                           use_repulsion=True, repulsion_freq=300, repulsion_lr=0.05)
        a1 = normalize_sum(a1, n)
        c1 = fft_convolve_self(a1)
        if c1.max() < global_best:
            global_best = c1.max()
            global_a = a1.copy()
            print("  Phase 1 best: %.6f" % global_best)

    # Phase 2: Hybrid FISTA + LP on the best so far
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("=== Phase 2: Hybrid FISTA+LP (remaining %.0fs) ===" % remaining)
            a2, s2, nlp = run_hybrid_optimization(global_a, n, S, remaining * 0.5, rng,
                                                   use_lp=True, lp_every=200)
            a2 = normalize_sum(a2, n)
            c2 = fft_convolve_self(a2)
            if c2.max() < global_best:
                global_best = c2.max()
                global_a = a2.copy()
                print("  Phase 2 best: %.6f (LP calls: %d)" % (global_best, nlp))
            else:
                print("  Phase 2: no improvement (best=%.6f, phase2=%.6f, LP calls: %d)" %
                      (global_best, c2.max(), nlp))

    # Phase 3: Try a different n (larger)
    remaining = deadline - time.monotonic()
    if remaining > 120:
        alt_n = n * 2 if n <= 8192 else n // 2
        if 2048 <= alt_n <= 32768:
            S_alt = np.sqrt(2.0 * alt_n)
            print("=== Phase 3: Trying n=%d (remaining %.0fs) ===" % (alt_n, remaining))
            if global_a is not None:
                alt_init = interpolate_sequence(global_a, alt_n)
                alt_init = normalize_sum(alt_init, alt_n)
            else:
                alt_init = np.ones(alt_n) * 0.01

            a3, s3 = run_fista(alt_init, alt_n, S_alt, remaining * 0.5, rng,
                               beta_init=15.0, beta_rate=0.002, beta_max=500.0,
                               use_repulsion=True, repulsion_freq=400, repulsion_lr=0.04)
            a3 = normalize_sum(a3, alt_n)
            c3 = fft_convolve_self(a3)
            obj3 = 2.0 * alt_n * c3.max() / (a3.sum() ** 2)
            if obj3 < global_best:
                global_best = obj3
                global_a = a3.copy()
                print("  Phase 3 (n=%d) best: %.6f" % (alt_n, global_best))
            else:
                print("  Phase 3 (n=%d): %.6f (no improvement vs %.6f)" % (alt_n, obj3, global_best))

    # Phase 4: Final intensive polishing
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_p = len(global_a)
            S_p = np.sqrt(2.0 * n_p)
            print("=== Phase 4: Final polish (remaining %.0fs, n=%d) ===" % (remaining, n_p))

            # High-beta FISTA
            a4, s4 = run_fista(global_a, n_p, S_p, remaining * 0.5, rng,
                               beta_init=100.0, beta_rate=0.001, beta_max=2000.0,
                               use_repulsion=False)
            a4 = normalize_sum(a4, n_p)
            c4 = fft_convolve_self(a4)
            obj4 = 2.0 * n_p * c4.max() / (a4.sum() ** 2)
            if obj4 < global_best:
                global_best = obj4
                global_a = a4.copy()
                print("  Phase 4a best: %.6f" % global_best)

            # LP polish
            remaining2 = deadline - time.monotonic()
            if remaining2 > 15:
                a5, s5, nlp5 = run_hybrid_optimization(global_a, n_p, S_p, remaining2 * 0.7, rng,
                                                       use_lp=True, lp_every=100)
                a5 = normalize_sum(a5, n_p)
                c5 = fft_convolve_self(a5)
                obj5 = 2.0 * n_p * c5.max() / (a5.sum() ** 2)
                if obj5 < global_best:
                    global_best = obj5
                    global_a = a5.copy()
                    print("  Phase 4b best: %.6f (LP calls: %d)" % (global_best, nlp5))

    # Phase 5: Try diverse initializations if time remains
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_a is not None:
        n_d = len(global_a)
        S_d = np.sqrt(2.0 * n_d)
        print("=== Phase 5: Diverse restarts (remaining %.0fs) ===" % remaining)
        inits = generate_initializations(n_d, rng)
        per_init = (remaining * 0.4) / len(inits)
        for i, init in enumerate(inits[:4]):
            if time.monotonic() > deadline - 30:
                break
            if per_init < 5:
                break
            ai, si = run_fista(init, n_d, S_d, per_init, rng,
                              beta_init=10.0, beta_rate=0.003, beta_max=600.0)
            ai = normalize_sum(ai, n_d)
            ci = fft_convolve_self(ai)
            obji = 2.0 * n_d * ci.max() / (ai.sum() ** 2)
            if obji < global_best:
                global_best = obji
                global_a = ai.copy()
                print("  Diverse init %d: %.6f" % (i, global_best))

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Final verification: %.6f" % ver if np.isfinite(ver) else "Final verification: inf")
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
    else:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    print("Returning sequence of length %d, best internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
