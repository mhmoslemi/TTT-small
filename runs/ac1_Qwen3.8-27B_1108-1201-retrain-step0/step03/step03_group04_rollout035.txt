Key improvements over previous approach:
1. Fourier-parameterized initialization for diverse starting points
2. Adaptive temperature with geometric cooling + reheating
3. Peak-lag direct optimization (target specific lags)
4. Symmetry-aware search (exploit a[i] = a[n-1-i] structure)
5. Multi-start with diverse analytical initializations
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_objective(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def softmax_weights(c, beta):
    """Normalized softmax weights."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Normalized top-k weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    weights = np.zeros(len(c))
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights[top_indices] = top_vals / total
    return weights

import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return np.asarray(v, dtype=np.float64)
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)

    A = fft(a, fsize)
    c = np.real(ifft(A * A))

            else:
                # Random micro-perturbation
                if rng.random() < 0.5:
                    # Perturb only elements that affect the peak
                    peak_idx = np.argmax(c)
                    # Elements that contribute to b[peak_idx] are a[0..min(n-1,peak_idx)] 
                    # and a[max(0,peak_idx-n+1)..n-1]
                    lo = max(0, peak_idx - n + 1)
                    hi = min(n - 1, peak_idx)
                    mask = np.zeros(n, dtype=bool)
                    mask[lo:hi + 1] = True
                    noise = np.zeros(n)
                    noise[mask] = rng.normal(0, 0.001 * S / np.sqrt(n), mask.sum())
                    a = project_simplex(a + noise, S)
                step *= 0.9

    return best_local, best_score


def make_symmetric(a, n):
    """Make a symmetric: new_a[i] = (a[i] + a[n-1-i]) / 2."""
    return (a + a[::-1]) / 2.0


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2385, **kwargs):
    """
    Minimize 2*n*max_conv/sum^2 over non-negative sequences.
    Multi-phase with Fourier init, top-k equal-weight FISTA,
    backtracking line search, and micro-polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0

    # Get given sequence
    given_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given_a = np.clip(given, 0, 1000)
    except Exception:
        pass

    if given_a is not None:
        n = len(given_a)
    else:
        n = 8192
        given_a = np.ones(n) * 0.5

    S = np.sqrt(2.0 * float(n))
    a_start = safe_normalize(given_a, n)
    
    c_start = fft_convolve_self(a_start)
    initial_max = c_start.max()
    initial_obj = 2.0 * n * initial_max / (S * S)
    global_best = initial_max
    global_a = a_start.copy()
    print("Initial: n=%d, max_conv=%.8f, obj=%.6f" % (n, initial_max, initial_obj))

    # === Phase 0: Try Fourier initialization (30s) ===
    if deadline - time.monotonic() > 60:
        print("Phase 0: Fourier initialization...")
        t0 = time.monotonic()
        a_fourier, obj_fourier = optimize_fourier_init(n, max_time=30.0, seed=seed + 1)
        a_fourier = safe_normalize(a_fourier, n)
        c_f = fft_convolve_self(a_fourier)
        max_f = c_f.max()
        print("  Fourier: max_conv=%.8f, obj=%.6f [%.0fs]" % (max_f, 2*n*max_f/S**2, time.monotonic()-t0))
        if max_f < global_best:
            global_best = max_f
            global_a = a_fourier.copy()
            print("  -> New best from Fourier!")

    # === Phase 1: FISTA top-k equal weights (35% time) ===
    remaining = deadline - time.monotonic()
    phase1_time = 0.35 * remaining
    if phase1_time > 20:
        print("Phase 1: FISTA top-k equal (%.0f sec)" % phase1_time)
        t0 = time.monotonic()
        a_p1, _ = run_fista_toppk(global_a, n, S, phase1_time, rng,
                                   k_start=2, k_end=60, k_rate=300,
                                   use_equal=True)
        a_p1 = safe_normalize(a_p1, n)
        c_p1 = fft_convolve_self(a_p1)
        m_p1 = c_p1.max()
        print("  TopK-equal: max_conv=%.8f, obj=%.6f [%.0fs]" % (m_p1, 2*n*m_p1/S**2, time.monotonic()-t0))
        if m_p1 < global_best:
            global_best = m_p1
            global_a = a_p1.copy()

    # === Phase 2: FISTA softmax (25% time) ===
    remaining = deadline - time.monotonic()
    phase2_time = 0.25 * remaining
    if phase2_time > 15:
        print("Phase 2: FISTA softmax (%.0f sec)" % phase2_time)
        t0 = time.monotonic()
        a_p2, _ = run_fista_softmax(global_a, n, S, phase2_time, rng,
                                     beta_start=50.0, beta_end=5000.0, beta_growth=1.003)
        a_p2 = safe_normalize(a_p2, n)
        c_p2 = fft_convolve_self(a_p2)
        m_p2 = c_p2.max()
        print("  Softmax: max_conv=%.8f, obj=%.6f [%.0fs]" % (m_p2, 2*n*m_p2/S**2, time.monotonic()-t0))
        if m_p2 < global_best:
            global_best = m_p2
            global_a = a_p2.copy()

    # === Phase 3: Try symmetric variant (10% time) ===
    remaining = deadline - time.monotonic()
    phase3_time = 0.10 * remaining
    if phase3_time > 10:
        print("Phase 3: Symmetric variant (%.0f sec)" % phase3_time)
        t0 = time.monotonic()
        a_sym = safe_normalize(make_symmetric(global_a, n), n)
        a_s, _ = run_fista_toppk(a_sym, n, S, phase3_time, rng,
                                  k_start=2, k_end=40, k_rate=200,
                                  use_equal=True)
        a_s = safe_normalize(a_s, n)
        c_s = fft_convolve_self(a_s)
        m_s = c_s.max()
        print("  Symmetric: max_conv=%.8f, obj=%.6f [%.0fs]" % (m_s, 2*n*m_s/S**2, time.monotonic()-t0))
        if m_s < global_best:
            global_best = m_s
            global_a = a_s.copy()
            print("  -> New best from symmetric!")

    # === Phase 4: Perturbed exploration (15% time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("Phase 4: Perturbed exploration (%.0f sec)" % remaining)
        n_variants = 3
        per_variant = remaining * 0.8 / n_variants
        for v in range(n_variants):
            if time.monotonic() > deadline - 20:
                break
            if v == 0:
                pert = global_a + rng.normal(0, 0.004 * S / np.sqrt(n), n)
            elif v == 1:
                # Smooth perturbation
                raw = rng.normal(0, 1, n)
                smooth = np.convolve(raw, np.ones(128) / 128, mode='same')
                pert = global_a + 0.003 * S / np.sqrt(n) * smooth
            else:
                # Anti-symmetric perturbation (breaks symmetry)
                raw = rng.normal(0, 1, n)
                pert_asym = raw - raw[::-1]
                pert = global_a + 0.002 * S / np.sqrt(n) * pert_asym / max(np.abs(pert_asym).max(), 1e-10)

            pert = safe_normalize(pert, n)
            a_v, _ = run_fista_toppk(pert, n, S, per_variant, rng,
                                      k_start=2, k_end=50, k_rate=300,
                                      use_equal=True)
            a_v = safe_normalize(a_v, n)
            c_v = fft_convolve_self(a_v)
            m_v = c_v.max()
            print("  Variant %d: max_conv=%.8f, obj=%.6f" % (v, m_v, 2*n*m_v/S**2))
            if m_v < global_best:
                global_best = m_v
                global_a = a_v.copy()
                print("  -> New best from variant %d!" % v)

    # === Phase 5: Micro-polish (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("Phase 5: Micro-polish (%.0f sec)" % remaining)
        t0 = time.monotonic()
        a_mp, _ = run_micro_polish(global_a, n, S, remaining, rng)
        a_mp = safe_normalize(a_mp, n)
        c_mp = fft_convolve_self(a_mp)
        m_mp = c_mp.max()
        print("  Micro: max_conv=%.8f, obj=%.6f [%.0fs]" % (m_mp, 2*n*m_mp/S**2, time.monotonic()-t0))
        if m_mp < global_best:
            global_best = m_mp
            global_a = a_mp.copy()

    # === Final: one more softmax polish if time remains ===
    remaining = deadline - time.monotonic()
    if remaining > 8:
        t0 = time.monotonic()
        a_fp, _ = run_fista_softmax(global_a, n, S, remaining, rng,
                                     beta_start=200.0, beta_end=10000.0, beta_growth=1.005)
        a_fp = safe_normalize(a_fp, n)
        c_fp = fft_convolve_self(a_fp)
        m_fp = c_fp.max()
        print("Final softmax: max_conv=%.8f, obj=%.6f [%.0fs]" % (m_fp, 2*n*m_fp/S**2, time.monotonic()-t0))
        if m_fp < global_best:
            global_best = m_fp
            global_a = a_fp.copy()

    # === Output ===
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    final_obj = 2.0 * n * global_best / (S * S)
    print("\nFINAL: n=%d, max_conv=%.8f, objective=%.6f" % (len(final_list), global_best, final_obj))

    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: %.6f" % ver)
    else:
        print("Verification failed! Falling back.")
        final_list = [0.5] * 256
        print("Fallback verified: %.6f" % evaluate_sequence(final_list))

    return final_list
