"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Uses Adam optimizer with LSE-smoothed max objective, multi-scale warm starts,
and coordinate-wise polishing for final improvement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    if s == S:
        return v
    # Simple proportional scaling if no clipping needed
    if (v * (S / s)).max() <= 1000.0:
        return v * (S / s)
    # Otherwise use proper simplex projection
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clamp to [0, 1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_lse_grad(a, beta):
    """Compute LSE value and gradient."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    lse_val = cm + np.log(tot) / beta
    w = e / tot
    # Gradient
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c


def run_adam_lse(a_init, n, S, max_time, rng, 
                 beta_start=10.0, beta_end=500.0, lr=0.01,
                 beta1=0.9, beta2=0.999, epsilon=1e-8,
                 restart_interval=2000, noise_scale=0.005):
    """Adam optimizer with LSE-smoothed max objective."""
    a = normalize_to_sum(a_init, S)
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    start = time.monotonic()
    best_val = np.inf
    best_a = a.copy()
    last_improve = 0
    
    total_iters = max_time / 0.01  # rough estimate
    
    while time.monotonic() - start < max_time:
        t += 1
        # Beta schedule: linear interpolation from beta_start to beta_end
        progress = min(1.0, t / (total_iters * 0.8))
        beta = beta_start + (beta_end - beta_start) * progress
        
        val, grad, c = compute_lse_grad(a, beta)
        
        # Track best actual max
        actual_max = c.max()
        if actual_max < best_val:
            best_val = actual_max
            best_a = a.copy()
            last_improve = t
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Learning rate with decay
        lr_t = lr / (1.0 + 0.001 * t)
        
        a_old = a.copy()
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + epsilon)
        a = project_nonneg_sum(a, S)
        
        # Periodic restart to escape local minima
        if t % restart_interval == 0 and (t - last_improve) > restart_interval // 2:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            a = project_nonneg_sum(a + noise, S)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            last_improve = 0
    
    # Final: return best
    return best_a, best_val


def run_subgradient_polish(a_init, n, S, max_time, rng, 
                           step_init=0.01, step_decay=0.999):
    """Polish using exact subgradient (one-hot at argmax) with FISTA acceleration."""
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_fista = 1.0
    step = step_init
    start = time.monotonic()
    best_val = np.inf
    best_a = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_val:
            best_val = cur
            best_a = y.copy()
        
        # Subgradient: one-hot at argmax
        k = int(np.argmax(c))
        grad = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        # Normalize gradient and use fixed step
        grad = grad / grad_norm
        x = project_nonneg_sum(y - step, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2))
        y = x + ((t_fista - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        a = x
        t_fista = t_new
        step *= step_decay
        
        # Occasional noise injection
        if rng.random() < 0.01:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_nonneg_sum(y + noise, S)
            t_fista = 1.0
    
    return best_a, best_val


def coordinate_polish(a, n, S, max_time, rng):
    """Coordinate-wise optimization: try small shifts of individual elements."""
    a = normalize_to_sum(a, S).copy()
    start = time.monotonic()
    best_val = fft_convolve_self(a).max()
    
    # Try coordinate descent with small perturbations
    n_try = max(1000, n * 10)  # number of coordinates to try
    coordinates = rng.permutation(n)
    
    for i in range(n_try):
        if time.monotonic() - start > max_time:
            break
        j = coordinates[i % n]
        
        # Try different values for a[j]
        current = a[j]
        # Try: current * 0.9, current * 0.95, current * 1.05, current * 1.1
        # Also try: 0, current/2, current*2
        candidates = [current * 0.85, current * 0.95, current * 1.05, current * 1.15]
        candidates.append(0.0)
        if current > 0:
            candidates.append(current * 0.5)
        
        best_candidate = current
        best_local = best_val
        
        for val in candidates:
            if val < 0 or val > 1000:
                continue
            a[j] = val
            # Renormalize
            a = normalize_to_sum(a, S)
            c = fft_convolve_self(a)
            local_val = c.max()
            if local_val < best_local:
                best_local = local_val
                best_candidate = val
        
        a[j] = best_candidate
        a = normalize_to_sum(a, S)
        if best_local < best_val:
            best_val = best_local
    
    return a, best_val


def multi_scale_warm_start(a_large, n_small):
    """Downsample a_large to n_small by block averaging."""
    n = len(a_large)
    nblocks = n // n_small
    a_small = a_large[:nblocks * n_small].reshape(n_small, nblocks).mean(axis=1)
    return a_small


def upsample(a_small, n_large):
    """Upsample a_small to n_large by linear interpolation."""
    if len(a_small) == n_large:
        return a_small.copy()
    t_old = np.linspace(0, 1, len(a_small))
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)


def generate_diverse_inits(n, rng):
    """Generate diverse initialization sequences."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)
    
    # Various smooth functions
    inits.append(np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * 3 * t), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * 5 * t + 0.7) + 0.2 * np.sin(2 * np.pi * 11 * t), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.sin(2 * np.pi * 2 * t - 0.3) - 0.2 * np.cos(2 * np.pi * 7 * t + 1.2), 0.01))
    inits.append(np.maximum(1.0 + 0.5 * np.sin(2 * np.pi * 4 * t + np.pi/4), 0.01))
    inits.append(np.maximum(1.0 + 0.3 * np.cos(4 * np.pi * t) + 0.3 * np.cos(6 * np.pi * t + 1), 0.01))
    
    # Bump functions
    inits.append(np.exp(-3.0 * (t - 0.3) ** 2) + 0.3 * np.exp(-3.0 * (t - 0.7) ** 2) + 0.05)
    inits.append(np.exp(-5.0 * (t - 0.5) ** 2) + 0.05)
    
    # Random smooth
    coeffs = rng.normal(0, 0.3, min(20, n // 4))
    freqs = np.arange(1, len(coeffs) + 1)
    rand_smooth = 1.0 + np.zeros(n)
    for k, (c, f) in enumerate(zip(coeffs, freqs)):
        rand_smooth += c * np.sin(2 * np.pi * f * t + rng.uniform(0, 2 * np.pi))
    inits.append(np.maximum(rand_smooth, 0.01))
    
    # Pure random
    inits.append(rng.random(n) * 0.5 + 0.5)
    inits.append(np.abs(rng.normal(1.0, 0.3, n)))
    
    return inits

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-phase optimization combining Adam+LSE exploration,
multi-scale warm starts, subgradient FISTA polishing, and coordinate descent.
Key improvements over previous: adaptive per-coordinate learning (Adam),
faster beta ramp, longer dedicated polish phases, and structural awareness
exploiting the fact that at optimum multiple lags tie at the max.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.full(n, S / n)
    if abs(s - S) < 1e-12:
        return v
    # Simple scaling if max stays below 1000
    scaled = v * (S / s)
    if scaled.max() <= 1000.0:
        return scaled
    # Proper simplex projection
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_lse_grad(a, beta):
    """Compute LSE value and gradient of max(c) approximation."""
    n = len(a)
    c = fft_convolve_self(a)
    cm = c.max()
    # Numerical stability: shift so max exponent is 0
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        lse_val = cm
        w = np.zeros_like(c)
        w[c == cm] = 1.0 / np.sum(c == cm)
    else:
        lse_val = cm + np.log(tot) / beta
        w = e / tot
    # Gradient: 2 * correlation(w, a)
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad, c


def compute_subgradient(a, c):
    """Compute subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def multi_max_subgradient(a, c, threshold_frac=0.999):
    """Subgradient averaged over all lags within threshold_frac of max."""
    n = len(a)
    cm = c.max()
    active = np.where(c >= threshold_frac * cm)[0]
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    grad /= len(active)
    return grad


def run_adam_lse(a_init, max_time, rng,
                 beta_start=10.0, beta_end=500.0, lr=0.01,
                 beta1=0.9, beta2=0.999, epsilon=1e-8,
                 restart_interval=3000, noise_scale=0.003):
    """Adam optimizer with LSE-smoothed max objective."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a_init, S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    best_val = np.inf
    best_a = a.copy()
    last_improve = 0
    stall_count = 0

    while time.monotonic() - start < max_time:
        t += 1
        # Beta schedule: smoothstep from start to end over 70% of time
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / (max_time * 0.7))
        # Smoothstep for smoother transition
        progress = progress * progress * (3 - 2 * progress)
        beta = beta_start + (beta_end - beta_start) * progress

        val, grad, c = compute_lse_grad(a, beta)
        actual_max = c.max()

        if actual_max < best_val:
            best_val = actual_max
            best_a = a.copy()
            last_improve = t
            stall_count = 0
        else:
            stall_count += 1

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Adaptive learning rate
        lr_t = lr / (1.0 + 0.0005 * t)
        # Reduce learning rate when stalled
        if stall_count > restart_interval:
            lr_t *= 0.5
            # Inject noise to escape
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            a = project_nonneg_sum(a + noise, S)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            last_improve = 0
            stall_count = 0
            continue

        a = a - lr_t * m_hat / (np.sqrt(v_hat) + epsilon)
        a = project_nonneg_sum(a, S)

    return best_a, best_val


def run_fista_subgrad(a_init, max_time, rng,
                      step_init=0.01, step_decay=0.9995,
                      use_multi_max=True, threshold=0.999):
    """FISTA with exact (or multi-max) subgradient."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t_fista = 1.0
    step = step_init
    start = time.monotonic()
    best_val = np.inf
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()

        if cur < best_val:
            best_val = cur
            best_a = y.copy()
            last_improve = it

        if it - last_improve > 5000:
            # Stuck: inject noise and reset
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_nonneg_sum(y + noise, S)
            t_fista = 1.0
            step = step_init * 0.5
            last_improve = it
            continue

        if use_multi_max:
            grad = multi_max_subgradient(y, c, threshold)
        else:
            grad = compute_subgradient(y, c)

        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        grad = grad / gn
        x = project_nonneg_sum(y - step * gn, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista))
        y = x + ((t_fista - 1.0) / t_new) * (x - a)
        y = np.maximum(y, 0.0)
        # Ensure sum constraint
        y = normalize_to_sum(y, S)
        a = x
        t_fista = t_new
        step *= step_decay

    return best_a, best_val


def coordinate_polish(a, max_time, rng):
    """Coordinate-wise local optimization."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_to_sum(a, S).copy()
    start = time.monotonic()
    best_val = fft_convolve_self(a).max()
    improved = True

    # Multiple passes
    while improved and time.monotonic() - start < max_time:
        improved = False
        # Try each coordinate
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - start > max_time:
                break
            orig = a[j]
            # Try multiplicative perturbations
            for factor in [0.8, 0.9, 1.1, 1.2, 0.7, 1.3]:
                a[j] = orig * factor
                a = normalize_to_sum(a, S)
                c = fft_convolve_self(a)
                val = c.max()
                if val < best_val - 1e-10:
                    best_val = val
                    improved = True
                    break
            else:
                # No improvement, restore
                a[j] = orig
                a = normalize_to_sum(a, S)

    return a, best_val


def multi_scale_warm_start(a_large, n_small):
    """Downsample by block averaging."""
    n = len(a_large)
    nblocks = n // n_small
    a_small = a_large[:nblocks * n_small].reshape(n_small, nblocks).mean(axis=1)
    return a_small


def upsample_interp(a_small, n_large):
    """Upsample by linear interpolation."""
    if len(a_small) == n_large:
        return a_small.copy()
    t_old = np.linspace(0, 1, len(a_small))
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)


def generate_diverse_inits(n, rng):

    print("=== Phase 1: Exploration (%.0fs) ===" % explore_time)

    # Weight time allocation
    weights = np.ones(n_cands)
    for i, (name, _) in enumerate(candidates):
        if name == "given":
            weights[i] = 4.0
        elif name.startswith("ms_"):
            weights[i] = 3.0
        else:
            weights[i] = 0.8
    weights /= weights.sum()
    times = explore_time * weights

    p1_start = time.monotonic()
    for i, (name, arr) in enumerate(candidates):
        if time.monotonic() > deadline - remaining * (1 - explore_frac) * 0.5:
            break
        t_alloc = min(times[i], deadline - time.monotonic() - 10)
        if t_alloc < 3:
            continue

        a_opt, val = run_adam_lse(
            arr, t_alloc, rng,
            beta_start=8.0, beta_end=200.0, lr=0.008,
            restart_interval=2000, noise_scale=0.005)

        a_norm = normalize_to_sum(a_opt, np.sqrt(2.0 * len(a_opt)))
        c_chk = fft_convolve_self(a_norm)
        actual = c_chk.max()

        if actual < global_best:
            global_best = actual
            global_a = a_norm.copy()
            print("  [%s] max_conv=%.6f (elapsed %.0fs)" % (name, global_best, time.monotonic() - p1_start))

    print("Phase 1 best: %.6f" % global_best)

    # Phase 2: Extended Adam on best candidate with higher beta
    if global_a is not None and time.monotonic() < deadline - 60:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        remaining2 = deadline - time.monotonic() - 60
        t2 = remaining2 * 0.35
        print("=== Phase 2: Extended Adam (%.0fs, beta 50->800) ===" % t2)

        a_opt2, val2 = run_adam_lse(
            global_a, t2, rng,
            beta_start=50.0, beta_end=800.0, lr=0.003,
            restart_interval=5000, noise_scale=0.002)

        a_norm2 = normalize_to_sum(a_opt2, S_best)
        c2 = fft_convolve_self(a_norm2)
        if c2.max() < global_best:
            global_best = c2.max()
            global_a = a_norm2.copy()
            print("  Phase 2 improved: %.6f" % global_best)

    # Phase 3: FISTA subgradient polish
    if global_a is not None and time.monotonic() < deadline - 30:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        remaining3 = deadline - time.monotonic() - 30
        t3 = remaining3 * 0.45
        print("=== Phase 3: FISTA subgradient polish (%.0fs) ===" % t3)

        # Try both single-max and multi-max
        for use_multi in [True, False]:
            if time.monotonic() > deadline - 15:
                break
            t_sub = min(t3 * 0.5, deadline - time.monotonic() - 10)
            a_opt3, val3 = run_fista_subgrad(
                global_a, t_sub, rng,
                step_init=0.003, step_decay=0.9998,
                use_multi_max=use_multi, threshold=0.9995)

            a_norm3 = normalize_to_sum(a_opt3, S_best)
            c3 = fft_convolve_self(a_norm3)
            if c3.max() < global_best:
                global_best = c3.max()
                global_a = a_norm3.copy()
                print("  FISTA (multi=%s) improved: %.6f" % (use_multi, global_best))

    # Phase 4: Coordinate polish
    if global_a is not None and time.monotonic() < deadline - 10:
        remaining4 = deadline - time.monotonic() - 5
        t4 = remaining4 * 0.9
        print("=== Phase 4: Coordinate polish (%.0fs) ===" % t4)

        a_coord, val_coord = coordinate_polish(global_a, t4, rng)
        a_coord = normalize_to_sum(a_coord, np.sqrt(2.0 * len(a_coord)))
        c_coord = fft_convolve_self(a_coord)
        if c_coord.max() < global_best:
            global_best = c_coord.max()
            global_a = a_coord.copy()
            print("  Coordinate polish improved: %.6f" % global_best)

    # Phase 5: Final tight polish with very small Adam steps
    if global_a is not None and time.monotonic() < deadline - 5:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        t5 = deadline - time.monotonic() - 3
        print("=== Phase 5: Final micro-polish (%.0fs) ===" % t5)

        a_fin, val_fin = run_adam_lse(
            global_a, t5, rng,
            beta_start=200.0, beta_end=2000.0, lr=0.001,
            beta1=0.95, beta2=0.9999,
            restart_interval=10000, noise_scale=0.001)

        a_fin = normalize_to_sum(a_fin, S_best)
        c_fin = fft_convolve_self(a_fin)
        if c_fin.max() < global_best:
            global_best = c_fin.max()
            global_a = a_fin.copy()
            print("  Final micro-polish improved: %.6f" % global_best)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification score: %.6f" % ver)
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            n_fb = 8192
            global_a = np.full(n_fb, np.sqrt(2.0 * n_fb) / n_fb)

    if global_a is None:
        n_fb = 8192
        global_a = np.full(n_fb, np.sqrt(2.0 * n_fb) / n_fb)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 8192
        global_a = np.full(n_fb, np.sqrt(2.0 * n_fb) / n_fb)

    total_time = time.monotonic() - overall_start
    print("Done in %.0fs. Returning length=%d, internal max_conv=%.6f" %
          (total_time, len(global_a), global_best))
    return [float(x) for x in global_a]
