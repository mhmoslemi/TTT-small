def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    # Standard simplex projection
    if v.sum() == S:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates < w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(np.asarray(v, dtype=np.float64), 0.0)
    if abs(v.sum() - S) < 1e-12 * max(1.0, S):
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates < w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient_fft(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a.
    grad[i] = (2/beta) * sum_k w[k] * a[k-i] (valid range)
    Computed via FFT cross-correlation.
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    # Cross-correlation: corr[i] = sum_j w[j+i] * a[j]
    # Using FFT: IFFT(FFT(w, N) * conj(FFT(a, N)))[:n]
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    corr = np.real(ifft(W * np.conj(A)))[:n]
    return (2.0 / beta) * corr


def topk_gradient_fft(a, c, top_k, temperature=1.0):
    """Gradient using top-k lags with softmax weighting."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices].copy()
    tv_max = top_vals.max()
    inv_temp = 1.0 / max(temperature, 1e-10)
    weights = np.exp((top_vals - tv_max) * inv_temp)
    weights = weights / weights.sum()
    w_full = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]

    N = 1 << (3 * n - 2 - 1).bit_length()
    W = fft(w_full, N)
    A = fft(a, N)
    corr = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * corr


def spectral_reflat(a, S, rng, n):
    """
    Attempt to flatten the power spectrum of a while maintaining non-negativity.
    Go to frequency domain, blend magnitudes toward flat, return time-domain result.
    """
    fsize = 1 << (2 * n - 1 - 1).bit_length()
    A = fft(a, fsize)
    # Current magnitudes
    mags = np.abs(A)
    # Target: flatten away from DC
    mag_mean = mags[1:].mean()
    if mag_mean < 1e-10:
        return a
    # Blend: new_mag = (1-alpha)*mag + alpha*mag_mean
    alpha = 0.1
    new_mags = mags.copy()
    new_mags[1:] = (1 - alpha) * mags[1:] + alpha * mag_mean
    # Keep DC fixed
    new_mags[0] = mags[0]
    # Reconstruct with original phases
    phases = np.angle(A)
    A_new = new_mags * np.exp(1j * phases)
    # Enforce conjugate symmetry
    A_new = 0.5 * (A_new[:fsize // 2 + 1])
    full = np.zeros(fsize, dtype=np.complex128)
    full[:fsize // 2 + 1] = A_new[:fsize // 2 + 1]
    if fsize % 2 == 0:
        full[-(fsize // 2 - 1):] = np.conj(full[2:-1][::-1])
    else:
        full[-(fsize // 2):] = np.conj(full[1:-1][::-1])
    a_new = np.real(ifft(full))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new * (S / s)
    return a_new


def run_lse_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=800.0,
                  reflat_interval=800):
    """LSE-FISTA with adaptive beta, backtracking, and spectral re-flattening."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = 1.0  # Lipschitz estimate

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()

        if beta > 50:
            grad = lse_gradient_fft(y, c, beta)
        else:
            grad = lse_gradient_fft(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / L
        f_y = fft_convolve_self(y).max()
        x_new = None
        for bt in range(6):
            x_cand = project_simplex(y - step * grad, S)
            if bt == 0:
                x_new = x_cand
            else:
                c_cand = fft_convolve_self(x_cand)
                f_cand = c_cand.max()
                if f_cand <= f_y - 0.5 * step * step * gnorm * gnorm * 0.1:
                    x_new = x_cand
                    break
                step *= 0.5
        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        # Adaptive L
        c_x = fft_convolve_self(x_new)
        if c_x.max() > f_y + 1e-10:
            L *= 1.5
        elif it % 50 == 0:
            L = max(L * 0.9, 0.5)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic spectral re-flattening
        if it % reflat_interval == 0 and rng.random() < 0.4:
            a_ref = spectral_reflat(a, S, rng, n)
            c_ref = fft_convolve_self(a_ref)
            if c_ref.max() < cur_max * 1.02:
                a = a_ref
                y = a_ref.copy()
                t_mom = 1.0

        # Periodic perturbation
        if it % 2000 == 0 and rng.random() < 0.2:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0

    return best_local, best_score


def run_topk_fista(a_init, n, S, max_time, rng, top_k=5, top_k_max=50):
    """FISTA using top-k gradient (sharper focus on peak lags)."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        top_k_curr = min(top_k_max, top_k + it // 500)
        temp = max(0.5, 3.0 * (1.0 - min(1.0, it / 5000.0)))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()

        grad = topk_gradient_fft(y, c, top_k_curr, temperature=temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / L
        f_y = cur_max
        x_new = None
        for bt in range(5):
            x_cand = project_simplex(y - step * grad, S)
            if bt == 0:
                x_new = x_cand
            else:
                c_cand = fft_convolve_self(x_cand)
                if c_cand.max() <= f_y + 1e-12:
                    x_new = x_cand
                    break
                step *= 0.5
        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        c_x = fft_convolve_self(x_new)
        if c_x.max() > f_y + 1e-10:
            L *= 1.4
        elif it % 30 == 0:
            L = max(L * 0.92, 0.3)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 3000 == 0 and rng.random() < 0.15:
            noise = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0

    return best_local, best_score


def run_pair_swap(a_init, n, S, max_time, rng, num_top_lags=5, max_swaps_per_iter=20):
    """
    Coordinate descent via pair swaps: find pairs (i,j) where moving mass
    from i to j (and vice versa for the conjugate) reduces the peak.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        # Get top lags
        top_lags = np.argsort(c)[::-1][:num_top_lags]
        improved = False

        for _ in range(max_swaps_per_iter):
            # Pick a random lag from top lags
            lag = int(top_lags[rng.integers(len(top_lags))])

            # Find contributing pairs for this lag
            i_min = max(0, lag - (n - 1))
            i_max = min(lag, n - 1)
            if i_min >= i_max:
                continue
            indices = np.arange(i_min, i_max + 1)
            partners = lag - indices
            products = a[indices] * a[partners]

            # Find the pair with largest product
            max_idx = np.argmax(products)
            i = int(indices[max_idx])
            j = int(partners[max_idx])
            if i == j:
                continue

            # Try: move small amount from max(a[i],a[j]) to min
            delta = 0.01 * max(a[i], a[j])
            if delta > a[i] * 0.5:
                delta = a[i] * 0.3
            if delta > a[j] * 0.5:
                delta = a[j] * 0.3
            if delta < 1e-10:
                continue

            # Try reducing the larger, increasing the smaller
            if a[i] >= a[j]:
                a[i] -= delta
                a[j] += delta
            else:
                a[j] -= delta
                a[i] += delta

            c_new = fft_convolve_self(a)
            if c_new.max() < cur_max - 1e-12:
                cur_max = c_new.max()
                improved = True
                if cur_max < best_score:
                    best_score = cur_max
                    best_local = a.copy()
            else:
                # Revert
                if a[i] >= a[j] - delta:
                    a[i] += delta
                    a[j] -= delta
                else:
                    a[j] += delta
                    a[i] -= delta

        if not improved and it % 100 == 0:
            # Try a different strategy: increase small elements, decrease large
            threshold = np.median(a)
            small_mask = a < threshold * 0.5
            large_mask = a > threshold * 2.0
            if np.any(small_mask) and np.any(large_mask):
                si = int(np.where(small_mask)[0][rng.integers(np.sum(small_mask))])
                li = int(np.where(large_mask)[0][rng.integers(np.sum(large_mask))])
                delta = 0.05 * a[li]
                a[li] -= delta
                a[si] += delta
                a = np.maximum(a, 0)
                a = a * (S / a.sum())
                c_new = fft_convolve_self(a)
                if c_new.max() > best_score * 1.05:
                    # Revert
                    a = project_simplex(a, S)

    return best_local, best_score


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.05
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum()) if a.sum() > 0 else np.ones(n) * S / n
    return a


def make_spectral_init(n, rng, S):
    """Spectral initialization: random phases in frequency domain."""
    fsize = 1 << (2 * n - 1 - 1).bit_length()
    A = np.zeros(fsize, dtype=np.complex128)
    A[0] = S
    # Random phases for non-DC components
    mags = S / np.sqrt(n) * (0.5 + 0.5 * rng.random(fsize // 2 + 1))
    mags[0] = S
    phases = rng.uniform(0, 2 * np.pi, fsize // 2 + 1)
    phases[0] = 0
    half_A = mags * np.exp(1j * phases)
    full_A = np.zeros(fsize, dtype=np.complex128)
    full_A[:fsize // 2 + 1] = half_A
    if fsize % 2 == 0:
        full_A[-(fsize // 2 - 1):] = np.conj(full_A[2:-1][::-1])
    else:
        full_A[-(fsize // 2):] = np.conj(full_A[1:-1][::-1])
    a = np.real(ifft(full_A))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    return a


def make_periodic_init(n, rng, S, num_freqs=3):
    """Periodic initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    for _ in range(num_freqs):
        freq = rng.integers(2, 25)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.4
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_winit(n, rng, S):
    """W-shaped initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    k = rng.integers(2, 6)
    a = np.abs(np.sin(np.pi * k * tt)) + 0.05
    a = a * (S / a.sum())
    return a


def generate_inits(n, rng, S, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and len(parent) >= 64:
        p = np.clip(parent.astype(np.float64), 0, 1000)
        if p.sum() > 0:
            inits.append(("parent", p * (S / p.sum())))
    for i in range(4):
        inits.append(("anti%d" % i, make_anti_palindromic_init(n, rng, S)))
    for i in range(3):
        inits.append(("spec%d" % i, make_spectral_init(n, rng, S)))
    for i in range(3):
        inits.append(("peri%d" % i, make_periodic_init(n, rng, S)))
    for i in range(2):
        inits.append(("w%d" % i, make_winit(n, rng, S)))
    rand_a = rng.random(n) + 0.05
    inits.append(("rand", rand_a * (S / rand_a.sum())))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None

    # Get parent
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    n = 8192
    if parent is not None and len(parent) >= 64:
        n = len(parent)
    S = np.sqrt(2.0 * float(n))

    print("n=%d, S=%.4f, parent=%s" % (n, S, "yes" if parent is not None else "no"))

    # Phase 1: Multi-initialization exploration
    inits = generate_inits(n, rng, S, parent)
    usable_time = max(30.0, deadline - time.monotonic())
    per_init = usable_time * 0.55 / len(inits)

    print("Phase 1: %d inits, ~%.1fs each" % (len(inits), per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 300:
            break
        t0 = time.monotonic()

        # LSE-FISTA
        seg1 = max(3.0, per_init * 0.55)
        a_f, s_f = run_lse_fista(arr, n, S, seg1, rng,
                                  beta_init=5.0, beta_max=500.0)
        c_f = fft_convolve_self(a_f)
        obj_f = c_f.max()
        if obj_f < global_best:
            global_best = obj_f
            global_a = a_f.copy()
            print("  LSE [%s]: %.6f (%.0fs)" % (name, obj_f, time.monotonic() - t0))

        # Top-k FISTA
        rem = per_init - (time.monotonic() - t0)
        if rem > 3:
            a_t, s_t = run_topk_fista(a_f, n, S, rem, rng,
                                       top_k=3, top_k_max=30)
            c_t = fft_convolve_self(a_t)
            if c_t.max() < global_best:
                global_best = c_t.max()
                global_a = a_t.copy()
                print("  TopK [%s]: %.6f" % (name, global_best))

    print("After Phase 1: best=%.6f" % global_best)

    # Phase 2: Intensive refinement on best
    if global_a is not None and time.monotonic() < deadline - 120:
        remaining = deadline - time.monotonic()
        print("Phase 2: Refinement with %.0fs remaining" % remaining)

        # 2a: High-beta LSE-FISTA
        seg = max(10.0, remaining * 0.25)
        a_h, s_h = run_lse_fista(global_a, n, S, seg, rng,
                                  beta_init=50.0, beta_max=2000.0,
                                  reflat_interval=1000)
        c_h = fft_convolve_self(a_h)
        if c_h.max() < global_best:
            global_best = c_h.max()
            global_a = a_h.copy()
            print("  HiBeta: %.6f" % global_best)

        # 2b: Top-k with very low temperature
        remaining2 = deadline - time.monotonic()
        seg2 = max(10.0, remaining2 * 0.20)
        a_tk, s_tk = run_topk_fista(global_a, n, S, seg2, rng,
                                     top_k=1, top_k_max=10)
        c_tk = fft_convolve_self(a_tk)
        if c_tk.max() < global_best:
            global_best = c_tk.max()
            global_a = a_tk.copy()
            print("  TopK1: %.6f" % global_best)

        # 2c: Pair swap
        remaining3 = deadline - time.monotonic()
        seg3 = max(10.0, remaining3 * 0.20)
        a_ps, s_ps = run_pair_swap(global_a, n, S, seg3, rng,
                                    num_top_lags=5, max_swaps_per_iter=50)
        c_ps = fft_convolve_self(a_ps)
        if c_ps.max() < global_best:
            global_best = c_ps.max()
            global_a = a_ps.copy()
            print("  PairSwap: %.6f" % global_best)

        # 2d: Another LSE pass with different params
        remaining4 = deadline - time.monotonic()
        if remaining4 > 10:
            seg4 = remaining4 * 0.25
            a_f2, s_f2 = run_lse_fista(global_a, n, S, seg4, rng,
                                        beta_init=100.0, beta_max=3000.0,
                                        reflat_interval=500)
            c_f2 = fft_convolve_self(a_f2)
            if c_f2.max() < global_best:
                global_best = c_f2.max()
                global_a = a_f2.copy()
                print("  LSE2: %.6f" % global_best)

        # 2e: Final pair swap polish
        remaining5 = deadline - time.monotonic()
        if remaining5 > 10:
            a_ps2, s_ps2 = run_pair_swap(global_a, n, S, remaining5 * 0.3, rng,
                                          num_top_lags=3, max_swaps_per_iter=100)
            c_ps2 = fft_convolve_self(a_ps2)
            if c_ps2.max() < global_best:
                global_best = c_ps2.max()
                global_a = a_ps2.copy()
                print("  PairSwap2: %.6f" % global_best)

    # Phase 3: Try longer sequence if time allows
    if time.monotonic() < deadline - 100:
        n_long = n * 2 if n <= 8192 else n * 3
        if n_long > 65536:
            n_long = 32768
        if n_long != n:
            S_long = np.sqrt(2.0 * float(n_long))
            t0 = time.monotonic()
            print("Phase 3: Trying n=%d" % n_long)

            if parent is not None and len(parent) >= 64:
                from numpy import interp
                tt_old = np.linspace(0, 1, len(parent))
                tt_new = np.linspace(0, 1, n_long)
                a_long = interp(tt_new, tt_old, parent)
                a_long = np.maximum(a_long, 0.01)
            else:
                a_long = make_anti_palindromic_init(n_long, rng, S_long)
            a_long = a_long * (S_long / a_long.sum())

            seg_long = min(200.0, max(30.0, (deadline - time.monotonic()) * 0.3))
            a_l, s_l = run_lse_fista(a_long, n_long, S_long, seg_long, rng,
                                      beta_init=10.0, beta_max=500.0)
            c_l = fft_convolve_self(a_l)
            obj_l = c_l.max()
            # Convert to comparable objective: 2*n*max_c / S^2 = 2*n*max_c/(2n) = max_c
            # So we just compare max_c values directly
            if obj_l < global_best:
                global_best = obj_l
                global_a = a_l.copy()
                print("  Long n=%d: %.6f" % (n_long, obj_l))
            print("  Phase 3 done in %.1fs" % (time.monotonic() - t0))

    # Final verification
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Final verify: %.6f (internal max_c=%.6f), n=%d" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver2 = evaluate_sequence(final_list)
        print("Fallback: %.6f" % (ver2 if np.isfinite(ver2) else float('inf')))

    return final_list
