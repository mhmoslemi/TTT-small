"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- Phase 1: L-BFGS-B with log-sum-exp continuation (warm start from given sequence)
- Phase 2: Nesterov accelerated subgradient with adaptive top-K geometric weighting
  and exact line search on the true (non-smooth) objective
- Phase 3: Structured perturbation (sinusoidal) + short L-BFGS re-optimization
- Phase 4: Final high-beta polish
Key improvement: direct subgradient on unsmoothed max with line search, adaptive K,
and sinusoidal perturbation for escaping local minima.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_from_weights(a, w):
    """
    Compute gradient of sum_k w[k] * c_k w.r.t. a, where c = full_conv(a, a).
    Uses FFT cross-correlation.
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z (softmax params)."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_grad_from_weights(a, w)

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=200):
    """Run L-BFGS-B with continuation on beta."""
    if beta_schedule is None:
        beta_schedule = [10, 30, 100, 300, 1000, 3000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break

        n_attempts = 1
        if beta >= 300 and max_time > 30:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 3:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.02, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            t0 = time.monotonic()
            try:
                result = scipy_minimize(
                    fun, z,
                    jac=jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_beta,
                        'maxfun': maxiter_per_beta * 2,
                        'ftol': 1e-14,
                        'gtol': 1e-9,
                        'maxls': 40,
                        'iprint': -1
                    }
                )
            except Exception:
                break

            z = result.x
            z = z - z.mean()

            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

        if time.monotonic() - start < max_time:
            print("    beta=%5d: score=%.8f (%.1fs)" %
                  (beta, best_score, time.monotonic() - start))

    return best_a, best_score


def run_nesterov_subgrad(a_init, n, S, max_time, rng):
    """
    Nesterov accelerated subgradient with adaptive top-K geometric weighting
    and periodic structured perturbations.
    minimizes max_k c_k on the simplex {a >= 0, sum(a) = S}.
    """
    a = a_init.copy()
    s = a.sum()
    a = a * S / s
    a_prev = a.copy()
    t = 1.0
    t_prev = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stagnant = 0
    K_max = 100
    K_decay = 3000  # iterations to go from K_max to 1

    # Step size parameters
    base_step = 2.0 * S / n

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive K: decrease from K_max to 1
        K = max(1, K_max - it // K_decay)

        # Nesterov lookahead
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = a + ((t - 1.0) / t_new) * (a - a_prev)
        y = np.maximum(y, 0.0)
        # Renormalize y to keep it on/near the simplex
        sy = y.sum()
        if sy > 1e-15:
            y = y * S / sy

        # Compute autocorrelation of y
        c = fft_convolve_self(y)
        c_max = c.max()

        # Find top-K indices
        if K == 1:
            top_indices = np.array([np.argmax(c)])
        else:
            top_indices = np.argpartition(c, -K)[-K:]

        # Compute geometric weights over top-K
        sorted_top = c[top_indices]
        sorted_top = sorted_top[sorted_top > 0]
        if len(sorted_top) == 0:
            sorted_top = np.array([c_max])
            top_indices = np.array([np.argmax(c)])

        # Geometric decay weights
        n_top = len(top_indices)
        if n_top > 1:
            max_c = c[top_indices].max()
            # Weight proportional to exp(-gamma * rank)
            ranks = np.arange(n_top, dtype=np.float64)
            gamma = 2.0  # sharpness
            weights = np.exp(-gamma * ranks / max(1, n_top - 1))
            weights = weights / weights.sum()
        else:
            weights = np.array([1.0])

        # Build full weight vector
        w_full = np.zeros(2 * n - 1)
        for i, k_idx in enumerate(top_indices):
            w_full[int(k_idx)] = weights[i]

        # Compute gradient
        grad = compute_grad_from_weights(y, w_full)

        # Step size (Nesterov optimal)
        step = 2.0 * S / (t_new * np.linalg.norm(grad, ord=np.inf) + 1e-30)
        step = min(step, 10.0 * S / n)

        # Take step and project
        x_new = project_simplex(y - step * grad, S)

        # Check objective at x_new periodically (every 10 iters or if promising)
        if it % 10 == 0 or it < 100:
            c_new = fft_convolve_self(x_new)
            score_new = c_new.max()
            if score_new < best_score:
                best_score = score_new
                best_a = x_new.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        # Update Nesterov state
        a_prev = a
        a = x_new
        t = t_new

        # Periodic structured perturbation for escaping local minima
        if stagnant > 5000:
            # Try a sinusoidal perturbation
            freq = rng.integers(2, 50)
            phase = rng.uniform(0, 2 * np.pi)
            tt = np.arange(n, dtype=np.float64)
            pert = np.cos(2.0 * np.pi * freq * tt / n + phase)
            pert = pert / np.linalg.norm(pert)
            eps = S / np.sqrt(n) * 0.05
            a_pert = a + eps * pert
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert * S / a_pert.sum()

            c_pert = fft_convolve_self(a_pert)
            if c_pert.max() < best_score:
                best_score = c_pert.max()
                best_a = a_pert.copy()
                a = a_pert.copy()
                a_prev = a_pert.copy()
                t = 1.0
                stagnant = 0
                print("      [Perturb freq=%d] score=%.8f" % (freq, best_score))
            else:
                # Revert and increase perturbation
                eps *= 2
                a_pert = a + eps * pert
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert * S / a_pert.sum()
                c_pert = fft_convolve_self(a_pert)
                if c_pert.max() < best_score:
                    best_score = c_pert.max()
                    best_a = a_pert.copy()
                    a = a_pert.copy()
                    a_prev = a_pert.copy()
                    t = 1.0
                    stagnant = 0

            # Reset stagnation counter regardless
            stagnant = 0

        # Also try random jump every 20000 iters
        if it % 20000 == 0:
            a_jump = a + rng.normal(0, 0.01 * S / np.sqrt(n), n)
            a_jump = np.maximum(a_jump, 0.0)
            a_jump = a_jump * S / a_jump.sum()
            c_jump = fft_convolve_self(a_jump)
            if c_jump.max() < best_score:
                best_score = c_jump.max()
                best_a = a_jump.copy()
                a = a_jump.copy()
                a_prev = a_jump.copy()
                t = 1.0
                stagnant = 0

    return best_a, best_score


def run_line_search_subgrad(a_init, n, S, max_time, rng):
    """
    Subgradient method with exact line search (no momentum).
    More robust than Nesterov for non-smooth objectives.
    """
    a = a_init.copy()
    s = a.sum()
    a = a * S / s
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    stagnant = 0
    base_alpha = 1.0

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive K
        K = max(1, 50 - it // 5000)

        # Compute autocorrelation
        c = fft_convolve_self(a)
        c_max = c.max()

        # Find top-K
        if K == 1:
            top_indices = np.array([np.argmax(c)])
        else:
            top_indices = np.argpartition(c, -K)[-K:]

        # Geometric weights
        n_top = len(top_indices)
        if n_top > 1:
            gamma = 3.0
            ranks = np.arange(n_top, dtype=np.float64)
            weights = np.exp(-gamma * ranks / max(1, n_top - 1))
            weights = weights / weights.sum()
        else:
            weights = np.array([1.0])

        w_full = np.zeros(2 * n - 1)
        for i, k_idx in enumerate(top_indices):
            w_full[int(k_idx)] = weights[i]

        grad = compute_grad_from_weights(a, w_full)

        # Line search: try several step sizes
        alpha = base_alpha / (1.0 + 0.001 * it)
        best_candidate = None
        best_candidate_score = c_max

        for trial in range(6):
            candidate = project_simplex(a - alpha * grad, S)
            c_trial = fft_convolve_self(candidate)
            score_trial = c_trial.max()
            if score_trial < best_candidate_score:
                best_candidate_score = score_trial
                best_candidate = candidate
            alpha *= 0.5

        if best_candidate is not None:
            a = best_candidate
            if best_candidate_score < best_score:
                best_score = best_candidate_score
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1
            # If line search fails, try a small random perturbation
            a = a + rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = np.maximum(a, 0.0)
            a = a * S / a.sum()

        # Restart on stagnation
        if stagnant > 8000:
            freq = rng.integers(2, 100)
            tt = np.arange(n, dtype=np.float64)
            pert = np.sin(2.0 * np.pi * freq * tt / n + rng.uniform(0, 2 * np.pi))
            eps = 0.03 * S / np.sqrt(n)
            a_pert = a + eps * pert
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert * S / a_pert.sum()
            c_pert = fft_convolve_self(a_pert)
            if c_pert.max() < best_score:
                best_score = c_pert.max()
                best_a = a_pert.copy()
                a = a_pert.copy()
            stagnant = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2300):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # --- Get initial sequence ---
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0.0, 1000.0)))
    except Exception:
        pass

    if inits:
        n = len(inits[0][1])
    else:
        n = 8192
        inits.append(("ones", np.ones(n)))

    S = np.sqrt(2.0 * float(n))

    print("=== Optimization: n=%d, budget=%.0fs ===" % (n, budget_s))
    print("Target: 1.5027, Previous: 1.508805")

    # Initial evaluation
    a0 = safe_normalize(inits[0][1], n)
    c0 = fft_convolve_self(a0)
    global_best = c0.max()
    global_a = a0.copy()
    print("Initial score: %.8f" % global_best)

    # --- Phase 1: L-BFGS Continuation (25% of time) ---
    p1_time = (deadline - time.monotonic()) * 0.25
    print("\n=== Phase 1: L-BFGS Continuation (%.0fs) ===" % p1_time)
    t0 = time.monotonic()

    a1, score1 = run_lbfgs_continuation(
        a0, n, S, p1_time, rng,
        beta_schedule=[10, 30, 100, 300, 1000, 3000, 5000],
        maxiter_per_beta=200
    )
    a1 = safe_normalize(a1, n)
    c1 = fft_convolve_self(a1)
    if c1.max() < global_best:
        global_best = c1.max()
        global_a = a1.copy()
        print("  Phase 1 improved: %.8f" % global_best)
    else:
        print("  Phase 1: no improvement (best=%.8f)" % global_best)
    print("  Time: %.1fs" % (time.monotonic() - t0))

    # --- Phase 2a: Nesterov Subgradient (35% of time) ---
    remaining = deadline - time.monotonic()
    p2a_time = remaining * 0.45
    print("\n=== Phase 2a: Nesterov Subgradient (%.0fs) ===" % p2a_time)
    t0 = time.monotonic()

    a2a, score2a = run_nesterov_subgrad(global_a, n, S, p2a_time, rng)
    a2a = safe_normalize(a2a, n)
    c2a = fft_convolve_self(a2a)
    if c2a.max() < global_best:
        global_best = c2a.max()
        global_a = a2a.copy()
        print("  2a improved: %.8f" % global_best)
    else:
        print("  2a: no improvement (best=%.8f, phase2a=%.8f)" % (global_best, score2a))
    print("  Time: %.1fs" % (time.monotonic() - t0))

    # --- Phase 2b: Line Search Subgradient (25% of time) ---
    remaining = deadline - time.monotonic()
    p2b_time = remaining * 0.45
    print("\n=== Phase 2b: Line Search Subgradient (%.0fs) ===" % p2b_time)
    t0 = time.monotonic()

    a2b, score2b = run_line_search_subgrad(global_a, n, S, p2b_time, rng)
    a2b = safe_normalize(a2b, n)
    c2b = fft_convolve_self(a2b)
    if c2b.max() < global_best:
        global_best = c2b.max()
        global_a = a2b.copy()
        print("  2b improved: %.8f" % global_best)
    else:
        print("  2b: no improvement (best=%.8f, phase2b=%.8f)" % (global_best, score2b))
    print("  Time: %.1fs" % (time.monotonic() - t0))

    # --- Phase 3: Perturbation + L-BFGS restart (15% of time) ---
    remaining = deadline - time.monotonic()
    if remaining > 30:
        p3_time = remaining * 0.7
        print("\n=== Phase 3: Perturbation + L-BFGS (%.0fs) ===" % p3_time)
        t0 = time.monotonic()

        n_perturb = max(2, int(p3_time / 30))
        per_perturb = p3_time / n_perturb

        for p in range(n_perturb):
            if time.monotonic() > deadline - 30:
                break

            # Structured perturbation
            freq = rng.integers(3, 80)
            phase = rng.uniform(0, 2 * np.pi)
            tt = np.arange(n, dtype=np.float64)
            pert = np.sin(2.0 * np.pi * freq * tt / n + phase)
            eps = 0.02 * S / np.sqrt(n)
            a_pert = global_a + eps * pert
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = safe_normalize(a_pert, n)

            a_new, score_new = run_lbfgs_continuation(
                a_pert, n, S, per_perturb, rng,
                beta_schedule=[100, 500, 2000, 5000],
                maxiter_per_beta=150
            )
            a_new = safe_normalize(a_new, n)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < global_best:
                global_best = c_new.max()
                global_a = a_new.copy()
                print("  Perturb %d (freq=%d): %.8f" % (p, freq, global_best))

        print("  Time: %.1fs" % (time.monotonic() - t0))

    # --- Phase 4: Final polish with high-beat L-BFGS ---
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("\n=== Phase 4: Final Polish (%.0fs) ===" % remaining)
        t0 = time.monotonic()

        a_final, score_final = run_lbfgs_continuation(
            global_a, n, S, remaining - 5, rng,
            beta_schedule=[500, 2000, 5000, 10000, 20000],
            maxiter_per_beta=300
        )
        a_final = safe_normalize(a_final, n)
        c_final = fft_convolve_self(a_final)
        if c_final.max() < global_best:
            global_best = c_final.max()
            global_a = a_final.copy()
            print("  Final: %.8f" % global_best)
        else:
            print("  Final: no improvement (best=%.8f)" % global_best)

    # --- Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
