    # Simplest correct approach for our case:
    # Construct the full weight matrix effect via FFT
    # w has length 2n-1. a has length n.
    # grad[j] = sum_{k=max(0,j)}^{min(2n-2,j+n-1)} w[k] * a[k-j] + sum_{k=max(0,j-n+1)}^{min(2n-2,j)} w[k] * a[j-k]
    
    # First sum: let k' = k-j, ranges from max(0,j)-j=0 to min(2n-2,j+n-1)-j = min(n-2, n-1) = n-1
    #   = sum_{k'=0}^{n-1} w[k'+j] * a[k'] (where k'+j is valid, i.e., k'+j <= 2n-2)
    # Second sum: let k' = j-k, ranges from j-max(0,j-n+1) to j-min(2n-2,j)
    #   If j <= 2n-2: k' from j-max(0,j-n+1) to j-min(2n-2,j) = 0 to min(j, 2n-2-j, n-1)
    #   = sum_{k'=0}^{min(j,n-1,2n-2-j)} w[j-k'] * a[k']
    
    # I'll just compute both sums using sliding window / FFT:
    # First sum is: for each j, dot product of a[0:n] with w[j:j+n] (where w has length 2n-1)
    # This is the "full cross-correlation" of w and a.
    
    # Using FFT: cross_correlation = IFFT(FFT(w, N) * conj(FFT(a, N)))
    # where N >= 2n-1
    
    N_fft = 1 << (2 * n - 1).bit_length()
    W_full = fft(w, N_fft)
    A_full = fft(a, N_fft)
    # cross[j] = sum_k w[k] * a[k-j] = IFFT(W * conj(A))[shifted]
    # Actually: IFFT(FFT(w) * conj(FFT(a)))[m] = sum_k w[k+m] * a[k] (circulant)
    # For linear: we need m from 0 to 2n-2
    cross = np.real(ifft(W_full * np.conj(A_full)))
    # cross[m] = sum_k w[k+m] * a[k] for m >= 0 (with w padded to N_fft)
    # cross[m] = sum_k w[k-m] * a[k+N_fft-m] ... for m < 0 (wraparound)
    
    # For our purposes:
    # First sum for grad[j] = cross[j] (for j = 0, ..., 2n-2), but we need j < n
    # cross[j] = sum_k w[k+j]*a[k] = sum_{k=0}^{n-1} w[k+j]*a[k] (since a[k]=0 for k>=n)
    # This is exactly our first sum! (with the constraint k+j <= 2n-2, which is k <= 2n-2-j < n for j > 0)
    
    first_sum = cross[:n]
    
    # Second sum: grad[j] second part = sum_k w[k]*a[j-k] for valid k, j-k
    # Let k' = j-k: = sum_{k'} w[j-k']*a[k'] for valid j-k' >= 0 and k' in [0,n-1]
    # = sum_{k'=0}^{min(j, n-1)} w[j-k']*a[k']
    # This equals cross[-j] in the circulant sense, or equivalently:
    # = IFFT(W * conj(A))[N_fft - j] for j > 0 (the wraparound part)
    # Actually: IFFT(W * conj(A))[m] for m > 0 is sum_k w[k+m mod N]*a[k]
    # For m = N_fft - j: sum_k w[k + N_fft - j mod N_fft]*a[k] = sum_k w[(k-j) mod N_fft]*a[k]
    # For k-j in [0, N_fft-1]: = sum_k w[k-j]*a[k] (for k >= j)
    # = sum_{k=j}^{j+n-1} w[k-j]*a[k] = sum_{m=0}^{n-1} w[m]*a[m+j]
    # Hmm that's not quite what I want.
    
    # Let me just compute the second sum differently:
    # sum_k w[k]*a[j-k] = (w * a_reversed)[j] convolved... 
    # Actually: sum_k w[k]*a[j-k] where 0<=j-k<n means k in [j-n+1, j]
    # = sum_{m=0}^{min(j,n-1)} w[j-m]*a[m]
    # This is the "first sum" of cross-correlation of a and w at lag j:
    # = np.correlate(w, a, 'full')[len(w)-1+j] ... I'm getting confused.
    
    # Let me just use a different FFT trick:
    # sum_k w[k]*a[j-k] = (convolution of w and a_rev) evaluated at appropriate position
    # where a_rev[i] = a[n-1-i]
    
    # Actually the simplest: both sums together give the gradient of the FULL convolution.
    # The full convolution c[k] = sum_i a[i]*a[k-i] has the property that:
    # The matrix J[k,j] = d c[k]/d a[j] is a banded matrix with (J[k,j] = a[k-j] if 0<=k-j<n) + (a[j-k] if 0<=j-k<n)
    # 
    # And grad = J^T * w (where w is the weight vector of length 2n-1)
    #
    # So grad[j] = sum_k J[k,j] * w[k] = sum_k w[k]*(a[k-j]+a[j-k]) for valid entries
    #
    # This is: (w correlated with a) + (w correlated with a reversed) ... 
    # Let me just define:
    # P1[j] = sum_k w[k]*a[k-j] (k-j in [0,n-1]) = cross_correlation of (w, a) at shift j
    # P2[j] = sum_k w[k]*a[j-k] (j-k in [0,n-1]) = cross_correlation of (w, a) at shift -j
    # 
    # In FFT: cross_correlation(w, a)[j] = IFFT(FFT(w) * conj(FFT(a)))[j] for j>=0
    # and = IFFT(FFT(w) * conj(FFT(a)))[N-j] for j<0 (wraparound)
    
    # So: grad = cross[:n] + cross[N_fft - np.arange(1, n)]  (the second part is the negative lags)
    # Wait: P2[j] = cross_correlation at shift -j = IFFT(W*conj(A))[N_fft - j]
    
    second_sum = np.zeros(n)
    for j in range(1, n):
        second_sum[j] = cross[N_fft - j]
    # For j=0: P2[0] = sum_k w[k]*a[0] (only k where j-k=0, i.e., k=j=0) = w[0]*a[0]
    # And P1[0] = sum_k w[k]*a[k] (k in [0, n-1]) = cross[0]
    # So grad[0] = cross[0] + w[0]*a[0]
    # Hmm, actually for j=0: P2[0] = sum_k w[k]*a[0-k] = w[0]*a[0] (only k=0 is valid since k>=0 and 0-k>=0 means k<=0)
    # So P2[0] = w[0]*a[0]
    
    # But from the FFT: cross[N_fft - 0] = cross[0] = sum_k w[k]*a[k]
    # That's P1[0], not P2[0].
    # P2[0] = w[0]*a[0] which is just the first term.
    
    # I think the issue is that at j=0, the two terms overlap (the "diagonal" term).
    # Let me handle it as: grad[j] = P1[j] + P2[j] for j > 0, and grad[0] = P1[0] + w[0]*a[0] - a[0]^2
    # (subtracting the double-counted a[0]^2 term)
    
    # Actually, let me reconsider. For the full linear convolution c[k] = sum_i a[i]*a[k-i]:
    # When we differentiate w.r.t. a[j]:
    # - j appears as the first index i=j: contributes a[k-j] for valid k-j
    # - j appears as the second index k-i=j, so i=k-j: contributes a[k-(k-j)] = a[j] for valid i=k-j
    #   Wait: if i = k-j is a valid index, then the term is a[k-j]*a[j], and d/d a[j] of this = a[k-j]
    # So: d c[k] / d a[j] = a[k-j] (if 0<=k-j<n, 0<=j<n) + a[j] (if 0<=k-j<n, 0<=j<n)
    # Wait that's not right either. Let me be very explicit.
    # 
    # c[k] = sum_{i=0}^{n-1} a[i]*a[k-i] * 1[0<=k-i<n]
    # d c[k] / d a[j] = a[k-j]*1[0<=k-j<n] + a[k-j]*1[0<=j<n] ... no
    # 
    # The term a[j] appears in c[k] in two ways:
    # 1. As a[i] with i=j: term is a[j]*a[k-j], valid if 0<=k-j<n. Derivative w.r.t. a[j] is a[k-j].
    # 2. As a[k-i] with k-i=j, i.e., i=k-j: term is a[k-j]*a[j], valid if 0<=k-j<n. Derivative w.r.t. a[j] is a[k-j].
    # 
    # So d c[k] / d a[j] = 2*a[k-j] * 1[0<=k-j<n, 0<=j<n]
    # EXCEPT when k-j = j (i.e., k=2j), the two appearances are the same term a[j]*a[j],
    # and the derivative is a[j] (not 2*a[j]).
    #
    # Wait no. When k=2j, the term in the sum is a[j]*a[j] (with i=j). It appears once in the sum.
    # d/d a[j] of a[j]*a[j] = 2*a[j]. So actually it IS 2*a[j] even in this case!
    # Because a[j]*a[j] = a[j]^2 and d/d a[j] of a[j]^2 = 2*a[j].
    # 
    # So in all cases: d c[k] / d a[j] = 2*a[k-j] * 1[0<=k-j<n, 0<=j<n]
    # (The indicator 0<=j<n is already satisfied since j is our index.)
    
    # So: grad[j] = 2 * sum_k w[k] * a[k-j] * 1[0<=k-j<n]
    # = 2 * sum_{m=0}^{n-1} w[m+j] * a[m] * 1[m+j <= 2n-2]
    # = 2 * cross[j] (for j where m+j <= 2n-2, which is always true since m <= n-1 and j <= n-1, so m+j <= 2n-2)
    
    # Great! So grad = 2 * cross[:n] where cross[m] = sum_k w[k+m]*a[k]
    
    grad = 2.0 * cross[:n]
    return grad, c_max


def multiplicative_step(a, grad, eta):
    """
    Mirror descent step: a_new = a * exp(-eta * grad) / Z
    Preserves non-negativity automatically.
    """
    # For numerical stability, subtract max
    f = -eta * grad
    f_max = f.max()
    a_new = a * np.exp(f - f_max)
    s = a_new.sum()
    if s < 1e-300:
        return a
    return a_new / s


def spectral_shap_step(a, N, top_k, alpha, rng):
    """
    One step of spectral shaping: modify |FFT(a)|^2 to reduce peak convolution.
    Returns new a.
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()

    m = np.arange(N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a, c_max

    weights = top_vals / total
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new_full, a.sum())
    return a_new, c_max


def lse_smoothed_grad_multi(a, c, betas):
    """
    Compute multi-scale LSE gradient: weighted average of LSE gradients at different beta levels.
    """
    n = len(a)
    grad_total = np.zeros(n)
    weights_sum = 0.0
    for beta in betas:
        g, _ = compute_grad_lse(a, c, float(beta))
        grad_total += g
        weights_sum += 1.0
    if weights_sum > 0:
        grad_total /= weights_sum
    return grad_total


def kkt_gap(a, c, n_active=20):
    """
    Compute a KKT gap measure: variance of the gradient on the support of a.
    Smaller gap = closer to KKT conditions.
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:n_active]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return 0.0
    weights = top_vals / total

    # Compute gradient on support
    fsize = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = weights[i]
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    cross = np.real(ifft(W * np.conj(A)))
    grad = 2.0 * cross[:n]

    support = a > 1e-10 * a.max()
    if support.sum() < 2:
        return 0.0
    grad_support = grad[support]
    return float(np.std(grad_support))


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def nesterov_lse_polish(a_init, n, N, max_time, rng, beta_schedule=None):
    """
    Nesterov-accelerated LSE gradient descent with adaptive beta.
    """
    a = np.clip(a_init, 1e-10, 1000.0).copy().astype(np.float64)
    S = a.sum()
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    if beta_schedule is None:
        beta_schedule = lambda it: 5.0 * (1.0 + 0.005 * it)
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(500.0, beta_schedule(it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        grad, c_max = compute_grad_lse(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        step = S / (n * (1.0 + 0.002 * it))
        step = min(step, 0.1 * S / max(gnorm, 1e-15))
        step = max(step, 1e-8)
        
        x_new = project_simplex(y - step * grad, S)
        
        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y * (S / s_y)
        a = x_new
        t = t_new
        
        if it % 2000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.004 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score


def mirror_descent_search(a_init, n, N, max_time, rng, eta_base=0.001):
    """
    Mirror descent (multiplicative weights) optimization.
    """
    a = np.clip(a_init, 1e-10, 1000.0).copy().astype(np.float64)
    S = a.sum()
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stall_count = 0
    last_improve = best_score
    
    while time.monotonic() - start < max_time:
        it += 1
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            stall_count = 0
            last_improve = cur
        else:
            stall_count += 1
        
        # Adaptive eta: decrease over time, increase when stalling
        eta = eta_base / (1.0 + 0.001 * it)
        if stall_count > 50:
            eta *= 2.0
        
        # Multi-scale gradient
        betas = [3.0, 10.0, 30.0]
        grad = lse_smoothed_grad_multi(a, c, betas)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_normed = grad / gnorm
        
        # Mirror descent step
        a_new = multiplicative_step(a, grad_normed, eta * gnorm)
        
        # Check if it improved, if not try smaller step
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur:
            a = a_new
        else:
            # Backtrack
            a_new2 = multiplicative_step(a, grad_normed, eta * gnorm * 0.5)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < cur:
                a = a_new2
            else:
                # Keep old, add small perturbation
                a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
        
        # Periodic spectral shaping
        if it % 500 == 0 and rng.random() < 0.3:
            a, _ = spectral_shap_step(a, N, top_k=8, alpha=0.15, rng=rng)
            a = project_simplex(a, S)
        
        # Restart if very stalled
        if stall_count > 200:
            a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            stall_count = 0
    
    return best_local, best_score


def coordinate_descent_local(a_init, n, max_time, rng, block_size=16):
    """
    Block coordinate descent: optimize small blocks of coordinates.
    For each block, try random perturbations and keep the best.
    """
    a = np.clip(a_init, 1e-10, 1000.0).copy().astype(np.float64)
    S = a.sum()
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        # Pick random block
        n_blocks = n // block_size
        if n_blocks == 0:
            break
        block_idx = rng.integers(0, n_blocks)
        start_idx = block_idx * block_size
        end_idx = start_idx + block_size
        
        # Try to improve this block
        best_block = a[start_idx:end_idx].copy()
        best_block_score = cur
        
        # The block's contribution to the convolution
        # Only lags affected by this block change need to be checked
        for trial in range(5):
            # Random perturbation of the block
            perturbation = rng.normal(0, 0.05 * S / n, block_size)
            candidate = a.copy()
            candidate[start_idx:end_idx] = np.maximum(0, a[start_idx:end_idx] + perturbation)
            
            # Renormalize
            s_cand = candidate.sum()
            if s_cand < 1e-10:
                continue
            candidate = candidate * (S / s_cand)
            
            # Evaluate only the affected lags
            # Lag k is affected if any element in [start_idx, end_idx] multiplies with a valid element
            # This is lags [start_idx, end_idx + n - 1] approximately
            c_full = fft_convolve_self(candidate)
            score = c_full.max()
            
            if score < best_block_score:
                best_block = candidate[start_idx:end_idx].copy()
                best_block_score = score
                a = candidate
        
    return best_local, best_score


def make_diverse_init(n, rng, init_type):
    """Create diverse initializations."""
    tt = np.linspace(0, 1, n, endpoint=False)
    if init_type == "uniform":
        return np.ones(n)
    elif init_type == "random":
        return rng.random(n) + 0.05
    elif init_type == "chirp":
        return np.clip(0.5 + 0.4 * np.sin(2 * np.pi * (5 * tt + tt * tt)) + 0.1 * rng.standard_normal(n), 0.01, None)
    elif init_type == "bell":
        return np.exp(-2.0 * (tt - 0.5) ** 2) + 0.05
    elif init_type == "u_shape":
        return 0.3 + 0.7 * (2 * tt - 1) ** 2 + 0.05 * rng.standard_normal(n)
    elif init_type == "bimodal":
        return np.exp(-5 * (tt - 0.25) ** 2) + np.exp(-5 * (tt - 0.75) ** 2) + 0.05
    elif init_type == "multi_sin":
        return 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.sin(2 * np.pi * 7 * tt) + 0.1 * np.sin(2 * np.pi * 13 * tt)
    elif init_type == "fejer_like":
        i = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (i - center) / center
        x[0] = 0.0
        x[n-1] = 0.0
        # Fejer kernel shape: (sin(m*x/2)/(m*x/2))^2
        m = 10
        xm = m * x / 2.0
        with np.errstate(divide='ignore', invalid='ignore'):
            kernel = np.sinc(xm / np.pi) ** 2  # sinc(x) = sin(pi*x)/(pi*x)
        return np.maximum(kernel, 0.01) + 0.05
    else:
        return np.ones(n)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses mirror descent + Nesterov LSE + spectral shaping + coordinate descent.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Determine length
    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d, budget=%.0fs" % (n, S, N, budget_s))

    # --- Phase 0: Get initial candidates ---
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 1:
            inits.append(("prev_best", np.clip(given, 0, 1000).astype(np.float64)))
            print("  Using height_sequence_1 as initial (score should be ~1.507)")
    except Exception:
        pass

    # Evaluate the initial candidates
    for name, arr in inits:
        a_norm = safe_normalize(arr, n)
        c = fft_convolve_self(a_norm)
        score = 2.0 * n * c.max() / (a_norm.sum() ** 2)
        print("  Initial [%s]: score = %.6f" % (name, score))
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()

    # If no good initial, create some
    if global_a is None:
        for itype in ["uniform", "chirp", "bell", "bimodal", "fejer_like"]:
            arr = make_diverse_init(n, rng, itype)
            a_norm = safe_normalize(arr, n)
            c = fft_convolve_self(a_norm)
            score = 2.0 * n * c.max() / (a_norm.sum() ** 2)
            if score < global_best:
                global_best = score
                global_a = a_norm.copy()
