def coordinate_swap_polish(a, n, S, max_time, rng):
    """Polish using coordinate swaps that reduce the max convolution."""
    a = normalize_sum(a, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    
    c = fft_convolve_self(a)
    best = c.max()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best - 1e-12:
            best = cur
            best_a = a.copy()

        # Gradient of C[k*] with respect to a[i] is 2*a[k*-i] for valid i
        # We want to decrease a[i] where a[k*-i] is large, and increase a[j] where a[k*-j] is small
        
        # Find the best pair to swap
        # Decreasing a[i] by delta changes C[k] by -2*delta*a[k-i] for the k* lag
        # But it also affects other lags
        
        # Simple approach: find i with largest a[i]*a[k*-i] and j with smallest a[j]*a[k*-j]
        # Move some mass from i to j
        
        valid_i = np.arange(max(0, k_star - n + 1), min(k_star, n - 1) + 1)
        products = a[valid_i] * a[k_star - valid_i]
        # Weight by how much this coordinate contributes to the max lag
        
        # Find top contributor
        i_top = valid_i[np.argmax(products)]
        # Find a coordinate that minimally affects C[k*]
        # a[j] * a[k*-j] should be small
        products_all = np.zeros(n)
        for j in range(n):
            ki = k_star - j
            if 0 <= ki < n:
                products_all[j] = a[j] * a[ki]
            else:
                products_all[j] = 0.0
        
        j_bot = np.argmin(products_all)
        
        # Move a small amount from i_top to j_bot
        delta = 0.01 * a[i_top]
        a[i_top] -= delta
        a[j_bot] += delta
        a = np.maximum(a, 0)
        # Re-normalize
        s = a.sum()
        if s > 0:
            a = a * (S / s)
    
    return best_a, best

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-resolution FISTA with high-beta LSE, exact subgradient
refinement, coordinate-swap local search, and adaptive sequence length.
Key improvements over previous: starts from best known, uses much higher
peak beta, multi-resolution warm-start, and a dedicated local search phase
that targets the residual max lag directly.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_value_and_grad(a, c, beta):
    """LSE value and gradient of the smoothed max."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    lse_val = cm + np.log(tot) / beta
    if tot < 1e-300:
        return lse_val, np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return lse_val, grad


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def topk_subgradient(a, c, k_count):
    """Subgradient using top-k lags with equal weight."""
    n = len(a)
    k_count = min(k_count, len(c))
    idx = np.argsort(c)[::-1][:k_count]
    grad = np.zeros(n)
    for k in idx:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        ii = np.arange(lo, hi + 1)
        grad[ii] += 2.0 * a[k - ii]
    grad /= k_count
    return grad


def run_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_rate=0.006,
              beta_max=3000.0, switch_to_exact_at=None):
    """FISTA with LSE then exact subgradient. Returns (best_a, best_score)."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    use_exact = False

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        if switch_to_exact_at is not None and it >= switch_to_exact_at:
            use_exact = True

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        if use_exact:
            grad = exact_subgradient(y, c)
        else:
            _, grad = lse_value_and_grad(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: for LSE use 1/(2*beta), for exact use Polyak-like
        if use_exact:
            # Polyak step: (f(y) - f*)/||g||^2, approximate f* with best
            excess = cur - best * 0.999
            if excess > 0:
                step = excess / (gnorm * gnorm + 1e-30)
                step = min(step, S / gnorm)
            else:
                step = S / (gnorm * (1.0 + 0.01 * np.log1p(it / 50.0)))
        else:
            step = 1.0 / (2.0 * beta)
            # Additional damping
            step *= 0.9

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

        # Stuck detection: inject noise
        if it - last_improve > 3000:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_k = 1.0
            last_improve = it

        # Periodic small perturbation
        if it % 5000 == 0 and not use_exact:
            decay = np.exp(-it / 30000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_k = 1.0

    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_exact_subgrad(a_init, n, S, max_time, rng, top_k=1):
    """Pure exact subgradient method with adaptive step."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

        c = fft_convolve_self(y)

            best_a = y.copy()
            last_improve = it

        k_star = int(np.argmax(c))
        # For lag k_star, C[k_star] = sum_{i} a[i]*a[k_star - i]
        # dC/d(a[i]) = a[k_star - i] (for valid i) + a[i + (n-1-k_star)] if k_star+i < n... 
        # Actually dC[k]/da[i] = a[k-i] + a[i-(k-n+1)]... let me just use the direct formula
        # C[k] = sum_j a[j]*a[k-j], dC[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-(k-n+1)]... 
        
        # Simpler: dC[k]/da[i] = a[k-i] for valid (k-i) in [0,n-1]
        # But also C[k] includes pairs where the "other" index is i: a[j]*a[k-j]
        # dC[k]/da[i] = a[k-i] (when k-i is valid) -- this is the partial derivative
        
        # The gradient of C[k_star] w.r.t. a is:
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        idx = np.arange(lo, hi + 1)
        grad_star = np.zeros(n)
        grad_star[idx] = a[k_star - idx]
        
        # Find coordinate to decrease: high a[i] * grad_star[i] (big contribution to max lag)
        # Find coordinate to increase: low a[i] * grad_star[i] (small contribution)
        contrib = a * grad_star
        
        # Also consider: increasing a[j] increases C[k] for other lags too
        # For now, just do a simple swap
        i_dec = np.argmax(contrib)
        i_inc = np.argmin(contrib)
        
        if i_dec == i_inc:
            break
        
        # Try multiple swap amounts
        a_try = a.copy()
        found_improvement = False
        for frac in [0.05, 0.02, 0.01, 0.005]:
            delta = frac * a[i_dec]
            if delta < 1e-15:
                continue
            a_try = a.copy()
            a_try[i_dec] -= delta
            a_try[i_inc] += delta
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try * (S / s)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < cur:
                a = a_try
                found_improvement = True
                break
        
        if not found_improvement:
            # Try random perturbation
            if rng.random() < 0.3:
                n_perturb = min(10, n // 10)
                idx_perturb = rng.choice(n, size=n_perturb, replace=False)
                a_try = a.copy()
                noise = rng.normal(0, 0.01 * S / np.sqrt(n), n_perturb)
                a_try[idx_perturb] += noise
                a_try = np.maximum(a_try, 0)
                a_try = project_simplex(a_try, S)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < cur:
                    a = a_try
                else:
                    # Stuck, try larger perturbation
                    if rng.random() < 0.1:
                        a_try = a + rng.normal(0, 0.005 * S / np.sqrt(n), n)
                        a_try = project_simplex(a_try, S)
                        a = a_try
                    else:
                        break

    return best_a, best


def downsample_upsample(a, n_target):
    """Downsample by averaging, then upsample by interpolation."""
    n = len(a)
    if n_target >= n:
        # Upsample by interpolation
        t_old = np.linspace(0, 1, n)
        t_new = np.linspace(0, 1, n_target)
        return np.interp(t_new, t_old, a)
    else:
        nblocks = n // n_target
        a_small = a[:nblocks * n_target].reshape(n_target, nblocks).mean(axis=1)
        t_old = np.linspace(0, 1, n_target)
        t_new = np.linspace(0, 1, n)
        return np.interp(t_new, t_old, a_small)


def generate_good_init(n, rng):
    """Generate a reasonable initialization."""
    tt = np.linspace(0, 1, n, endpoint=False)
    # Raised cosine
    f = 0.5 * (1 + np.cos(2 * np.pi * tt))
    f = np.maximum(f, 0.01)
    return f


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Try to get the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose main length
    if given is not None and 1024 <= len(given) <= 16384:
        main_len = len(given)
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * n)

    print("Starting optimization with n=%d, budget=%.0fs" % (n, budget_s))

    # Build list of starting points
    starts = []
    if given is not None and len(given) == n:
        starts.append(np.clip(given, 0, 1000))
    starts.append(generate_good_init(n, rng))

    # Multi-resolution start: downsample given, then handle at full res
    if given is not None and n > 2048:
        n_coarse = max(256, n // 8)
        coarse = downsample_upsample(given, n_coarse)
        coarse = downsample_upsample(coarse, n)  # back to n
        starts.append(coarse)

    # Time allocation
    total_time = max(60.0, deadline - time.monotonic())
    
    # Phase 1: Multi-resolution optimization (if starting from scratch or coarse)
    # Phase 2: Main FISTA optimization
    # Phase 3: Exact subgradient refinement
    # Phase 4: Local swap search
    # Phase 5: Try alternative length

    # Determine if we need multi-resolution
    if given is not None and len(given) == n:
        # We have a good starting point, skip multi-res
        multi_res_time = 0.0
    else:
        multi_res_time = total_time * 0.15
        print("=== Multi-resolution warm start (%.0fs) ===" % multi_res_time)
        
        # Optimize at coarse resolution
        n_res = [n // 16, n // 4, n // 2, n]
        n_res = [x for x in n_res if x >= 64]
        res_time = multi_res_time / len(n_res)
        current_a = starts[-1]
        
        for nr in n_res:
            S_r = np.sqrt(2.0 * nr)
            a_r = downsample_upsample(current_a, nr) if len(current_a) != nr else current_a.copy()
            a_r = np.clip(a_r, 0, 1000)
            a_r, _ = run_fista(a_r, nr, S_r, res_time, rng,
                               beta_init=5.0, beta_rate=0.005, beta_max=500.0)
            current_a = downsample_upsample(a_r, nr if len(a_r) == nr else n)
            c_chk = fft_convolve_self(normalize_sum(current_a, n))
            if c_chk.max() < global_best:
                global_best = c_chk.max()
                global_a = normalize_sum(current_a, n).copy()
            print("  Multi-res n=%d: score=%.6f" % (nr, c_chk.max()))

    # Phase 2: Main FISTA (use best starting point)
    remaining = deadline - time.monotonic()
    fista_time = remaining * 0.45
    print("=== Phase 2: Main FISTA (%.0fs) ===" % fista_time)

    best_start = global_a if global_a is not None else starts[0]
    best_start = np.clip(best_start, 0, 1000)

    # Run FISTA with high beta
    a_opt, score = run_fista(best_start, n, S, fista_time, rng,
                              beta_init=10.0, beta_rate=0.004, beta_max=5000.0)
    a_opt = normalize_sum(a_opt, n)
    c_chk = fft_convolve_self(a_opt)
    if c_chk.max() < global_best:
        global_best = c_chk.max()
        global_a = a_opt.copy()
    print("  After FISTA: %.6f" % c_chk.max())

    # Phase 3: Exact subgradient refinement
    remaining = deadline - time.monotonic()
    exact_time = min(remaining * 0.25, 300.0)
    print("=== Phase 3: Exact subgradient (%.0fs) ===" % exact_time)

    if exact_time > 10:
        a_exact, s_exact = run_exact_subgrad(global_a, n, S, exact_time, rng, top_k=1)
        c_ex = fft_convolve_self(a_exact)
        if c_ex.max() < global_best:
            global_best = c_ex.max()
            global_a = a_exact.copy()
        print("  After exact subgrad: %.6f" % c_ex.max())

    # Phase 3b: Top-k subgradient
    remaining = deadline - time.monotonic()
    topk_time = min(remaining * 0.15, 150.0)
    if topk_time > 10:
        a_topk, s_topk = run_exact_subgrad(global_a, n, S, topk_time, rng, top_k=3)
        c_tk = fft_convolve_self(a_topk)
        if c_tk.max() < global_best:
            global_best = c_tk.max()
            global_a = a_topk.copy()
        print("  After top-3 subgrad: %.6f" % c_tk.max())

    # Phase 4: Local swap search
    remaining = deadline - time.monotonic()
    swap_time = min(remaining * 0.3, 200.0)
    print("=== Phase 4: Local swap search (%.0fs) ===" % swap_time)

    if swap_time > 10:
        a_swap, s_swap = local_swap_search(global_a, n, S, swap_time, rng)
        c_sw = fft_convolve_self(a_swap)
        if c_sw.max() < global_best:
            global_best = c_sw.max()
            global_a = a_swap.copy()
        print("  After local swap: %.6f" % c_sw.max())

    # Phase 5: If time remains, try a slightly different length
    remaining = deadline - time.monotonic()
    if remaining > 120:
        # Try n*1.25 or n*0.75
        alts = []
        n_alt1 = int(n * 1.25 / 128) * 128
        n_alt2 = int(n * 0.75 / 128) * 128
        for na in [n_alt1, n_alt2]:
            if 1024 <= na <= 32768 and na != n:
                alts.append(na)
        
        for na in alts[:2]:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break
            Sa = np.sqrt(2.0 * na)
            # Downsample/upsample global_a to new length
            a_na = downsample_upsample(global_a, na)
            a_na = np.clip(a_na, 0, 1000)
            alt_time = min(remaining * 0.3, 120.0)
            
            a_alt, s_alt = run_fista(a_na, na, Sa, alt_time, rng,
                                      beta_init=50.0, beta_rate=0.003, beta_max=3000.0)
            a_alt = normalize_sum(a_alt, na)
            c_alt = fft_convolve_self(a_alt)
            
            # Convert to objective value
            sum_alt = a_alt.sum()
            obj_alt = 2.0 * na * c_alt.max() / (sum_alt ** 2)
            
            # Compare with global (need to compute global's objective too)
            sum_g = global_a.sum()
            obj_g = 2.0 * len(global_a) * global_best / (sum_g ** 2)
            
            if obj_alt < obj_g:
                global_best = c_alt.max()
                global_a = a_alt.copy()
                print("  Alt n=%d: obj=%.6f (beat %.6f)" % (na, obj_alt, obj_g))
            else:
                print("  Alt n=%d: obj=%.6f (not better than %.6f)" % (na, obj_alt, obj_g))

    # Final round of FISTA with very high beta on the best
    remaining = deadline - time.monotonic()
    if remaining > 30 and global_a is not None:
        n_f = len(global_a)
        S_f = np.sqrt(2.0 * n_f)
        final_time = remaining * 0.8
        print("=== Final high-beta FISTA (%.0fs) ===" % final_time)
        a_fin, s_fin = run_fista(global_a, n_f, S_f, final_time, rng,
                                  beta_init=200.0, beta_rate=0.002, beta_max=10000.0)
        a_fin = normalize_sum(a_fin, n_f)
        c_fin = fft_convolve_self(a_fin)
        sum_f = a_fin.sum()
        obj_fin = 2.0 * n_f * c_fin.max() / (sum_f ** 2)
        sum_g = global_a.sum()
        obj_g = 2.0 * len(global_a) * global_best / (sum_g ** 2)
        if obj_fin < obj_g:
            global_best = c_fin.max()
            global_a = a_fin.copy()
        print("  After final FISTA: obj=%.6f" % min(obj_fin, obj_g))

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    # Compute final objective for reporting
    c_final = fft_convolve_self(global_a)
    obj_final = 2.0 * len(global_a) * c_final.max() / (global_a.sum() ** 2)
    print("Final: n=%d, objective=%.6f" % (len(global_a), obj_final))
    return [float(x) for x in global_a]
