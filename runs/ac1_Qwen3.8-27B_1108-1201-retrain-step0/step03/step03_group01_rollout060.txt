"""
Minimize 2*n*max_conv/sum^2 using FISTA-LSE with continuation, adaptive spectral shaping,
and multi-start diverse optimization. Key improvements over previous: (1) continuation method
on LSE beta for smoother convergence, (2) power-flattening spectral shaping, (3) adaptive
step-size with backtracking, (4) coordinate-descent polish, (5) multi-length exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_lse_gradient(a, c, beta, L):
    """Compute gradient of LSE approximation using FFT cross-correlation."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total  # weights, length 2n-1

    a_pad = np.zeros(L)
    a_pad[:n] = a
    w_pad = np.zeros(L)
    w_pad[:len(w)] = w

    A = fft(a_pad)
    W = fft(w_pad)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad


def fista_lse_continuation(a_init, n, S, beta_levels, iters_per_level, deadline, rng):
    """
    FISTA with LSE smooth approximation using continuation on beta.
    Returns best (a, max_c) found.
    """
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    N = 1 << (3 * n - 2).bit_length()
    best_a = a.copy()
    best_score = np.inf

    for beta in beta_levels:
        if time.monotonic() > deadline:
            break

        # Local FISTA loop at this beta
        x = a.copy()
        y = a.copy()
        t = 1.0
        lr = 1.0 / (beta * 2.5)

        for it in range(iters_per_level):
            if time.monotonic() > deadline:
                break

            c = fft_convolve_self(y)
            cur = c.max()
            if cur < best_score:
                best_score = cur
                best_a = y.copy()

            grad = compute_lse_gradient(y, c, beta, N)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break

            # Gradient step with projection
            x_new = project_simplex(y - lr * grad, S)

            # Nesterov momentum update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = x_new + ((t - 1.0) / t_new) * (x_new - x)
            y = np.maximum(y, 0.0)
            x = x_new
            t = t_new

            # Occasionally restart momentum to avoid oscillation
            if it % 500 == 250 and rng.random() < 0.2:
                y = x.copy()
                t = 1.0

        a = best_a.copy()

        # After each beta level, do a few spectral shaping steps
        if time.monotonic() < deadline - 5:
            a = _spectral_flat_step(a, n, S, N, alpha=0.15, n_iters=50, rng=rng)
            c_check = fft_convolve_self(a)
            if c_check.max() < best_score:
                best_score = c_check.max()
                best_a = a.copy()

    return best_a, best_score


def _spectral_flat_step(a, n, S, N, alpha, n_iters, rng):
    """Spectral shaping: flatten |A|^2 to reduce peak autocorrelation."""
    a = a.copy()
    for it in range(n_iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2

        c = np.real(ifft(A * A))[:2 * n - 1]
        c_max = c.max()
        c_mean = c.sum() / len(c)

        if c_max <= c_mean * 1.01:
            break

        # Compute flattening direction: reduce where |A|^2 is high, increase where low
        P_mean = P.mean()
        if P_mean < 1e-30:
            break

        # Multiplicative flattening
        ratio = P / (P_mean + 1e-30)
        log_ratio = np.log(ratio)
        # Damp the high-frequency content
        P_new = P * np.exp(-alpha * log_ratio)

        # Ensure we don't change total power too much
        P_new = P_new * (P.sum() / (P_new.sum() + 1e-30))

        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(P_new, 0.0)) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)

    return a


def spectral_shaping_toppk(a, n, S, N, top_k, alpha, n_iters, rng, target=1.50):
    """
    Spectral shaping targeting top-k lags of the convolution.
    Modifies |FFT(a)|^2 to reduce power at frequencies contributing to peak lags.
    """
    a = a.copy()
    best_a = a.copy()
    best_score = np.inf

    for it in range(n_iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(A * A))[:2 * n - 1]
        c_max = c.max()

        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        if c_max <= target:
            break

        # Top-k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # Effective cosine
        m = np.arange(N)
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

        # Adaptive alpha
        excess = max(0.0, c_max - target) / max(c_max, 1e-10)
        alpha_eff = alpha * (0.2 + 0.8 * excess)

        P_new = np.maximum(P * (1.0 - alpha_eff * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)

        # Occasional noise perturbation
        if it % 200 == 100 and rng.random() < 0.25:
            a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)

    return best_a, best_score


def coordinate_descent_polish(a, n, S, max_time, rng, n_coords=200):

    Coordinate descent: randomly pick elements, adjust them to reduce max(c).

    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    start = time.monotonic()

    # Precompute the "influence" vectors for each coordinate
    # influence_i[k] = d(c[k])/d(a[i])
    # For efficiency, compute on the fly

    iterations = 0
    while time.monotonic() - start < max_time:
        iterations += 1
        i = rng.integers(0, n)
        old_val = a[i]
        if old_val < 1e-10:

        # The sum of other elements is S - old_val
        # If we set a[i] = v, others scale by (S-v)/(S-old_val)

        # Optimal v is roughly where the peak is minimized
        # Try: v = old_val * (1 - delta) for a few deltas
        deltas = [0.01, 0.02, 0.05, 0.1, 0.2, 0.0, -0.01, -0.05]
        best_v = old_val
        best_cmax = np.inf

        c_current = fft_convolve_self(a)

        for delta in deltas:
            v = old_val * (1.0 + delta)
            if v < 0 or v > 1000:
                continue
            scale = (S - v) / (S - old_val) if abs(S - old_val) > 1e-15 else 1.0
            if scale < 0:
                continue

            a_new = a * scale
            a_new[i] = v

            # Compute new c efficiently: c_new = c_old * scale^2 + adjustments
            # Actually just compute directly for simplicity (n=8192 is fast enough)
            c_new = fft_convolve_self(a_new)
            cm = c_new.max()
            if cm < best_cmax:
                best_cmax = cm
                best_v = v

        if best_v != old_val:
            scale = (S - best_v) / (S - old_val) if abs(S - old_val) > 1e-15 else 1.0
            a = a * scale
            a[i] = best_v
            a = np.maximum(a, 0.0)

            c_new = fft_convolve_self(a)
            if c_new.max() < best_score:
                best_score = c_new.max()
                best_a = a.copy()

    return best_a, best_score


def make_diverse_init(n, rng, init_type):
    """Create diverse initialization patterns."""
    t = np.linspace(0, 1, n, endpoint=False)

    if init_type == "uniform":
        return np.ones(n)
    elif init_type == "random":
        return rng.random(n) + 0.01
    elif init_type == "chirp":
        return 0.5 + 0.4 * np.abs(np.sin(2 * np.pi * (7 * t + 0.3 * t * t))) + 0.05 * rng.standard_normal(n)
    elif init_type == "sin_multi":
        return np.abs(0.8 + 0.3 * np.sin(2 * np.pi * 5 * t) + 0.2 * np.sin(2 * np.pi * 13 * t) + 0.1 * np.sin(2 * np.pi * 29 * t)) + 0.01
    elif init_type == "gaussian_bumps":
        centers = rng.random(5) * n
        amplitudes = rng.random(5) + 0.5
        sigmas = rng.random(5) * n * 0.1 + n * 0.02
        a = np.zeros(n)
        for c, amp, sig in zip(centers, amplitudes, sigmas):
            a += amp * np.exp(-0.5 * ((np.arange(n) - c) / sig) ** 2)
        return a + 0.01
    elif init_type == "alternating":
        a = np.ones(n)
        a[::2] = 0.3
        a += 0.05 * rng.standard_normal(n)
        return np.abs(a) + 0.01
    elif init_type == "ramped":
        a = np.linspace(0.1, 2.0, n)
        a += 0.05 * rng.standard_normal(n)
        return np.abs(a) + 0.01
    elif init_type == "fourier_low":
        # Low-frequency Fourier series (smooth)
        coeffs = rng.standard_normal(10) * np.arange(1, 11)**(-0.5)
        a = np.ones(n)
        for k, ck in enumerate(coeffs, 1):
            a += ck * np.cos(2 * np.pi * k * t)
        return np.abs(a) + 0.01
    elif init_type == "mseq_like":
        # M-sequence inspired: balanced binary-like
        a = (rng.random(n) > 0.5).astype(float) * 0.8 + 0.2
        # Smooth slightly
        kernel = np.ones(3) / 3.0
        a = np.convolve(a, kernel, mode='same')
        return a + 0.01
    else:
        return np.ones(n)


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def fista_single_beta(a_init, n, S, beta, max_iters, deadline, rng, lr_scale=1.0):
    """FISTA with single fixed beta. Returns (best_a, best_max_c)."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    N = 1 << (3 * n - 2).bit_length()
    x = a.copy()
    y = a.copy()
    t = 1.0
    lr = lr_scale / (beta * 2.0)
    best_a = a.copy()
    best_score = np.inf

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()

        grad = compute_lse_gradient(y, c, beta, N)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        x_new = project_simplex(y - lr * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0.0)
        x = x_new
        t = t_new

        # Adaptive: restart momentum periodically
        if it % 400 == 200:
            y = x.copy()
            t = 1.0

        # Very occasional noise
        if it % 2000 == 1000 and rng.random() < 0.15:
            y = project_simplex(y + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses FISTA-LSE continuation + spectral shaping + coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Determine main length
    n = 8192
    S = np.sqrt(2.0 * float(n))
    N_fft = 1 << (2 * n - 1).bit_length()  # For convolution
    N_grad = 1 << (3 * n - 2).bit_length()  # For gradient cross-correlation

    print("n=%d, S=%.4f, N_fft=%d, N_grad=%d" % (n, S, N_fft, N_grad))

    # Try to use given sequence as starting point
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", safe_normalize(np.clip(given, 0, 1000), n)))
            print("Loaded given sequence of length %d" % n)
    except Exception:
        pass

    # Add diverse initializations
    init_types = ["random", "chirp", "sin_multi", "gaussian_bumps", "alternating",
                  "fourier_low", "mseq_like", "ramped"]
    for itype in init_types:
        arr = make_diverse_init(n, rng, itype)
        inits.append((itype, safe_normalize(arr, n)))

    # Add a constant init
    inits.append(("uniform", np.ones(n) * S / n))

    print("=== Phase 1: Multi-start exploration (%d inits) ===" % len(inits))
    t_phase1_start = time.monotonic()
    phase1_budget = (deadline - t_phase1_start) * 0.35
    per_init_time = phase1_budget / len(inits)

    for name, a_init in inits:
        if time.monotonic() > deadline - 60:
            break

        t0 = time.monotonic()
        score_init = fft_convolve_self(a_init).max()
        if score_init < global_best:
            global_best = score_init
            global_a = a_init.copy()

        # Quick FISTA with moderate beta
        rem = per_init_time - (time.monotonic() - t0) - 2
        if rem > 1:
            a_f, sc_f = fista_single_beta(a_init, n, S, beta=20.0,
                                           max_iters=int(rem / 0.003),
                                           deadline=t0 + per_init_time, rng=rng)
            if sc_f < global_best:
                global_best = sc_f
                global_a = a_f.copy()
                print("  [%s] FISTA(beta=20): %.6f" % (name, global_best))

    print("=== Phase 1 done. Best: %.6f ===" % global_best)

    # --- Phase 2: Continuation optimization on best candidate ---
    print("=== Phase 2: Continuation FISTA ===")
    t_phase2_start = time.monotonic()
    phase2_budget = (deadline - t_phase2_start) * 0.45

    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = 2.0

    a_curr = global_a.copy()

    # Continuation: increase beta geometrically
    beta_schedule = [20, 50, 100, 200, 400, 800]
    iters_per = int(phase2_budget / 0.004)  # estimate iterations per level
    iters_per = max(200, iters_per // len(beta_schedule))

    for beta in beta_schedule:
        if time.monotonic() > deadline - 30:
            break

        # FISTA at this beta
        rem = min(60.0, (deadline - 30.0) - time.monotonic())
        n_iters = max(100, int(rem / 0.003))
        a_f, sc_f = fista_single_beta(a_curr, n, S, beta=beta,
                                       max_iters=n_iters,
                                       deadline=time.monotonic() + rem, rng=rng)
        if sc_f < global_best:
            global_best = sc_f
            global_a = a_f.copy()
            a_curr = a_f.copy()
            print("  FISTA beta=%d: %.6f" % (beta, global_best))
        else:
            # Try spectral shaping instead
            rem2 = min(20.0, (deadline - 30.0) - time.monotonic())
            if rem2 > 2:
                a_s, sc_s = spectral_shaping_toppk(a_curr, n, S, N_fft,
                                                    top_k=20, alpha=0.2,
                                                    n_iters=int(rem2 / 0.002),
                                                    rng=rng, target=global_best * 0.999)
                if sc_s < global_best:
                    global_best = sc_s
                    global_a = a_s.copy()
                    a_curr = a_s.copy()
                    print("  SS toppk: %.6f" % global_best)
                # Also try power flattening
                rem3 = min(10.0, (deadline - 30.0) - time.monotonic())
                if rem3 > 2:
                    a_p = _spectral_flat_step(a_curr, n, S, N_fft,
                                               alpha=0.1, n_iters=int(rem3 / 0.002),
                                               rng=rng)
                    sc_p = fft_convolve_self(a_p).max()
                    if sc_p < global_best:
                        global_best = sc_p
                        global_a = a_p.copy()
                        a_curr = a_p.copy()
                        print("  SS flat: %.6f" % global_best)

    print("=== Phase 2 done. Best: %.6f ===" % global_best)

    # --- Phase 3: Focused polish ---
    print("=== Phase 3: Polish ===")
    t_phase3_start = time.monotonic()
    remaining3 = deadline - t_phase3_start

    if global_a is not None and remaining3 > 3:
        a_curr = global_a.copy()

        # 3a: High-beta FISTA
        rem_a = min(remaining3 * 0.3, 120.0)
        if rem_a > 3:
            a_h, sc_h = fista_single_beta(a_curr, n, S, beta=1000.0,
                                           max_iters=int(rem_a / 0.003),
                                           deadline=time.monotonic() + rem_a,
                                           rng=rng, lr_scale=0.5)
            if sc_h < global_best:
                global_best = sc_h
                global_a = a_h.copy()
                a_curr = a_h.copy()
                print("  High-beta FISTA: %.6f" % global_best)

        # 3b: Spectral shaping (fine-tune)
        rem_b = min((deadline - time.monotonic()) * 0.3, 60.0)
        if rem_b > 2:
            a_s, sc_s = spectral_shaping_toppk(a_curr, n, S, N_fft,
                                                top_k=50, alpha=0.05,
                                                n_iters=int(rem_b / 0.002),
                                                rng=rng, target=global_best * 0.998)
            if sc_s < global_best:
                global_best = sc_s
                global_a = a_s.copy()
                a_curr = a_s.copy()
                print("  Fine SS: %.6f" % global_best)

        # 3c: Power flattening
        rem_c = min((deadline - time.monotonic()) * 0.2, 30.0)
        if rem_c > 2:
            a_p = _spectral_flat_step(a_curr, n, S, N_fft,
                                       alpha=0.05, n_iters=int(rem_c / 0.002),
                                       rng=rng)
            sc_p = fft_convolve_self(a_p).max()
            if sc_p < global_best:
                global_best = sc_p
                global_a = a_p.copy()
                a_curr = a_p.copy()
                print("  Fine flatten: %.6f" % global_best)

        # 3d: Coordinate descent polish
        rem_d = deadline - time.monotonic()
        if rem_d > 5:
            a_cd, sc_cd = coordinate_descent_polish(a_curr, n, S,
                                                     max_time=min(rem_d * 0.5, 60.0),
                                                     rng=rng)
            if sc_cd < global_best:
                global_best = sc_cd
                global_a = a_cd.copy()
                a_curr = a_cd.copy()
                print("  Coord descent: %.6f" % global_best)

            # 3e: Final FISTA with very high beta
            rem_e = deadline - time.monotonic()
            if rem_e > 3:
                a_f, sc_f = fista_single_beta(a_curr, n, S, beta=2000.0,
                                               max_iters=int(rem_e / 0.004),
                                               deadline=deadline, rng=rng, lr_scale=0.3)
                if sc_f < global_best:
                    global_best = sc_f
                    global_a = a_f.copy()
                    print("  Final FISTA: %.6f" % global_best)

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
