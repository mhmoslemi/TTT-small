    c = np.real(ifft(A * A))

def adam_descent(a0, time_budget, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                 temp_init=5.0, temp_final=0.3, temp_decay=0.9999,
                 stagnation_patience=5000, perturb_strength=0.15,
                 seed=42, verbose=False):
    """Adam-based subgradient descent with adaptive softmax."""
    rng = np.random.default_rng(seed)
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 1e-10)
    S = a.sum()
    a = a * (S / a.sum())
    
    c = fft_conv_self(a)
    best_obj = compute_obj(a, c)
    best_a = a.copy()
    best_iter = 0
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t += 1
        grad = softmax_subgrad(c, a, temp)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Multiplicative (log-space) update
        a = a * np.exp(-lr * m_hat / (np.sqrt(v_hat) + eps))
        a = np.maximum(a, 1e-10)
        a = a * (S / a.sum())
        
        c = fft_conv_self(a)
        cur_obj = compute_obj(a, c)
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = t
        elif t - best_iter > stagnation_patience:
            # Restart with perturbation
            noise = rng.standard_normal(n) * perturb_strength
            a = a * (1.0 + noise)
            a = np.maximum(a, 1e-10)
            a = a * (S / a.sum())
            c = fft_conv_self(a)
            m = np.zeros(n)
            v = np.zeros(n)
            temp = temp_init  # Reset temperature
            best_iter = t
            if verbose:
                print(f"        [restart] t={t}, obj={cur_obj:.6f}, best={best_obj:.6f}")
        
        temp *= temp_decay
        
        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - start
            print(f"        t={t}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"temp={temp:.3f}, lr_eff={lr:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def init_cosine(n, freq, amp, seed):
    rng = np.random.default_rng(seed)
    i = np.arange(n)
    phase = rng.random() * 2 * np.pi
    a = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.01)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a


def init_random(n, seed):
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.01
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a


def init_bimodal(n, seed):
    """Two Gaussian bumps."""
    rng = np.random.default_rng(seed)
    i = np.arange(n)
    c1 = rng.random() * 0.4 + 0.3
    c2 = 1.0 - c1
    w = 0.05 + 0.05 * rng.random()
    a = np.exp(-0.5 * ((i - c1 * n) / (w * n)) ** 2) + \
        np.exp(-0.5 * ((i - c2 * n) / (w * n)) ** 2)
    a = np.maximum(a, 0.01)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using Adam subgradient descent with
adaptive softmax peak weighting, diverse multi-start initialization,
structured perturbations, and multi-phase optimization.
Key differences from prior approach: log-space Adam updates, adaptive 
temperature annealing, structured (not just Gaussian) restarts, and
a final ultra-low-temperature peak-chasing phase.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_obj(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_conv_self(a)
    return 2.0 * n * c.max() / (S * S)


def softmax_subgrad(c, a, temp=1.0):
    """
    Compute smoothed subgradient of max(c) w.r.t. a using softmax weights.
    grad[i] = 2 * sum_m w[m] * a[m-i] for valid m-i in [0, n-1].
    """
    n = len(a)
    c_max = c.max()
    log_w = (c - c_max) / temp
    log_w = log_w - log_w.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-300:
        idx = np.argmax(c)
        w = np.zeros(len(c))
        w[idx] = 1.0
    else:
        w = w / w_sum

    a_rev = a[::-1]
    out_len = len(w) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    conv = np.real(ifft(W * R))
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def topk_subgrad(c, a, top_k=5):
    """Compute subgradient using uniform weight over top-k peaks."""
    n = len(a)
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / top_k

    a_rev = a[::-1]
    out_len = len(w) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    conv = np.real(ifft(W * R))
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def adam_logspace(a0, time_budget, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                  temp_init=5.0, temp_decay=0.99995, temp_min=0.1,
                  stagnation_patience=4000, perturb_strength=0.15,
                  use_topk=False, top_k=10, seed=42, verbose=False,
                  lr_decay=0.99999, oscillation_scale=0.1):
    """
    Adam optimizer in log-space with adaptive softmax subgradient.
    """
    rng = np.random.default_rng(seed)
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 1e-10)
    S = a.sum()
    a = a * (S / a.sum())

    c = fft_conv_self(a)
    best_obj = compute_obj(a, c)
    best_a = a.copy()
    best_iter = 0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temp = temp_init
    cur_lr = lr
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        t += 1

        # Compute gradient
        if use_topk:
            grad = topk_subgrad(c, a, top_k)
        else:
            grad = softmax_subgrad(c, a, temp)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # Log-space multiplicative update
        a = a * np.exp(-cur_lr * m_hat / (np.sqrt(v_hat) + eps))
        a = np.maximum(a, 1e-10)
        a = a * (S / a.sum())

        c = fft_conv_self(a)
        cur_obj = compute_obj(a, c)

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = t
        elif t - best_iter > stagnation_patience:
            # Structured perturbation
            strategy = t % 4
            if strategy == 0:
                # Gaussian noise
                noise = rng.standard_normal(n) * perturb_strength
                a = a * (1.0 + noise)
            elif strategy == 1:
                # Frequency-domain perturbation
                fsize = 1 << (2 * n - 1).bit_length()
                A = fft(a, fsize)
                phases = rng.standard_normal(fsize) * 0.3
                A *= np.exp(1j * phases)
                a = np.real(ifft(A, n))
                a = np.maximum(a, 1e-10)
            elif strategy == 2:
                # Block shuffle
                block = max(1, n // 8)
                perm = [i for i in range(0, n, block)]
                rng.shuffle(perm)
                a_new = np.zeros(n)
                idx = 0
                for start_i in perm:
                    a_new[idx:idx + (n - idx) if start_i == perm[-1] and (start_i + block) > n else start_i + block] = \
                        a[start_i:min(start_i + block, n)]
                    idx += min(block, n - start_i)
                a = a_new[idx:] if idx < n else a_new
                a = np.pad(a, (0, n - len(a))) if len(a) < n else a[:n]
                a = np.maximum(a, 1e-10)
            else:
                # Interpolation with fresh random
                fresh = rng.random(n) + 0.01
                fresh = fresh * (S / fresh.sum())
                a = 0.7 * a + 0.3 * fresh
                a = np.maximum(a, 1e-10)

            a = a * (S / a.sum())
            c = fft_conv_self(a)
            m = np.zeros(n)
            v = np.zeros(n)
            temp = max(temp_init * 0.5, temp_min)
            cur_lr = lr
            best_iter = t
            if verbose:
                print(f"        [restart_{strategy}] t={t}, obj={cur_obj:.6f}, best={best_obj:.6f}")

        # Anneal temperature
        temp = max(temp * temp_decay, temp_min)
        # Oscillate learning rate slightly to escape plateaus
        cur_lr = lr * lr_decay ** (t % 5000) * (1.0 + oscillation_scale * np.sin(0.005 * t))

        if verbose and t % 20000 == 0:
            elapsed = time.monotonic() - start
            print(f"        t={t}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"temp={temp:.3f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def sgd_nesterov(a0, time_budget, lr=0.003, momentum=0.9, temp=1.0,
                 seed=42, verbose=False, stagnation_patience=5000):
    """Nesterov accelerated projected SGD with softmax subgradient."""
    rng = np.random.default_rng(seed)
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())

    c = fft_conv_self(a)
    best_obj = compute_obj(a, c)
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)
    t = 0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        t += 1
        grad = softmax_subgrad(c, a, temp)

        v = momentum * v - lr * grad
        a = a + v * 0.5  # Damped step to stay in feasible region
        a = np.maximum(a, 1e-10)
        a = a * (S / a.sum())

        c = fft_conv_self(a)
        cur_obj = compute_obj(a, c)

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = t
        elif t - best_iter > stagnation_patience:
            noise = rng.standard_normal(n) * 0.1
            a = a * (1.0 + noise)
            a = np.maximum(a, 1e-10)
            a = a * (S / a.sum())
            c = fft_conv_self(a)
            v = np.zeros(n)
            best_iter = t

    return best_a, best_obj


def init_cosine(n, freq, amp, seed):
    """Cosine-modulated initialization."""
    rng = np.random.default_rng(seed)
    i = np.arange(n)
    phase = rng.random() * 2 * np.pi
    a = 1.0 + amp * np.cos(2 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.01)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a


def init_multicosine(n, freqs, amps, seed):
    """Multiple cosine components."""
    rng = np.random.default_rng(seed)
    i = np.arange(n)
    a = np.ones(n)
    for f, amp in zip(freqs, amps):
        phase = rng.random() * 2 * np.pi
        a += amp * np.cos(2 * np.pi * f * i / n + phase)
    a = np.maximum(a, 0.01)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a


def init_random(n, seed):
    """Random initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.01
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a


def init_gaussian_bumps(n, n_bumps, seed):
    """Multiple Gaussian bumps."""
    rng = np.random.default_rng(seed)
    i = np.arange(n)
    centers = sorted(rng.random(n_bumps))
    a = np.zeros(n)
    for c in centers:
        w = 0.03 + 0.04 * rng.random()
        a += np.exp(-0.5 * ((i - c * n) / (w * n)) ** 2)
    a = np.maximum(a, 0.01)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a


def init_flat_spectrum(n, seed):
    """Initialize with flat power spectrum (random phases)."""
    rng = np.random.default_rng(seed)
    fsize = 1 << (2 * n - 1).bit_length()
    phases = 2 * np.pi * rng.random(fsize)
    F = np.exp(1j * phases)
    # Enforce conjugate symmetry
    for i in range(1, fsize // 2 + 1):
        F[fsize - i] = np.conj(F[i])
    a = np.real(ifft(F, n))
    a = np.maximum(a, 0.01)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a
