Minimize 2*n*max_conv/sum^2 using adaptive weighted quadratic optimization.

Key approach:
1. Reformulate as minimizing a weighted sum of autocorrelation lags, where weights

2. The weighted sum is a quadratic form a^T Q_w a where Q_w is a Hankel matrix,

import numpy as np

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a, N):
    """Full self-convolution of a via FFT. N is the FFT size."""
    n = len(a)
    a_padded = np.zeros(N)
    a_padded[:n] = a
    A = fft(a_padded)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]
