    n = len(a)

import numpy as np

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax(theta):
    """Numerically stable softmax."""
    t = theta - theta.max()
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(theta)) / len(theta)
    return e / s


def get_a_from_theta(theta, S):
    """Convert log-parameter to amplitude sequence with sum S."""
    return S * softmax(theta)


def compute_cross_corr(w, a, n):
    """Compute cross-correlation (w cross a)[m] = sum_j w[j+m]*a[j] via FFT."""
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return corr[:n]


def lse_obj_and_grad(theta, T, S):
    """Log-sum-exp smoothed objective and gradient w.r.t. theta."""
    n = len(theta)
    a = get_a_from_theta(theta, S)
    c = fft_convolve_self(a)
    c_max = c.max()
    scaled = T * (c - c_max)
    e = np.exp(scaled)
    e_sum = e.sum()
    obj = c_max + np.log(e_sum) / T
    w = e / e_sum
    g_a = 2.0 * compute_cross_corr(w, a, n)
    dot = np.dot(g_a, a)
    g = g_a * a - (a / S) * dot
    return obj, g


def topk_obj_and_grad(theta, k, S):
    """Sum of top-k autocorrelation values, with gradient w.r.t. theta."""
    n = len(theta)

    else:

    g_a = 2.0 * compute_cross_corr(w, a, n)
    dot = np.dot(g_a, a)
    g = g_a * a - (a / S) * dot
    return obj, g

    Uses L-BFGS on smoothed objectives with temperature curriculum,

    """
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 10.0)
    deadline = time.monotonic() + total_budget

    global_best = np.inf
    global_a = None

    # Load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw.copy()
            n_given = len(given_a)
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given sequence: " + str(e))

    if global_a is None:
        # Fallback: create a reasonable initial sequence
        n = 4096
        global_a = np.ones(n)
        global_best = evaluate_sequence(global_a.tolist())
        print("Fallback uniform: n=%d, obj=%.7f" % (n, global_best))

    n = len(global_a)
    S = np.sqrt(2.0 * n)
    a = global_a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a * (S / a.sum())

    print("\nStarting optimization: n=%d, current obj=%.7f" % (n, global_best))
    print("Budget: %.0f seconds" % total_budget)

    # ============ PHASE 1: Quick exploration of other lengths (10%) ============
    explore_budget = 0.08 * total_budget
    explore_deadline = time.monotonic() + explore_budget

    if n > 1024 and explore_budget > 15:
        print("\nPhase 1: Exploring other lengths (budget=%.0fs)" % explore_budget)
        for test_n in [2048, 8192]:
            if time.monotonic() > explore_deadline - 5:
                break
            test_S = np.sqrt(2.0 * test_n)
            # Initialize with a cosine-mix pattern
            i = np.arange(test_n)
            test_a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * i / test_n + rng.random()) + \
                     0.2 * np.cos(2 * np.pi * 7 * i / test_n + rng.random()) + \
                     0.1 * np.cos(2 * np.pi * 13 * i / test_n + rng.random())
            test_a = np.maximum(test_a, 0.01)
            test_a = test_a * (test_S / test_a.sum())

            # Quick L-BFGS at moderate T
            theta = a_to_theta(test_a, test_S)
            t0 = time.monotonic()
            try:
                theta_new, _ = run_lbfgs_lse(theta, T=50.0, S=test_S, maxfun=2000)
            except Exception:
                theta_new = theta
            test_a = get_a_from_theta(theta_new, test_S)
            test_a = np.clip(test_a, 0, 1000)
            ver = evaluate_sequence(test_a.tolist())
            elapsed = time.monotonic() - t0
            print("  n=%d: obj=%.7f (%.1fs)" % (test_n, ver, elapsed))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = test_a.copy()
                n = test_n
                S = np.sqrt(2.0 * n)

    # ============ PHASE 2: L-BFGS multi-round refinement (55%) ============
    exploit_budget = 0.55 * total_budget
    exploit_deadline = time.monotonic() + exploit_budget

    n = len(global_a)
    S = np.sqrt(2.0 * n)
    a = global_a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a * (S / a.sum())
    theta = a_to_theta(a, S)

    print("\nPhase 2: L-BFGS refinement (budget=%.0fs, n=%d)" % (exploit_budget, n))

    # Temperature curriculum
    T_schedule = [20, 50, 100, 200, 500, 1000, 2000, 5000]
    n_rounds = len(T_schedule)
    per_round = exploit_budget / n_rounds

    for r, T in enumerate(T_schedule):
        if time.monotonic() > exploit_deadline - 5:
            break
        round_budget = min(per_round, exploit_deadline - time.monotonic())
        if round_budget < 3:
            break

        maxfun = max(500, int(round_budget * 300))  # ~300 evals/sec
        t0 = time.monotonic()
        try:
            theta_new, fun_val = run_lbfgs_lse(theta, T=T, S=S, maxfun=maxfun)
        except Exception as e:
            print("    L-BFGS T=%d failed: %s" % (T, str(e)))
            theta_new = theta
            fun_val = np.inf

        a = get_a_from_theta(theta_new, S)
        a = np.clip(a, 0, 1000)
        ver = evaluate_sequence(a.tolist())
        elapsed = time.monotonic() - t0

        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a.copy()
            theta = theta_new.copy()
            print("    T=%5d: obj=%.7f (LSE=%.6f, %.1fs)" % (T, ver, fun_val, elapsed))
        else:
            # Keep the old theta if this round didn't help
            print("    T=%5d: obj=%.7f (no improvement, LSE=%.6f, %.1fs)" % (T, ver, fun_val, elapsed))

    # ============ PHASE 3: Top-k active set optimization (15%) ============
    topk_budget = 0.15 * total_budget
    topk_deadline = time.monotonic() + topk_budget

    n = len(global_a)
    S = np.sqrt(2.0 * n)
    a = global_a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a * (S / a.sum())
    theta = a_to_theta(a, S)

    if topk_budget > 10:
        print("\nPhase 3: Top-k optimization (budget=%.0fs)" % topk_budget)
        c = fft_convolve_self(a)
        c_max = c.max()
        # Determine k based on how many values are near the max
        threshold = c_max * 0.99
        k_near = int(np.sum(c >= threshold))
        k_values = [max(1, k_near), max(3, k_near // 2), 10, 50]
        k_values = sorted(set([max(1, min(k, 2*n-1)) for k in k_values]))

        per_k = topk_budget / len(k_values)
        for k in k_values:
            if time.monotonic() > topk_deadline - 3:
                break
            round_budget = min(per_k, topk_deadline - time.monotonic())
            if round_budget < 2:
                break
            maxfun = max(300, int(round_budget * 200))
            t0 = time.monotonic()
            try:
                theta_new, fun_val = run_lbfgs_topk(theta, k=k, S=S, maxfun=maxfun)
            except Exception as e:
                print("    Top-k=%d failed: %s" % (k, str(e)))
                continue
            a = get_a_from_theta(theta_new, S)
            a = np.clip(a, 0, 1000)
            ver = evaluate_sequence(a.tolist())
            elapsed = time.monotonic() - t0
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a.copy()
                theta = theta_new.copy()
                print("    Top-%d: obj=%.7f (%.1fs)" % (k, ver, elapsed))
            else:
                print("    Top-%d: obj=%.7f (no improvement, %.1fs)" % (k, ver, elapsed))

    # ============ PHASE 4: Coordinate descent polishing (12%) ============
    cd_budget = 0.12 * total_budget
    cd_deadline = time.monotonic() + cd_budget

    n = len(global_a)
    S = np.sqrt(2.0 * n)
    a = global_a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a * (S / a.sum())

    if cd_budget > 5:
        print("\nPhase 4: Coordinate descent (budget=%.0fs)" % cd_budget)
        a_best = coordinate_descent_incremental(a, n, S, cd_budget, verbose=True)
        a_best = np.clip(a_best, 0, 1000)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  CD result: obj=%.7f" % ver)
        else:
            print("  CD result: obj=%.7f (no improvement)" % ver)

    # ============ PHASE 5: Final L-BFGS at very high T + CD (10%) ============
    final_budget = deadline - time.monotonic()
    if final_budget > 10:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = global_a.copy().astype(np.float64)
        a = np.maximum(a, 0)
        a = a * (S / a.sum())
        theta = a_to_theta(a, S)

        print("\nPhase 5: Final push (budget=%.0fs)" % final_budget)

        # Very high T L-BFGS
        t_final = final_budget * 0.5
        maxfun = max(500, int(t_final * 200))
        try:
            theta_new, _ = run_lbfgs_lse(theta, T=10000, S=S, maxfun=maxfun)
            a = get_a_from_theta(theta_new, S)
            a = np.clip(a, 0, 1000)
            ver = evaluate_sequence(a.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a.copy()
                print("  Final L-BFGS: obj=%.7f" % ver)
        except Exception:
            pass

        # Final CD
        remaining = deadline - time.monotonic() - 2
        if remaining > 3:
            a = global_a.copy().astype(np.float64)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 0:
                a = a * (S / s)
            a_best = coordinate_descent_incremental(a, n, S, remaining)
            a_best = np.clip(a_best, 0, 1000)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  Final CD: obj=%.7f" % ver)

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f (verified=%.7f, n=%d)" % (global_best, ver, len(final_a)))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)

    return final_a
