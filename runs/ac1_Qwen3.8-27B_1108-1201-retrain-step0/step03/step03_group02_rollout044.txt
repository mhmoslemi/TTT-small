import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient(a, c, k_list):
    """
    Compute weighted subgradient of max(c) using specified lags.
    For lag k: grad[m] += 2 * a[k-m] (when 0 <= k-m < n)
    """
    n = len(a)
    g = np.zeros(n)
    for k in k_list:
        k = int(k)
        # grad[m] += 2 * a[k - m] for 0 <= k-m < n
        # i.e., m ranges from max(0, k-n+1) to min(k, n-1)
        m_lo = max(0, k - n + 1)
        m_hi = min(k, n - 1)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi + 1)
            src_indices = k - indices
            g[indices] += 2.0 * a[src_indices]
    return g


def compute_subgradient_fft(a, c, top_k, temperature):
    """
    Compute subgradient using top-k lags with softmax weighting.
    temperature controls how sharply we focus on the max.
    """
    n = len(a)
    c_max = c.max()
    # Weight lags by exp(-temperature * (c_max - c[k]))
    if temperature < 1e-10:
        # Exact top-k
        top_indices = np.argsort(c)[::-1][:top_k]
        k_list = top_indices.tolist()
        return compute_subgradient(a, c, k_list)
    else:
        # Soft weighting over top-k
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        exp_w = np.exp(-temperature * (c_max - top_vals))
        weights = exp_w / exp_w.sum()
        
        # Build weighted subgradient via FFT for efficiency
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]
        
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        return g

    c = fft_convolve_self(a)
    return c.max()

            x_new = project_simplex(y - step * g, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur + 1e-14:
                accepted = True
                break
            step *= 0.5
            if step < 1e-12 * S:
                x_new = project_simplex(y - step * g, S)
                accepted = True
                break

        if not accepted:
            step = base_step / (1.0 + 0.01 * it) / max(gnorm / S, 1e-10)
            x_new = project_simplex(y - step * g, S)

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic noise injection (more aggressive when stuck)
        if no_improve > 500 and it % 20 == 0:
            noise_amp = noise_scale * (1.0 + 0.5 * min(no_improve / 1000.0, 3.0))
            noise = rng.normal(0, noise_amp, n)
            y = project_simplex(y + noise, S)
            a = y.copy()
            t_mom = 1.0
            noise_scale *= noise_decay

        # Gradually reduce noise
        noise_scale *= 0.9995

    return best_a, best_score


def run_focused_peaks(a_init, n, S, max_time, rng):
    """
    Focused peak reduction: repeatedly target the single largest peak,
    then move to the next, cycling through them.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    t_mom = 1.0
    y = a.copy()

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = y.copy()

        # Find top peaks and weight them
        top_n = min(16, 2 + it // 200)
        top_indices = np.argsort(c)[::-1][:top_n]
        top_vals = c[top_indices]
        
        # Weight: peaks close to max get more weight
        c_max = c.max()
        if c_max > 0:
            gaps = (c_max - top_vals) / c_max
            weights = np.exp(-20.0 * gaps)
            weights /= weights.sum()
        else:
            weights = np.ones(top_n) / top_n

        # Compute weighted gradient
        g = np.zeros(n)
        for i, k in enumerate(top_indices):
            k = int(k)
            m_lo = max(0, k - n + 1)
            m_hi = min(k, n - 1)
            if m_lo <= m_hi:
                indices = np.arange(m_lo, m_hi + 1)
                src_indices = k - indices
                g[indices] += 2.0 * weights[i] * a[src_indices]

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break

        # Step with mild backtracking
        step = S / (4.0 * np.sqrt(n)) / (1.0 + 0.001 * it)
        x_new = project_simplex(y - step * g, S)
        
        c_check = fft_convolve_self(x_new)
        if c_check.max() > cur * 1.005:
            step *= 0.3
            x_new = project_simplex(y - step * g, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Noise for exploration
        if it % 2000 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0

    return best_a, best_score


def run_softmax_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=400.0):
    """FISTA with log-sum-exp smoothing of max(c)."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = y.copy()

        # Log-sum-exp gradient
        c_shifted = c - c.max()
        exp_bc = np.exp(beta * c_shifted)
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total

        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(y, fsize)
        g = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        step = 1.0 / (beta * 2.0 * (1.0 + 0.001 * it))
        step = max(step, 1e-8 * S / n)

        x_new = project_simplex(y - step * g, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_mom

        if it % 3000 == 0 and rng.random() < 0.15:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0

    return best_a, best_score


def low_res_warm_start(n_target, S_target, max_time, rng):
    """
    Solve at low resolution, then upsample to target resolution.
    """
    # Try a few low-res sizes
    for n_low in [128, 256]:
        if time.monotonic() > max_time:
            break
        S_low = np.sqrt(2.0 * n_low)
        
        # Asymmetric init
        t = np.linspace(0, 1, n_low, endpoint=False)
        a_low = np.exp(2.0 * t) + 0.1
        a_low = a_low * (S_low / a_low.sum())
        
        # Optimize at low res (short time)
        t_budget = max(2.0, max_time * 0.3 / n_low * 128)
        a_opt, score = run_refined_subgradient(a_low, n_low, S_low, min(t_budget, 30), rng,
                                                k_start=2, k_end=8, step_scale=1.0,
                                                noise_start=0.01, noise_decay=0.995)
        
        # Upsample to target resolution
        a_up = np.interp(np.linspace(0, 1, n_target, endpoint=False),
                         np.linspace(0, 1, n_low, endpoint=False),
                         a_opt)
        a_up = a_up * (S_target / a_up.sum())
        
        # Fine-tune at full resolution
        a_fine, score_fine = run_refined_subgradient(a_up, n_target, S_target,
                                                      max_time * 0.7, rng,
                                                      k_start=4, k_end=64,
                                                      step_scale=1.0,
                                                      noise_start=0.002,
                                                      noise_decay=0.999)
        return a_fine, score_fine
    
    return None, np.inf


def try_diverse_inits(n, S, max_time, rng):
    """Try several diverse asymmetric initializations briefly."""
    t = np.linspace(0, 1, n, endpoint=False)
    inits = []
    
    # Various asymmetric shapes
    for alpha in [1.5, 2.5, 4.0, 6.0]:
        inits.append(np.exp(alpha * t) + 0.01)
        inits.append(np.exp(alpha * (1.0 - t)) + 0.01)
    
    # Power law
    inits.append((t + 0.05) ** 2.0)
    inits.append((1.0 - t + 0.05) ** 2.0)
    inits.append((t + 0.05) ** 3.0)
    
    # Sine-modulated ramps
    inits.append(0.5 + t + 0.3 * np.sin(4 * np.pi * t))
    inits.append(0.5 + (1-t) + 0.3 * np.cos(6 * np.pi * t))
    
    # Two-bump asymmetric
    inits.append(np.exp(-50*(t-0.2)**2) + 0.3*np.exp(-30*(t-0.7)**2) + 0.01)
    inits.append(np.exp(-30*(t-0.3)**2) + 0.5*np.exp(-50*(t-0.8)**2) + 0.01)
    
    # Step-like
    step = np.ones(n)
    step[:n//2] = 2.0
    inits.append(step)
    step2 = np.ones(n)
    step2[n//3:] = 2.5
    inits.append(step2)
    
    best_a = None
    best_score = np.inf
    per_init_time = max_time / len(inits)
    
    for arr in inits:
        if time.monotonic() > max_time:
            break
        arr = np.clip(arr, 0, 1000)
        s = arr.sum()
        if s < 1e-10:
            continue
        arr = arr * (S / s)
        
        a_opt, score = run_refined_subgradient(arr, n, S, per_init_time, rng,
                                                k_start=2, k_end=32,
                                                step_scale=1.0, noise_start=0.005,
                                                noise_decay=0.997)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
    
    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses refined subgradient with backtracking, adaptive active set,
    Nesterov acceleration, and diverse exploration strategies.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # --- Get starting sequence ---
    n = 8192
    S = np.sqrt(2.0 * n)
    start_seqs = []
    
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            gf = np.clip(given, 0, 1000)
            # Resample to our target n if different
            if len(gf) != n:
                gf = np.interp(np.linspace(0, 1, n, endpoint=False),
                               np.linspace(0, 1, len(gf), endpoint=False), gf)
            start_seqs.append(gf)
            print("Loaded parent sequence, length=%d" % len(gf))
    except Exception as e:
        print("No parent sequence available: %s" % str(e))

    # Also prepare a few asymmetric fallbacks
    tt = np.linspace(0, 1, n, endpoint=False)
    start_seqs.append(np.exp(2.0 * tt) + 0.01)
    start_seqs.append(np.exp(3.0 * (1.0 - tt)) + 0.01)

    print("=== Starting optimization, n=%d, S=%.4f ===" % (n, S))
    print("Budget: %.1f seconds" % (budget_s - 20.0))

    # --- Phase 1: Explore with diverse inits (25% time) ---
    phase1_time = min(300.0, (deadline - time.monotonic()) * 0.25)
    print("\n=== Phase 1: Diverse exploration (%.1f sec) ===" % phase1_time)
    
    # Try parent first with short refinement
    if start_seqs:
        a0 = safe_normalize(start_seqs[0], n)
        c0 = fft_convolve_self(a0)
        print("  Parent initial max_c = %.6f" % c0.max())
        
        a_p, s_p = run_refined_subgradient(a0, n, S, min(phase1_time * 0.4, 60), rng,
                                            k_start=4, k_end=64, step_scale=1.0,
                                            noise_start=0.003, noise_decay=0.998)
        c_p = fft_convolve_self(safe_normalize(a_p, n))
        if c_p.max() < global_best:
            global_best = c_p.max()
            global_a = safe_normalize(a_p, n).copy()
            print("  Parent refined: max_c = %.6f" % global_best)

    # Try diverse inits with remaining phase 1 time
    remaining_p1 = deadline - time.monotonic() - (budget_s - 20.0) * 0.75
    if remaining_p1 > 10:
        a_div, s_div = try_diverse_inits(n, S, min(remaining_p1, phase1_time * 0.6), rng)
        if a_div is not None:
            a_div = safe_normalize(a_div, n)
            c_div = fft_convolve_self(a_div)
            if c_div.max() < global_best:
                global_best = c_div.max()
                global_a = a_div.copy()
                print("  Diverse init best: max_c = %.6f" % global_best)

    # --- Phase 2: Main optimization (50% time) ---
    phase2_end = deadline - (budget_s - 20.0) * 0.25
    phase2_time = min(phase2_end - time.monotonic(), (budget_s - 20.0) * 0.50)
    print("\n=== Phase 2: Main subgradient optimization (%.1f sec) ===" % phase2_time)
    print("  Current best: %.6f" % global_best)

    if global_a is None:
        global_a = safe_normalize(np.exp(2.0 * tt), n)

    # Run main optimization with multiple configurations
    configs = [
        {"k_start": 2, "k_end": 32, "step_scale": 1.0, "noise_start": 0.002, "noise_decay": 0.999},
        {"k_start": 8, "k_end": 128, "step_scale": 0.8, "noise_start": 0.001, "noise_decay": 0.999},
    ]
    
    for cfg_i, cfg in enumerate(configs):
        remaining = min(phase2_end - time.monotonic(), phase2_time / (len(configs) - cfg_i))
        if remaining < 10:
            break
        
        a_curr = global_a.copy()
        a_opt, score = run_refined_subgradient(a_curr, n, S, remaining, rng, **cfg)
        a_opt = safe_normalize(a_opt, n)
        c_opt = fft_convolve_self(a_opt)
        if c_opt.max() < global_best:
            global_best = c_opt.max()
            global_a = a_opt.copy()
            print("  Config %d: max_c = %.6f" % (cfg_i, global_best))

    # --- Phase 3: Focused peak reduction (15% time) ---
    phase3_time = (deadline - time.monotonic()) * 0.4
    print("\n=== Phase 3: Focused peaks (%.1f sec) ===" % phase3_time)
    print("  Current best: %.6f" % global_best)

    if phase3_time > 10:
        a_p3, s3 = run_focused_peaks(global_a, n, S, min(phase3_time, (budget_s-20)*0.15), rng)
        a_p3 = safe_normalize(a_p3, n)
        c3 = fft_convolve_self(a_p3)
        if c3.max() < global_best:
            global_best = c3.max()
            global_a = a_p3.copy()
            print("  Focused peaks: max_c = %.6f" % global_best)

    # --- Phase 4: Softmax FISTA polish (10% time) ---
    phase4_time = (deadline - time.monotonic()) * 0.35
    print("\n=== Phase 4: Softmax FISTA polish (%.1f sec) ===" % phase4_time)

    if phase4_time > 10:
        a_p4, s4 = run_softmax_fista(global_a, n, S, min(phase4_time, (budget_s-20)*0.10), rng,
                                      beta_init=50.0, beta_max=500.0)
        a_p4 = safe_normalize(a_p4, n)
        c4 = fft_convolve_self(a_p4)
        if c4.max() < global_best:
            global_best = c4.max()
            global_a = a_p4.copy()
            print("  Softmax FISTA: max_c = %.6f" % global_best)

    # --- Phase 5: Final ultra-fine polish ---
    phase5_time = deadline - time.monotonic() - 5.0
    print("\n=== Phase 5: Final polish (%.1f sec) ===" % phase5_time)

    if phase5_time > 5:
        a_p5, s5 = run_refined_subgradient(global_a, n, S, phase5_time, rng,
                                            k_start=32, "k_end": 256,
                                            step_scale=0.5, noise_start=0.0005,
                                            noise_decay=0.9999)
        a_p5 = safe_normalize(a_p5, n)
        c5 = fft_convolve_self(a_p5)
        if c5.max() < global_best:
            global_best = c5.max()
            global_a = a_p5.copy()
            print("  Final polish: max_c = %.6f" % global_best)

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(n, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(n, dtype=np.float64)

    # Ensure sum is reasonable (not too small for evaluate_sequence)
    if global_a.sum() < 1.0:
        global_a = global_a * (1.0 / global_a.sum())

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\n=== Final Result ===")
    print("Internal score (max_c at normalized sum): %.6f" % global_best)
    print("Verified score via evaluate_sequence: %.6f" % ver if np.isfinite(ver) else "VERIFIED: inf")

    print("Sum: %.4f" % sum(final_list))

    if not np.isfinite(ver):
        print("  WARNING: evaluate_sequence rejected! Using fallback.")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback verified: %.6f" % ver)

    return final_list
