"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using Adam log-parameter optimization
with exponential peak weighting, symmetry-breaking bias, geometric-decay
initializations, coordinate descent polish, and multi-scale exploration.

Key insight: the optimal non-negative sequence is highly asymmetric (front-loaded),
so we initialize with decay profiles and use a bias term to break symmetry.
The gradient focuses on the top peaks of the self-convolution via exponential weighting.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def soft_weights(c, T):
    """Compute exponential peak weights."""
    c_max = c.max()
    if c_max < 1e-30:
        w = np.zeros(len(c))
        w[0] = 1.0
        return w
    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    return w


def compute_gradient_soft(c, w, a, n, S):
    """Compute gradient of sum_k w_k * c_k w.r.t. log-parameters.
    
    Returns g_theta such that minimizing sum(w_k * c_k) in log-param space.
    """
    # g_a[i] = 2 * sum_k w_k * a_{k-i}
    # = 2 * cross_corr(a, w)[i] where cross_corr[i] = sum_j a[j] * w[j+i]
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A_f = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A_f)))
    g_a = 2.0 * corr[:n]
    
    # Chain rule: d/dtheta_i = g_a[i]*a[i] - (a[i]/S)*dot(g_a, a)
    dot_ga = np.dot(g_a, a)
    g_theta = g_a * a - (a / S) * dot_ga
    
    return g_theta


def adam_log_optimize(a_init, n, S, time_budget, T_init=10.0, T_final=1000.0,
                      lr_init=0.02, beta1=0.9, beta2=0.999, bias=0.0,
                      verbose=False, label=""):
    """Adam optimizer on log-parameters with exponential peak weighting."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a * (S / a.sum())
    
    theta = np.log(a / S).copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8
    
    start = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    
    i_arr = np.arange(n, dtype=np.float64)
    bias_vec = bias * (n / 2.0 - i_arr) / n
    
    no_improve = 0
    patience = 200
    
    while time.monotonic() - start < time_budget:
        t += 1
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.6)
        
        # Convert theta to a
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300 or not np.isfinite(total):
            theta = np.zeros(n)
            a = np.full(n, S / n)
        else:
            a = S * exp_theta / total
        
        # Compute convolution
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Patience-based restart
        if no_improve > patience:
            noise = 0.05 * (1.0 - progress)
            theta += noise * np.random.standard_normal(n)
            theta = theta - theta.mean()
            no_improve = 0
            if verbose:
                print("    %s: restart at t=%d" % (label, t))
            continue
        
        # Compute weights and gradient
        c_max = c.max()
        if c_max < 1e-30:
            break
        w = soft_weights(c, T)
        g_theta = compute_gradient_soft(c, w, a, n, S)
        
        # Add symmetry-breaking bias
        if bias > 0:
            g_theta = g_theta + bias_vec
        
        # Adam update
        gnorm = np.linalg.norm(g_theta)
        if gnorm < 1e-15:
            break
        
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr_t = lr_init * (1.0 - 0.6 * progress)
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodic renormalization
        if t % 200 == 0:
            theta = theta - theta.mean()
        
        if verbose and t % 1000 == 0:
            print("    %s t=%d: obj=%.7f, T=%.1f, lr=%.4f" % (label, t, best_obj, T, lr_t))
    
    return best_a, best_obj


def coordinate_descent(a, n, S, time_budget, max_sweeps=20, verbose=False):
    """Coordinate descent with multiplicative factor search."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(777)
    factors = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0]
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for idx in order:
            if time.monotonic() - start > time_budget:
                break
            
            original = a[idx]
            if original < 1e-10:
                continue
            
            best_local = original
            best_local_obj = best_obj
            
            for factor in factors:
                a[idx] = original * factor
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-10:
                    best_local_obj = obj_check
                    best_local = original * factor
            
            a[idx] = best_local
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            
            if c.max() < best_obj - 1e-12:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        if not improved and sweep >= 2:
            if verbose:
                print("    CD converged at sweep %d, obj=%.7f" % (sweep, best_obj))
            break
    
    return best_a, best_obj


def get_geometric_init(n, S, r, rng):
    """Geometric decay initialization: a_i proportional to r^i."""
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    # Add small random perturbation
    a *= (1.0 + 0.01 * rng.standard_normal(n))
    a = np.maximum(a, 1e-10)
    return a * (S / a.sum())


def get_exp_decay_init(n, S, alpha, rng):
    """Exponential decay: a_i proportional to exp(-alpha * i/n)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i / n)
    a *= (1.0 + 0.01 * rng.standard_normal(n))
    a = np.maximum(a, 1e-10)
    return a * (S / a.sum())


def get_power_decay_init(n, S, p, rng):
    """Power law decay: a_i proportional to (1 - i/n)^p."""
    i = np.arange(n, dtype=np.float64)
    x = 1.0 - i / (n - 1)
    a = np.power(np.maximum(x, 1e-10), p)
    a *= (1.0 + 0.01 * rng.standard_normal(n))
    a = np.maximum(a, 1e-10)
    return a * (S / a.sum())


def get_mixed_init(n, S, rng):
    """Mixed initialization combining decay with oscillation."""
    i = np.arange(n, dtype=np.float64)
    r = 0.995 + 0.004 * rng.random()
    alpha = 1.0 + 2.0 * rng.random()
    a = np.power(r, i) * (1.0 + 0.1 * np.sin(2 * np.pi * rng.integers(1, 5) * i / n))
    a = np.maximum(a, 1e-10)
    return a * (S / a.sum())


def peak_equalize_gradient(c, a, n, S, top_k=10):
    """Gradient that focuses on equalizing the top-k peaks."""
    # Find top-k peaks
    k = min(top_k, len(c))
    top_indices = np.argpartition(c, -k)[-k:]
    
    # Equal weight on top peaks
    w = np.zeros(len(c))
    w[top_indices] = 1.0 / k
    
    # Compute gradient as before
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A_f = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A_f)))
    g_a = 2.0 * corr[:n]
    
    dot_ga = np.dot(g_a, a)
    g_theta = g_a * a - (a / S) * dot_ga
    
    return g_theta

            obj = evaluate_sequence(a_best.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_best.copy()
                print("  CD2: obj=%.7f" % obj)
    
    # ============ PHASE 4: Random restarts (15% time) ============
    restart_budget = deadline - time.monotonic()
    print("\n=== Phase 4: Restarts (%.0fs) ===" % max(0, restart_budget))
    
    if global_a is not None and restart_budget > 20:
        best_a = global_a.copy()
        n_rounds = max(3, int(restart_budget / 30))
        per_round = restart_budget / (n_rounds + 1)
        
        for r in range(n_rounds):
            if time.monotonic() > deadline - 10:
                break
            
            # Different perturbation strategies
            strategy = r % 4
            if strategy == 0:
                # Small Gaussian noise
                noise = 0.003 * (1 + 0.5 * r)
                a_noisy = best_a + noise * rng.standard_normal(n)
            elif strategy == 1:
                # Geometric re-blend
                i = np.arange(n, dtype=np.float64)
                r_new = 0.995 + 0.003 * rng.random()
                geom = np.power(r_new, i)
                blend = 0.7 * best_a + 0.3 * geom * (S / geom.sum())
                a_noisy = blend
            elif strategy == 2:
                # Coordinate shuffle
                a_noisy = best_a.copy()
                idx1 = rng.integers(0, n)
                idx2 = rng.integers(0, n)
                a_noisy[idx1], a_noisy[idx2] = a_noisy[idx2], a_noisy[idx1]
            else:
                # Random geometric restart blended
                r_new = 0.98 + 0.03 * rng.random()
                i = np.arange(n, dtype=np.float64)
                geom = np.power(r_new, i)
                geom = geom * (S / geom.sum())
                blend = 0.5 * best_a + 0.5 * geom
                a_noisy = blend
            
            a_noisy = normalize_nonneg(a_noisy, S)
            
            # Quick optimization
            a_best_r, _ = adam_log_optimize(
                a_noisy, n, S, per_round,
                T_init=100, T_final=1500, lr_init=0.015, bias=0.2,
                label="R%d" % r
            )
            a_best_r = normalize_nonneg(a_best_r, S)
            obj = evaluate_sequence(a_best_r.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_best_r.copy()
                best_a = a_best_r.copy()
                print("  Restart %d (str%d): obj=%.7f" % (r, strategy, obj))
            elif verbose:
                print("  Restart %d (str%d): obj=%.7f (no improve)" % (r, strategy, obj if np.isfinite(obj) else 0))
    
    # ============ Final verification ============
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f (verified=%.7f)" % (global_best, ver))
    
    if not np.isfinite(ver):
        print("WARNING: final verification failed, using fallback")
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)
    
    return final_a
