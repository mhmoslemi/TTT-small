"""
Minimize 2*n*max_conv/sum_a^2 using adaptive LSE-FISTA with spectral shaping and coordinate descent polish.

Key improvements over previous:
1. Verified LSE gradient via FFT with correct indexing
2. Adaptive step size with backtracking line search

7. Periodic restarts with structured perturbations to escape local minima

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient(a, w, fft_size):
    """
    Compute gradient of LSE(beta*c) at point a.
    w = softmax(beta * c), shape (2n-1,)
    Returns gradient of shape (n,).
    Formula: grad[j] = 2 * conv(w, a_rev)[n-1+j]
    """
    n = len(a)
    a_rev = a[::-1].copy()
    w_pad = np.zeros(fft_size)
    w_pad[:2 * n - 1] = w
    a_rev_pad = np.zeros(fft_size)
    a_rev_pad[:n] = a_rev
    W = fft(w_pad)
    A_rev = fft(a_rev_pad)
    conv_res = np.real(ifft(W * A_rev))
    grad = 2.0 * conv_res[n - 1: 2 * n - 1]
    return grad


def compute_score(a):
    """Compute the actual objective: 2*n*c_max/sum_a^2"""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def safe_normalize(a, n, S):
    """Clip to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def spectral_shaping(a, N_fft, top_k, alpha):
    """
    One spectral shaping step: modify |FFT(a)|^2 to reduce peak.
    Returns modified a (same sum).
    """
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N_fft)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]

    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a

    weights = top_vals / total
    m = np.arange(N_fft)
    cos_eff = np.zeros(N_fft)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N_fft)

    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def fista_lse(a_init, n, S, max_time, rng, beta_start=5.0, beta_end=500.0,
              step_start=0.3, momentum_decay=0.99):
    """
    FISTA with LSE smoothing and adaptive parameters.
    """
    a = safe_normalize(a_init, n, S)
    N_fft = 1 << (3 * n - 2).bit_length()

    y = a.copy()
    t_mom = 1.0
    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()
    it = 0
    stall = 0
    last_best = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: increase over time
        frac = min(1.0, it / max(1, int(max_time / 0.003)))
        beta = beta_start * (beta_end / beta_start) ** frac

        c = fft_convolve_self(y)
        c_max = c.max()
        score_curr = 2.0 * n * c_max / (S * S)

        if score_curr < best_score - 1e-8:
            best_score = score_curr
            best_a = y.copy()
            stall = 0
        else:
            stall += 1

        if stall > 5000:
            break

        # Compute LSE weights (with numerical stability)
        c_shifted = c - c.max()
        beta_c = beta * c_shifted
        # Clip to avoid overflow
        beta_c = np.clip(beta_c, -500, 500)
        exp_bc = np.exp(beta_c)
        w = exp_bc / exp_bc.sum()

        # Compute gradient
        grad = lse_gradient(y, w, N_fft)

        # Adaptive step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = step_start / (1.0 + 0.001 * it)
        step = min(step, 1.0 / max(gnorm / S, 1e-10))
        step = min(step, 2.0 * S / (n * gnorm / max(gnorm, 1e-10)))
        step = max(step, 1e-6)

        x_new = project_simplex(y - step * grad, S)

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y = project_simplex(y, S)
        a = x_new
        t_mom = t_new

        # Periodic perturbation to escape local minima
        if it % 3000 == 0 and rng.random() < 0.2:

            y = project_simplex(y + rng.normal(0, scale, n), S)
            a = y.copy()
            t_mom = 1.0
            stall //= 2

    return best_a, best_score


def coordinate_descent_polish(a_init, n, S, max_time, rng, n_sweeps_max=50):
    """
    Coordinate descent polish: for each element, try optimal value
    while maintaining sum constraint.
    """
    a = safe_normalize(a_init, n, S).copy()
    c = fft_convolve_self(a)
    best_score = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    start = time.monotonic()
    sweep = 0

    # Find active lags (top peaks)
    top_lags = np.argsort(c)[::-1][:20]

    while time.monotonic() - start < max_time and sweep < n_sweeps_max:
        sweep += 1
        improved = False
        # Randomize element order
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() - start > max_time:
                break
            old_val = a[j]
            if old_val < 1e-10:
                continue

            # Try a few candidate values for a[j]
            # When we change a[j] to a[j]+delta, we need to compensate
            # by reducing all other elements proportionally
            # For simplicity, try: set a[j] to candidate, redistribute the difference
            candidates = [old_val * 0.5, old_val * 1.5, 0.0,
                          old_val / (1 + 0.1), old_val * (1 + 0.1)]

            best_candidate = old_val
            best_local_score = best_score

            for cand in candidates:
                if cand < 0 or cand > 1000:
                    continue
                delta = cand - old_val
                if abs(delta) < 1e-12:
                    continue
                # Compensate: reduce other elements proportionally
                a_trial = a.copy()
                a_trial[j] = cand
                remaining = S - cand
                other_sum = S - old_val
                if other_sum < 1e-10 or remaining < 0:
                    continue
                a_trial[np.arange(n) != j] = a_trial[np.arange(n) != j] * (remaining / other_sum)
                a_trial = np.clip(a_trial, 0, 1000)
                if a_trial.sum() < 0.01:
                    continue
                # Quick score check using only affected lags
                # Full convolution is expensive, so just check top lags
                c_trial = fft_convolve_self(a_trial)
                sc = 2.0 * n * c_trial.max() / (a_trial.sum() ** 2)
                if sc < best_local_score:
                    best_local_score = sc
                    best_candidate = a_trial.copy()

            if best_candidate is not old_val and isinstance(best_candidate, np.ndarray):
                a = best_candidate
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    improved = True

        if not improved and sweep > 10:
            break

    return best_a, best_score


def make_initializations(n, rng, S):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    # Random
    inits.append(("rand", rng.random(n) + 0.05))

    # Chirp
    inits.append(("chirp", 0.5 + 0.4 * np.sin(2 * np.pi * (5 * t + 0.5 * t * t)) + 0.05 * rng.standard_normal(n)))

    # Multi-sine
    inits.append(("msine", 1.0 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.sin(2 * np.pi * 7 * t)
                  + 0.1 * np.sin(2 * np.pi * 13 * t) + 0.05 * rng.standard_normal(n)))

    # Bump
    inits.append(("bump", np.exp(-0.5 * ((t - 0.5) * 4.0) ** 2) + 0.15))

    # Multi-segment (5 segments)
    seg_len = n // 5
    best_h = None
    best_sc = np.inf
    for _ in range(500):
        heights = rng.random(5) + 0.1
        arr = np.zeros(n)
        for i in range(5):
            start_i = i * seg_len
            end_i = start_i + seg_len if i < 4 else n
            arr[start_i:end_i] = heights[i]
        c = fft_convolve_self(arr)
        sc = c.max() / (arr.sum() ** 2 / (2 * n))
        if sc < best_sc:
            best_sc = sc
            best_h = heights.copy()
    arr = np.zeros(n)
    for i in range(5):
        start_i = i * seg_len
        end_i = start_i + seg_len if i < 4 else n
        arr[start_i:end_i] = best_h[i]
    inits.append(("seg5", arr))

    # Multi-segment (9 segments)
    seg_len9 = n // 9
    best_h9 = None
    best_sc9 = np.inf
    for _ in range(300):
        heights = rng.random(9) + 0.1
        arr = np.zeros(n)
        for i in range(9):
            start_i = i * seg_len9
            end_i = start_i + seg_len9 if i < 8 else n
            arr[start_i:end_i] = heights[i]
        c = fft_convolve_self(arr)
        sc = c.max() / (arr.sum() ** 2 / (2 * n))
        if sc < best_sc9:
            best_sc9 = sc
            best_h9 = heights.copy()
    arr = np.zeros(n)
    for i in range(9):
        start_i = i * seg_len9
        end_i = start_i + seg_len9 if i < 8 else n
        arr[start_i:end_i] = best_h9[i]
    inits.append(("seg9", arr))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Uses adaptive LSE-FISTA + spectral shaping + coordinate descent.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Determine main length from provided sequence or default
    n = 8192
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if 256 <= given.size <= 32768:
            n = len(given)
    except Exception:
        pass

    S = np.sqrt(2.0 * float(n))
    N_fft = 1 << (3 * n - 2).bit_length()
    N_conv = 1 << (2 * n - 2).bit_length()

    print("n=%d, S=%.4f, N_fft=%d" % (n, S, N_fft))

    # Add the given sequence as an initialization
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            g = np.clip(given, 0, 1000)
            if g.sum() > 0.01:
                inits.append(("given", g))
    except Exception:
        pass

    other_inits = make_initializations(n, rng, S)
    inits.extend(other_inits)

    print("=== Phase 1: Multi-start exploration (%d inits) ===" % len(inits))
    t_p1 = time.monotonic()
    p1_budget = (deadline - t_p1) * 0.30

    per_init = p1_budget / len(inits)

    for name, arr in inits:
        if time.monotonic() > deadline - 120:
            break
        a_init = safe_normalize(arr, n, S)
        t0 = time.monotonic()
        alloc = min(per_init, max(2.0, (deadline - 120) - t0))

        # Short FISTA with moderate beta
        a_out, sc = fista_lse(a_init, n, S, alloc, rng,
                               beta_start=3.0, beta_end=100.0, step_start=0.25)
        sc_actual = compute_score(a_out)
        if sc_actual < global_best:
            global_best = sc_actual
            global_a = a_out.copy()
            print("  [%s] FISTA: %.6f" % (name, global_best))

    print("=== Phase 1 done. Best: %.6f ===" % global_best)

    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = compute_score(global_a)
        print("  Fallback: uniform, score=%.6f" % global_best)

    # Phase 2: Intensive optimization on best candidate
    print("=== Phase 2: Intensive FISTA-LSE ===")
    t_p2 = time.monotonic()
    p2_budget = (deadline - t_p2) * 0.35

    a_curr = global_a.copy()
    # Run FISTA with increasing beta
    rem = p2_budget * 0.4
    a_out, sc = fista_lse(a_curr, n, S, rem, rng,
                           beta_start=10.0, beta_end=200.0, step_start=0.2)
    sc_actual = compute_score(a_out)
    if sc_actual < global_best:
        global_best = sc_actual
        global_a = a_out.copy()
        print("  FISTA coarse->mid: %.6f" % global_best)

    # Add spectral shaping interspersed
    rem = deadline - time.monotonic() - (deadline - t_p2) * 0.25
    if rem > 10:
        a_curr = global_a.copy()
        c = fft_convolve_self(a_curr)
        c_max = c.max()
        n_ss = min(int(rem / 0.005), 20000)
        for i in range(n_ss):
            if time.monotonic() > t_p2 + p2_budget:
                break
            top_k = 10 + 5 * (i // 100)
            alpha = 0.15 * (1.0 + 0.5 * (c_max / 1.5 - 1.0))
            alpha = max(0.02, min(alpha, 0.5))
            a_curr = spectral_shaping(a_curr, N_conv, min(top_k, 50), alpha)
            if i % 200 == 100:
                c = fft_convolve_self(a_curr)
                c_max = c.max()
                sc_check = 2.0 * n * c_max / (S * S)
                if sc_check < global_best:
                    global_best = sc_check
                    global_a = a_curr.copy()
                    if i % 1000 == 0:
                        print("  Spectrum: %.6f" % global_best)

    print("=== Phase 2 done. Best: %.6f ===" % global_best)

    # Phase 3: Fine FISTA + coordinate descent polish
    print("=== Phase 3: Fine polish ===")
    t_p3 = time.monotonic()
    rem_total = deadline - t_p3

    # Fine FISTA with high beta
    rem_fista = min(rem_total * 0.5, 600.0)
    if rem_fista > 5:
        a_out, sc = fista_lse(global_a, n, S, rem_fista, rng,
                               beta_start=100.0, beta_end=1000.0, step_start=0.1)
        sc_actual = compute_score(a_out)
        if sc_actual < global_best:
            global_best = sc_actual
            global_a = a_out.copy()
            print("  Fine FISTA: %.6f" % global_best)

    # Coordinate descent
    rem_cd = deadline - time.monotonic() - 10.0
    if rem_cd > 5:
        a_cd, sc_cd = coordinate_descent_polish(global_a, n, S, rem_cd, rng, n_sweeps_max=30)
        sc_actual = compute_score(a_cd)
        if sc_actual < global_best:
            global_best = sc_actual
            global_a = a_cd.copy()
            print("  Coord descent: %.6f" % global_best)

    # Final verification and return
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified score: %.6f" % ver)
    else:
        print("WARNING: evaluate_sequence returned non-finite, using fallback")
        global_a = np.ones(256, dtype=np.float64)
        final_list = [1.0] * 256

    print("Returning sequence of length %d" % len(final_list))
    return final_list
