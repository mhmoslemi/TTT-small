import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def softmax(x):
    x = x - x.max()
    ex = np.exp(x)
    return ex / ex.sum()

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_gradient_fft(a, weights):
    n = len(a)
    fsize = 1 << (len(weights) + n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def logsumexp(x, beta):
    m = x.max()
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def lnsbfgs_optimize(b_init, n, max_time, beta_schedule):
    """L-BFGS optimization with annealed beta."""
    b = b_init.copy()
    p = softmax(b)
    best_p = p.copy()
    best_obj = 2.0 * n * (fft_convolve_self(p)).max()
    
    start = time.monotonic()
    
    for beta, max_iter in beta_schedule:
        if time.monotonic() - start > max_time:
            break
        
        def obj_grad(b):
            p = softmax(b)
            c = fft_convolve_self(p)
            m = c.max()
            exp_bc = np.exp(beta * (c - m))
            logsumexp_val = m + np.log(exp_bc.sum()) / beta
            w = exp_bc / exp_bc.sum()
            gp = compute_gradient_fft(p, w)
            grad_b = p * (gp - np.dot(p, gp))
            return logsumexp_val, grad_b
        
        result = minimize(obj_grad, b, method='L-BFGS-B', 
                          options={'maxiter': max_iter, 'gtol': 1e-8, 'maxfun': max_iter})
        b = result.x
        p = softmax(b)
        c = fft_convolve_self(p)
        obj = 2.0 * n * c.max()
        if obj < best_obj:
            best_obj = obj
            best_p = p.copy()
    
    return best_p, best_obj

"""
Minimize 2*n*max_conv/sum^2 using L-BFGS with annealed log-sum-exp smoothing.
Key innovations: smooth objective optimization via log-sum-exp, softmax reparameterization
for constraint handling, multiple structured initializations, multi-length search,
and hybrid FISTA+L-BFGS approach for both coarse and fine optimization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def softmax_fn(x):
    x = np.asarray(x, dtype=np.float64)
    x = x - x.max()
    ex = np.exp(x)
    s = ex.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return ex / s


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    n = len(a)
    fsize = 1 << (len(weights) + n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(a):
    n = len(a)
    c = fft_convolve_self(a)
    max_c = c.max()
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * max_c / (s * s)


def lnsbfgs_optimize(b_init, n, max_time, beta_schedule):
    b = b_init.copy()
    p = softmax_fn(b)
    best_p = p.copy()
    best_obj = compute_objective(p)

    start = time.monotonic()
    print("    L-BFGS: initial obj=%.6f" % best_obj)

    for beta, max_iter in beta_schedule:
        if time.monotonic() - start > max_time:
            break
        if time.monotonic() - start > max_time - 10:
            break

        t0 = time.monotonic()

        def obj_grad(b):
            p = softmax_fn(b)
            c = fft_convolve_self(p)
            m = c.max()
            exp_bc = np.exp(beta * (c - m))
            total = exp_bc.sum()
            if total < 1e-300:
                val = m
                w = np.full_like(c, 1.0 / len(c))
            else:
                val = m + np.log(total) / beta
                w = exp_bc / total
            gp = compute_gradient_fft(p, w)
            pgp = np.dot(p, gp)
            grad_b = p * (gp - pgp)
            return val, grad_b

        try:
            result = minimize(obj_grad, b, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'gtol': 1e-10,
                                       'maxfun': max_iter * 2, 'ftol': 1e-12})
            b = result.x
        except Exception as e:
            print("    L-BFGS error at beta=%d: %s" % (beta, str(e)))
            break

        p = softmax_fn(b)
        obj = compute_objective(p)
        elapsed = time.monotonic() - t0
        if obj < best_obj:
            best_obj = obj
            best_p = p.copy()
        print("    beta=%6d: obj=%.6f (%.1fs, success=%s)" % (beta, obj, elapsed, result.success))

    return best_p, best_obj


def fista_coarse(a_init, n, S, max_time, rng, beta_init=30.0, beta_max=500.0):
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = compute_objective(a)
    best_local = a.copy()
    it = 0
    step = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))

        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max() / (y.sum() ** 2)
        if cur_obj < best_score - 1e-12:
            best_score = cur_obj
            best_local = y.copy()

        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            w = np.full_like(c, 1.0 / len(c))
        else:
            w = exp_bc / total

        grad = compute_gradient_fft(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = project_simplex(y - step * grad / gnorm, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 100 == 0:
            step *= 0.999

        if it % 5000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0

    return best_local, best_score


def get_initialization(n, init_type, rng, given_seq=None):
    if init_type == 'given' and given_seq is not None and len(given_seq) == n:
        a = np.clip(given_seq, 0, 1000).copy().astype(np.float64)
        s = a.sum()
        if s > 1e-15:
            return a / s
        else:
            return np.ones(n) / n
    elif init_type == 'uniform':
        return np.ones(n) / n
    elif init_type == 'cosine':
        t = np.arange(n) / (n - 1)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 0)
        s = a.sum()
        return a / s
    elif init_type == 'triangular':
        t = np.arange(n) / (n - 1)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        s = a.sum()
        return a / s
    elif init_type == 'random':
        a = np.abs(rng.standard_normal(n))
        s = a.sum()
        return a / s
    elif init_type == 'bimodal':
        t = np.arange(n) / (n - 1)
        a = np.exp(-10 * (t - 0.3) ** 2) + np.exp(-10 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        s = a.sum()
        return a / s
    elif init_type == 'skewed':
        t = np.arange(n) / (n - 1)
        a = np.exp(-3 * t) + 0.5 * np.exp(-3 * (1 - t))
        s = a.sum()
        return a / s
    else:
        return np.ones(n) / n
