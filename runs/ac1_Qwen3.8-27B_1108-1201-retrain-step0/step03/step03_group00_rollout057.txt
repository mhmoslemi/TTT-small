    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[idx] = np.power(c[idx], p-1)

    f_val = np.sum(w * c)

    # We want sum_j a[j]*w[j+i], which is the correlation at lag -i

import numpy as np

def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def topk_subgradient(a, c, k):
    """Compute subgradient of sum of top-k c values w.r.t. a.
    Returns gradient vector of length n."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for ki in idx:
        lo = max(0, ki - n + 1)
        hi = min(ki, n - 1)
        if lo <= hi:
            # d(c[ki])/da[i] = 2 * a[ki - i] for i in [lo, hi]
            # a[ki - i] for i = lo, lo+1, ..., hi
            # = a[ki-lo], a[ki-lo-1], ..., a[ki-hi]
            grad[lo:hi+1] += 2.0 * a[lo:ki+1][::-1]
    return grad


def weighted_topk_subgradient(a, c, k, p=2.0):
    """Compute subgradient of (sum of top-k c^p)^(1/p) w.r.t. a.
    Weighted by c^(p-1)."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    weights = np.power(np.maximum(vals, 0), p - 1)
    wsum = np.sum(weights * vals)
    if wsum < 1e-30:
        return np.zeros(n), 0.0
    scale = wsum ** (1.0/p - 1.0)
    
    grad = np.zeros(n)
    for i_k in range(k):
        ki = idx[i_k]
        w = weights[i_k]
        lo = max(0, ki - n + 1)
        hi = min(ki, n - 1)
        if lo <= hi:
            grad[lo:hi+1] += 2.0 * w * a[lo:ki+1][::-1]
    return scale * grad, wsum ** (1.0/p)

def fourier_init(n, rng):
    """Initialize via random phase spectrum + projection to non-negative."""
    # Create a spectrum with flat magnitude (except DC)
    fsize = n  # use n-point FFT
    phases = rng.uniform(0, 2*np.pi, fsize)
    # Make it Hermitian (for real output)
    A = np.zeros(fsize, dtype=complex)
    for k in range(fsize // 2 + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k > 0 and k < fsize // 2:
            A[fsize - k] = np.conj(A[k])
    
    a = np.real(ifft(A))
    a = np.maximum(a, 0)  # project to non-negative
    a_sum = a.sum()
    if a_sum > 0:
        a = a / a_sum
    else:
        a = np.ones(n) / n
    return a

def upsample(a, n_new):
    """Upsample sequence to new length via interpolation."""
    n_old = len(a)
    if n_new <= n_old:
        return a[:n_new].copy()
    t_old = np.linspace(0, 1, n_old)
    t_new = np.linspace(0, 1, n_new)
    return np.interp(t_new, t_old, a)
