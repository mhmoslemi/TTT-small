def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_at_peak(a, peak_idx):
    """Compute subgradient of conv(a,a)[peak_idx] w.r.t. a.
    For full conv, grad[j] = a[peak_idx - j] if 0 <= peak_idx - j < n else 0.
    Since it's self-conv, multiply by 2 for the symmetric contribution.
    """
    n = len(a)
    grad = np.zeros(n)
    # peak_idx is in [0, 2n-2]
    lo = max(0, peak_idx - (n - 1))
    hi = min(n - 1, peak_idx)
    if lo <= hi:
        j_range = np.arange(lo, hi + 1)
        a_idx = peak_idx - j_range
        grad[j_range] = 2.0 * a[a_idx]
    return grad


def compute_grad_multi_peak(a, peak_indices, weights):
    """Weighted sum of subgradients at multiple peak locations."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(peak_indices, weights):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            j_range = np.arange(lo, hi + 1)
            a_idx = k - j_range
            grad[j_range] += w * 2.0 * a[a_idx]
    return grad


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z (softmax params)."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    # Gradient of smoothed max w.r.t. a
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=200):
    """L-BFGS-B with continuation on beta. Returns best (a, score)."""
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        # More attempts at high beta
        n_attempts = 1
        if beta >= 500 and max_time > 120:
            n_attempts = 2
        if beta >= 2000 and max_time > 240:
            n_attempts = 3

        elapsed_beta = 0.0
        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 5:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.015, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            t0 = time.monotonic()
            try:
                result = scipy_minimize(
                    fun, z,
                    jac=jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_beta,
                        'maxfun': maxiter_per_beta * 3,
                        'ftol': 1e-15,
                        'gtol': 1e-10,
                        'maxls': 50,
                        'iprint': -1
                    }
                )
            except Exception:
                break

            elapsed_beta += time.monotonic() - t0
            z = result.x
            z = z - z.mean()

            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

        if elapsed_beta > 0:
            print("    beta=%5d: best=%.8f (%d att, %.1fs)" %
                  (beta, best_score, n_attempts, elapsed_beta))

    return best_a, best_score


def run_nesterov_linesearch(a_init, n, S, max_time, rng,
                            top_k_init=20, top_k_max=2048,
                            threshold_factor=1.001):
    """Nesterov accelerated subgradient with backtracking line search."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)

        c_y = fft_convolve_self(y)
        cur = c_y.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Compute subgradient based on top peaks with soft weighting
        threshold = c_y.max() * threshold_factor
        above = c_y > threshold
        if not np.any(above):
            # Use top-k as fallback
            top_idx = np.argsort(c_y)[::-1][:top_k]
            top_vals = c_y[top_idx]
            w = top_vals / top_vals.sum()
            grad = compute_grad_multi_peak(a, top_idx.astype(int), w)
        else:
            top_idx = np.where(above)[0].astype(int)
            if len(top_idx) > top_k:
                # Keep only top_k
                top_idx_full = list(top_idx)
                top_idx = np.array(sorted(top_idx_full, key=lambda i: -c_y[i])[:top_k])
            top_vals = c_y[top_idx]
            w = top_vals / top_vals.sum()
            grad = compute_grad_multi_peak(a, top_idx, w)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 2.0 * S / (n * (1.0 + 0.001 * it))
        direction = -grad / gnorm

        # Try to find good step size
        best_x = a.copy()
        best_val = cur
        for ls_attempt in range(8):
            trial = a + step * direction * n  # scale direction to proper magnitude
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                step *= 0.5
                continue
            trial = trial * (S / ts)
            c_trial = fft_convolve_self(trial)
            trial_max = c_trial.max()
            if trial_max < best_val * 0.999:
                best_x = trial
                best_val = trial_max
                break
            step *= 0.7

        if best_val < cur:
            x_new = best_x
        else:
            x_new = a + step * 0.1 * direction * n
            x_new = np.maximum(x_new, 0.0)
            x_new = project_simplex(x_new, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y * (S / ys)
        a = x_new
        t = t_new

        # Random restart
        if it % 2000 == 0 and rng.random() < 0.1:
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0

    return best_local, best_score


def run_peak_shaving(a_init, n, S, max_time, rng):
    """Targeted peak shaving: iteratively reduce the top convolution values
    by adjusting the sequence elements that contribute most to them."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        # Find top peaks (use a slightly wider set for robustness)
        n_peaks = min(50, 10 + it // 100)
        peak_indices = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[peak_indices]
        total = peak_vals.sum()
        weights = peak_vals / total

        # Compute aggregated subgradient
        grad = compute_grad_multi_peak(a, peak_indices.astype(int), weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Multiple step sizes (geometric)
        base_step = S / (n * (1.0 + 0.005 * it))
        step_multipliers = [1.0, 0.5, 0.25, 0.125, 0.0625]

        best_trial = None
        best_trial_score = cur

        for mult in step_multipliers:
            step = base_step * mult
            trial = a - step * grad / gnorm
            trial = np.maximum(trial, 0.0)
            trial = project_simplex(trial, S)
            c_trial = fft_convolve_self(trial)
            trial_score = c_trial.max()
            if trial_score < best_trial_score:
                best_trial_score = trial_score
                best_trial = trial

        if best_trial is not None:
            a = best_trial
        else:
            # No improvement from subgradient; try random perturbation
            if it % 100 == 0:
                noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
            else:
                # Take minimum step
                a = a - base_step * 0.0625 * grad / gnorm
                a = np.maximum(a, 0.0)
                a = project_simplex(a, S)

        if it % 500 == 0:
            print("      PeakShave it=%d: score=%.8f" % (it, best_score))

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def thue_morse_sequence(n):
    """Generate Thue-Morse based initialization (good autocorrelation properties)."""
    tm = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    # Map to [0.3, 1.7] range for non-negative values with some spread
    return 0.3 + 1.4 * tm


def generate_initializations(n, rng):
    """Generate diverse initializations for optimization."""
    tt = np.linspace(0, 1, n, endpoint=False)
    inits = {}

    # Random
    inits['rand'] = rng.random(n) + 0.1

    # Thue-Morse
    inits['thue'] = thue_morse_sequence(n)

    # Multi-frequency
    inits['multi_sin'] = (1.0 + 0.3 * np.sin(2*np.pi*3*tt) +
                          0.2 * np.cos(2*np.pi*7*tt) +
                          0.15 * np.sin(2*np.pi*13*tt) +
                          0.1 * np.cos(2*np.pi*21*tt) +
                          0.05 * np.sin(2*np.pi*47*tt))

    # Periodic block pattern (blocks of different lengths)
    block = np.array([1.5, 0.5, 1.2, 0.3, 1.0, 0.7, 1.3, 0.4])
    repeats = n // len(block) + 1
    inits['block'] = np.tile(block, repeats)[:n]

    # Gaussian bumps
    inits['bumps'] = (np.exp(-0.5*((tt-0.2)*8.0)**2) * 1.5 +
                      np.exp(-0.5*((tt-0.5)*6.0)**2) * 1.2 +
                      np.exp(-0.5*((tt-0.8)*8.0)**2) * 1.5 + 0.05)

    # Checker-like with decay
    inits['checker'] = 0.5 + 0.4 * np.cos(2*np.pi*4*tt) * np.exp(-0.5*(tt-0.5)**2) + 0.05

    # Power-law decay from center
    center = np.abs(tt - 0.5)
    inits['power'] = 1.0 + 0.5 * (1 - center)**0.5 + 0.1 * np.random.random(n)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None
    global_n = 0

    # --- Get initial sequence if available ---
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given_clipped = np.clip(given, 0.0, 1000.0)
            n_given = len(given_clipped)
            S_given = np.sqrt(2.0 * float(n_given))
            given_norm = given_clipped * (S_given / given_clipped.sum())
            c_given = fft_convolve_self(given_norm)
            score_given = c_given.max()
            global_best = score_given
            global_a = given_norm.copy()
            global_n = n_given
            print("Starting from given sequence: n=%d, score=%.8f" % (n_given, score_given))
    except Exception:
        pass

    if global_a is None:
        print("No given sequence, starting fresh")

    # --- Determine search lengths ---
    # Try both n=4096 and n=8192; allocate time based on potential
    search_n = 8192
    S = np.sqrt(2.0 * float(search_n))

    print("\n=== Main Search: n=%d, budget=%.0fs ===" % (search_n, budget_s))
    print("Target: 1.5027, Current best: %.6f" % global_best)

    # Generate initializations
    inits = generate_initializations(search_n, rng)

    # If we have a given sequence, add it as primary initialization
    if 'given' not in inits and global_a is not None and global_n == search_n:
        inits['given'] = global_a

    # Also try the given sequence at our target n if length differs
    if 'given' not in inits and global_a is not None and global_n != search_n:
        # Upsample or downsample
        if global_n > search_n:
            inits['given'] = global_a[:search_n]
        else:
            # Linear interpolation upsampling
            idx_old = np.linspace(0, 1, global_n, endpoint=False)
            idx_new = np.linspace(0, 1, search_n, endpoint=False)
            inits['given'] = np.interp(idx_new, idx_old, global_a)

    n_inits = len(inits)
    print("  %d initializations: %s" % (n_inits, list(inits.keys())))

    # --- Phase 1: L-BFGS on each initialization ---
    # Allocate more time to 'given' init if it exists and is good
    usable = deadline - time.monotonic()
    # Give 40% most time to best init, split rest among others
    if 'given' in inits:
        given_budget = usable * 0.35
        other_budget = usable * 0.35 / (n_inits - 1)
    else:
        given_budget = 0
        other_budget = usable * 0.45 / n_inits

    print("\n=== Phase 1: L-BFGS Continuation ===")
    phase1_results = {}

    # Optimize 'given' first if available (most likely to be best)
    init_order = ['given'] + [k for k in inits if k != 'given']

    for name in init_order:
        if time.monotonic() > deadline - 300:
            break

        t0 = time.monotonic()
        budget_i = given_budget if name == 'given' else other_budget
        budget_i = max(budget_i, 15.0)

        a_norm = safe_normalize(inits[name], search_n)
        c = fft_convolve_self(a_norm)
        init_score = c.max()

        if init_score < global_best:
            global_best = init_score
            global_a = a_norm.copy()
            global_n = search_n

        # L-BFGS continuation with adaptive schedule
        if name == 'given':
            # Longer schedule for the best init
            beta_sched = [10, 20, 50, 100, 200, 500, 1000, 2000, 4000, 8000]
            maxiter = 250
        else:
            beta_sched = [5, 10, 20, 50, 100, 200, 500, 1000, 2000]
            maxiter = 150

        a_new, score = run_lbfgs_continuation(
            a_norm, search_n, S, budget_i, rng,
            beta_schedule=beta_sched, maxiter_per_beta=maxiter
        )
        a_new = safe_normalize(a_new, search_n)
        c_new = fft_convolve_self(a_new)
        final_score = c_new.max()
        elapsed = time.monotonic() - t0

        phase1_results[name] = (a_new, final_score)

        if final_score < global_best:
            global_best = final_score
            global_a = a_new.copy()
            global_n = search_n
            print("  [NEW BEST] %s: %.8f (init=%.4f, %.1fs)" %
                  (name, global_best, init_score, elapsed))
        else:
            print("  [%s: %.8f, best=%.6f, %.1fs]" %
                  (name, final_score, global_best, elapsed))

    print("\n=== After Phase 1: global_best=%.8f ===" % global_best)

    # --- Phase 2: Peak shaving on best solution ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            peak_budget = min(remaining * 0.3, 400)
            print("\n=== Phase 2: Peak Shaving (%.0fs) ===" % peak_budget)

            a_shaved, score = run_peak_shaving(
                global_a, search_n, S, peak_budget, rng
            )
            a_shaved = safe_normalize(a_shaved, search_n)
            c = fft_convolve_self(a_shaved)
            if c.max() < global_best:
                global_best = c.max()
                global_a = a_shaved.copy()
                print("  Peak shaving: %.8f" % global_best)
            else:
                print("  Peak shaving: no improvement (%.8f)" % c.max())

    # --- Phase 3: Nesterov with line search ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 40:
            nest_budget = min(remaining * 0.3, 350)
            print("\n=== Phase 3: Nesterov Line Search (%.0fs) ===" % nest_budget)

            a_nest, score = run_nesterov_linesearch(
                global_a, search_n, S, nest_budget, rng,
                top_k_init=20, top_k_max=1024
            )
            a_nest = safe_normalize(a_nest, search_n)
            c = fft_convolve_self(a_nest)
            if c.max() < global_best:
                global_best = c.max()
                global_a = a_nest.copy()
                print("  Nesterov: %.8f" % global_best)
            else:
                print("  Nesterov: no improvement (%.8f)" % c.max())

    # --- Phase 4: High-beta L-BFGS polish with multiple restarts ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            polish_budget = min(remaining * 0.35, 400)
            print("\n=== Phase 4: High-Beta Polish (%.0fs) ===" % polish_budget)

            # Multiple perturbed restarts at very high beta
            n_restarts = max(2, int(polish_budget / 60))
            per_restart = polish_budget / n_restarts

            for r in range(n_restarts):
                if time.monotonic() > deadline - 30:
                    break

                if r == 0:
                    a_start = global_a.copy()
                else:
                    noise_scale = 0.02 / (1.0 + 0.5 * r)
                    a_start = global_a + rng.normal(0, noise_scale * S / np.sqrt(search_n), search_n)
                    a_start = project_simplex(np.maximum(a_start, 0), S)

                a_pol, score = run_lbfgs_continuation(
                    a_start, search_n, S, per_restart, rng,
                    beta_schedule=[500, 1000, 2000, 5000, 10000, 20000],
                    maxiter_per_beta=200
                )
                a_pol = safe_normalize(a_pol, search_n)
                c = fft_convolve_self(a_pol)
                if c.max() < global_best:
                    global_best = c.max()
                    global_a = a_pol.copy()
                    print("  Restart %d: %.8f [NEW BEST]" % (r, global_best))
                else:
                    print("  Restart %d: %.8f" % (r, c.max()))

    # --- Phase 5: Final peak shaving ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            final_peel = min(remaining * 0.5, 200)
            print("\n=== Phase 5: Final Peak Shaving (%.0fs) ===" % final_peel)

            a_final, score = run_peak_shaving(
                global_a, search_n, S, final_peel, rng
            )
            a_final = safe_normalize(a_final, search_n)
            c = fft_convolve_self(a_final)
            if c.max() < global_best:
                global_best = c.max()
                global_a = a_final.copy()
                print("  Final shave: %.8f" % global_best)

    # --- Phase 6: Quick Nesterov on very best ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            quick_budget = min(remaining * 0.6, 100)
            print("\n=== Phase 6: Quick Nesterov (%.0fs) ===" % quick_budget)
            a_q, score = run_nesterov_linesearch(
                global_a, search_n, S, quick_budget, rng,
                top_k_init=50, top_k_max=4096
            )
            a_q = safe_normalize(a_q, search_n)
            c = fft_convolve_self(a_q)
            if c.max() < global_best:
                global_best = c.max()
                global_a = a_q.copy()
                print("  Quick Nesterov: %.8f" % global_best)

    # --- Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\n=== Verification ===")
        print("evaluate_sequence: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  Rejected! Using fallback.")
            global_a = np.ones(256, dtype=np.float64)
            global_best = 2.0
    else:
        global_a = np.ones(256, dtype=np.float64)
        global_best = 2.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("\nFinal: length=%d, internal score=%.8f" % (len(global_a), global_best))
    print("Time remaining: %.1fs" % (deadline - time.monotonic()))
    return [float(x) for x in global_a]
