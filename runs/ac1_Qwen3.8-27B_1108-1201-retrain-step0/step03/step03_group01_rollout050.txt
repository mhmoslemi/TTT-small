import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta_val = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta_val, 0.0)

def full_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2*n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]

def get_score(a):
    """Compute 2n*max(conv)/sum^2."""
    n = len(a)
    c = full_conv(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def nesterov_subgrad(a_init, n, S, max_time, rng, top_k=20):
    """Nesterov-accelerated subgradient method."""
    a = a_init.copy()
    a = project_simplex(a, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_a = a.copy()
    best_cmax = full_conv(a).max()
    N = 1 << (2*n - 2).bit_length()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = full_conv(y)
        cmax = c.max()
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = y.copy()
        
        # Top-k subgradient
        k_eff = min(top_k + it // 100, 2*n - 1)
        top_idx = np.argsort(c)[::-1][:k_eff]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        # Gradient: grad[j] = 2 * sum_{k in top} w_k * a[k-j]
        # This is a weighted sum of shifted copies of a
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            k = int(k)
            # grad[j] += 2 * w_k * a[k-j] (where 0 <= k-j < n)
            j_start = max(0, k - (n-1))
            j_end = min(n-1, k)
            if j_start <= j_end:
                indices = j_end - j_start + 1
                a_segment = a[k - j_end : k - j_start + 1]  # a[k-j_end], ..., a[k-j_start] = a[k-j] for j in [j_start, j_end]
                grad[j_start:j_end+1] += 2.0 * weights[i] * a_segment
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-20:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        a_new = project_simplex(y - step * grad / gnorm, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0*t*t))
        y = a_new + ((t - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        a = a_new
        t = t_new
        
        # Occasional restart
        if it % 2000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.005*S/np.sqrt(n), n), S)
            t = 1.0
    
    # Final check
    c = full_conv(best_a)
    return best_a, c.max()
