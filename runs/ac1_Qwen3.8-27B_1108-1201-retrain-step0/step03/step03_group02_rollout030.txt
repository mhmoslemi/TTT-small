"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
- Peak-pushing subgradient method with adaptive thresholding
- FISTA acceleration for faster convergence
- Multiple sequence lengths (n) explored
- Backtracking line search for robust step selection
- Periodic noise injection to escape local minima
- Fine polishing with coordinate descent at the end
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full linear self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def grad_topk(a, c, weights):
    """
    Compute gradient of sum_k weights[k] * c_k w.r.t. a.
    grad[m] = 2 * sum_k weights[k] * a[k-m] (for valid indices)
    This is 2 * cross-correlation of (weights, a).
    """
    n = len(a)
    fsize = 1 << (len(weights) + n - 1).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    # cross-correlation: (W * conj(A)) gives sum_k w[k] * a[k-m]
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.zeros(n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_objective(a):
    """Compute 2n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def peak_push_optimize(a_init, n, S, max_time, rng, 
                        init_top_k=8, max_top_k=512,
                        init_threshold_frac=0.5, 
                        noise_period=500, noise_scale=0.01,
                        alpha_accel=True):
    """
    Peak-pushing subgradient optimization with FISTA.
    
    At each step:
    1. Compute convolution c
    2. Find peaks above threshold
    3. Compute weighted subgradient
    4. Take FISTA step with backtracking
    """
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    no_improve_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Adaptive parameters
        frac = min(1.0, it / (max_time * 60))  # rough progress fraction
        top_k = min(max_top_k, int(init_top_k + (max_top_k - init_top_k) * frac))
        # Threshold: start high (only top peaks), decrease to include more
        threshold_frac = init_threshold_frac * (1.0 - 0.5 * frac)
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Compute weights for top peaks
        sorted_idx = np.argsort(c)[::-1][:top_k]
        sorted_vals = c[sorted_idx]
        
        # Threshold: peaks must be above threshold_frac * max
        threshold = threshold_frac * cur_max
        above = sorted_vals > threshold
        if not np.any(above):
            # Use just the top 1
            above = np.zeros(top_k, dtype=bool)
            above[0] = True
        
        # Weights proportional to excess above threshold
        excess = np.maximum(sorted_vals - threshold, 0.0)
        total_excess = excess.sum()
        if total_excess < 1e-30:
            w = np.zeros(top_k)
            w[0] = 1.0
        else:
            w = excess / total_excess
        
        # Build full weight vector on the convolution
        w_full = np.zeros(2 * n - 1)
        for i in range(top_k):
            if above[i]:
                w_full[int(sorted_idx[i])] = w[i]
        
        # Compute gradient
        grad = grad_topk(y, w_full, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Step size with backtracking
        # Initial step: based on problem scale
        step = S / (gnorm * (1.0 + 0.1 * it))
        step = min(step, S / 10.0)
        
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        # Backtracking: ensure we don't increase max too much
        c_new = fft_convolve_self(x_new)
        new_max = c_new.max()
        backtrack_count = 0
        while new_max > cur_max * 1.05 and backtrack_count < 10:
            step *= 0.5
            x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
            c_new = fft_convolve_self(x_new)
            new_max = c_new.max()
            backtrack_count += 1
        
        if new_max < best_score:
            best_score = new_max
            best_local = x_new.copy()
            no_improve_count = 0
        
        # FISTA update
        if alpha_accel:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = x_new + ((t - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t = t_new
        else:
            a = x_new
            y = x_new.copy()
        
        # Noise injection
        if it % noise_period == 0 and rng.random() < 0.3:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            a = y.copy()
            t = 1.0
        
        # Reset acceleration if stuck
        if no_improve_count > 200:
            t = 1.0
            no_improve_count = 0
    
    return best_local, best_score

"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
- Peak-pushing subgradient method with adaptive excess-weighting
- FISTA acceleration with periodic restarts
- Multiple sequence lengths explored (n=8192, 16384)
- Excess-quadratic peak weighting for better peak flattening
- Backtracking line search for robust step selection
- Periodic noise injection and restarts to escape local minima
- Final coordinate-polishing phase for fine-tuning
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def grad_weighted(a, w_full):
    """
    Compute gradient of sum_k w_full[k] * c_k w.r.t. a.
    grad[m] = 2 * sum_k w_full[k] * a[k-m]
    Computed via FFT: 2 * ifft(fft(w_full) * conj(fft(a)))
    """
    n = len(a)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    if n == 1:
        return np.array([S])
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.zeros(n)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_max_conv(a):
    """Compute max of self-convolution."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return c.max()


def normalize_to_sum(a, S):
    """Clamp to [0,1000] and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def peak_push_fista(a_init, n, S, max_time, rng,
                    init_top_k=16, max_top_k=1024,
                    init_thresh=0.6, final_thresh=0.1,
                    noise_period=300, noise_scale=0.008,
                    p_excess=2.0):
    """
    FISTA peak-pushing optimizer with adaptive excess-weighted subgradients.
    
    Key features:
    - Excess-weighted peaks (weights proportional to (c_k - threshold)^p)
    - Adaptive threshold (starts high, decreases over time)
    - Backtracking line search
    - Periodic noise and momentum restarts
    """
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = compute_max_conv(a)
    best_local = a.copy()
    it = 0
    stagnation = 0

    # Precompute FFT size for gradient
    fsize = 1 << ((2 * n - 1) + n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, it / max(1, max_time * 30))

        # Adaptive parameters
        top_k = int(init_top_k + (max_top_k - init_top_k) * progress)
        top_k = max(4, min(top_k, 2 * n - 1))
        thresh_frac = init_thresh + (final_thresh - init_thresh) * progress

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_local = y.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Compute excess-weighted subgradient
        sorted_idx = np.argsort(c)[::-1][:top_k]
        sorted_vals = c[sorted_idx]
        threshold = thresh_frac * cur_max

        excess = np.maximum(sorted_vals - threshold, 0.0)
        # Use excess^p weighting
        weights = excess ** p_excess
        total_w = weights.sum()

        if total_w < 1e-30:
            # Fallback: uniform on top few
            w_full = np.zeros(2 * n - 1)
            n_top = max(1, top_k // 8)
            for i in range(n_top):
                w_full[int(sorted_idx[i])] = 1.0 / n_top
        else:
            w_full = np.zeros(2 * n - 1)
            for i in range(top_k):
                if weights[i] > 0:
                    w_full[int(sorted_idx[i])] = weights[i] / total_w

        # Compute gradient via FFT
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            break

        # Step size: adaptive based on problem scale
        base_step = S / (n * 0.5)
        step = base_step / (1.0 + 0.5 * progress)

        x_new = project_simplex(y - step * grad / gnorm, S)

        # Backtracking: ensure max doesn't increase too much
        c_new = fft_convolve_self(x_new)
        new_max = c_new.max()
        bt = 0
        while new_max > cur_max * 1.02 and bt < 8:
            step *= 0.5
            x_new = project_simplex(y - step * grad / gnorm, S)
            c_new = fft_convolve_self(x_new)
            new_max = c_new.max()
            bt += 1

        if new_max < best_score:
            best_score = new_max
            best_local = x_new.copy()
            stagnation = 0

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t - 1.0) / t_new
        y = x_new + momentum * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Noise injection to escape local minima
        if it % noise_period == 0 and rng.random() < 0.4:
            scale = noise_scale * (1.0 + progress)
            noise = rng.normal(0, scale * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            a = y.copy()
            t = 1.0
            stagnation = 0

        # Reset momentum if stagnating
        if stagnation > 100:
            t = 1.0
            y = a.copy()
            stagnation = 0

    return best_local, best_score


def fine_polish(a_init, n, S, max_time, rng, top_k=256):

    Fine polishing: very small steps, high top_k, minimal noise.
    For the final refinement phase.
    """
    a = np.clip(a_init.astype(np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:

    else:

    y = a.copy()
    t = 1.0
    start = time.monotonic()

    best_local = a.copy()
    it = 0
    fsize = 1 << ((2 * n - 1) + n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_score - 1e-14:
            best_score = cur_max
            best_local = y.copy()

        # Use all significant peaks
        k = min(top_k, 2 * n - 1)
        sorted_idx = np.argsort(c)[::-1][:k]
        sorted_vals = c[sorted_idx]

        # Low threshold to include many peaks
        threshold = 0.05 * cur_max
        excess = np.maximum(sorted_vals - threshold, 0.0)
        total_w = excess.sum()

        if total_w < 1e-30:
            w_full = np.zeros(2 * n - 1)
            w_full[int(sorted_idx[0])] = 1.0
        else:
            w_full = np.zeros(2 * n - 1)
            for i in range(k):
                if excess[i] > 0:
                    w_full[int(sorted_idx[i])] = excess[i] / total_w

        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)

        if gnorm < 1e-15:
            break

        # Small step for fine tuning
        step = S / (n * 2.0 * (1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad / gnorm, S)

        c_new = fft_convolve_self(x_new)
        new_max = c_new.max()
        bt = 0
        while new_max > cur_max * 1.001 and bt < 6:

            c_new = fft_convolve_self(x_new)
            new_max = c_new.max()
            bt += 1

        if new_max < best_score:
            best_score = new_max
            best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t - 1.0) / t_new
        y = x_new + momentum * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    return best_local, best_score


def generate_inits(n, rng):
    """Generate diverse asymmetric initializations."""
    tt = np.linspace(0, 1, n, endpoint=False)
    inits = []

    # Exponential ramps
    for alpha in [1.0, 2.0, 3.0, 5.0, 8.0]:
        inits.append(np.exp(-alpha * tt))
        inits.append(np.exp(alpha * (tt - 1.0)))

    # Power-law
    for p in [0.5, 1.0, 2.0, 4.0]:
        inits.append((tt + 0.01) ** p)
        inits.append(((1.0 - tt) + 0.01) ** p)

    # Strongly front-loaded
    inits.append(np.exp(-8.0 * tt))
    inits.append(np.exp(-12.0 * tt))

    # Bump functions (concentrated in portion)
    for center, width in [(0.2, 0.1), (0.3, 0.15), (0.7, 0.1), (0.8, 0.15)]:
        inits.append(np.exp(-((tt - center) / width) ** 2))

    # Sinusoidal with trend
    inits.append(1.0 + 0.5 * np.sin(2 * np.pi * 3 * tt) + 0.5 * tt)
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 7 * tt) + 0.3 * tt)

    # Random with ramp
    inits.append(rng.random(n) * np.linspace(0.5, 2.0, n))

    # Linear ramp
    inits.append(tt + 0.1)
    inits.append(1.0 - tt + 0.1)

    # Hyperbolic
    inits.append(1.0 / (tt + 0.1))
    inits.append(1.0 / (1.0 - tt + 0.1))

    return inits
