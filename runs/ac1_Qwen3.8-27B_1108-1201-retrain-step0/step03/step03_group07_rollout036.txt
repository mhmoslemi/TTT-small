**Key mathematical observation**: The objective 2n*max(conv(a,a))/sum(a)² is the same as the "peak factor" of the sequence's aperiodic autocorrelation (up to a constant). This is a well-studied quantity in signal processing and radar.

For binary sequences, the minimum peak factor is related to the "sidelobe" level of the autocorrelation. For real-valued non-negative sequences, we have more flexibility.

One interesting result: for sequences with elements from {0, 1}, the minimum peak factor is approximately 2 (achieved by Barker-like sequences). For real-valued sequences, we can do better.

Minimize 2*n*max(conv(a,a))/sum(a)^2 using a hybrid approach:
- Mirror descent with adaptive subgradient (fast iterations)
- LP-based direction finding (occasional precise steps)
- Random perturbations for diversity
- Multi-phase optimization with different strategies

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def obj(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_conv_self(a)
    return 2.0 * n * c.max() / (S * S)

def normalize(a, S=None):
    """Normalize a so that sum(a) = S."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    cur = a.sum()
    if cur < 1e-30:
        return a
    return a * (S / cur)

def subgradient(c, a, top_k, soft_beta=0.0):
    """Compute subgradient of max(c) w.r.t. a."""
    n = len(a)
    k = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    
    if soft_beta > 0:
        logits = soft_beta * (c[top_idx] - c[top_idx[0]])
        w = np.exp(logits)
        w /= w.sum()
    else:
        w = np.full(k, 1.0 / k)
    
    # grad[m] = 2 * sum_j w[j] * a[top_idx[j] - m]
    # This is 2 * conv(w_full, a_rev)[n-1 : 2n-1]
    w_full = np.zeros(len(c))
    w_full[top_idx] = w
    
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w_full, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    
    return 2.0 * full_conv[n-1:2*n-1]

def lp_direction(a, c, n_peaks=10, allow_increase=True):
    """Find best descent direction using LP."""
    n = len(a)
    S = a.sum()
    
    top_idx = np.argsort(c)[::-1][:n_peaks]
    k_target = top_idx[0]
    
    # Build constraint matrix
    # (a * d)[k] = sum_i a[i] * d[k-i] for valid i
    A_mat = np.zeros((n_peaks, n))
    for idx, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_mat[idx, lo:hi+1] = a[k - hi:k - lo + 1][::-1]
    
    # Objective: minimize A_mat[0] @ d (reduce target peak)
    c_obj = A_mat[0].copy()
    
    # Inequality constraints:
    # 1. A_mat[1:] @ d <= 0 (don't increase other peaks)
    # 2. If not allow_increase: d <= 0
    # 3. If allow_increase: |d| <= 0.1 * a (limit relative change)
    
    constraints_A = [A_mat[1:]]
    constraints_b = [np.zeros(n_peaks - 1)]
    
    if allow_increase:
        # d[i] <= 0.1 * a[i] and d[i] >= -0.1 * a[i]
        # -d[i] <= 0.1 * a[i] and d[i] <= 0.1 * a[i]
        constraints_A.append(np.eye(n))
        constraints_b.append(0.1 * a)
        constraints_A.append(-np.eye(n))
        constraints_b.append(0.1 * a)
    else:
        constraints_A.append(np.eye(n))
        constraints_b.append(np.zeros(n))
    
    A_ub = np.vstack(constraints_A)
    b_ub = np.concatenate(constraints_b)
    
    # Equality: sum(d) = 0 (keep sum constant)
    A_eq = np.array([[1.0]] * n).reshape(1, -1)
    b_eq = np.array([0.0])
    
    bounds = [(None, None)] * n
    if not allow_increase:
        bounds = [(None, 0.0)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'maxiter': 5000, 'dual_feasibility_tolerance': 1e-6})
        if result.success and np.linalg.norm(result.x) > 1e-15:
            return result.x
    except Exception:
        pass
    
    # Fallback: uniform decrease of the top contributors
    d = np.zeros(n)
    lo = max(0, k_target - n + 1)
    hi = min(n - 1, k_target)
    d[lo:hi+1] = -a[k_target - hi:k_target - lo + 1][::-1]
    d /= (n - (hi - lo + 1))
    return d

def line_search(a, d, c_best, max_steps=10, alpha_init=1.0, alpha_ratio=0.5):
    """Find the best step size along direction d."""
    n = len(a)

    best_obj = c_best
    best_a = a.copy()
    best_t = 0.0
    alpha = alpha_init

            best_a = a_trial.copy()
            best_t = alpha
            # Try a slightly larger step
            alpha *= 1.2
        else:
            alpha *= alpha_ratio
    
    return best_a, best_obj, best_t
