"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Hybrid spectral-spatial optimization combining:

- Adaptive softmax subgradient (broad peak reduction)

- Multi-phase polishing with momentum and periodic restarts

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def softmax_subgrad(a, c, beta):
    """
    Subgradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a.
    Uses FFT for cross-correlation.
    """
    n = len(a)
    c_max = c.max()
    bc = beta * (c - c_max)
    # Clip to avoid overflow
    bc = np.clip(bc, -700, 700)
    exp_bc = np.exp(bc)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_subgrad(a, c, k):
    """Subgradient from top-k active lags."""
    n = len(a)
    k = min(k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total

    w_full = np.zeros(len(c))
    for i, idx in enumerate(top_indices):
        w_full[int(idx)] = weights[i]

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def all_above_threshold_subgrad(a, c, threshold):
    """Subgradient from all lags above threshold."""
    n = len(a)
    mask = c >= threshold
    if not np.any(mask):
        mask = c >= c.max() * 0.99
    if not np.any(mask):
        return topk_subgrad(a, c, 10)
    vals = c[mask]
    total = vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = vals / total

    w_full = np.zeros(len(c))
    w_full[mask] = weights

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_flatten(a, S, strength=0.3):
    """
    One step of spectral magnitude flattening.
    Modifies |FFT(a)| to be more uniform, then projects back to non-negative.
    """
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    mag = np.abs(A)
    phase = A / np.maximum(mag, 1e-30)
    
    # Target: flatten the magnitude toward the geometric mean
    log_mag = np.log(np.maximum(mag, 1e-30))
    target_log = np.mean(log_mag)
    new_log_mag = (1 - strength) * log_mag + strength * target_log
    new_mag = np.exp(new_log_mag)
    
    # Blend with original to avoid too drastic changes
    A_new = new_mag * phase
    a_new = np.real(ifft(A_new))[:n]
    
    # Project to non-negative and renormalize
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new * (S / s)
    else:
        a_new = np.ones(n) * S / n
    return a_new


def objective_from_conv(c, n):
    """Compute objective from convolution (assuming sum = sqrt(2n))."""
    return c.max()  # Since 2n * max_c / S^2 = max_c when S = sqrt(2n)


def run_optimizer(a_init, n, S, max_time, rng, mode="hybrid"):
    """
    Multi-phase optimizer with backtracking line search.
    mode: 'hybrid', 'softmax', 'topk', 'spectral'
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    no_improve = 0
    best_no_improve_score = a.sum() * 0  # dummy

    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()
    best_no_improve_score = best_score

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic momentum reset if stuck
        if no_improve > 5000:
            t_momentum = 1.0
            y = a + 0.01 * S / np.sqrt(n) * rng.normal(0, 1, n)
            y = project_simplex(y, S)
            no_improve = 0
            continue

        # Adaptive parameters
        if mode == "hybrid":
            # Alternate between softmax and topk
            if it % 4 < 2:
                beta = min(500.0, 5.0 * (1.0 + 0.01 * it))
                grad = softmax_subgrad(y, c, beta)
            else:
                k = min(512, 10 + it // 20)
                grad = topk_subgrad(y, c, k)
        elif mode == "softmax":
            beta = min(500.0, 5.0 * (1.0 + 0.01 * it))
            grad = softmax_subgrad(y, c, beta)
        elif mode == "topk":
            k = min(512, 10 + it // 20)
            grad = topk_subgrad(y, c, k)
        else:
            grad = topk_subgrad(y, c, 32)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14 * S:
            break

        # Backtracking line search
        step = S / (2.0 * gnorm + 1e-15)
        accepted = False
        for _ in range(8):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_max * (1.0 - 1e-7 * step * gnorm / S) + 1e-12:
                accepted = True
                break
            step *= 0.5

        if not accepted:
            # Accept anyway if it's not much worse
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() > cur_max * 1.05:
                # Take a smaller step
                step *= 0.1
                x_new = project_simplex(y - step * grad, S)

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 0:
            y = y * (S / s_y)
        a = x_new
        t_momentum = t_new

        # Occasional spectral flattening step
        if mode == "hybrid" and it % 200 == 0 and rng.random() < 0.3:
            a_spec = spectral_flatten(a, S, strength=0.2)
            c_spec = fft_convolve_self(a_spec)
            if c_spec.max() < best_score:
                best_score = c_spec.max()
                best_local = a_spec.copy()
                a = a_spec
                y = a_spec.copy()
                t_momentum = 1.0

        # Noise injection for exploration
        if it % 5000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_momentum = 1.0

    return best_local, best_score


def run_spectral_phase(a_init, n, S, max_time, rng):
    """Dedicated spectral flattening phase."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()

    best_local = a.copy()
    strength = 0.3
    it = 0

    c = fft_convolve_self(a)
    best_score = c.max()
    best_local = a.copy()

    while time.monotonic() - start < max_time:
        it += 1
        # Decay strength over time
        elapsed_frac = (time.monotonic() - start) / max_time
        s_current = max(0.05, strength * (1.0 - elapsed_frac))
        
        a_new = spectral_flatten(a, S, strength=s_current)
        c_new = fft_convolve_self(a_new)
        
        if c_new.max() < best_score:
            best_score = c_new.max()
            best_local = a_new.copy()
            a = a_new
        else:
            # Try a gradient step instead
            grad = topk_subgrad(a, c_new, 32)
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-14:
                step = S / (2.0 * gnorm + 1e-15)
                a_grad = project_simplex(a - step * grad, S)
                c_grad = fft_convolve_self(a_grad)
                if c_grad.max() < best_score:
                    best_score = c_grad.max()
                    best_local = a_grad.copy()
                    a = a_grad
                else:
                    a = a_new  # Keep the spectral step even if slightly worse

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

            # Final ultra-fine polish
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_f, s_f = run_optimizer(global_a, n_best, S_best, remaining2, rng, mode="topk")
                a_f = safe_normalize(a_f, n_best)
                c_f = fft_convolve_self(a_f)
                if c_f.max() < global_best:
                    global_best = c_f.max()
                    global_a = a_f.copy()
                    print("  Final polish: %.6f" % global_best)

    # --- Verification and return ---
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * 1.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64) * 1.0

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        global_a = np.ones(8192, dtype=np.float64) * 1.0
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d, internal score=%.6f, verified=%.6f" %
          (len(final_list), global_best, ver if np.isfinite(ver) else -1))
    return final_list
