"""
Minimize 2*n*max_conv / sum^2 using FISTA-LSE with adaptive parameters, 
Nesterov restarts, peak deflation, and diverse initialization.
Key improvements over previous: adaptive beta, backtracking step size,
Nesterov restarts, peak deflation heuristic, and softmax-based perturbations.

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a, N=None):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    if N is None:
        N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))[:size]
    return c, N

def compute_grad_lse(a, c, beta, N):
    """Gradient of LSE(c(a)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(A) * W))[:n]
    return grad, c_max

def fista_lse_adaptive(a_init, n, S, max_time, rng, 
                        beta_init=20.0, beta_final=500.0, 
                        step_init=0.01, restart_threshold=1.0):
    """
    FISTA with LSE penalty, adaptive beta, Nesterov restarts, 
    and backtracking step size.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    N = 1 << (2 * n - 1).bit_length()
    
    y = a.copy()
    t = 1.0
    step = step_init
    beta = beta_init
    beta_growth = (beta_final / beta_init) ** (0.7)  # per 1000 iters
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    obj_prev = np.inf
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        if it % 1000 == 0:
            beta = min(beta_final, beta * beta_growth)
        
        # Compute convolution and objective
        c = np.zeros(2*n-1)
        fsize = 1 << (2*n-1).bit_length()
        A = fft(y, fsize)
        c = np.real(ifft(A * A))[:2*n-1]
        c_max = c.max()
        
        LSE = c_max + np.log(np.exp(beta * (c - c_max)).sum()) / beta
        
        if c_max < best_score:
            best_score = c_max
            best_local = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Nesterov restart
        if LSE > obj_prev * restart_threshold:
            t = 1.0
            y = a.copy()
        
        obj_prev = LSE
        
        # Gradient
        c_max_local = c.max()
        exp_bc = np.exp(beta * (c - c_max_local))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total
        
        fsize2 = 1 << (len(w) + n - 2).bit_length()
        W = fft(w, fsize2)
        A2 = fft(y, fsize2)
        grad = 2.0 * np.real(ifft(np.conj(A2) * W))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking step size
        # L = Lipschitz constant estimate (use spectral radius of Hessian approx)
        # For now, use a simple decrease
        x_new = project_simplex(y - step * grad, S)
        
        # Check if step was too big (optional: compute LSE at x_new)
        # For efficiency, skip this and just use adaptive step
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Adaptive step: decrease if no improvement, increase if stuck
        if no_improve > 500:
            step *= 0.8
            no_improve = 0
        elif it % 5000 == 0 and c_max > 1.4:
            # Periodic perturbation to escape
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)
            y = a.copy()
            t = 1.0
            step *= 0.9
    
    return best_local, best_score


def peak_deflation(a, n, S, max_time, rng):
    """
    Peak deflation: iteratively reduce the elements that contribute most 
    to the peak autocorrelation lag.
    """
    a = np.clip(a, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    fsize = 1 << (2*n-1).bit_length()
    
    while time.monotonic() - start < max_time:
        A = fft(a, fsize)
        c = np.real(ifft(A * A))[:2*n-1]
        c_max = c.max()
        
        if c_max < best_score:
            best_score = c_max
            best_local = a.copy()
        
        # Find peak lag
        k_peak = int(np.argmax(c))
        
        # Find contributing elements
        # c[k_peak] = sum_{i} a[i] * a[k_peak - i]
        # Valid i: max(0, k_peak - n + 1) <= i <= min(k_peak, n-1)
        i_min = max(0, k_peak - n + 1)
        i_max = min(k_peak, n - 1)
        
        if i_max < i_min:
            break
        
        # Compute contributions
        i_range = np.arange(i_min, i_max + 1)
        contributions = a[i_range] * a[k_peak - i_range]
        total_contrib = contributions.sum()
        
        if total_contrib < 1e-10:
            break
        
        # Find the top contributors
        top_idx_in_range = np.argsort(contributions)[::-1][:min(10, len(i_range))]
        top_global_idx = i_range[top_idx_in_range]
        
        # Reduce the top contributors
        reduction = 0.01 * total_contrib / c_max  # small reduction
        for idx in top_global_idx[:5]:
            a[idx] *= (1.0 - 0.02)  # reduce by 2%
        
        # Renormalize
        s = a.sum()
        if s < 1e-15:
            break
        a = a * (S / s)
    
    return best_local, best_score
