def fft_convolve_self(a):
    n = len(a)

    fsize = 1 << (size-1).bit_length()
    A = fft(a, fsize)

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def run_toppk_subgrad(a, n, S, top_k, max_iter, time_limit, start_time=None):
    """Top-K subgradient method for the max function."""
    if start_time is None:
        start_time = time.monotonic()
    
    a = np.clip(a.copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a * (S / s)
    
    best_a = a.copy()
    best_c = fft_convolve_self(a).max()
    step = S / (2.0 * np.sqrt(n))
    
    for it in range(max_iter):
        if time.monotonic() - start_time > time_limit:
            break
        
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_c:
            best_c = cur
            best_a = a.copy()
        
        # Top-K subgradient
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        w_sparse = np.zeros(2*n-1)
        for i, k in enumerate(top_idx):
            w_sparse[int(k)] = top_vals[i] / top_vals.sum()
        
        fsize = 1 << (3*n).bit_length()
        W = fft(w_sparse, fsize)
        A = fft(a, fsize)
        g = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
