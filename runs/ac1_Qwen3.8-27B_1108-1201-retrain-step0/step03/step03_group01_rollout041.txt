import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def compute_score(a):
    """Compute the score 2n * max(conv) / sum^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))[:size]
    return 2.0 * n * c.max() / (S ** 2)

def ifta_step(a, N, S, beta, rng):
    """
    One step of IFTA-like spectral flattening.
    In frequency domain: flatten |A|^2 using softmax with temperature beta.
    In time domain: project to non-negative simplex.
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    
    # Flatten: P_new = P^alpha / sum(P^alpha) * sum(P)
    # This is equivalent to reducing the "temperature" of the spectrum
    # alpha < 1 flattens, alpha > 1 sharpens
    # We want to flatten, so use alpha < 1
    P_pos = np.maximum(P, 1e-30)
    P_alpha = P_pos ** (1.0 / beta)  # beta > 1 flattens
    P_new = P_alpha / P_alpha.sum() * P.sum()
    
    # Preserve phase
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new

"""
Minimize 2*n*max_conv/sum^2 using LSE-smoothed gradient descent with backtracking
line search, IFTA spectral equalization, and Nesterov acceleration.

Key improvements over previous approach:
1. Backtracking ARMS line search for adaptive step sizes
2. Full LSE gradient via FFT (no top-k approximation)
3. Temperature annealing (beta scheduling)
4. IFTA-style power spectrum equalization as global moves
5. Multi-scale: try different sequence lengths
6. Better initialization from known low-autocorrelation structures
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_lse_grad(a, c, fsize, beta):
    """
    Compute gradient of LSE(c; beta) w.r.t. a.
    LSE(c; beta) = (1/beta) * log(sum(exp(beta * c)))
    grad[i] = 2 * sum_k w[k] * (a[k-i] + a[i-k]) where w = softmax(beta*c)
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max
    w = exp_bc / total

    # Compute grad via FFT
    w_pad = np.zeros(fsize)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(fsize)
    a_pad[:n] = a

    W = fft(w_pad)
    A = fft(a_pad)

    # corr[m] = sum_k w[k] * a[k-m]  (cross-correlation)
    corr = np.real(ifft(W * np.conj(A)))[:n]
    # conv[m] = sum_k w[k] * a[m-k]  (convolution)
    conv = np.real(ifft(W * A))[:n]

    grad = 2.0 * (corr + conv)
    return grad, c_max


def lse_value(c, beta):
    """Compute LSE(c; beta) = (1/beta) * log(sum(exp(beta*c)))"""
    c_max = c.max()
    if c_max == 0:
        return 0.0
    exp_bc = np.exp(beta * (c - c_max))
    return c_max + np.log(exp_bc.sum()) / beta


def score_from_conv(c, n):
    """Compute score from convolution: 2n * max(c) / (sum)^2.
    Here we assume sum(a) = sqrt(2n) for normalization, so score = max(c).
    Actually let's just return 2n*max(c)/S^2 where S is the sum.
    But since we normalize, let's just return max(c) when S=sqrt(2n).
    """
    return c.max()


def ifta_equality_step(a, N, S, power, rng):
    """
    IFTA-style step: modify |FFT(a)|^2 to be flatter, then project.
    power < 1 flattens the spectrum.
    """
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    P_total = P.sum()

    # Flatten: P_new = P^power / sum(P^power) * P_total
    P_pos = np.maximum(P, 1e-30)
    P_powered = P_pos ** power
    P_powered_sum = P_powered.sum()
    if P_powered_sum < 1e-30:
        return a
    P_new = P_powered / P_powered_sum * P_total

    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def lse_gradient_descent(a_init, n, S, fsize, max_time, rng,
                         beta_init=10.0, beta_max=1000.0,
                         alpha_init=1.0, rho=0.5, c_armijo=1e-4,
                         noise_start=0, noise_freq=0, noise_scale=0):
    """
    LSE-smoothed gradient descent with backtracking ARMS line search.
    Beta increases over time (temperature annealing).
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start_time = time.monotonic()
    best_a = a.copy()
    best_score = np.inf
    alpha = alpha_init
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        # Temperature schedule: beta increases linearly then stays at max
        progress = min(1.0, it / max(1, int(max_time / 0.01)))  # rough estimate
        beta = min(beta_max, beta_init * (1.0 + it * 0.05))

        # Compute convolution and gradient
        c = fft_convolve_self(a)
        cur_score = c.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        grad, c_max = compute_lse_grad(a, c, fsize, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        f_curr = lse_value(c, beta)
        best_alpha_trial = 0
        best_a_trial = a.copy()
        best_f_trial = f_curr

        for _ in range(30):
            alpha_trial = alpha
            a_trial = project_simplex(a - alpha_trial * grad / max(gnorm, 1e-15), S)
            c_trial = fft_convolve_self(a_trial)
            f_trial = lse_value(c_trial, beta)

            # Armijo condition
            # directional derivative: <grad, a_trial - a> / ||grad|| ... 
            # Actually for projected gradient, use: f(a_trial) <= f(a) - c * alpha * ||grad||^2 / ||grad||
            # Simplified: f(a_trial) <= f(a) - c_armijo * alpha_trial * gnorm
            if f_trial < best_f_trial:
                best_f_trial = f_trial
                best_a_trial = a_trial.copy()
                best_alpha_trial = alpha_trial

            if f_trial <= f_curr - c_armijo * alpha_trial * gnorm:
                break
            alpha_trial *= rho

        a = best_a_trial
        alpha = min(alpha * 2.0, alpha_init * 2.0)  # grow step size (capped)

        # Periodic noise injection
        if noise_freq > 0 and it % noise_freq == noise_start and rng.random() < 0.5:
            a = project_simplex(a + rng.normal(0, noise_scale * S / np.sqrt(n), n), S)

    return best_a, best_score


def nesterov_lse(a_init, n, S, fsize, max_time, rng,
                 beta_init=50.0, beta_max=2000.0, step_init=0.5):
    """
    Nesterov-accelerated LSE gradient descent with fixed (decaying) step.
    For final polishing when we're close to the optimum.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()

def ifta_phase(a, n, S, N, max_time, rng, power_init=0.5, power_min=0.8):

def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

"""
Minimize 2*n*max_conv/sum^2 using:
1. IFTA-style spectral equalization (global moves)
2. LSE-smoothed gradient descent with backtracking ARMS line search (local)
3. Nesterov-accelerated LSE for polishing
4. Adaptive temperature (beta) scheduling
5. Multi-start with diverse initializations
6. Full gradient via FFT cross-correlation (no top-k truncation)

Key improvements over previous:
- Backtracking line search replaces fixed decaying step
- Adaptive beta: low when far from optimum, high when close
- IFTA power-spectrum flattening as complementary global moves
- Better initialization diversity

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_lse_grad(a, c, fsize, beta):
    """
    Compute gradient of LSE(c; beta) w.r.t. a.
    grad[i] = 2 * sum_k w[k] * (a[k-i] + a[i-k]) where w = softmax(beta*c)
    Uses FFT for efficiency. Requires fsize >= len(w) + len(a) - 1 = 3n - 2.
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    w_pad = np.zeros(fsize)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(fsize)
    a_pad[:n] = a

    W = fft(w_pad)
    A = fft(a_pad)

    # Cross-correlation: ifft(W * conj(A))[m] = sum_k w[k] * a[k-m]
    corr = np.real(ifft(W * np.conj(A)))[:n]
    # Convolution: ifft(W * A)[m] = sum_k w[k] * a[m-k]
    conv = np.real(ifft(W * A))[:n]

    grad = 2.0 * (corr + conv)
    return grad


def lse_value(c, beta):
    """Compute LSE(c; beta) = (1/beta) * log(sum(exp(beta*c)))"""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    return c_max + np.log(exp_bc.sum()) / beta


def ifta_step(a, N_fft, S, power):
    """
    IFTA-style step: flatten |FFT(a)|^2 by raising to power < 1,
    preserve phase, project to simplex.
    """
    n = len(a)
    a_pad = np.zeros(N_fft)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    P_total = P.sum()

    P_pos = np.maximum(P, 1e-30)
    P_powered = P_pos ** power
    P_sum = P_powered.sum()
    if P_sum < 1e-30:
        return a
    P_new = P_powered / P_sum * P_total

    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)

    c = fft_convolve_self(a_curr)
    if c.max() < best_score:
        best_score = c.max()
        best_a = a_curr.copy()
    return best_a, best_score


def lse_gd_linesearch(a_init, n, S, fsize_grad, fsize_conv, max_time, rng,
                      beta_init=10.0, beta_max=500.0,
                      alpha_init=2.0, alpha_max=10.0, rho=0.5, c_armijo=1e-4):
    """
    LSE-smoothed gradient descent with backtracking ARMS line search.
    Beta increases over time (temperature annealing).
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start_time = time.monotonic()
    best_a = a.copy()
    best_score = np.inf
    alpha = alpha_init
    it = 0
    stall = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        # Adaptive beta: increase over time, faster when score is high
        beta = min(beta_max, beta_init * (1.0 + 0.03 * it))

        # Compute convolution
        c = fft_convolve_self(a)
        cur_score = c.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # If stalled a lot, add noise and reset alpha
        if stall > 200:
            a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            alpha = alpha_init
            stall = 0
            continue

        # Compute gradient
        grad = compute_lse_grad(a, c, fsize_grad, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        f_curr = lse_value(c, beta)
        a_best_trial = a.copy()
        f_best_trial = f_curr
        alpha_curr = min(alpha, alpha_max)

        for _ in range(25):
            a_trial = project_simplex(a - alpha_curr * grad / max(gnorm, 1e-15), S)
            c_trial = fft_convolve_self(a_trial)
            f_trial = lse_value(c_trial, beta)

            if f_trial < f_best_trial:
                f_best_trial = f_trial
                a_best_trial = a_trial.copy()

            # Armijo sufficient decrease
            if f_trial <= f_curr - c_armijo * alpha_curr * gnorm / max(1.0, gnorm):
                break
            alpha_curr *= rho

        a = a_best_trial

        # Grow alpha if we took a full step
        if f_best_trial < f_curr - c_armijo * alpha * gnorm:
            alpha = min(alpha * 1.5, alpha_max)
        else:
            alpha = max(alpha * 0.7, alpha_init * 0.5)

    return best_a, best_score


def nesterov_lse(a_init, n, S, fsize_grad, fsize_conv, max_time, rng,
                 beta_init=100.0, beta_max=3000.0, step_init=0.3):
    """
    Nesterov-accelerated projected gradient on LSE objective.
    For final polishing with high beta.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    step = step_init
    start_time = time.monotonic()
    best_a = a.copy()
    best_score = np.inf
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()

        grad = compute_lse_grad(y, c, fsize_grad, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = project_simplex(y - step * grad / gnorm, S)

        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_a = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        step *= 0.9998

        if it % 5000 == 4500 and rng.random() < 0.15:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_a, best_score


def spectral_grad_polish(a_init, n, S, N_fft, max_time, rng, top_k=20):
    """
    Spectral shaping with top-k lag identification and adaptive alpha.
    Complements gradient methods by making global spectral changes.
    """
    a_curr = a_init.copy()
    S_curr = a_curr.sum()
    start_time = time.monotonic()
    best_a = a_curr.copy()
    best_score = fft_convolve_self(a_curr).max()
    it = 0

    while time.monotonic() - start_time < max_time:
        it += 1

        a_pad = np.zeros(N_fft)
        a_pad[:n] = a_curr
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = c.max()

        if c_max < best_score:
            best_score = c_max
            best_a = a_curr.copy()

        # Identify top-k lags
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # Spectral deflation
        m = np.arange(N_fft)
        cos_eff = np.zeros(N_fft)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N_fft)

        # Adaptive alpha
        excess = max(0.0, c_max - 0.9) / max(c_max, 1e-10)
        alpha = 0.15 * (0.3 + 0.7 * excess) / (1.0 + 0.05 * it)

        P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_curr = project_simplex(a_new, S_curr)

        if it % 2000 == 1900 and rng.random() < 0.2:
            a_curr = project_simplex(a_curr + rng.normal(0, 0.005 * S_curr / np.sqrt(n), n), S_curr)

    return best_a, best_score


def make_initializations(n, rng):
    """Create diverse initial sequences."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(("uniform", np.ones(n) * 0.5))

    # Pure random
    inits.append(("random", rng.random(n) + 0.01))

    # Smooth random
    r = rng.standard_normal(n)
    fsize = 1 << (n - 1).bit_length()
    R = fft(r, fsize)
    freqs = np.arange(fsize)
    cutoff = fsize // 4
    mask = np.ones(fsize)
    mask[cutoff:fsize - cutoff] = 0.0
    R *= mask
    r_smooth = np.real(ifft(R))[:n]
    r_smooth = np.clip(r_smooth + abs(r_smooth.min()) + 0.01, 0.01, None)
    inits.append(("smooth", r_smooth))

    # Multi-frequency sine
    s = 0.5
    for k in range(1, 8):
        s += 0.1 / k * np.sin(2 * np.pi * k * tt + rng.random() * 2 * np.pi)
    s = np.clip(s, 0.01, None)
    inits.append(("sines", s))

    # Chirp
    s = 0.5 + 0.3 * np.sin(2 * np.pi * (3.0 * tt + 0.5 * tt * tt * 10.0))
    s = np.clip(s + 0.05 * rng.standard_normal(n), 0.01, None)
    inits.append(("chirp", s))

    # Wide blocks
    n_blocks = 16
    blen = n // n_blocks
    s = np.zeros(n)
    for i in range(n_blocks):
        s[i * blen:min((i + 1) * blen, n)] = 0.3 + 0.4 * rng.random()
    inits.append(("blocks", s))

    # Inverse triangular
    s = 0.4 + 0.5 * (1.0 - np.sin(np.pi * tt))
    inits.append(("inv_tri", s))

    # Two-segment
    s = np.zeros(n)
    s[:n // 2] = 1.0
    s[n // 2:] = 0.3
    inits.append(("two_seg", s))

    return inits


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing evaluate_sequence.
    Combines IFTA spectral equalization, LSE gradient descent with line search,
    spectral deflation, and Nesterov acceleration.
    """
    global_best = np.inf
    global_a = None
    rng = np.random.default_rng(seed)

    # Determine sequence length
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        main_len = len(given) if given.size >= 64 else 8192
    except Exception:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * float(n))
    N_fft = 1 << (2 * n - 1).bit_length()       # For convolution and IFTA
    fsize_grad = 1 << (3 * n - 3).bit_length()  # For gradient computation

    print("n=%d, S=%.4f, N_fft=%d, fsize_grad=%d" % (n, S, N_fft, fsize_grad))

    deadline = time.monotonic() + max(0.0, budget_s - 25.0)
    t_start = time.monotonic()
    total_budget = deadline - t_start

    # --- Phase 1: Exploration with diverse initializations ---
    print("=== Phase 1: Exploration ===")
    inits = make_initializations(n, rng)

    # Add given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.insert(0, ("given", np.clip(given, 0, 1000).astype(np.float64)))
    except Exception:
        pass

    n_inits = len(inits)
    explore_budget = total_budget * 0.30
    per_init = explore_budget / n_inits

    phase1_end = t_start + explore_budget
    for name, arr in inits:
        if time.monotonic() > phase1_end - 10:
            break
        a_norm = safe_normalize(arr, n)
        t0 = time.monotonic()

        # Quick IFTA
        ifta_t = min(per_init * 0.25, 20.0)
        if ifta_t > 1:
            a_if, sc_if = ifta_phase(a_norm, n, S, N_fft, ifta_t, rng,
                                      power_init=0.3, power_end=0.7)
            a_if = safe_normalize(a_if, n)
            c_if = fft_convolve_self(a_if)
            if c_if.max() < global_best:
                global_best = c_if.max()
                global_a = a_if.copy()
                print("  [%s] IFTA: %.6f" % (name, global_best))

        # LSE GD with line search
        rem = per_init - (time.monotonic() - t0)
        if rem > 3:
            a_ld, _ = lse_gd_linesearch(a_if, n, S, fsize_grad, N_fft, rem, rng,
                                         beta_init=5.0, beta_max=200.0,
                                         alpha_init=2.0, alpha_max=8.0)
            a_ld = safe_normalize(a_ld, n)
            c_ld = fft_convolve_self(a_ld)
            if c_ld.max() < global_best:
                global_best = c_ld.max()
                global_a = a_ld.copy()
                print("  [%s] LSE-GD: %.6f" % (name, global_best))

    print("=== Phase 1 done. Best: %.6f ===" % global_best)

    # --- Phase 2: Focused optimization ---
    if global_a is None:
        global_a = safe_normalize(np.ones(n), n)
        global_best = 2.0

    print("=== Phase 2: Focused optimization ===")
    focus_budget = total_budget * 0.45
    t_phase2 = time.monotonic()
    phase2_end = t_phase2 + focus_budget

    a_curr = global_a.copy()
    it_outer = 0
    stall_outer = 0

    while time.monotonic() < phase2_end:
        it_outer += 1
        rem = phase2_end - time.monotonic()
        if rem < 10:
            break

        # Adaptive parameters based on progress
        beta_init = 10.0 + 20.0 * it_outer
        beta_max = 200.0 + 200.0 * it_outer

        # LSE gradient descent block
        block_t = min(rem * 0.4, 150.0)
        a_new, _ = lse_gd_linesearch(a_curr, n, S, fsize_grad, N_fft, block_t, rng,
                                      beta_init=beta_init, beta_max=beta_max,
                                      alpha_init=1.5, alpha_max=6.0)
        a_new = safe_normalize(a_new, n)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < global_best:
            global_best = c_new.max()
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall_outer = 0
            if it_outer % 3 == 0:
                print("  [P2-%d] LSE-GD: %.6f" % (it_outer, global_best))
        else:
            stall_outer += 1

        # Spectral deflation block
        rem2 = phase2_end - time.monotonic()
        if rem2 > 15:
            spec_t = min(rem2 * 0.3, 60.0)
            a_sp, _ = spectral_grad_polish(a_curr, n, S, N_fft, spec_t, rng,
                                            top_k=10 + 5 * it_outer)
            a_sp = safe_normalize(a_sp, n)
            c_sp = fft_convolve_self(a_sp)
            if c_sp.max() < global_best:
                global_best = c_sp.max()
                global_a = a_sp.copy()
                a_curr = a_sp.copy()
                stall_outer = 0
                print("  [P2-%d] Spectral: %.6f" % (it_outer, global_best))

        # IFTA block
        rem3 = phase2_end - time.monotonic()
        if rem3 > 10:
            ifta_t = min(rem3 * 0.2, 30.0)
            a_if, _ = ifta_phase(a_curr, n, S, N_fft, ifta_t, rng,
                                  power_init=0.6 + 0.05 * it_outer,
                                  power_end=0.85 + 0.02 * min(it_outer, 5))
            a_if = safe_normalize(a_if, n)
            c_if = fft_convolve_self(a_if)
            if c_if.max() < global_best:
                global_best = c_if.max()
                global_a = a_if.copy()
                a_curr = a_if.copy()
                stall_outer = 0
                print("  [P2-%d] IFTA: %.6f" % (it_outer, global_best))

        # Nesterov
        rem4 = phase2_end - time.monotonic()
        if rem4 > 15:
            ng_t = min(rem4 * 0.3, 60.0)
            a_ng, _ = nesterov_lse(a_curr, n, S, fsize_grad, N_fft, ng_t, rng,
                                    beta_init=50 + 30 * it_outer,
                                    beta_max=500 + 200 * it_outer,
                                    step_init=0.3 / (1.0 + 0.1 * it_outer))
            a_ng = safe_normalize(a_ng, n)
            c_ng = fft_convolve_self(a_ng)
            if c_ng.max() < global_best:
                global_best = c_ng.max()
                global_a = a_ng.copy()
                a_curr = a_ng.copy()
                stall_outer = 0
                print("  [P2-%d] Nesterov: %.6f" % (it_outer, global_best))
            else:
                stall_outer += 1

        # Perturb if stalled
        if stall_outer >= 4:
            print("  [P2-%d] Stalled, perturbing" % it_outer)
            a_curr = project_simplex(a_curr + rng.normal(0, 0.015 * S / np.sqrt(n), n), S)
            stall_outer = 0

    print("=== Phase 2 done. Best: %.6f ===" % global_best)

    # --- Phase 3: Final polishing ---
    print("=== Phase 3: Polishing ===")
    t_phase3 = time.monotonic()

    if global_a is not None:
        # High-beta Nesterov
        rem = deadline - time.monotonic()
        seg1 = min(rem * 0.35, 300.0)
        a_p1, _ = nesterov_lse(global_a, n, S, fsize_grad, N_fft, seg1, rng,
                               beta_init=300.0, beta_max=5000.0, step_init=0.1)
        a_p1 = safe_normalize(a_p1, n)
        c1 = fft_convolve_self(a_p1)
        if c1.max() < global_best:
            global_best = c1.max()
            global_a = a_p1.copy()
            print("  Polish Nesterov: %.6f" % global_best)

        # LSE GD with very high beta
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            seg2 = min(rem2 * 0.35, 300.0)
            a_p2, _ = lse_gd_linesearch(global_a, n, S, fsize_grad, N_fft, seg2, rng,
                                         beta_init=500.0, beta_max=5000.0,
                                         alpha_init=0.5, alpha_max=3.0)
            a_p2 = safe_normalize(a_p2, n)
            c2 = fft_convolve_self(a_p2)
            if c2.max() < global_best:
                global_best = c2.max()
                global_a = a_p2.copy()
                print("  Polish LSE-GD: %.6f" % global_best)

        # Spectral fine-tuning
        rem3 = deadline - time.monotonic()
        if rem3 > 5:
            a_p3, _ = spectral_grad_polish(global_a, n, S, N_fft, min(rem3 * 0.3, 120.0),
                                            rng, top_k=30)
            a_p3 = safe_normalize(a_p3, n)
            c3 = fft_convolve_self(a_p3)
            if c3.max() < global_best:
                global_best = c3.max()
                global_a = a_p3.copy()
                print("  Polish Spectral: %.6f" % global_best)

        # Final Nesterov with highest beta
        rem4 = deadline - time.monotonic()
        if rem4 > 3:
            a_p4, _ = nesterov_lse(global_a, n, S, fsize_grad, N_fft, rem4, rng,
                                    beta_init=1000.0, beta_max=10000.0, step_init=0.05)
            a_p4 = safe_normalize(a_p4, n)
            c4 = fft_convolve_self(a_p4)
            if c4.max() < global_best:
                global_best = c4.max()
                global_a = a_p4.copy()
                print("  Final Nesterov: %.6f" % global_best)

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.6f" % ver)
        else:
            print("Verification: inf - using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
