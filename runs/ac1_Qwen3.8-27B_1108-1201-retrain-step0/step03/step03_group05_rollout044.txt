        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Determine active set and weights
        if K_frac < 1.0:
            K = max(1, int(K_frac * conv_len))
            top_idx = np.argsort(c)[::-1][:K]
            w = np.zeros(conv_len)
            if w_type == 'proportional':
                vals = c[top_idx]
                total = vals.sum()
                if total > 1e-30:
                    w[top_idx] = vals / total
                else:
                    w[top_idx] = 1.0 / K
            else:
                w[top_idx] = 1.0 / K
        else:
            total_c = c.sum()
            if w_type == 'proportional':
                w = np.maximum(c, 0) / max(total_c, 1e-30)
            else:
                w = np.ones(conv_len) / conv_len
        
        # Adaptive dampening
        if schedule == 'annealing':
            t = dampening * (0.1 + 0.9 * (1.0 - it / max(max_iter, 1)))
        elif schedule == 'constant':
            t = dampening
        else:
            t = dampening
        
        # Frequency domain step
        a_natural = freq_domain_solve(w, fsize, D_rect, S, n)
        
        # Project to non-negative
        a_new = np.maximum(a_natural, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * S / s_new
        
        # Dampened update
        a = (1.0 - t) * a + t * a_new
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a * S / s
    
    # Final check
    c = fft_convolve_self(a)
    M = c.max()
    score = 2.0 * n * M / (S * S)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z (unconstrained softmax params)."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    # grad_a[j] = 2 * sum_k w[k] * a[j-k] (cross-correlation)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs_polish(a_init, n, S, max_time, rng, beta_schedule=None, maxiter_per_beta=200):
    """L-BFGS-B with log-sum-exp continuation for polishing."""
    if beta_schedule is None:
        beta_schedule = [50, 100, 200, 500, 1000, 2000, 5000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break

        def fun(z_val):
            return softmax_obj_grad(z_val, beta, S)[0]

        def jac(z_val):
            return softmax_obj_grad(z_val, beta, S)[1]

        try:
            result = scipy_minimize(
                fun, z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-14,
                         'gtol': 1e-10, 'maxls': 40, 'iprint': -1}
            )
            z = result.x
            z = z - z.mean()
        except Exception:
            break

        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = S * exp_z / exp_z.sum()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a * S / s

        c = fft_convolve_self(a)
        if c.max() < best_score:
            best_score = c.max()
            best_a = a.copy()

    return best_a, best_score


def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)
