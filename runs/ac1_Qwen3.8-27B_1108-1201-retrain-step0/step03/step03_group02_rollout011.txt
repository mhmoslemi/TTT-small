    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)

        # Nesterov momentum update
        x_new = project_simplex(y - alpha * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_next = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y_next = np.maximum(y_next, 0.0)

        # Periodic momentum reset
        if stall_count > 200 or it % 400 == 0:
            t_mom = 1.0
            y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            stall_count = 0
            # Add noise to escape local minima
            if stall_count > 0 or rng.random() < 0.3:
                noise_scale = 0.005 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
        else:
            y = y_next

        a = x_new
        t_mom = t_new

        # Adaptive step: if long stall, increase exploration
        if stall_count > 100:
            alpha = max(alpha_min, alpha * 0.8)
            if stall_count > 200:
                noise_scale = 0.01 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_mom = 1.0
                a = y.copy()

    return best_local, best_score


def run_peak_shave(a_init, n, S, max_time, rng):
    """
    Targeted peak-shaving: identify top peaks, reduce contributing elements,
    redistribute mass to non-contributing positions.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()

        if cur_max < best_score:
            best_score = cur_max
            best_local = a.copy()

        # Find top peaks
        n_peaks = min(8, 2 + it // 50)
        peak_lags = np.argsort(c)[::-1][:n_peaks]

        # Compute reduction targets
        reduction = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            # Contributions: a[i] * a[k-i]
            ii = np.arange(i_start, i_end + 1)
            contribs = a[ii] * a[k - ii]
            # Find top contributors
            n_top = max(1, len(ii) // 50)
            top_local = np.argsort(contribs)[::-1][:n_top]
            for ti in top_local:
                i = ii[ti]
                j = k - i
                # Reduce proportionally to reduce product
                reduce_amt = 0.05 * min(a[i], a[j])
                reduction[i] += reduce_amt
                if j != i:
                    reduction[j] += reduce_amt

        # Apply reduction
        a_new = a - reduction
        a_new = np.maximum(a_new, 0.0)

        # Redistribute saved mass
        saved = S - a_new.sum()
        if saved > 1e-12:
            # Find "safe" positions (low contribution to peaks)
            # Heuristic: positions with below-average value
            mean_val = a_new.sum() / n
            safe_mask = a_new < mean_val
            if safe_mask.sum() > 0:
                per = saved / safe_mask.sum()
                a_new[safe_mask] += per
                a_new = np.clip(a_new, 0, 1000)
                a_new = a_new * (S / a_new.sum())

        # Check improvement
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur_max * 1.005:
            a = a_new
        else:
            # Reject and try smaller reduction
            a_new2 = a - 0.3 * reduction
            a_new2 = np.maximum(a_new2, 0.0)
            saved2 = S - a_new2.sum()
            if saved2 > 1e-12:
                mean_val = a_new2.sum() / n
                safe_mask = a_new2 < mean_val
                if safe_mask.sum() > 0:
                    a_new2[safe_mask] += saved2 / safe_mask.sum()
                    a_new2 = a_new2 * (S / a_new2.sum())
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < cur_max * 1.005:
                a = a_new2
            else:
                # Add noise to escape
                noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)

        # Periodic full optimization restart
        if it % 200 == 0:
            a = a * (S / a.sum())

    return best_local, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses multi-phase optimization with diverse starts and peak-shaving.
    Tries both n=8192 and n=16384.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    def try_sequence(a_init, n, time_alloc):
        nonlocal global_best, global_a
        S = np.sqrt(2.0 * float(n))
        a_init = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
        s = a_init.sum()
        if s < 1e-15:
            a_init = np.ones(n) * S / n
        else:
            a_init = a_init * (S / s)

        # Run main optimization
        a_opt, score = run_optimization(a_init, n, S, time_alloc * 0.6, rng)
        a_opt = safe_normalize(a_opt, n)
        c = fft_convolve_self(a_opt)
        obj = c.max()

        if obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
            print("  [opt] n=%d: obj=%.6f" % (n, obj))

        # Run peak shaving on the result
        remaining = time_alloc * 0.4
        if remaining > 5:
            a_shave, score_shave = run_peak_shave(a_opt, n, S, remaining, rng)
            a_shave = safe_normalize(a_shave, n)
            c_shave = fft_convolve_self(a_shave)
            obj_shave = c_shave.max()
            if obj_shave < global_best:
                global_best = obj_shave
                global_a = a_shave.copy()
                print("  [shave] n=%d: obj=%.6f" % (n, obj_shave))

    # --- Gather candidates ---
    candidates = []

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if parent.size >= 64 and np.sum(np.clip(parent, 0, 1000)) > 0:
            candidates.append(("parent", np.clip(parent, 0, 1000)))
            print("Using parent sequence of length %d" % parent.size)
    except Exception:
        pass

    # New initializations for n=8192
    n8 = 8192
    tt = np.linspace(0, 1, n8, endpoint=False)

    # Exponential ramps
    for alpha in [2.0, 4.0, 6.0]:
        candidates.append(("exp_fwd_%0.1f" % alpha, np.exp(alpha * tt)))
        candidates.append(("exp_bwd_%0.1f" % alpha, np.exp(alpha * (1.0 - tt))))

    # Power law
    for p in [1.0, 2.0, 4.0]:
        candidates.append(("pow_fwd_%0.1f" % p, (tt + 0.02) ** p))
        candidates.append(("pow_bwd_%0.1f" % p, (1.0 - tt + 0.02) ** p))

    # Two-hump asymmetric
    h1 = np.exp(-((tt - 0.2) / 0.15) ** 2)
    h2 = np.exp(-((tt - 0.7) / 0.1) ** 2)
    candidates.append(("two_hump", 0.7 * h1 + 1.3 * h2 + 0.05))

    # Random with trend
    candidates.append(("rand_trend", rng.random(n8) * (0.5 + tt)))
    candidates.append(("rand_trend2", rng.random(n8) * (1.5 - tt)))

    # New: n=16384 candidates
    n16 = 16384
    tt16 = np.linspace(0, 1, n16, endpoint=False)
    candidates.append(("n16_exp", np.exp(3.0 * tt16)))
    candidates.append(("n16_exp_rev", np.exp(3.0 * (1.0 - tt16))))
    candidates.append(("n16_pow", (tt16 + 0.01) ** 2))
    candidates.append(("n16_rand", rng.random(n16) * (0.8 + 0.4 * tt16)))

    # If parent exists, add perturbed versions
    for name, arr in list(candidates):
        if "parent" in name:
            pert = arr * (1.0 + rng.normal(0, 0.1, len(arr)))
            candidates.append(("parent_pert1", np.clip(pert, 0, None)))
            pert2 = arr * (1.0 + rng.normal(0, 0.2, len(arr)))
            candidates.append(("parent_pert2", np.clip(pert2, 0, None)))

    n_cands = len(candidates)
    usable_time = max(10.0, deadline - time.monotonic())

    # Time allocation: prioritize parent and best candidates
    # Give parent 40% of time, others split the rest
    parent_time = usable_time * 0.35
    other_time = usable_time * 0.55 / max(1, n_cands - 1)

    print("=== Search: %d candidates, total time %.1fs ===" % (n_cands, usable_time))

    # Process parent first
    idx = 0
    for name, arr in candidates:
        if time.monotonic() > deadline - 30:
            break
        n_cur = len(arr)
        if n_cur < 64:
            continue
        if "parent" in name:
            t_alloc = parent_time if "parent_pert" not in name else parent_time * 0.5
        else:
            t_alloc = other_time
        t_alloc = min(t_alloc, deadline - time.monotonic() - 30)
        if t_alloc < 5:
            break

        print("[%d/%d] Trying '%s' (n=%d, t=%.1fs)" % (idx, n_cands, name, n_cur, t_alloc))
        try_sequence(arr, n_cur, t_alloc)
        idx += 1

    print("=== After exploration: best=%.6f ===" % global_best)

    # Final polish on the best candidate
    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * float(n_best))
        remaining = deadline - time.monotonic()

        if remaining > 10:
            print("=== Final polish (%.1fs) ===" % remaining)

            # Polish 1: full optimization from best
            seg1 = remaining * 0.4
            a_p1, s1 = run_optimization(global_a, n_best, S_best, seg1, rng)
            a_p1 = safe_normalize(a_p1, n_best)
            c1 = fft_convolve_self(a_p1)
            if c1.max() < global_best:
                global_best = c1.max()
                global_a = a_p1.copy()
                print("  Polish 1: %.6f" % global_best)

            # Polish 2: peak shaving
            remaining2 = deadline - time.monotonic()
            if remaining2 > 5:
                seg2 = remaining2 * 0.5
                a_p2, s2 = run_peak_shave(global_a, n_best, S_best, seg2, rng)
                a_p2 = safe_normalize(a_p2, n_best)
                c2 = fft_convolve_self(a_p2)
                if c2.max() < global_best:
                    global_best = c2.max()
                    global_a = a_p2.copy()
                    print("  Polish 2 (shave): %.6f" % global_best)

            # Polish 3: another optimization round with high-beta
            remaining3 = deadline - time.monotonic()
            if remaining3 > 5:
                a_p3, s3 = run_optimization(global_a, n_best, S_best, remaining3, rng)
                a_p3 = safe_normalize(a_p3, n_best)
                c3 = fft_convolve_self(a_p3)
                if c3.max() < global_best:
                    global_best = c3.max()
                    global_a = a_p3.copy()
                    print("  Polish 3: %.6f" % global_best)

    # Fallback
    if global_a is None:
        print("  WARNING: No valid candidate found, using uniform fallback")
        global_a = np.ones(256, dtype=np.float64)

    # Final validation
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)

    print("Final: length=%d, internal=%.6f, verified=%.6f" %
          (len(final_list), global_best, ver if np.isfinite(ver) else -1))
    return final_list
