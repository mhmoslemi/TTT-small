def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a."""
    n = len(a)
    cmax = c.max()
    exp_bc = np.exp(beta * (c - cmax))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def project_box(v, lo, hi):

def coordinate_descent(a, n, S, max_time, rng, iterations_per_coord=3):
    """Coordinate descent: for each pair of indices, try to reduce the max convolution."""
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()
    
    for it in range(200):
        if time.monotonic() - start > max_time:
            break
        
        # Find the current max lag
        peak_lag = int(c.argmax())
        
        # Try moving some mass from high-contribution elements to low ones
        # For the peak lag, find which pairs contribute most
        lag = peak_lag
        i_min = max(0, lag - (n - 1))
        i_max = min(lag, n - 1)
        
        # Get contribution of each pair to the peak
        contributions = np.zeros(n)
        for i in range(i_min, i_max + 1):
            j = lag - i
            contributions[i] += a[j]
            if j != i:
                contributions[j] += a[i]
        
        # Try swapping: reduce a[high_contrib], increase a[low_contrib]
        high_idx = int(np.argmax(contributions))
        low_candidates = np.where(contributions < np.median(contributions) + 1e-10)[0]
        if len(low_candidates) == 0:
            continue
        low_idx = low_candidates[rng.integers(len(low_candidates))]
        
        # Try small swap
        for frac in [0.05, 0.02, 0.01]:
            delta = frac * a[high_idx]
            a_new = a.copy()
            a_new[high_idx] -= delta
            a_new[low_idx] += delta
            a_new = np.maximum(a_new, 0)
            # Renormalize
            s = a_new.sum()
            if s > 0:
                a_new = a_new * (S / s)
            
            c_new = fft_convolve_self(a_new)
            if c_new.max() < best_score:
                best_score = c_new.max()
                a = a_new
                c = c_new
                break
    
    return a, best_score

def make_binary_anti_pal_init(n, rng, S):
    """Create a binary anti-palindromic initialization."""
    half_n = n // 2
    mu = S / half_n
    # Random binary pattern for first half
    pattern = (rng.random(half_n) > 0.5).astype(float) * mu
    # Ensure some are 0 and some are mu
    if pattern.sum() == 0:
        pattern[0] = mu
    a = np.zeros(n)
    a[:half_n] = pattern
    for i in range(half_n):
        a[n - 1 - i] = mu - pattern[i]
    a = a * (S / a.sum())
    return a

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.

Key strategy:
- Exploit anti-palindromic parameterization (a[i]+a[n-1-i]=mu) which forces
  the central convolution term to zero, reducing the problem to half the variables.
- FISTA with log-sum-exp smoothing in the reduced space with correct chain-rule gradients.
- Full-space FISTA polish, coordinate descent, and symmetry-breaking perturbation.
- Multiple structured initializations and length exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_gradient_full(a, c, beta):
    """Gradient of smooth surrogate w.r.t. full sequence a."""
    n = len(a)
    cmax = c.max()
    exp_bc = np.exp(beta * (c - cmax))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def reconstruct_from_half(half_a, n):
    """Reconstruct full anti-palindromic sequence from first half."""
    half_n = n // 2
    mu = np.sqrt(2.0 * n) / half_n
    a = np.zeros(n)
    a[:half_n] = half_a
    for i in range(half_n):
        a[n - 1 - i] = mu - half_a[i]
    return a


def fista_reduced(half_init, n, S, max_time, rng, beta_init=10.0, beta_max=600.0):
    """FISTA in anti-palindromic reduced space with box constraint [0, mu]."""
    half_n = n // 2
    mu = S / half_n

    x = np.clip(half_init, 0.0, mu).copy()
    y = x.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_x = x.copy()
    it = 0
    L = beta_init * 3.0
    stall = 0
    last_best = np.inf

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.004 * it))

        # Build full sequence from y
        a_y = reconstruct_from_half(y, n)

        c = fft_convolve_self(a_y)
        cur = c.max()

        if cur < best_score - 1e-12:
            best_score = cur
            best_x = x.copy()
            stall = 0
        elif it % 100 == 0:
            if cur < last_best - 1e-10:
                last_best = cur
                stall = 0
            else:
                stall += 1
        last_best = min(last_best, cur)

        # Compute gradient in full space, then reduce
        grad_full = lse_gradient_full(a_y, c, beta)
        # Chain rule: d(f)/d(half_a[j]) = grad_full[j] - grad_full[n-1-j]
        grad_half = grad_full[:half_n] - grad_full[n - 1 - np.arange(half_n)][::-1]

        gnorm = np.linalg.norm(grad_half)
        if gnorm < 1e-15:
            break

        step = 1.0 / L
        x_new = np.clip(y - step * grad_half, 0.0, mu)

        # FISTA momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        y = np.clip(y, 0.0, mu)
        x = x_new
        t_mom = t_new

        # Adaptive L

            a_check = reconstruct_from_half(x, n)
            c_check = fft_convolve_self(a_check)
            if c_check.max() > cur * 1.005:
                L *= 1.3
            else:
                L = max(L * 0.93, beta * 2.0)

        if stall > 40 and rng.random() < 0.4:
            noise_scale = 0.03 * mu
            y = np.clip(y + rng.normal(0, noise_scale, half_n), 0.0, mu)
            t_mom = 1.0
            stall = 0
            L = beta * 3.0

    return best_x, best_score

    s = a.sum()
    if s < 1e-15:

    else:

    t_mom = 1.0

    L = beta_init * 3.0

            best_score = cur
            best_a = y.copy()

            if cur < last_best - 1e-10:
                last_best = cur
                stall = 0
            else:
                stall += 1

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)

            t_mom = 1.0
            stall = 0

def topk_subgradient_step(a, n, S, k, step_size):
    """One subgradient step targeting the top-k peaks."""
    c = fft_convolve_self(a)
    top_idx = np.argsort(c)[::-1][:k]
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / k
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    a_new = project_simplex(a - step_size * grad, S)
    return a_new


def coordinate_descent(a, n, S, max_time, rng):
    """Coordinate descent targeting peak-reducing pair swaps."""
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_score = c.max()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur

        peak_lag = int(c.argmax())
        lag = peak_lag
        i_min = max(0, lag - (n - 1))
        i_max = min(lag, n - 1)
        if i_min >= i_max:
            break

        # Find pairs contributing to peak
        indices = np.arange(i_min, i_max + 1)
        partners = lag - indices
        products = a[indices] * a[partners]
        
        # Identify high and low contribution indices
        prod_order = np.argsort(products)[::-1]
        n_high = min(30, len(prod_order))
        high_idx = indices[prod_order[:n_high]]
        
        # Candidates to receive mass: low values
        low_mask = a < np.percentile(a, 30)
        low_candidates = np.where(low_mask)[0]
        if len(low_candidates) == 0:
            break

        improved = False
        for hi in high_idx:
            if time.monotonic() - start > max_time:
                break
            for frac in [0.1, 0.05, 0.02]:
                delta = frac * a[hi]
                if delta < 1e-12:
                    continue
                for _ in range(3):
                    lo = int(low_candidates[rng.integers(len(low_candidates))])
                    a_new = a.copy()
                    a_new[hi] -= delta
                    a_new[lo] += delta
                    a_new = np.maximum(a_new, 0)
                    s = a_new.sum()
                    a_new = a_new * (S / s)
                    c_new = fft_convolve_self(a_new)
                    if c_new.max() < best_score:
                        best_score = c_new.max()
                        a = a_new
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and rng.random() < 0.1:
            # Random perturbation
            noise_scale = 0.005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return a, best_score


def make_binary_anti_pal(n, rng, S):

def make_lattice_anti_pal(n, rng, S):

def make_from_parent(parent, target_n, S):

    half_n = n // 2
    mu = S / half_n

    # === Phase 1: Reduced-space FISTA (main optimization) ===
    phase1_time = (deadline - time.monotonic()) * 0.50
    print("=== Phase 1: Reduced FISTA, n=%d, budget=%.0fs ===" % (n, phase1_time))

    # Build initializations
    inits = []
    if parent is not None:
        inits.append(("parent", make_from_parent(parent, n, S)))
    for i in range(3):
        inits.append(("bin%d" % i, make_binary_anti_pal(n, rng, S)))
    for i in range(4):
        inits.append(("smooth%d" % i, make_smooth_anti_pal(n, rng, S)))
    for i in range(2):
        inits.append(("lattice%d" % i, make_lattice_anti_pal(n, rng, S)))
    # Random
    inits.append(("rand", rng.random(half_n) * mu * 0.5 + mu * 0.05))

    time_per_init = max(5.0, min(40.0, phase1_time / len(inits)))
    best_half = None
    best_half_score = np.inf

    for name, h_init in inits:
        if time.monotonic() > deadline - 100:
            break
        t0 = time.monotonic()
        t_alloc = min(time_per_init, deadline - time.monotonic() - 80)
        if t_alloc < 3:
            break

        h_opt, score = fista_reduced(h_init, n, S, t_alloc, rng,
                                      beta_init=10.0, beta_max=500.0)
        a_opt = reconstruct_from_half(h_opt, n)
        a_opt = np.maximum(a_opt, 0.0)
        s = a_opt.sum()
        a_opt = a_opt * (S / s)

        c_opt = fft_convolve_self(a_opt)
        actual = c_opt.max()
        obj = 2.0 * n * actual / (S ** 2)

        if actual < best_half_score:
            best_half_score = actual
            best_half = h_opt.copy()

        if actual < global_best:
            global_best = actual
            global_a = a_opt.copy()
            print("  [%s] reduced: max_b=%.6f, obj=%.6f (%.1fs)" %
                  (name, actual, obj, time.monotonic() - t0))

    # Continue with best half if available
    if best_half is not None:
        rem1 = deadline - time.monotonic() - 80
        if rem1 > 5:
            extra_time = min(60.0, rem1 * 0.3)
            h_opt, score = fista_reduced(best_half, n, S, extra_time, rng,
                                          beta_init=50.0, beta_max=800.0)
            a_opt = reconstruct_from_half(h_opt, n)
            a_opt = np.maximum(a_opt, 0.0)
            a_opt = a_opt * (S / a_opt.sum())
            c_opt = fft_convolve_self(a_opt)
            if c_opt.max() < global_best:
                global_best = c_opt.max()
                global_a = a_opt.copy()
                print("  [extra reduced] max_b=%.6f, obj=%.6f" %
                      (global_best, 2.0 * n * global_best / (S ** 2)))

    print("After Phase 1: best=%.6f, obj=%.6f" % (global_best, 2 * n * global_best / S ** 2))

    # === Phase 2: Full-space FISTA polish ===
    if global_a is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            print("=== Phase 2: Full FISTA polish (%.0fs) ===" % rem)
            seg = min(120.0, rem * 0.30)
            a_f, s_f = fista_full(global_a, n, S, seg, rng,
                                    beta_init=50.0, beta_max=1000.0)
            c_f = fft_convolve_self(a_f)
            if c_f.max() < global_best:
                global_best = c_f.max()
                global_a = a_f.copy()
                print("  Full FISTA: %.6f" % global_best)

    # === Phase 3: Coordinate descent ===
    if global_a is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            print("=== Phase 3: Coord descent (%.0fs) ===" % rem)
            seg = min(60.0, rem * 0.20)
            a_cd, s_cd = coordinate_descent(global_a, n, S, seg, rng)
            c_cd = fft_convolve_self(a_cd)
            if c_cd.max() < global_best:
                global_best = c_cd.max()
                global_a = a_cd.copy()
                print("  Coord desc: %.6f" % global_best)

    # === Phase 4: Symmetry breaking ===
    if global_a is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            print("=== Phase 4: Symmetry breaking (%.0fs) ===" % rem)
            for attempt in range(4):
                if time.monotonic() > deadline - 20:
                    break
                # Small perturbation breaking anti-palindromic symmetry
                scale = [0.002, 0.005, 0.01, 0.02][min(attempt, 3)]
                pert = rng.normal(0, scale * S / np.sqrt(n), n)
                a_pert = np.maximum(global_a + pert, 0)
                a_pert = a_pert * (S / a_pert.sum())

                seg = min(40.0, max(10.0, (deadline - time.monotonic()) * 0.25))
                a_bp, s_bp = fista_full(a_pert, n, S, seg, rng,
                                         beta_init=100.0, beta_max=1500.0)
                c_bp = fft_convolve_self(a_bp)
                if c_bp.max() < global_best:
                    global_best = c_bp.max()
                    global_a = a_bp.copy()
                    print("  Break %d (scale=%.3f): %.6f" % (attempt, scale, global_best))

    # === Phase 5: Alternative length ===
    if time.monotonic() < deadline - 60:
        # Try half or double length
        candidates_n = []
        if primary_n > 4096:
            candidates_n.append(primary_n // 2)
        if primary_n < 16384:
            candidates_n.append(min(primary_n * 2, 16384))

        for alt_n in candidates_n:
            if time.monotonic() > deadline - 40:
                break
            if alt_n < 512 or alt_n > 32768:
                continue
            print("=== Phase 5: Try n=%d ===" % alt_n)
            S_alt = np.sqrt(2.0 * alt_n)

            if parent is not None and len(parent) >= 64:
                a_alt = make_from_parent(parent, alt_n, S_alt)
                a_alt_full = reconstruct_from_half(a_alt, alt_n)
            else:
                h_alt = make_smooth_anti_pal(alt_n, rng, S_alt)
                a_alt_full = reconstruct_from_half(h_alt, alt_n)
            a_alt_full = np.maximum(a_alt_full, 0)
            a_alt_full = a_alt_full * (S_alt / a_alt_full.sum())

            # Quick reduced FISTA
            seg = min(50.0, (deadline - time.monotonic()) * 0.4)
            h_opt_alt, _ = fista_reduced(a_alt_full[:alt_n // 2], alt_n, S_alt, seg, rng,
                                          beta_init=20.0, beta_max=400.0)
            a_alt_opt = reconstruct_from_half(h_opt_alt, alt_n)
            a_alt_opt = np.maximum(a_alt_opt, 0)
            a_alt_opt = a_alt_opt * (S_alt / a_alt_opt.sum())
            c_alt = fft_convolve_self(a_alt_opt)
            obj_alt = 2.0 * alt_n * c_alt.max() / (S_alt ** 2)
            obj_cur = 2.0 * n * global_best / (S ** 2)

            if obj_alt < obj_cur:
                print("  n=%d: obj=%.6f < %.6f (improvement!)" % (alt_n, obj_alt, obj_cur))
                global_a = a_alt_opt.copy()
                global_best = c_alt.max()
                n = alt_n
                S = S_alt
            else:
                print("  n=%d: obj=%.6f >= %.6f (no improvement)" % (alt_n, obj_alt, obj_cur))

    # === Phase 6: Final intensive polish ===
    if global_a is not None and time.monotonic() < deadline - 10:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        rem = deadline - time.monotonic()
        print("=== Phase 6: Final polish (%.0fs, n=%d) ===" % (rem, n))

        # Very high beta FISTA
        seg = max(5.0, rem * 0.4)
        a_fin, _ = fista_full(global_a, n, S, seg, rng,
                               beta_init=300.0, beta_max=3000.0)
        c_fin = fft_convolve_self(a_fin)
        if c_fin.max() < global_best:
            global_best = c_fin.max()
            global_a = a_fin.copy()
            print("  Final FISTA hi-beta: %.6f" % global_best)

        # Final coord descent
        if time.monotonic() < deadline - 5:
            seg2 = max(3.0, (deadline - time.monotonic()) * 0.5)
            a_cd2, _ = coordinate_descent(global_a, n, S, seg2, rng)
            c_cd2 = fft_convolve_self(a_cd2)
            if c_cd2.max() < global_best:
                global_best = c_cd2.max()
                global_a = a_cd2.copy()
                print("  Final CD: %.6f" % global_best)

        # Top-k subgradient refinement
        if time.monotonic() < deadline - 3:
            c = fft_convolve_self(global_a)
            step = 0.1 * S / n
            for _ in range(50):
                if time.monotonic() > deadline - 2:
                    break
                c = fft_convolve_self(global_a)
                for k in [5, 10, 20]:
                    a_new = topk_subgradient_step(global_a, n, S, k, step)
                    c_new = fft_convolve_self(a_new)
                    if c_new.max() < global_best:
                        global_best = c_new.max()
                        global_a = a_new.copy()
                step *= 0.95

    # === Verification ===
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL: %.6f (n=%d) ===" % (ver if np.isfinite(ver) else float('inf'), len(final_list)))

    if not np.isfinite(ver):
        print("  WARNING: Invalid result, using fallback")
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback: %.6f" % ver)

    return final_list
