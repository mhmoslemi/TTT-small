"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- Multi-phase optimization: L-BFGS-B with log-sum-exp continuation,
  direct subgradient with top-K weighting and line search,
  multiplicative (exponentiated gradient) updates, and LP-based polishing.
- FFT-accelerated convolution and gradient computation.
- Multiple initializations and restarts to escape local minima.
- Aggressive beta schedule for tight approximation of the max function.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_conv_grad(a, w):
    """
    Compute gradient of sum_k w[k] * (a*a)[k] w.r.t. a.
    w is the weight vector over convolution indices (length 2n-1).
    Result: grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) (with boundary handling).
    Computed via FFT as cross-correlation.
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # grad[j] = 2 * sum_k w[k] * a[k-j] (with valid indices)
    # This is a cross-correlation: (w cross a)[j] = sum_k w[k] * a[k-j]
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """
    Compute log-sum-exp smoothed objective and gradient w.r.t. z.
    a = S * softmax(z), obj = (1/beta)*log(sum exp(beta*c_k))
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_conv_grad(a, w)

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=300):
    """
    Run L-BFGS-B with continuation on beta (smooth-to-sharp).
    """
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 4000, 8000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)
    z = z - z.mean()

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        n_attempts = 1
        if beta >= 500:
            n_attempts = 2
        if beta >= 2000:
            n_attempts = 3

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 5:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.01, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            try:
                result = scipy_minimize(
                    fun, z,
                    jac=jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_beta,
                        'maxfun': maxiter_per_beta * 3,
                        'ftol': 1e-15,
                        'gtol': 1e-10,
                        'maxls': 50,
                        'iprint': -1
                    }
                )
            except Exception:
                break

            z = result.x
            z = z - z.mean()

            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s_check

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def run_direct_subgradient(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=200):
    """
    Direct subgradient method with top-K peak weighting,
    Nesterov momentum, and adaptive step size.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step_base = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)

        c_y = fft_convolve_self(y)
        cur = c_y.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Get top-K peaks
        sorted_idx = np.argsort(c_y)[::-1][:top_k]
        top_vals = c_y[sorted_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = (top_vals / total) ** 2  # Square for more emphasis on peaks

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(sorted_idx):
            w_full[int(k)] = weights[i]

        # Compute subgradient at a (not y, for stability)
        c_a = fft_convolve_self(a)
        # Use the same weight structure but evaluated at current a
        grad = compute_conv_grad(a, w_full)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size with decay
        step = step_base * S / (np.sqrt(n) * (1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Check x_new directly
        if it % 2 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        # Random restart to escape local minima
        if it % 3000 == 0 and rng.random() < 0.1:
            noise_scale = 0.01 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0

        # Decay step base slowly
        if it % 1000 == 0:
            step_base = max(0.3, step_base * 0.85)

    return best_local, best_score


def run_multiplicative_grad(a_init, n, S, max_time, rng, eta_init=0.05):
    """
    Multiplicative (exponentiated gradient) update in the log domain.
    a[i] <- a[i] * (S/n / contrib[i])^eta, then renormalize.
    This is equivalent to mirror descent with entropy regularizer.
    """
    a = np.clip(a_init, 1e-15, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * (S / s)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    it = 0
    eta = eta_init

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Compute importance weights (top-K)
        top_k = min(100, 10 + it // 100)
        sorted_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[sorted_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(sorted_idx):
            w_full[int(k)] = weights[i]

        # Gradient of sum_k w[k] * c[k] w.r.t. a
        grad = compute_conv_grad(a, w_full)
        # In multiplicative update: a[i] <- a[i] * exp(-eta * grad[i])
        # This is mirror descent with negative entropy
        log_update = -eta * grad / max(np.abs(grad).max(), 1e-15)
        log_update = np.clip(log_update, -2.0, 2.0)  # Prevent extreme updates
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a * (S / a.sum())

        # Decay eta
        eta = max(eta_init * 0.01, eta * (1.0 - it / max_time * 1.5))

        if it % 50 == 0 and rng.random() < 0.05:
            a = a * np.exp(rng.normal(0, 0.005, n))
            a = np.maximum(a, 1e-15)
            a = a * (S / a.sum())

    return best_local, best_score


def run_lp_polish(a_init, n, S, max_time, rng):
    """
    LP-based direction finding (simplified version of the SOTA approach).
    Solve: max sum(b) s.t. (a*b)_k <= M for top constraints, b >= 0.
    Then interpolate: a <- (1-t)*a + t*b_normalized.
    """
    from scipy.optimize import linprog

    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a * (S / s)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()

    n_iterations = 0
    while time.monotonic() - start < max_time:
        n_iterations += 1
        c = fft_convolve_self(a)
        M = c.max()
        if M < best_score:
            best_score = M
            best_local = a.copy()

        # Select top constraints (where convolution is close to M)
        threshold = M * 0.9 + (1 - 0.9) * c.min()
        active_idx = np.where(c >= threshold)[0]
        if len(active_idx) < 10:
            active_idx = np.argsort(c)[::-1][:10]
        # Cap at 200 constraints for LP tractability
        if len(active_idx) > 200:
            active_idx = active_idx[:200]

        # Build LP: max sum(b) s.t. (a*b)_k <= M for k in active_idx, b >= 0
        # (a*b)_k = sum_j a[k-j] * b[j] for valid j
        # This is a matrix-vector product: A_lp @ b <= M
        # A_lp has shape (len(active_idx), n)
        # A_lp[i, j] = a[active_idx[i] - j] if 0 <= active_idx[i] - j < n, else 0

        m_constraints = len(active_idx)
        # Build sparse constraint matrix
        from scipy.sparse import csr_matrix
        rows = []
        cols = []
        vals = []
        for i, k in enumerate(active_idx):
            k = int(k)
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                rows.append(i)
                cols.append(j)
                vals.append(a[k - j])

        A_lp = csr_matrix((vals, (rows, cols)), shape=(m_constraints, n))

        # Solve: min -sum(b) s.t. A_lp @ b <= M, b >= 0
        c_obj = -np.ones(n)
        res = linprog(
            c_obj,
            A_ub=A_lp,
            b_ub=np.full(m_constraints, M),
            bounds=[(0, None)] * n,
            method='highs',
            options={'time_limit': max(1.0, (max_time - (time.monotonic() - start)) / 3.0)}
        )

        if not res.success:
            break

        b_sol = np.asarray(res.x, dtype=np.float64)
        b_sum = b_sol.sum()
        if b_sum < 1e-15:
            break

        # Normalize b to have sum = S
        b_norm = b_sol * (S / b_sum)

        # Check if this direction improves
        # Try several interpolation factors
        improved = False
        for t_blend in [0.1, 0.2, 0.3, 0.5]:
            a_try = (1 - t_blend) * a + t_blend * b_norm
            a_try = np.maximum(a_try, 0.0)
            a_try = a_try * (S / a_try.sum())
            c_try = fft_convolve_self(a_try)
            if c_try.max() < best_score:
                best_score = c_try.max()
                best_local = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            # Try smaller step
            for t_blend in [0.05, 0.02]:
                a_try = (1 - t_blend) * a + t_blend * b_norm
                a_try = np.maximum(a_try, 0.0)
                a_try = a_try * (S / a_try.sum())
                c_try = fft_convolve_self(a_try)
                if c_try.max() < best_score:
                    best_score = c_try.max()
                    best_local = a_try.copy()
                    a = a_try.copy()
                    break
            else:

    return best_local, best_score

                print("  Subgradient: %.8f (no improvement)" % c_sg.max())

            # Nesterov variant
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                seg2 = min(remaining2 * 0.15, 150.0)
                a_ne, score_ne = run_nesterov_toppk(
                    global_a, n, S, seg2, rng,
                    top_k_init=20, top_k_max=512
                )
                a_ne = safe_normalize(a_ne, n)
                c_ne = fft_convolve_self(a_ne)
                if c_ne.max() < global_best:
                    global_best = c_ne.max()
                    global_a = a_ne.copy()
                    print("  Nesterov: %.8f" % global_best)
                else:
                    print("  Nesterov: %.8f (no improvement)" % c_ne.max())

    print("\nAfter Phase 2: best=%.8f" % global_best)

    # --- Phase 3: Multiplicative gradient (mirror descent) ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            seg = min(remaining * 0.12, 120.0)
            print("\n=== Phase 3: Multiplicative Gradient (%.0fs) ===" % seg)
            t0 = time.monotonic()

            a_mp, score_mp = run_multiplicative_grad(
                global_a, n, S, seg, rng, eta_init=0.03
            )
            a_mp = safe_normalize(a_mp, n)
            c_mp = fft_convolve_self(a_mp)
            if c_mp.max() < global_best:
                global_best = c_mp.max()
                global_a = a_mp.copy()
                print("  MultGrad: %.8f (%.1fs)" % (global_best, time.monotonic() - t0))
            else:
                print("  MultGrad: %.8f (no improvement, %.1fs)" % (c_mp.max(), time.monotonic() - t0))

    print("\nAfter Phase 3: best=%.8f" % global_best)

    # --- Phase 4: Perturbation + L-BFGS restart ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print("\n=== Phase 4: Perturbation + L-BFGS Restart ===")

            n_perturb = max(2, int(remaining / 60))
            perturb_budget = remaining * 0.45 / n_perturb

            for p in range(n_perturb):
                if time.monotonic() > deadline - 60:
                    break

                # Varying noise scales
                noise_scales = [0.02, 0.04, 0.01, 0.03, 0.015]
                ns = noise_scales[p % len(noise_scales)] / (1.0 + 0.2 * p)
                a_pert = global_a + rng.normal(0, ns * S / np.sqrt(n), n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = safe_normalize(a_pert, n)

                a_new, score = run_lbfgs_continuation(
                    a_pert, n, S, perturb_budget, rng,
                    beta_schedule=[50, 100, 300, 800, 2000, 5000, 10000],
                    maxiter_per_beta=200
                )
                a_new = safe_normalize(a_new, n)
                c_new = fft_convolve_self(a_new)
                if c_new.max() < global_best:
                    global_best = c_new.max()
                    global_a = a_new.copy()
                    print("  Perturb %d (ns=%.4f): %.8f" % (p, ns, global_best))

    print("\nAfter Phase 4: best=%.8f" % global_best)

    # --- Phase 5: LP-based polishing ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            seg = min(remaining * 0.15, 180.0)
            print("\n=== Phase 5: LP Polishing (%.0fs) ===" % seg)
            t0 = time.monotonic()

            try:
                a_lp, score_lp = run_lp_polish(
                    global_a, n, S, seg, rng
                )
                a_lp = safe_normalize(a_lp, n)
                c_lp = fft_convolve_self(a_lp)
                if c_lp.max() < global_best:
                    global_best = c_lp.max()
                    global_a = a_lp.copy()
                    print("  LP: %.8f (%.1fs)" % (global_best, time.monotonic() - t0))
                else:
                    print("  LP: %.8f (no improvement, %.1fs)" % (c_lp.max(), time.monotonic() - t0))
            except Exception as e:
                print("  LP failed: %s" % str(e))

    print("\nAfter Phase 5: best=%.8f" % global_best)

    # --- Phase 6: Final high-beta L-BFGS polish ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print("\n=== Phase 6: Final Polish (%.0fs) ===" % remaining)

            # Very high beta L-BFGS
            a_final, score_final = run_lbfgs_continuation(
                global_a, n, S, remaining * 0.6, rng,
                beta_schedule=[500, 1000, 3000, 8000, 15000, 30000],
                maxiter_per_beta=300
            )
            a_final = safe_normalize(a_final, n)
            c_final = fft_convolve_self(a_final)
            if c_final.max() < global_best:
                global_best = c_final.max()
                global_a = a_final.copy()
                print("  Final L-BFGS: %.8f" % global_best)

            # One more subgradient pass
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                a_sg2, _ = run_direct_subgradient(
                    global_a, n, S, remaining2 * 0.8, rng,
                    top_k_init=50, top_k_max=500
                )
                a_sg2 = safe_normalize(a_sg2, n)
                c_sg2 = fft_convolve_self(a_sg2)
                if c_sg2.max() < global_best:
                    global_best = c_sg2.max()
                    global_a = a_sg2.copy()
                    print("  Final SubGrad: %.8f" % global_best)

    # --- Phase 7: Cross-pollination between best results ---
    if global_a is not None and len(phase1_results) >= 2:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print("\n=== Phase 7: Cross-pollination (%.0fs) ===" % remaining)
            sorted_results = sorted(phase1_results.items(), key=lambda x: x[1][1])

            for xp in range(min(3, len(sorted_results) - 1)):
                if time.monotonic() > deadline - 10:
                    break
                a1 = sorted_results[xp][1][0]
                a2 = sorted_results[xp + 1][1][0]
                for blend in [0.4, 0.5, 0.6]:
                    if time.monotonic() > deadline - 10:
                        break
                    a_blend = blend * a1 + (1 - blend) * a2
                    a_blend = safe_normalize(a_blend, n)
                    c_blend = fft_convolve_self(a_blend)
                    if c_blend.max() < global_best:
                        global_best = c_blend.max()
                        global_a = a_blend.copy()
                        print("  XPoll %d blend=%.1f: %.8f" % (xp, blend, global_best))

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("\nVerification score: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * 10.0
    else:
        global_a = np.ones(256, dtype=np.float64) * 10.0

    print("Returning sequence of length %d, internal score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
