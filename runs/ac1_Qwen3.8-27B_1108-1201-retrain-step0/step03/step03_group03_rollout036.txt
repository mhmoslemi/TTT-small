"""
Anti-palindromic FISTA with adaptive LSE smoothing and peak-directed refinement.

Key insight: For non-negative sequences, the convolution peak can be reduced
below the Cauchy-Schwarz bound by using anti-palindromic structure (a[i] + a[n-1-i] = const),
which makes the center lag b[n-1] small. The optimization is performed in the
reduced anti-palindromic subspace (dimension n/2) where projection is simple box clipping.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_full_gradient(a, c, beta):
    """
    Compute gradient of LSE-smoothed max of conv(a,a) w.r.t. a.
    
    Parameters:
    - a: sequence of length n
    - c: convolution of a with a, length 2n-1
    - beta: smoothing parameter
    
    Returns:
    - grad: array of length n
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total  # softmax weights over lags, length 2n-1
    
    # grad[i] = 2 * sum_j w[i+j] * a[j] for j=0,...,n-1 (where i+j < 2n-1)
    # Using cross-correlation via FFT:
    # corr[i] = sum_j w_rev[i+j] * a[j] where w_rev[k] = w[2n-2-k]
    # Actually: grad[i] = 2 * corr[2n-2-i] where corr = ifft(fft(w[::-1]) * fft(a))
    
    w_rev = w[::-1]  # length 2n-1
    fsize = 1 << (2*n - 1 + n - 1).bit_length()  # >= 3n-2
    W = fft(w_rev, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * A))
    
    # grad[i] = 2 * corr[2n-2-i]
    indices = 2*n - 2 - np.arange(n)
    grad = 2.0 * corr[indices]
    return grad


def anti_pal_to_full(h, mu):
    """Convert anti-palindromic parameters h (length m) to full sequence a (length n=2m)."""
    m = len(h)
    n = 2 * m
    a = np.zeros(n)
    a[:m] = h
    a[m:] = mu - h[::-1]
    return a


def project_box(v, lo, hi):
    """Project onto box [lo, hi] element-wise."""
    return np.clip(v, lo, hi)


def run_anti_pal_fista(h_init, mu, max_time, rng, beta_init=1.0, beta_max=1000.0):
    """
    FISTA in the anti-palindromic subspace.
    
    Parameters:
    - h_init: initial h values (length m = n/2)
    - mu: anti-palindromic constant (a[i] + a[n-1-i] = mu)
    - max_time: time budget in seconds
    - rng: random number generator
    - beta_init, beta_max: LSE smoothing parameters
    
    Returns:
    - h_best: best h found
    - score_best: best max_b achieved
    """
    m = len(h_init)
    n = 2 * m
    h = project_box(h_init.copy(), 0.0, mu)
    
    y = h.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_h = h.copy()
    it = 0
    L = beta_init * 4.0
    stall = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))
        
        a = anti_pal_to_full(y, mu)
        c = fft_convolve_self(a)
        cur = c.max()
        
        if cur < best_score:
            best_score = cur
            best_h = y.copy()
            stall = 0
        
        # Compute gradient in reduced space
        grad_a = compute_full_gradient(a, c, beta)
        grad_h = grad_a[:m] - grad_a[m:][::-1]  # grad[i] - grad[n-1-i]
        
        gnorm = np.linalg.norm(grad_h)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = 1.0 / L
        for _ in range(10):
            h_new = project_box(y - step * grad_h, 0.0, mu)
            if step < 1e-10:
                break
            # Check for sufficient decrease (optional, skip for speed)
            break
        
        h_new = project_box(y - step * grad_h, 0.0, mu)
        
        # Check if we made progress
        a_new = anti_pal_to_full(h_new, mu)
        c_new = fft_convolve_self(a_new)
        f_new = c_new.max()
        
        if f_new > cur * 1.01:
            L *= 2.0
            h_new = project_box(y - (1.0/L) * grad_h, 0.0, mu)
        elif it % 10 == 0:
            L = max(L * 0.9, beta * 2.0)
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = h_new + ((t_mom - 1.0) / t_new) * (h_new - h)
        y = project_box(y, 0.0, mu)  # Keep y in the box
        h = h_new
        t_mom = t_new
        
        # Periodic perturbation
        if stall > 50 and it % 100 == 0 and rng.random() < 0.3:
            noise = 0.01 * mu * rng.standard_normal(m)
            y = project_box(h + noise, 0.0, mu)
            t_mom = 1.0
            stall = 0
        elif it % 50 == 0:
            if cur >= best_score + 1e-10:
                stall += 1
    
    return best_h, best_score

def run_full_space_fista(a_init, n, S, max_time, rng, beta_init=5.0):

    """

    else:
        a = a * (S / s)

    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    L = beta_init * 2 * S / n

        it += 1
        beta = min(500.0, beta_init * (1.0 + 0.005 * it))

        cur = c.max()
        if cur < best_score:

        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-14:
            break
        
        step = 1.0 / L
        h_new = y - step * grad_a
        # Project onto {a >= 0, sum(a) = S}
        h_new = project_simplex(h_new, S)
        
        a_new = h_new
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0.0)
        a = a_new
        t_mom = t_new
        
        # Adaptive L
        if it % 20 == 0:
            c_check = fft_convolve_self(a)
            if c_check.max() > cur * 1.1:
                L *= 1.5
            else:
                L = max(L * 0.95, beta * 2 * S / n)

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def run_coordinate_descent(a, n, S, max_time, rng):
    """
    Coordinate descent: adjust pairs (i, n-1-i) to reduce the peak.
    """
    a = np.clip(a, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
        
        # Find the peak lag
        peak_lag = int(np.argmax(c))
        
        # Find contributing pairs
        i_min = max(0, peak_lag - (n - 1))
        i_max = min(peak_lag, n - 1)
        if i_min > i_max:
            break
        
        indices = np.arange(i_min, i_max + 1)
        partners = peak_lag - indices
        products = a[indices] * a[partners]
        
        # Reduce the largest contributing pair
        max_prod_idx = int(np.argmax(products))
        idx = indices[max_prod_idx]
        pidx = partners[max_prod_idx]
        
        # Transfer a small amount from the larger to a random smaller element
        delta = 0.01 * S / n
        if a[idx] >= a[pidx]:
            a[idx] -= delta
            # Add to a random element that's not at the peak
            candidates = np.where(np.abs(np.arange(n) - peak_lag) > n//4)[0]
            if len(candidates) > 0:
                add_idx = candidates[rng.integers(len(candidates))]
                a[add_idx] += delta
        else:
            a[pidx] -= delta
            candidates = np.where(np.abs(np.arange(n) - peak_lag) > n//4)[0]
            if len(candidates) > 0:
                add_idx = candidates[rng.integers(len(candidates))]
                a[add_idx] += delta
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (S / s)
    
    return best_a, best_score


def make_anti_pal_init(m, mu, rng, style='random'):

"""

Minimizes 2*n*max(conv(a,a))/sum(a)^2 by exploiting the anti-palindromic constraint
a[i]+a[n-1-i]=mu which reduces the center convolution lag below the Cauchy-Schwarz bound.
Optimization is performed in the reduced h-space (dimension n/2) with box projection [0,mu].

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_lse(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, where c = conv(a,a).
    Uses FFT cross-correlation for efficiency.
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total  # length 2n-1

    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]  (where i+j valid)
    # = 2 * corr[2n-2-i] where corr = ifft(fft(w[::-1], F) * fft(a, F))
    w_rev = w[::-1]
    fsize = 1 << (3 * n - 3).bit_length()
    W = fft(w_rev, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * A))
    indices = 2 * n - 2 - np.arange(n)
    return 2.0 * corr[indices]


def compute_gradient_topk(a, c, top_indices):
    """Gradient using uniform weights on top_indices."""
    n = len(a)
    w = np.zeros(len(c))
    k = len(top_indices)
    w[top_indices] = 1.0 / k
    w_rev = w[::-1]
    fsize = 1 << (3 * n - 3).bit_length()
    W = fft(w_rev, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * A))
    indices = 2 * n - 2 - np.arange(n)
    return 2.0 * corr[indices]


def anti_pal_to_full(h, mu):
    """h (length m) -> a (length 2m) with a[i]+a[n-1-i]=mu."""
    m = len(h)
    a = np.empty(2 * m)
    a[:m] = h
    a[m:] = mu - h[::-1]
    return a

def project_simplex(v, S):
    """Projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cands = (cumsum - S) / idx
    valid = theta_cands <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def run_full_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_max=500.0):
    """FISTA in full space with simplex projection."""
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    L = beta_init * 2 * S / n

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = y.copy()

        grad_a = compute_gradient_lse(y, c, beta)
        gnorm = np.linalg.norm(grad_a)
        if gnorm < 1e-15:
            break

        step = 1.0 / L
        x_new = project_simplex(y - step * grad_a, S)

        # Adaptive L
        if it % 15 == 0:
            c_chk = fft_convolve_self(x_new)
            if c_chk.max() > cur * 1.1:
                L *= 1.5
                x_new = project_simplex(y - (1.0 / L) * grad_a, S)
            elif it % 50 == 0:
                L = max(L * 0.93, beta * 2 * S / n)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 1000 == 0 and rng.random() < 0.15:
            noise = 0.005 * S / np.sqrt(n)
            y = project_simplex(a + noise * rng.standard_normal(n), S)
            t_mom = 1.0

    return best_a, best_score


def run_pair_swaps(a, n, S, max_time, rng):
    """
    Swap-based local search: move mass from peaks to valleys in the convolution.
    Works in full space, maintaining sum constraint.
    """
    a = np.clip(a.copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)
    best_score = c.max()
    eps = S / n * 0.005

    while time.monotonic() - start < max_time:
        cur = c.max()
        peak_lag = int(np.argmax(c))

        # Find pairs contributing to the peak
        i_min = max(0, peak_lag - (n - 1))
        i_max = min(peak_lag, n - 1)
        if i_min > i_max:
            break

        indices = np.arange(i_min, i_max + 1)
        partners = peak_lag - indices
        products = a[indices] * a[partners]

        # Top contributors
        n_top = min(20, len(indices))
        top_order = np.argsort(products)[::-1][:n_top]

        improved = False
        for ti in top_order:
            if time.monotonic() - start > max_time:
                break
            idx = int(indices[ti])
            pidx = int(partners[ti])

            # Try reducing the larger of the pair
            bigger = idx if a[idx] >= a[pidx] else pidx
            smaller = pidx if bigger == idx else idx

            if a[bigger] < eps:
                continue

            # Find a good place to add: where it won't create new peaks
            # Add to a position far from all current peaks
            top_peaks = np.argsort(c)[::-1][:5]
            # Score each candidate position by how much it would increase peaks
            # Simple heuristic: add to position with lowest average product with peak partners
            candidates = np.where(a < 0.5 * np.median(a) + eps)[0]
            if len(candidates) < 10:
                candidates = np.where(a < np.percentile(a, 40))[0]
            if len(candidates) == 0:
                candidates = np.arange(n)[::16]

            # Pick candidate that minimizes increase in top peaks
            best_cand = candidates[0]
            best_cost = np.inf
            sample_cands = candidates[rng.integers(0, len(candidates), min(50, len(candidates)))]
            for cand in sample_cands:
                cost = 0
                for pk in top_peaks[:3]:
                    j1 = int(pk) - int(cand)
                    j2 = int(cand)
                    if 0 <= j1 < n and 0 <= j2 < n:
                        cost += a[j1] + a[j2]
                if cost < best_cost:
                    best_cost = cost
                    best_cand = int(cand)

            delta = min(eps, a[bigger] * 0.5)
            a_test = a.copy()
            a_test[bigger] -= delta
            a_test[best_cand] += delta
            a_test = np.maximum(a_test, 0.0)
            a_test = a_test * (S / a_test.sum())

            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur - 1e-10:
                a = a_test
                c = c_test
                if c.max() < best_score:
                    best_score = c.max()
                    best_a = a.copy()
                improved = True
                break

        if not improved and rng.random() < 0.1:
            # Random pair swap
            i1 = rng.integers(n)
            i2 = rng.integers(n)
            if i1 != i2:
                delta = min(eps, a[i1] * 0.3, a[i2] * 0.3 + eps * 0.5)
                a_test = a.copy()
                a_test[i1] -= delta
                a_test[i2] += delta
                a_test = np.maximum(a_test, 0.0)
                a_test = a_test * (S / a_test.sum())
                c_test = fft_convolve_self(a_test)
                if c_test.max() <= cur * 1.001:
                    a = a_test
                    c = c_test

    return best_a, best_score
