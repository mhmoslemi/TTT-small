"""
Minimize 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy: Multi-phase optimization combining:
1. Coarse-to-fine FISTA with adaptive LSE smoothing
2. Excess annealing (hinge-loss on convolution peaks)
3. Exact subgradient peak shaving on top-k lags
4. Multi-length exploration (n = 4096, 8192, 16384)
5. Spectral-augmented update for faster convergence

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_to_sum(a, S):
    """Clip and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_smooth_max(c, beta):
    """Smooth approximation of max(c) using log-sum-exp."""
    cm = c.max()
    if beta * (c - cm).max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
    terms = np.exp(beta * (c - cm))
    return cm + np.log(terms.sum()) / beta


def lse_gradient(a, c, beta):
    """Gradient of LSE-smoothed max of convolution."""
    n = len(a)
    cm = c.max()
    scaled = beta * (c - cm)
    if scaled.max() > 500:
        beta = 500.0 / max(1e-15, (c - cm).max())
        scaled = beta * (c - cm)
    e = np.exp(scaled)
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def excess_gradient(a, c, T, p=2.0):
    """Gradient of sum((b_k - T)^p_+) — the 'excess' above threshold T."""
    n = len(a)
    excess = np.maximum(c - T, 0.0)
    if excess.max() < 1e-15:
        return np.zeros(n), 0.0
    if p == 1.0:
        w = np.where(excess > 0, 1.0, 0.0)
    else:
        w = p * excess ** (p - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n), 0.0
    w = w / wsum
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    obj = np.sum(excess ** p)
    return grad, obj


def peak_shave_gradient(a, c, k=16):
    """Exact subgradient from the top-k lags, weighted by excess."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    # Weight by how much above the mean
    mean_b = c.mean()
    weights = np.maximum(c[idx] - mean_b, 0.0)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights = weights / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = weights
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def run_fista_phase(a_init, n, S, max_time, rng, mode="lse",
                    beta_init=10.0, beta_rate=0.005, beta_max=5000.0):
    """FISTA with adaptive smoothing."""
    a = normalize_to_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = lse_smooth_max(c, beta) if mode == "lse" else c.max()
        if c.max() < best:
            best = c.max()
            best_a = y.copy()
            last_improve = it

        if mode == "lse":
            grad = lse_gradient(y, c, beta)
            step = 1.0 / (2.0 * beta * 0.3)
        elif mode == "excess":
            T = c.mean() + 0.5 * (c.max() - c.mean())
            grad, _ = excess_gradient(y, c, T, p=2.0)
            step = 1.0 / (0.5 * beta * 0.3)
        else:
            grad = peak_shave_gradient(y, c, k=min(32, n // 8))
            step = 2.0 * S / (n * (1.0 + 0.002 * it))

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: reduce if stuck

            step *= 0.3
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it

        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15) * min(gnorm, step * 0.5) * 2.0, S)

        # Simpler: just use step directly
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic restart

            noise_scale = 0.003 * S / np.sqrt(n) * decay

def run_excess_annealing(a_init, n, S, max_time, rng):
    """Anneal a threshold T downward, minimizing excess above T."""
    a = normalize_to_sum(a_init, S)
    start = time.monotonic()

    best_a = a.copy()

    y = a.copy()
    t = 1.0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max_time)
        # Anneal T: start high, decrease to near-mean

        T = max(T, T_end)

def beta_max_est(c, S, n):
    """Estimate a reasonable Lipschitz constant."""
    return 2.0 * S * S / (2 * n)

def upscale_sequence(a_small, n_large):
    """Upscale a sequence by interpolation."""
    n_small = len(a_small)
    t_old = np.linspace(0, 1, n_small)
    t_new = np.linspace(0, 1, n_large)
    return np.interp(t_new, t_old, a_small)
