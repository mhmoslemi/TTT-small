import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, dtype=np.float64)
    theta_cands = (cumsum - S) / idx
    valid = theta_cands <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def self_convolve(a):
    """Compute full self-convolution via FFT."""
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def gs_flatten(a, S, n_iter, gamma=0.5):
    """Gerchberg-Saxton style spectral flattening."""
    a = project_simplex(a.copy(), S)
    fsize = 1 << (2*len(a) - 1).bit_length()
    for it in range(n_iter):
        # Time to frequency
        A = fft(a, fsize)
        # Flatten magnitude
        mag = np.abs(A)
        # Extend to full linear convolution length
        # The DFT of the linear convolution is A^2
        # So the "target" magnitude for A^2 is constant
        # |A^2| = mag^2 should be constant -> mag should be constant
        # Blend: mag_new = (1-gamma)*mag + gamma*mean(mag)
        mag_mean = mag.mean()
        mag_new = (1 - gamma) * mag + gamma * mag_mean
        # Reconstruct with original phase
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a = np.real(ifft(A_new))[:len(a)]
        # Project to non-negative simplex
        a = project_simplex(a, S)
    return a

    c = self_convolve(a)
    # Find top-k lags

    grad = np.zeros(len(a))

        if 0 <= lag < n:

            valid_start = max(0, lag - n + 1)
            valid_end = min(n - 1, lag)
            for j in range(valid_start, valid_end + 1):

            # a[j-lag] is valid when 0 <= j-lag < n, i.e., lag <= j <= lag+n-1

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Hybrid optimization combining adaptive subgradient descent with
backtracking line search, spectral smoothing (Gerchberg-Saxton), power-law
weighting, simulated annealing, and multi-start diversity.
Key improvements over previous: backtracking line search, power-law weights
(smooth transition from top-k to all-lags), periodic spectral flattening,
adaptive top-k scheduling, and more diverse initializations.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_cands = (cumsum - S) / idx
    valid = theta_cands <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def self_convolve(a):
    """Full linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgrad_fft(a, weights):
    """
    Compute subgradient of max_lag b[lag] w.r.t. a.
    weights: array of length 2n-1 with non-negative weights on lags.
    grad[j] = 2 * sum_{k} weights[k] * a[k-j] for valid (k-j) in [0, n-1].
    This is 2 * cross_correlation(a, weights).
    """
    n = len(a)
    fsize = 1 << (n + len(weights) - 2).bit_length()
    A = fft(a, fsize)
    W = fft(weights, fsize)
    grad = np.real(ifft(np.conj(A) * W))[:n]
    return 2.0 * grad


def power_weights(c, p):
    """Compute power-law weights: w[k] = c[k]^p / sum(c[k']^p)."""
    c_clipped = np.maximum(c, 0.0)
    if p == 0:
        return np.full(len(c), 1.0 / len(c))
    max_c = c_clipped.max()
    if max_c < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    scaled = c_clipped / max_c
    wp = scaled ** p
    total = wp.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    return wp / total


def spectral_smooth(a, S, gamma, n_iters=5):
    """Apply Gerchberg-Saxton style spectral flattening on zero-padded sequence."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    for _ in range(n_iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad, N)
        mag = np.abs(A)
        phase = np.angle(A)
        mag_mean = mag.mean()
        # Blend towards flat spectrum
        mag_new = (1.0 - gamma) * mag + gamma * mag_mean
        # Avoid zeros in magnitude (keep phase meaningful)
        mag_new = np.maximum(mag_new, 1e-10)
        a_pad_new = np.real(ifft(mag_new * np.exp(1j * phase), N))
        a = a_pad_new[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a * (S / s)
        else:
            a = np.ones(n) * S / n
    return a

                   step_init=0.02, step_decay=0.9995,
                   smooth_every=2000, smooth_gamma=0.1,
                   anneal_every=3000, anneal_prob=0.3):

    Hybrid optimizer: adaptive subgradient + backtracking + spectral smoothing.
    Uses power-law weights (smooth version of top-k) with increasing exponent.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start_time = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    step = step_init * S / np.sqrt(n)
    no_improve = 0
    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start_time < max_time:
        it += 1

        # Compute convolution and score
        c = self_convolve(a)
        cur_score = c.max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive power exponent (increases over time to focus more on peaks)
        p = min(p_max, p_init * (1.0 + p_rate * it))

        # Compute weights and gradient
        weights = power_weights(c, p)
        grad = compute_subgrad_fft(a, weights)

        # Remove mean to maintain sum constraint
        grad = grad - grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Normalize gradient
        grad_dir = -grad / gnorm

        # Backtracking line search
        a_trial = project_simplex(a + step * grad_dir, S)
        c_trial = self_convolve(a_trial)
        trial_score = c_trial.max()

        # If trial is worse, try smaller step (up to 3 halvings)
        reductions = 0
        while trial_score > cur_score and reductions < 3:
            step *= 0.5
            a_trial = project_simplex(a + step * grad_dir, S)
            c_trial = self_convolve(a_trial)
            trial_score = c_trial.max()
            reductions += 1

        if trial_score <= cur_score:
            a = a_trial
        else:
            # If even tiny step doesn't help, try perturbation
            if no_improve > 500:
                noise_scale = 0.002 * S / np.sqrt(n)
                a = project_simplex(a + rng.normal(0, noise_scale, n), S)
            step *= 0.7

        # Slow step decay
        step *= step_decay
        step = max(step, 0.001 * S / np.sqrt(n))

        # Periodic spectral smoothing
        if it % smooth_every == 0:
            a = spectral_smooth(a, S, smooth_gamma, n_iters=3)
            no_improve = 0

        # Simulated annealing perturbation
        if it % anneal_every == 0 and rng.random() < anneal_prob:
            # Larger perturbation when stalled
            pert_scale = min(0.01, 0.001 * (1 + no_improve / 500)) * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, pert_scale, n), S)
            step = step_init * S / np.sqrt(n)

    return best_a, best_score


def run_fine_polish(a_init, n, S, max_time, rng):
    """
    Fine polishing: very small steps, focus on single peak,
    with careful backtracking.
    """
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start_time = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    step = 0.005 * S / np.sqrt(n)
    no_improve = 0

    while time.monotonic() - start_time < max_time:
        it += 1
        c = self_convolve(a)
        cur_score = c.max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            step = min(step * 1.005, 0.02 * S / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve > 200:
                step *= 0.8

        # Focus on the single peak (top-1 equivalent with very high power)
        p = 20.0 + 0.05 * it
        weights = power_weights(c, p)
        grad = compute_subgrad_fft(a, weights)
        grad = grad - grad.mean()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        grad_dir = -grad / gnorm
        a_trial = project_simplex(a + step * grad_dir, S)
        c_trial = self_convolve(a_trial)
        trial_score = c_trial.max()

        if trial_score <= cur_score:
            a = a_trial
        else:
            # Try half step
            half = step * 0.5
            a_half = project_simplex(a + half * grad_dir, S)
            c_half = self_convolve(a_half)
            if c_half.max() <= cur_score:
                a = a_half
                step = half
            else:
                # Micro-perturbation
                if rng.random() < 0.3:
                    a = project_simplex(a + rng.normal(0, 0.0003 * S / np.sqrt(n), n), S)
                step = max(step * 0.5, 0.0005 * S / np.sqrt(n))

    return best_a, best_score


def gen_init_uniform(n, S, rng):
    """Uniform with small random perturbation."""
    a = np.ones(n) * S / n
    noise = rng.normal(0, 0.01 * S / n, n)
    a = np.maximum(a + noise, 0)
    a = a * (S / a.sum())
    return a


def gen_init_random(n, S, rng):
    """Random non-negative, normalized."""
    a = rng.random(n)
    a = a * (S / a.sum())
    return a


def gen_init_bimodal(n, S, rng):
    """Bimodal: large at edges, small in middle."""
    t = np.linspace(0, 1, n)
    a = 1.0 + 0.5 * np.cos(2 * np.pi * t)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def gen_init_oscillatory(n, S, rng):
    """Oscillatory with random frequency."""
    freq = rng.integers(3, 20)
    t = np.arange(n) / n
    a = 1.0 + 0.4 * np.cos(2 * np.pi * freq * t) + 0.1 * rng.normal(0, 1, n)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def gen_init_from_parent(parent, n, S, rng):
    """Resample/interpolate parent to length n with perturbation."""
    if len(parent) == n:
        a = parent.copy().astype(np.float64)
    else:
        # Resample using interpolation
        old_t = np.linspace(0, 1, len(parent))
        new_t = np.linspace(0, 1, n)
        a = np.interp(new_t, old_t, parent)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    # Small perturbation
    noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
    a = np.maximum(a + noise, 0)
    a = a * (S / a.sum())
    return a


def propose_candidate(seed=42, budget_s=2385):
    """
    Multi-phase optimization:
    Phase 1: Quick exploration with diverse initializations (15% time)
    Phase 2: Best initialization refined with hybrid optimizer (45% time)
    Phase 3: Fine polishing (25% time)
    Phase 4: Additional refinement with different strategies (15% time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 25.0

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if p.size >= 64:
            parent = np.clip(p, 0, 1000)
    except Exception:
        pass

    n = 8192
    S = np.sqrt(2.0 * n)

    # If parent exists, use its length
    if parent is not None and parent.size >= 2048:
        n_candidate = parent.size
        if n_candidate in [4096, 8192, 16384, 32768]:
            n = n_candidate
            S = np.sqrt(2.0 * n)

    best_obj = np.inf
    best_a = None

    # === Phase 1: Quick exploration (15% of budget) ===
    phase1_time = 0.15 * (deadline - time.monotonic())
    per_init = phase1_time / 5

    inits = []
    inits.append(gen_init_uniform(n, S, rng))
    inits.append(gen_init_random(n, S, rng))
    inits.append(gen_init_bimodal(n, S, rng))
    inits.append(gen_init_oscillatory(n, S, rng))
    if parent is not None:
        inits.append(gen_init_from_parent(parent, n, S, rng))
    else:
        # Second oscillatory with different frequency
        freq = rng.integers(5, 30)
        t = np.arange(n) / n
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t) + 0.2 * np.cos(2 * np.pi * 7 * t)
        a = np.maximum(a, 0.01)
        inits.append(a * (S / a.sum()))

    phase1_results = []
    for i, a_init in enumerate(inits):
        if time.monotonic() > deadline - 60:
            break
        remaining = min(per_init, deadline - time.monotonic())
        if remaining < 5:
            break
        a_res, score = run_hybrid_opt(a_init, n, S, remaining, rng,
                                       p_init=5.0, p_max=50.0, p_rate=0.005,
                                       step_init=0.03, step_decay=0.999,
                                       smooth_every=1000, smooth_gamma=0.15,
                                       anneal_every=1500, anneal_prob=0.4)
        a_res = np.clip(a_res, 0, 1000)
        s = a_res.sum()
        if s > 0:
            a_res = a_res * (S / s)
        c = self_convolve(a_res)
        obj = 2.0 * n * c.max() / (S * S)
        phase1_results.append((obj, a_res.copy()))
        print("  Init %d: obj=%.6f (score=%.6f)" % (i, obj, c.max()))

    # Pick best from phase 1
    if phase1_results:
        phase1_results.sort(key=lambda x: x[0])
        best_obj = phase1_results[0][0]
        best_a = phase1_results[0][1]
        print("Phase 1 best: obj=%.6f" % best_obj)

    # === Phase 2: Refine best with longer hybrid run (40% time) ===
    remaining = deadline - time.monotonic()
    phase2_time = 0.40 * remaining
    if phase2_time > 20:
        print("Phase 2: Hybrid refinement (%.0f sec)" % phase2_time)
        t0 = time.monotonic()
        a_res, score = run_hybrid_opt(best_a, n, S, phase2_time, rng,
                                       p_init=10.0, p_max=200.0, p_rate=0.001,
                                       step_init=0.015, step_decay=0.9995,
                                       smooth_every=3000, smooth_gamma=0.08,
                                       anneal_every=4000, anneal_prob=0.2)
        a_res = np.clip(a_res, 0, 1000)
        s = a_res.sum()
        if s > 0:
            a_res = a_res * (S / s)
        c = self_convolve(a_res)
        obj = 2.0 * n * c.max() / (S * S)
        if obj < best_obj:
            best_obj = obj
            best_a = a_res
        print("  Phase 2: obj=%.6f [%.0fs]" % (obj, time.monotonic() - t0))

    # Also try second-best from phase 1 if available
    if len(phase1_results) > 1:
        remaining = deadline - time.monotonic()
        phase2b_time = 0.15 * remaining
        if phase2b_time > 10:
            print("Phase 2b: Refine second-best (%.0f sec)" % phase2b_time)
            t0 = time.monotonic()
            a_res, score = run_hybrid_opt(phase1_results[1][1], n, S, phase2b_time, rng,
                                           p_init=8.0, p_max=150.0, p_rate=0.002,
                                           step_init=0.02, step_decay=0.999,
                                           smooth_every=2000, smooth_gamma=0.1,
                                           anneal_every=3000, anneal_prob=0.25)
            a_res = np.clip(a_res, 0, 1000)
            s = a_res.sum()
            if s > 0:
                a_res = a_res * (S / s)
            c = self_convolve(a_res)
            obj = 2.0 * n * c.max() / (S * S)
            if obj < best_obj:
                best_obj = obj
                best_a = a_res
            print("  Phase 2b: obj=%.6f [%.0fs]" % (obj, time.monotonic() - t0))

    # === Phase 3: Fine polishing (25% time) ===
    remaining = deadline - time.monotonic()
    phase3_time = 0.25 * remaining
    if phase3_time > 10:
        print("Phase 3: Fine polishing (%.0f sec)" % phase3_time)
        t0 = time.monotonic()
        a_res, score = run_fine_polish(best_a, n, S, phase3_time, rng)
        a_res = np.clip(a_res, 0, 1000)
        s = a_res.sum()
        if s > 0:
            a_res = a_res * (S / s)
        c = self_convolve(a_res)
        obj = 2.0 * n * c.max() / (S * S)
        if obj < best_obj:
            best_obj = obj
            best_a = a_res
        print("  Phase 3: obj=%.6f [%.0fs]" % (obj, time.monotonic() - t0))

    # === Phase 4: Final micro-optimization and perturbation (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        print("Phase 4: Final refinement (%.0f sec)" % remaining)

        # Try a few perturbations followed by fine polish
        per_variant = remaining * 0.6 / 3
        for v in range(3):
            if time.monotonic() > deadline - 10:
                break
            if v == 0:
                perturbed = best_a + rng.normal(0, 0.003 * S / np.sqrt(n), n)
            elif v == 1:
                # Smooth perturbation
                noise = np.convolve(rng.normal(0, 1, n), np.ones(32) / 32, mode='same')
                perturbed = best_a + 0.002 * S / np.sqrt(n) * noise
            else:
                # Frequency-domain perturbation
                a_pad = np.zeros(1 << (2 * n - 1).bit_length())
                a_pad[:n] = best_a
                A = fft(a_pad)
                A *= (1.0 + 0.005 * rng.normal(0, 1, len(A)))
                perturbed = np.real(ifft(A))[:n]

            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps > 0:
                perturbed = perturbed * (S / ps)
            else:
                perturbed = best_a.copy()

            a_res, score = run_fine_polish(perturbed, n, S, per_variant, rng)
            a_res = np.clip(a_res, 0, 1000)
            s = a_res.sum()
            if s > 0:
                a_res = a_res * (S / s)
            c = self_convolve(a_res)
            obj = 2.0 * n * c.max() / (S * S)
            if obj < best_obj:
                best_obj = obj
                best_a = a_res
            print("  Variant %d: obj=%.6f" % (v, obj))

        # Very final polish with whatever time remains
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_res, score = run_fine_polish(best_a, n, S, remaining, rng)
            a_res = np.clip(a_res, 0, 1000)
            s = a_res.sum()
            if s > 0:
                a_res = a_res * (S / s)
            c = self_convolve(a_res)
            obj = 2.0 * n * c.max() / (S * S)
            if obj < best_obj:
                best_obj = obj
                best_a = a_res
            print("  Final: obj=%.6f" % obj)

    # === Output ===
    final_list = [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
    print("FINAL: length=%d, objective=%.6f" % (len(final_list), best_obj))

    # Verify with evaluate_sequence
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: %.6f" % ver)
        if ver < best_obj:
            best_obj = ver
    else:
        print("Verification failed! Using fallback.")
        # Fallback: uniform sequence
        final_list = [S / n] * n
        ver2 = evaluate_sequence(final_list)
        print("Fallback verified: %.6f" % ver2)

    return final_list
