"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. FISTA (Fast Iterative Shrinkage-Thresholding) with adaptive temperature subgradient
2. Smooth log-sum-exp optimization with Adam on log-parameters
3. Coordinate descent polish
4. Multi-length exploration (n=2048, 4096, 8192)

Key insight: The problem min max_m a^T Q_m a s.t. a>=0, sum(a)=S is convex.
FISTA provides O(1/k^2) convergence. We use adaptive temperature to transition
from multi-peak awareness to single-peak focus.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def proj_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S} using water-filling."""
    n = len(a)
    # Sort in descending order
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(a - theta, 0.0)


def fista_autocorr(a_init, S, time_budget, T_init=20.0, T_final=2000.0,
                   verbose=False, seed=42):
    """
    FISTA with adaptive temperature for min max autocorrelation.
    
    The problem: min f(a) = max_m c[m]  s.t. a >= 0, sum(a) = S
    where c = conv(a, a).
    
    Uses log-sum-exp smoothing for the gradient direction.
    """
    n = len(a_init)
    a = proj_simplex(a_init.copy().astype(np.float64), S)
    
    # Compute initial objective
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # FISTA state
    a_prev = a.copy()
    t = 1.0
    rng = np.random.default_rng(seed)
    
    # Adaptive step size
    step = S / np.sqrt(n)  # initial guess
    max_step = 2 * S / np.sqrt(n)
    min_step = 1e-6 * S / np.sqrt(n)
    
    start = time.monotonic()
    iter_count = 0
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.5)
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + (t - 1.0) / t_new * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * S / s_y
        
        # Compute autocorrelation at y
        c_y = fft_convolve_self(y)
        
        # Compute weighted gradient using log-sum-exp
        c_max = c_y.max()
        if c_max < 1e-30:
            break
        
        scaled = T * c_y / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c_y))
            w[np.argmax(c_y)] = 1.0
        else:
            w /= w_sum
        
        # Gradient: sum_m w[m] * 2 * Q_m * y
        # Q_m * y gives the cross-correlation of w with y
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        Y = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(Y)))[:n]
        
        # Backtracking line search
        alpha = step
        obj_y = c_y.max()
        accepted = False
        
        for bt in range(20):
            a_trial = y - alpha * grad
            a_trial = proj_simplex(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            
            if obj_trial <= obj_y + 1e-12 * obj_y:
                accepted = True
                break
            alpha *= 0.5
            if alpha < min_step:
                break
        
        if not accepted:
            a_trial = proj_simplex(y - min_step * grad, S)
        
        a_prev = a.copy()
        a = a_trial
        t = t_new
        step = alpha  # carry over step size
        
        # Update best
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        iter_count += 1
        
        # Restart if stuck
        if no_improve > max_no_improve:
            noise_scale = 0.01 * max(0.1, 1.0 - progress)
            a = best_a + noise_scale * S * rng.standard_normal(n) / np.sqrt(n)
            a = proj_simplex(a, S)
            a_prev = a.copy()
            t = 1.0
            no_improve = 0
        
        if verbose and iter_count % 100 == 0:
            print("    FISTA iter %d: obj=%.7f, T=%.1f, step=%.4f" % (iter_count, best_obj, T, alpha))
    
    return best_a, best_obj


def adam_log_autocorr(a_init, S, time_budget, T_init=10.0, T_final=500.0,
                      lr_init=0.01, beta1=0.9, beta2=0.999, verbose=False, seed=42):
    """
    Adam optimization on log-parameters with log-sum-exp smoothed objective.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-12)
    a = a * S / a.sum()
    
    # Log-parameterization: a_i = S * exp(theta_i) / sum(exp(theta))
    theta = np.log(a / a.sum())
    theta = theta - theta.mean()  # center for numerical stability
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.6)
        
        # Compute a from theta
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
        else:
            a = S * exp_theta / total
        
        # Compute autocorrelation
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Compute gradient of log-sum-exp
        c_max = c.max()
        if c_max < 1e-30:
            break
        
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        
        # Gradient w.r.t. a: sum_m w[m] * 2 * Q_m * a
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        g_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        # Chain rule: dL/dtheta_j = g_a[j]*a[j] - (a[j]/S) * sum_i g_a[i]*a[i]
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        lr = lr_init * (1.0 + 2.0 * (1.0 - progress))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
    
    return best_a, best_obj


def coordinate_descent(a_init, S, time_budget, verbose=False, seed=42):
    """Coordinate descent with local search around each coordinate."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = proj_simplex(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    sweep = 0
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        improved = False
        min_improvement = 1e-12
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            orig = a[i]
            if orig < 1e-14:
                continue
            
            best_val = orig
            best_local = best_obj
            
            # Try several multiplicative adjustments
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0, 0.9, 1.1]:
                a[i] = orig * factor
                a = proj_simplex(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local - min_improvement:
                    best_local = obj_check
                    best_val = orig * factor
            
            a[i] = best_val
            a = proj_simplex(a, S)
            
            if best_local < best_obj - min_improvement:
                best_obj = best_local
                best_a = a.copy()
                improved = True
        
        if not improved and sweep >= 5:
            break
    
    return best_a, best_obj


def get_initialization(name, n, S, rng):
    """Generate various initializations."""
    i = np.arange(n)
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "gaussian":
        a = np.exp(-0.5 * ((i - n / 2) / (n / 8)) ** 2)
        return a * (S / a.sum())
    elif name == "chirp":
        # Linear chirp (different frequency at different positions)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * (i * i) / (2 * n))
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "multi_chirp":
        a = 1.0 + 0.2 * np.cos(2 * np.pi * i * i / (2 * n)) + \
            0.1 * np.cos(2 * np.pi * i * i / n + 0.5)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "msequence_like":
        # Approximate m-sequence: alternating pattern with some randomness
        a = np.ones(n)
        pattern = rng.choice([0.5, 1.5], size=n)
        a = a * pattern
        return a * (S / a.sum())
    elif name == "cosine_window":
        a = 0.5 + 0.5 * np.cos(2 * np.pi * i / n)
        a = np.maximum(a, 0.001)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using convex-structured optimization.

Key insights:
1. The problem min max_k conv(a,a)[k] s.t. a>=0, sum(a)=S is CONVEX on the
   non-negative orthant (each conv(a,a)[k] is a convex quadratic there).
2. FISTA provides O(1/k^2) convergence for this convex problem.
3. The max of conv(a,a) is NOT necessarily at the center for asymmetric
   sequences, giving additional degrees of freedom.
4. Adaptive temperature in the log-sum-exp smoothing transitions from
   multi-peak awareness to single-peak focus.

Strategy:
- Phase 1: Quick multi-length exploration with diverse initializations
- Phase 2: Deep FISTA + Adam-log alternating optimization
- Phase 3: Targeted peak reduction with high-temperature subgradient
- Phase 4: Coordinate descent polish

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution conv(a,a) via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def proj_simplex(a, S):
    """Project onto {x >= 0, sum(x) = S} using sorted water-filling."""
    n = len(a)
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - S)
    if not np.any(cond):
        return np.full(n, S / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(a - theta, 0.0)


def compute_smoothed_grad(c, a, T):
    """
    Compute gradient of smoothed objective:
    f_T(a) = (1/T) * log(sum_m exp(T * c[m]))
    grad = sum_m w[m] * grad_m where w = softmax(T*c)
    grad_j = sum_m w[m] * a[m-j] (valid indices) = cross_corr(w, a)[j]
    Returns 2 * cross_corr(w, a)[:n] (factor 2 for symmetric quadratic).
    """
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)

    scaled = T * c / c_max
    scaled = scaled - scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    fsize = 1 << (3 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n], w


def fista_optimize(a_init, S, time_budget, T_init=20.0, T_final=2000.0,
                   verbose=False, seed=42, max_no_improve=300):
    """
    FISTA (accelerated projected subgradient) for min max conv(a,a)
    subject to a >= 0, sum(a) = S.
    """
    n = len(a_init)
    a = proj_simplex(a_init.copy().astype(np.float64), S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    a_prev = a.copy()
    t = 1.0
    rng = np.random.default_rng(seed)

    step = S * 0.5 / np.sqrt(n)
    max_step = S * 2.0 / np.sqrt(n)
    min_step = S * 1e-8 / np.sqrt(n)

    start = time.monotonic()
    iter_count = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.5)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + (t - 1.0) / t_new * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y * S / s_y
        else:
            y = a.copy()

        c_y = fft_convolve_self(y)
        obj_y = c_y.max()

        grad, w = compute_smoothed_grad(c_y, y, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        alpha = step
        accepted = False
        a_trial = y.copy()

        for _ in range(25):
            a_candidate = y - alpha * grad
            a_candidate = proj_simplex(a_candidate, S)
            c_trial = fft_convolve_self(a_candidate)
            obj_trial = c_trial.max()

            if obj_trial <= obj_y + 1e-10 * obj_y + 1e-15:
                a_trial = a_candidate
                accepted = True
                break
            alpha *= 0.5
            if alpha < min_step:
                a_trial = proj_simplex(y - min_step * grad, S)
                break

        a_prev = a.copy()
        a = a_trial
        t = t_new
        if accepted:
            step = min(alpha * 1.5, max_step)
        else:
            step = max(alpha * 0.5, min_step)

        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        iter_count += 1

        if no_improve > max_no_improve:
            noise_scale = 0.02 * S * max(0.05, 1.0 - progress) / np.sqrt(n)
            a = best_a + noise_scale * rng.standard_normal(n)
            a = proj_simplex(a, S)
            a_prev = a.copy()
            t = 1.0
            no_improve = 0

        if verbose and iter_count % 200 == 0:
            print("    FISTA iter %d: obj=%.7f, T=%.1f, step=%.4g" %
                  (iter_count, best_obj, T, step))

    return best_a, best_obj


def adam_log_optimize(a_init, S, time_budget, T_init=10.0, T_final=500.0,
                      lr_init=0.01, beta1=0.9, beta2=0.999,
                      verbose=False, seed=42):
    """
    Adam optimizer on log-parameters with log-sum-exp smoothed objective.
    Parameterizes a = S * softmax(theta) to ensure a > 0, sum(a) = S.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-10)
    a = a * S / a.sum()

    theta = np.log(a) - np.log(a.sum())
    theta -= theta.mean()

    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    t = 0
    eps = 1e-8

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        t += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.6)

        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
            theta = np.zeros(n)
        else:
            a = S * exp_theta / total

        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()

        c_max = c.max()
        if c_max < 1e-30:
            break

        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

        fsize = 1 << (3 * n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        g_a = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot

        m_arr = beta1 * m_arr + (1.0 - beta1) * g_theta
        v_arr = beta2 * v_arr + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m_arr / (1.0 - beta1 ** t)
        v_hat = v_arr / (1.0 - beta2 ** t)

        lr = lr_init * (0.5 + 1.5 * (1.0 - progress))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

    return best_a, best_obj


def targeted_peak_reduction(a_init, S, time_budget, verbose=False, seed=42):
    """
    Specifically target the top K peaks in the convolution and reduce them
    using a multi-peak subgradient approach with careful step control.
    """
    n = len(a_init)
    a = proj_simplex(a_init.copy().astype(np.float64), S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    start = time.monotonic()
    K = min(5, len(c))

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()

        # Find top K peaks
        top_idx = np.argsort(c)[::-1][:K]
        top_vals = c[top_idx]

        # Build weighted gradient focusing on top peaks
        # Use inverse-square weighting to emphasize the largest peaks
        w_peak = 1.0 / (top_vals + 1e-10)
        w_peak /= w_peak.sum()

        grad = np.zeros(n)
        for ki, idx in enumerate(top_idx):
            m = idx
            # grad_j += w_peak[ki] * a[m-j] (valid)
            lo = max(0, m - (n - 1))
            hi = min(m, n - 1)
            if lo <= hi:
                j_range = np.arange(lo, hi + 1)
                a_idx = m - j_range
                grad[j_range] += w_peak[ki] * a[a_idx]
        grad *= 2.0

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        grad_dir = grad / gnorm

        # Backtracking
        alpha = S * 0.1 / np.sqrt(n) * (1.0 - 0.8 * progress)
        for _ in range(15):
            a_trial = a - alpha * grad_dir
            a_trial = proj_simplex(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() <= cur_obj + 1e-10:
                a = a_trial
                break
            alpha *= 0.5
        else:
            a = proj_simplex(a - (alpha * grad_dir), S)

    return best_a, best_obj


def coordinate_descent(a_init, S, time_budget, verbose=False, seed=42):
    """
    Coordinate descent: for each coordinate, try several multiplicative
    adjustments and keep the best one.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = proj_simplex(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        order = rng.permutation(n)
        improved = False
        min_imp = 1e-11

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            orig = a[i]
            if orig < 1e-14:
                continue

            best_val = orig
            best_local = best_obj

            factors = [0.5, 0.7, 0.85, 1.15, 1.33, 1.5, 0.25, 2.0, 0.9, 1.1]
            for factor in factors:
                a[i] = orig * factor
                a = proj_simplex(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local - min_imp:
                    best_local = obj_check
                    best_val = orig * factor

            a[i] = best_val
            a = proj_simplex(a, S)
            c = fft_convolve_self(a)

            if c.max() < best_obj - min_imp:
                best_obj = c.max()
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_obj


def edge_taper_initialization(n, S, rng, style="cosine"):
    """Create a sequence with tapered edges to avoid large convolution peaks at boundaries."""
    i = np.arange(n)
    t = i / (n - 1)

    if style == "cosine":
        taper = 0.5 * (1 + np.cos(np.pi * (2 * t - 1)))
        inner = 1.0 + 0.3 * np.cos(2 * np.pi * 7 * t + rng.random() * 2) + \
                0.15 * np.cos(2 * np.pi * 13 * t + rng.random() * 2)
        a = taper * inner
    elif style == "quadratic":
        taper = 1.0 - (2 * t - 1) ** 2
        inner = 1.0 + 0.25 * np.sin(2 * np.pi * 11 * t + rng.random() * 2)
        a = taper * inner
    elif style == "smoothstep":
        taper = t * t * (3 - 2 * t) * (1 - t) * (1 - t) * (3 - 2 * (1 - t))
        taper = 0.5 * (1 + np.cos(np.pi * (2 * t - 1)))
        inner = 1.0 + 0.2 * np.cos(2 * np.pi * 5 * t) + 0.1 * np.cos(2 * np.pi * 17 * t)
        a = taper * inner
    else:
        a = np.ones(n)

    a = np.maximum(a, 1e-6)
    return a * S / a.sum()


def get_diverse_init(n, S, rng, init_id):
    """Generate diverse initializations based on init_id."""
    i = np.arange(n)
    styles = init_id % 6

    if styles == 0:
        return edge_taper_initialization(n, S, rng, "cosine")
    elif styles == 1:
        return edge_taper_initialization(n, S, rng, "quadratic")
    elif styles == 2:
        return edge_taper_initialization(n, S, rng, "smoothstep")
    elif styles == 3:
        a = edge_taper_initialization(n, S, rng, "cosine")
        a = a + 0.1 * S / n * rng.standard_normal(n)
        a = proj_simplex(a, S)
        return a
    elif styles == 4:
        # Asymmetric: one edge more tapered than the other
        t = i / (n - 1)
        left_taper = 0.5 * (1 + np.cos(np.pi * t * 1.5))
        right_taper = 0.5 * (1 + np.cos(np.pi * t * 0.8))
        taper = left_taper * right_taper
        a = taper * (1.0 + 0.2 * np.cos(2 * np.pi * 9 * t))
        a = np.maximum(a, 1e-6)
        return a * S / a.sum()
    else:
        # Random with edge suppression
        a = rng.random(n)
        edge_mask = np.ones(n)
        edge_frac = 0.1
        n_edge = int(n * edge_frac)
        edge_mask[:n_edge] = np.linspace(0.1, 1.0, n_edge)
        edge_mask[-n_edge:] = np.linspace(1.0, 0.1, n_edge)
        a = a * edge_mask
        a = np.maximum(a, 1e-10)
        return a * S / a.sum()


def propose_candidate(seed=42, budget_s=2350):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.

    Multi-phase strategy:
    1. Quick exploration of lengths and initializations
    2. Deep FISTA + Adam alternating
    3. Targeted peak reduction
    4. Coordinate descent polish
    5. Final high-temperature refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, budget_s - 40.0)

    global_best = np.inf
    global_a = None

    # Load provided sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            ng = len(given_a)
            Sg = np.sqrt(2.0 * ng)
            given_a = given_a * (Sg / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, eval=%.7f" % (ng, ver))
            if np.isfinite(ver):
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: " + str(e))

    # === PHASE 1: Exploration (10% time) ===
    p1_time = 0.10 * budget_s
    p1_deadline = time.monotonic() + p1_time

    print("=== Phase 1: Exploration (%.0fs) ===" % p1_time)

    lengths = [4096]
    if given_a is not None:
        ng = len(given_a)
        if ng not in lengths:
            lengths.insert(0, ng)
    for L in [2048, 8192]:
        if L not in lengths:
            lengths.append(L)

    n_attemps_total = len(lengths) * 4
    per_try = max(2.0, p1_time / max(1, n_attemps_total))

    init_counter = 0
    for n in lengths:
        if time.monotonic() > p1_deadline - 3:
            break
        S = np.sqrt(2.0 * n)

        for trial in range(4):
            if time.monotonic() > p1_deadline - 2:
                break
            a_init = get_diverse_init(n, S, rng, init_counter)
            init_counter += 1

            # Quick FISTA
            a_res, _ = fista_optimize(a_init, S, per_try, T_init=20, T_final=300,
                                      max_no_improve=100)
            a_res = proj_simplex(a_res, S)
            obj = evaluate_sequence(a_res.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_res.copy()
                print("  P1 n=%d trial=%d: obj=%.7f" % (n, trial, obj))

    # === PHASE 2: Deep optimization (50% time) ===
    p2_time = 0.50 * budget_s
    p2_deadline = time.monotonic() + p2_time

    if global_a is not None and p2_time > 20:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = proj_simplex(global_a.copy(), S)

        print("\n=== Phase 2: Deep optimization n=%d (%.0fs) ===" % (n, p2_time))

        # Round 1: FISTA moderate T
        r1 = 0.20 * p2_time
        a_res, _ = fista_optimize(a, S, r1, T_init=30, T_final=500, seed=1,
                                  max_no_improve=200)
        a = proj_simplex(a_res, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  R1 FISTA T30-500: obj=%.7f" % obj)

        # Round 2: Adam-log
        r2 = 0.20 * p2_time
        a_res, _ = adam_log_optimize(a, S, r2, T_init=50, T_final=800,
                                     lr_init=0.008, seed=2)
        a = proj_simplex(a_res, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  R2 AdamLog T50-800: obj=%.7f" % obj)

        # Round 3: FISTA high T (focus on dominant peak)
        r3 = 0.20 * p2_time
        a_res, _ = fista_optimize(a, S, r3, T_init=200, T_final=3000, seed=3,
                                  max_no_improve=200)
        a = proj_simplex(a_res, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  R3 FISTA T200-3000: obj=%.7f" % obj)

        # Round 4: Adam-log low T (broad smoothing)
        r4 = 0.20 * p2_time
        a_res, _ = adam_log_optimize(a, S, r4, T_init=10, T_final=200,
                                     lr_init=0.01, seed=4)
        a = proj_simplex(a_res, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  R4 AdamLog T10-200: obj=%.7f" % obj)

        # Round 5: FISTA very high T
        r5 = 0.20 * p2_time
        a_res, _ = fista_optimize(a, S, r5, T_init=500, T_final=5000, seed=5,
                                  max_no_improve=300)
        a = proj_simplex(a_res, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  R5 FISTA T500-5000: obj=%.7f" % obj)

    # === PHASE 3: Targeted peak reduction (15% time) ===
    p3_time = 0.15 * budget_s
    p3_deadline = time.monotonic() + p3_time

    if global_a is not None and p3_time > 10:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = proj_simplex(global_a.copy(), S)

        print("\n=== Phase 3: Targeted peak reduction (%.0fs) ===" % p3_time)

        a_res, _ = targeted_peak_reduction(a, S, p3_time, seed=10)
        a = proj_simplex(a_res, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  Targeted: obj=%.7f" % obj)

        # Also try with different K
        a_res, _ = targeted_peak_reduction(global_a, S, 5, seed=11)
        a = proj_simplex(a_res, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  Targeted2: obj=%.7f" % obj)

    # === PHASE 4: Coordinate descent polish (12% time) ===
    p4_time = 0.12 * budget_s
    p4_deadline = time.monotonic() + p4_time

    if global_a is not None and p4_time > 5:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = proj_simplex(global_a.copy(), S)

        print("\n=== Phase 4: CD polish (%.0fs) ===" % p4_time)

        a_res, _ = coordinate_descent(a, S, p4_time, seed=100)
        a = proj_simplex(a_res, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  CD: obj=%.7f" % obj)

    # === PHASE 5: Final alternating refinement (remaining time) ===
    p5_time = max(0.0, deadline - time.monotonic())

    if global_a is not None and p5_time > 15:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = proj_simplex(global_a.copy(), S)

        print("\n=== Phase 5: Final refinement (%.0fs) ===" % p5_time)

        n_rounds = max(2, int(p5_time / 20))
        per_round = p5_time / (n_rounds + 0.5)

        for r in range(n_rounds):
            if time.monotonic() > deadline - 3:
                break

            if r % 3 == 0:
                a_res, _ = fista_optimize(a, S, per_round, T_init=300, T_final=8000,
                                          seed=200 + r, max_no_improve=150)
            elif r % 3 == 1:
                a_res, _ = adam_log_optimize(a, S, per_round, T_init=300, T_final=8000,
                                             lr_init=0.004, seed=200 + r)
            else:
                a_res, _ = fista_optimize(a, S, per_round, T_init=100, T_final=2000,
                                          seed=200 + r, max_no_improve=150)

            a = proj_simplex(a_res, S)
            obj = evaluate_sequence(a.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a.copy()
                print("  P5 R%d: obj=%.7f" % (r, obj))

            # Occasional CD between rounds
            if r % 3 == 2 and p5_time > 30:
                remaining = deadline - time.monotonic()
                if remaining > 10:
                    cd_b = min(10.0, remaining * 0.3)
                    a_res, _ = coordinate_descent(a, S, cd_b, seed=300 + r)
                    a = proj_simplex(a_res, S)
                    obj = evaluate_sequence(a.tolist())
                    if np.isfinite(obj) and obj < global_best:
                        global_best = obj
                        global_a = a.copy()
                        print("  P5 R%d CD: obj=%.7f" % (r, obj))

    # Final validation
    if global_a is None:
        n_fb = 4096
        S_fb = np.sqrt(2.0 * n_fb)
        global_a = np.full(n_fb, S_fb / n_fb)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(4096)

    final_obj = evaluate_sequence(global_a.tolist())
    print("\nFINAL: n=%d, best=%.7f, verified=%.7f" %
          (len(global_a), global_best, final_obj))

    if not np.isfinite(final_obj):
        global_a = np.ones(4096)
        final_obj = evaluate_sequence(global_a.tolist())
        print("Fallback: obj=%.7f" % final_obj)

    return global_a.tolist()
