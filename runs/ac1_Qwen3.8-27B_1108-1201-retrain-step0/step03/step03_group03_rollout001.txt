import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_objective(a):
    """Compute the full objective value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_subgradient(a, c, top_k=10):

    n = len(a)

    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

                    new_a_lag_tgt = (a[tgt] + delta) * a[lag - tgt] if lag - tgt != tgt else (a[tgt] + delta)**2
                    new_contrib = new_a_lag_tgt if lag - tgt != tgt else 0
                    new_contrib = (a[tgt] + delta) * (a[lag - tgt] if lag - tgt != tgt else (a[tgt] + delta))
                gain = old_contrib - new_contrib + 0.1 * delta * (a[reduce_idx] - a[tgt])
                if gain > best_gain:
                    best_gain = gain
                    best_target = tgt

            if best_target >= 0 and best_gain > 0:
                delta = 0.02 * a[reduce_idx]
                a[reduce_idx] -= delta
                a[best_target] += delta
                a = np.maximum(a, 0.0)
                s_now = a.sum()
                a = a * (S / s_now)
                improved = True

        if not improved:
            # Try a random small perturbation
            noise = rng.normal(0, 0.005 * S / np.sqrt(n), n)
            a_new = project_simplex(a + noise, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur:
                a = a_new
            else:
                stall += 100

    return best_local, best_score


def run_spectral_flatten(a, n, S, max_time, rng, verbose=False):
    """
    Spectral error reduction: modify the FFT magnitude to reduce time-domain peaks.
    Alternates between time-domain projection (non-negativity) and frequency-domain flattening.
    """
    a = np.clip(a, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    N = 1 << (2 * n - 1).bit_length()
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    alpha = 0.3  # Blend factor for spectral modification

    while time.monotonic() - start < max_time:
        it += 1
        # Pad and FFT
        a_padded = np.zeros(N)
        a_padded[:n] = a
        A = fft(a_padded)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compute current |A|^2 and its IFFT (= circular autocorrelation)
        A_sq = A * A
        circ_corr = ifft(A_sq).real

        # Find peak in circular correlation (first n-1 lags are the "forward" part)
        # The linear convolution peak should be in circ_corr[0:n-1] or circ_corr[N-n+1:N]
        # Focus on the peak
        peak_val = circ_corr.max()
        peak_pos = circ_corr.argmax()

        # Modify magnitude: reduce frequencies that contribute most to the peak
        # Contribution of frequency f to time-domain lag k: |A[f]|^2 * exp(2*pi*i*f*k/N) / N
        # To reduce the peak at peak_pos, reduce |A[f]| where cos(2*pi*f*peak_pos/N) is positive
        f_indices = np.arange(N)
        contrib = np.cos(2 * np.pi * f_indices * peak_pos / N)
        # Reduce |A| where contrib > 0
        reduce_mask = contrib > 0.5
        mag[reduce_mask] *= (1 - alpha * 0.1)
        mag[~reduce_mask] *= (1 + alpha * 0.05)

        # Reconstruct
        A_new = mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * (S / s_new)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < best_score:
                best_score = c_new.max()
                best_local = a_new.copy()
            a = a_new
        else:
            break

        # Adaptive alpha
        if it % 100 == 0:
            alpha = max(0.05, alpha * 0.95)

    return best_local, best_score


def run_fista_refined(a_init, n, S, max_time, rng, beta_init=2.0, beta_max=1000.0):
    """
    Refined FISTA with:
    - Tighter backtracking
    - Adaptive beta ramp
    - Periodic restart from best
    - Better momentum schedule
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = beta_init * 3.0
    stall_count = 0
    last_best_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            last_best_it = it
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 500:
            # Restart from best with perturbation
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(best_local + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_momentum = 1.0
            stall_count = 0
            L = beta * 3.0
            continue

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search (more aggressive)
        step = 1.0 / L
        c_y = c  # Already computed
        f_y = c_y.max()
        x_new = None
        for bt in range(6):
            x_candidate = project_simplex(y - step * grad, S)
            if bt == 0:
                x_new = x_candidate
                continue
            c_cand = fft_convolve_self(x_candidate)
            f_cand = c_cand.max()
            if f_cand <= f_y - 0.1 * step * gnorm * gnorm:
                x_new = x_candidate
                break
            step *= 0.


4

        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        # Adaptive L
        c_x = fft_convolve_self(x_new)
        if c_x.max() > f_y + 1e-10:
            L *= 1.3
        elif it % 50 == 0:
            L = max(L * 0.9, beta * 2.0)

        # Nesterov momentum with damping
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        momentum_coeff = (t_momentum - 1.0) / t_new
        if it > 1000:
            momentum_coeff *= 0.8  # Damp momentum in later stages
        y = x_new + momentum_coeff * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Periodic structured perturbation
        if stall_count > 200 and it % 100 == 0:
            # Try a "segment swap" perturbation
            seg_len = rng.integers(n // 16, n // 4)
            seg_start = rng.integers(0, n - seg_len)
            a[seg_start:seg_start + seg_len] = a[seg_start:seg_start + seg_len][::-1]
            a = project_simplex(a, S)
            y = a.copy()
            t_momentum = 1.0

    return best_local, best_score


def run_active_set_v2(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=64):
    """
    Active-set method with:
    - Softer subgradient (top-k with decay)
    - Better step size adaptation
    - Periodic "pair rotation" perturbations
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (n * 5.0)

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()

        # Softer subgradient: weight peaks by their excess over second peak
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        base = top_vals[1] if len(top_vals) > 1 else top_vals[0] * 0.9
        excess = np.maximum(top_vals - base, 0.1 * base)
        w_full = np.zeros(2 * n - 1)
        w_full[top_indices] = excess / excess.sum()

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step
        step = 2.0 * S / (n * max(gnorm / (S / n), 0.05) * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        if it % 1000 == 0:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_momentum = 1.0

    return best_local, best_score


def make_anti_palindromic_init(n, rng, S):
    """Create an anti-palindromic initialization."""
    half = n // 2
    h = rng.random(half) * 0.8 + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 1e-10:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_multisine_init(n, rng, S, n_freqs=5):
    """Multi-sinusoidal initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n) * 0.5
    for _ in range(n_freqs):
        freq = rng.integers(2, 30)
        phase = rng.random() * 2 * np.pi
        amp = rng.random() * 0.3
        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_chirp_init(n, rng, S):
    """Chirp (quadratic phase) initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    chirp_rate = rng.integers(3, 15)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * (chirp_rate * tt * tt + rng.random() * 2 * np.pi))
    a = np.maximum(a, 0.05)
    a = a * (S / a.sum())
    return a


def safe_normalize(a, n):
    """Clamp and rescale."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * 1.0
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    
    Strategy: Multi-phase optimization combining LSE-FISTA, active-set subgradient,
    mass-transfer coordinate descent, and spectral flattening.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best_obj = np.inf
    global_a = None
    global_n = 0

    def update_best(a, n):
        nonlocal global_best_obj, global_a, global_n
        s = a.sum()
        if s < 0.01:
            return
        c = fft_convolve_self(a)
        obj = 2.0 * n * c.max() / (s * s)
        if obj < global_best_obj:
            global_best_obj = obj
            global_a = a.copy()
            global_n = n
            return True
        return False

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    primary_n = 8192
    if parent is not None and len(parent) >= 64:
        primary_n = len(parent)

    print("Primary length: %d, parent available: %s" % (primary_n, parent is not None))
    print("Starting at: %.6f" % time.strftime("%H:%M:%S"))

    S = np.sqrt(2.0 * primary_n)

    # === Phase 1: Initial optimization from parent or best init ===
    if parent is not None:
        a_start = parent * (S / parent.sum())
        t0 = time.monotonic()
        seg = min(300.0, max(60.0, (deadline - t0) * 0.2))
        print("Phase 1a: FISTA on parent (%.0f sec)" % seg)
        a1, s1 = run_fista_refined(a_start, primary_n, S, seg, rng,
                                    beta_init=5.0, beta_max=500.0)
        a1 = safe_normalize(a1, primary_n)
        if update_best(a1, primary_n):
            print("  FISTA parent: obj=%.6f" % global_best_obj)

        # Active set on result
        remaining = deadline - time.monotonic()
        seg2 = min(200.0, remaining * 0.15)
        if seg2 > 10:
            print("Phase 1b: ActiveSet on result (%.0f sec)" % seg2)
            a2, s2 = run_active_set_v2(a1, primary_n, S, seg2, rng,
                                        top_k_init=3, top_k_max=50)
            a2 = safe_normalize(a2, primary_n)
            if update_best(a2, primary_n):
                print("  ActiveSet: obj=%.6f" % global_best_obj)

        # Mass transfer
        remaining = deadline - time.monotonic()
        seg3 = min(200.0, remaining * 0.1)
        if seg3 > 10:
            print("Phase 1c: MassTransfer (%.0f sec)" % seg3)
            a3, s3 = run_mass_transfer(global_a, primary_n, S, seg3, rng)
            a3 = safe_normalize(a3, primary_n)
            if update_best(a3, primary_n):
                print("  MassTransfer: obj=%.6f" % global_best_obj)

    # === Phase 2: Try diverse initializations ===
    if parent is None or time.monotonic() < deadline - 600:
        print("Phase 2: Diverse initializations")
        inits = []
        inits.append(make_anti_palindromic_init(primary_n, rng, S))
        inits.append(make_multisine_init(primary_n, rng, S, n_freqs=3))
        inits.append(make_multisine_init(primary_n, rng, S, n_freqs=7))
        inits.append(make_chirp_init(primary_n, rng, S))
        inits.append(make_anti_palindromic_init(primary_n, rng, S))
        # Random
        rand_a = (rng.random(primary_n) + 0.1)
        inits.append(rand_a * (S / rand_a.sum()))

        remaining = deadline - time.monotonic()
        per_init = max(10.0, remaining * 0.3 / len(inits))

        for i, a_init in enumerate(inits):
            if time.monotonic() > deadline - 300:
                break
            t0 = time.monotonic()
            a_r, s_r = run_fista_refined(a_init, primary_n, S, per_init, rng,
                                          beta_init=2.0, beta_max=300.0)
            a_r = safe_normalize(a_r, primary_n)
            if update_best(a_r, primary_n):
                print("  Init %d: obj=%.6f" % (i, global_best_obj))

    print("After Phase 1+2: best obj=%.6f, n=%d" % (global_best_obj, global_n))

    # === Phase 3: Intensive polishing on best ===
    if global_a is not None and time.monotonic() < deadline - 60:
        n = global_n
        S_n = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()

        # 3a: High-beta FISTA
        seg = max(20.0, remaining * 0.25)
        a_p = run_fista_refined(global_a, n, S_n, seg, rng,
                                 beta_init=50.0, beta_max=2000.0)
        a_p = safe_normalize(a_p, n)
        if update_best(a_p, n):
            print("Phase 3a (hi-beta FISTA): %.6f" % global_best_obj)

        # 3b: Active set with high k
        remaining = deadline - time.monotonic()
        seg = max(20.0, remaining * 0.2)
        a_p = run_active_set_v2(global_a, n, S_n, seg, rng,
                                 top_k_init=10, top_k_max=100)
        a_p = safe_normalize(a_p, n)
        if update_best(a_p, n):
            print("Phase 3b (ActiveSet hi-k): %.6f" % global_best_obj)

        # 3c: Mass transfer
        remaining = deadline - time.monotonic()
        seg = max(20.0, remaining * 0.2)
        a_p = run_mass_transfer(global_a, n, S_n, seg, rng)
        a_p = safe_normalize(a_p, n)
        if update_best(a_p, n):
            print("Phase 3c (MassTransfer): %.6f" % global_best_obj)

        # 3d: Spectral flattening
        remaining = deadline - time.monotonic()
        seg = max(10.0, remaining * 0.1)
        a_p = run_spectral_flatten(global_a, n, S_n, seg, rng)
        a_p = safe_normalize(a_p, n)
        if update_best(a_p, n):
            print("Phase 3d (Spectral): %.6f" % global_best_obj)

        # 3e: Final FISTA polish
        remaining = deadline - time.monotonic()
        seg = max(10.0, remaining * 0.15)
        a_p = run_fista_refined(global_a, n, S_n, seg, rng,
                                 beta_init=200.0, beta_max=5000.0)
        a_p = safe_normalize(a_p, n)
        if update_best(a_p, n):
            print("Phase 3e (Final FISTA): %.6f" % global_best_obj)

    # === Phase 4: Try different lengths ===
    remaining = deadline - time.monotonic()
    if remaining > 200 and global_a is not None:
        for n_try in [4096, 16384]:
            if time.monotonic() > deadline - 150:
                break
            S_try = np.sqrt(2.0 * n_try)
            if n_try > len(global_a):
                # Upsample
                tt_old = np.linspace(0, 1, len(global_a))
                tt_new = np.linspace(0, 1, n_try)
                a_try = np.interp(tt_new, tt_old, global_a)
                a_try = np.maximum(a_try, 0.0)
                a_try = a_try * (S_try / a_try.sum())
            else:
                # Downsample
                tt_new = np.linspace(0, 1, len(global_a))
                tt_old = np.linspace(0, 1, n_try)
                a_try = np.interp(tt_old, tt_new, global_a)
                a_try = np.maximum(a_try, 0.0)
                a_try = a_try * (S_try / a_try.sum())

            seg = min(150.0, remaining * 0.3)
            a_r, s_r = run_fista_refined(a_try, n_try, S_try, seg, rng,
                                          beta_init=10.0, beta_max=500.0)
            a_r = safe_normalize(a_r, n_try)
            if update_best(a_r, n_try):
                print("Phase 4 (n=%d): %.6f" % (n_try, global_best_obj))

    # === Phase 5: Final intensive coordinate polish ===
    if global_a is not None and time.monotonic() < deadline - 30:
        n = global_n
        S_n = np.sqrt(2.0 * n)
        remaining = deadline - time.monotonic()
        seg = remaining * 0.5

        # Alternating mass transfer and small FISTA steps
        a_cur = global_a.copy()
        for round_i in range(3):
            if time.monotonic() > deadline - 30:
                break
            seg_sub = seg / 3.0
            a_p, s_p = run_mass_transfer(a_cur, n, S_n, seg_sub, rng)
            a_p = safe_normalize(a_p, n)
            if update_best(a_p, n):
                a_cur = a_p.copy()
                print("Phase 5 round %d: %.6f" % (round_i, global_best_obj))

    # === Final verification ===
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL: evaluate_sequence=%.6f, internal obj=%.6f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best_obj, len(global_a)))
    print("Time used: %.0f sec" % (time.monotonic() - (time.monotonic() - (deadline - time.monotonic()))))

    if not np.isfinite(ver):
        print("  WARNING: evaluate_sequence rejected, using fallback")
        final_list = [1.0] * 1024

    return final_list
