def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx

    if not np.any(valid):

    else:

def fft_convolve_self(a):

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

    """Gradient of sum_k weights[k] * (a*a)[k] with respect to a."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

    """
    Objective and gradient for the log-sum-exp smoothed max of autocorrelation.
    x is the log-parameterization: a_i = exp(x_i)
    We minimize log(sum_k exp(beta * (a*a)[k])) / beta

    c = fft_convolve_self(a)
    c_max = c.max()
    # log-sum-exp for numerical stability
    log_sum_exp = c_max / beta + np.log(np.sum(np.exp(beta * (c - c_max))))

    w = exp_bc / total

def lbfgs_direction(g, ses, m=10):
    """L-BFGS two-loop recursion.
    g: gradient
    ses: list of (s, y) pairs, most recent last
    Returns the search direction.
    """
    if not ses:
        return -g
    q = g.copy()
    alphas = []
    for s, y in reversed(ses):
        alpha = np.dot(s, y)
        if abs(alpha) > 1e-15:
            alpha = np.dot(s, q) / alpha
        alphas.append(alpha)
        q -= alpha * y
    alphas.reverse()
    # Initial Hessian approximation: H0 = I (or rho_0 * I)
    r = q  # H0 * q
    for (s, y), alpha in zip(ses, alphas):
        beta = np.dot(y, r)
        r += (alpha - beta) * s
    return -r

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)

    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient of sum_k weights[k] * (a*a)[k] w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def logsumexp_smoothed_obj_and_grad(a, beta):
    """
    Compute the log-sum-exp smoothed max of autocorrelation and its gradient.
    Returns (f_value, gradient).
    f = logsumexp(beta * c) / beta where c = conv(a,a)
    grad = (2/beta) * FFT-convolve(a, softmax(beta*c))
    """
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:

        f_val = (c_max + np.log(total)) / beta
    
    grad = (2.0 / beta) * compute_gradient_fft(a, w)

def lbfgs_two_loop(g, ses, m=20):
    """L-BFGS two-loop recursion."""
    if not ses:
        return -g
    
    q = g.copy()
    alphas = []
    
    for s_k, y_k in reversed(ses):
        yk_norm = np.dot(y_k, y_k)
        if yk_norm < 1e-15:
            alphas.append(0.0)
            continue
        rho = 1.0 / yk_norm
        alpha = rho * np.dot(s_k, q)
        alphas.append(alpha)
        q -= alpha * y_k
    
    alphas.reverse()
    
    # Initial Hessian dampening

"""
Minimize 2*n*max_conv/sum^2 using L-BFGS in log-space with continuation
on log-sum-exp smoothing parameter, followed by subgradient polishing.
Key innovation: L-BFGS superlinear convergence on smoothed surrogate in
softmax parameterization (no projection needed), continuation strategy,
and diverse restarts for robustness.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient of sum_k weights[k] * (a*a)[k] w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_safe(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    total = exp_x.sum()
    if total < 1e-300:
        return np.full(len(x), 1.0 / len(x))
    return exp_x / total


def lbfgs_direction(g, ses):
    """L-BFGS two-loop recursion to compute search direction."""
    if not ses:
        return -g

    q = g.copy()
    alphas = []

    for s_k, y_k in reversed(ses):
        yy = np.dot(y_k, y_k)
        if yy < 1e-30:
            alphas.append(0.0)
            continue
        rho = 1.0 / yy
        alpha = rho * np.dot(s_k, q)
        alphas.append(alpha)
        q = q - alpha * y_k

    alphas.reverse()

    # Gamma estimate from most recent pair
    if ses:
        s_last, y_last = ses[-1]
        sy = np.dot(s_last, y_last)
        yy = np.dot(y_last, y_last)
        if yy > 1e-30 and sy > 1e-30:
            gamma = sy / yy
        else:
            gamma = 1.0 / max(np.dot(g, g), 1e-30)
    else:
        gamma = 1.0

    r = gamma * q

    for (s_k, y_k), alpha in zip(ses, alphas):
        yy = np.dot(y_k, y_k)
        if yy < 1e-30:
            beta_k = 0.0
        else:
            rho = 1.0 / yy
            beta_k = rho * np.dot(y_k, r)
        r = r + (alpha - beta_k) * s_k

    return -r


def soft_obj_grad_x(x, beta):
    """
    Compute smoothed objective and gradient in log-space.
    x: log-parameterization (a = softmax(x))
    Returns (f_value, grad_x) where f approximates max_k conv(a,a)[k]
    """
    a = softmax_safe(x)
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        f_val = c_max
        w = np.full_like(c, 1.0 / len(c))
    else:
        f_val = (c_max + np.log(total)) / beta
        w = exp_bc / total

    # Gradient in a-space (treating a as free variables)
    grad_a = compute_gradient_fft(a, w)

    # Chain rule through softmax: grad_x[j] = a[j] * (grad_a[j] - dot(grad_a, a))
    ga_dot_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - ga_dot_a)

    return f_val, grad_x


def run_soft_lbfgs(a_init, n, S, max_time, beta_schedule, m=10, seed=42):
    """
    L-BFGS in log-space with continuation on beta.
    Works on the softmax parameterization (no projection needed).
    """
    rng = np.random.default_rng(seed)
    a = np.clip(a_init, 1e-10, 1000.0).copy()
    a = a / a.sum()
    x = np.log(a)
    x = x - x.mean()  # Gauge fixing: center x

    start = time.monotonic()
    best_c_max = np.inf
    best_a = a.copy()
    no_progress_count = 0
    last_best = best_c_max

    for beta_idx, beta in enumerate(beta_schedule):
        if time.monotonic() - start > max_time:
            break

        ses = []  # L-BFGS memory
        it = 0
        x = x.copy()  # Fresh start for each beta level
        x = x - x.mean()

        # Time budget per beta level
        remaining_levels = len(beta_schedule) - beta_idx
        time_per_level = (max_time - (time.monotonic() - start)) / max(1, remaining_levels)

        level_start = time.monotonic()
        while time.monotonic() - level_start < time_per_level:
            it += 1
            if time.monotonic() - start > max_time:
                break

            f, grad_x = soft_obj_grad_x(x, beta)
            gnorm = np.linalg.norm(grad_x)
            if gnorm < 1e-14:
                break

            # Check actual objective
            a_cur = softmax_safe(x)
            c = fft_convolve_self(a_cur)
            cur = c.max()
            if cur < best_c_max - 1e-12:
                best_c_max = cur
                best_a = a_cur.copy() * (S / a_cur.sum())
                no_progress_count = 0
                last_best = best_c_max
            else:
                no_progress_count += 1

            # L-BFGS direction
            d = lbfgs_direction(grad_x, ses)
            gd = np.dot(grad_x, d)
            if gd >= 0:
                # Not a descent direction, use neg gradient
                d = -grad_x
                gd = -gnorm * gnorm

            # Backtracking line search
            alpha = min(1.0, (2.0 * (f - cur) + 1e-6) / max(gd * (-1), 1e-15) if gd < 0 else 1.0)
            alpha = min(alpha, 5.0 / max(gnorm, 1e-10))
            improved = False
            for ls in range(50):
                x_try = x + alpha * d
                x_try = np.clip(x_try, -50.0, 50.0)
                x_try = x_try - x_try.mean()  # Keep gauge
                f_try, _ = soft_obj_grad_x(x_try, beta)
                if f_try < f + 1e-4 * alpha * gd:
                    improved = True
                    break
                alpha *= 0.5

            if not improved:
                # Fallback: small gradient step
                step = 0.01 / max(gnorm, 1e-10)
                x_try = x - step * grad_x
                x_try = np.clip(x_try, -50.0, 50.0)
                x_try = x_try - x_try.mean()
                if rng.random() < 0.1:
                    x_try += rng.normal(0, 0.01, n)
                    x_try = x_try - x_try.mean()

            # Update L-BFGS memory
            s_k = x_try - x
            _, grad_x_new = soft_obj_grad_x(x_try, beta)
            y_k = grad_x_new - grad_x
            sy = np.dot(s_k, y_k)
            ss = np.dot(s_k, s_k)
            if sy > 1e-14 and ss > 1e-14:
                ses.append((s_k, y_k))
                if len(ses) > m:
                    ses.pop(0)

            x = x_try

            # Convergence check
            if no_progress_count > 500 and it % 100 == 0:
                # Perturb and reset
                x += rng.normal(0, 0.02, n)
                x = x - x.mean()
                ses = []
                no_progress_count = 0

        a_check = softmax_safe(x) * S
        c_check = fft_convolve_self(a_check)
        if c_check.max() < best_c_max:
            best_c_max = c_check.max()
            best_a = a_check.copy()
        print("  L-BFGS beta=%d: best=%.6f (time=%.0fs)" % (beta, best_c_max, time.monotonic() - start))

    return best_a


def run_subgradient_polish(a_init, n, S, max_time, rng):
    """Fine-tune with subgradient method targeting the true max."""
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (n * 5.0)
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()
            stall = 0
            step = min(step * 1.005, S / n)
        else:
            stall += 1
            if stall > 200:
                step *= 0.7
                if stall > 2000:
                    break

        # Use very sharp softmax (essentially targeting top-1)
        beta = 500.0
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            w = np.full_like(c, 1.0 / len(c))
        else:
            w = exp_bc / total

        grad = compute_gradient_fft(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = project_simplex(a - step * grad / max(gnorm, 1e-15), S)
        c_new = fft_convolve_self(x_new)
        if c_new.max() < cur:
            a = x_new
        else:
            x_half = project_simplex(a - 0.3 * step * grad / max(gnorm, 1e-15), S)
            c_half = fft_convolve_self(x_half)
            if c_half.max() < cur:
                a = x_half
            elif rng.random() < 0.3:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)

        if it % 5000 == 0:
            print("    Subgrad it=%d: score=%.6f step=%.6e" % (it, best_score, step))

    return best_local, best_score


def run_fista_refined(a_init, n, S, max_time, rng, beta_init=200.0, beta_max=5000.0):
    """Refined FISTA with higher initial beta for final polishing."""
    a = np.clip(a_init, 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 1.0 / (3.0 * beta_init)
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.02 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            stall = 0
        else:
            stall += 1

        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            w = np.full_like(c, 1.0 / len(c))
        else:
            w = exp_bc / total

        grad = compute_gradient_fft(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if stall > 300:
            step *= 0.5
            stall = 0

        if it % 2000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

        a_final, _ = run_subgradient_polish(global_a, n, S, remaining, rng)
        a_final = safe_normalize(a_final, n)
        c_final = fft_convolve_self(a_final)
        if c_final.max() < global_best:
            global_best = c_final.max()
            global_a = a_final.copy()
            print("  Final: %.6f (obj %.6f)" % (global_best, 2*n*global_best/S**2))

    # === Output ===
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    final_obj = 2.0 * n * global_best / (S * S)
    print("FINAL: n=%d, score=%.6f, obj=%.6f" % (len(final_list), global_best, final_obj))

    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified: %.6f" % ver)
    else:
        print("Verification failed, fallback")
        final_list = [0.5] * 256

    return final_list
