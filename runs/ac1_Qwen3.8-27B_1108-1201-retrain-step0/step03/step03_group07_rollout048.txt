```python
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a combination of:
1. Log-sum-exp smoothed gradient descent with Nesterov momentum
2. LP-based best-response direction finding
3. Subgradient mirror descent with adaptive parameters
4. Coordinate descent polishing

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_ab(a, b):
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def smooth_max(c, T):
    """Compute T * log(sum(exp(c/T))) and its gradient."""
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    log_sum = c_max + T * np.log(exp_c.sum())
    p = exp_c / exp_c.sum()
    return log_sum, p


def smooth_max_grad(c, a, p):
    """Gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    # grad[m] = 2 * sum_k p[k] * a[k-m] (for valid indices)
    # = 2 * (p * a_rev)[n-1+m] where a_rev = a[::-1]
    a_rev = a[::-1]
    out_len = len(c) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    P = fft(p, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(P * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a hybrid approach:
1. Log-sum-exp smoothed gradient descent with Nesterov momentum for fast initial convergence
2. LP-based best-response direction finding (cross-convolution minimization) for breaking plateaus
3. Adaptive subgradient mirror descent with multi-scale peak chasing
4. Coordinate descent polishing for final refinement
The key innovation is the LP step which finds the optimal "partner" sequence g
that minimizes the peak of (f*g), providing a strong descent direction.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_ab(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def smooth_max_and_probs(c, T):
    """Compute smoothed max and probability weights."""
    c_shifted = (c - c.max()) / T
    exp_c = np.exp(c_shifted)
    p = exp_c / exp_c.sum()
    log_sum = c.max() + T * np.log(exp_c.sum())
    return log_sum, p


def smooth_max_grad(c, a, p):
    """Gradient of smoothed max of self-convolution w.r.t. a."""
    n = len(a)
    a_rev = a[::-1]
    out_len = len(c) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    P = fft(p, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(P * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def subgrad_topk(c, a, top_k):
    """Subgradient using top-k peaks (uniform weights)."""
    n = len(a)
    top_k = max(1, min(top_k, len(c)))
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    p = np.zeros(len(c))
    p[top_idx] = 1.0 / top_k
    a_rev = a[::-1]
    out_len = len(c) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    P = fft(p, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(P * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def mirror_descent_smoothed(a0, time_budget, T_start=None, T_end=None,
                            lr0=0.1, momentum=0.9, verbose=False):
    """
    Smoothed gradient descent with Nesterov momentum.
    Uses log-sum-exp to smooth the max, with adaptive temperature.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    best_iter = 0

    if T_start is None:
        T_start = c.max() / 5.0
    if T_end is None:
        T_end = c.max() / 50.0

    v = np.zeros(n)
    lr = lr0
    start = time.monotonic()
    iter_no = 0

    # Pre-compute total iterations estimate for T schedule
    total_iters_estimate = int(time_budget * 500)  # rough estimate

    while time.monotonic() - start < time_budget:
        progress = min(1.0, iter_no / total_iters_estimate)
        T_cur = T_start + (T_end - T_start) * progress

        _, p = smooth_max_and_probs(c, T_cur)
        grad = smooth_max_grad(c, a, p)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Nesterov momentum
        v = momentum * v + grad / gnorm
        vnorm = np.linalg.norm(v)
        if vnorm > 0:
            v = v / vnorm

        # Adaptive step with backtracking
        accepted = False
        for bt in range(10):
            lr_try = lr * (0.7 ** bt)
            a_trial = a - lr_try * v
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)
            if obj_trial <= obj_value(a, c) * (1.0 + 1e-8):
                a = a_trial
                c = c_trial
                accepted = True
                break

        if not accepted:
            lr *= 0.5
            continue

        if accepted and bt == 0:
            lr = min(lr * 1.05, lr0 * 3.0)

        cur_obj = obj_value(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no

        if iter_no - best_iter > 2000:
            # Perturb
            noise = np.random.randn(n) * 0.02 * S / np.sqrt(n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.1
            best_iter = iter_no

        iter_no += 1
        if verbose and iter_no % 1000 == 0:
            elapsed = time.monotonic() - start
            print(f"      [smooth] iter={iter_no}, obj={cur_obj:.6f}, "
                  f"best={best_obj:.6f}, T={T_cur:.4f}, t={elapsed:.1f}s")

    return best_a, best_obj


def lp_best_response(f, S, max_time=10.0):
    """
    Solve LP: minimize t subject to (f*g)[k] <= t for all k, sum(g) = S, g >= 0.
    Returns (g, t_opt) or (None, None) if LP fails.
    """
    n = len(f)
    num_constraints = 2 * n - 1
    num_vars = n + 1  # g[0..n-1] + t

    # Build sparse constraint matrix
    row_indices = []
    col_indices = []
    values = []

    for k in range(num_constraints):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            row_indices.append(k)
            col_indices.append(j)
            values.append(f[j])
        # -t column
        row_indices.append(k)
        col_indices.append(n)
        values.append(-1.0)

    A_rows = np.array(row_indices, dtype=np.int32)
    A_cols = np.array(col_indices, dtype=np.int32)
    A_vals = np.array(values, dtype=np.float64)

    A = lil_matrix((num_constraints, num_vars))
    A[A_rows, A_cols] = A_vals
    A_sparse = A.tocsr()

    b = np.zeros(num_constraints)
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    # Objective: minimize t (last variable)
    c_obj = np.zeros(num_vars)
    c_obj[n] = 1.0

    # Bounds: g[i] >= 0, t is free
    bounds = []
    for i in range(n):
        bounds.append((0, None))
    bounds.append((None, None))  # t is free

    start = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A_sparse, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        elapsed = time.monotonic() - start

        if res.status == 0 and elapsed < max_time:
            g = res.x[:n]
            t_opt = res.x[n]
            g = np.maximum(g, 0.0)
            g = g * (S / g.sum()) if g.sum() > 1e-10 else f.copy()
            return g, t_opt
    except Exception:
        pass
    return None, None


def lp_iterate(a0, time_budget, max_lp_time=5.0, alpha=0.3, verbose=True):
    """
    Iterate: solve LP for best response g given current f, then update f.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    iter_no = 0

    while time.monotonic() - start < time_budget:
        # Solve LP
        g, t_opt = lp_best_response(a, S, max_time=max_lp_time)
        if g is None:
            break

        # Update
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new * (S / a_new.sum())

        c_new = fft_convolve_self(a_new)
        obj_new = obj_value(a_new, c_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
            # Slightly increase alpha on success
            alpha = min(alpha * 1.02, 0.8)
        else:
            # Try smaller alpha
            alpha = max(alpha * 0.7, 0.05)
            a_new2 = (1.0 - alpha) * a + alpha * g
            a_new2 = np.maximum(a_new2, 0.0)
            a_new2 = a_new2 * (S / a_new2.sum())
            c_new2 = fft_convolve_self(a_new2)
            obj_new2 = obj_value(a_new2, c_new2)
            if obj_new2 < best_obj:
                best_obj = obj_new2
                best_a = a_new2.copy()
                a = a_new2
                c = c_new2

        iter_no += 1
        if verbose and iter_no % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"      [LP] iter={iter_no}, obj={obj_value(a, c):.6f}, "
                  f"best={best_obj:.6f}, t_lp={t_opt:.4f}, t={elapsed:.1f}s")

    return best_a, best_obj


def subgrad_mirror(a0, time_budget, eta0=0.3, momentum=0.85,
                   top_k_start=200, top_k_end=1, soft_T=0.0,
                   stagnation_limit=1500, verbose=False):
    """Adaptive subgradient mirror descent with multiplicative updates."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)
    eta = eta0
    iter_no = 0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        progress = min(1.0, iter_no / 50000)
        top_k_cur = max(1, int(top_k_start + (top_k_end - top_k_start) * progress))

        if soft_T > 0:
            _, p = smooth_max_and_probs(c, soft_T)
            grad = smooth_max_grad(c, a, p)
        else:
            grad = subgrad_topk(c, a, top_k_cur)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        # Multiplicative update with backtracking
        accepted = False
        for bt in range(8):
            eta_try = eta * (0.7 ** bt)
            a_trial = a * np.exp(-eta_try * v)
            a_trial = np.maximum(a_trial, 1e-300)
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_convolve_self(a_trial)
            obj_trial = obj_value(a_trial, c_trial)
            if obj_trial <= obj_value(a, c) * (1.0 + 1e-10):
                a = a_trial
                c = c_trial
                accepted = True
                break

        if not accepted:
            eta = max(eta * 0.8, eta0 * 0.05)
        else:
            if bt == 0:
                eta = min(eta * 1.03, eta0 * 2.5)

        cur_obj = obj_value(a, c)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no

        if iter_no - best_iter > stagnation_limit:
            noise_scale = 0.03 * S / np.sqrt(n)
            noise = np.random.randn(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.2
            eta = eta0
            best_iter = iter_no

        eta *= 0.99998
        iter_no += 1
        if verbose and iter_no % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"      [mirror] iter={iter_no}, obj={cur_obj:.6f}, "
                  f"best={best_obj:.6f}, top_k={top_k_cur}, t={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_polish(a0, S, time_budget, max_sweeps=500):
    """Coordinate descent: try adjusting one element at a time."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()

    factors = [0.8, 0.9, 1.1, 1.2]

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        # Process in random order for diversity
        idx_order = np.random.permutation(n)
        for i in idx_order:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-15:
                continue
            for factor in factors:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    continue
                a_norm = a * (S / s)
                c_new = fft_convolve_self(a_norm)
                obj_new = c_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_norm.copy()
                    a = a_norm
                    c = c_new
                    improved = True
                    break
            a[i] = a[i]  # keep current value
            s = a.sum()
            if s > 0:
                a = a * (S / s)

        if not improved:
            break

    return best_a, best_obj * S * S / (2 * n)

    a_2h = (np.exp(-0.5 * ((i - n * 0.3) / (n * 0.06)) ** 2) +
            0.7 * np.exp(-0.5 * ((i - n * 0.7) / (n * 0.08)) ** 2))
    a_2h = np.maximum(a_2h, 0.001)
    a_2h *= S / a_2h.sum()
    candidates.append(('2hump', a_2h))

    # Random
    a_rand = rng.random(n)
    a_rand = np.maximum(a_rand, 0.01)
    a_rand *= S / a_rand.sum()
    candidates.append(('rand', a_rand))

    # "Belt" pattern - concentrated in middle
    a_belt = np.exp(-0.5 * ((np.arange(n) - n / 2) / (n * 0.15)) ** 2)
    a_belt = np.maximum(a_belt, 0.001)
    a_belt *= S / a_belt.sum()
    candidates.append(('belt', a_belt))

    # --- Phase 1: Quick exploration (10% time) ---
    explore_time = 0.10 * (deadline - time.monotonic())
    print(f"\nPhase 1: Exploration ({explore_time:.0f}s)")

    per_candidate_time = explore_time / len(candidates)
    for name, a_cand in candidates:
        if time.monotonic() > deadline - 60:
            break
        a_best, obj_best = subgrad_mirror(
            a_cand, per_candidate_time,
            eta0=0.2, momentum=0.8,
            top_k_start=100, top_k_end=20,
            soft_T=0.0, verbose=False
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  {name}: obj={obj_best:.6f} (NEW BEST)")
        else:
            print(f"  {name}: obj={obj_best:.6f}")

    # --- Phase 2: Smoothed gradient descent (20% time) ---
    smooth_time = 0.20 * (deadline - time.monotonic())
    print(f"\nPhase 2: Smoothed gradient descent ({smooth_time:.0f}s)")

    if global_a is not None and smooth_time > 10:
        c_cur = fft_convolve_self(global_a)
        T_start = c_cur.max() / 5.0
        T_end = c_cur.max() / 80.0

        a_best, obj_best = mirror_descent_smoothed(
            global_a, smooth_time,
            T_start=T_start, T_end=T_end,
            lr0=0.05, momentum=0.9, verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  -> {obj_best:.6f} (NEW BEST)")

    # --- Phase 3: LP-based optimization (35% time) ---
    lp_time = 0.35 * (deadline - time.monotonic())
    print(f"\nPhase 3: LP best-response ({lp_time:.0f}s)")

    if global_a is not None and lp_time > 30:
        # Split into two sub-phases
        lp_time_a = lp_time * 0.6
        lp_time_b = lp_time * 0.4

        # 3a: LP with moderate alpha
        print("  3a: LP (alpha=0.3)")
        a_best, obj_best = lp_iterate(
            global_a, lp_time_a,
            max_lp_time=5.0, alpha=0.3, verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f} (NEW BEST)")

        # 3b: LP with smaller alpha (more conservative)
        print("  3b: LP (alpha=0.1)")
        a_best, obj_best = lp_iterate(
            global_a, lp_time_b,
            max_lp_time=5.0, alpha=0.1, verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f} (NEW BEST)")

    # --- Phase 4: Mirror descent exploration (20% time) ---
    mirror_time = 0.20 * (deadline - time.monotonic())
    print(f"\nPhase 4: Mirror descent ({mirror_time:.0f}s)")

    if global_a is not None and mirror_time > 10:
        # Multiple rounds with different parameters
        sub_time = mirror_time / 3.0

        # 4a: Broad (many peaks)
        print("  4a: Broad focus")
        a_best, obj_best = subgrad_mirror(
            global_a, sub_time,
            eta0=0.25, momentum=0.85,
            top_k_start=300, top_k_end=50,
            soft_T=3.0, verbose=False
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f} (NEW BEST)")

        # 4b: Medium
        print("  4b: Medium focus")
        a_best, obj_best = subgrad_mirror(
            global_a, sub_time,
            eta0=0.15, momentum=0.88,
            top_k_start=30, top_k_end=5,
            soft_T=0.0, verbose=False
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f} (NEW BEST)")

        # 4c: Tight (single peak)
        print("  4c: Tight focus")
        a_best, obj_best = subgrad_mirror(
            global_a, sub_time,
            eta0=0.1, momentum=0.92,
            top_k_start=3, top_k_end=1,
            soft_T=0.0, verbose=False
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f} (NEW BEST)")

    # --- Phase 5: Final polish (15% time) ---
    polish_time = deadline - time.monotonic() - 10.0
    print(f"\nPhase 5: Polish ({polish_time:.0f}s)")

    if global_a is not None and polish_time > 5:
        # 5a: Coordinate descent
        polish_5a = polish_time * 0.4
        print(f"  5a: Coordinate descent ({polish_5a:.0f}s)")
        S_cur = global_a.sum()
        a_best, obj_best = coordinate_polish(global_a, S_cur, polish_5a)
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f} (NEW BEST)")

        # 5b: Very fine mirror descent
        polish_5b = polish_time * 0.35
        print(f"  5b: Fine mirror ({polish_5b:.0f}s)")
        a_best, obj_best = subgrad_mirror(
            global_a, polish_5b,
            eta0=0.03, momentum=0.95,
            top_k_start=2, top_k_end=1,
            soft_T=0.0, stagnation_limit=500, verbose=False
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f} (NEW BEST)")

        # 5c: Final LP with very small alpha
        polish_5c = polish_time * 0.25
        print(f"  5c: Final LP ({polish_5c:.0f}s)")
        if polish_5c > 10:
            a_best, obj_best = lp_iterate(
                global_a, polish_5c,
                max_lp_time=4.0, alpha=0.05, verbose=False
            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print(f"    -> {obj_best:.6f} (NEW BEST)")

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    elapsed_total = time.monotonic() - start_total
    print(f"\nFINAL: local_obj={global_best:.6f}, verified={ver:.6f}, "
          f"n={len(final_a)}, time={elapsed_total:.1f}s")

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.6f}")

    return final_a
```
