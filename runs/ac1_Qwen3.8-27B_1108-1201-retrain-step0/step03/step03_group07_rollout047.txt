    n = len(a)
    size = 1 << (2*n - 2).bit_length()
    A = fft(a, size)

    for key in params:
        m = state['m'][key] if key in state['m'] else np.zeros_like(params[key])
        v = state['v'][key] if key in state['v'] else np.zeros_like(params[key])
        t = state['t']

        state['m'][key] = m
        state['v'][key] = v
    state['t'] += 1

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using smooth log-sum-exp optimization
with Adam, multi-scale T-annealing, diverse initializations, and peak-targeted
polishing. Key insight: use log-sum-exp to create a differentiable surrogate
for the max of the self-convolution, enabling efficient gradient-based search
with adaptive per-coordinate learning rates.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def linear_conv(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    return np.real(ifft(fft(a, size) * fft(b, size)))[:n]


def self_conv(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    A = fft(a, size)
    return np.real(ifft(A * A))[:2 * n - 1]


def obj_val(a):
    """Exact objective: 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * c.max() / (S * S)


def smooth_obj_grad(a, T):
    """Compute log-sum-exp smooth objective and gradient w.r.t. a."""
    c = self_conv(a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    log_sum = Tc_max + np.log(exp_Tc.sum())
    obj_smooth = log_sum / T
    p = exp_Tc / exp_Tc.sum()
    # grad_a[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # = 2 * (a * p_rev)[2n-2-i]
    n = len(a)
    p_rev = p[::-1]
    ap = linear_conv(a, p_rev)
    idx = 2 * n - 2 - np.arange(n)
    grad_a = 2.0 * ap[idx]
    return obj_smooth, grad_a


def adam_update(z, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update step."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * grad * grad
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v


def optimize_smooth(a0, time_budget, T_start=0.3, T_end=80.0,
                    lr=0.5, warmup_frac=0.1, anneal_power=1.5,
                    renorm_interval=200, verbose=True, tag=""):
    """
    Smooth optimization using log-sum-exp with Adam and T-annealing.
    Parameterizes a = exp(z) for guaranteed positivity.
    """
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    n = len(a)
    S_target = np.sqrt(2.0 * n)
    a = a / a.sum() * S_target

    z = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 1

    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    total_iters = 0
    stagnation_counter = 0
    best_iter_at = 0

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        # T schedule: quadratic ease from T_start to T_end
        progress = min(1.0, total_iters / max(1, int(time_budget * 50)))
        # Use a fraction of elapsed time for annealing
        time_progress = min(1.0, elapsed / (time_budget * 0.7))
        T = T_start + (T_end - T_start) * (time_progress ** anneal_power)

        # Compute smooth gradient
        _, grad_a = smooth_obj_grad(a, T)
        grad_z = a * grad_a

        # Adaptive learning rate: decrease as T increases (sharper peaks need smaller steps)
        lr_cur = lr * (T_start / max(T, T_start)) ** 0.5
        lr_cur = max(lr_cur, 0.01)

        # Adam update
        z, m, v = adam_update(z, grad_z, m, v, t_count, lr_cur)
        t_count += 1
        total_iters += 1

        # Update a from z
        a = np.exp(z)

        # Periodic renormalization for numerical stability
        if total_iters % renorm_interval == 0:
            S_cur = a.sum()
            a = a / S_cur * S_target
            z = np.log(np.maximum(a, 1e-15))

        # Track best
        cur_obj = obj_val(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter_at = total_iters
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        if stagnation_counter > 2000 and elapsed < time_budget * 0.85:
            noise_scale = 0.05 * S_target / np.sqrt(n)
            a = a + np.random.randn(n) * noise_scale
            a = np.maximum(a, 1e-15)
            a = a / a.sum() * S_target
            z = np.log(a)
            m *= 0.1
            v *= 0.1
            stagnation_counter = 0

        if verbose and total_iters % 2000 == 0:
            print(f"    [{tag}] iter={total_iters}, T={T:.1f}, obj={cur_obj:.6f}, "
                  f"best={best_obj:.6f}, t={elapsed:.1f}s")

    return best_a, best_obj


def peak_polish(a0, time_budget, verbose=True, tag=""):
    """
    Targeted polish: repeatedly find the peak position, compute exact subgradient,
    and apply multiplicative updates to reduce it.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_obj = obj_val(a)
    start = time.monotonic()
    v_buf = np.zeros(n)
    lr = 0.02

    while time.monotonic() - start < time_budget:
        c = self_conv(a)
        cur_obj = obj_val(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # Find top peaks
        top_k = min(5, 2 * n - 1)
        peak_idx = np.argsort(c, kind='stable')[::-1][:top_k]
        peak_vals = c[peak_idx]
        peak_max = peak_vals[0]

        # Weighted subgradient
        weights = np.exp((peak_vals - peak_max) * 3.0)
        weights /= weights.sum()

        # grad[i] = 2 * sum_k w_k * a[k-i] for valid k-i
        w_full = np.zeros(2 * n - 1)
        w_full[peak_idx] = weights
        # grad[i] = 2 * (a * w_rev)[2n-2-i]
        w_rev = w_full[::-1]
        aw = linear_conv(a, w_rev)
        idx = 2 * n - 2 - np.arange(n)
        grad = 2.0 * aw[idx]

        # Normalized momentum
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_norm = grad / gnorm
        v_buf = 0.85 * v_buf + 0.15 * grad_norm
        vnorm = np.linalg.norm(v_buf)
        if vnorm < 1e-15:
            break
        v_buf /= vnorm

        # Multiplicative update with backtracking
        accepted = False
        for bt in range(5):
            lr_try = lr * (0.5 ** bt)
            a_try = a * np.exp(-lr_try * v_buf)
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum() * S
            c_try = self_conv(a_try)
            obj_try = c_try.max()
            if obj_try < c.max() * (1.0 + 1e-8):
                a = a_try
                accepted = True
                break
        if not accepted:
            a = a * np.exp(-lr * 0.05 * v_buf)
            a = np.maximum(a, 1e-15)
            a = a / a.sum() * S

        # Decay lr
        lr *= 0.999

        if verbose and (total_iters := total_iters + 1) % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    [{tag}] iter={total_iters}, obj={obj_val(a):.6f}, "
                  f"best={best_obj:.6f}, t={elapsed:.1f}s")

    return best_a, best_obj

    return a
