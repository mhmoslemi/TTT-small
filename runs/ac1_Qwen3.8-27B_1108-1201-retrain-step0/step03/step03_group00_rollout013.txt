"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using multi-start FISTA
with aggressive LSE annealing, subgradient refinement, shape-parameter initialization,
and multi-length search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def subgradient_at_peak(a, c):
    """Exact subgradient: gradient of (a*a) at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def convolution_gradient_single_lag(a, k):
    """Gradient of (a*a)_k with respect to a."""
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def run_fista_aggressive(a_init, n, S, max_time, rng, beta_init=5.0,
                         beta_rate=0.006, beta_max=10000.0,
                         subgrad_switch=200.0):
    """FISTA with aggressive LSE and switch to subgradient at high beta."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-12:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Check convergence
        if it - last_improve > 3000:
            # Perturb to escape
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 10000.0)
            if noise_scale < 1e-8:
                break
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it
            continue

        if beta < subgrad_switch:
            # FISTA with LSE
            grad = lse_gradient(y, c, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            step = 1.0 / (2.0 * beta)
            # Adaptive: reduce step if not improving
            if it - last_improve > 500:
                step *= 0.5

            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = x_new + ((t - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t = t_new
        else:
            # Subgradient method with backtracking
            k_star = int(np.argmax(c))
            grad = convolution_gradient_single_lag(y, k_star)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break

            # Backtracking line search
            step = S / (gnorm * 10.0)
            for _ in range(20):
                x_trial = project_simplex(y - step * grad, S)
                c_trial = fft_convolve_self(x_trial)
                if c_trial.max() < cur + 1e-10:
                    break
                step *= 0.5
            else:
                x_trial = project_simplex(y - step * grad, S)

            c_trial = fft_convolve_self(x_trial)
            if c_trial.max() < best:
                best = c_trial.max()
                best_a = x_trial.copy()
                last_improve = it

            y = x_trial
            a = x_trial
            t = 1.0

        # Periodic restart
        if it % 5000 == 0 and beta < subgrad_switch:
            decay = np.exp(-it / 20000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def run_subgradient_polish(a_init, n, S, max_time, rng):
    """Pure subgradient polishing for high-precision refinement."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-12:
            best = cur
            best_a = y.copy()

        # Find top few peaks and use weighted subgradient
        top_k = min(8, len(c))
        idx_top = np.argsort(c)[::-1][:top_k]
        vals = c[idx_top]
        # Weight by exponential of values
        weights = np.exp(5.0 * (vals - vals[0]))
        weights /= weights.sum()

        # Combined subgradient
        grad = np.zeros(n)
        for j, k in enumerate(idx_top):
            grad += weights[j] * convolution_gradient_single_lag(y, k)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Decreasing step size
        step = 2.0 * S / (gnorm * np.sqrt(it + 1.0))
        # Backtracking
        for _ in range(15):
            x_trial = project_simplex(y - step * grad, S)
            c_trial = fft_convolve_self(x_trial)
            if c_trial.max() <= cur + 1e-12:
                break

        if c_trial.max() < cur - 1e-12:
            y = x_new
        else:
            # Average with previous (heavy ball)
            y = project_simplex(0.7 * y + 0.3 * x_new, S)

        a = y

def shape_sequence(n, p, q, rng=None):
    """Generate sequence a_i proportional to (i+1)^p * (n-i)^q."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = np.power(i, p) * np.power(j, q)
    a = np.maximum(a, 1e-10)
    return a

        print("=== Phase 2: Polishing (n=%d, remaining %.0fs) ===" % (n_best, remaining))

        if remaining > 30:
            # Subgradient polish
            t_pol = remaining * 0.5
            a_pol, s_pol = run_subgradient_polish(global_a, n_best, S_best, t_pol, rng)
            a_pol = normalize_sum(a_pol, n_best)
            s_real = quick_score(a_pol)
            if s_real < global_best:
                global_best = s_real
                global_a = a_pol.copy()
                print("  After subgrad polish: %.6f" % global_best)

        # Final FISTA micro-polish with very high beta
        remaining2 = total_deadline - time.monotonic()
        if remaining2 > 20 and global_a is not None:
            n_best = len(global_a)
            S_best = np.sqrt(2.0 * n_best)
            a_fin, s_fin = run_fista_aggressive(
                global_a, n_best, S_best, remaining2 * 0.7, rng,
                beta_init=500.0, beta_rate=0.001, beta_max=50000.0,
                subgrad_switch=10.0
            )
            a_fin = normalize_sum(a_fin, n_best)
            s_fin_real = quick_score(a_fin)
            if s_fin_real < global_best:
                global_best = s_fin_real
                global_a = a_fin.copy()
                print("  After micro-polish: %.6f" % global_best)

    # Phase 3: Try longer sequences if time allows
    remaining = total_deadline - time.monotonic()
    if remaining > 120:
        for n_try in [16384, 32768]:
            if time.monotonic() > total_deadline - 60:
                break
            if n_try == len(global_a) if global_a is not None else False:
                continue
            S_try = np.sqrt(2.0 * n_try)
            # Use the best shape as initialization (upsampled or re-sampled)
            if global_a is not None:
                t_old = np.linspace(0, 1, len(global_a))
                t_new = np.linspace(0, 1, n_try)
                init_long = np.interp(t_new, t_old, global_a)
            else:
                init_long = shape_sequence(n_try, -0.3, -0.3)
            init_long = normalize_sum(init_long, n_try)

            t_allot = min(remaining * 0.3, total_deadline - time.monotonic() - 60)
            if t_allot < 20:
                break

            print("=== Phase 3: Trying n=%d (%.0fs) ===" % (n_try, t_allot))
            a_try, s_try = run_fista_aggressive(
                init_long, n_try, S_try, t_allot, rng,
                beta_init=10.0, beta_rate=0.006, beta_max=5000.0,
                subgrad_switch=100.0
            )
            a_try = normalize_sum(a_try, n_try)
            s_real = quick_score(a_try)
            if s_real < global_best:
                global_best = s_real
                global_a = a_try.copy()
                print("  n=%d: %.6f (improved!)" % (n_try, global_best))

    # Final subgradient refinement if time remains
    remaining = total_deadline - time.monotonic()
    if remaining > 15 and global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        a_final, _ = run_subgradient_polish(global_a, n_best, S_best, remaining * 0.8, rng)
        a_final = normalize_sum(a_final, n_best)
        s_final = quick_score(a_final)
        if s_final < global_best:
            global_best = s_final
            global_a = a_final.copy()
            print("  Final refinement: %.6f" % global_best)

    # Verify and return
    if global_a is None:
        n_fb = 256
        global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2 * n_fb) / n_fb)
        global_best = 2.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        n_fb = 256
        global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2 * n_fb) / n_fb)

    # Verify with the actual evaluator
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        n_fb = 256
        global_a = np.ones(n_fb, dtype=np.float64) * (np.sqrt(2 * n_fb) / n_fb)
        final_list = [float(x) for x in global_a]
        ver2 = evaluate_sequence(final_list)
        print("  Fallback verification: %.6f" % ver2)

    print("Returning sequence of length %d, score=%.6f" % (len(global_a), ver if np.isfinite(ver) else global_best))
    return final_list
