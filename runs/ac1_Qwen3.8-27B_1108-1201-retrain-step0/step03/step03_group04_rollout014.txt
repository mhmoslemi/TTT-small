"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization from parent sequence:
1. Nesterov with exact top-k subgradient (k=1 to increasing)
2. Coordinate descent polish
3. Random restarts with perturbations
4. Final ultra-fine refinement

import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_gradient_fft(a, weights):
    """Compute gradient given weights on convolution lags using FFT."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def toppk_weights(c, top_k):
    """Compute normalized top-k weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights

def exact_peaks_weights(c, k):
    """Uniform weights on the top-k peaks."""
    k = min(k, len(c))
    top_indices = np.argpartition(c, -k)[-k:]
    weights = np.zeros(len(c))
    weights[top_indices] = 1.0 / k
    return weights

def nesterov_subgrad(a_init, n, S, max_time, rng, top_k_init=1, top_k_max=50, 
                      top_k_increase_every=500, step_init=None, use_exact=True):
    """Nesterov accelerated subgradient method with adaptive top-k."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    if step_init is None:
        step = S / (n * 2.0)
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // top_k_increase_every)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = y.copy()
        
        if use_exact:
            weights = exact_peaks_weights(c, top_k)
        else:
            weights = toppk_weights(c, top_k)
        
        grad = compute_gradient_fft(y, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Normalized step
        step_size = step_init / (1.0 + 0.01 * it) if step_init else S / (n * 2.0) / (1.0 + 0.01 * it)
        
        x_new = project_simplex(y - step_size * grad / gnorm, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Perturb if stalled
        if it % 1000 == 0:
            c_check = fft_convolve_self(x_new)
            if c_check.max() > best_score * 1.001:
                noise_scale = 0.001 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
    
    return best_local, best_score

        c = fft_convolve_self(a)
        cur_peak = c.max()
        peak_lag = c.argmax()

            # But we need to maintain sum(a) = S, so we increase another element
            
            # Simple approach: try decreasing a[i] and increasing a[j] for random j
            j = rng.integers(n)
            if j == i:
                continue
            
            for delta_frac in [0.01, 0.005, 0.001, 0.0005]:
                delta = delta_frac * a[i]
                if delta <= 0:
                    continue
                a_trial = a.copy()
                a_trial[i] = a[i] - delta
                a_trial[j] = a[j] + delta
                if a_trial[i] < 0:
                    continue
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur_peak:
                    a = a_trial
                    cur_peak = c_trial.max()
                    if cur_peak < best_score:
                        best_score = cur_peak
                        best_local = a.copy()
                    break
    
    return best_local, best_score

def coordinate_descent_v2(a_init, n, S, max_time, rng):
    """Faster coordinate descent using cached convolution."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_peak = c.max()
        if cur_peak < best_score:
            best_score = cur_peak
            best_local = a.copy()
        
        peak_lag = c.argmax()
        
        # Find elements that contribute to the peak
        # R[peak_lag] = sum_j a[j] * a[peak_lag - j]
        # The "partners" are the pairs (j, peak_lag - j)
        
        # Try to reduce the peak by adjusting the largest contributor
        # The largest contributor to R[peak_lag] is max_j a[j]*a[peak_lag-j]
        
        # Get the contribution of each pair
        min_j = max(0, peak_lag - (n - 1))
        max_j = min(n - 1, peak_lag)
        if min_j > max_j:
            continue
        
        js = np.arange(min_j, max_j + 1)
        partners = peak_lag - js
        contributions = a[js] * a[partners]
        
        # Find the pair with largest contribution
        idx_max = np.argmax(contributions)
        j_star = js[idx_max]
        p_star = partners[idx_max]
        
        # Try to reduce a[j_star] (move some mass to a random other position)
        j = rng.integers(n)
        if j == j_star or j == p_star:
            continue
        
        for frac in [0.1, 0.05, 0.02, 0.01]:
            delta = frac * a[j_star]
            if delta <= 0 or a[j_star] - delta < 0:
                continue
            a_trial = a.copy()
            a_trial[j_star] -= delta
            a_trial[j] += delta
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_peak:
                a = a_trial
                if c_trial.max() < best_score:
                    best_score = c_trial.max()
                    best_local = a.copy()
                break
    
    return best_local, best_score

def peak_reduction_step(a, S, n, peak_lag, rng, n_trials=10):
    """Try to reduce the peak at a specific lag by moving mass."""
    c = fft_convolve_self(a)
    cur_peak = c[peak_lag]
    
    # Find the largest contributor to this peak
    min_j = max(0, peak_lag - (n - 1))
    max_j = min(n - 1, peak_lag)
    if min_j > max_j:
        return a, cur_peak
    
    js = np.arange(min_j, max_j + 1)
    partners = peak_lag - js
    contributions = a[js] * a[partners]
    
    for _ in range(n_trials):
        idx = rng.integers(len(js))
        # Weight by contribution
        probs = contributions / contributions.sum()
        idx = rng.choice(len(js), p=probs)
        
        j_star = js[idx]
        p_star = partners[idx]
        
        # Move mass from j_star to a random position far from peak
        j = rng.integers(n)
        if j == j_star or j == p_star:
            continue
        
        for frac in [0.05, 0.02, 0.01, 0.005]:
            delta = frac * a[j_star]
            if delta <= 0:
                break
            a_trial = a.copy()
            a_trial[j_star] -= delta
            a_trial[j] += delta
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < c.max():
                return a_trial, c_trial.max()
    
    return a, cur_peak

def greedy_peak_reduction(a_init, n, S, max_time, rng):
    """Greedy: repeatedly find the peak and try to reduce it."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        peak_lag = c.argmax()
        
        # Try to reduce the peak
        a, new_peak = peak_reduction_step(a, S, n, peak_lag, rng, n_trials=20)
        
        # Small chance of global perturbation
        if rng.random() < 0.01:

    return best_local, best_score

def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization starting from parent sequence:
1. Nesterov with exact top-k subgradient (continuation from k=1)
2. Inverse-weighted reweighting optimization
3. Greedy peak reduction via targeted mass transfer
4. Random restarts with diverse perturbations
5. Final ultra-fine Nesterov polish
Key innovations: peak-tracking memory, adaptive step with backtracking,
structured coordinate descent for final polish, and multi-start diversity.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient of weighted sum of convolution lags w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def exact_topk_weights(c, k):
    """Uniform weights on top-k lags."""
    k = min(k, len(c))
    top_indices = np.argpartition(c, -k)[-k:]
    weights = np.zeros(len(c))
    weights[top_indices] = 1.0 / k
    return weights


def power_weights(c, gamma):
    """Weights proportional to c^gamma (emphasizes peaks for large gamma)."""
    c_safe = np.maximum(c, 1e-10)
    weights = c_safe ** gamma
    total = weights.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    return weights / total


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def nesterov_subgrad(a_init, n, S, max_time, rng,
                     top_k_init=1, top_k_max=10, top_k_every=300,
                     step_decay=0.003, use_power=False, gamma_start=2.0,
                     gamma_end=8.0, gamma_rate=0.002,
                     perturb_every=2000, perturb_prob=0.2):
    """Nesterov accelerated subgradient with adaptive top-k or power weights."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    base_step = S / (n * 2.0)

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-14:
            best_score = cur
            best_local = y.copy()

        if use_power:
            gamma = min(gamma_end, gamma_start + gamma_rate * it)
            weights = power_weights(c, gamma)
        else:
            top_k = min(top_k_max, top_k_init + it // top_k_every)
            weights = exact_topk_weights(c, top_k)

        grad = compute_gradient_fft(y, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = base_step / (1.0 + step_decay * it)
        x_new = project_simplex(y - step * grad / gnorm, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic perturbation to escape local minima
        if it % perturb_every == 0 and rng.random() < perturb_prob:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t = 1.0

    return best_local, best_score


def greedy_peak_reduction(a_init, n, S, max_time, rng, n_trials=30):
    """Greedy: repeatedly identify peak and transfer mass to reduce it."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        peak_lag = c.argmax()

        # Find contributors to the peak lag
        min_j = max(0, peak_lag - (n - 1))
        max_j = min(n - 1, peak_lag)
        if min_j > max_j:
            break

        js = np.arange(min_j, max_j + 1)
        partners = peak_lag - js
        contributions = a[js] * a[partners]
        total_contrib = contributions.sum()
        if total_contrib < 1e-15:
            break

        # Try to reduce the peak by moving mass from high contributors
        for _ in range(n_trials):
            # Pick a contributor weighted by its contribution
            probs = contributions / total_contrib
            idx = rng.choice(len(js), p=probs)
            j_star = int(js[idx])
            p_star = int(partners[idx])

            # Choose destination: prefer positions that don't create new peaks
            j_dest = rng.integers(n)
            if j_dest == j_star or j_dest == p_star:
                continue

            # Try several step sizes
            for frac in [0.08, 0.04, 0.02, 0.01]:
                delta = frac * a[j_star]
                if delta <= 0:
                    break
                a_trial = a.copy()
                a_trial[j_star] -= delta
                a_trial[j_dest] += delta
                c_trial = fft_convolve_self(a_trial)
                trial_peak = c_trial.max()
                if trial_peak < cur:
                    a = a_trial
                    cur = trial_peak
                    break

        # Occasional global perturbation
        if it % 50 == 0 and rng.random() < 0.05:
            noise_scale = 0.001 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_local, best_score


def coordinate_descent_peaks(a_init, n, S, max_time, rng):
    """Coordinate descent focusing on peak-reducing moves."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Get top-3 peaks
        top3 = np.argpartition(c, -3)[-3:]
        top3_sorted = top3[np.argsort(c[top3])[::-1]]

        improved = False
        for m in top3_sorted:
            # Find the largest contributor to lag m
            min_j = max(0, m - (n - 1))
            max_j = min(n - 1, m)
            if min_j > max_j:
                continue
            js = np.arange(min_j, max_j + 1)
            partners = m - js
            contributions = a[js] * a[partners]
            idx_max = np.argmax(contributions)
            j_star = int(js[idx_max])
            p_star = int(partners[idx_max])

            # Try to reduce a[j_star] by moving to a far position
            # Pick destination from bottom quartile of a to minimize new peaks
            bottom_q = np.argsort(a)[:n // 4]
            for attempt in range(5):
                j_dest = int(bottom_q[rng.integers(len(bottom_q))])
                if j_dest == j_star or j_dest == p_star:
                    continue
                for frac in [0.05, 0.02, 0.01, 0.005]:
                    delta = frac * a[j_star]
                    if delta <= 0:
                        break
                    a_trial = a.copy()
                    a_trial[j_star] -= delta
                    a_trial[j_dest] += delta
                    c_trial = fft_convolve_self(a_trial)
                    if c_trial.max() < cur:
                        a = a_trial
                        cur = c_trial.max()
                        improved = True
                        break
                if improved:
                    break

        if not improved and rng.random() < 0.1:
            noise_scale = 0.0005 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)

    return best_local, best_score


def multi_start_search(a_init, n, S, max_time, rng, n_starts=4):
    """Try multiple perturbed starts and return the best result."""
    best_a = a_init.copy()
    best_score = fft_convolve_self(a_init).max()

    per_start = max_time / n_starts

    for s_idx in range(n_starts):
        if time.monotonic() > (start_time := (start_time if 'start_time' in dir() else time.monotonic())) + max_time - 5:
            break

        # Generate different perturbations
        if s_idx == 0:
            pert = a_init + rng.normal(0, 0.004 * S / np.sqrt(n), n)
        elif s_idx == 1:
            smooth_noise = np.convolve(rng.normal(0, 1, n), np.ones(256) / 256, mode='same')
            pert = a_init + 0.003 * S / np.sqrt(n) * smooth_noise
        elif s_idx == 2:
            pert = a_init * (1.0 + rng.normal(0, 0.008, n))
        else:
            tt = np.linspace(0, 1, n, endpoint=False)
            freq = 3 + 2 * s_idx
            pert = a_init + 0.004 * S / np.sqrt(n) * np.sin(2 * np.pi * freq * tt)

        pert = safe_normalize(pert, n)

        # Quick Nesterov
        a_r, s_r = nesterov_subgrad(pert, n, S, per_start, rng,
                                    top_k_init=1, top_k_max=5,
                                    top_k_every=300, step_decay=0.002)
        a_r = safe_normalize(a_r, n)
        c_r = fft_convolve_self(a_r)
        if c_r.max() < best_score:
            best_score = c_r.max()
            best_a = a_r.copy()

    return best_a, best_score
