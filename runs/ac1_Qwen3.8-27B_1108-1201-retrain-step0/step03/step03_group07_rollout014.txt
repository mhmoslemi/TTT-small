```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a multi-strategy approach:
1. Mirror descent with adaptive top-k peak chasing
2. Spectral flattening via frequency-domain projection
3. Multi-scale simulated annealing for escaping local minima
4. Coordinate descent polishing
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def compute_subgrad(c, a, top_k, soft_temperature=0.0):
    n = len(a)
    top_k = max(1, min(top_k, len(c)))
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    c_max = c[top_idx[0]]
    
    if soft_temperature > 0:
        w = np.exp(soft_temperature * (c[top_idx] - c_max))
        w /= w.sum()
    else:
        w = np.full(top_k, 1.0 / top_k)
    
    w_full = np.zeros(len(c))
    w_full[top_idx] = w
    
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 2).bit_length()
    W = fft(w_full, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def mirror_descent_v2(a0, time_budget, eta0=0.5, eta_min=0.01,
                      top_k_start=500, top_k_end=1, top_k_ramp_frac=0.5,
                      momentum=0.9, soft_T=0.0, perturb_every=500,
                      perturb_scale=0.1, verbose=True, stagnation_limit=3000):
    """Improved mirror descent with:
    - Adaptive eta (not just decay, but also increase on progress)
    - Better perturbation (structured, not just Gaussian)
    - Periodic spectral projections
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    
    c = fft_convolve_self(a)
    cur_obj = obj_value(a, c)
    best_obj = cur_obj
    best_a = a.copy()
    best_iter = 0
    
    v = np.zeros(n)
    eta = eta0
    iter_no = 0
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        # Compute subgradient
        progress = min(1.0, iter_no / max(1, int(stagnation_limit * top_k_ramp_frac)))
        top_k_cur = max(1, int(top_k_start + (top_k_end - top_k_start) * progress))
        
        grad = compute_subgrad(c, a, top_k_cur, soft_T)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Momentum
        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm
        
        # Mirror descent step
        a_trial = a * np.exp(-eta * v)
        a_trial = np.maximum(a_trial, 1e-300)
        a_trial = a_trial * (S / a_trial.sum())
        
        c_trial = fft_convolve_self(a_trial)
        obj_trial = obj_value(a_trial, c_trial)
        
        if obj_trial < cur_obj:
            a = a_trial
            c = c_trial
            cur_obj = obj_trial
            # Increase eta
            eta = min(eta * 1.01, eta0)
        else:
            # Backtrack
            for bt in range(5):
                eta_try = eta * (0.5 ** (bt + 1))
                a_trial = a * np.exp(-eta_try * v)
                a_trial = np.maximum(a_trial, 1e-300)
                a_trial = a_trial * (S / a_trial.sum())
                c_trial = fft_convolve_self(a_trial)
                obj_trial = obj_value(a_trial, c_trial)
                if obj_trial < cur_obj:
                    a = a_trial
                    c = c_trial
                    cur_obj = obj_trial
                    eta = eta_try * 1.5  # Keep eta at the successful level
                    break
            else:
                # Accept anyway with tiny step
                eta = max(eta * 0.5, eta_min)
                a = a * np.exp(-eta * 0.1 * v)
                a = np.maximum(a, 1e-300)
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                cur_obj = obj_value(a, c)
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
        elif iter_no - best_iter > stagnation_limit:
            # Stagnation: structured perturbation
            # Option 1: Gaussian noise
            # Option 2: Random coordinate flips (multiply by random factors)
            # Option 3: Spectral perturbation
            pert_choice = np.random.random()
            
            if pert_choice < 0.33:
                # Gaussian perturbation
                noise = np.random.randn(n) * perturb_scale * S / np.sqrt(n)
                a = a + noise
                a = np.maximum(a, 0.0)
                a = a * (S / a.sum())
            elif pert_choice < 0.66:
                # Random block perturbation
                block_size = max(1, n // 16)
                for _ in range(4):
                    start_idx = np.random.randint(0, n - block_size)
                    a[start_idx:start_idx+block_size] *= np.random.uniform(0.5, 1.5)
                a = np.maximum(a, 0.0)
                a = a * (S / a.sum())
            else:
                # Spectral perturbation
                fsize = 1 << (2*n - 1).bit_length()
                A = fft(a, fsize)
                mag = np.abs(A)
                phase = np.angle(A)
                # Add random phase noise
                phase_noise = np.random.randn(fsize) * 0.1
                A_new = mag * np.exp(1j * (phase + phase_noise))
                a_new = np.real(ifft(A_new, fsize))[:n]
                a_new = np.maximum(a_new, 0.0)
                if a_new.sum() > 0:
                    a = a_new * S / a_new.sum()
                else:
                    a = np.ones(n) * S / n
            
            c = fft_convolve_self(a)
            cur_obj = obj_value(a, c)
            v *= 0.1
            eta = eta0
            best_iter = iter_no
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    [perturb] iter={iter_no}, obj={cur_obj:.6f}, "
                      f"best={best_obj:.6f}, t={elapsed:.1f}s")
        
        # Slow decay
        eta = max(eta * 0.99999, eta_min)
        
        iter_no += 1
        if verbose and iter_no % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"eta={eta:.4f}, top_k={top_k_cur}, t={elapsed:.1f}s")
    
    return best_a, best_obj


def spectral_flatten(a0, time_budget, T_init=2.0, T_final=0.1):
    """Spectral flattening via soft power spectral equalization."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()
    
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    
    total_iters = 20000
    
    for it in range(total_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Annealing
        T = T_init * (T_final / T_init) ** (it / total_iters)
        
        # Compute DFT
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat power spectrum
        # sum(|A|^2) = fsize * sum(a^2) (Parseval)
        # Target |A|^2 = sum(|A|^2) / fsize
        target_power = np.sum(mag**2) / fsize
        target_mag = np.sqrt(target_power)
        
        # Soft equalization: new_mag = mag * (target/mag)^(1/T)
        alpha = 1.0 / max(T, 0.01)
        mag_safe = np.maximum(mag, 1e-10)
        new_mag = mag_safe * (target_mag / mag_safe) ** alpha
        # This is equivalent to: new_mag = target_mag^alpha * mag_safe^(1-alpha)
        
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fsize))[:n]
        
        # Enforce non-negativity and normalization
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() > 1e-10:
            a_new = a_new * S / a_new.sum()
        else:
            continue
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        obj_new = obj_value(a_new, c_new)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        else:
            # Accept with probability exp(-(obj_new - best_obj)/T) for exploration
            if np.random.random() < np.exp(-min((obj_new - best_obj) / max(T * 0.01, 0.001), 50)):
                a = a_new
                c = c_new
                continue
        
        a = a_new
        c = c_new
    
    return best_a, best_obj


def coordinate_polish_v2(a0, time_budget):
    """Coordinate descent: try adjusting each element to reduce the max."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    
    # Sort indices by their contribution to peaks
    # Elements that appear in the worst peaks should be adjusted first
    peak_idx = np.argmax(c)
    
    while time.monotonic() - start < time_budget:
        improved = False
        
        # Process in order of peak involvement
        # The peak at position k involves a[max(0,k-n+1):min(n,k+1)]
        lo = max(0, peak_idx - n + 1)
        hi = min(n, peak_idx + 1)
        
        for i in range(lo, hi):
            if time.monotonic() - start > time_budget:
                break
            
            orig = a[i]
            if orig < 1e-15:
                continue
            
            # Try reducing this element (since it contributes to the peak)
            for factor in [0.5, 0.7, 0.85]:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_norm = a * (S / s)
                c_new = fft_convolve_self(a_norm)
                obj_new = obj_value(a_norm, c_new)
                
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_norm.copy()
                    a = a_norm
                    c = c_new
                    improved = True
                    # Update peak
                    peak_idx = np.argmax(c)
                    lo = max(0, peak_idx - n + 1)
                    hi = min(n, peak_idx + 1)
                    break
            a[i] = a[i]  # keep current state
            s = a.sum()
            if s > 0:
                a = a * (S / s)
        
        if not improved and time.monotonic() - start > time_budget * 0.5:
            # Try increasing some elements that are small
            small_idx = np.argsort(a)[:n // 8]
            for i in small_idx:
                if time.monotonic() - start > time_budget:
                    break
                orig = a[i]
                for factor in [2.0, 3.0, 5.0]:
                    a[i] = orig * factor
                    s = a.sum()
                    if s < 1e-10:
                        a[i] = orig
                        break
                    a_norm = a * (S / s)
                    c_new = fft_convolve_self(a_norm)
                    obj_new = obj_value(a_norm, c_new)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_norm.copy()
                        a = a_norm
                        c = c_new
                        improved = True
                        peak_idx = np.argmax(c)
                        break
                a[i] = a[i]
                s = a.sum()
                if s > 0:
                    a = a * (S / s)
        
        if not improved:
            # Try a random sweep
            idx = np.random.choice(n, min(n // 4, 512), replace=False)
            for i in idx:
                if time.monotonic() - start > time_budget:
                    break
                orig = a[i]
                factor = np.random.uniform(0.3, 2.0)
                a[i] = orig * factor
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 0:
                    a = a * (S / s)
                c = fft_convolve_self(a)
                obj_new = obj_value(a, c)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
    
    return best_a, best_obj


def generate_diverse_candidates(n, rng):
    """Generate diverse initial candidates."""
    S = np.sqrt(2.0 * n)
    i = np.arange(n)
    cands = []
    
    # Cosine with different frequencies
    for freq in [3, 5, 7, 11, 13]:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * i / n)
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
        cands.append(a)
    
    # Multi-frequency
    a = 1.0 + 0.2 * np.cos(2 * np.pi * 5 * i / n) + 0.1 * np.cos(2 * np.pi * 13 * i / n)
    a = np.maximum(a, 0.01)
    a *= S / a.sum()
    cands.append(a)
    
    # Random with different distributions
    a = rng.random(n)
    a = np.maximum(a, 0.01)
    a *= S / a.sum()
    cands.append(a)
    
    a = rng.exponential(1.0, n)
    a = np.maximum(a, 0.01)
    a *= S / a.sum()
    cands.append(a)
    
    # Binary-like (values close to 0 or 2*S/n)
    target = 2.0 * S / n
    a = (rng.random(n) > 0.5).astype(float) * target
    a = np.maximum(a, 0.01)
    a *= S / a.sum()
    cands.append(a)
    
    # Smooth noise
    freqs = rng.integers(1, n // 2, size=n // 4)
    amplitudes = rng.random(n // 4) * 0.3
    phases = rng.random(n // 4) * 2 * np.pi
    a = np.ones(n)
    for f, amp, ph in zip(freqs, amplitudes, phases):
        a += amp * np.cos(2 * np.pi * f * i / n + ph)
    a = np.maximum(a, 0.01)
    a *= S / a.sum()
    cands.append(cands[-1] if len(cands) > 0 else a)
    cands.append(a)
    
    return cands
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a multi-strategy approach:
1. Mirror descent with adaptive top-k peak chasing and structured perturbations
2. Spectral flattening via soft power-spectrum equalization in frequency domain
3. Multi-scale coordinate descent polishing
4. Diverse candidate generation with population-based selection
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def compute_subgrad(c, a, top_k, soft_temperature=0.0):
    """Compute subgradient of max_k c[k] using top-k weighted average."""
    n = len(a)
    top_k = max(1, min(top_k, len(c)))
    top_idx = np.argsort(c, kind='stable')[::-1][:top_k]
    c_max = c[top_idx[0]]

    if soft_temperature > 0:
        w = np.exp(soft_temperature * (c[top_idx] - c_max))
        w /= w.sum()
    else:
        w = np.full(top_k, 1.0 / top_k)

    w_full = np.zeros(len(c))
    w_full[top_idx] = w

    a_rev = a[::-1]
    conv_len = len(c) + n - 1
    fsize = 1 << (conv_len - 1).bit_length()
    W = fft(w_full, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))

    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def mirror_descent_v2(a0, time_budget, eta0=0.5, eta_min=0.01,
                      top_k_start=500, top_k_end=1, top_k_ramp_frac=0.5,
                      momentum=0.9, soft_T=0.0, perturb_scale=0.1,
                      verbose=True, stagnation_limit=2000):
    """Mirror descent with adaptive step size, momentum, and structured perturbations."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.full(n, S / n)

    c = fft_convolve_self(a)
    cur_obj = obj_value(a, c)
    best_obj = cur_obj
    best_a = a.copy()
    best_iter = 0

    v = np.zeros(n)
    eta = eta0
    iter_no = 0
    start = time.monotonic()
    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < time_budget:
        progress = min(1.0, iter_no / max(1, int(stagnation_limit * top_k_ramp_frac)))
        top_k_cur = max(1, int(top_k_start + (top_k_end - top_k_start) * progress))

        grad = compute_subgrad(c, a, top_k_cur, soft_T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        v = momentum * v + (1.0 - momentum) * grad
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        # Try step with current eta
        a_trial = a * np.exp(-eta * v)

        c_trial = fft_convolve_self(a_trial)
        obj_trial = obj_value(a_trial, c_trial)

            a = a_trial
            c = c_trial
            cur_obj = obj_trial
            eta = min(eta * 1.005, eta0)
        else:
            accepted = False
            for bt in range(6):
                eta_try = eta * (0.5 ** (bt + 1))
                a_trial = a * np.exp(-eta_try * v)
                a_trial = np.maximum(a_trial, 1e-300)
                a_trial = a_trial * (S / a_trial.sum())
                c_trial = fft_convolve_self(a_trial)

                    c = c_trial

                    eta = max(eta_try, eta_min)
                    accepted = True
                    break

                c = fft_convolve_self(a)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            best_iter = iter_no
        elif iter_no - best_iter > stagnation_limit:
            pert_choice = np.random.random()
            if pert_choice < 0.35:
                noise = np.random.randn(n) * perturb_scale * S / np.sqrt(n)
                a = a + noise
                a = np.maximum(a, 0.0)

                a = a * (S / a.sum())
            else:
                A = fft(a, fsize)
                mag = np.abs(A)

                phase += np.random.randn(fsize) * 0.15

                    a = a_new * S / a_new.sum()
            c = fft_convolve_self(a)
            cur_obj = obj_value(a, c)
            v *= 0.1
            eta = eta0
            best_iter = iter_no
            if verbose:
                elapsed = time.monotonic() - start
                print(f"    [perturb] iter={iter_no}, obj={cur_obj:.6f}, "
                      f"best={best_obj:.6f}, t={elapsed:.1f}s")

        eta = max(eta * 0.99998, eta_min)
        iter_no += 1
        if verbose and iter_no % 3000 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"eta={eta:.4f}, top_k={top_k_cur}, t={elapsed:.1f}s")

    return best_a, best_obj


def spectral_flatten(a0, time_budget, T_init=2.0, T_final=0.1, verbose=False):
    """Spectral flattening via soft power-spectrum equalization with simulated annealing."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()

    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    total_iters = 30000

    for it in range(total_iters):
        if time.monotonic() - start > time_budget:
            break

        T = T_init * (T_final / T_init) ** (it / total_iters)

        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)

        target_power = np.sum(mag ** 2) / fsize
        target_mag = np.sqrt(target_power)

        alpha = min(1.0 / max(T, 0.05), 8.0)
        mag_safe = np.maximum(mag, 1e-10)
        new_mag = target_mag ** alpha * mag_safe ** (1.0 - alpha)

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fsize))[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-10:
            continue
        a_new = a_new * S / a_new.sum()

        c_new = fft_convolve_self(a_new)
        obj_new = obj_value(a_new, c_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
        else:
            delta = obj_new - best_obj
            accept_prob = np.exp(-min(delta / max(T * 0.005, 0.0005), 100.0))
            if np.random.random() < accept_prob:
                a = a_new
                c = c_new
            # else: keep old a

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"    [spectral] it={it}, obj={obj_new:.6f}, best={best_obj:.6f}, "
                  f"T={T:.4f}, t={elapsed:.1f}s")

    return best_a, best_obj


def hybrid_step(a0, time_budget, n_md=3, n_sf=2, verbose=False):
    """Alternate between mirror descent and spectral flattening."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    fsize = 1 << (2 * n - 1).bit_length()

    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()

    iter_no = 0
    while time.monotonic() - start < time_budget:
        # Mirror descent steps
        for _ in range(n_md):
            if time.monotonic() - start > time_budget:
                break
            peak_idx = np.argmax(c)
            c_max_val = c[peak_idx]

            grad = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(n - 1, peak_idx)
            for m in range(lo, hi + 1):
                grad[m] += 2.0 * a[peak_idx - m]

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad = grad / gnorm

            lr = 0.05 / np.sqrt(iter_no + 10)
            a = a * np.exp(-lr * grad)
            a = np.maximum(a, 1e-300)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            cur_obj = obj_value(a, c)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
        # Spectral flattening steps
        for _ in range(n_sf):
            if time.monotonic() - start > time_budget:
                break
            A = fft(a, fsize)
            mag = np.abs(A)
            phase = np.angle(A)
            target_mag = np.sqrt(np.sum(mag ** 2) / fsize)
            alpha = 0.3
            mag_safe = np.maximum(mag, 1e-10)
            new_mag = target_mag ** alpha * mag_safe ** (1.0 - alpha)
            A_new = new_mag * np.exp(1j * phase)
            a_new = np.real(ifft(A_new, fsize))[:n]
            a_new = np.maximum(a_new, 0.0)
            if a_new.sum() > 1e-10:
                a_new = a_new * S / a_new.sum()
                c_new = fft_convolve_self(a_new)
                obj_new = obj_value(a_new, c_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                a = a_new
                c = c_new
        iter_no += 1
        if verbose and iter_no % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    [hybrid] it={iter_no}, best={best_obj:.6f}, t={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_polish_v2(a0, time_budget):
    """Coordinate descent focused on elements contributing to the worst peaks."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()

    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        improved = False
        peak_idx = np.argmax(c)
        lo = max(0, peak_idx - n + 1)
        hi = min(n, peak_idx + 1)

        for i in range(lo, hi):
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-15:
                continue
            for factor in [0.6, 0.8, 1.2, 1.5]:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_norm = a * (S / s)
                c_new = fft_convolve_self(a_norm)
                obj_new = obj_value(a_norm, c_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_norm.copy()
                    a = a_norm
                    c = c_new
                    improved = True
                    peak_idx = np.argmax(c)
                    lo = max(0, peak_idx - n + 1)
                    hi = min(n, peak_idx + 1)
                    break
            a[i] = orig if not improved else a[i]
            s = a.sum()
            if s > 0:
                a = a * (S / s)

        if not improved:
            try_random = time.monotonic() - start < time_budget * 0.8
            if try_random:
                idx = np.random.choice(n, min(256, n // 8), replace=False)
                for i in idx:
                    if time.monotonic() - start > time_budget:
                        break
                    orig = a[i]
                    factor = np.random.uniform(0.3, 2.5)
                    a[i] = orig * factor
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 0:
                        a = a * (S / s)
                    c = fft_convolve_self(a)
                    obj_new = obj_value(a, c)
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a.copy()
                        improved = True
            if not improved:
                break

    return best_a, best_obj


def generate_diverse_candidates(n, rng):
    """Generate diverse initial candidates for exploration."""
    S = np.sqrt(2.0 * n)
    i = np.arange(n)
    cands = []

    for freq in [3, 7, 11, 17]:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * i / n)
        a = np.maximum(a, 0.01)
        a *= S / a.sum()
        cands.append(a)

    a = 1.0 + 0.2 * np.cos(2 * np.pi * 5 * i / n) + 0.15 * np.cos(2 * np.pi * 13 * i / n)
    a = np.maximum(a, 0.01)
    a *= S / a.sum()
    cands.append(a)

    a = np.maximum(rng.random(n), 0.01)
    a *= S / a.sum()
    cands.append(a)

    a = np.maximum(rng.exponential(1.0, n), 0.01)
    a *= S / a.sum()
    cands.append(a)

    # Smooth random
    freqs = rng.integers(1, n // 2, size=n // 4)
    amps = rng.random(n // 4) * 0.25
    phases = rng.random(n // 4) * 2 * np.pi
    a = np.ones(n)
    for f, amp, ph in zip(freqs, amps, phases):
        a += amp * np.cos(2 * np.pi * f * i / n + ph)
    a = np.maximum(a, 0.01)
    a *= S / a.sum()
    cands.append(a)

    return cands

            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

    # --- Final verification ---
    if global_a is None:
        global_a = np.full(256, np.sqrt(2 * 256) / 256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.full(256, np.sqrt(2 * 256) / 256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: best_obj={global_best:.6f}, verified={ver:.6f}")

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.6f}")

    return final_a
```
