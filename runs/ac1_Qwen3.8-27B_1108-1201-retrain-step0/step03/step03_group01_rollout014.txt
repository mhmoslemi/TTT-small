import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, float)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1, float)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        return np.maximum(v - (cumsum[-1] - S) / n, 0)
    rho = np.max(np.where(valid)[0]) + 1
    theta_val = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta_val, 0)

def convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def fista_refined(a_init, n, S, max_time, rng, 
                   beta_start=5.0, beta_end=500.0,
                   top_k=20, use_top_k_grad=False):

    Refined FISTA with LSE approximation.
    Key improvements:
    - Adaptive beta with geometric increase
    - Top-k gradient for efficiency
    - Backtracking line search for step size
    """
    a = np.clip(a_init, 0, 1000).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: geometric increase
        beta = beta_start * (beta_end / beta_start) ** (it / max(it, 100))
        beta = min(beta, beta_end)
        
        # Compute convolution
        c = convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = y.copy()
        
        # Compute gradient of LSE approximation
        if use_top_k_grad and it > 50:
            # Use top-k lags only
            k_eff = min(top_k + it // 20, 2*n-1)
            top_idx = np.argsort(c)[::-1][:k_eff]
            c_shifted = c[top_idx] - c[top_idx].max()
            w = np.exp(beta * c_shifted)
            w = w / w.sum()
            # Build sparse weight vector
            w_full = np.zeros(2*n-1)
            w_full[top_idx] = w
        else:
            c_shifted = c - c.max()
            w = np.exp(beta * c_shifted)
            total = w.sum()
            if total < 1e-300:
                break
            w = w / total
            w_full = w

        # Actually, grad of LSE = 2 * sum_k w_k * Q_k * y
        # which is 2 * convolve(y, w_full) (with appropriate interpretation)
        # More precisely: grad_a [sum_k w_k * (a^T Q_k a)] = 2 * sum_k w_k * Q_k * a
        # Q_k * a at position i = a[i+k] + a[i-k] (with bounds)
        # So sum_k w_k * (Q_k * a) = convolve-ish operation

        # (Q_k * a)[i] = a[i+k] + a[i-k] for valid i
        # sum_k w_k * (Q_k * a)[i] = sum_k w_k * a[i+k] + sum_k w_k * a[i-k]
        # = (w_rev * a)[i] + (w * a)[i] (convolutions)
        # where w is defined on [-n+1, n-1]

        W = fft(w_full, fsize)
        A = fft(y, fsize)

        if gnorm < 1e-15:
            break

        # Simplified: use 1/(2*beta) as before, but with backtracking
        step = 1.0 / (2.0 * beta * max(gnorm / (2*S), 1e-10))

        if y_sum > 0:

        a = x_new
        t = t_new
        
        # Restart periodically
        if it % 500 == 0:
            t = 1.0
    
    return best_a, best_score

What about a "sparse" sequence with a specific pattern (like a m-sequence)? Binary m-sequences have good autocorrelation properties. For a binary sequence of length n (values 0 and 1), the autocorrelation is nearly flat (except at lag 0). This is exactly what we want!

For a binary m-sequence of length n = 2^m - 1:
- The periodic autocorrelation is -1/(2^m-1) for all non-zero lags.
- The aperiodic (linear) autocorrelation has a peak at lag 0 of n, and the other values are bounded.

If we use a binary m-sequence as our sequence (with 0s and 1s), the sum is about n/2 (half 1s), and the max convolution value is about n (at lag 0, since a[0]^2 = 1 if a[0] = 1, but wait, the max is at the middle lag where all terms contribute).

Hmm, for a binary sequence, conv(n-1) = sum a[i]*a[n-1-i]. If the sequence is "balanced" (roughly half 1s), this is about n/4. And conv(0) = a[0]^2 which is 0 or 1.

So for a balanced binary sequence: max ≈ n/4 (at the middle), sum ≈ n/2, score = 2*n*(n/4)/(n/2)^2 = 2*n^2/4 / (n^2/4) = 2. Still 2!
