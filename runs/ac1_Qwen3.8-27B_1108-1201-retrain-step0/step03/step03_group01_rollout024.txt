import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full linear self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_sumexp_gradient(a, c, beta):
    """
    Compute gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a.
    Uses FFT cross-correlation for efficiency.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Softmax weights over convolution values
    c_shifted = c - c.max()
    exp_bc = np.exp(beta * c_shifted)
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total  # length 2n-1

    # Gradient: grad_a[j] = 2 * sum_i w[j+i] * a[i] for valid i
    # This is cross-correlation of w with a, evaluated at lags 0..n-1
    # Via FFT: need N >= (2n-1) + n - 1 = 3n-2
    N2 = 1 << (3 * n - 3).bit_length()
    W = fft(w, N2)
    A = fft(a, N2)
    cross = np.real(ifft(np.conj(W) * A))[:n]
    grad = 2.0 * cross
    return grad


def sumexp_smooth_max(c, beta):
    """Compute (1/beta)*log(sum(exp(beta*c))) as a smooth approximation of max(c)."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    return c_max + np.log(exp_bc.sum()) / beta


def nesterov_sumexp(a_init, n, S, max_time, rng, beta_start=3.0, beta_end=50.0,
                     step_init=None, momentum_init=0.9):
    """
    Nesterov-accelerated gradient descent on sum-exp approximation of max(conv).
    Returns (best_a, best_score).
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    # Compute initial convolution and set initial step size
    c0 = fft_convolve_self(a)
    max0 = c0.max()
    if step_init is None:
        step = S / (n * 2.0)
    else:
        step = step_init

    y = a.copy()
    a_prev = a.copy()
    momentum = momentum_init
    start = time.monotonic()
    best_score = max0
    best_a = a.copy()
    it = 0
    patience = 5000
    last_improve_it = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: increase over time
        progress = min(1.0, it / max(1, int(max_time / 0.05)))
        beta = beta_start + (beta_end - beta_start) * progress

        # Nesterov extrapolation
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0.0)
        # Renormalize y to maintain sum constraint (soft)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)

        # Compute gradient at y
        c_y = fft_convolve_self(y)
        cur_smooth = sumexp_smooth_max(c_y, beta)

        if c_y.max() < best_score:
            best_score = c_y.max()
            best_a = a.copy()
            last_improve_it = it

        grad = compute_sumexp_gradient(y, c_y, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Gradient step with projection
        a_new = project_simplex(y - step * grad, S)

        # Evaluate new point
        if it % 3 == 0:
            c_new = fft_convolve_self(a_new)
            if c_new.max() < best_score:
                best_score = c_new.max()
                best_a = a_new.copy()
                last_improve_it = it

        # Update momentum (Nesterov)
        momentum = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        momentum = min(momentum, 50.0)  # Cap momentum

        # Adaptive step: reduce if objective increases
        c_a_new = fft_convolve_self(a_new)
        if c_a_new.max() > cur_smooth * 1.5 + max0:
            step *= 0.5
        elif it % 100 == 0:
            step *= 1.01  # Slowly increase

        a_prev = a
        a = a_new

        # Periodic reset to escape local minima
        if it - last_improve_it > patience:
            if rng.random() < 0.3:
                a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
                a_prev = a.copy()
                momentum = momentum_init
                step *= 0.5
                last_improve_it = it

        # Patience timeout
        if it - last_improve_it > patience * 3:
            break

    return best_a, best_score


def spectral_flatten(a_init, n, S, max_time, rng, top_k=20, alpha=0.5):
    """
    Spectral domain optimization: directly modify |FFT(a)|^2 to flatten autocorrelation.
    Uses the fact that conv(a,a) = IDFT(|FFT(a_padded)|^2).
    Gradient of max(c) w.r.t. P=|A|^2 is just the DFT row at argmax.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    N = 1 << (2 * n - 2).bit_length()
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    step = 0.1

    while time.monotonic() - start < max_time:
        it += 1
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(A * A))[:2 * n - 1]
        c_max = c.max()

        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        if best_score < 1.45:
            break

        # Sum-exp weights for smooth max
        beta = min(100.0, 2.0 + 0.05 * it)
        c_shifted = c - c_max
        w = np.exp(beta * c_shifted)
        w /= w.sum()

        # Gradient of sum-exp(c) w.r.t. P:
        # c = IDFT(P), so dc[k]/dP[m] = (1/N)*exp(2πimk/N) (real part for symmetric P)
        # grad_P = sum_k w[k] * IDFT basis at k = DFT of w (conjugated and scaled)
        W = fft(w, N)
        grad_P = np.real(np.conj(W)) / N  # length N

        # Update P
        P_new = np.maximum(P - step * grad_P, 0.0)

        # Preserve phase, update magnitude
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a_new = project_simplex(a_new_full, S)

        # Check improvement
        c_check = fft_convolve_self(a_new)
        if c_check.max() < c_max * 0.999:
            a = a_new
            step *= 1.02
        else:
            # Backtrack
            a = project_simplex(a + 0.5 * (a_new - a), S)
            step *= 0.7

        # Occasionally perturb
        if it % 500 == 0 and rng.random() < 0.2:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)

    return best_a, best_score

def make_init_chirp(n, freq1, freq2, rng):
    """Chirp-based initialization."""
    t = np.arange(n) / n
    a = 1.0 + 0.3 * np.sin(2 * np.pi * (freq1 * t + freq2 * t * t))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_block(n, n_blocks, rng):
    """Block-structured initialization with optimized heights."""
    block_size = n // n_blocks
    remainder = n % n_blocks
    best_a = None
    best_score = np.inf

    for trial in range(100):
        heights = rng.random(n_blocks) + 0.1
        a = np.zeros(n)
        pos = 0
        for i in range(n_blocks):
            sz = block_size + (1 if i < remainder else 0)
            a[pos:pos + sz] = heights[i]
            pos += sz
        c = fft_convolve_self(a)
        score = c.max() / (a.sum() ** 2)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a


def make_init_gaussian_mix(n, rng, n_comp=3):
    """Gaussian mixture initialization."""
    a = np.zeros(n)
    for _ in range(n_comp):
        center = rng.random() * n
        width = rng.uniform(n * 0.02, n * 0.15)
        amp = rng.uniform(0.5, 2.0)
        t = np.arange(n)
        a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.05  # Small base to avoid zeros
    return a


def safe_normalize(a, n):
    """Clamp to [0,1000] and normalize."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

            c_p = fft_convolve_self(a_p)
            obj_p = 2.0 * n * c_p.max() / (a_p.sum() ** 2)
            if obj_p < global_best:
                global_best = obj_p
                global_a = a_p.copy()
                print(f"  [FISTA] obj={global_best:.6f}")

        # Fine spectral tuning
        rem = deadline - time.monotonic()
        if rem > 5:
            a_p2, _ = spectral_flatten(global_a, n, S, min(rem * 0.4, 150.0), rng,
                                       top_k=30, alpha=0.2)
            a_p2 = safe_normalize(a_p2, n)
            c_p2 = fft_convolve_self(a_p2)
            obj_p2 = 2.0 * n * c_p2.max() / (a_p2.sum() ** 2)
            if obj_p2 < global_best:
                global_best = obj_p2
                global_a = a_p2.copy()
                print(f"  [SPEC fine] obj={global_best:.6f}")

        # Final Nesterov with very high beta
        rem = deadline - time.monotonic()
        if rem > 5:
            a_p3, _ = nesterov_sumexp(global_a, n, S, min(rem * 0.4, 150.0), rng,
                                       beta_start=50.0, beta_end=200.0)
            a_p3 = safe_normalize(a_p3, n)
            c_p3 = fft_convolve_self(a_p3)
            obj_p3 = 2.0 * n * c_p3.max() / (a_p3.sum() ** 2)
            if obj_p3 < global_best:
                global_best = obj_p3
                global_a = a_p3.copy()
                print(f"  [NG high-beta] obj={global_best:.6f}")

        # Very final top-k polish
        rem = deadline - time.monotonic()
        if rem > 3:
            a_p4, _ = topk_subgradient_step(global_a, n, S, rem, rng, top_k_init=50)
            a_p4 = safe_normalize(a_p4, n)
            c_p4 = fft_convolve_self(a_p4)
            obj_p4 = 2.0 * n * c_p4.max() / (a_p4.sum() ** 2)
            if obj_p4 < global_best:
                global_best = obj_p4
                global_a = a_p4.copy()
                print(f"  [FINAL topk] obj={global_best:.6f}")

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)

        final_list = [float(x) for x in global_a]
        try:
            ver = evaluate_sequence(final_list)
            print(f"Verification: {ver:.6f}" if np.isfinite(ver) else "Verification: inf")
            if not np.isfinite(ver):
                print("  evaluate_sequence rejected, using fallback")
                global_a = np.ones(256, dtype=np.float64)
        except Exception as e:
            print(f"  Verification error: {e}, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print(f"Returning sequence of length {len(global_a)}, internal obj={global_best:.6f}")
    return [float(x) for x in global_a]
