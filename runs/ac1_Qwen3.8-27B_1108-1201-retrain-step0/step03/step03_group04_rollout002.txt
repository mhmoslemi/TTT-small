import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_from_weights(a, weights):
    """
    Compute 2 * cross_correlation(weights, a) which equals the gradient
    of sum_k weights[k] * c[k] with respect to a.

    W = fft(weights, fsize)
    A = fft(a, fsize)

    return grad

            break
        pm = peak_at(mid)
        if pm <= best_peak:
            best_peak = pm
            best_alpha = mid
        else:
            lo = mid
        if lo >= hi:
            break
    
    return best_alpha, best_peak


def spectral_perturb(a, rng, strength=0.1):
    """
    Perturb the phase of the DFT of a to explore different basins.
    Returns a perturbed (and projected) version of a.
    """
    n = len(a)
    S = a.sum()
    fsize = 1 << (n - 1).bit_length()
    A = fft(a, fsize)
    magnitude = np.abs(A)
    phase = np.angle(A)
    
    # Add random phase perturbation (smooth in frequency)
    n_freq = fsize // 2 + 1
    smooth_noise = np.convolve(rng.normal(0, 1, max(n_freq, 32)),
                                np.ones(8) / 8, mode='same')[:n_freq]
    phase_pert = phase + strength * smooth_noise
    
    # Reconstruct with new phase
    A_new = np.zeros(fsize, dtype=np.complex128)
    A_new[:n_freq] = magnitude[:n_freq] * np.exp(1j * phase_pert)
    # Enforce Hermitian symmetry for real output
    for k in range(1, n_freq):
        if k < fsize - k:
            A_new[fsize - k] = np.conj(A_new[k])
    
    a_new = np.real(ifft(A_new, n))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def smooth_spectral_perturb(a, rng, strength=0.05):
    """
    Smoothly perturb the magnitude spectrum to flatten it.
    """
    n = len(a)
    S = a.sum()
    fsize = 1 << (n - 1).bit_length()
    A = fft(a, fsize)
    magnitude = np.abs(A)
    phase = np.angle(A)
    
    # Compute power spectrum and its mean
    power = magnitude ** 2
    power_mean = power.mean()
    
    # Damp peaks in the spectrum
    damp_factor = 1.0 - strength * np.maximum(power - power_mean, 0) / (power.max() + 1e-10)
    magnitude_new = magnitude * damp_factor
    
    # Add small random phase perturbation
    n_freq = fsize // 2 + 1
    small_noise = rng.normal(0, strength * 0.1, n_freq)
    phase_new = phase[:n_freq] + small_noise
    
    A_new = np.zeros(fsize, dtype=np.complex128)
    A_new[:n_freq] = magnitude_new[:n_freq] * np.exp(1j * phase_new)
    for k in range(1, n_freq):
        if k < fsize - k:
            A_new[fsize - k] = np.conj(A_new[k])
    
    a_new = np.real(ifft(A_new, n))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new * (S / s)


def optimize_peaks(a_init, n, S, max_time, rng, 
                   eps_init=0.0, eps_max=0.5, eps_growth=0.001,
                   stall_threshold=200, perturb_interval=500):
    """
    Main optimization loop with threshold-based active set and line search.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    start = time.monotonic()
    best_peak = np.inf
    best_local = a.copy()
    it = 0
    epsilon = eps_init
    stall_count = 0
    last_best = best_peak
    
    # Initial computation
    c = fft_convolve_self(a)
    current_peak = c.max()
    best_peak = current_peak
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        it += 1
        
        # Update active set threshold
        if it % 100 == 0:
            epsilon = min(eps_max, epsilon + eps_growth * (1 + stall_count / 50.0))
        
        # Find active lags
        threshold = current_peak * (1.0 - epsilon)
        if epsilon == 0.0:
            active_mask = (c >= current_peak - 1e-10)
        else:
            active_mask = (c >= threshold)
        
        if not np.any(active_mask):
            active_mask = np.zeros(len(c), dtype=bool)
            active_mask[np.argmax(c)] = True
        
        weights = np.zeros(len(c))
        weights[active_mask] = 1.0
        # Could also weight by proximity to peak
        if epsilon > 0.01:
            weights[active_mask] = (c[active_mask] - threshold) / (current_peak - threshold)
            if weights.sum() > 0:
                pass  # keep as is (normalized implicitly in gradient)
        
        # Compute gradient
        grad = compute_gradient_from_weights(a, weights)
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-14:
            break
        
        # Direction: negative gradient, projected to sum-0
        direction = -grad
        direction = direction - np.mean(direction)  # sum to 0
        
        # Compute max feasible alpha
        neg_mask = direction < -1e-15
        if np.any(neg_mask):
            max_alpha = np.min(a[neg_mask] / (-direction[neg_mask])) * 0.99
        else:
            max_alpha = S / (n * 1.0)
        
        if max_alpha < 1e-15:
            # Can't move in this direction, perturb
            a = project_simplex(a + rng.normal(0, 0.001 * S / np.sqrt(n), n), S)
            c = fft_convolve_self(a)
            current_peak = c.max()
            stall_count += 10
            continue
        
        # Line search
        best_alpha, new_peak = line_search_peaks(a, c, direction, n, max_alpha)
        
        if best_alpha > 1e-15 and new_peak < current_peak - 1e-12:
            a = a + best_alpha * direction
            a = project_simplex(a, S)  # Re-project to handle numerical issues
            c = fft_convolve_self(a)
            current_peak = c.max()
            stall_count = 0
        else:
            # Try simplex projected step with decreasing step sizes
            improved = False
            step = S / (n * 2.0)
            for try_count in range(5):
                a_try = project_simplex(a - step * grad / max(gnorm, 1e-15), S)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < current_peak - 1e-12:
                    a = a_try
                    c = c_try
                    current_peak = c.max()
                    improved = True
                    stall_count = 0
                    break
                step *= 0.5
            if not improved:
                stall_count += 1
        
        # Track best
        if current_peak < best_peak - 1e-12:
            best_peak = current_peak
            best_local = a.copy()
            if best_peak < last_best * 0.999:
                last_best = best_peak
        
        # Diversification: when stuck, try spectral perturbation
        if stall_count > stall_threshold:
            if stall_count % 50 == 0:
                if rng.random() < 0.5:
                    a_pert = spectral_perturb(a, rng, strength=min(0.3, 0.05 * stall_count / 100.0))
                else:
                    a_pert = smooth_spectral_perturb(a, rng, strength=min(0.2, 0.03 * stall_count / 100.0))
                
                c_pert = fft_convolve_self(a_pert)
                if c_pert.max() < best_peak * 1.01:
                    a = a_pert
                    c = c_pert
                    current_peak = c.max()
                    epsilon = eps_init  # Reset threshold
                    stall_count = 0
                else:
                    # Try blend
                    blend = 0.5 * a + 0.5 * a_pert
                    blend = project_simplex(blend, S)
                    c_blend = fft_convolve_self(blend)
                    if c_blend.max() < best_peak * 1.005:
                        a = blend
                        c = c_blend
                        current_peak = c.max()
                        stall_count = 0
            
            stderr = stall_count > stall_threshold * 3
    
    return best_local, best_peak


def optimize_with_restarts(a_init, n, S, max_time, rng, n_restarts=3):
    """
    Run optimization with periodic restarts from perturbed versions of best.
    """
    a = a_init.copy()
    best_a = a_init.copy()
    best_peak = np.inf
    
    # Phase 1: Main optimization
    time_main = max_time * 0.6
    a_opt, peak = optimize_peaks(a, n, S, time_main, rng,
                                  eps_init=0.0, eps_max=0.3, eps_growth=0.002,
                                  stall_threshold=150, perturb_interval=400)
    best_a = a_opt.copy()
    best_peak = peak
    print("  Main phase: peak=%.6f, obj=%.6f" % (best_peak, 2.0 * n * best_peak / (S * S)))
    
    remaining = max_time - (time.monotonic() - (max_time - time_main))  # approximate
    time_per_restart = max_time * 0.4 / n_restarts
    
    # Phase 2: Restarts from perturbed best
    for r in range(n_restarts):
        if time.monotonic() > 0:  # Just to track
            pass
        pert_strength = 0.02 + 0.03 * r
        a_pert = a_opt + rng.normal(0, pert_strength * S / np.sqrt(n), n)
        a_pert = project_simplex(a_pert, S)
        
        start_r = time.monotonic()
        a_r, peak_r = optimize_peaks(a_pert, n, S, time_per_restart, rng,
                                      eps_init=0.0, eps_max=0.2, eps_growth=0.003,
                                      stall_threshold=100, perturb_interval=300)
        elapsed_r = time.monotonic() - start_r
        if peak_r < best_peak:
            best_a = a_r.copy()
            best_peak = peak_r
            print("  Restart %d: peak=%.6f, obj=%.6f [%.0fs]" % (
                r, best_peak, 2.0 * n * best_peak / (S * S), elapsed_r))
    
    return best_a, best_peak


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2385, **kwargs):
    """
    Minimize 2*n*max_conv/sum^2 over non-negative sequences.
    Uses adaptive peak-chasing with threshold-based active sets,
    explicit line search, and spectral diversification.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s - 30.0)
    
    global_best_obj = np.inf
    global_a = None
    
    # Get the given sequence
    given_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given_a = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    # Try multiple sequence lengths
    candidate_lengths = []
    if given_a is not None:
        n_parent = len(given_a)
        candidate_lengths.append(n_parent)
        # Also try other lengths
        for nl in [4096, 8192, 16384]:
            if nl != n_parent:
                candidate_lengths.append(nl)
    else:
        candidate_lengths = [8192]
    
    # Remove duplicates and sort by priority (parent first)
    candidate_lengths = list(dict.fromkeys(candidate_lengths))
    
    # Time allocation: give more time to parent length
    n_candidates = len(candidate_lengths)
    if n_candidates > 1:
        # 70% to parent, 30% split among others
        times = [float(budget_s - 40) * 0.7]
        remaining_time = float(budget_s - 40) * 0.3
        times += [remaining_time / (n_candidates - 1)] * (n_candidates - 1)
    else:
        times = [float(budget_s - 40)]
    
    for idx, n in enumerate(candidate_lengths):
        if time.monotonic() > deadline - 10:
            break
        
        S = np.sqrt(2.0 * float(n))
        
        # Initialize
        if idx == 0 and given_a is not None:
            a_start = safe_normalize(given_a, n)
        else:
            # Create initial sequence: blend of parent (interpolated) and random
            a_random = rng.random(n) * 0.5 + 0.25
            a_random = safe_normalize(a_random, n)
            if given_a is not None and n != len(given_a):
                # Interpolate parent to new length
                from numpy import interp
                parent_resized = interp(np.linspace(0, 1, n),
                                        np.linspace(0, 1, len(given_a)),
                                        given_a)
                a_parent = safe_normalize(parent_resized, n)
                a_start = safe_normalize(0.7 * a_parent + 0.3 * a_random, n)
            elif given_a is not None:
                a_start = safe_normalize(given_a, n)
            else:
                a_start = a_random
        
        c_start = fft_convolve_self(a_start)
        init_peak = c_start.max()
        init_obj = 2.0 * n * init_peak / (S * S)
        
        if init_obj < global_best_obj:
            global_best_obj = init_obj
            global_a = a_start.copy()
        
        print("n=%d: initial peak=%.6f, obj=%.6f" % (n, init_peak, init_obj))
        
        # Run optimization
        time_alloc = min(times[idx], deadline - time.monotonic() - 5)
        if time_alloc < 30:
            continue
        
        print("Optimizing n=%d with %.0f sec budget..." % (n, time_alloc))
        t0 = time.monotonic()
        
        a_opt, peak_opt = optimize_with_restarts(a_start, n, S, time_alloc, rng, n_restarts=3)
        
        a_opt = safe_normalize(a_opt, n)
        c_opt = fft_convolve_self(a_opt)
        peak_final = c_opt.max()
        obj_final = 2.0 * n * peak_final / (S * S)
        
        print("n=%d: final peak=%.6f, obj=%.6f [%.0fs]" % (n, peak_final, obj_final, time.monotonic() - t0))
        
        if obj_final < global_best_obj:
            global_best_obj = obj_final
            global_a = a_opt.copy()
    
    # Final verification and output
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        print("FINAL: length=%d, objective=%.6f" % (len(final_list), global_best_obj))
        
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver) and ver < 100:
            print("Verified: %.6f" % ver)
        else:
            print("Verification gave %.6f, checking..." % ver)
            if not np.isfinite(ver):
                # Fallback
                final_list = [float(x) for x in np.clip(given_a, 0.0, 1000.0)] if given_a is not None else [0.5] * 256
                print("Using fallback, verified: %.6f" % evaluate_sequence(final_list))
        
        return final_list
    else:
        return [0.5] * 256
