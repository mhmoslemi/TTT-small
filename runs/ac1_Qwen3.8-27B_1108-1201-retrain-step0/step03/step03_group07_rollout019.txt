"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using smooth gradient descent with Adam
on log-parameterization, temperature annealing, and Fourier-based initialization.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def next_power_of_2(n):
    return 1 << (n - 1).bit_length()


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = next_power_of_2(out_len)
    A = fft(a, fsize)
    B = fft(b, fsize)
    return np.real(ifft(A * B))[:out_len]


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    fsize = next_power_of_2(2 * n - 1)
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:2 * n - 1]


def compute_smooth_grad(a, c, temperature):
    """
    Compute gradient of smooth max of c w.r.t. a.
    Uses softmax-weighted average of individual c[k] gradients.
    Returns gradient vector of length n.
    """
    n = len(a)
    # Softmax weights
    shifted = temperature * (c - c.max())
    w = np.exp(shifted)
    w /= w.sum()
    
    # grad[m] = 2 * sum_k w[k] * a[k-m] (with a[j]=0 for j outside [0,n-1])
    # = 2 * cross-correlation of w and a at lag m
    fsize = next_power_of_2(3 * n)
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def softmax_stable(z):
    z_max = z.max()
    e = np.exp(z - z_max)
    return e / e.sum()


def objective_value(a):
    """Compute the objective 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def smooth_objective(a, temperature):
    """Compute smooth approximation of objective and its gradient w.r.t. a."""
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    # Smooth max
    shifted = temperature * (c - c.max())
    log_sum_exp = c.max() + np.log(np.exp(shifted).sum()) / temperature
    obj = 2.0 * n * log_sum_exp / (S * S)
    grad = compute_smooth_grad(a, c, temperature)
    return obj, grad, c


def adam_update(params, grads, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer update. state is a dict with 'm', 'v', 't'."""
    if 't' not in state:
        state['m'] = np.zeros_like(params)
        state['v'] = np.zeros_like(params)
        state['t'] = 0
    state['t'] += 1
    t = state['t']
    state['m'] = beta1 * state['m'] + (1 - beta1) * grads
    state['v'] = beta2 * state['v'] + (1 - beta2) * grads ** 2
    m_hat = state['m'] / (1 - beta1 ** t)
    v_hat = state['v'] / (1 - beta2 ** t)
    return params - lr * m_hat / (np.sqrt(v_hat) + eps)


def optimize_from_z(z0, S, n, time_budget, T_start=50.0, T_end=500.0,
                    lr_start=0.01, lr_end=0.0001, verbose=True):
    """
    Optimize using Adam on log-parameterization z where a = S * softmax(z).
    """
    z = z0.copy().astype(np.float64)
    state = {}
    start_time = time.monotonic()
    total_iters = 0
    
    # Estimate how many iterations we can do
    # Each iteration: FFT of size ~2n + FFT of size ~3n ≈ O(n log n)
    # For n=4096: maybe 50k-100k iters per second
    # Let's just run until time budget
    
    while time.monotonic() - start_time < time_budget:
        elapsed = time.monotonic() - start_time
        progress = min(1.0, elapsed / time_budget)
        
        # Anneal temperature
        T = T_start + (T_end - T_start) * progress
        # Anneal learning rate
        lr = lr_start + (lr_end - lr_start) * progress
        
        # Compute a from z
        a = S * softmax_stable(z)
        
        # Compute objective and gradient
        c = fft_convolve_self(a)
        c_max = c.max()
        S_actual = a.sum()
        obj = 2.0 * n * c_max / (S_actual * S_actual)
        
        # Smooth gradient
        grad_a = compute_smooth_grad(a, c, T)
        
        # Chain rule: grad_z[j] = a[j]*grad_a[j] - (a[j]/S)*sum_i(a[i]*grad_a[i])
        dot = np.dot(a, grad_a)
        grad_z = a * grad_a - (a / S_actual) * dot
        
        # Adam update
        z = adam_update(z, grad_z, state, lr)
        
        total_iters += 1
        if verbose and total_iters % 5000 == 0:
            print(f"    iter={total_iters}, T={T:.1f}, lr={lr:.5f}, "
                  f"obj={obj:.6f}, elapsed={elapsed:.1f}s")
    
    # Final evaluation
    a = S * softmax_stable(z)
    c = fft_convolve_self(a)
    obj = 2.0 * n * c.max() / (a.sum() ** 2)
    return a, obj


def fourier_init(n, seed=42):
    """
    Create an initial sequence based on Fourier design.
    Goal: make |A(f)|^2 as flat as possible (giving flat auto-correlation).
    """
    rng = np.random.default_rng(seed)
    
    # Start with a flat power spectrum (white noise in frequency domain)
    # A(f) = sqrt(n) * exp(i*phase(f)) for f != 0, A(0) = S
    # Taking ifft gives a sequence with flat auto-correlation (approximately)
    
    # But we need a >= 0, so we add a large DC component
    # a = alpha * 1 + ifft(random_phase_spectrum)
    
    S = np.sqrt(2.0 * n)
    
    # Random phase spectrum
    fsize = next_power_of_2(2 * n)
    A = np.zeros(fsize, dtype=complex)
    A[0] = S * 0.5  # DC component (will be adjusted)
    for f in range(1, n):
        A[f] = np.exp(1j * rng.uniform(0, 2 * np.pi))
        A[fsize - f] = np.conj(A[f])
    
    a_osc = np.real(ifft(A))[:n]
    
    # Add constant to ensure non-negativity
    offset = -a_osc.min() + 0.1
    a = a_osc + offset
    
    # Normalize
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    
    return a


def cosine_init(n, freq=1, seed=42):
    """Cosine-based initialization."""
    i = np.arange(n)
    S = np.sqrt(2.0 * n)
    a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * i / n)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def random_init(n, seed=42):
    """Random initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = np.maximum(a, 0.01)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a

            if time.monotonic() - start > time_budget:
                break

                continue

            best_factor = 1.0

            # Try several scaling factors
            for factor in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
                a[i] = orig * factor
                a_norm = a * (S / a.sum())
                c_new = fft_convolve_self(a_norm)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using smooth gradient descent with Adam
on log-parameterization, temperature annealing, and multi-phase optimization.
Key innovations over prior work: softmax-smoothed max (vs top-k), Adam optimizer
on log-space (vs multiplicative mirror descent), aggressive temperature scheduling,
and structured multi-restart strategy.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def next_power_of_2(n):
    """Return smallest power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    fsize = next_power_of_2(2 * n - 1)
    A = fft(a, fsize)
    return np.real(ifft(A * A))[:2 * n - 1]


def compute_smooth_grad(a, c, temperature):
    """
    Compute gradient of smooth max of c w.r.t. a using softmax weighting.
    grad[m] = 2 * sum_k softmax(T*c)[k] * a[k-m] (a[j]=0 outside [0,n-1]).
    Uses FFT cross-correlation for efficiency.
    """
    n = len(a)
    # Softmax weights (temperature-scaled)
    shifted = temperature * (c - c.max())
    # Clip to avoid underflow issues
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: uniform weight on argmax
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # Cross-correlation: grad[m] = 2 * sum_k w[k] * a[k-m]
    # = 2 * ifft(fft(w) * conj(fft(a)))[m] with zero-padding
    fsize = next_power_of_2(3 * n)
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def stable_softmax(z):
    """Numerically stable softmax."""
    z_max = z.max()
    if z_max > 700:
        z = z - z_max
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def obj_from_conv(a, c):
    """Compute objective from precomputed convolution."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def adam_step(params, grads, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. Returns updated params and updated state."""
    t = state['t'] + 1
    state['t'] = t
    state['m'] = beta1 * state['m'] + (1.0 - beta1) * grads
    state['v'] = beta2 * state['v'] + (1.0 - beta2) * grads * grads
    m_hat = state['m'] / (1.0 - beta1 ** t)
    v_hat = state['v'] / (1.0 - beta2 ** t)
    return params - lr * m_hat / (np.sqrt(v_hat) + eps), state


def optimize_log_space(z0, S, n, time_budget, T_start=20.0, T_end=500.0,
                       lr_start=0.005, lr_end=0.0003, verbose=True,
                       stagnation_patience=3000, perturbation_scale=0.15):
    """
    Optimize a = S * softmax(z) to minimize 2n*max(conv(a,a))/S^2.
    Uses Adam on z with temperature-annealed smooth gradient.
    """
    z = z0.copy().astype(np.float64)
    state = {'m': np.zeros_like(z), 'v': np.zeros_like(z), 't': 0}
    start_time = time.monotonic()
    total_iters = 0
    best_obj = np.inf
    best_z = z.copy()
    last_improve_iter = 0

    while time.monotonic() - start_time < time_budget:
        elapsed = time.monotonic() - start_time
        progress = min(1.0, elapsed / max(1e-6, time_budget))

        # Anneal temperature (linear in progress)
        T = T_start + (T_end - T_start) * progress
        # Anneal learning rate
        lr = lr_start * (lr_end / lr_start) ** progress

        # Compute a from z
        a = S * stable_softmax(z)

        # Compute convolution and objective
        c = fft_convolve_self(a)
        cur_obj = obj_from_conv(a, c)

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_z = z.copy()
            last_improve_iter = total_iters

        # Compute smooth gradient w.r.t. a
        grad_a = compute_smooth_grad(a, c, T)

        # Chain rule: d/dz_j [f(a(z))] = sum_i (df/da_i)(da_i/dz_j)
        # da_i/dz_j = a_i * (delta_ij - a_j / S)
        # So grad_z_j = a_j * grad_a_j - (a_j / S) * sum_i(a_i * grad_a_i)
        dot = np.dot(a, grad_a)
        grad_z = a * grad_a - (a / S) * dot

        # Gradient clipping for stability
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm > 10.0:
            grad_z = grad_z * (10.0 / grad_norm)

        # Adam update
        z, state = adam_step(z, grad_z, state, lr)

        # Stagnation detection: perturb if no improvement for a while
        if total_iters - last_improve_iter > stagnation_patience:
            noise = np.random.randn(n) * perturbation_scale
            z = z + noise
            state['m'] *= 0.1
            state['v'] *= 0.1
            last_improve_iter = total_iters
            if verbose:
                print(f"      [perturb] iter={total_iters}, obj={cur_obj:.6f}, "
                      f"best={best_obj:.6f}, t={elapsed:.1f}s")

        total_iters += 1
        if verbose and total_iters % 2000 == 0:
            c_best = fft_convolve_self(S * stable_softmax(best_z))
            obj_best_now = obj_from_conv(S * stable_softmax(best_z), c_best)
            print(f"      iter={total_iters}, T={T:.1f}, lr={lr:.5f}, "
                  f"cur={cur_obj:.6f}, best={obj_best_now:.6f}, t={elapsed:.1f}s")

    # Return best found
    a_best = S * stable_softmax(best_z)
    c_best = fft_convolve_self(a_best)
    obj_best = obj_from_conv(a_best, c_best)
    return a_best, obj_best


def fourier_init(n, seed=0):
    """
    Initialize using random-phase spectrum to get oscillatory + DC structure.
    The auto-correlation will be relatively flat away from the DC peak.
    """
    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * n)

    fsize = next_power_of_2(2 * n)
    A = np.zeros(fsize, dtype=complex)
    # Set DC to control the offset
    A[0] = S * 0.6
    # Random phases for other frequencies
    phases = rng.uniform(0, 2 * np.pi, size=n)
    amplitudes = np.ones(n) * (S * 0.4) / np.sqrt(n)
    for f in range(1, n):
        A[f] = amplitudes[f] * np.exp(1j * phases[f])
        A[fsize - f] = np.conj(A[f])

    a_osc = np.real(ifft(A))[:n]
    # Ensure non-negativity
    offset = -a_osc.min() + abs(S) * 0.01
    a = a_osc + offset
    a = np.maximum(a, 0.001)
    a = a * (S / a.sum())
    return a


def multi_cosine_init(n, seed=0):
    """Sum of multiple cosines at different frequencies."""
    rng = np.random.default_rng(seed)
    i = np.arange(n)
    S = np.sqrt(2.0 * n)
    a = np.ones(n)
    n_components = rng.integers(2, 6)
    for _ in range(n_components):
        freq = rng.integers(1, n // 2)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.001)
    a = a * (S / a.sum())
    return a


def random_init(n, seed=0):
    """Simple random initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = np.maximum(a, 0.001)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())
    return a


def welch_init(n, seed=0):
    """
    Welch-like initialization: design |A(f)|^2 to be flat, then recover a.
    This gives a sequence whose auto-correlation is close to a delta (flat top).
    """
    rng = np.random.default_rng(seed)
    S = np.sqrt(2.0 * n)

    # Target: |A(f)|^2 = S^2/n for all f (flat spectrum)
    # This means a is like a delta (all energy at one point) which is bad.
    # Instead, use: |A(f)|^2 = S^2 * (1 + alpha * cos(2*pi*f*p/n)) / n
    # which gives a auto-correlation with a plateau.

    # Simpler: use a raised-cosine window in frequency
    fidx = np.arange(n)
    # Flat spectrum with a dip at high frequencies (smooth rolloff)
    spec_mag = np.ones(n)
    rolloff_start = int(n * 0.7)
    if rolloff_start < n - 1:
        t = (fidx[rolloff_start:] - rolloff_start) / (n - rolloff_start)
        spec_mag[rolloff_start:] = 0.5 * (1 + np.cos(np.pi * t))

    # Convert to time domain (this gives the auto-correlation)
    # Then we need a such that a*a (conv) has this shape
    # Actually, let's just use ifft of |A|^2 as a starting "target correlation"
    # and use its sqrt (in frequency) as our |A|
    A_mag = np.sqrt(spec_mag)
    phases = rng.uniform(0, 2 * np.pi, size=n)
    phases[0] = 0
    # Ensure Hermitian symmetry
    A = np.zeros(n, dtype=complex)
    A[0] = A_mag[0]
    for f in range(1, (n + 1) // 2):
        A[f] = A_mag[f] * np.exp(1j * phases[f])
        A[n - f] = np.conj(A[f])
    if n % 2 == 0:
        A[n // 2] = A_mag[n // 2]

    a = np.real(ifft(A))
    a = np.maximum(a, 0.001)
    a = a * (S / a.sum())
    return a


def coordinate_descent_polish(a0, time_budget, max_sweeps=50):
    """
    Coordinate descent: iteratively scale individual elements to reduce peak.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_from_conv(a, c)
    start = time.monotonic()
    factors = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        sweep_improved = False

        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            if a[i] < 1e-14:
                continue

            orig = a[i]
            best_c_max = c.max()
            best_factor = 1.0

            for f in factors:
                a[i] = orig * f
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    continue
                a_tmp = a * (S / s)
                c_tmp = fft_convolve_self(a_tmp)
                if c_tmp.max() < best_c_max:
                    best_c_max = c_tmp.max()
                    best_factor = f

            if best_factor != 1.0:
                a[i] = a[i] * (best_factor / (a[i] / orig))
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
                cur_obj = obj_from_conv(a, c)
                if cur_obj < best_obj:
                    best_obj = cur_obj
                sweep_improved = True
            else:
                a[i] = orig

        if not sweep_improved:
            break

    return a, best_obj


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    
    Multi-phase strategy:
    Phase 1 (10%): Quick screening of diverse initializations
    Phase 2 (50%): Main smooth gradient descent with annealing on best candidate
    Phase 3 (20%): Multi-restart with perturbation to escape local minima
    Phase 4 (20%): Polish with coordinate descent and fine-tuning
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    global_best = np.inf
    global_a = None
    global_n = 0

    # --- Load given sequence ---
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw.copy()
            n_given = len(given_a)
            ver = evaluate_sequence(given_a.tolist())
            print(f"Given sequence: n={n_given}, obj={ver:.6f}")
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
                global_n = n_given
    except Exception as e:
        print(f"Could not load given sequence: {e}")

    # --- Phase 1: Screening (10% of time) ---
    p1_frac = 0.10
    p1_time = p1_frac * (deadline - time.monotonic())
    print(f"\n=== Phase 1: Screening ({p1_time:.0f}s) ===")

    # Generate diverse initializations
    inits = []
    if given_a is not None:
        inits.append(('given', given_a))

    for n in [4096, 8192]:
        S = np.sqrt(2.0 * n)
        a_f = fourier_init(n, seed=seed)
        inits.append((f'fourier_{n}', a_f))
        a_m = multi_cosine_init(n, seed=seed + 100)
        inits.append((f'mcos_{n}', a_m))
        a_w = welch_init(n, seed=seed + 200)
        inits.append((f'welch_{n}', a_w))
        a_r = random_init(n, seed=seed + 300)
        inits.append((f'rand_{n}', a_r))

    per_init_time = p1_time / len(inits)
    for name, a_init in inits:
        if time.monotonic() > deadline - 300:
            break
        n_init = len(a_init)
        S_init = a_init.sum()
        z_init = np.log(np.maximum(a_init, 1e-10))

        t0 = time.monotonic()
        a_out, obj_out = optimize_log_space(
            z_init, S_init, n_init, per_init_time,
            T_start=10.0, T_end=100.0,
            lr_start=0.004, lr_end=0.0004,
            verbose=False
        )
        elapsed = time.monotonic() - t0
        # Verify
        c_v = fft_convolve_self(a_out)
        obj_v = obj_from_conv(a_out, c_v)
        print(f"  {name}: obj={obj_v:.6f} ({elapsed:.1f}s)")

        if obj_v < global_best:
            global_best = obj_v
            global_a = a_out.copy()
            global_n = n_init
            print(f"    *** New best: {obj_v:.6f} (n={n_init}) ***")

    # --- Phase 2: Main optimization (50% of time) ---
    p2_frac = 0.50
    p2_time = p2_frac * (deadline - time.monotonic())
    print(f"\n=== Phase 2: Main optimization ({p2_time:.0f}s, n={global_n}) ===")

    if global_a is not None and p2_time > 30:
        n = global_n
        S = global_a.sum()
        z_init = np.log(np.maximum(global_a, 1e-10))

        # Sub-phase 2a: Broad (low temperature, wider search)
        t_2a = 0.30 * p2_time
        if t_2a > 10:
            print(f"  2a: Broad (T: 5->80, {t_2a:.0f}s)")
            a_out, obj_out = optimize_log_space(
                z_init, S, n, t_2a,
                T_start=5.0, T_end=80.0,
                lr_start=0.003, lr_end=0.0003,
                verbose=True, stagnation_patience=4000
            )
            c_v = fft_convolve_self(a_out)
            obj_v = obj_from_conv(a_out, c_v)
            print(f"    Result: {obj_v:.6f}")
            if obj_v < global_best:
                global_best = obj_v
                global_a = a_out.copy()
                z_init = np.log(np.maximum(global_a, 1e-10))
                print(f"    *** New best: {obj_v:.6f} ***")

        # Sub-phase 2b: Medium (medium temperature)
        t_2b = 0.40 * p2_time
        if t_2b > 10 and time.monotonic() < deadline - 60:
            print(f"  2b: Medium (T: 80->400, {t_2b:.0f}s)")
            a_out, obj_out = optimize_log_space(
                z_init, S, n, t_2b,
                T_start=80.0, T_end=400.0,
                lr_start=0.002, lr_end=0.0002,
                verbose=True, stagnation_patience=3000
            )
            c_v = fft_convolve_self(a_out)
            obj_v = obj_from_conv(a_out, c_v)
            print(f"    Result: {obj_v:.6f}")
            if obj_v < global_best:
                global_best = obj_v
                global_a = a_out.copy()
                z_init = np.log(np.maximum(global_a, 1e-10))
                print(f"    *** New best: {obj_v:.6f} ***")

        # Sub-phase 2c: Sharp (high temperature, focus on worst peak)
        t_2c = deadline - time.monotonic() - p2_time + 0.30 * p2_time - (0.30 * p2_time)
        t_2c = p2_time - t_2a - t_2b
        if t_2c > 10 and time.monotonic() < deadline - 60:
            print(f"  2c: Sharp (T: 400->2000, {t_2c:.0f}s)")
            a_out, obj_out = optimize_log_space(
                z_init, S, n, t_2c,
                T_start=400.0, T_end=2000.0,
                lr_start=0.001, lr_end=0.0001,
                verbose=True, stagnation_patience=2000
            )
            c_v = fft_convolve_self(a_out)
            obj_v = obj_from_conv(a_out, c_v)
            print(f"    Result: {obj_v:.6f}")
            if obj_v < global_best:
                global_best = obj_v
                global_a = a_out.copy()
                print(f"    *** New best: {obj_v:.6f} ***")

    # --- Phase 3: Multi-restart (20% of time) ---
    p3_time = 0.20 * (deadline - time.monotonic())
    print(f"\n=== Phase 3: Multi-restart ({p3_time:.0f}s) ===")

    if global_a is not None and p3_time > 20:
        n = global_n
        S = global_a.sum()
        rng = np.random.default_rng(seed + 999)
        n_restarts = max(2, int(p3_time / 40))
        per_restart = p3_time / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break

            # Create perturbation with increasing scale
            scale = 0.05 + 0.3 * (r + 1) / n_restarts
            a_pert = global_a + rng.normal(0, scale, size=n) * (S / np.sqrt(n))
            a_pert = np.maximum(a_pert, 1e-6)
            a_pert = a_pert * (S / a_pert.sum())
            z_pert = np.log(a_pert)

            a_out, obj_out = optimize_log_space(
                z_pert, S, n, per_restart,
                T_start=50.0, T_end=500.0,
                lr_start=0.002, lr_end=0.0002,
                verbose=False, stagnation_patience=2000,
                perturbation_scale=0.1
            )
            c_v = fft_convolve_self(a_out)
            obj_v = obj_from_conv(a_out, c_v)
            print(f"  Restart {r+1}/{n_restarts} (scale={scale:.3f}): obj={obj_v:.6f}")

            if obj_v < global_best:
                global_best = obj_v
                global_a = a_out.copy()
                print(f"    *** New best: {obj_v:.6f} ***")

    # --- Phase 4: Polish (remaining time) ---
    p4_time = deadline - time.monotonic() - 15.0
    print(f"\n=== Phase 4: Polish ({p4_time:.0f}s) ===")

    if global_a is not None and p4_time > 10:
        n = global_n
        S = global_a.sum()

        # 4a: Coordinate descent
        cd_time = 0.4 * p4_time
        if cd_time > 5:
            print(f"  4a: Coordinate descent ({cd_time:.0f}s)")
            t0 = time.monotonic()
            a_cd, obj_cd = coordinate_descent_polish(global_a, cd_time)
            c_v = fft_convolve_self(a_cd)
            obj_cd = obj_from_conv(a_cd, c_v)
            elapsed = time.monotonic() - t0
            print(f"    Result: {obj_cd:.6f} ({elapsed:.1f}s)")
            if obj_cd < global_best:
                global_best = obj_cd
                global_a = a_cd.copy()
                print(f"    *** New best: {obj_cd:.6f} ***")

        # 4b: Very fine smooth gradient (high T, tiny lr)
        fine_time = deadline - time.monotonic() - 10.0
        if fine_time > 10:
            print(f"  4b: Fine-tune ({fine_time:.0f}s)")
            z_fine = np.log(np.maximum(global_a, 1e-10))
            S_fine = global_a.sum()
            a_fine, obj_fine = optimize_log_space(
                z_fine, S_fine, n, fine_time,
                T_start=200.0, T_end=2000.0,
                lr_start=0.0005, lr_end=0.00005,
                verbose=False, stagnation_patience=1500,
                perturbation_scale=0.05
            )
            c_v = fft_convolve_self(a_fine)
            obj_fine = obj_from_conv(a_fine, c_v)
            print(f"    Result: {obj_fine:.6f}")
            if obj_fine < global_best:
                global_best = obj_fine
                global_a = a_fine.copy()
                print(f"    *** New best: {obj_fine:.6f} ***")

    # --- Final safety & verification ---
    if global_a is None:
        print("WARNING: No valid sequence found, using fallback.")
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\n{'='*50}")
    print(f"FINAL: local_obj={global_best:.6f}, verified={ver:.6f}")
    print(f"Sequence length: {len(final_a)}")
    print(f"{'='*50}")

    if not np.isfinite(ver):
        print("Verification failed, using fallback [1.0]*256")
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback verified: {ver:.6f}")

    return final_a
