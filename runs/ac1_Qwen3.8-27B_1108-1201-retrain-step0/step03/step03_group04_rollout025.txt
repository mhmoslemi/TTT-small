import numpy as np

from scipy.fft import fft, ifft

def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_grad_conv(a, w):
    """Gradient of sum_k w[k] * (a*a)[k] with respect to a."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

def softmax_weights(c, beta):
    c_max = c.max()

    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total

    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.interpolate import interp1d


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_weighted(a, weights):
    """Compute gradient of sum_k weights[k]*(a*a)[k] w.r.t. a using FFT."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_grad_variance(a):
    """
    Compute gradient of sum(b^2) where b = a*a (self-convolution).
    Uses FFT: grad[j] = 2 * (b cross-correlate a)[j]
    In Fourier: FFT(grad) = 2 * FFT(b) * conj(FFT(a)) = 2 * FFT(a)^2 * conj(FFT(a))
    """
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    # b = a*a, so FFT(b) = A^2
    # grad = 2 * IFFT(conj(A) * A^2) = 2 * IFFT(A * conj(A) * conj(A) ...) 
    # Actually: d/da sum(b^2) = 2 * (b * a_reversed) in some sense
    # For self-convolution: grad[j] = 4 * sum_k b[k] * a[|k-j|] (approx)
    # In Fourier: FFT(grad) = 4 * conj(FFT(a)) * FFT(b) = 4 * conj(A) * A^2
    # But we need to be careful with the full vs cyclic convolution
    B = A * A  # FFT of self-convolution (cyclic)
    grad = 4.0 * np.real(ifft(B * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(np.clip(beta * (c - c_max), -700, 0))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Compute normalized top-k weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def get_score(a):

    c = fft_convolve_self(a)
    return c.max()


def run_backtracking_opt(a_init, n, S, max_time, rng, beta_init=40.0, beta_max=2000.0,

    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / n * 0.3
    stall_counter = 0
    best_stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.003 * it))

        c = fft_convolve_self(y)
        cur_max = c.max()

        if cur_max < best_score - 1e-14:
            best_score = cur_max
            best_local = y.copy()
            stall_counter = 0
        else:
            stall_counter += 1

        # Compute gradient based on mode
        if use_variance:
            grad = compute_grad_variance(y)
        elif mode == 'topk':
            weights = toppk_weights(c, top_k)
            grad = compute_grad_weighted(y, weights)
        else:
            weights = softmax_weights(c, beta)
            grad = compute_grad_weighted(y, weights)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search (limited)
        step_trial = step
        x_new = y
        c_new_max = cur_max
        for bt in range(6):
            x_candidate = project_simplex(y - step_trial * grad / max(gnorm, 1e-15), S)
            c_candidate = fft_convolve_self(x_candidate)
            c_new_max = c_candidate.max()
            if c_new_max <= cur_max + 1e-13:
                x_new = x_candidate
                break
            step_trial *= 0.5
            x_new = x_candidate
        else:
            x_new = project_simplex(y - step_trial * grad / max(gnorm, 1e-15), S)
            c_new_max = fft_convolve_self(x_new).max()

        # Update step size
        if c_new_max < cur_max * (1.0 - 1e-8):
            step = min(step * 1.02, S / n * 2.0)
        elif stall_counter > 200:
            step *= 0.7

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic perturbation to escape local minima
        if stall_counter > 500 and rng.random() < 0.3:
            noise_scale = 0.004 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
            stall_counter = 0

        # Reset momentum if oscillating
        if it > 100 and it % 500 == 0 and c_new_max > cur_max * 1.01:
            t_mom = 1.0
            y = a.copy()

    return best_local, best_score

        it += 1

        c = fft_convolve_self(y)
        cur_peak = c.max()

            best_peak = cur_peak
            best_local = y.copy()

        grad = compute_grad_variance(y)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking
        step_trial = step
        x_new = y
        for bt in range(5):
            x_candidate = project_simplex(y - step_trial * grad / max(gnorm, 1e-15), S)
            c_candidate = fft_convolve_self(x_candidate)
            if c_candidate.max() <= cur_peak + 1e-13:
                x_new = x_candidate
                break
            step_trial *= 0.5
            x_new = x_candidate

        c_new = fft_convolve_self(x_new)
        if c_new.max() < cur_peak:
            step = min(step * 1.01, S / n)
        else:
            step *= 0.8

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 2000 == 0 and rng.random() < 0.2:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
