def _fft_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    """Objective and gradient for L-BFGS."""
    n = _gs_n
    S = _gs_S
    T = _gs_T
    
    # Convert theta to a (normalized softmax)
    theta_shift = theta - theta.max()
    e = np.exp(theta_shift)
    a = S * e / e.sum()

    c = _fft_conv(a)
    c_max = c.max()
    
    # Smoothed objective: L_T = c_max + (1/T)*log(sum exp(T*(c - c_max)))
    scaled = T * (c - c_max)
    scaled = scaled - scaled.max()  # additional shift for stability (should be <= 0)
    log_sum = scaled.max() + np.log(np.exp(scaled).sum())
    obj = c_max + log_sum / T

    p_raw = T * c - T * c_max  # = T*(c - c_max), max is 0

- If k != 2i and 0 <= k-i < n: delta_c[k] = 2*(v'-v)*a[k-i]

import numpy as np
import time
import scipy.optimize
from scipy.fft import fft, ifft

# Global state for L-BFGS objective/gradient
_gs_n = 0
_gs_S = 1.0
_gs_T = 1.0


def _fft_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def _smoothed_obj_grad(theta):
    """
    Smoothed objective and gradient for L-BFGS.
    a = S * softmax(theta), objective = L_T(a*a) = smooth approx of max(a*a).
    """
    n = _gs_n
    S = _gs_S
    T = _gs_T

    # Convert theta to a (normalized)
    theta_shift = theta - theta.max()
    e = np.exp(theta_shift)
    e_sum = e.sum()
    if e_sum < 1e-300:
        a = np.full(n, S / n)
    else:
        a = S * e / e_sum

    # Convolution
    c = _fft_conv(a)
    c_max = c.max()
    if c_max < 1e-30:
        return 0.0, np.zeros(n)

    # Smoothed objective: L_T = c_max + (1/T)*log(sum_k exp(T*(c_k - c_max)))
    scaled = T * (c - c_max)
    max_scaled = scaled.max()
    log_sum_exp = max_scaled + np.log(np.exp(scaled - max_scaled).sum())
    obj = c_max + log_sum_exp / T

    # Softmax weights for the "which peaks matter" distribution
    p_raw = T * c
    p_raw = p_raw - p_raw.max()
    p = np.exp(p_raw)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(c))
        p[np.argmax(c)] = 1.0
    else:
        p /= p_sum

    # Gradient w.r.t. a: g_a[l] = sum_k p[k]*a[k-l]
    fsize = 1 << (3 * n - 2).bit_length()
    P = fft(p, fsize)
    A = fft(a, fsize)
    g_a = np.real(ifft(P * np.conj(A)))[:n]

    # Gradient w.r.t. theta: dL/dtheta[j] = a[j]*g_a[j] - (a[j]/S)*sum(a*g_a)
    dot = np.dot(g_a, a)
    g_theta = a * g_a - (a / S) * dot

    return obj, g_theta


def run_lbfgs_multi_temp(a_init, n, S, time_budget, T_values=None):
    """
    Run L-BFGS with a schedule of increasing temperatures.
    Returns best (a, objective) found.
    """
    global _gs_n, _gs_S, _gs_T

    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a * (S / a.sum())
    theta = np.log(a)

    best_obj = evaluate_sequence(a.tolist())
    best_a = a.copy()

    if T_values is None:
        T_values = [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096]

    _gs_n = n
    _gs_S = S
    start = time.monotonic()

    for T in T_values:
        if time.monotonic() - start > time_budget:
            break
        _gs_T = T

        n_iter = max(50, min(500, int(time_budget / (len(T_values) * 0.01))))

        try:
            result = scipy.optimize.minimize(
                _smoothed_obj_grad, theta, method='L-BFGS-B',
                jac=True,
                options={'maxiter': n_iter, 'maxfun': n_iter * 2,
                         'ftol': 1e-14, 'gtol': 1e-12, 'maxls': 20}
            )
            theta = result.x
        except Exception:
            pass

        # Convert back to a
        theta_shift = theta - theta.max()
        e = np.exp(theta_shift)
        a_new = S * e / e.sum()
        a_new = np.maximum(a_new, 0.0)

            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_obj


def fast_coordinate_descent(a_init, n, time_budget, max_sweeps=5000):
    """
    Fast coordinate descent with incremental convolution updates.
    Tries multiplying each coordinate by various factors.
    """
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a_init, np.inf
    c = _fft_conv(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    start = time.monotonic()
    rng = np.random.default_rng(42)

    factors = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0, 0.6, 1.4, 0.75, 1.25, 0.4, 1.8])

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        order = rng.permutation(n)
        improved = False

        for i in order:
            v = a[i]
            if v < 1e-14:
                continue

            best_obj_local = best_obj
            best_f = 1.0

            lo = i
            hi = min(2 * n - 2, i + n - 1)
            affected_len = hi - lo + 1

            for f in factors:
                v_new = v * f
                S_new = S + (v_new - v)
                if S_new < 1e-3:
                    continue
                delta = v_new - v

                # Compute delta_c for affected range
                # delta_c[k] = 2*delta*a[k-i] for k in [lo,hi], k != 2i
                # delta_c[2i] = delta*(v_new+v)
                delta_c = 2.0 * delta * a[:affected_len]
                if i < affected_len:
                    delta_c = delta_c.copy()
                    delta_c[i] = delta * (v_new + v)

                new_affected = c[lo:hi + 1] + delta_c
                local_max = new_affected.max()
                left_max = c[:lo].max() if lo > 0 else -np.inf
                right_max = c[hi + 1:].max() if hi + 1 < 2 * n - 1 else -np.inf
                new_max = max(local_max, left_max, right_max)

                new_obj = 2.0 * n * new_max / (S_new * S_new)
                if new_obj < best_obj_local:
                    best_obj_local = new_obj
                    best_f = f

            if best_f != 1.0:
                # Apply the change
                v_old = v
                v_new = v * best_f
                delta = v_new - v_old
                a[i] = v_new
                S += delta

                # Update c incrementally
                delta_c = 2.0 * delta * a[:affected_len]
                if i < affected_len:
                    delta_c = delta_c.copy()
                    delta_c[i] = delta * (v_new + v_old)
                c[lo:hi + 1] += delta_c

                improved = True
                if best_obj_local < best_obj:
                    best_obj = best_obj_local
                    best_a = a.copy()

        if not improved:
            break

    # Final: renormalize to get correct objective
    S_final = a.sum()
    if S_final > 1e-15:
        a = a * (S / S_final)
    final_obj = 2.0 * n * _fft_conv(a).max() / (S * S)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()

    return best_a, best_obj


def get_initialization(name, n, S, rng=None):

        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "geometric":
        r = 1.0 + rng.random() * 0.01
        a = r ** (i / n * 10)
        return a * (S / a.sum())
    elif name == "asym_exp":
        # Asymmetric exponential
        rate = rng.uniform(0.5, 3.0)
        a = np.exp(rate * (i / n - 0.5))
        return a * (S / a.sum())
    elif name == "cosine_asym":
        # Asymmetric cosine mix
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.randint(1, 10) * i / n + rng.random() * 6.28)
        a += 0.2 * np.cos(2 * np.pi * rng.randint(5, 30) * i / n + rng.random() * 6.28)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "skew_gaussian":
        # Skewed Gaussian
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(0.1, 0.4) * n
        a = np.exp(-0.5 * ((i - center) / width) ** 2)
        a += 0.5 * np.exp(-0.5 * ((i - center * 1.5) / (width * 0.5)) ** 2)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "two_sided":
        # Two-sided structure
        a = np.exp(-0.5 * (i / (n * 0.15)) ** 2) + np.exp(-0.5 * ((n - 1 - i) / (n * 0.1)) ** 2)
        return a * (S / a.sum())
    elif name == "power_law":
        alpha = rng.uniform(0.5, 2.0)
        a = (i + 1) ** (-alpha)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)

            print("\nFinal polish: obj=%.7f" % obj_final)

    # Final verification
    if global_a is None:
        global_a = np.ones(256).tolist()

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: best=%.7f, verified=%.7f" % (global_best, ver if np.isfinite(ver) else float('inf')))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback used: obj=%.7f" % ver)

    return final_a
