    lr = 0.5
    momentum = 0.9
    v = np.zeros(n)
    global_iter = 0

    for p_val in p_schedule:
        # Reset momentum when p changes significantly
        if p_val > 4:
            v *= 0.5
            momentum = min(0.99, 0.9 + 0.002 * (p_val - 4))
        
        for it in range(iters_per_p):
            if time.monotonic() - start_time > time_budget:
                break
            
            # Compute gradient
            pn, grad = pnorm_and_grad(c, a, float(p_val))
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad = grad / gnorm  # normalize for stable step size
            
            # Nesterov step
            v = momentum * v + (1.0 - momentum) * grad
            a_new = a - lr * v
            a_new = project_simplex(a_new, 1.0)
            
            # Evaluate
            c_new = fft_convolve_self(a_new)
            pn_new = np.sum(c_new ** p_val) ** (1.0 / p_val)
            
            # Adaptive learning rate
            if pn_new > pn * 1.001 + 1e-15:
                lr = max(lr * 0.7, 1e-6)
                # Revert
                a_new = a
                c_new = c
            else:
                a = a_new
                c = c_new
                lr = min(lr * 1.03, 5.0)
            
            # Track best using exact max
            exact_obj = 2.0 * n * c.max()
            if exact_obj < best_obj:
                best_obj = exact_obj
                best_a = a.copy()
                best_iter = global_iter
            
            global_iter += 1
        
        if verbose:
            elapsed = time.monotonic() - start_time
            print(f"    p={p_val}: obj={best_obj:.6f}, lr={lr:.4f}, t={elapsed:.1f}s")

    return best_a, best_obj


def subgradient_nesterov(a0, time_budget, top_k_start=200, top_k_end=1, verbose=True):
    """
    Subgradient method with Nesterov momentum on the exact max function.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = a / S

    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    best_iter = 0

    lr = 0.3
    momentum = 0.9
    v = np.zeros(n)
    start_time = time.monotonic()
    iter_no = 0
    stagnation = 0
    max_stagnation = 500

    while time.monotonic() - start_time < time_budget:
        # Progress: ramp top_k from start to end
        elapsed_frac = (time.monotonic() - start_time) / time_budget
        top_k = max(1, int(top_k_start * (1.0 - elapsed_frac) ** 2 + top_k_end * elapsed_frac))
        
        # Compute subgradient (softmax-weighted average of active peaks)
        c_max = c.max()
        # Softmax weights over all (or top) peaks
        top_idx = np.argsort(c, kind='stable')[::-1][:max(1, min(top_k, len(c)))]
        # Temperature for softmax
        T = max(1.0, (c_max - c[top_idx[-1]]) * 5.0 if len(top_idx) > 1 else 1.0)
        w = np.exp((c[top_idx] - c_max) * T)
        w /= w.sum()
        
        # Build full weight vector
        w_full = np.zeros(len(c))
        w_full[top_idx] = w
        
        # Gradient via FFT
        a_rev = a[::-1]
        fsize = 1 << (len(c) + n - 1 - 1).bit_length()
        W = fft(w_full, fsize)
        R = fft(a_rev, fsize)
        full_conv = np.real(ifft(W * R))
        grad = 2.0 * full_conv[n - 1:2 * n - 1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Nesterov step
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = project_simplex(a_new, 1.0)
        
        c_new = fft_convolve_self(a_new)
        obj_new = 2.0 * n * c_new.max()
        
        if obj_new < best_obj * (1.0 + 1e-10):
            a = a_new
            c = c_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
                best_iter = iter_no
                stagnation = 0
                lr = min(lr * 1.02, 2.0)
            else:
                stagnation += 1
        else:
            lr = max(lr * 0.8, 1e-5)
            stagnation += 1
            # Revert
            # a stays the same
        
        if stagnation > max_stagnation:
            # Perturbation
            noise = np.random.randn(n) * 0.01 / np.sqrt(n)
            a = a + noise
            a = project_simplex(a, 1.0)
            c = fft_convolve_self(a)
            v *= 0.5
            lr = 0.3
            stagnation = 0
        
        iter_no += 1
        if verbose and iter_no % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    iter={iter_no}, obj={2*n*c.max():.6f}, best={best_obj:.6f}, "
                  f"lr={lr:.4f}, top_k={top_k}, t={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_descent(a0, time_budget, verbose=True):
    """
    Coordinate descent: for each coordinate, try to find the optimal value
    by scanning a small range.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()
    start_time = time.monotonic()
    sweep = 0

    while time.monotonic() - start_time < time_budget:
        improved = False
        # Process in random order
        idx = np.random.permutation(n)
        
        for i in idx:
            if time.monotonic() - start_time > time_budget:
                break
            
            orig = a[i]
            if orig < 1e-10:
                continue
            
            # Try a few multiplicative factors
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
                a[i] = orig * factor
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a_norm = a * (1.0 / s)
                c_new = fft_convolve_self(a_norm)
                obj_new = 2.0 * n * c_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_norm.copy()
                    a = a_norm
                    c = c_new
                    improved = True
                    break
            a[i] = a[i]  # keep current (already set above if improved)
            # Renormalize
            s = a.sum()
            if s > 0:
                a = a * (1.0 / s)
        
        sweep += 1
        if not improved:
            if verbose:
                print(f"    CD sweep {sweep}: no improvement, stopping.")
            break
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    CD sweep {sweep}: obj={best_obj:.6f}, t={elapsed:.1f}s")

    return best_a, best_obj
