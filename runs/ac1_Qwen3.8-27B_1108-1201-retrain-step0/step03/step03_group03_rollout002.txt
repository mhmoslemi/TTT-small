import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.maximum(v - (cumsum[n] - S) / n, 0.0)
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
        return np.maximum(v - theta, 0.0)

For a single quadratic f(a) = a^T P a where P is a permutation matrix (shifted/reverse), the gradient is 2Pa (or just Pa depending on how we define it). The Hessian is 2P (or P).

The optimal step size for gradient descent on a quadratic is:

where L is the largest eigenvalue and mu is the smallest eigenvalue of the Hessian.

For a permutation matrix P:
- All eigenvalues have magnitude 1
- L = 2 (or 1), mu = -2 (or -1) if P has negative eigenvalues

Wait, for a real permutation matrix, the eigenvalues are on the unit circle. For the "shift and reverse" operation, the eigenvalues are e^{i*k*pi/(n-1)} type things.

d(conv_k)/da_i = a_{k-i} (if k-i is in [0,n-1])

Wait, but conv_k also has the term where a_i appears as a_{k-i} when i and k-i are both valid:
conv_k = sum_j a_j * a_{k-j}

The terms involving a_i are:

- When k-j = i, i.e., j = k-i: a_{k-i} * a_i

d(conv_k)/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i}

Actually, more carefully:

For a specific i, a_i appears in the sum when:
- j = i: term is a_i * a_{k-i} (requires 0 <= k-i < n)
- k-j = i, i.e., j = k-i: term is a_{k-i} * a_i (requires 0 <= k-i < n)

If i != k-i (i ≠ k/2): both terms are present, contribution is 2*a_i*a_{k-i}, derivative is 2*a_{k-i}
If i = k/2: single term a_i^2, derivative is 2*a_i

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy:
1. Start from parent sequence (or structured init)
2. Peak-chasing gradient descent with momentum (exact subgradient)
3. Multi-lag targeting to prevent peak migration
4. Coordinate descent polish
5. Try multiple lengths (8192, 16384)

import numpy as np
import time
from scipy.fft import fft, ifft

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        theta = (cumsum[n-1] - S) / n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def peak_gradient(a, peak_idx, n):
    """Gradient of conv[peak_idx] w.r.t. a."""
    k = peak_idx
    g = np.zeros(n)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    # g[i] = 2*a[k-i] for i in [i_min, i_max]
    # k-i ranges from k-i_max to k-i_min
    src_start = k - i_max
    src_end = k - i_min
    g[i_min:i_max+1] = 2.0 * a[src_start:src_end+1][::-1]
    return g

def run_peak_chase(a_init, n, S, max_time, rng, verbose=True):

    a = np.clip(a_init, 0, 1000).astype(np.float64)

    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    it = 0
    stall = 0
    top_k_lags = 3
    
    while time.monotonic() - start < max_time:
        it += 1
        
        if it % 20 == 1:
            c = fft_convolve_self(a)
            score = c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 0.05
        
        # Compute convolution of y for gradient direction
        c_y = fft_convolve_self(y)
        
        # Get top-k lags for robustness
        top_indices = np.argsort(c_y)[::-1][:top_k_lags]
        
        # Weighted gradient: weight by how close each lag is to the peak
        max_c = c_y[top_indices[0]]
        weights = np.zeros(len(top_indices))
        for i, idx in enumerate(top_indices):
            weights[i] = max(0, 1.0 - (max_c - c_y[idx]) / max_c) ** 2
        if weights.sum() < 1e-10:
            weights[0] = 1.0
        weights = weights / weights.sum()

        g = np.zeros(n)
        for i, idx in enumerate(top_indices):
            g += weights[i] * peak_gradient(y, int(idx), n)

            break
        
        # Adaptive step size
        # For the dominant quadratic (single peak), optimal step ~ 1/L where L ~ 2*||a||_inf or so
        # Use a conservative step

        x_new = None

            if bt == 0:

                c_cand = fft_convolve_self(x_candidate)
                if c_cand.max() <= f_y + 1e-3:
                    x_new = x_candidate
                    break
            step *= 0.5
        
        if x_new is None:
            x_new = project_simplex(y - step * g, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Perturbation to escape local minima

            noise = intensity * S / np.sqrt(n) * rng.standard_normal(n)
            y = project_simplex(y + noise, S)

            stall = 0
        
        if verbose and it % 5000 == 0:
            c_check = fft_convolve_self(a)
            print("  peak_chase it=%d score=%.6f best=%.6f" % (it, c_check.max(), best_score))
    
    return best_a, best_score

def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic initialization."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a

def make_periodic_init(n, rng, S, num_freqs=4):
    """Periodic initialization."""
    tt = np.arange(n, dtype=np.float64) / n

        a += amp * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a
