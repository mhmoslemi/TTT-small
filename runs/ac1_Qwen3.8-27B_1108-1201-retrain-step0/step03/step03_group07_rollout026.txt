def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_correlate(w, a):
    """Compute g[i] = sum_k w[k] * a[k-i] for i=0..n-1 (valid k-i in range)."""
    n = len(a)
    a_rev = a[::-1]
    out_len = len(w) + n - 1
    fsize = 1 << (out_len - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full = np.real(ifft(W * R))
    return full[n - 1:n - 1 + n]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v_clipped = np.clip(v, 0, None)
    s = v_clipped.sum()
    if s <= S:
        return v_clipped + (S - s) / n
    lo, hi = 0.0, v_clipped.max()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s_mid = np.maximum(v_clipped - mid, 0.0).sum()
        if s_mid > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.maximum(v_clipped - tau, 0.0)

            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.2
            eta = eta0
            best_iter = iter_no

        eta *= eta_decay
        iter_no += 1

        if verbose and iter_no % 1000 == 0 and time.monotonic() - last_print > 5:
            print(f"      MD iter={iter_no}, obj={cur_obj:.6f}, best={best_obj:.6f}, "
                  f"eta={eta:.4f}, top_k={top_k_cur}, t={time.monotonic()-start:.1f}s")
            last_print = time.monotonic()

    return best_a, best_obj


def coordinate_polish(a0, S, time_budget, max_rounds=500):
    """Coordinate-wise optimization: try reducing each element to lower max peak."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    start = time.monotonic()

    # Find the peak location
    k_max = c.argmax()

    for rd in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break
        improved = False
        # Focus on indices that contribute most to the peak
        # grad[i] = 2*a[k_max - i] for valid i
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        # Try modifying indices in order of their contribution
        indices = list(range(hi, lo - 1, -1))
        # Add some random indices
        indices.extend(np.random.randint(0, n, min(n // 4, 512)).tolist())

        tried = set()
        for i in indices:
            if time.monotonic() - start > time_budget:
                break
            if i in tried or i < 0 or i >= n:
                continue
            tried.add(i)
            orig = a[i]
            if orig < 1e-15:
                continue
            for factor in [0.8, 0.9, 1.1, 1.2]:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-10:
                    a[i] = orig
                    break
                a = a * (S / s)
                # Quick check: only recompute affected part of convolution
                c = fft_convolve_self(a)
                cur_max = c.max()
                if cur_max < best_obj * S * S / (2.0 * n) * (1.0 - 1e-10):
                    best_obj = 2.0 * n * cur_max / (S * S)
                    best_a = a.copy()
                    improved = True
                    k_max = c.argmax()
                    break
                else:
                    a[i] = orig * (S / a.sum())  # restore
            # Restore and renormalize
            a = best_a.copy() if improved else a
            if not improved:
                a[i] = orig
                a = a * (S / a.sum())

        if not improved and rd > 20:
            break

    return best_a, best_obj


def generate_initializations(rng, n):
    """Generate diverse initializations for a given length n."""
    S = np.sqrt(2.0 * n)
    i = np.arange(n)
    candidates = []

    # Uniform
    a = np.ones(n) * S / n
    candidates.append(('uniform', a.copy()))

    # Cosine with various frequencies
    for freq in [3, 5, 7, 11, 13, 17]:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * i / n)
        a = np.maximum(a, 0.01)
        a = a * S / a.sum()
        candidates.append((f'cos_{freq}', a.copy()))

    # Sum of two cosines
    for f1, f2 in [(3, 7), (5, 13), (7, 19)]:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * f1 * i / n) + 0.2 * np.cos(2 * np.pi * f2 * i / n)
        a = np.maximum(a, 0.01)
        a = a * S / a.sum()
        candidates.append((f'cos2_{f1}_{f2}', a.copy()))

    # Gaussian windows
    for center_frac in [0.3, 0.5]:
        for width_frac in [0.1, 0.15, 0.2]:
            a = np.exp(-0.5 * ((i - n * center_frac) / (n * width_frac)) ** 2)
            a = np.maximum(a, 0.001)
            a = a * S / a.sum()
            candidates.append((f'gauss_{center_frac}_{width_frac}', a.copy()))

    # Random (with smoothing)
    for trial in range(3):
        a = rng.random(n)
        # Smooth it
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.001)
        a = a * S / a.sum()
        candidates.append((f'rand_smooth_{trial}', a.copy()))

    # Two humps
    a = (np.exp(-0.5 * ((i - n * 0.25) / (n * 0.08)) ** 2) +
         np.exp(-0.5 * ((i - n * 0.75) / (n * 0.08)) ** 2))
    a = np.maximum(a, 0.001)
    a = a * S / a.sum()
    candidates.append(('two_humps', a.copy()))

    # Ramp
    a = np.linspace(0.1, 1.0, n)
    a = a * S / a.sum()
    candidates.append(('ramp', a.copy()))

    return candidates


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    global_best = np.inf
    global_a = None

    # --- Load given sequence ---
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print(f"Given sequence: n={n_given}, obj={ver:.6f}")
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print(f"Could not load given sequence: {e}")

    # --- Phase 1: Exploration with Adam smooth optimization ---
    explore_time = 0.25 * (deadline - time.monotonic())
    print(f"\nPhase 1: Adam exploration ({explore_time:.0f}s)")

    # Generate candidates for multiple lengths
    all_candidates = []
    if given_a is not None:
        all_candidates.append(('given', given_a, len(given_a)))

    for n_try in [2048, 4096, 8192]:
        cands = generate_initializations(rng, n_try)
        for name, a in cands:
            all_candidates.append((f'{name}_n{n_try}', a, n_try))

    # Evaluate all candidates quickly
    print(f"  Evaluating {len(all_candidates)} initial candidates...")
    best_cand_name = None
    for name, a, n in all_candidates:
        S = a.sum()
        c = fft_convolve_self(a)
        obj = 2.0 * n * c.max() / (S * S)
        if obj < global_best:
            global_best = obj
            global_a = a.copy()
            best_cand_name = name
        if time.monotonic() > deadline - explore_time - 60:
            break

    print(f"  Best initial: {best_cand_name}, obj={global_best:.6f}")

    # Run Adam on top candidates
    top_candidates = []
    for name, a, n in all_candidates:
        S = a.sum()
        c = fft_convolve_self(a)
        obj = 2.0 * n * c.max() / (S * S)
        top_candidates.append((obj, name, a, n))
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:8]  # Top 8

    per_cand_time = explore_time / len(top_candidates)
    for rank, (obj_init, name, a, n) in enumerate(top_candidates):
        if time.monotonic() > deadline - 120:
            break
        S = np.sqrt(2.0 * n)
        a = a * (S / a.sum())
        t0 = time.monotonic()
        a_best, obj_best = adam_smooth_optimize(
            a, S, per_cand_time,
            beta_start=5.0, beta_end=300.0,
            lr=0.5, verbose=(rank < 3)
        )
        elapsed = time.monotonic() - t0
        # Verify
        ver = obj_value(a_best)
        if ver < global_best:
            global_best = ver
            global_a = a_best.copy()
        print(f"  [{rank}] {name}: {obj_init:.6f} -> {ver:.6f} ({elapsed:.1f}s)")

    # --- Phase 2: Main optimization with mirror descent ---
    main_time = 0.40 * (deadline - time.monotonic())
    print(f"\nPhase 2: Mirror descent main ({main_time:.0f}s)")

    if global_a is not None and main_time > 10:
        n = len(global_a)
        S = global_a.sum()

        # Run multiple MD configurations
        md_configs = [
            {'eta0': 0.25, 'eta_decay': 0.99995, 'top_k_start': 500, 'top_k_end': 50,
             'momentum': 0.85, 'soft_T': 3.0, 'stagnation_limit': 2000,
             'perturb_frac': 0.04, 'perturb_every': 1500},
            {'eta0': 0.15, 'eta_decay': 0.99997, 'top_k_start': 100, 'top_k_end': 10,
             'momentum': 0.9, 'soft_T': 0.0, 'stagnation_limit': 1500,
             'perturb_frac': 0.03, 'perturb_every': 1200},
            {'eta0': 0.1, 'eta_decay': 0.99998, 'top_k_start': 20, 'top_k_end': 1,
             'momentum': 0.92, 'soft_T': 0.0, 'stagnation_limit': 1000,
             'perturb_frac': 0.02, 'perturb_every': 800},
        ]

        sub_time = main_time / len(md_configs)
        for ci, cfg in enumerate(md_configs):
            if time.monotonic() > deadline - 60:
                break
            print(f"  MD config {ci}: eta0={cfg['eta0']}, top_k={cfg['top_k_start']}->{cfg['top_k_end']}")
            a_best, obj_best = mirror_descent_v2(global_a, sub_time, verbose=True, **cfg)
            ver = obj_value(a_best)
            if ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print(f"    -> improved to {ver:.6f}")
            else:
                print(f"    -> no improvement (best stays {global_best:.6f}, got {ver:.6f})")

    # --- Phase 3: Adam polish with high beta ---
    adam_time = 0.15 * (deadline - time.monotonic())
    print(f"\nPhase 3: Adam polish ({adam_time:.0f}s)")

    if global_a is not None and adam_time > 5:
        n = len(global_a)
        S = global_a.sum()
        a_best, obj_best = adam_smooth_optimize(
            global_a, S, adam_time,
            beta_start=100.0, beta_end=1000.0,
            lr=0.1, verbose=True
        )
        ver = obj_value(a_best)
        if ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print(f"  -> improved to {ver:.6f}")
        else:
            print(f"  -> no improvement (best stays {global_best:.6f})")

    # --- Phase 4: Coordinate polish ---
    polish_time = deadline - time.monotonic() - 15.0
    print(f"\nPhase 4: Coordinate polish ({polish_time:.0f}s)")

    if global_a is not None and polish_time > 3:
        n = len(global_a)
        S = global_a.sum()
        a_best, obj_best = coordinate_polish(global_a, S, polish_time)
        ver = obj_value(a_best)
        if ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print(f"  -> improved to {ver:.6f}")
        else:
            print(f"  -> no improvement (best stays {global_best:.6f})")

    # --- Phase 5: Final perturbation + quick MD ---
    final_time = deadline - time.monotonic() - 10.0
    if global_a is not None and final_time > 5:
        n = len(global_a)
        S = global_a.sum()
        # Try a few random perturbations followed by short MD
        for attempt in range(5):
            if time.monotonic() > deadline - 10:
                break
            a_pert = global_a.copy()
            noise = rng.randn(n) * 0.01 * S / np.sqrt(n)
            a_pert = np.maximum(a_pert + noise, 0.0)
            a_pert = a_pert * (S / a_pert.sum())
            obj_pert = obj_value(a_pert)
            t_sub = max(1.0, (deadline - time.monotonic() - 10) / (5 - attempt))
            a_best, obj_best = mirror_descent_v2(a_pert, t_sub,
                                                 eta0=0.1, eta_decay=0.9999,
                                                 top_k_start=10, top_k_end=1,
                                                 momentum=0.9, stagnation_limit=500,
                                                 verbose=False)
            ver = obj_value(a_best)
            if ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print(f"  Perturb attempt {attempt}: improved to {ver:.6f}")

    # --- Final verification ---
    print(f"\n{'='*50}")
    print(f"Best found: {global_best:.6f}")

    if global_a is None:
        global_a = np.ones(256)
        S = np.sqrt(512.0)
        global_a = global_a * S / global_a.sum()

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
        S = np.sqrt(512.0)
        global_a = global_a * S / global_a.sum()

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"FINAL verified: {ver:.6f} (n={len(final_a)}, sum={np.sum(global_a):.4f})")

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.6f}")

    return final_a
