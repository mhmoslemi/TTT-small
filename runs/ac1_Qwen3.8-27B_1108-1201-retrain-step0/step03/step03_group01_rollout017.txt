So the gradient of c[m] with respect to a is: 2 * reverse_shift(a, m), where reverse_shift means we take a[m-j] for each j (with 0 for out-of-bounds).

In code:
```python
def autocorr_gradient(a, m):
    n = len(a)
    grad = np.zeros(n)
    # j ranges from max(0, m-n+1) to min(m, n-1)
    # grad[j] = 2 * a[m-j]
    lo = max(0, m - n + 1)
    hi = min(m, n - 1)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * a[m - j]
    return grad
```

Or vectorized:
```python
def autocorr_gradient(a, m):
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, m - n + 1)
    hi = min(m, n - 1)
    js = np.arange(lo, hi + 1)
    grad[js] = 2.0 * a[m - js]
    return grad
```

This is O(n) per gradient evaluation, which is fine.

Now, for the "top-k" gradient (sum of gradients of the top-k lags weighted by their values):
```python
def topk_gradient(a, c, top_k):
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    grad = np.zeros(n)
    for i, k in enumerate(top_indices):
        grad += weights[i] * autocorr_gradient(a, int(k))
    return grad
```

This is O(k*n) which for k=50 and n=8192 is about 400K operations, very fast.

```python
"""
Minimize 2*n*max(self_conv) / sum^2 for non-negative sequences.
Uses exact gradient computation, adaptive top-k subgradient with Nesterov,
and multi-start exploration with diverse initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    # Sort in descending order
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def exact_grad_c(a, m):
    """Exact gradient of c[m] = sum_i a[i]*a[m-i] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, m - n + 1)
    hi = min(m, n - 1)
    if lo > hi:
        return grad
    js = np.arange(lo, hi + 1)
    grad[js] = 2.0 * a[m - js]
    return grad


def topk_gradient_exact(a, c, top_k):
    """Weighted sum of exact gradients for top-k lags."""
    n = len(a)
    top_k = min(top_k, 2 * n - 1)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    grad = np.zeros(n)
    for i in range(len(top_indices)):
        k = int(top_indices[i])
        grad += weights[i] * exact_grad_c(a, k)
    return grad


def normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def spectral_shaping(a, N, top_k, alpha, n_iters, rng, target):
    """Spectral shaping: modify |FFT|^2 to reduce peak autocorrelation."""
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score = np.inf
    
    for it in range(n_iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = c.max()
        
        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()
        
        if c_max <= target:
            break
        
        # Get top-k lag weights
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        m = np.arange(N)
        cos_eff = np.zeros(N)
        for i in range(len(top_indices)):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(top_indices[i]) / N)
        
        # Adaptive alpha
        excess = max(0.0, c_max - target) / max(c_max, 1e-10)
        alpha_curr = alpha * (0.3 + 0.7 * excess)
        
        P_new = np.maximum(P * (1.0 - alpha_curr * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new, S)
        
        # Occasional noise
        if it % 500 == 250 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.002 * S / np.sqrt(n), n), S)
    
    return best_a, best_score


def nesterov_optimize(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=256, target=1.0):
    """Nesterov-accelerated top-k subgradient with exact gradients."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        if best_score <= target:
            break
        
        grad = topk_gradient_exact(y, c, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic restart
        if it % 2000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0
    
    return best_local, best_score


def peak_shave(a, n, S, n_iters, rng):
    """Directly slash the single peak by moving in the negative gradient direction."""
    best_a = a.copy()
    S_curr = a.sum()
    best_score = np.inf
    
    for it in range(n_iters):
        c = fft_convolve_self(a)
        if c.max() < best_score:
            best_score = c.max()
            best_a = a.copy()
        
        # Find the single peak lag
        k_star = int(np.argmax(c))
        grad = exact_grad_c(a, k_star)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Take a step in the negative gradient direction
        for alpha_try in [0.01, 0.005, 0.002, 0.001]:
            a_try = project_simplex(a - alpha_try * S_curr / gnorm * grad / gnorm, S_curr)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < best_score:
                a = a_try
                break
        else:
            # No improvement, try spectral nudge
            break
    
    return best_a, best_score


def multi_sine_init(n, rng, n_sines=3):
    """Create a multi-sine initialization."""
    a = np.ones(n)
    for _ in range(n_sines):
        freq = rng.uniform(0.1, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        a += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    a = np.maximum(a, 0.01)
    return a


def chirp_init(n, rng):
    """Create a chirp-based initialization."""
    t = np.arange(n)
    beta = rng.uniform(0.5, 5.0)
    a = 1.0 + 0.3 * np.cos(np.pi * beta * t * t / n)
    a = np.maximum(a, 0.01)
    return a


def random_gap_init(n, rng, gap_frac=0.3):
    """Create an initialization with random gaps."""
    a = np.ones(n)
    n_gaps = int(n * gap_frac)
    gap_positions = rng.choice(n, n_gaps, replace=False)
    a[gap_positions] = 0.0
    a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    return a
