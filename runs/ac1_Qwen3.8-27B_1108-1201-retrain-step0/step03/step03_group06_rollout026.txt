import numpy as np

from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_weighted_subgrad(c, a, T, top_k=None):
    """Compute weighted subgradient with optional top-k peak focus."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)

    if top_k is not None and top_k < len(c):
        # Focus on top-k peaks with high weight, rest with low weight
        thresh = np.sort(c)[-top_k]
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        # Boost top-k peaks
        top_mask = c >= thresh
        w[top_mask] *= (1 + T * 0.5)
    else:
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)

    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]

def peak_targeted_cd(a_init, n, S, time_budget, verbose=False):
    """
    Peak-targeted coordinate descent using two-point mass transfers.
    For the maximum peak at lag k, identify responsible coordinates and
    transfer mass to safe coordinates.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)

    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    iteration = 0
    rng = np.random.default_rng(54321)

    while time.monotonic() - start < time_budget:
        iteration += 1
        c = fft_convolve_self(a)
        cur_obj = c.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()

        # Find top peaks
        k_max = np.argmax(c)
        peak_val = c[k_max]

        # Find responsible coordinates for the peak at k_max
        # c[k_max] = sum_i a[i] * a[k_max - i]
        # Contribution of coordinate i is a[i] * a[k_max - i]
        valid_i_start = max(0, k_max - n + 1)
        valid_i_end = min(k_max, n - 1)
        if valid_i_start > valid_i_end:
            break

        indices = np.arange(valid_i_start, valid_i_end + 1)
        contributions = a[indices] * a[k_max - indices]

        # Top responsible coordinates
        n_responsible = min(20, len(indices))
        top_resp_local = np.argsort(contributions)[-n_responsible:]
        top_resp = indices[top_resp_local]

        # Find safe coordinates: those with low contribution to ALL top peaks
        # Approximate: low a[i] * a[k_max - i] AND low a[i] * a[i] (to not create new peaks)
        safe_scores = a.copy()
        # Penalize coordinates that contribute to the current peak
        for i in top_resp:
            if 0 <= k_max - i < n:
                safe_scores[i] += a[k_max - i]
                safe_scores[k_max - i] += a[i]

        # Try to move mass from responsible to safe coordinates
        n_safe_candidates = min(30, n)
        safe_candidates = np.argsort(safe_scores)[:n_safe_candidates]
        # Filter out the responsible ones
        safe_mask = np.array([s not in set(top_resp.tolist()) for s in safe_candidates])
        safe_candidates = safe_candidates[safe_mask]

        if len(safe_candidates) == 0:
            safe_candidates = np.array([np.argmin(safe_scores)])

        improved = False
        n_tries = min(20, len(top_resp))
        try_order = rng.permutation(n_responsible)[:n_tries]

        for ti in try_order:
            if time.monotonic() - start > time_budget:
                break
            i_src = top_resp[ti]
            # Try moving mass from i_src to various safe targets
            for j_dst in safe_candidates[:5]:
                if j_dst == i_src:
                    continue
                # Bisection for optimal delta
                best_delta = 0
                best_delta_obj = cur_obj
                a_orig = a[i_src]

                for frac in [0.1, 0.05, 0.02, 0.01, 0.005]:
                    delta = a_orig * frac
                    if delta < 1e-15:
                        continue
                    a[i_src] = a_orig - delta
                    a[j_dst] = a[j_dst] + delta
                    a = normalize_nonneg(a, S)
                    c_check = fft_convolve_self(a)
                    obj_check = c_check.max()
                    if obj_check < best_delta_obj - 1e-12:
                        best_delta_obj = obj_check
                        best_delta = delta
                    a[i_src] = a_orig

                if best_delta > 0:
                    a[i_src] = a_orig - best_delta
                    a[j_dst] = a[j_dst] + best_delta
                    a = normalize_nonneg(a, S)
                    improved = True
                    c = fft_convolve_self(a)
                    cur_obj = c.max()
                    if cur_obj < best_obj - 1e-12:
                        best_obj = cur_obj
                        best_a = a.copy()

        if not improved and iteration > 5:
            # Try random perturbations to escape
            if iteration % 10 == 0:
                n_perturb = rng.integers(1, min(20, n // 10))
                perturb_idx = rng.choice(n, size=n_perturb, replace=False)
                for idx in perturb_idx:
                    old = a[idx]
                    for factor in [0.5, 1.5, 0.3, 0.7, 2.0]:
                        a[idx] = old * factor
                        a = normalize_nonneg(a, S)
                        c_check = fft_convolve_self(a)
                        if c_check.max() < best_obj - 1e-12:
                            best_obj = c_check.max()
                            best_a = a.copy()
                            cur_obj = best_obj
                            break
                    else:
                        a[idx] = old
                        a = normalize_nonneg(a, S)

    return best_a, best_obj


def spectral_init(n, S, rng):
    """
    Initialize using spectral method: random phases in frequency domain,
    inverse DFT, clip to non-negative.
    """
    fsize = 1 << (2 * n - 1).bit_length()
    # Random phases
    phases = rng.random(fsize) * 2 * np.pi
    # Magnitude spectrum: try to make it somewhat flat
    mags = np.ones(fsize)
    mags[0] = 0  # DC component handled by normalization
    A = mags * np.exp(1j * phases)
    a = np.real(ifft(A))[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.full(n, S / n)
    else:
        a = a * (S / s)
    return a


def cosine_init(n, S, rng):
    """Initialize with cosine structure (asymmetric)."""
    i = np.arange(n)
    n_terms = rng.integers(3, 8)
    a = np.ones(n)
    for _ in range(n_terms):
        freq = rng.integers(1, n // 2)
        amp = rng.random() * 0.3
        phase = rng.random() * 2 * np.pi
        a += amp * np.cos(2 * np.pi * freq * i / n + phase)
    a = np.maximum(a, 0.01)
    return a * (S / a.sum())


def two_point_refine(a, n, S, time_budget, verbose=False):
    """
    Fine-grained two-point refinement: try small mass transfers between pairs.
    Focus on reducing the current peak.
    """
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(9999)
    iteration = 0

    while time.monotonic() - start < time_budget:
        iteration += 1
        c = fft_convolve_self(a)
        cur_obj = c.max()
        k_max = np.argmax(c)

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()

        # Try a few random two-point transfers
        n_transfers = max(5, n // 500)
        for _ in range(n_transfers):
            if time.monotonic() - start > time_budget:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            a_orig_i = a[i]
            if a_orig_i < 1e-12:
                continue

            # Try small transfers
            for frac in [0.02, 0.01, 0.005, -0.02, -0.01]:
                delta = a_orig_i * frac
                if a_orig_i + delta < 0 or (frac > 0 and delta < 0):
                    continue
                a[i] = a_orig_i - delta
                a[j] = a[j] + delta
                a = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a)
                if c_check.max() < best_obj - 1e-13:
                    best_obj = c_check.max()
                    best_a = a.copy()
                    cur_obj = best_obj
                    break
                else:
                    a[i] = a_orig_i
                    a[j] = a[j] - delta
                    a = normalize_nonneg(a, S)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, min(float(budget_s), 2380.0) - 30.0)
    deadline = time.monotonic() + total_budget

    global_best = np.inf
    global_a = None

    # Load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw.copy()
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: " + str(e))

    # ============ PHASE 1: Quick multi-length exploration (18% time) ============
    explore_budget = 0.18 * total_budget
    explore_deadline = time.monotonic() + explore_budget

    lengths_to_try = []
    if given_a is not None:
        lengths_to_try.append(len(given_a))
    for n in [2048, 3072, 4096, 5120, 6144, 8192]:
        if n not in lengths_to_try:
            lengths_to_try.append(n)

    lengths_to_try = sorted(set(lengths_to_try))
    n_candidates = len(lengths_to_try)
    per_candidate_time = max(5.0, explore_budget / n_candidates)

    print("Phase 1: Exploration (%.0fs, %d lengths: %s)" %
          (explore_budget, n_candidates, lengths_to_try))

    candidates = []  # (n, a_init, quick_obj)

    for n in lengths_to_try:
        if time.monotonic() > explore_deadline - 10:
            break
        S = np.sqrt(2.0 * float(n))

        # Try up to 3 initializations per length
        n_inits = min(3, max(1, int(per_candidate_time / 10)))
        inits = []

        # Init 1: From given sequence (resampled) or random
        if given_a is not None and len(given_a) != n:
            # Resample using linear interpolation
            old_n = len(given_a)
            new_indices = np.linspace(0, old_n - 1, n)
            idx_floor = np.floor(new_indices).astype(int)
            idx_ceil = np.minimum(idx_floor + 1, old_n - 1)
            frac = new_indices - idx_floor
            a_resamp = given_a[idx_floor] * (1 - frac) + given_a[idx_ceil] * frac
            a_resamp = np.maximum(a_resamp, 0)
            inits.append(a_resamp * (S / a_resamp.sum()))
        elif given_a is not None and len(given_a) == n:
            inits.append(given_a.copy())
        else:
            inits.append(np.full(n, S / n))

        # Init 2: Spectral
        inits.append(spectral_init(n, S, rng))
        # Init 3: Cosine
        inits.append(cosine_init(n, S, rng))

        inits = inits[:n_inits]
        per_init_time = per_candidate_time / max(1, len(inits))

        for init_a in inits:
            if time.monotonic() > explore_deadline - 5:
                break
            if per_init_time < 3:
                break
            # Quick optimization
            a_best, obj_best = adaptive_subgradient(
                init_a, n, S, per_init_time,
                T_init=10, T_final=300, lr_init=0.06
            )
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver):
                candidates.append((n, a_best.copy(), ver))
                if ver < global_best:
                    global_best = ver
                    global_a = a_best.copy()
                    print("  Explore: n=%d obj=%.7f" % (n, ver))

    # Sort candidates by objective
    candidates.sort(key=lambda x: x[2])
    print("  Top candidates: %s" % [(c[0], "%.5f" % c[2]) for c in candidates[:5]])

    # ============ PHASE 2: Deep optimization (55% time) ============
    exploit_budget = 0.55 * total_budget
    exploit_deadline = time.monotonic() + exploit_budget

    # Pick top 2-3 distinct-length candidates
    top_candidates = []
    seen_lengths = set()
    for n, a_cand, obj_cand in candidates:
        if n not in seen_lengths:
            top_candidates.append((n, a_cand))
            seen_lengths.add(n)
        if len(top_candidates) >= 3:
            break

    # Always include the current global best
    if global_a is not None:
        n_gb = len(global_a)
        if n_gb not in seen_lengths:
            top_candidates.append((n_gb, global_a.copy()))

    # Also add the given sequence if not already included
    if given_a is not None and len(given_a) not in seen_lengths and len(top_candidates) < 4:
        top_candidates.append((len(given_a), given_a.copy()))

    n_top = len(top_candidates)
    per_top_budget = exploit_budget / max(1, n_top)

    print("\nPhase 2: Deep optimization (%.0fs, %d candidates)" % (exploit_budget, n_top))

    for idx, (n, a_start) in enumerate(top_candidates):
        if time.monotonic() > exploit_deadline - 30:
            break
        S = np.sqrt(2.0 * float(n))
        a = normalize_nonneg(a_start, S)
        remaining = min(per_top_budget, exploit_deadline - time.monotonic() - 5)

        print("  Candidate %d: n=%d, starting obj=%.7f" %
              (idx, n, evaluate_sequence(a.tolist())))

        # Round 1: Adaptive subgradient (40% of budget)
        r1_time = 0.40 * remaining
        a_best, _ = adaptive_subgradient(
            a, n, S, r1_time,
            T_init=20, T_final=500, lr_init=0.06,
            use_average=True, top_k=10
        )
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("    R1: obj=%.7f" % obj)

        # Round 2: Peak-targeted CD (30% of budget)
        r2_time = 0.30 * remaining
        a_best, _ = peak_targeted_cd(
            a, n, S, r2_time
        )
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("    R2: obj=%.7f" % obj)

        # Round 3: More subgradient with different params (20% of budget)
        r3_time = 0.20 * remaining
        a_best, _ = adaptive_subgradient(
            a, n, S, r3_time,
            T_init=100, T_final=3000, lr_init=0.03,
            use_average=True, top_k=5
        )
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("    R3: obj=%.7f" % obj)

        # Round 4: Peak-targeted CD again (10% of budget)
        r4_time = 0.10 * remaining
        a_best, _ = peak_targeted_cd(
            a, n, S, r4_time
        )
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("    R4: obj=%.7f" % obj)

    # ============ PHASE 3: Polish the best (17% time) ============
    polish_budget = 0.17 * total_budget
    polish_deadline = time.monotonic() + polish_budget

    if global_a is not None and polish_budget > 20:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a = normalize_nonneg(global_a, S)

        print("\nPhase 3: Polish (budget=%.0fs, n=%d)" % (polish_budget, n))

        # Fine two-point refinement
        p1_time = 0.40 * polish_budget
        a_best, _ = two_point_refine(a, n, S, p1_time)
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  Two-point refine: obj=%.7f" % obj)

        # Peak-targeted CD
        p2_time = 0.35 * polish_budget
        a_best, _ = peak_targeted_cd(a, n, S, p2_time)
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  Peak CD: obj=%.7f" % obj)

        # Final subgradient with very high T
        p3_time = 0.25 * polish_budget
        a_best, _ = adaptive_subgradient(
            a, n, S, p3_time,
            T_init=500, T_final=5000, lr_init=0.01,
            use_average=True, top_k=3
        )
        a = normalize_nonneg(a_best, S)
        obj = evaluate_sequence(a.tolist())
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a.copy()
            print("  HighT SG: obj=%.7f" % obj)

    # ============ PHASE 4: Random restarts and final push (10% time) ============
    restart_budget = deadline - time.monotonic()
    if global_a is not None and restart_budget > 20:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        best_a = global_a.copy()

        print("\nPhase 4: Restarts (budget=%.0fs)" % restart_budget)
        n_rounds = max(2, int(restart_budget / 40))
        per_round = restart_budget / (n_rounds + 1)

        for r in range(n_rounds):
            if time.monotonic() > deadline - 10:
                break

            # Different perturbation strategies
            strategy = r % 3
            if strategy == 0:
                # Gaussian noise
                noise_scale = 0.01 * (1 + r * 0.2)
                a_noisy = best_a + noise_scale * rng.standard_normal(n)
            elif strategy == 1:
                # Sparse perturbation (change a few coordinates)
                a_noisy = best_a.copy()
                n_pert = rng.integers(5, 30)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                for idx in pert_idx:
                    a_noisy[idx] *= rng.random() * 1.5 + 0.25
            else:
                # Peak-targeted perturbation
                a_noisy = best_a.copy()
                c_check = fft_convolve_self(best_a)
                k_peak = np.argmax(c_check)
                # Perturb coordinates near the peak contributors
                valid_start = max(0, k_peak - n + 1)
                valid_end = min(k_peak, n - 1)
                if valid_start <= valid_end:
                    center = (valid_start + valid_end) // 2
                    n_pert = min(20, n // 10)
                    pert_idx = np.random.default_rng(r * 31).choice(
                        n, size=n_pert, replace=False
                    )
                    for idx in pert_idx:
                        a_noisy[idx] *= rng.random() * 1.3 + 0.35

            a_noisy = normalize_nonneg(a_noisy, S)

            a_best, _ = adaptive_subgradient(
                a_noisy, n, S, per_round,
                T_init=50, T_final=1000, lr_init=0.04,
                use_average=True, top_k=8
            )
            a_best = normalize_nonneg(a_best, S)
            # Quick peak CD
            a_best, _ = peak_targeted_cd(a_best, n, S, min(10, per_round * 0.2))
            a_best = normalize_nonneg(a_best, S)

            obj = evaluate_sequence(a_best.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_best.copy()
                best_a = a_best.copy()
                print("  Restart %d (str=%d): obj=%.7f" % (r, strategy, obj))

    # Final verification
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f (verified=%s)" % (
        global_best, "%.7f" % ver if np.isfinite(ver) else "inf"))

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)

    return final_a
