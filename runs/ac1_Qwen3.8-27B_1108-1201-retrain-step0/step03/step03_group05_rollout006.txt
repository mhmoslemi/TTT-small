"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses L-BFGS continuation with log-sum-exp smoothing, top-k subgradient polishing,
and multi-restart strategy. Key improvements: higher beta ceiling, adaptive
top-k subgradient with line search, structured perturbations, and block coordinate descent.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_from_weights(a, w):
    """Compute grad[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation via FFT)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z (softmax params)."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_grad_from_weights(a, w)

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=200):
    """L-BFGS-B with continuation on beta."""
    if beta_schedule is None:
        beta_schedule = [3, 5, 10, 20, 50, 100, 200, 400, 800, 1500, 3000, 6000, 10000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break

        n_attempts = 1
        if beta >= 500 and max_time > 30:
            n_attempts = 2
        if beta >= 3000 and max_time > 60:
            n_attempts = 3

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 3:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.01 / (1 + attempt * 0.5), n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            t0 = time.monotonic()
            try:
                result = scipy_minimize(
                    fun, z,
                    jac=jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_beta,
                        'maxfun': maxiter_per_beta * 3,
                        'ftol': 1e-16,
                        'gtol': 1e-10,
                        'maxls': 50,
                        'iprint': -1
                    }
                )
            except Exception:
                break

            z = result.x
            z = z - z.mean()

            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def run_toppk_subgradient(a_init, n, S, max_time, rng,
                          top_k_init=3, top_k_max=128, decay=0.999):
    """
    Top-k subgradient method with Nesterov acceleration and line search.
    Focuses gradient on the top-k largest convolution values.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step_scale = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 40)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        # Soft weighting: emphasize the top values
        top_vals_norm = top_vals - top_vals.min() + 1e-10
        weights = top_vals_norm ** 2
        weights /= weights.sum()

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        grad = compute_grad_from_weights(a, w_full)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = step_scale * S / (n ** 0.5 * (1.0 + 0.001 * it))

        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        # Simple line search: try a few step multipliers
        best_x = x_new
        best_obj = fft_convolve_self(x_new).max()
        for mult in [0.5, 2.0]:
            x_try = project_simplex(y - step * mult * grad / max(gnorm, 1e-15), S)
            obj_try = fft_convolve_self(x_try).max()
            if obj_try < best_obj:
                best_obj = obj_try
                best_x = x_try

        if best_obj < best_score:
            best_score = best_obj
            best_local = best_x.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = best_x + ((t - 1.0) / t_new) * (best_x - a)
        y = np.maximum(y, 0.0)
        a = best_x
        t = t_new

        # Adaptive step scaling
        if best_obj < cur * 0.999:
            step_scale = min(step_scale * 1.1, 5.0)
        elif best_obj > cur * 1.001:
            step_scale = max(step_scale * decay, 0.1)

        # Occasional restart to escape stagnation
        if it % 8000 == 0 and rng.random() < 0.15:
            noise = rng.normal(0, 0.01 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            a = y.copy()
            t = 1.0

    return best_local, best_score


def run_block_descent(a_init, n, S, max_time, rng, block_size=64):
    """Block coordinate descent: optimize blocks of coordinates at a time."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()

    n_blocks = n // block_size
    remaining = n % block_size

    while time.monotonic() - start < max_time:
        # Shuffle block order
        block_order = rng.permutation(n_blocks)
        if remaining > 0:
            # Add the last partial block
            block_order = np.append(block_order, n_blocks)

        for bi in block_order:
            if time.monotonic() - start > max_time - 2:
                break

            if bi < n_blocks:
                start_idx = bi * block_size
                end_idx = start_idx + block_size
            else:
                start_idx = n_blocks * block_size
                end_idx = n

            # Optimize this block using a few L-BFGS iterations
            # Fix all other coordinates, optimize the block
            a_fixed = a.copy()
            S_block_target = a[start_idx:end_idx].sum()

            def block_obj(block_vals):
                a_test = a_fixed.copy()
                a_test[start_idx:end_idx] = np.maximum(block_vals, 0)
                # Rescale only the block to maintain total sum
                old_block_sum = a_fixed[start_idx:end_idx].sum()
                if old_block_sum < 1e-15:
                    a_test[start_idx:end_idx] = np.maximum(block_vals, 0)
                    a_test[start_idx:end_idx] = a_test[start_idx:end_idx] - a_test[start_idx:end_idx].min()
                    if a_test[start_idx:end_idx].max() < 1e-15:
                        a_test[start_idx:end_idx] = S_block_target / len(block_vals)
                    else:
                        a_test[start_idx:end_idx] = S_block_target * a_test[start_idx:end_idx] / a_test[start_idx:end_idx].sum()
                else:
                    a_test[start_idx:end_idx] = S_block_target * block_vals / max(block_vals.sum(), 1e-15)
                c = fft_convolve_self(a_test)
                return c.max()

            current_block = a[start_idx:end_idx].copy()
            try:
                result = scipy_minimize(
                    block_obj, current_block,
                    method='Nelder-Mead',
                    options={'maxiter': 100, 'xatol': 1e-10, 'fatol': 1e-10}
                )
                new_block = np.maximum(result.x, 0)
                new_block_sum = new_block.sum()
                if new_block_sum > 1e-15 and S_block_target > 1e-15:
                    a[start_idx:end_idx] = S_block_target * new_block / new_block_sum
                else:
                    a[start_idx:end_idx] = S_block_target / len(new_block)
            except Exception:
                pass

            c = fft_convolve_self(a)
            score = c.max()
            if score < best_score:
                best_score = score
                best_local = a.copy()

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # --- Get initial sequence ---
    inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("given", np.clip(given, 0.0, 1000.0)))
    except Exception:
        pass

    if inits:
        main_len = len(inits[0][1])
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * float(n))
    tt = np.linspace(0, 1, n, endpoint=False)

    # Diverse initializations
    inits.append(("rand", rng.random(n) + 0.1))
    # Multi-frequency with optimized structure
    inits.append(("sin_multi",
                  1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) +
                  0.25 * np.cos(2 * np.pi * 5 * tt) +
                  0.2 * np.sin(2 * np.pi * 11 * tt) +
                  0.15 * np.cos(2 * np.pi * 17 * tt) +
                  0.1 * np.sin(2 * np.pi * 23 * tt) +
                  0.08 * np.cos(2 * np.pi * 31 * tt)))
    # Edge-biased pattern (exploits asymmetry to reduce center peak)
    inits.append(("edge_bias",
                  0.5 + 0.4 * np.cos(np.pi * tt) +
                  0.2 * np.cos(2 * np.pi * tt) +
                  0.1 * np.cos(4 * np.pi * tt) + 0.05))
    # Two-peak structure
    inits.append(("two_peak",
                  np.exp(-20.0 * ((tt - 0.15) / 0.3) ** 2) +
                  np.exp(-20.0 * ((tt - 0.65) / 0.3) ** 2) + 0.05))
    # Sawtooth-like (creates destructive interference in convolution)
    inits.append(("sawtooth",
                  0.5 + 0.3 * np.abs(2 * (tt * 7 - np.floor(tt * 7 + 0.5))) - 0.15 + 0.1))

    n_inits = len(inits)
    print("=== Optimization: n=%d, %d inits, budget=%.0fs ===" % (n, n_inits, budget_s))
    print("Target: 1.5027, Previous best: 1.508805")

    # Evaluate initial scores
    for name, arr in inits:
        a_norm = safe_normalize(arr, n)
        c = fft_convolve_self(a_norm)
        score = c.max()
        if score < global_best:
            global_best = score
            global_a = a_norm.copy()
        print("  Init '%s': score=%.6f" % (name, score))

    print("\nInitial best: %.8f" % global_best)

    # --- Phase 1: L-BFGS continuation on each init ---
    usable = deadline - time.monotonic()
    # Give more time to the best init
    weights = [2.0 if i == 0 else 1.0 for i in range(n_inits)]
    total_w = sum(weights)
    per_init_times = [usable * 0.5 * w / total_w for w in weights]
    for i in range(n_inits):
        per_init_times[i] = max(per_init_times[i], 8.0)

    print("\n=== Phase 1: L-BFGS Continuation ===")
    phase1_results = []

    for idx, (name, arr) in enumerate(inits):
        if time.monotonic() > deadline - 120:
            break

        t0 = time.monotonic()
        a_norm = safe_normalize(arr, n)

        a_new, score = run_lbfgs_continuation(
            a_norm, n, S, per_init_times[idx], rng,
            beta_schedule=[3, 5, 10, 20, 50, 100, 200, 400, 800, 2000, 5000, 10000],
            maxiter_per_beta=250
        )
        a_new = safe_normalize(a_new, n)
        c_new = fft_convolve_self(a_new)
        final_score = c_new.max()

        phase1_results.append((name, a_new, final_score))

        if final_score < global_best:
            global_best = final_score
            global_a = a_new.copy()
            print("  [NEW BEST] %s: %.8f (%.1fs)" % (name, global_best, time.monotonic() - t0))
        else:
            print("  %s: %.8f (best=%.6f, %.1fs)" % (name, final_score, global_best, time.monotonic() - t0))

    print("\nAfter Phase 1: global_best=%.8f" % global_best)

    # --- Phase 2: Top-k subgradient polishing on best result ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            seg2 = min(remaining * 0.35, 300)
            print("\n=== Phase 2: Top-k Subgradient (%.1f sec) ===" % seg2)

            a_polish, score = run_toppk_subgradient(
                global_a, n, S, seg2, rng,
                top_k_init=3, top_k_max=64, decay=0.9995
            )
            a_polish = safe_normalize(a_polish, n)
            c = fft_convolve_self(a_polish)
            if c.max() < global_best:
                global_best = c.max()
                global_a = a_polish.copy()
                print("  TopK polish: %.8f" % global_best)
            else:
                print("  TopK polish: %.8f (no improvement)" % c.max())

    # --- Phase 3: Multi-restart with diverse perturbations ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_perturb = max(4, int(remaining / 30))
            perturb_budget = remaining * 0.4 / n_perturb
            print("\n=== Phase 3: Multi-restart (%d restarts, %.1f sec each) ===" %
                  (n_perturb, perturb_budget))

            for p in range(n_perturb):
                if time.monotonic() > deadline - 60:
                    break

                # Different perturbation strategies
                strategy = p % 4
                if strategy == 0:
                    # Gaussian noise
                    noise_scale = 0.05 / (1.0 + 0.2 * p)
                    a_pert = global_a + rng.normal(0, noise_scale * S / np.sqrt(n), n)
                elif strategy == 1:
                    # Block perturbation: move mass between blocks
                    n_blocks = 8
                    block_size = n // n_blocks
                    a_pert = global_a.copy()
                    for b in range(n_blocks - 1):
                        donor = slice(b * block_size, (b + 1) * block_size)
                        receiver = slice((b + 1) * block_size, (b + 2) * block_size)
                        amount = a_pert[donor].sum() * rng.uniform(0.05, 0.2)
                        a_pert[donor] -= amount / len(donor)
                        a_pert[receiver] += amount / len(receiver)
                elif strategy == 2:
                    # Frequency-domain perturbation
                    A = fft(global_a)
                    phase_noise = rng.normal(0, 0.1, A.size)
                    A_pert = A * np.exp(1j * phase_noise)
                    a_pert = np.real(ifft(A_pert)[:n])
                    a_pert = np.maximum(a_pert, 0)
                else:
                    # Spike perturbation: add/remove spikes at random positions
                    a_pert = global_a.copy()
                    n_spikes = n // 256
                    spike_positions = rng.choice(n, n_spikes, replace=False)
                    for sp in spike_positions:
                        a_pert[sp] += rng.normal(0, 0.1 * S / np.sqrt(n))
                    a_pert = np.maximum(a_pert, 0)

                a_pert = safe_normalize(a_pert, n)

                a_new, score = run_lbfgs_continuation(
                    a_pert, n, S, perturb_budget, rng,
                    beta_schedule=[50, 200, 1000, 5000],
                    maxiter_per_beta=150
                )
                a_new = safe_normalize(a_new, n)
                c_new = fft_convolve_self(a_new)
                new_score = c_new.max()

                if new_score < global_best:
                    global_best = new_score
                    global_a = a_new.copy()
                    print("  Restart %d (strat=%d): %.8f *** NEW BEST ***" % (p, strategy, global_best))
                else:
                    print("  Restart %d (strat=%d): %.8f" % (p, strategy, new_score))

    # --- Phase 4: Fine-tuning with high-beta L-BFGS + subgradient ---
    if global_a is not None:
        remaining = deadline - time.monotonic()
        print("\n=== Phase 4: Final Finishing (%.1f sec) ===" % remaining)

        if remaining > 20:
            # Very high beta L-BFGS
            seg4a = min(remaining * 0.5, 120)
            a_final, score = run_lbfgs_continuation(
                global_a, n, S, seg4a, rng,
                beta_schedule=[500, 2000, 5000, 10000, 20000],
                maxiter_per_beta=300
            )
            a_final = safe_normalize(a_final, n)
            c_final = fft_convolve_self(a_final)
            if c_final.max() < global_best:
                global_best = c_final.max()
                global_a = a_final.copy()
                print("  Final L-BFGS: %.8f *** NEW BEST ***" % global_best)
            else:
                print("  Final L-BFGS: %.8f" % c_final.max())

            # Subgradient polish
            remaining2 = deadline - time.monotonic()
            if remaining2 > 15:
                seg4b = min(remaining2 * 0.7, 100)
                a_polish, score = run_toppk_subgradient(
                    global_a, n, S, seg4b, rng,
                    top_k_init=1, top_k_max=16, decay=0.9999
                )
                a_polish = safe_normalize(a_polish, n)
                c = fft_convolve_self(a_polish)
                if c.max() < global_best:
                    global_best = c.max()
                    global_a = a_polish.copy()
                    print("  Final subgradient: %.8f *** NEW BEST ***" % global_best)

    # --- Phase 5: Verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        print("\nVerification: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
