        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Temperature schedule: exponential increase with periodic reheat
        base_T = T_init * (T_final / T_init) ** (progress ** 0.6)
        if use_reheat and (step % 200 == 0) and step > 0:
            T = base_T * 0.05  # sudden cool to explore
        else:
            T = base_T
        
        # Compute gradient
        g = compute_smooth_grad(c, a, T)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adam update
        step += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        
        # Learning rate: high early, decays
        lr = lr_init * (1.0 + 0.3 * np.cos(np.pi * progress / 2.0))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = normalize_nonneg(a_new, S)
        
        # Evaluate
        c_new = fft_convolve_self(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        c = c_new
        
        # Reheat with noise if stuck
        if no_improve > max_no_improve:
            noise_scale = 0.005 * (0.5 + 0.5 * (1.0 - progress))
            a = best_a + noise_scale * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            m = np.zeros(n)
            v = np.zeros(n)
            step = 0
            no_improve = 0
    
    return best_a, best_obj


def run_adam_log(a_init, n, S, time_budget, T_init=10.0, T_final=2000.0,
                 lr_init=0.05):
    """Adam optimization in log-space (ensures strict positivity)."""
    a = normalize_nonneg(a_init, S)
    a = np.maximum(a, 1e-12)
    a = a * (S / a.sum())
    theta = np.log(a) - np.log(S)  # log of proportions (sum theta constraints handled by softmax)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0
    no_improve = 0
    max_no_improve = 400
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(77)
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.5)
        
        # Convert theta to a via softmax
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        a = S * exp_theta / total
        
        # Compute gradient in a-space
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Gradient wrt a
        g_a = compute_smooth_grad(c, a, T)
        
        # Chain rule: dL/dtheta[j] = g_a[j]*a[j] - (a[j]/S)*sum(g_a*a)
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot
        
        # Adam update
        step += 1
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta * g_theta)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        
        lr = lr_init * (0.5 + 0.5 * (1.0 - progress))
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Restart if stuck
        if no_improve > max_no_improve:
            noise = 0.01 * (1.0 - progress * 0.5)
            theta = theta + noise * rng.standard_normal(n)
            m = np.zeros(n)
            v = np.zeros(n)
            step = 0
            no_improve = 0
    
    return best_a, best_obj


def coordinate_descent(a_init, n, S, time_budget, max_sweeps=200):
    """Fast coordinate descent: try adjusting individual coordinates with subgradient guidance."""
    a = normalize_nonneg(a_init.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    stagnation = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            a_i = a[i]
            if a_i < 1e-14:
                continue
            
            best_factor = 1.0
            best_local = best_obj
            
            # Try a set of multiplicative factors (biased toward reduction)
            factors = [0.8, 0.9, 0.95, 0.7, 0.98, 0.5, 0.85]
            for factor in factors:
                saved = a[i]
                a[i] = a_i * factor
                a_norm = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                a[i] = saved
                if obj_check < best_local - 1e-14:
                    best_local = obj_check
                    best_factor = factor
            
            if best_factor != 1.0:
                a[i] = a_i * best_factor
                a = normalize_nonneg(a, S)
                c = fft_convolve_self(a)
                if c.max() < best_obj - 1e-13:
                    best_obj = c.max()
                    best_a = a.copy()
                    sweep_improved = True
        
        if sweep_improved:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation >= 3:
                break
    
    return best_a, best_obj


def structured_perturbation(a, n, S, rng, scale=0.02):
    """Apply a structured (low-frequency) perturbation to escape local minima."""
    # Generate smooth random perturbation via random Fourier features
    n_freq = 8
    freqs = rng.integers(1, 20, size=n_freq) * 2 * np.pi / n
    phases = rng.uniform(0, 2 * np.pi, size=n_freq)
    amps = rng.uniform(0.5, 1.5, size=n_freq)
    
    i = np.arange(n)
    pert = np.zeros(n)
    for f, ph, amp in zip(freqs, phases, amps):
        pert += amp * np.cos(f * i + ph)
    
    pert = scale * pert / (np.abs(pert).max() + 1e-10)
    a_new = a + pert
    return normalize_nonneg(a_new, S)


def get_initialization(n, S, rng, mode="auto"):
    """Generate diverse initializations."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    if mode == "auto":
        mode = ["raised_cos", "gaussian", "two_cos", "flat"][rng.integers(4)]
    
    if mode == "raised_cos":
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / (n - 1)))
    elif mode == "gaussian":
        sigma = n / 5.0
        a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    elif mode == "two_cos":
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / (n - 1)))
        a *= 0.5 * (1.0 + np.cos(4.0 * np.pi * i / (n - 1) + rng.uniform(0, 1)))
    elif mode == "flat":
        a = np.ones(n)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 1e-6)
    return normalize_nonneg(a, S)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search for non-negative sequence minimizing the evaluation function.
    
    Phase 1 (15%): Quick exploration with diverse initializations
    Phase 2 (40%): Main Adam optimization (direct space, aggressive T)
    Phase 3 (25%): Log-space Adam for fine-tuning
    Phase 4 (20%): Coordinate descent polish + final restarts
    """
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s) - 30.0)
    
    n = 4096
    S = np.sqrt(2.0 * n)
    
    global_best = np.inf
    global_a = None
    
    # Load given sequence
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size == n and given_raw.sum() > 0.01:
            given_a = normalize_nonneg(given_raw, S)
            ver = evaluate_sequence(given_a.tolist())
            if np.isfinite(ver):
                global_best = ver
                global_a = given_a.copy()
                print("Given sequence: obj=%.7f" % ver)
    except Exception as e:
        print("Could not load given sequence: %s" % str(e))
    
    if global_a is None:
        global_a = get_initialization(n, S, rng, mode="raised_cos")
        global_best = evaluate_sequence(global_a.tolist())
        print("Initial: obj=%.7f" % global_best)
    
    deadline = time.monotonic() + total_budget
    
    # ================================================================
    # Phase 1: Quick exploration (15% time)
    # Try a few different initializations with short Adam runs
    # ================================================================
    p1_budget = 0.15 * total_budget
    p1_deadline = time.monotonic() + p1_budget
    print("\n=== Phase 1: Exploration (budget=%.0fs) ===" % p1_budget)
    
    init_modes = ["raised_cos", "gaussian", "two_cos", "flat"]
    n_inits = min(len(init_modes), max(2, int(p1_budget / 15)))
    per_init_time = p1_budget / n_inits - 1.0
    
    for idx in range(n_inits):
        if time.monotonic() > p1_deadline - 5:
            break
        
        mode = init_modes[idx % len(init_modes)]
        a_init = get_initialization(n, S, rng, mode=mode)
        
        a_res, obj_res = run_adam_direct(
            a_init, n, S, per_init_time,
            T_init=10.0, T_final=500.0, lr_init=0.1, use_reheat=False
        )
        a_res = normalize_nonneg(a_res, S)
        ver = evaluate_sequence(a_res.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_res.copy()
            print("  Explore %s: obj=%.7f" % (mode, global_best))
    
    # ================================================================
    # Phase 2: Main direct-space Adam (40% time)
    # Aggressive temperature, periodic reheat
    # ================================================================
    p2_budget = 0.40 * total_budget
    p2_deadline = time.monotonic() + p2_budget
    print("\n=== Phase 2: Main Adam Direct (budget=%.0fs) ===" % p2_budget)
    
    # Split into two sub-phases with different T ranges
    sub1_budget = 0.6 * p2_budget
    sub2_budget = 0.4 * p2_budget
    
    # Sub-phase 1: Moderate T
    a_res, _ = run_adam_direct(
        global_a, n, S, sub1_budget,
        T_init=30.0, T_final=2000.0, lr_init=0.08, use_reheat=True
    )
    a_res = normalize_nonneg(a_res, S)
    ver = evaluate_sequence(a_res.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_res.copy()
        print("  P2a: obj=%.7f" % global_best)
    
    # Sub-phase 2: High T (focus on the max peak)
    if time.monotonic() < p2_deadline - 10:
        a_res, _ = run_adam_direct(
            global_a, n, S, sub2_budget,
            T_init=200.0, T_final=10000.0, lr_init=0.04, use_reheat=True
        )
        a_res = normalize_nonneg(a_res, S)
        ver = evaluate_sequence(a_res.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_res.copy()
            print("  P2b: obj=%.7f" % global_best)
    
    # ================================================================
    # Phase 3: Log-space Adam (25% time)
    # Ensures strict positivity, different optimization landscape
    # ================================================================
    p3_budget = 0.25 * total_budget
    p3_deadline = time.monotonic() + p3_budget
    print("\n=== Phase 3: Log-space Adam (budget=%.0fs) ===" % p3_budget)
    
    a_res, _ = run_adam_log(
        global_a, n, S, p3_budget,
        T_init=50.0, T_final=5000.0, lr_init=0.03
    )
    a_res = normalize_nonneg(a_res, S)
    ver = evaluate_sequence(a_res.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_res.copy()
        print("  P3: obj=%.7f" % global_best)
    
    # ================================================================
    # Phase 4: Coordinate descent polish + restarts (20% time)
    # ================================================================
    p4_budget = deadline - time.monotonic()
    print("\n=== Phase 4: Polish (budget=%.0fs) ===" % p4_budget)
    
    if p4_budget > 10:
        # CD Round 1
        cd_budget1 = 0.4 * p4_budget
        a_res, _ = coordinate_descent(global_a, n, S, time.monotonic() + cd_budget1, max_sweeps=100)
        a_res = normalize_nonneg(a_res, S)
        ver = evaluate_sequence(a_res.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_res.copy()
            print("  CD1: obj=%.7f" % global_best)
        
        # Try structured restart
        remaining = deadline - time.monotonic() - 5
        if remaining > 30:
            n_restarts = max(2, int(remaining / 40))
            per_restart = remaining / (n_restarts + 1)
            
            for r in range(n_restarts):
                if time.monotonic() > deadline - 10:
                    break
                
                a_pert = structured_perturbation(global_a, n, S, rng, scale=0.01 + 0.005 * r)
                a_res, _ = run_adam_direct(
                    a_pert, n, S, per_restart,
                    T_init=100.0, T_final=5000.0, lr_init=0.05, use_reheat=False
                )
                a_res = normalize_nonneg(a_res, S)
                ver = evaluate_sequence(a_res.tolist())
                if np.isfinite(ver) and ver < global_best:
                    global_best = ver
                    global_a = a_res.copy()
                    print("  Restart %d: obj=%.7f" % (r, global_best))
            
            # Final CD
            final_cd_time = deadline - time.monotonic() - 5
            if final_cd_time > 10:
                a_res, _ = coordinate_descent(global_a, n, S, time.monotonic() + final_cd_time, max_sweeps=50)
                a_res = normalize_nonneg(a_res, S)
                ver = evaluate_sequence(a_res.tolist())
                if np.isfinite(ver) and ver < global_best:
                    global_best = ver
                    global_a = a_res.copy()
                    print("  Final CD: obj=%.7f" % global_best)
    
    # Final validation
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_list = global_a.tolist()
    ver = evaluate_sequence(final_list)
    print("\nFINAL: obj=%.7f" % (ver if np.isfinite(ver) else global_best))
    
    if not np.isfinite(ver):
        final_list = [1.0] * 256
        ver = evaluate_sequence(final_list)
        print("Fallback: obj=%.7f" % ver)
    
    return final_list
