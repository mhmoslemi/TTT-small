This is a first-order approximation. We could use:
P_new = P * exp(-alpha * cos_eff)  (exponential, always positive)
or
P_new = max(P - alpha * P * cos_eff, P_min)  (with a floor)

The exponential form is more stable and always keeps P positive.

**Another idea: "Flat spectrum" target**
Instead of trying to reduce the peak directly, we can define a target spectrum T(f) and minimize |||FFT(a)|^2 - T(f)||^2 subject to a >= 0, sum(a) = S. This is a convex-ish problem (the constraint a >= 0 makes it non-convex in the spectral domain, but we can iterate).

The optimal T(f) for minimizing peak autocorrelation would be the one that minimizes the peak of IFFT(T). This is a fixed-point problem: T* = argmin_f peak(IFFT(T)) subject to IFFT(sqrt(T) * exp(j*phase)) >= 0.

This is essentially what the spectral shaping does, but we could make it more explicit.

"""
Minimize 2*n*max_conv/sum^2 using adaptive LSE-FISTA with spectral preconditioning and entropy regularization.

Key improvements over previous approach:
1. Exponential spectral shaping (always positive, more stable)
2. Adaptive two-scale LSE: alternating between peak-focused and uniformly-focused gradients
3. Entropy regularization in early phase to prevent premature concentration
4. Geometric-weighted top-k gradient (sharper focus on peak lags)
5. Simulated-annealing style restarts with adaptive temperature
6. Backtracking step size for FISTA
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
        theta = 0.0
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_toppk(a, c, top_k, beta=1.0, geometric_power=1.0):
    """
    Gradient of LSE_beta approximation to max(c), using top-k active lags.
    Uses geometric weighting to emphasize the peak.
    """
    n = len(a)
    if top_k > len(c):
        top_k = len(c)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    
    # Softmax weights with geometric emphasis
    max_val = top_vals[0]
    if max_val <= 1e-30:
        return np.zeros(n)
    # Normalize and apply geometric power for sharper peak focus
    normalized = np.maximum(top_vals, 0.0) / max_val
    weights = normalized ** geometric_power
    total_w = weights.sum()
    if total_w < 1e-30:
        return np.zeros(n)
    weights = weights / total_w
    
    # Build full weight vector and compute gradient via FFT
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def spectral_shaping_step(a, N, top_k, alpha, geometric_power=1.0):
    """
    One spectral shaping step using exponential update.
    P_new = P * exp(-alpha * cos_eff), which is always positive.
    """
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()
    
    if c_max < 1e-30:
        return a, c, c_max
    
    m = np.arange(N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    max_val = top_vals[0]
    if max_val < 1e-30:
        return a, c, c_max
    
    normalized = np.maximum(top_vals, 0.0) / max_val
    weights = normalized ** geometric_power
    total_w = weights.sum()
    if total_w < 1e-30:
        return a, c, c_max
    weights = weights / total_w
    
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
    
    # Exponential update - always positive
    P_new = P * np.exp(-alpha * cos_eff)
    
    # Preserve phase
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new_full = np.real(ifft(A_new))[:n]
    
    # Project to simplex
    a_new = project_simplex(a_new_full, S)
    return a_new, c, c_max


def lse_fista_optimize(a_init, n, S, max_time, rng, beta_start=20.0, beta_end=500.0,
                       top_k=20, geometric_power=2.0, use_entropy=False, entropy_weight=0.01):
    """
    FISTA with LSE approximation and adaptive parameters.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    N = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: start moderate, increase over time
        frac = min(1.0, it / (max_time * 100))
        beta = beta_start + (beta_end - beta_start) * frac
        # Adaptive top_k
        cur_top_k = min(200, top_k + it // 20)
        # Adaptive geometric power: increase to focus more on peak over time
        cur_geo = geometric_power + 0.5 * frac
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        if cur < 1e-30:
            break
        
        grad = compute_grad_toppk(y, c, cur_top_k, beta=beta, geometric_power=cur_geo)
        
        if use_entropy and it < max_time * 20:
            # Add entropy regularization gradient
            pos = y > 1e-10
            ent_grad = np.zeros(n)
            ent_grad[pos] = -np.log(y[pos] + 1e-10)
            grad += entropy_weight * ent_grad / n
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: 1/(2*beta) is the theoretical Lipschitz-based step
        # Use a conservative fraction
        step = 0.9 / (beta * 2.0)
        x_new = project_simplex(y - step * grad, S)
        
        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        momentum = (t_mom - 1.0) / t_new
        y = x_new + momentum * (x_new - a)
        y = np.maximum(y, 0.0)
        # Renormalize to prevent drift
        ys = y.sum()
        if ys > 0:
            y = y * (S / ys)
        a = x_new
        t_mom = t_new
        
        # Periodic kick to escape local minima
        if it % 5000 == 2500 and rng.random() < 0.3:
            noise_scale = 0.005 * S / np.sqrt(n) * max(0.1, 1.0 - frac)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_mom = 1.0
    
    return best_local, best_score


def spectral_phase(a_init, N, max_time, rng, alpha_start=0.5, alpha_end=0.05,
                   top_k_start=5, top_k_end=50, target=1.45):
    """
    Spectral shaping phase with adaptive parameters.
    Returns best sequence found.
    """
    n = len(a_init)
    S = a_init.sum()
    a = a_init.copy()
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / max(1, int(max_time / 0.003)))
        
        # Adaptive alpha: decrease over time
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        # Adaptive top_k: increase over time
        cur_top_k = min(top_k_end, top_k_start + int(frac * (top_k_end - top_k_start)))
        
        a, c, c_max = spectral_shaping_step(a, N, cur_top_k, alpha, geometric_power=2.0)
        
        # Compute actual score
        score = c_max  # We track max_conv; will compute full score at the end
        
        # Use actual convolution max for tracking
        if c_max < best_score:
            best_score = c_max
            best_local = a.copy()
        
        # Early stop if we're well below target
        if c_max <= target and frac > 0.5:
            break
        
        # Occasional noise injection
        if it % 300 == 150 and rng.random() < 0.25:
            noise_scale = 0.003 * S / np.sqrt(n) * (1.0 - frac * 0.7)
            a = project_simplex(a + rng.normal(0, noise_scale, n), S)
    
    return best_local, best_score


def get_score(a):
    """Compute the actual score for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s ** 2)


def safe_normalize(a, n):
    """Clamp to [0,1000], ensure sum > 0."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * np.sqrt(2.0 * n) / n
    return a

def multi_segment_init(n, k, rng, n_trials=100):
    """Create optimized multi-segment initialization."""
    seg_len = n // k
    remainder = n % k
    boundaries = [0]
    for i in range(k):
        boundaries.append(boundaries[-1] + seg_len + (1 if i < remainder else 0))
    
    best_score = np.inf
    best_a = None
    
    for _ in range(n_trials):
        # Random heights with some structure
        heights = rng.random(k) + 0.1
        # Sometimes try alternating high/low
        if rng.random() < 0.3:
            heights = np.where(np.arange(k) % 2 == 0, rng.random(k) + 0.5, rng.random(k) * 0.3 + 0.05)
        
        a = np.zeros(n)
        for i in range(k):
            a[boundaries[i]:boundaries[i + 1]] = heights[i]
        
        c = fft_convolve_self(a)
        score = c.max() / (a.sum() ** 2)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a if best_a is not None else np.ones(n)

        # Determine which method to use based on stall and progress
        elapsed_frac = (time.monotonic() - t_p2) / max(p2_budget, 1)
        
        # FISTA block
        fista_time = min(30.0 * (1.0 + elapsed_frac), p2_budget - (time.monotonic() - t_p2))
        if fista_time > 3:
            # Adaptive parameters based on current state
            beta_s = 50 + 200 * elapsed_frac
            beta_e = 200 + 800 * elapsed_frac
            top_k_cur = 15 + int(50 * elapsed_frac)
            geo_cur = 2.0 + 2.0 * elapsed_frac
            
            a_f, _ = lse_fista_optimize(a_curr, n, S, fista_time, rng,
                                        beta_start=beta_s, beta_end=beta_e,
                                        top_k=top_k_cur, geometric_power=geo_cur,
                                        use_entropy=False)
            if update_best(a_f):
                a_curr = global_a.copy()
                stall_count = 0
                if stall_count % 3 == 0 or last_best - global_best > 0.001:
                    print("  [P2] FISTA: %.6f" % global_best)
                    last_best = global_best
            else:
                stall_count += 1
        
        # Spectral kickoff between FISTA blocks
        if time.monotonic() < t_p2 + p2_budget - 5:
            ss_time = min(15.0, (t_p2 + p2_budget) - time.monotonic())
            if ss_time > 2:
                alpha_start = max(0.05, 0.3 * (1.0 - elapsed_frac))
                alpha_end = max(0.02, 0.05 * (1.0 - elapsed_frac))
                a_s, _ = spectral_phase(a_curr, N, ss_time, rng,
                                        alpha_start=alpha_start, alpha_end=alpha_end,
                                        top_k_start=10, top_k_end=40, target=global_best / (2.0 * n) * 0.9)
                if update_best(a_s):
                    a_curr = global_a.copy()
                    stall_count = 0
                    print("  [P2] Spectral: %.6f" % global_best)
                    last_best = global_best
                else:
                    stall_count += 1
        
        # Restart if stalled
        if stall_count >= 5:
            print("  [P2] Stalled at %.6f, restarting" % global_best)
            # Perturb and restart
            noise_scale = 0.02 * S / np.sqrt(n)
            a_curr = project_simplex(global_a + rng.normal(0, noise_scale, n), S)
            stall_count = 0
            # Brief FISTA to recover
            a_rec, _ = lse_fista_optimize(a_curr, n, S, 10.0, rng,
                                          beta_start=30, beta_end=100,
                                          top_k=10, geometric_power=2.0)
            update_best(a_rec)
            a_curr = global_a.copy() if global_a is not None else a_rec
    
    print("Phase 2 best: %.6f" % global_best)
    
    # --- Phase 3: Intensive polish ---
    print("\n=== Phase 3: Intensive Polish ===")
    t_p3 = time.monotonic()
    remaining = deadline - t_p3
    
    if global_a is not None and remaining > 10:
        # Multiple FISTA passes with increasing beta
        for pass_i in range(4):
            rem = deadline - time.monotonic()
            if rem < 5:
                break
            
            pass_time = min(rem * 0.3, 120.0)
            beta_s = 100 + 200 * pass_i
            beta_e = 500 + 500 * pass_i
            top_k_cur = 30 + 30 * pass_i
            geo_cur = 3.0 + pass_i
            
            a_p, _ = lse_fista_optimize(global_a, n, S, pass_time, rng,
                                        beta_start=beta_s, beta_end=beta_e,
                                        top_k=top_k_cur, geometric_power=geo_cur)
            if update_best(a_p):
                print("  [P3] Pass %d: %.6f" % (pass_i + 1, global_best))
            
            # Interleave with spectral
            rem2 = deadline - time.monotonic()
            if rem2 > 5:
                ss_time = min(rem2 * 0.15, 30.0)
                if ss_time > 2:
                    a_s, _ = spectral_phase(global_a, N, ss_time, rng,
                                            alpha_start=0.05, alpha_end=0.01,
                                            top_k_start=20, top_k_end=60,
                                            target=global_best / (2.0 * n) * 0.95)
                    update_best(a_s)
        
        # Final ultra-high beta FISTA
        rem3 = deadline - time.monotonic()
        if rem3 > 10:
            a_final, _ = lse_fista_optimize(global_a, n, S, min(rem3, 180.0), rng,
                                            beta_start=500, beta_end=2000,
                                            top_k=50, geometric_power=5.0)
            if update_best(a_final):
                print("  [P3] Final FISTA: %.6f" % global_best)
        
        # Final spectral micro-tuning
        rem4 = deadline - time.monotonic()
        if rem4 > 5:
            a_micro, _ = spectral_phase(global_a, N, min(rem4, 60.0), rng,
                                        alpha_start=0.02, alpha_end=0.005,
                                        top_k_start=30, top_k_end=80,
                                        target=global_best / (2.0 * n) * 0.98)
            if update_best(a_micro):
                print("  [P3] Micro spectral: %.6f" % global_best)
    
    # --- Verification and return ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("\nVerification: %.6f" % ver)
            print("Internal best: %.6f" % global_best)
        else:
            print("Verification failed, using fallback")
            final_list = [float(x) for x in np.ones(256, dtype=np.float64)]
    else:
        final_list = [float(x) for x in np.ones(256, dtype=np.float64)]
    
    print("Returning sequence of length %d" % len(final_list))
    return final_list
