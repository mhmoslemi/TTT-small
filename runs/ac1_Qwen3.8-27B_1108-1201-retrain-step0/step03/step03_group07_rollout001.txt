def softmax_obj(c, T):
    """Soft approximation to max."""
    c_shifted = c - c.max()
    return c.max() + np.log(np.sum(np.exp(T * c_shifted))) / T

def softmax_grad(c, a, T):
    """Gradient of soft-max objective."""
    n = len(a)
    c_shifted = c - c.max()
    w = np.exp(T * c_shifted)
    w /= w.sum()
    # grad = 2 * conv(w, a_reversed) at the right offset
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n-1:2*n-1]

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS with soft-max annealing,
combined with mirror descent and coordinate polish.
The key innovation is using a smooth approximation (soft-max) of the peak
objective, optimized with a quasi-Newton method in log-space, with an
annealing schedule that sharpens the approximation over time.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_conv_self(a)
    return 2.0 * n * c.max() / (S * S)


def softmax_obj_grad(b, S, T):
    """
    Compute soft-max objective and gradient in log-space.
    b is the log-parameter, a = S * softmax(b).
    """
    n = len(b)
    b = b - b.max()  # numerical stability
    a = np.exp(b)
    a_sum = a.sum()
    a = a * (S / a_sum)  # normalize to sum S
    
    c = fft_conv_self(a)
    c_max = c.max()
    c_shifted = c - c_max
    
    # Soft max value
    log_sum_exp = np.log(np.sum(np.exp(T * c_shifted)))
    obj_val = c_max + log_sum_exp / T
    
    # Weights for soft max
    w = np.exp(T * c_shifted)
    w /= w.sum()
    
    # Gradient of soft-max w.r.t. a
    # d(softmax)/d(c_k) = w_k
    # d(c_k)/d(a_m) = 2*a_{k-m} (for valid k-m)
    # So grad_a = 2 * sum_k w_k * a_{k-m} = 2 * (w * a_rev) convolved appropriately
    
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1 - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad_a = 2.0 * full_conv[n - 1:2 * n - 1]
    
    # Transform to b-space
    # a_i = S * exp(b_i) / sum(exp(b))
    # d(softmax_obj)/d(b_j) = sum_i (d obj/d a_i) * (d a_i / d b_j)
    # d a_i / d b_j = a_i * (delta_ij - a_j / S)
    # Sum_i grad_a_i * a_i * (delta_ij - a_j/S)
    # = a_j * (grad_a_j - sum_i grad_a_i * a_i / S)
    
    inner = np.dot(grad_a, a) / S
    grad_b = a * (grad_a - inner)
    
    # Remove flat direction (adding constant to b doesn't change a)
    grad_b = grad_b - grad_b.mean()
    
    return obj_val, grad_b


def lbfgs_anneal(a0, S, T_schedule, time_budget, maxiter_per_phase=500):
    """
    Run L-BFGS with annealed soft-max temperature.
    """
    n = len(a0)
    b = np.log(np.maximum(a0, 1e-300))
    b = b - b.mean()  # center
    
    best_a = a0.copy()
    best_obj = obj_value(a0)
    start = time.monotonic()
    
    for T in T_schedule:
        if time.monotonic() - start > time_budget:
            break
        
        phase_time = min(30, (time_budget - (time.monotonic() - start)) / max(1, len(T_schedule)))
        
        def fun_and_grad(b_vec):
            val, grad = softmax_obj_grad(b_vec, S, T)
            return val, grad
        
        result = minimize(
            fun_and_grad, b, method='L-BFGS-B',
            jac=True,
            options={'maxiter': maxiter_per_phase, 'ftol': 1e-12, 'gtol': 1e-8,
                     'maxls': 20}
        )
        
        b = result.x - result.x.mean()
        a_new = np.exp(b)
        a_new = a_new * (S / a_new.sum())
        a_new = np.maximum(a_new, 0.0)
        
        obj_new = obj_value(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        print(f"    T={T}: obj={obj_new:.6f}, best={best_obj:.6f}, "
              f"iters={result.nit}, status={result.status}, "
              f"t={time.monotonic()-start:.1f}s")
    
    return best_a, best_obj

def coordinate_polish(a0, S, time_budget):
    """
    Coordinate descent: try to reduce each element that contributes to peaks.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    c = fft_conv_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    start = time.monotonic()
    
    # Precompute: for each index i, which peaks does it contribute to?
    # c_k includes a_i * a_{k-i}, so a_i contributes to c_k for k in [i, i+n-1]
    
    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        improved = False
        c_max = c.max()
        
        # Process indices that contribute to the top peaks
        top_peaks = np.argsort(c)[::-1][:100]
        
        # Indices that appear in the top peak computations
        affected = set()
        for k in top_peaks:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                affected.add(i)
        
        # Try reducing high-contributing indices
        for i in sorted(affected, key=lambda x: a[x], reverse=True)[:200]:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-10:
                continue
            
            # Try a few factors
            best_local = best_obj
            best_factor = 1.0
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3]:
                a[i] = orig * factor
                s = a.sum()
                if s < 0.01:
                    a[i] = orig
                    break
                a_norm = a * (S / s)
                c_new = fft_conv_self(a_norm)
                obj_new = c_new.max()  # proportional to objective when S fixed
                if obj_new < best_local:
                    best_local = obj_new
                    best_factor = factor
            
            if best_factor != 1.0:
                a[i] = orig * best_factor
                a = a * (S / a.sum())
                c = fft_conv_self(a)
                obj_cur = obj_value(a, c)
                if obj_cur < best_obj:
                    best_obj = obj_cur
                    best_a = a.copy()
                    improved = True
        
        if not improved and sweep > 3:
            break
    
    return best_a, best_obj
