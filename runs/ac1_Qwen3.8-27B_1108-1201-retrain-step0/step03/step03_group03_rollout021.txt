"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy:
- Use log-sum-exp smooth approximation with geometric beta schedule
- Anti-palindromic and chirp-based structured initializations
- FISTA with adaptive backtracking and periodic structured perturbations
- Multi-phase optimization: explore -> refine -> polish
- Coordinate descent polish targeting peak-contributing pairs
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def lse_value_and_grad(a, c, beta):
    """Compute log-sum-exp value and gradient w.r.t. a.
    Returns (value, gradient)."""
    n = len(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    # Numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_bc = np.exp(shifted)
    total = exp_bc.sum()
    if total < 1e-300:
        return c_max, np.zeros(n)
    w = exp_bc / total
    value = (c_max + np.log(total)) * (1.0 / beta) * beta  # = c_max + log(sum exp)/beta * beta
    # Actually: (1/beta)*log(sum exp(beta*c)) = c_max + (1/beta)*log(sum exp(beta*(c-c_max)))
    value = c_max + np.log(total) / beta
    
    # Gradient: (1/beta) * (1/total) * sum_k exp(beta*c_k) * 2 * (Q_k @ a)
    # = 2 * ifft(FFT(w) * conj(FFT(a)))[:n] (with proper padding)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return value, grad


def run_fista_geometric(a_init, n, S, max_time, rng, beta_start=1.0, beta_end=500.0, 
                        perturbation_interval=200, perturbation_prob=0.25):
    """FISTA with geometric beta schedule and structured perturbations."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    L = beta_start * 4.0  # Initial Lipschitz estimate
    stall_count = 0

    # Precompute chirp perturbation base
    tt = np.arange(n, dtype=np.float64)
    chirp = np.cos(2.0 * np.pi * tt * tt / n)

    while time.monotonic() - start < max_time:
        it += 1
        # Geometric beta schedule
        frac = min(1.0, it / (perturbation_interval * 8))
        beta = beta_start * (beta_end / beta_start) ** frac
        beta = min(beta, beta_end)

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_local = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        val, grad = lse_value_and_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking line search
        step = 1.0 / L
        x_new = None
        f_y = val
        for bt in range(6):
            x_candidate = project_simplex(y - step * grad, S)
            if bt == 0:
                x_new = x_candidate
            else:
                c_cand = fft_convolve_self(x_candidate)
                val_cand, _ = lse_value_and_grad(x_candidate, c_cand, beta)
                if val_cand <= f_y - 1e-10 * step * gnorm * gnorm:
                    x_new = x_candidate
                    break
                step *= 0.4

        if x_new is None:
            x_new = project_simplex(y - step * grad, S)

        # Adaptive L
        c_x = fft_convolve_self(x_new)
        if c_x.max() > cur_max * (1.0 + 1e-6):
            L *= 1.3
        elif it % 50 == 0:
            L = max(L * 0.97, beta * 2.0)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Structured perturbation to escape local minima
        if stall_count > perturbation_interval and rng.random() < perturbation_prob:
            # Chirp-based perturbation
            scale = 0.02 * S / np.sqrt(n)
            y = project_simplex(y + scale * (0.5 * chirp + 0.5 * rng.standard_normal(n)), S)
            t_momentum = 1.0
            stall_count = 0

    return best_local, best_score


def run_active_set_polish(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=256):
    """Active-set subgradient focusing on top lags."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_momentum = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    lr = S / (n * 5.0)

    while time.monotonic() - start < max_time:
        it += 1
        k = min(top_k_max, top_k_init + it // 100)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Top-k subgradient with softmax weighting
        top_indices = np.argsort(c)[::-1][:k]
        top_vals = c[top_indices]
        tv_max = top_vals.max()
        weights = np.exp((top_vals - tv_max) / 1.0)
        weights = weights / weights.sum()
        
        w_full = np.zeros(2 * n - 1)
        for i, idx in enumerate(top_indices):
            w_full[int(idx)] = weights[i]
        
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Polyak-style step
        lr = 2.0 * (cur - best_score * 0.999) / max(gnorm * gnorm, 1e-10)
        lr = min(lr, S / (n * 0.5))
        lr = max(lr, S / (n * 100.0))

        x_new = project_simplex(y - lr * grad, S)

        # Verify improvement
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

    return best_local, best_score


def run_coordinate_polish(a_init, n, S, max_time, rng):
    """Coordinate descent: try adjusting individual elements to reduce peak."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    c = fft_convolve_self(a)
    best_score = c.max()
    peak_lag = int(np.argmax(c))
    
    # Find top contributing pairs for the peak lag
    i_min = max(0, peak_lag - (n - 1))
    i_max = min(peak_lag, n - 1)
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            peak_lag = int(np.argmax(c))
            i_min = max(0, peak_lag - (n - 1))
            i_max = min(peak_lag, n - 1)

        # Get top contributing indices for current peak
        if i_min <= i_max:
            indices = np.arange(i_min, i_max + 1)
            partner = peak_lag - indices
            products = a[indices] * np.clip(a, 0, 1000)[partner]
            
            # Try reducing the largest contributors
            top_idx = np.argsort(products)[::-1][:min(20, len(indices))]
            
            for ti in top_idx:
                idx = int(indices[ti])
                pidx = int(partner[ti])
                if idx == pidx:
                    continue
                    
                # Try shifting mass from idx to a random small element
                delta = 0.01 * S / n
                candidates = np.where(a < np.median(a))[0]
                if len(candidates) == 0:
                    continue
                add_idx = int(candidates[rng.integers(len(candidates))])
                
                a_trial = a.copy()
                a_trial[idx] = max(0, a_trial[idx] - delta)
                a_trial[add_idx] = a_trial[add_idx] + delta
                # Renormalize
                s_trial = a_trial.sum()
                if s_trial > 0:
                    a_trial = a_trial * (S / s_trial)
                    
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur - 1e-12:
                    a = a_trial
                    break
                else:
                    # Try opposite
                    a_trial2 = a.copy()
                    a_trial2[idx] = min(1000.0, a_trial2[idx] + delta)
                    a_trial2[add_idx] = max(0, a_trial2[add_idx] - delta)
                    s_trial2 = a_trial2.sum()
                    if s_trial2 > 0:
                        a_trial2 = a_trial2 * (S / s_trial2)
                    c_trial2 = fft_convolve_self(a_trial2)
                    if c_trial2.max() < cur - 1e-12:
                        a = a_trial2
                        break

    return best_local, best_score


def make_anti_palindromic_chirp(n, rng, S):
    """Anti-palindromic initialization with chirp structure."""
    half = n // 2
    tt = np.arange(half, dtype=np.float64)
    # Chirp + random for the first half
    h = 0.5 + 0.3 * np.cos(2.0 * np.pi * tt * tt / n) + 0.2 * rng.standard_normal(half)
    h = np.maximum(h, 0.01)
    
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu * 0.5
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_chirp_positive(n, rng, S):
    """Positive chirp: a_i = 1 + eps*cos(2*pi*i^2/n), normalized."""
    tt = np.arange(n, dtype=np.float64)
    eps = rng.uniform(0.3, 0.8)
    freq = rng.uniform(0.8, 1.2)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * tt * tt / n)
    a = np.maximum(a, 0.01)
    a = a * (S / a.sum())
    return a


def make_bandpass_random(n, rng, S):
    """Random sequence with band-pass frequency content."""
    a = rng.standard_normal(n)
    # Apply band-pass filter in frequency domain
    A = fft(a)
    fsize = len(A)
    # Zero out low and high frequencies
    low_cut = fsize // (rng.integers(4, 10))
    high_cut = fsize - low_cut
    A[:low_cut] = 0
    A[high_cut:] = 0
    a = np.real(ifft(A))
    a = np.abs(a) + 0.1
    a = a * (S / a.sum())
    return a


def make_periodic_mixed(n, rng, S):
    """Mix of periodic components at different frequencies."""
    tt = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    n_freqs = rng.integers(2, 6)
    for _ in range(n_freqs):
        freq = rng.integers(2, 30)
        phase = rng.random() * 2 * np.pi
        amp = rng.uniform(0.1, 0.5)
        a += amp * np.cos(2.0 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.05)
    a = a * (S / a.sum())
    return a


def generate_inits(n, rng, S, parent=None):
    """Generate diverse initializations."""
    inits = []
    
    if parent is not None and len(parent) >= 64:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if p.sum() > 0:
            inits.append(("parent", p * (S / p.sum())))
            # Perturbed parent
            p2 = p + 0.05 * S / np.sqrt(n) * rng.standard_normal(n)
            p2 = np.maximum(p2, 0)
            if p2.sum() > 0:
                inits.append(("parent_pert", p2 * (S / p2.sum())))
    
    # Anti-palindromic chirp
    for i in range(3):
        inits.append(("apal_chirp_%d" % i, make_anti_palindromic_chirp(n, rng, S)))
    
    # Pure chirp
    for i in range(2):
        inits.append(("chirp_%d" % i, make_chirp_positive(n, rng, S)))
    
    # Band-pass random
    for i in range(2):
        inits.append(("bandpass_%d" % i, make_bandpass_random(n, rng, S)))
    
    # Periodic mixed
    for i in range(2):
        inits.append(("periodic_%d" % i, make_periodic_mixed(n, rng, S)))
    
    # Simple random
    r = rng.random(n) + 0.1
    inits.append(("random", r * (S / r.sum())))
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)
    
    global_best = np.inf
    global_a = None
    
    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    n = 8192
    if parent is not None and len(parent) >= 64:
        n = len(parent)
    
    S = np.sqrt(2.0 * n)
    print("Starting optimization: n=%d, parent=%s, budget=%.0fs" % (n, parent is not None, budget_s))
    
    # === Phase 1: Explore with diverse initializations ===
    inits = generate_inits(n, rng, S, parent)
    n_inits = len(inits)
    phase1_time = (deadline - time.monotonic()) * 0.25
    per_init_time = phase1_time / n_inits
    
    print("Phase 1: %d inits, %.1f sec each" % (n_inits, per_init_time))
    
    for name, arr in inits:
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break
        
        alloc = min(per_init_time, remaining * 0.9)
        a_f, s_f = run_fista_geometric(arr, n, S, alloc, rng, 
                                         beta_start=1.0, beta_end=50.0)
        a_f = np.clip(a_f, 0, 1000)
        s_f = a_f.sum()
        if s_f > 0:
            a_f = a_f * (S / s_f)
        c_f = fft_convolve_self(a_f)
        score_f = c_f.max()
        if score_f < global_best:
            global_best = score_f
            global_a = a_f.copy()
            print("  Phase1 [%s]: max_conv=%.6f" % (name, global_best))
    
    print("After Phase 1: best=%.6f" % global_best)
    
    # === Phase 2: Refine the best with medium beta ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        phase2_time = remaining * 0.40
        print("Phase 2: refining for %.0f seconds" % phase2_time)
        
        # Run multiple FISTA passes with increasing beta
        best_in_phase2 = global_a.copy()
        best_score_phase2 = global_best
        
        for pass_idx in range(3):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            alloc = min(phase2_time / (3 - pass_idx), rem * 0.9)
            b_start = 20.0 * (pass_idx + 1)
            b_end = 100.0 * (pass_idx + 1)
            
            a_p, s_p = run_fista_geometric(best_in_phase2, n, S, alloc, rng,
                                             beta_start=b_start, beta_end=b_end)
            a_p = np.clip(a_p, 0, 1000)
            s_p = a_p.sum()
            if s_p > 0:
                a_p = a_p * (S / s_p)
            c_p = fft_convolve_self(a_p)
            if c_p.max() < best_score_phase2:
                best_score_phase2 = c_p.max()
                best_in_phase2 = a_p.copy()
                print("  Phase2 pass %d: %.6f" % (pass_idx, best_score_phase2))
        
        if best_score_phase2 < global_best:
            global_best = best_score_phase2
            global_a = best_in_phase2.copy()
    
    print("After Phase 2: best=%.6f" % global_best)
    
    # === Phase 3: High-beta polishing ===
    if global_a is not None:
        remaining = deadline - time.monotonic()
        
        # 3a: Very high beta FISTA
        alloc3a = min(remaining * 0.30, 300.0)
        if alloc3a > 10:
            a_p, s_p = run_fista_geometric(global_a, n, S, alloc3a, rng,
                                             beta_start=200.0, beta_end=2000.0,
                                             perturbation_interval=100,
                                             perturbation_prob=0.15)
            a_p = np.clip(a_p, 0, 1000)
            s_p = a_p.sum()
            if s_p > 0:
                a_p = a_p * (S / s_p)
            c_p = fft_convolve_self(a_p)
            if c_p.max() < global_best:
                global_best = c_p.max()
                global_a = a_p.copy()
                print("  Phase3a (hi-beta): %.6f" % global_best)
        
        # 3b: Active set polish
        remaining = deadline - time.monotonic()
        alloc3b = min(remaining * 0.30, 200.0)
        if alloc3b > 10:
            a_p, s_p = run_active_set_polish(global_a, n, S, alloc3b, rng,
                                               top_k_init=5, top_k_max=512)
            a_p = np.clip(a_p, 0, 1000)
            s_p = a_p.sum()
            if s_p > 0:
                a_p = a_p * (S / s_p)
            c_p = fft_convolve_self(a_p)
            if c_p.max() < global_best:
                global_best = c_p.max()
                global_a = a_p.copy()
                print("  Phase3b (active-set): %.6f" % global_best)
        
        # 3c: Coordinate polish
        remaining = deadline - time.monotonic()
        alloc3c = min(remaining * 0.25, 150.0)
        if alloc3c > 10:
            a_p, s_p = run_coordinate_polish(global_a, n, S, alloc3c, rng)
            a_p = np.clip(a_p, 0, 1000)
            s_p = a_p.sum()
            if s_p > 0:
                a_p = a_p * (S / s_p)
            c_p = fft_convolve_self(a_p)
            if c_p.max() < global_best:
                global_best = c_p.max()
                global_a = a_p.copy()
                print("  Phase3c (coordinate): %.6f" % global_best)
        
        # 3d: Final FISTA squeeze
        remaining = deadline - time.monotonic()
        if remaining > 20:
            alloc3d = remaining * 0.8
            a_p, s_p = run_fista_geometric(global_a, n, S, alloc3d, rng,
                                             beta_start=500.0, beta_end=5000.0,
                                             perturbation_interval=50,
                                             perturbation_prob=0.1)
            a_p = np.clip(a_p, 0, 1000)
            s_p = a_p.sum()
            if s_p > 0:
                a_p = a_p * (S / s_p)
            c_p = fft_convolve_self(a_p)
            if c_p.max() < global_best:
                global_best = c_p.max()
                global_a = a_p.copy()
                print("  Phase3d (final squeeze): %.6f" % global_best)
    
    # === Phase 4: Verification ===
    if global_a is None:
        global_a = np.ones(n, dtype=np.float64) * (S / n)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(n, dtype=np.float64) * (S / n)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification: eval=%.6f, internal_max=%.6f, n=%d" % 
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))
    
    if not np.isfinite(ver):
        print("WARNING: evaluate_sequence rejected, using fallback")
        global_a = np.ones(1024, dtype=np.float64) * 1.0
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
    
    print("Final result: %.6f" % (ver if np.isfinite(ver) else float('inf')))
    return final_list
