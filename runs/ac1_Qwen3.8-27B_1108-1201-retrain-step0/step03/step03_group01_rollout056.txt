"""

spectral equalization, and multi-start optimization.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates < w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Fix numerical drift
    diff = S - result.sum()
    if abs(diff) > 1e-12:
        result += diff / rho
        result = np.maximum(result, 0.0)
    return result


def compute_conv(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_lse(a, c, w):
    """
    Compute gradient of weighted sum of c[k] w.r.t. a.
    w has length 2n-1 (same as c).
    Returns gradient of length n.
    """
    n = len(a)
    len_c = 2 * n - 1
    fsize = 1 << (len_c + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def lse_weights(c, beta):
    """Compute LSE weights (softmax of c with temperature 1/beta)."""
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    total = exp_c.sum()
    if total < 1e-300:
        # Fallback: uniform over top elements
        top_idx = np.argsort(c)[::-1][:10]
        w = np.zeros(len(c))
        w[top_idx] = 1.0 / len(top_idx)
        return w, c_max
    return exp_c / total, c_max


def fista_lse_optimize(a_init, n, S, max_time, rng, beta_init=50, beta_max=50000,
                       step_scale=1.0, noise_period=500, noise_prob=0.1):
    """
    FISTA with LSE smoothing. Aggressive beta schedule for sharper max approximation.
    """
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    backtrack_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: exponential growth
        beta = min(beta_max, beta_init * np.exp(0.02 * it))

        c = compute_conv(y)
        cur_max = c.max()
        score = 2.0 * n * cur_max / S**2
        if score < best_score:
            best_score = score
            best_local = y.copy()

        if cur_max <= 1.5020 * S**2 / (2.0 * n):
            break

        # Compute LSE weights
        w, c_max_val = lse_weights(c, beta)

        # Gradient
        grad = compute_gradient_lse(y, c, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: 1/(2*beta) is the standard for LSE with smoothness beta
        # But our function has larger smoothness, so use a smaller step
        step = step_scale / (beta * 2.0)
        step = min(step, S / (2.0 * n))  # Don't take too large a step
        step = max(step, 1e-10)

        # Nesterov update
        x_new = project_simplex(y - step * grad, S)

        # Backtracking: check if function value increased
        if it % 5 == 0:
            c_x = compute_conv(x_new)
            score_x = 2.0 * n * c_x.max() / S**2
            if score_x < best_score:
                best_score = score_x
                best_local = x_new.copy()
            if score_x > score * 1.01:
                # Increase (worsened), reduce step
                step *= 0.5
                backtrack_count += 1
                x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t - 1.0) / t_new
        y = x_new + momentum * (x_new - a)
        y = np.maximum(y, 0.0)
        # Renormalize to maintain sum = S
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        else:
            y = a.copy()
        a = x_new
        t = t_new

        # Periodic noise to escape local minima
        if it % noise_period == 0 and rng.random() < noise_prob:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t = 1.0

    return best_local, best_score


def topk_subgradient_optimize(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=200,
                              step_init=None):
    """
    Top-k subgradient with Nesterov acceleration.
    Focuses on the top-k lags of the convolution.
    """
    a = np.clip(a_init.copy(), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    if step_init is None:
        step = S / (n * 2.0)
    else:
        step = step_init

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)

        c = compute_conv(y)
        cur_max = c.max()
        score = 2.0 * n * cur_max / S**2
        if score < best_score:
            best_score = score
            best_local = y.copy()

        if score <= 1.5020:
            break

        # Top-k weights: proportional to c values (or just uniform over top-k)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        # Use soft weights: proportional to values
        w = np.zeros(len(c))
        for i, k in enumerate(top_idx):
            w[k] = top_vals[i] / total

        grad = compute_gradient_lse(y, c, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step
        step_curr = step / (1.0 + 0.001 * it)
        x_new = project_simplex(y - step_curr * grad / max(gnorm, 1e-15), S)

        if it % 10 == 0:
            c_x = compute_conv(x_new)
            score_x = 2.0 * n * c_x.max() / S**2
            if score_x < best_score:
                best_score = score_x
                best_local = x_new.copy()
            if score_x > score * 1.02:
                step *= 0.7

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t - 1.0) / t_new
        y = x_new + momentum * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y * (S / y_sum)
        else:
            y = a.copy()
        a = x_new
        t = t_new

        # Noise
        if it % 300 == 0 and rng.random() < 0.15:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0

    return best_local, best_score


def spectral_equalize(a, n, N, n_iters=100, alpha=0.3, rng=None):
    """
    Spectral equalization: try to make |FFT(a)|^2 more uniform at non-DC frequencies.
    This flattens the autocorrelation sidelobes.
    """
    S = a.sum()
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A)**2
    P[0] = 0  # Don't touch DC

    for it in range(n_iters):
        # Compute current autocorrelation
        c = np.real(ifft(P))[:2*n-1]
        c_max = c.max()

        # Power spectrum "target": inverse of current power (water-filling)
        # Suppress frequencies that contribute most to the peak
        m = np.arange(N)
        # Find which frequencies contribute most to the peak lag
        peak_lag = c.argmax()
        cos_peak = np.cos(2.0 * np.pi * m * peak_lag / N)

        # Reduce power at frequencies that align with the peak
        P = P * (1.0 - alpha * np.maximum(cos_peak, 0))
        P[0] = 0  # Keep DC fixed

        # Reconstruct
        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(P, 0)) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new, S)

    return a


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_diverse_init(n, rng, base):
    """Create diverse initializations from a base sequence."""
    inits = []
    S = np.sqrt(2.0 * float(n))

    # The base sequence itself
    inits.append(("base", base.copy()))

    # Small perturbations
    for scale in [0.005, 0.01, 0.02]:
        perturbed = base + rng.normal(0, scale * S / np.sqrt(n), n)
        inits.append(("pert_%.0f" % (scale*10000), perturbed))

    # Geometric modulation
    t = np.arange(n) / n
    geo = base * (1.0 + 0.05 * np.sin(2*np.pi*3*t))
    inits.append(("geo3", geo))
    geo2 = base * (1.0 + 0.03 * np.cos(2*np.pi*7*t + 0.5))
    inits.append(("geo7", geo2))

    # Logarithmic taper
    taper = 1.0 + 0.1 * np.log1p(np.abs(base - base.mean()) / (base.std() + 1e-10))
    inits.append(("taper", base * taper))

    return inits

            global_a = a_p.copy()
            print("  3c result: %.6f" % global_best)
        a_curr = global_a.copy()

    # 3d: Final micro-FISTA
    rem = deadline - time.monotonic()
    if rem > 5:
        print("  3d: Final micro-FISTA (%.0f sec)" % rem)
        a_p, sc_p = fista_lse_optimize(a_curr, n, S, rem, rng,
                                        beta_init=10000, beta_max=50000,
                                        step_scale=0.3, noise_period=2000, noise_prob=0.02)
        a_p = safe_normalize(a_p, n)
        c_p = compute_conv(a_p)
        sc = 2.0 * n * c_p.max() / S**2
        if sc < global_best:
            global_best = sc
            global_a = a_p.copy()
            print("  3d result: %.6f" % global_best)

    # === Verification ===
    print("\n=== Verification ===")
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        print("WARNING: sum too small, using fallback")
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    if np.isfinite(ver):
        print("Verified score: %.6f" % ver)
        print("Sequence length: %d" % len(final_list))
    else:
        print("WARNING: evaluate_sequence rejected, using fallback")
        global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Fallback score: %.6f" % ver)

    print("Returning sequence of length %d, internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
