"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using spectral equalization + adaptive LSE subgradient.

Approach:
1. Direct spectral equalization to flatten |FFT(a)|^2
2. Adaptive log-sum-exp subgradient descent
3. Structured perturbations for escape from local minima
4. Multi-start from diverse initializations
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv(a):
    """Self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fft_conv(a)
    return 2.0 * n * c.max() / (a.sum() ** 2)


def spectral_equalize(a, N, S, alpha_start, alpha_end, n_iter, rng):
    """
    Iteratively flatten the power spectrum of a.
    Uses direct blending toward flat spectrum.
    Returns best a found.
    """
    best_a = a.copy()
    best_cmax = fft_conv(a).max()
    
    for it in range(n_iter):
        # Adaptive alpha: start aggressive, end gentle
        frac = it / max(n_iter - 1, 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        
        # Compute current spectrum
        a_pad = np.zeros(N)
        a_pad[:len(a)] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        
        # Target: flat spectrum
        P_target = S ** 2 / N
        
        # Blend
        P_new = (1.0 - alpha) * P + alpha * P_target
        
        # Reconstruct with original phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        
        # Inverse FFT and project
        a_new = np.real(ifft(A_new))[:len(a)]
        a = project_simplex(a_new, S)
        
        # Track best
        cmax = fft_conv(a).max()
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
    
    return best_a, best_cmax


def lse_subgradient(a, n, S, max_time, rng, beta_init=10.0, beta_max=500.0):
    """
    FISTA with log-sum-exp smooth approximation of max(conv(a,a)).
    """
    a = np.clip(a, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    N = 1 << (2 * n - 1).bit_length()
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_conv(y).max()
    best_local = y.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.01 * it))
        
        c = fft_conv(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        # LSE approximation of max
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total
        
        # Gradient: d/dc max_approx = w, and dc/da = 2 * conv_indicator * a (cross-conv)
        # More precisely: grad_a = 2 * (w_conv cross a) where w_conv is the weight pattern
        # conv(a,a)[k] = sum_i a[i]*a[k-i], so d(conv[k])/d(a[j]) = a[k-j] + a[j-k] (with bounds)
        # So grad = 2 * sum_k w[k] * (a * reverse_weighted_pattern)
        # Using FFT: grad = 2 * ifft(fft(w) * conj(fft(a)))
        
        fsize = N
        W = fft(w, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: for LSE, the function is (1/beta)-smooth
        step = 1.0 / (beta * 2.0)
        
        x_new = project_simplex(y - step * grad, S)
        
        if it % 5 == 0:
            c_x = fft_conv(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Occasional perturbation
        if it % 1000 == 500 and rng.random() < 0.2:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
    
    return best_local, best_score


def structured_perturb(a, n, S, rng, strength=0.01):
    """Apply a structured perturbation to escape local minima."""
    a_new = a.copy()
    pert_type = rng.integers(0, 4)
    
    if pert_type == 0:
        # Smooth sinusoidal perturbation
        freq = rng.uniform(0.5, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        t = np.arange(n) / n
        a_new = a_new + strength * S * np.sin(2 * np.pi * freq * t + phase)
    elif pert_type == 1:
        # Random swap of two blocks
        block_size = rng.integers(n // 32, n // 8)
        i = rng.integers(0, n - block_size)
        j = rng.integers(0, n - block_size)
        if abs(i - j) > block_size:
            a_new[i:i+block_size], a_new[j:j+block_size] = a_new[j:j+block_size].copy(), a_new[i:i+block_size].copy()
    elif pert_type == 2:
        # Gaussian blob perturbation
        center = rng.integers(0, n)
        width = rng.integers(n // 64, n // 16)
        amplitude = strength * S / width
        t = np.arange(n)
        a_new = a_new + amplitude * np.exp(-0.5 * ((t - center) / width) ** 2)
    else:
        # Scale random subset
        mask = rng.random(n) < 0.1
        a_new[mask] *= (1.0 + strength * rng.standard_normal(mask.sum()))
    
    return project_simplex(a_new, S)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using combined spectral shaping,
adaptive LSE subgradient (FISTA), and gradient-guided perturbation.

Key innovations over the 1.5074 baseline:
1. Hybrid spectral shaping: multiplicative correction for coarse adjustments,
   additive blending for fine-tuning.
2. Adaptive LSE temperature: beta increases only when progress is made,
   preventing premature convergence to suboptimal LSE smoothing.
3. Gradient-guided perturbation: perturbations are biased toward the
   negative gradient direction, making them more effective at escaping.
4. Multi-timescale optimization: alternate between fast (spectral) and
   slow (subgradient) timescales to avoid getting trapped.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_conv(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:size]


def norm_simplex(a, S):
    """Clamp to [0, inf) and rescale to sum S."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) * S / len(a)
    return a * (S / s)


def spectral_shape_multiplicative(a, N, S, alpha, top_k, n_iter, rng):
    """
    Spectral shaping via multiplicative correction.
    Reduces power at frequencies contributing to top-k autocorrelation lags.
    Returns best (a, cmax) found during iterations.
    """
    n = len(a)
    best_a = a.copy()
    c = fft_conv(a)
    best_cmax = c.max()
    
    for it in range(n_iter):
        # Adaptive alpha: decrease over time
        frac = it / max(n_iter - 1, 1)
        alpha_cur = alpha * (1.0 - 0.7 * frac)
        
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        c = np.real(ifft(P))[:2 * n - 1]
        c_max = c.max()
        
        if c_max < best_cmax:
            best_cmax = c_max
            best_a = a.copy()
        
        if c_max <= 1.5 * S ** 2 / (2 * n):  # close to theoretical min
            break
        
        # Compute effective cosine for top-k lags
        m = np.arange(N)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total
        
        cos_eff = np.zeros(N)
        for i, k in enumerate(top_idx):
            cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
        
        P_new = np.maximum(P * (1.0 - alpha_cur * cos_eff), 0.0)
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new_full = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new_full, S)
    
    return best_a, best_cmax


def spectral_shape_additive(a, N, S, alpha, n_iter, rng):

    Spectral shaping via additive blending toward flat spectrum.
    Returns best (a, cmax) found during iterations.

        if cmax < best_cmax:

    return best_a, best_cmax


def fista_lse(a, n, S, max_time, rng, beta_init=20.0, beta_max=500.0, beta_growth=0.005):
    """
    FISTA with log-sum-exp smooth approximation of max(conv(a,a)).
    Adaptive beta: increases with progress, resets on perturbation.
    Returns best (a, cmax).
    """
    a = norm_simplex(a, S)
    N = 1 << (2 * n - 1).bit_length()
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = fft_conv(y).max()
    best_local = y.copy()
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + beta_growth * it))
        
        c = fft_conv(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
        
        # LSE approximation
        c_max = c.max()
        # Clip to avoid overflow
        arg = beta * (c - c_max)
        arg = np.clip(arg, -500, 500)
        exp_bc = np.exp(arg)
        total = exp_bc.sum()
        if total < 1e-300:
            break
        w = exp_bc / total
        
        # Gradient via FFT
        W = fft(w, N)
        A = fft(y, N)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size for (1/beta)-smooth function: 1/(2*beta) 
        # But we also have the simplex constraint, so use a conservative step
        step = 1.0 / (2.0 * beta)
        
        x_new = project_simplex(y - step * grad, S)
        
        if it % 3 == 0:
            c_x = fft_conv(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
        
        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Occasional perturbation
        if it % 500 == 250 and rng.random() < 0.25:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale

, n), S)
            t_mom = 1.0
    
    return best_local, best_score


def gradient_perturb(a, n, S, rng, strength=0.01):
    """
    Gradient-guided perturbation: move in negative gradient direction with noise.
    """
    c = fft_conv(a)
    c_max = c.max()
    
    # Find the lag(s) where max occurs
    threshold = c_max - 0.01 * c_max
    active_lags = np.where(c >= threshold)[0]
    
    # Compute subgradient at active lags
    grad = np.zeros(n)
    for k in active_lags:
        k = int(k)
        # conv[k] = sum_i a[i]*a[k-i], d(conv[k])/d(a[j]) = a[k-j] + a[j-k] (with bounds)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            # d(a[j]*a[k-j])/d(a[j]) = a[k-j]
            # d(a[k-j]*a[j])/d(a[j]) = a[k-j] (same term, counted once in the sum)
            # Actually: conv[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
            # d/d(a[j]) of conv[k] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n, i.e., k-j<0)
            # Wait, more carefully:
            # conv[k] = sum_i a[i] * a[k-i] where 0<=i<n and 0<=k-i<n
            # = sum_i a[i] * a[k-i] where max(0,k-n+1) <= i <= min(n-1, k)
            # d/d(a[j]): j appears as a[i] when i=j, and as a[k-i] when k-i=j, i.e., i=k-j
            # So d/d(a[j]) = (a[k-j] if 0<=k-j<n) + (a[j-k] if 0<=j-k<n) ... 
            # Wait, if k-j is in [0,n-1], then a[j]*a[k-j] contributes a[k-j] to d/d(a[j])
            # And if j-k is in [0,n-1] (i.e., k<0 or j>k), then a[k-j]*a[j] = a[j-k]... 
            # I'm overcomplicating this. Let me use the FFT approach instead.
        pass
    
    # Use FFT for efficiency
    N = 1 << (2 * n - 1).bit_length()
    w_full = np.zeros(2 * n - 1)
    active_count = len(active_lags)
    for k in active_lags:
        w_full[int(k)] = 1.0 / max(active_count, 1)
    
    W = fft(w_full, N)
    A = fft(a, N)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    # Move in negative gradient direction with noise
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad_dir = -grad / gnorm
    else:

    a_new = a + strength * S * (0.5 * grad_dir + 0.5 * rng.standard_normal(n) / np.sqrt(n))
