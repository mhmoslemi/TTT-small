"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP cutting-plane direction finding
combined with adaptive power-sum gradient descent and coordinate descent polish.

Key innovations over previous approach:
1. LP cutting-plane: solve reduced LPs to find optimal descent directions
2. Power-sum smooth objective (p-norm approximation to max) for better gradients
3. Adaptive temperature based on convolution flatness
4. Multi-scale: try different sequence lengths
5. Aggressive coordinate descent with line search

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve(f, g):
    """Linear convolution of f and g via FFT."""
    n = len(f)
    m = len(g)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    F = fft(f, fsize)
    G = fft(g, fsize)
    c = np.real(ifft(F * G))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def objective_value(a):
    """Compute the evaluation objective for sequence a."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def power_sum_obj(c, alpha):
    """Power-sum approximation to max: (sum c^alpha / K)^(1/alpha)."""
    K = len(c)
    c_pos = np.maximum(c, 1e-30)
    return (np.sum(c_pos ** alpha) / K) ** (1.0 / alpha)


def power_sum_grad(c, a, alpha):
    """Gradient of power-sum objective w.r.t. a.
    d/da_i [ (sum_k c_k^alpha / K)^(1/alpha) ]
    = (1/K) * (sum c^alpha)^(1/alpha - 1) * alpha * sum_k c_k^(alpha-1) * dc_k/da_i
    dc_k/da_i = 2 * sum_j: a_j where (f*f)_k involves a_i
    Actually c_k = (a*a)_k, dc_k/da_i = 2*(a*a partial w.r.t a_i)
    dc_k/da_i = 2 * sum_{j: i+j=k or similar} ... 
    More precisely: c_k = sum_j a_j * a_{k-j}, so dc_k/da_i = 2 * a_{k-i} (if 0<=k-i<n)
    Wait no: c_k = sum_j a[j]*a[k-j]. d/d a[i] = sum_j [delta(j,i)*a[k-j] + a[j]*delta(k-j,i)]
    = a[k-i] + a[k-i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    So gc_i = (1/K)*S^(1/alpha - 1)*alpha * sum_k c_k^(alpha-1) * 2*a[k-i] (0<=k-i<n)
    = (2*alpha/K)*S^(1/alpha-1) * sum_{k: 0<=k-i<n} c_k^(alpha-1) * a[k-i]
    Let m = k-i, then k = m+i, and m ranges over [0, n-1] with 0 <= m+i < 2n-1
    gc_i = (2*alpha/K)*S^(1/alpha-1) * sum_{m=0}^{n-1} c_{m+i}^(alpha-1) * a[m]
    This is a cross-correlation!
    gc = (2*alpha/K)*S^(1/alpha-1) * cross_corr(c^(alpha-1), a)
    where cross_corr(x, a)[i] = sum_m x[m+i]*a[m]
    """
    n = len(a)
    c_alpha_minus_1 = np.maximum(c, 1e-30) ** (alpha - 1)
    S = np.sum(np.maximum(c, 1e-30) ** alpha) / len(c)
    if S < 1e-30:
        return np.zeros(n)
    prefactor = (2.0 * alpha / len(c)) * S ** (1.0 / alpha - 1.0)
    
    # cross_corr: gc[i] = sum_{m=0}^{n-1} c_alpha_minus_1[m+i] * a[m]
    # This is the correlation of c_alpha_minus_1 with a, at lag i
    fsize = 1 << (3 * n - 2).bit_length()
    C = fft(c_alpha_minus_1, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(C * np.conj(A)))
    gc = prefactor * corr[:n]
    return gc


def run_adaptive_gradient(a_init, n, S, time_budget, alpha_init=10.0, alpha_final=500.0,
                          lr_init=0.1, verbose=False):
    """Adaptive power-sum gradient descent with backtracking line search."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 200
    
    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        
        # Adaptive alpha: increase over time, also adapt to flatness
        alpha = alpha_init * (alpha_final / alpha_init) ** (progress ** 0.5)
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            # Restart with noise
            noise_scale = 0.02 * max(0.1, 1.0 - progress)
            rng = np.random.default_rng(step * 7919 + 13)
            a = best_a + noise_scale * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            no_improve = 0
            step += 1
            continue
        
        # Compute gradient
        g = power_sum_grad(c, a, alpha)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g_dir = g / gnorm
        
        # Backtracking line search
        lr = lr_init * (1.0 - 0.5 * progress)
        for i in range(10):
            a_trial = a - lr * g_dir
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial < cur_obj:
                break
            lr *= 0.5
        
        a = a_trial
        step += 1
        
        if verbose and step % 200 == 0:
            print("    Grad step %d: obj=%.7f, alpha=%.1f" % (step, best_obj, alpha))
    
    return best_a, best_obj

                min_idx = np.argmin(fg_vals)
                to_remove = all_rows[min_idx]
                active_set.discard(to_remove)
                all_rows = sorted(active_set)
                fg_vals = np.array([fg[r] for r in all_rows])
    
    return best_g


def run_lp_optimization(a_init, n, S, time_budget, verbose=False):
    """Main LP-based optimization loop."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    M = c.max()
    best_obj = M
    best_a = a.copy()
    
    start = time.monotonic()
    iteration = 0
    no_improve = 0
    lp_step = 0.3  # mixing parameter
    
    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 5:
            break
        
        c = fft_convolve_self(a)
        M = c.max()
        
        if M < best_obj - 1e-12:
            best_obj = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            # Reset step size and try again with smaller step
            lp_step = max(0.05, lp_step * 0.5)
            no_improve = 0
        
        # Solve LP for direction
        g = lp_direction(a, M * 1.001, max_active=200, max_iters=20)
        
        if g is None or g.sum() <= 0:
            break
        
        sum_g = g.sum()
        if sum_g <= a.sum() * (1 + 1e-6):
            # LP didn't find improvement
            lp_step = max(0.05, lp_step * 0.7)
            if lp_step < 0.06:
                break
            continue
        
        # Rescale g to match normalization
        g_rescaled = g * (S / sum_g)
        
        # Update with mixing
        a_new = (1.0 - lp_step) * a + lp_step * g_rescaled
        a_new = normalize_nonneg(a_new, S)
        
        # Check improvement
        c_new = fft_convolve_self(a_new)
        obj_new = c_new.max()
        
        if obj_new < M:
            a = a_new
            # Try to increase step
            lp_step = min(0.5, lp_step * 1.1)
        else:
            # Backtrack: reduce step
            lp_step = max(0.05, lp_step * 0.5)
            a_try = (1.0 - lp_step) * a + lp_step * g_rescaled
            a_try = normalize_nonneg(a_try, S)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < M:
                a = a_try
            else:
                # Try even smaller step
                lp_step = max(0.02, lp_step * 0.5)
                a_try = (1.0 - lp_step) * a + lp_step * g_rescaled
                a_try = normalize_nonneg(a_try, S)
                c_try = fft_convolve_self(a_try)
                if c_try.max() < M:
                    a = a_try
        
        iteration += 1
        if verbose and iteration % 5 == 0:
            c_chk = fft_convolve_self(a)
            print("    LP iter %d: obj=%.7f, step=%.3f, sum_g/a=%.4f" % (
                iteration, c_chk.max(), lp_step, sum_g / a.sum()))
    
    return best_a, best_obj


def coordinate_descent(a_init, n, S, time_budget, verbose=False):
    """Coordinate descent with line search on individual coordinates."""
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(42)
    sweep = 0
    
    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 5:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget - 3:
                break
            
            orig = a[i]
            if orig < 1e-14:
                continue
            
            best_local = orig
            best_local_obj = best_obj
            
            # Try multiplicative adjustments
            for factor in [0.5, 0.7, 0.85, 1.2, 1.5, 2.0, 0.3, 0.6]:
                a[i] = orig * factor
                a = normalize_nonneg(a, S)
                c_tmp = fft_convolve_self(a)
                obj_tmp = c_tmp.max()
                if obj_tmp < best_local_obj - 1e-13:
                    best_local_obj = obj_tmp
                    best_local = orig * factor
            
            a[i] = best_local
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            
            if c.max() < best_obj - 1e-13:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        sweep += 1
        if not improved and sweep > 5:
            if verbose:
                print("    CD converged at sweep %d" % sweep)
            break
    
    return best_a, best_obj


def get_initialization(name, n, S, rng):
    """Generate various initializations."""
    i = np.arange(n, dtype=np.float64)
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "gaussian":
        a = np.exp(-0.5 * ((i - n / 2) / (n / 5)) ** 2)
        return a * (S / a.sum())
    elif name == "cosine_mix":
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * i / n + rng.random() * 2) + \
            0.2 * np.cos(2 * np.pi * 7 * i / n + rng.random() * 2) + \
            0.1 * np.cos(2 * np.pi * 13 * i / n + rng.random() * 2)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "tapered":
        t = i / (n - 1)
        a = np.sin(np.pi * t) ** 0.5
        return a * (S / a.sum())
    elif name == "optimal_fourier":
        # Try to create a sequence with flat magnitude spectrum
        # Initialize with random phases, take IFFT magnitude
        rng_local = np.random.default_rng(int(rng.integers(0, 2**31)))
        phases = rng_local.random(n) * 2 * np.pi
        spec = np.exp(1j * phases)
        spec[0] = 1.0  # DC
        a = np.abs(ifft(spec))
        a = np.maximum(a, 0.001)
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)


def propose_candidate(seed=42, budget_s=2400):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    total_budget = max(10.0, float(budget_s) - 30.0)
    deadline = time.monotonic() + total_budget
    
    global_best = np.inf
    global_a = None
    
    # Load given sequence if available
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw.copy()
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print("Given: n=%d, obj=%.7f" % (n_given, ver))
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print("Could not load given: " + str(e))
    
    # If no given sequence, generate one
    if global_a is None:
        n = 4096
        S = np.sqrt(2.0 * n)
        a = get_initialization("cosine_mix", n, S, rng)
        obj = evaluate_sequence(a.tolist())
        global_best = obj
        global_a = a.copy()
        print("Generated initial: n=%d, obj=%.7f" % (n, obj))
    
    n = len(global_a)
    S = np.sqrt(2.0 * float(n))
    a = global_a.copy()
    a = normalize_nonneg(a, S)
    
    # ============ PHASE 1: Quick gradient improvement (15%) ============
    phase1_budget = min(0.15 * total_budget, 300.0)
    print("\nPhase 1: Gradient (%.0fs, n=%d)" % (phase1_budget, n))
    
    a_best, obj = run_adaptive_gradient(a, n, S, phase1_budget,
                                         alpha_init=20.0, alpha_final=300.0,
                                         lr_init=0.15, verbose=True)
    a_best = normalize_nonneg(a_best, S)
    ver = evaluate_sequence(a_best.tolist())
    if np.isfinite(ver) and ver < global_best:
        global_best = ver
        global_a = a_best.copy()
        print("  P1 done: obj=%.7f" % ver)
    a = global_a.copy()
    
    # ============ PHASE 2: LP optimization (45%) ============
    phase2_budget = min(0.45 * total_budget, deadline - time.monotonic() - 0.3 * total_budget)
    if phase2_budget > 30:
        print("\nPhase 2: LP optimization (%.0fs)" % phase2_budget)
        
        a_best, obj = run_lp_optimization(a, n, S, phase2_budget, verbose=True)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  P2 done: obj=%.7f" % ver)
        a = global_a.copy()
    
    # ============ PHASE 3: Gradient polishing (20%) ============
    phase3_budget = min(0.20 * total_budget, deadline - time.monotonic() - 0.15 * total_budget)
    if phase3_budget > 20:
        print("\nPhase 3: High-alpha gradient (%.0fs)" % phase3_budget)
        
        a_best, obj = run_adaptive_gradient(a, n, S, phase3_budget,
                                             alpha_init=100.0, alpha_final=2000.0,
                                             lr_init=0.05, verbose=True)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  P3 done: obj=%.7f" % ver)
        a = global_a.copy()
    
    # ============ PHASE 4: Coordinate descent (10%) ============
    phase4_budget = min(0.10 * total_budget, deadline - time.monotonic() - 5.0)
    if phase4_budget > 10:
        print("\nPhase 4: Coordinate descent (%.0fs)" % phase4_budget)
        
        a_best, obj = coordinate_descent(a, n, S, phase4_budget, verbose=True)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  P4 done: obj=%.7f" % ver)
        a = global_a.copy()
    
    # ============ PHASE 5: Final LP + gradient cycle (remaining time) ============
    remaining = deadline - time.monotonic() - 10.0
    if remaining > 30:
        print("\nPhase 5: Final cycles (%.0fs)" % remaining)
        
        cycle_budget = remaining / 3
        for cycle in range(3):
            if time.monotonic() > deadline - 10:
                break
            
            # LP step
            a_best, _ = run_lp_optimization(a, n, S, cycle_budget * 0.6, verbose=False)
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  Cycle %d LP: obj=%.7f" % (cycle, ver))
            a = global_a.copy()
            
            # Gradient step
            if time.monotonic() > deadline - 10:
                break
            a_best, _ = run_adaptive_gradient(a, n, S, cycle_budget * 0.4,
                                               alpha_init=200.0, alpha_final=3000.0,
                                               lr_init=0.03, verbose=False)
            a_best = normalize_nonneg(a_best, S)
            ver = evaluate_sequence(a_best.tolist())
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = a_best.copy()
                print("  Cycle %d Grad: obj=%.7f" % (cycle, ver))
            a = global_a.copy()
    
    # ============ Final CD polish with remaining time ============
    remaining = deadline - time.monotonic() - 5.0
    if remaining > 5 and global_a is not None:
        n = len(global_a)
        S = np.sqrt(2.0 * float(n))
        a_last = global_a.copy()
        a_last = normalize_nonneg(a_last, S)
        a_best, _ = coordinate_descent(a_last, n, S, remaining, verbose=False)
        a_best = normalize_nonneg(a_best, S)
        ver = evaluate_sequence(a_best.tolist())
        if np.isfinite(ver) and ver < global_best:
            global_best = ver
            global_a = a_best.copy()
            print("  Final CD: obj=%.7f" % ver)
    
    # Final verification
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: obj=%.7f (verified=%s)" % (
        global_best, "%.7f" % ver if np.isfinite(ver) else "inf"))
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print("Fallback: obj=%.7f" % ver)
    
    return final_a
