"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-scale FISTA with adaptive LSE smoothing, coordinate 
descent polishing, diverse initializations, and multi-length search.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, S):
    """Clamp and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def compute_objective(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c))."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def fista_lse(a_init, n, S, max_time, rng, beta_init=5.0, beta_rate=0.005, 
              beta_max=5000.0, momentum=True):
    """FISTA with LSE smoothing."""
    a = normalize_sum(a_init, S)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    last_improve = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur_peak = c.max()
        frac = cur_peak * n / (S * S)
        obj = 2.0 * frac
        if obj < best:
            best = obj
            best_a = y.copy()
            last_improve = it
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: use 1/(2*beta*max_f) as conservative estimate
        # Actually, the Lipschitz constant of the LSE gradient w.r.t. f
        # is bounded by 2*beta*||f||_1 (roughly), but we use empirical scaling
        step = 1.0 / (2.0 * beta)
        # Clip step to avoid overshooting
        step = min(step, 5.0 / (n * 0.01))  # don't take steps larger than this
        
        x_new = project_simplex(y - step * grad, S)
        
        if momentum:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = x_new + ((t - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t = t_new
        else:
            a = x_new
            y = a.copy()
        
        # Periodic restart with noise
        if it % 2000 == 0 and it - last_improve > 500:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 10000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
                a = y.copy()
        
        # Adaptive: if stuck for long, increase noise
        if it - last_improve > 3000:
            noise_scale = 0.01 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            a = y.copy()
            last_improve = it  # reset counter
    
    return normalize_sum(best_a, S), best

                    m2 = lo + golden * (hi - lo)

                    c_a1[is_diag] = c_base[is_diag] + m1 * m1

                    # Evaluate g at m2

def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    
    # Window functions
    inits.append(0.5 * (1 + np.cos(2 * np.pi * tt)))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(0.42 - 0.5 * np.cos(2 * np.pi * tt) + 0.08 * np.cos(4 * np.pi * tt))
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(np.sin(np.pi * tt) ** 6)
    
    # Multi-sinusoidal
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * 3 * tt + 0.5))
    
    # Ramp (asymmetric - important!)
    inits.append(1.0 + 0.5 * tt)
    inits.append(1.0 - 0.5 * tt)
    inits.append(0.5 + tt)
    inits.append(1.0 + 0.3 * np.sin(np.pi * tt))
    
    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.5, n)))
    
    # Two-block with different heights
    m = n // 2
    two_block = np.ones(n)
    two_block[:m] = 1.5
    two_block[m:] = 0.5
    inits.append(two_block)
    
    # Three-block
    n3 = n // 3
    three_block = np.ones(n)
    three_block[:n3] = 2.0
    three_block[n3:2*n3] = 0.5
    three_block[2*n3:] = 1.0
    inits.append(three_block)
    
    # Triangle (peaks in middle)
    inits.append(1.0 - 2.0 * np.abs(tt - 0.5))
    
    return [np.maximum(x, 0.01) for x in inits]


def upscale(a, n_target):
    """Upsample a to n_target using interpolation."""
    n_src = len(a)
    if n_src == n_target:
        return a.copy()
    t_old = np.linspace(0, 1, n_src)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a)

def coordinate_descent_polish(a, n, S, max_time, rng, n_sweeps=50):

    a = normalize_sum(a, S).copy()
    start = time.monotonic()
    best_obj = compute_objective(a)
    best_a = a.copy()
    
    # Precompute convolution
    c = fft_convolve_self(a)  # length 2n-1
    
    sweep = 0
    while time.monotonic() - start < max_time and sweep < n_sweeps:
        sweep += 1
        improved = 0
        order = rng.permutation(n)
        
        for i in order:
            x_old = a[i]
            
            # Find optimal x for coordinate i
            # C[k] is affected for k in [i, i+n-1] (where 0<=k-i<n)
            # Contribution of f[i] to C[k]:
            #   if k-i != i: 2*f[i]*f[k-i]
            #   if k-i == i (k=2i): f[i]^2
            # Base C (without f[i]):
            #   C_base[k] = C[k] - 2*x_old*f[k-i] (for k-i != i, valid)
            #   C_base[2i] = C[2i] - x_old^2
            #   C_base[k] = C[k] (for unaffected k)
            
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            # Get the affected portion
            j_range = np.arange(k_lo - i, k_hi - i + 1)  # = 0, 1, ..., min(n-1, ...)
            f_affected = a[j_range]
            is_diag = (j_range == i)
            
            c_base = c[k_lo:k_hi+1].copy()
            c_base[~is_diag] -= 2 * x_old * f_affected[~is_diag]
            if np.any(is_diag):
                c_base[is_diag] -= x_old * x_old
            
            # Unaffected max
            max_unaff = 0.0
            if i > 0:
                max_unaff = max(max_unaff, float(np.max(c[:i])))
            if k_hi + 1 < 2*n - 1:
                max_unaff = max(max_unaff, float(np.max(c[k_hi+1:])))
            
            S_base = S - x_old
            
            # Find optimal x by golden section
            # g(x) = 2*n*max(max_unaff, max(c_base + 2*x*f_aff (non-diag)), c_base_diag + x^2) / (S_base + x)^2
            
            # First, try a few candidates to bracket
            candidates = [0.0, x_old * 0.5, x_old, x_old * 1.5, max(x_old * 2.0, S_base / n)]
            best_x = x_old
            best_g = np.inf
            for xc in candidates:
                xc = max(0.0, xc)
                c_aff = c_base + 2 * xc * f_affected
                if np.any(is_diag):
                    c_aff[is_diag] = c_base[is_diag] + xc * xc
                max_c = max(max_unaff, float(np.max(c_aff)))
                s_new = S_base + xc
                g_c = 2.0 * n * max_c / (s_new * s_new) if s_new > 1e-15 else np.inf
                if g_c < best_g:
                    best_g = g_c
                    best_x = xc
            
            # Golden section refinement
            lo = 0.0
            hi = max(x_old * 3.0, S_base / max(n // 10, 1))
            # Make sure lo and hi bracket the minimum
            # g(0) and g(hi) should be > g(best_x)

                golden = 0.381966
                a_gs = hi - golden * (hi - lo)
                b_gs = lo + golden * (hi - lo)
                
                def eval_g(x):

                    if np.any(is_diag):

                    max_c = max(max_unaff, float(np.max(c_aff)))

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Multi-scale FISTA with adaptive LSE smoothing, coordinate descent
polishing, top-k subgradient refinement, and multi-length search.
Key improvements:
- Multi-scale warm start for better global exploration
- Faster beta annealing with adaptive step sizing
- Coordinate descent with incremental convolution updates for final polish
- Top-k subgradient for concentrated peak reduction
- Multiple sequence lengths explored

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, S):
    """Clamp and rescale to sum = S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) * S / len(a)
    return a * (S / s)


def compute_objective(a):
    """Compute 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s * s)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c))."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(a, c, k=16):
    """Subgradient using top-k lags with equal weight."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros(len(c))
    w[idx] = 1.0 / k
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_power_subgradient(a, c, k=32, p=2.0):
    """Subgradient using top-k lags with power weighting."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    w_k = np.power(np.maximum(vals - c[idx[-1]] + 1e-10, 1e-10), p)
    w_k = w_k / w_k.sum()
    w = np.zeros(len(c))
    w[idx] = w_k
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def fista_adaptive(a_init, n, S, max_time, rng, mode="lse",

    a = normalize_sum(a_init, S)

            last_improve = it

                t = 1.0
