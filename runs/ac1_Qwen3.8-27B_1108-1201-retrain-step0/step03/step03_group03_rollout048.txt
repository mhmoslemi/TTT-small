import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve(a):
    """Self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * c.max() / (S * S)


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        return np.maximum(v - v.mean(), 0.0)
    rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_smoothing(c, beta):
    """Compute smooth approximation of max(c) using log-sum-exp."""
    c_max = c.max()
    log_sum = c_max + np.log(np.sum(np.exp(beta * (c - c_max)))) / beta
    return log_sum


def lse_gradient(a, c, beta):
    """Gradient of smooth max via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def topk_subgradient(a, c, top_k):
    """Subgradient using top-k lags with uniform weight."""
    n = len(a)
    k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[top_indices] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 2).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def make_anti_palindromic_init(n, rng, S):

    a[:half] = h

    a = np.maximum(a, 0.0)
    s = a.sum()

        # The peak in time domain corresponds to certain frequency patterns.
        # A simple approach: compute the gradient as if we're reducing c[peak_lag]
        # c[peak_lag] = sum_i a[i]*a[peak_lag-i]
        # dc[peak_lag]/da[i] = a[peak_lag-i] + a[peak_lag-i] (for i != peak_lag/2)
        #                     = 2*a[i] (for i = peak_lag/2)

        i_min = max(0, peak_lag - (n - 1))
        i_max = min(peak_lag, n - 1)
        if i_min > i_max:
            break
        indices = np.arange(i_min, i_max + 1)
        partners = peak_lag - indices

        # Gradient of c[peak_lag] w.r.t. a
        grad_lag = np.zeros(n)
        for idx, pt in zip(indices, partners):
            if idx == pt:
                grad_lag[idx] += 2.0 * a[idx]
            else:
                grad_lag[idx] += a[pt]
                grad_lag[pt] += a[idx]

        # Also consider second-highest lag
        sorted_lags = np.argsort(c)[::-1]
        lag2 = int(sorted_lags[1])
        if lag2 != peak_lag:
            i_min2 = max(0, lag2 - (n - 1))
            i_max2 = min(lag2, n - 1)
            if i_min2 <= i_max2:
                indices2 = np.arange(i_min2, i_max2 + 1)
                partners2 = lag2 - indices2
                for idx, pt in zip(indices2, partners2):
                    if idx == pt:
                        grad_lag[idx] += 0.5 * 2.0 * a[idx]
                    else:
                        grad_lag[idx] += 0.5 * a[pt]
                        grad_lag[pt] += 0.5 * a[idx]

        gnorm = np.linalg.norm(grad_lag)
        if gnorm < 1e-15:
            break

        # Normalized gradient descent step
        step = S / (n * 5.0) / max(gnorm / (S / n), 0.1)
        a_new = project_simplex(a - step * grad_lag, S)

        c_new = fft_convolve(a_new)
        if c_new.max() < cur:
            a = a_new
        else:
            # Try smaller step
            a_new2 = project_simplex(a - step * 0.3 * grad_lag, S)
            c_new2 = fft_convolve(a_new2)
            if c_new2.max() < cur:
                a = a_new2
            else:
                # Try random direction in the null space of the gradient
                rand = rng.normal(0, 1, n)
                rand = rand - a * (rand @ a) / a.sum()  # orthogonalize
                a_new3 = project_simplex(a + 0.01 * step * grad_lag * 0 + 0.001 * S / np.sqrt(n) * rand, S)
                c_new3 = fft_convolve(a_new3)
                if c_new3.max() < cur:
                    a = a_new3

    return best_local, best_score


def run_pairwise_swap(a_init, n, S, max_time, rng):
    """
    Pairwise swap optimization: try swapping small amounts of mass
    between pairs of indices that are "far apart" in the convolution sense.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    swap_size = S / n * 0.05

    # Precompute which lags are problematic
    c = fft_convolve(a)
    top_lags = set(np.argsort(c)[::-1][:10].tolist())

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
            top_lags = set(np.argsort(c)[::-1][:10].tolist())

        # Try random pairwise swaps
        n_attempts = 50
        for _ in range(n_attempts):
            if time.monotonic() - start > max_time:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            if a[i] < swap_size or a[j] < swap_size:
                continue

            # Propose: move swap_size from i to j (or symmetric)
            delta = min(swap_size, a[i] * 0.3)
            a[i] -= delta
            a[j] += delta
            c_test = fft_convolve(a)
            if c_test.max() < cur:
                if c_test.max() < best_score:
                    best_score = c_test.max()
                    best_local = a.copy()
            else:
                a[i] += delta
                a[j] -= delta

        # Gradually decrease swap size
        swap_size = max(swap_size * 0.999, S / n * 0.005)

        # Periodic refresh of top lags
        if it % 50 == 0:
            c = fft_convolve(a)
            top_lags = set(np.argsort(c)[::-1][:10].tolist())

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp and normalize to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * n)
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except Exception:
        pass

    n = 8192
    S = np.sqrt(2.0 * n)
    print("n=%d, S=%.4f, parent=%s" % (n, S, "yes" if parent is not None else "no"))

    best_score = np.inf
    best_a = None

    def update_best(a_candidate):
        nonlocal best_score, best_a
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
        c = fft_convolve(a_candidate)
        sc = c.max()
        if sc < best_score:
            best_score = sc
            best_a = a_candidate.copy()
            print("  NEW BEST: max(c)=%.8f, obj=%.6f" % (sc, 2.0 * n * sc / S**2))

    # === Phase 1: Coarse optimization (40% of time) ===
    phase1_time = max(30.0, (deadline - time.monotonic()) * 0.35)
    print("=== Phase 1: Coarse FISTA (%.0f sec) ===" % phase1_time)

    # Start from parent or best initialization
    if parent is not None and len(parent) == n:
        a_start = safe_normalize(parent, n)
    else:
        a_start = make_anti_palindromic_init(n, rng, S)

    # Run FISTA
    a1, s1 = run_fista(a_start, n, S, phase1_time, rng, beta_start=10.0, beta_end=300.0)
    a1 = safe_normalize(a1, n)
    update_best(a1)

    # Try a few alternative initializations with short FISTA runs
    if time.monotonic() < deadline - 60:
        alts = [
            make_w_shape_init(n, rng, S),
            make_sinusoidal_init(n, rng, S),
        ]
        per_alt = max(5.0, (deadline - time.monotonic()) * 0.08)
        for alt in alts:
            if time.monotonic() > deadline - 60:
                break
            a_alt, _ = run_fista(alt, n, S, per_alt, rng, beta_start=10.0, beta_end=200.0)
            a_alt = safe_normalize(a_alt, n)
            update_best(a_alt)

    # === Phase 2: Peak-pair coordinate descent (35% of time) ===
    phase2_time = max(20.0, (deadline - time.monotonic()) * 0.35)
    print("=== Phase 2: Peak-pair descent (%.0f sec) ===" % phase2_time)

    a2, _ = run_peak_pair_descent(best_a, n, S, phase2_time, rng,
                                   top_k_lags=5, max_steps_per_lag=100)
    a2 = safe_normalize(a2, n)
    update_best(a2)

    # === Phase 3: Spectral polish (15% of time) ===
    phase3_time = max(10.0, (deadline - time.monotonic()) * 0.20)
    print("=== Phase 3: Spectral polish (%.0f sec) ===" % phase3_time)

    a3, _ = run_spectral_polish(best_a, n, S, phase3_time, rng)
    a3 = safe_normalize(a3, n)
    update_best(a3)

    # === Phase 4: Pairwise swap (10% of time) ===
    phase4_time = max(5.0, (deadline - time.monotonic()) * 0.15)
    print("=== Phase 4: Pairwise swap (%.0f sec) ===" % phase4_time)

    a4, _ = run_pairwise_swap(best_a, n, S, phase4_time, rng)
    a4 = safe_normalize(a4, n)
    update_best(a4)

    # === Phase 5: Final FISTA polish with high beta ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("=== Phase 5: Final FISTA polish (%.0f sec) ===" % remaining)
        a5, _ = run_fista(best_a, n, S, remaining * 0.8, rng,
                           beta_start=100.0, beta_end=1000.0)
        a5 = safe_normalize(a5, n)
        update_best(a5)

    # === Verification ===
    if best_a is None:
        best_a = np.ones(256, dtype=np.float64)

    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:
        best_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in best_a]
    ver = evaluate_sequence(final_list)
    print("=== Final verification: %.6f (internal obj: %.6f), length=%d ===" %
          (ver if np.isfinite(ver) else float('inf'),
           2.0 * len(best_a) * fft_convolve(best_a).max() / best_a.sum()**2,
           len(best_a)))

    if not np.isfinite(ver):
        print("  WARNING: Using fallback")
        final_list = [1.0] * 1024

    return final_list
