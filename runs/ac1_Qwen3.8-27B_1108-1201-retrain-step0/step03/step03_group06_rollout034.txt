```python
    a = S * exp_theta / total

    c = fft_convolve_self(a)

    c_max = c.max()
    scaled = T * (c - c_max)  # shift to avoid overflow
    w = np.exp(scaled)
    S_w = w.sum()
    obj = c_max + np.log(S_w) / T

    w_norm = w / S_w
    # grad_a[m] = 2 * sum_k w_norm[k] * a[k-m]

    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    grad_a = 2.0 * corr[:n]
    
    # Gradient w.r.t. theta
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    
    return obj, grad_theta
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS on smoothed objective
with adaptive temperature, multi-start, and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def softmax_a(theta, S):
    """Convert theta to non-negative a with sum S."""
    t = theta - theta.max()
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * e / s


def compute_smoothed_obj_grad(theta, T, n, S):
    """Compute smoothed objective and gradient w.r.t. theta."""
    a = softmax_a(theta, S)
    c = fft_convolve_self(a)
    
    # Smoothed objective: (1/T) log(sum exp(T*c))
    c_max = c.max()
    scaled = T * (c - c_max)
    w = np.exp(scaled)
    sum_w = w.sum()
    obj = c_max + np.log(sum_w) / T
    
    # Gradient w.r.t. a: grad_a[m] = 2 * sum_k (w[k]/sum_w) * a[k-m]
    w_norm = w / sum_w
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w_norm, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    grad_a = 2.0 * corr[:n]
    
    # Gradient w.r.t. theta: grad_theta[j] = grad_a[j]*a[j] - (a[j]/S)*sum_i grad_a[i]*a[i]
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    
    return obj, grad_theta


def compute_true_obj(a):
    """Compute the true (unsmoothed) objective."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def run_lbfgs_smoothed(theta_init, n, S, T, maxiter=500, tol=1e-10):
    """Run L-BFGS on the smoothed objective."""
    state = {'obj': 0, 'grad': None, 'count': 0}
    
    def fun_and_grad(theta):
        obj, grad = compute_smoothed_obj_grad(theta, T, n, S)
        state['obj'] = obj
        state['grad'] = grad
        state['count'] += 1
        return obj, grad
    
    result = minimize(fun_and_grad, theta_init, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'gtol': tol, 'maxls': 50})
    
    return result.x, state['count']

def coordinate_descent(a, n, S, time_budget, max_sweeps=100):
    """Coordinate descent with FFT recomputation."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(42)
    start = time.monotonic()
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            orig = a[i]
            if orig < 1e-14:
                continue
            
            best_local = orig
            best_local_obj = best_obj
            
            # Try multiplicative perturbations
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]:
                a[i] = orig * factor
                a = normalize_nonneg(a, S)
                c_tmp = fft_convolve_self(a)
                obj_tmp = c_tmp.max()
                if obj_tmp < best_local_obj - 1e-14:
                    best_local_obj = obj_tmp
                    best_local = orig * factor
            
            a[i] = best_local
            a = normalize_nonneg(a, S)
            c = fft_convolve_self(a)
            
            if c.max() < best_obj - 1e-14:
                best_obj = c.max()
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj
```

```python
def peak_directed_step(a, n, S, time_budget, max_iters=1000):
    """Directly minimize the peak by finding the direction that reduces it most."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        # Find the peak
        peak_idx = np.argmax(c)
        
        # Find a direction that reduces c[peak_idx]
        # c[peak_idx] = sum_i a[i] * a[peak_idx - i]
        # dc[peak_idx]/da[m] = 2 * a[peak_idx - m]
        # We want to decrease c[peak_idx], so we want to decrease a in positions
        # where a[peak_idx - m] is large.
        
        # Simple approach: find the m that maximizes a[peak_idx - m] (with valid indices)
        # and decrease a[m], while increasing some other a[j] to maintain the sum.
        
        # Better approach: use the gradient of c[peak_idx]
        # grad[m] = 2 * a[peak_idx - m] (with a[j]=0 for j not in [0,n-1])
        grad_peak = np.zeros(n)
        valid_m = np.arange(max(0, peak_idx - (n-1)), min(n, peak_idx + 1))
        for m in valid_m:
            j = peak_idx - m
            if 0 <= j < n:
                grad_peak[m] = 2.0 * a[j]
        
        # Direction: -grad_peak (to decrease c[peak_idx])
        # But we need to maintain sum(a) = S, so project to the tangent space
        # of the simplex: d = -grad_peak + mean(grad_peak) * ones
        d = -grad_peak
        d -= d.mean()  # tangent to sum constraint
        
        # Also need d to point in a feasible direction (a + t*d >= 0)
        # Find max t such that a + t*d >= 0
        negative = d < 0
        if negative.any():
            max_t = (a[negative] / (-d[negative])).min()
        else:
            max_t = np.inf
        
        # Try a few step sizes
        best_local_obj = best_obj
        best_local_a = a.copy()
        
        for frac in [0.1, 0.05, 0.01, 0.005, 0.001]:
            t = frac * (max_t if max_t < np.inf else 1.0)
            if t <= 0:
                continue
            a_trial = a + t * d
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_convolve_self(a_trial)
            obj_trial = c_trial.max()
            if obj_trial < best_local_obj:
                best_local_obj = obj_trial
                best_local_a = a_trial.copy()
        
        if best_local_obj < best_obj - 1e-14:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_a
            c = fft_convolve_self(a)
        else:
            break
    
    return best_a, best_obj
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS on log-sum-exp smoothed
objective with adaptive temperature scheduling, peak-directed refinement,
multi-start strategy, and coordinate descent polishing.

Key innovations over previous approach:
1. L-BFGS (quasi-Newton) on smoothed objective - much faster convergence than first-order methods
2. Theta (log-softmax) parameterization for natural simplex constraint handling
3. Peak-directed steps that directly attack the maximum autocorrelation lag
4. Adaptive temperature schedule with early stopping per T value
5. Multi-scale perturbation for escaping local minima

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def softmax_a(theta, S):
    """Convert theta to non-negative a with sum S."""
    t = theta - theta.max()
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(theta), S / len(theta))
    return S * e / s


def a_to_theta(a, S):
    """Convert non-negative a to theta parameterization."""
    a_safe = np.maximum(a, 1e-15)
    return np.log(a_safe / a_safe.sum())


def compute_smoothed_obj_grad(theta, T, n, S):
    """Compute smoothed objective and gradient w.r.t. theta.
    
    Smoothed objective: L_T(a) = (1/T) log(sum_k exp(T * c[k]))
    where c = conv(a,a)
    
    Uses the identity: dc[k]/da[m] = 2*a[k-m] (with a[j]=0 for j notin [0,n-1])
    """
    a = softmax_a(theta, S)
    c = fft_convolve_self(a)
    
    # Smoothed objective with overflow protection
    c_max = c.max()
    scaled = T * (c - c_max)
    w = np.exp(scaled)
    sum_w = w.sum()
    obj = c_max + np.log(sum_w) / T
    
    # Gradient w.r.t. a: grad_a[m] = 2 * sum_k (w[k]/sum_w) * a[k-m]
    w_norm = w / sum_w
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w_norm, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    grad_a = 2.0 * corr[:n]
    
    # Gradient w.r.t. theta via chain rule:
    # dL/dtheta[j] = grad_a[j]*a[j] - (a[j]/S)*sum_i grad_a[i]*a[i]
    dot = np.dot(grad_a, a)
    grad_theta = grad_a * a - (a / S) * dot
    
    return obj, grad_theta


def compute_true_obj_normalized(a):
    """Compute max(conv(a,a)) assuming sum(a) = sqrt(2n)."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return c.max()


def compute_true_obj_full(a):
    """Compute the full evaluation objective: 2*n*max_conv/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

            if time.monotonic() > deadline - 30:
                break
            
            a_init = normalize_nonneg(a_init.copy(), S)
            t0 = time.monotonic()
            a_best, obj_norm = run_adaptive_lbfgs(
                a_init, n, S, per_init_time,
                T_values=[20, 100, 500, 2000, 10000],
                maxiter_per_T=200
            )
            elapsed = time.monotonic() - t0
            true_obj = compute_true_obj_full(a_best)
            
            if true_obj < global_best:
                global_best = true_obj
                global_a = a_best.copy()
                print(f"  Init '{name}': obj={true_obj:.8f} ({elapsed:.1f}s) *NEW BEST*")
            else:
                print(f"  Init '{name}': obj={true_obj:.8f} ({elapsed:.1f}s)")
    
    # ============ PHASE 4: High-T L-BFGS + peak-directed (15% time) ============
    if global_a is not None:
        p4_budget = min(300, 0.15 * total_budget)
        remaining_for_p4 = min(p4_budget, deadline - time.monotonic() - 30)
        if remaining_for_p4 > 20:
            print(f"\n=== Phase 4: High-T refinement ({remaining_for_p4:.0f}s) ===")
            a = global_a.copy()
            a = normalize_nonneg(a, S)
            
            # Very high T L-BFGS
            t0 = time.monotonic()
            a_best, obj_norm = run_adaptive_lbfgs(
                a, n, S, remaining_for_p4 * 0.5,
                T_values=[10000, 30000, 100000, 300000],
                maxiter_per_T=500
            )
            elapsed1 = time.monotonic() - t0
            true_obj = compute_true_obj_full(a_best)
            print(f"  High-T L-BFGS: obj={true_obj:.8f} ({elapsed1:.1f}s)")
            
            if true_obj < global_best:
                global_best = true_obj
                global_a = a_best.copy()
            
            # Peak-directed on the result
            remaining2 = remaining_for_p4 * 0.5 - elapsed1
            if remaining2 > 5:
                a_best2, obj_norm2 = peak_directed_refine(
                    global_a, n, S, remaining2, max_iters=3000
                )
                true_obj2 = compute_true_obj_full(a_best2)
                print(f"  Peak-directed: obj={true_obj2:.8f}")
                
                if true_obj2 < global_best:
                    global_best = true_obj2
                    global_a = a_best2.copy()
                    print(f"  *NEW BEST*")
    
    # ============ PHASE 5: Coordinate descent polish (5% time) ============
    if global_a is not None:
        p5_budget = min(120, 0.05 * total_budget)
        remaining_for_p5 = min(p5_budget, deadline - time.monotonic() - 10)
        if remaining_for_p5 > 5:
            print(f"\n=== Phase 5: Coordinate descent ({remaining_for_p5:.0f}s) ===")
            a = global_a.copy()
            a = normalize_nonneg(a, S)
            
            a_best, obj_norm = coordinate_descent_fast(a, n, S, remaining_for_p5, max_sweeps=200)
            true_obj = compute_true_obj_full(a_best)
            print(f"  CD: obj={true_obj:.8f}")
            
            if true_obj < global_best:
                global_best = true_obj
                global_a = a_best.copy()
    
    # ============ PHASE 6: Final perturbation attempts (5% time) ============
    if global_a is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            print(f"\n=== Phase 6: Final attempts ({remaining:.0f}s) ===")
            n_rounds = max(2, int(remaining / 20))
            per_round = remaining / (n_rounds + 1)
            
            for r in range(n_rounds):
                if time.monotonic() > deadline - 5:
                    break
                
                # Different perturbation strategies
                strategy = r % 3
                if strategy == 0:
                    # Gaussian perturbation
                    noise = 0.003 * (1 + 0.3 * r) * rng.standard_normal(n)
                    a_try = global_a + noise
                elif strategy == 1:
                    # Coordinate swap
                    a_try = global_a.copy()
                    i, j = rng.integers(0, n, 2)
                    delta = 0.01 * S / n
                    a_try[i] += delta
                    a_try[j] -= delta
                else:
                    # Frequency-domain perturbation
                    A = fft(global_a)
                    phase_noise = 0.001 * rng.standard_normal(len(A))
                    A += phase_noise
                    a_try = np.real(ifft(A))[:n]
                
                a_try = normalize_nonneg(a_try, S)
                
                # Quick L-BFGS
                theta = a_to_theta(a_try, S)
                T_pick = [500, 2000, 10000][r % 3]
                theta_new, _ = run_lbfgs_smoothed(theta, n, S, T_pick, maxiter=200, gtol=1e-10)
                a_final = softmax_a(theta_new, S)
                true_obj = compute_true_obj_full(a_final)
                
                if true_obj < global_best:
                    global_best = true_obj
                    global_a = a_final.copy()
                    print(f"  Round {r} (strategy={strategy}): obj={true_obj:.8f} *NEW BEST*")
                else:
                    print(f"  Round {r} (strategy={strategy}): obj={true_obj:.8f}")
    
    # ============ PHASE 7: Try different n (if time permits) ============
    if time.monotonic() < deadline - 60:
        for n_try in [3072, 5120, 8192]:
            if time.monotonic() > deadline - 30:
                break
            S_try = np.sqrt(2.0 * n_try)
            budget_n = min(30, deadline - time.monotonic() - 15)
            if budget_n < 10:
                break
            
            print(f"\n=== Trying n={n_try} ({budget_n:.0f}s) ===")
            if global_a is not None:
                # Interpolate current best to new length
                old_idx = np.linspace(0, 1, len(global_a))
                new_idx = np.linspace(0, 1, n_try)
                a_init = np.interp(new_idx, old_idx, global_a)
            else:
                a_init = np.full(n_try, S_try / n_try)
            a_init = normalize_nonneg(a_init, S_try)
            
            theta = a_to_theta(a_init, S_try)
            # Quick optimization
            for T in [100, 1000]:
                theta, _ = run_lbfgs_smoothed(theta, n_try, S_try, T, maxiter=100, gtol=1e-8)
            a_final = softmax_a(theta, S_try)
            true_obj = compute_true_obj_full(a_final)
            print(f"  n={n_try}: obj={true_obj:.8f}")
            
            if true_obj < global_best:
                global_best = true_obj
                global_a = a_final.copy()
                print(f"  *NEW BEST with different n!*")
    
    # Final verification
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\n{'='*60}")
    print(f"FINAL: computed_best={global_best:.8f}, evaluate_sequence={ver:.8f}")
    print(f"Sequence length: {len(final_a)}")
    print(f"{'='*60}")
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.8f}")
    
    return final_a
```
