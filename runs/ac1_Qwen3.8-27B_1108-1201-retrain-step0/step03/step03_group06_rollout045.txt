"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LSE-smoothed L-BFGS-B optimization
with log-softmax parameterization, peak-targeting coordinate descent, and
multi-scale T-schedule. Key innovations over previous approach:
1. L-BFGS-B quadratic convergence on LSE approximation (vs subgradient)
2. Log-softmax parameterization for smooth constraint handling
3. Adaptive T-schedule with L-BFGS restarts
4. Peak-targeted coordinate descent for final polishing
5. Multiple n-values exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def lse_objective_and_grad(theta, T, n, S):
    a = S * softmax(theta)
    c = fft_convolve_self(a)
    c_scaled = T * c
    c_max = c_scaled.max()
    # Numerical stability
    w = np.exp(c_scaled - c_max)
    w_sum = w.sum()
    if w_sum < 1e-300:
        obj = c_max / T
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        obj = (c_max + np.log(w_sum)) / T
        w /= w_sum
    
    # Gradient of LSE wrt a: sum_j w_j * dc_j/da
    # dc_j/da_i = a_{j-i} (for 0<=j-i<n) + a_{i-j} (for 0<=i-j<n)
    # This is: for each lag j, the gradient contribution to a_i is w_j * (a[(j-i)%...])
    # More precisely: c_j = sum_i a_i * a_{j-i} (with a_k=0 outside [0,n-1])
    # dc_j/da_i = a_{j-i} + a_{i-j} (where a_k = 0 if k < 0 or k >= n)
    # So dLSE/da_i = sum_j w_j * (a_{j-i} + a_{i-j})
    # = sum_j w_j * a_{(j-i)} + sum_j w_j * a_{(i-j)}
    # First term: for fixed i, sum_j w_j * a_{j-i} = (w * a)[i] at appropriate offset
    # This is cross-correlation of w and a
    
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # cross_corr(w, a)[i] = sum_j w[j] * a[j-i]
    corr = np.real(ifft(W * np.conj(A)))
    g_a = 2.0 * corr[:n]
    
    # Chain rule for theta
    dot = np.dot(g_a, a)
    g_theta = a * (g_a - dot / S)
    
    return obj, g_theta


def lse_objective_only(theta, T, n, S):
    a = S * softmax(theta)
    c = fft_convolve_self(a)
    c_scaled = T * c
    c_max = c_scaled.max()
    w = np.exp(c_scaled - c_max)
    return (c_max + np.log(w.sum())) / T


def run_lbfgs_lse(a_init, n, S, T, maxiter=200, time_limit=None, start_time=None):
    """Run L-BFGS-B on LSE approximation with given T."""
    theta = np.log(np.maximum(a_init, 1e-30))
    theta = theta - theta.mean()  # center
    
    if time_limit is None:
        start_time = time.monotonic()
    
    def fun_and_grad(theta):
        if time_limit is not None and time.monotonic() - start_time > time_limit:
            raise TimeoutError()
        return lse_objective_and_grad(theta, T, n, S)
    
    try:
        result = minimize(
            fun_and_grad, theta, method='L-BFGS-B',
            jac=True, options={'maxiter': maxiter, 'maxfun': maxiter * 2,
                             'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
        )
        a = S * softmax(result.x)
        return a, result.fun, result.success
    except (TimeoutError, Exception) as e:
        a = S * softmax(theta)
        return a, None, False


def peak_targeted_cd(a, n, S, time_budget, top_k=10):
    """Coordinate descent targeting the top-k peaks."""
    a = a.copy()
    start = time.monotonic()
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    rng = np.random.default_rng(99)
    
    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        # Find top-k peaks
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            orig = a[i]
            if orig < 1e-10:
                continue
            
            best_factor = 1.0
            best_local_obj = cur_obj
            
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]:
                a[i] = orig * factor
                s = a.sum()
                if s < 1e-15:
                    a[i] = orig
                    continue
                a = a * (S / s)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-12:
                    best_local_obj = obj_check
                    best_factor = factor
            
            a[i] = orig * best_factor
            a = a * (S / a.sum())
            
            if best_factor != 1.0:
                improved = True
                if best_local_obj < best_obj - 1e-12:
                    best_obj = best_local_obj
                    best_a = a.copy()
        
        if not improved and sweep >= 2:
            break
    
    return best_a, best_obj


def get_normalized(a, S):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LSE-smoothed L-BFGS-B optimization
with log-softmax parameterization, adaptive T-schedule, peak-targeted coordinate
descent, and multi-start strategy. Key innovations over previous approach:
1. L-BFGS-B quadratic convergence on LSE approximation (faster than subgradient)
2. Log-softmax parameterization for smooth constraint handling
3. Focused T-schedule in the range where L-BFGS is most effective
4. Peak-targeted coordinate descent for final precision
5. Diverse initializations and n-value exploration
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_stable(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(x), 1.0 / len(x))
    return e / s


def get_normalized(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_objective_and_grad(theta, T, n, S):
    """Compute LSE(T*conv(a,a)) and its gradient w.r.t. theta, where a = S*softmax(theta)."""
    a = S * softmax_stable(theta)
    c = fft_convolve_self(a)
    c_scaled = T * c
    c_max = c_scaled.max()
    shifted = c_scaled - c_max
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        obj = c_max / T
        w = np.zeros(len(c))
        idx = np.argmax(c)
        w[idx] = 1.0
    else:
        obj = (c_max + np.log(w_sum)) / T
        w = w / w_sum

    # Gradient of LSE wrt a: sum_j w_j * dc_j/da
    # dc_j/da_i = a_{j-i} + a_{i-j} (with a_k=0 outside [0,n-1])
    # This equals 2 * cross_corr(w, a)[i]
    fsize = 1 << (2 * n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    g_a = 2.0 * corr[:n]

    # Chain rule: a_i = S * exp(theta_i) / Z
    # da_i/dtheta_j = a_i * (delta_ij - a_j/S)
    # dLSE/dtheta_j = a_j * (g_a_j - sum_i g_a_i * a_i / S)
    dot = np.dot(g_a, a)
    g_theta = a * (g_a - dot / S)

    return obj, g_theta


def run_lbfgs(a_init, n, S, T, maxiter=150, time_limit=30.0):
    """Run L-BFGS-B on LSE approximation with given T."""
    a_safe = np.maximum(a_init, 1e-15)
    theta = np.log(a_safe / a_safe.sum())
    
    start = time.monotonic()
    
    def fun_and_grad(th):
        if time.monotonic() - start > time_limit:
            raise TimeoutError()
        return lse_objective_and_grad(th, T, n, S)

    try:
        result = minimize(
            fun_and_grad, theta, method='L-BFGS-B',
            jac=True,
            options={'maxiter': maxiter, 'maxfun': maxiter * 3,
                     'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30,
                     'maxcor': 20}
        )
        a_out = S * softmax_stable(result.x)
        return a_out, result.fun, result.success, result.n_iter
    except (TimeoutError, Exception):
        # Return current theta if timeout
        a_out = S * softmax_stable(theta)
        return a_out, None, False, 0


def peak_targeted_cd(a, n, S, time_budget, top_k=10, rng=None):
    """Coordinate descent targeting the top-k peaks with adaptive factors."""
    if rng is None:
        rng = np.random.default_rng(42)
    a = a.copy().astype(np.float64)
    a = get_normalized(a, S)

    best_obj = c.max()
    best_a = a.copy()
    
    sweep = 0
    no_improve_sweeps = 0
    
    while time.monotonic() - start < time_budget:
        sweep += 1
        c = fft_convolve_self(a)
        cur_obj = c.max()
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps > 3:
                break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:

            if orig < 1e-14:

            best_factor = 1.0

                if s < 1e-14:

                obj_check = c_check.max()

                s = a.sum()
                a = a * (S / s)

def nesterov_lse(a_init, n, S, T, time_budget, lr_init=0.01, momentum=0.9):
    """Nesterov accelerated gradient on LSE with log-softmax parameterization."""
    a = np.maximum(a_init, 1e-15)
    theta = np.log(a / a.sum())
    y = theta.copy()
    beta = momentum
    lr = lr_init
    start = time.monotonic()
    
    best_a = a_init.copy()
    best_true_obj = fft_convolve_self(a_init).max()
    
    step = 0
    while time.monotonic() - start < time_budget:
        step += 1
        a_y = S * softmax_stable(y)
        obj, grad = lse_objective_and_grad(y, T, n, S)
        
        theta = y - (beta - 1.0) / (1.0 + beta) * (y - theta)
        y = theta - lr * grad
        
        a_cur = S * softmax_stable(theta)
        true_obj = fft_convolve_self(a_cur).max()
        if true_obj < best_true_obj:
            best_true_obj = true_obj
            best_a = a_cur.copy()
        
        # Decay learning rate
        lr = lr_init / (1.0 + 0.01 * step)
    
    return best_a, best_true_obj

        a_cd, obj_cd = peak_targeted_cd(a, n_base, S_base, cd_b1, top_k=8, rng=rng)
        a_cd = get_normalized(a_cd, S_base)
        obj_cd = evaluate_sequence(a_cd.tolist())
        print("  CD after Phase1: obj=%.8f" % obj_cd)
        if np.isfinite(obj_cd) and obj_cd < global_best:
            global_best = obj_cd
            global_a = a_cd.copy()
            a = a_cd
    
    # ============ Phase 2: Nesterov + L-BFGS hybrid at different T ============
    phase2_budget = 0.25 * total_budget
    if time.monotonic() < deadline - 60 and phase2_budget > 30:
        print("\n=== Phase 2: Nesterov + L-BFGS hybrid (budget=%.0fs) ===" % phase2_budget)
        a = global_a.copy()
        
        # Nesterov at moderate T
        nest_budget = phase2_budget * 0.3
        a_nest, obj_nest = nesterov_lse(a, n_base, S_base, T=200, time_budget=nest_budget, lr_init=0.005, momentum=0.9)
        a_nest = get_normalized(a_nest, S_base)
        obj_nest = evaluate_sequence(a_nest.tolist())
        print("  Nesterov T=200: obj=%.8f" % obj_nest)
        if np.isfinite(obj_nest) and obj_nest < global_best:
            global_best = obj_nest
            global_a = a_nest.copy()
            a = a_nest
        
        # L-BFGS at T=500 from Nesterov result
        lbfgs_b2 = phase2_budget * 0.3
        a_new, _, _, _ = run_lbfgs(a, n_base, S_base, 500, maxiter=80, time_limit=lbfgs_b2)
        a_new = get_normalized(a_new, S_base)
        obj = evaluate_sequence(a_new.tolist())
        print("  L-BFGS T=500 (from Nest): obj=%.8f" % obj)
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_new.copy()
            a = a_new
        
        # CD again
        cd_b2 = phase2_budget * 0.25
        a_cd, obj_cd = peak_targeted_cd(a, n_base, S_base, cd_b2, top_k=5, rng=rng)
        a_cd = get_normalized(a_cd, S_base)
        obj_cd = evaluate_sequence(a_cd.tolist())
        print("  CD after Phase2: obj=%.8f" % obj_cd)
        if np.isfinite(obj_cd) and obj_cd < global_best:
            global_best = obj_cd
            global_a = a_cd.copy()
            a = a_cd
    
    # ============ Phase 3: Alternative initializations ============
    phase3_budget = 0.15 * total_budget
    if time.monotonic() < deadline - 30 and phase3_budget > 20:
        print("\n=== Phase 3: Alternative initializations (budget=%.0fs) ===" % phase3_budget)
        
        i = np.arange(n_base)
        n_inits = 4
        per_init = phase3_budget / n_inits
        
        # Different structured initializations
        inits = []
        
        # 1. Given + multiplicative noise
        mult_noise = np.exp(0.02 * rng.standard_normal(n_base))
        inits.append(("mult_noise", global_a * mult_noise))
        
        # 2. Cosine window (taper at edges)
        taper = 0.5 + 0.5 * np.cos(2 * np.pi * i / (2 * n_base))
        inits.append(("cosine_taper", global_a * (0.5 + 0.5 * taper)))
        
        # 3. Random with same mean
        inits.append(("random", rng.random(n_base)))
        
        # 4. Given with edge boost
        edge_boost = 1.0 + 0.3 * np.cos(2 * np.pi * i / n_base)
        inits.append(("edge_boost", global_a * edge_boost))
        
        for name, a_init in inits:
            if time.monotonic() > deadline - 30:
                break
            a_init = get_normalized(a_init, S_base)
            
            # Quick L-BFGS
            a_new, _, _, _ = run_lbfgs(a_init, n_base, S_base, 100, maxiter=40, time_limit=per_init * 0.7)
            a_new = get_normalized(a_new, S_base)
            obj = evaluate_sequence(a_new.tolist())
            print("  Init '%s': obj=%.8f" % (name, obj))
            
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_new.copy()
                # Refine
                a_new2, _, _, _ = run_lbfgs(a_new, n_base, S_base, 300, maxiter=40, time_limit=per_init * 0.3)
                a_new2 = get_normalized(a_new2, S_base)
                obj2 = evaluate_sequence(a_new2.tolist())
                print("  Init '%s' refined: obj=%.8f" % (name, obj2))
                if np.isfinite(obj2) and obj2 < global_best:
                    global_best = obj2
                    global_a = a_new2.copy()
    
    # ============ Phase 4: Try slightly different n ============
    phase4_budget = 0.10 * total_budget
    if time.monotonic() < deadline - 20 and phase4_budget > 15:
        print("\n=== Phase 4: Different n values (budget=%.0fs) ===" % phase4_budget)
        
        n_options = [n_base - 256, n_base + 256, n_base + 512]
        n_options = [x for x in n_options if 1024 <= x <= 12000]
        n_options = list(set(n_options))[:3]
        
        per_n = phase4_budget / max(1, len(n_options))
        
        for n_try in n_options:
            if time.monotonic() > deadline - 20:
                break
            S_try = np.sqrt(2.0 * n_try)
            
            # Interpolate global_a to new length
            if n_try > n_base:
                x_old = np.linspace(0, 1, n_base)
                x_new = np.linspace(0, 1, n_try)
                a_resampled = np.interp(x_new, x_old, global_a)
            else:
                x_old = np.linspace(0, 1, n_base)
                x_new = np.linspace(0, 1, n_try)
                a_resampled = np.interp(x_new, x_old, global_a)
            
            a_resampled = get_normalized(a_resampled, S_try)
            
            a_new, _, _, _ = run_lbfgs(a_resampled, n_try, S_try, 100, maxiter=40, time_limit=per_n * 0.8)
            a_new = get_normalized(a_new, S_try)
            obj = evaluate_sequence(a_new.tolist())
            print("  n=%d: obj=%.8f" % (n_try, obj))
            
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_new.copy()
    
    # ============ Phase 5: Final intense polishing ============
    remaining = deadline - time.monotonic() - 15
    if global_a is not None and remaining > 10:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        a = global_a.copy()
        
        print("\n=== Phase 5: Final polish (budget=%.0fs) ===" % remaining)
        
        # Very high T L-BFGS for precision
        t_hi = remaining * 0.35
        a_new, _, _, _ = run_lbfgs(a, n, S, 2000, maxiter=100, time_limit=t_hi)
        a_new = get_normalized(a_new, S)
        obj = evaluate_sequence(a_new.tolist())
        print("  High-T L-BFGS (T=2000): obj=%.8f" % obj)
        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_new.copy()
            a = a_new
        
        # Final CD with very tight tolerance
        t_cd = remaining * 0.35
        a_cd, obj_cd = peak_targeted_cd(a, n, S, t_cd, top_k=3, rng=rng)
        a_cd = get_normalized(a_cd, S)
        obj_cd = evaluate_sequence(a_cd.tolist())
        print("  Final CD (top3): obj=%.8f" % obj_cd)
        if np.isfinite(obj_cd) and obj_cd < global_best:
            global_best = obj_cd
            global_a = a_cd.copy()
        
        # One more L-BFGS at T=1000
        t_final = deadline - time.monotonic() - 10
        if t_final > 5:
            a_new, _, _, _ = run_lbfgs(global_a, n, S, 1000, maxiter=80, time_limit=t_final * 0.9)
            a_new = get_normalized(a_new, S)
            obj = evaluate_sequence(a_new.tolist())
            print("  Ultra-final L-BFGS (T=1000): obj=%.8f" % obj)
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_new.copy()
    
    # Final verification
    if global_a is None:
        global_a = np.ones(256)
    
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)
    
    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print("\nFINAL: n=%d, obj=%.8f" % (len(final_a), ver))
    
    if not np.isfinite(ver):
        final_a = [1.0] * 256
        print("Fallback used: obj=%.4f" % evaluate_sequence(final_a))
    
    return final_a
