def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def obj_value(a, c=None):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def power_spectrum_equalization(a, n_iters=100, aggressiveness=0.5):

    n = len(a)

    A = fft(a, fsize)
    # DC component is fixed (determines sum)
    P = np.abs(A)**2

        # Compute current convolution
        c = np.real(ifft(A * A))
        c = c[:2*n-1]
        
        # Find peaks in convolution and try to reduce them
        # by equalizing the power spectrum
        c_max = c.max()
        c_mean = c.mean()

def peak_chase_polish(a0, time_budget, lr=0.003):
    """Focused single-peak reduction."""
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = obj_value(a, c)
    best_a = a.copy()
    v = np.zeros(n)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        # Find the worst peak
        k_max = np.argmax(c)
        
        # Exact gradient at k_max
        grad = np.zeros(n)
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[k_max - m]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        v = 0.95 * v + 0.05 * grad
        
        a = a * np.exp(-lr * v)
        a = np.maximum(a, 1e-300)
        a = a * (S / a.sum())
        c = fft_convolve_self(a)
        cur_obj = obj_value(a, c)
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
    
    return best_a, best_obj
