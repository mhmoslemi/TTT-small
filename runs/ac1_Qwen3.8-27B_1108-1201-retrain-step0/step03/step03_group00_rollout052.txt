    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def subgradient_topk(a, c, k, power=1.0):
    """Compute top-k weighted subgradient of max(c).
    
    The subgradient of max_j c_j is a convex combination of gradients of active pieces.
    grad_k(c)[i] = 2*a[c_k - i] (cross-correlation interpretation).
    The weighted mixture is 2 * cross_corr(weights, a).
    """
    n = len(a)
    top_k = min(k, len(c))
    idx = np.argsort(c)[::-1][:top_k]
    vals = np.maximum(c[idx], 1e-30)
    weights = np.power(vals, power)
    wsum = weights.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    weights /= wsum

    weights_full = np.zeros(len(c))
    weights_full[idx] = weights

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    W = fft(weights_full, fsize)
    # Cross-correlation: ifft(fft(w) * conj(fft(a)))
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def compute_objective(a):
    """Compute the actual objective 2*n*max(b)/S^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (S * S)


def fista_topk(a0, n, S, max_time, rng, k_start=8, k_end=256, power=2.0,
               step_mode="adaptive", restart_interval=2000):
    """FISTA with top-k subgradient and adaptive step size.
    
    Uses continuation on k: start with large k (smooth), decrease to small k (sharp).
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    a = normalize_sum(a, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0

    # Estimate problem parameters
    c_init = fft_convolve_self(y)
    g_init = subgradient_topk(y, c_init, k_start)
    G_est = np.linalg.norm(g_init)
    if G_est < 1e-15:
        G_est = 1.0
    R_est = np.linalg.norm(a - np.full(n, S / n))
    if R_est < 1e-15:
        R_est = S / np.sqrt(n)

    base_step = 2.0 * R_est / (G_est * (1.0 + 1e-6))

    while time.monotonic() - start < max_time:
        it += 1

        # Continuation schedule for k: geometric from k_start to k_end
        progress = min(1.0, it / (restart_interval * 3))
        k = max(2, int(k_start * np.exp(np.log(k_end / k_start) * progress)))
        if k > len(c_init):
            k = len(c_init)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Compute subgradient
        g = subgradient_topk(y, c, k, power=power)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        # Adaptive step size
        if step_mode == "adaptive":
            eta = base_step / np.sqrt(1.0 + 0.1 * it)
            # Clamp to avoid too-large steps
            eta = min(eta, 2.0 * S / np.max(a) / max(k, 1))
        else:
            eta = base_step / np.sqrt(1.0 + 0.05 * it)

        x_new = project_simplex(y - eta * g, S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic restart with noise to escape stagnation
        if it - last_improve > restart_interval:
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it

    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def swap_optimal_delta(a, b, i, j, n_bisection=40):
    """Find optimal delta for transferring mass from i to j.
    
    After the swap, a[i] -= delta, a[j] += delta.
    The convolution changes by: db[k] = 2*delta*(a[k-j] - a[k-i])
    We want to minimize max_k (b[k] + delta * d[k]).
    
    Returns the optimal delta (clamped to [0, a[i]] for forward swap).
    """
    n = len(a)
    size = 2 * n - 1

    # Compute direction d[k] = 2*(a[k-j] - a[k-i])
    d = np.zeros(size)
    # a[k-j] is nonzero for k in [j, j+n-1]
    lo_j = j
    hi_j = min(size - 1, j + n - 1)
    d[lo_j:hi_j+1] += a[:hi_j+1-lo_j]
    # a[k-i] is nonzero for k in [i, i+n-1]
    lo_i = i
    hi_i = min(size - 1, i + n - 1)
    d[lo_i:hi_i+1] -= a[:hi_i+1-lo_i]
    d *= 2.0

    # Bisection on the subgradient of f(delta) = max_k(b[k] + delta*d[k])
    delta_lo = 0.0
    delta_hi = a[i]

    if delta_hi <= 1e-15:
        return 0.0

    tol = 1e-12

    for _ in range(n_bisection):
        mid = (delta_lo + delta_hi) / 2.0
        vals = b + mid * d
        max_idx = np.argmax(vals)
        max_val = vals[max_idx]

        # Find all (approximately) active indices
        active = np.where(vals >= max_val - 1e-8 * max(1.0, max_val))[0]
        d_active = d[active]

        if d_active.max() < -tol:
            # f is decreasing at mid, move right
            delta_lo = mid
        elif d_active.min() > tol:
            # f is increasing at mid, move left
            delta_hi = mid
        else:
            # 0 is in subgradient, optimal found
            break

    return (delta_lo + delta_hi) / 2.0


def pairwise_cd(a, n, S, max_time, rng, n_swaps_target=50000):
    """Pairwise coordinate descent using 2-opt swaps.
    
    Repeatedly select pairs (i,j) and find optimal mass transfer.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = normalize_sum(a, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    b = fft_convolve_self(a)
    best_max = b.max()

    # Precompute for faster swap evaluation
    n_active = np.sum(a > 1e-10)
    active_idx = np.where(a > 1e-10)[0]

    swaps_done = 0
    recent_improvements = 0
    stagnant = 0

    while time.monotonic() - start < max_time and swaps_done < n_swaps_target:
        # Select pair (i, j)
        # Strategy: i from high-mass coordinates, j from low-mass
        r = rng.random()
        if n_active > 1:
            if r < 0.5:
                # Random active i, random j
                i = active_idx[rng.integers(len(active_idx))]
                j = rng.integers(n)
            elif r < 0.8:
                # High-mass i, low-mass j
                i = active_idx[rng.integers(len(active_idx))]
                # Pick j from bottom half of mass distribution
                sorted_idx = np.argsort(a)
                j = sorted_idx[rng.integers(n // 2)]
            else:
                # Two random active coordinates
                i = active_idx[rng.integers(len(active_idx))]
                j = active_idx[rng.integers(len(active_idx))]
        else:
            break

        if i == j:
            continue

        # Try forward swap (i -> j)
        delta_f = swap_optimal_delta(a, b, i, j)

        # Try reverse swap (j -> i)
        if a[j] > 1e-15:
            delta_r = swap_optimal_delta(a, b, j, i)
        else:
            delta_r = 0.0

        # Choose the better swap
        # Evaluate: what's the new max after each swap?
        best_delta = 0.0
        best_new_max = b.max()

        if delta_f > 1e-12:
            # Quick evaluation
            size = 2 * n - 1
            d = np.zeros(size)
            d[j:j+n] += a[:min(n, size-j)]
            d[i:i+n] -= a[:min(n, size-i)]
            d *= 2.0
            new_b_f = b + delta_f * d
            new_max_f = new_b_f.max()
            if new_max_f < best_new_max:
                best_new_max = new_max_f
                best_delta = delta_f

        if delta_r > 1e-12:
            size = 2 * n - 1
            d = np.zeros(size)
            d[i:i+n] += a[:min(n, size-i)]
            d[j:j+n] -= a[:min(n, size-j)]
            d *= 2.0
            new_b_r = b + delta_r * d
            new_max_r = new_b_r.max()
            if new_max_r < best_new_max:
                best_new_max = new_max_r
                # Reverse: transfer from j to i means delta_r for (j,i) swap
                # Apply: a[j] -= delta_r, a[i] += delta_r
                best_delta = -delta_r  # negative means reverse

        if best_delta != 0.0:
            if best_delta > 0:
                a[i] -= best_delta
                a[j] += best_delta
            else:
                a[j] -= abs(best_delta)
                a[i] += abs(best_delta)

            # Ensure non-negativity
            a = np.maximum(a, 0.0)

            # Update convolution incrementally (exact)
            b = fft_convolve_self(a)
            cur_max = b.max()

            if cur_max < best_max - 1e-12:
                recent_improvements += 1
                stagnant = 0
            else:
                # Revert if no improvement
                if best_delta > 0:
                    a[i] += best_delta
                    a[j] -= best_delta
                else:
                    a[j] += abs(best_delta)
                    a[i] -= abs(best_delta)
                a = np.maximum(a, 0.0)
                b = fft_convolve_self(a)
                stagnant += 1

            if cur_max < best:
                best = cur_max
                best_a = a.copy()
        else:
            stagnant += 1

        swaps_done += 1

        # Periodic full recompute and normalization
        if swaps_done % 5000 == 0:
            a = normalize_sum(a, n)
            b = fft_convolve_self(a)
            best_max = b.max()
            active_idx = np.where(a > 1e-10)[0]
            n_active = len(active_idx)

            if stagnant > 20000:
                # Add noise to escape
                noise = rng.normal(0, 1e-4 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
                b = fft_convolve_self(a)
                stagnant = 0
                active_idx = np.where(a > 1e-10)[0]
                n_active = len(active_idx)

    return best_a, best


def fista_polish(a0, n, S, max_time, rng):
    """Final high-precision FISTA polish with very focused subgradient."""
    a = normalize_sum(a0, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0

    # Start with large k and rapidly decrease
    c0 = fft_convolve_self(y)
    g0 = subgradient_topk(y, c0, 64)
    G_est = np.linalg.norm(g0)
    if G_est < 1e-15:
        G_est = 1.0
    base_step = 1.0 / (G_est * 0.5)

    while time.monotonic() - start < max_time:
        it += 1

        # Rapidly decreasing k
        k = max(2, min(128, 128 // (1 + it // 50)))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()

        g = subgradient_topk(y, c, k, power=2.0)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        eta = base_step / (1.0 + 0.01 * it)
        x_new = project_simplex(y - eta * g, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Raised cosine window
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    # Hanning
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    # Hamming
    inits.append(0.54 - 0.46 * np.cos(2 * np.pi * tt))
    # Gaussian
    inits.append(np.exp(-8.0 * (tt - 0.5) ** 2))
    # Sin^2
    inits.append(np.sin(np.pi * tt) ** 2)
    # Multi-sinusoidal
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt) + 0.2 * np.cos(2 * np.pi * 7 * tt))
    # Random
    inits.append(rng.random(n) + 0.1)
    # Exponential random
    inits.append(np.exp(rng.normal(0, 0.5, n)))

    return [np.maximum(x, 0.01) for x in inits]


def downsample_upsample(a, n_target):
    """Downsample a to n_target/4 then interpolate back to n_target."""
    n = len(a)
    n_coarse = max(64, n_target // 4)
    # Average pooling
    blocks = n // n_coarse
    if blocks < 2:
        return a.copy()
    a_small = a[:blocks * n_coarse].reshape(n_coarse, blocks).mean(axis=1)
    # Interpolate
    t_old = np.linspace(0, 1, n_coarse)
    t_new = np.linspace(0, 1, n_target)
    return np.interp(t_new, t_old, a_small)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    total_deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Try to load the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose primary length
    if given is not None and 1024 <= len(given) <= 32768:
        main_len = len(given)
    else:
        main_len = 8192

    print("=== Starting optimization ===")
    print("Budget: %.0fs, Primary length: %d" % (budget_s, main_len))
    if given is not None:
        print("Loaded given sequence of length %d" % len(given))

    # Phase 1: Multi-start exploration with FISTA (50% of budget)
    phase1_time = (total_deadline - time.monotonic()) * 0.50
    p1_start = time.monotonic()

    # Build initialization list
    n = main_len
    S = np.sqrt(2.0 * n)

    init_list = []
    if given is not None:
        # Use the given sequence directly
        init_list.append(("given", normalize_sum(given, n)))
        # Downsampled version
        if n > 2048:
            coarse = downsample_upsample(given, n)
            init_list.append(("coarse", normalize_sum(coarse, n)))

    # Add structured inits
    struct_inits = generate_structured_inits(n, rng)
    for i, si in enumerate(struct_inits[:4]):
        init_list.append(("struct_%d" % i, normalize_sum(si, n)))

    # Random init
    init_list.append(("rand", normalize_sum(rng.random(n) + 0.1, n)))

    n_inits = len(init_list)
    per_init_time = phase1_time / n_inits
    # Give "given" more time
    if given is not None:
        per_init_time *= 0.8

    print("Phase 1: %d initializations, %.0fs each (total %.0fs)" %
          (n_inits, per_init_time, phase1_time))

    for idx, (name, arr) in enumerate(init_list):
        if time.monotonic() > total_deadline - (total_deadline - p1_start) * 0.5:
            break

        t_alloc = min(per_init_time * (2.0 if idx == 0 and given is not None else 1.0),
                      total_deadline - time.monotonic())
        if t_alloc < 5:
            continue

        n_arr = len(arr)
        S_arr = np.sqrt(2.0 * n_arr)

        # Run FISTA with different parameters for diversity
        k_start = 4 + 4 * (idx % 4)
        k_end = 128 + 64 * (idx % 3)

        a_opt, score = fista_topk(arr, n_arr, S_arr, t_alloc, rng,
                                   k_start=k_start, k_end=k_end, power=1.5,
                                   step_mode="adaptive", restart_interval=1500)
        a_opt = normalize_sum(a_opt, n_arr)
        obj = compute_objective(a_opt)

        if np.isfinite(obj) and obj < global_best:
            global_best = obj
            global_a = a_opt.copy()
            print("  [%s] n=%d: obj=%.6f (elapsed %.0fs)" %
                  (name, n_arr, obj, time.monotonic() - p1_start))

    print("After Phase 1: best=%.6f, time=%.0fs" % (global_best, time.monotonic() - p1_start))

    # Phase 2: Pairwise coordinate descent polishing (30% of budget)
    elapsed = time.monotonic() - p1_start
    phase2_time = (budget_s - 20.0 - elapsed) * 0.35
    p2_start = time.monotonic()

    if global_a is not None and phase2_time > 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("Phase 2: Pairwise CD (n=%d, %.0fs)" % (n_best, phase2_time))

        a_cd, cd_score = pairwise_cd(global_a, n_best, S_best, phase2_time, rng,
                                      n_swaps_target=200000)
        a_cd = normalize_sum(a_cd, n_best)
        obj_cd = compute_objective(a_cd)
        if obj_cd < global_best:
            global_best = obj_cd
            global_a = a_cd.copy()
            print("  After CD: obj=%.6f" % obj_cd)

    # Phase 3: Final FISTA polish (15% of budget)
    elapsed = time.monotonic() - p2_start
    phase3_time = (budget_s - 20.0 - elapsed - (time.monotonic() - p1_start)) * 0.20
    phase3_time = max(10.0, min(phase3_time, (total_deadline - time.monotonic()) * 0.3))
    p3_start = time.monotonic()

    if global_a is not None and phase3_time > 10:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)
        print("Phase 3: FISTA polish (n=%d, %.0fs)" % (n_best, phase3_time))

        a_pol, pol_score = fista_polish(global_a, n_best, S_best, phase3_time, rng)
        a_pol = normalize_sum(a_pol, n_best)
        obj_pol = compute_objective(a_pol)
        if obj_pol < global_best:
            global_best = obj_pol
            global_a = a_pol.copy()
            print("  After polish: obj=%.6f" % obj_pol)

        # Second round of CD
        remaining = total_deadline - time.monotonic()
        if remaining > 30:
            a_cd2, _ = pairwise_cd(global_a, n_best, S_best, remaining * 0.5, rng,
                                    n_swaps_target=50000)
            a_cd2 = normalize_sum(a_cd2, n_best)
            obj_cd2 = compute_objective(a_cd2)
            if obj_cd2 < global_best:
                global_best = obj_cd2
                global_a = a_cd2.copy()
                print("  After CD2: obj=%.6f" % obj_cd2)

            # Final micro-polish
            remaining2 = total_deadline - time.monotonic()
            if remaining2 > 15:
                a_fin, _ = fista_polish(global_a, n_best, S_best, remaining2 * 0.8, rng)
                a_fin = normalize_sum(a_fin, n_best)
                obj_fin = compute_objective(a_fin)
                if obj_fin < global_best:
                    global_best = obj_fin
                    global_a = a_fin.copy()
                    print("  After micro-polish: obj=%.6f" % obj_fin)

    # Phase 4: Try alternate length if time permits
    remaining = total_deadline - time.monotonic()
    if remaining > 120 and global_a is not None:
        n_main = len(global_a)
        alt_n = n_main * 2 if n_main <= 8192 else n_main // 2
        if 2048 <= alt_n <= 32768:
            S_alt = np.sqrt(2.0 * alt_n)
            print("Phase 4: Trying n=%d (remaining %.0fs)" % (alt_n, remaining))
            # Interpolate the best sequence to alt_n
            t_old = np.linspace(0, 1, len(global_a))
            t_new = np.linspace(0, 1, alt_n)
            alt_init = np.interp(t_new, t_old, global_a)
            alt_init = normalize_sum(alt_init, alt_n)

            a_alt, _ = fista_topk(alt_init, alt_n, S_alt, remaining * 0.5, rng,
                                   k_start=8, k_end=128, power=1.5)
            a_alt = normalize_sum(a_alt, alt_n)
            obj_alt = compute_objective(a_alt)
            if obj_alt < global_best:
                global_best = obj_alt
                global_a = a_alt.copy()
                print("  Alt n=%d: obj=%.6f" % (alt_n, obj_alt))

                # Polish the alt
                rem2 = total_deadline - time.monotonic()
                if rem2 > 20:
                    a_alt2, _ = fista_polish(global_a, alt_n, S_alt, rem2 * 0.7, rng)
                    a_alt2 = normalize_sum(a_alt2, alt_n)
                    obj_alt2 = compute_objective(a_alt2)
                    if obj_alt2 < global_best:
                        global_best = obj_alt2
                        global_a = a_alt2.copy()
                        print("  Alt polished: obj=%.6f" % obj_alt2)

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("=== FINAL: length=%d, verified objective=%.6f ===" % (len(final_list), ver))
        else:
            print("WARNING: evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
            final_list = [float(x) for x in global_a]
            print("Fallback objective: %.6f" % evaluate_sequence(final_list))
