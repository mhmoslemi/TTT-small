"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Adam with log-sum-exp smooth max and step-based continuation
2. Exact 1D coordinate descent (exploiting convexity of the 1D subproblem)

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_smooth_gradient(c, a, T):
    """
    Compute gradient of smooth max: (1/T)log(sum(exp(T*c_k)))
    grad = sum_k p_k * grad(c_k) where p_k = softmax(T*c)
    grad(c_k)[i] = 2*a[k-i] (with a[m]=0 for m outside [0,n-1])
    """
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    
    scaled = T * c / c_max - T  # = T*(c/c_max - 1), max is 0
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    
    # grad[i] = 2 * sum_k p[k] * a[k-i] (with a[m]=0 for m outside [0,n-1])
    # = 2 * (p convolved with reversed a)[i]... 
    # Actually: sum_k p[k] * a[k-i]
    # Let's compute this as a cross-correlation
    # cross_corr(p, a)[i] = sum_k p[k] * conj(a[k-i])  (for real a: sum_k p[k]*a[k-i])
    # But we need sum_k p[k] * a[k-i] where a is zero outside [0,n-1]
    # This is the same as (p * flip(a))[i + n - 1]... 
    # Let me just use FFT:
    # Define b[m] = a[m] for m in [0,n-1], 0 otherwise (length N)
    # Define q[k] = p[k] for k in [0, 2n-2], 0 otherwise (length N)
    # We want r[i] = sum_k q[k] * b[k-i] = cross_correlation(q, b)[i]
    # cross_correlation(q, b)[i] = sum_k q[k] * b[k-i]
    # = IFFT(FFT(q) * conj(FFT(b)))[i]
    
    fsize = 1 << (2 * (2*n-1) - 1).bit_length()  # enough for linear cross-correlation
    Q = fft(p, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(Q * np.conj(A)))
    return 2.0 * corr[:n]


def run_adam_continuation(a_init, n, S, time_budget, T_start=5.0, T_end=5000.0,
                          T_double_every=50, lr=0.05, beta1=0.9, beta2=0.999,
                          grad_clip=10.0, verbose=False):
    """
    Adam optimizer on the smooth max objective with step-based continuation.
    Works with a >= 0, sum(a) = S.
    """
    a = normalize_nonneg(a_init.copy(), S)
    
    # Use log-parameters: a = S * softmax(theta)
    # This ensures a > 0 and sum(a) = S automatically
    # But we also want to allow a[i] = 0, so use: a = S * exp(theta) / sum(exp(theta))
    # with a floor on theta
    
    theta = np.full(n, np.log(1.0/n))  # uniform init in log-space
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    eps = 1e-8
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        t += 1
        T = min(T_start * (2.0 ** (t // T_double_every)), T_end)
        
        # Convert theta to a
        theta_shifted = theta - theta.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            a = np.full(n, S / n)
        else:
            a = S * exp_theta / total
        
        # Compute dc = conv(a, a)
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Compute smooth gradient
        g_a = compute_smooth_gradient(c, a, T)
        
        # Convert to theta gradient:
        # a = S * exp(theta) / sum(exp(theta))
        # da[i]/dtheta[j] = a[i] * (delta_{ij} - a[j]/S)
        # dL/dtheta[j] = sum_i (dL/da[i]) * (da[i]/dtheta[j])
        # = sum_i g_a[i] * a[i] * (delta_{ij} - a[j]/S)
        # = g_a[j]*a[j] - a[j]/S * sum_i g_a[i]*a[i]
        dot = np.dot(g_a, a)
        g_theta = g_a * a - (a / S) * dot
        
        # Clip gradient
        gnorm = np.linalg.norm(g_theta)
        if gnorm > grad_clip:
            g_theta = g_theta * (grad_clip / gnorm)
        if gnorm < 1e-15:
            break
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_theta
        v = beta2 * v + (1.0 - beta2) * (g_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - start
            print(f"      Adam step {t}: obj={best_obj:.7f}, T={T:.1f}, time={elapsed:.1f}s")
    
    return best_a, best_obj
