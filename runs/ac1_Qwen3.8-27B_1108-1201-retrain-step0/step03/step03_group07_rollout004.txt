Phase 1: Mirror descent with softmax subgradient

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_conv_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_conv_ab(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def compute_obj(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_conv_self(a)
    return 2.0 * n * c.max() / (S * S)


def compute_subgrad_softmax(c, a, temperature):
    """
    Compute softmax-weighted subgradient of max_k c[k].
    grad[m] = 2 * sum_k w[k] * a[k-m] for valid indices.
    """
    n = len(a)
    c_max = c.max()
    # Use temperature to control concentration
    w = np.exp(temperature * (c - c_max))
    total = w.sum()
    if total < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = w / total
    
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def solve_lp_direction(a, M, n_active=100):
    """
    Solve: max sum(g) s.t. (a*g)[k] <= M for top-k most binding k, g >= 0.

    n = len(a)
    c = fft_conv_self(a)

    active_idx = np.argsort(c, kind='stable')[::-1][:n_active]

    rows = []
    cols = []
    vals = []

        m_hi = min(n - 1, k)
        for m in range(m_lo, m_hi + 1):
            rows.append(idx)
            cols.append(m)
            vals.append(a[m])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_idx), n))
    b = np.full(len(active_idx), M)
    c_obj = -np.ones(n)

def coordinate_descent_polish(a0, time_budget):

        improved = False
        c_max = c.max()

        k_max = np.argmax(c)
        m_lo = max(0, k_max - n + 1)
        m_hi = min(n - 1, k_max)
        
        # Try adjusting elements near the peak
        for m in range(m_lo, m_hi + 1):
            if time.monotonic() - start > time_budget:

            orig = a[m]
            if orig < 1e-12:

            # Try reducing this element
            for factor in [0.5, 0.7, 0.85]:

                s = a.sum()
                if s < 1e-10:
