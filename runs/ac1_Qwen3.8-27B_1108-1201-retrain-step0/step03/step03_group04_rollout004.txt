Strategy: Multi-phase optimization combining:

2. FISTA with excess-weighted gradient and BB step sizes

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute 2 * cross_correlation(weights, a) truncated to length n."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad

    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def excess_weights(c):
    """Weights proportional to excess above mean."""
    mean_c = c.mean()
    excess = np.maximum(0.0, c - mean_c)
    total = excess.sum()
    if total < 1e-20:
        return np.ones_like(c) / len(c)
    return excess / total


def toppk_weights(c, top_k):
    """Normalized top-k weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights

    rel_gap = gap / cur_max

    else:

        k = min(k, 100)
        return toppk_weights(c, k)

            else:

        # step * grad / gnorm gives a unit step in the gradient direction
        # We want the step magnitude to be ~ S/n (the "natural" scale)
        step_mag = base_step * (0.5 + 0.5 * min(1.0, base_step / (gnorm * base_step + 1e-20)))
        
        x_new = project_simplex(y - step_mag * (grad / max(gnorm, 1e-15)), S)
        
        # Check improvement
        if it % 10 == 0:
            c_check = fft_convolve_self(x_new)
            if c_check.max() < best_score:
                best_score = c_check.max()
                best_local = x_new.copy()
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y_new = np.maximum(y_new, 0.0)
        
        # Stalldetection: if no improvement for a long time, perturb
        if stall > 2000:
            noise_scale = 0.003 * S / np.sqrt(n) * min(1.0, stall / 5000.0)
            y_new = project_simplex(y_new + rng.normal(0, noise_scale, n), S)
            t = 1.0
            stall = 0
            prev_g = None
            prev_a = None
        
        prev_g = grad.copy()
        prev_a = a.copy()
        a = x_new
        y = y_new
        t = t_new
    
    return best_local, best_score


def run_exact_linesearch_polish(a_init, n, S, max_time, rng):
    """Polish with exact line search: try multiple step sizes, keep best."""
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    base_step = S / n
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()
        
        # Focus on top lags
        top_k = min(3, max(1, it // 5000 + 1))
        weights = toppk_weights(c, top_k)
        grad = compute_gradient_fft(a, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        grad_dir = -grad / max(gnorm, 1e-15)
        
        # Try several step sizes
        best_x = a
        best_val = cur
        for trial in range(5):
            step = base_step * (0.1 * (2.0 ** trial))
            x_try = project_simplex(a + step * grad_dir, S)
            c_try = fft_convolve_self(x_try)
            if c_try.max() < best_val:
                best_val = c_try.max()
                best_x = x_try
        
        if best_val < cur:
            a = best_x
        else:
            # Reduce step or perturb
            if rng.random() < 0.3:
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                a = project_simplex(a + noise, S)
            base_step *= 0.8
    
    return best_local, best_score


def run_block_preopt(a_init, n, S, block_size, max_time, rng):
    """Optimize a block-averaged version, then use as initialization."""
    n_blocks = n // block_size
    if n_blocks < 16:
        block_size = max(1, n // 256)
        n_blocks = n // block_size
    
    # Create block-averaged version
    n_use = n_blocks * block_size
    a_blocks = a_init[:n_use].reshape(n_blocks, block_size).mean(axis=1)
    S_blocks = S * n_blocks / n  # proportional sum
    
    # Optimize the block version
    best_blocks = a_blocks.copy()
    t0 = time.monotonic()
    
    y = best_blocks.copy()
    t = 1.0
    best_block_score = np.inf
    
    while time.monotonic() - t0 < max_time * 0.5:
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_block_score - 1e-12:
            best_block_score = cur
            best_blocks = y.copy()
        
        weights = excess_weights(c)
        grad = compute_gradient_fft(y, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 2.0 * S_blocks / (n_blocks * gnorm)
        x_new = project_simplex(y - step * (grad / max(gnorm, 1e-15)), S_blocks)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - best_blocks)
        y = np.maximum(y, 0.0)
        best_blocks = x_new
        t = t_new
    
    # Expand back to full resolution
    a_full = np.repeat(best_blocks, block_size)
    if n_use < n:
        a_full = np.concatenate([a_full, a_init[n_use:]])
    
    return safe_normalize(a_full, n)


def propose_candidate(seed=42, budget_s=2385, **kwargs):
    """
    Minimize 2*n*max_conv/sum^2 over non-negative sequences.
    Multi-phase: block preopt, FISTA v2 (adaptive weights + BB step),
    exact line search polish, and perturbation exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s - 25.0)
    
    # Get the given sequence
    given_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given_a = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    if given_a is not None:
        n = len(given_a)
    else:
        n = 8192
        given_a = np.ones(n) * 0.5
    
    S = np.sqrt(2.0 * float(n))
    a_start = safe_normalize(given_a, n)
    c_start = fft_convolve_self(a_start)
    global_best = c_start.max()
    global_a = a_start.copy()
    obj_start = 2.0 * n * global_best / (S * S)
    print("Initial: score=%.6f, obj=%.6f, n=%d" % (global_best, obj_start, n))
    
    # === Phase 0: Try alternative starting points (fast) ===
    alternatives = []
    # Given sequence
    alternatives.append(("given", a_start))
    
    # Modulated variants
    tt = np.linspace(0, 1, n, endpoint=False)
    # Gentle sine modulation
    alt1 = a_start * (1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt))
    alternatives.append(("sine3", safe_normalize(alt1, n)))
    
    # Gentle ramp
    alt2 = a_start * (1.0 + 0.5 * tt)
    alternatives.append(("ramp", safe_normalize(alt2, n)))
    
    # Anti-ramp
    alt3 = a_start * (1.0 + 0.5 * (1 - tt))
    alternatives.append(("antiramp", safe_normalize(alt3, n)))
    
    # Two-hump
    alt4 = a_start * (1.0 + 0.4 * np.sin(np.pi * tt))
    alternatives.append(("twohump", safe_normalize(alt4, n)))
    
    # Random
    alt5 = rng.dirichlet(np.ones(n) * 0.5)
    alternatives.append(("random", safe_normalize(alt5, n)))
    
    # Evaluate alternatives
    for name, alt in alternatives:
        c_alt = fft_convolve_self(alt)
        score = c_alt.max()
        if score < global_best:
            global_best = score
            global_a = alt.copy()
            print("  Alt '%s': %.6f (obj %.6f)" % (name, score, 2*n*score/S**2))
    
    # === Phase 1: Block pre-optimization (10% of time) ===
    remaining = deadline - time.monotonic()
    phase1_time = 0.08 * remaining
    if phase1_time > 10:
        print("Phase 1: Block preopt (%.0fs)" % phase1_time)
        t0 = time.monotonic()
        a_block = run_block_preopt(global_a, n, S, block_size=8, max_time=phase1_time, rng=rng)
        c_block = fft_convolve_self(a_block)
        if c_block.max() < global_best:
            global_best = c_block.max()
            global_a = a_block.copy()
            print("  Block: %.6f (obj %.6f)" % (global_best, 2*n*global_best/S**2))
    
    # === Phase 2: Main FISTA optimization (50% of time) ===
    remaining = deadline - time.monotonic()
    phase2_time = 0.50 * remaining
    if phase2_time > 30:
        print("Phase 2: FISTA v2 (%.0fs)" % phase2_time)
        t0 = time.monotonic()
        a_p2, _ = run_fista_v2(global_a, n, S, phase2_time, rng, use_adaptive=True)
        a_p2 = safe_normalize(a_p2, n)
        c_p2 = fft_convolve_self(a_p2)
        if c_p2.max() < global_best:
            global_best = c_p2.max()
            global_a = a_p2.copy()
            print("  FISTA-v2: %.6f (obj %.6f) [%.0fs]" % (global_best, 2*n*global_best/S**2, time.monotonic()-t0))
        else:
            print("  FISTA-v2: no improvement (best was %.6f)" % global_best)
    
    # === Phase 3: Second FISTA with different parameters (15% of time) ===
    remaining = deadline - time.monotonic()
    phase3_time = 0.15 * remaining
    if phase3_time > 15:
        print("Phase 3: FISTA v2 variant (%.0fs)" % phase3_time)
        t0 = time.monotonic()
        # Perturb slightly
        perturbed = global_a + rng.normal(0, 0.002 * S / np.sqrt(n), n)
        perturbed = safe_normalize(perturbed, n)
        a_p3, _ = run_fista_v2(perturbed, n, S, phase3_time, rng, use_adaptive=False)
        a_p3 = safe_normalize(a_p3, n)
        c_p3 = fft_convolve_self(a_p3)
        if c_p3.max() < global_best:
            global_best = c_p3.max()
            global_a = a_p3.copy()
            print("  FISTA-var: %.6f (obj %.6f)" % (global_best, 2*n*global_best/S**2))
    
    # === Phase 4: Exact line search polish (15% of time) ===
    remaining = deadline - time.monotonic()
    phase4_time = 0.15 * remaining
    if phase4_time > 10:
        print("Phase 4: Line search polish (%.0fs)" % phase4_time)
        t0 = time.monotonic()
        a_p4, _ = run_exact_linesearch_polish(global_a, n, S, phase4_time, rng)
        a_p4 = safe_normalize(a_p4, n)
        c_p4 = fft_convolve_self(a_p4)
        if c_p4.max() < global_best:
            global_best = c_p4.max()
            global_a = a_p4.copy()
            print("  LineSearch: %.6f (obj %.6f)" % (global_best, 2*n*global_best/S**2))
    
    # === Phase 5: Multi-restart exploration (10% of time) ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("Phase 5: Multi-restart (%.0fs)" % remaining)
        n_restarts = 5
        per_restart = remaining * 0.8 / n_restarts
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            # Different perturbation for each restart
            if r == 0:
                pert = global_a + rng.normal(0, 0.01 * S / np.sqrt(n), n)
            elif r == 1:
                # Swap some blocks
                pert = global_a.copy()
                for _ in range(3):
                    i = rng.integers(0, n - 256)
                    j = rng.integers(0, n - 256)
                    pert[i:i+128], pert[j:j+128] = pert[j:j+128].copy(), pert[i:i+128].copy()
            elif r == 2:
                # Smooth perturbation
                smooth = np.convolve(rng.normal(0, 1, n), np.ones(128)/128, mode='same')
                pert = global_a + 0.005 * S / np.sqrt(n) * smooth
            elif r == 3:
                # Frequency-domain perturbation
                A = fft(global_a)
                A_noisy = A * (1.0 + 0.02 * rng.normal(0, 1, n))
                pert = np.real(ifft(A_noisy))
            else:
                # Antisymmetric perturbation (breaks symmetry)
                pert = global_a.copy()
                for _ in range(100):
                    i = rng.integers(0, n)
                    pert[i] += 0.001 * S / np.sqrt(n) * rng.standard_normal()
            
            pert = safe_normalize(pert, n)
            a_r, _ = run_fista_v2(pert, n, S, per_restart, rng, use_adaptive=True)
            a_r = safe_normalize(a_r, n)
            c_r = fft_convolve_self(a_r)
            if c_r.max() < global_best:
                global_best = c_r.max()
                global_a = a_r.copy()
                print("  Restart %d: %.6f (obj %.6f)" % (r, global_best, 2*n*global_best/S**2))
    
    # === Phase 6: Final ultra-fine polish ===
    remaining = deadline - time.monotonic()
    if remaining > 5:
        print("Phase 6: Final polish (%.0fs)" % remaining)
        a_final, _ = run_exact_linesearch_polish(global_a, n, S, remaining, rng)
        a_final = safe_normalize(a_final, n)
        c_final = fft_convolve_self(a_final)
        if c_final.max() < global_best:
            global_best = c_final.max()
            global_a = a_final.copy()
            print("  Final: %.6f (obj %.6f)" % (global_best, 2*n*global_best/S**2))
    
    # === Output ===
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    final_obj = 2.0
