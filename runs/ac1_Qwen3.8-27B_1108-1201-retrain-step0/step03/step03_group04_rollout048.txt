"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
1. Continuation FISTA with increasing softmax temperature (smooth surrogate optimization)
2. Top-k subgradient polishing with Nesterov momentum
3. Coordinate descent final polish
4. Multiple restarts with diverse initializations and n values
Key insight: non-symmetric sequences can achieve max_conv < sum(a_i^2),
breaking the symmetric-sequence lower bound.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    """Compute gradient of weighted sum of convolution values."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Compute normalized top-k weights by value."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights


def uniform_peaks_weights(c, k):
    """Uniform weights on top-k peaks."""
    k = min(k, len(c))
    top_indices = np.argpartition(c, -k)[-k:]
    weights = np.zeros(len(c))
    weights[top_indices] = 1.0 / k
    return weights


def run_continuation_fista(a_init, n, S, max_time, rng,
                           beta_start=10.0, beta_end=2000.0,
                           n_beta_steps=30):
    """FISTA with continuation on beta parameter."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    best_score = np.inf
    best_local = a.copy()
    start = time.monotonic()

    # Create beta schedule
    betas = np.geomspace(beta_start, beta_end, n_beta_steps)

    for beta in betas:
        if time.monotonic() - start > max_time:
            break

        # Run FISTA for current beta
        x = a.copy()
        y = a.copy()
        t = 1.0
        step = 1.0 / (beta + 2.0)
        it = 0
        max_it_per_beta = max(100, int(max_time * 3000 / len(betas) / 0.5))

        while it < max_it_per_beta and time.monotonic() - start < max_time:
            it += 1
            c = fft_convolve_self(y)
            cur = c.max()
            if cur < best_score - 1e-12:
                best_score = cur
                best_local = y.copy()

            w = softmax_weights(c, beta)
            grad = compute_gradient_fft(y, w)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break

            x_new = project_simplex(y - step * grad, S)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = x_new + ((t - 1.0) / t_new) * (x_new - a)
            y = np.maximum(y, 0.0)
            a = x_new
            t = t_new

            # Adaptive step
            if it % 500 == 0:
                step = 1.0 / (beta + 2.0 + it * 0.001)

        # Rescale a to maintain normalization
        s_a = a.sum()
        if s_a > 1e-15:
            a = a * (S / s_a)

    return a.copy(), best_score


def run_toppk_nesterov(a_init, n, S, max_time, rng,
                       top_k_init=1, top_k_max=50, top_k_rate=100):
    """Nesterov-accelerated top-k subgradient method."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = S / (n * 0.5)
    stall = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // top_k_rate)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            stall = 0
        else:
            stall += 1

        weights = toppk_weights(c, top_k)
        grad = compute_gradient_fft(y, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        adaptive_step = step / (1.0 + 0.001 * it)
        x_new = project_simplex(y - adaptive_step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic check and perturbation
        if it % 500 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
            if stall > 500 and rng.random() < 0.3:
                noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
                y = project_simplex(y + noise, S)
                t = 1.0
                stall = 0

    return best_local, best_score


def run_coordinate_descent(a_init, n, S, max_time, rng):
    """Coordinate descent: optimize one coordinate at a time."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()

        # Try random coordinates
        n_tries = min(n, 50)
        indices = rng.choice(n, size=n_tries, replace=False)

        for idx in indices:
            if time.monotonic() - start > max_time:
                break
            old_val = a[idx]
            # Try a few perturbations of this coordinate
            for factor in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]:
                new_val = max(0.0, old_val * factor)
                delta = new_val - old_val
                a[idx] = new_val
                # Renormalize
                s_new = a.sum()
                if s_new > 1e-15:
                    a = a * (S / s_new)
                c_new = fft_convolve_self(a)
                if c_new.max() < cur - 1e-12:
                    cur = c_new.max()
                    if cur < best_score:
                        best_score = cur
                        best_local = a.copy()
                else:
                    # Revert
                    a[idx] = old_val
                    s_new = a.sum()
                    if s_new > 1e-15:
                        a = a * (S / s_new)
                    break

    return best_local, best_score


def run_multiplicative_weights(a_init, n, S, max_time, rng,
                               eta_init=0.1, top_k=5):
    """Mirror descent / multiplicative weights."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()

        eta = eta_init / (1.0 + 0.002 * it)
        weights = toppk_weights(c, top_k)
        grad = compute_gradient_fft(a, weights)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new * (S / s_new)

    return best_local, best_score


def generate_random_init(n, S, rng, style='uniform'):
    """Generate a random non-negative sequence with sum S."""
    if style == 'uniform':
        a = rng.uniform(0.1, 1.0, n)
    elif style == 'sparse':
        a = np.zeros(n)
        n_nonzero = n // 3
        indices = rng.choice(n, size=n_nonzero, replace=False)
        a[indices] = rng.uniform(0.5, 2.0, n_nonzero)
    elif style == 'smooth':
        a = np.convolve(rng.uniform(0.5, 1.5, n), np.ones(16) / 16, mode='same')
        a = np.maximum(a, 0.01)
    elif style == 'asymmetric':
        # Deliberately asymmetric: ramp up then down with offset
        t = np.linspace(0, 1, n)
        a = np.exp(-4 * (t - 0.3) ** 2) + 0.5 * np.exp(-4 * (t - 0.7) ** 2)
        a += 0.1 * rng.standard_normal(n)
        a = np.maximum(a, 0.01)
    elif style == 'Periodic':
        # Near-periodic with random phase
        freq = rng.uniform(3, 7)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq / n * np.arange(n) + phase)
        a = np.maximum(a, 0.01)
    else:
        a = rng.uniform(0.1, 1.0, n)

    s = a.sum()
    if s > 1e-15:
        a = a * (S / s)
    return a


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def propose_candidate(seed=42, budget_s=2385, **kwargs):
    """
    Multi-strategy optimization:
    1. Continuation FISTA from parent + diverse random inits
    2. Top-k Nesterov polishing
    3. Coordinate descent final polish
    4. Multiplicative weights for diversity
    Tries multiple n values and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s - 25.0)

    global_best = np.inf
    global_a = None
    global_n = 0

    # Try to get parent sequence
    parent_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            parent_a = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Determine n values to try
    n_candidates = [8192]
    if parent_a is not None:
        n_candidates.insert(0, len(parent_a))
    n_candidates.extend([4096, 16384])
    # Remove duplicates, keep order
    seen = set()
    n_unique = []
    for n in n_candidates:
        if n not in seen and 256 <= n <= 32768:
            seen.add(n)
            n_unique.append(n)

    # Phase allocation: spend most time on best n, some on others
    # Primary n (from parent or 8192)
    primary_n = n_unique[0]

    def try_optimize(a_init, n, time_budget, label=""):
        """Run full optimization pipeline on a starting sequence."""
        nonlocal global_best, global_a, global_n
        if time_budget < 5:
            return
        S = np.sqrt(2.0 * float(n))
        a_start = safe_normalize(a_init, n)
        c_start = fft_convolve_self(a_start)
        init_score = c_start.max()

        # Step 1: Continuation FISTA (50% of budget)
        t1 = time_budget * 0.5
        a1, s1 = run_continuation_fista(a_start, n, S, t1, rng,
                                         beta_start=5.0, beta_end=500.0,
                                         n_beta_steps=20)
        a1 = safe_normalize(a1, n)
        c1 = fft_convolve_self(a1)
        score1 = c1.max()
        if score1 < init_score:
            a_start = a1
            init_score = score1

        # Step 2: Top-k Nesterov (30% of budget)
        t2 = time_budget * 0.3
        a2, s2 = run_toppk_nesterov(a_start, n, S, t2, rng,
                                     top_k_init=1, top_k_max=30, top_k_rate=200)
        a2 = safe_normalize(a2, n)
        c2 = fft_convolve_self(a2)
        score2 = c2.max()
        if score2 < init_score:
            a_start = a2
            init_score = score2

        # Step 3: Multiplicative weights (10% of budget)
        t3 = time_budget * 0.1
        a3, s3 = run_multiplicative_weights(a_start, n, S, t3, rng,
                                             eta_init=0.05, top_k=3)
        a3 = safe_normalize(a3, n)
        c3 = fft_convolve_self(a3)
        score3 = c3.max()
        if score3 < init_score:
            a_start = a3
            init_score = score3

        # Step 4: Coordinate descent (10% of budget)
        t4 = time_budget * 0.1
        a4, s4 = run_coordinate_descent(a_start, n, S, t4, rng)
        a4 = safe_normalize(a4, n)
        c4 = fft_convolve_self(a4)
        score4 = c4.max()
        if score4 < init_score:
            a_start = a4
            init_score = score4

        # Final check
        if init_score < global_best:
            obj = 2.0 * n * init_score / (S * S)
            global_best = init_score
            global_a = a_start.copy()
            global_n = n
            print("  %s (n=%d): score=%.6f obj=%.6f" % (label, n, init_score, obj))

    # === Main optimization: Primary n from parent ===
    remaining = deadline - time.monotonic()
    print("Phase A: Primary optimization (n=%d, %.0f sec)" % (primary_n, remaining * 0.6))

    # Start from parent if available
    if parent_a is not None and len(parent_a) == primary_n:
        ptime = remaining * 0.35
        try_optimize(parent_a, primary_n, ptime, "Parent")
    else:
        ptime = remaining * 0.35
        a_rand = generate_random_init(primary_n, np.sqrt(2.0 * primary_n), rng, 'asymmetric')
        try_optimize(a_rand, primary_n, ptime, "Random-Asym")

    # Additional starts for primary n
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("Phase B: Additional starts for n=%d" % primary_n)
        S = np.sqrt(2.0 * float(primary_n))
        n_starts = 3
        per_start = remaining * 0.5 / n_starts
        styles = ['uniform', 'sparse', 'Periodic']
        for i, style in enumerate(styles):
            if time.monotonic() > deadline - 30:
                break
            a_rand = generate_random_init(primary_n, S, rng, style)
            try_optimize(a_rand, primary_n, per_start, "Random-%s" % style)

    # === Try other n values ===
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print("Phase C: Other n values (%.0f sec)" % (remaining * 0.3))
        for n_alt in n_unique[1:]:
            if time.monotonic() > deadline - 30:
                break
            S_alt = np.sqrt(2.0 * float(n_alt))
            a_rand = generate_random_init(n_alt, S_alt, rng, 'asymmetric')
            t_alt = min(remaining * 0.15, 60)
            try_optimize(a_rand, n_alt, t_alt, "Alt-n=%d" % n_alt)

    # === Final polish on best ===
    remaining = deadline - time.monotonic()
    if remaining > 10 and global_a is not None:
        print("Phase D: Final polish (%.0f sec)" % remaining)
        n_final = global_n
        S_final = np.sqrt(2.0 * float(n_final))

        # Top-k with very low k
        t_polish = remaining * 0.4
        a_p, s_p = run_toppk_nesterov(global_a, n_final, S_final, t_polish, rng,
                                       top_k_init=1, top_k_max=10, top_k_rate=500)
        a_p = safe_normalize(a_p, n_final)
        c_p = fft_convolve_self(a_p)
        if c_p.max() < global_best:
            global_best = c_p.max()
            global_a = a_p.copy()
            print("  Final top-k: %.6f" % global_best)

        # Coordinate descent
        remaining2 = deadline - time.monotonic()
        if remaining2 > 5:
            a_cd, s_cd = run_coordinate_descent(global_a, n_final, S_final, remaining2, rng)
            a_cd = safe_normalize(a_cd, n_final)
            c_cd = fft_convolve_self(a_cd)
            if c_cd.max() < global_best:
                global_best = c_cd.max()
                global_a = a_cd.copy()
                print("  Final CD: %.6f" % global_best)

    # === Output ===
    if global_a is None:
        # Fallback
        n_fallback = 256
        S_fallback = np.sqrt(2.0 * n_fallback)
        global_a = np.ones(n_fallback) * S_fallback / n_fallback
        global_n = n_fallback
        global_best = fft_convolve_self(global_a).max()

    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    S = np.sqrt(2.0 * float(global_n))
    final_obj = 2.0 * global_n * global_best / (S * S)
    print("FINAL: n=%d, max_conv=%.6f, objective=%.6f" % (global_n, global_best, final_obj))

    # Verify with evaluate_sequence
    try:
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified: %.6f" % ver)
        else:
            print("Verification failed, fallback")
            final_list = [0.5] * 256
    except Exception as e:
        print("Verification error: %s" % str(e))

    return final_list
