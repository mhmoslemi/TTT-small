def resample_array(a, target_n):
   target_n=int(max(1,target_n))
   a=np.asarray(a,dtype=np.float64).reshape(-1)
   n=a.size
   if n==0: return np.full(target_n, 1.0/target_n)
   if target_n == n: return normalize_sum(a,1.0)
   if target_n > n and target_n % n == 0:
       r=target_n//n; vals=np.repeat(a, r); vals = vals / float(r)
   elif target_n < n and n % target_n == 0:
       vals=a.reshape(target_n, n//target_n).mean(axis=1)
   else:
       old_pos=(np.arange(n)+0.5)/float(n)
       new_pos=(np.arange(target_n)+0.5)/float(target_n)
       vals=np.interp(new_pos, old_pos, a)
   return normalize_sum(vals,1.0)
```

"""
Search for non-negative sequences minimizing
    2*n*max(conv(a,a)) / sum(a)^2
using smooth-max natural mirror descent, multi-resolution warm starts,
multiple initializations, adaptive peak-weighting, restarts, and final
length extension.  The program is self-contained and only assumes that
evaluate_sequence is available in the global namespace for final checks.
"""

import numpy as np
import time

FLOOR_A = 1e-12


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def conv_self(a):
    """Linear self-convolution a * a via FFT, clipped to nonnegative."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fsize = next_pow2(size)
    A = np.fft.fft(a, fsize)
    c = np.real(np.fft.ifft(A * A))
    c = c[:size]
    np.maximum(c, 0.0, out=c)
    return c


def conv_qa(q, a_rev):
    """
    For q of length 2n-1 and reversed a of length n, return
    G[i] = sum_k q[k] * a[k-i] for valid k, i=0..n-1.
    """
    q = np.asarray(q, dtype=np.float64).reshape(-1)
    a_rev = np.asarray(a_rev, dtype=np.float64).reshape(-1)
    n = a_rev.size
    L = q.size
    if n == 0 or L == 0:
        return np.array([], dtype=np.float64)
    fsize = next_pow2(L + n - 1)
    Q = np.fft.fft(q, fsize)
    A = np.fft.fft(a_rev, fsize)
    h = np.real(np.fft.ifft(Q * A))
    return h[n - 1:n + n]


def normalize_sum(a, target=1.0):
    """Project to nonnegative vector with sum target, with a tiny floor."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0:
        return np.array([float(target)], dtype=np.float64)
    s = float(a.sum())
    if s <= 1e-300:
        return np.full(n, target / float(n), dtype=np.float64)
    a = a * (target / s)
    a = np.maximum(a, FLOOR_A)
    s2 = float(a.sum())
    if s2 <= 1e-300:
        return np.full(n, target / float(n), dtype=np.float64)
    a = a * (target / s2)
    return a


def local_obj(a, c=None):
    """Objective value, scale invariant."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    s = float(a.sum())
    if n == 0 or s <= 1e-100:
        return float("inf")
    if c is None:
        c = conv_self(a)
    if c.size == 0:
        return float("inf")
    m = float(np.max(c))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))


def perturb_sequence(a, sigma, rng):
    """Multiplicative lognormal perturbation, preserving nonnegativity."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    sigma = max(1e-6, float(sigma))
    noise = rng.normal(0.0, sigma, size=a.size)
    a2 = a * np.exp(noise)
    return normalize_sum(a2, 1.0)


def make_weights(c, n, lam, top_count, rng):
    """
    Build a probability distribution over convolution coefficients.
    Most mass is on near-maximal peaks, with some broad smoothing.
    """
    L = c.size
    if L == 0:
        return np.array([1.0], dtype=np.float64)

    # Soft peak weights.  c has already been clipped nonnegative.
    if lam > 0.0:
        z = float(lam) * (float(n) * c)
        zmax = float(np.max(z))
        if not np.isfinite(zmax):
            q = np.full(L, 1.0 / L, dtype=np.float64)
        else:
            e = np.exp(z - zmax)
            se = float(e.sum())
            if se <= 1e-300 or not np.isfinite(se):
                q = np.full(L, 1.0 / L, dtype=np.float64)
            else:
                q = e / se
    else:
        q = np.full(L, 1.0 / L, dtype=np.float64)

    # Uniform over a random-ish top block.
    top_count = int(max(1, min(L, int(top_count))))
    if top_count >= L:
        qtop = q.copy()
    else:
        qtop = np.zeros(L, dtype=np.float64)
        if top_count == 1:
            idx = int(np.argmax(c))
            qtop[idx] = 1.0
        else:
            idx = np.argpartition(c, L - top_count)[-top_count:]
            qtop[idx] = 1.0 / float(top_count)

    # Mix smooth and hard peak chasing.
    w_top = 0.5
    q = (1.0 - w_top) * q + w_top * qtop

    # Tiny uniform background to avoid dead zones.
    q += 1e-6 / float(L)
    qs = float(q.sum())
    if qs <= 1e-300 or not np.isfinite(qs):
        q = np.full(L, 1.0 / L, dtype=np.float64)
    else:
        q = q / qs
    return q


def optimize_log(a0, time_limit, seed=0, base_lr=0.15, lam_start=5.0,
                 lam_max=120.0, step_clip=1.0, patience_time=45.0,
                 perturb_sigma=0.20, verbose=False):
    """
    Adaptive natural-gradient mirror descent for smooth max convolution.
    Keeps sum(a)=1.  Returns the best array and its local objective.
    """
    a0 = np.asarray(a0, dtype=np.float64).reshape(-1)
    if a0.size == 0:
        return a0.copy(), float("inf")

    a = normalize_sum(a0, 1.0)
    n = a.size
    start = time.monotonic()
    deadline = start + max(0.0, float(time_limit))

    c = conv_self(a)
    obj = local_obj(a, c)
    best_a = a.copy()
    best_obj = obj

    rng = np.random.default_rng(seed)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    lam = float(lam_start)
    lr = float(base_lr)
    curr_step_clip = float(step_clip)
    base_step_clip = float(step_clip)
    sigma = float(perturb_sigma)
    last_improve = start
    iter_no = 0
    reject_count = 0

    b1 = 0.85
    b2 = 0.99
    eps = 1e-8
    d_clip = 50.0

    if verbose:
        print(f"    optimize_log start: n={n}, obj={obj:.6f}, limit={float(time_limit):.1f}s")

    while time.monotonic() < deadline:
        iter_no += 1

        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            last_improve = time.monotonic()
            lam = min(float(lam_max), lam * 1.015)
            lr = min(float(base_lr) * 3.0, lr * 1.005)
            curr_step_clip = min(base_step_clip, curr_step_clip * 1.03)
            sigma = max(0.03, sigma * 0.985)

        # If the trajectory has been stuck, restart around the current best.
        if time.monotonic() - last_improve > max(1.0, float(patience_time)):
            a = perturb_sequence(best_a, sigma, rng)
            c = conv_self(a)
            obj = local_obj(a, c)
            np.fill(m, 0.0)
            np.fill(v, 0.0)
            t = 0
            lam = max(max(1.0, float(lam_start) * 0.5), lam * 0.75)
            lr = float(base_lr)
            curr_step_clip = float(base_step_clip)
            sigma = min(1.0, sigma * 1.25)
            last_improve = time.monotonic()
            reject_count = 0
            if not np.isfinite(obj):
                a = best_a.copy()
                c = conv_self(a)
                obj = local_obj(a, c)
            if verbose:
                print(f"      restart n={n} t={time.monotonic()-start:.1f}s "
                      f"obj={obj:.6f} best={best_obj:.6f} lam={lam:.3f}")
            if time.monotonic() >= deadline:
                break

        # Adaptive peak mass.
        top_frac = 0.10 / max(1.0, float(lam))
        top_frac = min(0.25, max(1.0 / float(len(c)), top_frac))
        top_count = int(len(c) * top_frac)
        if rng.random() < 0.15:
            top_count = max(1, min(top_count, max(1, int(top_count * 0.25))))

        q = make_weights(c, n, lam, top_count, rng)
        G = conv_qa(q, a[::-1])
        sval = float(np.dot(q, c))

        den = np.maximum(a, 1e-8)
        d = G / den - sval
        d = np.nan_to_num(d, nan=0.0, posinf=d_clip, neginf=-d_clip)
        d = np.clip(d, -d_clip, d_clip)

        t += 1
        m = b1 * m + (1.0 - b1) * d
        v = b2 * v + (1.0 - b2) * d * d
        mhat = m / (1.0 - b1 ** t)
        vhat = v / (1.0 - b2 ** t)
        step = (lr * mhat) / np.sqrt(vhat + eps)
        step = np.nan_to_num(step, nan=0.0, posinf=curr_step_clip,
                             neginf=-curr_step_clip)
        step = np.clip(step, -curr_step_clip, curr_step_clip)

        if not np.any(step):
            step = rng.normal(0.0, 0.01, size=n)
            step = np.clip(step, -curr_step_clip, curr_step_clip)

        a_new = a * np.exp(-step)
        a_new = np.maximum(a_new, FLOOR_A)
        if not np.all(np.isfinite(a_new)):
            a_new = a.copy()
        ss = float(a_new.sum())
        if ss <= 1e-300:
            a_new = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a_new = a_new / ss

        c_new = conv_self(a_new)
        obj_new = local_obj(a_new, c_new)

        accept = np.isfinite(obj_new)
        if accept:
            if np.isfinite(obj):
                if np.isfinite(best_obj):
                    margin = max(0.02 * obj, 0.02 * best_obj, 1e-6)
                else:
                    margin = max(0.02 * obj, 1e-6)
            else:
                margin = 1e-3
            if obj_new > obj + margin and obj_new > best_obj + margin:
                accept = False

        if accept:
            a = a_new
            c = c_new
            obj = obj_new
            reject_count = 0
            if obj <= best_obj or reject_count == 0:
                curr_step_clip = min(base_step_clip, curr_step_clip * 1.02)
        else:
            reject_count += 1
            lr = max(base_lr * 0.3, lr * 0.8)
            curr_step_clip = max(0.2, curr_step_clip * 0.8)
            if reject_count > 12:
                a = perturb_sequence(best_a, max(sigma, 0.25), rng)
                c = conv_self(a)
                obj = local_obj(a, c)
                np.fill(m, 0.0)
                np.fill(v, 0.0)
                t = 0
                reject_count = 0
                lr = float(base_lr)
                curr_step_clip = float(base_step_clip)
                last_improve = time.monotonic()

        if verbose and (iter_no % 2000 == 0):
            print(f"      iter={iter_no} t={time.monotonic()-start:.1f}s "
                  f"obj={obj:.6f} best={best_obj:.6f} lam={lam:.3f} lr={lr:.4f}")

    return best_a, best_obj


def resample_array(a, target_n):
    """Resample to target length, preserving shape as much as possible."""
    target_n = int(max(1, target_n))
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.full(target_n, 1.0 / target_n, dtype=np.float64)
    if target_n == n:
        return normalize_sum(a, 1.0)

    if target_n > n and target_n % n == 0:
        r = target_n // n
        vals = np.repeat(a, r)
        vals = vals / float(r)
    elif target_n < n and n % target_n == 0:
        vals = a.reshape(target_n, n // target_n).mean(axis=1)
    else:
        old_pos = (np.arange(n) + 0.5) / float(n)
        new_pos = (np.arange(target_n) + 0.5) / float(target_n)
        vals = np.interp(new_pos, old_pos, a)
    return normalize_sum(vals, 1.0)


def random_shape(n, seed=0):
    """Smoothed random positive shape with one antipalindromic harmonic."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    k = max(1, min(n // 16, 64))
    if k > 1 and n > 1:
        a = np.convolve(a, np.ones(k) / float(k), mode="same")
    a = np.maximum(a, 0.0)
    mx = float(a.max())
    if mx > 0.0:
        a = a / mx
    else:
        a = np.ones(n, dtype=np.float64)

    i = np.arange(n, dtype=np.float64)
    freq = int(rng.integers(1, max(2, n // 4)))
    amp = 0.5 * float(rng.uniform(-1.0, 1.0))
    a = a * (1.0 + amp * np.sin(2.0 * np.pi * freq * i / float(n)))
    a = np.maximum(a, 0.01)
    return normalize_sum(a, 1.0)


def random_sine_shape(n, seed=0):
    """Random nonnegative sum of antipalindromic sine modes."""
    rng = np.random.default_rng(seed)
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    n_terms = min(6, max(0, n // 2))
    for r in range(1, n_terms + 1):
        amp = (0.8 / float(r + 1)) * float(rng.uniform(-1.0, 1.0))
        a = a + amp * np.sin(2.0 * np.pi * float(r) * i / float(n))
    a = a - float(a.min()) + 0.02
    return normalize_sum(a, 1.0)


def run_multires(time_limit, seed=0, given_a=None, max_n=1024, verbose=False):
    """
    Cheap multi-resolution search to create an alternative warm start.
    Returns best sequence found at any resolution.
    """
    time_limit = max(0.0, float(time_limit))
    start = time.monotonic()
    deadline = start + time_limit

    if time_limit < 5.0:
        if given_a is not None:
            return given_a.copy(), local_obj(given_a)
        a = random_sine_shape(64, seed)
        return a, local_obj(a)

    stages = [16, 32, 64, 128, 256, 512, 1024]
    stages = [s for s in stages if s <= max_n]
    if not stages:
        stages = [max(16, int(max_n))]

    n0 = stages[0]
    candidates = []
    if given_a is not None:
        candidates.append(resample_array(given_a, n0))
    for s in range(5):
        candidates.append(random_sine_shape(n0, seed + s))
        candidates.append(random_shape(n0, seed + 100 + s))

    per = max(0.2, (time_limit * 0.06) / max(1, len(candidates)))
    best_a = candidates[0].copy()
    best_obj = local_obj(best_a)

    idx = 0
    for cand in candidates:
        if time.monotonic() > deadline - 1.0:
            break
        a, obj = optimize_log(
            cand, per, seed=seed + idx,
            base_lr=0.30, lam_start=2.0, lam_max=30.0,
            step_clip=1.5, patience_time=8.0,
            perturb_sigma=0.50, verbose=False
        )
        if np.isfinite(obj) and obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        idx += 1

    a = best_a
    if len(stages) > 1:
        raw = list(stages[1:])
        total_raw = float(sum(raw))
        alloc = 0.94
        stage_times = [alloc * float(nn) / total_raw for nn in raw]
        for j, nn in enumerate(stages[1:]):
            if time.monotonic() > deadline - 2.0:
                break
            t_i = stage_times[j] if j < len(stage_times) else 1.0
            t_i = max(0.5, min(float(t_i), deadline - time.monotonic()))
            a = resample_array(a, nn)
            base_lr = 0.28 if nn <= 256 else 0.20
            lam_max = 50.0 if nn <= 256 else 80.0
            sig = 0.35 if nn <= 256 else 0.25
            a, obj = optimize_log(
                a, t_i, seed=seed + 2000 + j,
                base_lr=base_lr, lam_start=3.0, lam_max=lam_max,
                step_clip=1.2, patience_time=12.0,
                perturb_sigma=sig, verbose=False
            )
            if np.isfinite(obj) and obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            if verbose:
                print(f"      multires n={nn} obj={obj:.6f} best={best_obj:.6f}")

    return best_a, best_obj

            perturb_sigma=0.18, verbose=True
        )
        candidate_list.append(a3)
        if np.isfinite(obj3) and obj3 < global_obj:
            global_obj = obj3
            global_a = a3.copy()

    # Sharper refinement on the best 4096-length candidate.
    rem = deadline - time.monotonic()
    t = min(800.0, 0.32 * rem)
    if t > 60.0 and global_a is not None:
        if global_a.size != 4096:
            a_cur = resample_array(global_a, 4096)
            obj_cur = local_obj(a_cur)
            if np.isfinite(obj_cur) and obj_cur < global_obj:
                global_obj = obj_cur
                global_a = a_cur.copy()
        else:
            a_cur = global_a.copy()

        print(f"\nphase 4: refined 4096 polish ({t:.0f}s)")
        a4, obj4 = optimize_log(
            a_cur, t, seed=seed + 3000,
            base_lr=0.09, lam_start=10.0, lam_max=180.0,
            step_clip=0.80, patience_time=45.0,
            perturb_sigma=0.10, verbose=True
        )
        candidate_list.append(a4)
        if np.isfinite(obj4) and obj4 < global_obj:
            global_obj = obj4
            global_a = a4.copy()

    # Try extending the best sequence to 8192.  Longer sequences often have
    # a smaller finite-size penalty, but only keep them if they help.
    rem = deadline - time.monotonic()
    t = min(420.0, 0.35 * rem)
    if t > 80.0 and global_a is not None:
        a8_init = resample_array(global_a, 8192)
        obj8_init = local_obj(a8_init)
        candidate_list.append(a8_init)
        if np.isfinite(obj8_init) and obj8_init < global_obj:
            global_obj = obj8_init
            global_a = a8_init.copy()

        print(f"\nphase 5: extension to 8192 ({t:.0f}s)")
        a5, obj5 = optimize_log(
            a8_init, t, seed=seed + 4000,
            base_lr=0.08, lam_start=10.0, lam_max=160.0,
            step_clip=0.75, patience_time=40.0,
            perturb_sigma=0.10, verbose=True
        )
        candidate_list.append(a5)
        if np.isfinite(obj5) and obj5 < global_obj:
            global_obj = obj5
            global_a = a5.copy()

    # Final micro-polish of whatever is currently best, if time remains.
    rem = deadline - time.monotonic()
    t = max(0.0, rem - 5.0)
    if t > 20.0 and global_a is not None:
        a_len = global_a.size
        if a_len >= 8000:
            base_lr = 0.05
            step_clip = 0.5
            lam_start = 20.0
        elif a_len >= 2000:
            base_lr = 0.07
            step_clip = 0.6
            lam_start = 15.0
        else:
            base_lr = 0.10
            step_clip = 0.8
            lam_start = 10.0

        print(f"\nphase 6: final polish n={a_len} ({t:.0f}s)")
        a6, obj6 = optimize_log(
            global_a, t, seed=seed + 5000,
            base_lr=base_lr, lam_start=lam_start, lam_max=250.0,
            step_clip=step_clip, patience_time=25.0,
            perturb_sigma=0.07, verbose=False
        )
        candidate_list.append(a6)
        if np.isfinite(obj6) and obj6 < global_obj:
            global_obj = obj6
            global_a = a6.copy()

    candidate_list.append(global_a)

    # Final validation using the official evaluator when available.
    best_a = None
    best_ver = float("inf")

    # Deduplicate by size/object identity is unnecessary; evaluating a few
    # candidates is cheap near the end.
    for a in candidate_list:
        if a is None:
            continue
        a = np.asarray(a, dtype=np.float64).reshape(-1)
        if a.size == 0:
            continue
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s < 0.01:
            continue

        a = normalize_sum(a, 1.0)
        try:
            ver = evaluate_sequence(a.tolist())
        except Exception:
            ver = local_obj(a)

        if np.isfinite(ver) and ver < best_ver:
            best_ver = float(ver)
            best_a = a.copy()
        print(f"final candidate: n={a.size}, evaluated={best_ver:.6f} (this={ver:.6f if np.isfinite(ver) else 'inf'})")

    if best_a is None:
        best_a = np.ones(4096, dtype=np.float64)
        try:
            best_ver = evaluate_sequence(best_a.tolist())
        except Exception:
            best_ver = local_obj(best_a)
        print(f"fallback: n={best_a.size}, evaluated={best_ver:.6f}")

    final_list = [float(x) for x in best_a]
    if len(final_list) == 0 or sum(final_list) < 0.01:
        final_list = [1.0] * 256

    print(f"\nFINAL local/final bound: {best_ver:.6f}, length={len(final_list)}")
    return final_list
