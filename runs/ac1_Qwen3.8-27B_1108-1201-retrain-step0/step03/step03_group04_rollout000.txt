Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
1. L-BFGS on log-sum-exp smooth approximation (increasing beta)
2. Cutting-plane LP "kicks" to escape local minima
3. Multi-phase optimization with multi-start exploration
Key innovation: exponential parameterization + L-BFGS second-order info
+ LP-based feasible direction for global exploration.

import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_cross_convolve(a, b):
    """Full cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:size]


def compute_grad_a(a, weights):
    """Gradient of sum_k weights[k]*c[k] w.r.t. a, where c = a*a (linear conv)."""
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_arr(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z_max = z.max()
    e = np.exp(z - z_max)
    s = e.sum()
    if s < 1e-300:
        return np.full_like(z, 1.0 / len(z))
    return e / s


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n):
    """Clamp to [0, 1000], rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def smooth_obj_and_grad(z, beta, n):
    """
    Log-sum-exp smooth approximation of max(a*a), parameterized via z.
    a = S * softmax(z), S = sqrt(2n).
    Returns (objective, gradient_wrt_z).
    """
    S = np.sqrt(2.0 * n)
    p = softmax_arr(z)
    a = S * p
    c = fft_convolve_self(a)
    # Log-sum-exp
    c_max = c.max()
    shifted = beta * (c - c_max)
    e = np.exp(shifted)
    Z = e.sum()
    if Z < 1e-300:
        obj = c_max
        w = np.full_like(c, 1.0 / len(c))
    else:
        obj = (c_max + np.log(Z)) / beta
        w = e / Z
    # Gradient w.r.t. a
    grad_a = compute_grad_a(a, w)
    # Convert to z-gradient (softmax mirror)
    center = np.dot(a, grad_a) / S
    grad_z = a * (grad_a - center)
    return obj, grad_z


def run_lbfgs_smooth(a_init, n, max_time, beta_start=50.0, beta_end=50000.0,
                     lbfgs_maxiter=200, beta_schedule='log', rng=None):
    """
    L-BFGS optimization of log-sum-exp objective with increasing beta.
    Uses exponential parameterization (z = log(a/a_mean)) to handle simplex.
    """
    a = safe_normalize(a_init, n)
    S = np.sqrt(2.0 * n)
    # Initialize z from a: z_i = log(a_i) - log(mean(a))
    a_safe = np.maximum(a, 1e-15)
    z = np.log(a_safe) - np.log(a_safe.mean())
    
    start = time.monotonic()
    best_a = a.copy()
    best_score = fft_convolve_self(a).max()
    
    # Schedule beta values
    n_phases = max(1, int(max_time / 3.0))  # ~3 seconds per phase
    if beta_schedule == 'log':
        betas = np.geomspace(beta_start, beta_end, min(n_phases, 30))
    else:
        betas = np.linspace(beta_start, beta_end, min(n_phases, 30))
    
    for beta in betas:
        if time.monotonic() - start > max_time - 2:
            break
        
        def obj_grad(z_vec):
            return smooth_obj_and_grad(z_vec, beta, n)
        
        # Run L-BFGS
        try:
            res = minimize(
                fun=lambda z_vec: smooth_obj_and_grad(z_vec, beta, n)[0],
                x0=z,
                jac=lambda z_vec: smooth_obj_and_grad(z_vec, beta, n)[1],
                method='L-BFGS-B',
                options={'maxiter': lbfgs_maxiter, 'gtol': 1e-8,
                         'maxls': 50, 'ftol': 1e-10}
            )
            if res.x is not None:
                z = res.x
                p = softmax_arr(z)
                a_candidate = S * p
                a_candidate = np.clip(a_candidate, 0.0, 1000.0)
                sc = a_candidate.sum()
                if sc > 1e-15:
                    a_candidate = a_candidate * (S / sc)
                c = fft_convolve_self(a_candidate)
                score = c.max()
                if score < best_score:
                    best_score = score
                    best_a = a_candidate.copy()
        except Exception:
            pass
        
        # Update a from z for next phase
        p = softmax_arr(z)
        a = S * p
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s > 1e-15:
            a = a * (S / s)
    
    return best_a, best_score


def run_lp_kick(f, M, n, max_lp_iters=30, K_init=50, K_max=200, rng=None):
    """
    Cutting-plane LP to find a direction g with max sum(g) s.t. (f*g)[k] <= M.
    Returns the LP solution g (normalized) or None.
    """
    S = np.sqrt(2.0 * n)
    fg = fft_convolve_self(f)
    
    for lp_iter in range(max_lp_iters):
        # Get binding constraints
        K = min(K_init + lp_iter * 10, K_max, 2 * n - 1)
        top_idx = np.argpartition(fg, -K)[-K:]
        
        # Build constraint matrix
        A_ub = np.zeros((K, n))
        for i, k in enumerate(top_idx):
            m_start = max(0, k - n + 1)
            m_end = min(k, n - 1)
            if m_start <= m_end:
                A_ub[i, m_start:m_end + 1] = f[k - np.arange(m_start, m_end + 1)]
        
        # Solve LP: max sum(g) s.t. A_ub @ g <= M, g >= 0
        # Scale M to make it dimensionless
        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=np.full(K, M),
                          bounds=[(0, None)] * n, method='highs')
        except Exception:
            return None
        
        if res.status != 0:
            return None
        
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            return None
        
        # Check all constraints
        fg_check = fft_cross_convolve(f, g)
        violations = np.where(fg_check > M + 1e-8)[0]
        
        if len(violations) == 0:
            # Optimal - return normalized g
            return g * (S / g_sum)
        else:
            # Add violated constraints to active set
            # Update fg for next iteration
            fg = fg_check
            if len(violations) > 50:
                # Too many violations, stop
                break
    
    return None


def run_subgradient_polish(a_init, n, max_time, rng, top_k_start=1, top_k_max=50,
                           step_scale=1.0):
    """FISTA-style projected subgradient with top-k weights."""
    S = np.sqrt(2.0 * n)
    a = safe_normalize(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    it = 0
    step = S / (n * 0.5) * step_scale
    
    while time.monotonic() - start < max_time:
        it += 1
        k = min(top_k_max, top_k_start + it // 100)
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
        
        # Top-k weights
        top_idx = np.argpartition(c, -k)[-k:]
        weights = np.zeros(len(c))
        top_vals = c[top_idx]
        total = top_vals.sum()
        if total > 1e-30:
            weights[top_idx] = top_vals / total
        else:
            weights = np.full(len(c), 1.0 / len(c))
        
        grad = compute_grad_a(y, weights)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = S / (n * 0.5) * step_scale / (1.0 + 0.002 * it)
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic perturbation
        if it % 1000 == 0 and rng is not None and rng.random() < 0.2:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
    
    return best_local, best_score


def run_coordinate_descent(a_init, n, max_time, rng, n_rounds=10000):
    """
    Coordinate descent: for each coordinate, find optimal value
    that minimizes the peak convolution.
    """
    S = np.sqrt(2.0 * n)
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_local = a.copy()
    
    # Precompute which convolution lags are affected by each coordinate
    # Coordinate i affects lags k where 0 <= k-i < n, i.e., k in [i, i+n-1]
    # (bounded by [0, 2n-2])
    
    it = 0
    while time.monotonic() - start < max_time and it < n_rounds:
        it += 1
        # Random coordinate
        i = rng.integers(0, n)
        
        c = fft_convolve_self(a)
        cur_max = c.max()
        
        # Try a few alternative values for a[i]
        # The affected lags are k in [i, i+n-1] ∩ [0, 2n-2]
        k_min = max(0, i)
        k_max = min(2 * n - 2, i + n - 1)
        
        # Current contribution at each affected lag:
        # c[k] = sum_j a[j]*a[k-j], the term involving a[i] is a[i]*a[k-i]
        # (appears once if i != k-i, twice if i == k-i)
        # We want to reduce max c[k] for k in affected range.
        
        # Simple: try reducing a[i] by various amounts
        best_val = a[i]
        best_peak = cur_max
        
        # Binary-ish search on a[i]
        candidates = [a[i] * 0.5, a[i] * 0.7, a[i] * 0.85, a[i] * 1.1, a[i] * 1.3]
        candidates = [max(0.0, min(1000.0, v)) for v in candidates]
        
        for val in candidates:
            a[i] = val
            # Renormalize (approximately)
            s = a.sum()
            if s > 1e-15:
                a_scaled = a * (S / s)
            else:
                a_scaled = a
            c_new = fft_convolve_self(a_scaled)
            peak = c_new.max()
            if peak < best_peak:
                best_peak = peak
                best_val = val
        
        a[i] = best_val
        s = a.sum()
        if s > 1e-15:
            a = a * (S / s)
        
        if best_peak < best_score - 1e-12:
            best_score = best_peak
            best_local = a.copy()
    
    return best_local, best_score


def try_initialization(n, init_type, rng, S):

        freq = rng.integers(3, 20)
        a = 0.5 + 0.4 * np.sin(2 * np.pi * freq * t + rng.uniform(0, 2 * np.pi))
        a = np.abs(a) + 0.1
    elif init_type == 'block':
        a = np.zeros(n)
        n_blocks = rng.integers(4, 16)
        block_size = n // n_blocks
        for k in range(n_blocks):
            if rng.random() > 0.3:
                a[k * block_size:(k + 1) * block_size] = rng.uniform(0.5, 1.5)
    elif init_type == 'two_scale':
        a = rng.uniform(0.1, 0.3, n)
        n_spikes = n // 10
        spike_idx = rng.choice(n, n_spikes, replace=False)
        a[spike_idx] = rng.uniform(1.0, 2.0, n_spikes)
    else:
        a = np.ones(n) * 0.5
    return safe_normalize(a, n)


def propose_candidate(seed=42, budget_s=2385, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: L-BFGS on log-sum-exp (increasing beta) - 35%
    Phase 2: LP kicks + subgradient - 25%
    Phase 3: Multi-start exploration - 20%
    Phase 4: Final polish (L-BFGS high beta + coordinate descent) - 20%
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s - 25.0)
    
    # Get initial sequence
    given_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given_a = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    if given_a is not None:
        n = len(given_a)
    else:
        n = 8192
        given_a = np.ones(n) * 0.5
    
    S = np.sqrt(2.0 * n)
    a_start = safe_normalize(given_a, n)
    c_start = fft_convolve_self(a_start)
    initial_score = c_start.max()
    global_best = initial_score
    global_a = a_start.copy()
    obj_start = 2.0 * n * initial_score / (S * S)
    print("Initial: n=%d, max_conv=%.6f, objective=%.6f" % (n, initial_score, obj_start))
    
    # === Phase 1: L-BFGS on log-sum-exp (35% of time) ===
    remaining = deadline - time.monotonic()
    phase1_time = 0.35 * remaining
    if phase1_time > 15:
        print("Phase 1: L-BFGS log-sum-exp (%.0f sec)" % phase1_time)
        t0 = time.monotonic()
        a_p1, score_p1 = run_lbfgs_smooth(a_start, n, phase1_time,
                                           beta_start=50.0, beta_end=50000.0,
                                           lbfgs_maxiter=150, beta_schedule='log', rng=rng)
        a_p1 = safe_normalize(a_p1, n)
        c_p1 = fft_convolve_self(a_p1)
        if c_p1.max() < global_best:
            global_best = c_p1.max()
            global_a = a_p1.copy()
            print("  L-BFGS: max_conv=%.6f, obj=%.6f [%.0fs]" % (
                global_best, 2 * n * global_best / S**2, time.monotonic() - t0))
        else:
            print("  L-BFGS: no improvement (%.6f vs %.6f)" % (c_p1.max(), global_best))
    
    # === Phase 2: LP kicks + subgradient (25%) ===
    remaining = deadline - time.monotonic()
    phase2_time = 0.25 * remaining
    if phase2_time > 10:
        print("Phase 2: LP kicks + subgradient (%.0f sec)" % phase2_time)
        t0 = time.monotonic()
        
        current_a = global_a.copy()
        current_M = fft_convolve_self(current_a).max()
        
        # Try LP kicks
        lp_attempts = 0
        while time.monotonic() < t0 + phase2_time * 0.6 and lp_attempts < 10:
            if time.monotonic() - t0 > phase2_time * 0.6:
                break
            M = fft_convolve_self(current_a).max()
            g = run_lp_kick(current_a, M, n, max_lp_iters=15, K_init=50, K_max=150, rng=rng)
            lp_attempts += 1
            if g is not None:
                # Blend
                t_blend = 0.3
                new_a = (1 - t_blend) * current_a + t_blend * g
                new_a = safe_normalize(new_a, n)
                c_new = fft_convolve_self(new_a)
                if c_new.max() < global_best:
                    global_best = c_new.max()
                    global_a = new_a.copy()
                    current_a = new_a.copy()
                    print("  LP kick %d: max_conv=%.6f, obj=%.6f" % (
                        lp_attempts, global_best, 2 * n * global_best / S**2))
                else:
                    current_a = new_a.copy()
            else:
                break
        
        # Subgradient polish
        remaining2 = deadline - time.monotonic()
        if remaining2 > 5:
            a_p2, score_p2 = run_subgradient_polish(global_a, n, remaining2, rng,
                                                     top_k_start=1, top_k_max=30,
                                                     step_scale=1.0)
            a_p2 = safe_normalize(a_p2, n)
            c_p2 = fft_convolve_self(a_p2)
            if c_p2.max() < global_best:
                global_best = c_p2.max()
                global_a = a_p2.copy()
                print("  Subgrad: max_conv=%.6f, obj=%.6f" % (
                    global_best, 2 * n * global_best / S**2))
    
    # === Phase 3: Multi-start exploration (20%) ===
    remaining = deadline - time.monotonic()
    phase3_time = 0.20 * remaining
    if phase3_time > 15:
        print("Phase 3: Multi-start exploration (%.0f sec)" % phase3_time)
        t0 = time.monotonic()
        init_types = ['random_exp', 'sinusoidal', 'two_scale', 'block']
        per_init = phase3_time / len(init_types)
        
        for init_type in init_types:
            if time.monotonic() > deadline - 30:
                break
            a_init = try_initialization(n, init_type, rng, S)
            # Quick L-BFGS
            a_try, score_try = run_lbfgs_smooth(a_init, n, per_init,
                                                 beta_start=100.0, beta_end=20000.0,
                                                 lbfgs_maxiter=100, beta_schedule='log', rng=rng)
            a_try = safe_normalize(a_try, n)
            c_try = fft_convolve_self(a_try)
            if c_try.max() < global_best:
                global_best = c_try.max()
                global_a = a_try.copy()
                print("  %s: max_conv=%.6f, obj=%.6f" % (
                    init_type, global_best, 2 * n * global_best / S**2))
            else:
                print("  %s: %.6f (no improvement)" % (init_type, c_try.max()))
    
    # === Phase 4: Final polish (20%) ===
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print("Phase 4: Final polish (%.0f sec)" % remaining)
        t0 = time.monotonic()
        
        # L-BFGS with very high beta
        polish_time = remaining * 0.5
        a_f1, score_f1 = run_lbfgs_smooth(global_a, n, polish_time,
                                           beta_start=10000.0, beta_end=200000.0,
                                           lbfgs_maxiter=200, beta_schedule='log', rng=rng)
        a_f1 = safe_normalize(a_f1, n)
        c_f1 = fft_convolve_self(a_f1)
        if c_f1.max() < global_best:
            global_best = c_f1.max()
            global_a = a_f1.copy()
            print("  L-BFGS high-beta: max_conv=%.6f, obj=%.6f" % (
                global_best, 2 * n * global_best / S**2))
        
        # Coordinate descent
        remaining2 = deadline - time.monotonic()
        if remaining2 > 5:
            a_f2, score_f2 = run_coordinate_descent(global_a, n, remaining2, rng,
                                                     n_rounds=50000)
            a_f2 = safe_normalize(a_f2, n)
            c_f2 = fft_convolve_self(a_f2)
            if c_f2.max() < global_best:
                global_best = c_f2.max()
                global_a = a_f2.copy()
                print("  CoordDesc: max_conv=%.6f, obj=%.6f" % (
                    global_best, 2 * n * global_best / S**2))
        
        # One more subgradient pass
        remaining3 = deadline - time.monotonic()
        if remaining3 > 5:
            a_f3, score_f3 = run_subgradient_polish(global_a, n, remaining3, rng,
                                                     top_k_start=1, top_k_max=10,
                                                     step_scale=0.5)
            a_f3 = safe_normalize(a_f3, n)
            c_f3 = fft_convolve_self(a_f3)
            if c_f3.max() < global_best:
                global_best = c_f3.max()
                global_a = a_f3.copy()
                print("  Final subgrad: max_conv=%.6f, obj=%.6f" % (
                    global_best, 2 * n * global_best / S**2))
    
    # === Final output ===
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    final_obj = 2.0 * n * global_best / (S * S)
    print("FINAL: n=%d, max_conv=%.6f, objective=%.6f" % (n, global_best, final_obj))
    
    # Verify
    try:
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verified: %.6f" % ver)
        else:
            print("Verification failed!")
            final_list = [float(x) for x in a_start]
    except Exception as e:
        print("Verification error: %s" % str(e))
