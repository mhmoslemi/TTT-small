def project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0: return v
    # if all zeros
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    r = np.arange(1,n+1)
    cond = (u - (css - S)/r) > 0
    rho = max(idx where cond)
    theta = (css[rho-1]-S)/rho
    return np.maximum(v-theta,0)

```python
"""
Search for a non-negative sequence minimizing
    2*n*max(convolve(a,a)) / (sum(a)^2)

The main novelty over the previous state of the art is a direct
"convolution-box / square-root spectrum" alternating method:

1. Work on the normalized sequence a (sum = sqrt(2n)).
2. Let A = FFT(a), C = A^2, c = IFFT(C).  Then c is the true linear
   self-convolution.
3. Project c onto the convex box {0 <= c <= M, sum(c) = S^2}.
4. Use a damped Newton square-root update in the frequency domain to move
   A toward a square root of the new target spectrum C_target.
5. Project the resulting time-domain sequence back to the non-negative
   simplex and repeat while gradually lowering M.

This is combined with several conventional refinements:
- FFT FISTA on a smooth log-sum-exp approximation of the peak.
- Active-set top-lag sub-gradient line search.
- Cheap structural perturbations (anti-palindromic amplification).
- Multi-length search (typically 8192 and 16384).
"""

import time
import numpy as np
from scipy.fft import fft, ifft


def _next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x)=S}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if S <= 0:
        return np.zeros(n, dtype=np.float64)

    if not np.all(np.isfinite(v)):
        v = np.where(np.isfinite(v), v, 0.0)

    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (css - S) / rho > 0.0

    if not np.any(cond):
        return np.full(n, S / float(n), dtype=np.float64)

    k = int(np.max(np.where(cond)[0]))
    theta = (css[k] - S) / float(k + 1)
    out = np.maximum(v - theta, 0.0)

    # Remove tiny roundoff sum errors.
    diff = S - out.sum()
    if abs(diff) > 0.0:
        pos = np.where(out > 0.0)[0]
        if pos.size > 0:
            out[pos] += diff / float(pos.size)
            out[pos] = np.maximum(out[pos], 0.0)
        else:
            out = np.full(n, S / float(n), dtype=np.float64)

    return out


def _normalize_sum(a, S):
    """Clamp to [0,1000], make non-negative, and rescale to sum S."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a
    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.0)
    for _ in range(3):
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s <= 1e-15:
            return np.full(a.size, S / float(a.size), dtype=np.float64)
        a = a * (S / s)
    return np.clip(a, 0.0, 1000.0)


def _conv_from_A(A, n):
    """Return linear self-convolution from A = FFT(padded a)."""
    c = np.real(ifft(A * A))[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _conv_from_a(a, N):
    A = fft(a, N)
    return _conv_from_A(A, len(a))


def _score_from_c(c, n, S):
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    S2 = float(S) * float(S)
    if S2 <= 0.0:
        return np.inf
    return (2.0 * float(n) * m) / S2


def _clip_sum(c, theta, M):
    return np.clip(c - theta, 0.0, M).sum()


def _box_project_c_bisect(c, M, target_sum):
    """L2 projection of c onto {0<=x<=M, sum(x)=target_sum} by bisection."""
    c = np.maximum(np.asarray(c, dtype=np.float64), 0.0)
    L = c.size
    if target_sum <= 0.0:
        return np.zeros(L, dtype=np.float64)
    if M <= 0.0:
        return np.zeros(L, dtype=np.float64)
    if target_sum >= M * L:
        return np.full(L, M, dtype=np.float64)

    lo = float(np.min(c)) - M - 1.0
    hi = float(np.max(c)) + 1.0

    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = _clip_sum(c, mid, M)
        if s >= target_sum:
            lo = mid
        else:
            hi = mid

    theta = 0.5 * (lo + hi)
    return np.clip(c - theta, 0.0, M)


def _box_project_c(c, M, target_sum):
    """Fast projection with bisection as a fallback."""
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    if target_sum <= 0.0:
        return np.zeros(L, dtype=np.float64)
    if M <= 0.0:
        return np.zeros(L, dtype=np.float64)

    if target_sum >= M * L:
        return np.full(L, M, dtype=np.float64)

    x = np.minimum(c, M)
    x = np.maximum(x, 0.0)

    for _ in range(10):
        diff = target_sum - x.sum()
        if abs(diff) <= 1e-10 * max(1.0, float(target_sum)):
            break
        if diff > 0.0:
            mask = x < (M - 1e-12)
            if not np.any(mask):
                break
            add = diff / float(np.sum(mask))
            x[mask] += add
            np.minimum(x, M, out=x)
        else:
            mask = x > 1e-12
            if not np.any(mask):
                break
            sub = -diff / float(np.sum(mask))
            x[mask] -= sub
            np.maximum(x, 0.0, out=x)

    if abs(target_sum - x.sum()) > 1e-7 * max(1.0, float(target_sum)):
        x = _box_project_c_bisect(c, M, target_sum)
    return x


def _smooth_c1d(c, width):
    """Small moving-average smoothing, preserving sum and non-negativity."""
    if width <= 1:
        return c
    width = int(width)
    if width % 2 == 0:
        width += 1
    k = np.ones(width, dtype=np.float64) / float(width)
    out = np.convolve(c, k, mode="same")
    s = out.sum()
    cs = c.sum()
    if s > 1e-15:
        out = out * (cs / s)
    return np.maximum(out, 0.0)


def _square_root_step(A, C, alpha, eps_frac=1e-10):
    """
    Damped Newton step for updating A toward sqrt(C), preserving the
    current square-root branch as much as possible.
    """
    absA = np.abs(A)
    maxA = float(absA.max()) if absA.size else 1.0
    eps = eps_frac * max(1.0, maxA)

    denom = A.copy()
    small = absA < eps
    if np.any(small):
        phase = np.angle(A)
        denom[small] = eps * np.exp(1j * phase[small])

    step = 0.5 * (A + C / denom)
    return A + alpha * (step - A)


def _resample(a, n_target):
    """Resample a 1D array to length n_target by linear interpolation."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.ones(n_target, dtype=np.float64)
    n = a.size
    if n == int(n_target):
        return a.copy()
    if n == 1:
        return np.full(int(n_target), float(a[0]), dtype=np.float64)

    x_old = np.arange(n, dtype=np.float64) / float(max(n - 1, 1))
    x_new = np.arange(int(n_target), dtype=np.float64) / float(max(int(n_target) - 1, 1))
    return np.interp(x_new, x_old, a)


def _random_init(n, rng):
    i = np.arange(n, dtype=np.float64)
    x = i / float(max(n - 1, 1))
    phase = rng.uniform(0.0, 2.0 * np.pi)
    a = 1.0
    a += 0.35 * np.sin(2.0 * np.pi * 2.0 * x + phase)
    a += 0.20 * np.sin(2.0 * np.pi * 5.0 * x + rng.uniform(0.0, 2.0 * np.pi))
    a += 0.10 * rng.standard_normal(n)

    # Mild endpoint bumps, useful for edge lags.
    W = max(8, n // 64)
    a += 0.6 * np.exp(-i / float(W))
    a += 0.6 * np.exp(-i[::-1] / float(W))

    a = np.maximum(a, 0.01)
    return a


def _antisym_explode(a, S, gamma):
    """
    Amplify the anti-palindromic component of a:
        a = b + h, where b = (a + rev(a))/2, h = (a - rev(a))/2.
    Larger gamma makes the sequence more anti-symmetric, which often reduces
    the central convolution lag, subject to non-negativity projection.
    """
    rev = a[::-1]
    b = 0.5 * (a + rev)
    h = 0.5 * (a - rev)
    c = b + gamma * h
    c = np.maximum(c, 0.0)
    return _normalize_sum(c, S)


def _evaluate_fast(a, n, N, S):
    c = _conv_from_a(a, N)
    return _score_from_c(c, n, S)


def spectral_box_optimize(
    a_init,
    n,
    N,
    S,
    deadline,
    rng,
    alpha=0.25,
    smooth_width=5,
    min_target=1.05,
    m_rel=0.998,
    blend=0.5,
    max_stall=400,
):
    """
    Alternating convolution-box projection and frequency-domain square-root
    Newton steps.  This is the main new global/local optimizer.
    """
    a = _normalize_sum(a_init, S)
    A = fft(a, N)
    c = _conv_from_A(A, n)
    score = _score_from_c(c, n, S)

    best_a = a.copy()
    best_score = score

    L = 2 * n - 1
    S2 = float(S) * float(S)
    alpha_cur = float(alpha)
    stall = 0
    reject_streak = 0

    while time.monotonic() < deadline:
        c = _conv_from_A(A, n)
        if not np.all(np.isfinite(c)):
            a = _normalize_sum(best_a + rng.normal(0.0, 0.01 * S / np.sqrt(float(n)), n), S)
            A = fft(a, N)
            stall = 0
            continue

        sc = float(np.max(c))
        score = _score_from_c(c, n, S)

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            a = _normalize_sum(best_a + rng.normal(0.0, 0.005 * S / np.sqrt(float(n)), n), S)
            A = fft(a, N)
            stall = 0
            alpha_cur = min(0.6, alpha_cur * 1.2)

        current_best = min(sc, best_score)
        target = max(min_target, current_best * m_rel)
        if target >= sc:
            target = max(min_target, sc - 1e-5)

        ctarget = _box_project_c(c, target, S2)
        if smooth_width > 1:
            ctarget = _smooth_c1d(ctarget, smooth_width)
            ctarget = _box_project_c(ctarget, target, S2)

        Ctarget = fft(ctarget, N)
        Ccur = A * A
        Cnew = (1.0 - blend) * Ccur + blend * Ctarget

        accepted = False

        # Try a normal step, then a very small step.
        for trial in range(2):
            step_alpha = alpha_cur if trial == 0 else max(0.03, 0.25 * alpha_cur)
            A_step = _square_root_step(A, Cnew, step_alpha)
            a_step = np.real(ifft(A_step))[:n]

            if not np.all(np.isfinite(a_step)):
                a_step = np.zeros(n, dtype=np.float64)

            a_step = _normalize_sum(a_step, S)
            A_step = fft(a_step, N)
            c_step = _conv_from_A(A_step, n)
            score_step = _score_from_c(c_step, n, S)

            if np.isfinite(score_step) and (score_step <= score + 1e-9 or score_step < best_score):
                if score_step < best_score:
                    best_score = score_step
                    best_a = a_step.copy()
                    stall = 0
                a = a_step
                A = A_step
                accepted = True
                if score_step < score:
                    alpha_cur = min(0.6, alpha_cur * 1.04)
                break

        if not accepted:
            alpha_cur = max(0.05, alpha_cur * 0.75)
            reject_streak += 1

            if reject_streak > 80:
                reject_streak = 0
                if rng.random() < 0.7:
                    a = _normalize_sum(a + rng.normal(0.0, 0.002 * S / np.sqrt(float(n)), n), S)
                    A = fft(a, N)
                else:
                    a = _normalize_sum(best_a + rng.normal(0.0, 0.004 * S / np.sqrt(float(n)), n), S)
                    A = fft(a, N)
                stall = 0

        # Occasional random perturbation to escape branch/local fixed points.
        if rng.random() < 0.004:
            a = _normalize_sum(a + rng.normal(0.0, 0.002 * S / np.sqrt(float(n)), n), S)
            A = fft(a, N)
            stall = 0

    return best_a, best_score


def smooth_fista(
    a_init,
    n,
    N,
    S,
    deadline,
    rng,
    beta_start=30.0,
    beta_max=3000.0,
    beta_growth=1.0015,
    lr_start=0.3,
    lr_max=1.0,
    max_stall=2000,
):
    """
    FISTA-style descent on a log-sum-exp smooth approximation of the peak
    convolution.  The softmax weights become increasingly active-set-like as
    beta grows.
    """
    a = _normalize_sum(a_init, S)
    y = a.copy()
    t = 1.0
    beta = float(beta_start)
    lr = float(lr_start)

    A = fft(a, N)
    c = _conv_from_A(A, n)
    best_score = _score_from_c(c, n, S)
    best_a = a.copy()

    L = 2 * n - 1
    stall = 0

    while time.monotonic() < deadline:
        A_y = fft(y, N)
        c_y = _conv_from_A(A_y, n)

        if not np.all(np.isfinite(c_y)):
            break

        sc_y = float(np.max(c_y))
        score_y = _score_from_c(c_y, n, S)

        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        z = beta * (c_y - sc_y)
        p = np.exp(z)
        psum = float(p.sum())
        if psum <= 0.0 or not np.isfinite(psum):
            p = np.zeros(L, dtype=np.float64)
            p[int(np.argmax(c_y))] = 1.0
        else:
            p /= psum

        w = np.zeros(N, dtype=np.float64)
        w[:L] = p

        B = fft(w, N)
        grad = 2.0 * np.real(ifft(B * np.conj(A_y)))[:n]

        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n, dtype=np.float64)

        gnorm = float(np.linalg.norm(grad))
        if gnorm <= 1e-14:
            break

        accepted = False
        for factor in (1.0, 0.5, 0.25):
            step = lr * factor
            x = _project_simplex(y - step * grad, S)
            A_x = fft(x, N)
            c_x = _conv_from_A(A_x, n)
            score_x = _score_from_c(c_x, n, S)

            if not np.isfinite(score_x):
                continue

            if score_x <= score_y + 1e-9 or score_x < best_score:
                if score_x < best_score:
                    best_score = score_x
                    best_a = x.copy()
                    stall = 0
                    lr = min(lr_max, lr * 1.08)
                else:
                    stall += 1
                    lr = max(0.01, lr * 0.92)

                t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
                y = x + ((t - 1.0) / t_new) * (x - a)
                y = _project_simplex(y, S)
                a = x
                t = t_new
                beta = min(float(beta_max), beta * float(beta_growth))
                accepted = True
                break
            else:
                lr = max(0.01, lr * 0.7)

        if not accepted:
            lr = max(0.01, lr * 0.5)
            stall += 1
            y = a.copy()
            t = max(1.0, t * 0.5)

            if stall > max_stall:
                a = _normalize_sum(best_a + rng.normal(0.0, 0.003 * S / np.sqrt(float(n)), n), S)
                y = a.copy()
                t = 1.0
                beta = float(beta_start)
                lr = float(lr_start)
                stall = 0

    return best_a, best_score


def active_line_search(
    a_init,
    n,
    N,
    S,
    deadline,
    rng,
    top_frac=0.01,
):
    """
    1D line search along the sub-gradient of the near-active top lags.
    Useful for final polishing.
    """
    a = _normalize_sum(a_init, S)
    A = fft(a, N)
    c = _conv_from_A(A, n)
    best_score = _score_from_c(c, n, S)
    best_a = a.copy()

    L = 2 * n - 1
    base = 0.2
    stall = 0

    while time.monotonic() < deadline:
        A = fft(a, N)
        c = _conv_from_A(A, n)

        if not np.all(np.isfinite(c)):
            break

        sc = float(np.max(c))
        score = _score_from_c(c, n, S)

        if score < best_score:
            best_score = score
            best_a = a.copy()

        thresh = sc - max(1e-8, top_frac * sc)
        idx = np.where(c >= thresh)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(c))], dtype=int)

        w = np.zeros(N, dtype=np.float64)
        w[idx] = 1.0 / float(idx.size)

        B = fft(w, N)
        grad = 2.0 * np.real(ifft(B * np.conj(A)))[:n]

        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n, dtype=np.float64)

        gnorm = float(np.linalg.norm(grad))
        if gnorm <= 1e-14:
            break

        local_best_a = a
        local_best_score = score

        for mult in (0.25, 0.5, 1.0, 2.0, 4.0):
            at = _project_simplex(a - base * mult * grad, S)
            at = _normalize_sum(at, S)
            c_at = _conv_from_a(at, N)
            s_at = _score_from_c(c_at, n, S)
            if np.isfinite(s_at) and s_at < local_best_score:
                local_best_score = s_at
                local_best_a = at

        if local_best_score < score - 1e-12:
            a = local_best_a
            if local_best_score < best_score:
                best_score = local_best_score
                best_a = a.copy()
            base = min(2.0, base * 1.08)
            stall = 0
        else:
            base = max(0.02, base * 0.75)
            stall += 1
            if stall > 400:
                a = _normalize_sum(best_a + rng.normal(0.0, 0.002 * S / np.sqrt(float(n)), n), S)
                base = 0.2
                stall = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400.0):
    """
    Main search entry point.  Returns a Python list of non-negative floats.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0
    budget = min(budget, 2390.0)

    deadline = t_start + budget - 12.0

    # ------------------------------------------------------------------
    # Load any previous construction if available.
    # ------------------------------------------------------------------
    given = None
    given_n = 0
    try:
        g = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if g.size >= 1024 and float(np.sum(np.clip(g, 0.0, 1000.0))) > 0.0:
            given = np.clip(g, 0.0, 1000.0)
            given_n = int(given.size)
    except Exception:
        given = None
        given_n = 0

    global_a = None
    global_n = 8192
    global_score = np.inf

    if given is not None and 4096 <= given_n <= 16384:
        S = np.sqrt(2.0 * float(given_n))
        N = _next_pow2(2 * given_n - 1)
        a0 = _normalize_sum(given, S)
        s0 = _evaluate_fast(a0, given_n, N, S)
        if np.isfinite(s0):
            global_a = a0
            global_n = given_n
            global_score = float(s0)
            print("given length=%d score=%.8f" % (global_n, global_score))

    # Fallback best if we have nothing valid yet.
    if global_a is None:
        global_n = 8192
        S = np.sqrt(2.0 * float(global_n))
        global_a = np.full(global_n, S / float(global_n), dtype=np.float64)
        global_score = 2.0
        print("fallback uniform length=%d score=%.8f" % (global_n, global_score))

    # ------------------------------------------------------------------
    # Candidate lengths.  Cap at 16384 to keep final np.convolve robust.
    # ------------------------------------------------------------------
    candidate_lengths = []
    if 4096 <= given_n <= 16384:
        candidate_lengths.append(int(given_n))
    candidate_lengths.extend([8192, 16384])

    deduped = []
    for nval in candidate_lengths:
        if nval not in deduped:
            deduped.append(int(nval))
    candidate_lengths = deduped

    # ------------------------------------------------------------------
    # Quick phase: probe different lengths/initializations with the
    # convolution-box method.
    # ------------------------------------------------------------------
    t_quick_start = time.monotonic()
    quick_total = min(360.0, max(0.0, (deadline - t_quick_start) * 0.25))
    if quick_total > 30.0:
        fracs = [0.55, 0.30, 0.15]
        for i, n in enumerate(candidate_lengths):
            if time.monotonic() > deadline - 60.0:
                break
            frac = fracs[min(i, len(fracs) - 1)]
            alloc = quick_total * frac
            if alloc < 20.0:
                continue

            S = np.sqrt(2.0 * float(n))
            N = _next_pow2(2 * n - 1)

            if global_a is not None and len(global_a) == n:
                init = global_a.copy()
            elif global_a is not None:
                init = _resample(global_a, n)
                if len(init) != n:
                    init = _random_init(n, rng)
            else:
                init = _random_init(n, rng)

            init = _normalize_sum(init, S)
            init_score = _evaluate_fast(init, n, N, S)
            print("quick start n=%d score=%.8f" % (n, init_score))

            a_best, s_best = spectral_box_optimize(
                init,
                n,
                N,
                S,
                time.monotonic() + alloc,
                rng,
                alpha=0.30,
                smooth_width=3,
                min_target=1.05,
                m_rel=0.990,
                blend=0.60,
                max_stall=350,
            )

            if np.isfinite(s_best) and s_best < global_score:
                global_a = a_best
                global_n = n
                global_score = float(s_best)
                print("quick improved n=%d score=%.8f" % (global_n, global_score))

    # ------------------------------------------------------------------
    # Deep phase: focus on the current best but keep returning the best
    # found.  Alternate spectral-box, FISTA, active line search, and
    # structural anti-palindromic perturbations.
    # ------------------------------------------------------------------
    deep_iterations = 0
    while time.monotonic() < deadline - 30.0:
        deep_iterations += 1
        n = int(global_n)
        S = np.sqrt(2.0 * float(n))
        N = _next_pow2(2 * n - 1)
        rem = max(0.0, deadline - time.monotonic())

        # Block A: spectral-box refinement.
        t1 = min(rem * 0.42, 280.0)
        if t1 > 15.0:
            a1, s1 = spectral_box_optimize(
                global_a,
                n,
                N,
                S,
                time.monotonic() + t1,
                rng,
                alpha=0.13,
                smooth_width=5,
                min_target=1.02,
                m_rel=0.9995,
                blend=0.50,
                max_stall=500,
            )
            if np.isfinite(s1) and s1 < global_score:
                global_a = a1
                global_n = n
                global_score = float(s1)
                print("deep spectral n=%d iter=%d score=%.8f" % (global_n, deep_iterations, global_score))

        # Block B: cheap structural explorations from the best sequence.
        if time.monotonic() < deadline - 60.0:
            S = np.sqrt(2.0 * float(n))
            N = _next_pow2(2 * n - 1)
            best_struct_a = None
            best_struct_s = np.inf

            for gamma in (1.5, 2.0, 3.0, 5.0, 8.0):
                cand = _antisym_explode(global_a, S, gamma)
                sc = _evaluate_fast(cand, n, N, S)
                if np.isfinite(sc) and sc < best_struct_s:
                    best_struct_s = sc
                    best_struct_a = cand

            # Also try a slight overall smoothing.
            cand = _smooth_c1d(global_a, 5)
            cand = _normalize_sum(cand, S)
            sc = _evaluate_fast(cand, n, N, S)
            if np.isfinite(sc) and sc < best_struct_s:
                best_struct_s = sc
                best_struct_a = cand

            if best_struct_a is not None and best_struct_s < global_score:
                global_a = best_struct_a
                global_score = float(best_struct_s)
                print("deep structural n=%d iter=%d score=%.8f" % (global_n, deep_iterations, global_score))

            # If structural move helped a lot, give it a short spectral polish.
            if best_struct_a is not None and best_struct_s < global_score + 1e-12:
                t_struct = min(80.0, max(0.0, deadline - time.monotonic()) * 0.10)
                if t_struct > 10.0:
                    a2, s2 = spectral_box_optimize(
                        global_a,
                        n,
                        N,
                        S,
                        time.monotonic() + t_struct,
                        rng,
                        alpha=0.15,
                        smooth_width=4,
                        min_target=1.02,
                        m_rel=0.9997,
                        blend=0.50,
                        max_stall=300,
                    )
                    if np.isfinite(s2) and s2 < global_score:
                        global_a = a2
                        global_n = n
                        global_score = float(s2)
                        print("deep structural polish n=%d iter=%d score=%.8f" % (global_n, deep_iterations, global_score))

        # Block C: FISTA polish.
        t2 = min(rem * 0.35, 260.0)
        if t2 > 15.0:
            a3, s3 = smooth_fista(
                global_a,
                n,
                N,
                S,
                time.monotonic() + t2,
                rng,
                beta_start=40.0,
                beta_max=3500.0,
                beta_growth=1.0018,
                lr_start=0.25,
                lr_max=1.0,
                max_stall=2500,
            )
            if np.isfinite(s3) and s3 < global_score:
                global_a = a3
                global_n = n
                global_score = float(s3)
                print("deep fista n=%d iter=%d score=%.8f" % (global_n, deep_iterations, global_score))

        # Block D: active top-lag line search.
        t3 = min(rem * 0.15, 120.0)
        if t3 > 8.0:
            a4, s4 = active_line_search(
                global_a,
                n,
                N,
                S,
                time.monotonic() + t3,
                rng,
                top_frac=0.012,
            )
            if np.isfinite(s4) and s4 < global_score:
                global_a = a4
                global_n = n
                global_score = float(s4)
                print("deep line n=%d iter=%d score=%.8f" % (global_n, deep_iterations, global_score))

        # If one length is clearly best, occasionally try transferring to
        # the other common length.
        if deep_iterations % 3 == 2 and time.monotonic() < deadline - 240.0:
            if global_n == 8192:
                n_other = 16384
            elif global_n == 16384:
                n_other = 8192
            else:
                n_other = 16384

            S_other = np.sqrt(2.0 * float(n_other))
            N_other = _next_pow2(2 * n_other - 1)
            init_other = _resample(global_a, n_other)
            init_other = _normalize_sum(init_other, S_other)

            alloc_other = min(180.0, max(0.0, deadline - time.monotonic()) * 0.15)
            if alloc_other > 20.0:
                a_other, s_other = spectral_box_optimize(
                    init_other,
                    n_other,
                    N_other,
                    S_other,
                    time.monotonic() + alloc_other,
                    rng,
                    alpha=0.18,
                    smooth_width=4,
                    min_target=1.02,
                    m_rel=0.999,
                    blend=0.55,
                    max_stall=400,
                )
                if np.isfinite(s_other) and s_other < global_score:
                    global_a = a_other
                    global_n = n_other
                    global_score = float(s_other)
                    print("cross-length improved n=%d score=%.8f" % (global_n, global_score))

    # ------------------------------------------------------------------
    # Final verification and fallback.
    # ------------------------------------------------------------------
    if global_a is None or len(global_a) == 0:
        global_a = np.ones(8192, dtype=np.float64)

    global_a = np.asarray(global_a, dtype=np.float64)
    global_a = np.clip(global_a, 0.0, 1000.0)

    # If it became too large for safe external evaluation, resample down.
    if len(global_a) > 16384:
        global_a = _resample(global_a, 16384)

    S_final = np.sqrt(2.0 * float(len(global_a)))
    global_a = _normalize_sum(global_a, S_final)

    final_list = [float(x) for x in global_a]

    try:
        ver = evaluate_sequence(final_list)
    except Exception:
        ver = np.inf

    if not np.isfinite(ver):
        fallback_n = 8192
        fallback_S = np.sqrt(2.0 * float(fallback_n))
        final_list = [fallback_S / float(fallback_n)] * fallback_n
        print("final verification failed; returning uniform fallback")
    else:
        print("final length=%d verified score=%.8f internal score=%.8f" % (
            len(final_list), float(ver), global_score
        ))

    return final_list
```
