def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}"""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    # Sort in descending order
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = 0
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = 0
        theta = v.mean()
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_spectrum_features(a, N, n):
    """Compute FFT and power spectrum of a padded to size N."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    return A, P, c


def spectral_exp_shape(a, N, n, top_k, alpha, rng, target=1.45):
    """
    Single spectral shaping step using exponential modification.
    Targets top-k peak lags simultaneously.
    Returns (new_a, c_max).
    """
    S = a.sum()
    A, P, c = compute_spectrum_features(a, N, n)
    c_max = c.max()

    if c_max <= target:
        return a, c_max

    # Find top-k peak lags
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return a, c_max

    # Weight by excess above target
    excess = np.maximum(top_vals - target, 0.0)
    excess_sum = excess.sum()
    if excess_sum < 1e-30:
        excess = top_vals
        excess_sum = total

    weights = excess / excess_sum

    # Compute effective cosine (weighted sum over top lags)
    m = np.arange(N)
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_indices):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)

    # Exponential shaping: only reduce where cos_eff > 0
    alpha_eff = alpha * min(1.0, (c_max - target) / max(c_max, 1e-10) + 0.3)
    P_new = P * np.exp(-alpha_eff * np.maximum(cos_eff, 0.0))

    # Small random phase perturbation for exploration
    if rng is not None and rng.random() < 0.2:
        phase_noise = rng.normal(0, 0.05 * alpha_eff, N)
        A_new = np.sqrt(P_new) * np.exp(1j * (np.angle(A) + phase_noise))
    else:
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))

    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new, c_max


def spectral_waterfill_loop(a_init, n, N, max_time, rng, target=1.45,
                            top_k_init=5, alpha_init=0.4):
    """
    Iterative spectral water-filling with adaptive parameters.
    Returns (best_a, best_cmax).
    """
    a = a_init.copy()
    S = a.sum()
    best_a = a.copy()
    best_cmax = np.inf
    it = 0
    stall = 0
    last_best = np.inf

    while time.monotonic() - max_time < 0:
        it += 1
        # Adaptive parameters
        top_k = min(50, top_k_init + it // 20)
        alpha = alpha_init / (1.0 + 0.05 * it)

        a_new, c_max = spectral_exp_shape(a, N, n, top_k, alpha, rng, target)

        # Verify actual improvement
        c_check = fft_convolve_self(a_new)
        actual_cmax = c_check.max()

        if actual_cmax < best_cmax:
            best_cmax = actual_cmax
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            if it % 500 == 0:
                print("    SW iter %d: %.6f" % (it, best_cmax))
        else:
            a = a_new.copy()  # Still move forward
            stall += 1
            if stall > 100:
                # Perturb to escape
                a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
                a_new, c_max = spectral_exp_shape(a, N, n, top_k, alpha, rng, target)
                c_check = fft_convolve_self(a_new)
                actual_cmax = c_check.max()
                if actual_cmax < best_cmax:
                    best_cmax = actual_cmax
                    best_a = a_new.copy()
                    a = a_new.copy()
                stall = 0

        if best_cmax <= target:
            break

        # Occasional larger perturbation
        if it % 500 == 250 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)

    return best_a, best_cmax


def random_spectral_perturb(a, N, n, S, sigma, rng):
    """Add random complex noise to the spectrum, project back."""
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    noise = sigma * (rng.standard_normal(N) + 1j * rng.standard_normal(N))
    A_new = A + noise
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_simplex(a_new, S)
    return a_new


def nesterov_polish(a_init, n, S, max_time, rng, top_k_init=20, top_k_max=256):
    """Nesterov-accelerated projected gradient with top-k subgradient."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 20)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Compute top-k subgradient
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        # Gradient via FFT cross-correlation with weighted lags
        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]
        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.002 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 2000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.004 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def coordinate_descent_polish(a_init, n, S, max_time, rng, n_top=30, n_rounds=100):
    """
    Coordinate descent on top-contributing coordinates.
    For each selected coordinate, try several multipliers and keep the best.
    """
    a = a_init.copy()
    start = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    N = 1 << (2 * n - 1).bit_length()

    multipliers = [0.3, 0.5, 0.7, 1.3, 1.5, 0.1]

    for rnd in range(n_rounds):
        if time.monotonic() - start > max_time:
            break

        c = fft_convolve_self(a)
        c_max = c.max()
        peak_lag = int(np.argmax(c))

        if c_max < best_score:
            best_score = c_max
            best_a = a.copy()

        # Find top-contributing coordinates at the peak lag
        low = max(0, peak_lag - n + 1)
        high = min(peak_lag, n - 1)
        contribs = np.zeros(n)
        for i in range(low, high + 1):
            j = peak_lag - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        top_coords = np.argsort(contribs)[::-1][:n_top]

        improved = False
        for idx in top_coords:
            i = int(idx)
            if time.monotonic() - start > max_time:
                break
            j = peak_lag - i
            if j < 0 or j >= n or i == j:
                continue

            orig_val = a[i]
            best_frac = 1.0
            best_cmax = c_max

            for frac in multipliers:
                a[i] = orig_val * frac
                # Quick evaluation of affected lags
                # Only check a subset of lags for speed
                c_affected = c.copy()
                # Lags affected by changing a[i]: k=i to k=i+n-1
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                # Recompute affected lags
                for k in range(k_lo, k_hi + 1):
                    j2 = k - i
                    if 0 <= j2 < n:
                        c_affected[k] += (a[i] - orig_val) * a[j2]
                    j3 = i
                    k2 = k - j3
                    if 0 <= k2 < n and k2 != j2:
                        c_affected[k] += (a[i] - orig_val) * a[k2]

                new_max = c_affected[k_lo:k_hi + 1].max()
                # Also check if other lags were the peak
                other_max = max(c_affected[:k_lo].max() if k_lo > 0 else 0,
                               c_affected[k_hi + 1:].max() if k_hi + 1 < len(c_affected) else 0)
                new_cmax = max(new_max, other_max)

                if new_cmax < best_cmax:
                    best_cmax = new_cmax
                    best_frac = frac

            a[i] = orig_val * best_frac
            if best_frac != 1.0:
                improved = True

        # Rescale to maintain sum
        a = project_simplex(a, S)

        # If no improvement in a full round, perturb
        if not improved and rng.random() < 0.5:
            top_coords2 = np.argsort(contribs)[::-1][:min(10, n_top)]
            for idx in top_coords2:
                a[int(idx)] *= rng.uniform(0.5, 1.5)
            a = project_simplex(a, S)

    return best_a, best_score


def structured_init(n, rng, method="sin"):
    """Create structured initializations with good autocorrelation properties."""
    t = np.arange(n) / n
    if method == "sin":
        a = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * t) + 0.2 * np.sin(2 * np.pi * 7 * t) \
            + 0.1 * np.sin(2 * np.pi * 13 * t) + 0.05 * rng.standard_normal(n)
    elif method == "chirp":
        a = 0.5 + 0.3 * np.sin(2 * np.pi * (2 * t + 3 * t * t)) + 0.1 * rng.standard_normal(n)
    elif method == "bump":
        a = 0.3 + 0.5 * np.exp(-0.5 * ((t - 0.3) * 4) ** 2) + 0.2 * np.exp(-0.5 * ((t - 0.7) * 4) ** 2)
    elif method == "alternating":
        # Alternating high/low with random perturbation
        a = np.where(np.arange(n) % 2 == 0, 1.0, 0.3) + 0.1 * rng.standard_normal(n)
    elif method == "piecewise":
        # Random piecewise constant with many pieces
        n_pieces = 64
        piece_len = n // n_pieces
        heights = 0.3 + 0.7 * rng.random(n_pieces)
        a = np.repeat(heights, piece_len)
        if len(a) < n:
            a = np.concatenate([a, rng.random(n - len(a)) * 1.0 + 0.3])
        a = a[:n]
    else:
        a = 0.5 + 0.3 * rng.standard_normal(n)

    a = np.clip(a, 0.01, 2.0)
    return a


def safe_normalize(a, n):
    """Clamp to [0, 1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

                                                   top_k_init=5, alpha_init=0.4)
                if update_best(a_sw2):
                    a_curr = a_sw2.copy()
                    stall_count = 0
                    print("  [P2] perturb+wf(%.2f): %.6f" % (sigma, global_best))
                    break
            stall_count = 0

    print("=== Phase 2 done. Best: %.6f ===" % global_best)

    # === Phase 3: Final polish (25% time) ===
    print("=== Phase 3: Final polish ===")
    t_phase3 = time.monotonic()
    remaining = deadline - t_phase3

    if global_a is not None and remaining > 5:
        # Long Nesterov with high top-k
        seg1 = min(remaining * 0.30, 120.0)
        a_p1, _ = nesterov_polish(global_a, n, S, seg1, rng,
                                  top_k_init=50, top_k_max=300)
        if update_best(a_p1):
            print("  [P3] nesterov high-k: %.6f" % global_best)

        # Coordinate descent on best
        rem2 = deadline - time.monotonic()
        if rem2 > 5:
            seg2 = min(rem2 * 0.30, 60.0)
            a_p2, _ = coordinate_descent_polish(global_a, n, S, seg2, rng,
                                                n_top=40, n_rounds=200)
            if update_best(a_p2):
                print("  [P3] coord: %.6f" % global_best)

        # Final spectral water-filling with very small alpha (fine tuning)
        rem3 = deadline - time.monotonic()
        if rem3 > 3:
            a_p3, _ = spectral_waterfill_loop(global_a, n, N,
                                              time.monotonic() + min(rem3 * 0.3, 30.0),
                                              rng, target=1.45,
                                              top_k_init=20, alpha_init=0.1)
            if update_best(a_p3):
                print("  [P3] fine wf: %.6f" % global_best)

        # One more coordinate descent
        rem4 = deadline - time.monotonic()
        if rem4 > 3:
            a_p4, _ = coordinate_descent_polish(global_a, n, S, rem4, rng,
                                                n_top=50, n_rounds=300)
            if update_best(a_p4):
                print("  [P3] final coord: %.6f" % global_best)

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning sequence of length %d, internal score=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
