from scipy.optimize import linprog

def lp_find_direction(a, M, S):
    """
    Find g >= 0 such that conv(a,g)[k] <= M for all k.
    Maximize sum(g).
    
    a: array of length n
    M: bound on cross-convolution
    S: target sum (for rescaling later)
    
    Returns: g (unnormalized, will be rescaled by caller)
    """
    n = len(a)
    
    # conv(a, g)[k] = sum_i a[i] * g[i+k] for valid i, i+k
    # This is a linear function of g.
    # The coefficient of g[j] in conv(a,g)[k] is a[k-j] if 0 <= k-j < n
    
    # We need: for each k in 0..2n-2, sum_{j: 0<=k-j<n} a[k-j] * g[j] <= M
    # which is: (A @ g)[k] <= M where A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    
    # A is a (2n-1) x n matrix, banded with bandwidth n
    
    # For linprog: minimize -sum(g) s.t. A @ g <= M, g >= 0
    # c = -1 for all variables
    # A_ub = A (the convolution matrix)
    # b_ub = M * ones(2n-1)
    # bounds = [(0, None)] * n
    
    # But A is large: (2n-1) x n = 16383 x 8192 = ~134 million entries
    # However, it's sparse (each row has at most n non-zero entries, but actually each column has 2n-1 non-zero entries)
    # Actually, A is a Toeplitz-like matrix. Let me think...
    
    # A[k, j] = a[k-j] if 0 <= k-j < n, i.e., k-n < j <= k
    # So for row k, the non-zero entries are in columns max(0, k-n+1) to min(n-1, k)
    # Each row has at most min(n, k+1, 2n-1-k) non-zero entries
    
    # This is a banded matrix. We can use sparse format.
    
    from scipy.sparse import lil_matrix
    A_sparse = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A_sparse[k, j] = a[k - j]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    result = linprog(c, A_ub=A_sparse.tocsr(), b_ub=b_ub, 
                     bounds=[(0, None)] * n, method='highs')
    
    if result.success:
        return result.x
    else:

        return np.ones(n) * S / n

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    Compute subgradient of max(conv) using top-k active peaks.
    Returns a vector of length n.

    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total

        j_start = max(0, lag - n + 1)

        src_indices = lag - indices

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated subgradient
method with adaptive top-k peaks, backtracking line search, and periodic
exploration. Key insight: the self-convolution peak can be well below
sum(a_i^2), allowing objective values below 2. We exploit this by using
asymmetric sequences and adaptive peak-targeting strategies.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient_toppk(a, c, top_k):
    """
    Compute subgradient of max(conv(a,a)) using top-k active peaks.
    grad[j] = 2 * sum_l w_l * a[k_l - j] (when valid).
    """
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total

    grad = np.zeros(n)
    for i in range(len(top_idx)):
        w = weights[i]
        lag = int(top_idx[i])
        # a[lag - j] valid when 0 <= lag - j < n, i.e., lag-n+1 <= j <= lag
        j_start = max(0, lag - n + 1)
        j_end = min(n, lag + 1)
        if j_start < j_end:
            length = j_end - j_start
            src = a[lag - j_end + 1: lag - j_start + 1][::-1]
            grad[j_start:j_end] += 2.0 * w * src

    return grad


def compute_subgradient_softmax(a, c, beta):

    # This is the cross-correlation of w and a

def nesterov_subgradient(a_init, n, S, max_time, rng,

    """
    Nesterov-accelerated subgradient method with adaptive top-k.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    no_improve = 0

    if step_init is None:
        step_init = S / (2.0 * np.sqrt(n))

    step = step_init
    L_est = 2.0 * S  # Upper bound on Lipschitz constant

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive top-k: increase over time
        frac = min(1.0, it / (max_time * 10))
        k = max(k_start, int(k_start + frac * (k_end - k_start)))
        k = min(k, 512)

        # Compute convolution and objective
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute subgradient
        grad = compute_subgradient_toppk(y, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size: decrease slowly, reset on jumps
        step = step_init / (1.0 + 0.002 * it)
        step = max(step, 1e-8)

        # Nesterov lookahead
        x_new = project_simplex(y - step * grad, S)

        # Regular momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic noise injection to escape local minima
        if it % noise_period == 0 and rng.random() < 0.3:

            noise = rng.normal(0, scale, n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
            no_improve = 0

        if no_improve > noise_period * 3:

            noise = rng.normal(0, jump_scale, n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0

def equalize_peaks(a_init, n, S, max_time, rng, k_targets=8):
    """
    Phase that tries to equalize the top-k peaks.
    Uses a weighted subgradient that penalizes peaks that are above
    the median of the top-k.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Get top peaks
        top_k = min(k_targets + it // 20, 64)
        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]

        # Weight: peaks above median get proportional weight
        median_val = np.median(top_vals)
        excess = np.maximum(0, top_vals - median_val)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            # All equal, use uniform weights
            weights = np.ones(top_k) / top_k
        else:
            weights = excess / total_excess

        # Compute weighted subgradient
        grad = np.zeros(n)
        for i in range(len(top_idx)):
            w = weights[i]
            if w < 1e-12:
                continue
            lag = int(top_idx[i])
            j_start = max(0, lag - n + 1)
            j_end = min(n, lag + 1)
            if j_start < j_end:
                src = a[lag - j_end + 1: lag - j_start + 1][::-1]
                grad[j_start:j_end] += 2.0 * w * src

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = S / (4.0 * np.sqrt(n) * (1.0 + 0.001 * it))
        x_new = project_simplex(y - step * grad, S)

        # Simple momentum
        y = 0.5 * x_new + 0.5 * y
        y = np.maximum(y, 0.0)
        a = x_new

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""

    s = a.sum()
    S = np.sqrt(2.0 * float(n))

        return np.ones(n) * S / n
    return a * (S / s)

            parent_arr = np.clip(given, 0, 1000)
    except Exception:
        pass

    if parent_arr is not None:
        n = len(parent_arr)
    else:
        n = 8192

    S = np.sqrt(2.0 * float(n))

    print("=== Starting optimization, n=%d, S=%.4f ===" % (n, S))
    print("Budget: %.1f seconds" % budget_s)

    # --- Phase 1: Quick exploration with diverse initializations ---
    print("--- Phase 1: Exploration ---")
    explore_time = max(30.0, (deadline - time.monotonic()) * 0.35)
    per_init_time = max(5.0, explore_time / 12)

    inits = generate_initializations(n, rng)

    # Add parent as first initialization
    if parent_arr is not None:
        inits.insert(0, parent_arr)
        print("Using parent sequence of length %d" % n)

    # Add the best known pattern: strong asymmetry
    tt = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.exp(5.0 * tt) + 0.01)
    inits.append(np.exp(5.0 * (1.0 - tt)) + 0.01)

    results = []
    for i, init_arr in enumerate(inits):
        if time.monotonic() > deadline - 120:
            break

        arr = np.clip(init_arr, 0, 1000).astype(np.float64)
        if arr.sum() < 0.01:
            continue

        # Quick optimization: top-k subgradient
        a_opt, score = nesterov_subgradient(
            arr, n, S, per_init_time, rng,
            k_start=4, k_end=64,
            noise_period=300, noise_scale=0.005
        )
        a_norm = safe_normalize(a_opt, n)
        c = fft_convolve_self(a_norm)
        obj = c.max()

        results.append((obj, a_norm, i))
        if obj < global_best:
            global_best = obj
            global_a = a_norm.copy()
            print("  Init %d: obj=%.6f (new best)" % (i, obj))
        else:
            print("  Init %d: obj=%.6f" % (i, obj))

    print("After Phase 1: best=%.6f" % global_best)

    # --- Phase 2: Deep optimization on top candidates ---
    print("--- Phase 2: Deep optimization ---")
    # Sort results and take top 3
    results.sort(key=lambda x: x[0])
    top_results = results[:min(3, len(results))]

    remaining = deadline - time.monotonic()
    per_deep = max(10.0, remaining * 0.3 / max(1, len(top_results)))

    for rank, (score, arr, idx) in enumerate(top_results):
        if time.monotonic() > deadline - 60:
            break

        print("  Deep opt on candidate %d (score=%.6f), time=%.1fs" %
              (idx, score, per_deep))

        # Strategy A: Nesterov top-k with increasing k
        a_a, s_a = nesterov_subgradient(
            arr, n, S, per_deep * 0.4, rng,
            k_start=8, k_end=128,
            noise_period=400, noise_scale=0.003
        )
        a_a = safe_normalize(a_a, n)
        c_a = fft_convolve_self(a_a)
        if c_a.max() < global_best:
            global_best = c_a.max()
            global_a = a_a.copy()
            print("    Top-k: %.6f" % global_best)

        # Strategy B: Softmax
        remaining_b = deadline - time.monotonic() - 60
        if remaining_b > 10:
            a_b, s_b = nesterov_softmax(
                global_a, n, S, min(per_deep * 0.3, remaining_b * 0.4), rng,
                beta_start=20.0, beta_end=300.0,
                noise_period=300, noise_scale=0.002
            )
            a_b = safe_normalize(a_b, n)
            c_b = fft_convolve_self(a_b)
            if c_b.max() < global_best:
                global_best = c_b.max()
                global_a = a_b.copy()
                print("    Softmax: %.6f" % global_best)

        # Strategy C: Peak equalization
        remaining_c = deadline - time.monotonic() - 60
        if remaining_c > 10:
            a_c, s_c = equalize_peaks(
                global_a, n, S, min(per_deep * 0.3, remaining_c * 0.4), rng,
                k_targets=8
            )
            a_c = safe_normalize(a_c, n)
            c_c = fft_convolve_self(a_c)
            if c_c.max() < global_best:
                global_best = c_c.max()
                global_a = a_c.copy()
                print("    Equalize: %.6f" % global_best)

    # --- Phase 3: Final polishing ---
    print("--- Phase 3: Final polish ---")
    remaining = deadline - time.monotonic()
    if remaining > 10 and global_a is not None:
        print("  Remaining time: %.1f sec, current best: %.6f" % (remaining, global_best))

        # Polish 1: Very fine top-k (k=2 to 8)
        seg1 = min(remaining * 0.3, 300.0)
        a_p1, s_p1 = nesterov_subgradient(
            global_a, n, S, seg1, rng,
            k_start=2, k_end=16,
            step_init=S / (5.0 * np.sqrt(n)),
            noise_period=1000, noise_scale=0.0005
        )
        a_p1 = safe_normalize(a_p1, n)
        c_p1 = fft_convolve_self(a_p1)
        if c_p1.max() < global_best:
            global_best = c_p1.max()
            global_a = a_p1.copy()
            print("  Polish 1: %.6f" % global_best)

        # Polish 2: High-beta softmax
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            seg2 = min(remaining2 * 0.3, 200.0)
            a_p2, s_p2 = nesterov_softmax(
                global_a, n, S, seg2, rng,
                beta_start=100.0, beta_end=1000.0,
                step_init=S / (5.0 * np.sqrt(n)),
                noise_period=800, noise_scale=0.0003
            )
            a_p2 = safe_normalize(a_p2, n)
            c_p2 = fft_convolve_self(a_p2)
            if c_p2.max() < global_best:
                global_best = c_p2.max()
                global_a = a_p2.copy()
                print("  Polish 2: %.6f" % global_best)

        # Polish 3: Equalize with more peaks
        remaining3 = deadline - time.monotonic()
        if remaining3 > 10:
            seg3 = min(remaining3 * 0.3, 200.0)
            a_p3, s_p3 = equalize_peaks(
                global_a, n, S, seg3, rng, k_targets=16
            )
            a_p3 = safe_normalize(a_p3, n)
            c_p3 = fft_convolve_self(a_p3)
            if c_p3.max() < global_best:
                global_best = c_p3.max()
                global_a = a_p3.copy()
                print("  Polish 3: %.6f" % global_best)

        # Polish 4: Final ultra-fine
        remaining4 = deadline - time.monotonic()
        if remaining4 > 5:
            a_p4, s_p4 = nesterov_subgradient(
                global_a, n, S, remaining4, rng,
                k_start=1, k_end=4,
                step_init=S / (10.0 * np.sqrt(n)),
                noise_period=2000, noise_scale=0.0001
            )
            a_p4 = safe_normalize(a_p4, n)
            c_p4 = fft_convolve_self(a_p4)
            if c_p4.max() < global_best:
                global_best = c_p4.max()
                global_a = a_p4.copy()
                print("  Polish 4: %.6f" % global_best)

    # --- Verification ---
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * 1.0

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Verification: score=%s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
    print("Internal best: %.6f, length: %d" % (global_best, len(final_list)))

    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]

    return final_list
