"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- Multigrid initialization: optimize at small n, upsample, refine
- L-BFGS-B on softmax parameterization with log-sum-exp continuation
- Nesterov-accelerated top-k subgradient with adaptive step size
- Coordinate descent refinement (1D convex sub-problems)
- Multiple restarts with diverse initializations
- FFT-accelerated gradient computation O(n log n)
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_a(a, w):
    """
    Compute grad_a[j] = 2 * sum_k w[k] * a[j-k] (cross-correlation via FFT).
    This is the gradient of (1/2) * sum_k w[k] * b_k w.r.t. a[j],
    where b = a * a.
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """
    Compute smoothed objective and gradient w.r.t. z (unconstrained softmax params).
    a = S * softmax(z), objective = (1/beta)*log(sum exp(beta*c_k))
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_grad_a(a, w)

    # grad_z[k] = a[k] * (grad_a[k] - dot(grad_a, a)/S)
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                            beta_schedule=None, maxiter_per_beta=300):
    """
    Run L-BFGS-B with continuation on beta (smooth-to-sharp).
    Returns best (a, score) found.
    """
    if beta_schedule is None:
        beta_schedule = [3, 5, 10, 20, 50, 100, 200, 400, 800, 1500, 3000, 5000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        elapsed_beta = 0.0
        n_attempts = 1
        if beta >= 200 and max_time > 60:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 5:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.02, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            t0 = time.monotonic()
            try:
                result = scipy_minimize(
                    fun, z,
                    jac=jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_beta,
                        'maxfun': maxiter_per_beta * 3,
                        'ftol': 1e-15,
                        'gtol': 1e-10,
                        'maxls': 50,
                        'iprint': -1
                    }
                )
            except Exception:
                break

            elapsed_beta += time.monotonic() - t0
            z = result.x
            z = z - z.mean()

            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s

            c = fft_convolve_self(a)
            score = c.max()

            if score < best_score:
                best_score = score
                best_a = a.copy()

        if elapsed_beta > 0:
            print("    beta=%4d: score=%.8f (%d attempts, %.1fs)" %
                  (beta, best_score, n_attempts, elapsed_beta))

    return best_a, best_score


def run_nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=5, top_k_max=512):
    """Nesterov-accelerated top-k subgradient with adaptive step."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    base_step = 2.0 * S / n

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: decrease slowly
        step = base_step / (1.0 + 0.001 * it)
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 3 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 5000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def coordinate_descent_pass(a, n, S, n_sweeps=1, rng=None):
    """
    Cyclic coordinate descent. For each coordinate i, find optimal a_i
    given all others fixed (keeping sum = S by adjusting a_i and redistributing
    the excess to a random other coordinate... actually, let's just do free
    coordinate descent without the sum constraint, then renormalize).
    
    Actually, since the objective is scale-invariant, we can optimize max_b
    subject to a >= 0, and the sum will adjust naturally. But that's not right
    because the objective depends on the ratio max_b / sum^2.
    
    Let me think again. The objective is 2*n*max_b/sum^2. If we scale a by c,
    max_b scales by c^2 and sum^2 scales by c^2, so the ratio is unchanged.
    So we can fix sum(a) = S (any positive value) and just minimize max_b.
    
    For coordinate descent with fixed sum: change a_i by delta, change a_j by -delta
    for some j. This is a 2D problem.
    
    Alternatively, we can just do: for each i, minimize max_b subject to a_i >= 0
    and sum(a) = S, where all other a_j are fixed. The sum constraint means
    a_i is determined by the other a_j, so there's no freedom!
    
    OK so pure coordinate descent with a sum constraint doesn't work directly.
    
    Alternative: minimize the objective 2*n*max_b/sum^2 directly. This is
    scale-invariant, so we can fix one coordinate (say a_0 = 1) and optimize
    the rest. Or we can use a different parameterization.
    
    Simplest approach: for each i, try a few values of a_i (geometric spacing)
    and keep the best. This is a form of "coordinate search" rather than
    proper descent.
    """
    a = np.clip(a, 0.0, 1000.0).copy()
    s = a.sum()
    S_target = np.sqrt(2.0 * float(n))
    a = a * (S_target / s) if s > 1e-15 else np.ones(n) * S_target / n
    
    best_score = fft_convolve_self(a).max()
    
    for sweep in range(n_sweeps):
        for i in range(n):
            old_val = a[i]
            # Try a few multiplier values
            for mult in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                if time.monotonic() - (time.monotonic() - 0) > 10000:  # safety
                    break
                a[i] = old_val * mult
                if a[i] < 0:
                    a[i] = 0
                # Renormalize
                s_new = a.sum()
                if s_new < 1e-15:
                    break
                a_norm = a * (S_target / s_new)
                score = fft_convolve_self(a_norm).max()
                if score < best_score:
                    best_score = score
                    # Accept
                    a = a_norm.copy()
                else:
                    a[i] = old_val
            a = a * (S_target / a.sum())
    
    return a, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def multigrid_initialization(n_target, max_time, rng, seed=42):
    """
    Multigrid approach: optimize at small n, upsample to larger n.
    Returns a sequence of length n_target.
    """
    n = 128
    S_small = np.sqrt(2.0 * n)
    
    # Start with random at small scale
    a = rng.random(n) + 0.1
    a = a * S_small / a.sum()
    
    levels = []
    while n < n_target:
        # Optimize at current scale
        t0 = time.monotonic()
        a, score = run_lbfgs_continuation(
            a, n, S_small, min(30, max_time * 0.05), rng,
            beta_schedule=[5, 10, 20, 50, 100, 200],
            maxiter_per_beta=100
        )
        levels.append((n, a.copy(), score))
        print("  MG: n=%d, score=%.6f, %.1fs" % (n, score, time.monotonic() - t0))
        
        if time.monotonic() > max_time * 0.3:
            break
        
        # Upsample to 2n
        n_new = min(2 * n, n_target)
        # Use linear interpolation for upsample
        indices = np.linspace(0, n - 1, n_new)
        a_new = np.interp(indices, np.arange(n), a)
        S_small = np.sqrt(2.0 * n_new)
        a = a_new * S_small / a_new.sum()
        n = n_new
    
    # Final level
    if n < n_target:
        # Upsample to target
        indices = np.linspace(0, n - 1, n_target)
        a = np.interp(indices, np.arange(n), a)
    
    S = np.sqrt(2.0 * float(n_target))
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s > 1e-15:
        a = a * S / s
    else:
        a = np.ones(n_target) * S / n_target
    
    return a

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- L-BFGS-B on softmax parameterization with log-sum-exp continuation (increasing beta)
- Multiple diverse initializations + perturbation restarts
- Nesterov-accelerated top-k subgradient with adaptive step
- Aggressive high-beta polishing to converge to local optimum
- FFT-accelerated gradient computation O(n log n) per evaluation
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_a(a, w):
    """
    Compute grad_a[j] = 2 * sum_k w[k] * a_{k-j} via FFT cross-correlation.
    This is the gradient of sum_k w[k]*b_k w.r.t. a[j] where b = a*a.
    """
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """
    Compute smoothed objective and gradient w.r.t. z (unconstrained softmax params).
    a = S * softmax(z), objective approx = (1/beta)*log(sum exp(beta*c_k)) ~ max(c_k)
    """
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_grad_a(a, w)

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def run_lbfgs(a_init, n, S, max_time, beta_schedule, maxiter_per_beta=200, rng=None):
    """
    Run L-BFGS-B with continuation on beta. Returns (a, score).
    """
    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break

        # At very high beta, try a second attempt with perturbation
        n_attempts = 1
        if beta >= 500 and max_time > 60 and rng is not None:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 3:
                break
            if attempt > 0 and rng is not None:
                z = z + rng.normal(0, 0.01, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            try:
                result = scipy_minimize(
                    fun, z,
                    jac=jac,
                    method='L-BFGS-B',
                    options={
                        'maxiter': maxiter_per_beta,
                        'maxfun': maxiter_per_beta * 3,
                        'ftol': 1e-15,
                        'gtol': 1e-10,
                        'maxls': 50,
                        'iprint': -1
                    }
                )
                z = result.x
            except Exception:
                pass

            z = z - z.mean()
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s2 = a.sum()
            if s2 < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s2

            c = fft_convolve_self(a)
            score = c.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def run_nesterov(a_init, n, S, max_time, rng, top_k_init=3, top_k_max=200):
    """Nesterov-accelerated top-k subgradient with adaptive step."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 100)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        total = top_vals.sum()
        if total < 1e-30:
            break
        weights = top_vals / total

        w_full = np.zeros(2 * n - 1)
        for i, k in enumerate(top_indices):
            w_full[int(k)] = weights[i]

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step: base / (1 + gamma * it) with base calibrated to problem scale
        step = (S / np.sqrt(n)) / (1.0 + 0.001 * it)
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic restart with noise
        if it % 10000 == 0 and rng is not None and rng.random() < 0.15:
            y = project_simplex(y + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

        if np.isfinite(ver):
            print("\n=== VERIFIED SCORE: %.6f ===" % ver)
        else:
            print("evaluate_sequence returned inf, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning length=%d, internal score=%.8f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
