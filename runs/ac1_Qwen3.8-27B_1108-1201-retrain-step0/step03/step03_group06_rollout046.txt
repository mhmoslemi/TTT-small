def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}.
    O(n log n) algorithm."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)

                             alpha_scale=1.0, verbose=False):

    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s  # normalize to sum = 1
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    x_prev = a.copy()
    t = 1.0
    
    start = time.monotonic()
    step = 0
    
    while time.monotonic() - start < time_budget:
        # Update t
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_next
        
        # Extrapolation
        y = a + beta * (a - x_prev)
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-15:
            y = a.copy()
        else:
            y = y / s_y  # project onto simplex (approximate)
        
        # Compute objective at y
        c_y = fft_convolve_self(y)
        obj_y = c_y.max()
        
        if obj_y < best_obj - 1e-12:
            best_obj = obj_y
            best_a = y.copy()
        
        # Find top-K peaks
        if K_peaks == 1:
            k_peak = int(np.argmax(c_y))
            # Compute subgradient
            g = compute_subgrad_at_peak(y, k_peak, n)
        else:
            top_k = np.argsort(c_y)[-K_peaks:]
            # Weight subgradients by peak values
            g = np.zeros(n)
            w_sum = 0
            for k in top_k:
                w = c_y[k]
                g += w * compute_subgrad_at_peak(y, k, n)
                w_sum += w
            g /= w_sum if w_sum > 0 else 1
        
        # Projected step
        step_size = alpha_scale / t_next
        x_new = y - step_size * g
        # Project onto simplex
        x_new = project_simplex(x_new)
        
        x_prev = a
        a = x_new
        t = t_next
        step += 1
        
        if verbose and step % 1000 == 0:
            elapsed = time.monotonic() - start
            print("  Step %d: obj=%.8f, best=%.8f, elapsed=%.1fs" % (step, obj_y, best_obj, elapsed))
    
    return best_a, best_obj
