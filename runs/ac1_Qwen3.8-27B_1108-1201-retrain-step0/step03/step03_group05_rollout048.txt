"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using:
- Multi-length exploration (4096, 8192, 16384)
- Nesterov accelerated projected gradient with FFT and backtracking
- LP-based direction finding via cvxpy for global progress
- L-BFGS with log-sum-exp continuation for local refinement
- Multiple structured initializations
- Cross-length transfer and perturbation restarts
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
        theta = (cumsum[n-1] - S) / n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_cross_corr(a, w):
    """Cross-correlation: result[j] = sum_k w[k] * a[k-j] (via FFT)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    result = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return result

def softmax_obj_grad(z, beta, S):
    """Smoothed objective and gradient via log-sum-exp."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp
    
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta
    
    w = exp_bc / sum_exp_bc
    grad_a = compute_cross_corr(a, w)
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)
    
    return obj, grad_z

def run_nesterov_backtrack(a_init, n, S, max_time, rng, top_k_init=10, top_k_max=1024):
    """Nesterov with backtracking line search and FFT gradients."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    it = 0
    step = 1.0
    max_step = 2.0 * S / (n * 0.5)
    
    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = y.copy()
        
        # Get top-k weights
        if top_k >= 2*(n-1):
            # Use all
            w = c / c.sum() if c.sum() > 0 else np.ones(2*n-1)/(2*n-1)
        else:
            top_indices = np.argpartition(c, -top_k)[-top_k:]
            top_vals = c[top_indices]
            total = top_vals.sum()
            if total < 1e-30:
                w = np.ones(2*n-1) / (2*n-1)
            else:
                w = np.zeros(2*n-1)
                w[top_indices] = top_vals / total
        
        grad = compute_cross_corr(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        c_y = fft_convolve_self(y)
        f_y = c_y.max()
        
        accepted = False
        for _ in range(20):
            x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
            c_x = fft_convolve_self(x_new)
            f_x = c_x.max()
            
            if f_x < f_y - 1e-10:  # sufficient decrease
                accepted = True
                break
            step *= 0.5
        
        if not accepted:
            x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 0:
            y = y * S / y_sum
        a = x_new
        t_mom = t_new
        
        # Adaptive step size: increase if we accepted on first try
        if step < max_step and accepted:
            step = min(max_step, step * 1.05)
        
        # Occasional random restart
        if it % 2000 == 0 and rng.random() < 0.05:
            y = project_simplex(y + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            t_mom = 1.0
            step = 1.0
    
    return best_a, best_score

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
        theta = (cumsum[n - 1] - S) / n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
        theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_cross_corr(a, w):
    """Cross-correlation gradient: grad_a[j] = 2 * sum_k w[k] * a_shifted[j-k]."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):

    grad_a = compute_cross_corr(a, w)

         + 0.15 * np.cos(2 * np.pi * 17 * tt)
         + 0.1 * np.sin(2 * np.pi * 31 * tt)
         + 0.08 * np.cos(2 * np.pi * 43 * tt))
    a = np.maximum(a, 0.02)
    inits.append(("multi_sin", a * S / a.sum()))

    # Binary-like (low PAPR inspired)
    binary = rng.choice([0.3, 1.0], size=n)
    # Smooth it slightly
    from scipy.ndimage import uniform_filter1d
    smoothed = uniform_filter1d(binary, size=max(3, n // 512))
    smoothed = np.maximum(smoothed, 0.01)
    inits.append(("binary_smooth", smoothed * S / smoothed.sum()))

    # Edge-biased (energy near edges reduces wrap-around peaks)
    a = np.exp(-2.0 * np.abs(tt - 0.5) * 3.0) + 0.1
    inits.append(("edge_bump", a * S / a.sum()))

    # Checker pattern with envelope
    freq = max(2, n // 512)
    checker = 0.6 + 0.3 * np.cos(2 * np.pi * freq * tt)
    envelope = 1.0 - 0.3 * np.abs(tt - 0.5)
    a = checker * envelope + 0.05
    inits.append(("checker_env", a * S / a.sum()))

    # Random with slow envelope
    a = (rng.random(n) + 0.3) * (1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt))
    a = np.maximum(a, 0.01)
    inits.append(("rand_envelope", a * S / a.sum()))

    # Golay-inspired: use complementary sequence structure
    # For power-of-2 lengths, we can use recursive construction
    if n & (n - 1) == 0:  # power of 2
        # Recursive: start with [1], repeatedly apply [a, a] and [a, -a] -> [a+a, 0] and [a-a, 0] mapped to non-negative
        # Simplified: use a Kronecker product pattern
        base = np.array([1.0, 1.0, 0.5, 0.5])
        pattern = base.copy()
        while len(pattern) < n:
            pattern = np.concatenate([pattern, pattern * 0.7])
            pattern = pattern[:min(len(pattern), n)]
        pattern = pattern[:n]
        a = pattern + 0.1 * rng.random(n)
        inits.append(("golay_like", a * S / a.sum()))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses multi-length exploration with Nesterov + L-BFGS hybrid.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, budget_s - 25.0)

    global_best = np.inf
    global_a = None
    global_n = 0

    print("=== Starting optimization, budget=%.0fs ===" % budget_s)
    print("Target: 1.5027, Previous best: 1.508805")

    # --- Get initial sequences ---
    extra_inits = []
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            extra_inits.append(("given", np.clip(given, 0.0, 1000.0)))
    except Exception:
        pass

    # Determine which lengths to try
    # Strategy: focus on 8192 and 16384, with 4096 as backup
    lengths = [8192, 16384, 4096]

    # Time allocation (roughly proportional to expected value)
    # Give 45% to 8192, 40% to 16384, 15% to 4096
    time_alloc = {8192: 0.45, 16384: 0.40, 4096: 0.15}

    # Final reserve for polishing
    reserve = 120.0
    usable = max(0.0, budget_s - 25.0 - reserve)

    for n_target in lengths:
        if time.monotonic() > deadline - 60:
            break

        alloc_per_n = usable * time_alloc.get(n_target, 0.1)
        n = n_target
        S = np.sqrt(2.0 * float(n))

        print("\n--- Length n=%d, budget=%.0fs ---" % (n, alloc_per_n))

        # Generate initializations
        inits = generate_initializations(n, rng, S)

        # Add given sequence if length matches
        for name, arr in extra_inits:
            if len(arr) == n:
                inits.append(("given_exact", arr.copy()))
            elif len(arr) > n:
                # Downsample
                idx = np.linspace(0, len(arr) - 1, n).astype(int)
                inits.append(("given_sub", arr[idx]))
            elif len(arr) < n:
                # Upsample
                a_up = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(arr)), arr)
                inits.append(("given_up", a_up))

        # Evaluate all inits quickly
        init_scores = []
        for name, arr in inits:
            a_norm = safe_normalize(arr, n)
            c = fft_convolve_self(a_norm)
            score = c.max()
            init_scores.append((name, a_norm, score))
            if score < global_best:
                global_best = score
                global_a = a_norm.copy()
                global_n = n

        # Sort by score
        init_scores.sort(key=lambda x: x[2])
        print("  Best init: %s score=%.8f" % (init_scores[0][0], init_scores[0][2]))
        for nm, _, sc in init_scores[:4]:
            print("    %s: %.8f" % (nm, sc))

        # Phase A: Nesterov on top initializations
        nesterov_budget = alloc_per_n * 0.5
        best_nesterov_a = None
        best_nesterov_score = np.inf

        n_top_inits = min(3, len(init_scores))
        per_init_nest = nesterov_budget / n_top_inits

        for i in range(n_top_inits):
            if time.monotonic() > deadline - 60:
                break
            name, a_init, init_score = init_scores[i]
            print("  Nesterov [%s] (budget=%.0fs, init=%.6f):" % (name, per_init_nest, init_score))

            a_new, score = run_nesterov_adaptive(
                a_init, n, S, per_init_nest, rng,
                top_k_init=1, top_k_max=max(256, n // 16)
            )
            a_new = safe_normalize(a_new, n)
            c = fft_convolve_self(a_new)
            score = c.max()

            if score < best_nesterov_score:
                best_nesterov_score = score
                best_nesterov_a = a_new.copy()
            if score < global_best:
                global_best = score
                global_a = a_new.copy()
                global_n = n
                print("    NEW GLOBAL BEST: %.8f" % global_best)

        # Phase B: L-BFGS continuation on best Nesterov result
        lbfgs_budget = alloc_per_n * 0.45
        if best_nesterov_a is not None:
            remaining = min(lbfgs_budget, deadline - time.monotonic() - 30)
            if remaining > 10:
                print("  L-BFGS (budget=%.0fs):" % remaining)
                a_new, score = run_lbfgs_aggressive(
                    best_nesterov_a, n, S, remaining, rng,
                    beta_schedule=[10, 25, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 40000],
                    maxiter_per_beta=300
                )
                a_new = safe_normalize(a_new, n)
                c = fft_convolve_self(a_new)
                score = c.max()
                if score < best_nesterov_score:
                    best_nesterov_score = score
                    best_nesterov_a = a_new.copy()
                if score < global_best:
                    global_best = score
                    global_a = a_new.copy()
                    global_n = n
                    print("    NEW GLOBAL BEST: %.8f" % global_best)

        # Phase C: Perturbation restarts
        remaining_c = max(0.0, deadline - time.monotonic() - 30)
        if remaining_c > 20 and best_nesterov_a is not None:
            n_perturb = max(2, int(remaining_c / 40))
            perturb_budget = remaining_c * 0.7 / n_perturb
            print("  Perturbations (n=%d, budget=%.0fs each):" % (n_perturb, perturb_budget))

            for p in range(n_perturb):
                if time.monotonic() > deadline - 30:
                    break
                noise_scale = 0.02 / (1.0 + 0.5 * p)
                a_pert = best_nesterov_a + rng.normal(0, noise_scale * S / np.sqrt(n), n)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = safe_normalize(a_pert, n)

                # Quick Nesterov then L-BFGS
                half_budget = perturb_budget / 2
                a_mid, _ = run_nesterov_adaptive(
                    a_pert, n, S, half_budget, rng, top_k_init=1, top_k_max=128
                )
                a_new, score = run_lbfgs_aggressive(
                    a_mid, n, S, half_budget, rng,
                    beta_schedule=[50, 100, 500, 2000, 8000],
                    maxiter_per_beta=200
                )
                a_new = safe_normalize(a_new, n)
                c = fft_convolve_self(a_new)
                score = c.max()
                if score < best_nesterov_score:
                    best_nesterov_score = score
                    best_nesterov_a = a_new.copy()
                if score < global_best:
                    global_best = score
                    global_a = a_new.copy()
                    global_n = n
                    print("    Perturb %d: NEW GLOBAL BEST %.8f" % (p, global_best))

        print("  Best for n=%d: %.8f" % (n, best_nesterov_score))

    # --- Final polishing phase ---
    if global_a is not None:
        remaining = max(0.0, deadline - time.monotonic() - 5)
        if remaining > 20:
            print("\n=== Final polishing (budget=%.0fs) ===" % remaining)
            n_pol = global_n
            S_pol = np.sqrt(2.0 * float(n_pol))

            # High-beta L-BFGS
            half = remaining * 0.6
            a_pol, score = run_lbfgs_aggressive(
                global_a, n_pol, S_pol, half, rng,
                beta_schedule=[1000, 3000, 8000, 16000, 32000, 64000],
                maxiter_per_beta=400
            )
            a_pol = safe_normalize(a_pol, n_pol)
            c = fft_convolve_self(a_pol)
            if c.max() < global_best:
                global_best = c.max()
                global_a = a_pol.copy()
                print("  Polish L-BFGS: %.8f" % global_best)

            # Nesterov final
            remaining2 = max(0.0, deadline - time.monotonic() - 5)
            if remaining2 > 10:
                a_pol2, score2 = run_nesterov_adaptive(
                    global_a, n_pol, S_pol, remaining2, rng,
                    top_k_init=1, top_k_max=1024
                )
                a_pol2 = safe_normalize(a_pol2, n_pol)
                c2 = fft_convolve_self(a_pol2)
                if c2.max() < global_best:
                    global_best = c2.max()
                    global_a = a_pol2.copy()
                    print("  Polish Nesterov: %.8f" % global_best)

    # --- Verification ---
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    if sum(final_list) < 0.01:
        final_list = [1.0] * 256

    ver = evaluate_sequence(final_list)
    print("\nVerification score: %s" % ("%.6f" % ver if np.isfinite(ver) else "inf"))
    print("Sequence length: %d" % len(final_list))
    print("Time used: %.1fs" % (time.monotonic() - t_start))

    return final_list
