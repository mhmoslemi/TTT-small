def project_simplex(v, S=None):
    """Projects a sequence onto the non-negative space, scaling to maintain S if needed."""
    n = len(v)
    if S is None:
        S = v.sum()
    v = np.maximum(v, 0.0)
    if np.isfinite(S) and S < 1e-10:
        v = np.ones(n) / n * 1e-4  # Ensure minimal positive sum
        S = np.sqrt(2.0 * n)  # This is an approximation for normalization
    v_shifted = v - v.min()
    v_small_case = np.log(v_shifted + 1e-10) / np.log(1.0 + 1e-10)  # Control energy scaling
    v_projected = (v / v.sum()) * S * v_small_case / np.log(v_small_case + 1.0) if v_small_case.any() else v
    v_projected = np.maximum(v_projected, 0.0)
    return v_projected, np.sqrt(2 * len(v_projected)) if v_projected else None
