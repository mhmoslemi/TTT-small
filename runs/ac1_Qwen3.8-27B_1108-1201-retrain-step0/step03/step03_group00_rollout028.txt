"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Hybrid frequency-domain + time-domain FISTA with LSE annealing.
Phase 1: Frequency-domain optimization (minimize max |DFT|^2) for fast shape convergence.
Phase 2: Time-domain polishing with exact linear convolution LSE.
Phase 3: Direct subgradient refinement targeting the exact maximum lag.
Multi-length exploration with diverse initializations.

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def freq_lse_gradient(a, beta):
    """Gradient of LSE approximation of max |DFT(a)|^2."""
    n = len(a)
    A = fft(a)
    pwr = np.abs(A) ** 2
    pwr_max = pwr.max()
    e = np.exp(beta * (pwr - pwr_max))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), pwr_max
    p = e / tot
    grad = (2.0 / n) * np.real(fft(p * np.conj(A)))
    return grad, pwr_max


def time_lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max convolution."""
    n = len(a)
    c_max = c.max()
    e = np.exp(beta * (c - c_max))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n), c_max
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad, c_max


def exact_subgradient_time(a, c):
    """Exact subgradient at the argmax lag of the linear convolution."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def run_freq_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_rate=0.003, beta_max=200.0):
    """FISTA using frequency-domain LSE surrogate."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    it = 0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        grad, pwr_max = freq_lse_gradient(y, beta)
        cur = pwr_max
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Step size: gradient is (2/n)*Re(fft(p*conj(A)))
        # Lipschitz constant ~ 2*sum(a)/n = 2*S/n (rough)
        step = 1.0 / max(2.0 * beta * S / n, 1e-10)
        step = min(step, 10.0 / max(gnorm, 1e-15))
        
        # Adaptive: reduce step if stuck
        if it - last_improve > 5000:
            step *= 0.5
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it
        
        x_new = project_simplex(y - step * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Periodic restart
        if it % 5000 == 0:
            decay = np.exp(-it / 30000.0)
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best


def run_time_fista(a_init, n, S, max_time, rng, beta_init=5.0, beta_rate=0.004, beta_max=500.0, mode="lse"):
    """FISTA using time-domain LSE on exact linear convolution."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    it = 0

    best = np.inf
    best_a = a.copy()

def run_local_refinement(a_init, n, S, max_time, rng):
    """Local refinement: identify peaks and apply targeted reductions."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()
        
        # Find top k lags
        k = min(16, len(c) // 4)
        top_idx = np.argsort(c)[::-1][:k]
        top_vals = c[top_idx]
        
        # Weighted subgradient
        w = np.zeros(len(c))
        weights = np.exp(5.0 * (top_vals - top_vals[0]))
        w[top_idx] = weights
        w /= w.sum()
        
        # Compute gradient from weighted lags
        fsize = 1 << (len(w) + n - 1).bit_length()
        W = fft(w, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        step = 0.5 * S / (gnorm * (1.0 + 0.01 * np.log1p(it)))
        
        # Add small random perturbation for diversity
        noise = rng.normal(0, 0.0005 * S / np.sqrt(n), n)
        a_new = project_simplex(a - step * grad + noise, S)
        
        # Accept if better or with small probability (simulated annealing)
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur or rng.random() < np.exp(-0.1 * (c_new.max() - cur) / max(cur * 1e-6, 1e-10)):
            a = a_new
    
    return best_a, best


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []

    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-4.0 * (tt - 0.5) ** 2 / (0.5 * 0.5)))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(1.0 + 0.4 * np.sin(2 * np.pi * 3 * tt + 0.5) + 0.15 * np.cos(2 * np.pi * 7 * tt))
    inits.append(rng.random(n) + 0.1)
    inits.append(rng.random(n) ** 2 + 0.05)
    
    # Block structures
    for nblock in [3, 5, 7]:
        block_len = max(1, n // nblock)

        inits.append(block_seq[:n])
    
    # Triangle
    inits.append(1.0 - 2.0 * np.abs(tt - 0.5))
    
    # Broadband (many sinusoids)
    inits.append(1.0 + 0.2 * np.sin(2 * np.pi * 1 * tt) + 0.15 * np.sin(2 * np.pi * 3 * tt + 0.7) 
                  + 0.1 * np.sin(2 * np.pi * 5 * tt + 1.3) + 0.08 * np.sin(2 * np.pi * 7 * tt + 2.1))
    
    return [np.maximum(x, 0.01) for x in inits]

Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Adaptive-beta FISTA with frequency-domain warm start,
multiple lengths, and structured local refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_gradient(a, c, beta, n):
    """Compute LSE gradient for the time-domain convolution max."""
    c_max = c.max()
    e = np.exp(beta * (c - c_max))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]
