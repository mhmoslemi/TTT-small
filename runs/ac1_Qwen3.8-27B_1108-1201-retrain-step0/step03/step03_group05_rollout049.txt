def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_a(a, w):
    """Compute grad_a[j] = 2 * sum_k w[k] * a[j-k] (cross-correlation via FFT)."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z (unconstrained softmax params)."""
    n = len(z)

def build_toeplitz_band(a, top_indices):
    """Build sparse constraint matrix for (a*g)[k] <= M for k in top_indices."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in top_indices:
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start > j_end:

        rows.extend(np.full(len(js), k).tolist())
        cols.extend(js.tolist())
        vals.extend(a[idxs].tolist())
    
    A_sparse = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A_sparse


def solve_lp_direction(a, S, top_k=500, method='highs'):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for top-k lags, g >= 0.
    Returns normalized g (sum = S).
    """
    n = len(a)
    c = fft_convolve_self(a)
    M = c.max()
    
    # Top-k active lags
    top_indices = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix
    A_sparse = build_toeplitz_band(a, top_indices)
    
    # LP: max 1^T g s.t. A_sparse @ g <= M * ones, g >= 0
    # In standard form: min -1^T g s.t. A_sparse @ g <= M * ones, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    
    # Need to extract only the rows corresponding to top_indices
    # Actually, A_sparse already has all 2n-1 rows, so we need to select
    A_sel = A_sparse[top_indices, :]

        if result.success:

            s = g.sum()
            if s > 1e-15:

                return g, M
    except Exception as e:

    return a.copy(), M

Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Algorithm:
- Diverse initializations (spectral, structured, random, given)
- LP-based direction finding with top-k active constraints for main optimization
- L-BFGS with log-sum-exp continuation for smoothing-based polishing
- Nesterov top-k subgradient for final refinement
- FFT-accelerated gradient computation O(n log n) per evaluation
- Cross-pollination and adaptive perturbation

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if n == 1:
        return np.maximum(v, 0.0) * (S / max(v[0], 1e-30))
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_cross_corr(a, w):
    """Compute cross-correlation: result[j] = sum_k w[k] * a[k-j].
    This is the gradient of (a*w) peak w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    # cross-correlation: ifft(conj(W) * A) gives sum_k w[k] * a[k+j]... 
    # Actually let me think about this more carefully.
    # We want grad_a[j] = 2 * sum_k w[k] * a[k-j] (for the peak at lag k)
    # This is 2 * (w cross-correlate a) [j] where (w x a)[j] = sum_k w[k] a[k-j]
    # In FFT terms: ifft(conj(fft(w)) * fft(a)) gives the cross-correlation
    # But we need to be careful with the definition.
    # (w x a)[j] = sum_k w[k] * a[k-j]
    # = sum_k w[k] * a[k-j]
    # Let's define b[m] = w[-m] (with appropriate wrapping). Then (w x a) = b * a (convolution).
    # In FFT: ifft(fft(b) * fft(a)) = ifft(conj(fft(w)) * fft(a))
    # But this gives circular cross-correlation, not linear.
    # With zero-padding, it gives linear cross-correlation.
    corr = np.real(ifft(np.conj(W) * A))[:n]
    return 2.0 * corr


def softmax_obj_grad(z, beta, S):
    """Compute smoothed objective and gradient w.r.t. z (unconstrained softmax params)."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp
    
    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta
    
    w = exp_bc / sum_exp_bc
    grad_a = compute_cross_corr(a, w)
    
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)
    
    return obj, grad_z


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def build_conv_matrix(a, top_indices):
    """
    Build sparse matrix A (len(top_indices) x n) such that
    A @ g = [(a*g)[k] for k in top_indices].
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    
    for ri, k in enumerate(top_indices):
        k = int(k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        i_arr = np.arange(i_start, i_end + 1)
        a_idx = k - i_arr
        rows.extend([ri] * len(i_arr))
        cols.extend(i_arr.tolist())
        vals.extend(a[a_idx].tolist())
    
    A = sp.csc_matrix((vals, (rows, cols)), shape=(len(top_indices), n))
    return A


def solve_lp_direction(a, S, top_k=500):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in top_k active lags, g >= 0.
    Returns (g_normalized, M).
    """
    n = len(a)
    c = fft_convolve_self(a)
    M = c.max()
    
    if M < 1e-15:
        return a.copy(), M
    
    # Top-k active lags
    top_indices = np.argsort(c)[::-1][:top_k]
    
    # Build constraint matrix
    A_mat = build_conv_matrix(a, top_indices)
    
    # LP: min -1^T g s.t. A_mat @ g <= M * 1, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    
    try:
        result = linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs')
        if result.success and result.x is not None:
            g = result.x
            s = g.sum()
            if s > 1e-15:
                g = g * (S / s)
                g = np.clip(g, 0.0, 1000.0)
                return g, M
    except Exception as e:
        print("    LP error: %s" % str(e))
    
    return None, M
