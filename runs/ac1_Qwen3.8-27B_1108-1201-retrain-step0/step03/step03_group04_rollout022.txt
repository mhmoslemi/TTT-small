    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v

    w = np.sort(v)[::-1]

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Multi-phase approach: subgradient peak reduction, log-sum-exp annealing
with adaptive beta, L-BFGS polishing, multi-n exploration, and local refinement.
Key innovations: backtracking line search, adaptive smoothing schedule,
frequency-domain gradient computation, and diverse restart strategies.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def softmax_weights(c, beta):
    """Compute softmax weights on convolution values."""
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def toppk_weights(c, top_k):
    """Compute normalized top-k weights."""
    top_k = min(top_k, len(c))
    top_indices = np.argpartition(c, -top_k)[-top_k:]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.full(len(c), 1.0 / len(c))
    weights = np.zeros(len(c))
    weights[top_indices] = top_vals / total
    return weights


def compute_grad_from_weights(a, w):
    """Compute gradient of sum_m w[m]*c[m] where c = convolve(a,a).
    grad = 2 * IFFT(conj(FFT(w)) * FFT(a))[:n]
    """
    n = len(a)
    N = len(w) + n - 1
    fsize = 1 << (N - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(np.conj(W) * A))[:n]
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * float(n))
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def compute_objective(c_max, n, S):
    """Compute the full objective value."""
    return 2.0 * n * c_max / (S * S)


def run_subgradient_peak(a_init, n, S, max_time, rng, top_k_init=1, top_k_max=50):
    """Subgradient method focusing on top peaks with backtracking."""
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    alpha = S / (n * 2.0)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_local = a.copy()

        top_k = min(top_k_max, top_k_init + it // 200)
        w = toppk_weights(c, top_k)
        grad = compute_grad_from_weights(a, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        a_trial = project_simplex(a - alpha * grad / gnorm, S)
        c_trial = fft_convolve_self(a_trial)
        if c_trial.max() >= cur_max:
            alpha *= 0.7
            a_trial = project_simplex(a - alpha * grad / gnorm, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() >= cur_max:
                alpha *= 0.7
        else:
            alpha = min(alpha * 1.05, S / n)
            a = a_trial

        # Periodic perturbation to escape local minima
        if it % 1500 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_local, best_score


def run_logsumexp_fista(a_init, n, S, max_time, rng, beta_start=20.0, beta_end=500.0):
    """FISTA with log-sum-exp smoothed objective and adaptive beta."""
    a = safe_normalize(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 1.0 / (4.0 * beta_start)

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: geometric increase
        progress = min(1.0, (time.monotonic() - start) / max_time)
        beta = beta_start * (beta_end / beta_start) ** progress

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_local = y.copy()

        w = softmax_weights(c, beta)
        grad = compute_grad_from_weights(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Reset momentum periodically
        if it % 2000 == 0:
            t = 1.0
            y = a.copy()

        # Perturbation for diversity
        if it % 5000 == 0 and rng.random() < 0.15:
            noise = rng.normal(0, 0.002 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0

    return best_local, best_score


def run_ldfgs_refine(a_init, n, S, max_time, rng):
    """Use scipy L-BFGS-B to refine the sequence on the hard max objective."""
    from scipy.optimize import minimize

    a = safe_normalize(a_init, n)
    S_val = S

    def objective(x):
        x_nonneg = np.maximum(x, 0.0)
        # Rescale to maintain sum constraint approximately
        s = x_nonneg.sum()
        if s < 1e-10:
            return 1e6
        x_norm = x_nonneg * (S_val / s)
        c = fft_convolve_self(x_norm)
        return c.max()

    def gradient(x):
        x_nonneg = np.maximum(x, 0.0)
        s = x_nonneg.sum()
        if s < 1e-10:
            return np.zeros(n)
        x_norm = x_nonneg * (S_val / s)
        c = fft_convolve_self(x_norm)
        # Weight on top peaks
        top_k = min(10, len(c))
        w = toppk_weights(c, top_k)
        grad_c = compute_grad_from_weights(x_norm, w)
        # Chain rule for normalization
        grad = (grad_c * S_val / s) - (grad_c * (x_nonneg / s))
        # Zero out where x was negative (active constraint)
        grad[x < 0] = 0
        return grad

    bounds = [(0.0, 1000.0)] * n
    x0 = a.copy()

    best_x = x0.copy()
    best_val = objective(x0)
    start = time.monotonic()

    # Multiple L-BFGS runs with different maxiter
    while time.monotonic() - start < max_time:
        t0 = time.monotonic()
        try:
            res = minimize(objective, x0, method='L-BFGS-B',
                           jac=gradient, bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-10})
            if res.fun < best_val:
                best_val = res.fun
                best_x = res.x.copy()
                x0 = res.x.copy()
        except Exception:
            pass

        if time.monotonic() - t0 > 0.5:
            # L-BFGS is too slow, break
            break

        # Small perturbation for next run
        if time.monotonic() - start < max_time * 0.8:
            x0 = project_simplex(x0 + rng.normal(0, 0.001 * S_val / np.sqrt(n), n), S_val)

    return safe_normalize(best_x, n), best_val


def run_mirror_descent(a_init, n, S, max_time, rng, eta_init=0.1):
    """Mirror descent (multiplicative weights) for diversity."""
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = a.copy()

        eta = eta_init / (1.0 + 0.01 * it)
        top_k = min(5, len(c))
        w = toppk_weights(c, top_k)
        grad = compute_grad_from_weights(a, w)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break

        a_new = a * np.exp(-eta * grad / gmax)
        a_new = np.maximum(a_new, 1e-20)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.ones(n) * S / n
        else:
            a = a_new * (S / s_new)

    return best_local, best_score


def run_oscillation_expansion(a_init, n, S, max_time, rng):
    """Try adding structured oscillatory components to the sequence."""
    a = safe_normalize(a_init, n)
    best_local = a.copy()
    best_score = fft_convolve_self(a).max()
    start = time.monotonic()

    for freq in range(1, 20):
        if time.monotonic() - start > max_time:

        for amp in [0.001, 0.005, 0.01, 0.02]:

    return best_local, best_score
