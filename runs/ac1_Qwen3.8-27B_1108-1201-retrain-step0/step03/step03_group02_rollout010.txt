"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Approach: Hybrid of weighted subgradient descent, coordinate descent, 
and population-based exploration.
"""
import numpy as np
import time
from scipy.fft import fft, ifft

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def compute_objective(a):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)

def weighted_subgradient_step(a, c, n, S, top_k, beta, step_size):
    """
    Compute and apply a weighted subgradient step.
    Weights are based on exponential of (c_k - max_c) * beta for top peaks.
    """
    # Find top peaks
    order = np.argsort(c)[::-1][:top_k]
    peak_vals = c[order]
    
    # Exponential weights
    w = np.exp(beta * (peak_vals - peak_vals.max()))
    w = w / w.sum()
    
    # Build weight vector over all lags
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(order):
        w_full[int(k)] = w[i]
    
    # Gradient: grad[m] = 2 * sum_k w_full[k] * a[k-m] (cross-correlation)
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    
    # Apply step
    a_new = a - step_size * grad
    a_new = project_simplex(a_new, S)
    return a_new, grad

def coordinate_descent_step(a, coord_idx, c, n):
    """
    For a single coordinate, find the best value via 1D search.
    We use the fact that changing a[i] only affects c_k through a_i * a[k-i].
    """
    # For coordinate i, the contribution to c_k is a[i] * a[k-i] (for valid k-i)
    # Plus a[i]^2 if k = 2*i
    i = coord_idx
    a_i = a[i]
    
    # The "base" convolution (without a[i])
    # c[k] = base[k] + a[i]*a[k-i] + a[i]^2*[k==2*i]
    # To find optimal a[i], we minimize max_k c[k] over a[i] >= 0
    
    # For a given t (= a[i]), c[k] = base[k] + t * a[k-i] + t^2 * [k==2*i]
    # where base[k] = c[k] - a_i * a[k-i] - a_i^2 * [k==2*i]
    
    # Compute base
    base = c.copy()
    for k in range(2 * n - 1):
        j = k - i
        if 0 <= j < n:
            base[k] -= a_i * a[j]
    base[2 * i] -= a_i * a_i  # Remove the quadratic term
    
    # Now we want to minimize max_k (base[k] + t * a[k-i] + t^2 * [k==2*i]) over t >= 0
    # This is a 1D convex problem (max of convex functions + t^2 term)
    # Use golden section search
    
    # The optimal t is bounded. Let's say t in [0, 2 * a_i] (or [0, 1000])
    t_lo = 0.0
    t_hi = max(2.0 * a_i, 1.0)
    
    def f1d(t):
        # Compute max_k (base[k] + t * a[k-i] + t^2 * [k==2*i])
        contrib = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            j = k - i
            if 0 <= j < n:
                contrib[k] = base[k] + t * a[j]
        if 2 * i < 2 * n - 1:
            contrib[2 * i] += t * t
        return contrib.max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a_gs = t_hi - gr * (t_hi - t_lo)
    b_gs = t_hi - (1 - gr) * (t_hi - t_lo)
    fa = f1d(a_gs)
    fb = f1d(b_gs)
    
    for _ in range(50):
        if fa < fb:
            t_hi = b_gs
            b_gs = a_gs
            fb = fa
            a_gs = t_hi - gr * (t_hi - t_lo)
            fa = f1d(a_gs)
        else:
            t_lo = a_gs
            a_gs = b_gs
            fa = fb
            b_gs = t_hi - (1 - gr) * (t_hi - t_lo)
            fb = f1d(b_gs)
    
    t_opt = (t_lo + t_hi) / 2
    
    # Update
    a[i] = max(0.0, t_opt)
    
    # Recompute c
    c_new = fft_convolve_self(a)
    return a, c_new

def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    """Compute objective from convolution c, length n, sum S."""
    if S < 1e-15:

    return 2.0 * n * c.max() / (S * S)
