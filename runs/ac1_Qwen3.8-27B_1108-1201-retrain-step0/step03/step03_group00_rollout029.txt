"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-phase optimization combining FISTA with adaptive LSE smoothing,
subgradient averaging, and coordinate descent polishing. Explores multiple
sequence lengths and diverse initializations starting from the given sequence.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_value_and_grad(a, c, beta):
    """Log-sum-exp value and gradient of the LSE approximation to max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return cm, np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    lse_val = cm + np.log(tot) / beta
    return lse_val, grad


def subgradient_of_max(a, c):
    """Exact subgradient of max(c) w.r.t. a."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def run_fista(a_init, n, S, max_time, rng,
              beta_init=3.0, beta_final=800.0, n_iters=None):
    """FISTA with adaptive LSE smoothing and periodic restarts."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    stall_count = 0

    # Precompute FFT size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    gsize = 1 << (size + n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        # Adaptive beta: logarithmic ramp
        frac = min(1.0, it / max(100, n_iters if n_iters else 5000))
        beta = beta_init * (beta_final / beta_init) ** frac
        beta = min(beta, beta_final)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
            stall_count = 0
        else:
            stall_count += 1

        # Compute LSE gradient
        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            w = np.zeros(len(c))
            w[cm.argmax()] = 1.0
        else:
            w = e / tot

        W = fft(w, gsize)
        A = fft(y, gsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Adaptive step size
        step = 1.0 / (2.0 * beta + gnorm * 0.1)
        step = min(step, S / (gnorm + 1e-10))

        x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Stall detection and restart
        if stall_count > 3000:
            # Inject noise and restart momentum
            noise_scale = max(0.001, 0.01 * np.exp(-stall_count / 5000.0)) * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it
            stall_count = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best, it


def run_subgradient_average(a_init, n, S, max_time, rng):
    """Subgradient method with iterate averaging."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    # Averaged point
    a_avg = np.zeros(n)
    a = a.copy()
    y = a.copy()
    t = 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()

        # Subgradient
        grad = subgradient_of_max(y, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size: O(1/sqrt(t))
        step = 2.0 * S / (np.sqrt(it) * gnorm)
        step = min(step, 1.0)

        x_new = project_simplex(y - step * grad, S)

        # Fista momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic averaging
        if it % 500 == 0:
            a_avg = normalize_sum(a_avg + x_new, n)
            c_avg = fft_convolve_self(a_avg)
            if c_avg.max() < best:
                best = c_avg.max()
                best_a = a_avg.copy()

        # Occasional noise
        if it % 5000 == 0 and rng.random() < 0.3:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0

    return best_a, best, it


def coordinate_descent_polish(a_init, n, S, max_time, rng):
    """Coordinate descent: reduce elements that contribute most to the peak."""
    a = normalize_sum(a_init, n).copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    c = fft_convolve_self(a)

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()

        k_star = int(np.argmax(c))
        S_target = np.sqrt(2.0 * n)

        # Elements contributing to c[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        contributing = np.arange(lo, hi + 1)

        if len(contributing) == 0:
            break

        # For each contributing element, try to reduce it
        improved = False
        for j in contributing:
            # Current contribution to c[k_star]: a[j] * a[k_star - j]
            partner = a[k_star - j]
            if partner < 1e-15:
                continue
            # Try reducing a[j]
            # c_new[k_star] = c[k_star] - 2*delta*a[k_star-j] + delta^2*(k_star==2j)
            # We want c_new[k_star] to be as low as possible without
            # making other lags exceed the current max
            # Simple: reduce by a fraction
            frac = 0.01
            delta = -frac * a[j]
            a_test = a.copy()
            a_test[j] = max(0.0, a[j] + delta)
            a_test = normalize_sum(a_test, n)
            c_test = fft_convolve_self(a_test)
            if c_test.max() < cur:
                a = a_test
                cur = c_test.max()
                improved = True

        if not improved:
            # Try more aggressive: reduce the largest contributor
            partner_vals = a[k_star - contributing]
            idx_max = np.argmax(partner_vals * a[contributing])
            j = contributing[idx_max]
            # Binary search for max reduction
            lo_d = -a[j]
            hi_d = 0.0
            for _ in range(20):
                mid = (lo_d + hi_d) / 2
                a_test = a.copy()
                a_test[j] = max(0.0, a[j] + mid)
                a_test = normalize_sum(a_test, n)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < cur:
                    lo_d = mid
                    a = a_test
                    cur = c_test.max()
                    improved = True
                else:
                    hi_d = mid
            if not improved:
                break

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def multi_scale_fista(a_init, n, S, max_time, rng):
    """FISTA with multi-scale: solve coarsely then refine."""
    a = normalize_sum(a_init, n)

    # Coarse solve
    n_coarse = max(128, n // 8)
    nblocks = n // n_coarse
    a_coarse = a[:n_coarse * nblocks].reshape(n_coarse, nblocks).mean(axis=1)
    S_coarse = np.sqrt(2.0 * n_coarse)
    a_coarse = a_coarse * (S_coarse / a_coarse.sum())

    t_coarse = max_time * 0.2
    a_c_opt, _, _ = run_fista(a_coarse, n_coarse, S_coarse, t_coarse, rng,
                               beta_init=2.0, beta_final=200.0)

    # Upsample
    t_old = np.linspace(0, 1, n_coarse)
    t_new = np.linspace(0, 1, n)
    a_up = np.interp(t_new, t_old, a_c_opt)
    a_up = normalize_sum(a_up, n)

    # Fine solve
    t_fine = max(1.0, max_time - (time.monotonic() - t_coarse) - 1.0)
    return run_fista(a_up, n, S, t_fine, rng,
                     beta_init=5.0, beta_final=600.0)

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)

    """Exact subgradient of max(c) w.r.t. a."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def run_fista(a_init, n, S, max_time, rng,
              beta_init=3.0, beta_final=800.0):
    """FISTA with adaptive LSE smoothing and periodic restarts."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    stall_count = 0

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    gsize = 1 << (size + n - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / 5000.0)
        beta = beta_init * (beta_final / beta_init) ** frac
        beta = min(beta, beta_final)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        cm = c.max()
        e = np.exp(beta * (c - cm))
        tot = e.sum()
        if tot < 1e-300:
            w = np.zeros(len(c))
            w[int(cm.argmax())] = 1.0
        else:
            w = e / tot

        W = fft(w, gsize)
        A = fft(y, gsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / (2.0 * beta + 1e-10)
        step = min(step, 2.0 * S / (gnorm + 1e-10))

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if stall_count > 2000:
            noise_scale = max(0.0005, 0.005 * np.exp(-stall_count / 3000.0)) * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            stall_count = 0

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best, it


def run_subgradient_average(a_init, n, S, max_time, rng):
    """Subgradient method with FISTA momentum and iterate averaging."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    a_avg = a.copy()
    avg_count = 1
    stall_count = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            stall_count = 0
        else:
            stall_count += 1

        grad = subgradient_of_max(y, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (gnorm * (1.0 + 0.05 * np.log1p(it / 100.0)))
        step = min(step, 0.5)

        x_new = project_simplex(y - step * grad, S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 100 == 0:
            a_avg = (a_avg * avg_count + x_new) / (avg_count + 1)
            avg_count += 1
            a_avg = normalize_sum(a_avg, n)
            c_avg = fft_convolve_self(a_avg)
            if c_avg.max() < best:
                best = c_avg.max()
                best_a = a_avg.copy()
                stall_count = 0

        if stall_count > 5000:
            noise_scale = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            stall_count = 0

    return best_a, best, it


def coordinate_descent_polish(a_init, n, S, max_time, rng):
    """Coordinate descent: reduce elements contributing to the peak lag."""
    a = normalize_sum(a_init, n).copy()
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()

    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = a.copy()

        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        contributing = np.arange(lo, hi + 1)

        if len(contributing) == 0:
            break

        improved = False
        contrib_products = a[contributing] * a[k_star - contributing]
        order = np.argsort(contrib_products)[::-1]

        for idx in order[:min(8, len(order))]:
            j = contributing[idx]
            lo_d = -a[j] * 0.5
            hi_d = 0.0
            found = False
            for _bs in range(22):
                mid = (lo_d + hi_d) / 2.0
                a_test = a.copy()
                a_test[j] = max(0.0, a[j] + mid)
                a_test = normalize_sum(a_test, n)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < cur - 1e-10:
                    lo_d = mid
                    found = True
                else:
                    hi_d = mid
            if found:
                a[j] = max(0.0, a[j] + lo_d)
                a = normalize_sum(a, n)
                cur = fft_convolve_self(a).max()
                improved = True

        if not improved:
            j = rng.integers(n)
            lo_d = -a[j] * 0.3
            hi_d = 0.0
            found = False
            for _bs in range(18):
                mid = (lo_d + hi_d) / 2.0
                a_test = a.copy()
                a_test[j] = max(0.0, a[j] + mid)
                a_test = normalize_sum(a_test, n)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < cur - 1e-10:
                    lo_d = mid
                    found = True
                else:
                    hi_d = mid
            if found:
                a[j] = max(0.0, a[j] + lo_d)
                a = normalize_sum(a, n)
                improved = True
            else:
                noise_scale = 0.001 * S / np.sqrt(n)
                a_test = project_simplex(a + rng.normal(0, noise_scale, n), S)
                c_test = fft_convolve_self(a_test)
                if c_test.max() < cur:
                    a = a_test
                    improved = True
                else:
                    break

    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    return best_a, best


def generate_start(n, rng):
    """Generate a diverse random start."""
    tt = np.linspace(0, 1, n, endpoint=False)
    choice = rng.integers(8)
    if choice == 0:
        a = np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * tt), 0.01)
    elif choice == 1:
        a = np.exp(-4.0 * (tt - 0.5) ** 2)
    elif choice == 2:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * tt + rng.random() * 2 * np.pi)
    elif choice == 3:
        a = 1.0 + 0.2 * np.sin(2 * np.pi * 7 * tt) + 0.1 * np.cos(2 * np.pi * 13 * tt)
    elif choice == 4:
        a = rng.random(n) + 0.05
    elif choice == 5:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 10) * tt)
    elif choice == 6:
        a = np.maximum(1.0 - 0.5 * np.abs(2 * tt - 1), 0.01)
    else:
        a = 1.0 + 0.4 * np.cos(2 * np.pi * rng.uniform(0.5, 5) * tt)
    return np.maximum(a, 0.01)
