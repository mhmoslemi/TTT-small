"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive softmax mirror descent
with spectral flattening and active-set refinement.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_value(a, c=None):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    if c is None:
        c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def softmax_weights(c, T):
    """Compute softmax weights over convolution values."""
    c_shifted = c - c.max()
    w = np.exp(T * c_shifted)
    w /= w.sum()
    return w


def compute_grad_softmax(a, c, T):
    """Compute gradient using softmax weights."""
    n = len(a)
    w = softmax_weights(c, T)
    # grad[m] = 2 * sum_k w[k] * a[k-m] for valid k-m
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1 - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def spectral_flatten(a, alpha=0.5):
    """Flatten the power spectrum of a using amplitude adjustment."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    # |A|^2 is the power spectrum
    # We want to make |A|^2 more uniform
    # Update: A_new = A * (target / |A|^2)^alpha
    # where target = mean(|A|^2)
    power = np.abs(A) ** 2
    target = power.mean()
    # Avoid division by zero
    power_safe = np.maximum(power, 1e-30)
    correction = (target / power_safe) ** alpha
    A_new = A * correction
    a_new = np.real(ifft(A_new, fsize))[:n]
    return a_new

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive softmax mirror descent
with O(n) coordinate descent polishing and spectral flattening.
Key improvements: softmax gradients (vs top-k uniform), fast coordinate
descent with O(n) convolution updates, adaptive temperature schedule.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_from_conv(c, n, S):
    """Compute objective from precomputed convolution."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def compute_grad_softmax(a, c, T):
    """Compute gradient of soft-max objective using softmax weights."""
    n = len(a)
    c_shifted = c - c.max()
    w = np.exp(T * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w /= w_sum
    # grad[m] = 2 * sum_k w[k] * a[k-m] for valid k-m in [0, n-1]
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1 - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def spectral_flatten_step(a, alpha=0.3):
    """Apply spectral flattening to reduce peak-to-average ratio."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    target = power.mean()
    power_safe = np.maximum(power, 1e-30)
    correction = (target / power_safe) ** alpha
    A_new = A * correction
    a_new = np.real(ifft(A_new, fsize))[:n]
    return a_new

def fast_coordinate_polish(a0, time_budget, S=None, n_factors=6):
    """
    Coordinate descent with O(n) convolution update per evaluation.
    Tries to reduce each a[i] or increase it to lower the max convolution.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    if S is None:
        S = a.sum()
    C = fft_convolve_self(a).copy()
    c_len = 2 * n - 1
    best_obj = obj_from_conv(C, n, S)
    best_a = a.copy()
    start = time.monotonic()

    # Precompute prefix and suffix max of C for O(1) outside-window max
    prefix_max = np.maximum.accumulate(C)
    suffix_max = np.maximum.accumulate(C[::-1])[::-1]

    factors = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]
    # Sort by how "safe" they are (smaller changes first)
    
    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        improved = False
        # Process positions (maybe in random order for diversity)
        order = np.random.permutation(n) if sweep % 3 == 0 else np.arange(n)
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            orig = a[i]
            if orig < 1e-12:
                continue

            # Window: k in [i, i+n-1]
            lo_k = i
            hi_k = min(i + n - 1, c_len - 1)
            
            # A = C[lo_k:hi_k+1], v = a[0:hi_k-lo_k+1]
            window_len = hi_k - lo_k + 1
            A = C[lo_k:hi_k + 1]
            v = a[:window_len]
            
            # Outside max
            U = -np.inf
            if lo_k > 0:
                U = max(U, prefix_max[lo_k - 1])
            if hi_k + 1 < c_len:
                U = max(U, suffix_max[hi_k + 1])
            
            # Index of 2i in the window (for the delta^2 correction)
            idx_2i = 2 * i - lo_k  # = i
            has_2i = (0 <= idx_2i < window_len)
            
            for factor in factors:
                delta = orig * (factor - 1.0)
                new_val = orig + delta
                if new_val < -1e-12:
                    continue
                new_S = S + delta
                if new_S < 1e-10:
                    continue
                
                # C_new in window: A + 2*delta*v
                C_win_new = A + 2.0 * delta * v
                # Add delta^2 correction at position idx_2i
                if has_2i:
                    C_win_new[idx_2i] += delta * delta
                
                max_win = C_win_new.max()
                C_new_max = max(max_win, U)
                
                obj_new = 2.0 * n * C_new_max / (new_S * new_S)
                
                if obj_new < best_obj * (1.0 - 1e-10):
                    # Accept this change
                    a[i] = new_val
                    S = new_S
                    C[lo_k:hi_k + 1] = C_win_new
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    # Update prefix/suffix max (expensive, but we can skip for speed)
                    # Actually, let's just recompute them occasionally
                    break
        
        # Recompute prefix/suffix max every few sweeps
        if sweep % 5 == 0:
            prefix_max = np.maximum.accumulate(C)
            suffix_max = np.maximum.accumulate(C[::-1])[::-1]
        
        if not improved and sweep > 3:
            break

    return best_a, best_obj

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive softmax mirror descent
with O(n) coordinate descent polishing and spectral flattening.
Key improvements over prior: softmax gradients (vs top-k uniform), fast
coordinate descent with O(n) convolution updates, adaptive temperature

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def obj_from_conv(c, n, S):
    """Compute objective from precomputed convolution and sum."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * c.max() / (S * S)


def compute_grad_softmax(a, c, T):
    """Compute gradient of soft-max of convolution using softmax weights."""
    n = len(a)
    c_shifted = c - c.max()
    w = np.exp(T * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w /= w_sum
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1 - 1).bit_length()
    W = fft(w, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(W * R))
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def spectral_flatten_step(a, alpha=0.2):
    """Apply spectral flattening to reduce peak-to-average ratio in convolution."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    power = np.abs(A) ** 2
    target = power.mean()
    power_safe = np.maximum(power, 1e-30)
    correction = (target / power_safe) ** alpha
    A_new = A * correction
    a_new = np.real(ifft(A_new, fsize))[:n]
    return a_new

            if hi_k + 1 < c_len:
                U = max(U, suffix_max[hi_k + 1])

            idx_2i = 2 * i - lo_k
            has_2i = (0 <= idx_2i < window_len)

            for factor in factors:
                delta = orig * (factor - 1.0)
                new_val = orig + delta
                if new_val < -1e-12:
                    continue
                new_S = S + delta
                if new_S < 1e-10:
                    continue

                C_win_new = A + 2.0 * delta * v
                if has_2i:
                    C_win_new[idx_2i] += delta * delta

                max_win = C_win_new.max()
                C_new_max = max(max_win, U)

                obj_new = 2.0 * n * C_new_max / (new_S * new_S)

                if obj_new < best_obj * (1.0 - 1e-10):
                    a[i] = new_val
                    S = new_S
                    C[lo_k:hi_k + 1] = C_win_new
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
                    break

        if sweep % 10 == 0:
            prefix_max = np.maximum.accumulate(C)
            suffix_max = np.maximum.accumulate(C[::-1])[::-1]

        if improved:
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
            if consecutive_no_improve >= 5:
                break

        if verbose_cd and sweep % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"    coord sweep={sweep}, obj={best_obj:.7f}, t={elapsed:.1f}s")

    return best_a, best_obj

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Multi-phase: brief exploration -> adaptive softmax mirror descent ->
    fast O(n) coordinate polish -> final fine-tune.
    """
    global verbose_cd
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    global_best = np.inf
    global_a = None

    # Load given sequence
    given_a = None
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given_raw = np.clip(given_raw, 0.0, 1000.0)
        if given_raw.size >= 16 and given_raw.sum() > 0.01:
            given_a = given_raw
            n_given = len(given_a)
            S_given = np.sqrt(2.0 * n_given)
            given_a = given_a * (S_given / given_a.sum())
            ver = evaluate_sequence(given_a.tolist())
            print(f"Given sequence: n={n_given}, obj={ver:.7f}")
            if np.isfinite(ver) and ver < global_best:
                global_best = ver
                global_a = given_a.copy()
    except Exception as e:
        print(f"Could not load given sequence: {e}")

    if global_a is None:
        n = 4096
        S = np.sqrt(2.0 * n)
        global_a = np.ones(n) * (S / n)
        global_best = evaluate_sequence(global_a.tolist())
        print(f"Fallback init: n={n}, obj={global_best:.7f}")

    n = len(global_a)
    S_norm = np.sqrt(2.0 * n)
    global_a = global_a * (S_norm / global_a.sum())

    # Phase 1: Brief exploration (4% of time)
    explore_time = 0.04 * (deadline - time.monotonic())
    print(f"\nPhase 1: Exploration ({explore_time:.0f}s)")

    if explore_time > 3:
        i = np.arange(n)
        inits = []
        # Various structured inits
        a_cos = 1.0 + 0.7 * np.cos(2 * np.pi * 17 * i / n)
        a_cos = np.maximum(a_cos, 0.01)
        a_cos *= S_norm / a_cos.sum()
        inits.append(('cos17', a_cos))

        a_prime = 1.0 + 0.5 * np.cos(2 * np.pi * 3 * i / n) + 0.5 * np.cos(2 * np.pi * 11 * i / n)
        a_prime = np.maximum(a_prime, 0.01)
        a_prime *= S_norm / a_prime.sum()
        inits.append(('dprime', a_prime))

        per_init_time = explore_time / len(inits)
        for name, a_init in inits:
            if time.monotonic() > deadline - 30:
                break
            c_init = fft_convolve_self(a_init)
            obj_init = obj_from_conv(c_init, n, S_norm)
            print(f"  {name}: init obj={obj_init:.7f}")
            if obj_init < global_best:
                global_best = obj_init
                global_a = a_init.copy()
                print(f"    -> new best {obj_init:.7f}")

    # Phase 2: Main mirror descent (56% of time)
    main_time = 0.56 * (deadline - time.monotonic())
    print(f"\nPhase 2: Mirror descent ({main_time:.0f}s), best={global_best:.7f}")

    if main_time > 10:
        # 2a: Broad exploration (high temperature)
        t_2a = main_time * 0.35
        print(f"  2a: Broad T=[100,5] ({t_2a:.0f}s)")
        a_b, obj_b = mirror_descent_improved(
            global_a, t_2a,
            T_start=100.0, T_end=5.0,
            eta0=0.4, momentum=0.85,
            spectral_every=600, spectral_alpha=0.2,
            verbose=True
        )
        if obj_b < global_best:
            global_best = obj_b
            global_a = a_b.copy()
            print(f"    -> {obj_b:.7f}")

        # 2b: Medium focus
        t_2b = main_time * 0.35
        print(f"  2b: Medium T=[5,1] ({t_2b:.0f}s)")
        a_b, obj_b = mirror_descent_improved(
            global_a, t_2b,
            T_start=5.0, T_end=1.0,
            eta0=0.25, momentum=0.88,
            spectral_every=500, spectral_alpha=0.12,
            verbose=True
        )
        if obj_b < global_best:
            global_best = obj_b
            global_a = a_b.copy()
            print(f"    -> {obj_b:.7f}")

        # 2c: Focused (low temperature)
        t_2c = main_time * 0.30
        print(f"  2c: Focused T=[1,0.1] ({t_2c:.0f}s)")
        a_b, obj_b = mirror_descent_improved(
            global_a, t_2c,
            T_start=1.0, T_end=0.1,
            eta0=0.12, momentum=0.93,
            spectral_every=400, spectral_alpha=0.08,
            verbose=True
        )
        if obj_b < global_best:
            global_best = obj_b
            global_a = a_b.copy()
            print(f"    -> {obj_b:.7f}")

    # Phase 3: Fast coordinate polish (30% of time)
    polish_time = 0.30 * (deadline - time.monotonic())
    print(f"\nPhase 3: Coordinate polish ({polish_time:.0f}s), best={global_best:.7f}")

    if polish_time > 5 and global_a is not None:
        verbose_cd = True
        # Multiple rounds: first broad, then finer
        round_time = polish_time / 3.0

        print(f"  Round 1 ({round_time:.0f}s) - broad changes")
        a_b, obj_b = fast_coordinate_polish(
            global_a, round_time, S=global_a.sum()
        )
        if obj_b < global_best:
            global_best = obj_b
            global_a = a_b.copy()
            print(f"    -> {obj_b:.7f}")
        else:
            print(f"    (no improvement)")

        print(f"  Round 2 ({round_time:.0f}s)")
        a_b, obj_b = fast_coordinate_polish(
            global_a, round_time, S=global_a.sum()
        )
        if obj_b < global_best:
            global_best = obj_b
            global_a = a_b.copy()
            print(f"    -> {obj_b:.7f}")

        print(f"  Round 3 ({round_time:.0f}s)")
        a_b, obj_b = fast_coordinate_polish(
            global_a, round_time, S=global_a.sum()
        )
        if obj_b < global_best:
            global_best = obj_b
            global_a = a_b.copy()
            print(f"    -> {obj_b:.7f}")
        verbose_cd = False

    # Phase 4: Final fine mirror descent (remaining time minus buffer)
    final_time = deadline - time.monotonic() - 8.0
    if final_time > 5 and global_a is not None:
        print(f"\nPhase 4: Final fine-tune ({final_time:.0f}s)")
        a_b, obj_b = mirror_descent_improved(
            global_a, final_time,
            T_start=0.5, T_end=0.05,
            eta0=0.04, momentum=0.95,
            spectral_every=1000, spectral_alpha=0.03,
            verbose=True
        )
        if obj_b < global_best:
            global_best = obj_b
            global_a = a_b.copy()
            print(f"    -> {obj_b:.7f}")

    # Final verification
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: local_best={global_best:.7f}, verified={ver:.7f}")

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.7f}")

    return final_a
