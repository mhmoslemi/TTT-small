"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using active-set subgradient descent with
Hessian approximation, projected gradient with backtracking, and coordinate
descent polishing. Starting from the given sequence, we iteratively reduce the
peak autocorrelation by tracking the top-K peaks and using weighted subgradients.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgrad_topk(c, a, K=10):
    """Compute subgradient using top-K peaks with equal weight."""
    n = len(a)
    size = len(c)
    
    # Find top K peaks
    topk_idx = np.argsort(c)[-K:][::-1]
    topk_val = c[topk_idx]
    
    # Use exponential weighting: w_k = exp(c_k / c_max)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    
    # Weight: focus on top peaks but with some smoothness
    w = np.zeros(size)
    for idx in topk_idx:
        w[idx] = 1.0  # Equal weight on top K
    
    # Optionally add a small baseline to all peaks to avoid getting stuck
    w += 0.01 * (c / c_max)
    
    w_sum = w.sum()
    w /= w_sum
    
    # g[m] = 2 * sum_k w[k] * a[k-m] (cross-correlation)
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def project_nonneg_sum(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def active_set_subgradient(a_init, n, S, time_budget, K=20, lr_init=0.1, 
                           momentum=0.9, verbose=True):
    """Active-set subgradient descent with momentum and backtracking."""
    a = project_nonneg_sum(a_init.copy(), S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Momentum
    v = np.zeros(n)
    beta = momentum
    
    start = time.monotonic()
    step = 0
    
    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Compute subgradient using top-K
        g = compute_subgrad_topk(c, a, K)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        g_dir = g / gnorm
        
        # Backtracking line search
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        lr = lr_init * (1.0 - 0.5 * progress)
        
        a_trial = None
        for i in range(15):
            a_cand = a - lr * g_dir
            a_cand = project_nonneg_sum(a_cand, S)
            c_cand = fft_convolve_self(a_cand)
            obj_cand = c_cand.max()
            
            if obj_cand < cur_obj - 1e-10 * max(cur_obj, 1e-10):
                a_trial = a_cand
                break
            lr *= 0.5
        
        if a_trial is None:
            # No improvement found, try a smaller step or break
            break
        
        # Momentum update
        v = beta * v + (1 - beta) * (a_trial - a)
        a = a_trial + beta * v
        a = project_nonneg_sum(a, S)
        
        step += 1
        if verbose and step % 100 == 0:
            elapsed = time.monotonic() - start
            c_now = fft_convolve_self(a)
            print("    AS-SG step %d: obj=%.7f, best=%.7f, elapsed=%.1fs" % (step, c_now.max(), best_obj, elapsed))
    
    return best_a, best_obj

                               lr_init=0.05, verbose=True):
    """Smoothed subgradient with approximate Newton correction."""
    a = project_nonneg_sum(a_init.copy(), S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

            best_obj = cur_obj

        if c_max < 1e-30:
            break
        
        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)

        w /= w_sum
        
        # Subgradient

def coordinate_descent(a_init, n, S, time_budget, verbose=True):
    """Coordinate descent with binary search on each coordinate."""
    a = project_nonneg_sum(a_init.copy(), S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    sweep = 0
    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            orig = a[i]
            if orig < 1e-10:
                continue
            
            best_val = orig
            best_obj_local = best_obj
            
            # Try several multiplicative changes
            factors = [0.7, 0.85, 1.15, 1.3, 0.5, 1.5, 0.9, 1.1]
            for f in factors:
                a[i] = orig * f
                a = project_nonneg_sum(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_obj_local - 1e-12:
                    best_obj_local = obj_check
                    best_val = orig * f
            
            a[i] = best_val
            a = project_nonneg_sum(a, S)
            
            if best_obj_local < best_obj - 1e-12:
                best_obj = best_obj_local
                best_a = a.copy()
                improved = True
        
        if not improved and sweep > 2:
            break
    
    return best_a, best_obj


def random_perturbation_search(a_init, n, S, time_budget, n_trials=100, verbose=True):
    """Try random perturbations and keep improvements."""
    a = project_nonneg_sum(a_init.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(123)
    
    for trial in range(n_trials):
        if time.monotonic() - start > time_budget:
            break
        
        # Random perturbation
        scale = 0.01 * (1 + trial / n_trials)
        a_pert = a + scale * rng.standard_normal(n)
        a_pert = project_nonneg_sum(a_pert, S)

            c_ref = fft_convolve_self(a_ref)

                best_obj = c_ref.max()
                best_a = a_ref.copy()

def try_zero_masking(a, n, S, n_zero=50):

    a = project_nonneg_sum(a.copy(), S)

    best_a = a.copy()
    
    sorted_idx = np.argsort(a)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using a multi-stage optimization:
1. Balancing subgradient descent (equalize top-K autocorrelation peaks)
2. Smoothed subgradient with adaptive temperature
3. Coordinate descent for fine-tuning

Key improvements over previous approach:

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_nonneg_sum(a, S):
    """Project onto {x >= 0, sum(x) = S}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def compute_grad_weighted(w, a):
    """Compute gradient of sum_k w[k]*c[k] where c = conv(a,a)."""
    n = len(a)
    size = len(w)
    fsize = 1 << (size + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))
    return 2.0 * corr[:n]


def balancing_descent(a_init, n, S, time_budget, K_init=10, K_max=50,
                      lr_init=0.15, verbose=True):
    """
    Subgradient descent that balances top-K peaks to equal values.
    Uses mean-removed gradient for efficiency and adaptive K.
    """
    a = project_nonneg_sum(a_init.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    step = 0
    no_improve = 0
    max_no_improve = 300
    rng = np.random.default_rng(777)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)

        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            scale = 0.003 * (1.0 + 0.5 * progress)
            a = best_a + scale * rng.standard_normal(n)
            a = project_nonneg_sum(a, S)
            no_improve = 0
            step += 1
            continue

        # Adaptive K: start small, grow as we converge
        K = min(K_max, K_init + int(20 * progress))
        K = max(3, K)

        # Find top K peaks
        topk_idx = np.argsort(c)[-K:][::-1]
        topk_val = c[topk_idx]

        # Balance: target = mean of top K
        target = topk_val.mean()
        excess = np.maximum(topk_val - target, 0.0)

        if excess.sum() < 1e-12 * max(cur_obj, 1e-10):
            # Peaks are well-balanced, use uniform weight on top K
            w = np.zeros(len(c))
            w[topk_idx] = 1.0 / K
        else:
            w = np.zeros(len(c))
            for idx, ex in zip(topk_idx, excess):
                if ex > 0:
                    w[idx] = ex
            w /= w.sum()

        # Compute gradient
        g = compute_grad_weighted(w, a)

        # Remove mean component (more efficient since projection kills it anyway)
        g_dir = g - g.mean()
        gnorm = np.linalg.norm(g_dir)
        if gnorm < 1e-15:
            break

        g_dir /= gnorm

        # Backtracking line search with Armijo condition
        progress_lr = 1.0 - 0.6 * progress
        lr = lr_init * progress_lr
        slope = -np.dot(g_dir, g)  # directional derivative (should be negative)

        a_trial = None
        for i in range(18):
            a_cand = a - lr * g_dir
            a_cand = project_nonneg_sum(a_cand, S)
            c_cand = fft_convolve_self(a_cand)
            obj_cand = c_cand.max()

            if obj_cand <= cur_obj - 1e-4 * lr * abs(slope) + 1e-12:
                a_trial = a_cand
                break
            lr *= 0.5

        if a_trial is None:
            # Even smallest step doesn't help - try a tiny nudge
            lr = 1e-5
            a_cand = a - lr * g_dir
            a_cand = project_nonneg_sum(a_cand, S)
            c_cand = fft_convolve_self(a_cand)
            if c_cand.max() < cur_obj:
                a_trial = a_cand
            else:
                break

        a = a_trial
        step += 1

        if verbose and step % 500 == 0:
            c_now = fft_convolve_self(a)
            print("    BD step %d: obj=%.7f, best=%.7f, K=%d, elapsed=%.0fs" %
                  (step, c_now.max(), best_obj, K, elapsed))

    return best_a, best_obj


def smooth_descent(a_init, n, S, time_budget, T_init=50, T_final=3000,
                   lr_init=0.06, verbose=True):
    """Smoothed subgradient with log-sum-exp approximation and momentum."""
    a = project_nonneg_sum(a_init.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    step = 0
    no_improve = 0
    momentum_buf = np.zeros(n)
    beta_m = 0.85

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.7)

        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400:
            break

        # Temperature-weighted subgradient
        c_max = c.max()
        if c_max < 1e-30:
            break

        scaled = T * c / c_max
        scaled = scaled - scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        g = compute_grad_weighted(w, a)
        g_dir = g - g.mean()
        gnorm = np.linalg.norm(g_dir)
        if gnorm < 1e-14:
            break
        g_dir /= gnorm

        # Learning rate schedule
        lr = lr_init * (1.0 - 0.5 * progress) / (1.0 + 0.3 * np.log1p(T / 50.0))

        # Momentum + backtracking
        momentum_buf = beta_m * momentum_buf + (1 - beta_m) * (-g_dir)

        a_trial = None
        for i in range(12):
            a_cand = a + lr * momentum_buf
            a_cand = project_nonneg_sum(a_cand, S)
            c_cand = fft_convolve_self(a_cand)
            if c_cand.max() < cur_obj - 1e-10:
                a_trial = a_cand
                break
            lr *= 0.5

        if a_trial is None:
            break

        a = a_trial
        step += 1

        if verbose and step % 400 == 0:
            c_now = fft_convolve_self(a)
            print("    SD step %d: obj=%.7f, best=%.7f, T=%.0f, elapsed=%.0fs" %
                  (step, c_now.max(), best_obj, T, elapsed))

    return best_a, best_obj


def coordinate_descent(a_init, n, S, time_budget, n_factors=6, verbose=True):
    """Coordinate descent trying multiplicative changes on each coordinate."""
    a = project_nonneg_sum(a_init.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(999)

    factors = np.array([0.75, 0.85, 0.92, 1.08, 1.15, 1.25]) if n_factors == 6 else \
              np.array([0.7, 0.85, 0.95, 1.05, 1.15, 1.3])

    sweep = 0
    stagnant = 0

    while time.monotonic() - start < time_budget:
        sweep += 1
        order = rng.permutation(n)
        sweep_improvement = 0

        for i in order:
            if time.monotonic() - start > time_budget:
                break

            orig = a[i]
            if orig < 1e-12:
                continue

            best_val = orig
            best_obj_local = best_obj

            for f in factors:
                a[i] = orig * f
                a = project_nonneg_sum(a, S)
                c_check = fft_convolve_self(a)
                obj_check = c_check.max()
                if obj_check < best_obj_local - 1e-12:
                    best_obj_local = obj_check
                    best_val = orig * f

            a[i] = best_val
            a = project_nonneg_sum(a, S)

            if best_obj_local < best_obj - 1e-12:
                best_obj = best_obj_local
                best_a = a.copy()
                sweep_improvement += 1

        if sweep_improvement == 0 and sweep > 3:
            stagnant += 1
            if stagnant > 3:
                break
        else:
            stagnant = 0

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - start
            print("    CD sweep %d: obj=%.7f, elapsed=%.0fs" % (sweep, best_obj, elapsed))

    return best_a, best_obj


def fine_tune_peaks(a_init, n, S, time_budget, verbose=True):
    """
    Targeted optimization: identify the specific peaks and try to reduce them
    by adjusting coordinates that contribute most to those peaks.
    """
    a = project_nonneg_sum(a_init.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    rng = np.random.default_rng(555)

    while time.monotonic() - start < time_budget:
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-11:
            best_obj = cur_obj
            best_a = a.copy()

        # Find the peak with maximum value
        peak_idx = np.argmax(c)

        # The peak at index k is: sum_{i} a[i]*a[k-i]
        # To reduce it, we want to decrease coordinates that contribute heavily
        # Contribution of a[j] to c[k] is: a[k-j] (if valid) + a[j-k] (if valid, j!=k-j)
        # So coordinates j where a[j]*a[k-j] is large are the "important" ones

        # Find top contributing pairs
        if 0 <= peak_idx <= 2 * (n - 1):
            # c[peak_idx] = sum_{i=max(0,peak_idx-n+1)}^{min(n-1, peak_idx)} a[i]*a[peak_idx-i]
            i_lo = max(0, peak_idx - n + 1)
            i_hi = min(n - 1, peak_idx)
            if i_lo <= i_hi:
                contributions = a[i_lo:i_hi+1] * a[peak_idx - i_hi:peak_idx - i_lo:-1]
                # Top contributors
                top_contrib = np.argsort(contributions)[-5:]
                target_indices = (i_lo + top_contrib)

                # Try reducing these coordinates
                for idx in target_indices:
                    if a[idx] < 1e-10:
                        continue
                    for factor in [0.7, 0.8, 0.9, 1.1, 1.2]:
                        saved = a[idx]
                        a[idx] = saved * factor
                        a = project_nonneg_sum(a, S)
                        c_check = fft_convolve_self(a)
                        if c_check.max() < cur_obj - 1e-12:
                            cur_obj = c_check.max()
                            if cur_obj < best_obj:
                                best_obj = cur_obj
                                best_a = a.copy()
                        else:
                            a[idx] = saved
                            a = project_nonneg_sum(a, S)
                            break

    return best_a, best_obj


def spectrum_flatten(a_init, n, S, time_budget, verbose=True):
    """
    Try to flatten the power spectrum |FFT(a)|^2 to reduce autocorrelation peaks.
    Uses iterative amplitude/phase manipulation.
    """
    a = project_nonneg_sum(a_init.copy(), S)
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()

    start = time.monotonic()
    fsize = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - start < time_budget:
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)

        # Autocorrelation is IFFT(|A|^2)
        P = mag ** 2
        c_round = np.real(ifft(P))[:2 * n - 1]

        # Try to reduce the peak of c_round by modifying P
        # Push energy away from frequency components that cause peaks
        c_max = c_round.max()
        if c_max < 1e-10:
            break

        # Reduce power at frequencies that correlate with the peak
        # Simple approach: reduce overall power at high-magnitude frequencies
        # (since they contribute most to the autocorrelation peak)
        mag_threshold = np.percentile(mag, 90)
        reduction = 0.05
        mag_reduced = mag.copy()
        mag_reduced[mag > mag_threshold] *= (1 - reduction)

        # Reconstruct with modified magnitude, keep phase
        A_new = mag_reduced * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = project_nonneg_sum(a_new, S)

        c_new = fft_convolve_self(a_new)
        if c_new.max() < best_obj - 1e-12:
            best_obj = c_new.max()
            best_a = a_new.copy()
            a = a_new
        else:
            # Try the opposite: reduce low-magnitude frequencies
            mag_low = mag.copy()
            mag_low[mag < mag_threshold] *= (1 + reduction * 2)
            A_new2 = mag_low * np.exp(1j * phase)
            a_new2 = np.real(ifft(A_new2))[:n]
            a_new2 = project_nonneg_sum(a_new2, S)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < best_obj - 1e-12:
                best_obj = c_new2.max()
                best_a = a_new2.copy()
                a = a_new2
            else:
                break

    return best_a, best_obj

            a_best, _ = coordinate_descent(a, n, S, rem2, n_factors=4, verbose=False)
            a_best = project_nonneg_sum(a_best, S)
            obj = evaluate_sequence(a_best.tolist())
            if np.isfinite(obj) and obj < global_best:
                global_best = obj
                global_a = a_best.copy()
                print("  Final CD: obj=%.7f" % obj)

    # Zero masking attempt
    a = project_nonneg_sum(global_a.copy(), S)
    sorted_idx = np.argsort(a)
    improved_zm = False
    for idx in sorted_idx[:100]:
        if a[idx] < 1e-8:
            continue
        saved = a[idx]
        a[idx] = 0.0
        a = project_nonneg_sum(a, S)
        c_check = fft_convolve_self(a)
        obj_check = c_check.max()
        # Convert to full objective
        obj_full = 2.0 * n * obj_check / (S ** 2)
        if np.isfinite(obj_full) and obj_full < global_best - 1e-12:
            global_best = obj_full
            global_a = a.copy()
            improved_zm = True
        else:
            a[idx] = saved
            a = project_nonneg_sum(a, S)
    if improved_zm:
        print("  ZeroMask improved: obj=%.7f" % global_best)

    # Final verification
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    if np.isfinite(ver):
        print("\nFINAL: obj=%.7f (verified=%.7f)" % (global_best, ver))
    else:
        print("\nFINAL: obj=%.7f (verification failed, using fallback)" % global_best)
        final_a = [1.0] * 256
        print("Fallback: obj=%.7f" % evaluate_sequence(final_a))

    return final_a
