import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    result = np.maximum(v - theta, 0.0)
    # Ensure exact sum
    diff = S - result.sum()
    if abs(diff) > 1e-12:
        # Add diff to the largest element
        max_idx = np.argmax(result)
        result[max_idx] += diff
        result = np.maximum(result, 0.0)
    return result


def compute_conv(a, N):
    """Compute self-convolution via FFT."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    c = np.real(ifft(A * A))[:2*n-1]
    return c, A


def lse_gradient(a, N, beta):
    """Compute gradient of LSE-smoothed max of conv(a,a).
    
    Returns (gradient, lse_value, c, c_max, A).
    """
    n = len(a)
    c, A = compute_conv(a, N)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()

        return np.zeros(n), 0.0, c, c_max, A

    A_full = fft(a, fsize)

            inits.append(("given", np.clip(given, 0, 1000).astype(np.float64)))
    except Exception:
        pass
    inits.append(("rand", make_init_rand(n, rng)))
    inits.append(("chirp", make_init_chirp(n, rng)))
    inits.append(("multifreq", make_init_multi_freq(n, rng)))
    inits.append(("binary", make_init_binary_like(n, rng)))
    inits.append(("fourier", make_init_fourier(n, rng, N)))

    n_inits = len(inits)
    per_init = phase1_budget / n_inits
    print("  %d inits, ~%.1f sec each" % (n_inits, per_init))

    for name, arr in inits:
        if time.monotonic() > deadline - 120:
            break
        a_norm = safe_normalize(arr, n, S)
        t0 = time.monotonic()

        # Quick L-BFGS
        rem = max(1.0, per_init - (time.monotonic() - t0) - 0.5)
        a_l, sc_l = lbfgs_lse_block(a_norm, n, S, N, rem, rng,
                                     beta_init=5.0, beta_max=80.0, m_lbfgs=8)
        update_best(a_l, name + "/LBFGS")

    print("Phase 1 done. Best: %.6f" % global_best)

    # --- Phase 2: Focused L-BFGS optimization ---
    if global_a is None:
        global_a = np.ones(n) * S / n
        global_best = 2.0

    print("=== Phase 2: Focused L-BFGS ===")
    t2 = time.monotonic()
    phase2_budget = min(1200.0, (deadline - t2) * 0.5)

    a_curr = global_a.copy()
    stall = 0
    last_best = global_best

    while time.monotonic() < t2 + phase2_budget:
        block_time = min(120.0, (t2 + phase2_budget) - time.monotonic())
        if block_time < 5:
            break
        beta_start = 10.0 + 5.0 * stall
        beta_end = min(300.0, 50.0 * (1 + 0.5 * stall))
        a_new, sc_new = lbfgs_lse_block(a_curr, n, S, N, block_time, rng,
                                         beta_init=beta_start, beta_max=beta_end, m_lbfgs=12)
        if sc_new < global_best - 1e-9:
            global_best = sc_new
            global_a = a_new.copy()
            a_curr = a_new.copy()
            stall = 0
            print("  [P2] %.6f (beta %.0f-%.0f)" % (global_best, beta_start, beta_end))
        else:
            a_curr = a_new.copy()
            stall += 1
            if stall >= 3:
                a_curr = project_simplex(a_curr + rng.normal(0, 0.008 * S / np.sqrt(n), n), S)
                stall = 0

    # --- Phase 3: Spectral shaping refinement ---
    print("=== Phase 3: Spectral shaping ===")
    t3 = time.monotonic()
    phase3_budget = min(300.0, (deadline - t3) * 0.2)

    a_sp, sc_sp = spectral_block(global_a, n, S, N, phase3_budget, rng,
                                  top_k=10, alpha=0.15)
    update_best(a_sp, "spectral")

    a_sp2, sc_sp2 = spectral_block(global_a, n, S, N, min(150.0, (deadline - time.monotonic()) * 0.3),
                                    rng, top_k=5, alpha=0.08)
    update_best(a_sp2, "spectral_fine")

    # --- Phase 4: Nesterov polish ---
    print("=== Phase 4: Nesterov polish ===")
    t4 = time.monotonic()
    remaining = deadline - t4
    if remaining > 10:
        a_ng, sc_ng = nesterov_toppk_block(global_a, n, S, N, min(remaining - 5, 600.0),
                                            rng, top_k_init=20, top_k_max=300)
        update_best(a_ng, "nesterov")

    # --- Phase 5: Final high-beta L-BFGS ---
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_fin, sc_fin = lbfgs_lse_block(global_a, n, S, N, min(remaining - 3, 300.0),
                                         rng, beta_init=50.0, beta_max=500.0, m_lbfgs=10)
        update_best(a_fin, "final_LBFGS")

    # --- Verification ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("Verification: %.6f" % ver if np.isfinite(ver) else "Verification: inf")
        if not np.isfinite(ver):
            print("  Rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64)
    else:
        global_a = np.ones(256, dtype=np.float64)

    print("Returning length %d, internal=%.6f" % (len(global_a), global_best))
    return [float(x) for x in global_a]
