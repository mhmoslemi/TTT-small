import numpy as np
import time

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, fsize)
    c = np.real(np.fft.ifft(A * A))
    return c[:size]


def gradient_sum_c_squared(a, fsize):
    """Gradient of ||conv(a,a)||^2 with respect to a."""
    A = np.fft.fft(a, fsize)
    # grad = 4 * real(IFFT(A^2 * conj(A)))
    # For real a: conj(A(k)) = A(N-k)
    B = A * A * np.conj(A)
    grad = 4.0 * np.real(np.fft.ifft(B))[:len(a)]
    return grad

def lse_gradient(a, c, beta):
    """Gradient of smoothed max (log-sum-exp) of c with respect to a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = np.fft.fft(w, fsize)
    A = np.fft.fft(a, fsize)
    grad = 2.0 * np.real(np.fft.ifft(W * np.conj(A)))[:n]
    return grad

def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))

    return a * (S / s)


def spectral_balancing(a_init, n, S, max_time, rng, n_iter=None):
    """Gerchberg-Saxton style: alternate between time-domain (non-negative, sum=S)
    and frequency-domain (flat |A(k)|)."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    # Target: |A(k)| should be as flat as possible
    # The DC component A(0) = S is fixed.
    # For the self-convolution to be flat, we want |A(k)|^2 to be flat.
    # But A(0) is much larger than other components typically.
    # Let's target: |A(k)| = target for k != 0, and A(0) = S.
    # Sum |A(k)|^2 = N * sum(a^2) (Parseval)
    # If sum(a) = S, then sum(a^2) >= S^2/n (by QM-AM)
    # Let's just iterate: compute |A|, replace non-DC with median, keep DC.
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()
        
        A = np.fft.fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target magnitude: geometric mean of non-DC magnitudes
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        
        # Create new spectrum: flat magnitude for non-DC, keep DC
        A_new = np.zeros(fsize, dtype=complex)
        A_new[0] = A[0]
        for k in range(1, fsize):
            A_new[k] = target * np.exp(1j * phase[k])
        
        # Inverse FFT
        a_new = np.real(np.fft.ifft(A_new))[:n]
        
        # Project to non-negative with sum S
        a = project_simplex(a_new, S)
    
    return best_local, best_score


def make_anti_palindromic_init(n, rng, S):
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    a = a * (S / a.sum()) if a.sum() > 0 else np.ones(n) * S / n
    return a

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L2-spectral method.

Key innovation: Use ||conv(a,a)||^2 as a smooth quartic surrogate objective.
Since sum(conv(a,a)) = sum(a)^2 is fixed under normalization, minimizing
||conv||^2 is equivalent to minimizing the variance of the convolution values,
which drives the convolution toward flatness. The gradient is computed
efficiently via FFT: grad = 4*real(IFFT(A * |A|^2)) where A = FFT(a padded).

Then polish with LSE-FISTA on the actual max objective.
"""
import numpy as np
import time


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, fsize)
    c = np.real(np.fft.ifft(A * A))
    return c[:size]


def grad_l2_conv(a, fsize):
    """Gradient of ||conv(a,a)||^2 with respect to a (zero-padded FFT)."""
    A = np.fft.fft(a, fsize)
    B = A * np.abs(A) ** 2
    grad = 4.0 * np.real(np.fft.ifft(B))[:len(a)]
    return grad


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = np.fft.fft(w, fsize)
    A = np.fft.fft(a, fsize)
    grad = 2.0 * np.real(np.fft.ifft(W * np.conj(A)))[:n]
    return grad


def run_l2_fista(a_init, n, S, max_time, rng):
    """FISTA on ||conv(a,a)||^2 with projected gradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_max = np.inf
    best_local = a.copy()
    it = 0

    # Initial Lipschitz estimate
    grad0 = grad_l2_conv(y, fsize)
    L = np.linalg.norm(grad0) / max(S, 1e-10) * 10.0 + 1.0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_max:
            best_max = cur_max
            best_local = y.copy()

        grad = grad_l2_conv(y, fsize)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size with adaptive backtracking
        step = 1.0 / max(L, 1e-10)
        c_y = fft_convolve_self(y)
        f_y = np.sum(c_y ** 2)

        x_new = project_simplex(y - step * grad, S)

        # Backtrack if needed (check L2 objective, not max)
        if it % 4 == 0:
            c_cand = fft_convolve_self(x_new)
            f_cand = np.sum(c_cand ** 2)
            if f_cand > f_y + 0.01 * step * gnorm ** 2:
                L *= 1.3
                step = 1.0 / L
                x_new = project_simplex(y - step * grad, S)
        else:
            L = max(L * 0.99, gnorm / max(S, 1e-10) + 0.1)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        # Periodic perturbation
        if it % 200 == 0 and rng.random() < 0.15:
            noise = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0

    return best_local, best_max


def run_lse_fista(a_init, n, S, max_time, rng, beta_init=20.0, beta_max=1000.0):
    """FISTA on LogSumExp-smoothed max of convolution."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_max = np.inf
    best_local = a.copy()
    it = 0
    L = S * 5.0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.003 * it))

        c = fft_convolve_self(y)
        cur_max = c.max()
        if cur_max < best_max:
            best_max = cur_max
            best_local = y.copy()

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 1.0 / L
        x_new = project_simplex(y - step * grad, S)

        # Adaptive L
        if it % 10 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() > cur_max * 1.001:
                L *= 1.4
            else:
                L = max(L * 0.97, gnorm / max(S / n, 1e-10) + 0.5)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new

        if it % 300 == 0 and rng.random() < 0.1:
            noise = 0.003 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise, n), S)
            t_mom = 1.0

    return best_local, best_max


def run_topk_subgrad(a_init, n, S, max_time, rng, top_k=10):
    """Subgradient descent on top-k lags with momentum."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_max = np.inf
    best_local = a.copy()
    it = 0
    momentum = np.zeros(n)
    beta = 0.9
    lr = S / (n * 5.0)

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_max:
            best_max = cur_max
            best_local = a.copy()

        # Top-k subgradient
        k = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k]
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / k
        fsize = 1 << (len(w) + n - 2).bit_length()
        W = np.fft.fft(w, fsize)
        A = np.fft.fft(a, fsize)
        grad = 2.0 * np.real(np.fft.ifft(W * np.conj(A)))[:n]

        # Adam-style update
        momentum = beta * momentum + (1 - beta) * grad
        m_hat = momentum / (1 - beta ** (it + 1))
        denom = np.sqrt(m_hat ** 2).max() + 1e-10
        a_new = project_simplex(a - lr / denom * m_hat, S)

        # Adaptive lr
        if it % 50 == 0:
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur_max:
                lr *= 1.1
            else:
                lr *= 0.85
                a_new = a  # revert

        a = a_new

        if it % 500 == 0 and rng.random() < 0.1:
            noise = 0.002 * S / np.sqrt(n)
            a = project_simplex(a + rng.normal(0, noise, n), S)
            momentum = np.zeros(n)

    return best_local, best_max


def spectral_flat_iteration(a_init, n, S, max_time, rng, n_iter=200):
    """Gerchberg-Saxton: alternate time-domain projection and spectral flattening."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

    start = time.monotonic()
    best_max = np.inf
    best_local = a.copy()

    for it in range(n_iter):
        if time.monotonic() - start > max_time:
            break

        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_max:
            best_max = cur_max
            best_local = a.copy()

        A = np.fft.fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: make |A(k)| flat for k != 0
        # Use geometric mean of non-DC magnitudes
        non_dc = mag[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        # Blend: soften the flattening to avoid destroying the signal
        blend = 0.5
        new_mag = mag.copy()
        new_mag[1:] = blend * target + (1 - blend) * mag[1:]

        A_new = new_mag * np.exp(1j * phase)
        A_new[0] = A[0]  # keep DC fixed

        a_new = np.real(np.fft.ifft(A_new))[:n]
        a = project_simplex(a_new, S)

    return best_local, best_max


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_anti_palindromic_init(n, rng, S):
    """Anti-palindromic: a[i] + a[n-1-i] = const."""
    half = n // 2
    h = rng.random(half) + 0.1
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_random_init(n, rng, S):
    a = rng.random(n) + 0.05
    return a * (S / a.sum())


def make_periodic_init(n, rng, S, freq=None):
    if freq is None:
        freq = int(rng.integers(2, 15))
    tt = np.arange(n, dtype=np.float64) / n
    phase = rng.random() * 2 * np.pi
    a = 1.0 + 0.35 * np.cos(2 * np.pi * freq * tt + phase)
    a = np.maximum(a, 0.05)
    return a * (S / a.sum())


def make_bipolar_init(n, rng, S):
    """Initialize with a nearly binary structure (like Barker-type sequences)."""
    a = rng.choice([0.3, 1.0], size=n)
    # Ensure some structure: make it anti-palindromic-ish
    for i in range(n // 2):
        a[n - 1 - i] = 1.3 - a[i]
    a = np.maximum(a, 0.05)
    return a * (S / a.sum())

    inits.append(("random", make_random_init(primary_n, rng, S)))

    n_inits = len(inits)
    usable_time = max(30.0, deadline - time.monotonic())

    # === Phase 1: L2-FISTA (main workhorse) ===
    print("=== Phase 1: L2-FISTA on %d inits ===" % n_inits)
    per_init = usable_time * 0.30 / n_inits

    for name, arr in inits:
        if time.monotonic() > deadline - 90:
            break
        t0 = time.monotonic()
        seg = min(per_init, max(5.0, (deadline - time.monotonic()) * 0.5 / max(1, n_inits)))

        a_l2, s_l2 = run_l2_fista(arr, primary_n, S, seg, rng)
        a_l2 = safe_normalize(a_l2, primary_n)
        c = fft_convolve_self(a_l2)
        score = c.max()
        if score < global_best:
            global_best = score
            global_a = a_l2.copy()
            flatness = score / (S ** 2 / (2 * primary_n - 1))
            print("  L2 [%s]: max=%.6f, flatness_ratio=%.4f (t=%.0fs)" %
                  (name, score, flatness, time.monotonic() - t0))

    if global_a is None:
        global_a = make_random_init(primary_n, rng, S)
        global_best = fft_convolve_self(global_a).max()

    print("After Phase 1: best=%.6f" % global_best)

    # === Phase 2: Spectral flat iteration ===
    if time.monotonic() < deadline - 90:
        print("=== Phase 2: Spectral flat iteration ===")
        seg = min(45.0, max(10.0, (deadline - time.monotonic()) * 0.10))
        a_sf, s_sf = spectral_flat_iteration(global_a, primary_n, S, seg, rng)
        a_sf = safe_normalize(a_sf, primary_n)
        c_sf = fft_convolve_self(a_sf)
        if c_sf.max() < global_best:
            global_best = c_sf.max()
            global_a = a_sf.copy()
            print("  SpectralFlat: %.6f" % global_best)

    # === Phase 3: More L2-FISTA from best ===
    if time.monotonic() < deadline - 90:
        print("=== Phase 3: L2-FISTA polish ===")
        seg = min(60.0, max(10.0, (deadline - time.monotonic()) * 0.15))
        a_p, s_p = run_l2_fista(global_a, primary_n, S, seg, rng)
        a_p = safe_normalize(a_p, primary_n)
        c_p = fft_convolve_self(a_p)
        if c_p.max() < global_best:
            global_best = c_p.max()
            global_a = a_p.copy()
            print("  L2 polish: %.6f" % global_best)

    # === Phase 4: LSE-FISTA (smoothed max) ===
    if time.monotonic() < deadline - 60:
        print("=== Phase 4: LSE-FISTA ===")
        seg = min(90.0, max(15.0, (deadline - time.monotonic()) * 0.20))
        a_lse, s_lse = run_lse_fista(global_a, primary_n, S, seg, rng,
                                       beta_init=20.0, beta_max=800.0)
        a_lse = safe_normalize(a_lse, primary_n)
        c_lse = fft_convolve_self(a_lse)
        if c_lse.max() < global_best:
            global_best = c_lse.max()
            global_a = a_lse.copy()
            print("  LSE: %.6f" % global_best)

    # === Phase 5: Top-k subgradient ===
    if time.monotonic() < deadline - 60:
        print("=== Phase 5: Top-k subgradient ===")
        seg = min(45.0, max(10.0, (deadline - time.monotonic()) * 0.12))
        for tk in [5, 15]:
            if time.monotonic() > deadline - 60:
                break
            a_tk, s_tk = run_topk_subgrad(global_a, primary_n, S, seg / 2, rng, top_k=tk)
            a_tk = safe_normalize(a_tk, primary_n)
            c_tk = fft_convolve_self(a_tk)
            if c_tk.max() < global_best:
                global_best = c_tk.max()
                global_a = a_tk.copy()
                print("  TopK(%d): %.6f" % (tk, global_best))

    # === Phase 6: High-beta LSE final polish ===
    if time.monotonic() < deadline - 30:
        print("=== Phase 6: Final high-beta LSE ===")
        seg = max(10.0, (deadline - time.monotonic()) * 0.25)
        a_f, s_f = run_lse_fista(global_a, primary_n, S, seg, rng,
                                   beta_init=100.0, beta_max=3000.0)
        a_f = safe_normalize(a_f, primary_n)
        c_f = fft_convolve_self(a_f)
        if c_f.max() < global_best:
            global_best = c_f.max()
            global_a = a_f.copy()
            print("  Final LSE: %.6f" % global_best)

    # === Phase 7: Try alternative length ===
    if time.monotonic() < deadline - 60:
        for alt_n in [4096, 16384]:
            if time.monotonic() > deadline - 60:
                break
            S_alt = np.sqrt(2.0 * alt_n)
            if alt_n > primary_n:
                tt_old = np.linspace(0, 1, primary_n)
                tt_new = np.linspace(0, 1, alt_n)
                a_alt = np.interp(tt_new, tt_old, global_a)
            else:
                # Downsample by averaging
                chunk = primary_n // alt_n
                a_alt = np.zeros(alt_n)
                for i in range(alt_n):
                    a_alt[i] = global_a[i * chunk:(i + 1) * chunk].mean()
            a_alt = np.maximum(a_alt, 0.0)
            a_alt = safe_normalize(a_alt, alt_n)

            seg = max(10.0, min(30.0, (deadline - time.monotonic()) * 0.15))
            a_alt, _ = run_l2_fista(a_alt, alt_n, S_alt, seg, rng)
            a_alt = safe_normalize(a_alt, alt_n)
            c_alt = fft_convolve_self(a_alt)
            obj_alt = 2.0 * alt_n * c_alt.max() / (a_alt.sum() ** 2)
            obj_cur = global_best  # since S^2 = 2n, objective = max(c)

            if obj_alt < obj_cur:
                global_best = c_alt.max()
                global_a = a_alt.copy()
                primary_n = alt_n
                print("  Alt n=%d: obj=%.6f" % (alt_n, obj_alt))

    # === Final verification ===
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL: eval=%.6f, internal=%.6f, n=%d ===" %
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        print("  Fallback to uniform")
        global_a = np.ones(1024, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback eval: %.6f" % (ver if np.isfinite(ver) else float('inf')))

    return final_list
