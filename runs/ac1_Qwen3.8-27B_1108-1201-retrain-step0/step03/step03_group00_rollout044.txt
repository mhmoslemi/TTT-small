"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: Multi-length FISTA with adaptive LSE annealing, subgradient polishing,
and importance-weighted top-k gradients. Explores n in {8192, 16384, 32768}
with progressive refinement and exact subgradient final polish.

import numpy as np
import time


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_from_scipy(a, size=None):
    from scipy.fft import fft as sfft
    return sfft(a, size)


def ifft_from_scipy(a, size=None):
    from scipy.fft import ifft as sifft
    return sifft(a, size)


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation via FFT."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft_from_scipy(w, fsize)
    A = fft_from_scipy(a, fsize)
    return 2.0 * np.real(ifft_from_scipy(W * np.conj(A)))[:n]


def topk_gradient(a, c, k, power=1.0):
    """Top-k gradient with power-law weighting."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    vmax = vals[0]
    if vmax < 1e-300:
        return np.zeros(n)
    w = np.power(np.maximum(vals, 1e-30), power)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    w_full = np.zeros(len(c))
    w_full[idx] = w
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft_from_scipy(w_full, fsize)
    A = fft_from_scipy(a, fsize)
    return 2.0 * np.real(ifft_from_scipy(W * np.conj(A)))[:n]


def exact_subgradient(a, c):
    """Exact subgradient at the argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def compute_score(a):
    """Compute the actual evaluation score for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    max_c = c.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

            noise_scale = min(noise_scale, 0.05 * S / np.sqrt(n))
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_k = 1.0
            if staggard_count > 5:
                # More aggressive restart
                y = normalize_sum(best_a * (1 + 0.01 * rng.standard_normal(n)), n)
                noise_scale = 0.01 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_k = 1.0
                staggard_count = 0

        x_new = project_simplex(y - step * grad, S)

        # FISTA momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

        # Periodic small perturbations
        if it % 5000 == 0:
            decay = np.exp(-it / 30000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t_k = 1.0

    # Final check
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()

    return best_a, best


def run_subgradient_polish(a_init, n, S, max_time, rng):
    """Exact subgradient polishing with diminishing steps."""
    from scipy.fft import fft as sfft, ifft as sifft

    y = np.array(a_init, copy=True)
    y = normalize_sum(y, n)
    start = time.monotonic()
    best = np.inf
    best_y = y.copy()
    it = 0

    while time.monotonic() - start < max_time - 1:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-14:
            best = cur
            best_y = y.copy()

        grad = exact_subgradient(y, c)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Diminishing step size
        step = S / (gnorm * (1.0 + 0.001 * it ** 0.7))
        step = min(step, 0.5 / it ** 0.3)

        y_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        # Occasional noise to escape
        if it % 2000 == 0 and rng.random() < 0.3:
            noise_scale = 0.0005 * S / np.sqrt(n) / np.sqrt(1 + it / 5000.0)
            y_new = project_simplex(y_new + rng.normal(0, noise_scale, n), S)

        y = y_new

    c_final = fft_convolve_self(best_y)
    if c_final.max() < best:
        best = c_final.max()
        best_y = best_y.copy()

    return best_y, best


def run_topk_polish(a_init, n, S, max_time, rng):
    """Top-k gradient polishing with increasing k."""
    from scipy.fft import fft as sfft, ifft as sifft

    y = np.array(a_init, copy=True)
    y = normalize_sum(y, n)
    a = y.copy()
    t_k = 1.0
    start = time.monotonic()
    best = np.inf
    best_y = y.copy()
    it = 0

    while time.monotonic() - start < max_time - 1:
        it += 1
        k = min(1024, 16 + it // 20)
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-14:
            best = cur
            best_y = y.copy()

        grad = topk_gradient(y, c, k, power=1.5)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        step = 2.0 * S / (n * (1.0 + 0.001 * it ** 0.5))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new

    c_final = fft_convolve_self(best_y)
    if c_final.max() < best:
        best = c_final.max()
        best_y = best_y.copy()

    return best_y, best


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-6.0 * (tt - 0.5) ** 2))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(np.sin(np.pi * tt) ** 8)

    # Flat-top with smooth edges
    edge = 0.1
    flat = np.ones(n)
    for i in range(int(edge * n)):
        flat[i] = 0.5 * (1 + np.cos(np.pi * i / (edge * n)))
        flat[n - 1 - i] = flat[i]
    inits.append(flat)

    # Random
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(0.5 * rng.standard_normal(n)))

    # Two-level structure
    levels = rng.integers(3, 11)
    block = n // levels
    hts = rng.uniform(0.5, 1.5, levels)
    seq = np.repeat(hts, block)
    if len(seq) < n:
        seq = np.concatenate([seq, np.full(n - len(seq), hts[-1])])
    inits.append(seq[:n])

    # Triangular
    inits.append(np.maximum(1.0 - 2.0 * np.abs(tt - 0.5), 0.01))

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Try to use the given sequence as a warm start
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 512 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    print("Budget: %.0fs, deadline at t=%.0f" % (budget_s, budget_s))
    if given is not None:
        print("Given sequence length: %d" % len(given))
        gs = compute_score(given)
        print("Given sequence score: %.6f" % gs)

    # Define lengths to try
    lengths = [8192, 16384, 32768]

    # Phase 1: Multi-length exploration (40% of time)
    phase1_time = (deadline - time.monotonic()) * 0.40
    print("\n=== Phase 1: Exploration (%.0fs) ===" % phase1_time)
    p1_start = time.monotonic()

    # Build candidate list
    candidates = []

    if given is not None:
        # Use given sequence and resized versions
        for target_n in [8192, 16384]:
            if len(given) != target_n:
                # Resize by interpolation
                t_old = np.linspace(0, 1, len(given))
                t_new = np.linspace(0, 1, target_n)
                resized = np.interp(t_new, t_old, given)
                resized = np.maximum(resized, 0.0)
                if resized.sum() > 0.01:
                    candidates.append((target_n, resized, "given_%d" % target_n))
        candidates.append((len(given), given.copy(), "given"))

    # Generate inits for each length
    for n in lengths:
        inits = generate_structured_inits(n, rng)
        for i, init in enumerate(inits[:4]):  # Take a few per length
            candidates.append((n, init, "init_%d_%d" % (n, i)))

    # Allocate time to candidates
    n_cands = len(candidates)
    per_cand = phase1_time / n_cands

    for ci, (n_c, init_c, name) in enumerate(candidates):
        if time.monotonic() > deadline - (deadline - p1_start) * 0.60 - 30:
            break

        S_c = np.sqrt(2.0 * n_c)
        t_alloc = per_cand
        if t_alloc < 5:
            continue

        # Use different modes for diversity
        use_topk = (ci % 3 == 1)
        beta_init = 5.0 if not use_topk else 3.0
        beta_rate = 0.003
        beta_max = 2000.0

        a_opt, score_opt = run_fista_adaptive(
            init_c, n_c, S_c, t_alloc, rng,
            beta_init=beta_init, beta_rate=beta_rate,
            beta_max=beta_max, use_topk=use_topk, topk_k=128
        )

        a_norm = normalize_sum(a_opt, n_c)
        real_score = compute_score(a_norm)

        if real_score < global_best:
            global_best = real_score
            global_a = a_norm.copy()
            print("  [%s] n=%d: score=%.6f (elapsed %.0fs)" % (name, n_c, global_best, time.monotonic() - p1_start))

    print("=== After Phase 1: best=%.6f, elapsed=%.0fs ===" % (global_best, time.monotonic() - p1_start))

    # Phase 2: Focused optimization on best (35% of time)
    phase2_time = (deadline - time.monotonic()) * 0.55
    print("\n=== Phase 2: Focused Optimize (%.0fs) ===" % phase2_time)

    if global_a is not None:
        n_best = len(global_a)
        S_best = np.sqrt(2.0 * n_best)

        # 2a: High-beta FISTA
        t2a = phase2_time * 0.45
        a2, s2 = run_fista_adaptive(
            global_a, n_best, S_best, t2a, rng,
            beta_init=50.0, beta_rate=0.002,
            beta_max=5000.0, use_topk=False
        )
        a2 = normalize_sum(a2, n_best)
        score2 = compute_score(a2)
        if score2 < global_best:
            global_best = score2
            global_a = a2.copy()
            print("  After high-beta FISTA: %.6f" % global_best)

        # 2b: Top-k polishing
        t2b = phase2_time * 0.30
        a3, s3 = run_topk_polish(global_a, n_best, S_best, t2b, rng)
        a3 = normalize_sum(a3, n_best)
        score3 = compute_score(a3)
        if score3 < global_best:
            global_best = score3
            global_a = a3.copy()
            print("  After top-k polish: %.6f" % global_best)

        # 2c: Exact subgradient polishing
        t2c = phase2_time * 0.25
        a4, s4 = run_subgradient_polish(global_a, n_best, S_best, t2c, rng)
        a4 = normalize_sum(a4, n_best)
        score4 = compute_score(a4)
        if score4 < global_best:
            global_best = score4
            global_a = a4.copy()
            print("  After subgradient polish: %.6f" % global_best)

    # Phase 3: Try to improve with a different length if time remains
    remaining = deadline - time.monotonic()
    print("\n=== Phase 3: Remaining %.0fs ===" % remaining)

    if remaining > 30 and global_a is not None:
        # Try a slightly different length
        n_cur = len(global_a)
        alt_lengths = [x for x in [4096, 8192, 16384, 32768, 65536] if x != n_cur and 1024 <= x <= 65536]

        for alt_n in alt_lengths[:2]:
            if remaining < 20:
                break
            S_alt = np.sqrt(2.0 * alt_n)

            # Resize current best to new length
            t_old = np.linspace(0, 1, n_cur)
            t_new = np.linspace(0, 1, alt_n)
            init_alt = np.interp(t_new, t_old, global_a)
            init_alt = np.maximum(init_alt, 0.0)

            t_alt = min(remaining * 0.4, 120.0)
            a_alt, s_alt = run_fista_adaptive(
                init_alt, alt_n, S_alt, t_alt, rng,
                beta_init=30.0, beta_rate=0.003,
                beta_max=3000.0, use_topk=False
            )
            a_alt = normalize_sum(a_alt, alt_n)
            score_alt = compute_score(a_alt)

            if score_alt < global_best:
                global_best = score_alt
                global_a = a_alt.copy()
                n_cur = alt_n
                print("  Alt n=%d: %.6f" % (alt_n, global_best))

            remaining = deadline - time.monotonic()

            # Quick subgradient polish
            if remaining > 10:
                S_alt2 = np.sqrt(2.0 * alt_n)
                a_pol, s_pol = run_subgradient_polish(global_a, alt_n, S_alt2, min(remaining * 0.3, 30.0), rng)
                a_pol = normalize_sum(a_pol, alt_n)
                score_pol = compute_score(a_pol)
                if score_pol < global_best:
                    global_best = score_pol
                    global_a = a_pol.copy()
                    print("  After polish n=%d: %.6f" % (alt_n, global_best))

    # Final: one more subgradient polish on the best
    remaining = deadline - time.monotonic()
    if remaining > 5 and global_a is not None:
        n_fin = len(global_a)
        S_fin = np.sqrt(2.0 * n_fin)
        a_fin, s_fin = run_subgradient_polish(global_a, n_fin, S_fin, remaining * 0.7, rng)
        a_fin = normalize_sum(a_fin, n_fin)
        score_fin = compute_score(a_fin)
        if score_fin < global_best:
            global_best = score_fin
            global_a = a_fin.copy()
            print("  Final polish: %.6f" % global_best)

    # Verification
    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        global_best = compute_score(global_a)

    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    ver = evaluate_sequence(final_list)
    print("\nFinal score (internal): %.6f" % global_best)
    print("Verification score: %.6f" % ver if np.isfinite(ver) else "Verification: inf")

    if not np.isfinite(ver):
        print("Fallback to ones sequence")
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)
        final_list = [float(x) for x in global_a]

    print("Returning sequence of length %d" % len(final_list))
    return final_list
