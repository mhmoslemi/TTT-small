def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates < w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def linear_conv_self(a):
    """Full linear self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def spectral_equality_step(a, N, alpha, rng=None):

    then project back to non-negative simplex.
    """

    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    P_mean = P.mean()

    a_new = project_simplex(a_new_full, S)

def fista_logsumexp(a_init, S, max_time, rng, beta_init=5.0, beta_max=500.0):

def spectral_iteration(a_init, n, S, N, max_time, rng, alpha_init=0.1, alpha_decay=0.9995):
    """
    Iterative spectral shaping: repeatedly flatten the power spectrum.
    This is a "dual" approach - in the frequency domain, flatten |A|^2;
    in the time domain, project to non-negative.
    """
    a = a_init.copy()
    alpha = alpha_init
    start = time.monotonic()
    best_score = linear_conv_self(a).max()
    best_a = a.copy()
    stall = 0
    best_stall_score = best_score
    
    while time.monotonic() - start < max_time:
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = fft(a_pad)
        P = np.abs(A) ** 2
        P_mean = P[P > 1e-30].mean() if np.any(P > 1e-30) else P.mean()
        
        # Flatten: P_new = (1-alpha)*P + alpha*P_mean
        P_new = (1.0 - alpha) * P + alpha * P_mean
        P_new = np.maximum(P_new, 0.0)
        
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a = project_simplex(a_new, S)
        
        c = linear_conv_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            best_stall_score = cur
            stall = 0
        else:
            if cur > best_stall_score * 1.001:
                stall += 1
            else:
                stall += 1
        
        if stall > 200:
            break
        
        alpha *= alpha_decay
        alpha = max(alpha, 0.001)
    
    return best_a, best_score

Minimize 2*n*max_conv/sum^2 using multi-method optimization with 
spectral equalization, exact subgradient, and adaptive restarts.

Strategy:
1. Start from the provided sequence (already at ~1.507)
2. Apply iterative spectral equalization (flatten |FFT|^2)
3. Apply exact subgradient on the max (not top-k approximation)
4. Use adaptive step sizes with backtracking
5. Periodic restarts with controlled perturbations
6. Final polish with very small steps
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates < w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def linear_conv_self(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_linf(a, c):
    """Compute the (sub)gradient of max_k c[k] where c = conv(a,a).
    Returns gradient array of length n."""
    n = len(a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            if j == i:
                grad[i] = a[i]
            else:
                grad[i] = 2.0 * a[j]
    return grad


def compute_grad_toppk(a, c, top_k):
    """Weighted subgradient using top-k active lags via FFT."""
    n = len(a)
    top_k = min(top_k, len(c))
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    weights = top_vals / total
    
    # Build weight vector
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]
    
    # Gradient via cross-correlation
    L = 1 << (n + 2 * n - 2).bit_length()
    A_fft = fft(a, L)
    W_fft = fft(w_full, L)
    grad = 2.0 * np.real(ifft(np.conj(A_fft) * W_fft))[:n]
    return grad


def spectral_flatten_step(a, N, S, alpha):
    """One step: flatten the power spectrum, project back."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    P_mean = P.mean()
    P_new = (1.0 - alpha) * P + alpha * P_mean
    P_new = np.maximum(P_new, 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)


def spectral_damp_peaks(a, N, S, alpha, top_k):
    """One step: damp frequencies that contribute to peak autocorrelation."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()
    
    if c_max < 1e-30:
        return a
    
    m = np.arange(N)
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a
    weights = top_vals / total
    
    cos_eff = np.zeros(N)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N)
    
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    return project_simplex(a_new, S)
