import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, c, beta):
    """Gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, via FFT."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_contribution(a, k):
    """Compute how much each index contributes to lag k of self-convolution.
    contrib[i] = a[i+k] (if valid) + a[i-k] (if valid)
    Reducing a[i] by delta reduces c[k] by delta * contrib[i]."""
    n = len(a)
    contrib = np.zeros(n)
    if 0 < k < n:
        # a[i] * a[i+k]: for i in [0, n-1-k]
        if k < n:
            contrib[:n-k] += a[k:]
        # a[i-k] * a[i]: for i in [k, n-1]
        contrib[k:] += a[:n-k]
    elif k == 0:
        contrib[:] = 2.0 * a  # c[0] = sum a[i]^2, grad w.r.t. a[i] = 2*a[i]
    return contrib

        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        step = 1.0 / L
        f_y = c.max()
        x_new = project_simplex(y - step * grad, S)

        # Check improvement (periodically to save time)
        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            f_cand = c_x.max()
            bt = 0
            while f_cand > f_y + 1e-10 * S and bt < 4 and step > 1e-10:
                step *= 0.5
                x_new = project_simplex(y - step * grad, S)
                c_x = fft_convolve_self(x_new)
                f_cand = c_x.max()
                bt += 1
            if f_cand > f_y:
                L *= 1.5
            elif it % 50 == 0:
                L = max(L * 0.98, beta * 2.0)
        else:
            if it % 50 == 0:
                L = max(L * 0.98, beta * 2.0)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
        y = x_new + ((t_momentum - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_momentum = t_new

        # Periodic peak deflation
        if it % deflate_every == 0:
            a, _ = peak_deflation(a, S, n, max_swaps=deflate_swaps,
                                  budget_check=2.0, budget_start=time.monotonic())
            y = a.copy()
            t_momentum = 1.0
            c_def = fft_convolve_self(a)
            if c_def.max() < best_score:
                best_score = c_def.max()
                best_local = a.copy()
                stall_count = 0

        # Periodic perturbation to escape local minima
        if stall_count > 50 and rng.random() < 0.4:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t_momentum = 1.0
            stall_count = 0

    return best_local, best_score


def run_coord_descent(a_init, n, S, max_time, rng, batch_size=50):
    """Coordinate descent: select top-k problematic indices and update them.
    Uses second-order approximation for step size."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = a.copy()

        # Find top lags
        top_lags = np.argsort(c)[::-1][:5]
        
        # Compute gradient for top lags (hard selection)
        grad = np.zeros(n)
        for lag in top_lags:
            lag = int(lag)
            if lag == 0:
                grad[:n] += 2.0 * a[:n]
            else:
                k = lag
                if k < n:
                    grad[:n-k] += a[k:]
                    grad[k:] += a[:n-k]
        
        # Negative gradient direction: reduce high-gradient indices
        # But we need to maintain sum constraint, so do it as a swap
        # Pick indices to reduce (high grad) and increase (low grad)
        
        # Sort by gradient
        order = np.argsort(grad)
        # Top batch to reduce, bottom batch to increase
        reduce_idx = order[-batch_size:]
        increase_idx = order[:batch_size]
        
        # Compute total reduction and increase
        total_reduce = grad[reduce_idx].sum()
        total_increase = grad[increase_idx].sum()
        net_change = total_increase - total_reduce
        
        if net_change >= 0:
            # Try reverse: increase the high-grad ones? No, that's wrong direction.
            # This means the gradient is already pointing the right way for some.
            # Just do a small random perturbation
            pert = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a_new = project_simplex(a + pert, S)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < cur:
                a = a_new
            continue

        # Step size: we want to reduce c by approximately net_change * step
        # a_new[i] = a[i] - step * grad_weight[i] for reduce indices
        # The change in c[lag] for a swap of delta from i to j is:
        # -delta*contrib_i[lag] + delta*contrib_j[lag]
        # Total change = -delta * (sum contrib_i - sum contrib_j) / count
        # We want this to be a fraction of the current peak
        target_reduction = 0.01 * cur
        avg_diff = (total_reduce - total_increase) / (2 * batch_size)
        if avg_diff < 1e-15:
            break
        step = target_reduction / avg_diff
        step = min(step, 0.3 * a[reduce_idx].min())  # Can't go negative
        step = max(step, 1e-8)
        
        a_new = a.copy()
        per_delta = step
        a_new[reduce_idx] -= per_delta
        a_new[increase_idx] += per_delta
        a_new = np.clip(a_new, 0, 1000.0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new = a_new * (S / s_new)
        
        c_new = fft_convolve_self(a_new)
        if c_new.max() < cur:
            a = a_new
        else:
            # Try half step
            a_new2 = a.copy()
            a_new2[reduce_idx] -= per_delta * 0.5
            a_new2[increase_idx] += per_delta * 0.5
            a_new2 = np.clip(a_new2, 0, 1000.0)
            s_new2 = a_new2.sum()
            if s_new2 > 1e-10:
                a_new2 = a_new2 * (S / s_new2)
            c_new2 = fft_convolve_self(a_new2)
            if c_new2.max() < cur:
                a = a_new2

    return best_local, best_score


def make_anti_palindromic_init(n, rng, S, smooth=True):
    """Create anti-palindromic initialization."""
    half = n // 2
    if smooth:
        # Smooth random: use low-frequency Fourier components
        tt = np.arange(half, dtype=np.float64) / half
        h = 0.5 + 0.3 * np.sin(2 * np.pi * rng.integers(2, 8) * tt + rng.random() * 2 * np.pi)
        h += 0.15 * np.sin(2 * np.pi * rng.integers(3, 12) * tt + rng.random() * 2 * np.pi)
        h = np.maximum(h, 0.1)
    else:
        h = rng.random(half) + 0.1
    
    mu = S / half
    a = np.zeros(n)
    a[:half] = h
    for i in range(half):
        a[n - 1 - i] = mu - h[i]
    if n % 2 == 1:
        a[half] = mu / 2
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a = a * (S / a.sum())
    else:
        a = np.ones(n) * S / n
    return a


def make_cosine_taper_init(n, rng, S):
    """Cosine-tapered initialization: high in middle, low at edges."""
    tt = np.arange(n, dtype=np.float64) / n
    center = 0.5
    width = rng.uniform(0.3, 0.7)
    a = np.exp(-((tt - center) / width) ** 2)
    a += 0.05 * np.ones(n)
    a = a * (S / a.sum())
    return a


def make_dual_peak_init(n, rng, S):
    """Dual peak initialization."""
    tt = np.arange(n, dtype=np.float64) / n
    p1 = rng.uniform(0.2, 0.4)
    p2 = 1.0 - p1
    w = rng.uniform(0.1, 0.25)
    a = np.exp(-((tt - p1) / w) ** 2) + np.exp(-((tt - p2) / w) ** 2)
    a += 0.03 * np.ones(n)
    a = a * (S / a.sum())
    return a


def make_spectral_flat_init(n, rng, S):
    """Initialize by trying to make the spectrum flat (which would give flat autocorrelation).
    Use random phases in frequency domain, then take magnitude and impose structure."""
    # Generate random Fourier coefficients with flat magnitude
    fsize = 1 << (n - 1).bit_length()
    A = np.zeros(fsize, dtype=np.complex128)
    for k in range(fsize):
        A[k] = rng.random() * np.exp(1j * rng.random() * 2 * np.pi)
    a = np.real(ifft(A))[:n]
    a = np.abs(a) + 0.1
    a = a * (S / a.sum())
    return a


def try_length(n, a_init, S, rng, max_time):
    """Run optimization for a given length and initialization."""
    # Phase A: FISTA with deflation
    seg_a = max_time * 0.45
    a1, s1 = run_fista_deflate(a_init, n, S, seg_a, rng,
                                beta_init=5.0, beta_max=600.0,
                                deflate_every=40, deflate_swaps=20)
    a1 = project_simplex(a1, S)
    c1 = fft_convolve_self(a1)
    best_score = c1.max()
    best_a = a1.copy()
    
    # Phase B: Coordinate descent
    seg_b = max_time * 0.25
    a2, s2 = run_coord_descent(a1, n, S, seg_b, rng, batch_size=30)
    a2 = project_simplex(a2, S)
    c2 = fft_convolve_self(a2)
    if c2.max() < best_score:
        best_score = c2.max()
        best_a = a2.copy()
    
    # Phase C: More FISTA polish
    seg_c = max_time * 0.30
    a3, s3 = run_fista_deflate(best_a, n, S, seg_c, rng,
                                beta_init=50.0, beta_max=2000.0,
                                deflate_every=30, deflate_swaps=40)
    a3 = project_simplex(a3, S)
    c3 = fft_convolve_self(a3)
    if c3.max() < best_score:
        best_score = c3.max()
        best_a = a3.copy()
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 15.0)

    global_best = np.inf
    global_a = None
    global_n = 0

    # Try to get parent sequence
    parent = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            parent = np.clip(given, 0, 1000)
    except NameError:
        pass
    except Exception:
        pass

    print("Parent available: %s, length: %s" % (parent is not None, len(parent) if parent is not None else 0))

    # === Phase 1: Optimize from parent (n=8192) ===
    if parent is not None and len(parent) >= 64:
        n1 = len(parent)
        S1 = np.sqrt(2.0 * n1)
        a_init = parent * (S1 / parent.sum())
        a_init = np.clip(a_init, 0, 1000)
        
        t0 = time.monotonic()
        print("=== Phase 1: Optimize parent n=%d ===" % n1)
        time_avail = min(max_time_phase1(deadline, t0), 600)
        
        a_best, score = try_length(n1, a_init, S1, rng, time_avail)
        obj = 2.0 * n1 * score / S1 ** 2
        if obj < global_best:
            global_best = obj
            global_a = a_best.copy()
            global_n = n1
            print("  Parent optimized: obj=%.6f (max_b=%.2e)" % (obj, score))
        print("  Phase 1 done in %.1fs" % (time.monotonic() - t0))

    # === Phase 2: Try new anti-palindromic inits at n=8192 ===
    n2 = 8192
    S2 = np.sqrt(2.0 * n2)
    
    if time.monotonic() < deadline - 200:
        t0 = time.monotonic()
        print("=== Phase 2: New inits n=%d ===" % n2)
        
        inits = []
        for i in range(4):
            inits.append(make_anti_palindromic_init(n2, rng, S2, smooth=(i % 2 == 0)))
        inits.append(make_cosine_taper_init(n2, rng, S2))
        inits.append(make_dual_peak_init(n2, rng, S2))
        inits.append(make_spectral_flat_init(n2, rng, S2))
        
        # Also try smoothed parent if available
        if parent is not None and len(parent) >= 64:
            from scipy.ndimage import gaussian_filter1d
            p_smooth = gaussian_filter1d(parent, sigma=5)
            p_smooth = np.clip(p_smooth, 0, 1000)
            if p_smooth.sum() > 0:
                inits.append(p_smooth * (S2 / p_smooth.sum()))
        
        time_per_init = min(120.0, max(30.0, (deadline - time.monotonic() - 150) / len(inits)))
        
        for i, init in enumerate(inits):
            if time.monotonic() > deadline - 150:
                break
            t1 = time.monotonic()
            a_best, score = try_length(n2, init, S2, rng, time_per_init)
            obj = 2.0 * n2 * score / S2 ** 2
            if obj < global_best:
                global_best = obj
                global_a = a_best.copy()
                global_n = n2
                print("  Init %d: obj=%.6f" % (i, obj))
            print("  Init %d (t=%.0fs): obj=%.6f" % (i, time.monotonic() - t1, obj))
        
        print("  Phase 2 done in %.1fs" % (time.monotonic() - t0))

    # === Phase 3: Try n=16384 ===
    if time.monotonic() < deadline - 300:
        n3 = 16384
        S3 = np.sqrt(2.0 * n3)
        t0 = time.monotonic()
        print("=== Phase 3: Longer n=%d ===" % n3)
        
        # Upsample the best found
        if global_a is not None:
            from numpy import interp
            tt_old = np.linspace(0, 1, len(global_a))
            tt_new = np.linspace(0, 1, n3)
            a_long_init = interp(tt_new, tt_old, global_a)
            a_long_init = np.maximum(a_long_init, 0.0)
            a_long_init = a_long_init * (S3 / a_long_init.sum())
        else:
            a_long_init = make_anti_palindromic_init(n3, rng, S3, smooth=True)
        
        time_long = min(200.0, max(60.0, (deadline - time.monotonic() - 100) * 0.5))
        a_best, score = try_length(n3, a_long_init, S3, rng, time_long)
        obj = 2.0 * n3 * score / S3 ** 2
        if obj < global_best:
            global_best = obj
            global_a = a_best.copy()
            global_n = n3
            print("  Long: obj=%.6f" % obj)
        print("  Phase 3 done in %.1fs" % (time.monotonic() - t0))

    # === Phase 4: Intensive polish on best ===
    if global_a is not None and time.monotonic() < deadline - 30:
        n = len(global_a)
        S = np.sqrt(2.0 * n)
        t0 = time.monotonic()
        remaining = deadline - time.monotonic() - 20
        print("=== Phase 4: Final polish n=%d, remaining=%.0fs ===" % (n, remaining))
        
        # 4a: Very high beta FISTA
        seg = remaining * 0.35
        a_p, s_p = run_fista_deflate(global_a, n, S, seg, rng,
                                      beta_init=100.0, beta_max=5000.0,
                                      deflate_every=30, deflate_swaps=50)
        a_p = project_simplex(a_p, S)
        c_p = fft_convolve_self(a_p)
        obj_p = 2.0 * n * c_p.max() / S ** 2
        if obj_p < global_best:
            global_best = obj_p
            global_a = a_p.copy()
            print("  Polish A: obj=%.6f" % obj_p)
        
        # 4b: Coordinate descent
        remaining2 = deadline - time.monotonic() - 20
        seg2 = remaining2 * 0.30
        if seg2 > 5:
            a_cd, s_cd = run_coord_descent(global_a, n, S, seg2, rng, batch_size=20)
            a_cd = project_simplex(a_cd, S)
            c_cd = fft_convolve_self(a_cd)
            obj_cd = 2.0 * n * c_cd.max() / S ** 2
            if obj_cd < global_best:
                global_best = obj_cd
                global_a = a_cd.copy()
                print("  Polish B (coord): obj=%.6f" % obj_cd)
        
        # 4c: Peak deflation only (no gradient)
        remaining3 = deadline - time.monotonic() - 20
        if remaining3 > 5:
            a_def, _ = peak_deflation(global_a, S, n, max_swaps=500,
                                       budget_check=remaining3, budget_start=time.monotonic())
            a_def = project_simplex(a_def, S)
            c_def = fft_convolve_self(a_def)
            obj_def = 2.0 * n * c_def.max() / S ** 2
            if obj_def < global_best:
                global_best = obj_def
                global_a = a_def.copy()
                print("  Polish C (deflate): obj=%.6f" % obj_def)
        
        # 4d: Final tiny FISTA
        remaining4 = deadline - time.monotonic() - 20
        if remaining4 > 10:
            a_f, s_f = run_fista_deflate(global_a, n, S, remaining4 * 0.7, rng,
                                          beta_init=500.0, beta_max=10000.0,
                                          deflate_every=20, deflate_swaps=30)
            a_f = project_simplex(a_f, S)
            c_f = fft_convolve_self(a_f)
            obj_f = 2.0 * n * c_f.max() / S ** 2
            if obj_f < global_best:
                global_best = obj_f
                global_a = a_f.copy()
                print("  Polish D (final): obj=%.6f" % obj_f)
        
        print("  Phase 4 done in %.1fs, best obj=%.6f" % (time.monotonic() - t0, global_best))

    # === Fallback ===
    if global_a is None:
        global_a = np.ones(1024, dtype=np.float64)
        global_best = 2.0

    # === Final verification ===
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(1024, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("=== FINAL: evaluate=%.6f, internal_best=%.6f, n=%d ===" % 
          (ver if np.isfinite(ver) else float('inf'), global_best, len(global_a)))

    if not np.isfinite(ver):
        print("  WARNING: rejected, using uniform fallback")
        global_a = np.ones(1024, dtype=np.float64) * 1.0
        final_list = [float(x) for x in global_a]

    return final_list


def max_time_phase1(deadline, t0):
    """Calculate available time for phase 1."""
    return max(60.0, min(600.0, (deadline - time.monotonic()) * 0.35))
