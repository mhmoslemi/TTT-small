```python
def mwu_dual(f, T, eta):
    """Run MWU on the dual LP for T iterations.
    Returns approximate dual weights y.
    """
    n = len(f)
    num_lags = 2 * n - 1
    y = np.ones(num_lags) / num_lags
    
    for t in range(T):
        # Compute z = C^T y (the dual constraint values)
        # z[j] = sum_k C[k,j] * y[k] = sum_k f(k-j) * y[k]
        # This is the cross-correlation of y and f, restricted to j = 0..n-1
        z = compute_Cy(f, y)
        
        # Find the most violated constraint
        j_star = np.argmin(z)
        
        # Update: y_k *= exp(-eta * f(k - j_star))
        # f(k - j_star) is non-zero only for k in [j_star, j_star + n - 1]
        start = max(0, j_star)
        end = min(num_lags, j_star + n)
        if end > start:
            idx = np.arange(start, end)
            f_vals = f[idx - j_star]  # f(k - j_star) for k in [start, end)
            y[idx] *= np.exp(-eta * f_vals)
        
        # Normalize to prevent underflow
        y_max = y.max()
        if y_max < 1e-300:
            y = np.ones(num_lags) / num_lags
        elif y_max > 1e100:
            y /= y_max
    
    return y
```

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: MWU (Multiplicative Weight Update) on the dual LP to get
high-quality subgradient weights, combined with Nesterov-accelerated
projected subgradient, adaptive smoothing, and multi-phase polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def cross_corr_fft(a, b):
    """Compute cross-correlation: result[k] = sum_j a[j+k] * b[j] for k=0..len(a)-1.
    Using FFT: ifft(FFT(a) * conj(FFT(b)))
    """
    n = len(a)
    m = len(b)
    fsize = 1 << (n + m - 2).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    result = np.real(ifft(A * np.conj(B)))
    return result[:n]


def compute_Ct_y(f, y):
    """Compute C^T y where C[k,j] = f(k-j) (zero-padded).
    (C^T y)[j] = sum_k C[k,j] * y[k] = sum_k f(k-j) * y[k]
    = cross-correlation of y and f at lag j.
    Using: cross_corr(y, f)[j] = sum_k y[k] * f[k-j]
    """
    n = len(f)
    # cross_corr(y, f)[j] = sum_k y[k] * f[k-j] (with f zero-padded)
    # This is what we want: (C^T y)[j]
    return cross_corr_fft(y, f)


def mwu_dual_weights(f, T, eta, rng):
    """Run MWU on the dual LP to get approximate dual weights.
    
    Dual: min M * sum(y_k) s.t. (C^T y)[j] >= 1 for all j, y >= 0
    where C[k,j] = f(k-j).
    
    Returns normalized weight vector y (length 2n-1).
    """
    n = len(f)
    num_lags = 2 * n - 1
    M = 1.0  # We'll work with normalized constraints (C^T y >= 1)
    y = np.ones(num_lags) / num_lags
    
    for t in range(T):
        # Compute z[j] = (C^T y)[j]
        z = compute_Ct_y(f, y)
        
        # Find most violated constraint (smallest z[j])
        j_star = np.argmin(z)
        z_min = z[j_star]
        
        # If all constraints are satisfied (z >= 1), we're at a feasible point
        # Scale y to make it feasible: y *= 1 / min(z)
        # But for MWU, we just update based on the most violated constraint
        
        # Update: y[k] *= exp(-eta * f(k - j_star) / M)
        # f(k - j_star) is non-zero for k in [j_star, j_star + n - 1]
        start = max(0, j_star)
        end = min(num_lags, j_star + n)
        if end > start:
            k_idx = np.arange(start, end)
            f_vals = f[k_idx - j_star]
            y[k_idx] *= np.exp(-eta * f_vals)
        
        # Periodic renormalization to prevent underflow/overflow
        if t % 100 == 99:
            y_max = y.max()
            if y_max < 1e-100 or y_max > 1e100:
                y = y / y_max if y_max > 0 else np.ones(num_lags) / num_lags
    
    # Normalize to be a probability vector
    y_sum = y.sum()
    if y_sum < 1e-30:
        y = np.ones(num_lags) / num_lags
    else:
        y = y / y_sum
    
    return y


def run_mwu_subgradient(a_init, n, S, max_time, rng, mwu_T=500, mwu_eta=1.0):
    """Main optimization: MWU-based subgradient with Nesterov acceleration."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_nest = 1.0
    start_time = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    alpha = S / n  # Initial step size
    stall_count = 0
    no_improve = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        
        # Compute current objective
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Run MWU to get dual weights
        if it % 5 == 1 or it == 1:  # Recompute weights every 5 iterations
            T_mwu = min(2000, mwu_T + it // 10)
            mwu_eta = max(0.1, mwu_eta * (0.999 ** (it / 10000)))
            w = mwu_dual_weights(y, T_mwu, mwu_eta, rng)
        
        # Compute subgradient: s = C^T w
        s_grad = compute_Ct_y(y, w)
        gnorm = np.linalg.norm(s_grad)
        if gnorm < 1e-14:
            break
        
        # Normalize gradient
        s_grad = s_grad / gnorm
        
        # Polyak step size (use best_score as upper bound on optimal)
        # alpha_polyak = (cur - lower_bound) / gnorm
        lower_bound = 1.0  # Conservative lower bound on max_conv
        if cur > lower_bound + 1e-6:
            alpha_polyak = (cur - lower_bound) / gnorm
            alpha = min(alpha_polyak, alpha * 2)  # Cap the increase
        else:
            alpha *= 0.95
        
        # Projected gradient step
        x_new = project_simplex(y - alpha * s_grad, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        y = x_new + ((t_nest - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        
        # Ensure y is on the simplex (fix numerical issues)
        y_sum = y.sum()
        if y_sum > 0:
            y = y * (S / y_sum)
        else:
            y = np.ones(n) * S / n
        
        a = x_new
        t_nest = t_new
        
        # Adaptive step size adjustment
        if no_improve > 200:
            alpha *= 0.9
            stall_count += 1
            if stall_count > 5:
                # Perturb and restart momentum
                noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
                y = project_simplex(y + noise, S)
                t_nest = 1.0
                alpha = S / n
                stall_count = 0
                no_improve = 0
        
        # Check time
        if time.monotonic() - start_time > max_time - 1:
            break
    
    return best_local, best_score


def run_smoothed_fista(a_init, n, S, max_time, rng, beta_init=100.0, beta_max=5000.0):
    """Smoothed FISTA with log-sum-exp and backtracking."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t_nest = 1.0
    start_time = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.02 * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = y.copy()
        
        # Log-sum-exp gradient
        c_max = c.max()
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            w = np.full(len(c), 1.0 / len(c))
        else:
            w = exp_bc / total
        
        # Gradient: C^T w
        s_grad = compute_Ct_y(y, w)
        gnorm = np.linalg.norm(s_grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        step = 1.0 / (beta * 2.0)
        x_trial = project_simplex(y - step * s_grad, S)
        
        # Check if it improved (using the smoothed objective as proxy)
        c_trial = fft_convolve_self(x_trial)
        cur_trial = c_trial.max()
        
        if cur_trial < cur:
            x_new = x_trial
        else:
            # Try smaller step
            step *= 0.5
            x_trial = project_simplex(y - step * s_grad, S)
            c_trial = fft_convolve_self(x_trial)
            if c_trial.max() < cur:
                x_new = x_trial
            else:
                x_new = project_simplex(y - step * 0.5 * s_grad, S)
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        y = x_new + ((t_nest - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 0:
            y = y * (S / y_sum)
        else:
            y = np.ones(n) * S / n
        
        a = x_new
        t_nest = t_new
    
    return best_local, best_score


def run_local_polish(a_init, n, S, max_time, rng):

    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n

        it += 1
        c = fft_convolve_self(a)
        cur = c.max()

            best_score = cur
            best_local = a.copy()

        k_star = np.argmax(c)
        
        # Find the coordinates that contribute most to the peak
        # c[k_star] = sum_j a[j] * a[k_star - j]
        j_range = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        contributions = a[j_range] * a[k_star - j_range]
        
        # Try to reduce the peak by moving mass from high-contribution coordinates
        # to low-contribution coordinates

            bottom_contrib = np.argsort(contributions)
            
            for _ in range(10):  # Try a few moves
                if time.monotonic() - start_time > max_time:
                    break
                i = top_contrib[_ % len(top_contrib)]
                j = bottom_contrib[(_ + 3) % len(bottom_contrib)]
                if i == j:

                src = j_range[i]
                dst = j_range[j]
                
                # Move a small amount from src to dst

                a_trial = a.copy()

                a_trial[dst] += delta
                if a_trial[src] < 0:
                    a_trial[src] = 0
                    a_trial[dst] += a_trial[src]  # This won't happen since we checked
                
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur:
                    a = a_trial
                    cur = c_trial.max()
                    if cur < best_score:
                        best_score = cur
                        best_local = a.copy()
                else:
                    # Try moving in the other direction
                    a_trial = a.copy()
                    a_trial[dst] -= delta
                    a_trial[src] += delta
                    if a_trial[dst] >= 0:
                        c_trial = fft_convolve_self(a_trial)
                        if c_trial.max() < cur:
                            a = a_trial
                            cur = c_trial.max()
                            if cur < best_score:
                                best_score = cur
                                best_local = a.copy()
        
        # Also try random pairs
        for _ in range(5):
            if time.monotonic() - start_time > max_time:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:

            delta = rng.uniform(-0.002, 0.002) * S / n
            a_trial = a.copy()
            a_trial[i] += delta
            a_trial[j] -= delta
            if a_trial[i] >= 0 and a_trial[j] >= 0:
                c_trial = fft_convolve_self(a_trial)
                if c_trial.max() < cur:
                    a = a_trial
                    cur = c_trial.max()
                    if cur < best_score:
                        best_score = cur
                        best_local = a.copy()
    
    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    local coordinate descent (polish), multi-restart exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s - 30.0)
    
    global_best = np.inf
    global_a = None
    
    # Get the given sequence
    given_a = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64:
            given_a = np.clip(given, 0, 1000)
    except Exception:
        pass
    
    if given_a is not None:
        n = len(given_a)
    else:
        n = 8192
        given_a = np.ones(n) * 0.5
    
    S = np.sqrt(2.0 * float(n))
    a_start = safe_normalize(given_a, n)
    c_start = fft_convolve_self(a_start)
    initial_score = c_start.max()
    global_best = initial_score
    global_a = a_start.copy()
    obj_start = 2.0 * n * initial_score / (S * S)
    print("Initial: n=%d, score=%.6f, obj=%.6f" % (n, initial_score, obj_start))
    
    # === Phase 1: MWU Subgradient (50% of time) ===
    phase1_time = 0.50 * (deadline - time.monotonic())
    if phase1_time > 20:
        print("Phase 1: MWU Subgradient (%.0f sec)" % phase1_time)
        t0 = time.monotonic()
        a_p1, _ = run_mwu_subgradient(a_start, n, S, phase1_time, rng, mwu_T=500, mwu_eta=1.0)
        a_p1 = safe_normalize(a_p1, n)
        c_p1 = fft_convolve_self(a_p1)
        if c_p1.max() < global_best:
            global_best = c_p1.max()
            global_a = a_p1.copy()
            print("  MWU: %.6f (obj %.6f) [%.0fs]" % (global_best, 2*n*global_best/S**2, time.monotonic()-t0))
        else:
            print("  MWU: no improvement (best %.6f) [%.0fs]" % (global_best, time.monotonic()-t0))
    
    # === Phase 2: Smoothed FISTA (25% of time) ===
    remaining = deadline - time.monotonic()
    phase2_time = 0.25 * remaining
    if phase2_time > 10:
        print("Phase 2: Smoothed FISTA (%.0f sec)" % phase2_time)
        t0 = time.monotonic()
        a_p2, _ = run_smoothed_fista(global_a, n, S, phase2_time, rng, beta_init=200.0, beta_max=10000.0)
        a_p2 = safe_normalize(a_p2, n)
        c_p2 = fft_convolve_self(a_p2)
        if c_p2.max() < global_best:
            global_best = c_p2.max()
            global_a = a_p2.copy()
            print("  FISTA: %.6f (obj %.6f) [%.0fs]" % (global_best, 2*n*global_best/S**2, time.monotonic()-t0))
        else:
            print("  FISTA: no improvement (best %.6f) [%.0fs]" % (global_best, time.monotonic()-t0))
    
    # === Phase 3: Another MWU round with different params (15% of time) ===
    remaining = deadline - time.monotonic()
    phase3_time = 0.15 * remaining
    if phase3_time > 10:
        print("Phase 3: MWU round 2 (%.0f sec)" % phase3_time)
        t0 = time.monotonic()
        a_p3, _ = run_mwu_subgradient(global_a, n, S, phase3_time, rng, mwu_T=1000, mwu_eta=0.5)
        a_p3 = safe_normalize(a_p3, n)
        c_p3 = fft_convolve_self(a_p3)
        if c_p3.max() < global_best:
            global_best = c_p3.max()
            global_a = a_p3.copy()
            print("  MWU2: %.6f (obj %.6f) [%.0fs]" % (global_best, 2*n*global_best/S**2, time.monotonic()-t0))
    
    # === Phase 4: Local polish (10% of time) ===
    remaining = deadline - time.monotonic()
    phase4_time = 0.10 * remaining
    if phase4_time > 5:
        print("Phase 4: Local polish (%.0f sec)" % phase4_time)
        t0 = time.monotonic()
        a_p4, _ = run_local_polish(global_a, n, S, phase4_time, rng)
        a_p4 = safe_normalize(a_p4, n)
        c_p4 = fft_convolve_self(a_p4)
        if c_p4.max() < global_best:
            global_best = c_p4.max()
            global_a = a_p4.copy()
            print("  Polish: %.6f (obj %.6f) [%.0fs]" % (global_best, 2*n*global_best/S**2, time.monotonic()-t0))
    
    # === Phase 5: Multi-restart exploration (remaining time) ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("Phase 5: Multi-restart (%.0f sec)" % remaining)
        n_restarts = 3
        per_restart = remaining * 0.6 / n_restarts
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            # Different perturbation
            if r == 0:
                perturbed = global_a + rng.normal(0, 0.003 * S / np.sqrt(n), n)
            elif r == 1:
                # Smooth perturbation
                smooth = np.convolve(rng.normal(0, 1, n), np.ones(128)/128, mode='same')
                perturbed = global_a + 0.002 * S / np.sqrt(n) * smooth
            else:
                # Multiplicative
                perturbed = global_a * (1.0 + rng.normal(0, 0.005, n))
            
            perturbed = safe_normalize(perturbed, n)
            a_r, _ = run_mwu_subgradient(perturbed, n, S, per_restart, rng, mwu_T=300, mwu_eta=1.0)
            a_r = safe_normalize(a_r, n)
            c_r = fft_convolve_self(a_r)
            if c_r.max() < global_best:
                global_best = c_r.max()
                global_a = a_r.copy()
                print("  Restart %d: %.6f (obj %.6f)" % (r, global_best, 2*n*global_best/S**2))
        
        # Final FISTA on best
        remaining2 = deadline - time.monotonic()
        if remaining2 > 10:
            a_f, _ = run_smoothed_fista(global_a, n, S, remaining2, rng, beta_init=500.0, beta_max=20000.0)
            a_f = safe_normalize(a_f, n)
            c_f = fft_convolve_self(a_f)
            if c_f.max() < global_best:
                global_best = c_f.max()
                global_a = a_f.copy()
                print("  Final FISTA: %.6f (obj %.6f)" % (global_best, 2*n*global_best/S**2))
    
    # === Final output ===
    final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
    final_obj = 2.0 * n * global_best / (S * S)
    print("FINAL: n=%d, score=%.6f, obj=%.6f" % (n, global_best, final_obj))
    
    # Verify
    ver = evaluate_sequence(final_list)
