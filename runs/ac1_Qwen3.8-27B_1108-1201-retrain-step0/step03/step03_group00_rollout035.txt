def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses multi-phase FISTA with adaptive LSE smoothing, exact subgradient polishing,
and structured initializations. Key improvements over baseline:
- Adaptive step size with backtracking

import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(conv(a,a))."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_subgradient(a, c, k):
    """Averaged subgradient from top-k active constraints."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    w_full = np.zeros(len(c))
    w_full[idx] = 1.0 / k
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def objective_val(a, n, S):
    """Compute objective value efficiently."""
    b = fft_convolve_self(a)
    return 2.0 * n * b.max() / (S * S)


def fista_with_backtrack(a_init, n, S, time_limit, rng, beta_init=10.0,
                          beta_max=500.0, beta_rate=0.003, mode="lse"):
    """FISTA with adaptive beta and backtracking line search."""
    a = normalize_sum(a_init, n)
    y = a.copy()
    t_k = 1.0
    start = time.monotonic()
    best_obj = objective_val(a, n, S)
    best_a = a.copy()
    last_improve_time = start
    it = 0
    step_size = 1.0 / (2.0 * beta_init)
    
    while time.monotonic() - start < time_limit:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = y.copy()
            last_improve_time = time.monotonic()
        
        # Stuck detection: restart with perturbation
        if time.monotonic() - last_improve_time > 20:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t_k = 1.0
            step_size = 1.0 / (2.0 * beta)
            last_improve_time = time.monotonic()
            continue
        
        # Compute gradient
        if mode == "lse":
            grad = lse_gradient(y, c, beta)
        elif mode == "topk":
            k = min(64, 8 + it // 50)
            grad = topk_subgradient(y, c, k)
        else:  # hybrid
            if it % 3 == 0:
                grad = topk_subgradient(y, c, min(32, 4 + it // 100))
            else:
                grad = lse_gradient(y, c, beta)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        step = 1.0 / (2.0 * beta) if mode != "topk" else 2.0 * S / (n * gnorm)
        step = min(step, 1.0)
        
        # Backtracking: reduce step if function increases
        for _ in range(5):
            x_new = project_simplex(y - step * grad, S)
            c_new = fft_convolve_self(x_new)
            obj_new = 2.0 * n * c_new.max() / (S * S)
            if obj_new <= cur_obj + 1e-10 or step < 1e-10:
                break
            step *= 0.5
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_k = t_new
    
    # Final evaluation
    c_final = fft_convolve_self(a)
    obj_final = 2.0 * n * c_final.max() / (S * S)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a.copy()
    
    return best_a, best_obj


def polish_subgradient(a_init, n, S, time_limit, rng):
    """Polish using exact subgradient with decreasing step."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best_obj = objective_val(a, n, S)
    best_a = a.copy()
    it = 0
    
    while time.monotonic() - start < time_limit:
        it += 1

        cur_obj = 2.0 * n * c.max() / (S * S)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Use top-k subgradient with increasing k
        k = min(128, 4 + it // 10)
        grad = topk_subgradient(a, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Step size: decreasing over iterations

        step = min(step, 4.0 * S / (n * gnorm + 1e-10))
        step = max(step, 1e-8)
        
        a_new = project_simplex(a - step * grad, S)
        
        # Check improvement; if not, try smaller step

        obj_new = 2.0 * n * c_new.max() / (S * S)

                step *= 0.5
                a_new = project_simplex(a - step * grad, S)

        a = a_new
