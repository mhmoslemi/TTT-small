"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Hybrid FISTA + LP-based response oracle, with progressive
polishing and multi-length exploration. Starts from the given sequence
and uses the LP response oracle (from the prior SOTA) to make periodic
jumps, combined with FISTA for smooth convergence.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog

def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def project_simplex(v, S):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def normalize_sum(a, n):
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta):
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]

def run_fista(a_init, n, S, max_time, rng, beta_init=10.0, beta_rate=0.005, beta_max=3000.0):
    a = normalize_sum(a_init, n)
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * np.exp(beta_rate * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best:
            best = cur
            best_a = y.copy()
            last_improve = it
        
        grad = lse_gradient(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = 1.0 / (2.0 * beta)
        
        if it - last_improve > 3000:
            step *= 0.3
            noise_scale = 0.002 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            t = 1.0
            last_improve = it
        
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        if it % 5000 == 0:
            decay = np.exp(-it / 30000.0)
            noise_scale = 0.002 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
    
    c_final = fft_convolve_self(a)
    if c_final.max() < best:
        best = c_final.max()
        best_a = a.copy()
    
    return best_a, best

def solve_lp_response(f, M, n):
    """
    Solve: maximize sum(g) subject to g >= 0, (f*g)[k] <= M for all k.
    Returns the optimal g.
    """
    # Constraint matrix: for each k, the row is the coefficients of g in (f*g)[k]
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
    # Coefficient of g[m] in (f*g)[k] is f[k-m] if 0 <= k-m < n, else 0
    
    # Build the constraint matrix A (shape: (2n-1, n))
    # A[k, m] = f[k-m] if 0 <= k-m < n, else 0
    
    # This is a "convolution matrix". We can build it efficiently.
    # For each k, the non-zero entries are at positions m where 0 <= k-m < n
    # i.e., max(0, k-n+1) <= m <= min(k, n-1)
    
    # For efficiency, use sparse matrix
    from scipy.sparse import csr_matrix
    
    # Build row by row
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for m in range(lo, hi + 1):
            rows.append(k)
            cols.append(m)
            vals.append(f[k - m])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: A @ g <= M (for each k)
    b_vec = np.full(2 * n - 1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
    
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    else:
        # Fallback: return f itself
        return f.copy()

    # The inner product <g_k, g_l> = 4 * sum_j a[k-j] * a[l-j] (for valid j)

    # Let m = l - k. Then sum_j a[k-j] * a[l-j] = sum_j a[k-j] * a[k-j+m]
    # This is the autocorrelation of a at lag m, but with a different index range.
    # For the linear autocorrelation, (a*a)[l-k+n-1] = sum_j a[j] * a[l-k+n-1-j]

        lo = max(0, k - n + 1)
        hi = min(k, n - 1)

    # This is: minimize lambda^T G lambda subject to sum lambda = 1, lambda >= 0
    # where G[i,j] = <g_i, g_j>
    
    G = grads @ grads.T  # K x K matrix
    
    # Solve the QP
    # For small K, we can use a simple approach
    if K <= 1:
        grad = grads[0]
    else:

        # Actually, the KKT conditions are: 2*G*lambda + mu*1 = 0, sum(lambda) = 1, lambda >= 0
        # => G*lambda = -mu/2 * 1
        # => lambda = -mu/2 * G^{-1} * 1
        # With sum(lambda) = 1: -mu/2 * sum(G^{-1} * 1) = 1 => mu = -2 / sum(G^{-1} * 1)
        # => lambda = 1 / sum(G^{-1} * 1) * G^{-1} * 1
        # But we also need lambda >= 0. If any component is negative, we need to solve the full QP.

                    grad = lam @ grads
                else:

                    def qf(l):
                        return l @ G @ l

                                   bounds=[(0, 1)] * K, method='SLSQP')

Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy: Hybrid bundle/FISTA method with steepest subgradient computation,
LP-based response oracle on coarser scale for diversity jumps, adaptive
beta scheduling, and backtracking step-size. Extends prior SOTA by using
exact active-set geometry rather than Lse approximation alone.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def lse_gradient(a, c, beta):
    """Gradient of LSE approximation of max(c) w.r.t. a."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def compute_subgradient_at_lag(a, k):
    """Subgradient of max_k (a*a)[k] at a specific lag k."""
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    grad[lo:hi + 1] = 2.0 * a[k - (lo:hi + 1)]
    return grad


def steepest_subgradient(a, c, rel_tol=1e-4):
    """Compute steepest subgradient of max(c) w.r.t. a using active set."""
    n = len(a)
    M = c.max()
    # Active lags: c[k] >= M - rel_tol * M
    threshold = M - rel_tol * M
    active = np.where(c >= threshold)[0]

    if len(active) == 0:
        active = np.array([np.argmax(c)])

    if len(active) == 1:
        return compute_subgradient_at_lag(a, int(active[0]))

    # Cap active set size for efficiency
    if len(active) > 50:
        # Keep only the top 50
        active_sorted = active[np.argsort(c[active])[::-1]][:50]
        active = active_sorted

    K = len(active)
    # Compute gradients for each active lag
    grads = np.zeros((K, n))
    for i, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        grads[i, lo:hi + 1] = 2.0 * a[k - (lo:hi + 1)]

    # Gram matrix G[i,j] = <g_i, g_j>
    G = grads @ grads.T
    # Add small regularization for numerical stability
    G += 1e-12 * np.eye(K)

    # Solve: minimize lambda^T G lambda s.t. sum(lambda)=1, lambda>=0
    # Try unconstrained first
    try:
        G_inv_1 = np.linalg.solve(G, np.ones(K))
        s = G_inv_1.sum()
        if s > 1e-10:
            lam = G_inv_1 / s
            if np.all(lam >= -1e-8):
                lam = np.maximum(lam, 0.0)
                lam_sum = lam.sum()
                if lam_sum > 1e-10:
                    lam /= lam_sum
                    return lam @ grads
    except np.linalg.LinAlgError:
        pass

    # Fallback: uniform average
    return np.mean(grads, axis=0)

def downsample_avg(a, factor):
    """Downsample by averaging blocks of given factor."""
    n = len(a)
    n_small = n // factor
    a_trimmed = a[:n_small * factor]
    return a_trimmed.reshape(n_small, factor).mean(axis=1)


def upsample_interp(a_small, n):
    """Upsample via linear interpolation."""
    if len(a_small) == n:
        return a_small.copy()
    t_old = np.linspace(0, 1, len(a_small))
    t_new = np.linspace(0, 1, n)
    return np.interp(t_new, t_old, a_small)


def solve_lp_response(f, M, n):
    """
    Solve: maximize sum(g) s.t. g>=0, (f*g)[k] <= M for all k.
    Uses scipy linprog with HiGHS solver.
    Only feasible for moderate n (up to ~4096).
    """
    # Build constraint matrix: A[k, m] = f[k-m] if 0 <= k-m < n
    # Use sparse construction
    rows = []
    cols = []
    vals = []

    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for m in range(lo, hi + 1):
            rows.append(k)
            cols.append(m)
            vals.append(f[k - m])

    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds,
                  method='highs', options={'presolve': True})

    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return f.copy()

        if time.monotonic() - start + 60 > max_time:
            break

        g_coarse = solve_lp_response(a_coarse, M, n_coarse)
        t_lp_elapsed = time.monotonic() - t_lp_start
        print(f"    LP iter {it}: M={M:.6f}, solve time={t_lp_elapsed:.1f}s")

        if g_coarse.sum() > 1e-10:
            g_coarse = normalize_sum(g_coarse, n_coarse)
            g_fine = normalize_sum(upsample_interp(g_coarse, n), n)

            # Mix
            a_new = (1.0 - mix_frac) * a_coarse_full + mix_frac * g_fine if 'a_coarse_full' in dir() else (1.0 - mix_frac) * best_a + mix_frac * g_fine
            a_new = normalize_sum(a_new, n)
            c_new = fft_convolve_self(a_new)
            if c_new.max() < best:
                best = c_new.max()
                best_a = a_new.copy()
                print(f"    Improved: {best:.6f}")

            # Quick FISTA polish after LP jump
            remaining = max_time - (time.monotonic() - start)
            if remaining > 10:
                t_polish = min(20.0, remaining * 0.3)
                a_pol, s_pol = run_fista_bundle(best_a, n, S, t_polish, rng,
                                                 beta_init=100, beta_rate=0.004,
                                                 beta_max=5000, mode="auto")
                if s_pol < best:
                    best = s_pol
                    best_a = a_pol.copy()

        a_coarse_full = best_a  # Track for next iteration

    return best_a, best


def run_local_search(a_init, n, S, max_time, rng):
    """Random perturbation local search."""
    a = normalize_sum(a_init, n)
    start = time.monotonic()
    best = fft_convolve_self(a).max()
    best_a = a.copy()

    c = fft_convolve_self(a)
    M = c.max()
    # Focus perturbations on active lags
    active = np.where(c >= M - 0.01 * M)[0]

    while time.monotonic() - start < max_time:
        # Random perturbation
        scale = 0.01 * S / np.sqrt(n)
        a_pert = a + rng.normal(0, scale, n)
        a_pert = project_simplex(a_pert, S)

        c_pert = fft_convolve_self(a_pert)
        if c_pert.max() < best:
            best = c_pert.max()
            best_a = a_pert.copy()
            a = a_pert.copy()
            # Recompute active set
            M = c_pert.max()
            active = np.where(c_pert >= M - 0.01 * M)[0]
        else:
            # Simulated annealing: accept with some probability
            if rng.random() < 0.1:
                a = a_pert.copy()

    return best_a, best


def generate_diverse_inits(n, rng):
    """Generate diverse initializations for exploration."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)

    # Window functions
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.exp(-6.0 * (tt - 0.5) ** 2))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.5, n)))

    # Block structures
    for nblock in [3, 5, 7]:
        block_len = n // nblock
        heights = rng.uniform(0.3, 1.5, nblock)
        block_seq = np.repeat(heights, block_len)
        if len(block_seq) < n:
            block_seq = np.concatenate([block_seq, np.full(n - len(block_seq), heights[-1])])
        inits.append(block_seq[:n])

    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses hybrid FISTA/bundle method with LP-based jumps.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s) - 20.0

    global_best = np.inf
    global_a = None

    # Get initial sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    if given is not None:
        n = len(given)
    else:
        n = 8192

    S = np.sqrt(2.0 * n)
    print(f"Starting: n={n}, budget={budget_s:.0f}s")

    # Initialize from given or random
    if given is not None:
        a = normalize_sum(given, n)
    else:
        a = normalize_sum(np.ones(n) * (1 + 0.1 * rng.standard_normal(n)), n)

    # Evaluate initial
    c0 = fft_convolve_self(a)
    global_best = c0.max()
    global_a = a.copy()
    print(f"Initial: max_conv={global_best:.6f}, objective={2*n*global_best/S**2:.6f}")

    # Phase 1: FISTA exploration (40% of time)
    t1 = max(60.0, (deadline - time.monotonic()) * 0.40)
    print(f"=== Phase 1: FISTA ({t1:.0f}s) ===")
    p1_start = time.monotonic()

    a1, s1 = run_fista_bundle(a, n, S, t1 * 0.6, rng,
                               beta_init=5.0, beta_rate=0.005, beta_max=2000,
                               mode="lse")
    if s1 < global_best:
        global_best = s1
        global_a = a1.copy()
        print(f"  After FISTA-LSE: {global_best:.6f}")

    # Phase 1b: Bundle method (remaining Phase 1 time)
    t1b = deadline - time.monotonic() - (deadline - p1_start) * 0.4
    t1b = max(30.0, t1b * 0.6)
    a1b, s1b = run_fista_bundle(global_a, n, S, t1b, rng,
                                 beta_init=100.0, beta_rate=0.003, beta_max=5000,
                                 mode="bundle")
    if s1b < global_best:
        global_best = s1b
        global_a = a1b.copy()
        print(f"  After bundle: {global_best:.6f}")

    # Phase 2: LP-based polishing (30% of time)
    remaining = deadline - time.monotonic()
    t2 = max(60.0, remaining * 0.30)
    print(f"=== Phase 2: LP polish ({t2:.0f}s) ===")

    if n <= 4096:
        # Can do LP at full scale
        a2, s2 = run_fista_bundle(global_a, n, S, t2 * 0.4, rng,
                                   beta_init=200, beta_rate=0.002, beta_max=8000,
                                   mode="auto")
        if s2 < global_best:
            global_best = s2
            global_a = a2.copy()

        # LP at full scale if small enough
        if n <= 2048 and (deadline - time.monotonic()) > 30:
            t2b = min(60.0, deadline - time.monotonic() - 30)
            a2b, s2b = run_lp_polish(global_a, n, S, t2b, rng)
            if s2b < global_best:
                global_best = s2b
                global_a = a2b.copy()
                print(f"  After full LP: {global_best:.6f}")
    else:
        # Coarse-scale LP for large n
        n_coarse = min(1024, n // 4)
        a_coarse = downsample_avg(global_a, n // n_coarse)
        S_coarse = np.sqrt(2.0 * n_coarse)
        a_coarse = normalize_sum(a_coarse, n_coarse)

        # Optimize coarse with FISTA
        t2a = min(60.0, t2 * 0.3)
        a_coarse_opt, s_coarse = run_fista_bundle(a_coarse, n_coarse, S_coarse, t2a, rng,
                                                   beta_init=10, beta_rate=0.01,
                                                   beta_max=3000, mode="auto")
        # Upsample and use as guide
        guide = normalize_sum(upsample_interp(a_coarse_opt, n), n)
        # Mix guide with current best
        mix = 0.3
        a_mixed = (1 - mix) * global_a + mix * guide
        a_mixed = normalize_sum(a_mixed, n)
        c_mixed = fft_convolve_self(a_mixed)
        if c_mixed.max() < global_best:
            global_best = c_mixed.max()
            global_a = a_mixed.copy()
            print(f"  After coarse guide mix: {global_best:.6f}")

        # LP on coarse scale
        t2c = min(120.0, deadline - time.monotonic() - 60)
        if t2c > 10:
            a_lp, s_lp = run_lp_polish(global_a, n, S, t2c, rng)
            if s_lp < global_best:
                global_best = s_lp
                global_a = a_lp.copy()
                print(f"  After LP polish: {global_best:.6f}")

    # Phase 3: Final aggressive FISTA (20% of time)
    remaining = deadline - time.monotonic()
    t3 = max(30.0, remaining * 0.50)
    print(f"=== Phase 3: Final polish ({t3:.0f}s) ===")

    a3, s3 = run_fista_bundle(global_a, n, S, t3 * 0.5, rng,
                               beta_init=500, beta_rate=0.001, beta_max=20000,
                               mode="lse")
    if s3 < global_best:
        global_best = s3
        global_a = a3.copy()
        print(f"  After high-beta LSE: {global_best:.6f}")

    # Phase 3b: Bundle method final
    t3b = min(60.0, deadline - time.monotonic() - 30)
    if t3b > 5:
        a3b, s3b = run_fista_bundle(global_a, n, S, t3b, rng,
                                     beta_init=1000, beta_rate=0.0005, beta_max=50000,
                                     mode="bundle")
        if s3b < global_best:
            global_best = s3b
            global_a = a3b.copy()
            print(f"  After final bundle: {global_best:.6f}")

    # Phase 4: Local search (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 10:
        print(f"=== Phase 4: Local search ({remaining:.0f}s) ===")
        a4, s4 = run_local_search(global_a, n, S, min(remaining * 0.8, 60.0), rng)
        if s4 < global_best:
            global_best = s4
            global_a = a4.copy()
            print(f"  After local search: {global_best:.6f}")

    # Final verification
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        obj_internal = 2.0 * len(global_a) * global_best / (np.sum(np.clip(global_a, 0, 1000)) ** 2)
        print(f"Final: internal={obj_internal:.6f}, verify={ver:.6f}" if np.isfinite(ver)
              else f"Final: internal={obj_internal:.6f}, verify=inf")
        if not np.isfinite(ver):
            print("  Fallback to uniform")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    final_obj = 2.0 * len(global_a) * fft_convolve_self(global_a).max() / global_a.sum() ** 2
    print(f"Returning n={len(global_a)}, objective={final_obj:.6f}")
    return [float(x) for x in global_a]
