"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-guided optimization.
Key idea: alternate between solving an LP (maximize sum(g) s.t. f*g <= M, g>=0)
and updating f via convex combination. Polish with subgradient and coordinate descent.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns the optimal g vector."""
    n = len(f)
    
    # Build sparse constraint matrix A such that A @ g <= M * ones
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
    # For row k: A[k, j] = f[j] if 0 <= k-j <= n-1, else 0
    # i.e., A[k, j] = f[j] if max(0, k-n+1) <= j <= min(n-1, k)
    
    rows = []
    cols = []
    data = []
    
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[j])
    
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=(b * 0 + M), bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        # Fallback: return uniform
        return np.ones(n) / n


def lp_iteration(f, M, t=1.0):
    """One LP iteration: solve LP and update f."""
    n = len(f)
    S = f.sum()
    
    g = solve_lp_direction(f, M)
    g = g * (S / g.sum())  # Rescale to same sum
    
    f_new = (1 - t) * f + t * g
    f_new = np.maximum(f_new, 0)
    f_new = f_new * (S / f_new.sum())
    
    return f_new


def lp_optimize(f, time_budget, t=1.0, verbose=True):
    """Run LP-based optimization."""
    n = len(f)
    S = f.sum()
    f = f.copy().astype(np.float64)
    f = normalize_nonneg(f, S)
    
    c = fft_convolve_self(f)
    M = c.max()
    best_M = M
    best_f = f.copy()
    
    start = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        t_iter = time.monotonic() - start
        
        c = fft_convolve_self(f)
        M = c.max()
        
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and it % 10 == 0:
            print("  LP iter %d: M=%.8f, t=%.0fs, stall=%d" % (it, M, t_iter, stall))
        
        if stall > 20:
            # Restart with smaller t or perturbation
            f = normalize_nonneg(best_f + 0.01 * np.random.default_rng(it).standard_normal(n), S)
            stall = 0
            continue
        
        try:
            f_new = lp_iteration(f, M, t=t)
        except Exception as e:
            if verbose:
                print("  LP failed: " + str(e))
            break
        
        if not np.all(np.isfinite(f_new)):
            break
        
        f = f_new
    
    return best_f, best_M


def subgradient_polish(f, time_budget, T_init=50, T_final=1000, lr=0.05, verbose=False):
    """Weighted subgradient polish."""
    n = len(f)
    S = f.sum()
    f = f.copy().astype(np.float64)
    f = normalize_nonneg(f, S)
    
    c = fft_convolve_self(f)
    best_M = c.max()
    best_f = f.copy()
    
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_budget:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** progress
        
        c = fft_convolve_self(f)
        M = c.max()
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
        
        # Compute weighted gradient
        c_max = c.max()
        if c_max < 1e-30:
            break
        scaled = T * c / c_max
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        
        # Gradient: g[m] = 2 * sum_k w[k] * a[k-m]
        fsize = 1 << (3 * n - 2).bit_length()
        W = fft(w, fsize)
        A = fft(f, fsize)
        corr = np.real(ifft(W * np.conj(A)))
        grad = 2.0 * corr[:n]
        
        # Normalize and update
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Backtracking
        lr_cur = lr * (1.0 - 0.5 * progress)
        best_f_trial = f.copy()
        best_obj_trial = M
        for i in range(10):
            f_trial = f - lr_cur * grad
            f_trial = normalize_nonneg(f_trial, S)
            c_trial = fft_convolve_self(f_trial)
            obj_trial = c_trial.max()
            if obj_trial < M - 1e-10 * M:
                best_f_trial = f_trial
                best_obj_trial = obj_trial
                break
            lr_cur *= 0.5
        
        f = best_f_trial
    
    return best_f, best_M


def coordinate_descent(f, time_budget, max_sweeps=20):
    """Coordinate descent polish."""
    n = len(f)
    S = f.sum()
    f = f.copy().astype(np.float64)
    f = normalize_nonneg(f, S)
    
    c = fft_convolve_self(f)
    best_M = c.max()
    best_f = f.copy()
    
    start = time.monotonic()
    rng = np.random.default_rng(42)
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            orig = f[i]
            best_local = orig
            best_local_M = best_M
            
            for factor in [0.5, 0.75, 1.33, 1.5, 0.25, 2.0, 0.6, 1.4, 0.8, 1.25]:
                f[i] = orig * factor
                f_norm = normalize_nonneg(f, S)
                c_check = fft_convolve_self(f_norm)
                if c_check.max() < best_local_M - 1e-12:
                    best_local_M = c_check.max()
                    best_local = orig * factor
            
            f[i] = best_local
            f = normalize_nonneg(f, S)
            c = fft_convolve_self(f)
            if c.max() < best_M - 1e-12:
                best_M = c.max()
                best_f = f.copy()
                improved = True
        
        if not improved:
            break
    
    return best_f, best_M
