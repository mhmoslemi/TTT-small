"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS on power-mean objective
with spectral equalization, peak-chasing, and coordinate descent polish.

Key innovations over previous approach:
1. L-BFGS (second-order) for much faster convergence on smooth power-mean
2. Adaptive power p: start low (broad attack) then increase (peak focus)
3. Spectral equalization as periodic re-initialization within optimization
4. Exact peak subgradient for final polish
5. Multi-peak combined subgradient
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def power_mean_value(a, p):
    """Compute (mean(c^p))^(1/p) for the self-convolution of a."""
    c = fft_convolve_self(a)
    return (np.mean(c ** p)) ** (1.0 / p)


def compute_power_mean_grad(a, p):
    """Compute gradient of (mean(c^p))^(1/p) w.r.t. a using FFT."""
    n = len(a)
    c = fft_convolve_self(a)
    K = 2 * n - 1
    cp = c ** p
    mean_cp = np.mean(cp)
    if mean_cp < 1e-30:
        return np.zeros(n), c
    prefactor = 2.0 * (mean_cp ** (1.0 / p - 1.0)) / K
    d = c ** (p - 1)
    a_rev = a[::-1]
    N = 1 << (3 * n - 2).bit_length()
    A_rev_f = fft(a_rev, N)
    D_f = fft(d, N)
    conv = np.real(ifft(A_rev_f * D_f))
    grad = conv[n - 1:2 * n - 1] * prefactor
    return grad, c


def log_param_grad(theta, a, S, p):
    """Compute gradient of power-mean w.r.t. log-parameters theta.
    a = softmax(theta) * S (i.e., a_j = exp(theta_j) / sum(exp) * S)
    """
    n = len(a)
    grad_a, c = compute_power_mean_grad(a, p)
    # Chain rule: dL/dtheta_k = a_k * grad_a_k - (a_k/S) * sum_j a_j * grad_a_j
    dot = np.dot(a, grad_a)
    grad_theta = a * grad_a - (a / S) * dot
    return grad_theta


def spectral_equalize(a, beta=0.3, n_iter=1):
    """Apply spectral equalization: compress the power spectrum toward uniform."""
    n = len(a)
    S = a.sum()
    for _ in range(n_iter):
        N = 1 << (2 * n - 1).bit_length()  # at least 2n-1
        A = fft(a, N)
        P = np.abs(A) ** 2
        # Keep DC component, equalize the rest
        P_mean = P[1:].mean()
        P_new = P.copy()
        P_new[1:] = (1 - beta) * P[1:] + beta * P_mean
        # Also boost the very low ones
        P_new = np.maximum(P_new, 1e-10)
        mag_new = np.sqrt(P_new)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new))[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new * (S / s)
        else:
            a_new = a * 1.0
        a = a_new
    return a


def peak_chase(a, S, n_peaks=5, max_iters=200):
    """Chase the top peaks using exact subgradient."""
    n = len(a)
    a = a.copy()
    c = fft_convolve_self(a)
    best_obj = c.max()
    
    for it in range(max_iters):
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_obj:
            best_obj = cur_max
        
        # Find top peaks
        peak_threshold = cur_max * 0.95
        peak_indices = np.where(c >= peak_threshold)[0]
        
        if len(peak_indices) == 0:
            break
        
        # Compute combined subgradient
        g = np.zeros(n)
        for k in peak_indices:
            # dc[k]/da[m] = 2*a[k-m] if 0 <= k-m < n
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            g[valid] += 2.0 * a[idx[valid]]
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g_dir = -g / gnorm
        
        # Backtracking line search
        alpha = 0.1
        improved = False
        for _ in range(20):
            a_trial = a + alpha * g_dir
            a_trial = normalize_nonneg(a_trial, S)
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < cur_max * (1 - 1e-10):
                a = a_trial
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            break
    
    return a


def coordinate_descent(a, S, max_sweeps=10):
    """Coordinate descent with golden section search on each coordinate."""
    n = len(a)
    a = a.copy()
    c = fft_convolve_self(a)
    best_obj = c.max()
    RNG = np.random.default_rng(42)
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            if a[i] < 1e-12:
                continue
            # Golden section search around current value
            orig = a[i]
            lo = 0.0
            hi = orig * 3.0
            
            # Quick check: try a few values
            best_val = orig
            best_f = best_obj
            
            for factor in [0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5]:
                a[i] = orig * factor
                c_check = fft_convolve_self(a)
                f = c_check.max()
                if f < best_f:
                    best_f = f
                    best_val = orig * factor
            
            if best_val != orig:
                a[i] = best_val
                a = normalize_nonneg(a, S)
                improved = True
            else:
                a[i] = orig
        
        if not improved:
            break
    
    a = normalize_nonneg(a, S)
    c = fft_convolve_self(a)
    return a, c.max()


def run_lbfgs_on_logparam(theta, a, S, p, maxiter=200, tol=1e-10):
    """Run L-BFGS on log-parameters for power-mean objective."""
    n = len(a)
    S_val = S
    
    def objective_and_grad(theta_flat):
        # Convert theta to a
        theta_shifted = theta_flat - theta_flat.max()
        exp_theta = np.exp(theta_shifted)
        total = exp_theta.sum()
        if total < 1e-300:
            return 1e10, np.zeros(n)
        a_loc = S_val * exp_theta / total
        # Compute objective and gradient
        val = power_mean_value(a_loc, p)
        # Gradient w.r.t. theta
        grad_a, _ = compute_power_mean_grad(a_loc, p)
        dot = np.dot(a_loc, grad_a)
        grad_theta = a_loc * grad_a - (a_loc / S_val) * dot
        return val, grad_theta
    
    result = minimize(
        objective_and_grad,
        theta,
        jac=True,
        method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
    )
    
    # Convert back
    theta_opt = result.x
    theta_shifted = theta_opt - theta_opt.max()
    exp_theta = np.exp(theta_shifted)
    total = exp_theta.sum()
    a_opt = S_val * exp_theta / total
    a_opt = normalize_nonneg(a_opt, S_val)
    
    return a_opt
