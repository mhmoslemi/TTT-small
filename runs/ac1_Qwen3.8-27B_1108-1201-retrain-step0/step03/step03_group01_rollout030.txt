"""
Minimize 2*n*max_conv/sum^2 using spectral flattening + L-BFGS-B log-sum-exp + Nesterov polish.

Key innovations:
1. Spectral flattening: explicitly target uniform |FFT(p)| to minimize conv peak.
2. L-BFGS-B with softmax parameterization and log-sum-exp smooth approximation.

4. Adaptive beta schedule for the smooth approximation.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def spectral_flatten(a, N, eta=0.1, n_iter=1000, rng=None, target_flat=1.0):
    """
    Iteratively flatten the power spectrum of a.
    Target: |P[m]| ≈ target_flat for all m >= 1.
    """
    n = len(a)
    S = a.sum()
    best_a = a.copy()
    best_score = np.inf
    
    p = a / S  # work with probability vector
    
    for it in range(n_iter):
        p_pad = np.zeros(N)
        p_pad[:n] = p
        P = fft(p_pad)
        mag2 = np.abs(P) ** 2
        
        # Check current peak
        c = np.real(ifft(mag2))[:2*n-1]
        c_max = c.max()
        score = 2 * n * c_max  # since sum(p) = 1
        if score < best_score:
            best_score = score
            best_a = (p * S).copy()
        
        # Compute target magnitude
        # Ideal: |P[m]| = 1 for all m (since sum |P|^2 = N and |P[0]| = 1)
        # Actually sum |P[m]|^2 = N * sum(p^2) by Parseval for the zero-padded signal
        # Let's just target |P[m]| = sqrt((N-1)/(N-1)) = 1 for m >= 1
        # But we need to be careful: N * mean(|P|^2) = sum |P|^2
        # sum |P[m]|^2 = N * (sum_k c[k]) = N * 1 = N (since sum p = 1)
        # |P[0]|^2 = 1
        # So sum_{m>=1} |P[m]|^2 = N - 1
        # Target: |P[m]| = sqrt((N-1)/(N-1)) = 1 for m >= 1
        
        target_mag = np.ones(N)
        target_mag[0] = 1.0  # P[0] = 1 (DC component)
        
        # Linear interpolation of magnitudes
        mag = np.sqrt(mag2)
        mag_new = (1 - eta) * mag + eta * target_mag
        
        # Preserve phase
        phase = np.angle(P)
        P_new = mag_new * np.exp(1j * phase)
        
        # Ensure P[0] is real and = 1
        P_new[0] = 1.0
        
        # Inverse FFT
        p_new_full = np.real(ifft(P_new))[:n]
        
        # Project to simplex
        p_new = project_simplex(p_new_full, 1.0)
        
        # Check if projection is valid
        if p_new.sum() < 0.99:
            # Projection failed, reduce step
            p = p  # keep old
        else:
            p = p_new / p_new.sum()  # renormalize
        
        # Adaptive eta
        if it > 100 and it % 100 == 0:
            eta *= 0.95  # slowly decrease
        
        # Occasional noise for diversification
        if rng is not None and it % 500 == 250 and rng.random() < 0.2:
            p = project_simplex(p + rng.normal(0, 0.001 * np.sqrt(1.0/n), n), 1.0)
    
    return best_a, best_score

def log_sum_exp_objective(x, beta, n, N):
    """
    Objective and gradient for L-BFGS-B.
    x is the log-space variable, p = softmax(x).
    Minimizes (1/beta) * log(sum exp(beta * c[k])) where c = convolve(p,p).
    """
    # Softmax
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    Z = exp_x.sum()
    p = exp_x / Z
    
    # Convolution
    p_pad = np.zeros(N)
    p_pad[:n] = p
    P = fft(p_pad)
    mag2 = np.abs(P) ** 2
    c = np.real(ifft(mag2))[:2*n-1]
    
    # Log-sum-exp
    c_max = c.max()
    beta_c = beta * (c - c_max)
    # Clip for numerical stability
    beta_c = np.clip(beta_c, -500, 500)
    exp_bc = np.exp(beta_c)
    log_sum = c_max + np.log(exp_bc.sum()) / beta
    
    # Gradient w.r.t. p
    w = exp_bc / exp_bc.sum()  # weights for the log-sum-exp
    
    # grad c[k] / grad p[i] = p[k-i] (for valid k-i)
    # So grad_p[i] = 2 * sum_k w[k] * p[k-i] = 2 * (w * p)[i] (cross-correlation)
    # Using FFT: cross_correlation(w, p) = Ifft(fft(w) * conj(fft(p)))
    fsize = N  # same as before
    W = fft(w, fsize)
    grad_p = 2.0 * np.real(ifft(W * P))[:n]
    
    # Gradient w.r.t. x: grad_x[j] = p[j] * (grad_p[j] - sum_i grad_p[i] * p[i])
    grad_p_mean = (grad_p * p).sum()
    grad_x = p * (grad_p - grad_p_mean)
    
    return log_sum, grad_x

def lbfgsb_optimize(a_init, n, N, max_time, beta_schedule=None, rng=None):
    """
    L-BFGS-B optimization with softmax parameterization.
    """
    S = a_init.sum()
    p = a_init / S
    
    # Initialize x from p (x = log(p) approximately)
    x0 = np.log(p + 1e-30)
    
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500]
    
    best_a = a_init.copy()
    best_score = np.inf
    start = time.monotonic()
    
    for beta in beta_schedule:
        if time.monotonic() - start > max_time:
            break
        
        beta = float(beta)
        
        def fun(x):
            val, _ = log_sum_exp_objective(x, beta, n, N)
            return val
        
        def grad(x):
            _, g = log_sum_exp_objective(x, beta, n, N)
            return g
        
        # Run L-BFGS-B
        try:
            res = scipy_minimize(fun, x0, jac=grad, method='L-BFGS-B',
                               options={'maxiter': 500, 'gtol': 1e-8, 'maxls': 50})
            
            # Convert back to p
            x_opt = res.x
            x_shifted = x_opt - x_opt.max()
            exp_x = np.exp(x_shifted)
            p_opt = exp_x / exp_x.sum()
            p_opt = np.clip(p_opt, 0, 1000)
            p_opt = p_opt / p_opt.sum()
            
            a_opt = p_opt * S
            c = fft_convolve_self(a_opt)
            score = 2 * n * c.max() / (a_opt.sum() ** 2)
            
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
            
            x0 = x_opt  # warm start for next beta
        except Exception as e:
            print("  L-BFGS-B error: %s" % str(e))
            break
    
    return best_a, best_score

def nesterov_toppk(a_init, n, S, max_time, rng, top_k=50):
    """Nesterov-accelerated top-k subgradient."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    
    N = 1 << (2 * n - 1).bit_length()
    
    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        score = 2 * n * cur / (y.sum() ** 2)
        if score < best_score:
            best_score = score
            best_local = y.copy()
        
        # Top-k subgradient
        top_k_curr = min(top_k, len(c))
        top_indices = np.argsort(c)[::-1][:top_k_curr]
        
        # Gradient: sum of contributions from top peaks
        fsize = N
        W = np.zeros(fsize)
        for k in top_indices:
            W[int(k)] = 1.0
        Wf = fft(W, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(Wf * np.conj(A)))[:n]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
    
    return best_local, best_score

def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def make_chirp_init(n, rng):
    """Create a chirp-like initialization."""
    t = np.arange(n)
    # Real chirp: cos(pi * t^2 / n)
    c = np.cos(np.pi * t * t / n)
    p = 0.5 + 0.5 * c  # shift to [0, 1]
    p = p + rng.normal(0, 0.01, n)
    p = np.clip(p, 0.001, None)
    return p / p.sum()

def make_qr_init(n, rng):
    """Quadratic residue inspired initialization."""
    # For n that's prime, use Legendre symbol
    # For general n, use a pseudo-random structure
    p = np.ones(n) * 0.5
    # Add structured noise
    for k in range(1, min(10, n // 2)):
        freq = 2 * np.pi * k / n
        amp = 1.0 / (k + 1)
        p += amp * np.cos(freq * np.arange(n) + rng.random() * 2 * np.pi)
    p = np.clip(p, 0.001, None)
    return p / p.sum()

def block_structure_optimize(n, K, rng, n_trials=500):

    # Random block boundaries

    best_p = None
    
    for trial in range(n_trials):
        # Random partition
        if trial == 0:
            # Equal blocks
            bounds = np.linspace(0, n, K + 1, dtype=int)
        else:
            cuts = np.sort(rng.random(K - 1) * n).astype(int)
            cuts = np.clip(cuts, 0, n-1)
            cutoffs = np.zeros(K, dtype=int)
            all_cuts = np.concatenate([[0], cuts, [n]])
            bounds = all_cuts
        
        # Optimize block heights
        block_lens = np.diff(bounds)
        
        # Simple: random heights
        for h_trial in range(50):

            p = p / p.sum()
            c = fft_convolve_self(p)
            score = c.max() * 2 * n
            if score < best_score:
                best_score = score
                best_p = p.copy()
    
    return best_p, best_score

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_score_p(p, n):
    """Compute objective score for probability vector p (sum=1)."""
    c = fft_convolve_self(p)
    return 2.0 * n * c.max()


def spectral_flatten(a, N, rng, max_time, eta_init=0.12, target_flat=True):
    """
    Iteratively flatten the power spectrum of a to reduce conv peak.
    Targets |P[m]| ≈ 1 for all m (theoretically optimal flat spectrum).
    Uses acceptance criteria to prevent oscillation.
    """
    n = len(a)
    S = a.sum()
    p = a / S
    best_p = p.copy()

    accept_count = 0
    total_count = 0

        it += 1

        p_pad[:n] = p
        P = fft(p_pad)

        # (since |P[0]|^2 = 1 and sum|P|^2 = N, target is |P[m]| = 1 for all m)
        target_mag = np.ones(N)

        # Interpolate magnitudes towards target

        p_candidate = np.real(ifft(P_new))[:n]

        if p_prej.sum() < 0.9 * S / S:  # sanity check

        score_old = compute_score_p(p, n)

def lse_objective_and_grad(x, beta, n, N):
    """
    Log-sum-exp objective and gradient for L-BFGS-B.
    x: log-space variable of length n.
    Returns (objective_value, gradient_vector).
    """
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    Z = exp_x.sum()
    if Z < 1e-300:
        return 0.0, np.zeros(n)
    p = exp_x / Z

    # Convolution via FFT
    p_pad = np.zeros(N)
    p_pad[:n] = p
    P = fft(p_pad)
    mag2 = np.abs(P) ** 2
    c = np.real(ifft(mag2))[:2 * n - 1]

    # Log-sum-exp
    c_max = c.max()
    beta_c = beta * (c - c_max)
    beta_c = np.clip(beta_c, -500.0, 500.0)
    exp_bc = np.exp(beta_c)
    sum_exp = exp_bc.sum()
    if sum_exp < 1e-300:
        return c_max, np.zeros(n)
    log_sum = c_max + np.log(sum_exp) / beta

    # Weights
    w = exp_bc / sum_exp

    # Gradient w.r.t. p: grad_p[i] = 2 * sum_k w[k] * p[k-i]
    # This is 2 * cross_correlation(w, p)[i]
    fsize = N
    W = fft(w, fsize)
    grad_p = 2.0 * np.real(ifft(W * P))[:n]

    # Gradient w.r.t. x (softmax parameterization)
    # grad_x[j] = p[j] * (grad_p[j] - dot(grad_p, p))
    grad_p_dot_p = np.dot(grad_p, p)
    grad_x = p * (grad_p - grad_p_dot_p)

    return log_sum, grad_x


def lbfgsb_optimize(a_init, n, N, max_time, rng=None):
    """
    L-BFGS-B optimization with softmax parameterization and adaptive beta.
    """
    S = a_init.sum()
    p = a_init / S
    x0 = np.log(p + 1e-30)

    best_a = a_init.copy()
    best_score = 2.0 * n * fft_convolve_self(p).max()
    start = time.monotonic()

    beta_schedule = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]
    time_per_beta = max_time / len(beta_schedule)

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 2.0:
            break
        beta = float(beta)
        beta_max_iter = time_per_beta

        def fun_closure(x):
            val, _ = lse_objective_and_grad(x, beta, n, N)
            return val

        def grad_closure(x):
            _, g = lse_objective_and_grad(x, beta, n, N)
            return g

        t0 = time.monotonic()
        try:
            res = scipy_minimize(fun_closure, x0, jac=grad_closure, method='L-BFGS-B',
                                 options={'maxiter': 300, 'gtol': 1e-10, 'maxls': 30})

            x_opt = res.x
            x_shifted = x_opt - x_opt.max()
            exp_x = np.exp(x_shifted)
            p_opt = exp_x / exp_x.sum()
            p_opt = np.clip(p_opt, 0.0, None)
            if p_opt.sum() < 1e-10:
                p_opt = p  # fallback
            else:
                p_opt = p_opt / p_opt.sum()

            score = compute_score_p(p_opt, n)
            if score < best_score:
                best_score = score
                best_a = (p_opt * S).copy()

            x0 = x_opt  # warm start
        except Exception:
            break

    return best_a, best_score


def nesterov_toppk(a_init, n, S, max_time, rng, top_k_init=30, top_k_max=512):
    """Nesterov-accelerated top-k subgradient polishing."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    N = 1 << (2 * n - 1).bit_length()
    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 30)

        c = fft_convolve_self(y)
        cur_score = 2.0 * n * c.max() / (y.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_local = y.copy()

        # Top-k subgradient
        top_indices = np.argsort(c)[::-1][:top_k]
        fsize = N
        W = np.zeros(fsize)
        for k in top_indices:
            W[int(k)] = 1.0
        Wf = fft(W, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(Wf * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.005 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 2000 == 0 and rng is not None and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def make_chirp_init(n, rng):
    """Chirp-inspired non-negative initialization."""
    t = np.arange(n)
    c = np.cos(np.pi * t * t / n)
    p = 0.5 + 0.4 * c + 0.1 * rng.standard_normal(n)
    p = np.clip(p, 0.001, None)
    return p / p.sum()


def make_multisine_init(n, rng):
    """Multi-sine initialization with random phases."""
    t = np.arange(n)
    p = np.ones(n) * 0.3
    n_harm = 8
    for k in range(1, n_harm + 1):
        freq = 2 * np.pi * k / n
        amp = 0.8 / (k * 0.7 + 1)
        phase = rng.random() * 2 * np.pi
        p += amp * np.cos(freq * t + phase)
    p = np.clip(p, 0.001, None)
    return p / p.sum()


def make_block_init(n, K, rng):
    """Block-structured initialization with optimized heights."""
    bounds = np.linspace(0, n, K + 1, dtype=int)
    block_lens = np.diff(bounds)

    best_score = np.inf
    best_p = None

    for trial in range(300):
        heights = rng.random(K) + 0.2
        p = np.repeat(heights, block_lens)
        p = p / p.sum()
        score = 2.0 * n * fft_convolve_self(p).max()
        if score < best_score:
            best_score = score
            best_p = p.copy()

    return best_p, best_score


def make_ramp_modulated_init(n, rng):
    """Ramp modulated by cosine - creates asymmetric structure."""
    t = np.arange(n) / n
    ramp = t
    mod = 1.0 + 0.6 * np.cos(2 * np.pi * (2.0 * t + 0.3 * t * t))
    p = ramp * mod + 0.05 * rng.standard_normal(n)
    p = np.clip(p, 0.001, None)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing evaluate_sequence.
    Uses spectral flattening + L-BFGS-B + Nesterov with diverse multi-start.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 30.0)

    global_best = np.inf
    global_a = None

    n = 8192
    S = np.sqrt(2.0 * float(n))
    N = 1 << (2 * n - 1).bit_length()

    print("n=%d, S=%.4f, N(FFT)=%d" % (n, S, N))

    # --- Generate initializations ---
    print("Generating initializations...")
    inits = {}

    inits['uniform'] = np.ones(n) / n
    inits['chirp'] = make_chirp_init(n, rng)
    inits['multisine'] = make_multisine_init(n, rng)
    inits['ramp'] = make_ramp_modulated_init(n, rng)

    r = rng.random(n) + 0.05
    inits['random'] = r / r.sum()

    bp, _ = make_block_init(n, 5, rng)
    inits['block5'] = bp
    bp3, _ = make_block_init(n, 3, rng)
    inits['block3'] = bp3

    # Try to load previous solution
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size == n:
            g = np.clip(given, 0, 1000)
            if g.sum() > 0.01:
                inits['previous'] = g / g.sum()
                sc = 2.0 * n * fft_convolve_self(g / g.sum()).max()
                if sc < global_best:
                    global_best = sc
                    global_a = g.copy()
                print("  Previous solution score: %.6f" % sc)
    except Exception:
        pass

    print("  %d initializations" % len(inits))

    # --- Phase 1: Spectral Flattening ---
    print("=== Phase 1: Spectral Flattening ===")
    t_p1 = time.monotonic()
    phase1_budget = (deadline - t_p1) * 0.40
    per_init_p1 = phase1_budget / len(inits)

    sf_results = {}

    for name, p in inits.items():
        if time.monotonic() > deadline - 90:
            break
        max_t = min(per_init_p1, 150.0)
        a = p * S
        a_out, score = spectral_flatten(a, N, rng, max_time=max_t, eta_init=0.12)
        a_out = a_out / a_out.sum()
        sf_results[name] = a_out
        if score < global_best:
            global_best = score
            global_a = a_out.copy()
            print("  [%s] SF: %.6f" % (name, global_best))
        print("  [%s] done, cur_best=%.6f" % (name, global_best))

    print("Phase 1 done. Best: %.6f" % global_best)

    # --- Phase 2: L-BFGS-B on top results ---
    print("=== Phase 2: L-BFGS-B ===")
    t_p2 = time.monotonic()
    phase2_budget = (deadline - t_p2) * 0.30
    per_init_p2 = phase2_budget / min(5, len(sf_results))

    # Sort by current score
    scored = []
    for name, p in sf_results.items():
        sc = compute_score_p(p, n)
        scored.append((sc, name, p))
    scored.sort(key=lambda x: x[0])

    for sc, name, p in scored[:5]:
        if time.monotonic() > deadline - 60:
            break
        max_t = min(per_init_p2, 180.0)
        a = p * S
        a_out, score2 = lbfgsb_optimize(a, n, N, max_t=max_t, rng=rng)
        a_out = a_out / a_out.sum()
        if score2 < global_best:
            global_best = score2
            global_a = a_out.copy()
            print("  [%s] LBFGS: %.6f" % (name, global_best))
        print("  [%s] done, cur_best=%.6f" % (name, global_best))

    print("Phase 2 done. Best: %.6f" % global_best)

    # --- Phase 3: Alternating spectral + Nesterov polish ---
    print("=== Phase 3: Polish ===")
    t_p3 = time.monotonic()
    if global_a is not None:
        remaining = deadline - t_p3
        a_curr = global_a.copy()

        # Multiple Nesterov passes with increasing top-k
        nk_schedule = [20, 50, 100, 200, 400]
        seg_time = min(remaining / (len(nk_schedule) + 2), 120.0)

        for tk in nk_schedule:
            if time.monotonic() > deadline - 10:
                break
            a_out, _ = nesterov_toppk(a_curr, n, S, seg_time, rng,
                                       top_k_init=tk, top_k_max=tk + 50)
            a_out = a_out / a_out.sum()
            sc = compute_score_p(a_out, n)
            if sc < global_best:
                global_best = sc
                global_a = a_out.copy()
                a_curr = a_out.copy()
                print("  NG tk=%d: %.6f" % (tk, global_best))

        # Final spectral flatten with very small eta
        if time.monotonic() < deadline - 15:
            rem = deadline - time.monotonic() - 5
            a_out, _ = spectral_flatten(global_a, N, rng, max_time=rem, eta_init=0.03)
            a_out = a_out / a_out.sum()
            sc = compute_score_p(a_out, n)
            if sc < global_best:
                global_best = sc
                global_a = a_out.copy()
                print("  SF final: %.6f" % global_best)

        # Final L-BFGS-B with high beta
        if time.monotonic() < deadline - 10:
            rem = deadline - time.monotonic() - 3
            a_out, _ = lbfgsb_optimize(global_a, n, N, max_time=rem, rng=rng)
            a_out = a_out / a_out.sum()
            sc = compute_score_p(a_out, n)
            if sc < global_best:
                global_best = sc
                global_a = a_out.copy()
                print("  LBFGS final: %.6f" % global_best)

    print("Phase 3 done. Best: %.6f" % global_best)

    # --- Final verification and return ---
    if global_a is not None:
        global_a = np.clip(global_a, 0.0, 1000.0)
        if global_a.sum() < 0.01:
            global_a = np.ones(256, dtype=np.float64)
        final_a = global_a / global_a.sum() * S
        final_list = [float(x) for x in final_a]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Verification: %.8f" % ver)
            if ver < global_best:
                global_best = ver
        else:
            print("Verification failed, using fallback")
            final_a = np.ones(256, dtype=np.float64)
            final_list = [float(x) for x in final_a]
    else:
        final_a = np.ones(256, dtype=np.float64)
        final_list = [float(x) for x in final_a]

    print("FINAL: score=%.8f, length=%d" % (global_best, len(final_list)))
    return final_list
