"""
Minimize 2*n*max_conv/sum^2 using Frank-Wolfe with LP oracle + L-BFGS polishing + diverse restarts.
Key innovations over previous approach:
1. LP-based Frank-Wolfe direction (different from gradient-based L-BFGS)
2. Active constraint selection for efficient LP solves
3. Structured local search (swaps, subsystem scaling)
4. Multiple resolution strategy (coarse-to-fine)
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize, linprog


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_cross(a, w):
    """Compute cross-correlation: result[j] = sum_k w[k] * a[k-j] (via FFT).
    This gives grad of (1/2)*max_w w @ conv(a,a) w.r.t. a."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """Smoothed objective and gradient for softmax parameterization."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_grad_cross(a, w)
    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def solve_lp_oracle(f, M, active_indices, n):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_indices, g >= 0.
    Returns solution g or None if infeasible.
    """
    K = len(active_indices)
    if K == 0:
        return np.ones(n)
    
    # Build constraint matrix A (K x n)
    # A[idx, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    A = np.zeros((K, n))
    for idx, k in enumerate(active_indices):
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            # A[idx, j] = f[k-j] for j in [j_min, j_max]
            # So A[idx, j_min:j_max+1] = f[k-j_min : k-j_max+1] reversed
            A[idx, j_min:j_max+1] = f[k - j_min : k - j_max + 1][::-1]
    
    # min -1^T g s.t. A g <= M*1, 0 <= g
    c_obj = -np.ones(n)
    b_ub = M * np.ones(K)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def frank_wolfe_lp(f_init, n, S, max_time, active_K=30, rng=None):
    """Frank-Wolfe with LP oracle for active constraints."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    f = f_init.copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f * S / s
    
    best_f = f.copy()
    best_score = fft_convolve_self(f).max()
    
    t_start = time.monotonic()
    t = 0
    no_improve = 0
    
    while time.monotonic() - t_start < max_time:
        t += 1
        c = fft_convolve_self(f)
        M = c.max()
        
        if M < best_score:
            best_score = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        # Select active constraints: top K by value
        top_idx = np.argsort(c)[::-1][:active_K]
        
        # Solve LP
        g = solve_lp_oracle(f, M, top_idx, n)
        if g is None:
            # Fallback: use gradient-related direction
            top_vals = c[top_idx]
            total = top_vals.sum()
            if total < 1e-30:
                break
            weights = top_vals / total
            w_full = np.zeros(2 * n - 1)
            for i, k in enumerate(top_idx):
                w_full[int(k)] = weights[i]
            grad = compute_grad_cross(f, w_full)
            g = project_simplex(f - 0.1 * S / np.sqrt(n) * grad, S)
        else:
            g_sum = g.sum()
            if g_sum < 1e-15:
                break
            g = g * S / g_sum
        
        # Step size with backtracking
        base_step = 2.0 / (t + 2.0)
        step = min(base_step, 0.5)
        
        f_new = (1 - step) * f + step * g
        f_new = np.maximum(f_new, 0.0)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            break
        f_new = f_new * S / f_new_sum
        
        f = f_new
        
        if t % 100 == 0:
            cur = fft_convolve_self(f).max()
            print("    FW iter %d: score=%.8f, best=%.8f" % (t, cur, best_score))
    
    return best_f, best_score


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=200):
    """L-BFGS with log-sum-exp continuation."""
    if beta_schedule is None:
        beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 5:
            break

        n_attempts = 1
        if beta >= 500 and max_time > 120:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 5:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.01, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

                return softmax_obj_grad(z_val, beta, S)[1]

            try:

                    fun, z, jac=jac,
                    method='L-BFGS-B',

                             'maxls': 30, 'iprint': -1}

            except Exception:
                break

            a = np.maximum(a, 0.0)

            c = fft_convolve_self(a)
            score = c.max()
            if score < best_score:

def local_search_swap(a, n, S, rng, n_swaps=1000):
    """Local search using swap moves."""
    a = a.copy()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    
    for _ in range(n_swaps):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # Swap
        a[i], a[j] = a[j], a[i]
        c = fft_convolve_self(a)
        score = c.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        else:
            # Revert
            a[i], a[j] = a[j], a[i]
    
    return best_a, best_score


def local_search_move(a, n, S, rng, n_moves=1000):
    """Local search using move (redistribute mass between two positions)."""
    a = a.copy()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()
    
    for _ in range(n_moves):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # Move some mass from i to j (or vice versa)
        delta = rng.random() * 0.1 * max(a[i], a[j], S/n)
        if a[i] >= delta:
            a[i] -= delta
            a[j] += delta
        else:
            a[j] -= delta
            a[i] += delta
        a = np.maximum(a, 0)
        a = a * S / a.sum()
        
        c = fft_convolve_self(a)
        score = c.max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        else:
            # Revert (approximately)
            a = best_a.copy()
    
    return best_a, best_score


def safe_normalize(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def thue_morse_pattern(n):
    """Generate a Thue-Morse-inspired non-negative pattern."""
    # Thue-Morse: t(k) = popcount(k) mod 2
    # Map to values: t=0 -> high, t=1 -> low
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=np.float64)
    # Create a pattern: high where tm=0, low where tm=1
    return (1.0 - 0.5 * tm)

So the coefficient of g[i] in constraint k is f[k-i] if 0 <= k-i <= n-1, else 0.
