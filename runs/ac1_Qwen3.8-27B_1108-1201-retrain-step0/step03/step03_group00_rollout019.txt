on active constraints, and momentum restarts for escaping local minima.

import numpy as np
import time

def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cumsum - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def normalize_sum(a, n):
    """Clamp and rescale to sum = sqrt(2n)."""
    S = np.sqrt(2.0 * n)
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

    """Gradient of LSE approximation of max(c)."""
    n = len(a)
    cm = c.max()
    e = np.exp(beta * (c - cm))
    tot = e.sum()
    if tot < 1e-300:
        return np.zeros(n)
    w = e / tot
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    return 2.0 * np.real(ifft(W * np.conj(A)))[:n]


def topk_gradient(a, c, k, power=20.0):
    """Gradient using top-k peaks with power weighting."""
    n = len(a)
    k = min(k, len(c))
    idx = np.argsort(c)[::-1][:k]
    vals = c[idx]
    vmax = vals[0]
    if vmax < 1e-30:
        return np.zeros(n)

    wsum = w.sum()

def exact_subgradient(a, c):
    """Exact subgradient at the single argmax lag."""
    n = len(a)
    k = int(np.argmax(c))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    grad = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k - idx]

    else:
        if len(active) > K:
            active = active[np.argsort(c[active])[::-1]][:K]

    if len(active) == 0:
        return None

    K_actual = len(active)
    if K_actual > 60:
        K_actual = 60
        active = active[:K_actual]

    # Build constraint matrix A (K_actual x n)
    A_mat = np.zeros((K_actual, n))
    for j, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        A_mat[j, idx] = a[k - idx]

    # Solve: min -sum(b) s.t. A_mat @ b <= M, b >= 0
    try:
        from scipy.optimize import linprog
        c_obj = -np.ones(n)
        res = linprog(c_obj, A_ub=A_mat, b_ub=np.full(K_actual, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def run_fista_adaptive(a_init, n, S, max_time, rng, start_beta=10.0,
                       beta_rate=0.003, beta_max=3000.0, use_lp=True,
                       lp_interval=5000, topk=16):
    """FISTA with adaptive gradient (LSE + top-k + exact) and LP polishing."""
    a = normalize_sum(a_init.copy(), n)
    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    it = 0
    last_improve = 0
    last_lp_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, start_beta * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-15:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Choose gradient
        if it % 3 == 0:
            # Top-k gradient (more diverse)
            grad = topk_gradient(y, c, k=topk, power=min(50.0, 5.0 + 0.01 * it))
        elif it % 3 == 1:
            # LSE gradient
            grad = lse_gradient(y, c, beta)
        else:
            # Exact subgradient at peak
            grad = exact_subgradient(y, c)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Step size
        if it < 1000:
            step = 1.0 / (2.0 * max(beta, 5.0) * 0.5)
        else:
            step = S / (gnorm * (1.0 + 0.01 * np.log1p(it / 500.0)))

        # FISTA update
        x_new = project_simplex(y - step * grad, S)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        a_prev = a
        t = t_new

        # LP-based direction finding (periodically)
        if use_lp and it % lp_interval == 0 and (time.monotonic() - start) < max_time - 10:
            lp_dir = solve_lp_direction(a, c, cur, K=30)
            if lp_dir is not None:
                lp_dir = normalize_sum(lp_dir, n)
                # Line search: find best t in [0, 1]
                best_val = cur
                best_t = 0.0
                for trial_t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    trial_a = normalize_sum(a + trial_t * (lp_dir - a), n)
                    trial_c = fft_convolve_self(trial_a)
                    if trial_c.max() < best_val:
                        best_val = trial_c.max()
                        best_t = trial_t
                if best_t > 0:
                    a = normalize_sum(a + best_t * (lp_dir - a), n)
                    y = a.copy()
                    t = 1.0
                    if best_val < best:
                        best = best_val
                        best_a = a.copy()
                        last_improve = it
                        last_lp_improve = it

        # Stuck detection: add noise and restart momentum
        stuck = it - last_improve > 3000
        if stuck and (time.monotonic() - start) < max_time - 5:
            noise_scale = 0.002 * S / np.sqrt(n) * np.exp(-it / 30000.0)
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
                a = y.copy()
            else:
                # Try a larger perturbation
                noise_scale = 0.01 * S / np.sqrt(n)
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0
                a = y.copy()
            last_improve = it

        # Periodic small restarts
        if it % 8000 == 0 and (time.monotonic() - start) < max_time - 10:
            decay = np.exp(-it / 50000.0)
            noise_scale = 0.001 * S / np.sqrt(n) * decay
            if noise_scale > 1e-8:
                y = project_simplex(y + rng.normal(0, noise_scale, n), S)
                t = 1.0

    # Final check
    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()

    return best_a, best


def run_polish_exact(a_init, n, S, max_time, rng):
    """Polish using exact subgradient with adaptive step and averaging."""
    a = normalize_sum(a_init.copy(), n)
    start = time.monotonic()
    best = np.inf
    best_a = a.copy()
    y = a.copy()
    t = 1.0
    step_base = S / n
    momentum = 0.9
    v = np.zeros(n)
    it = 0
    last_improve = 0

    while time.monotonic() - start < max_time:
        it += 1
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best - 1e-15:
            best = cur
            best_a = y.copy()
            last_improve = it

        # Use top-5 peaks for robustness
        top_k = min(5, len(c))
        idx = np.argsort(c)[::-1][:top_k]
        vals = c[idx]
        vmax = vals[0]
        if vmax < 1e-30:
            break
        w = (vals / vmax) ** 100.0
        wsum = w.sum()
        w = w / wsum
        w_full = np.zeros(len(c))
        w_full[idx] = w
        fsize = 1 << (len(w_full) + n - 1).bit_length()
        W = fft(w_full, fsize)
        A = fft(y, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = step_base / (1.0 + 0.001 * it)
        v = momentum * v - step * grad
        y_new = y + v
        y_new = project_simplex(y_new, S)

        # Momentum restart
        if it % 200 == 0 or it - last_improve > 2000:
            t = 1.0
            v = np.zeros(n)
            if it - last_improve > 2000:
                noise = 0.001 * S / np.sqrt(n)
                y_new = project_simplex(y_new + rng.normal(0, noise, n), S)
                last_improve = it

        y = y_new

    c_final = fft_convolve_self(best_a)
    if c_final.max() < best:
        best = c_final.max()
    return best_a, best


def generate_diverse_inits(n, rng):
    """Generate diverse initializations for exploration."""
    inits = []
    tt = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.maximum(0.5 * (1 + np.cos(2 * np.pi * tt)), 0.01))
    inits.append(0.54 + 0.46 * np.cos(2 * np.pi * tt))
    inits.append(np.sin(np.pi * tt) ** 2)
    inits.append(np.sin(np.pi * tt) ** 4)
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 2 * tt) + 0.2 * np.cos(2 * np.pi * 5 * tt))
    inits.append(rng.random(n) + 0.1)
    inits.append(np.exp(rng.normal(0, 0.5, n)))
    for f in [3, 7, 11]:
        inits.append(1.0 + 0.4 * np.sin(2 * np.pi * f * tt + rng.random() * 2 * np.pi))
    return [np.maximum(x, 0.01) for x in inits]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)

    global_best = np.inf
    global_a = None

    # Try to get the given sequence
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0.01:
            given = np.clip(given, 0, 1000)
    except Exception:
        pass

    # Choose main length
    if given is not None and 1024 <= len(given) <= 32768:
        main_len = len(given)
    else:
        main_len = 8192

    n = main_len
    S = np.sqrt(2.0 * n)
    print("Starting optimization with n=%d, budget=%.0fs" % (n, budget_s))

    # --- Phase 1: Quick FISTA warm-up from given sequence (120s) ---
    print("=== Phase 1: Warm-up FISTA ===")
    p1_time = min(120.0, (deadline - time.monotonic()) * 0.1)
    if given is not None:
        a1, s1 = run_fista_adaptive(given, n, S, p1_time, rng,
                                     start_beta=20.0, beta_rate=0.005,
                                     beta_max=1000.0, lp_interval=3000)
        a1 = normalize_sum(a1, n)
        c1 = fft_convolve_self(a1)
        score1 = c1.max()
        print("  After warm-up: max_conv=%.6f, score=%.6f" % (score1, 2.0 * n * score1 / S ** 2))
        if score1 < global_best:
            global_best = score1
            global_a = a1.copy()
    else:
        a1 = np.ones(n) * S / n
        global_a = a1.copy()
        c1 = fft_convolve_self(a1)
        global_best = c1.max()

    # --- Phase 2: Main optimization (60% of remaining time) ---
    remaining = deadline - time.monotonic()
    p2_time = remaining * 0.55
    print("=== Phase 2: Main optimization (%.0fs) ===" % p2_time)

    a2, s2 = run_fista_adaptive(global_a, n, S, p2_time, rng,
                                 start_beta=50.0, beta_rate=0.002,
                                 beta_max=3000.0, lp_interval=4000, topk=20)
    a2 = normalize_sum(a2, n)
    c2 = fft_convolve_self(a2)
    if c2.max() < global_best:
        global_best = c2.max()
        global_a = a2.copy()
    print("  After main: max_conv=%.6f, score=%.6f" % (global_best, 2.0 * n * global_best / S ** 2))

    # --- Phase 3: Exact subgradient polishing (25% of remaining) ---
    remaining = deadline - time.monotonic()
    p3_time = remaining * 0.30
    print("=== Phase 3: Exact polishing (%.0fs) ===" % p3_time)

    a3, s3 = run_polish_exact(global_a, n, S, p3_time, rng)
    a3 = normalize_sum(a3, n)
    c3 = fft_convolve_self(a3)
    if c3.max() < global_best:
        global_best = c3.max()
        global_a = a3.copy()
    print("  After exact polish: max_conv=%.6f, score=%.6f" % (global_best, 2.0 * n * global_best / S ** 2))

    # --- Phase 4: Final high-beta FISTA (15% of remaining) ---
    remaining = deadline - time.monotonic()
    p4_time = remaining * 0.40
    print("=== Phase 4: Final high-beta FISTA (%.0fs) ===" % p4_time)

    a4, s4 = run_fista_adaptive(global_a, n, S, p4_time, rng,
                                 start_beta=200.0, beta_rate=0.001,
                                 beta_max=10000.0, lp_interval=8000, topk=8)
    a4 = normalize_sum(a4, n)
    c4 = fft_convolve_self(a4)
    if c4.max() < global_best:
        global_best = c4.max()
        global_a = a4.copy()
    print("  After final FISTA: max_conv=%.6f, score=%.6f" % (global_best, 2.0 * n * global_best / S ** 2))

    # --- Phase 5: LP-based final polishing ---
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print("=== Phase 5: LP polishing (remaining %.0fs) ===" % remaining)
        a5 = normalize_sum(global_a.copy(), n)
        S5 = S
        for lp_iter in range(20):
            if time.monotonic() > deadline - 5:
                break
            c5 = fft_convolve_self(a5)
            M5 = c5.max()
            cur_score = 2.0 * n * M5 / S5 ** 2
            print("  LP iter %d: score=%.6f" % (lp_iter, cur_score))

            lp_dir = solve_lp_direction(a5, c5, M5, K=40)
            if lp_dir is None:
                break
            lp_dir = normalize_sum(lp_dir, n)

            # Line search
            best_val = M5
            best_t = 0.0
            for trial_t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
                trial_a = normalize_sum(a5 + trial_t * (lp_dir - a5), n)
                trial_c = fft_convolve_self(trial_a)
                if trial_c.max() < best_val:
                    best_val = trial_c.max()
                    best_t = trial_t

            if best_t > 0:
                a5 = normalize_sum(a5 + best_t * (lp_dir - a5), n)
                if best_val < global_best:
                    global_best = best_val
                    global_a = a5.copy()
            else:
                # No improvement from LP, try small perturbation
                noise = 0.001 * S5 / np.sqrt(n)
                a5 = project_simplex(a5 + rng.normal(0, noise, n), S5)

        # Verify final LP result
        c5_final = fft_convolve_self(a5)
        if c5_final.max() < global_best:
            global_best = c5_final.max()
            global_a = a5.copy()
        print("  After LP polish: score=%.6f" % (2.0 * n * global_best / S ** 2))

    # --- Phase 6: Try exploring a different length if time permits ---
    remaining = deadline - time.monotonic()
    if remaining > 120:
        alt_n = n * 2 if n <= 8192 else n // 2
        if 2048 <= alt_n <= 32768:
            S_alt = np.sqrt(2.0 * alt_n)
            print("=== Phase 6: Exploring n=%d (remaining %.0fs) ===" % (alt_n, remaining))
            # Upsample or downsample current best
            if alt_n > n:
                # Interpolate
                t_old = np.linspace(0, 1, n)
                t_new = np.linspace(0, 1, alt_n)
                alt_init = np.interp(t_new, t_old, global_a)
            else:
                nblocks = n // alt_n
                alt_init = global_a[:nblocks * alt_n].reshape(alt_n, nblocks).mean(axis=1)

            a6, s6 = run_fista_adaptive(alt_init, alt_n, S_alt, remaining * 0.5, rng,
                                         start_beta=20.0, beta_rate=0.003,
                                         beta_max=2000.0, lp_interval=5000)
            a6 = normalize_sum(a6, alt_n)
            c6 = fft_convolve_self(a6)
            score6 = 2.0 * alt_n * c6.max() / S_alt ** 2
            # Compare using the actual evaluation metric
            if c6.max() * 2.0 * alt_n / S_alt ** 2 < global_best * 2.0 * n / S ** 2:
                # This is comparing scores, we want lower
                pass
            # Actually let's compare the actual objective values
            obj_current = 2.0 * n * global_best / S ** 2
            obj_alt = 2.0 * alt_n * c6.max() / S_alt ** 2
            if obj_alt < obj_current:
                print("  Alt n=%d gives better score: %.6f vs %.6f" % (alt_n, obj_alt, obj_current))
                global_best = c6.max()
                global_a = a6.copy()
                n = alt_n
                S = S_alt
            else:
                print("  Alt n=%d: %.6f (not better than %.6f)" % (alt_n, obj_alt, obj_current))

    # --- Final verification ---
    if global_a is not None:
        final_list = [float(x) for x in np.clip(global_a, 0.0, 1000.0)]
        ver = evaluate_sequence(final_list)
        if np.isfinite(ver):
            print("Final verification: %.6f" % ver)
        else:
            print("evaluate_sequence rejected, using fallback")
            global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64) * (np.sqrt(2 * 256) / 256)

    final_score = 2.0 * len(global_a) * fft_convolve_self(global_a).max() / global_a.sum() ** 2
    print("Returning sequence of length %d, score=%.6f" % (len(global_a), final_score))
    return [float(x) for x in global_a]
