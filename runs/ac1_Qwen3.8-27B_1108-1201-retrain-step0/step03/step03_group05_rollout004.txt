"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy:
- Try multiple sequence lengths (8192, 16384) for more degrees of freedom
- Use asymmetric initializations (exponential ramps) exploiting the insight
  that optimal sequences are front/back-loaded to reduce center convolution
- L-BFGS-B with log-sum-exp continuation (smooth-to-sharp max approximation)
- Subgradient polishing with top-k peak focusing
- Cross-length transfer and perturbation restarts

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize as scipy_minimize


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_grad_a(a, w):
    """Gradient of dot(w, conv(a,a)) w.r.t. a, via FFT cross-correlation."""
    n = len(a)
    fsize = 1 << (len(w) + n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_obj_grad(z, beta, S):
    """Smoothed objective and gradient w.r.t. softmax parameters z."""
    n = len(z)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    sum_exp = exp_z.sum()
    a = S * exp_z / sum_exp

    c = fft_convolve_self(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    exp_bc = np.exp(scaled)
    sum_exp_bc = exp_bc.sum()
    obj = (c_max + np.log(sum_exp_bc)) / beta

    w = exp_bc / sum_exp_bc
    grad_a = compute_grad_a(a, w)

    dot_ga_a = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot_ga_a / S)

    return obj, grad_z


def safe_normalize(a, n):
    """Clamp to [0,1000], rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def run_lbfgs_continuation(a_init, n, S, max_time, rng,
                           beta_schedule=None, maxiter_per_beta=200):
    """L-BFGS-B with continuation on beta. Returns (best_a, best_score)."""
    if beta_schedule is None:
        beta_schedule = [3, 5, 10, 20, 50, 100, 200, 400, 800, 1500, 3000]

    a = np.clip(a_init, 1e-15, 10000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * S / s
    z = np.log(a / S + 1e-30)

    start = time.monotonic()
    best_score = fft_convolve_self(a).max()
    best_a = a.copy()

    for beta in beta_schedule:
        if time.monotonic() - start > max_time - 3:
            break
        n_attempts = 1
        if beta >= 200 and max_time > 30:
            n_attempts = 2

        for attempt in range(n_attempts):
            if time.monotonic() - start > max_time - 3:
                break
            if attempt > 0:
                z = z + rng.normal(0, 0.015, n)
                z = z - z.mean()

            def fun(z_val):
                return softmax_obj_grad(z_val, beta, S)[0]

            def jac(z_val):
                return softmax_obj_grad(z_val, beta, S)[1]

            t0 = time.monotonic()
            try:
                result = scipy_minimize(
                    fun, z, jac=jac, method='L-BFGS-B',
                    options={'maxiter': maxiter_per_beta,
                             'maxfun': maxiter_per_beta * 3,
                             'ftol': 1e-14, 'gtol': 1e-10,
                             'maxls': 30, 'iprint': -1}
                )
            except Exception:
                break

            z = result.x
            z = z - z.mean()

            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = S * exp_z / exp_z.sum()
            a = np.maximum(a, 0.0)
            s2 = a.sum()
            if s2 < 1e-15:
                a = np.ones(n) * S / n
            else:
                a = a * S / s2

            score = fft_convolve_self(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def run_subgradient_polish(a_init, n, S, max_time, rng, top_k_init=8, top_k_max=1024):
    """Nesterov-accelerated top-k subgradient method."""
    a = np.clip(a_init, 0.0, 1000.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        top_k = min(top_k_max, top_k_init + it // 50)

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_local = y.copy()

        # Use soft weighting: weight proportional to c^p for large p
        p = max(1.0, 5.0 + it * 0.1)
        c_clipped = np.maximum(c, 1e-30)
        weights = c_clipped ** p
        total_w = weights.sum()
        if total_w < 1e-30:
            break
        weights = weights / total_w

        # Only use top-k for efficiency
        top_indices = np.argsort(c)[::-1][:top_k]
        w_full = np.zeros(len(c))
        for i, k in enumerate(top_indices):
            w_full[k] = weights[k]
        w_full = w_full / w_full.sum()

        fsize = 1 << (len(w_full) + n - 2).bit_length()
        W = fft(w_full, fsize)
        A = fft(a, fsize)
        grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        step = 2.0 * S / (n * (1.0 + 0.003 * it))
        x_new = project_simplex(y - step * grad / max(gnorm, 1e-15), S)

        if it % 5 == 0:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 3000 == 0 and rng.random() < 0.1:
            y = project_simplex(y + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
            t = 1.0

    return best_local, best_score


def generate_ramp_init(n, alpha, rng):
    """Generate a geometric ramp initialization: a[i] = exp(alpha * i/n)."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(alpha * (t - 0.5))
    # Add small noise for diversity
    a += rng.normal(0, 0.01, n)
    a = np.maximum(a, 0.001)
    return a


def generate_smooth_asymmetric(n, rng):
    """Generate smooth asymmetric initialization using polynomial basis."""
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 + 0.8 * (t - 0.3) + 0.3 * np.sin(2 * np.pi * 2 * t) \
        + 0.2 * np.cos(2 * np.pi * 5 * t) + 0.1 * np.sin(2 * np.pi * 11 * t)
    a = np.maximum(a, 0.01)
    a += rng.normal(0, 0.005, n)
    a = np.maximum(a, 0.001)
    return a
