def fft_convolve_self(a):

def normalize_nonneg(a, S):
    """Project to non-negative and normalize to sum S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(len(a), S / len(a))
    return a * (S / s)

def lse_objective(c, T):

    scaled = T * (c - max_c)
    scaled = np.clip(scaled, -700, 700)
    return max_c + np.log(np.exp(scaled).sum()) / T

def run_lse_lbfqsg(a_init, n, S, time_budget, T_init=5.0, T_final=500.0,

    best_obj = c.max()

    g_list = []  # gradient values

    step = 0
    no_improve = 0
    max_no_improve = 100

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    adam_t = 0
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        T = T_init * (T_final / T_init) ** (progress ** 0.8)
        
        c = fft_convolve_self(a)
        cur_obj = c.max()

            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            # Restart with noise

            a = best_a + noise_scale * np.random.default_rng(step).standard_normal(n)
            a = normalize_nonneg(a, S)
            s_list = []
            y_list = []
            no_improve = 0
            step += 1
            continue

        if len(s_list) > 0:
            q = g.copy()
            alphas = []
            rhos = []
            for i in range(len(s_list) - 1, -1, -1):
                r = s_list[i].dot(y_list[i])
                if r < 1e-30:
                    break
                rho = 1.0 / r
                alpha = rho * s_list[i].dot(q)
                alphas.append(alpha)
                rhos.append(rho)

            alphas.reverse()
            rhos.reverse()

        step_size = 1.0

                improved = True

            beta1, beta2, eps = 0.9, 0.999, 1e-8
            m_adam = beta1 * m_adam + (1 - beta1) * g
            v_adam = beta2 * v_adam + (1 - beta2) * (g ** 2)

            v_hat = v_adam / (1 - beta2 ** adam_t)

def hessian_direction(c, a, active_lags):
    """Compute Newton-like direction for active peaks.
    For each active lag m, the Hessian of c_m w.r.t. a is 2*Q_m where Q_m[i,j] = 1 if i+j=m.
    We want to find d that reduces all active c_m simultaneously.
    """
    n = len(a)
    # Gradient for active lags
    g_active = np.zeros(n)
    for m in active_lags:
        for j in range(n):
            k = m - j
            if 0 <= k < n:
                g_active[j] += 2.0 * a[k]
    
    # Simple Newton step: d = -H^{-1} g where H is the Hessian of the smoothed objective
    # For simplicity, use diagonal approximation: H_ii = 2*sum_m w_m * delta(m-2i in range)
    # Actually, H_ii for c_m is 2 if 2i=m and 0<m-2i... 
    # Let's just use gradient direction with step size from curvature
    gnorm = np.linalg.norm(g_active)
    if gnorm < 1e-15:
        return np.zeros(n)
    return -g_active / gnorm

def frequency_projection(a, S, n_iter=10):
    """Try to improve by working in frequency domain.

def run_adaptive_smoothed(a_init, n, S, time_budget, verbose=False):
    """
    Main optimization: adaptive log-sum-exp with multiple strategies.
    """
    a = a_init.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Track history for adaptive temperature
    obj_history = [best_obj]
    
    start = time.monotonic()
    step = 0
    no_improve = 0
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    while time.monotonic() - start < time_budget:
        progress = min(1.0, (time.monotonic() - start) / time_budget)
        t += 1
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 80:
            # More aggressive restart
            noise = 0.02 * max(0.05, 1.0 - progress)
            rng = np.random.default_rng(step * 1000)
            a = best_a + noise * rng.standard_normal(n)
            a = normalize_nonneg(a, S)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            continue
        
        # Adaptive temperature: based on spread of top peaks
        top_vals = np.sort(c)[::-1][:10]
        if top_vals[0] > 0:
            spread = (top_vals[0] - top_vals[min(9, len(top_vals)-1)]) / top_vals[0]
            # If peaks are close together, use higher T (more uniform weighting)
            # If one peak dominates, use lower T (focus on that peak)
            T = 5.0 / (spread + 0.1)
            T = max(1.0, min(2000.0, T * (1.0 + 2.0 * progress)))
        else:
            T = 10.0
        
        # Compute gradient
        g = compute_lse_gradient(c, a, T)
        
        # Adam update
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr = 0.5 * (1.0 - 0.5 * progress)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = normalize_nonneg(a_new, S)
        
        # Check if we should also try a "peak shaving" step
        if step % 50 == 0 and n > 100:
            c_new = fft_convolve_self(a_new)
            obj_new = c_new.max()
            
            # If the max is at a specific lag, try targeted reduction
            if obj_new < cur_obj:
                a = a_new
            else:
                # Keep old a, try peak shaving
                a = a_new  # still use new (Adam should be monotonically improving on LSE)
        
        a = a_new
        step += 1
        
        if verbose and step % 500 == 0:
            print("    Adaptive step %d: obj=%.7f, T=%.1f" % (step, best_obj, T))
    
    return best_a, best_obj

def coordinate_descent_peaks(a, n, S, time_budget, max_sweeps=30):
    """Coordinate descent focusing on peak reduction."""
    a = a.copy().astype(np.float64)
    a = normalize_nonneg(a, S)
    start = time.monotonic()
    
    c = fft_convolve_self(a)
    best_obj = c.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(42)
    
    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        
        c = fft_convolve_self(a)
        cur_obj = c.max()
        max_lag = np.argmax(c)
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - start > time_budget:
                break
            
            original = a[i]
            if original < 1e-12:
                continue
            
            best_local = original
            best_local_obj = cur_obj
            
            # Try multiplicative changes, biased towards reducing the peak
            peak_val = a[(max_lag - i) % n] if (max_lag - i) % n < n else 0
            factors = [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 0.8, 1.2, 0.4]
            
            for factor in factors:
                a[i] = original * factor
                a_norm = normalize_nonneg(a, S)
                c_check = fft_convolve_self(a_norm)
                obj_check = c_check.max()
                if obj_check < best_local_obj - 1e-14:
                    best_local_obj = obj_check
                    best_local = original * factor
            
            a[i] = best_local
            a = normalize_nonneg(a, S)
            
            if best_local != original:
                c = fft_convolve_self(a)
                if c.max() < best_obj - 1e-13:
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
        
        if not improved and sweep >= 5:
            break
    
    return best_a, best_obj

def get_initialization(name, n, S, rng):
    """Generate various initializations."""
    i = np.arange(n)
    if name == "uniform":
        return np.full(n, S / n)
    elif name == "random":
        a = rng.random(n)
        return a * (S / a.sum())
    elif name == "hamming":
        return np.hamming(n) * (S / np.hamming(n).sum())
    elif name == "hann":
        w = np.hanning(n)
        return w * (S / w.sum())
    elif name == "gaussian":
        sigma = n / 4
        a = np.exp(-0.5 * ((i - n/2) / sigma) ** 2)
        return a * (S / a.sum())
    elif name == "cosine_mix":
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * i / n + rng.random() * 2) + \
            0.1 * np.cos(2 * np.pi * 7 * i / n + rng.random() * 2)
        a = np.maximum(a, 0.01)
        return a * (S / a.sum())
    elif name == "raised_cosine":
        beta = 0.5
        t = i / n
        a = np.ones(n)
        transition = beta / 2
        for idx in range(n):
            tt = t[idx]
            if tt < transition:
                a[idx] = 0.5 * (1 + np.cos(np.pi * tt / transition))
            elif tt > 1 - transition:
                a[idx] = 0.5 * (1 + np.cos(np.pi * (tt - (1 - transition)) / transition))
        return a * (S / a.sum())
    elif name == "exponential_decay":
        a = np.exp(-i / (n * 0.3))
        return a * (S / a.sum())
    elif name == "barker_eval":
        # Evaluate Barker-like patterns
        a = np.zeros(n)
        a[0] = 1
        a[n-1] = 1
        for j in range(2, n-1, 2):
            a[j] = 0.5
        return a * (S / a.sum())
    else:
        return np.full(n, S / n)
