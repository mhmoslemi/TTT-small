```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using smooth log-sum-exp optimization
with multiplicative updates, Nesterov momentum, backtracking line search,
adaptive temperature, and multi-phase strategy.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_ab(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def logsumexp_smooth_obj(a, T, c=None):
    """Compute (1/T)*log(sum(exp(T*c))) where c = conv(a,a)."""
    n = len(a)
    if c is None:
        c = fft_convolve_self(a)
    cm = c.max()
    shifted = T * (c - cm)
    val = (cm + np.log(np.sum(np.exp(shifted)))) / T
    return val, c


def smooth_grad(a, T, c=None):
    """Compute gradient of (1/T)*log(sum(exp(T*c))) w.r.t. a."""
    n = len(a)
    if c is None:
        c = fft_convolve_self(a)
    cm = c.max()
    shifted = T * (c - cm)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    # grad_a_m = (1/T) * sum_k p_k * d(c_k)/da_m
    # d(c_k)/da_m = a_{k-m} + a_{k-m} (for self-conv, symmetric)
    # Approximation: grad ≈ (2/T) * conv(p, a_rev)[n-1:2n-1]
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n - 1).bit_length()
    P = fft(p, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(P * R))
    grad = (2.0 / T) * full_conv[n - 1:2 * n - 1]
    return grad


def nesterov_smooth_optimize(a0, time_budget, T_start=3.0, T_end=30.0,
                               eta0=0.5, momentum=0.85,
                               restart_every=2000, perturb_scale=0.03,
                               verbose=True):
    """
    Nesterov-accelerated smooth optimization with multiplicative updates.
    Uses log-sum-exp approximation to max with adaptive temperature.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 1e-15)
    a = a * (S / a.sum())
    
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum
    eta = eta0
    iter_no = 0
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        # Adaptive temperature
        progress = min(1.0, iter_no / max(1, int(time_budget * 200)))
        T_cur = T_start + (T_end - T_start) * progress
        
        # Compute smooth gradient
        grad = smooth_grad(a, T_cur, c)
        
        # Normalize gradient for stable updates
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad_norm = grad / gnorm
        else:
            break
        
        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad_norm
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm
        
        # Backtracking line search with multiplicative update
        cur_smooth_val, _ = logsumexp_smooth_obj(a, T_cur, c)
        accepted = False
        for bt in range(15):
            eta_try = eta * (0.5 ** bt)
            if eta_try < 1e-8:
                break
            a_trial = a * np.exp(-eta_try * v)
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial * (S / a_trial.sum())
            
            c_trial = fft_convolve_self(a_trial)
            smooth_trial, _ = logsumexp_smooth_obj(a_trial, T_cur, c_trial)
            
            if smooth_trial < cur_smooth_val:
                a = a_trial
                c = c_trial
                accepted = True
                break

            # Take a tiny step
            a = a * np.exp(-1e-6 * v)
            a = np.maximum(a, 1e-15)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
        
        # Track best (using hard max for evaluation)
        obj_hard = 2.0 * n * c.max() / (S * S)
        if obj_hard < best_obj:
            best_obj = obj_hard
            best_a = a.copy()
        
        # Decay step size slowly
        eta *= 0.9999
        
        # Periodic restart
        if iter_no % restart_every == 0 and iter_no > 0:
            noise = np.random.randn(n) * perturb_scale * S / np.sqrt(n)
            a = a + noise
            a = np.maximum(a, 1e-15)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.5
        
        iter_no += 1

            print(f"    iter={iter_no}, T={T_cur:.1f}, obj={obj_hard:.6f}, "

def coordinate_descent_polish(a0, time_budget, max_iter=50000, verbose=True):
    """
    Coordinate descent: for each index, try to reduce the max convolution value.
    Focus on indices that contribute most to the peak.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    start = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break
        
        # Find the peak
        k_max = np.argmax(c)
        c_max = c[k_max]
        
        # Find indices that contribute to this peak
        # c[k_max] = sum_j a[j] * a[k_max - j]
        # Indices j where 0 <= j < n and 0 <= k_max - j < n
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        
        # Try reducing the largest contributor
        contributions = np.zeros(hi - lo + 1)
        for j in range(lo, hi + 1):
            contributions[j - lo] = a[j] * a[k_max - j]
        
        if contributions.max() < 1e-10:
            continue
            
        # Index with largest contribution
        j_best = lo + np.argmax(contributions)
        
        # Try reducing a[j_best] by a small factor
        improved = False
        for factor in [0.9, 0.8, 0.7, 0.6]:
            a_trial = a.copy()
            a_trial[j_best] *= factor
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < c_max:
                a = a_trial
                c = c_trial
                obj = 2.0 * n * c.max() / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try a different peak
            # Reduce the second-largest peak
            c_sorted = np.sort(c)[::-1]
            if len(c_sorted) > 1:
                # Just perturb randomly
                j_rand = np.random.randint(n)
                a[j_rand] *= 0.8
                a = np.maximum(a, 1e-15)
                a = a * (S / a.sum())
                c = fft_convolve_self(a)
        
        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - start
            print(f"    coord_iter={it}, obj={best_obj:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_obj
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using smooth log-sum-exp optimization
with Nesterov momentum, backtracking line search, adaptive temperature,
multi-phase strategy, and coordinate descent polish.
Key improvements over previous: smooth differentiable objective, adaptive
step size, and temperature annealing for better convergence.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_convolve_ab(a, b):
    """Full convolution of a and b via FFT."""
    out_len = len(a) + len(b) - 1
    fsize = 1 << (out_len - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:out_len]


def compute_smooth_obj_and_grad(a, T, c=None):
    """
    Compute logsumexp(T*conv(a,a))/T and its gradient w.r.t. a.
    Uses the identity: grad_a[m] = (2/T) * (p * a_rev)[m+n-1]
    where p = softmax(T*c).
    """
    n = len(a)
    if c is None:
        c = fft_convolve_self(a)
    cm = c.max()
    shifted = T * (c - cm)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    p = exp_shifted / sum_exp
    obj_val = (cm + np.log(sum_exp)) / T

    # Gradient: grad_a[m] = (2/T) * sum_k p[k] * a[k-m] (for valid k-m)
    # = (2/T) * conv(p, a_rev)[m + n - 1]
    a_rev = a[::-1]
    fsize = 1 << (len(c) + n).bit_length()
    P = fft(p, fsize)
    R = fft(a_rev, fsize)
    full_conv = np.real(ifft(P * R))
    grad = (2.0 / T) * full_conv[n - 1:2 * n - 1]
    return obj_val, c, grad


def nesterov_smooth_optimize(a0, time_budget, T_start=3.0, T_end=50.0,
                              eta0=0.5, momentum=0.85,
                              restart_every=1500, perturb_scale=0.03,
                              verbose=True):
    """
    Nesterov-accelerated smooth optimization with multiplicative updates.
    Uses log-sum-exp approximation with adaptive temperature and
    backtracking line search.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    a = np.maximum(a, 1e-15)
    a = a * (S / a.sum())

    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()

    v = np.zeros(n)
    eta = eta0
    iter_no = 0
    start = time.monotonic()
    est_iters = int(time_budget * 150)  # rough estimate for temperature ramp

    while time.monotonic() - start < time_budget:
        # Adaptive temperature (exponential ramp)
        progress = min(1.0, iter_no / max(1, est_iters))
        T_cur = T_start * (T_end / T_start) ** progress

        # Compute smooth gradient
        smooth_val, c, grad = compute_smooth_obj_and_grad(a, T_cur, c)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad_norm = grad / gnorm

        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad_norm
        vnorm = np.linalg.norm(v)
        if vnorm < 1e-15:
            break
        v = v / vnorm

        # Backtracking line search
        accepted = False
        for bt in range(20):
            eta_try = eta * (0.5 ** bt)
            if eta_try < 1e-10:
                break
            a_trial = a * np.exp(-eta_try * v)
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial * (S / a_trial.sum())

            c_trial = fft_convolve_self(a_trial)
            smooth_trial, _, _ = compute_smooth_obj_and_grad(a_trial, T_cur, c_trial)

            if smooth_trial < smooth_val * (1.0 - 1e-12):
                a = a_trial
                c = c_trial
                accepted = True
                break

        if not accepted:
            a = a * np.exp(-1e-7 * v)
            a = np.maximum(a, 1e-15)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)

        obj_hard = 2.0 * n * c.max() / (S * S)
        if obj_hard < best_obj:
            best_obj = obj_hard
            best_a = a.copy()

        # Slow decay of step size
        eta *= 0.99995

        # Periodic restart with perturbation
        if iter_no > 0 and iter_no % restart_every == 0:
            scale = perturb_scale * (0.5 + 0.5 * progress)
            noise = np.random.randn(n) * scale * S / np.sqrt(n)
            a = a + noise
            a = np.maximum(a, 1e-15)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            v *= 0.3
            eta = max(eta, eta0 * 0.3)

        iter_no += 1
        if verbose and iter_no % 3000 == 0:
            elapsed = time.monotonic() - start
            print(f"    iter={iter_no}, T={T_cur:.1f}, obj={obj_hard:.6f}, "
                  f"best={best_obj:.6f}, eta={eta:.4f}, t={elapsed:.1f}s")

    return best_a, best_obj


def coordinate_polish(a0, time_budget, max_iter=200000, verbose=True):
    """
    Coordinate descent polish: iteratively reduce the peak of conv(a,a)
    by scaling down indices that contribute most to the current peak.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    start = time.monotonic()
    no_improve_count = 0
    max_no_improve = 5000

    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        # Find the current peak
        k_max = np.argmax(c)
        c_max = c[k_max]

        # Indices contributing to this peak
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        n_contrib = hi - lo + 1

        if n_contrib == 0:
            continue

        # Contribution of each index to c[k_max]
        # c[k_max] = sum_{j=lo}^{hi} a[j] * a[k_max - j]
        # Try reducing the index with the largest contribution
        # for efficiency, work with a small window around the peak
        j_vals = np.arange(lo, hi + 1)
        contribs = a[j_vals] * a[k_max - j_vals]

        # Sort by contribution (largest first)
        top_idx = np.argsort(contribs)[::-1]

        improved = False
        for idx_i in range(min(5, len(top_idx))):
            j = j_vals[top_idx[idx_i]]
            orig_val = a[j]
            if orig_val < 1e-12:
                continue

            for factor in [0.85, 0.7, 0.5]:
                a[j] = orig_val * factor
                a = np.maximum(a, 1e-15)
                a = a * (S / a.sum())
                c_new = fft_convolve_self(a)
                if c_new.max() < c_max * (1.0 - 1e-9):
                    c = c_new
                    obj = 2.0 * n * c.max() / (S * S)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                        no_improve_count = 0
                    else:
                        no_improve_count += 1
                    improved = True
                    break
                a[j] = orig_val  # revert

            if improved:
                break

        if not improved:
            # Try a random perturbation
            no_improve_count += 1
            j_rand = np.random.randint(n)
            a[j_rand] *= 0.8
            a = np.maximum(a, 1e-15)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)

        if no_improve_count > max_no_improve:
            # Larger restart
            noise = np.random.randn(n) * 0.05 * S / np.sqrt(n)
            a = a + noise
            a = np.maximum(a, 1e-15)
            a = a * (S / a.sum())
            c = fft_convolve_self(a)
            no_improve_count = 0
            max_no_improve = int(max_no_improve * 1.3)

        if verbose and it % 20000 == 0:
            elapsed = time.monotonic() - start
            print(f"    coord_it={it}, obj={best_obj:.6f}, no_improve={no_improve_count}, "
                  f"t={elapsed:.1f}s")

    return best_a, best_obj


def dual_peel(a0, time_budget, n_sweeps=100, verbose=True):
    """
    Dual-peel method: iteratively identify the set of "active" peaks
    and equalize them by adjusting the contributing indices.
    """
    a = a0.copy().astype(np.float64)
    n = len(a)
    S = a.sum()
    c = fft_convolve_self(a)
    best_obj = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    start = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break

        # Find all "hot" peaks (within 5% of max)
        c_max = c.max()
        threshold = c_max * 0.95
        hot_mask = c >= threshold
        n_hot = hot_mask.sum()

        if n_hot == 0:
            break

        # For each hot peak, compute the gradient contribution
        # and aggregate
        grad = np.zeros(n)
        for k in np.where(hot_mask)[0]:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += a[k - j]

        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        # Multiplicative update with small step
        for lr in [0.3, 0.15, 0.07]:
            a_trial = a * np.exp(-lr * grad)
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial * (S / a_trial.sum())
            c_trial = fft_convolve_self(a_trial)
            if c_trial.max() < c_max:
                a = a_trial
                c = c_trial
                obj = 2.0 * n * c.max() / (S * S)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                break

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"    dual_sweep={sweep}, n_hot={n_hot}, obj={best_obj:.6f}, "
                  f"t={elapsed:.1f}s")

    return best_a, best_obj

        a_best, obj_best = nesterov_smooth_optimize(
            global_a, phase2_time,
            T_start=20.0, T_end=100.0,
            eta0=0.3, momentum=0.9,
            restart_every=1500, perturb_scale=0.015,
            verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  Phase 2 result: {obj_best:.6f}")

    # --- Phase 3: Coordinate descent + dual peel (20% of time) ---
    phase3_time = 0.20 * (deadline - time.monotonic())
    print(f"\nPhase 3: Coord + dual ({phase3_time:.0f}s)")

    if global_a is not None and phase3_time > 15:
        # Split: 60% coord, 40% dual
        t_coord = phase3_time * 0.6
        t_dual = phase3_time * 0.4

        print(f"  Coordinate descent ({t_coord:.0f}s)")
        a_best, obj_best = coordinate_polish(
            global_a, t_coord, max_iter=100000, verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"    -> {obj_best:.6f}")

        t_dual = min(t_dual, deadline - time.monotonic() - 10)
        if t_dual > 5:
            print(f"  Dual peel ({t_dual:.0f}s)")
            a_best, obj_best = dual_peel(
                global_a, t_dual, n_sweeps=200, verbose=True
            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print(f"    -> {obj_best:.6f}")

    # --- Phase 4: Final fine-tuning (10% of time) ---
    phase4_time = deadline - time.monotonic() - 15.0
    print(f"\nPhase 4: Final polish ({phase4_time:.0f}s)")

    if global_a is not None and phase4_time > 5:
        a_best, obj_best = nesterov_smooth_optimize(
            global_a, phase4_time,
            T_start=50.0, T_end=150.0,
            eta0=0.15, momentum=0.93,
            restart_every=2000, perturb_scale=0.008,
            verbose=True
        )
        if obj_best < global_best:
            global_best = obj_best
            global_a = a_best.copy()
            print(f"  Phase 4 result: {obj_best:.6f}")

        # One more coordinate pass if time allows
        t_rem = deadline - time.monotonic() - 10
        if t_rem > 5:
            a_best, obj_best = coordinate_polish(
                global_a, t_rem, max_iter=50000, verbose=False
            )
            if obj_best < global_best:
                global_best = obj_best
                global_a = a_best.copy()
                print(f"  Final coord: {obj_best:.6f}")

    # --- Final verification ---
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_a = global_a.tolist()
    ver = evaluate_sequence(final_a)
    print(f"\nFINAL: local_best={global_best:.6f}, verified={ver:.6f}")

    if not np.isfinite(ver):
        final_a = [1.0] * 256
        ver = evaluate_sequence(final_a)
        print(f"Fallback: obj={ver:.6f}")

    return final_a
```
