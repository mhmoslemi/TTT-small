Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Strategy: FISTA with adaptive beta annealing + Newton refinement + support identification.
Key innovation: Hessian-vector products via FFT for second-order polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights):
    n = len(a)
    fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def smoothed_max_value(c, beta):
    c_max = c.max()
    return c_max + (1.0 / beta) * np.log(np.sum(np.exp(beta * (c - c_max))))

        # Anneal beta

dp[k]/da[i] = beta*p[k]*dc[k]/da[i] - p[k]*beta*sum_m p[m]*dc[m]/da[i]
= beta*p[k]*(dc[k]/da[i] - grad[i]/beta) ... hmm, this is getting circular.

Let me use a cleaner derivation. f = (1/beta)*log(Z) where Z = sum_k exp(beta*c[k]).
df/da[i] = (1/beta) * (1/Z) * sum_k beta*exp(beta*c[k]) * dc[k]/da[i]
= sum_k p[k] * dc[k]/da[i] ✓

d²f/da[i]da[j] = sum_k (dp[k]/da[i]) * dc[k]/da[j] + sum_k p[k] * d²c[k]/da[i]da[j]

For the second term: d²c[k]/da[i]da[j] is nonzero only when {i,j} is related to k. Specifically:
c[k] = sum_m a[m]*a[k-m]
d²c[k]/da[i]da[j] = delta related to i+j=k or |i-j| related to k.

This is getting very complicated. Let me just use the approximation:
H ≈ sum_k p[k] * g_k * g_k^T (the "covariance" of the gradient vectors)

This is the "Gauss-Newton" approximation, which works well for least-squares-like problems. It ignores the second derivative of c[k] (which is small) and the derivative of p[k] (which is of order 1/beta).

With this approximation:
(Hv)[i] = sum_k p[k] * g_k[i] * (sum_j g_k[j] * v[j])
= sum_k p[k] * g_k[i] * (Gv)[k]

where g_k[i] = dc[k]/da[i] ≈ 2*a[k-i] and (Gv)[k] = sum_j g_k[j]*v[j] ≈ 2*(a*v)[k]

So: (Hv)[i] = sum_k p[k] * 2*a[k-i] * 2*(a*v)[k]
= 4 * sum_k p[k] * (a*v)[k] * a[k-i]

Let u[k] = 4*p[k]*(a*v)[k]
Then (Hv)[i] = sum_k u[k]*a[k-i] = (u ⊛ a)[i] (cross-correlation)

This can be computed with two FFTs:
1. Compute (a*v)[k] for all k via FFT cross-correlation
2. Multiply pointwise by 4*p[k]
3. Cross-correlate with a via FFT

Let me verify the FFT computation for cross-correlation:
(a ⊛ v)[k] = sum_j a[j]*v[j-k] = sum_j a[j+k]*v[j]

Using FFT: if we zero-pad both to length L >= n + m - 1 (where m is the length of the other signal), then:
cross-correlation = IFFT(FFT(a, L) * conj(FFT(v, L)))

But for our case, both a and v have length n, and we want the result for k = 0 to 2n-2 (since c has length 2n-1).

Actually, let me just use the convolution approach:
(a*v)[k] = sum_j a[j]*v[k-j] (linear convolution)
This is: IFFT(FFT(a, L) * FFT(v, L))[:2n-1] where L >= 2n-1.

Then (Hv)[i] = sum_k u[k]*a[k-i] where u[k] = 4*p[k]*(a*v)[k]
This is: cross-correlation of u and a at lag i.
= IFFT(FFT(u, L2) * conj(FFT(a, L2)))[:n] where L2 >= len(u) + n - 1 = 3n-2.

So the Hessian-vector product requires:
- 1 FFT of size 2n (for a*v) + 1 IFFT
- 1 FFT of size 4n (for u ⊛ a) + 1 IFFT

For n=8192: FFT sizes 16384 and 32768. Very fast!

Let me now write the complete solution:

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Multi-phase: FISTA with beta annealing → Newton polish → coordinate refinement.
Key innovation: FFT-based Hessian-vector products for second-order polishing.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64).copy()
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_gradient_fft(a, weights, fsize=None):
    n = len(a)
    if fsize is None:
        fsize = 1 << (len(weights) + n - 2).bit_length()
    W = fft(weights, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def softmax_weights(c, beta):
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.full_like(c, 1.0 / len(c))
    return exp_bc / total


def hessian_vec_product(a, v, p, beta):
    """Compute (H v)[i] where H ≈ sum_k p[k] * g_k g_k^T (Gauss-Newton).
    g_k[i] ≈ 2*a[k-i] (cross-correlation kernel).
    (Hv)[i] = 4 * sum_k p[k] * (a*v)[k] * a[k-i]
    where (a*v)[k] is the linear convolution of a and v.
    """
    n = len(a)
    # Step 1: Compute u1[k] = (a*v)[k] for k=0..2n-2 (linear convolution)
    fsize1 = 1 << (2 * n - 1).bit_length()
    A = fft(a, fsize1)
    V = fft(v, fsize1)
    av = np.real(ifft(A * V))[:2*n-1]
    
    # Step 2: u2[k] = 4 * p[k] * av[k]
    u2 = 4.0 * p * av
    
    # Step 3: (Hv)[i] = sum_{k=0}^{2n-2} u2[k] * a[k-i] for i=0..n-1
    # This is cross-correlation of u2 and a: (u2 ⊛ a)[i] = sum_k u2[k]*a[k-i]
    # Using FFT: IFFT(FFT(u2) * conj(FFT(a)))
    fsize2 = 1 << (len(u2) + n - 1).bit_length()
    U2 = fft(u2, fsize2)
    A2 = fft(a, fsize2)
    hv = np.real(ifft(U2 * np.conj(A2)))[:n]
    
    return hv


def run_fista_anneal(a_init, n, S, max_time, rng, beta_start=100.0, beta_end=800.0):
    """FISTA with beta annealing, backtracking, and momentum restarts."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 1.0 / (beta_start * 2.0)
    no_improve_count = 0
    
    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / (max_time * 30.0))
        beta = beta_start + (beta_end - beta_start) * frac
        
        c = fft_convolve_self(y)
        cur_max = c.max()
        
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_local = y.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        w = softmax_weights(c, beta)
        grad = compute_gradient_fft(y, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        x_new = project_simplex(y - step * grad, S)
        
        # Periodic backtracking check
        if it % 20 == 0:
            c_new = fft_convolve_self(x_new)
            if c_new.max() > cur_max * 1.001 + 1e-10:
                step *= 0.7
                x_new = project_simplex(y - step * grad, S)
            elif c_new.max() < cur_max * 0.999 and step < 1.0 / (beta * 0.1):
                step *= 1.02
        
        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new
        
        # Momentum restart on stall
        if no_improve_count > 300:
            t = 1.0
            y = a.copy()
            no_improve_count = 0
            step = max(step * 0.8, 1e-6)
        
        # Random perturbation
        if it % 3000 == 0 and rng.random() < 0.2:
            noise_scale = 0.001 * S / np.sqrt(n)
            y = project_simplex(y + rng.normal(0, noise_scale, n), S)
            a = y.copy()
            t = 1.0
    
    return best_local, best_score


def run_newton_polish(a_init, n, S, max_time, beta=300.0):
    """Second-order polish using Gauss-Newton Hessian approximation."""
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_max = c.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_local = a.copy()
        
        p = softmax_weights(c, beta)
        grad = compute_gradient_fft(a, p)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Compute H * grad to estimate curvature
        Hg = hessian_vec_product(a, grad, p, beta)
        curvature = np.dot(grad, Hg) / (gnorm * gnorm + 1e-30)
        
        if curvature > 1e-8:
            direction = -grad / max(curvature, 1e-6)
        else:
            direction = -grad / gnorm
        
        # Line search
        best_alpha = 0.0
        best_val = cur_max
        for alpha in [0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
            trial = project_simplex(a + alpha * direction, S)
            c_trial = fft_convolve_self(trial)
            if c_trial.max() < best_val - 1e-12:
                best_val = c_trial.max()
                best_alpha = alpha
        
        if best_alpha > 0:
            a = project_simplex(a + best_alpha * direction, S)
        else:
            # Try pure gradient step
            for scale in [0.01, 0.005, 0.001]:
                trial = project_simplex(a - scale * grad / gnorm, S)
                c_trial = fft_convolve_self(trial)
                if c_trial.max() < cur_max - 1e-12:
                    a = trial
                    break
            else:
                break  # No improvement possible
    
    return best_local, best_score
