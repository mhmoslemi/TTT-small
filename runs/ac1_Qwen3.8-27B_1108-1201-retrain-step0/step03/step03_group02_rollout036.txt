"""

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_objective(a):
    """Compute objective without normalization constraint."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (S * S)


def fft_grad_soft(a, c, beta):
    """
    Gradient of (1/beta)*log(sum exp(beta*c[k])) w.r.t. a.
    grad[m] = 2 * sum_k softmax_k * a[k-m] (cross-correlation).
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_toppk(a, c, top_k):
    """Gradient using top-k active lags with weighted contribution."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    total = top_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    # Use exponential weighting to focus on the largest peaks
    weights = top_vals / total

    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_only_max(a, c):
    """Subgradient of max(conv) - focuses only on the argmax."""
    n = len(a)
    k_max = int(np.argmax(c))
    # Subgradient: grad[m] = 2*a[k_max - m] for valid m
    # This is 2 * correlation of a with delta at k_max
    grad = np.zeros(n)
    for m in range(n):
        idx = k_max - m
        if 0 <= idx < n:
            grad[m] = 2.0 * a[idx]
    return grad


def fft_grad_weighted_peaks(a, c, top_k, temperature=1.0):
    """Weighted peak gradient with temperature control."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    
    # Soft weighting based on temperature
    max_val = top_vals[0]
    weights = np.exp(-temperature * (max_val - top_vals) / max(max_val, 1e-15))
    weights = weights / weights.sum()
    
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_adaptive_fista(a_init, n, S, max_time, rng, 
                        beta_init=5.0, beta_max=2000.0, 
                        beta_growth=0.01, noise_scale=0.003):
    """
    FISTA with aggressive adaptive beta and backtracking.
    Key improvement: much faster beta growth for sharper max approximation.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stall_count = 0
    last_improvement = best_score

    while time.monotonic() - start < max_time:
        it += 1
        # Aggressive beta growth
        beta = min(beta_max, beta_init * np.exp(beta_growth * it))
        
        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_local = y.copy()
            stall_count = 0
            last_improvement = cur
        else:
            stall_count += 1

        grad = fft_grad_soft(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step size with backtracking
        step = 1.0 / (beta * 2.0)
        step = min(step, S / (2.0 * gnorm))
        step = max(step, 1e-8)
        
        x_new = project_simplex(y - step * grad, S)
        
        # Verify improvement with actual objective
        if it % 20 == 1:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
                stall_count = 0

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic perturbation to escape local minima
        if it % 500 == 0 and rng.random() < 0.3:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
            a = y.copy()

        # If stalled for too long, restart momentum
        if stall_count > 2000:
            t = 1.0
            y = a.copy()
            stall_count = 0
            noise = rng.normal(0, noise_scale * 2 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)

    return best_local, best_score

def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
- Start from parent sequence for local refinement
- Multi-phase optimization: adaptive FISTA with aggressive beta scheduling
- Peak-targeted subgradient with top-k identification and exponential weighting
- Spectral equalization as refinement
- Multi-restart with diverse perturbations to escape local minima
- Final high-precision polish with very high beta

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_grad_soft(a, c, beta):
    """
    Gradient of smooth max approximation (1/beta)*log(sum exp(beta*c[k])).
    grad[m] = 2 * sum_k w[k] * a[k-m] where w is softmax(beta*c).
    """
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total

    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_grad_toppk_exp(a, c, top_k, tau=1.0):
    """
    Gradient using top-k peaks with exponential weighting.
    tau controls how much we focus on the largest peak vs others.
    """
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    max_val = top_vals[0]
    if max_val < 1e-30:
        return np.zeros(n)
    # Exponential weighting: weights proportional to exp(-tau * (max - val)/max)
    weights = np.exp(-tau * (max_val - top_vals) / max_val)
    weights = weights / weights.sum()

    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_indices):
        w_full[int(k)] = weights[i]

    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def run_fista_adaptive(a_init, n, S, max_time, rng,
                       beta_init=5.0, beta_max=5000.0,
                       beta_rate=0.003, noise_scale=0.002,
                       use_toppk=False, top_k=16, tau=0.5):
    """
    FISTA with adaptive beta and backtracking line search.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    stall = 0
    beta = beta_init

    while time.monotonic() - start < max_time:
        it += 1

        # Adaptive beta growth: faster when stalled
        if stall > 500:
            beta = min(beta_max, beta * 1.5)
            stall = 0
            t = 1.0
            y = a.copy()
        else:
            beta = min(beta_max, beta_init * np.exp(beta_rate * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = y.copy()
            stall = 0
        else:
            stall += 1

        # Choose gradient method
        if use_toppk and beta > 50:
            grad = fft_grad_toppk_exp(y, c, top_k, tau)
        else:
            grad = fft_grad_soft(y, c, beta)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Adaptive step with backtracking
        step = 1.0 / (2.0 * beta * (1.0 + 0.0001 * it))
        step = min(step, 0.5 * S / gnorm)
        step = max(step, 1e-9)

        x_new = project_simplex(y - step * grad, S)

        # Periodic verification
        if it % 25 == 1:
            c_x = fft_convolve_self(x_new)
            if c_x.max() < best_score:
                best_score = c_x.max()
                best_local = x_new.copy()
                stall = 0

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        # Periodic perturbation
        if it % 800 == 0 and rng.random() < 0.25:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
            a = y.copy()

    return best_local, best_score


def run_peak_elimination(a_init, n, S, max_time, rng,
                         top_k=4, tau=0.3, noise_scale=0.001):
    """
    Focused peak elimination: iteratively identify and reduce the
    largest convolution peaks using direct subgradient steps.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    y = a.copy()
    t = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1

        k_use = min(256, top_k + (it // 50))
        tau_use = max(0.05, tau * np.exp(-0.0005 * it))

        c = fft_convolve_self(y)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = y.copy()

        grad = fft_grad_toppk_exp(y, c, k_use, tau_use)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Step size based on peak value and gradient
        step = cur / (gnorm * gnorm + 1e-10)
        step = min(step, S / (np.sqrt(n) * 2.0))
        step = max(step, 1e-10)

        x_new = y - step * grad
        x_new = project_simplex(x_new, S)

        # Backtracking verification
        if it % 3 == 0:
            c_new = fft_convolve_self(x_new)
            if c_new.max() > cur * 1.01:
                x_new = y - 0.25 * step * grad
                x_new = project_simplex(x_new, S)
            elif c_new.max() < best_score:
                best_score = c_new.max()
                best_local = x_new.copy()

        # Nesterov
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = x_new + ((t - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t = t_new

        if it % 2000 == 0 and rng.random() < 0.15:
            noise = rng.normal(0, noise_scale * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t = 1.0
            a = y.copy()

    return best_local, best_score


def run_spectral_flattening(a_init, n, S, max_time, rng, alpha_init=0.3):
    """
    Spectral flattening: modify the magnitude of the DFT to reduce
    variance of |A(f)|^2, which makes the convolution flatter.
    """
    a = np.clip(a_init, 0.0, 1000.0).copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)

    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    fsize = 1 << (2 * n - 1 - 1).bit_length()

    while time.monotonic() - start < max_time:
        it += 1
        alpha = alpha_init * np.exp(-0.002 * it)

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score - 1e-10:
            best_score = cur
            best_local = a.copy()

        # Compute DFT
        A = fft(a, fsize)
        mag = np.abs(A)
        phase = np.angle(A)
        power = mag ** 2
        mean_power = power.mean()

        if mean_power < 1e-30:
            break

        # Compress dynamic range of magnitude
        # New magnitude: mag * (mean / mag^2)^(alpha/2)
        # This reduces large mag values and increases small ones
        ratio = mean_power / (power + 1e-10)
        compression = np.power(ratio, alpha / 4.0)
        mag_new = mag * compression

        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_candidate = np.real(ifft(A_new, fsize))[:n]
        a_candidate = project_simplex(a_candidate, S)

        # Verify
        c_new = fft_convolve_self(a_candidate)
        if c_new.max() < cur:
            a = a_candidate
        else:
            # Blend
            blend = 0.3
            a_candidate = project_simplex((1.0 - blend) * a + blend * a_candidate, S)
            c_blend = fft_convolve_self(a_candidate)
            if c_blend.max() < cur:
                a = a_candidate

        # Small perturbation to avoid getting stuck
        if it % 200 == 0:
            noise = rng.normal(0, 0.001 * S / np.sqrt(n), n)
            a = project_simplex(a + noise, S)

    return best_local, best_score


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

        obj6 = c6.max()
        print("  Final FISTA: %.8f" % obj6)
        if obj6 < global_best:
            global_best = obj6
            global_a = a6.copy()

        remaining = deadline - time.monotonic()
        if remaining > 5:
            a7, s7 = run_peak_elimination(global_a, n, S, remaining, rng,
                                           top_k=32, tau=0.1, noise_scale=0.00005)
            a7 = safe_normalize(a7, n)
            c7 = fft_convolve_self(a7)
            obj7 = c7.max()
            print("  Final peak: %.8f" % obj7)
            if obj7 < global_best:
                global_best = obj7
                global_a = a7.copy()

    # === Verification ===
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("\n=== VERIFICATION ===")
    print("Internal score: %.8f" % global_best)
    print("Verified score: %.8f" % (ver if np.isfinite(ver) else float('inf')))

    if not np.isfinite(ver):
        print("  Rejected! Using fallback.")
        global_a = np.ones(8192, dtype=np.float64)
        final_list = [float(x) for x in global_a]
        ver = evaluate_sequence(final_list)
        print("  Fallback score: %.6f" % ver)

    print("Final: length=%d, score=%.8f" % (len(final_list), ver))
    return final_list
