"""
Minimize 2*n*max(conv(a,a)) / sum(a)^2 over non-negative sequences.

Key insights:
1. Asymmetric ramp/step initializations are far superior to symmetric ones
2. Two-level step functions with H/L ratio near 2 give initial objective ~1.78
3. Smooth ramps (linear, exponential) allow the optimizer to find better shapes
4. Adaptive smoothed-max FISTA with backtracking + coordinate descent polish
5. The optimal shape is a smooth asymmetric ramp, not a sharp step
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def fft_subgradient(a, c, top_k):
    """
    Subgradient of max(c) using top-k active lags.
    grad[j] = (1/|K|) * sum_{k in K} 2*a[k-j] (with valid range check)
    """
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    
    # For each active lag k, the contribution to grad[j] is 2*a[k-j] if 0<=k-j<=n-1
    # Equivalently: grad[j] = 2 * sum_{k in K} a[k-j] * [0 <= k-j <= n-1]
    # This is a cross-correlation between a and the indicator of K (placed at positions K)
    
    w_full = np.zeros(2 * n - 1)
    for k in top_indices:
        w_full[int(k)] = 1.0 / len(top_indices)
    
    fsize = 1 << (len(w_full) + n - 1).bit_length()
    W = fft(w_full, fsize)
    A = fft(a, fsize)
    # cross-correlation: grad[j] = sum_k w[k] * a[k-j]
    # = ifft(fft(w) * conj(fft(a)))[j]
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def fft_smooth_grad(a, c, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a."""
    n = len(a)
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exp_bc / total
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    grad = 2.0 * np.real(ifft(W * np.conj(A)))[:n]
    return grad


def compute_objective_maxc(a):
    """Return max of self-convolution (objective when sum = sqrt(2n))."""
    c = fft_convolve_self(a)
    return c.max()


def safe_normalize(a, n):
    """Clip to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)


def make_two_level_step(n, ratio, step_frac):
    """Two-level step: 1 for i < step_frac*n, ratio for i >= step_frac*n."""
    a = np.ones(n)
    m = int(n * step_frac)
    a[m:] = ratio
    return a


def make_linear_ramp(n, ratio):
    """Linear ramp from 1 to ratio."""
    a = np.linspace(1.0, ratio, n)
    return a


def make_smooth_step(n, ratio, step_frac, width):
    """Smooth step using sigmoid transition."""
    x = np.arange(n, dtype=np.float64) / n
    center = step_frac
    a = 1.0 + (ratio - 1.0) / (1.0 + np.exp(-(x - center) / width))
    return np.clip(a, 0.1, None)


def make_quadratic_ramp(n, ratio):
    """Quadratic ramp (slower start, faster end)."""
    t = np.linspace(0, 1, n)
    a = 1.0 + (ratio - 1.0) * t * t
    return a


def run_fista_smooth(a_init, n, max_time, rng, beta_init=5.0, beta_max=500.0):
    """FISTA with adaptive smoothed max and backtracking."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 1.0 / (4.0 * S)  # Initial step
    
    while time.monotonic() - start < max_time:
        it += 1
        beta = min(beta_max, beta_init * (1.0 + 0.005 * it))
        
        c = fft_convolve_self(y)
        cur_maxc = c.max()
        if cur_maxc < best_score:
            best_score = cur_maxc
            best_local = y.copy()
        
        grad = fft_smooth_grad(y, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Backtracking line search
        s_try = step
        improved = False
        for _ in range(20):
            x_new = project_simplex(y - s_try * grad, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_maxc + 1e-10:
                improved = True
                break
            s_try *= 0.5
        
        if not improved:
            x_new = project_simplex(y - step * 0.01 * grad, S)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Occasionally reset momentum to escape local minima
        if it % 500 == 0 and rng.random() < 0.1:
            t_mom = 1.0
            y = a.copy()
        
        # Adaptive step: increase if we used full step, decrease otherwise
        if s_try > step * 0.9:
            step = min(step * 1.05, 1.0 / (2.0 * S))
        else:
            step = max(s_try, 1e-10)
    
    return best_local, best_score


def run_peak_pincer_bt(a_init, n, max_time, rng, top_k_init=4, top_k_max=64):
    """Peak-pincer with backtracking line search."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    y = a.copy()
    t_mom = 1.0
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    it = 0
    step = 2.0 * S / (n * top_k_init)
    
    while time.monotonic() - start < max_time:
        it += 1
        k = min(top_k_max, top_k_init + it // 100)
        
        c = fft_convolve_self(y)
        cur_maxc = c.max()
        if cur_maxc < best_score:
            best_score = cur_maxc
            best_local = y.copy()
        
        grad = fft_subgradient(y, c, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Normalize gradient for step control
        grad_normed = grad / max(gnorm, 1e-15)
        
        # Backtracking
        s_try = step
        improved = False
        for _ in range(15):
            x_new = project_simplex(y - s_try * grad_normed, S)
            c_new = fft_convolve_self(x_new)
            if c_new.max() <= cur_maxc + 1e-10:
                improved = True
                break
            s_try *= 0.5
        
        if not improved:
            break  # Can't make progress
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        if s_try > step * 0.9:
            step = min(step * 1.02, 5.0 * S / n)
        else:
            step = max(s_try * 1.1, 1e-8)
        
        # Noise injection to escape local minima
        if it % 2000 == 0 and rng.random() < 0.08:
            noise = rng.normal(0, 0.003 * S / np.sqrt(n), n)
            y = project_simplex(y + noise, S)
            t_mom = 1.0
    
    return best_local, best_score


def coordinate_descent(a_init, n, max_time, rng):
    """Coordinate descent: move mass from high-gradient to low-gradient positions."""
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a_init, n)
    start = time.monotonic()
    best_score = np.inf
    best_local = a.copy()
    
    # Precompute which positions matter for the top peaks
    while time.monotonic() - start < max_time:
        c = fft_convolve_self(a)
        cur_maxc = c.max()
        if cur_maxc < best_score:
            best_score = cur_maxc
            best_local = a.copy()
        
        # Find top peaks
        top_k = 8
        top_indices = np.argsort(c)[::-1][:top_k]
        
        # Compute gradient
        grad = fft_subgradient(a, c, top_k)
        
        # Find best single-coordinate move (respecting sum constraint)
        # Move delta from position i to position j
        # Effect on b[k]: delta * (grad[i] - grad[j]) approximately
        # We want to minimize max(b), so move from high grad to low grad
        
        # Sample a few candidate moves
        n_tries = 50
        best_move = 0  # no move
        best_delta = 0
        best_score_try = cur_maxc
        
        for _ in range(n_tries):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            a_i = a[i]
            a_j = a[j]
            if a_i < 1e-10:
                continue
            
            # Try moving a small fraction of a[i] to a[j]
            frac = 0.01 + 0.04 * rng.random()
            delta = frac * a_i
            
            a_try = a.copy()
            a_try[i] -= delta
            a_try[j] += delta
            a_try = np.maximum(a_try, 0.0)
            # Renormalize (shouldn't be needed if we're just redistributing)
            s_try = a_try.sum()
            if abs(s_try - S) > 1e-10:
                a_try = a_try * S / s_try
            
            c_try = fft_convolve_self(a_try)
            if c_try.max() < best_score_try:
                best_score_try = c_try.max()
                best_move = (i, j)
                best_delta = delta
        
        if best_move == 0:
            # No improvement found, try a random perturbation
            break
        
        a[best_move[0]] -= best_delta
        a[best_move[1]] += best_delta
        a = np.maximum(a, 0.0)
        a = a * S / a.sum()
    
    return best_local, best_score


def run_optimizer(a_init, n, max_time, rng):
    """Combined optimizer: FISTA -> peak-pincer -> coordinate descent."""
    start = time.monotonic()
    best_score = np.inf
    best_local = safe_normalize(a_init, n)
    
    # Phase 1: FISTA (50% of time)
    seg1 = max_time * 0.45
    a1, s1 = run_fista_smooth(a_init, n, seg1, rng, beta_init=10.0, beta_max=300.0)
    if s1 < best_score:
        best_score = s1
        best_local = a1.copy()
    
    # Phase 2: Peak-pincer (30% of time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        seg2 = remaining * 0.45
        a2, s2 = run_peak_pincer_bt(a1, n, seg2, rng, top_k_init=4, top_k_max=32)
        if s2 < best_score:
            best_score = s2
            best_local = a2.copy()
    
    # Phase 3: More peak-pincer with higher k (remaining)
    remaining2 = max_time - (time.monotonic() - start)
    if remaining2 > 5:
        a3, s3 = run_peak_pincer_bt(best_local, n, remaining2, rng, top_k_init=16, top_k_max=128)
        if s3 < best_score:
            best_score = s3
            best_local = a3.copy()
    
    # Phase 4: Coordinate descent polish
    remaining3 = max_time - (time.monotonic() - start)
    if remaining3 > 3:
        a4, s4 = coordinate_descent(best_local, n, remaining3, rng)
        if s4 < best_score:
            best_score = s4
            best_local = a4.copy()
    
    return best_local, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses asymmetric ramp/step initializations with multi-phase optimization.
    """
    rng = np.random.default_rng(seed)
    budget_s = float(budget_s)
    deadline = time.monotonic() + max(0.0, budget_s - 20.0)
    
    global_best = np.inf
    global_a = None
    
    # Collect initializations
    inits = []
    
    # Try to use the parent sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        if given.size >= 64 and np.sum(np.clip(given, 0, 1000)) > 0:
            inits.append(("parent", np.clip(given, 0, 1000)))
    except Exception:
        pass
    
    # Determine n from parent or default
    if inits:
        n_default = len(inits[0][1])
    else:
        n_default = 8192
    
    # Generate asymmetric initializations for the default n
    n = n_default
    for ratio in [1.3, 1.5, 1.7, 2.0, 2.5, 3.0]:
        for step_frac in [0.35, 0.45, 0.5, 0.55, 0.65]:
            inits.append((
                "step_r%.1f_f%.2f" % (ratio, step_frac),
                make_two_level_step(n, ratio, step_frac)
            ))
    
    for ratio in [1.5, 2.0, 2.5, 3.0]:
        inits.append(("ramp_r%.1f" % ratio, make_linear_ramp(n, ratio)))
        inits.append(("quad_r%.1f" % ratio, make_quadratic_ramp(n, ratio)))
    
    for ratio in [1.5, 2.0]:
        for width in [0.05, 0.1, 0.2]:
            inits.append((
                "smooth_r%.1f_w%.1f" % (ratio, width),
                make_smooth_step(n, ratio, 0.5, width)
            ))
    
    # Random perturbations of the ramp
    base_ramp = make_linear_ramp(n, 2.0)
    for _ in range(4):
        noise = rng.normal(0, 0.1, n)
        inits.append(("rand_ramp", np.maximum(base_ramp + noise, 0.1)))
    
    # Allocate time
    n_inits = len(inits)
    usable_time = max(10.0, deadline - time.monotonic())
    
    # We'll spend most time on the best few candidates
    # Phase 1: Quick screen of all inits (short time each)
    screen_time = min(3.0, usable_time / n_inits)
    print("=== Phase 1: Screening %d inits, %.1f sec each ===" % (n_inits, screen_time))
    
    results = []
    for name, arr in inits:
        if time.monotonic() > deadline - 120:
            break
        n_i = len(arr)
        if n_i < 64:
            continue
        a_opt, score = run_optimizer(arr, n_i, screen_time, rng)
        a_norm = safe_normalize(a_opt, n_i)
        c = fft_convolve_self(a_norm)
        obj = c.max()
        
        if obj < global_best:
            global_best = obj
            global_a = a_norm.copy()
            print("  [screen %s] n=%d: obj=%.6f" % (name, n_i, obj))
        
        results.append((obj, a_norm.copy()))
    
    print("=== After screening: best=%.6f ===" % global_best)
    
    # Phase 2: Deep optimization on the top 3 candidates
    if results:
        results.sort(key=lambda x: x[0])
        top_candidates = results[:3]
        
        remaining = deadline - time.monotonic()
        per_candidate = max(10.0, (remaining - 60.0) / max(1, len(top_candidates)))
        
        print("=== Phase 2: Deep optimization on top %d, %.1f sec each ===" %
              (len(top_candidates), per_candidate))
        
        for rank, (score_init, a_cand) in enumerate(top_candidates):
            if time.monotonic() > deadline - 60:
                break
            n_c = len(a_cand)
            print("  Candidate %d (initial score=%.6f):" % (rank, score_init))
            
            # Run full optimization with different parameters
            a_opt, score = run_optimizer(a_cand, n_c, per_candidate, rng)
            a_norm = safe_normalize(a_opt, n_c)
            c = fft_convolve_self(a_norm)
            obj = c.max()
            
            if obj < global_best:
                global_best = obj
                global_a = a_norm.copy()
                print("    Deep opt: obj=%.6f" % obj)
            else:
                print("    Deep opt: obj=%.6f (no improvement)" % obj)
    
    # Phase 3: Final polish on the global best
    if global_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_best = len(global_a)
            print("=== Phase 3: Final polish (%.1f sec) ===" % remaining)
            
            # Polish 1: Peak-pincer with very high k
            seg = remaining * 0.4
            a_p1, s1 = run_peak_pincer_bt(global_a, n_best, seg, rng,
                                           top_k_init=32, top_k_max=256)
            a_p1 = safe_normalize(a_p1, n_best)
            c1 = fft_convolve_self(a_p1)
            if c1.max() < global_best:
                global_best = c1.max()
                global_a = a_p1.copy()
                print("  Polish 1: %.6f" % global_best)
            
            # Polish 2: FISTA with high beta
            remaining2 = deadline - time.monotonic()
            if remaining2 > 5:
                seg2 = remaining2 * 0.4
                a_p2, s2 = run_fista_smooth(global_a, n_best, seg2, rng,
                                             beta_init=100.0, beta_max=1000.0)
                a_p2 = safe_normalize(a_p2, n_best)
                c2 = fft_convolve_self(a_p2)
                if c2.max() < global_best:
                    global_best = c2.max()
                    global_a = a_p2.copy()
                    print("  Polish 2: %.6f" % global_best)
            
            # Polish 3: Coordinate descent
            remaining3 = deadline - time.monotonic()
            if remaining3 > 3:
                a_p3, s3 = coordinate_descent(global_a, n_best, remaining3, rng)
                a_p3 = safe_normalize(a_p3, n_best)
                c3 = fft_convolve_self(a_p3)
                if c3.max() < global_best:
                    global_best = c3.max()
                    global_a = a_p3.copy()
                    print("  Polish 3: %.6f" % global_best)
    
    # Fallback
    if global_a is None:
        global_a = make_linear_ramp(8192, 2.0)
    
    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(8192, dtype=np.float64)
    
    final_list = [float(x) for x in global_a]
    ver = evaluate_sequence(final_list)
    print("Final: len=%d, internal=%.6f, verified=%.6f" %
          (len(final_list), global_best, ver if np.isfinite(ver) else float('inf')))
    
    if not np.isfinite(ver):
        print("  evaluate_sequence rejected, using fallback")
        global_a = make_linear_ramp(8192, 2.0)
        final_list = [float(x) for x in global_a]
    
    return final_list
