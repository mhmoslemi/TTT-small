def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if S <= 0:
        return np.zeros(n)
    # Water-filling
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    rho = int(np.max(np.where(valid)[0]) + 1) if np.any(valid) else n
    rho_f = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - rho_f, 0.0)


def linear_convolve_fft(a):
    """Compute linear self-convolution of a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def lse_gradient_toppk(a, c, beta, top_k):
    """Compute LSE gradient using only top-k lags (faster but approximate)."""
    n = len(a)
    c_max = c.max()
    top_idx = np.argsort(c)[::-1][:top_k]
    top_c = c[top_idx].copy()
    exp_bc = np.exp(beta * (top_c - c_max))
    total = exp_bc.sum()
    if total < 1e-300:
        return np.zeros(n), c_max
    w_top = exp_bc / total
    
    # Build full w array (sparse)
    w_full = np.zeros(2 * n - 1)
    for i, k in enumerate(top_idx):
        w_full[int(k)] = w_top[i]
    
    fsize2 = 1 << (3 * n - 2).bit_length()
    W2 = fft(w_full, fsize2)
    A2 = fft(a, fsize2)
    term1 = np.real(ifft(np.conj(W2) * A2))[:n]
    term2 = np.real(ifft(W2 * A2))[:n]
    
    return term1 + term2, c_max


def spectral_flatten(a, N_fft, n_target_flat, rng, alpha=0.1):
    """
    Flatten the power spectrum |FFT(a)|^2 to reduce autocorrelation peaks.
    Returns modified a.
    """
    n = len(a)
    S = a.sum()
    a_pad = np.zeros(N_fft)
    a_pad[:n] = a
    A = fft(a_pad)
    P = np.abs(A) ** 2
    c = np.real(ifft(P))[:2 * n - 1]
    c_max = c.max()
    
    # Find the lags with highest c values and create a "repulsive" pattern
    n_top = min(n_target_flat, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:n_top]
    top_vals = c[top_idx]
    total = top_vals.sum()
    if total < 1e-30:
        return a, c_max
    
    weights = top_vals / total
    m = np.arange(N_fft)
    cos_eff = np.zeros(N_fft)
    for i, k in enumerate(top_idx):
        cos_eff += weights[i] * np.cos(2.0 * np.pi * m * int(k) / N_fft)
    
    # Reduce power at frequencies that contribute to high autocorrelation
    P_new = np.maximum(P * (1.0 - alpha * cos_eff), 0.0)
    
    # Also add a small "Burg" step: increase power where it's low
    P_mean = P.mean()
    P_new = P_new + 0.01 * alpha * P_mean * (1.0 - P_new / (P_mean + 1e-10))
    P_new = np.maximum(P_new, 0.0)
    
    phase = np.angle(A)
    A_new = np.sqrt(P_new) * np.exp(1j * phase)
    a_new = np.real(ifft(A_new))[:n]
    a_new = project_nonneg_sum(a_new, S)
    return a_new, c_max


def nesterov_lse_full(a_init, n, S, max_time, rng, beta_start=5.0, beta_end=200.0,
                       n_beta_steps=500, use_full_grad=False, top_k_start=20, top_k_end=200):
    """
    Nesterov-accelerated LSE minimization with adaptive beta.
    """
    a = project_nonneg_sum(a_init.copy(), S)
    y = a.copy()
    t_mom = 1.0
    start_time = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    
    fsize2 = 1 << (3 * n - 2).bit_length()  # Precompute FFT size
    
    it = 0
    while time.monotonic() - start_time < max_time:
        it += 1
        
        # Adaptive beta
        frac = min(1.0, it / n_beta_steps)
        beta = beta_start + (beta_end - beta_start) * frac
        beta = min(beta, 500.0)
        
        # Adaptive top_k
        top_k = int(top_k_start + (top_k_end - top_k_start) * min(1.0, it / (2 * n_beta_steps)))
        
        # Compute convolution
        c = linear_convolve_fft(y)
        cur_max = c.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = y.copy()
        
        # Compute gradient
        if use_full_grad and it % 5 == 0:
            grad, c_max = lse_gradient_full(y, c, beta)
        else:
            grad, c_max = lse_gradient_toppk(y, c, beta, top_k)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Nesterov step with backtracking
        grad_normed = grad / gnorm
        step = 2.0 * S / (n * (1.0 + 0.01 * it))
        
        # Try a few step sizes (backtracking-like)
        best_x = None
        best_x_score = np.inf
        for trial in range(3):
            x_new = project_nonneg_sum(y - step * grad_normed, S)
            c_x = linear_convolve_fft(x_new)
            x_score = c_x.max()
            if x_score < best_x_score:
                best_x_score = x_score
                best_x = x_new.copy()
            step *= 0.7  # Reduce step
        
        x_new = best_x
        if best_x_score < best_score:
            best_score = best_x_score
            best_a = x_new.copy()
        
        # Nesterov momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - a)
        y = np.maximum(y, 0.0)
        a = x_new
        t_mom = t_new
        
        # Periodic restart
        if it % 2000 == 0 and rng.random() < 0.2:
            noise_scale = 0.005 * S / np.sqrt(n)
            y = project_nonneg_sum(y + rng.standard_normal(n) * noise_scale, S)
            t_mom = 1.0
    
    return best_a, best_score


def nuclea...(a_init, n, S, max_time, rng, beta=50.0):
    """
    NUCLEA-style alternating: alternate between minimizing LSE and projecting.
    Uses the structure that minimizing max c[k] is equivalent to finding a point
    where the subgradient of max contains the normal cone of the constraint.
    """
    a = project_nonneg_sum(a_init.copy(), S)
    start_time = time.monotonic()
    best_score = np.inf
    best_a = a.copy()
    
    fsize2 = 1 << (3 * n - 2).bit_length()
    
    # Initialize with the argmax lag
    c = linear_convolve_fft(a)
    active_lags = [int(np.argmax(c))]
    
    step = S / n * 0.1
    it = 0
    
    while time.monotonic() - start_time < max_time:
        it += 1
        c = linear_convolve_fft(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
        
        # Update active set: include all lags within 1% of max
        threshold = cur * 0.99
        active_mask = c >= threshold
        active_lags = np.where(active_mask)[0]
        
        if len(active_lags) == 0:
            active_lags = [int(np.argmax(c))]
        
        # Gradient of average of active c[k] values
        w = np.zeros(2 * n - 1)
        active_count = len(active_lags)
        for k in active_lags:
            w[int(k)] = 1.0 / active_count
        
        # Compute gradient via FFT
        W = fft(w, fsize2)
        A = fft(a, fsize2)
        term1 = np.real(ifft(np.conj(W) * A))[:n]
        term2 = np.real(ifft(W * A))[:n]
        grad = term1 + term2
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Proximal gradient step with backtracking
        alpha = step
        for _ in range(5):
            x_new = project_nonneg_sum(a - alpha * grad, S)
            c_new = linear_convolve_fft(x_new)
            if c_new.max() < cur:
                a = x_new
                break
            alpha *= 0.5
        else:
            a = project_nonneg_sum(a - alpha * grad, S)
        
        # Adaptive step size
        c_check = linear_convolve_fft(a)
        if c_check.max() < cur * 0.999:
            step = min(step * 1.1, S / n)
        else:
            step *= 0.8
    
    return best_a, best_score


def make_msequence_init(n, rng):
    """Create an m-sequence inspired initialization."""
    t = np.arange(n) / n
    # Chirp-like modulation
    a = 0.5 + 0.4 * np.sin(2 * np.pi * (4 * t + 0.5 * t * t)) + 0.15 * rng.standard_normal(n)
    a = np.clip(a, 0.01, 1.0)
    return a


def make_zadoff_chu_like(n, rng):
    """Create a Zadoff-Chu inspired initialization (complex, take magnitude)."""
    root = rng.integers(1, max(2, n // 4))
    t = np.arange(n)
    # Zadoff-Chu: x[n] = exp(-pi*i*root*n*(n+1)/N) for N odd
    N = n if n % 2 == 1 else n + 1
    t_N = np.arange(N)
    x = np.exp(-1j * np.pi * root * t_N * (t_N + 1) / N)
    # Take real part and shift to positive
    a = np.real(x[:n]) + 0.5
    a = np.clip(a, 0.01, 1.0)
    return a


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        return np.ones(n) * S / n
    return a * (S / s)

def lse_gradient(a, c, beta, top_k=None):
    """Compute gradient of LSE(beta * c) with respect to a."""
    n = len(a)
    c_max = c.max()
    
    if top_k is not None:
        top_idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros(2 * n - 1)
        top_c = c[top_idx] - c_max
        exp_bc = np.exp(beta * top_c)
        total = exp_bc.sum()
        if total < 1e-300:
            return np.zeros(n)
        w[top_idx] = exp_bc / total
    else:
        exp_bc = np.exp(beta * (c - c_max))
        total = exp_bc.sum()
        if total < 1e-300:
            return np.zeros(n)
        w = exp_bc / total
    
    # grad = 2 * cross_correlation(w, a)
    # cross_correlation[w, a](i) = sum_j w[i+j] * a[j]
    # This is the "anti-diagonal" sum, computable via FFT as:
    # cross_corr = ifft(conj(FFT(w)) * FFT(a)) (circular, with padding)
    
    fsize = 1 << (len(w) + n - 1).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    cross_corr = np.real(ifft(np.conj(W) * A))[:n]
    
    return 2.0 * cross_corr
