    """Solve: min t s.t. (g*f)_k <= t for all k, f >= 0, sum(f) = 1."""
    from scipy.optimize import linprog
    
    # Variables: f_0, ..., f_{n-1}, t (total n+1 variables)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # (g*f)_k <= t  =>  sum_j g[k-j] * f_j - t <= 0
    m = 2 * n - 1
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for k in range(m):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = g[idx]
        A_ub[k, -1] = -1.0  # -t
    
    # Equality constraint: sum(f) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: f_j >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    return result.x[:n], result.x[-1]

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using alternating LP 
(celebrration procedure) combined with multiplicative mirror descent polishing.
The alternating LP solves a sequence of linear programs where one copy of the
sequence is fixed, making the convolution linear in the other copy.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def solve_conv_lp(g, n):
    """
    Solve: minimize t subject to (g*f)_k <= t for all k,
           f >= 0, sum(f) = 1.
    Returns (f, t).
    """
    # Variables: [f_0, ..., f_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Build constraint matrix: (g*f)_k <= t for k = 0, ..., 2n-2
    # (g*f)_k = sum_j g[k-j] * f_j (where g[k-j] = 0 if k-j out of [0,n-1])
    m = 2 * n - 1
    A_ub = np.zeros((m, n + 1))
    
    for k in range(m):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A_ub[k, j] = g[k - j]
        A_ub[k, n] = -1.0  # coefficient of t
    
    b_ub = np.zeros(m)
    
    # Equality: sum(f) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    return result.x[:n], result.x[n]


def alternating_lp(f0, max_iter=5000, damping=0.5, time_limit=None):
    """
    Alternating LP procedure to minimize max(f*f).
    f0 should be normalized (sum=1, non-negative).
    Returns best f found.
    """
    n = len(f0)
    f = f0.copy()
    f = np.maximum(f, 0)
    f /= f.sum()
    
    best_f = f.copy()
    best_conv_max = fast_convolve(f).max()
    
    t0 = time.monotonic()
    g = f.copy()
    
    for it in range(max_iter):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: min t s.t. g*f' <= t, f' >= 0, sum(f') = 1
        f_new, t_val = solve_conv_lp(g, n)
        if f_new is None:
            break
        
        # Damped update
        f = (1 - damping) * f + damping * f_new
        f = np.maximum(f, 0)
        f /= f.sum()
        
        # Update target
        g = f.copy()
        
        # Check improvement
        conv_max = fast_convolve(f).max()
        if conv_max < best_conv_max - 1e-12:
            best_conv_max = conv_max
            best_f = f.copy()
    
    return best_f, best_conv_max

def solve_conv_lp(g, n):
    """min t s.t. (g*f)_k <= t ∀k, f>=0, sum(f)=1. Returns (f, t) or (None, None)."""
    c = np.zeros(n + 1)
    c[-1] = 1.0
    m = 2 * n - 1
    A_ub = np.zeros((m, n + 1))
    for k in range(m):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        A_ub[k, j_min:j_max+1] = g[k - j_max:k - j_min+1][::-1]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    return res.x[:n].copy(), res.x[n]


def alternating_lp(f0, max_iter=2000, damping=0.5, time_limit=10.0):

        f /= s

        w = np.exp(scaled)

            sg[j_start:j_end+1] += w[k] * a[k - j_end:k - j_start+1][::-1]
        sg_c = sg - sg.mean()
        lf = -eta * sg_c
        lf -= lf.max()
        a = a * np.exp(lf)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n

            best_a = a.copy()
            patience = 0

            current_beta *= 1.5
            if eta < 0.01:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Alternating Linear Programming (ALP) - for fixed g, min max(g*f) over simplex is an LP
2. Multiplicative Mirror Descent (MMD) with log-sum-exp smoothing - polishing engine
3. Block-structured and diverse initializations
4. Adaptive multi-restart with time-aware budget allocation
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def conv_max_val(a):
    return fast_convolve(a).max()


def norm_obj(a):
    n = len(a)
    return 2.0 * n * conv_max_val(a)


def solve_lp_fixed_g(g, n):
    """min t s.t. (g*f)_k <= t for all k, f>=0, sum(f)=1. Returns f or None."""
    c = np.zeros(n + 1)
    c[n] = 1.0
    m = 2 * n - 1
    A_ub = np.zeros((m, n + 1))
    for k in range(m):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A_ub[k, js:je + 1] = g[k - je:k - js + 1][::-1]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status != 0:
        return None
    f = np.maximum(res.x[:n], 0.0)
    s = f.sum()
    if s < 1e-15:
        return None
    return f / s


def alternating_lp(f0, max_iter=3000, damping=0.5, time_limit=10.0):
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    f = f / s if s > 1e-15 else np.ones(n) / n
    best_f = f.copy()
    best_cm = conv_max_val(f)
    t0 = time.monotonic()
    g = f.copy()
    stall = 0
    for _ in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        f_new = solve_lp_fixed_g(g, n)
        if f_new is None:
            break
        f = (1.0 - damping) * f + damping * f_new
        f = np.maximum(f, 0.0)
        s = f.sum()
        f = f / s if s > 1e-15 else np.ones(n) / n
        g = f.copy()
        cm = conv_max_val(f)
        if cm < best_cm - 1e-10:
            best_cm = cm
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        if stall > 200:
            break
    return best_f, best_cm


def mirror_descent(a0, time_limit, beta=40.0, eta0=0.5):
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    conv = fast_convolve(a)
    best_obj = norm_obj(a)
    best_a = a.copy()
    eta = eta0
    patience = 0
    cur_beta = beta
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        mc = conv.max()
        if mc < 1e-15:
            break
        scaled = cur_beta * conv / mc
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        sg = np.zeros(n)
        thresh = w.max() * 0.001
        active = np.where(w > thresh)[0]
        for ki in active:
            k = int(ki)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
        sc = sg - sg.mean()
        lf = -eta * sc
        lf -= lf.max()
        a = a * np.exp(lf)
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 1e-15 else np.ones(n) / n
        conv = fast_convolve(a)
        co = norm_obj(a)

def generate_inits(n, rng, parent=None):
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n
    for sf in [0.1, 0.15, 0.2, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 0)
    if cos_w.sum() > 0:
        inits['cosine'] = cos_w / cos_w.sum()
    # Bimodal
    bi = np.zeros(n)
    bi[:max(1, n // 4)] = 1.0
    bi[-max(1, n // 4):] = 1.0
    inits['bimodal'] = bi / bi.sum()
    # Block structures with random heights
    for nb in [2, 4, 8, 16]:
        bs = n // nb
        if bs > 0:
            b = np.zeros(n)
            for i in range(nb):
                b[i * bs:(i + 1) * bs] = rng.uniform(0.3, 2.0)
            b = np.maximum(b, 0)
            inits[f'block_{nb}'] = b / b.sum()
    # Random Dirichlet
    inits['rand_a'] = rng.dirichlet(np.ones(n) * 0.05)
    inits['rand_b'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand_c'] = rng.dirichlet(np.ones(n) * 2.0)
    # Sparse random
    sp = np.zeros(n)
    idx = rng.choice(n, size=max(4, n // 8), replace=False)
    sp[idx] = rng.uniform(0.5, 2.0, size=len(idx))
    inits['sparse'] = sp / sp.sum()
    # Parent
    if parent is not None:
        pp = np.asarray(parent, dtype=np.float64)
        pp = np.clip(pp, 0, 1000)
        if pp.size == n:
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        elif pp.size >= n:
            pp = np.maximum(pp[:n], 0)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, pp.size), pp)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
    return inits
