def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute the objective: 2n * max(conv(a,a)) / (sum(a))^2"""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))

def smooth_obj_and_grad(a, tau):
    """
    Compute the smoothed objective and its gradient.
    Objective: 2n * softmax(conv(a,a), tau) / (sum(a))^2
    """
    n = len(a)
    sum_a = np.sum(a)
    c = fft_convolve(a, a)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    soft_max_val = c_max + tau * np.log(sum_exp)
    
    obj = 2.0 * n * soft_max_val / (sum_a ** 2)
    
    # Gradient of soft_max w.r.t. a
    a_rev = a[::-1]
    corr = fft_convolve(weights, a_rev)
    grad_softmax = 2.0 * corr[n-1:2*n-1]
    
    # Gradient of full objective
    grad = 2.0 * n * (grad_softmax * sum_a - 2.0 * soft_max_val) / (sum_a ** 3)

targeted local search with peak-directed perturbations.
Exploits scale invariance and uses BFGS quasi-Newton updates for fast convergence
on the smooth surrogate, followed by stochastic local search for final refinement.

import numpy as np
import time
import math
from scipy.optimize import minimize as sp_minimize


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the objective: 2n * max(conv(a,a)) / (sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))


def smooth_obj_and_grad(a, tau):
    """
    Compute smoothed objective and gradient.
    Objective: 2n * softmax(conv(a,a), tau) / (sum(a))^2
    Gradient computed analytically.
    """
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return 1e10, np.zeros(n)
    c = fft_convolve(a, a)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    soft_max_val = c_max + tau * np.log(sum_exp)

    obj = 2.0 * n * soft_max_val / (sum_a ** 2)

    # Gradient of soft_max w.r.t. a_j = 2 * (weights * a_reflected) correlation
    a_rev = a[::-1]
    corr = fft_convolve(weights, a_rev)
    grad_softmax = 2.0 * corr[n - 1: 2 * n - 1]

    # Full gradient: d/d a_j [2n * SM / S^2] = 2n * (grad_SM_j * S - 2*SM) / S^3
    grad = 2.0 * n * (grad_softmax * sum_a - 2.0 * soft_max_val) / (sum_a ** 3)

    return obj, grad


def l-bfgs_wrap(a, tau, maxiter, ftol, gtol):
    """Wrapper that calls L-BFGS-B with given parameters (top-level to avoid closures)."""
    n = len(a)

    def fun(x):
        return smooth_obj_and_grad(x, tau)[0]

    def jac(x):
        return smooth_obj_and_grad(x, tau)[1]

    result = sp_minimize(
        fun=fun,
        x0=a.copy(),
        jac=jac,
        method='L-BFGS-B',
        bounds=[(0.0, 1000.0)] * n,
        options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol, 'iprint': -1}
    )
    return result.x, result.nit, result.fun


def make_chirp(n, alpha, phase):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n + phase)
    return np.clip(a, 0.01, None)


def make_multi_tone(n, rng):
    """Multi-tone initialization with random phases."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    harmonics = [(1, 0.25), (2, 0.18), (3, 0.14), (4, 0.10), (5, 0.08), (7, 0.06), (11, 0.04)]
    for k, amp in harmonics:
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_zadoff_chu_real(n, rng):
    """Real component of Zadoff-Chu, shifted to be non-negative."""
    t = np.arange(n, dtype=np.float64)
    k = int(rng.integers(1, max(2, n // 2)))
    phase = -np.pi * k * t * (t + 1) / n
    a = 1.0 + 0.6 * np.cos(phase)
    return np.clip(a, 0.01, None)


def make_barker_like(n, rng):
    """Barker-inspired: blocks of high/low values."""
    a = np.ones(n)
    n_blocks = int(rng.integers(4, 12))
    block_size = max(2, n // n_blocks)
    for i in range(0, n, block_size):
        amp = rng.uniform(0.2, 0.6)
        sign = rng.choice([-1, 1])
        end = min(i + block_size, n)
        a[i:end] += sign * amp
    return np.clip(a, 0.01, None)


def make_raised_cosine(n, freq, rng):
    """Raised cosine modulation."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.1 * np.cos(4.0 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
    return np.clip(a, 0.01, None)


def local_score_fast(a):
    """Fast local score computation (same as compute_score but assumes valid input)."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))

    """
    Simulated annealing with multiple perturbation strategies.
    Focuses on reducing the peak of the autocorrelation.
    """
    current = a_init.astype(np.float64).copy()

    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()

    temperature = 0.0015
    cooling = 0.99985
    min_temp = 1e-7
    evals = 0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.25:
            # Targeted: reduce top contributors to peak
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(4, n // 10)
            bottom_k = max(4, n // 12)
            sorted_idx = np.argsort(contribs)
            reduce_idx = sorted_idx[-top_k:]
            increase_idx = sorted_idx[:bottom_k]
            candidate[reduce_idx] *= np.exp(rng.normal(-0.035, 0.012, size=len(reduce_idx)))
            candidate[increase_idx] *= np.exp(rng.normal(0.035, 0.012, size=len(increase_idx)))
        elif r < 0.45:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.60:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 12))))
                start = int(rng.integers(0, n - width))

        elif r < 0.75:

            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            src = int(rng.integers(0, n))
            dst = int(rng.integers(0, n))
            attempts = 0
            while dst == src and attempts < 10:
                dst = int(rng.integers(0, n))
                attempts += 1
            if dst != src:
                amt = current[src] * rng.uniform(0.02, 0.08)
                candidate[src] = max(0.001, current[src] - amt)
                candidate[dst] = current[dst] + amt
        elif r < 0.88:
            # Symmetric perturbation
            delta = rng.normal(0.0, 0.018, size=n)
            delta = 0.5 * (delta + delta[::-1])
            candidate = current * np.exp(delta)
        else:
            # Global small noise
            candidate = current * np.exp(rng.normal(0.0, 0.012, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score_fast(candidate)
        evals += 1

        if np.isfinite(new_score) and new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
                if evals % 5000 == 0:
                    print(f"    LS: eval={evals}, score={best_score:.6f}, t={time.monotonic()-t0:.0f}s")
        elif np.isfinite(new_score) and rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score

        temperature = max(min_temp, temperature * cooling)

    return best, best_score


def adam_optimize(a0, time_limit, n, tau_start, tau_end, rng):
    """
    Adam optimizer on the smooth objective with annealed temperature.
    Works directly on a (non-negative) with projection.
    """
    current = a0.astype(np.float64).copy()
    n = len(current)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    best = current.copy()
    best_score = local_score_fast(current)
    t0 = time.monotonic()
    lr_base = 0.5
    plateau = 0
    plateau_best = best_score

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac * 1.2)

        obj, grad = smooth_obj_and_grad(current, tau)
        step += 1
        m1 = beta1 * m1 + (1 - beta1) * grad
        m2 = beta2 * m2 + (1 - beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - beta1 ** step)
        m2_hat = m2 / (1.0 - beta2 ** step)
        lr = lr_base / (np.sqrt(step) + 1)
        current = current - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        current = np.clip(current, 0.0, 1000.0)

        # Periodic score check
        if step % 50 == 0:
            score = local_score_fast(current)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = current.copy()
                plateau = 0
            else:
                plateau += 1
                # Reheat if stuck
                if plateau > 200:
                    current = best * np.exp(rng.normal(0.0, 0.05, size=n))
                    current = np.clip(current, 0.0, 1000.0)
                    m1[:] = 0
                    m2[:] = 0
                    step = 0
                    plateau = 0

    return best, best_score

def make_barker_like(n, rng):
    """Block-structured initialization."""
    a = np.ones(n)
    n_blocks = int(rng.integers(4, 12))
    block_size = max(2, n // n_blocks)
    for i in range(0, n, block_size):
        amp = rng.uniform(0.2, 0.6)
        sign = rng.choice([-1, 1])
        end = min(i + block_size, n)
        a[i:end] += sign * amp
    return np.clip(a, 0.01, None)


def make_raised_cosine(n, freq, rng):
    """Raised cosine modulation."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.1 * np.cos(4.0 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
    return np.clip(a, 0.01, None)


def local_score_fast(a):
    """Fast local score (assumes valid non-negative input)."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))


def local_search_polish(a_init, time_limit, rng):
    """Simulated annealing with targeted peak-reduction perturbations."""
    current = a_init.astype(np.float64).copy()
    current_score = local_score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()

    temperature = 0.0015
    cooling = 0.99985
    min_temp = 1e-7
    evals = 0

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.25:
            # Targeted: reduce top contributors to the main peak
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(4, n // 10)
            bottom_k = max(4, n // 12)
            sorted_idx = np.argsort(contribs)
            reduce_idx = sorted_idx[-top_k:]
            increase_idx = sorted_idx[:bottom_k]
            candidate[reduce_idx] *= np.exp(rng.normal(-0.035, 0.012, size=len(reduce_idx)))
            candidate[increase_idx] *= np.exp(rng.normal(0.035, 0.012, size=len(increase_idx)))
        elif r < 0.45:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.60:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 12))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.75:
            # Mass transfer between positions
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            src = int(rng.integers(0, n))
            dst = int(rng.integers(0, n))
            att = 0
            while dst == src and att < 10:
                dst = int(rng.integers(0, n))
                att += 1
            if dst != src:
                amt = current[src] * rng.uniform(0.02, 0.08)
                candidate[src] = max(0.001, current[src] - amt)
                candidate[dst] = current[dst] + amt
        elif r < 0.88:
            # Symmetric perturbation
            delta = rng.normal(0.0, 0.018, size=n)
            delta = 0.5 * (delta + delta[::-1])
            candidate = current * np.exp(delta)
        else:
            # Global small noise
            candidate = current * np.exp(rng.normal(0.0, 0.012, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        new_score = local_score_fast(candidate)
        evals += 1

        if np.isfinite(new_score) and new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif np.isfinite(new_score) and rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score

        temperature = max(min_temp, temperature * cooling)

    return best, best_score
