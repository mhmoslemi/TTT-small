```python
"""
Search for non-negative coefficient sequences minimizing

    2 * n * max(fft_convolution(a, a)) / sum(a)^2

by optimizing on the probability simplex.

Main ideas:
  * Several strong initializations:
      - the provided previous construction, resized;
      - analytic one-sided / bilateral sequences from (1-z)^(-1/2);
      - a few power-law blends for diversity.
  * A smoothed multiplicative / mirror-descent optimizer.  It minimizes a
    log-sum-exp surrogate of the maximum convolution coefficient using an
    FFT-based gradient derived from weighted cross-correlations.
  * A shorter Adam polish with exact top-lag subgradients and simplex
    projection.
  * Multi-resolution exploration over several lengths, with polishing of the
    global best and cheap final shifts / reversals / symmetrizations.
"""

import time
import numpy as np

try:
    from scipy.fft import fft, ifft
except Exception:
    from numpy.fft import fft, ifft


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def normalize_one(a):
    arr = np.asarray(a, dtype=np.float64).reshape(-1)
    if arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(arr.size, 1.0 / float(arr.size), dtype=np.float64)
    return arr / s


def radical_coeffs(n):
    n = int(max(1, n))
    if n == 1:
        return np.ones(1, dtype=np.float64)
    ratios = (2.0 * np.arange(1, n, dtype=np.float64) - 1.0) / (
        2.0 * np.arange(1, n, dtype=np.float64)
    )
    c = np.empty(n, dtype=np.float64)
    c[0] = 1.0
    c[1:] = np.cumprod(ratios)
    return c


def get_parent():
    try:
        hs = height_sequence_1
    except NameError:
        return None
    try:
        arr = np.asarray(hs, dtype=np.float64).reshape(-1)
    except Exception:
        return None
    if arr.size < 16:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.sum() <= 0.0:
        return None
    return arr


def resize_sequence(a, m):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    m = int(max(1, m))
    s = float(a.sum())
    if a.size == 0 or s <= 1e-300:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    a = a / s

    x_old = np.linspace(0.0, 1.0, a.size + 1, dtype=np.float64)
    old_cdf = np.empty(a.size + 1, dtype=np.float64)
    old_cdf[0] = 0.0
    old_cdf[1:] = np.cumsum(a)

    x_new = np.linspace(0.0, 1.0, m + 1, dtype=np.float64)
    new_cdf = np.interp(x_new, x_old, old_cdf)
    b = np.diff(new_cdf)
    b = np.maximum(b, 0.0)
    sb = float(b.sum())
    if sb <= 1e-300:
        return np.full(m, 1.0 / float(m), dtype=np.float64)
    return b / sb


def make_starts(n, parent, rng):
    n = int(max(16, n))
    starts = []

    if parent is not None:
        starts.append(("parent", resize_sequence(parent, n)))

    c = radical_coeffs(n)
    cl = c
    cr = c[::-1]

    starts.append(("rad_left", normalize_one(cl)))
    starts.append(("rad_right", normalize_one(cr)))
    starts.append(("rad_blend", normalize_one(0.5 * (cl + cr))))
    starts.append(("rad_heavy_left", normalize_one(0.72 * cl + 0.28 * cr)))
    starts.append(("rad_heavy_right", normalize_one(0.28 * cl + 0.72 * cr)))

    x = np.arange(n, dtype=np.float64) + 1.0
    for alpha in (0.35, 0.65):
        left = x ** (-alpha)
        right = left[::-1]
        starts.append(("pow_left_%.2f" % alpha, normalize_one(left)))
        starts.append(("pow_blend_%.2f" % alpha, normalize_one(0.5 * (left + right))))

    if parent is None:
        starts.append(("rand", rng.random(n)))

    return starts


def convolve_only(p, L=None):
    n = int(len(p))
    if L is None:
        L = next_power_of_two(2 * n - 1)
    P = fft(p, n=L)
    c = np.real(ifft(P * P, n=L))
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def score_fft(a, L=None):
    p = normalize_one(a)
    n = int(len(p))
    if L is None:
        L = next_power_of_two(2 * n - 1)
    c = convolve_only(p, L)
    return float(2.0 * n * float(c.max()))


def conv_gradient_smooth(p, beta, L):
    """Full convolution and smooth log-sum-exp gradient.

    Returns (c, maxc, grad).  If beta <= 0, grad is None.
    """
    n = int(len(p))
    M = 2 * n - 1

    P = fft(p, n=L)
    c = np.real(ifft(P * P, n=L))[:M]
    c = np.maximum(c, 0.0)
    maxc = float(c.max())

    if beta <= 0.0 or maxc <= 1e-300:
        return c, maxc, None

    shifted = beta * (c - maxc)
    e = np.exp(shifted)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        w = np.zeros(M, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    W = fft(w, n=L)
    corr = np.real(ifft(P * np.conj(W), n=L))

    grad = np.empty(n, dtype=np.float64)
    grad[0] = corr[0]
    if n > 1:
        grad[1:] = corr[L - n + 1:L][::-1]

    return c, maxc, 2.0 * grad


def top_subgradient(p, c, idx):
    n = int(len(p))
    grad = np.zeros(n, dtype=np.float64)
    if idx.size == 0:
        return grad

    vals = c[idx].astype(np.float64, copy=True)
    s = float(vals.sum())
    if s <= 1e-300:
        vals = np.ones(idx.size, dtype=np.float64) / float(idx.size)
    else:
        vals = vals / s

    for wt, kk in zip(vals, idx):
        k = int(kk)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = p[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += wt * 2.0 * seg

    return grad


def project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    n = int(v.size)
    if n == 0:
        return v

    w = np.sort(v)[::-1]
    cum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cum - S) / idx
    valid = theta <= w

    if not np.any(valid):
        rho = n
    else:
        rho = int(np.flatnonzero(valid).max() + 1)

    theta_val = (cum[rho - 1] - S) / float(rho)
    u = np.maximum(v - theta_val, 0.0)
    su = float(u.sum())
    if su <= 1e-300:
        u = np.full(n, S / float(n), dtype=np.float64)
    else:
        u = u * (S / su)
    return u


def run_mirror_smooth(a0, max_time, rng, alpha0=1.0, alpha1=40.0,
                      eta0=1.0, eta1=0.05, jitter=0.0):
    p = normalize_one(a0)
    n = int(len(p))
    L = next_power_of_two(2 * n - 1)

    c = convolve_only(p, L)
    maxc = float(c.max())
    if not np.isfinite(maxc):
        p = np.full(n, 1.0 / float(n), dtype=np.float64)
        maxc = 1.0 / float(n)
    best = float(2.0 * n * maxc)
    bestp = p.copy()

    if max_time <= 1e-9:
        return bestp, best

    start = time.monotonic()
    t = 0
    last_improve = 0.0
    floor = max(1e-300, 1e-12 / float(n))

    while True:
        elapsed = time.monotonic() - start
        if elapsed >= max_time:
            break

        frac = elapsed / max_time
        if frac >= 1.0:
            frac = 0.999999

        alpha = alpha0 * (alpha1 / alpha0) ** frac
        beta = 2.0 * n * alpha

        c, maxc, grad = conv_gradient_smooth(p, beta, L)
        cur = float(2.0 * n * maxc)

        if np.isfinite(cur) and cur < best - 1e-15:
            best = cur
            bestp = p.copy()
            last_improve = frac

        if grad is None:
            break

        centered = grad - float(grad.mean())
        hmax = float(np.max(np.abs(centered)))
        if not np.isfinite(hmax) or hmax <= 1e-300:
            break

        eta = eta0 + (eta1 - eta0) * frac
        if frac - last_improve > 0.10:
            eta *= max(0.2, 1.0 - 3.0 * (frac - last_improve))

        p = p * np.exp(-eta * (centered / hmax))
        p = np.maximum(p, floor)
        s = float(p.sum())
        if not np.isfinite(s) or s <= 1e-300:
            p = np.full(n, 1.0 / float(n), dtype=np.float64)
            break
        p = p / s

        if jitter > 0.0 and (t % 200 == 0) and rng.random() < 0.08:
            noise = rng.normal(0.0, float(jitter), size=n)
            noise -= float(noise.mean())
            noise = np.clip(noise, -1.0, 1.0)
            p = np.maximum(p * np.exp(noise), floor)
            p = p / p.sum()

        t += 1

    c = convolve_only(p, L)
    cur = float(2.0 * n * float(c.max()))
    if np.isfinite(cur) and cur < best:
        best = cur
        bestp = p.copy()

    return bestp, best


def run_adam_top(a0, max_time, rng, top_k_max=120, top_k_min=8, lr_scale=1.0):
    p = normalize_one(a0)
    n = int(len(p))
    L = next_power_of_two(2 * n - 1)

    c = convolve_only(p, L)
    best = float(2.0 * n * float(c.max()))
    bestp = p.copy()

    if max_time <= 1e-9:
        return bestp, best

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1 = 0.85
    b2 = 0.999
    eps = 1e-12
    start = time.monotonic()

    while True:
        elapsed = time.monotonic() - start
        if elapsed >= max_time:
            break

        c = convolve_only(p, L)
        cur = float(2.0 * n * float(c.max()))
        if np.isfinite(cur) and cur < best - 1e-15:
            best = cur
            bestp = p.copy()

        frac = min(1.0, elapsed / max_time)
        top_k = int(round(top_k_min + (top_k_max - top_k_min) * (1.0 - frac)))
        top_k = max(int(top_k_min), min(int(top_k), int(c.size)))

        if top_k >= c.size:
            idx = np.arange(c.size, dtype=np.int64)
        else:
            idx = np.argpartition(c, -top_k)[-top_k:]

        grad = top_subgradient(p, c, idx)
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm <= 1e-300:
            break
        grad = grad / gnorm

        t += 1
        lr = (float(lr_scale) / float(n)) * (0.5 ** (float(t) / 2000.0))

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        p = p - lr * (m_hat / (np.sqrt(v_hat) + eps))
        p = project_simplex(p, 1.0)

        if (t % 5000 == 0) and rng.random() < 0.03:
            p = normalize_one(p + rng.normal(0.0, 0.5 / float(n), size=n))

    c = convolve_only(p, L)
    cur = float(2.0 * n * float(c.max()))
    if np.isfinite(cur) and cur < best:
        best = cur
        bestp = p.copy()

    return bestp, best


def search_length(n, budget, parent, rng, global_best, global_a, deadline):
    n = int(max(16, n))
    if budget <= 5.0:
        return global_a, global_best

    starts = make_starts(n, parent, rng)
    if global_a is not None:
        starts.append(("global_seed", resize_sequence(global_a, n)))

    L = next_power_of_two(2 * n - 1)
    scored = []
    for i, item in enumerate(starts):
        name, arr = item
        arr = normalize_one(arr)
        sc = score_fft(arr, L)
        if not np.isfinite(sc):
            sc = np.inf
        scored.append((sc, i, name, arr))

    scored.sort()

    best = global_best
    best_a = global_a

    if budget > 700.0:
        k = 3
    elif budget > 250.0:
        k = 3
    elif budget > 70.0:
        k = 2
    else:
        k = 1
    k = min(k, len(scored))

    if k == 1:
        allocs = [1.0]
    elif k == 2:
        allocs = [0.65, 0.35]
    else:
        allocs = [0.50, 0.32, 0.18]

    for i in range(k):
        run_left = deadline - time.monotonic()
        alloc = budget * allocs[i]
        run_budget = min(run_left, alloc)
        if run_budget < 10.0:
            break

        alpha1 = 30.0 if n <= 8192 else (45.0 if n <= 16384 else 65.0)
        mirror_budget = run_budget * 0.72
        p1, s1 = run_mirror_smooth(
            scored[i][3],
            mirror_budget,
            rng,
            alpha0=1.0,
            alpha1=alpha1,
            eta0=1.0,
            eta1=0.05,
            jitter=0.0 if i == 0 else 0.02
        )
        if s1 < best:
            best = s1
            best_a = p1.copy()

        left = deadline - time.monotonic()
        if left > 20.0 and n <= 16384:
            adam_budget = min(run_budget * 0.28, left - 5.0)
            if adam_budget > 15.0:
                topkmax = 120 if n <= 8192 else 60
                p2, s2 = run_adam_top(
                    p1,
                    adam_budget,
                    rng,
                    top_k_max=topkmax,
                    top_k_min=8,
                    lr_scale=1.0
                )
                if s2 < best:
                    best = s2
                    best_a = p2.copy()

    return best_a, best


def authoritative_score(a):
    f = globals().get("evaluate_sequence", None)
    if f is None:
        return np.inf
    try:
        return float(f([float(x) for x in a]))
    except Exception:
        return np.inf


def propose_candidate(budget_seconds=2400.0):
    t0 = time.monotonic()
    total = min(float(budget_seconds), 2400.0) - 40.0
    if total <= 120.0:
        total = max(60.0, min(float(budget_seconds), 2400.0) - 10.0)

    hard_end = t0 + total
    polish_budget = max(60.0, 0.08 * total)
    explore_budget = total - polish_budget
    if explore_budget <= 60.0:
        explore_budget = 0.7 * total
        polish_budget = total - explore_budget

    rng = np.random.default_rng(123456)
    parent = get_parent()

    global_best = np.inf
    global_a = None

    print("total=%.1f explore=%.1f polish=%.1f" %
          (total, explore_budget, polish_budget))

    plans = [
        (4096, 0.07),
        (8192, 0.31),
        (16384, 0.37),
        (32768, 0.25),
    ]

    used_explore = 0.0
    for n_plan, frac_plan in plans:
        max_allowed = explore_budget - used_explore
        b = min(explore_budget * frac_plan, max_allowed, hard_end - time.monotonic())
        if b <= 10.0:
            break

        t_plan = time.monotonic()
        global_a, global_best = search_length(
            n_plan,
            b,
            parent,
            rng,
            global_best,
            global_a,
            hard_end
        )
        used_explore += time.monotonic() - t_plan
        print("after len=%d best=%.8f time_left=%.1f" %
              (n_plan, global_best, hard_end - time.monotonic()))

    if global_a is not None and time.monotonic() < hard_end - 10.0:
        n_best = int(len(global_a))
        left = hard_end - time.monotonic()

        mirror_b = min(left * 0.55, 520.0)
        alpha1 = 80.0 if n_best <= 8192 else 120.0
        p1, s1 = run_mirror_smooth(
            global_a,
            mirror_b,
            rng,
            alpha0=20.0,
            alpha1=alpha1,
            eta0=0.6,
            eta1=0.02,
            jitter=0.0
        )
        if s1 < global_best:
            global_best = s1
            global_a = p1.copy()

        left = hard_end - time.monotonic()
        if left > 15.0:
            if int(len(global_a)) <= 16384:
                b = min(left * 0.65, 420.0)
                p2, s2 = run_adam_top(
                    global_a,
                    b,
                    rng,
                    top_k_max=80,
                    top_k_min=4,
                    lr_scale=1.0
                )
                if s2 < global_best:
                    global_best = s2
                    global_a = p2.copy()
            else:
                b = min(left * 0.65, 420.0)
                p2, s2 = run_mirror_smooth(
                    global_a,
                    b,
                    rng,
                    alpha0=40.0,
                    alpha1=140.0,
                    eta0=0.3,
                    eta1=0.02,
                    jitter=0.0
                )
                if s2 < global_best:
                    global_best = s2
                    global_a = p2.copy()

    if global_a is None:
        global_a = np.ones(256, dtype=np.float64)

    final_a = np.asarray(global_a, dtype=np.float64).reshape(-1)
    final_a = normalize_one(final_a)
    n_final = int(len(final_a))

    if n_final >= 16:
        L_final = next_power_of_two(2 * n_final - 1)
        base = final_a
        rev = base[::-1].copy()
        best_final = score_fft(base, L_final)
        best_candidate = base.copy()

        for k in (0, 1, 2, 8, max(1, n_final // 16)):
            cand = np.roll(base, k)
            sc = score_fft(cand, L_final)
            if sc < best_final:
                best_final = sc
                best_candidate = cand.copy()

            cand = np.roll(rev, k)
            sc = score_fft(cand, L_final)
            if sc < best_final:
                best_final = sc
                best_candidate = cand.copy()

        sym = normalize_one(0.5 * (base + rev))
        sc = score_fft(sym, L_final)
        if sc < best_final:
            best_final = sc
            best_candidate = sym.copy()

        if best_final < global_best:
            global_best = best_final
            global_a = best_candidate

        if n_final <= 8192 and (hard_end - time.monotonic()) > 30.0:
            val = authoritative_score(global_a)
            if np.isfinite(val) and val < global_best:
                global_best = val

    final_a = np.clip(np.asarray(global_a, dtype=np.float64).reshape(-1), 0.0, 1000.0)
    final_a = normalize_one(final_a)
    if final_a.sum() < 0.01:
        final_a = np.ones(256, dtype=np.float64)

    print("final best=%.8f length=%d time_left=%.2f" %
          (global_best, len(final_a), hard_end - time.monotonic()))

    return [float(x) for x in final_a]
```
