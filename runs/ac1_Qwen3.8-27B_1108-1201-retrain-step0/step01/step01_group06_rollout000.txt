def lp_direction_finding(a, M):
    """
    Solve LP: maximize sum(b) subject to b >= 0, (a*b)[k] <= M for all k.
    This gives a direction to move that keeps conv <= M.
    """
    n = len(a)
    conv_len = 2 * n - 1
    
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: (a*b)[k] <= M for k = 0, ..., 2n-2
    # (a*b)[k] = sum_j a[j] * b[k-j] (valid j)
    # For each k, the constraint is: sum_{j valid} a[j] * b[k-j] <= M
    # In terms of b: sum_{j valid} b[j] * a[k-j] <= M
    # (substituting j -> k-j)
    
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    A = np.zeros((conv_len, n))
    for k in range(conv_len):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    b_const = np.full(conv_len, M)
    
    # b >= 0 is handled by bounds
    bounds = [(0, None)] * n
    
    from scipy.optimize import linprog
    result = linprog(c, A_ub=A, b_ub=b_const, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def obj_normalized(a):
    """2*n*max(conv) assuming sum(a)=1"""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def subgradient(a, conv, peak_weights=None):
    """Compute weighted subgradient of max(conv)."""
    n = len(a)
    if peak_weights is None:
        # Single peak
        k = int(np.argmax(conv))
        sg = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        sg[j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
    else:
        # Weighted combination
        w = peak_weights
        sg = np.zeros(n)
        thresh = w.max() * 0.001
        for k_int in np.where(w > thresh)[0]:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end+1] += w[k] * a[k - j_end:k - j_start + 1][::-1]
    return sg

def softmax_weights(conv, temperature):
    scaled = temperature * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    return w / w.sum()

def top_k_weights(conv, k):
    idx = np.argsort(conv)[-k:]
    w = np.zeros(len(conv))
    w[idx] = conv[idx]
    return w / w.sum()

def mirror_descent(a, conv, sg, eta):
    """One mirror descent step on the simplex."""
    n = len(a)
    sg_centered = sg - sg.mean()
    log_factor = -eta * sg_centered
    log_factor -= log_factor.max()
    a_new = a * np.exp(log_factor)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-300:
        return a
    return a_new / s

def adam_mirror_descent(a0, max_time, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                        temperature=50, use_topk=False, topk=5):
    """Adam-adapted mirror descent."""
    a = a0.copy()
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = obj_normalized(a)
    best_a = a.copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Compute weights
        if use_topk:
            w = top_k_weights(conv, topk)
        else:
            w = softmax_weights(conv, temperature)
        
        # Subgradient
        sg = subgradient(a, conv, w)
        sg_centered = sg - sg.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * sg_centered
        v = beta2 * v + (1 - beta2) * sg_centered**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Update in log-domain (mirror descent with Adam)
        log_factor = -lr * m_hat / (np.sqrt(v_hat) + eps)
        log_factor -= log_factor.max()
        a_new = a * np.exp(log_factor)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-300:
            break
        a = a_new / s
        
        conv = fast_convolve(a)
        current_obj = obj_normalized(a)
        
        if current_obj < best_obj:
            best_obj = current_obj
            best_a = a.copy()
    
    return best_a, best_obj

                if a_lp2 is not None:
                    conv_lp2 = fast_convolve(a_lp2)
                    obj_lp2 = obj_normalized(a_lp2)
                    if obj_lp2 < best_obj:
                        best_obj = obj_lp2
                        best_a = a_lp2.copy()
                        a = a_lp2

        count += 1
        # Gradually reduce LP mixing
        lp_t = max(0.1, lp_t * 0.95)

    return best_a, best_obj


def get_parent_sequence():
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p
    except Exception:
        pass
    return None


def run_full_search(seed, budget_s):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    parent = get_parent_sequence()

    # Determine which lengths to try
    lengths = [64, 96, 128, 160, 192, 256]

    # Phase 1: Quick scan (first 30% of time)
    phase1_budget = 0.30 * (deadline - time.monotonic())
    phase1_deadline = time.monotonic() + phase1_budget
    print(f"Phase 1: Quick scan (budget {phase1_budget:.0f}s)")

    scan_results = []
    inits_per_length = 5

    for n in lengths:
        # Select diverse initializations
        init_types = ['uniform', 'gauss_1', 'gauss_2', 'random', 'bimodal']
        if parent is not None:
            init_types.append('parent')

        for init_type in init_types:
            if time.monotonic() > phase1_deadline - 30:
                break

            if init_type == 'parent' and parent is not None:
                if parent.size >= n:
                    a0 = normalize_seq(parent[:n])
                else:
                    a0 = np.interp(np.linspace(0, 1, n),
                                   np.linspace(0, 1, parent.size), parent)
                    a0 = normalize_seq(a0)
            else:
                a0 = normalize_seq(generate_init(n, init_type, rng))

            t_alloc = min(4.0, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 1.0:
                break

            a_opt, obj_opt = mirror_descent_optimize(
                a0, t_alloc, lr=0.05, temp0=80.0, temp1=10.0,
                use_adam=True, use_topk=False
            )
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, init_type, a_opt))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan: n={n}, {init_type}, score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Deep optimization with LP cycles (next 50% of remaining time)
    remaining = deadline - time.monotonic()
    phase2_budget = 0.50 * remaining
    phase2_deadline = time.monotonic() + phase2_budget
    print(f"Phase 2: Deep LP+MD cycles (budget {phase2_budget:.0f}s)")

    # Select top candidates for deep optimization
    top_candidates = []
    seen = set()
    for score, n, init_type, a_opt in scan_results:
        key = (n, init_type)
        if key not in seen:
            top_candidates.append((n, a_opt))
            seen.add(key)
        if len(top_candidates) >= 8:
            break

    # Also add parent if not already included
    if parent is not None:
        pn = len(parent)
        if not any(n == pn for n, _ in top_candidates):
            top_candidates.append((pn, normalize_seq(parent)))

    # Include current best
    if best_sequence is not None:
        bn = len(best_sequence)
        if not any(n == bn for n, _ in top_candidates):
            top_candidates.append((bn, best_sequence.copy()))

    n_cands = len(top_candidates)
    time_per_candidate = phase2_budget / max(1, n_cands)

    for i, (n, a0) in enumerate(top_candidates):
        if time.monotonic() > phase2_deadline - 30:
            break
        t_alloc = min(time_per_candidate, phase2_deadline - time.monotonic() - 20)
        if t_alloc < 10:
            break

        print(f"  Deep {i+1}/{n_cands}: n={n}, alloc={t_alloc:.0f}s")

        # Run LP+MD cycle
        a_opt, obj_opt = run_lp_mirror_cycle(a0, t_alloc, md_time=6.0, lp_t=0.4)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"    Improved! score={score:.6f}")

        # Also try with different parameters
        if time.monotonic() < phase2_deadline - 20:
            t_alloc2 = min(t_alloc * 0.5, phase2_deadline - time.monotonic() - 20)
            if t_alloc2 > 5:
                a_opt2, obj_opt2 = mirror_descent_optimize(
                    a_opt, t_alloc2, lr=0.03, temp0=50.0, temp1=2.0,
                    use_adam=True, use_topk=True, topk=5
                )
                score2 = float(evaluate_sequence(a_opt2.tolist()))
                if np.isfinite(score2) and score2 < best_score:
                    best_score = score2
                    best_sequence = a_opt2.copy()
                    print(f"    Improved (topk)! score={score2:.6f}")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # Phase 3: Final polish (remaining time)
    remaining = deadline - time.monotonic()
    print(f"Phase 3: Final polish (budget {remaining:.0f}s)")

    if best_sequence is not None and remaining > 10:
        # Coordinate descent polish
        cd_budget = min(remaining * 0.4, 120.0)
        if cd_budget > 5:
            a_cd, obj_cd = coordinate_descent(best_sequence, cd_budget, lr_init=0.01)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD improved: {score_cd:.6f}")

        remaining = deadline - time.monotonic()

        # More LP+MD cycles on the best
        if remaining > 20:
            a_opt, obj_opt = run_lp_mirror_cycle(best_sequence, remaining * 0.5,
                                                  md_time=5.0, lp_t=0.2)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP+MD improved: {score:.6f}")

            remaining = deadline - time.monotonic()

        # Final coordinate descent
        if remaining > 10:
            a_cd, obj_cd = coordinate_descent(best_sequence, remaining * 0.6, lr_init=0.005)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  Final CD improved: {score_cd:.6f}")

    # Phase 4: Try a few fresh starts if time remains
    remaining = deadline - time.monotonic()
    if remaining > 60 and best_sequence is not None:
        print(f"Phase 4: Fresh restarts (budget {remaining:.0f}s)")
        bn = len(best_sequence)
        for attempt in range(5):
            if time.monotonic() > deadline - 10:
                break
            # Perturb the best sequence
            noise = rng.random(bn) * 0.1
            a_pert = normalize_seq(best_sequence + noise)
            t_alloc = min(20.0, deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break
            a_opt, obj_opt = run_lp_mirror_cycle(a_pert, t_alloc, md_time=5.0, lp_t=0.5)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Restart {attempt+1} improved: {score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f}")
    return best_sequence


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Uses hybrid Adam mirror descent + LP perturbation
    + coordinate descent to minimize the auto-convolution peak ratio.
    """
    result = run_full_search(seed=seed, budget_s=budget_s)
    return [float(x) for x in result]
