"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.

Strategy:
  - Use log-sum-exp smoothed objective for early phase (smooth landscape),
    transitioning to hard max (top-k subgradient) for late refinement.
  - Adam optimizer with adaptive learning rate and simplex projection.
  - Multiple diverse initializations, then focus time on the best candidate.
  - Progressive top-k: start broad (smooth gradient), narrow to top-1 (sharp).
  - Final polish phase with very small steps for convergence.
  - Try lengths 4096 and 8192, starting from given sequence and smooth inits.

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def compute_subgradient_topk(a, n, top_indices, c, k_count):
    """Weighted subgradient using top-k active lags (hard weights)."""
    grad = np.zeros(n, dtype=np.float64)
    top_indices_k = top_indices[:k_count]
    top_vals = c[top_indices_k]
    total = top_vals.sum()
    if total <= 1e-15:
        return grad
    weights = top_vals / total
    for w_val, k in zip(weights, top_indices_k):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += w_val * 2.0 * seg
    return grad


def compute_gradient_lse(a, n, c, alpha):
    """Gradient of log-sum-exp(alpha * c) w.r.t. a, using all lags with softmax weights."""
    # Softmax weights over lags
    c_shifted = c - c.max()
    exp_c = np.exp(alpha * c_shifted)
    total = exp_c.sum()
    if total <= 1e-30:
        return np.zeros(n, dtype=np.float64)
    weights = alpha * exp_c / total
    
    # Use only lags with significant weight for efficiency
    threshold = weights.max() * 0.001
    active = np.where(weights > threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(weights))])
    
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        w_val = weights[k]
        if w_val < 1e-12:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[k - hi:k - lo + 1][::-1]
            grad[lo:hi + 1] += w_val * 2.0 * seg
    return grad


def safe_normalize(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def objective_value(a):
    """Compute the objective 2*n*max(conv)/sum^2 directly."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return float(2.0 * n * c.max() / (s * s))


def run_optimization(a_init, n, S, max_time, rng, phase_type="mixed"):
    """
    Run Adam optimization with adaptive strategy.
    phase_type: 'smooth' (lse), 'hard' (top-k), 'mixed' (transition)
    Returns (best_a, best_objective).
    """
    a = safe_normalize(a_init, n)
    c = fft_convolve_self(a)
    best_obj = objective_value(a)
    best_local = a.copy()
    
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    # Adaptive learning rate schedule
    lr_init = 0.8 * S / n
    lr_decay = 0.97
    
    # Phase parameters
    if phase_type == "smooth":
        alpha_start, alpha_end = 3.0, 10.0
        k_start, k_end = 32, 16
    elif phase_type == "hard":
        alpha_start, alpha_end = 50.0, 200.0
        k_start, k_end = 1, 4
    else:  # mixed
        alpha_start, alpha_end = 5.0, 100.0
        k_start, k_end = 16, 2
    
    start = time.monotonic()
    patience = 0
    patience_limit = 5000
    last_best = best_obj
    
    while time.monotonic() - start < max_time:
        t += 1
        elapsed = time.monotonic() - start
        progress = elapsed / max_time
        
        # Interpolate parameters
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        k_use = max(1, int(k_start + (k_end - k_start) * progress))
        lr = lr_init * (lr_decay ** t)
        lr = max(lr, lr_init * 0.01)
        
        # Compute convolution
        c = fft_convolve_self(a)
        cur_obj = float(2.0 * n * c.max() / (S * S))
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_local = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > patience_limit:
            # Perturb to escape
            a = project_simplex(a + rng.normal(0, 0.005 * S / n, n), S)
            patience = 0
        if patience > patience_limit * 3:
            break
        
        # Compute gradient

        global_best = obj
        global_a = a_res.copy()
        print("  After mixed: obj=%.6f" % obj)
    
    # Phase 3: Hard optimization (focus on exact max)
    left = deadline - time.monotonic()
    phase3_time = min(900.0, left * 0.5)
    
    if phase3_time > 10:
        print("Phase 3: Hard optimization for %.0f seconds" % phase3_time)
        a_res, obj = run_optimization(global_a, n, S, phase3_time, rng, "hard")
        if obj < global_best:
            global_best = obj
            global_a = a_res.copy()
            print("  After hard: obj=%.6f" % obj)
    
    # Phase 4: Fine polish
    left = deadline - time.monotonic()
    polish_time = min(400.0, left * 0.7)
    
    if polish_time > 10:
        print("Phase 4: Fine polish for %.0f seconds" % polish_time)
        a_res, obj = run_few_step_polish(global_a, n, S, polish_time, rng)
        if obj < global_best:
            global_best = obj
            global_a = a_res.copy()
            print("  After polish: obj=%.6f" % obj)
    
    # Phase 5: Try a restart from perturbed best (if time allows)
    left = deadline - time.monotonic()
    if left > 120:
        print("Phase 5: Restart from perturbed best for %.0f seconds" % (left * 0.8))
        pert = global_a + rng.normal(0, 0.002 * S / n, n)
        pert = np.maximum(pert, 0)
        a_res, obj = run_optimization(pert, n, S, left * 0.5, rng, "hard")
        if obj < global_best:
            global_best = obj
            global_a = a_res.copy()
            print("  After restart (hard): obj=%.6f" % obj)
        
        left = deadline - time.monotonic()
        if left > 60:
            a_res, obj = run_few_step_polish(global_a, n, S, left * 0.8, rng)
            if obj < global_best:
                global_best = obj
                global_a = a_res.copy()
                print("  After restart polish: obj=%.6f" % obj)
    
    # Final verification with evaluate_sequence
    global_a = np.clip(np.asarray(global_a, dtype=np.float64), 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)
    
    try:
        verify = evaluate_sequence(global_a.tolist())
        print("Final verify: %.6f (internal: %.6f)" % (verify, global_best))
        if np.isfinite(verify):
            global_best = verify
    except Exception as e:
        print("Verification failed: %s" % str(e))
    
    # Ensure we return before deadline
    if time.monotonic() > deadline - 5:
        print("Returning near deadline")
    
    return [float(x) for x in global_a]
