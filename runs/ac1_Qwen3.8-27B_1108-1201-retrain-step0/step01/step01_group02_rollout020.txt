"""

Uses soft-max gradient, adaptive step size, and local search polishing.

def _score(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def _normalize(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def _init_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def _init_uniform(n):
    return np.ones(n, dtype=np.float64)


def _init_random_smooth(n, rng, n_bumps=4):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(2, n // 15), max(3, n // 4)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return _init_raised_cosine(n)
    return a / s

    """Subgradient descent with soft-max weights and momentum."""
    rng = np.random.default_rng(seed)
    a = _normalize(a.copy().astype(np.float64))
    n = len(a)
    start = time.monotonic()
    
    best_a = a.copy()
    best_score = _score(a)
    if verbose:
        print(f"    SG start: score={best_score:.6f}, n={n}")
    
    # Hyperparameters
    momentum = 0.85
    eta = 0.3
    T_frac_start = 0.05  # temperature as fraction of max_conv
    T_frac_end = 0.002
    min_improvement = 1e-10
    
    m = np.zeros(n)
    no_improve = 0
    it = 0
    total_iters = max(1000, int(time_budget * 2000))  # rough estimate
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(a, a)
        max_conv = conv.max()
        
        # Cooling
        frac = min(1.0, it / total_iters)
        T_frac = T_frac_start * (T_frac_end / T_frac_start) ** frac
        T = T_frac * max_conv
        
        # Softmax weights
        shifted = (conv - max_conv) / T
        shifted = np.clip(shifted, -50, 0)  # prevent overflow
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        else:
            weights /= wsum
        
        # Gradient
        g = np.correlate(weights, a, mode='valid')
        
        # Momentum
        m = momentum * m + (1.0 - momentum) * g
        m_sum = m.sum()
        if m_sum > 1e-15:
            m_dir = m / m_sum  # normalize to sum to 1
        else:
            m_dir = np.ones(n) / n
        
        # Update
        a_new = a - eta * m_dir
        a_new = np.maximum(0.0, a_new)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new /= s
        
        score_new = _score(a_new)
        
        if score_new < best_score - min_improvement:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slightly increase eta if making progress

            a = a_new

    if verbose:

        # Choose segment
        seg_len = int(rng.integers(max(2, n // 200), max(3, n // 5)))

        noise = rng.standard_normal(seg_len)

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start:.0f}s")
    return best_a, best_score


def _coordinate_descent(a, time_budget, verbose=True):
    """Try coordinate-wise updates: adjust a_i to reduce the peak."""
    a = _normalize(a.copy().astype(np.float64))
    n = len(a)
    start = time.monotonic()
    
    best_a = a.copy()
    best_score = _score(a)
    
    it = 0
    while time.monotonic() - start < time_budget:
        conv = np.convolve(a, a)
        max_conv = conv.max()
        
        # Find the peak position
        k_star = int(np.argmax(conv))
        
        # For the peak, (a*a)_{k_star} = sum_i a_i * a_{k_star - i}
        # To reduce it, decrease a_i for i that contribute most
        # The contribution of a_i is a_i * a_{k_star - i}
        contributions = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        
        # Decrease the top contributors slightly, increase others
        top_k = max(1, n // 100)
        top_indices = np.argsort(contributions)[-top_k:]
        
        a_new = a.copy()
        a_new[top_indices] *= 0.99
        a_new = _normalize(a_new)
        
        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        
        it += 1
    
    if verbose:
        print(f"    CD done: best={best_score:.6f}, iters={it}, {time.monotonic()-start:.0f}s")
    return best_a, best_score

The objective (with $\sum a = 1$) is: $f(a) = 2n \cdot \max_k (a*a)_k$

The subgradient of $\max_k c_k$ w.r.t. $a_i$ is:
- If $c_{k^*} = \max$ at a unique point $k^*$: $g_i = a_{k^* - i}$ (for valid $k^* - i$)
- With soft-max: $g_i = \sum_k p_k \cdot a_{k-i}$ where $p_k$ are softmax weights

The direction of steepest descent is $-g$ (we want to decrease $f$).
After the update $a_{new} = a - \eta g$, we project onto the simplex.

But there's a subtlety: the subgradient of $\max_k c_k$ is not a true gradient (it's a subgradient), so the function might not decrease monotonically. The momentum and step size control help with this.

Also, I should note that the "soft max" $\frac{1}{\tau} \log \sum_k e^{\tau c_k}$ is a smooth upper bound on $\max_k c_k$, and its gradient is a convex combination of the subgradients at each $k$. As $\tau \to \infty$, it approaches the true max. So optimizing the soft max is a valid relaxation.

One potential issue: when the temperature is too low, the weights become very concentrated on a single peak, and the optimization effectively only tries to reduce that one peak. Actually, I realize I should be more careful. The mirror descent update $a_i \leftarrow a_i e^{-\eta g_i}$ followed by normalization is:
$a_i^{new} = \frac{a_i e^{-\eta g_i}}{\sum_j a_j e^{-\eta g_j}}$

This is a multiplicative update that reduces $a_i$ more when $g_i$ is large (i.e., when $a_i$ contributes more to the peak). This makes intuitive sense.

The step size $\eta$ should be related to the scale of $g$. Since $\sum_i g_i = \sum_k p_k \sum_i a_{k-i} = \sum_k p_k \cdot 1 = 1$ (because $\sum_i a_i = 1$ for each valid $k$), we have $\sum_i g_i = 1$ on average. So $g$ is already "normalized" in some sense.

Wait, that's not quite right. $\sum_i g_i = \sum_i \sum_k p_k a_{k-i} = \sum_k p_k \sum_i a_{k-i}$. Now, $\sum_i a_{k-i}$ where $0 \leq i < n$ and $0 \leq k-i < n$ means $k-n+1 \leq i \leq k$. So the sum is $\sum_{i=\max(0,k-n+1)}^{\min(n-1,k)} a_i$, which is not necessarily 1 (it's the sum of a window of $a$).

For $k$ in the range $[n-1, n-1]$ (the center of the convolution), $\sum_i a_{k-i} = \sum_{i=0}^{n-1} a_i = 1$. For $k$ near the edges, the sum is less than 1.

So $\sum_i g_i = \sum_k p_k \cdot (\text{window sum of } a \text{ at position } k)$. Since $p_k$ is concentrated near the center (where the window sum is 1), we have $\sum_i g_i \approx 1$.

OK so the gradient is roughly normalized. A step size of $\eta = 0.1$ to $1.0$ should be reasonable.

Let me finalize the code with mirror descent as the primary optimization method.

Actually, you know what, let me think about this problem from a higher level. What kinds of sequences minimize the peak of the auto-convolution?

For the non-negative constraint, the optimal sequence should be "spread out" (not concentrated) to avoid creating a large peak in the convolution. The most "spread out" non-negative sequence is the uniform sequence (all ones), but that gives a score of 2 (triangle convolution).

The next best would be a sequence that's somewhat concentrated in the middle (like a raised cosine or Gaussian), which gives a more "rounded" convolution with a less sharp peak.

The optimal sequence is a balance between:
1. Being spread out (to reduce the peak)
2. Being "smooth" (to avoid creating sharp features in the convolution)

I suspect the optimal sequence is somewhere between a raised cosine and a slightly "flattened" version.

For longer sequences, we have more "degrees of freedom" to shape the convolution, so we might be able to get a flatter peak. But the $n$ factor in the score penalizes longer sequences.

Let me also think about whether there's a theoretical result. The problem is related to the "Loomis-Whitney inequality" or "Brunn-Minkowski" type results, but I don't think there's a closed-form solution.

One more idea: **spectral approach**. In the frequency domain:
- Let $A(\omega) = \sum_{j=0}^{n-1} a_j e^{-2\pi i \omega j / N}$ where $N \geq 2n-1$ (to avoid aliasing)
- Then $(a*a)_k$ is the $k$-th coefficient of the inverse DFT of $|A(\omega)|^2$
- We want the inverse DFT of $|A|^2$ to be flat
- A flat sequence in time domain corresponds to a delta in frequency domain
- So we want $|A(\omega)|^2$ to be a delta at $\omega = 0$
- This means $A(\omega)$ should be concentrated at $\omega = 0$
- $A(0) = \sum a_j = 1$ (normalized)
- $A(\omega) \approx 0$ for $\omega \neq 0$

But this is the condition for $a$ to be "as constant as possible", which leads to the uniform sequence and a score of 2. So the spectral approach confirms that the uniform sequence is the "most spread out" option, but it's not optimal.

The issue is that we can't have $|A|^2$ be exactly a delta (that would require $a$ to be exactly uniform, which gives a triangle convolution, not flat). We need $|A|^2$ to be "almost" a delta but with a specific shape that makes the inverse DFT flat.

This is a complex optimization problem, and I think the subgradient/mirror descent approach is the way to go.

"""
Mirror descent + subgradient optimization for minimizing peak auto-convolution
of non-negative sequences. Uses soft-max gradient in frequency/time domain,
adaptive temperature cooling, and local search polishing. Multiple initializations
and sequence lengths are explored.
"""

import numpy as np
import time


def _score(a):
    """Compute the objective: 2*n*max(a*a)/(sum a)^2."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def _normalize(a):
    """Project to non-negative simplex (sum=1)."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def _init_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def _init_uniform(n):
    return np.ones(n, dtype=np.float64)


def _init_random_smooth(n, rng, n_bumps=4):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(2, n // 15), max(3, n // 4)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return _init_raised_cosine(n)
    return a / s


def _mirror_descent_optimize(a, time_budget, seed=42, verbose=True):
    """
    Mirror descent (multiplicative) with soft-max gradient.
    Update: a_i <- a_i * exp(-eta * g_i), then normalize.
    """
    rng = np.random.default_rng(seed)
    a = _normalize(a.copy().astype(np.float64))
    n = len(a)
    start = time.monotonic()

    best_a = a.copy()
    best_score = _score(a)
    if verbose:
        print(f"    MD start: score={best_score:.6f}, n={n}")

    # Hyperparameters
    eta = 0.5
    T_frac_start = 0.03
    T_frac_end = 0.001
    total_iters = max(2000, int(time_budget * 3000))

    no_improve = 0
    it = 0
    g_ema = np.zeros(n)  # exponential moving average of gradient

    while time.monotonic() - start < time_budget:
        conv = np.convolve(a, a)
        max_conv = conv.max()

        # Cooling schedule (geometric)
        frac = min(1.0, it / total_iters)
        T_frac = T_frac_start * (T_frac_end / T_frac_start) ** frac
        T = T_frac * max_conv

        # Softmax weights on the convolution
        shifted = (conv - max_conv) / T
        shifted = np.clip(shifted, -60.0, 0.0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        else:
            weights /= wsum

        # Gradient via cross-correlation
        g = np.correlate(weights, a, mode='valid')

        # Smooth gradient with EMA
        g_ema = 0.7 * g_ema + 0.3 * g
        g_ema_sum = g_ema.sum()
        if g_ema_sum > 1e-15:
            g_use = g_ema / g_ema_sum
        else:
            g_use = g / (g.sum() + 1e-30)

        # Mirror descent update: multiplicative
        log_update = -eta * g_use
        log_update -= log_update.max()  # numerical stability
        a_new = a * np.exp(log_update)
        a_new = _normalize(a_new)

        score_new = _score(a_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slowly increase step size when making progress
            eta = min(1.5, eta * 1.002)
        elif score_new < _score(a) + 1e-12:
            a = a_new
            no_improve += 1
            eta = max(0.05, eta * 0.99)
        else:
            no_improve += 1
            eta = max(0.05, eta * 0.95)

        # Restart if stuck for too long
        if no_improve > 800:
            pert = rng.standard_normal(n) * 0.03
            a = a + pert
            a = _normalize(a)
            g_ema *= 0.3
            eta = 0.3
            no_improve = 0

        it += 1
        if verbose and it % 10000 == 1:
            elapsed = time.monotonic() - start
            print(f"    MD iter {it}: score={best_score:.6f}, eta={eta:.3f}, T_frac={T_frac:.5f}, {elapsed:.0f}s")

    final_score = _score(best_a)
    if final_score < best_score:
        best_score = final_score
    if verbose:
        print(f"    MD done: best={best_score:.6f}, iters={it}, {time.monotonic()-start:.0f}s")

    return best_a, best_score


def _gradient_descent_optimize(a, time_budget, seed=42, verbose=True):
    """
    Projected gradient descent with momentum and soft-max gradient.
    Alternative to mirror descent - sometimes finds different basins.
    """
    rng = np.random.default_rng(seed)
    a = _normalize(a.copy().astype(np.float64))
    n = len(a)
    start = time.monotonic()

    best_a = a.copy()
    best_score = _score(a)
    if verbose:
        print(f"    GD start: score={best_score:.6f}, n={n}")

    momentum = 0.8
    eta = 0.3
    T_frac = 0.02
    total_iters = max(1000, int(time_budget * 2000))

    m = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(a, a)
        max_conv = conv.max()

        # Cooling
        frac = min(1.0, it / total_iters)
        T_frac = 0.03 * (0.002 / 0.03) ** frac
        T = T_frac * max_conv

        shifted = (conv - max_conv) / T
        shifted = np.clip(shifted, -60.0, 0.0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        else:
            weights /= wsum

        g = np.correlate(weights, a, mode='valid')

        # Momentum
        m = momentum * m + (1.0 - momentum) * g
        m_sum = m.sum()
        if m_sum > 1e-15:
            m_dir = m / m_sum
        else:
            m_dir = np.ones(n) / n

        a_new = a - eta * m_dir
        a_new = _normalize(a_new)
        score_new = _score(a_new)

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            eta = min(0.6, eta * 1.001)
        else:
            no_improve += 1
            eta = max(0.02, eta * 0.97)
            # Try half step
            a_half = a - 0.5 * eta * m_dir
            a_half = _normalize(a_half)
            if _score(a_half) < _score(a):
                a = a_half

        if no_improve > 600:
            pert = rng.standard_normal(n) * 0.02
            a = a + pert
            a = _normalize(a)
            m *= 0.5
            no_improve = 0
            eta = 0.2

        it += 1
        if verbose and it % 10000 == 1:
            elapsed = time.monotonic() - start
            print(f"    GD iter {it}: score={best_score:.6f}, eta={eta:.3f}, {elapsed:.0f}s")

    if verbose:
        print(f"    GD done: best={best_score:.6f}, iters={it}, {time.monotonic()-start:.0f}s")
    return best_a, best_score


def _local_search(a, time_budget, seed=0, verbose=True):
    """Local search with multiplicative segment perturbations."""
    rng = np.random.default_rng(seed)
    a = _normalize(a.copy().astype(np.float64))
    n = len(a)
    start = time.monotonic()

    best_a = a.copy()
    best_score = _score(a)
    if verbose:
        print(f"    LS start: score={best_score:.6f}")

    accept = 0
    while time.monotonic() - start < time_budget:
        seg_len = int(rng.integers(max(2, n // 300), max(3, n // 8)))
        pos = int(rng.integers(0, max(1, n - seg_len)))

        candidate = a.copy()
        noise = rng.standard_normal(seg_len)
        candidate[pos:pos + seg_len] *= np.exp(0.04 * noise)

        if rng.random() < 0.25:
            candidate = 0.5 * (candidate + candidate[::-1])

        candidate = _normalize(candidate)
        score = _score(candidate)

        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate
            accept += 1
            if verbose and accept % 200 == 0:
                print(f"    LS accept {accept}: score={best_score:.6f}")

    if verbose:
        print(f"    LS done: best={best_score:.6f}, {time.monotonic()-start:.0f}s")
    return best_a, best_score


def _peak_targeting(a, time_budget, verbose=True):
    """Directly target the peak positions: reduce contributions at peak."""
    a = _normalize(a.copy().astype(np.float64))
    n = len(a)
    start = time.monotonic()

    best_a = a.copy()
    best_score = _score(a)
    it = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(a, a)
        max_conv = conv.max()
        k_star = int(np.argmax(conv))

        # Find which a_i contribute most to the peak
        contributions = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]

        # Reduce top contributors
        n_reduce = max(1, n // 50)
        top_idx = np.argsort(contributions)[-n_reduce:]
        a_new = a.copy()
        a_new[top_idx] *= 0.98
        a_new = _normalize(a_new)

        score = _score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new

        it += 1

    if verbose:
        print(f"    PT done: best={best_score:.6f}, iters={it}, {time.monotonic()-start:.0f}s")
    return best_a, best_score
