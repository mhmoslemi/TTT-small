$\frac{\partial f}{\partial a_j} = 4 \sum_m (b_{j+m} - \bar{b}) a_m$ (for valid indices)

This is $4 \cdot \text{correlate}(b - \bar{b}, a, \text{'valid'})$ (with appropriate truncation of $b$).

This is a "variance minimization" approach, which encourages flatness of the convolution.

**Combined approach:**
1. First, minimize the variance (to get a flat convolution).
2. Then, minimize the max (to reduce the remaining peak).
3. Use soft-max as an intermediate (smooth) approximation to the max.

Let me now write the full implementation. I'll use a multi-phase approach:

Phase 1: Variance minimization (few iterations)
Phase 2: Soft-max optimization with increasing beta
Phase 3: Subgradient on the max
Phase 4: Diversification

"""
Multi-phase optimization to minimize the normalized autocorrelation peak
of a non-negative sequence. Uses soft-max gradient descent, variance
flattening, and subgradient refinement with multiple restarts.
"""

import numpy as np
import time

def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (S * S)

def _b_max(a):
    """Compute max of conv(a,a) assuming sum(a)=1."""
    b = np.convolve(a, a)
    return np.max(b)

def _soft_max_grad(a, beta):
    """Compute (soft_max_value, gradient) for the soft-max of conv(a,a)."""
    n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    exp_b = np.exp(beta * (b - b_max))
    sum_exp = np.sum(exp_b)
    w = exp_b / sum_exp
    sm_val = b_max + np.log(sum_exp) / beta
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return sm_val, grad

def _variance_grad(a):
    """Compute (variance, gradient) for the variance of conv(a,a)."""
    n = len(a)
    b = np.convolve(a, a)
    b_mean = np.mean(b)
    grad = 4.0 * np.correlate(b - b_mean, a, mode='valid')
    return np.sum((b - b_mean) ** 2), grad

def _project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _adam_step(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v

def _optimize_soft_max(a, beta, max_time, lr=0.01):
    """Optimize using soft-max objective with Adam."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    
    best_a = a.copy()
    best_score = 2.0 * n * _b_max(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        _, grad = _soft_max_grad(a, beta)
        a, m, v = _adam_step(a, grad, m, v, t, lr)
        a = _project_simplex(a, 1.0)
        
        if t % 100 == 0:
            score = 2.0 * n * _b_max(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            lr *= 0.95
    
    score = 2.0 * n * _b_max(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

def _subgrad_max(a, max_time, M=20, eta_init=0.05):
    """Subgradient descent on the max convolution value."""
    n = len(a)
    a = a.copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    
    best_a = a.copy()
    best_score = 2.0 * n * _b_max(a)
    eta = eta_init
    no_improve = 0
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        b = np.convolvesa, a)
        b_max_val = np.max(b)
        score = 2.0 * n * b_max_val
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 200:
            eta *= 0.8
            no_improve = 0
            if eta < 1e-10:
                break
        
        # Get top M peak locations
        if M >= len(b):
            k_idx = np.arange(len(b))
            wts = np.ones(len(b)) / len(b)
        else:
            k_idx = np.argpartition(b, -M)[-M:]
            b_sel = b[k_idx]
            wts = np.exp(-10.0 * (b_max_val - b_sel) / max(b_max_val, 1e-15))
            wts /= np.sum(wts)
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for ki in range(len(k_idx)):
            w = wts[ki]
            if w < 1e-8:
                continue
            k = k_idx[ki]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * w * a[k - idx]
        
        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        a = a - eta * grad / grad_norm
        a = _project_simplex(a, 1.0)
    
    score = 2.0 * n * _b_max(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

def _make_anti_symmetric(n, profile):
    """Create anti-symmetric sequence from a profile function."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = profile(i)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:

    return a / S
