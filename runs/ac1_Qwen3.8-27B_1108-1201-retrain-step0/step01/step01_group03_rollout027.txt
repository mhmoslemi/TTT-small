import time
from scipy.optimize import linprog


def _next_pow2(n):
    """Next power of 2 >= n."""
    p = 1
    while p < n:
        p *= 2
    return p


def _autocorr(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)
    return c[:2 * n - 1]

    c = _autocorr(a)
    S = a.sum()
    if S < 0.01:

    """Scale a to have sum = target."""
    S = a.sum()
    if S < 1e-12:
        return np.full_like(a, target / len(a))
    return a * (target / S)

    n = len(a)

    """One Nesterov momentum step minimizing c[k_star]."""
    n = len(a)
    g = _grad_peak(a, k_star)
    # Project gradient onto sum=0 (to preserve sum)
    g = g - g.mean()
    # Nesterov momentum
    v = beta * v + g
    # Update
    a = a - lr * v
    # Project onto non-negative
    a = np.maximum(a, 0.0)

One more thought: the "peak shaving" idea from PAPR reduction. In the PAPR literature, a common approach is:

1. Compute the time-domain signal `x = IFFT(X)`
2. Find peaks where `|x[n]| > threshold`
3. Modify the spectrum: `X_new = X - IFFT(peak_mask * x)` (subtract the peak from the time domain and transform back)
4. Iterate

For our problem, the "time domain" is the autocorrelation `c = IFFT(|A|^2)` and the "spectrum" is `|A|^2`. But modifying `|A|^2` directly is hard because we need to maintain the non-negativity of `a = IFFT(A)`.

An alternative "PAPR-style" approach:
1. Compute `c = conv(a, a)`
2. Find the peak at lag `k*`
3. The value `c[k*] = sum_j a[j]*a[k*-j]`
4. Create a "peak signal" `p[j]` that is proportional to `a[k*-j]` (the "partner" values)
5. Subtract: `a_new[j] = a[j] - alpha * p[j]` for j in the valid range
6. This reduces `c[k*]` because we're reducing the product `a[j]*a[k*-j]`
7. But it might increase other lags
8. Project to non-negative and renormalize

This is essentially the same as gradient descent with a specific step size! The gradient `grad[j] = a[j] + a[k*-j]` is the "peak signal" plus the value itself.

So I think my gradient approach is already capturing this. The key improvement is in the convergence strategy (momentum, adaptive lr, etc.).

Actually, one more important idea: **the "push-apart" strategy**. When the peak is at lag `k*`, the gradient is `a[j] + a[k*-j]`. This means:
- If `a[j]` is large AND `a[k*-j]` is large, the gradient at `j` is large → we reduce both
- This is like "pushing apart" the high values that are at distance `k*` from each other

The most effective strategy is to reduce the values that are "paired" with large values. This is exactly what the gradient does.

But there's a subtlety: reducing `a[j]` also affects other lags. Specifically, `a[j]` appears in `c[k]` for all `k` where `0 <= k-j < n` (as `a[j]*a[k-j]`) and for all `k` where `0 <= j-k < n` (as `a[j-k]*a[j]`). So reducing `a[j]` reduces many autocorrelation values, not just the peak. This is good - it's a "global" reduction.

The issue is that the gradient only looks at the single peak lag. A better approach would be to look at ALL lags where `a[j]*a[k-j]` contributes significantly. This is what the soft-max approach does.

"""
Optimizes a non-negative sequence to minimize 2n*max(convolve(a,a))/(sum(a))^2.
Uses FFT-accelerated gradient descent with Nesterov momentum, multi-peak soft-max
smoothing, spectral initialization, LP-based refinement, and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    p = 1
    while p < n:
        p *= 2
    return p


def _autocorr(a):
    n = len(a)
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)
    return c[:2 * n - 1]


def _objective(a):
    n = len(a)
    c = _autocorr(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    return 2.0 * n * c.max() / (S * S)


def _normalize(a, target):
    S = a.sum()
    if S < 1e-12:
        return np.full_like(a, target / len(a))
    return a * (target / S)


def _grad_at_lag(a, k_star):
    n = len(a)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    i = np.arange(lo, hi + 1)
    return a[i] + a[k_star - i]


def _spectral_init(n, target, rng):
    """Initialize with approximately flat power spectrum."""
    size = n
    half = size // 2 + 1
    A = np.zeros(half, dtype=complex)
    A[0] = target
    mag = target / np.sqrt(n) * 0.8
    phases = rng.uniform(0, 2 * np.pi, half - 1)
    A[1:] = mag * np.exp(1j * phases)
    A_full = np.zeros(size, dtype=complex)
    A_full[:half] = A
    if size % 2 == 0:
        A_full[half:] = np.conj(A_full[1:half][::-1])
    else:
        A_full[half:] = np.conj(A_full[1:half][::-1])
    a = np.fft.irfft(A_full, size)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-6:
        a = rng.random(n) + 0.1
    a = _normalize(a, target)
    return a


def _optimize_momentum(a0, deadline, lr_init=0.08, beta=0.85, max_iters=100000):
    """Nesterov momentum gradient descent minimizing peak autocorrelation."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    target = np.sqrt(2.0 * n)
    a = _normalize(a, target)

    c = _autocorr(a)
    peak = c.max()
    best_peak = peak
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    stall_count = 0
    rng_local = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        k_star = c.argmax()
        g = _grad_at_lag(a, k_star)
        g = g - g.mean()
        v = beta * v + g
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, target)

        c_new = _autocorr(a_new)
        peak_new = c_new.max()

        if peak_new < peak:
            a = a_new
            c = c_new
            peak = peak_new
            stall_count = 0
            if peak < best_peak:
                best_peak = peak
                best_a = a.copy()
        else:
            stall_count += 1
            lr *= 0.7
            if stall_count > 50:
                # Random perturbation to escape local min
                a = best_a.copy()
                noise = 0.02 * rng_local.standard_normal(n)
                a = a + noise
                a = np.maximum(a, 0.0)
                a = _normalize(a, target)
                c = _autocorr(a)
                peak = c.max()
                v = np.zeros(n)
                lr = lr_init * 0.5
                stall_count = 0

        if (it + 1) % 20000 == 0:
            obj = 2.0 * n * best_peak / (target * target)
            print(f"      iter={it}, obj={obj:.6f}, lr={lr:.6f}")

    obj = 2.0 * n * best_peak / (target * target)
    return best_a, obj


def _optimize_multipeak(a0, deadline, max_iters=50000):
    """Gradient descent optimizing weighted sum of top-k autocorrelation peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    target = np.sqrt(2.0 * n)
    a = _normalize(a, target)

    best_a = a.copy()
    best_peak = _autocorr(a).max()

    beta = 0.9
    v = np.zeros(n)
    tau = 15.0
    n_top = max(5, n // 4)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _autocorr(a)
        # Soft-max weights
        c_shifted = c - c.max()
        w = np.exp(tau * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(c)
            w[c.argmax()] = 1.0
        else:
            w = w / w_sum

        # Compute gradient: grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) where valid
        # Use top-k approximation for speed
        top_idx = np.argpartition(c, -n_top)[-n_top:]
        w_top = w[top_idx]
        w_top_norm = w_top.sum()
        if w_top_norm > 0:
            w_top = w_top / w_top_norm

        grad = np.zeros(n)
        for idx, k in enumerate(top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                i = np.arange(lo, hi + 1)
                grad[i] += w_top[idx] * (a[i] + a[k - i])

        grad = grad - grad.mean()
        v = beta * v + grad

        # Adaptive lr
        lr = 0.05 * (1.0 + np.cos(np.pi * it / max_iters)) * 0.5
        lr = max(lr, 0.002)

        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, target)

        peak_new = _autocorr(a_new).max()
        if peak_new < best_peak:
            best_peak = peak_new
            best_a = a_new.copy()

        a = a_new
        # Anneal tau
        tau = 15.0 + 30.0 * (it / max_iters)

    obj = 2.0 * n * best_peak / (target * target)
    return best_a, obj


def _lp_refine(a, deadline, max_iters=100):
    """LP-based refinement alternating between a and g."""
    n = len(a)
    if n > 600:
        return a, _objective(a)

    target = np.sqrt(2.0 * n)
    a = _normalize(a, target)
    best_a = a.copy()
    best_obj = _objective(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _autocorr(a)
        M = c.max()

        c_obj = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                A_ub[k, lo:hi + 1] = a[lo:hi + 1]  # a[k-j] for j in [lo,hi] means a[k-hi:k-lo+1]
                # Actually A_ub[k,j] = a[k-j] for j in [lo, hi]
                for j in range(lo, hi + 1):
                    A_ub[k, j] = a[k - j]

        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            remaining = max(1, int(deadline - time.monotonic() - 5))
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': remaining})
            if res.status != 0:
                break
            g = res.x
            if g is None or not np.all(np.isfinite(g)):
                break

            improved = False
            for t in [0.6, 0.4, 0.8, 0.2, 0.3]:
                a_new = (1.0 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                a_new = _normalize(a_new, target)
                obj_new = _objective(a_new)
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    break

            if not improved:
                # Try gradient nudge
                c2 = _autocorr(a)
                ks = c2.argmax()
                gr = _grad_at_lag(a, ks)
                gr = gr - gr.mean()
                a = a - 0.01 * gr
                a = np.maximum(a, 0.0)
                a = _normalize(a, target)
                obj_chk = _objective(a)
                if obj_chk < best_obj:
                    best_obj = obj_chk
                    best_a = a.copy()
        except Exception:
            break

    return best_a, best_obj


def _optimize_log_param(a0, deadline, max_iters=30000):
    """Optimize using log parameterization a[j]=exp(x[j]) to avoid projection issues."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-8)
    target = np.sqrt(2.0 * n)
    a = _normalize(a, target)

    x = np.log(a)
    best_a = a.copy()
    best_peak = _autocorr(a).max()

    v = np.zeros(n)
    beta = 0.9
    lr = 0.1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        a = np.exp(x)
        a = _normalize(a, target)
        c = _autocorr(a)
        k_star = c.argmax()

        # Gradient w.r.t. x: d(max(c))/dx[j] = a[j] * d(max(c))/da[j]
        g_a = _grad_at_lag(a, k_star)
        g_x = a * g_a
        g_x = g_x - g_x.mean()  # approximately preserve sum under log param

        v = beta * v + g_x
        x = x - lr * v

        # Decay
        lr = 0.1 * (1.0 + np.cos(np.pi * it / max_iters)) * 0.5
        lr = max(lr, 0.001)

        a_check = np.exp(x)
        a_check = _normalize(a_check, target)
        peak_check = _autocorr(a_check).max()
        if peak_check < best_peak:
            best_peak = peak_check
            best_a = a_check.copy()

    obj = 2.0 * n * best_peak / (target * target)
    return best_a, obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Explores multiple starting sequences of various lengths
    using momentum gradient descent, multi-peak optimization, log-parameterization,
    and LP refinement. Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(120.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def _update(seq, score):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  *** NEW BEST: score={best_score:.6f}, n={len(seq)} ***")

    # --- Build starting points ---
    tasks = []  # (method, a0, name, time_alloc)

    # Use previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size >= 16 and np.all(np.isfinite(prev)) and np.all(np.maximum(prev, 0) >= 0):
            prev = np.maximum(prev, 0)
            if prev.sum() > 0.01:
                tasks.append(('momentum', prev.copy(), "prev_best", 0))
                tasks.append(('multi', prev.copy(), "prev_multi", 0))
                tasks.append(('log', prev.copy(), "prev_log", 0))
                # Perturbed versions
                for pidx in range(4):
                    p = prev * (1.0 + 0.15 * rng.standard_normal(len(prev)))
                    p = np.maximum(p, 0.0)
                    tasks.append(('momentum', p, f"prev_pert{pidx}", 0))
    except (NameError, AttributeError):
        pass

    # Spectral initializations
    for n in [256, 512, 1024]:
        target = np.sqrt(2.0 * n)
        a_spec = _spectral_init(n, target, rng)
        tasks.append(('momentum', a_spec, f"spec_{n}", 0))
        if n <= 512:
            tasks.append(('multi', a_spec, f"spec_multi_{n}", 0))

    # Random starts
    for n in [128, 256, 512, 1024]:
        a = rng.random(n) * 0.4 + 0.3
        tasks.append(('momentum', a, f"rand_{n}", 0))

    # Structured starts
    for n in [256, 512]:
        i = np.arange(n, dtype=np.float64)
        tasks.append(('momentum', 0.5 + 0.5 * np.cos(2 * np.pi * 2 * i / n), f"cos2_{n}", 0))
        tasks.append(('multi', 0.5 + 0.5 * np.cos(2 * np.pi * 7 * i / n), f"cos7_{n}", 0))
        tasks.append(('momentum', np.abs(np.sin(4 * np.pi * i / n)) + 0.2, f"sine_{n}", 0))
        tasks.append(('log', np.exp(-0.5 * ((i - n / 2) / (n * 0.25)) ** 2) + 0.01, f"gauss_{n}", 0))

    # Sort by length (shorter first)
    tasks.sort(key=lambda x: len(x[1]))

    print(f"Total optimization tasks: {len(tasks)}, time budget: {budget_s}s")

    # --- Run optimizations ---
    num_tasks = len(tasks)
    for tidx, (method, a0, name, _) in enumerate(tasks):
        remaining = deadline - time.monotonic()
        if remaining < 45.0:
            print(f"  Time running out ({remaining:.0f}s), stopping task loop.")
            break

        n = len(a0)
        # Allocate time: longer for larger n, less for later tasks
        frac = 1.0 / max(1, num_tasks - tidx)
        if n <= 128:
            time_alloc = min(90.0, remaining * 0.25)
        elif n <= 256:
            time_alloc = min(150.0, remaining * 0.25)
        elif n <= 512:
            time_alloc = min(250.0, remaining * 0.22)
        else:
            time_alloc = min(350.0, remaining * 0.2)

        local_deadline = min(deadline - 40.0, time.monotonic() + time_alloc)
        t0 = time.monotonic()
        print(f"  [{tidx+1}/{num_tasks}] {name}, n={n}, alloc={time_alloc:.0f}s", end="")

        try:
            if method == 'momentum':
                seq, score = _optimize_momentum(a0, local_deadline, lr_init=0.08, beta=0.85)
            elif method == 'multi':
                seq, score = _optimize_multipeak(a0, local_deadline)
            elif method == 'log':
                seq, score = _optimize_log_param(a0, local_deadline)
            else:
                seq, score = a0, _objective(a0)
        except Exception as e:
            print(f" (error: {e})")
            continue

        _update(seq, score)

        # LP refinement for promising small candidates
        if score < 1.53 and n <= 400 and time.monotonic() < deadline - 80.0:
            lp_deadline = min(deadline - 60.0, time.monotonic() + 60.0)
            seq2, score2 = _lp_refine(seq, lp_deadline, max_iters=40)
            if score2 < score:
                _update(seq2, score2)
                seq, score = seq2, score2

        elapsed = time.monotonic() - t0
        print(f" -> score={score:.6f} ({elapsed:.1f}s), best={best_score:.6f}")

    # --- Final refinement ---
    if best_seq is not None and time.monotonic() < deadline - 25.0:
        print(f"\n=== Final refinement (current best={best_score:.6f}) ===")
        n = len(best_seq)

        # Multiple refinement passes with different methods
        for pass_i in range(4):
            if time.monotonic() > deadline - 20.0:
                break
            ref_deadline = min(deadline - 15.0, time.monotonic() + (deadline - time.monotonic()) * 0.4)
            print(f"  Pass {pass_i+1}: momentum refinement...")
            seq, score = _optimize_momentum(best_seq, ref_deadline, lr_init=0.03, beta=0.9, max_iters=30000)
            if score < best_score:
                _update(seq, score)

        if n <= 400 and time.monotonic() < deadline - 40.0:
            print(f"  LP refinement...")
            seq, score = _lp_refine(best_seq, deadline - 15.0, max_iters=50)
            if score < best_score:
                _update(seq, score)

        if time.monotonic() < deadline - 25.0:
            print(f"  Multi-peak refinement...")
            seq, score = _optimize_multipeak(best_seq, deadline - 15.0, max_iters=20000)
            if score < best_score:
                _update(seq, score)

    # --- Final output ---
    if best_seq is not None:
        final_score = _objective(best_seq)
        print(f"\n=== FINAL: n={len(best_seq)}, internal_score={final_score:.6f} ===")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence score: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform sequence of length 256.")
    return [1.0] * 256
