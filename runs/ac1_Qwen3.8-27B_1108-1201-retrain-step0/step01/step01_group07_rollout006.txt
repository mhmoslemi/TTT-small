    # This is np.correlate(q, a, 'full')[j]
    
    # Great! So first term = np.correlate(q, a, 'full')[j]
    # 
    # Second term for position j: sum_k q[k] * a[j-k]
    # Let m = j-k, so k = j-m: sum_m q[j-m] * a[m]
    # Let n=m: sum_n q[j-n] * a[n]
    # This is np.correlate(a, q, 'full')[j] = sum_n a[n] * q[n-j]... 
    # Hmm, np.correlate(a, q, 'full')[j] = sum_n a[n] * q[n-j]
    # That's sum_n a[n] * q[n-j], not sum_n q[j-n] * a[n]
    # But for real sequences, sum_n q[j-n] * a[n] = sum_n a[n] * q[j-n]
    # And np.correlate(a, q, 'full')[j] = sum_n a[n] * q[n-j]
    # These are different unless q is symmetric.
    
    # Wait, let me reconsider. 
    # sum_k q[k] * a[j-k]: this is the convolution (q * a)[j]
    # In numpy: np.convolve(q, a, 'full')[j] = sum_k q[k] * a[j-k]
    # Yes! That's the second term.
    
    # So: grad[j] = np.correlate(q, a, 'full')[j] + np.convolve(q, a, 'full')[j]
    
    # But wait, np.convolve(q, a, 'full') has length (2n-1) + n - 1 = 3n-2
    # And np.correlate(q, a, 'full') has the same length.
    # We only need indices 0 to n-1 for grad.
    
    # Let me verify: for j in [0, n-1], both terms should be well-defined.
    # np.convolve(q, a, 'full')[j] for j in [0, n-1]:
    #   = sum_k q[k] * a[j-k] where 0<=j-k<n, i.e., j-n+1 <= k <= j
    #   Since k >= 0 and k <= 2n-2, and j <= n-1:
    #   k ranges from max(0, j-n+1) to min(j, 2n-2) = j
    #   This is correct.
    
    # Great, so the gradient is:
    # grad = np.correlate(q, a, 'full')[:n] + np.convolve(q, a, 'full')[:n]
    
    full_corr = np.correlate(q, a, 'full')[:n]
    full_conv = np.convolve(q, a, 'full')[:n]
    grad = full_corr + full_conv
    return grad

def run_pgd(a0, tau_start, tau_end, lr_start, lr_end, t_budget, S=None, verbose=False):
    """Adaptive projected gradient descent with soft-max."""
    n = a0.size
    if S is None:
        S = np.sqrt(2.0 * n)
    
    a = a0.astype(np.float64).copy()
    a = a / max(a.sum(), 1e-12) * S
    a = project_simplex(a, S)
    
    best = a.copy()
    best_val, _, _ = compute_objective(best)
    best_score = evaluate_sequence(best.tolist())
    
    t0 = time.monotonic()
    steps = 0
    stagnation = 0
    last_best = best_val
    
    while time.monotonic() - t0 < t_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(t_budget, 1e-6))
        
        # Anneal tau and learning rate
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        c = fast_convolve(a)
        c_max = c.max()
        
        if c_max < best_val - 1e-10:
            best_val = c_max
            best = a.copy()
            best_score = evaluate_sequence(best.tolist())
            stagnation = 0
        else:
            stagnation += 1
        
        # Compute gradient
        grad = soft_max_grad(a, c, int(np.argmax(c)), tau)
        
        # Step
        a_new = a - lr * grad
        a_new = project_simplex(a_new, S)
        a = a_new
        steps += 1
        
        if verbose and steps % 1000 == 0:
            cur_val = fast_convolve(a).max()
            print(f"    step={steps}, tau={tau:.4f}, lr={lr:.4f}, max_c={cur_val:.6f}, best={best_val:.6f}")
    
    # Final score
    final_score = evaluate_sequence(best.tolist())
    if final_score < best_score:
        best_score = final_score
        best = a.copy()
    
    if verbose:
        print(f"  PGD done: steps={steps}, best_max_c={best_val:.6f}, score={best_score:.6f}")
    
    return best, best_score

def coordinate_descent(a, S, n_iter=50, verbose=False):
    """Coordinate descent to polish the solution."""
    n = a.size
    a = a.copy()
    c = fast_convolve(a)
    
    for it in range(n_iter):
        improved = False
        for j in range(n):
            # Save old value
            old_val = a[j]
            if old_val == 0:
                #Check if setting to small positive helps
                a[j] = S / (n * 10)
                new_c = fast_convolve(a)
                if new_c.max() < c.max() - 1e-10:
                    improved = True
                    c = new_c
                else:
                    a[j] = 0
                continue
            
            # The objective involves c[k] for all k where a[j] appears
            # c[k] contains a[j]*a[k-j] for k-j in [0,n-1] and a[j]*a[j-k] for j-k in [0,n-1]
            # So c[k] is affected for k in [j, j+n-1] (from a[j]*a[k-j]) and k in [0, 2*j] (from a[j]*a[j-k])
            # More precisely: k-j in [0,n-1] means k in [j, j+n-1]
            # j-k in [0,n-1] means k in [j-n+1, j], clipped to [0, 2n-2]
            
            # For a single change a[j] -> a[j] + delta:
            # c[k] changes by: delta * a[k-j] (if 0<=k-j<n) + delta * a[j-k] (if 0<=j-k<n)
            # Plus if k=2j: additional delta * ... no, c[2j] = ... + a[j]*a[j] + ...
            # Actually c[k] = sum_i a[i]*a[k-i], so if we change a[j], 
            # c[k] changes by: a[k-j]*delta + delta*a[j-k] for k != 2j
            # and c[2j] changes by 2*a[j]*delta + delta^2 (the delta^2 from a[j]*a[j])
            
            # To minimize max(c), we want to decrease the largest c[k] values.
            # The most effective j to change is the one that appears in the max position.
            
            # For coordinate descent, let's try: for each j, find the optimal a[j]
            # given all other coordinates fixed.
            # The max convolution value as a function of a[j] (with a[j] = x) is:
            # max_k (c0[k] + x * (a[k-j] + a[j-k]) + [k==2j] * x^2)
            # where c0 is the convolution with a[j] = 0.
            
            # This is complex. Let me use a simpler approach: try a few candidate values.
            
            # Candidate values based on the gradient direction
            c_kj = np.zeros(2*n-1)
            for k in range(max(0, j), min(2*n-1, j+n)):
                if k - j < n:
                    c_kj[k] += a[k-j]
            for k in range(max(0, j-n+1), min(2*n-1, j+1)):
                if j - k < n:
                    c_kj[k] += a[j-k]
            # c_kj[k] is the coefficient of x (new a[j] value) in c[k]
            # c_new[k] = c[k] - old_val * c_kj[k] + x * c_kj[k] + (x^2 if k==2j else 0)
            # = c[k] + (x - old_val) * c_kj[k] + (x^2 - old_val^2 if k==2j else 0)
            # Hmm this is getting complicated. Let me just do a simple 1D search.
            
            # Try a few perturbation values
            best_x = old_val
            best_max_c = c.max()
            
            # Current c with a[j] = old_val
            # If we change a[j] to x, the new c is:
            # c_new[k] = c[k] - old_val * c_kj[k] + x * c_kj[k]
            # (ignoring the x^2 term for simplicity, or handling it)
            # Actually: c[k] = ... + a[j]*a[k-j] + a[j-k]*a[j] + ... (the two cross terms)
            # If k = 2j, these are the same term: a[j]*a[j] = a[j]^2
            # So: c[k] = base[k] + a[j] * c_kj[k] + (a[j]^2 if k==2j else 0)
            # where base[k] = c[k] - a[j]*c_kj[k] - (a[j]^2 if k==2j else 0)
            
            # Let me just compute base once:
            base = c - old_val * c_kj
            if 2*j < 2*n-1:
                base[2*j] -= old_val * old_val  # Remove a[j]^2 contribution
            
            # Now c_new[k] = base[k] + x * c_kj[k] + (x^2 if k==2j else 0)
            # We want to minimize max_k c_new[k] over x >= 0
            
            # Try a few values:
            candidates = [old_val * 0.5, old_val * 0.8, old_val, old_val * 1.2, old_val * 1.5,
                         max(0, old_val - 0.01*old_val), max(0, old_val + 0.01*old_val)]
            # Also try the value that equalizes the max
            # If c_kj at the argmax position is positive, decreasing x helps
            max_k = int(np.argmax(c))
            if c_kj[max_k] > 0:
                # Decreasing x will decrease c[max_k]
                # Find x such that c_new[max_k] = second_max
                second_c = np.sort(c)[-2]
                # base[max_k] + x * c_kj[max_k] (+ x^2 if max_k==2j) = second_c
                if max_k == 2*j:
                    # x^2 + c_kj[max_k]*x + base[max_k] - second_c = 0
                    A, B, Cc = 1.0, c_kj[max_k], base[max_k] - second_c
                    disc = B*B - 4*A*Cc
                    if disc >= 0:
                        x1 = (-B + np.sqrt(disc)) / (2*A)
                        x2 = (-B - np.sqrt(disc)) / (2*A)
                        for xv in [x1, x2]:
                            if xv >= 0:
                                candidates.append(xv)
                else:
                    x_star = (second_c - base[max_k]) / c_kj[max_k]
                    if x_star >= 0:
                        candidates.append(x_star)
            elif c_kj[max_k] < 0:
                # Increasing x will decrease c[max_k] - but we need to check other positions
                pass
            
            # Evaluate candidates
            for x in candidates:
                x = max(0.0, min(x, 1000.0))
                c_new = base + x * c_kj
                if 2*j < 2*n-1:
                    c_new[2*j] += x * x
                c_new = np.maximum(c_new, 0)
                if c_new.max() < best_max_c - 1e-12:
                    best_max_c = c_new.max()
                    best_x = x
            
            if best_x != old_val:
                a[j] = best_x
                c = base + best_x * c_kj
                if 2*j < 2*n-1:
                    c[2*j] += best_x * best_x
                improved = True
        
        # Renormalize
        a = project_simplex(a, S)
        
        if verbose and it % 10 == 0:
            print(f"  CD iter {it}: max_c = {fast_convolve(a).max():.6f}")
        
        if not improved and it > 5:
            break
    
    return a

def init_chirp(n, seed=0):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = 1.0 + 0.8 * np.cos(2.0 * np.pi * (t - center)**2 / (4.0 * n))
    rng = np.random.default_rng(seed)
    a += 0.05 * rng.standard_normal(n)
    return np.maximum(a, 0.0)

def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1 + np.cos(2 * np.pi * t / n))

def init_prime_pattern(n, seed=0):
    """Use a prime-based pattern for good autocorrelation properties."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    # Add harmonic components at prime-related frequencies
    for k in range(1, min(20, n // 10)):
        freq = k * (1 + rng.random() * 0.1)
        a += 0.3 * np.cos(2 * np.pi * freq * np.arange(n) / n + rng.random() * 2 * np.pi)
    return np.maximum(a, 0.0)

Wait, I have a problem. The `coordinate_descent` function is O(n^2) per iteration because for each coordinate j, I'm computing the effect on all 2n-1 convolution positions. For n=3000, that's 9 million operations per coordinate, and 3000 coordinates, so 27 billion per full CD iteration. That's way too slow.

Let me optimize the CD. The key observation is:
- For coordinate j, the convolution values that are affected are only those where k-j or j-k is in [0, n-1]
- That's at most n values for each term, so at most 2n values total
- But we can precompute the "base" convolution and then just update the affected positions

Actually, a simpler approach: for each j, the new convolution is:
`c_new = c + (x - old) * c_kj + (x^2 - old^2) * e_{2j}`

where `c_kj` is a vector of length 2n-1 with at most 2n non-zero entries.

So for each j, updating c is O(n) (sparse update), and finding the max is O(n) (or we can maintain a heap). Total per full CD pass: O(n^2).

For n=3000, that's 9 million per pass, which is fine.

But the issue is the `fast_convolve` I was calling inside CD. Let me rewrite CD to not recompute the full convolution each time, but instead maintain it incrementally.

Actually, let me reconsider. For n=3000:
- `fast_convolve` using FFT takes O(n log n) ≈ 30000 operations
- A full CD pass with n coordinates, each requiring O(n) work, is O(n^2) = 9M operations
- But if I recompute the full convolution after each coordinate update (using FFT), it's n * O(n log n) = 3000 * 30000 = 90M per pass

That's still feasible but slow. Let me use the incremental update approach instead.

Let me rewrite CD more carefully:

def coordinate_descent_v2(a, S, n_iter=20, verbose=False, time_budget=None, t0=None):
    """Coordinate descent with incremental convolution updates."""
    n = a.size
    a = a.copy()
    c = fast_convolve(a)
    c_max = c.max()
    
    for it in range(n_iter):
        if time_budget is not None and time.monotonic() - t0 > time_budget:
            break
        
        improved_any = False
        # Process coordinates in random order for better mixing
        order = np.random.default_rng(it).permutation(n)
        
        for j in order:
            old = a[j]
            if old < 1e-15:
                continue
            
            # Compute c_kj: the coefficient vector
            # c[k] = sum_i a[i]*a[k-i]
            # Partial of c[k] with respect to a[j]:
            # = a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n)
            # Note: if k-j == j (i.e., k = 2j), both terms give a[j], so coefficient is 2*a[j]
            # Wait no: c[k] = sum_i a[i]*a[k-i]
            # The terms involving a[j] are: a[j]*a[k-j] (when i=j) + a[j-k]*a[j] (when k-i=j, i.e., i=k-j)
            # These are the same term when j = k-j, i.e., k = 2j
            # So: dc[k]/da[j] = a[k-j] + a[j-k] (both 0 if index out of range)
            # But when k=2j: dc[k]/da[j] = a[j] + a[j] = 2*a[j]
            # However, c[k] itself contains a[j]^2 when k=2j, so:
            # c_new[k] = c_old[k] + (x-old)*dc[k]/da[j] + [k==2j]*(x^2 - old^2)
            # Wait, that's not right either. Let me be more careful.
            #
            # c[k] = sum_i a[i] * a[k-i]
            # If we change a[j] from old to x:
            # c_new[k] = c_old[k] - old*(a[k-j] + a[j-k]) + x*(a[k-j] + a[j-k]) + [k==2j]*(x^2 - old^2)
            # Wait, that overcounts when k=2j. Let me redo.
            #
            # c[k] = sum_i a[i]*a[k-i]
            # The terms where i=j: a[j]*a[k-j]
            # The terms where k-i=j (i.e., i=k-j): a[k-j]*a[j]
            # If j != k-j (i.e., k != 2j): these are two separate terms, each contributing a[j]*a[k-j]
            #   So c[k] contains 2*a[j]*a[k-j] (with appropriate bounds)
            #   Changing a[j] to x: change is (x-old)*a[k-j] + a[k-j]*(x-old) = 2*(x-old)*a[k-j]
            # If j == k-j (i.e., k == 2j): it's a single term a[j]*a[j] = a[j]^2
            #   Changing a[j] to x: change is x^2 - old^2
