import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def score_sequence(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def compute_subgradient(a, h):
    """
    Compute g_i = 2 * sum_k h_k * a_{k-i} (cross-correlation of h and a, scaled by 2).
    h is the indicator of active (near-max) convolution positions.
    """
    n = len(a)
    size = 1
    while size < len(h) + n - 1:
        size *= 2
    H = np.fft.rfft(h, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(H * np.conj(A), size)
    return 2.0 * corr[:n]

def build_conv_matrix_sparse(a, n_constraints=None):
    """Build sparse convolution matrix A where (A@g)_k = conv(a,g)_k."""
    n = len(a)
    n_rows = 2 * n - 1
    if n_constraints is not None and n_constraints < n_rows:
        n_rows = n_constraints

    # For each variable j, it contributes a[j] to rows j, j+1, ..., j+n-1
    # (within bounds)
    rows = []
    cols = []
    data = []

    for j in range(n):
        row_start = j
        row_end = min(j + n, n_rows)
        for k in range(row_start, row_end):
            rows.append(k)
            cols.append(j)
            data.append(a[j])

    return csr_matrix((data, (rows, cols)), shape=(n_rows, n))


def build_conv_matrix_fast(a):
    """Build sparse convolution matrix efficiently."""
    n = len(a)
    n_rows = 2 * n - 1

    # Variable j contributes to rows j through j+n-1 (clipped to [0, 2n-2])
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, min(j + n, n_rows), dtype=np.int32)
                              for j in range(n)])
    # Filter out of bounds
    mask = (row_idx >= 0) & (row_idx < n_rows)
    col_idx = col_idx[mask]
    row_idx = row_idx[mask]
    data = np.tile(a, n).astype(np.float64)[mask]

    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0, top_frac=1.0):
    """
    Solve the LP: maximize sum(g) subject to conv(a,g) <= M, g >= 0.
    Only include the top_frac fraction of constraints (those near the peak).
    """
    n = len(a)
    n_rows = 2 * n - 1

    # Compute conv(a,a) to identify important constraints
    conv_aa = fft_convolve(a, a)
    c_max = conv_aa.max()

    # Include constraints where conv(a,a) is in the top fraction
    # (these are the ones most likely to be binding)
    if top_frac < 1.0:
        threshold = np.percentile(conv_aa, (1 - top_frac) * 100)
        important = conv_aa >= threshold
        # Map to constraint indices
        # The LP has constraints conv(a,g)_k <= M for k=0,...,2n-2
        # We only include important ones
        important_idx = np.where(important)[0]
    else:
        important_idx = np.arange(n_rows)

    # Build constraint matrix for only important rows
    A_full = build_conv_matrix_fast(a)
    A = A_full[important_idx, :]

    c = -np.ones(n)
    b = np.full(len(important_idx), M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception as e:
        pass
    return None


def lp_refine(a, time_limit_s=300.0, verbose=True, top_frac=0.5):
    """
    LP-based iterative refinement. At each step, find a direction g that
    maximizes sum(g) subject to conv(a,g) <= current peak.
    """
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"      LP start: score={best_score:.6f}, n={n}")

    stagnant = 0
    it = 0
    for it in range(2000):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break

        conv = fft_convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M  # sum(a) = 1

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 30:
            break

        remaining = time_limit_s - elapsed
        if remaining < 10.0:
            break

        timeout = max(5.0, min(60.0, remaining * 0.25))
        g = solve_lp_step(a, M, timeout=timeout, top_frac=top_frac)
        if g is None:
            break

        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-8:
            break

        g = g / sum_g  # normalize to sum=1

        # Mix with current a
        alpha = min(0.5, max(0.05, 2.0 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)

        new_score = score_sequence(a_new)
        if new_score <= score + 1e-10:
            a = a_new
        else:
            # Try smaller step
            alpha2 = alpha * 0.1
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"      LP done: best={best_score:.6f}, {it+1} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search_refine(a, time_limit_s=60.0, rng=None, verbose=True):
    """Random perturbation local search."""
    if rng is None:
        rng = np.random.default_rng(42)
    start_time = time.monotonic()
    n = len(a)
    a = normalize_nonneg(a.copy().astype(np.float64))
    best_a = a.copy()
    best_score = score_sequence(a)

    if verbose:
        print(f"      LS start: score={best_score:.6f}")

    accept_count = 0
    while time.monotonic() - start_time < time_limit_s:
        # Random segment perturbation
        width = max(2, int(rng.integers(max(2, n // 100), max(3, n // 5))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.15, size=stop - start))

        # Occasionally enforce symmetry
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])

        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 100 == 0:
                print(f"      LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"      LS done: best={best_score:.6f}, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_flat_top(n):
    """Flat-top window coefficients (designed for flat passband)."""
    if n == 1:
        return np.ones(1)
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    a0 = 0.21557895
    a1 = 0.41663158
    a2 = 0.277263158
    a3 = 0.083578947
    a4 = 0.006947368
    w = a0 - a1 * np.cos(2 * np.pi * x) + a2 * np.cos(4 * np.pi * x) \
        - a3 * np.cos(6 * np.pi * x) + a4 * np.cos(8 * np.pi * x)
    return np.maximum(0.0, w)


def initial_bell(n, power=2.0):
    """Gaussian raised to a power."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / (n / 2.0)
    return np.exp(-power * x * x)


def initial_erpw(n):
    """Exponentially raised power window - designed for flat auto-convolution."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / (n / 2.0 + 1)
    # This gives a shape that's wider than Gaussian
    return np.exp(-1.5 * x**2 - 0.5 * x**4)


def random_smooth_init(n, rng, n_bumps=3):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 8), max(2, n // 2)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2) * rng.uniform(0.5, 1.5)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return initial_raised_cosine(n)
    return a / s


def add_random_symmetric_perturbation(a, rng, sigma=0.1):
    """Add a random symmetric perturbation."""
    n = len(a)
    half = n // 2
    pert = rng.normal(0, sigma, half)
    pert_full = np.zeros(n)
    pert_full[:half] = pert
    pert_full[-half:] = pert[::-1]
    if n % 2 == 1:
        pert_full[half] = pert[-1]
    new_a = a + pert_full * a
    return normalize_nonneg(new_a)


def run_single_config(n, init_type, time_budget, rng, verbose=True, parent_seq=None):
    """Run the full optimization pipeline for a single (n, init_type) config."""
    if verbose:
        print(f"    --- n={n}, init={init_type}, budget={time_budget:.0f}s ---")

    # Generate initial sequence
    if init_type == 'raised_cosine':
        a = initial_raised_cosine(n)
    elif init_type == 'gaussian':
        a = initial_gaussian(n, sigma_frac=rng.uniform(0.2, 0.4))
    elif init_type == 'flat_top':
        a = initial_flat_top(n)
    elif init_type == 'bell':
        a = initial_bell(n, power=rng.uniform(1.0, 3.0))
    elif init_type == 'erpw':
        a = initial_erpw(n)
    elif init_type == 'random':
        a = random_smooth_init(n, rng, n_bumps=int(rng.integers(2, 6)))
    elif init_type == 'parent':
        a = parent_seq.copy()
    else:
        a = initial_raised_cosine(n)

    a = normalize_nonneg(a)
    init_score = score_sequence(a)
    if verbose:
        print(f"    Init score: {init_score:.6f}")

    # Phase 1: Fast gradient (40% of budget)
    t1 = time_budget * 0.35
    a, s1 = fast_gradient_optimizer(a, time_limit=t1, verbose=verbose)

    # Phase 2: LP refinement (40% of budget)
    t2 = time_budget * 0.40
    a, s2 = lp_refine(a, time_limit_s=t2, verbose=verbose, top_frac=rng.uniform(0.3, 0.7))

    # Phase 3: Local search (15% of budget)
    t3 = time_budget * 0.15
    a, s3 = local_search_refine(a, time_limit_s=t3, rng=rng, verbose=verbose)

    # Phase 4: Final LP polish (10% of budget)
    t4 = time_budget * 0.10
    a, s4 = lp_refine(a, time_limit_s=t4, verbose=False, top_frac=1.0)

    score = score_sequence(a)
    if verbose:
        print(f"    Final score for n={n}, init={init_type}: {score:.6f}")

    return a, score


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for the non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses multi-phase optimization: fast gradient + LP polishing + local search,
    across multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(30.0, budget_s - 20.0)

    best_score = float('inf')
    best_seq = None

    # Try to use parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.maximum(0, parent)
        if parent.sum() > 0.01:
            parent = parent / parent.sum()
    except Exception:
        parent = None

    # Define search configurations
    # (n, init_type, time_fraction)
    configs = []

    if parent is not None:
        pn = len(parent)
        configs.append((pn, 'parent', 0.12))
        # Also try perturbing the parent
        configs.append((pn, 'random', 0.08))

    configs.extend([
        (500, 'raised_cosine', 0.10),
        (500, 'erpw', 0.10),
        (500, 'flat_top', 0.08),
        (1000, 'raised_cosine', 0.12),
        (1000, 'erpw', 0.10),
        (1000, 'gaussian', 0.08),
        (2000, 'raised_cosine', 0.12),
        (2000, 'erpw', 0.10),
        (2000, 'flat_top', 0.08),
        (4000, 'raised_cosine', 0.12),
        (4000, 'erpw', 0.10),
        (8000, 'raised_cosine', 0.10),
    ])

    # Sort by time fraction descending to prioritize important configs
    configs.sort(key=lambda x: -x[2])

    if verbose_ok():
        print(f"Search starting with {len(configs)} configs, budget={budget_s}s")

    for n, init_type, time_frac in configs:
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break

        time_budget = min(time_frac * budget_s, remaining * 0.9)
        if time_budget < 20.0:
            continue

        try:
            a, score = run_single_config(n, init_type, time_budget, rng,
                                         verbose=True, parent_seq=parent)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                elapsed = time.monotonic() - total_start
                print(f"    *** NEW BEST: score={score:.6f}, n={n}, "
                      f"init={init_type}, elapsed={elapsed:.0f}s")
        except Exception as e:
            print(f"    Error in config ({n}, {init_type}): {e}")
            continue

    # Final polishing on the best sequence found
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            print(f"\n=== Final polishing (remaining={remaining:.0f}s) ===")
            # LP polish
            t_lp = remaining * 0.4
            a, s = lp_refine(best_seq, time_limit_s=t_lp, verbose=True, top_frac=1.0)
            if s < best_score:
                best_score = s
                best_seq = a.copy()

            # Local search
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20.0:
                a, s = local_search_refine(best_seq, time_limit_s=remaining2 * 0.4,
                                           rng=rng, verbose=True)
                if s < best_score:
                    best_score = s
                    best_seq = a.copy()

            # One more LP
            remaining3 = deadline - time.monotonic()
            if remaining3 > 15.0:
                a, s = lp_refine(best_seq, time_limit_s=remaining3 * 0.7,
                                 verbose=True, top_frac=0.5)
                if s < best_score:
                    best_score = s
                    best_seq = a.copy()

    # Fallback
    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)
        print(f"    Fallback: score={best_score:.6f}")

    # Convert to list of floats for return
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Verify
    verify_score = evaluate_sequence(result)
    print(f"\n=== FINAL RESULT ===")
    print(f"Score (internal): {best_score:.6f}")
    print(f"Score (verified): {verify_score:.6f}")
    print(f"Sequence length: {len(result)}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")

    return result


def verbose_ok():
    """Always verbose for debugging."""
    return True
