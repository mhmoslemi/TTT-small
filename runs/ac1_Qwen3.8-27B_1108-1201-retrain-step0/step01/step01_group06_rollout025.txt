import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def build_A(a):
    """Build the constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0"""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp_direction(a, M):
    """Solve the LP to find direction b."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    
    # Maximize sum(b) subject to A*b <= M, b >= 0
    # linprog minimizes, so we negate the objective
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return res.x

"""
LP-guided optimization combined with multiplicative gradient descent and
L-BFGS polishing to minimize 2*n*max(auto-conv)/(sum)^2.
Multi-start over diverse initializations and sequence lengths (64-256).
Key innovation: periodic LP direction-finding (from the state-of-the-art LP
formulation) provides globally-informed update directions, complemented by
local multiplicative gradient steps and occasional L-BFGS refinement.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj(a):
    """Compute the normalized objective 2n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_constraint_matrix(a):
    """Build A where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1]
    return A


def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. A*b <= M, b >= 0. Returns b or None."""
    n = len(a)
    m = 2 * n - 1
    A = build_constraint_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(m, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True,
                                               'dual_feasibility_tolerance': 1e-9,
                                               'primal_feasibility_tolerance': 1e-9})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def softmax_subgradient(a, tau):
    """Compute the soft-max weighted subgradient of max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    if M < 1e-15:
        return np.zeros(n), M
    scaled = tau * conv / M
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-12:
        idx = np.argmax(conv)
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    thresh = w.max() * 0.005
    active = np.where(w > thresh)[0]
    for k_int in active:
        k = int(k_int)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        sg[j_lo:j_hi + 1] += w[k] * a[k - j_hi:k - j_lo + 1][::-1]
    return sg, M


def mul_grad_step(a, tau, eta):
    """Multiplicative (exponential weights) gradient step."""
    n = len(a)
    sg, M = softmax_subgradient(a, tau)
    if M < 1e-15:
        return a
    sg_c = sg - sg.mean()
    lf = -eta * sg_c
    lf -= lf.max()
    a_new = a * np.exp(lf)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-12:
        a_new /= s
    else:
        a_new = a.copy()
    return a_new


def smooth_obj_and_grad(x, tau):
    """Smooth log-sum-exp objective and its full gradient."""
    n = len(x)
    S = x.sum()
    if S < 1e-12:
        return 1e10, np.zeros(n)
    conv = fast_convolve(x)
    tconv = tau * conv
    tconv -= tconv.max()
    et = np.exp(tconv)
    et_sum = et.sum()
    if et_sum < 1e-15:
        return 1e10, np.zeros(n)
    w = et / et_sum
    LSE = np.log(et_sum) / tau
    gLSE = np.zeros(n)
    thresh = w.max() * 0.01
    active = np.where(w > thresh)[0]
    for k_int in active:
        k = int(k_int)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        gLSE[j_lo:j_hi + 1] += w[k] * x[k - j_hi:k - j_lo + 1][::-1]
    f = 2.0 * n * LSE / (S * S)
    grad = (2.0 * n / (S * S)) * (gLSE - 2.0 * LSE / S)
    return f, grad


def lbfgs_optimize(x0, tau, max_time, maxiter=300):
    """Run L-BFGS-B on the smoothed objective with time limit."""
    x0 = np.maximum(x0.astype(np.float64), 0.0)
    if x0.sum() < 1e-12:
        x0 = np.ones(len(x0)) / len(x0)
    t_start = time.monotonic()
    x_curr = x0.copy()

    def _obj(x):
        if time.monotonic() - t_start > max_time:
            raise TimeoutError()
        return smooth_obj_and_grad(x, tau)[0]

    def _grad(x):
        if time.monotonic() - t_start > max_time:
            raise TimeoutError()
        return smooth_obj_and_grad(x, tau)[1]

    try:
        res = minimize(_obj, x0, jac=_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1000.0)] * len(x0),
                       options={'maxiter': maxiter, 'gtol': 1e-12,
                                'maxls': 50, 'ftol': 1e-12})
        if res.x is not None and res.fun < _obj(x0):
            x_curr = np.maximum(res.x, 0.0)
    except (TimeoutError, Exception):
        pass
    return x_curr

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(('uni', np.ones(n) / n))

    # Gaussians with various widths
    for sf in [0.08, 0.12, 0.18, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        inits.append((f'g{sf:.2f}', g / g.sum()))

    # Bimodal (edges)
    bi = np.zeros(n)
    cut = max(1, n // 5)
    bi[:cut] = 1.0
    bi[-cut:] = 1.0
    inits.append(('bim', bi / bi.sum()))

    # Trimeodal
    tri = np.zeros(n)
    c = max(1, n // 7)
    tri[:c] = 1.0
    tri[n//2 - c:n//2 + c] = 1.0
    tri[-c:] = 1.0
    inits.append(('tri', tri / tri.sum()))

    # Cosine bump
    cb = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    inits.append(('cos', cb / cb.sum()))

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(('ramp', ramp / ramp.sum()))

    # Random
    inits.append(('rand', rng.dirichlet(np.ones(n) * 0.3)))
    inits.append(('rand2', rng.dirichlet(np.ones(n) * 2.0)))

    # Power law
    pw = (x + 1) ** (-0.5)
    inits.append(('pwr', pw / pw.sum()))

    # From parent
    if parent is not None:
        if parent.size == n:
            pp = parent.copy()
        elif parent.size > n:
            pp = parent[:n].copy()
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
        pp = np.maximum(pp, 0)
        if pp.sum() > 1e-10:
            inits.append(('par', pp / pp.sum()))

    return inits

        # Periodic L-BFGS
        if it % 30 == 0 and time.monotonic() < deadline - 20:
            t_lb = min(15.0, deadline - time.monotonic() - 10)
            if t_lb > 3:
                a_lb = lbfgs_optimize(a, tau=150, max_time=t_lb, maxiter=100)
                conv_lb = fast_convolve(a_lb)
                if 2.0 * n * conv_lb.max() < obj:
                    a = a_lb
                    obj = 2.0 * n * conv_lb.max()
                    tau_grad = 80.0
                    eta = 0.15

        # Evaluate periodically
        if it % 10 == 0:
            score = float(evaluate_sequence(a.tolist()))
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"    iter {it}: {score:.6f}")

    # Final evaluation
    score = float(evaluate_sequence(a.tolist()))
    if score < best_score:
        best_score = score
        best_seq = a.copy()

    return best_seq, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Multi-phase search combining LP-guided updates,
    multiplicative gradient descent, and L-BFGS polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, sum={parent.sum():.2f}")
        # Evaluate parent
        s = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(s):
            best_score = s
            best_seq = parent.copy()
            print(f"Parent score: {s:.6f}")

    # Phase 1: Quick scan
    lengths = [64, 96, 128, 160, 192, 256]
    results = do_phase1_scan(lengths, parent, rng, deadline)

    # Update best from Phase 1
    for score, name, n, a in results:
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()

    # Phase 2: Deep optimization
    best_seq, best_score = do_phase2_deep(results, parent, deadline,
                                          best_seq, best_score)

    # Phase 3: Final polish
    best_seq, best_score = do_phase3_polish(best_seq, best_score, deadline)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
