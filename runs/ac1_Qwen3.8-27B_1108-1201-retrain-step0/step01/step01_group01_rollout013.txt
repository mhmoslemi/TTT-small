Mirror descent with reversal symmetry for minimizing normalized autocorrelation peak.
Exploits the problem's reversal invariance to halve effective dimensionality.
Uses adaptive multiplicative weight updates with softmax-weighted active constraints.
"""

import numpy as np
import time


def _compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _subgrad_k(a, k, n):
    """Subgradient of b_k = sum_i a_i*a_{k-i} w.r.t. a."""
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        g[m] = 2.0 * a[k - m]
    return g


def _symmetrize(a):
    """Enforce a[i] = a[n-1-i]."""
    return (a + a[::-1].copy()) / 2.0


def _normalize(a):
    s = np.sum(a)
    if s > 1e-15:
        return a / s
    return np.ones(len(a)) / len(a)


def _mirror_step(a, g, eta):
    """Multiplicative update: a * exp(-eta*g), then normalize."""
    # Use log-space for numerical stability
    log_a = np.log(a + 1e-30)
    log_new = log_a - eta * g
    log_new -= np.max(log_new)
    new_a = np.exp(log_new)
    s = np.sum(new_a)
    if s < 1e-30:
        return a
    return new_a / s

    """First-order mirror step: a * (1 - eta*g), clip to non-negative, normalize."""
    new_a = a * (1.0 - eta * g)
    new_a = np.maximum(new_a, 0.0)
    s = np.sum(new_a)
    if s < 1e-30:

    return new_a / s
