3. Nesterov-accelerated gradient descent with adaptive step size
4. Multiple initializations (given sequence, random, structured)
5. Coordinate descent polishing at the end

import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - S) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def get_fft_size(n, mode='conv'):
    """Get FFT size for convolution or cross-correlation."""
    if mode == 'conv':
        size = 2 * n - 1
    else:  # 'xcorr'
        size = 3 * n - 2
    return 1 << (size - 1).bit_length()


def self_convolve(a):
    """Full self-convolution (length 2n-1)."""
    n = len(a)
    N = get_fft_size(n, 'conv')
    A = fft(a, N)
    c = np.real(ifft(A * A))
    return c[:2 * n - 1]


def compute_full_gradient(a, tau=None):
    """
    Compute gradient of the objective w.r.t. a.
    If tau is given, returns gradient of log-sum-exp smoothed max.
    If tau is None, returns subgradient of max (using all active lags).
    """
    n = len(a)
    N_conv = get_fft_size(n, 'conv')
    N_xcorr = get_fft_size(n, 'xcorr')
    
    A = fft(a, N_conv)
    c = np.real(ifft(A * A))[:2 * n - 1]

        shifted = tau * (c - c_max)

        log_w = shifted - m - np.log(np.exp(shifted - m).sum())
        w = np.exp(log_w)

        c_max = c.max()
        active = (c >= c_max - 1e-12).astype(np.float64)

    grad = np.real(ifft(W * np.conj(A2)))[:n]

            t_global += 1

            momentum = (t_phase - 1) / (t_phase + 1.0)
            y = a + momentum * (a - a_prev)
            y = project_simplex(y, S)
            
            # Compute gradient
            grad, c = compute_full_gradient(y, tau=tau)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            
            # Adaptive step size
            eta = eta_base / (1.0 + 0.005 * t_phase)
            
            # Line search (simple: try the step, if it doesn't help, reduce)
            a_new = project_simplex(y - eta * grad, S)
            
            # Evaluate
            c_new = self_convolve(a_new)
            val = c_new.max()
            
            if val < best_val:
                best_val = val
                best_a = a_new.copy()
            
            a_prev = a
            a = a_new
            
            # Occasional random perturbation
            if t_phase % 500 == 0 and rng.random() < 0.05:
                a = project_simplex(a + rng.normal(0, 0.001 * S / n, n), S)
                a_prev = a.copy()
    
    return best_a, best_val


def run_subgrad_descent(a_init, n, S, time_budget, rng, top_k=10):
    """
    Subgradient descent on the exact max with Nesterov acceleration.
    """
    a = project_simplex(a_init.copy(), S)
    a_prev = a.copy()
    
    best_a = a.copy()
    best_val = self_convolve(a).max()
    
    start = time.monotonic()
    t = 0
    eta = 1.0 / (n + 1.0)
    
    while time.monotonic() - start < time_budget:
        t += 1
        
        # Nesterov extrapolation
        momentum = (t - 1) / (t + 1.0)
        y = a + momentum * (a - a_prev)
        y = project_simplex(y, S)
        
        # Subgradient using top-k
        c = self_convolve(y)
        top = np.argsort(c)[::-1][:top_k]
        
        # Weighted subgradient
        N_xcorr = get_fft_size(n, 'xcorr')
        w = np.zeros(2 * n - 1)
        vals = c[top]
        w[top] = vals / vals.sum()  # Weight proportional to value
        
        W = fft(w, N_xcorr)
        A = fft(y, N_xcorr)
        grad = np.real(ifft(W * np.conj(A)))[:n]
        grad *= 2.0
        
        # Step
        eta = 1.0 / (np.sqrt(t) * (n + 1.0))
        a_new = project_simplex(y - eta * grad, S)
        
        # Evaluate
        c_new = self_convolve(a_new)
        val = c_new.max()
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
        
        a_prev = a
        a = a_new
    
    return best_a, best_val


def coordinate_descent_polish(a, n, S, budget, rounds=1000):
    """
    Coordinate descent: for each coordinate, find the best value
    (1D line search) while keeping others fixed.
    """
    a = a.copy()
    best_val = self_convolve(a).max()
    start = time.monotonic()
    
    for round_idx in range(rounds):
        if time.monotonic() - start > budget:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - start > budget:
                    break
            
            # Compute the quadratic in a_i
            # (a*a)[k] = a_i * (2*a_{k-i} - a_i*delta(k,2i)) + rest_k
            # Let's compute the effect of changing a_i
            old_ai = a[i]
            
            # The contribution of a_i to (a*a)[k]:
            # For each k, the terms involving a_i are:
            #   a_i * a[k-i] (when j=i in the sum)
            #   a[k-i] * a_i (when j=k-i, i.e., the term at position k-i times a_i)
            
            # (a*a)[k] = sum_j a[j] * a[k-j]
            # Terms with a[i]:
            #   j = i: a[i] * a[k-i] (valid if 0 <= k-i < n)
            #   j = k-i: a[k-i] * a[i] (valid if 0 <= k-i < n) -- same term!
            # So the total contribution is 2 * a[i] * a[k-i] (when k != 2i)
            # or a[i]^2 (when k = 2i)
            
            # Let's compute: for each k, the "base" (a*a)[k] without a_i,
            # and the "slope" of (a*a)[k] w.r.t. a_i
            base = np.zeros(2 * n - 1)
            slope = np.zeros(2 * n - 1)
            
            # base[k] = (a*a)[k] - 2*a[i]*a[k-i] + a[i]^2 * delta(k, 2i)
            # slope[k] = 2*a[k-i] - a[i] * delta(k, 2i)
            
            # More directly:
            # (a*a)[k] = base[k] + a_i * slope[k] + a_i^2 * delta(k, 2i)
            # where base[k] = sum_{j != i} a[j] * a[k-j]  (with appropriate handling)
            
            # Actually, let me just compute it numerically:
            # Set a[i] = 0, compute c0 = a*a
            # Set a[i] = 1, compute c1 = a*a
            # Then (a*a)[k] = c0[k] + a_i * (c1[k] - c0[k])
            # Wait, that's only linear in a_i, but the self-convolution is quadratic.
            
            # Let me reconsider. (a*a)[k] = sum_j a[j] a[k-j]
            # = a[i]*a[k-i] + sum_{j!=i} a[j]*a[k-j]
            # The second sum also includes j = k-i (if k-i != i), giving a[k-i]*a[i]
            # So: (a*a)[k] = (if k-i in [0,n)): a[i]*a[k-i] + (if k-i in [0,n) and k-i != i): a[k-i]*a[i] + rest
            # = 2*a[i]*a[k-i] (if k != 2i and 0 <= k-i < n) + rest
            # = a[i]^2 (if k = 2i) + rest
            
            # So (a*a)[k] = A_k * a_i + B_k * a_i^2 + C_k
            # where:
            #   B_k = 1 if k = 2i, else 0
            #   A_k = 2*a[k-i] if 0 <= k-i < n and k != 2i, else 0
            #         (when k = 2i, A_k = 0 since the linear term is a_i * a[i] = a_i^2 which goes to B_k)
            #   C_k = (a*a)[k] - A_k * a[i] - B_k * a[i]^2
            
            # To find the best a_i, we want to minimize max_k (A_k * a_i + B_k * a_i^2 + C_k)
            # subject to a_i >= 0 and a.sum() = S (so a_i <= S - (S - a[i]) = a[i]... 
            # actually if we change a_i, we need to adjust another coordinate to maintain the sum)
            
            # This is getting complicated. For simplicity, let's just do a small line search:
            # Try a few values around the current a_i and pick the best.
            
            old_val = a[i]
            best_use = old_val
            best_obj = best_val  # Current best
            
            # Try a few perturbations
            deltas = [0.5 * old_val, 0.75 * old_val, 1.25 * old_val, 1.5 * old_val,
                      0.25 * old_val, old_val + S/(10*n), old_val - S/(10*n)]
            deltas = [d for d in deltas if d >= 0]
            
            for new_ai in deltas:
                if abs(new_ai - old_ai) < 1e-15:
                    continue
                a[i] = new_ai
                # Renormalize to maintain sum
                s = a.sum()
                if s > 0:
                    a = a * (S / s)
                c = self_convolve(a)
                val = c.max()
                if val < best_obj:
                    best_obj = val
                    best_use = new_ai
                a[i] = old_ai
            
            if best_use != old_ai:
                a[i] = best_use
                a = project_simplex(a, S)
                improved = True
        
        if not improved:
            break
    
    return a, best_val
