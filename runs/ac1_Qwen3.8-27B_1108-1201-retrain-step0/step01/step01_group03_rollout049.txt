import time

def _compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def _normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)

def _multi_peak_grad(a, K=10, decay=0.8):
    """Compute gradient using top-K peaks of autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    
    # Get top K peaks
    top_indices = np.argsort(c)[-K:][::-1]
    weights = np.array([decay**i for i in range(K)])
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, m in zip(weights, top_indices):
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * w * a[m - j]
    
    grad -= grad.mean()
    return grad

def _optimize_multi_peak(a0, deadline, max_iters=5000):
    """Optimize using multi-peak gradient descent."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    a = _normalize(a)
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    S_t = np.sqrt(2.0 * n)
    
    K = min(10, n)
    lr_init = 0.2
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 10.0:
            break
        
        # Cosine annealing
        progress = it / max_iters
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, 0.001)
        
        # Increase K over time (start focused, broadening)
        K_cur = min(int(K * (0.5 + 0.5 * progress)), n)
        
        grad = _multi_peak_grad(a, K=K_cur)
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        a = _normalize(a, S_t)
        
        if (it + 1) % 50 == 0:

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based soft-peak gradient descent, coordinate descent, LP refinement,

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar


def _conv(a):
    """Compute full convolution using FFT for speed."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fsize = 1
    while fsize < size:
        fsize *= 2
    fa = np.fft.rfft(a, fsize)
    c = np.fft.irfft(fa * fa, fsize)[:size]
    return c


def _compute_obj(a):
    n = len(a)
    c = _conv(a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a
    return a * (target / S)


def _soft_peak_grad(a, K=15, beta=3.0):
    """
    Compute gradient using softmax-weighted peaks.
    Uses exp(beta * (c[m] - c_max)) weighting on top-K peaks.
    """
    n = len(a)
    c = _conv(a)
    c_max = c.max()
    
    # Softmax weights
    shifted = beta * (c - c_max)
    w = np.exp(shifted)
    
    # Keep only top K for efficiency
    if K < len(c):
        top_idx = np.argsort(c)[-K:]
        w_masked = np.zeros_like(w)
        w_masked[top_idx] = w[top_idx]
        w = w_masked
    
    w_sum = w.sum()
    if w_sum < 1e-15:
        return np.zeros(n)
    w = w / w_sum
    
    # Gradient: grad[k] = 2 * sum_m w[m] * a[m-k] * [0<=m-k<n]
    grad = np.zeros(n)
    for m in range(len(c)):
        if w[m] < 1e-12:

        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        for k in range(lo, hi + 1):
            grad[k] += 2.0 * w[m] * a[m - k]
    
    grad[np.isnan(grad)] = 0.0
    grad -= grad.mean()
    return grad

def _lp_direction(f, M, max_n=1536):

                if stall_count % 3 == 0:
                    grad_hard = _hard_peak_grad(a, K=3)
                    a = a - 0.05 * grad_hard
                    a = np.maximum(a, 0.0)
                    a = _normalize(a, S_t)
    
    return best, best_score


def _freq_init(n, rng):
    """Frequency-domain inspired initialization.
    Creates a sequence whose autocorrelation is close to flat."""
    # Target: flat autocorrelation means flat power spectrum (plus DC)
    # Generate a random "flat-ish" spectrum and take inverse FFT
    fsize = 2 * n
    freqs = np.fft.rfftfreq(fsize)
    
    # Create a spectrum that's somewhat flat but with random phase
    magnitude = np.ones(n // 2 + 1)
    # Add some random variation
    magnitude = magnitude * (0.5 + 0.5 * rng.random(n // 2 + 1))
    # Boost DC component
    magnitude[0] = 1.0
    phase = 2 * np.pi * rng.random(n // 2 + 1)
    
    spectrum = magnitude * np.exp(1j * phase)
    sig = np.fft.irfft(spectrum, fsize)[:n]
    
    # Make non-negative
    a = np.maximum(sig, 0)
    if a.sum() < 1e-6:
        a = np.abs(sig)
    return a


def _asymmetric_init(n, rng):
    """Create asymmetric initialization to break symmetry."""
    # Two-bump at asymmetric positions
    pos1 = int(n * rng.uniform(0.1, 0.3))
    pos2 = int(n * rng.uniform(0.6, 0.9))
    width = n * rng.uniform(0.1, 0.25)
    
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - pos1) / width) ** 2) + \
        np.exp(-0.5 * ((i - pos2) / width) ** 2)
    a += 0.1 * rng.random(n)
    return a


def _ramp_init(n, rng):
    """Asymmetric ramp initialization."""
    i = np.arange(n, dtype=np.float64)
    direction = rng.choice([1, -1])
    if direction == 1:
        a = 1.0 + i / n
    else:
        a = 1.0 - i / n
    a += 0.2 * rng.random(n)
    return np.maximum(a, 0.01)


def _optimize_single(a0, deadline, use_lp=True):
    """Full optimization pipeline for a single sequence."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n)
    S_t = np.sqrt(2.0 * n)
    a = _normalize(a, S_t)
    
    score = _compute_obj(a)
    best = a.copy()
    best_score = score
    
    # Phase 1: Gradient descent
    if time.monotonic() < deadline - 30:
        a, score = _gradient_descent(a, S_t, min(deadline, time.monotonic() + 120),
                                      max_iters=3000, K=15, beta=3.0, lr_init=0.15)
        if score < best_score:
            best_score = score
            best = a.copy()
    
    # Phase 2: Coordinate descent
    if time.monotonic() < deadline - 20:
        time_per_pass = 30.0 / max(1, min(n, 512) // 128)
        coord_deadline = min(deadline, time.monotonic() + min(60.0, n * 0.05))
        a, score = _coordinate_optimize(a, S_t, max_passes=min(3, max(1, int(n / 256))))
        if score < best_score:
            best_score = score
            best = a.copy()
    
    # Phase 3: More gradient descent (finer)
    if time.monotonic() < deadline - 20:
        grad_deadline = min(deadline - 20, time.monotonic() + 60)
        a, score = _gradient_descent(a, S_t, grad_deadline,
                                      max_iters=3000, K=8, beta=8.0, lr_init=0.05)
        if score < best_score:
            best_score = score
            best = a.copy()
    
    # Phase 4: LP refinement
    if use_lp and n <= 1024 and time.monotonic() < deadline - 10:
        lp_deadline = min(deadline, time.monotonic() + 60)
        a, score = _lp_refine(a, S_t, lp_deadline, max_iters=40)
        if score < best_score:
            best_score = score
            best = a.copy()
    
    # Phase 5: Final coordinate polish
    if time.monotonic() < deadline - 5:
        a, score = _coordinate_optimize(a, S_t, max_passes=2)
        if score < best_score:
            best_score = score
            best = a.copy()
    
    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Explores multiple initialization strategies and
    sequence lengths using gradient descent, coordinate descent, and LP.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 10.0)
    
    best_seq = None
    best_score = float('inf')
    
    def _try(a0, label=""):
        nonlocal best_seq, best_score
        if time.monotonic() > deadline - 30:
            return
        t0 = time.monotonic()
        use_lp = len(a0) <= 1024
        seq, score = _optimize_single(a0, deadline, use_lp=use_lp)
        elapsed = time.monotonic() - t0
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  New best: {score:.6f} (n={len(seq)}, {label}, {elapsed:.1f}s)")
        else:
            print(f"  No improvement: {score:.6f} (n={len(a0)}, {label}, {elapsed:.1f}s)")
    
    # === Generate diverse initializations ===
    print(f"Starting search, budget={budget_s}s, deadline in {deadline - t_start:.0f}s")
    
    # Load previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size >= 8 and np.all(np.isfinite(prev)):
            prev_norm = _normalize(prev.copy())
            s_prev = _compute_obj(prev_norm)
            print(f"Previous sequence: n={len(prev)}, score={s_prev:.6f}")
            if s_prev < best_score:
                best_score = s_prev
                best_seq = prev_norm.copy()
    except NameError:
        pass
    
    # Sequence lengths to try
    # Allocate time based on length (longer = more time per optimization)
    lengths_to_try = []
    
    # Always try the previous length if available
    try:
        prev_len = len(height_sequence_1)
        if prev_len not in lengths_to_try:
            lengths_to_try.append(prev_len)
    except NameError:
        pass
    
    # Add other lengths
    for L in [256, 384, 512, 768]:
        if L not in lengths_to_try:
            lengths_to_try.append(L)
    
    # Sort by length (try medium first, then adjust)
    lengths_to_try.sort()
    
    # Time allocation: give more time to promising lengths
    # Start with 512 (known to work), then explore others
    lengths_to_try.sort(key=lambda x: abs(x - 512))
    
    # Truncate to fit budget
    time_per_len = max(120.0, (deadline - t_start - 60) / len(lengths_to_try))
    
    print(f"Trying lengths: {lengths_to_try}")
    
    for n in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        
        n_deadline = min(deadline - 30, time.monotonic() + time_per_len)
        
        # Multiple initializations per length
        inits = []
        
        # 1. Random
        inits.append(('random', rng.random(n) * 0.5 + 0.25))
        
        # 2. Frequency domain
        inits.append(('freq', _freq_init(n, rng)))
        
        # 3. Asymmetric
        inits.append(('asym', _asymmetric_init(n, rng)))
        
        # 4. Ramp
        inits.append(('ramp', _ramp_init(n, rng)))
        
        # 5. Noise + structure
        i = np.arange(n, dtype=np.float64)
        inits.append(('chirp', 0.5 + 0.5 * np.cos(2 * np.pi * i * rng.uniform(1, 5) / n) + 0.1 * rng.random(n)))
        
        # 6. Two-bump asymmetric
        inits.append(('bump2', _asymmetric_init(n, rng)))
        
        # If we have a previous sequence of this length, use it
        try:
            if len(height_sequence_1) == n:
                inits.append(('prev', _normalize(height_sequence_1.copy())))
        except NameError:
            pass
        
        # Try each initialization within the time budget for this length
        # Prioritize: prev > freq > asym > random
        order = ['prev', 'freq', 'asym', 'ramp', 'chirp', 'bump2', 'random']
        inits.sort(key=lambda x: order.index(x[0]) if x[0] in order else 99)
        
        for label, a0 in inits:
            if time.monotonic() > n_deadline:
                break
            _try(a0, label=f"{label},n={n}")
        
        # After trying all inits for this length, do a final refinement of the best
        if best_seq is not None and len(best_seq) != n and time.monotonic() < deadline - 60:
            pass  # Only refine the global best at the end
    
    # === Final refinement of the overall best ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\nFinal refinement of best (score={best_score:.6f}, n={len(best_seq)})")
        n_best = len(best_seq)
        S_t = np.sqrt(2.0 * n_best)
        
        # Multiple refinement passes
        for pass_i in range(8):
            if time.monotonic() > deadline - 10:
                break
            
            # Alternate between gradient and coordinate
            if pass_i % 2 == 0:
                d = min(deadline - 10, time.monotonic() + 60)
                a, score = _gradient_descent(best_seq, S_t, d,
                                              max_iters=2000, K=10, beta=6.0, lr_init=0.03)
            else:
                d = min(deadline - 10, time.monotonic() + 60)
                a, score = _coordinate_optimize(best_seq, S_t, max_passes=2)
            
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"  Refine pass {pass_i}: {best_score:.6f}")
            else:
                # Try perturbation
                if pass_i < 4:
                    a = best_seq + 0.02 * rng.randn(n_best)
                    a = np.maximum(a, 0)
                    a = _normalize(a, S_t)
                    d = min(deadline - 10, time.monotonic() + 30)
                    a, score = _gradient_descent(a, S_t, d,
                                                  max_iters=1000, K=8, beta=5.0, lr_init=0.05)
                    if score < best_score:
                        best_score = score
                        best_seq = a.copy()
                        print(f"  Perturb pass {pass_i}: {best_score:.6f}")
        
        # LP refinement if size allows
        if n_best <= 1024 and time.monotonic() < deadline - 10:
            d = min(deadline, time.monotonic() + 60)
            a, score = _lp_refine(best_seq, S_t, d, max_iters=30)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"  Final LP: {best_score:.6f}")
    
    # === Final validation ===
    if best_seq is not None:
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = np.minimum(best_seq, 1000.0)
        final_score = _compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
        
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        
        return [float(x) for x in best_seq]
    
    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
